```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Main ideas:
- Work in probability simplex (sum(a)=1).
- Use the exact log-convolution natural gradient for the nonsmooth peak
  objective, smoothed by a softmax over lag variables:
      q_i = a_i * sum_k w_k a_{k-i} / sum_k w_k c_k,
  direction q - a.
- Combine mirror-descent log steps with projected Euclidean steps for robustness.
- Use a structure-specific Gerchberg-Saxton-style iteration in the squared
  Fourier domain A^2, not |A|^2, to drive the linear self-convolution toward
  a low-peak water-filled target.
- Added restarts, length interpolation, and final high-beta polish.
"""

import numpy as np
import time


def next_pow2(m: int) -> int:
    """Smallest power of two >= m."""
    m = int(max(1, m))
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_plain(a) -> np.ndarray:
    """Clip nonnegative and normalize to sum 1, without adding a floor."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    a = a / s
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def floor_normalize(a, floor_total: float = 1e-9) -> np.ndarray:
    """
    Normalize to sum 1 and add a tiny positive floor to every coordinate.
    The total floor mass is floor_total, so it is negligible for the score
    but prevents log/update lock-in of exact zeros.
    """
    a = normalize_plain(a)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = a + float(floor_total) / a.size
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a
    return a / s


def score_a(a) -> float:
    """Exact internal score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    m = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m).real[:2 * n - 1]
    c = 0.5 * (c + c[::-1])
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float(2.0 * n * cm / (s * s))


def compute_direction(a, beta: float):
    """
    Compute current score and normalized natural-gradient vector q for the
    smoothed peak-convolution loss.

    For a probability vector a (sum 1), c = a * a (linear convolution).
    Let w_k be softmax weights over c. Then
        q_i = a_i * sum_k w_k a_{k-i} / sum_k w_k c_k,
    and q is a probability vector. The log-score gradient direction is q - a.
    """
    a = floor_normalize(a, 1e-9)
    n = int(a.size)
    L = 2 * n - 1
    if n == 1:
        return 2.0, np.array([1.0], dtype=np.float64)

    # Use one FFT length large enough for both self-convolution (2n-1) and
    # the auxiliary cross convolution needed for q (3n-2).
    m = next_pow2(3 * n - 2)

    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m).real[:L]
    c = 0.5 * (c + c[::-1])

    s = float(a.sum())
    if s <= 1e-300:
        s = 1.0
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    sc = float(2.0 * n * cmax / (s * s))

    # Softmax weights over lags.  beta controls peak focus.
    shifted = float(beta) * (c - cmax)
    shifted = np.clip(shifted, -700.0, 700.0)
    w = np.exp(shifted)
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 1e-300:
        w = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        w = w / ws
    w = 0.5 * (w + w[::-1])
    ws2 = float(w.sum())
    if ws2 <= 1e-300:
        w = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        w = w / ws2

    denom = float(np.dot(w, c))
    if not np.isfinite(denom) or denom <= 1e-300:
        q = a.copy()
    else:
        # raw_i = sum_k w_k a_{k-i}.
        # This is a cross-correlation of a with w.  Compute it as
        # conv(a_rev, w)[n-1 + i].
        a_rev = a[::-1]
        W = np.fft.rfft(w, m)
        C = np.fft.irfft(np.fft.rfft(a_rev, m) * W, m).real[:3 * n - 2]
        raw = C[n - 1:2 * n - 1]
        raw = np.maximum(raw, 0.0)

        q = a * raw / denom
        q = np.maximum(q, 0.0)
        qs = float(q.sum())
        if not np.isfinite(qs) or qs <= 1e-300:
            q = a.copy()
        else:
            q = q / qs

    return sc, q


def natural_candidate(a, q, eta: float) -> np.ndarray:
    """Mirror-descent step in log variables: p <- p * exp(-eta(q-p))."""
    a = floor_normalize(a, 1e-9)
    q = normalize_plain(q)
    if eta <= 0.0:
        return a
    e = -float(eta) * (q - a)
    e = np.clip(e, -50.0, 50.0)
    b = a * np.exp(e)
    b = np.maximum(b, 1e-9 / max(1, b.size))
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a
    return b / s


def proj_candidate(a, q, eta: float) -> np.ndarray:
    """Simple projected Euclidean step on the simplex; useful for support movement."""
    a = floor_normalize(a, 1e-9)
    q = normalize_plain(q)
    if eta <= 0.0:
        return a
    b = a - float(eta) * (q - a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return a
    return b / s


def perturb_log(a, rng, sigma: float) -> np.ndarray:
    """Multiplicative log-space perturbation."""
    a = floor_normalize(a, 1e-9)
    v = np.log(np.maximum(a, 1e-300))
    v = v + float(sigma) * rng.standard_normal(a.size)
    v = v - float(np.mean(v))
    b = np.exp(np.clip(v - float(np.max(v)), -60.0, 60.0))
    return floor_normalize(b, 1e-9)


def perturb_blend(a, rng, sigma: float) -> np.ndarray:
    """Blend with a random positive distribution to activate new support."""
    a = floor_normalize(a, 1e-9)
    sigma = float(np.clip(sigma, 0.0, 1.0))
    noise = np.abs(rng.standard_normal(a.size))
    ns = float(noise.sum())
    if ns <= 1e-300:
        noise = np.ones_like(a)
    else:
        noise = noise / ns
    b = (1.0 - sigma) * a + sigma * noise
    return floor_normalize(b, 1e-9)


def interpolate_sequence(a, new_n: int) -> np.ndarray:
    """Resample a sequence to length new_n by linear interpolation."""
    a = floor_normalize(a, 1e-9)
    old_n = int(a.size)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return a
    xo = np.linspace(0.0, 1.0, old_n)
    xn = np.linspace(0.0, 1.0, new_n)
    b = np.interp(xn, xo, a)
    return floor_normalize(b, 1e-9)


def natural_refine(a, beta: float, eta_log: float, eta_lin: float, deadline: float,
                   max_no_improve: int = 70, rng=None, tag: str = ""):
    """
    Run adaptive natural/projection descent for a fixed beta until deadline.
    Returns (best_a, best_score, final_a).
    """
    a = floor_normalize(a, 1e-9)
    n = int(a.size)
    best_a = a.copy()
    best_s = score_a(a)
    final_a = a.copy()

    eta_log = max(1e-6, float(eta_log))
    eta_lin = max(0.0, float(eta_lin))
    noimp = 0

    while time.monotonic() < deadline:
        sc, q = compute_direction(a, beta)

        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = float(sc)
            best_a = a.copy()
            print(f"    [{tag}] natural improved: {best_s:.9f}, n={n}, beta={beta:.0f}")

        best_cand = None
        best_cand_s = float(sc)

        for mult in (1.0, 0.5, 0.25):
            e_log = eta_log * mult
            e_lin = eta_lin * mult

            if e_log > 1e-12:
                cand = natural_candidate(a, q, e_log)
                scand = score_a(cand)
                if np.isfinite(scand) and scand < best_cand_s - 1e-14:
                    best_cand_s = float(scand)
                    best_cand = cand

            if e_lin > 1e-12:
                cand = proj_candidate(a, q, e_lin)
                scand = score_a(cand)
                if np.isfinite(scand) and scand < best_cand_s - 1e-14:
                    best_cand_s = float(scand)
                    best_cand = cand

            if best_cand_s < float(sc) - 1e-12:
                break

        if best_cand is not None and best_cand_s < float(sc) - 1e-12:
            a = best_cand
            sc = best_cand_s
            noimp = 0
            if sc < best_s - 1e-12:
                best_s = float(sc)
                best_a = a.copy()
                print(f"    [{tag}] step improved: {best_s:.9f}, eta_log={eta_log:.5f}, eta_lin={eta_lin:.5f}")

            eta_log = min(3.0, eta_log * 1.07)
            eta_lin = min(0.05, eta_lin * 1.04)
        else:
            noimp += 1
            eta_log *= 0.62
            eta_lin *= 0.62
            if eta_log < 1e-4:
                eta_log = 0.05
            if eta_lin < 1e-5:
                eta_lin = 0.001

            if noimp >= max_no_improve:
                if rng is None:
                    break
                sigma = 0.018 + 0.006 * min(6, noimp // 25)
                a = perturb_log(a, rng, sigma)
                sc = score_a(a)
                eta_log = max(eta_log, 0.10)
                eta_lin = max(eta_lin, 0.004)
                noimp = 35
                if sc < best_s - 1e-12:
                    best_s = float(sc)
                    best_a = a.copy()

        final_a = a.copy()

    if np.isfinite(sc) and sc < best_s:
        best_s = float(sc)
        best_a = a.copy()

    final_a = a.copy()
    return best_a, best_s, final_a


def conv_fienup(a, deadline: float, alpha: float, tau_frac: float, tag: str = ""):
    """
    Gerchberg-Saxton style iteration tailored to linear self-convolution.

    In the zero-padded Fourier domain, C = A^2 is the transform of c = a*a.
    We blend C with the transform of a water-filled time-domain target c_des,
    take a square root to get A_new, and project back to nonnegative a.
    """
    a = floor_normalize(a, 1e-9)
    n = int(a.size)
    L = 2 * n - 1
    m = next_pow2(L)

    best_a = a.copy()
    best_s = score_a(a)
    c_des = None

    it = 0
    alpha = float(np.clip(alpha, 0.05, 0.95))
    tau_frac = float(np.clip(tau_frac, 0.30, 0.995))

    while time.monotonic() < deadline:
        A = np.fft.rfft(a, m)
        C = A * A
        c = np.fft.irfft(C, m).real[:L]
        c = 0.5 * (c + c[::-1])
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax < 0.0:
            cmax = 0.0
        sc = float(2.0 * n * cmax)

        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = float(sc)
            best_a = a.copy()
            print(f"    [{tag}] conv_fienup improved: {best_s:.9f}, alpha={alpha:.2f}, tau={tau_frac:.2f}")

        # If the trajectory blew up, revert to the best seen.
        if not np.isfinite(sc) or sc > best_s * 3.0 + 0.10:
            a = best_a.copy()
            c_des = None
            it = 0
            continue

        # Refresh the desired time-domain shape every few iterations.
        if c_des is None or it % 12 == 0:
            delta = cmax * (1.0 - tau_frac)
            c_des = np.maximum(c - delta, 0.0)
            sd = float(c_des.sum())
            if not np.isfinite(sd) or sd <= 1e-12:
                c_des = np.full(L, 1.0 / L, dtype=np.float64)
            else:
                c_des = c_des / sd
            c_des = 0.5 * (c_des + c_des[::-1])

        C_des = np.fft.rfft(c_des, m)
        C_new = (1.0 - alpha) * C + alpha * C_des

        # DC bin must be positive real.
        C_new = C_new.astype(np.complex128, copy=False)
        C_new[0] = complex(max(float(np.real(C_new[0])), 1e-300), 0.0)

        if not np.all(np.isfinite(C_new)):
            break

        A_new = np.sqrt(C_new)

        if not np.all(np.isfinite(A_new)):
            a = best_a.copy()
            c_des = None
            continue

        # Choose square-root signs to stay close to the previous phase.
        align = np.real(A_new * np.conj(A))
        A_new[align < 0.0] = -A_new[align < 0.0]

        a_new = np.fft.irfft(A_new, m).real[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = floor_normalize(a_new, 1e-9)
        a = a_new
        it += 1

        if it % 250 == 0:
            cur = score_a(a)
            print(f"    [{tag}] conv_fienup iter={it}, cur={cur:.9f}, best={best_s:.9f}")

    fin_s = score_a(a)
    if np.isfinite(fin_s) and fin_s < best_s:
        best_s = float(fin_s)
        best_a = a.copy()

    return best_a, best_s, a


def propose_candidate(seed: int = 42, budget_s: float = 2400.0, **kwargs) -> list:

        sf = score_a(ftr)

    # ------------------------------------------------------------------

    )

    for f0, f1, kind, sigma, beta, elog, elin in restart_ranges:
        d0 = t0 + f0 * T
        d1 = t0 + f1 * T
        if time.monotonic() >= d1 - 3.0:
            continue

        if kind == "log":
            start = perturb_log(best_a, rng, sigma + 0.005 * rng.standard_normal())
        else:
            start = perturb_blend(best_a, rng, sigma)

        a1, s1, ftr = natural_refine(start, beta=beta,
                                     eta_log=elog, eta_lin=elin,
                                     deadline=d1 - 1.0,
                                     max_no_improve=70, rng=rng, tag=f"R-{kind}")
        if np.isfinite(s1) and s1 < best_s:
            best_s = float(s1)
            best_a = np.asarray(a1, dtype=np.float64).copy()
        sf = score_a(ftr)
        if np.isfinite(sf) and sf < best_s:
            best_s = float(sf)

    # ------------------------------------------------------------------
    # Phase 4: length exploration.
    # Try a moderate upsampling/downsampling around the current length.
    # ------------------------------------------------------------------
    length_section_end = t0 + 0.905 * T
    cur_n = int(best_a.size)
    candidates_n = []

    if cur_n < 16384:
        candidates_n.append(min(20000, max(cur_n + 1, int(round(cur_n * 1.6)))))
    if cur_n > 12500:
        candidates_n.append(10000)
    if cur_n < 4096:
        candidates_n.append(8192)

    # De-duplicate while preserving order.
    seen_n = set()
    uniq_n = []
    for nn in candidates_n:
        if nn not in seen_n and nn > 511:
            seen_n.add(nn)
            uniq_n.append(nn)

    if len(uniq_n) > 0:
        # Split available time among candidate lengths, but cap total.
        l0 = max(time.monotonic(), t0 + 0.905 * T)
        l1 = length_section_end - 1.0
        if l1 - l0 > 25.0:
            per = (l1 - l0) / float(len(uniq_n))
            for nn in uniq_n:
                d_end = l0 + per - 1.0
                if time.monotonic() >= d_end - 2.0:
                    break
                a_interp = interpolate_sequence(best_a, int(nn))
                s0 = score_a(a_interp)
                if np.isfinite(s0) and s0 < best_s:
                    best_s = float(s0)
                    best_a = a_interp.copy()
                if s0 <= best_s * 1.04 + 1e-6:
                    a1, s1, ftr = natural_refine(a_interp, beta=20000.0,
                                                 eta_log=0.14, eta_lin=0.005,
                                                 deadline=d_end,
                                                 max_no_improve=55, rng=rng,
                                                 tag=f"LEN-{nn}")
                    if np.isfinite(s1) and s1 < best_s:
                        best_s = float(s1)
                        best_a = np.asarray(a1, dtype=np.float64).copy()
                    sf = score_a(ftr)
                    if np.isfinite(sf) and sf < best_s:
                        best_s = float(sf)
                        best_a = np.asarray(ftr, dtype=np.float64).copy()

    # ------------------------------------------------------------------
    # Phase 5: final high-beta polish.
    # ------------------------------------------------------------------
    polish = (
        (0.915, 0.955, 60000.0, 0.10, 0.003, "F1-60k"),
        (0.955, 0.985, 250000.0, 0.08, 0.002, "F2-250k"),
        (0.985, 1.000, 100000.0, 0.06, 0.002, "F3-100k"),
    )

    for f0, f1, beta, elog, elin, tag in polish:
        d0 = t0 + f0 * T
        d1 = t0 + f1 * T
        if time.monotonic() >= d1 - 2.0:
            continue
        a1, s1, ftr = natural_refine(best_a.copy(), beta=beta,
                                     eta_log=elog, eta_lin=elin,
                                     deadline=d1 - 0.75,
                                     max_no_improve=45, rng=rng, tag=tag)
        if np.isfinite(s1) and s1 < best_s:
            best_s = float(s1)
            best_a = np.asarray(a1, dtype=np.float64).copy()
        sf = score_a(ftr)
        if np.isfinite(sf) and sf < best_s:
            best_s = float(sf)
            best_a = np.asarray(ftr, dtype=np.float64).copy()

    # ------------------------------------------------------------------
    # Final safe output.
    # ------------------------------------------------------------------
    best_a = floor_normalize(best_a, 0.0)
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]

    if len(out) == 0 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = float("inf")
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float("inf")

    # If evaluation disagrees badly, try a normalized copy.
    if not np.isfinite(verified) or verified > best_s + 1e-6:
        alt = [float(max(0.0, min(1000.0, x))) for x in normalize_plain(np.asarray(best_a, dtype=np.float64))]
        if len(alt) > 0 and sum(alt) >= 0.01:
            try:
                valt = float(evaluate_sequence(alt))
                if np.isfinite(valt) and valt < verified:
                    out = alt
                    verified = valt
            except Exception:
                pass

    elapsed = time.monotonic() - t0
    print("=" * 70)
    print(f"FINAL internal best = {best_s:.9f}")
    print(f"FINAL evaluated     = {verified:.9f}")
    print(f"Length              = {len(out)}")
    print(f"Time used           = {elapsed:.1f} / {budget:.1f}s")
    print("=" * 70)

    return out
```
