def next_power_of_two(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def conv_gradient(a, w):
    """
    Gradient of sum_k w[k]*conv(k) w.r.t. a[j].
    conv = full_convolution(a, a), length 2n-1.
    grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices.

    # d(sum w[k]*conv[k])/da[j] = 2 * (a * w_padded)[j] clipped

    # Simpler: just compute via two FFTs

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def argmax_weights(conv, top_k=5, eps_frac=0.02):
    """Get weights concentrated on the argmax region of conv."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    if active.sum() < top_k:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        active[idx] = True
    max_count = max(top_k, int(0.02 * conv.size))
    if active.sum() > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_step(a, conv, top_k, eps, eta):
    """Exponentiated gradient step: reduce a where it contributes to peak conv."""
    w = argmax_weights(conv, top_k=top_k, eps_frac=eps)
    grad = conv_gradient(a, w)
    gmax = float(np.max(grad))
    if gmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * grad / gmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def lse_objective_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    # Score = 2n*c/S^2, so minimize max(c)/S^2
    # LSE of c: L = (1/beta)*log(sum(exp(beta*c)))
    # Score_LSE = 2n*L/S^2
    # Gradient: d(2n*L/S^2)/da = (2n/S^2)*dL/da - 4n*L*a/S^3
    # dL/da = (1/beta) * exp(beta*c)/sum(exp(beta*c)) * beta * dc/da
    #       = w_soft * dc/da where w_soft[k] = exp(beta*c[k])/sum(exp(beta*c))
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w_soft = e / sum_e
    L = cmax + np.log(sum_e) / beta
    # dc/da = conv_gradient(a, 1) essentially, but we need weighted version
    # d(sum_k w_soft[k]*c[k])/da = conv_gradient(a, w_soft)
    grad_c = conv_gradient(a, w_soft)
    # d(score)/da = 2n/S^2 * grad_c - 4n*L*a/S^3
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n / (S * S)) * grad_c - (4.0 * n * L / (S ** 3)) * a
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e10, 1e10)
    return float(score), grad


def run_lbfgs(a, beta, deadline, maxfun=300):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        x = np.clip(x, 0.0, None)
        return lse_objective_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-13,
        "maxiter": max(20, maxfun // 3),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def peak_peel_step(a, conv, eta):
    """Targeted step: identify peak lag, apply anti-pressure at that lag."""
    n = len(a)
    peak_k = int(np.argmax(conv))
    # Create a delta-like weight at the peak
    w = np.zeros(2 * n - 1)
    # Spread slightly around peak for numerical stability
    spread = max(1, n // 512)
    lo = max(0, peak_k - spread)
    hi = min(2 * n - 2, peak_k + spread)
    w[lo:hi+1] = 1.0
    w /= w.sum()
    grad = conv_gradient(a, w)
    gmax = float(np.max(grad))
    if gmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * grad / gmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent=None, cap=30):

    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):

            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        add(pr ** 1.5)
        add(pr ** 0.5)

    # Uniform
    add(np.ones(n))

    # Asymmetric ramps (KEY: asymmetry helps!)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        add(x ** p)
        add((1.0 - x) ** p)
    # Linear ramps
    add(0.1 + 0.9 * x)
    add(0.1 + 0.9 * (1.0 - x))

    # V-shapes (asymmetric)
    for c in (0.3, 0.4, 0.5, 0.6, 0.7):
        for w in (0.05, 0.1, 0.2, 0.3):
            add(np.exp(-((x - c) / w) ** 2) + 0.05)
    # Bimodal asymmetric
    for c1, c2, w1, w2 in [(0.2, 0.8, 0.1, 0.1), (0.3, 0.7, 0.15, 0.15),
                            (0.2, 0.6, 0.1, 0.2), (0.4, 0.9, 0.1, 0.08)]:
        add(np.exp(-((x - c1) / w1) ** 2) + np.exp(-((x - c2) / w2) ** 2) + 0.05)
    # Power shapes asymmetric
    for p in (1.0, 2.0, 3.0, 5.0):
        add((x ** p) + (1 - x) ** (2 * p) + 0.01)
        add((x ** (2 * p)) + (1 - x) ** p + 0.01)
    # Cosine asymmetric
    for f in (1, 2, 3):
        add(0.5 + 0.5 * np.cos(2 * np.pi * f * x - np.pi / 3))
        add(0.5 + 0.5 * np.cos(2 * np.pi * f * x + np.pi / 4))
    # Exponential taper asymmetric
    for r in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-r * x) + 0.01)
        add(np.exp(-r * (1.0 - x)) + 0.01)
    # Random smooth (asymmetric via random phase)
    Nf = next_power_of_two(n)
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        decay = rng.uniform(1.5, 4.0)
        power = 1.0 / ((np.abs(freqs) + 1.0) ** decay)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(np.abs(noise) + 0.1)
        add(0.5 + 0.5 * noise / (np.max(np.abs(noise)) + 1e-10) + 0.1)
    # Random (noisy)
    for _ in range(4):

    # Step functions

        arr = np.ones(n)
        cut = int(split * n)
        arr[cut:] = 0.3
        add(arr)
        arr2 = np.ones(n)
        arr2[:cut] = 0.3
        add(arr2)

    return inits

        best_eta = eta

        # Try multiple etas with active-set exp-grad
        etas = [eta * 0.3, eta * 0.5, eta, eta * 1.5, eta * 2.5]
        for eta_try in etas:
            if eta_try < 0.005:
                continue
            new_a = exp_grad_step(a, conv, top_k, eps_frac, eta_try)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_iter:
                best_sc_iter = new_sc
                best_a_iter = new_a
                best_eta = eta_try

        # Peak peeling
        if use_peel:
            for eta_p in [eta * 0.3, eta * 0.7, eta * 1.2]:
                new_a = peak_peel_step(a, conv, eta_p)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter:
                    best_sc_iter = new_sc
                    best_a_iter = new_a

        if best_sc_iter < best_sc - 1e-12:
            best_sc = best_sc_iter
            best_a = best_a_iter.copy()
            a = best_a_iter
            eta = best_eta
            stagnant = max(0, stagnant - 2)
            if verbose and iteration % 100 == 0:
                print("  [{} i{}] score={:.9f} k={} eps={:.3f} eta={:.3f}".format(
                    tag, iteration, best_sc, top_k, eps_frac, eta))
        else:
            stagnant += 1
            eta *= 0.8
            if stagnant >= max_stagnant:
                # Adapt parameters
                top_k = min(int(top_k * 1.4) + 1, 50)
                eps_frac = min(eps_frac * 1.2, 0.12)
                eta = max(0.05, eta * 0.6)
                # Try a random perturbation to escape
                if rng is not None:
                    noise = 0.05 * rng.standard_normal(n)
                    a = normalize_a(a * np.exp(noise))
                stagnant = 0
                if verbose:
                    print("  [{} i{}] RESTART k={} eps={:.3f}".format(tag, iteration, top_k, eps_frac))
        # Slowly tighten active set
        top_k = max(2, int(top_k * 0.995))
        eps_frac = max(0.005, eps_frac * 0.998)

    return float(best_sc), best_a


def subgrad_polish(a, deadline, tag=""):
    """Fine-polish: systematic sweep of (top_k, eta) combos."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    top_ks = [1, 2, 3, 4, 5, 8, 12, 20, 40]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.8, 1.5]
    eps_list = [0.0, 0.005, 0.01, 0.02, 0.05]
    peel_etas = [0.01, 0.03, 0.08, 0.2, 0.5]

    improved = True
    rnum = 0
    while improved and time.monotonic() < deadline and rnum < 10:
        improved = False
        rnum += 1
        conv = fast_self_conv(best_a)
        for tk in top_ks:
            if time.monotonic() > deadline - 3:
                break
            for ef in eps_list:
                for eta in etas:
                    if time.monotonic() > deadline - 3:
                        break
                    new_a = exp_grad_step(best_a, conv, tk, ef, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = new_a
                        improved = True
        for eta_p in peel_etas:
            if time.monotonic() > deadline - 3:
                break
            new_a = peak_peel_step(best_a, conv, eta_p)
            new_a = normalize_a(new_a)
            sc = exact_score(new_a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = new_a
                improved = True

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    best_global_sc = float('inf')
    best_global_a = None

    # Try multiple sequence lengths
    n_candidates = [10240]
    if parent is not None:
        pn = int(parent.size)
        if pn >= 1024:
            n_candidates.insert(0, pn)
    # Add some alternatives
    for nl in [8192, 12288, 16384]:
        if nl not in n_candidates:
            n_candidates.append(nl)

    # Time allocation per length
    time_per_n = (total_budget - 100.0) / max(1, min(3, len(n_candidates)))

    for ni, n in enumerate(n_candidates[:3]):  # Try at most 3 lengths
        n_dead = t_start + 30.0 + time_per_n * (ni + 1)
        n_dead = min(n_dead, deadline - 50.0)
        if time.monotonic() > n_dead - 30:
            break

        print("=== Length n={} (deadline +{:.0f}s) ===".format(n, n_dead - t_start))

        # Generate inits
        inits = generate_inits(n, rng, parent, cap=25)
        best_init_sc = float('inf')
        best_init_a = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init_a = init
        print("  Best init: {:.9f}".format(best_init_sc))

        a = normalize_a(best_init_a)

        # Phase 1: Active-set optimization (main)
        p1_dead = min(n_dead - 60.0, t_start + 30.0 + time_per_n * 0.6)
        sc, a = active_set_optimize(a, deadline=p1_dead, verbose=True, tag="n{}p1".format(n))
        print("  After active-set: {:.9f}".format(sc))

        # Phase 2: Subgradient polish
        p2_dead = min(n_dead - 40.0, t_start + 30.0 + time_per_n * 0.75)
        sc, a = subgrad_polish(a, deadline=p2_dead, tag="n{}p2".format(n))
        print("  After subgrad: {:.9f}".format(sc))

        # Phase 3: Multi-beta LSE-LBFGS
        p3_dead = min(n_dead - 10.0, t_start + 30.0 + time_per_n * 0.92)
        a = normalize_a(a)
        for beta in [1000, 5000, 20000, 50000, 100000, 300000]:
            if time.monotonic() > p3_dead - 20:
                break
            rem = p3_dead - time.monotonic()
            if rem < 30:
                break
            mf = max(50, int(rem * 0.6 / 0.2))
            mf = min(mf, 800)
            sc, a = run_lbfgs(a, beta, p3_dead, mf)
            a = normalize_a(a)
            if sc < best_global_sc:
                pass  # track per-length
            print("    LSE beta={}: {:.9f}".format(beta, sc))

        # Phase 4: Final active-set + subgrad
        p4_dead = n_dead - 20.0
        if p4_dead > time.monotonic() + 20:
            sc, a = active_set_optimize(a, deadline=p4_dead - 10, verbose=False, tag="n{}p4".format(n))
            sc, a = subgrad_polish(a, deadline=p4_dead, tag="n{}p4b".format(n))

        # Final LSE polish on best
        if time.monotonic() < n_dead - 15:
            for beta in [100000, 500000]:
                if time.monotonic() > n_dead - 15:
                    break
                sc, a = run_lbfgs(a, beta, n_dead - 10, 200)

        final_sc = exact_score(a)
        print("  Final for n={}: {:.9f}".format(n, final_sc))

        if final_sc < best_global_sc:
            best_global_sc = final_sc
            best_global_a = a.copy()

    # Use remaining time for final refinement on best
    if best_global_a is not None and time.monotonic() < deadline - 30:
        final_dead = deadline - 10
        best_a = best_global_a.copy()
        best_sc = best_global_sc
        n = len(best_a)

        # Quick active-set
        sc, best_a = active_set_optimize(best_a, deadline=time.monotonic() + (final_dead - time.monotonic()) * 0.4, verbose=False, tag="fin1")
        if sc < best_sc:
            best_sc = sc

        # Subgrad
        rem = final_dead - time.monotonic()
        if rem > 30:
            sc, best_a = subgrad_polish(best_a, deadline=time.monotonic() + rem * 0.4, tag="fin2")
            if sc < best_sc:
                best_sc = sc

        # Final LSE
        rem = final_dead - time.monotonic()
        if rem > 50:
            best_a = normalize_a(best_a)
            for beta in [200000, 500000, 1000000]:
                if time.monotonic() > final_dead - 20:
                    break
                sc, best_a = run_lbfgs(best_a, beta, final_dead - 10, 300)
                if sc < best_sc:
                    best_sc = sc
                    print("  Final LSE beta={}: {:.9f}".format(beta, sc))

    final_sc = exact_score(best_a)
    print("Best overall: {:.9f}, n={}".format(final_sc, len(best_a)))

    # Output
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = final_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        final_sc, verified, len(out), time.monotonic() - t_start))
    return out
