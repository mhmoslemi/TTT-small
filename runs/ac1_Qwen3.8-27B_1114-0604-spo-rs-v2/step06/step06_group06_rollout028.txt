def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Compute full linear self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def compute_active_weights(conv, top_k=10, eps_frac=0.02):
    """Compute active set weights proportional to excess above threshold."""
    dmax = float(np.max(conv))
    if dmax <= 0:

def exp_grad_update(a, conv, w, eta):
    """Exponential gradient step in log-space."""
    h = corr_pair(a, w)
    hmax = float(np.max(h))

    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))

    """Apply momentum to the direction."""
    if prev_direction is None:
        return direction_a, direction_a.copy()
    new_dir = momentum * prev_direction + (1.0 - momentum) * direction_a
    return new_dir, new_dir.copy()


def spectral_flatten(a, alpha=0.1):
    """Frequency-domain step: shift power spectrum toward flatness."""
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = np.abs(A) ** 2
    # Shift P toward its mean (flatten)
    P_mean = float(np.mean(P))
    P_new = (1.0 - alpha) * P + alpha * P_mean
    # Ensure non-negative (it should be already, but just in case)
    P_new = np.maximum(P_new, 0.0)
    # Reconstruct with original phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    return a_new


def spectral_sharpen(a, alpha=0.1):
    """Alternatively: sharpen the spectrum (concentrate at DC) to flatten time-domain conv."""
    n = len(a)
    N = next_power_of_two(2 * n - 1)

    P = np.abs(A) ** 2
    # Concentrate power: boost low frequencies, reduce high
    freqs = np.arange(len(P))
    # Weight that's high at low freq, low at high freq
    weight = 1.0 / (1.0 + 0.1 * freqs)
    weight = weight / weight[0]  # normalize so DC = 1

    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)

    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    return a_new

            best_a = a.copy()

        # Compute active weights
        w = compute_active_weights(conv, top_k=top_k, eps_frac=eps_frac)
        
        # Compute direction
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            break
        
        direction = h / hmax
        
        # Apply momentum
        dir_mom, prev_direction = momentum_update(a * 0 + direction, direction, prev_direction, momentum)
        
        # Try multiple step sizes (geometric)
        best_a_iter = a
        best_sc_iter = sc
        base_eta = eta
        
        # Fine geometric search
        etas = []
        for mult in [0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.4, 2.0, 3.0]:
            etas.append(base_eta * mult)
        
        for eta_try in etas:
            if eta_try < 0.005:
                continue
            log_a = np.log(a + 1e-300)
            log_a -= eta_try * dir_mom
            log_a -= float(np.max(log_a))
            new_a = np.exp(log_a)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_iter:
                best_sc_iter = new_sc
                best_a_iter = new_a
                base_eta = eta_try

        if best_sc_iter < best_sc - 1e-10:
            a = best_a_iter
            eta = base_eta * 1.1  # slightly increase if it worked
            momentum = min(0.7, momentum + 0.01)  # increase momentum
            stagnant = 0
            if verbose and it % 100 == 0:
                print("  [{} i{}] score={:.9f} eta={:.3f} mom={:.2f}".format(
                    tag, it, best_sc, eta, momentum))
        else:
            stagnant += 1
            eta *= 0.8
            if stagnant >= max_stagnant:
                # Try spectral step
                if stagnant % 2 == 0:
                    spec_a = spectral_flatten(a, alpha=0.05)
                    spec_a = normalize_a(spec_a)
                    spec_sc = exact_score(spec_a)
                    if spec_sc < best_sc:
                        best_sc = spec_sc
                        best_a = spec_a
                        a = spec_a
                else:
                    spec_a = spectral_sharpen(a, alpha=0.05)
                    spec_a = normalize_a(spec_a)
                    spec_sc = exact_score(spec_a)
                    if spec_sc < best_sc:
                        best_sc = spec_sc
                        best_a = spec_a
                        a = spec_a
                
                # Random perturbation
                rng = np.random.default_rng(hash((tag, it)) & 0xFFFFFFFF)
                a = normalize_a(a * np.exp(0.1 * rng.standard_normal(n)))
                
                # Reset parameters
                top_k = min(top_k + 5, 50)
                eps_frac = min(eps_frac * 1.5, 0.1)
                eta = max(0.1, eta * 0.5)
                momentum = 0.2
                stagnant = 0
                prev_direction = None
                if verbose:
                    print("  [{} i{}] diversify: top_k={}, eps={:.3f}".format(
                        tag, it, top_k, eps_frac))
        # Periodic spectral step
        if it % spectral_interval == spectral_interval - 1 and stagnant < max_stagnant:
            spec_a = spectral_flatten(a, alpha=0.03)
            spec_a = normalize_a(spec_a)
            spec_sc = exact_score(spec_a)
            if spec_sc < best_sc:
                best_sc = spec_sc
                best_a = spec_a.copy()
                a = spec_a
                if verbose:
                    print("  [{} i{}] spectral: {:.9f}".format(tag, it, best_sc))

    return float(best_sc), best_a


def lse_score(a, beta):
    """LSE-smoothed objective value."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10
    L = dmax + np.log(sum_e) / beta
    return float(2.0 * n * L)


def lse_grad(a, beta):
    """Gradient of LSE-smoothed objective."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return np.zeros(n)
    w = e / sum_e
    # Gradient of 2n * (dmax + log(sum_e)/beta) w.r.t. a
    # d/d a_j of d[k] = 2 * a[k-j] / S^2 * n (for the conv part)
    # Actually let me just use the subgradient of max and smooth it
    h = corr_pair(a, w)
    # The gradient of the LSE approximation to max(d) is:
    # sum_k w[k] * d(d[k])/da
    # d(d[k])/da[j] = 2 * n * a[k-j] / S^2 (approximately, for the linear conv)
    # But we also have the -4n/S * a[j] * (sum derivative) term from S dependence
    # For simplicity, just use the correlation with w:
    grad = 2.0 * n * h / (S * S) - 4.0 * n * a * float(np.dot(w, d)) / (S * S)
