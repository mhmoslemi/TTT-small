"""
Multi-phase optimizer for minimizing the step function evaluation score.
Uses LP-based best response, LSE-LBFGS, subgradient, GS, and coordinate descent.
Key innovation: LP-based best response iteration for escaping local minima.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

def lp_best_response(f, M, S):
    """Solve LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0.
    Returns g normalized to have sum S.
    """
    n = len(f)
    num_cons = 2 * n - 1
    
    # Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for k in range(num_cons):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(num_cons, n))
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b_rhs = np.full(num_cons, M)
    
    res = linprog(
        c_obj,
        A_ub=A_sparse,
        b_ub=b_rhs,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True}
    )
    
    if not res.success:
        return None
    
    g = res.x.copy()
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return None
    g = g * (S / g_sum)
    return g

                # d(c_k)/da_m: the term a_j * a_{k-j} depends on a_m when j=m or k-j=m (i.e., j=k-m).
                # If j=m is in range: contribution is a_{k-m}
                # If j=k-m is in range (and k-m != m): contribution is a_m
                # If K=2m (i.e., j=m=k-m): contribution is 2*a_m
                # So: d(c_k)/da_m = a_{k-m} * I[0<=k-m<n and m in [j_start,j_end]] + a_m * I[0<=k-m<n and k-m in [j_start,j_end] and k-m != m]
                # Simplifying: if 0<=k-m<n:
                #   If k-m == m: contribution = 2*a_m (but only if m is in [j_start, j_end])
                #   Else: contribution = a_{k-m} + a_m (if both m and k-m are in [j_start, j_end])
                #         or = a_{k-m} (if only m is in range)
                #         or = a_m (if only k-m is in range)
                
                # Simpler approach: just use the FFT-based computation
                pass
        
        # Use FFT-based subgradient (simpler and correct)
        # d(c_k)/da = 2 * fconv(a, delta_k) where delta_k is a spike at position k
        # Actually: d(c_k)/da_j = a_{k-j} (treating a as extended by zeros)
        # The subgradient of max(c) is the average of subgradients at all argmax points
        # For simplicity, use the top peaks weighted equally
        
        # More efficient: use the fact that d(c_k)/da = 2 * circconv(a, e_k) restricted to [0,n)
        # where e_k is a one-hot at position k
        # This equals 2 * [a shifted by k positions] (with wrap-around handled by bounds)
        
        # Let's just do it directly for the top peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # d(c_k)/da_j = a_{k-j} if 0 <= k-j < n, else 0
            # Plus a_j if 0 <= k-j < n (the "other" occurrence)
            # Actually, let me just use: d(c_k)/da = 2 * (a reflected and shifted)
            # For the linear convolution c_k = sum_j a_j a_{k-j}:
            # d(c_k)/da_m = a_{k-m} (if 0<=k-m<n) [from j=m term]
            #              + a_{k-m} (if 0<=k-m<n and k-m != m) [from j=k-m term, contributing a_m]
            # Wait, I'm overcomplicating this.
            # c_k = sum_{j: valid} a_j * a_{k-j}
            # Let's write it as: c_k = sum_j a_j * a_{k-j} (with a out of range = 0)
            # d(c_k)/da_m = a_{k-m} (from j=m) + a_m (from j=k-m, if k-m != m)
            #              = a_{k-m} + a_m * [k-m != m and 0<=k-m<n]
            # If k-m == m (i.e., k=2m): d(c_k)/da_m = 2*a_m
            # If k-m != m and 0<=k-m<n: d(c_k)/da_m = a_{k-m} + a_m
            # If k-m < 0 or k-m >= n: d(c_k)/da_m = 0 (if the only occurrence is the out-of-range one)
            #   Actually if 0<=k-m<n is false, then j=m gives a_{k-m}=0, and j=k-m is out of range, so contribution is 0.
            
            # Simplified: for each m in [0,n):
            #   if 0 <= k-m < n: sg[m] += a[k-m] + (a[m] if k-m != m else a[m])
            #   = a[k-m] + a[m] if k-m != m, or 2*a[m] if k-m == m
            #   But we need to be careful: the sum c_k = sum_j a_j * a_{k-j} only includes valid j.
            #   So d(c_k)/da_m = a_{k-m} * [0<=k-m<n and m in valid range for j]
            #                  + a_m * [0<=k-m<n and k-m in valid range for j and k-m != m]
            #   The valid range for j is [max(0,k-n+1), min(k, n-1)].
            #   m is in valid range iff max(0,k-n+1) <= m <= min(k, n-1).
            #   k-m is in valid range iff max(0,k-n+1) <= k-m <= min(k, n-1).
            #     => k - min(k,n-1) <= m <= k - max(0,k-n+1)
            #     => max(1, k-n+1) <= m <= min(k, n-1)  [hmm this is getting complicated]
            
            # Let me just do it simply: for the peak at lag k, the gradient w.r.t. a[m] is:
            # a[k-m] if 0 <= k-m < n (this is the j=m term)
            # + a[m] if 0 <= k-m < n and k-m != m (this is the j=k-m term)
            # = a[k-m] + a[m] if 0<=k-m<n and k-m!=m
            # = 2*a[m] if k-m==m (i.e., k=2m)
            # = 0 if k-m < 0 or k-m >= n
            # (This assumes both m and k-m are in [0,n), which they are by the condition)
            
            lo = max(0, k - n + 1)
           高 hi = min(k, n - 1)
            for m in range(lo, hi + 1):
                other = k - m
                if 0 <= other < n:
                    if other == m:
                        sg[m] += 2.0 * a[other]
                    else:
                        sg[m] += a[other] + a[m]
                # If other is out of [0,n), this m doesn't appear in the sum at all
                # Actually wait: if m is in [lo, hi], then by definition max(0,k-n+1) <= m <= min(k,n-1)
                # This means: k-m >= 0 (since m <= k) and k-m <= k-lo = min(k,n-1) - max(0,k-n+1)
                # Hmm, let me just check: if m >= 0 and m < n and k-m >= 0 and k-m < n, then m appears in the sum.
                # k-m >= 0 iff m <= k
                # k-m < n iff m > k-n
                # So m appears iff max(0, k-n+1) <= m <= min(k, n-1) AND m is in [0, n)
                # Which is exactly [lo, hi] intersected with [0, n-1], which is [lo, hi] since lo >= 0 and hi <= n-1.
                # And for m in [lo, hi]: k-m is in [0, n-1] automatically? 
                # k-m >= 0: m <= k, and hi <= k, so yes.
                # k-m < n: m > k-n, and lo = max(0, k-n+1) > k-n, so yes.
                # So for all m in [lo, hi], both m and k-m are in [0, n).
                pass
        
        # Actually, the above loop is O(n * n_peaks) which is too slow for n=6962.
        # Let me use FFT instead.
        
        # Better approach: the subgradient of c_k w.r.t. a is 2 * (a * delta_k)[n-1:n+n-1] 
        # where delta_k is a spike at position k. This is just 2 * a reversed and shifted.
        # For the sum over top peaks: sg = 2 * sum_k (a reversed, shifted by k)
        
        # Even better: use the weighted convolution approach
        # sg = 2 * (a_rev * p) where p is the "peak weight" vector
        # This is what the LSE gradient does with p being the softmax of peaks
        
        # For the exact subgradient (uniform weight on top peaks):
        p = np.zeros(2 * n - 1)
        for k in top_idx:
            p[int(k)] = 1.0 / len(top_idx)
        
        # sg = 2 * (a_rev convolved with p) [n-1 : 2n-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(a[::-1], M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        sg = 2.0 * conv_res
        
        # Full gradient of score = 2n * c_max / S²
        # d(score)/da = (2n/S²) * d(c_max)/da - (2n * c_max * 2S)/S⁴ * da/da
        # = (2n/S²) * sg - (4n * c_max / S³) * 1
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n / S2) * sg - (4.0 * n * c_max / S3) * np.ones(n)
        
        # Polyak step size
        # alpha = (sc - lower_bound) / ||grad||²
        gn = float(np.dot(grad, grad))
        if gn < 1e-20:
            break
        alpha = min(0.1 * (sc - lower_bound) / gn, 0.5)
        alpha = max(alpha, 1e-8)
        
        if use_momentum:
            v = momentum * v + alpha * grad
            a_new = a - v
        else:
            a_new = a - alpha * grad
        
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
    
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a


def gs_adaptive(a0, n_iters=500, alpha_init=0.5, alpha_final=0.05):
    """GS-Fienup with adaptive convolution target that accounts for boundary effects."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        # Adaptive target: for the first few iterations, use flat target
        # Later, use a target that allows higher values at the boundaries
        c_current = fast_conv(a)
        c_max = float(c_current.max())
        c_mean = float(c_current.mean())
        
        # Target: start flat, then allow boundary peaks
        # target_k = c_mean * (1 + boundary_boost * exp(-k / width) + boundary_boost * exp(-(2n-2-k) / width))
        boundary_boost = 0.5 * progress  # increase boundary tolerance over time
        width = n / 3.0
        k = np.arange(2 * n - 1, dtype=np.float64)
        target = c_mean * (1.0 + boundary_boost * (np.exp(-k / width) + np.exp(-(2 * n - 2 - k) / width)))
        
        b_target = np.zeros(M, dtype=np.float64)
        b_target[:2 * n - 1] = np.maximum(target, c_mean * 0.5)
        
        T = np.fft.fft(b_target)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def coordinate_descent(a0, n_rounds=200, top_k=20):
    """Coordinate descent: reduce top-contributing elements of each peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contribution of a[j] to c[k]
            contrib = np.array([a[j] * a[k - j] for j in range(j_start, j_end + 1)])
            
            # Find the top contributing element
            max_contrib_idx = np.argmax(contrib)
            j = j_start + max_contrib_idx
            other = k - j
            
            # Reduce a[j] and increase a[other] (or a different element)
            # Actually, reduce a[j] slightly
            reduction = 0.01 * a[j]
            a[j] -= reduction
            # Add the mass to a random element that doesn't contribute much to this peak
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            if other != j:
                mask[other] = False
            # Also exclude elements that contribute to this peak
            for jj in range(j_start, j_end + 1):
                mask[jj] = False
            if mask.any():
                candidates = np.where(mask)[0]
                target_j = candidates[np.random.randint(len(candidates))]
                a[target_j] += reduction
        
        a = np.maximum(a, 0.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if (rnd + 1) % 50 == 0:
            print(f"    CD round {rnd+1}: score={best_sc:.8f}")
    
    return best_a
