import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    ess = exp_s.sum()
    p = exp_s / ess
    L_val = c_max + np.log(ess) / beta
    
    # grad_L[i] = sum_k p[k] * db_k/da_i = sum_k p[k] * 2*a[k-i] (if 0<=k-i<n)
    # = 2 * (p * a_rev)[n-1+i] ... let me use the convolution approach
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    P = np.fft.rfft(p.astype(np.float64), M3_fft)
    R = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(P * R, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    score = 2.0 * n * L_val / (S * S)
    grad = (2.0 * n / (S * S)) * grad_L - (4.0 * n * L_val / (S ** 3)) * 1.0
    return score, grad

def subsgrad_nesterov(a0, n_iters, deadline):
    """Nesterov-accelerated subgradient on the raw max."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    y = a.copy()
    t = 1.0
    lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(y)
        S = float(y.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient at the maximizing lag
        # grad_i = 4n*(a[k*-i]*S - M_val) / S^3 (if 0<=k*-i<n)
        grad = np.full(n, -4.0 * n * M_val / (S ** 3))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            # For i in [j_start, j_end]: k*-i is in [k*-j_end, k*-j_start] = [k*-min(k*,n-1), k*-max(0,k*-n+1)]
            # We need 0 <= k*-i <= n-1, i.e., k*-(n-1) <= i <= k*
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            if i_hi >= i_lo:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += 4.0 * n * y[k_star - idx] / (S * S)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, 0.02)
        step = max(step, 1e-12)
        
        x = y - step * grad
        x = np.maximum(x, 0.0)
        Sx = x.sum()
        if Sx < 1e-15:
            break
        x = x / Sx
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        
        a = x
        t = t_new
    
    return best_a, best_sc


def lbfgs_lse(a0, beta, maxiter=3000, deadline=None):

def symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return a_sym


def perturb_sin(a, rng, n_modes=5, scale=0.01):
    """Add sinusoidal perturbation."""
    n = len(a)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for m in range(n_modes):
        freq = rng.integers(2, 30)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / (m + 1)
        pert += amp * np.sin(2.0 * np.pi * freq * x + phase)
    return normalize(a * (1.0 + pert))


def perturb_sparse(a, rng, n_sparse=100, scale=0.005):
    """Add sparse random perturbation."""
    n = len(a)
    a_new = a.copy()
    idx = rng.choice(n, min(n_sparse, n), replace=False)
    a_new[idx] += scale * rng.standard_normal(len(idx))
    return normalize(a_new)


def downsample(a, factor):
    """Downsample by averaging over blocks."""
    n = len(a)
    new_n = n // factor
    a_padded = np.zeros(new_n * factor)
    a_padded[:n] = a
    return a_padded.reshape(new_n, factor).mean(axis=1)


def upsample(a, new_n):
    """Upsample via linear interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
FISTA + LSE annealing optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses fast iterative shrinkage-thresholding (FISTA) as the primary engine with
adaptive step sizes, aggressive multi-phase LSE annealing, multi-scale
exploration, diverse structured perturbations, and direct subgradient polish.
Key improvements over previous: faster per-iteration cost (FISTA vs L-BFGS),
more aggressive annealing schedule, symmetry exploitation, and SPSA-based
local search for escaping plateaus.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    ess = float(exp_s.sum())
    p = exp_s / ess
    L_val = c_max + np.log(ess) / beta

    # Gradient of L w.r.t. a: 2 * sum_k p[k] * a[k-i] (valid indices)
    a_rev = a[::-1]
    M3 = (2 * n - 1) + (n - 1)
    M3_fft = 1 << (M3 - 1).bit_length()
    P = np.fft.rfft(p.astype(np.float64), M3_fft)
    R = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(P * R, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    score = 2.0 * n * L_val / (S * S)
    grad = (2.0 * n / (S * S)) * grad_L - (4.0 * n * L_val / (S ** 3)) * 1.0
    return score, grad


def fista_lse(a0, beta, n_iters, deadline, verbose=False):
    """
    FISTA on LSE-smoothed objective with adaptive step size.
    Normalizes to sum=1 (score is scale-invariant).
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    y = a.copy()
    t = 1.0

    # Initial step size estimate
    _, g0 = lse_score_grad(y, beta)
    g0n = float(np.linalg.norm(g0))
    if g0n < 1e-15:
        return best_a, best_sc

    # Conservative step: for LSE with parameter beta, Lipschitz ~ O(n*beta)
    # Use adaptive: start small, grow if stable
    step = 1.0 / max(4.0 * n * min(beta, 200.0), g0n * 4.0)
    step = min(step, 0.5 / (g0n + 1e-10))
    step = max(step, 1e-10)

    no_improve = 0
    sc_threshold = best_sc

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        f, g = lse_score_grad(y, beta)

        # Gradient step with projection
        x = y - step * g
        x = np.maximum(x, 0.0)
        Sx = x.sum()
        if Sx < 1e-15:
            break
        x = x / Sx

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy > 1e-15:
            y = y / Sy

        a = x
        t = t_new

        # Periodic check and adaptive step
        if it % 5 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
                # Gradually grow step if improving
                step = min(step * 1.3, 2.0 / (g0n + 1e-10))
            else:
                no_improve += 1
                if no_improve > 20:
                    # Restart from best
                    a = best_a.copy()
                    y = best_a.copy()
                    t = 1.0
                    step = step * 0.7
                    no_improve = 0

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a, best_sc


def subsgrad_nesterov(a0, n_iters, deadline, top_k=1):
    """
    Nesterov-accelerated subgradient on the raw max (or top-k).
    Directly attacks the peak autocorrelation.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    y = a.copy()
    t = 1.0
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(y)
        S = float(y.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-8:
            break

        # Build subgradient from top-k peaks
        if top_k == 1:
            top_idx = np.array([k_star])
            weights = np.array([1.0])
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            weights = c[top_idx] / c[top_idx].sum()

        grad = np.full(n, -4.0 * n * np.sum(weights * c[top_idx]) / (S ** 3))
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += 4.0 * n * w * y[k - idx] / (S * S)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with cap
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, 0.015)
        step = max(step, 1e-12)

        x = y - step * grad
        x = np.maximum(x, 0.0)
        Sx = x.sum()
        if Sx < 1e-15:
            break
        x = x / Sx

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy > 1e-15:
            y = y / Sy

        a = x
        t = t_new

    return best_a, best_sc


def lbfgs_lse(a0, beta, maxiter=2000, deadline=None):
    """L-BFGS-B on LSE-smoothed objective for precise local optimization."""
    n = len(a0)
    a0 = normalize(a0.copy())

    def obj_grad(x):
        s, g = lse_score_grad(x, beta)
        return s, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12, 'maxiter': maxiter,
                     'maxfun': maxiter * 30, 'maxls': 50}
        )
        a_out = normalize(res.x)
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def symmetrize(a):
    """Enforce a[i] = a[n-1-i] (palindromic)."""
    return (a + a[::-1]) / 2.0


def perturb_sin(a, rng, n_modes=5, scale=0.01):
    """Add sinusoidal perturbation with random frequencies."""
    n = len(a)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for m in range(n_modes):
        freq = int(rng.integers(2, 40))
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / (m + 1)
        pert += amp * np.sin(2.0 * np.pi * freq * x + phase)
    return normalize(a * (1.0 + pert))


def perturb_sparse(a, rng, n_sp=150, scale=0.004):
    """Add sparse random perturbation."""
    n = len(a)
    a_new = a.copy()
    idx = rng.choice(n, min(n_sp, n), replace=False)
    a_new[idx] += scale * rng.standard_normal(len(idx))
    return normalize(a_new)


def perturb_poly(a, rng, deg=5, scale=0.008):
    """Add polynomial perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    coeffs = rng.standard_normal(deg + 1) * scale
    # Evaluate polynomial
    pert = np.zeros(n)
    for i, cf in enumerate(coeffs):
        pert += cf * (x ** i)
    # Remove mean to preserve approximate shape
    pert -= pert.mean()
    return normalize(a * (1.0 + pert))


def perturb_envelope(a, rng, scale=0.02):
    """Perturb the envelope (smooth random modulation)."""
    n = len(a)
    # Generate a smooth random modulation
    k_freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.5, 1.0, size=4) * scale
    x = np.arange(n) / n
    mod = np.zeros(n)
    for kf, ph, am in zip(k_freqs, phases, amps):
        mod += am * np.cos(2 * np.pi * kf * x + ph)
    return normalize(a * (1.0 + mod))


def downsample(a, factor):
    """Downsample by block averaging."""
    n = len(a)
    new_n = n // factor
    a_pad = np.zeros(new_n * factor)
    a_pad[:n] = a
    return a_pad.reshape(new_n, factor).mean(axis=1)


def upsample(a, new_n):
    """Upsample via linear interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def coord_refine(a0, n_sweeps=5, n_top=50, deadline=None):
    """
    Coordinate descent: reduce top contributors to the peak lag.
    Multiplicative updates with evaluation.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_hi < i_lo:
            break

        js = np.arange(i_lo, i_hi + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = js[top_local]

        for j in top_global:
            if deadline is not None and time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.4, 0.6, 0.75, 0.85, 0.92]:
                a[j] = a_old * f
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def spsa_local(a0, n_iters=2000, c0=0.02, deadline=None):
    """
    Simultaneous Perturbation Stochastic Approximation (SPSA).
    Good for escaping plateaus in non-smooth objectives.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    alpha = 0.1
    c = c0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # SPSA parameters (standard schedule)
        a_t = alpha / (10 + it + 1) ** 0.6
        c_t = c / (1 + it / 100.0) ** 0.6

        delta = np.where(np.random.random(n) < 0.5, 1.0, -1.0)

        a_plus = normalize(a + c_t * delta)
        a_minus = normalize(a - c_t * delta)

        f_plus = compute_score(a_plus)
        f_minus = compute_score(a_minus)

        if f_plus < best_sc:
            best_sc = f_plus
            best_a = a_plus.copy()
        if f_minus < best_sc:
            best_sc = f_minus
            best_a = a_minus.copy()

        g = (f_plus - f_minus) / (2.0 * c_t + 1e-30) * delta
        a = a - a_t * g
        a = normalize(a)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    1. FISTA + LSE annealing (fast, broad)
    2. L-BFGS refinement (precise, local)
    3. Multi-scale exploration (escape local minima)
    4. Diverse perturbations + local refine (global search)
    5. SPSA + subgradient (plateau escape)
    6. Final polish (coordinate descent + high-beta FISTA)
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            prev = best_score
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            improvement = prev - sc if np.isfinite(prev) else 0
            print(f"  [{label}] NEW BEST: {sc:.8f} (improved by {improvement:.6f}, n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        best_a = np.ones(10000) / 10000.0
        best_score = 2.0
        print("No parent. Starting uniform n=10000.")

    if best_a is None:
        best_a = np.ones(10000) / 10000.0

    # Try symmetrization
    a_sym = symmetrize(normalize(best_a))
    sc_sym = compute_score(a_sym)
    update(sc_sym, a_sym, "sym")

    a_work = normalize(best_a.copy())
    print(f"Starting best: {best_score:.8f}")

    # ============================================================
    # Phase 1: FISTA + LSE Annealing (35% time)
    # ============================================================
    phase1_end = t_start + 0.35 * (budget_s - 30.0)
    print(f"\n=== Phase 1: FISTA+LSE Annealing (to t={phase1_end - t_start:.0f}s) ===")

    betas_p1 = [15, 40, 100, 250, 600, 1500, 4000, 10000, 25000]
    for i, beta in enumerate(betas_p1):
        if time.monotonic() > phase1_end - 5:
            break
        remaining = phase1_end - 5 - time.monotonic()
        if remaining < 5:
            break
        # More iterations for mid-range betas
        weight = 1.0 + 0.5 * max(0, 1 - abs(i - 3) / 3.0)
        iters = int(min(8000, remaining * 40 * weight))
        iters = max(iters, 300)
        print(f"  FISTA beta={beta:>6}, iters={iters}")
        t0 = time.monotonic()
        a_out, sc = fista_lse(a_work, beta, iters, phase1_end - 2)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_out, f"F1 b={beta}")
        a_work = a_out.copy()

    # ============================================================
    # Phase 2: L-BFGS Precise Refinement (20% time)
    # ============================================================
    phase2_end = t_start + 0.55 * (budget_s - 30.0)
    print(f"\n=== Phase 2: L-BFGS Refinement ===")

    for beta in [5000, 20000, 80000]:
        if time.monotonic() > phase2_end - 10:
            break
        remaining = phase2_end - 10 - time.monotonic()
        if remaining < 15:
            break
        maxiter = max(200, int(remaining * 2.5))
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_out = lbfgs_lse(best_a, beta, maxiter=maxiter)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_out, f"L-BFGS b={beta}")

        # Also try from a slightly perturbed version
        if time.monotonic() < phase2_end - 15:
            a_p = perturb_sin(best_a, rng, n_modes=3, scale=0.003)
            maxiter2 = max(100, int((phase2_end - 15 - time.monotonic()) * 2))
            sc2, a_out2 = lbfgs_lse(a_p, beta, maxiter=maxiter2)
            update(sc2, a_out2, f"L-BFGS+P b={beta}")

    # ============================================================
    # Phase 3: Multi-scale Exploration (15% time)
    # ============================================================
    phase3_end = t_start + 0.70 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Multi-scale Exploration ===")

    for fac in [2, 4]:
        if time.monotonic() > phase3_end - 20:
            break
        small_n = len(best_a) // fac
        if small_n < 256:
            continue
        a_small = downsample(best_a, fac)
        print(f"  n={small_n} (factor {fac})")

        for beta in [200, 1000, 5000]:
            if time.monotonic() > phase3_end - 10:
                break
            remaining = phase3_end - 10 - time.monotonic()
            if remaining < 10:
                break
            iters = int(min(20000, remaining * 80))
            a_opt, sc = fista_lse(a_small, beta, iters, phase3_end - 5)

            # Upsample and refine
            a_up = upsample(a_opt, len(best_a))
            a_up = normalize(a_up)
            sc_up = compute_score(a_up)
            print(f"    n={small_n} b={beta}: small={sc:.8f}, upsampled={sc_up:.8f}")
            update(sc_up, a_up, f"SCALE {fac}x b={beta}")

            # Refine at full scale
            if time.monotonic() < phase3_end - 5:
                sc_ref, a_ref = fista_lse(a_up, 10000, 3000, phase3_end - 2)
                update(sc_ref, a_ref, f"SCALE+REF {fac}x")

    # ============================================================
    # Phase 4: Diverse Perturbations + Local Refine (15% time)
    # ============================================================
    phase4_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Perturbations + Refine ===")

    n_trials = 30
    pert_types = [perturb_sin, perturb_sparse, perturb_poly,
                  perturb_envelope, perturb_sin]

    for r in range(n_trials):
        if time.monotonic() > phase4_end - 5:
            break
        remaining = phase4_end - 5 - time.monotonic()
        if remaining < 10:
            break

        pt = r % len(pert_types)
        if pt == 0:
            a_p = perturb_sin(best_a, rng, n_modes=3 + (r % 5),
                              scale=0.003 + 0.002 * (r % 4))
        elif pt == 1:
            a_p = perturb_sparse(best_a, rng, n_sp=50 + 50 * (r % 5),
                                 scale=0.002 + 0.002 * (r % 3))
        elif pt == 2:
            a_p = perturb_poly(best_a, rng, deg=3 + (r % 4),
                               scale=0.004 + 0.003 * (r % 3))
        elif pt == 3:
            a_p = perturb_envelope(best_a, rng, scale=0.01 + 0.01 * (r % 3))
        else:
            a_p = perturb_sin(best_a, rng, n_modes=8, scale=0.002)

        # FISTA refinement
        iters = max(800, int(remaining * 25))
        iters = min(iters, 5000)
        beta = 3000 + 2000 * (r % 4)
        a_opt, sc = fista_lse(a_p, beta, iters, phase4_end - 2)
        update(sc, a_opt, f"P4-{r}")

        # occasionally also try subgradient
        if r % 5 == 4 and time.monotonic() < phase4_end - 5:
            a_sg, sc_sg = subsgrad_nesterov(a_opt, 2000, phase4_end - 3, top_k=5)
            update(sc_sg, a_sg, f"P4-SG-{r}")

    # ============================================================
    # Phase 5: SPSA + Subgradient Plateau Escape (5% time)
    # ============================================================
    phase5_end = t_start + 0.92 * (budget_s - 30.0)
    print(f"\n=== Phase 5: SPSA + Subgradient ===")

    if time.monotonic() < phase5_end - 5:
        # SPSA from best
        a_sp, sc_sp = spsa_local(best_a, n_iters=1500, c0=0.03,
                                  deadline=phase5_end - 10)
        update(sc_sp, a_sp, "SPSA")
        # Refine SPSA result
        if sc_sp < best_score and time.monotonic() < phase5_end - 5:
            a_rf, sc_rf = fista_lse(a_sp, 10000, 2000, phase5_end - 3)
            update(sc_rf, a_rf, "SPSA+REF")

    # Subgradient with top-k=5
    if time.monotonic() < phase5_end - 3:
        a_sg5, sc_sg5 = subsgrad_nesterov(best_a, 5000, phase5_end - 2, top_k=5)
        update(sc_sg5, a_sg5, "SG k=5")

    # ============================================================
    # Phase 6: Final Polish (3% time)
    # ============================================================
    print(f"\n=== Phase 6: Final Polish ===")

    # Coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd, sc_cd = coord_refine(best_a, n_sweeps=5, n_top=40,
                                    deadline=deadline - 10)
        update(sc_cd, a_cd, "COORD")

    # Very high beta FISTA
    if time.monotonic() < deadline - 5:
        a_f, sc_f = fista_lse(best_a, 100000, 3000, deadline - 3)
        update(sc_f, a_f, "FINAL-FISTA-100k")

    # Symmetrization of final best
    a_s = symmetrize(best_a)
    sc_s = compute_score(a_s)
    update(sc_s, a_s, "SYM-FINAL")

    # Final L-BFGS with very high beta if time allows
    if time.monotonic() < deadline - 8:
        sc_lb, a_lb = lbfgs_lse(best_a, 200000, maxiter=1000)
        update(sc_lb, a_lb, "FINAL-LBFGS-200k")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 10000

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
