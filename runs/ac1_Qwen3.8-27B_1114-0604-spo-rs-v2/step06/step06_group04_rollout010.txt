Minimizing $a \cdot s$ subject to $a \ge 0$, $\sum a = 1$ is just: put all mass on the coordinate with the smallest $s_i$. That's too extreme.

Better: minimize $(a - a_0) \cdot s + \frac{\rho}{2}\|a - a_0\|^2$ subject to $a \ge 0$, $\sum a = 1$. This is a QP with a single linear constraint and box constraints. The solution is:
$a_i = \text{clip}(a_{0,i} - s_i/\rho - \mu, 0, a_{0,i})$ where $\mu$ is chosen so $\sum a_i = 1$.

Wait, that's not quite right. The KKT conditions for $\min (a-a_0) \cdot s + \frac{\rho}{2}\|a-a_0\|^2$ s.t. $a \ge 0$, $1^T a = 1$:
$\rho(a - a_0) + s + \mu 1 - \nu = 0$ (where $\nu \ge 0$ is for non-negativity)
$a_i > 0 \implies \nu_i = 0$
$a_i = 0 \implies \nu_i \ge 0$

So $a_i = \max(0, a_{0,i} - s_i/\rho - \mu/\rho)$ and $\mu$ is chosen for the sum constraint.

This is a "water-filling" solution. I can implement it with binary search on $\mu$.

But wait, I want to minimize the max, not the sum. Minimizing the sum of the active peaks makes them more equal (by the pigeonhole principle, if the total is fixed and you minimize the sum, they equalize). But the "peaks" outside the active set might grow.

A better approach: minimize $\max(c_k)$ directly using a bisection on the target value $t$:
- Binary search on $t$: is there $a \ge 0$, $\sum a = 1$ with $c_k \le t$ for all $k$?
- The feasibility check is a QP: $\min \sum_k \max(0, c_k - t)$ s.t. $a \ge 0$, $\sum a = 1$. If the min is 0, it's feasible.
- But this is non-convex (quadratic constraints).

Alternatively, for the active set $\mathcal{A}$ only: $\min \max_{k \in \mathcal{A}} c_k$ s.t. $a \ge 0$, $\sum a = 1$, $\|a - a_0\| \le \epsilon$. This is a trust-region QP with a max objective. Can be solved by bisection on $t$ and checking feasibility of $c_k \le t$ for $k \in \mathcal{A}$.

For simplicity, I'll use the "sum of active peaks" approach as an approximation. It's not exact but should help.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Diverse structured initialization + low-frequency parameterization
2. Progressive LSE smoothing with L-BFGS-B
3. Peak-aware Nesterov with adaptive active set and Polyak-like steps
4. Active-set flattening via water-filling QP
5. Ultra-fine final polishing with single-peak focus

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _active_flatten(a, k_peaks=8, tol_rel=0.01):
    """
    Active-set flattening: find peaks near max, compute their gradients,
    and take a water-filling step to equalize/reduce them.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = _fast_self_conv(a)
    cmax = float(c.max())
    threshold = cmax * (1.0 - tol_rel)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    active = active[:k_peaks]

    # Compute "pressure" vector: s_i = sum_{k in active} a_{k-i} (with zero pad)
    N2 = _next_pow2(3 * n)
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    s = conv_full[n - 1:2 * n - 1]

    # Water-filling: a_new_i = clip(a_i - eta * s_i, 0, None), normalize
    eta = 0.05
    a_new = np.clip(a - eta * s, 0.0, None)
    Snew = float(a_new.sum())
    if Snew < 1e-12:
        return a
    a_new /= Snew
    return a_new


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6,
                       use_flatten_every=100):
    """
    Nesterov accelerated subgradient method with adaptive active set
    and periodic active-flattening.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t = 1.0
    y = a.copy()

    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = float(np.linalg.norm(g0))
    if g_norm < 1e-12:
        return best_score, best_a

    # Polyak-like: estimate optimal value as lower bound
    # Lower bound: 2n * (1/(2n-1)) = 2n/(2n-1) ~ 1
    f_opt = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound for score
    f_opt = max(f_opt, best_score - 0.05)  # practical lower bound

    eta0 = 0.03 / g_norm
    prev_score = best_score
    stale = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * (a - y + v) if it > 0 else a.copy()
        # Corrected Nesterov: y = a + (t-1)/(t_new) * (a - a_prev)
        # Simplified: use standard Nesterov
        y = a + beta_mom * y if it > 0 else a.copy()

        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy

        score_y, grad_y = _subgrad_exact(y, k_active, tol)
        if not np.isfinite(score_y):
            break

        # Polyak-like step size
        gap = score_y - f_opt
        gn = float(np.linalg.norm(grad_y))
        if gn < 1e-14:
            break
        eta = min(eta0 / (1.0 + 0.05 * it ** 0.5), max(1e-8, gap / (gn * gn + 1e-12) * 0.5))
        eta = min(eta, 0.1)  # cap step size

        x_new = y - eta * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t = t_new

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        # Periodic active flattening
        if use_flatten_every > 0 and it > 0 and it % use_flatten_every == 0:
            a_flat = _active_flatten(a, k_peaks=6, tol_rel=0.02)
            sc_flat = _exact_score(a_flat)
            if sc_flat < best_score:
                best_score = sc_flat
                best_a = a_flat.copy()
                a = a_flat

        # If stuck, perturb slightly
        if stale > 200:
            rng_local = np.random.default_rng(it)
            a = _normalize(a + 0.01 * rng_local.standard_normal(n))
            stale = 0

    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    a = _normalize(a0)
    n = a.size
    if maxfun is None:
        maxfun = 300

    _cache = {'calls': 0}

    def fun(x):
        _cache['calls'] += 1
        if deadline is not None and time.monotonic() > float(deadline):
            raise ValueError("timeout")
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _lowfreq_optimize(a0, n_coeffs, deadline):
    """
    Optimize using a low-frequency cosine parameterization.
    a_i = d0 + sum_{k=1}^{K} c_k * cos(2*pi*k*i/n)
    This reduces dimensionality and helps escape local minima.
    """
    a = np.asarray(a0, dtype=np.float64).ravel()
    n = a.size

    # Fit cosine coefficients
    x = np.arange(n)
    A = np.ones((n, n_coeffs + 1))
    for k in range(1, n_coeffs + 1):
        A[:, k] = np.cos(2.0 * np.pi * k * x / n)

    # Least squares fit (allowing negative values temporarily)
    coeffs, _, _, _ = np.linalg.lstsq(A, a, rcond=None)

    def _eval_params(params):
        d0 = params[0]
        cs = params[1:]
        af = np.full(n, d0)
        for k in range(1, n_coeffs + 1):
            af += cs[k - 1] * np.cos(2.0 * np.pi * k * x / n)
        af = np.clip(af, 0.0, None)
        S = af.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n_coeffs + 1)
        af = af / S
        sc = _exact_score(af)
        # Approximate gradient via finite differences on the LSE-smoothed objective
        # Actually, let's use the chain rule: grad w.r.t. params
        c = _fast_self_conv(af)
        d = n * c / (S * S)  # S is already 1 after normalization
        dmax = float(d.max())
        # Subgradient of max
        idx = np.argmax(d)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        N2 = _next_pow2(3 * n)
        a_rev = af[::-1]
        AR = np.fft.rfft(a_rev, N2)
        WR = np.fft.rfft(w, N2)
        conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
        grad_a = conv_full[n - 1:2 * n - 1]
        # Chain rule: d(af)/dparams
        grad_params = A.T @ grad_a  # (n_coeffs+1,)
        # Subtract mean to handle normalization (approximate)
        grad_params -= np.mean(grad_params) * A.T @ np.ones(n) / n
        # Actually, for simplicity, just use the direct gradient
        grad_params = np.nan_to_num(grad_params, nan=0.0)
        return 2.0 * dmax, grad_params * (2.0 * n)

    def obj_gradient(params):
        d0 = params[0]
        cs = params[1:]
        af = np.full(n, d0)
        for k in range(1, n_coeffs + 1):
            af += cs[k - 1] * np.cos(2.0 * np.pi * k * x / n)
        af = np.clip(af, 0.0, None)
        S = af.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n_coeffs + 1)
        af = af / S
        c = _fast_self_conv(af)
        d = n * c / (S * S)
        dmax = float(d.max())
        # Use LSE with moderate beta for smoothness
        beta = 500.0
        z = beta * (d - dmax)
        z = np.clip(z, -300, 300)
        e = np.exp(z)
        se = e.sum()
        w = e / se
        T = w.dot(c)
        N2 = _next_pow2(3 * n)
        a_rev = af[::-1]
        AR = np.fft.rfft(a_rev, N2)
        WR = np.fft.rfft(w, N2)
        conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
        R = conv_full[n - 1:2 * n - 1]
        grad_a = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
        grad_params = A.T @ grad_a
        grad_params = np.nan_to_num(grad_params, nan=0.0)
        score = 2.0 * (dmax + np.log(se) / beta)
        return score, grad_params

    p0 = coeffs.copy()
    try:
        res = minimize(obj_gradient, p0, method="L-BFGS-B", jac=True,
                       options={"maxiter": 200, "ftol": 1e-12, "gtol": 1e-10,
                                "maxfun": 500})
        if res.x is not None:
            p_out = res.x
            af = np.full(n, p_out[0])
            for k in range(1, n_coeffs + 1):
                af += p_out[k] * np.cos(2.0 * np.pi * k * x / n)
            af = _normalize(af)
            sc = _exact_score(af)
            return sc, af
    except Exception:
        pass
    return _exact_score(a), a


def _generate_inits(n, rng, parent, cap=30):
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
        _add(0.7 * pr + 0.3 * pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Cosine shapes (key for this problem)
    for alpha in [0.2, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]:
        _add(1.0 + alpha * np.cos(np.pi * x))
        _add(1.0 + alpha * np.cos(2.0 * np.pi * x))
        _add(1.0 - alpha * np.cos(np.pi * x))

    # Combined shapes
    for a1 in [0.3, 0.5, 0.7]:
        for a2 in [0.1, 0.2, 0.3]:
            _add(1.0 + a1 * np.cos(np.pi * x) + a2 * np.cos(2.0 * np.pi * x))
            _add(1.0 + a1 * np.cos(np.pi * x) - a2 * np.cos(2.0 * np.pi * x))

    # V-shapes
    for alpha in [0.3, 0.5, 0.8, 1.2]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Ramp
    _add(x + 0.1)
    _add(1.0 - x + 0.1)

    # Uniform
    _add(np.ones(n))

    # Bimodal
    for w in [0.1, 0.15, 0.25]:
        for ctr in [0.2, 0.3, 0.35]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 2.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.2)) * noise)

    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    return inits


def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 5))

    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = float(np.linalg.norm(g))
            if gn > 1e-12:
                q = a - strength * 0.5 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Low-frequency noise
        x = np.arange(n)
        K = int(rng.integers(2, 8))
        noise = np.zeros(n)
        for k in range(1, K + 1):
            amp = rng.uniform(0.1, 0.5) / k
            noise += amp * np.cos(2.0 * np.pi * k * x / n + rng.uniform(0, 2 * np.pi))
        q = a * np.exp(strength * 0.3 * noise)
    else:
        q = a.copy()
        n_swaps = int(rng.integers(5, max(10, n // 50)))
        for _ in range(n_swaps):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = a * np.exp(0.03 * rng.standard_normal(n))

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize(a, deadline):
    """Quick multi-phase optimization for a candidate."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    # LSE stages

            break

            best_a = a2.copy()
        a = _normalize(a2)

    # Nesterov
    rem = deadline - time.monotonic()
    if rem > 15.0:
        n_it = int(max(200, rem * 25))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                    k_active=3, tol=1e-7, use_flatten_every=80)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    return best_score, best_a


def _intensive_optimize(a, deadline, label=""):
    """
    Full intensive optimization pipeline.
    """
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()

    print("[{}] start: score={:.10f} n={}".format(label, best_score, a.size))

    # Stage 1: LSE with progressive beta
    rem_total = deadline - t0
    lse_end = t0 + 0.30 * rem_total
    betas = [200, 1000, 5000, 20000, 100000, 400000]
    for beta in betas:
        rem = lse_end - time.monotonic()
        if rem < 15.0:
            break
        maxfun = int(max(50, min(800, rem * 12)))
        sc, a2 = _lbfgs_lse(a, beta, lse_end, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] LSE beta={:.0f}: {:.10f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Low-frequency re-optimization (escape local minima)
    rem = deadline - time.monotonic()
    if rem > 60.0:
        n_coeffs = min(30, a.size // 100)
        lf_dead = time.monotonic() + min(40.0, rem * 0.15)
        sc_lf, a_lf = _lowfreq_optimize(a, n_coeffs, lf_dead)
        if np.isfinite(sc_lf) and sc_lf < best_score:
            best_score = sc_lf
            best_a = a_lf.copy()
            print("[{}] LowFreq: {:.10f}".format(label, best_score))
        a = _normalize(a_lf)

    # Stage 2: Nesterov with adaptive k
    nesterov_end = t0 + 0.65 * rem_total
    for k_act in [1, 2, 3, 5]:
        rem = nesterov_end - time.monotonic()
        if rem < 15.0:
            break
        n_it = int(max(300, rem * 20))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=nesterov_end,
                                    k_active=k_act, tol=1e-7, use_flatten_every=100)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] Nesterov k={}: {:.10f}".format(label, k_act, best_score))
        a = _normalize(a2)

    # Stage 3: Active-set flattening + LSE polish
    polish_end = t0 + 0.85 * rem_total
    for _ in range(3):
        rem = polish_end - time.monotonic()
        if rem < 15.0:
            break
        a_flat = _active_flatten(a, k_peaks=10, tol_rel=0.005)
        sc_flat = _exact_score(a_flat)
        if sc_flat < best_score:
            best_score = sc_flat
            best_a = a_flat.copy()
            a = a_flat
        for beta in [200000, 800000]:
            rem = polish_end - time.monotonic()
            if rem < 10.0:
                break
            maxfun = int(max(30, min(500, rem * 10)))
            sc, a2 = _lbfgs_lse(a, beta, polish_end, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Polish LSE beta={:.0f}: {:.10f}".format(label, beta, best_score))
            a = _normalize(a2)

    # Stage 4: Ultra-fine Nesterov (single peak focus)
    rem = deadline - time.monotonic()
    if rem > 20.0:
        n_it = int(max(500, rem * 30))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                    k_active=1, tol=1e-9, use_flatten_every=200)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] Final Nesterov k=1: {:.10f}".format(label, best_score))
        a = _normalize(a2)

    # Final active flattening
    a_flat = _active_flatten(a, k_peaks=12, tol_rel=0.002)
    sc_flat = _exact_score(a_flat)
    if sc_flat < best_score:
        best_score = sc_flat
        best_a = a_flat.copy()
        print("[{}] Final flatten: {:.10f}".format(label, best_score))

    print("[{}] done: score={:.10f} time={:.1f}s".format(label, best_score, time.monotonic() - t0))
    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns list of non-negative floats minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    Uses progressive LSE + Nesterov + active-set flattening + low-freq re-optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 60.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = min(parent.size, 16384)
        if parent_len < 2048:
            parent_len = 4096
        lengths = [parent_len, 10240, 8192]
    else:
        lengths = [10240, 8192, 12288]
    lengths = list(dict.fromkeys(lengths))  # deduplicate preserving order
    lengths = [L for L in lengths if 2048 <= L <= 16384][:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate and evaluate inits
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = _generate_inits(n, rng, parent, cap=25)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)
                all_lens.append(n)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(all_scores)
    print("Best init: {:.10f} (len={})".format(all_scores[order[0]], all_lens[order[0]]))

    # Select top starts (prefer different lengths and different shapes)
    starts = []
    used_lens = set()
    for idx in order[:15]:
        if len(starts) >= 4:
            break
        L = all_lens[idx]
        # Allow same length but different shape
        if len(starts) < 2 or L not in used_lens:
            starts.append(all_seqs[idx].copy())
            if len(starts) >= 2:
                used_lens.add(L)

    while len(starts) < 3:
        starts.append(_perturb(starts[0], rng))

    # Initial best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    print("Initial best: {:.10f}".format(best_score))

    # Phase 2: Main optimization on top starts
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))

    # Time allocation
    main_end = deadline - 500.0
    main_budget = max(200.0, main_end - t_start)

    if len(starts_sorted) == 1:
        fracs = [1.0]
    elif len(starts_sorted) == 2:
        fracs = [0.65, 0.35]
    elif len(starts_sorted) == 3:
        fracs = [0.50, 0.30, 0.20]
    else:
        fracs = [0.40, 0.25, 0.20, 0.15]

    cum = 0.0
    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 30.0:
            continue

        sq = starts[si].copy()

        # Blend secondary starts with current best
        if rank > 0 and len(best_seq) != len(sq):
            best_resampled = _resample(best_seq, len(sq))
            sq = _normalize(0.6 * _normalize(sq) + 0.4 * _normalize(best_resampled))

        sc, seq = _intensive_optimize(sq, sub_dead, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.10f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: Smart perturbation restarts
    n_perturb = 6
    perturb_end = deadline - 400.0
    if perturb_end > time.monotonic() + 80.0:
        chunk = (perturb_end - time.monotonic()) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = time.monotonic() + chunk
            if chunk_dead > perturb_end:
                chunk_dead = perturb_end
            if time.monotonic() > chunk_dead - 5.0:
                break

            # Smart perturbation: use gradient-informed direction
            p = _perturb(best_seq, rng, strength=float(rng.uniform(0.05, 0.2)))
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.10f}".format(pi, best_score))

    # Phase 4: Final intensive polish on the best sequence
    final_dead = deadline - 20.0
    if time.monotonic() < final_dead - 60.0:
        sc, seq = _intensive_optimize(best_seq, final_dead, "final")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Final micro-polish: active flattening + tiny Nesterov
    for _ in range(3):
        a_flat = _active_flatten(best_seq, k_peaks=15, tol_rel=0.001)
        sc_f = _exact_score(a_flat)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_flat.copy()

    rem = final_dead - time.monotonic()
    if rem > 10.0:
        n_it = int(max(200, rem * 30))
        sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                    k_active=1, tol=1e-10, use_flatten_every=150)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
        best_seq = _normalize(a2)

    # Final output formatting
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
