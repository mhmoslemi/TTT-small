def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a, left=0.0, right=0.0).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + 1e-3 * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * 1e-3 * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def estimate_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 5.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    mf = max(64, min(5000, mf))
    return mf


def subgrad_at_k_fast(p, k_star):
    """Compute subgradient of (p*p)[k_star] w.r.t. p.
    Result: grad[j] = 2*p[k_star-j] if 0 <= k_star-j < n, else 0."""
    n = len(p)
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n, k_star + 1)
    if j_start < j_end:
        idxs = k_star - np.arange(j_start, j_end)
        idxs = idxs[(idxs >= 0) & (idxs < n)]
        valid_j = np.arange(j_start, j_end)
        valid_j = valid_j[(k_star - valid_j >= 0) & (k_star - valid_j < n)]
        grad[valid_j] = 2.0 * p[k_star - valid_j]
    return grad


def spectral_flattening(a, n_iter=20, alpha=0.5, deadline=None):
    """Iteratively flatten the spectrum of a while maintaining non-negativity.
    Uses soft magnitude equalization in the DFT domain."""
    a = normalize_a(a.copy())
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute target: geometric mean of magnitudes (soft flatten)
        log_mag = np.log(mag + 1e-30)
        target_mag = np.exp(np.mean(log_mag))
        
        # Blend: new_mag = mag^(1-alpha) * target_mag^alpha
        new_mag = mag ** (1.0 - alpha) * target_mag ** alpha
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Soft projection: exponential reweighting toward non-negative
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return float(best_sc), best_a


def spectral_flattening_hard(a, n_iter=30, deadline=None):
    """Hard spectral flattening: set all magnitudes to their mean."""
    a = normalize_a(a.copy())
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Hard: set all to mean magnitude
        mean_mag = float(np.mean(mag))
        A_new = mean_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
    
    return float(best_sc), best_a


def exact_subgrad_method(a, max_iters=100, deadline=None, k_list=None):
    """Exact subgradient method for the max-convolution objective."""
    a = normalize_a(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    eta_decay = 0.95
    eta = 0.1
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        d, S = d_from_a(a)
        if d.size == 0:
            break
        
        # Find active set (top-k lags)
        if k_list is None:
            k_list = [1, 3, 5]
        
        improved = False
        for k_active in k_list:
            # Get top-k lags
            m = len(d)
            if k_active >= m:
                idxs = np.arange(m)
            else:
                idxs = np.argpartition(d, m - k_active)[m - k_active:]
            
            # Average subgradient over active set
            grad = np.zeros(n)
            for ki in idxs:
                grad += subgrad_at_k_fast(a, int(ki))
            grad /= float(len(idxs))
            
            # Normalized step
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                continue
            
            # Try several step sizes
            for scale in (eta, eta * 0.5, eta * 0.1, eta * 2.0):
                a_trial = a - scale * grad / gnorm
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize_a(a_trial)
                sc = exact_score_a(a_trial)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
        
        if not improved and it > 10:
            eta *= eta_decay
    
    return float(best_sc), best_a


def peak_shaving(a, max_iters=100, deadline=None):
    """Targeted peak shaving: reduce contributions to the maximum lag."""
    a = normalize_a(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        S = float(a.sum())
        
        # Compute contribution of each pair (i, k_star-i)
        # c[k_star] = sum_i a[i]*a[k_star-i]
        # Reduce the largest contributing pairs
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                pairs.append((contrib, i, j))
        
        if not pairs:
            break
        
        pairs.sort(reverse=True)
        
        # Reduce top contributions
        improved = False
        n_reduce = max(1, min(10, len(pairs) // 4))
        
        for contrib, i, j in pairs[:n_reduce]:
            if contrib < 1e-15:
                break
            
            # Reduce the larger of a[i], a[j]
            if a[i] >= a[j]:
                # Reduce a[i]
                for factor in (0.95, 0.9, 0.8):
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = normalize_a(a_trial)
                    sc = exact_score_a(a_trial)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
            else:
                # Reduce a[j]
                for factor in (0.95, 0.9, 0.8):
                    a_trial = a.copy()
                    a_trial[j] *= factor
                    a_trial = normalize_a(a_trial)
                    sc = exact_score_a(a_trial)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
        
        if not improved and it > 20:
            break
    
    return float(best_sc), best_a


def coordinate_descent(a, max_iters=200, deadline=None, window=200):
    """Coordinate descent: optimize small windows of coordinates."""
    a = normalize_a(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Pick a random window to optimize
        start = int(np.random.randint(0, max(1, n - window)))
        end = min(n, start + window)
        
        # Try perturbations in this window
        improved = False
        for _ in range(5):
            a_trial = a.copy()
            delta = np.random.uniform(-0.1, 0.1, end - start)
            a_trial[start:end] = np.maximum(a_trial[start:end] + delta, 0.0)
            a_trial = normalize_a(a_trial)
            sc = exact_score_a(a_trial)
            if np.isfinite(sc) and sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
        
        if not improved and it > 50:
            break
    
    return float(best_sc), best_a


def generate_asymmetric_inits(n, rng, parent=None, cap=20):
    """Generate diverse ASYMMETRIC initializations."""
    inits = []
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        arr = np.maximum(arr, 0.0)
        if arr.sum() < 1e-12:
            arr = np.ones(n)
        p = normalize_a(arr)
        inits.append(p)
    
    x = np.linspace(0.0, 1.0, n)
    
    # Parent
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr * (1.0 + 0.3 * x))  # Asymmetric weighting
        add(pr * (1.0 + 0.3 * (1.0 - x)))
    
    # Linear ramps (key asymmetric structures)
    for slope in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(1.0 + slope * x)
        add(1.0 + slope * (1.0 - x))
        add(np.exp(slope * x))
        add(np.exp(slope * (1.0 - x)))
        add(np.exp(slope * (x - 0.5)) + 0.1)
    
    # Off-center Gaussians
    for center in (0.2, 0.3, 0.4, 0.6, 0.7, 0.8):
        for width in (0.05, 0.1, 0.15, 0.25, 0.4):
            add(np.exp(-(((x - center) / width) ** 2)) + 0.01)
    
    # Two asymmetric peaks
    for c1, c2 in ((0.2, 0.6), (0.3, 0.7), (0.1, 0.5), (0.4, 0.8), (0.15, 0.55)):
        add(np.exp(-(((x - c1) / 0.1) ** 2)) + 0.5 * np.exp(-(((x - c2) / 0.1) ** 2)) + 0.01)
    
    # Bathtub shapes (low at edges, high in middle) - but asymmetric
    for asym in (0.3, 0.5, 0.7):
        add(1.0 + 2.0 * x * (1.0 - x) * (asym + (1.0 - asym) * np.sin(np.pi * x)))
        add(1.0 + 3.0 * np.minimum(x, (1.0 - x) * asym))
    
    # Power law shapes
    for p in (0.5, 1.0, 2.0, 3.0):
        add(x ** p + 0.01)
        add((1.0 - x) ** p + 0.01)
        add(0.5 * x ** p + 0.5 * (1.0 - x) ** (p + 0.5) + 0.01)
    
    # Sine-based asymmetric
    for f in (1, 2, 3):
        add(1.0 + 0.5 * np.sin(np.pi * f * x) + 0.3 * x)
        add(1.0 + 0.5 * np.sin(np.pi * f * x + np.pi/4) + 0.2)
    
    # Random smooth (Fourier-based)
    Nf = next_power_of_two(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        decay = float(rng.uniform(1.5, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** decay)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.3, 1.5))
        add(1.0 + amp * noise)
    
    return inits


def perturb_best(a, rng, strength=None):
    """Generate a perturbation of the best sequence."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 10))
    
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Asymmetric multiplicative perturbation
        x = np.linspace(0, 1, n)
        q = p * np.exp(strength * 0.5 * rng.standard_normal(1) * x)
    elif mode == 2:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 3:
        # Reverse and perturb
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 4:
        # Smoothed noise
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, n // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - noise.mean()
        logp = logp + strength * 0.5 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # Shift the peak
        shift = int(rng.integers(-n // 10, n // 10 + 1))
        q = np.roll(p, shift)
        # Fade edges
        fade = np.ones(n)
        edge = max(1, n // 20)
        for i in range(edge):
            fade[i] = i / edge
            fade[n-1-i] = i / edge
        q = q * fade
    elif mode == 6:
        # Localized perturbation
        center = int(rng.integers(0, n))
        width = int(rng.integers(n // 20, n // 5))
        mask = np.exp(-(((np.arange(n) - center) / width) ** 2))
        q = p * (1.0 + strength * mask * rng.standard_normal(1))
    elif mode == 7:
        # Logistic warp
        t = float(rng.uniform(0.5, 2.0))
        x = np.linspace(0, 1, n)
        warped = 1.0 / (1.0 + np.exp(-t * (2*x - 1)))
        q = resample_sequence(p, n) * (0.5 + 0.5 * warped)
    else:
        # Random recency bias
        decay = float(rng.uniform(0.5, 3.0))
        bias = np.exp(-decay * np.linspace(0, 1, n))
        q = p * (1.0 + 0.3 * (bias - bias.mean() / bias.std() + 1))
    
    return normalize_a(np.clip(q, 0.0, None))


def optimize_trajectory(a_start, deadline, betas=(200, 1000, 5000, 20000, 100000, 500000)):
    """Run LSE optimization with increasing beta."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    t0 = time.monotonic()
    total = float(max(10.0, deadline - t0))
    n_betas = len(betas)
    
    for i, beta in enumerate(betas):
        sub_dead = t0 + ((i + 1) / n_betas) * total
        sub_dead = min(sub_dead, deadline - 3.0)
        
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, len(a))
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)
    
    return float(best_sc), best_a

    candidates = []
    time_per_init = max(10.0, (t1_end - t_start) / len(inits))
    
    for i, init in enumerate(inits):
        if time.monotonic() > t1_end - 10.0:
            break
        sub_dead = min(t1_end - 5.0, time.monotonic() + time_per_init)
        
        sc, seq = optimize_trajectory(init, sub_dead, betas=(500, 5000, 50000))
        sc_exact = exact_score_a(seq)
        candidates.append((sc_exact, seq.copy()))
        print("Init {}: score={:.8f}".format(i, sc_exact))
    
    if not candidates:
        a0 = normalize_a(inits[0]) if inits else np.ones(n_primary) / n_primary
        candidates = [(exact_score_a(a0), a0)]
    
    candidates.sort(key=lambda x: x[0])
    best_sc = candidates[0][0]
    best_seq = candidates[0][1].copy()
    print("Phase 1 best: {:.8f}".format(best_sc))
    
    # Phase 2: Deep-optimize top candidates (30% time)
    t2_end = t_start + 0.50 * total_budget
    top_k = min(5, len(candidates))
    time_per_cand = max(20.0, (t2_end - t_start) / top_k)
    
    for i in range(top_k):
        if time.monotonic() > t2_end - 15.0:
            break
        sub_dead = min(t2_end - 10.0, time.monotonic() + time_per_cand)
        
        a = candidates[i][1].copy()
        sc, seq = optimize_trajectory(a, sub_dead, betas=(10000, 50000, 200000, 1000000))
        sc_exact = exact_score_a(seq)
        
        if sc_exact < best_sc:
            best_sc = sc_exact
            best_seq = seq.copy()
            print("Phase 2 cand {}: {:.8f}".format(i, best_sc))
    
    # Phase 3: Spectral flattening on best (10% time)
    t3_end = t_start + 0.60 * total_budget
    if time.monotonic() < t3_end - 10.0:
        sc, seq = spectral_flattening(best_seq, n_iter=30, alpha=0.3, deadline=t3_end)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 3 (spec soft): {:.8f}".format(best_sc))
        
        if time.monotonic() < t3_end - 5.0:
            sc, seq = spectral_flattening_hard(best_seq, n_iter=20, deadline=t3_end)
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("Phase 3 (spec hard): {:.8f}".format(best_sc))
    
    # Phase 4: Peak shaving + subgradient (15% time)
    t4_end = t_start + 0.75 * total_budget
    if time.monotonic() < t4_end - 10.0:
        sc, seq = peak_shaving(best_seq, max_iters=100, deadline=t4_end)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 4 (peak shave): {:.8f}".format(best_sc))
        
        if time.monotonic() < t4_end - 5.0:
            sc, seq = exact_subgrad_method(best_seq, max_iters=80, deadline=t4_end,
                                           k_list=[1, 3, 5, 8])
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("Phase 4 (subgrad): {:.8f}".format(best_sc))
    
    # Phase 5: More LSE with very high beta (15% time)
    t5_end = t_start + 0.90 * total_budget
    if time.monotonic() < t5_end - 10.0:
        a = best_seq.copy()
        for beta in (500000, 2000000, 10000000):
            if time.monotonic() > t5_end - 10.0:
                break
            rem = float(t5_end - time.monotonic())
            if rem < 15.0:
                break
            sub_dead = min(t5_end - 5.0, time.monotonic() + rem * 0.4)
            sc, a2 = optimize_trajectory(a, sub_dead, betas=(float(beta),))
            sc_exact = exact_score_a(a2)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_seq = a2.copy()
                print("Phase 5 (beta={:.0f}): {:.8f}".format(beta, best_sc))
            a = normalize_a(a2)
    
    # Phase 6: Perturbation restarts (10% time)
    t6_end = deadline - 30.0
    n_perturb = 5
    while time.monotonic() < t6_end and n_perturb > 0:
        left = float(t6_end - time.monotonic())
        if left < 30.0:
            break
        chunk = min(t6_end - 5.0, time.monotonic() + left / n_perturb)
        n_perturb -= 1
        
        p = perturb_best(best_seq, rng, strength=float(rng.uniform(0.03, 0.15)))
        
        # Quick optimize
        if time.monotonic() < chunk - 10.0:
            sc, a2 = optimize_trajectory(p, min(chunk - 5.0, time.monotonic() + 60.0),
                                         betas=(10000, 100000))
            sc_exact = exact_score_a(a2)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_seq = a2.copy()
                print("Perturb: {:.8f}".format(best_sc))
        
        # Spectral refinement
        if time.monotonic() < chunk - 5.0:
            sc, a2 = spectral_flattening(p, n_iter=15, alpha=0.3, deadline=chunk)
            if sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Perturb spec: {:.8f}".format(best_sc))
    
    # Phase 7: Final polish
    if time.monotonic() < deadline - 10.0:
        sc, seq = exact_subgrad_method(best_seq, max_iters=50, deadline=deadline - 10.0,
                                       k_list=[1, 2, 3])
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
        
        if time.monotonic() < deadline - 5.0:
            sc, seq = peak_shaving(best_seq, max_iters=30, deadline=deadline - 5.0)
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
        
        # Final LSE with max beta
        if time.monotonic() < deadline - 20.0:
            rem = float(deadline - time.monotonic() - 15.0)
            if rem > 30.0:
                sc, seq = optimize_trajectory(best_seq, deadline - 15.0, betas=(5000000,))
                sc_exact = exact_score_a(seq)
                if sc_exact < best_sc:
                    best_sc = sc_exact
                    best_seq = seq.copy()
    
    # Final verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc
    
    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
