"""
Enhanced optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key improvements over previous version:
1. FISTA (Nesterov accelerated gradient) replacing L-BFGS for LSE - 10-50x more iterations
2. Aggressive adaptive peak shaving with multiplicative updates and blame vectors
3. Interleaved LSE + peak shaving refinement
4. Smart multi-scale perturbations for restarts
5. Higher beta values for more faithful max approximation

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score_LSE = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    # Softmax weights
    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a:
    # dL/da_j = 2 * sum_k p_k * a_{k-j}
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def lse_value_only(a, beta):
    """Compute LSE score without gradient (faster)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    return (2.0 * n * L) / (S * S)


def fista_lse(a0, beta, n_iters=50000, step_init=None, check_interval=50):
    """
    FISTA (Nesterov accelerated gradient) on LSE-smoothed objective.
    Much faster than L-BFGS: no curvature estimation, no line search overhead.
    """
    n = len(a0)
    x = normalize(a0.copy())
    x_prev = x.copy()
    t = 1.0

    if step_init is None:
        # Estimate Lipschitz constant
        # Gradient involves convolution, roughly O(n/beta) for large beta
        step = 1.0 / max(10.0, n / beta * 2)
        step = min(step, 0.5)
    else:
        step = step_init

    best_sc = compute_score(x)
    best_x = x.copy()
    lse_best = lse_value_only(x, beta)

    for it in range(n_iters):
        # Nesterov extrapolation
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t_prev - 1.0) / t * (x - x_prev)
        y = np.maximum(y, 0.0)

        # Gradient at y
        lse_y, grad = lse_score_and_grad(y, beta)

        # Backtracking line search (limited)
        step = min(step * 1.3, 1.0)
        for _ in range(10):
            x_new = y - step * grad
            x_new = np.maximum(x_new, 0.0)
            S_new = x_new.sum()
            if S_new < 1e-12:
                x_new = y.copy()
                break
            x_new = x_new / S_new
            # Sufficient decrease check
            lse_new = lse_value_only(x_new, beta)
            if lse_new <= lse_y - 0.5 * step * float(np.dot(grad, grad)) + 1e-14:
                break
            step *= 0.5

        x_prev = x
        x = x_new

        # Periodic exact score check
        if (it + 1) % check_interval == 0 or it == n_iters - 1:
            sc = compute_score(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

    return best_sc, best_x


def fista_lse_with_restart(a0, beta, n_iters=30000, restart_every=5000,
                           step_init=None, check_interval=50):
    """FISTA with periodic restarts (resets momentum to escape cycles)."""
    n = len(a0)
    x = normalize(a0.copy())
    best_sc = compute_score(x)
    best_x = x.copy()

    for start in range(0, n_iters, restart_every):
        end = min(start + restart_every, n_iters)
        n_sub = end - start
        sc, x_opt = fista_lse(x, beta, n_iters=n_sub, step_init=step_init,
                              check_interval=check_interval)
        if sc < best_sc:
            best_sc = sc
            best_x = x_opt.copy()
        x = x_opt.copy()

    return best_sc, best_x


def peak_shave_aggressive(a, n_iters=5000, top_k=20, alpha_init=0.05):
    """
    Aggressive peak shaving with multiplicative updates.
    For each top peak, compute blame and apply multiplicative reduction.
    Add mass to "free" positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute total blame for each index
        blame = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / c_max  # Weight by peak height
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                blame[js] += w * a[k - js]

        # Normalize blame
        blame_max = blame.max()
        if blame_max < 1e-15:
            break
        blame = blame / blame_max

        # Multiplicative update: reduce high-blame, slightly boost low-blame
        a_new = a * np.exp(-alpha * blame)
        # Boost low-blame positions
        free_mask = blame < 0.3
        if free_mask.sum() > 10:
            a_new[free_mask] *= (1.0 + alpha * 0.5)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        # Decay alpha
        alpha *= 0.9995

        # Periodic score check (every 5 iters for speed)
        if (it + 1) % 5 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def peak_shave_targeted(a, n_iters=3000, n_peaks=10):
    """
    Targeted peak shaving: for each peak, find the single most contributing pair
    and reduce it. More surgical than the blame approach.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k - js]
            # Find the top contributing pair and reduce both members slightly
            top_local = np.argmax(pairs)
            j = js[top_local]
            j2 = k - j
            if j == j2:
                # Reduce a[j]
                factor = 0.95
                a[j] *= factor
            else:
                # Reduce both
                factor = 0.97
                a[j] *= factor
                a[j2] *= factor
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def interleaved_refine(a, beta_start, beta_end, n_total_iters=30000,
                       fista_chunk=2000, shave_chunk=200):
    """
    Interleave FISTA LSE steps with peak shaving steps.
    Gradually increase beta from beta_start to beta_end.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    total_steps = n_total_iters
    n_fista_rounds = total_steps // (fista_chunk + shave_chunk)
    remaining = total_steps % (fista_chunk + shave_chunk)

    for r in range(n_fista_rounds + 1):
        progress = r / max(1, n_fista_rounds)
        beta = beta_start * (beta_end / beta_start) ** progress if beta_start > 0 else beta_end

        # FISTA chunk
        sc, a_opt = fista_lse(a, beta, n_iters=fista_chunk)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a = a_opt.copy()

        # Peak shaving chunk (if not the last round)
        if r < n_fista_rounds:
            a_shaved = peak_shave_aggressive(a, n_iters=shave_chunk, top_k=10)
            sc_sh = compute_score(a_shaved)
            if sc_sh < best_sc:
                best_sc = sc_sh
                best_a = a_shaved.copy()
            a = a_shaved.copy()

    return best_sc, best_a


def coord_descent_aggressive(a, n_sweeps=15, n_top=50):

def fienup_flat(a, n_iters=200):
    """Fienup: iteratively flatten the power spectrum."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        alpha = 0.2 + 0.6 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        if (it + 1) % 10 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def perturb_mult(a, rng, scale=0.02):
    """Multiplicative perturbation with low-frequency structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 8):
        freq = rng.integers(1, 30)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_peak_aware(a, rng, scale=0.01):
    """Perturbation that adds to non-peak-contributing positions."""
    n = len(a)
    c = fast_conv(a)
    k_peak = int(np.argmax(c))
    j_start = max(0, k_peak - n + 1)
    j_end = min(k_peak, n - 1)

    a_new = a.copy()
    # Add to positions far from the peak contributors
    for _ in range(20):
        idx = rng.integers(0, n)
        if j_start <= idx <= j_end:
            continue
        a_new[idx] += scale / n
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

        n_iters = max(5000, min(n_iters, 50000))

        t0 = time.monotonic()
        sc, a_opt = fista_lse_with_restart(a_work, beta, n_iters=n_iters,
                                            restart_every=10000)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"FISTA b={beta}")
        a_work = a_opt.copy()

    # ==================================================================
    # Phase 3: Aggressive Peak Shaving (15% time)
    # ==================================================================
    phase3_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Aggressive Peak Shaving (budget {phase3_end - t_start:.0f}s) ===")

    a_ps1 = peak_shave_aggressive(best_a, n_iters=3000, top_k=15, alpha_init=0.04)
    sc = compute_score(a_ps1)
    update(sc, a_ps1, "PSAVE-aggr k=15")

    a_ps2 = peak_shave_aggressive(best_a, n_iters=3000, top_k=30, alpha_init=0.03)
    sc = compute_score(a_ps2)
    update(sc, a_ps2, "PSAVE-aggr k=30")

    a_ps3 = peak_shave_targeted(best_a, n_iters=2000, n_peaks=5)
    sc = compute_score(a_ps3)
    update(sc, a_ps3, "PSAVE-target")

    # FISTA after peak shaving
    if time.monotonic() < phase3_end - 20:
        sc2, a2 = fista_lse(best_a, 30000, n_iters=15000)
        update(sc2, a2, "PSAVE+FISTA")

    # ==================================================================
    # Phase 4: Interleaved Refinement (20% time)
    # ==================================================================
    phase4_end = t_start + 0.70 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Interleaved Refinement (budget {phase4_end - t_start:.0f}s) ===")

    remaining_iters = int((phase4_end - 20 - time.monotonic()) * 8)
    remaining_iters = max(20000, min(remaining_iters, 60000))
    print(f"  Interleaved: beta 5k->100k, {remaining_iters} total steps")

    sc4, a4 = interleaved_refine(best_a, beta_start=5000, beta_end=100000,
                                  n_total_iters=remaining_iters,
                                  fista_chunk=3000, shave_chunk=300)
    update(sc4, a4, "INTERLEAVED")

    # ==================================================================
    # Phase 5: Multi-Restart (10% time)
    # ==================================================================
    phase5_end = t_start + 0.80 * (budget_s - 30.0)
    print(f"\n=== Phase 5: Multi-Restart ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 20:
            break
        remaining = phase5_end - 20 - time.monotonic()
        if remaining < 25:
            break

        # Different perturbation strategies
        if r % 4 == 0:
            a_pert = perturb_mult(best_a, rng, scale=0.015)
        elif r % 4 == 1:
            a_pert = perturb_peak_aware(best_a, rng, scale=0.02)
        elif r % 4 == 2:
            # Gaussian perturbation
            a_pert = best_a + 0.01 * rng.standard_normal(n)
            a_pert = normalize(a_pert)
        else:
            # Spectral perturbation
            A = np.fft.rfft(a_pert if 'a_pert' in locals() else best_a, 1 << (2*n-1).bit_length())
            A *= np.exp(0.01j * rng.standard_normal(len(A)))
            a_pert = np.fft.irfft(A, 1 << (2*n-1).bit_length())[:n]
            a_pert = normalize(a_pert)

        n_iters = max(5000, int(remaining * 5))
        beta = 2000 + r * 3000
        sc, a_opt = fista_lse(a_pert, beta, n_iters=n_iters)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 6: Final Polish (10% time)
    # ==================================================================
    print(f"\n=== Phase 6: Final Polish ===")

    # Very high beta FISTA
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        n_iters = max(20000, int(remaining * 6))
        sc, a_final = fista_lse_with_restart(best_a, 100000, n_iters=n_iters,
                                              restart_every=10000)
        update(sc, a_final, "FINAL-FISTA-100k")

    # Final peak shave
    if time.monotonic() < deadline - 30:
        remaining_iters = max(2000, int((deadline - 30 - time.monotonic()) * 10))
        a_ps = peak_shave_aggressive(best_a, n_iters=min(remaining_iters, 10000),
                                      top_k=10, alpha_init=0.02)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    # Final FISTA on best
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        n_iters = max(5000, int(remaining * 8))
        sc, a_final = fista_lse(best_a, 200000, n_iters=n_iters)
        update(sc, a_final, "FINAL-FISTA-200k")

    # Final targeted shave
    if time.monotonic() < deadline - 5:
        remaining_iters = max(500, int((deadline - 5 - time.monotonic()) * 15))
        a_ps = peak_shave_targeted(best_a, n_iters=min(remaining_iters, 5000), n_peaks=5)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-TARGET")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
