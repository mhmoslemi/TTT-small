def fast_conv(a, M=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0: return np.array([])
    if M is None: M = next_pow2(2*n - 1)

def conv_cross(p,a,M3=None):
    n=len(a)
    if n==0: return np.array([])
    if M3 is None: M3=next_pow2(3*n-2)
    a_rev=a[::-1]
    P=np.fft.rfft(p, M3)
    R=np.fft.rfft(a_rev, M3)
    conv=np.fft.irfft(P*R, M3)
    return conv[n-1:2*n-1]

def lse_objective(x, beta, M, M3):
    if time.monotonic() > EVAL_TIME_LIMIT:
        raise RuntimeError("eval timed out")
    x=np.asarray(x,dtype=np.float64)
    a=x.copy()
    if (a < 0.0).any():
        a=np.maximum(a,0.0)
    n=a.size
    if n==0: return 1e100, np.array([])
    S=float(a.sum())
    if S < 1e-15:
        return 1e100, np.zeros(n)
    A=np.fft.rfft(a, M)
    q=np.fft.irfft(A*A, M)[:2*n-1]
    qmax=float(np.max(q)) if q.size else 0.0
    if beta <= 0: beta=1.0
    sh=beta*(q - qmax)
    sh=np.clip(sh, -600.0, 600.0)
    ex=np.exp(sh)
    exsum=float(ex.sum())
    if exsum <= 0.0 or not np.isfinite(exsum):
       ex=np.full_like(q,1.0); exsum=float(ex.size)
    L=qmax + float(np.log(exsum))/beta

    p=ex/exsum
    g=2.0*conv_cross(p,a,M3)
    S2=S*S; S3=S2*S
    grad=(2.0*n/S2)*g - (4.0*n*L/S3)

    if not np.all(np.isfinite(grad)):

    score=float(2.0*n*L/S2)
    if not np.isfinite(score): score=1e100
    return score, grad

def cap_rebalance(q, cap, uniform_blend=0.0):
   q=np.asarray(q,dtype=np.float64).copy()
   L=q.size
   if L==0: return q
   q=np.maximum(q,0.0)
   S=float(q.sum())
   if S <= 1e-300 or not np.isfinite(S):
       return np.full(L, 1.0/L)
   q /= S
   cap=abs(float(cap))
   if cap <= 0: cap=1.0/(L+1)
   if cap >= float(q.max()):
       q_new=q
   else:
       q_new=q.copy()
       over=q_new > cap
       excess=float((q_new[over]-cap).sum())
       q_new[over]=cap
       head=np.maximum(cap - q_new, 0.0)
       hsum=float(head.sum())
       if hsum > 1e-300:
          frac=excess/hsum
          if frac <= 1.0:
             q_new = q_new + frac*head
          else:
             q_new = q_new + head
             residual=excess - hsum
             if residual > 0:
                q_new += residual/L
       else:
          q_new += max(0.0, excess)/L
       q_new=np.maximum(q_new,0.0)
       s=float(q_new.sum())
       if s <= 0: q_new=np.full(L, cap if cap>0 else 1.0/L)
       else: q_new /= s
   if uniform_blend > 0:
       u=1.0/L
       q_new=(1.0-uniform_blend)*q_new + uniform_blend*u
       q_new/=q_new.sum()
   return q_new
```

"""
Optimizer for min 2n*max(conv(a,a))/(sum(a)^2), a >= 0.

Approach:
- Use the parent sequence when available and also build asymmetric non-negation
  initializers, because symmetric sequences are stuck at score >= 2.
- Polynomial-convolutions are handled with FFTs.
- Main driver is an L-smoothed (LSE) annealing schedule optimized by L-BFGS-B
  on the original nonnegative variables, using projected gradients and a wall-clock
  interrupt to fit inside the time budget.
- Additional refiners: top-peak subgradient shaving, capped spectral-factorization
  iterations, pairwise mass transfers, and multi-restart perturbations.
- Optionally grows the length (interpolation/block-upsample) when it appears
  beneficial.
"""

import numpy as np
from scipy.optimize import minimize
import time

TARGET_N = 10000
TARGET_SCORE = 1.5027

BEST_SCORE = np.inf
BEST_SEQ = None
BEST_FIXED_SCORE = np.inf
BEST_FIXED_SEQ = None

DEADLINE = 0.0
T_START = 0.0

ACTUAL_COUNT = 0
EVAL_TIME_LIMIT = 1e100
LAST_LBFGS_X = None


def remaining_seconds():
    return max(0.0, DEADLINE - time.monotonic())


def next_pow2(L):
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a, M=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if M is None:
        M = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def conv_cross(p, a, M3=None):
    """
    Returns g_i = sum_j a_j p_{i+j}, i=0..n-1, with p length 2n-1.
    This is (p conv a_reversed)[n-1 : 2n-1].
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if M3 is None:
        M3 = next_pow2(3 * n - 2)
    a_rev = a[::-1]
    P = np.fft.rfft(np.asarray(p, dtype=np.float64), M3)
    R = np.fft.rfft(a_rev, M3)
    conv = np.fft.irfft(P * R, M3)
    return conv[n - 1:2 * n - 1]


def compute_score_fast(a, M=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    S = float(a.sum())
    if not np.isfinite(S) or S < 1e-15:
        return np.inf
    q = fast_conv(a, M)
    if q.size == 0:
        return np.inf
    val = 2.0 * n * float(np.max(q)) / (S * S)
    return val if np.isfinite(val) else np.inf


def lse_objective(x, beta, M, M3):
    if time.monotonic() > EVAL_TIME_LIMIT:
        raise RuntimeError("eval timed out")

    x = np.asarray(x, dtype=np.float64).ravel()
    a = x.copy()
    if np.any(a < 0.0):
        a = np.maximum(a, 0.0)

    n = a.size
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)

    S = float(a.sum())
    if not np.isfinite(S) or S < 1e-15:
        return 1e100, np.zeros(n, dtype=np.float64)

    A = np.fft.rfft(a, M)
    q = np.fft.irfft(A * A, M)[:2 * n - 1]
    qmax = float(np.max(q)) if q.size else 0.0

    if beta <= 0.0 or not np.isfinite(beta):
        beta = 1.0

    sh = beta * (q - qmax)
    sh = np.clip(sh, -600.0, 600.0)
    ex = np.exp(sh)
    exsum = float(ex.sum())
    if not np.isfinite(exsum) or exsum <= 0.0:
        ex = np.full_like(q, 1.0)
        exsum = float(ex.size)

    L = qmax + float(np.log(exsum)) / beta
    if not np.isfinite(L):
        L = qmax

    p = ex / exsum
    grad_L = 2.0 * conv_cross(p, a, M3)

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * grad_L - (4.0 * n * L / S3)

    # Keep the direction tangent to the scale/sum direction as much as possible.
    grad -= float(np.mean(grad))

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)

    score = float(2.0 * n * L / S2)
    if not np.isfinite(score):
        score = 1e100
    return score, grad


def lbfgs_callback(xk):
    global LAST_LBFGS_X
    try:
        if xk is not None:
            LAST_LBFGS_X = np.asarray(xk, dtype=np.float64).copy()
    except Exception:
        pass
    if time.monotonic() > EVAL_TIME_LIMIT:
        raise RuntimeError("lbfgs callback timeout")
    return False


def run_lbfgs_lse(a0, beta, budget_s):
    global LAST_LBFGS_X, EVAL_TIME_LIMIT

    if a0 is None:
        return (None, None)

    now = time.monotonic()
    if budget_s < 5.0 or now > DEADLINE - 30.0:
        return (None, None)

    a = normalize(a0)
    n = a.size
    if n == 0:
        return (None, None)

    M = next_pow2(2 * n - 1)
    M3 = next_pow2(3 * n - 2)

    LAST_LBFGS_X = a.copy()
    EVAL_TIME_LIMIT = max(now + 3.0, min(DEADLINE - 8.0, now + max(5.0, float(budget_s)) - 1.0))

    # Let the wall-clock limit dominate; large maxfun/maxiter are handled by the
    # timeout in objective/callback.
    maxfun = 1000000
    maxiter = 100000

    bounds = [(0.0, None)] * n

    try:
        res = minimize(
            lse_objective,
            a,
            args=(beta, M, M3),
            jac=True,
            method="L-BFGS-B",
            bounds=bounds,
            callback=lbfgs_callback,
            options={
                "ftol": 1e-18,
                "gtol": 1e-12,
                "maxiter": maxiter,
                "maxfun": maxfun,
                "maxls": 50,
                "disp": False,
            },
        )
        out = res.x
    except Exception:
        out = LAST_LBFGS_X

    out = np.asarray(out, dtype=np.float64).ravel()
    if out.size != n or not np.all(np.isfinite(out)):
        out = a

    out = np.clip(out, 0.0, 1000.0)
    out = normalize(out)
    sc = compute_score_fast(out)
    if not np.isfinite(sc):
        return (None, out)
    return (sc, out)


def update_best(a, label="", force_actual=False):
    global BEST_SCORE, BEST_SEQ, ACTUAL_COUNT, BEST_FIXED_SCORE, BEST_FIXED_SEQ

    if a is None:
        return

    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return

    a = a / s
    sc = compute_score_fast(a)
    if not np.isfinite(sc):
        return

    # Ground truth check on meaningful improvements, but not too often for large n.
    use_actual = force_actual or (
        ACTUAL_COUNT < 80 and a.size <= 15000 and sc < BEST_SCORE + 1e-9
    )

    if use_actual:
        try:
            lst = [float(x) for x in a]
            av = evaluate_sequence(lst)
            ACTUAL_COUNT += 1
            if np.isfinite(av):
                sc = float(av)
        except Exception:
            pass

    if sc < BEST_SCORE:
        BEST_SCORE = sc
        BEST_SEQ = a.copy()
        print(f"  [{label}] new best {sc:.9f} n={a.size}")

    if a.size == TARGET_N and sc < BEST_FIXED_SCORE:
        BEST_FIXED_SCORE = sc
        BEST_FIXED_SEQ = a.copy()


def interpolate_to(a, n):
    a = normalize(a)
    if a.size == n:
        return a.copy()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, n)
    return normalize(np.interp(new, old, a))


def power_shape(n, alpha):
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.arange(1, n + 1, dtype=np.float64)
    try:
        arr = np.power(x, -alpha) * np.power(n - x + 1, -alpha)
    except Exception:
        arr = np.ones(n, dtype=np.float64)
    arr = arr + 1e-3
    return normalize(arr)


def skew_power_shape(n, alpha, skew=0.0):
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.arange(1, n + 1, dtype=np.float64)
    try:
        left = np.power(x, -alpha)
        right = np.power(n - x + 1, -(alpha + skew))
        arr = left * right
    except Exception:
        arr = np.ones(n, dtype=np.float64)
    arr = arr + 1e-3
    return normalize(arr)


def random_bump_shape(n, rng):
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    a = np.full(n, 0.02, dtype=np.float64)
    for _ in range(7):
        c = float(rng.random())
        w = float(rng.uniform(0.004, 0.18))
        amp = float(rng.uniform(0.3, 3.0))
        a += amp * np.exp(-((x - c) / w) ** 2)
    a = np.abs(a) + 0.01
    return normalize(a)


def lowfreq_perturb(a, rng, scale):
    a = normalize(a)
    n = a.size
    if n == 0:
        return a
    x = np.linspace(0.0, 1.0, n)
    pert = np.zeros(n, dtype=np.float64)
    for m in range(1, 8):
        f = int(rng.integers(1, 32))
        ph = 2.0 * np.pi * float(rng.random())
        amp = scale / float(m)
        pert += amp * np.cos(2.0 * np.pi * f * x + ph)
    return normalize(a * (1.0 + pert))


def small_random_perturb(a, rng, scale):
    a = normalize(a)
    if a.size == 0:
        return a
    return normalize(a + scale * rng.standard_normal(a.size))


def mix_two(a, b, w):
    a = normalize(a)
    b = normalize(b)
    if a.size != b.size:
        b = interpolate_to(b, a.size)
    return normalize(w * a + (1.0 - w) * b)


def upsample_block(a, m):
    a = normalize(a)
    if m <= 1:
        return a.copy()
    n = a.size
    out = np.zeros(n * m, dtype=np.float64)
    block = a / float(m)
    for r in range(m):
        out[r::m] = block
    return normalize(out)


def cap_rebalance(q, cap, uniform_blend=0.0):
    q = np.asarray(q, dtype=np.float64).ravel().copy()
    L = q.size
    if L == 0:
        return q

    q = np.maximum(q, 0.0)
    S = float(q.sum())
    if not np.isfinite(S) or S <= 1e-300:
        return np.full(L, 1.0 / L, dtype=np.float64)

    q /= S
    cap = abs(float(cap))
    if cap <= 0.0 or not np.isfinite(cap):
        cap = 1.0 / (L + 1)

    if cap >= float(np.max(q)):
        q_new = q.copy()
    else:
        q_new = q.copy()
        over = q_new > cap
        excess = float((q_new[over] - cap).sum())
        q_new[over] = cap

        head = np.maximum(cap - q_new, 0.0)
        hsum = float(head.sum())

        if hsum > 1e-300:
            frac = excess / hsum
            if frac <= 1.0:
                q_new = q_new + frac * head
            else:
                q_new = q_new + head
                residual = excess - hsum
                if residual > 0.0:
                    q_new += residual / L
        else:
            q_new += max(0.0, excess) / L

        q_new = np.maximum(q_new, 0.0)
        s = float(q_new.sum())
        if not np.isfinite(s) or s <= 0.0:
            q_new = np.full(L, cap if cap > 0 else 1.0 / L, dtype=np.float64)
        else:
            q_new /= s

    if uniform_blend > 0.0:
        u = 1.0 / L
        q_new = (1.0 - uniform_blend) * q_new + uniform_blend * u
        s = float(q_new.sum())
        if s > 0.0 and np.isfinite(s):
            q_new /= s

    return q_new


def spectral_square_root(q, n, M):
    q = np.asarray(q, dtype=np.float64).ravel()
    if q.size < M:
        qp = np.zeros(M, dtype=np.float64)
        qp[:q.size] = q
        q = qp
    else:
        q = q[:M]

    Q = np.fft.rfft(q, M)
    B = np.lib.scimath.sqrt(Q)

    if B.size > 0:
        B[0] = complex(float(np.real(B[0])), 0.0)
        if M % 2 == 0:
            idx = M // 2
            ny = B[idx]
            B[idx] = complex(max(0.0, float(np.real(ny))), 0.0)

    b = np.fft.irfft(B, M)[:n]
    return b


def spectral_cap_run(a, target_score=None, max_iters=200, seconds=60.0):
    a = normalize(a.copy())
    n = a.size
    if n == 0:
        return a

    M = next_pow2(2 * n - 1)
    best = compute_score_fast(a, M)
    best_a = a.copy()

    cap = best / (2.0 * n) if np.isfinite(best) and best > 0 else 1.0 / (2 * n - 1)
    if target_score is not None and np.isfinite(target_score):
        cap_limit = float(target_score) / (2.0 * n)
    else:
        cap_limit = (best * 0.998) / (2.0 * n)

    lb_cap = 1.0 / (2 * n - 1)
    cap_limit = max(lb_cap, cap_limit)
    cap = min(cap, max(cap_limit, lb_cap))

    end = time.monotonic() + max(1.0, float(seconds))
    it = 0

    while it < max_iters and time.monotonic() < min(end, DEADLINE - 5.0):
        it += 1
        q = fast_conv(a, M)
        cap = max(cap_limit, cap * (1.0 - 0.015))
        q_t = cap_rebalance(q, cap, uniform_blend=0.0)
        b = spectral_square_root(q_t, n, M)

        tries = []
        pos = normalize(np.maximum(b, 0.0))
        if pos.size == n:
            tries.append(pos)

        absb = normalize(np.abs(b))
        if absb.size == n:
            tries.append(absb)

        mix = normalize(0.6 * best_a + 0.4 * absb)
        if mix.size == n:
            tries.append(mix)

        improved = False
        for tr in tries:
            sc = compute_score_fast(tr, M)
            if np.isfinite(sc) and sc < best:
                best = sc
                best_a = tr.copy()
                improved = True

        if improved:
            a = best_a.copy()
        else:
            a = normalize(0.7 * best_a + 0.3 * tries[0])

    return best_a


def subgrad_polish(a, seconds=60.0, top_k=8, beta=50000.0, step_init=0.01):
    a = normalize(a.copy())
    n = a.size
    if n == 0:
        return a

    M = next_pow2(2 * n - 1)
    M3 = next_pow2(3 * n - 2)

    best = compute_score_fast(a, M)
    best_a = a.copy()
    step = step_init
    end = time.monotonic() + max(1.0, float(seconds))
    lb = 2.0 * n / (2.0 * n - 1.0)

    while time.monotonic() < min(end, DEADLINE - 5.0):
        q = fast_conv(a, M)
        Mval = float(np.max(q))
        sc = 2.0 * n * Mval

        if np.isfinite(sc) and sc < best:
            best = sc
            best_a = a.copy()

        if not np.isfinite(sc) or sc - lb < 1e-12:
            break

        sh = beta * (q - Mval)
        sh = np.clip(sh, -600.0, 600.0)
        ex = np.exp(sh)

        if top_k < q.size:
            idx = np.argpartition(q, -top_k)[-top_k:]
            mask = np.zeros_like(ex, dtype=bool)
            mask[idx] = True
            ex *= mask

        exsum = float(ex.sum())
        if not np.isfinite(exsum) or exsum <= 0.0:
            break

        p = ex / exsum
        g = 2.0 * conv_cross(p, a, M3)
        g -= float(np.mean(g))

        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn) or gn < 1e-14:
            break

        ref = lb
        if TARGET_SCORE < sc:
            ref = max(lb, TARGET_SCORE)
        if sc - ref <= 0.0:
            ref = lb

        step_polyak = (sc - ref) / (2.0 * n * gn * gn + 1e-30)
        if not np.isfinite(step_polyak) or step_polyak <= 0.0:
            step_polyak = step

        step = min(step, step_polyak, 0.01)
        step = max(step, 1e-10)

        a = a - step * g
        a = np.maximum(a, 0.0)
        a = normalize(a)
        step *= 0.998

    return best_a


def pair_polish(a, rng, max_evals=400, seconds=60.0):
    a = normalize(a.copy())
    n = a.size
    if n == 0:
        return a

    M = next_pow2(2 * n - 1)
    best = compute_score_fast(a, M)
    best_a = a.copy()
    evals = 0
    end = time.monotonic() + max(1.0, float(seconds))

    while evals < max_evals and time.monotonic() < min(end, DEADLINE - 5.0):
        a = best_a.copy()
        q = fast_conv(a, M)
        k = int(np.argmax(q))

        j1 = max(0, k - n + 1)
        j2 = min(k, n - 1)
        if j2 < j1:
            break

        js = np.arange(j1, j2 + 1, dtype=np.int64)
        partners = k - js
        valid = (partners >= 0) & (partners < n)
        js = js[valid]
        partners = partners[valid]
        if js.size == 0:
            break

        contribs = a[js] * a[partners]
        top_count = min(12, contribs.size)
        order = np.argsort(contribs)[-top_count:]

        for oi in order[:4]:
            if evals >= max_evals:
                break

            j = int(js[oi])
            partner = int(partners[oi])
            base = a[j]
            if base < 1e-13:
                continue

            chosen_l = None
            if n > 2:
                cands = rng.integers(0, n, 60)
                for c in cands:
                    c = int(c)
                    if c != j and c != partner:
                        chosen_l = c
                        break

            if chosen_l is None:
                chosen_l = 0 if j != 0 else 1

            for f in (0.90, 0.80, 0.65, 0.45):
                if evals >= max_evals:
                    break

                a_try = a.copy()
                eps = base * (1.0 - f)
                a_try[j] -= eps
                a_try[chosen_l] += eps
                a_try = np.maximum(a_try, 0.0)
                s = float(a_try.sum())
                if not np.isfinite(s) or s <= 1e-15:
                    continue
                a_try /= s

                sc = compute_score_fast(a_try, M)
                evals += 1

                if np.isfinite(sc) and sc < best:
                    best = sc
                    best_a = a_try.copy()
                    break

    return best_a


def propose_candidate(seed=20250525, budget_s=2400.0):
    global BEST_SCORE, BEST_SEQ, DEADLINE, T_START, ACTUAL_COUNT
    global BEST_FIXED_SCORE, BEST_FIXED_SEQ

    T_START = time.monotonic()
    DEADLINE = T_START + max(60.0, float(budget_s) - 30.0)

    BEST_SCORE = np.inf
    BEST_SEQ = None
    BEST_FIXED_SCORE = np.inf
    BEST_FIXED_SEQ = None
    ACTUAL_COUNT = 0

    rng = np.random.default_rng(seed)
    print(f"Start search: budget={budget_s:.0f}s, deadline margin=30s")

    parent = None
    try:
        h = height_sequence_1
        if h is not None:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size >= 20:
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        if parent.size == TARGET_N:
            update_best(parent, "parent")
            # Small cyclic wraps can create useful asymmetric variants.
            for shft in (1, 2, 5, 13):
                if parent.size > shft:
                    update_best(np.roll(parent, shft), f"parent-roll {shft}")
        else:
            update_best(interpolate_to(parent, TARGET_N), "parent-resize")
            if parent.size <= 25000:
                update_best(parent, "parent-orig")

    # If no good sequence has been found, start from explicitly asymmetric
    # low-pass / power-tail shapes. Uniform and pure symmetric shapes are bad
    # because symmetric nonnegative sequences have score >= 2.
    if BEST_SEQ is None or np.isfinite(BEST_FIXED_SCORE):
        alphas = (0.10, 0.18, 0.25, 0.35)
        skews = (-0.08, 0.0, 0.08)
        for alpha in alphas:
            for skew in skews:
                if remaining_seconds() < 2.0:
                    break
                a = skew_power_shape(TARGET_N, alpha, skew)
                update_best(a, f"skew {alpha}/{skew}")
            if remaining_seconds() < 2.0:
                break

        for r in range(6):
            if remaining_seconds() < 2.0:
                break
            a = random_bump_shape(TARGET_N, rng)
            update_best(a, f"random {r}")

    if BEST_SEQ is None:
        BEST_SEQ = normalize(np.ones(TARGET_N, dtype=np.float64))
        BEST_SCORE = compute_score_fast(BEST_SEQ)
        print("Fallback to uniform sequence")

    # For the main polish, prefer a length-10000 candidate if it is competitive.
    if BEST_FIXED_SEQ is not None:
        if (BEST_SEQ.size != TARGET_N) and (BEST_FIXED_SCORE <= BEST_SCORE + 1e-8):
            BEST_SEQ = BEST_FIXED_SEQ.copy()
            BEST_SCORE = BEST_FIXED_SCORE
        elif (BEST_SEQ.size == TARGET_N) and (BEST_FIXED_SCORE < BEST_SCORE):
            BEST_SEQ = BEST_FIXED_SEQ.copy()
            BEST_SCORE = BEST_FIXED_SCORE

    print(f"Initial main best: {compute_score_fast(BEST_SEQ):.9f}, n={len(BEST_SEQ)}")

    # ------------------------------------------------------------------
    # Main LSE annealing.
    # Reserve time for later local refiners and larger-n exploration.
    # ------------------------------------------------------------------
    betas = (500, 1500, 4000, 10000, 20000, 50000, 100000, 200000, 400000, 800000)
    a_cur = BEST_SEQ.copy()
    local_score = compute_score_fast(a_cur)

    main_reserve = 900.0

    for beta in betas:
        if time.monotonic() > DEADLINE - main_reserve:
            break

        for rep in range(3):
            if time.monotonic() > DEADLINE - main_reserve:
                break

            avail = max(0.0, DEADLINE - main_reserve - time.monotonic())
            if avail < 20.0:
                break

            budget = min(55.0, max(18.0, avail))
            sc, a_out = run_lbfgs_lse(a_cur, beta, budget)

            if a_out is not None:
                sc_after = compute_score_fast(a_out)
                update_best(a_out, f"LSE beta={beta} rep={rep}")

                if np.isfinite(sc_after) and sc_after < local_score - 1e-11:
                    local_score = sc_after
                    a_cur = a_out.copy()
                else:
                    a_cur = BEST_SEQ.copy()
                    local_score = compute_score_fast(a_cur)
                    break

    # ------------------------------------------------------------------
    # Local refiners on the current best.
    # ------------------------------------------------------------------
    if remaining_seconds() > 700.0 and BEST_SEQ is not None:
        a_pg = subgrad_polish(BEST_SEQ, seconds=min(90.0, remaining_seconds() - 650.0),
                              top_k=8, beta=50000.0)
        update_best(a_pg, "subgrad top8")

    if remaining_seconds() > 700.0 and BEST_SEQ is not None:
        target = max(TARGET_SCORE, BEST_SCORE * 0.998)
        a_sp = spectral_cap_run(BEST_SEQ, target_score=target,
                                max_iters=160,
                                seconds=min(120.0, remaining_seconds() - 650.0))
        update_best(a_sp, "spectral cap")

    if remaining_seconds() > 600.0 and BEST_SEQ is not None:
        a_pp = pair_polish(BEST_SEQ, rng, max_evals=260,
                           seconds=min(70.0, remaining_seconds() - 550.0))
        update_best(a_pp, "pair polish")

    # ------------------------------------------------------------------
    # Multi-restart perturbations from the incumbent.
    # ------------------------------------------------------------------
    attempts = 0
    while remaining_seconds() > 500.0 and attempts < 12 and BEST_SEQ is not None:
        attempts += 1
        n_cur = len(BEST_SEQ)

        if attempts % 3 == 1:
            a0 = lowfreq_perturb(BEST_SEQ, rng, 0.008 + 0.003 * (attempts % 5))
        elif attempts % 3 == 2:
            a0 = small_random_perturb(BEST_SEQ, rng, 0.005 + 0.002 * (attempts % 5))
        else:
            alpha = float(rng.uniform(0.12, 0.42))
            a0 = mix_two(BEST_SEQ, power_shape(n_cur, alpha), 0.78 + 0.03 * (attempts % 3))

        beta = 20000 + attempts * 18000
        budget = min(45.0, max(18.0, remaining_seconds() - 450.0))
        sc, a_out = run_lbfgs_lse(a0, beta, budget)
        if a_out is not None:
            update_best(a_out, f"restart {attempts}")

    # ------------------------------------------------------------------
    # Larger length exploration.
    # Scaling is neutral, but more coefficients can offer better finite-n
    # platforms. Try interpolated and block-upsampled versions if time remains.
    # ------------------------------------------------------------------
    if remaining_seconds() > 380.0 and BEST_SEQ is not None:
        n_best = len(BEST_SEQ)
        if n_best < 18000:
            if n_best <= 10000:
                new_n = 20000
            else:
                new_n = min(20000, n_best + 5000)

            a_interp = interpolate_to(BEST_SEQ, new_n)
            update_best(a_interp, f"large interp {new_n}")

            a_block = upsample_block(BEST_SEQ, 2)
            if a_block.size != new_n:
                a_block = interpolate_to(a_block, new_n)
            update_best(a_block, f"large block {len(a_block)}")

            big_start = a_interp
            big_sc = compute_score_fast(a_interp)

            cands = (a_block,)
            for c in cands:
                sc = compute_score_fast(c)
                if sc < big_sc:
                    big_sc = sc
                    big_start = c

            a_lse_big = big_start.copy()
            local_big = big_sc

            for beta in (5000, 20000, 60000, 150000):
                if remaining_seconds() < 300.0:
                    break

                for rep in range(2):
                    if remaining_seconds() < 250.0:
                        break

                    avail = max(0.0, remaining_seconds() - 220.0)
                    if avail < 20.0:
                        break

                    budget = min(70.0, max(18.0, avail))
                    sc, a_out = run_lbfgs_lse(a_lse_big, beta, budget)
                    if a_out is not None:
                        sc_after = compute_score_fast(a_out)
                        update_best(a_out, f"large LSE beta={beta} rep={rep}")
                        if np.isfinite(sc_after) and sc_after < local_big - 1e-10:
                            local_big = sc_after
                            a_lse_big = a_out.copy()
                        else:
                            break

    # ------------------------------------------------------------------
    # Final high-beta polish.
    # ------------------------------------------------------------------
    if remaining_seconds() > 80.0 and BEST_SEQ is not None:
        final_beta = 400000 if len(BEST_SEQ) <= 12000 else 150000
        avail = max(0.0, remaining_seconds() - 35.0)
        budget = min(180.0, max(25.0, avail))
        sc, a_out = run_lbfgs_lse(BEST_SEQ, final_beta, budget)
        if a_out is not None:
            update_best(a_out, "final LSE")

    if remaining_seconds() > 55.0 and BEST_SEQ is not None:
        a_pg = subgrad_polish(BEST_SEQ, seconds=min(30.0, remaining_seconds() - 25.0),
                              top_k=5, beta=100000.0)
        update_best(a_pg, "final subgrad")

    # Prefer length 10000 if it is essentially tied, to match the stated
    # construction length; otherwise keep the best found length.
    final_seq = BEST_SEQ
    if final_seq is None:
        final_seq = np.ones(TARGET_N, dtype=np.float64) / float(TARGET_N)

    if BEST_FIXED_SEQ is not None:
        if (final_seq.size != TARGET_N) and (BEST_FIXED_SCORE <= BEST_SCORE + 1e-6):
            final_seq = BEST_FIXED_SEQ
        elif (final_seq.size == TARGET_N) and (BEST_FIXED_SCORE < BEST_SCORE):
            final_seq = BEST_FIXED_SEQ

    arr = np.asarray(final_seq, dtype=np.float64).ravel()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        arr = np.ones(TARGET_N, dtype=np.float64) / float(TARGET_N)

    out = [float(x) for x in arr]

    if time.monotonic() < DEADLINE - 5.0:
        try:
            av = evaluate_sequence(out)
            print(f"Final verified score: {av:.9f}, length={len(out)}")
        except Exception:
            print("Final verification skipped/failed")
        print(f"Elapsed: {time.monotonic() - T_START:.1f}s of {budget_s:.0f}s")
    else:
        print("Time exhausted; returning current best")

    return out
