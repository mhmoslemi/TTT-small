Multi-strategy optimizer for minimum peak-to-mean autocorrelation.
Uses exponential parameterization, FISTA, adaptive LSE, spectral shaping,

def fft_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def score(a):
    """Compute 2n*max(conv)/S^2."""
    n = len(a)
    c = fft_conv(a)
    S = a.sum()
    if S < 1e-14:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def norm1(a):
    """Normalize to sum=1, clip negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        return a
    return a / s

Optimizer combining exponential-parameterization FISTA, spectral peak suppression,
Nesterov subgradient, and diverse multi-restart for minimum peak autocorrelation.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def get_score(a):
    n = len(a)
    c = fft_conv(a)
    S = float(a.sum())
    if S < 1e-14:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-14:
        return a
    return a / S


def lse_obj_grad_theta(theta, beta):
    """LSE score in exp parameterization. Returns (value, gradient)."""
    t = theta - theta.max()
    a = np.exp(t)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(len(theta))
    a = a / a_sum
    n = len(a)
    
    c = fft_conv(a)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -600, 600)
    e = np.exp(shifted)
    e_sum = e.sum()
    logsum = float(np.log(e_sum))
    L = c_max + logsum / beta
    p = e / e_sum
    
    a_rev = a[::-1]
    L3 = (2 * n - 1) + n - 1
    Mf3 = 1 << (L3 - 1).bit_length()
    PA = np.fft.rfft(p, Mf3)
    RA = np.fft.rfft(a_rev, Mf3)
    cr = np.fft.irfft(PA * RA, Mf3)[:L3]
    grad_L = 2.0 * cr[n - 1:2 * n - 1]
    
    grad_a = 2.0 * n * grad_L
    
    # Chain rule for softmax-like param
    ga = grad_a * a
    ga_s = ga.sum()
    grad_theta = ga - a * ga_s
    
    return 2.0 * n * L, grad_theta


def fista_exp(theta0, beta, n_iter=2000):

            best_y = y.copy()

def nesterov_subgrad(a, n_iters=3000, top_k=8, lr=0.003, decay=0.9999):

def lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000):

def _lse_wrap(a, beta):

    n = len(a)

def interpolate_seq(a, new_n):
