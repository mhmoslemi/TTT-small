```python
def fast_conv(a):

    n = len(a)

    M_fft = 1 << (M - 1).bit_length()
```

```python
def a2_fienup(a, n_iters=500, alpha_start=0.95, alpha_end=0.5):
    """
    A²-Fienup: iterative spectral flattening working with A² directly.
    For each iteration:
    1. Compute C = DFT(conv(a,a)) = A²
    2. In spatial domain, compress c to be flatter
    3. Recover A = sqrt(DFT(c_new))
    4. a = IDFT(A), clip to [0, ∞), normalize
    """
    n = len(a)
    a = normalize(a.copy())
    L = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        # Compute convolution and its DFT
        A = np.fft.rfft(a, L)
        C = A * A  # DFT of linear conv
        c = np.fft.irfft(C, L)[:2 * n - 1]
        
        c_mean = float(c.mean())
        if c_mean < 1e-15:
            break
        
        # Power compression: c_new = (c/c_mean)^alpha * c_mean
        c_ratio = c / c_mean
        c_new = np.power(np.maximum(c_ratio, 0), alpha) * c_mean
        
        # Also ensure c_new is non-negative (it should be since c ≥ 0 for a ≥ 0)
        c_new = np.maximum(c_new, 0)
        
        # Recover A from modified C
        C_new = np.fft.rfft(c_new, L)
        A_new = np.sqrt(C_new)  # complex square root
        
        # Recover a
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
```

```python
def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    # dL/da_j = 2 * sum_k p_k * a_{k-j}
    # This is 2 * (p * a_reversed) convolved, extracted at the right offset
    a_rev = a[::-1]
    L3 = (2 * n - 1) + n - 1
    L3_fft = 1 << (L3 - 1).bit_length()
    PA = np.fft.rfft(p, L3_fft)
    RA = np.fft.rfft(a_rev, L3_fft)
    conv_res = np.fft.irfft(PA * RA, L3_fft)[:L3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad
```

            # Hybrid: clip + power
            cf = 2.0 - 1.0 * progress
            c_clipped = np.minimum(c, c_mean * cf)
            c_new = np.power(np.maximum(c_clipped / c_mean, 1e-30), alpha) * c_mean
        
        c_new = np.maximum(c_new, 0.0)
        
        C_new = np.fft.rfft(c_new, L)
        # Complex square root
        A_new = np.sqrt(C_new)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L_val = c_max + log_sum / beta

    p = exp_s / exp_s.sum()

    a_rev = a[::-1]
    L3 = (2 * n - 1) + n - 1
    L3_fft = 1 << (L3 - 1).bit_length()
    PA = np.fft.rfft(p, L3_fft)
    RA = np.fft.rfft(a_rev, L3_fft)
    conv_res = np.fft.irfft(PA * RA, L3_fft)[:L3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L_val / S3)

    score = (2.0 * n * L_val) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    n = len(a0)
    a0 = normalize_seq(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except:
        return compute_score(a0), a0


def peak_shave(a, n_iters=3000, top_k=8, step_init=0.005):
    n = len(a)
    a = normalize_seq(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient targeting top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        total_w = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_w += w
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += w * a[je-1:js-1:-1]  # reversed portion
        
        S2 = S * S
        S3 = S2 * S
        avg_w = total_w / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        lb = 2.0 * n / (2.0 * n - 1.0)
        step_p = (sc - lb) / (gn * gn + 1e-30)
        step = min(step, step_p, 0.02)
        step = max(step, 1e-10)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
    
    return best_a
