def nesterov_loop(a, obj_fn, max_iters, deadline):
    n = len(a)
    a_prev = a.copy()
    t = 1.0
    step_size = 1.0 / (np.max(np.abs(obj_fn(a)[1])) + 1e-8)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        momentum = (t - 1) / t_new
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)

        obj, grad = obj_fn(y)

        a_new = np.maximum(a_new, 0)

        a_prev = a
        a = a_new
        t = t_new
    
    return a

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:

    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def lse_obj_grad(a, beta):
    """Return (objective, gradient) for Nesterov use. Objective includes normalization penalty."""
    a = normalize_a(a)
    score, grad = lse_score_grad(a, beta)
    return score, grad


def nesterov_accelerated(a_start, beta, max_iters, deadline, Step0=None):
    """Nesterov-accelerated gradient descent on the LSE objective."""
    a = normalize_a(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = lse_obj_grad(a, beta)[0]
    
    _, grad0 = lse_obj_grad(a, beta)
    gnorm = float(np.max(np.abs(grad0)))
    if gnorm < 1e-15:
        return best_sc, best_a
    if Step0 is None:
        step_size = 0.5 / gnorm
    else:
        step_size = float(Step0)
    
    a_prev = a.copy()
    t = 1.0
    patience = 0
    best_iter_sc = best_sc
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy
        
        sc_y, grad_y = lse_obj_grad(y, beta)
        gnorm_y = float(np.max(np.abs(grad_y)))
        if gnorm_y < 1e-15:
            break
            
        a_trial = y - step_size * grad_y
        a_trial = np.maximum(a_trial, 0.0)
        St = float(a_trial.sum())
        if St < 1e-12:
            a_trial = y.copy()
        else:
            a_trial = a_trial / St
        
        sc_trial = lse_obj_grad(a_trial, beta)[0]
        
        if sc_trial < sc_y - 1e-14:
            # Accept step, maybe increase step size slightly
            step_size = min(step_size * 1.05, 5.0 / gnorm_y)
            patience = 0
            if sc_trial < best_iter_sc:
                best_iter_sc = sc_trial
                best_sc = sc_trial
                best_a = a_trial.copy()
        else:
            # Reject, reduce step size
            step_size *= 0.5
            patience += 1
            if patience > 30:
                break
        
        a_prev = a
        a = a_trial
        t = t_new
    
    final_exact = exact_score_a(best_a)
    if np.isfinite(final_exact):
        return final_exact, best_a
    return best_sc, best_a


def adaptive_mirror_descent(a_start, max_iters, deadline, k_list=(1, 2, 3, 5, 8, 12)):
    """Mirror descent with adaptive step size and active set."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        d, S = d_from_a(a)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax <= 1e-15:
            break
        
        # Compute effective active set size from LSE with moderate beta
        beta_eff = 50.0
        z = beta_eff * (d - dmax)
        z = np.clip(z, -30.0, 30.0)
        w_lse = np.exp(z)
        w_lse = w_lse / w_lse.sum()
        # Effective number of active constraints
        entropy = -float(np.sum(w_lse * np.log(w_lse + 1e-30)))
        n_eff = int(max(1, min(d.size, np.exp(entropy))))
        
        # Try several k values around n_eff
        ks = sorted(set(max(1, k) for k in [n_eff // 2, n_eff, 2 * n_eff] + list(k_list)))
        ks = [k for k in ks if 1 <= k <= d.size]
        
        for k in ks[:4]:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            idx = np.argpartition(d, d.size - k)[d.size - k:]
            w = np.zeros(d.size)
            w[idx] = 1.0
            w = w / float(k)
            
            h = corr_pair(a, w)
            h = np.maximum(h, 1e-15)
            
            # Try multiple power values
            for t_power in (0.02, 0.05, 0.1, 0.15, 0.25, 0.4):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                log_a = np.log(np.maximum(a, 1e-15))
                log_h = np.log(h)
                log_new = (1.0 - t_power) * log_a + t_power * log_h
                a_new = np.exp(log_new - np.max(log_new))
                a_new = normalize_a(a_new)
                sc = exact_score_a(a_new)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
    
    return float(best_sc), best_a


def exact_subgrad_polish(a_start, max_iters, deadline, k_list=(1, 2, 3, 4, 5, 7, 10)):
    """Polish using exact max subgradient with adaptive step sizes."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            d, S = d_from_a(a)
            if d.size == 0:
                break
            dmax = float(np.max(d))
            if dmax <= 1e-15:
                break
            idx = np.argpartition(d, min(d.size, k) - k if k < d.size else 0)
            if k < d.size:
                idx = np.argpartition(d, d.size - k)[d.size - k:]
            else:
                idx = np.arange(d.size)
            w = np.zeros(d.size)
            w[idx] = 1.0
            w = w / float(len(idx))
            
            h = corr_pair(a, w)
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-15:
                continue
            grad = (4.0 * float(n) * h) / (S * S)
            LDS = float(np.dot(w, d))
            grad -= (4.0 * LDS) / S
            
            for eta in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = a - eta * grad
                q = normalize_a(np.maximum(q, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q
        if not improved and it > 10:
            break
    return float(best_sc), best_a


def projected_gradient_nesterov_full(a_start, beta, max_iters, deadline):
    """Full Nesterov with proper backtracking and normalization."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_f = lse_obj_grad(a, beta)[0]
    
    n = len(a)
    f0, g0 = lse_obj_grad(a, beta)
    L = float(np.max(np.abs(g0))) + 1e-8
    step = 1.0 / L
    
    a_old = a.copy()
    t_k = 1.0
    stall = 0
    
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        
        # Nesterov extrapolation
        y = a + alpha * (a - a_old)
        y = np.maximum(y, 0.0)
        Sy = float(y.sum())
        if Sy > 1e-12:
            y = y / Sy
        
        f_y, g_y = lse_obj_grad(y, beta)
        
        # Gradient step
        a_new = y - step * g_y
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a_new = a_new / S_new
        
        f_new = lse_obj_grad(a_new, beta)[0]
        
        if f_new <= f_y - 1e-15:
            stall = 0
            if f_new < best_f:
                best_f = f_new
                best_a = a_new.copy()
        else:
            step *= 0.5
            stall += 1
            if stall > 50:
                break
        
        a_old = a
        a = a_new
        t_k = t_new
    
    sc = exact_score_a(best_a)
    if np.isfinite(sc):
        return sc, best_a
    return best_f, best_a


def generate_diverse_inits(n, rng, parent, cap=30):
    """Generate diverse initializations with emphasis on asymmetric shapes."""
    inits = []
    n = int(n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        if float(p.sum()) > 1e-12:
            inits.append(p)
    
    x = np.linspace(0.0, 1.0, n)
    
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    # Ramp shapes (key: continuous limit analysis shows ramp is near-optimal)
    add(x)  # linear ramp 0->1
    add(1.0 - x)  # linear ramp 1->0
    add(x + 0.1)  # slightly offset ramp
    add(np.maximum(x, 0.05))  # clamped ramp
    
    # Power-law ramps
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(x ** p)
        add((1.0 - x) ** p)
        add(x ** p + 0.1)
        add((1.0 - x) ** p + 0.1)
    
    # Exponential shapes
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + 0.01)
        add(np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(rate * x) / np.exp(rate) + 0.01)
    
    # V-shapes (for comparison)
    add(1.0 + 0.5 * np.abs(2.0 * x - 1.0))
    add(1.0 + 1.0 * np.abs(2.0 * x - 1.0))
    
    # Cosine shapes
    add(1.0 + 0.5 * np.cos(np.pi * x))
    add(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    
    # Asymmetric cosine
    add(1.0 + 0.5 * np.cos(np.pi * (2.0 * x - 0.3)))
    
    # Gaussian bumps (asymmetric positions)
    for pos in (0.2, 0.3, 0.7, 0.8):
        for width in (0.1, 0.2, 0.3):
            add(np.exp(-((x - pos) / width) ** 2) + 0.05)
    
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    
    # Asymmetric random smooth
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        # Bias towards low frequencies
        power *= np.exp(-0.5 * np.abs(freqs) / Nf)
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.min())
        add(noise + 0.1)
    
    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.4))
    mode = int(rng.integers(0, 10))
    x = np.linspace(0.0, 0.0, n)  # placeholder
    
    if mode == 0:
        q = p * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.linspace(0.1, 1.0, n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        # Shift in log-space (circular shift effect)
        shift = int(rng.integers(1, max(2, n // 20)))
        logp = np.log(p + 1e-300)
        logp_shifted = np.roll(logp, shift)
        # Handle wrap-around by fading
        fade = np.ones(n)
        fade[:min(shift, n // 4)] = np.linspace(0.5, 1.0, min(shift, n // 4))
        fade[-min(shift, n // 4):] = np.linspace(1.0, 0.5, min(shift, n // 4))
        q = np.exp(0.5 * logp + 0.5 * logp_shifted * fade)
    elif mode == 4:
        # Smoothed noise
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.2 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # Multiply by a ramp
        ramp = np.linspace(0.5, 1.5, n)
        q = p * ramp
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Targeted: reduce peaks in convolution
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_lse(d, 50.0)
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * 0.2 * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 8:
        # Blend with a ramp
        ramp = np.linspace(0.2, 1.0, n)
        t = float(rng.uniform(0.2, 0.6))
        q = normalize_a((1 - t) * p + t * ramp)
    else:
        # Invert and perturb
        q = (1.0 - p / p.max()) + 0.05
        q = q * np.exp(0.1 * rng.standard_normal(n))
    
    return normalize_a(np.clip(q, 1e-10, None))


def active_weights_lse(d, beta):
    """Compute LSE-based active weights."""
    dmax = float(np.max(d))
    z = beta * (d - dmax)
    z = np.clip(z, -30.0, 30.0)
    w = np.exp(z)
    s = w.sum()
    if s < 1e-30:
        return np.ones(d.size) / d.size
    return w / s


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def multi_stage_optimize(a_start, deadline, n_stages=6):
    """Multi-stage LSE optimization with increasing beta, using both L-BFGS-B and Nesterov."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    # Beta schedule: start low, increase geometrically
    beta_schedule = [50.0, 200.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0, 500000.0]
    # Time fractions for each stage
    frac_schedule = [0.10, 0.12, 0.14, 0.14, 0.14, 0.14, 0.12, 0.10]
    
    t0 = time.monotonic()
    total_time = max(10.0, float(deadline - t0 - 5.0))
    n_use = min(n_stages, len(beta_schedule))
    
    for stage in range(n_use):
        if time.monotonic() > deadline - 5.0:
            break
        beta = beta_schedule[stage]
        sub_dead = t0 + (sum(frac_schedule[:stage + 1]) / sum(frac_schedule[:n_use])) * total_time
        sub_dead = min(sub_dead, deadline - 5.0)
        
        # Alternate between L-BFGS-B and Nesterov
        use_lbfgs = (stage % 2 == 0)
        
        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 10.0:
                break
            
            if use_lbfgs:
                mf = max(64, min(2000, int(rem * 0.7 / max(0.01, 0.02 * len(a) / 8192.0))))
                sc, a2 = run_lbfgs(a, beta, sub_dead, mf)
            else:
                max_it = max(50, min(500, int(rem * 2.0)))
                sc, a2 = nesterov_accelerated(a, beta, max_it, sub_dead)
            
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            
            a = normalize_a(a2)
    
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None
    
    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))
    
    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = parent.size
        if n_primary > 16384:
            n_primary = 16384
        if n_primary < 2048:
            n_primary = 2048
    
    # === Phase 0: Quick multi-start with diverse inits (10% of time) ===
    t0_end = t_start + 0.10 * total_budget
    
    if parent is not None:
        a_main = normalize_a(resample_sequence(parent, n_primary))
    else:
        x = np.linspace(0.0, 1.0, n_primary)
        a_main = normalize_a(x + 0.1)  # ramp initialization
    
    best_sc = exact_score_a(a_main)
    best_seq = a_main.copy()
    print("Initial score: {:.8f} (length {})".format(best_sc, len(best_seq)))
    
    # Try a few quick inits
    inits = generate_diverse_inits(n_primary, rng, parent, cap=12)
    for init in inits[:8]:
        if time.monotonic() > t0_end - 20.0:
            break
        sc = exact_score_a(init)
        if sc < best_sc:
            best_sc = sc
            best_seq = init.copy()
            print("Better init found: {:.8f}".format(best_sc))
        # Quick LSE on this init
        if time.monotonic() < t0_end - 10.0:
            mf = max(100, min(500, int((t0_end - time.monotonic() - 5.0) / 0.03)))
            sc2, a2 = run_lbfgs(init, 200.0, t0_end - 5.0, mf)
            if sc2 < best_sc:
                best_sc = sc2
                best_seq = a2.copy()
                print("Quick LSE improved: {:.8f}".format(best_sc))
    
    # === Phase 1: Main multi-stage optimization (40% of time) ===
    t1_end = t_start + 0.50 * total_budget
    sc, seq = multi_stage_optimize(best_seq, t1_end, n_stages=6)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 1 (multi-stage) best: {:.8f}".format(best_sc))
    else:
        print("Phase 1 ended at: {:.8f}".format(exact_score_a(seq)))
    
    # === Phase 2: Mirror descent with adaptive active set (15% of time) ===
    t2_end = t_start + 0.65 * total_budget
    sc, seq = adaptive_mirror_descent(best_seq, max_iters=200, deadline=t2_end,
                                      k_list=(1, 2, 3, 4, 5, 7, 10, 15))
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 2 (mirror descent) best: {:.8f}".format(best_sc))
    
    # Also try Nesterov with high beta
    if time.monotonic() < t2_end - 20.0:
        max_it = max(100, int((t2_end - time.monotonic() - 10.0) * 1.5))
        sc, seq = projected_gradient_nesterov_full(best_seq, 100000.0, max_it, t2_end - 10.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 2b (Nesterov high-beta) best: {:.8f}".format(best_sc))
    
    # === Phase 3: Exact subgradient polish (10% of time) ===
    t3_end = t_start + 0.75 * total_budget
    sc, seq = exact_subgrad_polish(best_seq, max_iters=60, deadline=t3_end,
                                   k_list=(1, 2, 3, 4, 5, 6, 8, 10, 12))
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 3 (subgrad polish) best: {:.8f}".format(best_sc))
    
    # === Phase 4: Very high beta LSE (15% of time) ===
    t4_end = t_start + 0.90 * total_budget
    a = normalize_a(best_seq)
    for beta in (100000.0, 500000.0, 1000000.0, 2000000.0):
        if time.monotonic() > t4_end - 15.0:
            break
        rem = float(t4_end - time.monotonic())
        if rem < 15.0:
            break
        
        # Use L-BFGS-B for this
        mf = max(100, min(1500, int(rem * 0.5 / max(0.01, 0.02 * len(a) / 8192.0))))
        sc, a2 = run_lbfgs(a, beta, t4_end - 5.0, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("Phase 4 (LSE b={:.0f}) best: {:.8f}".format(beta, best_sc))
        
        # Also try Nesterov
        if time.monotonic() < t4_end - 10.0:
            max_it = max(50, int((t4_end - time.monotonic() - 5.0)))
            sc, a2 = nesterov_accelerated(a2, beta, max_it, t4_end - 5.0)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Phase 4 (Nest b={:.0f}) best: {:.8f}".format(beta, best_sc))
        
        a = normalize_a(a2)
    
    # === Phase 5: Perturbation restarts (10% of time) ===
    t5_end = deadline - 15.0
    perturb_count = 0
    while time.monotonic() < t5_end and perturb_count < 5:
        left = float(t5_end - time.monotonic())
        if left < 40.0:
            break
        chunk = time.monotonic() + left / float(5 - perturb_count) - 5.0
        
        p = perturb_best(best_seq, rng, strength=float(rng.uniform(0.03, 0.2)))
        
        # Quick multi-stage on perturbed
        if time.monotonic() < chunk - 10.0:
            sc, a2 = multi_stage_optimize(p, chunk, n_stages=4)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Perturb {} best: {:.8f}".format(perturb_count, best_sc))
            p = normalize_a(a2)
        
        # Mirror descent
        if time.monotonic() < chunk:
            sc, a2 = adaptive_mirror_descent(p, max_iters=30, deadline=chunk,
                                             k_list=(1, 2, 3, 5, 8))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Perturb {} MD best: {:.8f}".format(perturb_count, best_sc))
        
        perturb_count += 1
    
    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 10.0:
        sc, seq = exact_subgrad_polish(best_seq, max_iters=30, deadline=deadline - 5.0,
                                       k_list=(1, 2, 3, 4, 5, 7))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
        
        if time.monotonic() < deadline - 3.0:
            sc, seq = adaptive_mirror_descent(best_seq, max_iters=20, deadline=deadline - 3.0,
                                             k_list=(1, 2, 3))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
    
    # Final verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc
    
    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
