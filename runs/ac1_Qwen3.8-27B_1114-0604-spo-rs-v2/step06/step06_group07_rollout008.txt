import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    """Fast linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_flat(a, n_iters, alpha_start, alpha_end, M_fft=None):
    """Fienup with full-spectrum flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        
        # Use mean of non-DC as target
        C = np.mean(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_subgrad_step(a, n_peaks, step):
    """One subgradient step targeting top peaks."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
    S = float(a.sum())
    M_val = float(c.max())
    sc = 2.0 * n * M_val / (S * S)
    
    if n_peaks >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
    
    # Subgradient of max(conv) w.r.t. a
    sg = np.zeros(n)
    for k in top_idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            sg[js:je+1] += 2.0 * a[js:je+1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
    
    a -= step * grad
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, sc
    a = a / S
    return a, sc

    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(np.clip(scaled, -700, 0))
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. x
    # L = log-sum-exp of beta*c, shifted
    # dL/dx_i = sum_k p_k * dc_k/dx_i where p_k = exp(beta*(c_k-c_max))/sum
    p = exp_s / exp_sum

    # d/dx_i = x[k-i] (if valid) + x[k-i] (if valid, different term)

    # d c_k / d x_i:

    #     j=i: x[k-i] (derivative of x[i]*x[k-i] w.r.t. x[i])
    #     k-j=i => j=k-i: x[i] (derivative of x[k-i]*x[i] w.r.t. x[i])
    #   If i == k-i (2i = k): only one term, derivative is x[i]
    #   Otherwise: derivative is 2*x[k-i]

    """Run L-BFGS-B with our gradient."""
    n = len(a0)
    x0 = np.clip(a0, 0, 1000).copy()
    if x0.sum() < 1e-6:
        x0[:] = 0.5
    x0 = x0 / x0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        v, _ = lse_obj_grad(x, beta, M_fft)
        return v
    
    def grad(x):
        _, g = lse_obj_grad(x, beta, M_fft)
        return g
    
    res = minimize(obj, x0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-12, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 30})
    x_out = np.clip(res.x, 0, 1000)
    if x_out.sum() < 1e-6:
        x_out = x0.copy()
    return x_out

def conv_equalize(a, n_iters, target_ratio=0.95):
    """
    Iteratively reduce peaks in the convolution by adjusting a multiplicatively.
    For each peak position k, reduce a[j] proportional to a[k-j] (the partner).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2*n-1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A*A, M_fft)[:2*n-1]
        S = float(a.sum())
        c_mean = (S*S) / (2*n-1)  # average conv value
        c_target = target_ratio * float(c.max())
        
        # Find positions above target
        above = c > c_target
        if not above.any():
            break
        
        # Compute "pressure" on each a[i] from the peaks
        pressure = np.zeros(n)
        above_idx = np.where(above)[0]
        for k in above_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                # a[j] and a[k-j] both contribute
                pressure[js:je+1] += a[js:je+1][::-1]
        
        # Multiplicative update: reduce elements with high pressure
        p_max = pressure.max()
        if p_max > 1e-20:
            reduction = 1.0 - 0.1 * (pressure / p_max)
            a = a * reduction
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def fast_conv(a):
    """Fast linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_flat(a, n_iters=300, alpha_start=0.05, alpha_end=0.9, M_fft=None):
    """Fienup iteration with full-spectrum geometric flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def peak_subgrad(a, n_peaks=10, step=0.01):
    """One subgradient step targeting top convolution peaks."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
    S = float(a.sum())
    M_val = float(c.max())
    if n_peaks >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
    sg = np.zeros(n)
    for k in top_idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a
    return a_new / S_new


def peak_directed_descent(a, n_iters=2000, n_peaks=10, step_init=0.01, decay=0.99995):
    """Iterative peak-directed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        step = min(step, 0.02)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:

        a = a / S
        step *= decay

def lse_obj_grad(x, beta, M_fft):

    LSE-smoothed objective: 2n * LSE(beta*c) / S^2, with exact gradient.
    LSE approximates max(c) smoothly.

    A = np.fft.rfft(x, M_fft)

    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    # Gradient of L w.r.t. x[i] = 2 * sum_m x[m] * p[m+i] (cross-correlation)
    # Compute via FFT: g_L[i] = 2 * conv(x, p_flip)[n-1+i] ... actually:
    # g_L[i] = 2 * sum_{m=0}^{n-1} x[m] * p[m+i]
    # = 2 * conv_full[n-1 : 2*n-1][::-1] where conv_full = conv(x, p[::-1])
    p_flip = p[::-1]
    M2 = n + M - 1  # = 3n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    PX = np.fft.rfft(x, M2_fft)
    PF = np.fft.rfft(p_flip, M2_fft)
    conv_full = np.fft.irfft(PX * PF, M2_fft)[:M2]
    g_L = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_beta(a0, beta, M_fft, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective with given beta."""
    n = len(a0)
    x0 = np.clip(a0, 0, 1000).astype(np.float64).copy()
    if x0.sum() < 1e-6:
        x0[:] = 0.5
    x0 = x0 / x0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        v, _ = lse_obj_grad(x, beta, M_fft)
        return v

    def jac(x):
        _, g = lse_obj_grad(x, beta, M_fft)
        return g

    try:
        res = minimize(obj, x0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun,
                                'maxls': 30})
        x_out = np.clip(res.x, 0, 1000)
    except Exception:
        x_out = x0.copy()
    if x_out.sum() < 1e-6:
        x_out = x0.copy()
    return x_out


def conv_equalize(a, n_iters=200, max_reduction=0.05):
    """
    Reduce convolution peaks by multiplicatively reducing the elements
    that contribute most to each peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        c_max = float(c.max())
        c_target = 0.98 * c_max
        above_idx = np.where(c > c_target)[0]
        if len(above_idx) == 0:
            break
        pressure = np.zeros(n)
        for k in above_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                pressure[js:je + 1] += a[js:je + 1][::-1]
        p_max = pressure.max()
        if p_max > 1e-20:
            reduction = 1.0 - max_reduction * (pressure / p_max)
            a = a * reduction
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * x) +
                            0.3 * np.cos(4.0 * np.pi * x), 0.01))
    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    inits.append(np.maximum(0.2 + 0.6 * x ** 0.7 + 0.6 * (1 - x) ** 0.7, 0.01))
    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
    return inits


def stochastic_hill_climb(a, n_trials=500, sigma_start=0.02, sigma_end=0.002):
    """
    Stochastic hill climbing: perturb random subsets of elements and
    accept if score improves.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)
    n_perturb = max(3, n // 50)
    for it in range(n_trials):
        progress = it / max(1, n_trials)
        sigma = sigma_start + (sigma_end - sigma_start) * progress
        idx = rng.integers(0, n, size=n_perturb)
        a_test = a.copy()
        a_test[idx] *= (1.0 + sigma * rng.standard_normal(n_perturb))
        a_test = np.maximum(a_test, 0.0)
        S_test = float(a_test.sum())
        if S_test < 1e-12:
            continue
        a_test = a_test / S_test
        sc_test = compute_score(a_test)
        if sc_test < best_sc - 1e-12:
            best_sc = sc_test
            best_a = a_test.copy()
            a = a_test
    return best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: multi-stage optimization combining Fienup spectral shaping,
    L-BFGS with LSE smoothing (beta annealing), peak-directed subgradient,
    convolution equalization, and stochastic hill climbing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 45.0

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    M_fft = 1 << (2 * n_work - 1).bit_length()

    best_score = np.inf
    best_a = np.ones(n_work) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Working length: {n_work}, M_fft={M_fft}")
    print(f"Parent available: {parent is not None}")

    if parent is not None and len(parent) == n_work:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ============================================================
    # Phase 1: Diverse spectral shaping (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - 45.0)
    print(f"\n=== Phase 1: Spectral Shaping (deadline t={phase1_end - t_start:.0f}s) ===")

    inits = build_initializations(n_work, rng, parent)
    print(f"  {len(inits)} initializations generated")

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        sc0 = compute_score(a0)
        print(f"  Init {i}: score={sc0:.6f}")
        update_best(sc0, a0, f"INIT {i}")

        # Fienup flat
        a_g = fienup_flat(a0, n_iters=200, alpha_start=0.05, alpha_end=0.9, M_fft=M_fft)
        sc = compute_score(a_g)
        update_best(sc, a_g, f"FLAT i={i}")

        # Power-law Fienup
        if time.monotonic() < phase1_end - 30:
            a_p = fienup_power_law(a0, n_iters=200, gamma=0.3, M_fft=M_fft)
            sc = compute_score(a_p)
            update_best(sc, a_p, f"POWL i={i}")

    # Additional Fienup rounds from best
    if time.monotonic() < phase1_end:
        a_f = fienup_flat(best_a, n_iters=300, alpha_start=0.02, alpha_end=0.7, M_fft=M_fft)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FLAT2")
        a_f2 = fienup_power_law(best_a, n_iters=300, gamma=0.25, M_fft=M_fft)
        sc = compute_score(a_f2)
        update_best(sc, a_f2, "POWL2")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: L-BFGS with beta annealing (45% time)
    # ============================================================
    phase2_end = t_start + 0.70 * (total_budget - 45.0)
    print(f"\n=== Phase 2: L-BFGS Beta Annealing (deadline t={phase2_end - t_start:.0f}s) ===")

    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 90:
            break
        maxiter = min(6000, int(remaining * 25))
        maxfun = maxiter * 20
        print(f"  beta={beta}, remaining={remaining:.0f}s, maxiter={maxiter}")
        a_opt = run_lbfgs_beta(best_a, beta, M_fft, maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            # Interleave a subgradient step
            if time.monotonic() < phase2_end - 30:
                for _ in range(200):
                    best_a = peak_subgrad(best_a, n_peaks=10, step=0.005)
                c = fast_conv(best_a)
                S = float(best_a.sum())
                sc = 2.0 * n_work * float(c.max()) / (S * S)
                update_best(sc, best_a, f"SUBGRAD b={beta}")

    # Additional L-BFGS from parent
    if parent is not None and len(parent) == n_work:
        for beta in [100, 500, 2000]:
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 60:
                break
            print(f"  [parent] beta={beta}, remaining={remaining:.0f}s")
            a_opt = run_lbfgs_beta(parent, beta, M_fft, maxiter=3000, maxfun=60000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt, f"LBFGS-P b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Peak-directed + Conv equalization (15% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 45.0)
    print(f"\n=== Phase 3: Peak-Directed + Equalization (deadline t={phase3_end - t_start:.0f}s) ===")

    for n_peaks in [5, 10, 20, 50, 100]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        n_iters = min(1500, int(remaining * 5))
        if n_iters < 100:
            n_iters = 100
        print(f"  peak_directed n_peaks={n_peaks}, iters={n_iters}")
        a_pd = peak_directed_descent(best_a, n_iters=n_iters, n_peaks=n_peaks,
                                     step_init=0.008, decay=0.99998)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

        # Conv equalization
        if time.monotonic() < phase3_end - 30:
            a_eq = conv_equalize(best_a, n_iters=150, max_reduction=0.03)
            sc = compute_score(a_eq)
            update_best(sc, a_eq, f"EQUI m={n_peaks}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish + stochastic (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # More L-BFGS at high beta
    if time.monotonic() < deadline - 120:
        for beta in [5000, 15000, 30000]:
            if time.monotonic() > deadline - 60:
                break
            remaining = min(deadline - 60 - time.monotonic(), 180)
            if remaining < 30:
                break
            maxiter = int(remaining * 20)
            print(f"  FINAL LBFGS beta={beta}, maxiter={maxiter}")
            a_opt = run_lbfgs_beta(best_a, beta, M_fft, maxiter=maxiter,
                                   maxfun=max(maxiter * 15, 1000))
            sc = compute_score(a_opt)
            update_best(sc, a_opt, f"FIN-LBFGS b={beta}")

    # Stochastic hill climbing
    if time.monotonic() < deadline - 60:
        print("  Stochastic hill climbing...")
        a_shc = stochastic_hill_climb(best_a, n_trials=300,
                                      sigma_start=0.015, sigma_end=0.003)
        sc = compute_score(a_shc)
        update_best(sc, a_shc, "STOCH")

    # More peak-directed refinement
    if time.monotonic() < deadline - 40:
        a_fin = peak_directed_descent(best_a, n_iters=500, n_peaks=15,
                                      step_init=0.003, decay=0.99999)
        sc = compute_score(a_fin)
        update_best(sc, a_fin, "FIN-PEAK")

    # Final Fienup polish
    if time.monotonic() < deadline - 30:
        a_fp = fienup_flat(best_a, n_iters=100, alpha_start=0.02, alpha_end=0.4, M_fft=M_fft)
        sc = compute_score(a_fp)
        update_best(sc, a_fp, "FIN-FLAT")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.01 * (1 + trial * 0.5)
            a_p = best_a * (1.0 + noise * rng.standard_normal(n_work))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            a_r = fienup_flat(a_p, n_iters=80, alpha_start=0.05, alpha_end=0.5, M_fft=M_fft)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
