    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if target_sum <= 0:
        return np.zeros(n)
    # Bisection on threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find rho such that sum(max(v - t, 0)) = target_sum
    # Lo and hi for bisection
    lo = min(v.min() - 1.0, 0.0)
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.maximum(v - t, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    else:
        result = np.full(n, target_sum / n)
    return result


def subgradient_peak_reduction(a, n_iters=500, eta_init=0.1):
    """
    Subgradient method targeting the max convolution peak directly.
    Uses projection onto the simplex (non-negativity + sum constraint).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        c = np.fft.irfft(np.fft.rfft(a, M_fft) ** 2, M_fft)[:2 * n - 1]
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all positions within 1% of max (to handle ties)
        threshold = c_max * 0.995
        peak_positions = np.where(c >= threshold)[0]

        # Compute subgradient (average over peak positions)
        sg = np.zeros(n)
        for kp in peak_positions:
            kp = int(kp)
            # d(a*a)[kp]/da[m] = 2*a[kp-m] if 0 <= kp-m < n
            j_start = max(0, kp - n + 1)
            j_end = min(kp, n - 1)
            if j_end >= j_start:
                # sg[m] += 2 * a[kp - m] for m in [j_start, j_end]
                # kp - m ranges from kp-j_start to kp-j_end
                vals = a[kp - j_end:kp - j_start + 1][::-1] * 2.0
                sg[j_start:j_end + 1] += vals

        sg /= len(peak_positions)

        # Score gradient includes the sum normalization
        # score = 2n * c_max / S^2
        # d score / da = 2n * (sg / S^2 - 2*c_max / S^3)
        S2 = S * S
        S3 = S2 * S
        full_grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)

        # Adaptive step size
        grad_norm = np.linalg.norm(full_grad)
        if grad_norm < 1e-15:
            break
        step = eta * full_grad / grad_norm

        # Try the step
        a_new = project_simplex(a - step, target_sum)
        sc_new = compute_score(a_new)

        # Backtracking line search
        for shrink in range(10):
            if sc_new < sc:
                a = a_new
                eta = min(eta * 1.05, 1.0)
                break
            eta *= 0.7
            step = eta * full_grad / grad_norm
            a_new = project_simplex(a - step, target_sum)
            sc_new = compute_score(a_new)
        else:
            # No improvement, try smaller steps
            for shrink2 in range(5):
                eta *= 0.5
                step = eta * full_grad / grad_norm
                a_new = project_simplex(a - step, target_sum)
                sc_new = compute_score(a_new)
                if sc_new < sc:
                    a = a_new
                    break

        eta = max(eta, 0.001)

    return best_a


def build_conv_matrix(f):
    """Build the sparse convolution constraint matrix A where (A@ g)[k] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # g[j] contributes to (f*g)[k] with coefficient f[k-j] for k in [j, j+n-1]
        k_vals = np.arange(j, j + n)
        valid = k_vals < 2 * n - 1
        k_vals = k_vals[valid]
        f_idx = k_vals - j
        rows.extend(k_vals.tolist())
        cols.extend([j] * len(k_vals))
        vals.extend(f[f_idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_improvement_step(f, max_seconds=8.0):
    """
    LP-based step: find g maximizing sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, M) or (None, None) on failure.
    """
    n = len(f)
    c = fast_conv(f)
    M = float(c.max())
    S = float(f.sum())
    if S < 1e-12 or M < 1e-12:
        return None, None

    # Build constraint matrix
    t0 = time.monotonic()
    A = build_conv_matrix(f)

    # Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0
    # Equivalent to: minimize -sum(g) s.t. A@g <= M, g >= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_seconds})
        if res.success:
            g = res.x
            g = np.maximum(g, 0.0)
            # Verify
            g_conv = fast_conv(g)
            if float(g_conv.max()) <= M * (1 + 1e-6):
                return g, M
            else:
                # Slightly violated, scale down
                scale = M / (float(g_conv.max()) + 1e-30)
                g = g * min(1.0, scale * 0.999)
                return g, M
    except Exception:
        pass
    return None, None


def lp_based_optimization(a, n_iters=100, max_lp_time=5.0, deadline=None):
    """
    Iterative LP-based optimization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / S

    best_a = a.copy()
    best_sc = compute_score(a)
    print(f"  LP-opt start: score={best_sc:.8f}")

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        if time.monotonic() - t0 > max_lp_time * 3:
            break

        g, M = lp_improvement_step(a, max_seconds=max_lp_time)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g < 1e-12:
            break

        # Rescale g to have target sum
        g_scaled = g * target_sum / sum_g

        # Blend
        t = 0.1 + 0.1 * min(it / 10.0, 1.0)  # Start with small blend, increase
        a_new = (1 - t) * a + t * g_scaled
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * target_sum / S_new

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            if it % 10 == 0:
                print(f"  LP-opt it={it}: score={sc_new:.8f}")
        a = a_new

    return best_a


def fienup_enhanced(a, n_iters=500, deadline=None):
    """Enhanced Fienup with adaptive blending and non-negativity repair."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Adaptive alpha: aggressive early, gentle late
        alpha = 0.7 * (1.0 - progress) ** 2 + 0.05
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        # Gentle spectral update
        ratio = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * ratio ** alpha
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        # Repair: add small uniform floor to maintain all elements positive
        floor = 1e-6 * P_mean / n
        a_new = np.maximum(a_new, floor)

        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * target_sum / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_descent_polish(a, n_sweeps=5, deadline=None):
    """
    Coordinate descent: for each element, find the optimal value
    that minimizes the max convolution, holding others fixed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Randomize order
        order = np.random.permutation(n)

        for idx in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            j = int(idx)
            old_val = a[j]
            if old_val < 1e-10:
                # Try to increase from zero
                candidates = [old_val, 0.5 * S / n, S / n]
            else:
                # Try scaling the current value
                candidates = [old_val * f for f in [0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.1, 1.3, 1.5]]
                candidates.append(0.0)

            best_val = old_val
            best_local_sc = compute_score(a)

            for cv in candidates:
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = old_val
                    continue
                a_norm = a / S_tmp
                sc_tmp = compute_score(a_norm)
                if sc_tmp < best_local_sc:
                    best_local_sc = sc_tmp
                    best_val = cv
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def stochastic_restarts(a, rng, n_restarts=8, deadline=None, lse_beta=5000):
    """Random perturbations followed by quick L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(n_restarts):
        if deadline is not None and time.monotonic() > deadline:
            break

        strategy = r % 4
        if strategy == 0:
            a_pert = a * (1.0 + 0.02 * rng.standard_normal(n))
        elif strategy == 1:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.05 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 2:
            # Block perturbation
            block_size = max(1, n // 32)
            block = rng.integers(0, n - block_size)
            a_pert = a.copy()
            a_pert[block:block + block_size] *= (1.0 + 0.15 * rng.standard_normal(block_size))
        else:
            # Peak targeting: perturb elements contributing to the peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            noise = rng.standard_normal(n) * 0.01
            noise[j_start:j_end + 1] *= 3.0
            a_pert = a * (1.0 + noise)

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        sc, a_opt = run_lbfgs(a_pert, lse_beta, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Restart {r}: {sc:.8f}")

    return best_a, best_sc


def adaptive_lbfgs_schedule(a, deadline, betas=None):
    """
    Run L-BFGS with increasing beta schedule.
    """
    if betas is None:
        betas = [50, 200, 500, 1000, 3000, 10000, 30000, 100000, 300000]

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate iterations based on remaining time and beta
        frac = (i + 1) / len(betas)
        maxiter = int(min(20000, remaining * 10 * (0.5 + 0.5 * frac)))
        maxiter = max(maxiter, 2000)

        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 30)
        dt = time.monotonic() - t0
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    L-BFGS beta={beta}: {sc:.8f} ({dt:.1f}s)")
        else:
            print(f"    L-BFGS beta={beta}: {sc:.8f} (no improvement, {dt:.1f}s)")

    return best_a, best_sc


def symmetry_exploit(a, deadline=None):
    """Try symmetric optimization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    half = (n + 1) // 2
    a_sym = 0.5 * (a + a[::-1])
    x = a_sym[:half].copy()
    x = np.maximum(x, 0.0)
    sx = x.sum()
    if sx < 1e-12:
        return a
    x = x / sx

    def expand_sym(x_vec):
        a_full = np.zeros(n)
        a_full[:half] = x_vec
        a_full[half:] = x_vec[:n - half][::-1]
        return a_full

    a_full = expand_sym(x)
    S_full = a_full.sum()
    a_full = a_full / S_full
    best_sc = compute_score(a_full)
    best_x = x.copy()

    # LSE gradient on the symmetric half
    M_fft = 1 << (2 * n - 1).bit_length()
    for it in range(3000):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a_full)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S_full * S_full)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()

        beta = 2000
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + np.log(exp_s.sum()) / beta

        S2 = S_full * S_full
        S3 = S2 * S_full
        a_rev = a_full[::-1]
        PA = np.fft.rfft(p, M_fft)
        AA = np.fft.rfft(a_rev, M_fft)
        h = np.fft.irfft(AA * PA, M_fft)[:2 * n - 1][n - 1:2 * n - 1]
        grad_full = 2.0 * n * (h / S2 - 2.0 * L / S3)

        # Project to symmetric half
        grad_sym = 0.5 * (grad_full[:half] + grad_full[:half][::-1])

        gn = np.linalg.norm(grad_sym)
        if gn < 1e-15:
            break
        step = 0.01 / gn
        x_new = x - step * grad_sym
        x_new = np.maximum(x_new, 1e-12)
        x_new = x_new / x_new.sum()
        x = x_new
        a_full = expand_sym(x)
        S_full = a_full.sum()
        a_full = a_full / S_full

    best_a = expand_sym(best_x)
    best_a = best_a / best_a.sum()
    return best_a


def multi_length_explore(best_a, best_sc, deadline, seed=42):
    """Try different sequence lengths by interpolation."""
    rng = np.random.default_rng(seed)
    n_orig = len(best_a)
    candidate_lengths = [n_orig // 2, n_orig, n_orig * 2]
    # Add some random lengths
    for dl in [int(n_orig * 0.75), int(n_orig * 1.5)]:
        if dl not in candidate_lengths and dl >= 128:
            candidate_lengths.append(dl)

    for target_n in candidate_lengths:
        if deadline is not None and time.monotonic() > deadline:
            break
        if target_n < 64 or target_n > 16384:
            continue
        if target_n == n_orig:
            continue

        # Interpolate
        x_old = np.linspace(0, 1, n_orig)
        x_new = np.linspace(0, 1, target_n)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.0)
        S = a_interp.sum()
        if S < 1e-12:
            continue
        a_interp = a_interp / S

        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_interp, 5000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Multi-length n={target_n}: {sc:.8f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization combining LP steps, subgradient method,
    LSE-based L-BFGS, spectral equalization, coordinate descent, and restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform n=2048")

    n = len(best_a)
    print(f"Initial best: {best_score:.8f}")

    # ================================================================
    # Phase 1: Adaptive L-BFGS (25% of time)
    # ================================================================
    p1_end = t_start + 0.25 * (total_budget - 60.0)
    print(f"\n=== Phase 1: L-BFGS schedule (until t={p1_end - t_start:.0f}s) ===")
    a_lb, sc_lb = adaptive_lbfgs_schedule(best_a, p1_end)
    update(sc_lb, a_lb, "LBFGS-schedule")

    # ================================================================
    # Phase 2: LP-based optimization (20% of time)
    # ================================================================
    p2_end = t_start + 0.45 * (total_budget - 60.0)
    print(f"\n=== Phase 2: LP-based optimization (until t={p2_end - t_start:.0f}s) ===")
    t0 = time.monotonic()
    a_lp = lp_based_optimization(best_a, n_iters=200, max_lp_time=5.0, deadline=p2_end)
    sc_lp = compute_score(a_lp)
    update(sc_lp, a_lp, "LP-opt")

    # ================================================================
    # Phase 3: Subgradient + Spectral (20% of time)
    # ================================================================
    p3_end = t_start + 0.65 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Subgradient + Fienup (until t={p3_end - t_start:.0f}s) ===")

    if time.monotonic() < p3_end:
        a_sg = subgradient_peak_reduction(best_a, n_iters=300, eta_init=0.05)
        sc_sg = compute_score(a_sg)
        update(sc_sg, a_sg, "SUBGRAD")

    if time.monotonic() < p3_end:
        a_f = fienup_enhanced(best_a, n_iters=400, deadline=p3_end)
        sc_f = compute_score(a_f)
        update(sc_f, a_f, "FIENUP")
        # L-BFGS polish after Fienup
        sc_f2, a_f2 = run_lbfgs(a_f, 5000, maxiter=5000, maxfun=100000)
        update(sc_f2, a_f2, "FIENUP+LBFGS")

    # ================================================================
    # Phase 4: Coordinate descent + L-BFGS polish (15% of time)
    # ================================================================
    p4_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Coord descent (until t={p4_end - t_start:.0f}s) ===")

    if time.monotonic() < p4_end:
        a_cd = coordinate_descent_polish(best_a, n_sweeps=3, deadline=p4_end)
        sc_cd = compute_score(a_cd)
        update(sc_cd, a_cd, "CD")

    if sc_cd < best_score and time.monotonic() < p4_end:
        sc_cd2, a_cd2 = run_lbfgs(a_cd, 10000, maxiter=8000, maxfun=150000)
        update(sc_cd2, a_cd2, "CD+LBFGS")

    # ================================================================
    # Phase 5: Stochastic restarts + symmetry + multi-length (10%)
    # ================================================================
    p5_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Restart + Symmetry + Multi-length (until t={p5_end - t_start:.0f}s) ===")

    a_sr, sc_sr = stochastic_restarts(best_a, rng, n_restarts=6, deadline=p5_end, lse_beta=5000)
    update(sc_sr, a_sr, "RESTART")

    if time.monotonic() < p5_end:
        a_sym = symmetry_exploit(best_a, deadline=p5_end)
        sc_sym = compute_score(a_sym)
        update(sc_sym, a_sym, "SYMMETRY")
        if sc_sym < best_score:
            sc_sym2, a_sym2 = run_lbfgs(a_sym, 5000, maxiter=5000, maxfun=100000)
            update(sc_sym2, a_sym2, "SYMMETRY+LBFGS")

    if time.monotonic() < p5_end:
        best_a, best_score = multi_length_explore(best_a, best_score, p5_end)
        print(f"    After multi-length: {best_score:.8f}")

    # ================================================================
    # Phase 6: Final high-beta polish (5%)
    # ================================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = int(min(15000, remaining * 12))
        maxiter = max(maxiter, 2000)
        sc, a_fin = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 25)
        update(sc, a_fin, "FINAL-LBFGS-200k")

    if time.monotonic() < deadline - 15:
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=5000, maxfun=100000)
        update(sc, a_fin, "FINAL-LBFGS-100k")

    if time.monotonic() < deadline - 10:
        a_cd = coordinate_descent_polish(best_a, n_sweeps=2, deadline=deadline - 5)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
