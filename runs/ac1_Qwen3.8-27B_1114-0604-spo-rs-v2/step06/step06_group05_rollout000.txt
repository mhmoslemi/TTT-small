Adaptive continuation LSE + exact subgradient + 1D coordinate descent
for minimizing self-convolution peak. Uses multiple n values and progressive
smoothing parameter increase.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    """Score = 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a_i = 2 * (p * a)_i (truncated conv)
    rev_a = a[::-1]
    L_rev = 2 * n - 2 + 1  # length of rev_a is n, p is 2n-1
    # We need (p * rev_a) at positions [n-1, 2n-2] which gives length n
    M2 = 2 * n - 1  # result length should be (2n-1) + n - 1 = 3n - 2
    M2_fft = 1 << (3 * n - 2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    # Full gradient of score
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

def lse_obj_func(x, beta):
    """Objective function for scipy minimize."""
    return lse_score_grad(x, beta)

def weighted_lse_score_grad(a, w, beta):
    """Weighted LSE score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    # Weighted max-like: LSE of w_k * c_k
    wc = w * c
    wc_max = float(wc.max())
    scaled = beta * (wc - wc_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = wc_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient
    rev_a = a[::-1]
    M2_fft = 1 << (3 * n - 3).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3

    return weighted_lse_score_grad(x, w, beta)

    """Run L-BFGS-B on LSE objective."""
    n = len(a0)

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)

    res = minimize(
        weighted_lse_obj, a0, args=(w, beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)

def subgradient_exact(a0, n_iters=5000, n_peaks=10, step_init=0.01):
    """Exact subgradient descent on max(c)/(sum)^2."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Polyak step size parameters
    f_best = best_sc
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            f_best = best_sc
        
        # Polyak step size
        if f_best > 1.0:
            step_polyak = (sc - f_best) / max(np.sum((c_max / (S*S)) * 2.0 * n * np.ones(1))**2, 1e-20)
            step = min(step, max(step_polyak, 1e-8))
        
        # Subgradient: average gradient over top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        count = 0
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
                count += 1
        if count > 0:
            sg /= count
        
        # Full gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg * S - 2.0 * c_max) / S3
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        grad = grad / gn  # normalize
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
        
        # Add small random perturbation to escape
        if it % 500 == 0 and it > 0:
            a += 0.001 * np.random.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a

def coordinate_descent_1d(a0, n_sweeps=20, top_k=5):
    """1D coordinate descent on elements contributing to top peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        # Find elements contributing to these peaks
        active_elements = set()
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                if a[k - j] > 0.01 * float(np.max(a)):
                    active_elements.add(j)
        
        active_list = sorted(active_elements)[:200]  # limit for speed
        
        for j in active_list:
            # Try different values for a[j]
            old_val = a[j]
            S_old = float(a.sum())
            
            # Try a grid of values
            candidates = [old_val * f for f in [0.5, 0.7, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.3, 1.5]]
            candidates.append(0.0)
            
            best_val = old_val
            best_local_sc = compute_score(a)
            
            for cv in candidates:
                a[j] = cv
                S_new = float(a.sum())
                if S_new < 0.01:
                    a[j] = old_val
                    continue
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            
            a[j] = best_val
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def gs_flat(a0, n_iters=300, alpha_init=0.5, alpha_final=0.1):
    """GS-Fienup with flat convolution target."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    
    return inits
