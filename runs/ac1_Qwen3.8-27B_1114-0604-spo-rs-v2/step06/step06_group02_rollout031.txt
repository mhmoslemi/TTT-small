"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses: adaptive LSE-LBFGS, subgradient polishing, multi-length exploration,
and active-set refinement with SLSQP.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    
    def func_and_grad(x):
        sc, g = lse_grad(x, beta)
        return sc, g
    
    res = minimize(
        func_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def subgradient_polish(a, n_iters=50000, alpha_init=0.1, alpha_decay=0.9999, 
                       top_k=10, deadline=None):
    """
    Subgradient method with averaging for the exact max objective.
    Uses a distribution over the top-K peaks for a "smoothed" subgradient.
    """
    n = len(a)
    a_s = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a_s.sum()
    if S < 1e-12:
        return a
    a_s = a_s / S
    
    best_a = a_s.copy()
    best_sc = compute_score(a_s)
    alpha = alpha_init
    
    # Running average (for the subgradient averaging trick)
    a_avg = a_s.copy()
    weight_sum = 1.0
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a_s)
        c_max = float(c.max())
        
        if it % 1000 == 0:
            sc = 2.0 * n * c_max / (float(a_s.sum()) ** 2)
            if sc < best_sc:
                best_sc = sc
                best_a = a_s.copy()
        
        # Get top-K peaks
        if top_k <= len(c):
            top_idx = np.argsort(c)[::-1][:top_k]
        else:
            top_idx = np.arange(len(c))
        
        # Create distribution over top-K peaks (weight by proximity to max)
        top_vals = c[top_idx]
        margin = top_vals - top_vals[-1] + 1e-10
        p = margin / margin.sum()
        
        # Compute "smoothed" subgradient
        # This is equivalent to LSE with a discrete distribution
        a_rev = a_s[::-1]
        M = 2 * n - 1
        M_fft = 1 << (M - 1).bit_length()
        PA = np.fft.rfft(p, M_fft)
        AA = np.fft.rfft(a_rev, M_fft)
        h_full = np.fft.irfft(AA * PA, M_fft)[:M]
        h = h_full[n - 1:2 * n - 1]
        
        # Constrained gradient (orthogonal to 1)
        h_c = h - h.mean()
        
        # Adaptive step size (backtracking)
        gnorm = np.linalg.norm(h_c)
        if gnorm < 1e-15:
            break
        
        step_dir = -h_c / gnorm
        
        # Try a few step sizes
        best_step_alpha = alpha
        best_step_score = compute_score(a_s)
        for trial_alpha in [alpha, alpha * 0.5, alpha * 2.0]:
            a_trial = a_s + trial_alpha * step_dir
            a_trial = project_simplex(a_trial)
            sc_trial = compute_score(a_trial)
            if sc_trial < best_step_score:
                best_step_score = sc_trial
                best_step_alpha = trial_alpha
        
        a_s = a_s + best_step_alpha * step_dir
        a_s = project_simplex(a_s)
        
        # Update running average (Polyak averaging)
        weight_sum += 1.0
        a_avg = a_avg * (1.0 - 1.0 / weight_sum) + a_s * (1.0 / weight_sum)
        
        # Decay step size slowly
        alpha *= alpha_decay
        alpha = max(alpha, 0.001)
    
    # Check the average iterate
    sc_avg = compute_score(a_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = a_avg.copy()
    
    return best_a

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Aggressive LSE-beta schedule with per-level convergence checks
2. Subgradient polish with adaptive backtracking and Polyak averaging
3. Active-set SLSQP refinement targeting top peaks
4. Spectral shaping initialization and multi-restart exploration
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    def func_and_grad(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(
        func_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_polish(a, deadline, top_k=8, alpha_init=0.05, n_max_iters=200000):
    n = len(a)
    a_s = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a_s.sum()
    if S < 1e-12:
        return a
    a_s = a_s / S

    best_a = a_s.copy()
    best_sc = compute_score(a_s)
    alpha = alpha_init

    a_avg = a_s.copy()
    weight_sum = 1.0
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a_s)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (float(a_s.sum()) ** 2)

        if sc < best_sc:
            best_sc = sc
            best_a = a_s.copy()

        if it % 5000 == 0:
            sc_avg = compute_score(a_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = a_avg.copy()

        if top_k <= len(c):
            top_idx = np.argsort(c)[::-1][:top_k]
        else:
            top_idx = np.arange(len(c))

        top_vals = c[top_idx]
        spread = top_vals - top_vals[-1]
        p = (spread + 0.1) / (spread + 0.1).sum()

        a_rev = a_s[::-1]
        M = 2 * n - 1
        PA = np.fft.rfft(p, M_fft)
        AA = np.fft.rfft(a_rev, M_fft)
        h_full = np.fft.irfft(AA * PA, M_fft)[:M]
        h = h_full[n - 1:2 * n - 1]

        h_c = h - h.mean()
        gnorm = np.linalg.norm(h_c)
        if gnorm < 1e-15:
            break

        step_dir = -h_c / gnorm

        best_a_trial = a_s
        best_sc_trial = sc
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.1, alpha * 2.0]:
            a_trial = a_s + trial_alpha * step_dir
            a_trial = project_simplex(a_trial)
            sc_trial = compute_score(a_trial)
            if sc_trial < best_sc_trial:
                best_sc_trial = sc_trial
                best_a_trial = a_trial

        a_s = best_a_trial.copy()

        weight_sum += 1.0
        a_avg = a_avg * (1.0 - 1.0 / weight_sum) + a_s * (1.0 / weight_sum)

        if best_sc_trial < sc:
            alpha = min(alpha * 1.02, 0.3)
        else:
            alpha *= 0.95
        alpha = max(alpha, 0.001)

    sc_avg = compute_score(a_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = a_avg.copy()

    return best_a


def slsqp_refine(a0, n_active=8, maxiter=200):
    n = len(a0)
    a0 = a0.copy()
    S = a0.sum()
    if S < 1e-12:
        return compute_score(a0), a0
    a0 = a0 / S

    c = fast_conv(a0)
    top_idx = np.argsort(c)[::-1][:n_active]

    def objective(x):
        return np.sum((x - a0) ** 2)

    def objective_grad(x):
        return 2.0 * (x - a0)

    constraints = []
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1).astype(int)
        ks = (k - js).astype(int)

        def make_con(js, ks):
            def con_fun(x):
                return target_M - np.dot(x[js], x[ks])
            def con_jac(x):
                g = np.zeros(n)
                g[js] -= x[ks]
                g[ks] -= x[js]
                return g
            return con_fun, con_jac

        con_fun, con_jac = make_con(js, ks)
        constraints.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac})

    def eq_fun(x):
        return np.sum(x) - 1.0

    def eq_jac(x):
        return np.ones(n)

    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})

    target_M = c_max * 0.9999
    bounds = [(0.0, 1000.0)] * n

    def target_M_wrapper(fn):
        def wrapped(x):
            return fn(x)
        return wrapped

    # We need target_M to be accessible in con_fun, but we can't use closure
    # So let's restructure: use a global-like approach via the constraints list
    # Actually, let me just use a different approach: penalty method

    # Penalty approach: minimize ||x - a0||^2 + mu * sum_k max(0, c_k - target_M)^2
    target_M = float(c.max()) * 0.9999
    active_js = []
    active_ks = []
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1).astype(int)
        ks = (k - js).astype(int)
        active_js.append(js)
        active_ks.append(ks)

    mu = 0.01

    for _ in range(5):
        def pen_obj(x):
            val = np.sum((x - a0) ** 2)
            for js, ks in zip(active_js, active_ks):
                cv = np.dot(x[js], x[ks])
                viol = max(0.0, cv - target_M)
                val += mu * viol * viol
            return val

        def pen_grad(x):
            g = 2.0 * (x - a0)
            for js, ks in zip(active_js, active_ks):
                cv = np.dot(x[js], x[ks])
                if cv > target_M:
                    viol = cv - target_M
                    g[js] += 2.0 * mu * viol * x[ks]
                    g[ks] += 2.0 * mu * viol * x[js]
            return g

        res = minimize(pen_obj, a0, jac=pen_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

        a_new = np.clip(res.x, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new > 1e-12:
            a_new = a_new / S_new
        else:
            break
        a0 = a_new
        mu *= 10.0

    sc = compute_score(a0)
    return sc, a0


def spectral_init(n, rng):
    """Create a sequence with near-flat spectrum as initialization."""
    M_fft = 1 << (2 * n - 1).bit_length()
    # Random phase with constant magnitude
    n_freqs = M_fft // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    # Make it conjugate symmetric for real output
    A = np.ones(n_freqs)
    A *= np.exp(1j * phases)
    # Enforce conjugate symmetry
    A_full = np.zeros(M_fft, dtype=complex)
    A_full[:n_freqs] = A
    A_full[n_freqs:] = A[1:n_freqs][::-1].conj()
    if M_fft % 2 == 0:
        A_full[M_fft // 2] = np.abs(A_full[M_fft // 2])
    a = np.fft.irfft(A_full)[:n]
    a = np.abs(a)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    else:
        a = np.ones(n) / n
    return a


def fienup_quick(a, n_iters=200):
    """Quick Fienup iteration for initialization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.5 * (1.0 - it / n_iters) + 0.05
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
