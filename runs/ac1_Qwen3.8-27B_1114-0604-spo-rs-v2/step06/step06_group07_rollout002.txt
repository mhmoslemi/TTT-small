import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_shaping(a, n_iters, target='flat', alpha_schedule=None, n_modes=1):
    """
    Spectral shaping: iteratively adjust the amplitude spectrum toward a target.
    target: 'flat', 'tapered', 'raised_cosine'
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) * 0.5
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    k = np.r_[0, 1:len(a) // 2]  # frequency indices for rfft

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        
        if alpha_schedule is None:
            alpha = 0.3 + 0.5 * progress
        else:
            alpha = alpha_schedule(it, n_iters)

        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)

        # Compute target amplitude
        if target == 'flat':
            C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
            C = max(C, 1e-20)
            target_amp = np.full_like(amp, C)
            # Don't modify DC too much
            target_amp[0] = amp[0]
        elif target == 'tapered':
            C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
            C = max(C, 1e-20)
            # Slightly boost low frequencies (helps with non-negativity)
            t = np.linspace(0, 1, len(amp))
            target_amp = C * (1.0 + 0.3 * (1.0 - t))
            target_amp[0] = amp[0]
        else:  # 'raised_cosine'
            C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
            C = max(C, 1e-20)
            t = np.linspace(0, 1, len(amp))
            target_amp = C * (0.8 + 0.4 * np.cos(np.pi * t))
            target_amp[0] = amp[0]

        # Blend: amp_new = amp * (target/amp)^alpha
        amp_safe = np.maximum(amp, 1e-20)
        target_safe = np.maximum(target_amp, 1e-20)
        amp_new = amp * (target_safe / amp_safe) ** alpha

        # Reconstruct
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_subgradient(a, n_top=10, n_iters=5000, step_init=0.01, decay=0.99995):
    """
    Peak-directed subgradient descent.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        if it % 100 == 0:
            A = np.fft.rfft(a, M_fft)
            c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
            S = float(a.sum())
            M_val = float(c.max())
            sc = 2.0 * n * M_val / (S * S)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Compute subgradient at current peaks
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())

        # Find top peaks (only count peaks near the max)
        threshold = M_val * 0.99
        peak_mask = c >= threshold
        if peak_mask.sum() == 0:
            peak_mask[np.argmax(c)] = True

        # Subgradient: sg[i] = sum_{k in peaks} d c_k / d a_i
        # d c_k / d a_i = a[k-i] (if 0 <= k-i < n) + a[k-i] (if 0 <= k-i... )
        # Actually c_k = sum_j a_j a_{k-j}, so d c_k / d a_i = a[k-i] (from j=i)
        # plus a[k-i] (from k-j=i, i.e., j=k-i), unless i = k-i
        # Simplified: d c_k / d a_i = 2*a[k-i] if i != k-i, a[i] if i == k-i
        # For the "off-diagonal" case, it's 2*a[k-i]
        # For the "diagonal" case (2i=k), it's a[i]
        # Since we're summing over peaks, let's just use:
        # sg[i] += a[k-i] for each peak k (this counts each term once)
        # Actually, the gradient of c_k w.r.t. a is:
        # (grad)_[i] = a[(k-i) % ... ] ... this is getting confusing
        # Let me use the convolution approach:
        # grad of c_k w.r.t. a = 2 * circshift(a, k-i) restricted to valid indices
        # For linear convolution: d c_k / d a_i = a[k-i] if 0 <= k-i < n
        # But we need to be careful: c_k = sum_j a_j * a[k-j]
        # d/d a_i: the term a_j * a[k-j] is affected when j=i or k-j=i (j=k-i)
        # Case j=i: derivative is a[k-i]
        # Case j=k-i: derivative is a[i] (only if k-i != i, i.e., 2i != k)
        # So if 2i != k: d c_k / d a_i = a[k-i] + a[i]
        # If 2i == k: d c_k / d a_i = a[i]  (the two cases are the same)
        # Wait no. If 2i == k, then k-i = i, so:
        # Case j=i: a[k-i] = a[i]
        # Case j=k-i=i: a[i]
        # But these are the SAME term (j=i and j=k-i are the same when 2i=k)
        # So d c_k / d a_i = a[i] (counted once) when 2i=k
        # And d c_k / d a_i = 2*a[k-i] when 2i != k
        
        # For simplicity and vectorization, let's use:
        # sg[i] = sum_{k in peaks} a[k-i] (for valid k-i)
        # This is approximately correct (off by at most a factor of 2 for diagonal terms)
        
        sg = np.zeros(n)
        peaks_idx = np.where(peak_mask)[0]
        for k in peaks_idx:
            k = int(k)
            # a[k-i] is valid for i in [max(0, k-n+1), min(k, n-1)]
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                # a[k-i] for i in [i_start, i_end] = a[k-i_end : k-i_start+1][::-1]
                idx = k - np.arange(i_start, i_end + 1)
                sg[i_start:i_end + 1] += 2.0 * a[idx][:][::-1]
                # Wait, let me re-think:
                # For i in [i_start, i_end], a[k-i] gives:
                # a[k - i_start], a[k - i_start - 1], ..., a[k - i_end]
                # which is a[k-i_end : k-i_start+1][::-1]
        
        # Actually let me redo this more carefully
        # sg[i] should accumulate a[k-i] for each peak k where 0 <= k-i < n
        # i.e., i is in [k-n+1, k] intersected with [0, n-1]
        # Let me just loop over peaks (should be few)
        
        # Recompute sg properly
        sg = np.zeros(n)
        for k in peaks_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_hi >= i_lo:
                # For each i in [i_lo, i_hi], contribute a[k-i]
                # a[k-i] for i = i_lo, i_lo+1, ..., i_hi
                # = a[k-i_lo], a[k-i_lo-1], ..., a[k-i_hi]
                vals = a[k - i_hi : k - i_lo + 1][::-1]
                sg[i_lo:i_hi + 1] += 2.0 * vals

        # Gradient of score = 2n * (dg / S^2 - 2*M/S^3) where g = peak value
        # Actually, the subgradient of max(c) is in the cone over the gradients of active c_k
        # For LSE approximation, we'd weight by softmax. For hard max, we weight uniformly.
        # Let's use: subgrad_of_score = (2n/S^2) * sg - (4n*M/S^3) * ones
        # Wait: score = 2n * M / S^2
        # d score / d a_i = 2n * (dM/da_i * S^2 - M * 2S * da_i) / S^4
        # = 2n * (dM/da_i / S^2 - 2M / S^3)
        # where dM/da_i = subgradient of max = weighted sum of dc_k/da_i
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step
        sc_curr = 2.0 * n * M_val / S2
        step = min(step, max((sc_curr - sc_lb) / (gnorm * gnorm + 1e-30), 1e-12))
        step = min(step, 0.02)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

        if it % 1000 == 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
            print(f"    iter {it}: sc={sc_check:.8f}, best={best_sc:.8f}, step={step:.6f}")

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()
    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # For numerical stability with large beta
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient
    p = exp_scaled / exp_sum  # softmax weights
    # dL/dx_i = sum_k p_k * dc_k/dx_i
    # dc_k/dx_i = 2*x[k-i] (for valid k-i, and x[i] for diagonal)
    # Approximate: dL/dx_i ≈ 2 * sum_k p_k * x[k-i] (for valid k-i)
    # This is 2 * conv(p, x_reversed)[i + n - 1]
    
    # Use FFT: if P = FFT(p), X = FFT(x), then conv(p, x_rev) can be computed
    # Actually, let's use the direct approach with FFT
    # sum_k p_k * x[k-i] for i = 0, ..., n-1
    # This is the "cross-correlation" of p and x
    # = IFFT(FFT(p) * conj(FFT(x))) ... but sizes differ
    
    # p has length M = 2n-1, x has length n
    # We want g[i] = sum_{k=max(i,0)}^{min(M-1, i+n-1)} p[k] * x[k-i]
    # = sum_j p[i+j] * x[j] for j = 0, ..., min(n-1, M-1-i)
    # This is a "sliding dot product"
    
    # Use FFT: zero-pad x to length M_fft_x, compute cross-correlation
    M2 = M + n - 1  # length of full convolution of p and x
    M2_fft = 1 << (M2 - 1).bit_length()
    Px = np.fft.rfft(p, M2_fft)
    Xr = np.fft.rfft(x[::-1], M2_fft)  # reversed x
    corr = np.fft.irfft(Px * Xr, M2_fft)
    # corr[m] = sum_j p[j] * x_rev[m-j] = sum_j p[j] * x[n-1-m+j]
    # We want g[i] = sum_k p[k] * x[k-i]
    # Let's verify: corr[i + n - 1] = sum_j p[j] * x[n-1-(i+n-1)+j] = sum_j p[j] * x[j-i]
    # Hmm, that's not quite right. Let me think more carefully.
    
    # corr = conv(p, x_rev) where x_rev[j] = x[n-1-j]
    # corr[m] = sum_j p[j] * x_rev[m-j] = sum_j p[j] * x[n-1-(m-j)] = sum_j p[j] * x[n-1-m+j]
    # We want g[i] = sum_k p[k] * x[k-i] (for valid k-i, i.e., 0 <= k-i < n)
    # Let j = k, then g[i] = sum_j p[j] * x[j-i]
    # corr[m] = sum_j p[j] * x[n-1-m+j]
    # Set n-1-m = -i, i.e., m = n-1+i:
    # corr[n-1+i] = sum_j p[j] * x[j-i]
    # Yes! So g[i] = corr[n-1+i]
    
    g_L = 2.0 * corr[n - 1: 2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad
