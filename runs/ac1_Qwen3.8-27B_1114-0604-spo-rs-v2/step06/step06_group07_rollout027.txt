"""
Optimized step function construction using LSE-annealed optimization with
exact gradients, geometric-mean spectral shaping, and peak-directed
coordinate descent. Key improvements over previous approach:
1. Geometric beta annealing for LSE (smooth to sharp)
2. Exact gradient with diagonal correction
3. Geometric-mean targeting in Fienup (vs median)
4. Backtracking line search for peak-directed steps
5. Multi-start with structured perturbations
6. Final coordinate descent on peak contributors

import numpy as np
from scipy.optimize import minimize
import time


def fft_size(m):
    """Next power of 2 >= m."""
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mf = fft_size(M)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def compute_score(a):
    """Compute the objective value 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_geo_mean(a, n_iters=500, alpha_start=0.05, alpha_end=0.9):
    """Fienup with geometric-mean amplitude targeting."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters - 1))
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        # Geometric mean of non-DC amplitudes
        non_dc = amp[1:]
        if len(non_dc) > 0:
            C = np.exp(np.mean(np.log(np.maximum(non_dc, 1e-30))))
        else:
            C = max(amp[0], 1e-20)
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-30)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_flat_target(a, n_iters=500, alpha=0.5):
    """Fienup targeting exactly flat non-DC spectrum (amplitude = 1)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    # Target amplitude: sqrt(n) for normalized signal (Parseval)
    # If sum(a) = 1, then sum(a^2) <= 1, and |A(0)| = 1
    # For flat spectrum, |A(k)| = 1/sqrt(n) for all k (by Parseval)
    target = 1.0 / np.sqrt(n)

    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        amp_safe = np.maximum(amp, 1e-30)
        # Push non-DC toward target, keep DC
        amp_new = amp_safe * (target / amp_safe) ** alpha
        amp_new[0] = amp[0]  # Keep DC
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and exact gradient.
    Objective: 2*n*L / S^2 where L = (1/beta)*log(sum exp(beta*c_k))
    c = conv(x, x)
    """
    n = len(x)
    M = 2 * n - 1
    Mf = fft_size(M)
    A = np.fft.rfft(x, Mf)
    c = np.fft.irfft(A * A, Mf)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Probability weights
    p = exp_scaled / exp_sum

    # Compute g[i] = sum_j p[i+j] * x[j] for j=0..n-1
    # = conv(p, x_rev)[n-1 : 2n-1]
    x_rev = x[::-1]
    M2 = M + n - 1  # (2n-1) + n - 1 = 3n-2
    M2f = fft_size(M2)
    P = np.fft.rfft(p, M2f)
    XR = np.fft.rfft(x_rev, M2f)
    conv_res = np.fft.irfft(P * XR, M2f)[n - 1:2 * n - 1]

    # Exact gradient of L: 2*g[i] - p[2i]*x[i]
    diag_mask = np.arange(0, M, 2)  # 0, 2, 4, ..., 2n-2
    grad_L = 2.0 * conv_res - p[diag_mask] * x

    # Gradient of score = 2n*L/S^2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs_lse(a0, betas, maxiters_per_beta=5000):
    """Run L-BFGS-B with geometric beta annealing."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
        S = n * 0.5
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)
    bounds = [(0.0, 1000.0)] * n

    for beta in betas:
        res = minimize(
            lse_obj_grad, a.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-12,
                     'maxiter': maxiters_per_beta, 'maxfun': maxiters_per_beta * 30,
                     'maxls': 50}
        )
        a_opt = np.clip(res.x, 0.0, 1000.0)
        S_opt = a_opt.sum()
        if S_opt < 1e-6:
            a_opt = a.copy()
        else:
            a_opt = a_opt / S_opt
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        a = a_opt

    return best_score, best_a


def peak_coordinate_descent(a, n_iters=5000, n_peaks=10, max_step=0.05):
    """
    Coordinate descent targeting the peaks of conv(a,a).
    For each peak, identify the elements that contribute most and
    reduce them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        c = np.fft.irfft(A * A, Mf)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        k_peaks = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        Sgrad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Coordinate descent: find the element with largest positive gradient
        # and reduce it
        pos_mask = Sgrad > 0
        if not pos_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        pos_grads = Sgrad[pos_idx]
        # Pick the top few
        k_cd = min(20, len(pos_idx))
        cd_idx = pos_idx[np.argsort(pos_grads)[::-1][:k_cd]]

        # Try reducing each candidate
        best_improvement = 0.0
        best_new_a = None
        for idx in cd_idx:
            for frac in [0.5, 0.3, 0.1, 0.05]:
                a_trial = a.copy()
                a_trial[idx] *= (1.0 - frac)
                if a_trial[idx] < 0:
                    a_trial[idx] = 0.0
                S_trial = float(a_trial.sum())
                if S_trial < 1e-12:
                    continue
                a_trial = a_trial / S_trial
                A_t = np.fft.rfft(a_trial, Mf)
                c_t = np.fft.irfft(A_t * A_t, Mf)[:2 * n - 1]
                sc_t = 2.0 * n * float(c_t.max()) / (S_trial * S_trial)
                if sc < sc_t - 1e-12:
                    # This made it worse? No, we want lower
                    pass
                if sc_t < sc - best_improvement:
                    best_improvement = sc - sc_t
                    best_new_a = a_trial

        if best_new_a is not None:
            a = best_new_a
        else:
            # No improvement from coordinate descent, try gradient step
            gn = float(np.linalg.norm(Sgrad))
            if gn < 1e-15:
                break
            step = min(max_step, max((sc - 2.0) / (gn * gn + 1e-30), 1e-8))
            a_new = a - step * Sgrad
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
            else:
                # Backtrack
                for _ in range(5):
                    step *= 0.5
                    a_new = a - step * Sgrad
                    a_new = np.maximum(a_new, 0.0)
                    S_new = float(a_new.sum())
                    if S_new < 1e-12:
                        break
                    a_new = a_new / S_new
                    sc_new = compute_score(a_new)
                    if sc_new < sc:
                        a = a_new
                        break

    return best_a


def spectral_peak_shave(a, n_iters=200, n_top=20, strength=0.3):
    """
    Directly shave the top peaks in the power spectrum.
    This is equivalent to reducing the top peaks of the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        # Identify top amplitude bins (excluding DC)
        non_dc_idx = np.arange(1, len(amp))
        k_top = min(n_top, len(non_dc_idx))
        top_bins = non_dc_idx[np.argsort(amp[non_dc_idx])[::-1][:k_top]]
        # Shave them
        C = np.median(amp[non_dc_idx])
        for b in top_bins:
            amp[b] = amp[b] * (1.0 - strength) + C * strength
        A_new = amp * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

"""
Optimized step function construction using LSE-annealed optimization with
exact gradients (including diagonal correction), geometric-mean spectral
shaping, peak-directed coordinate descent, and adaptive time allocation.
Key improvements: (1) exact LSE gradient with diagonal correction,
(2) geometric beta annealing from smooth to sharp, (3) peak CD with
backtracking, (4) multi-scale spectral refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fft_size(m):
    """Next power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mf = fft_size(M)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_geom_mean(a, n_iters=500, alpha_start=0.03, alpha_end=0.95):
    """Fienup iteration with geometric-mean amplitude targeting."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters - 1))
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        non_dc = amp[1:]
        if len(non_dc) > 0:
            C = np.exp(np.mean(np.log(np.maximum(non_dc, 1e-30))))
        else:
            C = max(amp[0], 1e-20)
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-30)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def fienup_weighted(a, n_iters=400, n_top=50, strength=0.3):
    """Fienup that preferentially shaves the top-amplitude spectral bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        non_dc_idx = np.arange(1, len(amp))
        C = np.median(amp[non_dc_idx])
        C = max(C, 1e-20)
        k_top = min(n_top, len(non_dc_idx))
        top_bins = non_dc_idx[np.argsort(amp[non_dc_idx])[::-1][:k_top]]

        A_new = amp_new * np.exp(1j * phase)

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()
    return best_a


def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and exact gradient with diagonal correction.
    f(x) = 2n * L / S^2, L = (1/beta)*log(sum_k exp(beta*c_k)), c = conv(x,x)
    """
    n = len(x)
    M = 2 * n - 1
    Mf = fft_size(M)
    A = np.fft.rfft(x, Mf)
    c = np.fft.irfft(A * A, Mf)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # g[i] = sum_j p[i+j]*x[j], computed as conv(p, x_rev)[n-1 : 2n-1]
    x_rev = x[::-1]
    M2 = M + n - 1
    M2f = fft_size(M2)
    P = np.fft.rfft(p, M2f)
    XR = np.fft.rfft(x_rev, M2f)
    conv_res = np.fft.irfft(P * XR, M2f)[n - 1:2 * n - 1]

    # Exact gradient: 2*g[i] - p[2i]*x[i]
    diag_idx = np.arange(0, M, 2)
    grad_L = 2.0 * conv_res - p[diag_idx] * x

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=None):
    """Run L-BFGS-B on LSE objective at given beta."""
    if maxfun is None:
        maxfun = maxiter * 30
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
        S = n * 0.5
    a = a / S
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_grad, a.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        a_opt = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_opt = a.copy()
    S_opt = float(a_opt.sum())
    if S_opt < 1e-6:
        a_opt = a.copy()
    else:
        a_opt = a_opt / S_opt
    sc = compute_score(a_opt)
    return sc, a_opt


def peak_cd(a, n_iters=5000, n_peaks=10, n_candidates=10):
    """Peak-directed coordinate descent with backtracking."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    stall = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        c = np.fft.irfft(A * A, Mf)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

        k_peaks = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k_peaks)[-k_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        Sgrad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        pos_mask = Sgrad > 0
        if not pos_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        k_cd = min(n_candidates, len(pos_idx))
        cd_idx = pos_idx[np.argsort(Sgrad[pos_idx])[::-1][:k_cd]]

        improved = False
        for idx in cd_idx:
            for frac in [0.15, 0.08, 0.04]:
                new_val = a[idx] * (1.0 - frac)
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[idx] = new_val
                S_trial = float(a_trial.sum())
                if S_trial < 1e-12:
                    continue
                A_t = np.fft.rfft(a_trial, Mf)
                c_t = np.fft.irfft(A_t * A_t, Mf)[:2 * n - 1]
                sc_t = 2.0 * n * float(c_t.max()) / (S_trial * S_trial)
                if sc_t < sc - 1e-14:
                    a = a_trial
                    sc = sc_t
                    improved = True
                    break
            if improved:
                break

        if not improved:
            gn = float(np.linalg.norm(Sgrad))
            if gn < 1e-15:
                break
            step = min(0.03, (sc - 2.0) / (gn * gn + 1e-30))
            step = max(step, 1e-10)
            for _ in range(8):
                a_new = a - step * Sgrad
                a_new = np.maximum(a_new, 0.0)
                S_new = float(a_new.sum())
                if S_new < 1e-12:
                    break
                a_new = a_new / S_new
                sc_new = compute_score(a_new)
                if sc_new < sc:
                    a = a_new
                    sc = sc_new
                    improved = True
                    break
                step *= 0.5

    return best_a


def spectral_refine(a, n_iters=300, n_top=30, strength=0.15):
    """Progressive spectral peak shaving with decreasing strength."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    Mf = fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        s = strength * (1.0 - it / n_iters * 0.7)
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        non_dc_idx = np.arange(1, len(amp))
        C = np.median(amp[non_dc_idx])
        C = max(C, 1e-20)
        k_top = min(n_top, len(non_dc_idx))
        top_bins = non_dc_idx[np.argsort(amp[non_dc_idx])[::-1][:k_top]]
        amp_new = amp.copy()
        for b in top_bins:
            amp_new[b] = amp[b] * (1.0 - s) + C * s
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())

        a = a_new

    return best_a
