def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _fast_cross_conv(a, b):
    n = a.size
    m = b.size
    M = n + m - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return x
    mu = np.sort(x)[::-1]
    cv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu - (cv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    theta = (cv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(x - theta, 0.0)


def _lse_value_grad(a, beta):
    """LSE-smoothed objective 2*max(n*c/S^2) and its gradient (on simplex)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=5, tol=1e-6):
    """Exact subgradient of 2*n*max(c)/S^2 with top-k active set."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _bb_nesterov_optimize(a0, n_iters=3000, deadline=None, k_active=3, tol=1e-7, use_momentum=True):
    """Nesterov-accelerated subgradient with BB step size and exact simplex projection."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-14:
        return best_score, best_a
    eta = 0.1 / max(g_norm, 1e-10)
    prev_a = a.copy()
    prev_g = g0.copy()
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break
        if use_momentum:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            beta_mom = (t - 1.0) / t_new
            y = a + beta_mom * v
            y = _simplex_project(y)
        else:
            y = a.copy()
        score_y, grad_y = _subgrad_exact(y, k_active, tol)
        if not np.isfinite(score_y):
            break
        # BB step size
        if it > 0 and use_momentum:
            diff = a - prev_a
            gdiff = grad_y - prev_g
            dd = float(diff @ diff)
            dg = float(diff @ gdiff)
            if dd > 1e-16:
                eta_bb = dd / max(abs(dg), 1e-10)
                eta = np.clip(eta_bb, 0.01 * eta, 10.0 * eta + 0.01)
        prev_a = a.copy()
        prev_g = grad_y.copy()
        # Gradient step with projection
        x_new = y - eta * grad_y
        x_new = _simplex_project(x_new)
        # Update momentum
        v = x_new - a
        a = x_new
        if use_momentum:
            t = t_new
        # Decay step size slowly
        if it > 100:
            eta *= 0.999
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            eta *= 0.8
            no_improve = 0
    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=500):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size
    if maxfun is None:
        maxfun = 500

    def fun(x):
        x = _simplex_project(x)
        val, grad = _lse_value_grad(x, beta)
        return val, grad

    bounds = [(0.0, 1.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(maxfun // 3, 100),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = _simplex_project(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc):
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _lp_direction(a, M=None, max_iter=100):
    """
    LP-based direction finding: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k.
    Uses projected subgradient on the dual (sliding window).
    Returns the primal solution g (normalized).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    a = a / S  # normalize
    if M is None:
        c = _fast_self_conv(a)
        M = float(c.max())

    # Dual: min M * sum(y) s.t. window_sum(y, j) >= 1/a[j] for j=0..n-1, y >= 0
    # y has length 2n-1
    req = np.zeros(n)
    mask = a > 1e-12
    req[mask] = 1.0 / a[mask]
    req[~mask] = 0.0

    # Greedy: process from right to left, add deficit to rightmost available position
    y = np.zeros(2 * n - 1)
    # Precompute window sums efficiently using cumulative sum
    # Process j from n-1 down to 0
    # Window for j: y[j], y[j+1], ..., y[j+n-1]
    # When processing from right to left, positions y[j+n-1], y[j+n-2], ... may already be set
    running_sum = np.zeros(2 * n)  # use cumulative sum for efficiency

    # Simple O(n^2) approach (fine for n up to ~16384)
    # Actually let's use a smarter approach with a running sum
    window_sum = 0.0
    # Initialize: process j = n-1
    # Window is y[n-1..2n-2]
    for j in range(n - 1, -1, -1):
        # Current window sum for constraint j
        # y[j] through y[j+n-1]
        # When we move from j+1 to j, we lose y[j+n-1] from the right and gain y[j] on the left
        # This is tricky, let's just compute directly for small n or use prefix sums

    # For efficiency, use the approach: process j from 0 to n-1
    # At step j, the "new" position is y[j+n-1] (rightmost in window)
    # running_sum tracks sum(y[j..j+n-1])
    # Start: running_sum = 0 for j=0 (window y[0..n-1], all zero initially)
    running_sum = 0.0
    for j in range(n):
        # Ensure window sum >= req[j]
        deficit = req[j] - running_sum
        if deficit > 0:
            # Add to y[j+n-1] (rightmost in current window)
            y[j + n - 1] = deficit
            running_sum += deficit
        # Move to next window: lose y[j], gain y[j+n]
        running_sum -= y[j]
        if j + n < 2 * n - 1:
            running_sum += y[j + n]
        # (y[j] was added only if j+n-1 == j, i.e., n=1, so we subtract it)

    # Recover primal from dual: complementary slackness
    # If (a*g)[k] < M, then y[k] = 0. If y[k] > 0, then (a*g)[k] = M.
    # This is complex. Instead, use y as a "weight" to construct g.
    # Actually, let's use a different approach: use the dual solution to guide the primal.

    # Alternative: use the subgradient of the dual to iteratively construct g.
    # Simpler: just use the greedy y as a fallback direction.
    # Actually, let's construct g from the constraint structure:
    # The optimal g satisfies: for each j, a[j] * sum_{k=j}^{j+n-1} y_dual[k] = 1 (if g[j] > 0)
    # This is the complementary slackness condition.

    # Let me use a simpler heuristic: g[j] proportional to req[j]
    # This is equivalent to g[j] proportional to 1/a[j]
    # Normalize to sum = 1
    g = req.copy()
    g_sum = float(g.sum())
    if g_sum > 1e-12:
        g = g / g_sum
    else:
        g = np.ones(n) / n
    return g


def _lp_blend_step(a, deadline, n_inner=50):
    """One LP-based blend step: find g via LP dual, blend with a."""
    a = _normalize(a)
    n = a.size
    c = _fast_self_conv(a)
    M = float(c.max())
    g = _lp_direction(a, M)
    # Try different blend factors
    best_a = a.copy()
    best_score = _exact_score(a)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
        candidate = _normalize((1.0 - t) * a + t * g)
        sc = _exact_score(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = candidate
    return best_score, best_a


def _quick_optimize(a, deadline):
    """Quick multi-phase optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)
    # Phase 1: LSE with moderate beta
    rem = deadline - time.monotonic()
    if rem > 10.0:
        for beta in [500, 5000, 50000, 200000]:
            rem = deadline - time.monotonic()
            if rem < 8.0:
                break
            maxfun = int(max(30, min(400, rem * 8)))
            sc, a2 = _lbfgs_lse(a, beta, deadline, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
            a = _normalize(a2)
    # Phase 2: Subgradient
    rem = deadline - time.monotonic()
    if rem > 8.0:
        n_it = int(max(100, rem * 20))
        sc, a2 = _bb_nesterov_optimize(a, n_iters=n_it, deadline=deadline, k_active=3, tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)
    # Phase 3: LP blend
    rem = deadline - time.monotonic()
    if rem > 5.0:
        sc, a2 = _lp_blend_step(a, deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
    return best_score, best_a


def _full_optimize(a, deadline, label=""):
    """Full multi-stage optimization pipeline."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total_time = max(20.0, float(deadline - t0 - 5.0))
    print("[{}] start: score={:.8f} n={}".format(label, best_score, a.size))

    stages = [
        ("lse_low", [200, 1000, 5000], 0.25),
        ("lse_mid", [20000, 80000, 300000], 0.25),
        ("subgrad1", None, 0.20),
        ("lse_high", [1000000, 3000000], 0.15),
        ("subgrad2", None, 0.15),
    ]

    cum = 0.0
    for name, betas, frac in stages:
        cum += frac
        sub_dead = t0 + cum * total_time
        sub_dead = min(sub_dead, float(deadline - 5.0))
        if time.monotonic() > sub_dead - 3.0:
            continue

        if name.startswith("lse"):
            for beta in betas:
                rem = sub_dead - time.monotonic()
                if rem < 5.0:
                    break
                maxfun = int(max(40, min(800, rem * 12)))
                sc, a2 = _lbfgs_lse(a, beta, sub_dead - 2.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] {} beta={:.0f}: {:.8f}".format(label, name, beta, best_score))
                a = _normalize(a2)
        elif name.startswith("subgrad"):
            rem = sub_dead - time.monotonic()
            if rem < 5.0:
                continue
            n_it = int(max(200, rem * 25))
            for k_act in [3, 5, 10]:
                rem = sub_dead - time.monotonic()
                if rem < 5.0:
                    break
                sc, a2 = _bb_nesterov_optimize(a, n_iters=int(max(100, rem * 15)),
                                               deadline=sub_dead, k_active=k_act, tol=1e-8)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] {} k={}: {:.8f}".format(label, name, k_act, best_score))
                a = _normalize(a2)

    return best_score, best_a


def _generate_inits(n, rng, parent, cap=40):
    """Generate diverse initial sequences including parametric families."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)
    idx = np.arange(n, dtype=np.float64)

    # Power laws: (i+1)^(-alpha)
    for alpha in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        _add(np.power(idx + 1.0, -alpha) + 0.01)

    # 1/sqrt with shift
    for delta in [0, 0.5, 1.0, 2.0, 5.0, 10.0]:
        _add(1.0 / np.sqrt(idx + 1.0 + delta))

    # Exponential decay
    for rate in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:
        _add(np.exp(-rate * idx / n) + 0.01)

    # 1/sqrt with cosine modulation
    for freq in [1, 2, 3, 5, 8, 13]:
        for amp in [0.05, 0.1, 0.2, 0.3]:
            base = 1.0 / np.sqrt(idx + 1.0)
            _add(base * (1.0 + amp * np.cos(2.0 * np.pi * freq * x)))

    # 1/sqrt with linear taper
    for gamma in [0.1, 0.2, 0.3, 0.5]:
        _add(1.0 / np.sqrt(idx + 1.0) * (1.0 - gamma * x))

    # Product of power law and exponential
    for alpha in [0.3, 0.4, 0.5]:
        for rate in [0.5, 1.0, 2.0]:
            _add(np.power(idx + 1.0, -alpha) * np.exp(-rate * idx / n) + 0.001)

    # V-shaped and asymmetric
    for alpha in [0.5, 1.0, 2.0]:
        _add(1.0 + alpha * x)
        _add(1.0 + alpha * (1.0 - x))
        _add(np.exp(alpha * x) + 0.01)
        _add(np.exp(alpha * (1.0 - x)) + 0.01)

    # Bimodal (two peaks)
    for w in [0.1, 0.2, 0.3]:
        for ctr in [0.15, 0.25, 0.35]:
            g1 = np.exp(-(((x - (0.5 - ctr)) / w) ** 2))
            g2 = np.exp(-(((x - (0.5 + ctr)) / w) ** 2))
            _add(g1 + g2 + 0.01)

    # Uniform
    _add(np.ones(n))

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 2.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    return inits


def _perturb(a, rng, strength=None):
    """Generate perturbation of a for restarts."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 5))
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=5, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.3 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        x = np.linspace(0.0, 1.0, n)
        freq = float(rng.uniform(1, 10))
        q = a * (1.0 + strength * 0.5 * np.cos(2.0 * np.pi * freq * x))
    else:
        # Shift perturbation
        shift = int(rng.integers(1, max(2, n // 100)))
        q = np.roll(a, shift)
        q = q * np.exp(0.02 * rng.standard_normal(n))
    return _normalize(np.clip(q, 0.0, None))

    parent = None

            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
