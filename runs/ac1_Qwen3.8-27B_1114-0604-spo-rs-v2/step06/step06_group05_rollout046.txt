    c_max = float(c.max())

    log_sum_exp = max_scaled + np.log(exp_s.sum())
    L = c_max + log_sum_exp / beta
    p = exp_s / exp_s.sum()

    # L = c_max + (1/beta) * log(sum exp(beta*(c_k - c_max)))
    # dL/da_i = sum_k p_k * dc_k/da_i
    # dc_k/da_i = 2 * a_{k-i} (for valid indices)
    # So dL/da = 2 * (rev_a * p) in the appropriate range
    
    rev_a = a[::-1]
    M2 = 3*n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    # Score = 2n * L / S^2
    # d(score)/da_i = (2n * S^2 * g_L_i - 2n * L * 2 * S) / S^4
    # = (2n / S^3) * (g_L_i * S - 2*L)
    grad = (2.0 * n / (S**3)) * (g_L * S - 2.0 * L)
    return 2.0 * n * L / (S*S), grad

def solve_bal_step(b, active_set, n):
    """Given b, solve LP to find optimal a."""
    # Variables: a_0, ..., a_{n-1}, t (total n+1)
    # Minimize: t (i.e., c_obj = [0, 0, ..., 0, 1])
    # Subject to:
    #   sum(a) = 1
    #   a >= 0
    #   For each k in active_set: sum_j a_j * b[max(0,k-n+1):min(k+1,n)] [reversed appropriately] <= t
    
    from scipy.optimize import linprog
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequalities: for each active k, d_k^T a - t <= 0
    A_ub = []
    b_ub = []
    for k in active_set:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        d = np.zeros(n)
        # d[j] = b[k-j] for valid j
        for j in range(j_start, j_end + 1):
            d[j] = b[k - j]
        row = np.zeros(n + 1)
        row[:n] = d
        row[-1] = -1.0  # d^T a - t <= 0
        A_ub.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub) if A_ub else np.zeros((0, n+1))
    b_ub = np.array(b_ub) if b_ub else np.zeros(0)
    
    bounds = [(0, None)] * n + [(0, None)]  # a >= 0, t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    else:
        return None

"""
Step function optimization via Bilinear Alternating Linear Programming (BALM) + LSE refinement.
Key innovation: Instead of smooth approximations (LSE) to the max, we solve exact LP subproblems
in a bilinear alternating framework. Each subproblem is: min_{a in simplex} max_{k in active} (a*b)_k,
which is a linear program. Alternating between a and b drives both toward a common fixed point
where the cross-convolution a*b is as flat as possible.
Combined with multi-resolution initialization and high-precision LSE polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a, b=None):
    """Linear convolution of a with b (or with itself if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = 1 << (n + m - 2).bit_length()
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    return np.fft.irfft(A * B, M)[:n + m - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_s = np.exp(scaled - max_scaled)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def solve_balm_step(fixed_vec, active_indices, n):
    """
    Solve the LP: min t s.t. x in simplex, (x * fixed_vec)_k <= t for k in active_indices.
    Returns the optimal x (length n) or None if infeasible.
    """
    # Variables: x_0, ..., x_{n-1}, t  (n+1 total)
    # Minimize: t
    # Subject to: sum(x) = 1, x >= 0, t >= 0
    #             (x * fixed_vec)_k - t <= 0 for k in active_indices
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Equality constraint: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality constraints
    rows = []
    for k in active_indices:
        k = int(k)
        # (x * fixed_vec)_k = sum_{j: 0<=j<n, 0<=k-j<n} x_j * fixed_vec[k-j]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        row = np.zeros(n + 1)
        for j in range(j_start, j_end + 1):
            row[j] = fixed_vec[k - j]
        row[-1] = -1.0  # (x*f)(k) - t <= 0
        rows.append(row)
    
    if not rows:
        return np.ones(n) / n
    
    A_ub = np.array(rows)
    b_ub = np.zeros(len(rows))
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 10, 'presolve': True})
    
    if res.success:
        return res.x[:n]
    return None


def balm_optimize(a0, n_max_iter=5000, active_size=300, tol=1e-10, time_limit=300.0):
    """
    Bilinear Alternating Linear Programming (BALM).
    Alternates between solving LP subproblems for a and b.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    b = a.copy()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    for it in range(n_max_iter):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Step 1: Update a (fix b)
        c_cross = fast_conv(a, b)
        top_k = min(active_size, len(c_cross))
        active = np.argsort(c_cross)[::-1][:top_k]
        
        a_new = solve_balm_step(b, active, n)
        if a_new is not None:
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new / S_new
            a = a_new
        
        # Step 2: Update b (fix a)
        c_cross = fast_conv(a, b)
        top_k = min(active_size, len(c_cross))
        active = np.argsort(c_cross)[::-1][:top_k]
        
        b_new = solve_balm_step(a, active, n)
        if b_new is not None:
            b_new = np.maximum(b_new, 0.0)
            S_new = b_new.sum()
            if S_new < 1e-12:
                b_new = b.copy()
            else:
                b_new = b_new / S_new
            b = b_new
        
        # Check convergence
        if it % 50 == 0:
            diff = float(np.linalg.norm(a - b))
            # Evaluate the "balanced" solution
            a_bal = (a + b) / 2.0
            sc = compute_score(a_bal)
            if sc < best_score:
                best_score = sc
                best_a = a_bal.copy()
            
            if diff < tol:
                break
        
        if it % 200 == 0:
            a_bal = (a + b) / 2.0
            sc = compute_score(a_bal)
            if sc < best_score:
                best_score = sc
                best_a = a_bal.copy()
            if it % 1000 == 0:
                print(f"      BALM iter={it}, diff={diff:.2e}, score={best_score:.8f}")
    
    # Final evaluation
    a_bal = (a + b) / 2.0
    sc = compute_score(a_bal)
    if sc < best_score:
        best_score = sc
        best_a = a_bal.copy()
    
    return best_a


def gs_flat_target(a, n_iters=300, alpha_init=0.6, alpha_final=0.05):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def run_lbfgs(a0, maxiter=10000, maxfun=50000, beta=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def multi_resolution_balm(n_target, time_budget, rng):
    """Solve BALM at increasing resolutions."""
    resolutions = []
    # Build resolution ladder
    n_curr = max(64, n_target // 16)
    while n_curr < n_target:
        resolutions.append(n_curr)
        n_curr = min(n_curr * 2, n_target)
    if resolutions[-1] != n_target:
        resolutions.append(n_target)
    
    # Ensure minimum resolution
    if not resolutions or resolutions[0] > 128:
        resolutions = [128] + resolutions
    
    print(f"  Multi-res ladder: {resolutions}")
    
    a = np.ones(resolutions[0]) / resolutions[0]
    t_start = time.monotonic()
    
    for i, n_res in enumerate(resolutions):
        time_left = time_budget - (time.monotonic() - t_start)
        if time_left < 10:
            break
        
        # Upsample if needed
        if len(a) != n_res:
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, n_res)
            a = np.interp(x_new, x_old, a)
            a = np.maximum(a, 0.0)
            a = a / max(a.sum(), 1e-12)
        
        # Determine iterations based on time
        frac_time = (time_budget * (0.5 ** (len(resolutions) - 1 - i))) / len(resolutions)
        n_iter = min(5000, max(200, int(frac_time * 500)))
        time_limit = time_left * 0.8
        
        print(f"  BALM at n={n_res}: {n_iter} iters, time_limit={time_limit:.0f}s")

                         time_limit=time_limit)
        
        sc = compute_score(a)
        print(f"  n={n_res}: score={sc:.8f}")
    
    # Upsample to target
    if len(a) != n_target:
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, n_target)
        a = np.interp(x_new, x_old, a)
        a = np.maximum(a, 0.0)
        a = a / max(a.sum(), 1e-12)
    
    return a

def peak_surgery(a, n_rounds=50, top_k=50, reduction=0.1):
    """Directly reduce elements contributing to top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        # Collect contributions
        reduce_mask = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            reduce_mask[j_start:j_end + 1] += contrib
        
        # Reduce top contributors
        top_contributors = np.argsort(reduce_mask)[::-1][:min(100, n // 10)]
        for j in top_contributors:
            a[j] *= (1.0 - reduction)
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

        time_left = t_phase4_end - time.monotonic()
        a_bal = balm_optimize(best_a, n_max_iter=5000, active_size=500,
                             time_limit=time_left * 0.7)
        sc = compute_score(a_bal)
        update_best(sc, a_bal)
    
    print(f"After Phase 4: best = {best_score:.8f}")
    
    # ================================================================
    # Phase 5: Final polish interleaving (5% time)
    # ================================================================
    t_phase5_end = deadline
    print(f"\n=== Phase 5: Final polish ===")
    
    for round_idx in range(10):
        if time.monotonic() > t_phase5_end:
            break
        
        # Short LSE
        a_opt = run_lbfgs(best_a, maxiter=2000, maxfun=5000, beta=100000 + round_idx * 50000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        
        # Quick BALM
        if time.monotonic() > t_phase5_end:
            break
        time_left = t_phase5_end - time.monotonic()
        if time_left > 5:
            a_bal = balm_optimize(best_a, n_max_iter=2000, active_size=500,
                                 time_limit=min(time_left * 0.5, 30))
            sc = compute_score(a_bal)
            update_best(sc, a_bal)
        
        # GS
        if time.monotonic() > t_phase5_end:
            break
        a_gs = gs_flat_target(best_a, n_iters=100, alpha_init=0.2, alpha_final=0.01)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)
        
        if (round_idx + 1) % 3 == 0:
            print(f"  Polish {round_idx}: best = {best_score:.8f}")
    
    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
