```python
"""
Hybrid constrained/non-constrained search for self-convolution peak minimization.

The returned sequence is non-negative.  The routine starts from the supplied
parent construction when available, then combines:
  1. actual-evaluator-driven multi-start sampling and local transformations;
  2. fixed-active-set smooth peak penalty refinement (L-BFGS-B or fallback);
  3. peak shaving, Fourier smoothing, power transforms, directional moves.

The actual evaluator is used only on full candidate sequences, so the program
stays close to the true objective even if the internals use fast proxies.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False

_GLOBAL_BEST_SCORE = np.inf
_GLOBAL_BEST_SEQ = None
_GLOBAL_POP = []
_EVAL_COUNT = 0
_EVAL_COST = 0.1
_MAX_EVALS_TOTAL = 60000


def fft_len(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    M = fft_len(2 * n - 1)
    A = np.fft.rfft(x, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def make_sequence(a, target_sum=1.0):
    if a is None:
        return None
    try:
        x = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    S = float(np.sum(x))
    if not np.isfinite(S) or S <= 1e-15:
        return None
    if target_sum is None or target_sum <= 0.0 or not np.isfinite(target_sum):
        target_sum = 1.0

    scale = float(target_sum) / S
    if scale > 0.0:
        mx = float(np.max(x))
        if mx > 0.0 and mx * scale > 1000.0:
            scale = 1000.0 / mx
            target_sum = float(np.sum(x)) * scale
    x = x * scale
    x = np.clip(x, 0.0, 1000.0)
    S2 = float(np.sum(x))
    if not np.isfinite(S2) or S2 < 0.01:
        if S2 > 0.0:
            y = x / S2
            y = np.clip(y, 0.0, 1000.0)
            if float(np.sum(y)) >= 0.01:
                x = y
                target_sum = 1.0
        else:
            return None
        S2 = float(np.sum(x))
        if S2 < 0.01:
            return None

    scale2 = float(target_sum) / S2
    if scale2 > 0.0 and np.isfinite(scale2):
        mx = float(np.max(x))
        if mx == 0.0 or mx * scale2 <= 1000.0:
            x = x * scale2
            x = np.clip(x, 0.0, 1000.0)

    S3 = float(np.sum(x))
    if not np.isfinite(S3) or S3 < 0.01:
        return None
    x = np.clip(x, 0.0, 1000.0)
    return x


def uniform_seq(n):
    n = int(n)
    if n < 1:
        n = 1
    return np.ones(n, dtype=np.float64) / float(n)


def interpolate_seq(a, n):
    n = int(n)
    if n < 1:
        return None
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    if x.size == n:
        return x
    x_old = np.linspace(0.0, 1.0, x.size)
    x_new = np.linspace(0.0, 1.0, n)
    return make_sequence(np.interp(x_new, x_old, x), 1.0)


def pop_key(item):
    return item[0]


def pop_insert(x, score):
    global _GLOBAL_POP
    if x is None or not np.isfinite(score):
        return
    arr = make_sequence(x, 1.0)
    if arr is None:
        return
    _GLOBAL_POP.append((float(score), arr.copy()))
    _GLOBAL_POP.sort(key=pop_key)
    if len(_GLOBAL_POP) > 8:
        _GLOBAL_POP = _GLOBAL_POP[:8]


def actual_evaluate(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.inf
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    S = float(np.sum(x))
    if not np.isfinite(S) or S < 0.01:
        return np.inf
    try:
        val = evaluate_sequence(x.tolist())
    except Exception:
        return np.inf
    if not np.isfinite(val):
        return np.inf
    return float(val)


def eval_candidate(x, label=""):
    global _GLOBAL_BEST_SCORE, _GLOBAL_BEST_SEQ, _GLOBAL_POP, _EVAL_COUNT, _EVAL_COST
    arr = make_sequence(x, 1.0)
    if arr is None:
        return np.inf
    t0 = time.monotonic()
    sc = actual_evaluate(arr)
    cost = max(1e-6, time.monotonic() - t0)
    if _EVAL_COUNT <= 2:
        _EVAL_COST = float(cost)
    else:
        _EVAL_COST = 0.8 * _EVAL_COST + 0.2 * float(cost)
    _EVAL_COUNT += 1
    if np.isfinite(sc):
        if sc < _GLOBAL_BEST_SCORE:
            _GLOBAL_BEST_SCORE = float(sc)
            _GLOBAL_BEST_SEQ = arr.copy()
            print("NEW BEST %.10f len=%d label=%s" % (_GLOBAL_BEST_SCORE, arr.size, label))
        pop_insert(arr, float(sc))
    return sc


def create_diverse_inits(n, rng, parent):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    raw = []
    if parent is not None:

        if p is not None:
            if p.size == n:
                raw.append(p)
            else:
                xo = np.linspace(0.0, 1.0, p.size)
                raw.append(np.interp(x, xo, p))

    raw.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * math.pi * x), 1e-3))
    raw.append(0.3 + 0.7 * np.cos(2.0 * math.pi * x) ** 2)
    raw.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 1e-3, None))
    raw.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * math.pi * x) + 0.2 * np.cos(2.0 * math.pi * x), 1e-3))
    for _ in range(3):
        nhar = int(rng.integers(2, 8))
        sig = np.zeros(n, dtype=np.float64)
        for i in range(1, nhar + 1):

            amp = float(rng.standard_normal())
            sig = sig + amp * np.cos(2.0 * math.pi * i * x + ph)

def fourier_smooth(a, sigma):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    n = x.size
    if n < 2 or sigma <= 0.0:
        return x
    f = np.fft.rfftfreq(n) * 2.0
    filt = np.exp(-(f / float(sigma)) ** 2)
    A = np.fft.rfft(x)
    y = np.fft.irfft(A * filt, n)
    return make_sequence(y, 1.0)


def power_transform(a, p):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    p = float(p)
    if p <= 0.0 or not np.isfinite(p):
        return x
    y = np.power(x, p)
    return make_sequence(y, 1.0)


def mix_uniform(a, lam):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    n = x.size
    lam = float(np.clip(lam, 0.0, 1.0))
    u = np.ones(n, dtype=np.float64) / float(n)
    y = (1.0 - lam) * x + lam * u
    return make_sequence(y, 1.0)


def add_cos(a, m, lam, phase=0.0):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    n = x.size
    if n < 2:
        return x
    tt = np.arange(n, dtype=np.float64) / float(n)
    pat = np.cos(2.0 * math.pi * int(m) * tt + float(phase))
    y = x * (1.0 + float(lam) * pat)
    return make_sequence(y, 1.0)


def block_perturb(a, rng, n_blocks=3, scale_min=0.3, scale_max=2.0):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    x = x.copy()
    n = x.size
    if n < 4:
        return x
    lo = int(max(2, int(n * 0.01)))
    hi = int(max(lo + 1, min(n, int(n * 0.20))))
    for _ in range(int(n_blocks)):
        blen = int(rng.integers(lo, hi + 1))
        if blen < 1:
            continue
        if blen >= n:
            blen = max(1, n // 2)
        start = int(rng.integers(0, n - blen + 1))
        sc = float(rng.uniform(scale_min, scale_max))
        x[start:start + blen] *= sc
    return make_sequence(x, 1.0)


def shave_peaks(a, top_k=20, factor=0.85, mode="full"):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    c = fast_conv(x)
    vals = c.copy()
    if mode != "full" and vals.size > 2:
        vals[0] = -np.inf
        vals[-1] = -np.inf
    idx = np.argsort(vals)[::-1][:int(top_k)]
    n = x.size
    for kk in idx:
        k = int(kk)
        i0 = max(0, k - n + 1)
        i1 = min(k, n - 1)
        if i1 < i0:
            continue
        seg = x[i0:i1 + 1]
        other = x[k - i1:k - i0 + 1][::-1]
        contrib = seg * other
        count = min(8, contrib.size)
        if count <= 0:
            continue
        top_local = np.argsort(contrib)[::-1][:count]
        for tt in top_local:
            j = i0 + int(tt)
            if 0 <= j < n:
                x[j] *= float(factor)
    return make_sequence(x, 1.0)


def normalize_dir(d):
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size == 0:
        return None
    d = np.where(np.isfinite(d), d, 0.0)
    s = float(np.sum(np.abs(d)))
    if s <= 1e-15:
        return None
    return d / s


def add_scaled_dir(a, d, lam):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size != x.size:
        return None
    d = np.where(np.isfinite(d), d, 0.0)
    y = x + float(lam) * d
    return make_sequence(y, 1.0)


def direct_active_Hg(x, params):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    H = 0.0
    g = np.zeros(n, dtype=np.float64)
    for item in params:
        i0 = int(item[0])
        i1 = int(item[1])
        k = int(item[2])
        w = float(item[3])
        if w <= 0.0 or i1 < i0:
            continue
        seg = x[i0:i1 + 1]
        other = x[k - i1:k - i0 + 1][::-1]
        ck = float(np.dot(seg, other))
        H += w * ck
        g[i0:i1 + 1] += (2.0 * w) * other
    if H < 0.0 and H > -1e-18:
        H = 0.0
    return H, g


def build_params_from_active(active_idx, active_w, n):
    n = int(n)
    params = []
    for pos in range(len(active_idx)):
        k = int(active_idx[pos])
        w = float(active_w[pos])
        if w <= 0.0 or k < 0 or k >= 2 * n - 1:
            continue
        i0 = max(0, k - n + 1)
        i1 = min(k, n - 1)
        if i1 >= i0:
            params.append((i0, i1, k, w))
    return params


def select_active(c, mode, top_max, p, eps, n):
    active_idx = []
    active_w = []
    n = int(n)
    if n <= 0:
        return active_idx, active_w
    c = np.asarray(c, dtype=np.float64)
    if mode == "full":
        if c.size == 0:
            return active_idx, active_w
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            return [0], [1.0]
        order = np.argsort(c)[::-1][:int(top_max)]
        vals = c[order].astype(np.float64)
        vmax = float(np.max(vals))
        if vmax <= 0.0:
            vmax = 1e-300
        w = (vals / vmax) ** float(p)
        s = float(np.sum(w))
        if not np.isfinite(s) or s <= 0.0:
            w = np.ones(order.size, dtype=np.float64)
        else:
            w = w / s
        for i in range(order.size):
            active_idx.append(int(order[i]))
            active_w.append(float(w[i]))
        return active_idx, active_w

    eps_eff = float(min(max(float(eps), 0.0), 0.3))
    if c.size <= 2:
        return [0], [1.0]
    interior = c[1:-1]
    if interior.size:
        M = float(np.max(interior))
        if np.isfinite(M) and M > 0.0:
            order = np.argsort(interior)[::-1][:int(top_max)]
            vals = interior[order].astype(np.float64)
            vmax = float(np.max(vals))
            if vmax <= 0.0:
                vmax = 1e-300
            w = (vals / vmax) ** float(p)
            s = float(np.sum(w))
            if not np.isfinite(s) or s <= 0.0:
                w = np.ones(order.size, dtype=np.float64)
            else:
                w = w / s
            mass = 1.0 - 2.0 * eps_eff
            if mass <= 0.0:
                mass = 0.5
            for i in range(order.size):
                active_idx.append(int(1 + order[i]))
                active_w.append(float(w[i] * mass))
    for k_extra in (0, 2 * n - 2):
        if 0 <= k_extra < c.size:
            active_idx.append(int(k_extra))
            active_w.append(eps_eff)
    total = 0.0
    for v in active_w:
        total += float(v)
    if total <= 0.0 or not np.isfinite(total):
        return [0], [1.0]
    out_idx = []
    out_w = []
    for i in range(len(active_idx)):
        out_idx.append(int(active_idx[i]))
        out_w.append(float(active_w[i] / total))
    return out_idx, out_w


def full_score_fast(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.inf
    S = float(np.sum(x))
    if S <= 1e-15:
        return np.inf
    c = fast_conv(x)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * float(n) * m / (S * S)


def proxy_score(x, mode, eps=0.1):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.inf
    S = float(np.sum(x))
    if S <= 1e-15:
        return np.inf
    c = fast_conv(x)
    if c.size == 0:
        return np.inf
    if mode == "full":
        m = float(np.max(c))
    else:
        if c.size > 2:
            m = float(np.max(c[1:-1]))
        else:
            m = float(np.max(c))
        m = float(eps) * float(c[0]) + m
    v = 2.0 * float(n) * m / (S * S)
    if not np.isfinite(v):
        return np.inf
    return v


def peak_direction(a, mode, top_max, p, eps):
    x = make_sequence(a, 1.0)
    if x is None:
        return None
    c = fast_conv(x)
    idx, w = select_active(c, mode, top_max, p, eps, x.size)
    if not idx:
        return None
    params = build_params_from_active(idx, w, x.size)
    if not params:
        return None
    H, g = direct_active_Hg(x, params)
    S = float(np.sum(x))
    if S <= 1e-12:
        return None
    grad = (g * S - 2.0 * H * x) / (S ** 3)
    return normalize_dir(-grad)


def active_objective(x, params, w_fft, M2, use_fft):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    S = float(np.sum(x))
    if S <= 1e-12:
        return 1e12, np.zeros(n, dtype=np.float64)

    if use_fft:
        H, _ = direct_active_Hg(x, params)
        rev = x[::-1]
        RA = np.fft.rfft(rev, M2)
        gconv = np.fft.irfft(RA * w_fft, M2)[n - 1:2 * n - 1]
        g = 2.0 * gconv
    else:
        H, g = direct_active_Hg(x, params)

    if not np.isfinite(H):
        return 1e12, np.zeros(n, dtype=np.float64)
    S2 = S * S
    S3 = S2 * S
    C = 2.0 * float(n)
    val = C * H / S2
    grad = C * (g * S - 2.0 * H * x) / S3
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if not np.isfinite(val) or val < 0.0:
        val = 1e12
    return float(val), grad


def gradient_active_step(a0, params, num_iters=50, lr=0.25):
    x = make_sequence(a0, 1.0)
    if x is None:
        return None
    for _ in range(int(num_iters)):
        S = float(np.sum(x))
        if S <= 1e-12:
            break
        H, g = direct_active_Hg(x, params)
        grad = (g * S - 2.0 * H * x) / (S ** 3)
        norm = float(np.linalg.norm(grad))
        if not np.isfinite(norm) or norm <= 1e-15:
            break
        y = x - (float(lr) / norm) * grad
        x = make_sequence(y, 1.0)
        if x is None:
            break
    return x


def run_lbfgs_active(a0, params, w_fft, M2, use_fft, maxiter, maxfun):
    if not _HAS_SCIPY:
        return gradient_active_step(a0, params, 60, 0.25)
    x0 = make_sequence(a0, 1.0)
    if x0 is None:
        return None
    n = x0.size
    bounds = [(0.0, 100.0)] * n
    args = (params, w_fft, M2, bool(use_fft))
    try:
        res = minimize(
            active_objective,
            x0,
            method="L-BFGS-B",
            jac=True,
            args=args,
            bounds=bounds,
            options={
                "ftol": 1e-13,
                "gtol": 1e-11,
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 20,
            }
        )
        x1 = res.x
    except Exception:
        x1 = x0
    x1 = make_sequence(x1, 1.0)
    if x1 is None:
        return x0
    return x1


def active_refine(a0, mode, max_outers, top_max, p_start, eps, inner_maxiter, inner_maxfun, deadline):
    a = make_sequence(a0, 1.0)
    if a is None:
        return None
    n = a.size
    if n < 2:
        return a
    best = a.copy()
    best_p = proxy_score(a, mode, eps)
    if not np.isfinite(best_p):
        best_p = 1e30
    M2 = fft_len(3 * n - 2)
    p = float(p_start)
    stagnant = 0
    for _ in range(int(max_outers)):
        if time.monotonic() > deadline - 10.0:
            break
        c = fast_conv(a)
        idx, w = select_active(c, mode, top_max, p, eps, n)
        if not idx:
            break
        params = build_params_from_active(idx, w, n)
        if not params:
            break

        if _HAS_SCIPY:
            warr = np.zeros(2 * n - 1, dtype=np.float64)
            for i in range(len(idx)):
                k = int(idx[i])
                if 0 <= k < 2 * n - 1:
                    warr[k] += float(w[i])
            sw = float(np.sum(warr))
            if sw > 0.0:
                warr = warr / sw
            wfft = np.fft.rfft(warr, M2)
            use_fft = len(params) > 8
            mf = int(inner_maxfun)
            mi = int(inner_maxiter)
            if n > 9000:
                mf = max(100, int(mf * 0.6))
            a_new = run_lbfgs_active(a, params, wfft, M2, use_fft, mi, mf)
        else:
            a_new = gradient_active_step(a, params, 60, 0.25)

        if a_new is None:
            break
        a_new = make_sequence(a_new, 1.0)
        if a_new is None:
            break
        sc = proxy_score(a_new, mode, eps)
        if np.isfinite(sc) and sc < best_p:
            best_p = float(sc)
            best = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            a = a_new
            stagnant += 1
            if stagnant >= 4:
                break
        p += 0.4
    return best


def get_parent_sequence():
    try:
        if "height_sequence_1" in globals():
            p = globals()["height_sequence_1"]
            arr = np.asarray(p, dtype=np.float64).ravel()
            if arr.size > 0:
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.clip(arr, 0.0, 1000.0)
                if float(arr.sum()) > 0.0:
                    return make_sequence(arr, 1.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400):
    global _GLOBAL_BEST_SCORE, _GLOBAL_BEST_SEQ, _GLOBAL_POP, _EVAL_COUNT, _EVAL_COST

    _GLOBAL_BEST_SCORE = np.inf
    _GLOBAL_BEST_SEQ = None
    _GLOBAL_POP = []
    _EVAL_COUNT = 0
    _EVAL_COST = 0.1

    t_start = time.monotonic()
    budget = float(budget_s)
    deadline = t_start + max(120.0, budget - 45.0)
    rng = np.random.default_rng(seed)

    parent = get_parent_sequence()
    if parent is not None:
        base_n = int(parent.size)
    else:
        base_n = 6962

    print("Starting optimized sequence search, base_n=%d, budget=%.0fs" % (base_n, budget))

    if parent is not None:
        eval_candidate(parent, "parent")
    else:
        inits = create_diverse_inits(base_n, rng, None)
        for i, a in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            eval_candidate(a, "init%d" % i)

    eval_candidate(uniform_seq(base_n), "uniform_base")

    if _GLOBAL_BEST_SEQ is None:
        fb = uniform_seq(base_n)
        eval_candidate(fb, "fallback_uniform")

    if time.monotonic() < deadline - 600 and _EVAL_COST < 0.6:
        for factor in (0.5, 1.5, 2.0):
            if time.monotonic() > deadline - 300:
                break
            n2 = int(round(base_n * factor))
            n2 = max(512, min(16384, n2))
            if parent is not None:
                cand = interpolate_seq(parent, n2)
            else:
                cand = interpolate_seq(_GLOBAL_BEST_SEQ, n2) if _GLOBAL_BEST_SEQ is not None else uniform_seq(n2)
            if cand is not None:
                eval_candidate(cand, "len%d" % n2)

    if not _GLOBAL_POP and _GLOBAL_BEST_SEQ is not None and np.isfinite(_GLOBAL_BEST_SCORE):
        pop_insert(_GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE)

    active_specs = (
        ("full", 28, 1.0, 0.0, 4, 650),
        ("interior", 28, 1.0, 0.05, 3, 500),
        ("full", 64, 1.8, 0.0, 3, 450),
        ("interior", 64, 1.8, 0.02, 3, 450),
    )
    for spec in active_specs:
        if time.monotonic() > deadline - 240:
            break
        mode, top_max, p_start, eps, max_outers, inner_fun = spec
        if _GLOBAL_BEST_SEQ is not None:
            base = _GLOBAL_BEST_SEQ
        elif parent is not None:
            base = parent
        else:
            base = uniform_seq(base_n)
        a_ref = active_refine(
            base, mode,
            max_outers=max_outers,
            top_max=top_max,
            p_start=p_start,
            eps=eps,
            inner_maxiter=max(20, inner_fun // 2),
            inner_maxfun=inner_fun,
            deadline=deadline - 60.0
        )
        if a_ref is not None:
            eval_candidate(a_ref, "active_" + mode)

    print("After active phase, best=%r" % _GLOBAL_BEST_SCORE)

    round_no = 0
    while time.monotonic() < deadline - 120 and _EVAL_COUNT < _MAX_EVALS_TOTAL:
        round_no += 1
        if not _GLOBAL_POP:
            if _GLOBAL_BEST_SEQ is not None:
                pop_insert(_GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE)
            else:
                break
        top = _GLOBAL_POP[:3]
        for base_score, a_base in top:
            if time.monotonic() > deadline - 120 or _EVAL_COUNT >= _MAX_EVALS_TOTAL:
                break
            nbase = a_base.size
            if nbase < 8:
                continue

            for sigma in (0.02, 0.05, 0.10, 0.20, 0.50):
                if time.monotonic() > deadline - 120:
                    break
                cand = fourier_smooth(a_base, sigma)
                if cand is not None:
                    eval_candidate(cand, "r%d_sm%.2f" % (round_no, sigma))

            for pw in (0.25, 0.50, 0.80, 1.25, 1.60, 2.50):
                if time.monotonic() > deadline - 120:
                    break
                cand = power_transform(a_base, pw)
                if cand is not None:
                    eval_candidate(cand, "r%d_pw%.2f" % (round_no, pw))

            for lam in (0.05, 0.15, 0.30, 0.55, 0.80):
                if time.monotonic() > deadline - 120:
                    break
                cand = mix_uniform(a_base, lam)
                if cand is not None:
                    eval_candidate(cand, "r%d_mu%.2f" % (round_no, lam))

            for m in (1, 2, 3, 5, 8, 13, 21):
                for lam in (0.2, 0.5, 0.8):
                    if time.monotonic() > deadline - 120:
                        break
                    cand = add_cos(a_base, m, lam, 0.0)
                    if cand is not None:
                        eval_candidate(cand, "r%d_cos%d_lam%.1f" % (round_no, m, lam))

            for mode in ("full", "interior"):
                for factor in (0.80, 0.90):
                    for kk in (10, 30):
                        if time.monotonic() > deadline - 120:
                            break
                        cand = shave_peaks(a_base, top_k=kk, factor=factor, mode=mode)
                        if cand is not None:
                            eval_candidate(cand, "r%d_shave_%s_%.*f_%d" % (round_no, mode, 1, factor, kk))

            for _ in range(6):
                if time.monotonic() > deadline - 120:
                    break
                cand = block_perturb(a_base, rng, n_blocks=3, scale_min=0.3, scale_max=2.0)
                if cand is not None:
                    eval_candidate(cand, "r%d_block%03d" % (round_no, _))

            if time.monotonic() < deadline - 180:
                dirs = []
                d1 = peak_direction(a_base, "full", 16, 1.0, 0.0)
                if d1 is not None:
                    dirs.append(d1)
                d2 = peak_direction(a_base, "interior", 16, 1.0, 0.05)
                if d2 is not None:
                    dirs.append(d2)

                sm = fourier_smooth(a_base, 0.06)
                if sm is not None:
                    d3 = normalize_dir(sm - a_base)
                    if d3 is not None:
                        dirs.append(d3)

                sh = shave_peaks(a_base, top_k=16, factor=0.80, mode="full")
                if sh is not None:
                    d4 = normalize_dir(sh - a_base)
                    if d4 is not None:
                        dirs.append(d4)

                pw = power_transform(a_base, 0.75)
                if pw is not None:
                    d5 = normalize_dir(pw - a_base)
                    if d5 is not None:
                        dirs.append(d5)

                for di, d in enumerate(dirs[:6]):
                    for lam in (0.02, 0.1, 0.3, 1.0):
                        if time.monotonic() > deadline - 180:
                            break
                        cand = add_scaled_dir(a_base, d, lam)
                        if cand is not None:
                            eval_candidate(cand, "r%d_dir%d_lam%.2f" % (round_no, di, lam))

        if round_no % 4 == 0 and time.monotonic() < deadline - 120 and _GLOBAL_BEST_SEQ is not None:
            for factor in (0.5, 1.5, 2.0):
                if time.monotonic() > deadline - 120:
                    break
                n2 = int(round(_GLOBAL_BEST_SEQ.size * factor))
                n2 = max(512, min(20480, n2))
                cand = interpolate_seq(_GLOBAL_BEST_SEQ, n2)
                if cand is not None:
                    eval_candidate(cand, "r%d_best_len%d" % (round_no, n2))

    if time.monotonic() < deadline - 90 and _GLOBAL_BEST_SEQ is not None:
        a_ref = active_refine(
            _GLOBAL_BEST_SEQ, "full",
            max_outers=3, top_max=32, p_start=1.6, eps=0.0,
            inner_maxiter=300, inner_maxfun=400,
            deadline=deadline - 30.0
        )
        if a_ref is not None:
            eval_candidate(a_ref, "final_active_full")

    if time.monotonic() < deadline - 60 and _GLOBAL_BEST_SEQ is not None:
        a_ref = active_refine(
            _GLOBAL_BEST_SEQ, "interior",
            max_outers=3, top_max=32, p_start=1.6, eps=0.03,
            inner_maxiter=300, inner_maxfun=350,
            deadline=deadline - 20.0
        )
        if a_ref is not None:
            eval_candidate(a_ref, "final_active_interior")

    best = _GLOBAL_BEST_SEQ
    if best is None:
        best = uniform_seq(base_n)
        eval_candidate(best, "last_fallback")

    best = make_sequence(best, 1.0)
    if best is None:
        best = uniform_seq(base_n)

    out1 = make_sequence(best, 1.0)
    out2 = make_sequence(best, math.sqrt(2.0 * float(best.size)))

    sc1 = actual_evaluate(out1) if out1 is not None else np.inf
    sc2 = actual_evaluate(out2) if out2 is not None else np.inf

    if np.isfinite(sc2) and (not np.isfinite(sc1) or sc2 < sc1):
        out_arr = out2
    else:
        out_arr = out1

    if out_arr is None:
        out_arr = uniform_seq(base_n)

    out_list = []
    for v in out_arr:
        out_list.append(float(np.clip(v, 0.0, 1000.0)))

    # As a safety net, if the final list is invalid, return a simple uniform.
    if len(out_list) == 0:
        out_list = [1.0]

    final_score = actual_evaluate(out_list)
    elapsed = time.monotonic() - t_start
    print("FINAL: best_internal=%r final_eval=%r len=%d time=%.1fs/%.0fs" %
          (_GLOBAL_BEST_SCORE, final_score, len(out_list), elapsed, budget))

    return out_list
```
