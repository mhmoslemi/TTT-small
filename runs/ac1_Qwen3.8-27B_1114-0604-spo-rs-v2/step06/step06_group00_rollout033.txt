The sum_j z_j * [k in [j, j+n-1]] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} z_j
This is a sliding window sum! Can be computed in O(n) with a prefix sum.

y_k *= exp(eta * Z_k) where Z_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} z_j

def lp_project_mirror(f, M, n_iters=1000, eta_init=1.0):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Using mirror descent (exponentiated gradient) on the dual.
    
    Returns g (normalized to sum=1).
    """
    n = len(f)
    eta = eta_init
    
    # Initialize dual variables
    y = np.zeros(2 * n - 1)
    
    # Initialize primal
    g = np.ones(n) / n
    
    best_g = g.copy()
    
    for it in range(n_iters):
        # Compute (f*g) - the constraint violations
        fg = fast_cross_conv(f, g)
        max_fg = np.max(fg)
        
        if max_fg <= M * (1 + 1e-10):
            break  # Feasible
        
        # Compute (Wy) - the dual constraint LHS
        # (Wy)_j = sum_{k=j}^{j+n-1} y_k
        y_prefix = np.concatenate([[0.0], np.cumsum(y)])
        Wy = np.zeros(n)
        for j in range(n):
            k_end = min(j + n, 2 * n - 1)
            Wy[j] = y_prefix[k_end] - y_prefix[j]
        
        # Compute dual constraint violations: f[j] * Wy[j] - 1
        dual_slack = f * Wy - 1.0  # positive means satisfied
        
        # Mirror descent update on y:
        # y_k *= exp(eta * Z_k) where Z_k = sum of max(-dual_slack_j, 0) over j where k is in [j, j+n-1]
        
        # z_j = max(1 - f[j]*Wy[j], 0) = max(-dual_slack_j, 0)
        z = np.maximum(-dual_slack, 0)
        
        # Z_k = sum_{j: max(0,k-n+1) <= j <= min(k, n-1)} z_j
        z_prefix = np.concatenate([[0.0], np.cumsum(z)])
        Z = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                Z[k] = z_prefix[j_end + 1] - z_prefix[j_start]
        
        # Update y
        log_update = eta * Z
        log_update = np.clip(log_update, -50, 50)  # prevent overflow
        y *= np.exp(log_update)
        
        # Normalize y to prevent overflow
        max_y = np.max(y)
        if max_y > 1e10:
            y /= max_y
            eta /= max_y  # compensate
        
        # Update primal from dual (complementary slackness approximation)
        # g_j = exp(-f[j] * Wy[j]) (up to normalization)
        # Recompute Wy after y update
        y_prefix = np.concatenate([[0.0], np.cumsum(y)])
        Wy = np.zeros(n)
        for j in range(n):
            k_end = min(j + n, 2 * n - 1)
            Wy[j] = y_prefix[k_end] - y_prefix[j]
        
        log_g = -f * Wy
        log_g -= np.max(log_g)  # for numerical stability
        g = np.exp(log_g)
        g_sum = g.sum()
        if g_sum < 1e-30:
            break
        g = g / g_sum
        
        eta *= 0.995  # anneal
    
    return g

def compute_Wy(y, n):
    """Compute (Wy)_j = sum_{k=j}^{min(j+n, 2n-1)} y_k for j = 0,...,n-1."""
    y_prefix = np.concatenate([[0.0], np.cumsum(y)])
    j = np.arange(n)
    k_end = np.minimum(j + n, len(y))
    return y_prefix[k_end] - y_prefix[j]

def compute_Z(z, n):
    """Compute Z_k = sum_{j=max(0,k-n+1)}^{min(k, n-1)} z_j for k = 0,...,2n-2."""
    z_prefix = np.concatenate([[0.0], np.cumsum(z)])
    k = np.arange(2 * n - 1)
    j_start = np.maximum(0, k - n + 1)

    # But need to handle j_end < j_start (shouldn't happen for 0<=k<=2n-2 with 0<=j<=n-1)

Actually, one more important consideration: the LP projection as described finds g that maximizes sum(g) subject to (f*g)[k] <= M. But we want g that will reduce ||g*g||_inf when blended with f. The LP constraint is on f*g, not g*g. So the LP gives a "safe direction" in the sense that f_new * g = (1-t)f * g + t*g*g, and if f*g <= M and g*g <= M (approximately), then f_new * g is bounded.

"""
Advanced optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
- LP projection via mirror descent (novel approach from SOTA description)
- Aggressive Fienup spectral shaping
- LSE annealing with adaptive beta schedule
- Nesterov-averaged subgradient for direct peak attack
- Multi-scale exploration and diverse restarts

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via real FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score_LSE = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def lp_project_mirror(f, M, n_iters=500, eta_init=1.0, eta_decay=0.997):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Uses mirror descent (exponentiated gradient) on the dual.
    Returns g normalized to sum=1.
    """
    n = len(f)
    eta = eta_init
    y = np.zeros(2 * n - 1)
    g = np.ones(n) / n
    best_g = g.copy()
    best_sum = 1.0

    for it in range(n_iters):
        # Compute constraint values f*g
        fg = fast_cross_conv(f, g)
        max_fg = float(np.max(fg))

        if max_fg <= M * (1.0 + 1e-9):
            break

        # Compute (Wy)_j = sum_{k=j}^{min(j+n, 2n-1)} y_k
        y_prefix = np.concatenate([[0.0], np.cumsum(y)])
        j_idx = np.arange(n)
        k_end = np.minimum(j_idx + n, 2 * n - 1)
        Wy = y_prefix[k_end] - y_prefix[j_idx]

        # Dual slack: f[j] * Wy[j] - 1 (positive = satisfied)
        dual_slack = f * Wy - 1.0
        z = np.maximum(-dual_slack, 0.0)

        # If all dual constraints satisfied, we're at optimum (approximately)
        if np.max(z) < 1e-12:
            break

        # Z_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} z_j
        z_prefix = np.concatenate([[0.0], np.cumsum(z)])
        k_idx = np.arange(2 * n - 1)
        j_start = np.maximum(0, k_idx - n + 1)
        j_end = np.minimum(k_idx, n - 1)
        Z = z_prefix[j_end + 1] - z_prefix[j_start]

        # Mirror descent update: y_k *= exp(eta * Z_k)
        log_update = eta * Z
        log_update = np.clip(log_update, -100, 100)
        y = y + log_update  # additive in log-space equivalent for stability

        # Actually use multiplicative for correctness
        # Reset and redo
        # y_new = y_old * exp(eta * Z)
        # For numerical stability, work in log-space
        # Let's just use: y = y + eta * Z (gradient step on dual)
        # This is subgradient, not mirror descent. Let me use proper multiplicative.

    # Restart with proper multiplicative updates
    y = np.zeros(2 * n - 1)
    g = np.ones(n) / n

    for it in range(n_iters):
        fg = fast_cross_conv(f, g)
        max_fg = float(np.max(fg))
        if max_fg <= M * (1.0 + 1e-9):
            break

        y_prefix = np.concatenate([[0.0], np.cumsum(y)])
        Wy = y_prefix[k_end] - y_prefix[j_idx]
        dual_slack = f * Wy - 1.0
        z = np.maximum(-dual_slack, 0.0)

        if np.max(z) < 1e-14:
            break

        z_prefix = np.concatenate([[0.0], np.cumsum(z)])
        Z = z_prefix[j_end + 1] - z_prefix[j_start]

        # Multiplicative update in log-space
        # log(y_new) = log(y_old) + eta * Z
        # But y can be zero, so use: y_new = y_old + eta * Z (subgradient on dual)
        y = y + eta * Z
        y = np.maximum(y, 0.0)

        # Recover primal from dual via complementary slackness
        y_prefix = np.concatenate([[0.0], np.cumsum(y)])
        Wy = y_prefix[k_end] - y_prefix[j_idx]

        # g_j proportional to exp(-f[j] * Wy[j]) (from mirror descent duality)
        log_g = -f * Wy
        max_log_g = np.max(log_g)
        if max_log_g < -700:
            log_g = log_g - max_log_g
            g = np.exp(log_g)
        else:
            g = np.exp(log_g - max_log_g)
        g_sum = g.sum()
        if g_sum < 1e-30:
            break
        g = g / g_sum

        eta *= eta_decay

    return g


def fienup_aggressive(a, n_iters=400):
    """Aggressive Fienup with progressive spectral clipping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Progressive clipping: start clipping top 30%, tighten to top 5%
        q_clip = 0.30 - 0.25 * progress
        thresh = np.quantile(P, 1.0 - q_clip)

        P_new = np.minimum(P, thresh)

        # Add floor to prevent spectral zeros
        floor = np.quantile(P, 0.05) * 0.05
        P_new = np.maximum(P_new, floor)

        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def fienup_blend(a, n_iters=300):
    """Fienup with blending: alternate between flatten and boost-low."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if it % 4 == 0:
            # Flatten
            alpha = 0.2 + 0.6 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean
        elif it % 4 == 1:
            # Clip top
            q = 0.40 - 0.30 * progress
            thresh = np.quantile(P, 1.0 - q)
            P_new = np.minimum(P, thresh)
        elif it % 4 == 2:
            # Boost bottom
            thresh = np.quantile(P, 0.2)
            P_new = P.copy()
            P_new[P < thresh] = thresh
        else:
            # Moderate flatten
            alpha = 0.1 + 0.3 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def nesterov_subgrad(a, n_iters=5000, top_k=5):
    """
    Subgradient descent with Nesterov averaging for peak minimization.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    # Nesterov averaging
    a_avg = a.copy()
    t = 0
    eta = 0.01

    for it in range(n_iters):
        t += 1
        c = fast_self_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Build subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = float(np.mean([c[int(k)] / (M_val + 1e-30) for k in top_idx]))
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        # Nesterov step size
        eta_t = 2.0 / (t + 2.0)
        # Scale by problem size
        eta_t = min(eta_t, 0.05) * 100.0 / n

        a = a - eta_t * grad
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # Nesterov average
        a_avg = (t * a_avg + a) / (t + 1)

        # Use the average for the next iterate (acceleration)
        if t % 10 == 0:
            sc_avg = compute_score(a_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = a_avg.copy()
            a = a_avg.copy()

    return best_a, best_sc


def peak_shave_focused(a, n_iters=2000, top_k=3):
    """
    Focused peak shaving: directly reduce contributions to the top peaks.
    Uses multiplicative updates on the contributing entries.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_self_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        for k in top_idx:
            k = int(k)
            if c[k] < M_val * 0.9:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue

            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k - js]
            total = pairs.sum()
            if total < 1e-15:
                continue

            # Reduce top contributors
            n_reduce = max(3, min(10, len(js) // 20))
            top_local = np.argsort(pairs)[::-1][:n_reduce]
            reduce_idx = j_start + top_local

            # Multiplicative reduction
            factor = 0.95
            for j in reduce_idx:
                if a[j] > 1e-12:
                    a[j] *= factor

            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Every 100 iterations, try to redistribute mass to low-peak areas
        if it % 100 == 99:
            c = fast_self_conv(a)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            peak_mask = np.ones(n, dtype=bool)
            peak_mask[j_start:j_end + 1] = False
            if np.sum(peak_mask) > 10:
                add_idx = np.where(peak_mask)[0]
                n_add = min(50, len(add_idx))
                chosen = np.random.choice(add_idx, n_add, replace=False)
                delta = 0.001 / n
                a[chosen] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a, best_sc


def coord_descent_multiplicative(a, n_sweeps=8, n_top=30):
    """
    Coordinate descent with multiplicative updates and line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.5, 0.7, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def mixed_lp_fienup_cycle(a, M, n_cycles=50, lp_iters_per_cycle=200):
    """
    Alternate between LP projection and Fienup polishing.
    """
    n = len(a)
    a_work = normalize(a.copy())
    best_a = a_work.copy()
    best_sc = compute_score(a_work)

    for cyc in range(n_cycles):
        # LP projection step
        g = lp_project_mirror(a_work, M, n_iters=lp_iters_per_cycle,
                              eta_init=0.5, eta_decay=0.998)

        # Blend
        for t in [0.1, 0.3, 0.5]:
            a_try = (1.0 - t) * a_work + t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
            if sc < compute_score(a_work):
                a_work = a_try
                break

        # Update M
        c = fast_self_conv(a_work)
        M = float(np.max(c))

        # Fienup polish
        a_f, sc_f = fienup_aggressive(a_work, n_iters=50)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_f.copy()
            a_work = a_f.copy()

    return best_a, best_sc
