"""

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _cross_corr(a, w):
    """Compute s[i] = sum_l a[l] * w[l+i] for i=0..n-1 using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    lw = w.size
    N2 = _next_pow2(n + lw - 1)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:n + lw - 1]
    s = conv_full[n - 1:n + lw - 1]
    return s


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

        w_active[active] = 1.0 / len(active)
    elif weight_mode == "inverse":
        w_vals = 1.0 / c[active]
        w_active[active] = w_vals / w_vals.sum()
    elif weight_mode == "sqrt_inverse":
        w_vals = 1.0 / np.sqrt(c[active])
        w_active[active] = w_vals / w_vals.sum()
    else:
        w_active[active] = 1.0 / len(active)

    R = _cross_corr(a, w_active)
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * 2.0 * R - (2.0 * n * T) * 2.0 / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    return score, grad, len(active)


def _mirror_nesterov(a0, deadline, tol=1e-5, eta0=0.5, verbose=True,
                     weight_mode="uniform", max_iter=None):
    """
    Mirror descent (exponential weights) with Nesterov acceleration.
    Operates on the simplex (a >= 0, sum(a) = 1).
    Minimizes f(a) = 2*n*max(conv(a,a)) (since S=1).
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    if verbose:
        print("  Mirror-Nesterov start: score={:.8f}, n={}".format(best_score, n))

    # Nesterov parameters
    t = 1.0
    a_prev = a.copy()
    eta = float(eta0)

    # Track improvement
    no_improve_count = 0
    improve_threshold = 1e-10

    iter_count = 0
    if max_iter is None:
        max_iter = 100000

    for it in range(max_iter):
        if time.monotonic() > float(deadline) - 2.0:
            break

        iter_count += 1

        # Nesterov extrapolation
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            beta = (t - 1.0) / t_new
            y = a + beta * (a - a_prev)
            a_prev = a
            t = t_new
        else:
            y = a.copy()

        # Project y to simplex (clip and normalize)
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y = y / Sy

        # Compute subgradient at y
        score_y, grad_y, n_active = _subgradient_weighted(y, tol=tol, weight_mode=weight_mode)
        if not np.isfinite(score_y):
            break

        # Shift gradient to be relative (for mirror descent on simplex,
        # we can subtract the mean to maintain the same direction)
        # Actually for exponential weights, the absolute level of grad matters
        # (it affects Z). Let's center it to make the step more stable.
        g_centered = grad_y - grad_y.mean()

        # Mirror descent step: a_new[i] = y[i] * exp(-eta * g_centered[i]) / Z
        log_step = -eta * g_centered
        # Clip for numerical stability
        log_step = np.clip(log_step, -50.0, 50.0)
        weights = np.exp(log_step)
        Z = y.sum() * float(weights.sum())  # Actually: Z = sum(y[i] * weights[i])
        Z = float((y * weights).sum())
        if Z < 1e-30:
            # Fallback: no update
            a_new = y.copy()
        else:
            a_new = y * weights / Z

        # Ensure feasibility
        a_new = np.clip(a_new, 0.0, None)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / S_new

        # Evaluate
        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            if sc < best_score - improve_threshold:
                no_improve_count = 0
            else:
                no_improve_count += 1
            best_score = sc
            best_a = a_new.copy()
            if verbose and (iter_count % 100 == 0 or sc < best_score - 1e-6):
                elapsed = time.monotonic()
                print("    iter {:5d}: score={:.8f} active={:4d} eta={:.4f}".format(
                    iter_count, sc, n_active, eta))
        else:
            no_improve_count += 1

        a = a_new

        # Adaptive step size
        if no_improve_count >= 30:
            eta *= 0.7
            no_improve_count = 0
            # Also try a "kick": add small noise to escape plateau
            if eta < 0.01:
                a = _normalize(a * (1.0 + 0.01 * np.random.default_rng(it).standard_normal(n)))
                eta = 0.1

        # Periodic tolerance tightening
        if iter_count % 500 == 0 and tol > 1e-8:
            tol = max(1e-8, tol * 0.5)

    return best_score, best_a


def _mirror_descent_basic(a0, deadline, tol=1e-5, eta0=0.3, verbose=True):
    """Basic mirror descent without Nesterov (more stable)."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    if verbose:
        print("  Mirror-Descent start: score={:.8f}".format(best_score))

    eta = float(eta0)
    no_improve = 0

    for it in range(1000000):
        if time.monotonic() > float(deadline) - 2.0:
            break

        score, grad, n_act = _subgradient(a, tol=tol)
        if not np.isfinite(score):
            break

        g = grad - grad.mean()
        log_w = np.clip(-eta * g, -40.0, 40.0)
        w = np.exp(log_w)
        Z = float((a * w).sum())
        if Z < 1e-30:
            break
        a_new = a * w / Z
        a_new = np.clip(a_new, 0.0, None)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new /= S

        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 40:
            eta *= 0.75
            no_improve = 0
            if eta < 0.005:
                break

    return best_score, best_a


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == n:
        return a.copy()
    if m == 0:
        return np.ones(n) / n
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _generate_inits(n, rng, parent):
    """Generate diverse initial sequences for the given length."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def _add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _normalize(_resample(parent, n))
        _add(pr)
        _add(pr[::-1])

    # Uniform
    _add(np.ones(n))

    # Linear ramp (lopsided - key for score < 2)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        _add(1.0 + alpha * x)
        _add(1.0 + alpha * (1.0 - x))

    # Exponential ramp
    for rate in [1, 2, 3, 5, 8, 12]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1 - x)) + 0.01)

    # Quadratic
    for alpha in [1.0, 2.0, 4.0]:
        _add(1.0 + alpha * (x - 0.5) ** 2)
        _add(1.0 + alpha * (x - 0.3) ** 2)
        _add(1.0 + alpha * (x - 0.7) ** 2)

    # Sine-based
    for f in [1, 2, 3]:
        _add(1.0 + 0.5 * np.sin(2 * np.pi * f * x))
        _add(1.0 + 0.5 * np.sin(np.pi * f * x))

    # "Stairs" - step function
    for nsteps in [2, 4, 8]:
        step = (x * nsteps).astype(int) / nsteps
        _add(1.0 + step * 2)
        _add(1.0 + (1 - step) * 2)

    # Cosine with phase
    for phase in [0, 0.3, 0.5]:
        _add(1.0 + 0.8 * np.cos(2 * np.pi * x + phase * np.pi))

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf)
        gamma = float(rng.uniform(1.5, 3.5))
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        _add(1.0 + 0.8 * noise)

    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    return inits


def _dual_refine(a, max_iters=50):
    """
    Try to find better active set weights using iterative refinement.
    Goal: find y >= 0 on active set such that (y correlates a) is as flat as possible
    on the support of a.
    Returns the modified a (unchanged) and the weights (for subgradient computation).
    Actually, this just returns improved weights for the subgradient.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    c = _fast_self_conv(a)
    cmax = float(c.max())
    threshold = cmax * (1.0 - 1e-5)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

    # Initialize weights
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)

    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        support = np.arange(n)

    for _ in range(max_iters):
        # Compute h[i] = sum_k w[k] * a[k-i] on support
        R = _cross_corr(a, w)
        h_support = R[support]

        if h_support.min() < 1e-15:
            # Some support points have near-zero "dual value" - try to increase their weights
            # This is heuristic
            break

        # Compute the "ratio" that would make h flat
        # We want: sum_k w[k]*a[k-i] = const for all i in support
        # Update: w[k] *= (target / contribution of k to h)
        # Simplified: w[k] proportional to "how much it helps equalize"

        # For each active k, compute its "contribution profile" on support
        # prof_k[i] = a[k-i] for i in support
        # We want to increase w[k] where prof_k is small (to boost low h values)

        # Heuristic: weight by 1/mean(contribution on support)
        contributions = np.zeros(len(active))
        for idx, k in enumerate(active):
            # contribution of k to h[i] is w[k]*a[k-i]
            # mean over support: mean(a[k-support])
            vals = np.array([a[int(k - i)] if 0 <= int(k - i) < n else 0.0 for i in support])
            contributions[idx] = vals.mean()

        if contributions.min() < 1e-15:
            break

        # New weights proportional to 1/contributions
        new_w = 1.0 / (contributions + 1e-10)
        new_w /= new_w.sum()

        # Damping
        w_old = w.copy()
        w[active] = 0.5 * w[active] + 0.5 * new_w
        w[~np.isin(np.arange(2*n-1), active)] = 0.0

    return w


def _coarse_to_fine(coarse_a, target_n, deadline, verbose=True):
    """Upscale a coarse solution and refine."""
    fine_a = _normalize(_resample(coarse_a, target_n))
    if verbose:
        print("  Coarse-to-fine: n={} -> n={}".format(coarse_a.size, target_n))
    sc, a = _mirror_nesterov(fine_a, deadline, tol=1e-5, eta0=0.3,
                             verbose=verbose, weight_mode="uniform")
    return sc, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns list of non-negative floats minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 60.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    if parent is not None:
        p_score = _exact_score(parent)
        print("Parent score: {:.8f}".format(p_score))

    # Determine lengths to try
    if parent is not None:
        plen = parent.size
        lengths = []
        # Try the parent length and nearby
        lengths.append(plen)
        for L in [8192, 10000, 10240, 12000, 12288, 16384, 4096, 6144]:
            if L not in lengths and 2048 <= L <= 16384:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288, 6144]

    lengths = lengths[:4]
    print("Trying lengths: {}".format(lengths))

    # Global best
    best_score = float(np.inf)
    best_seq = None

    # Phase 1: Evaluate initializations and pick best starts
    all_candidates = []
    for n in lengths:
        inits = _generate_inits(n, rng, parent)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_candidates.append((sc, arr))

    if not all_candidates:
        n0 = lengths[0]
        all_candidates.append((2.0, np.ones(n0) / n0))

    all_candidates.sort(key=lambda x: x[0])
    print("Best init: {:.8f} (n={})".format(all_candidates[0][0], all_candidates[0][1].size))

    # Select top diverse starts (different lengths preferred)
    starts = []
    used_lens = set()
    for sc, arr in all_candidates:
        L = arr.size
        if L not in used_lens:
            starts.append(arr.copy())
            used_lens.add(L)
        if len(starts) >= 3:
            break

    while len(starts) < 2:
        starts.append(_normalize(_resample(starts[0], lengths[0])))

    # Initialize global best from starts
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Coarse-to-fine for a good start
    # First, quickly optimize a small version
    small_n = min(512, lengths[0])
    if time.monotonic() < deadline - 200:
        small_start = _normalize(_resample(best_seq, small_n))
        small_dead = min(deadline - 180, t_start + 30.0)
        sc_small, a_small = _mirror_nesterov(small_start, small_dead, tol=1e-4,
                                             eta0=0.5, verbose=False)
        if sc_small < best_score:
            # Upscale and use as new start
            upscaled = _normalize(_resample(a_small, best_seq.size))
            sc_up = _exact_score(upscaled)
            if sc_up < best_score:
                best_score = sc_up
                best_seq = upscaled
                print("Coarse-to-fine improved: {:.8f}".format(best_score))

    # Phase 3: Main optimization on top starts
    # Allocate time based on number of starts
    n_starts = len(starts)
    remaining = deadline - time.monotonic() - 300.0  # Reserve 300s for final polish
    if remaining < 30:
        remaining = 30

    # Sort starts by current score
    start_scores = [_exact_score(s) for s in starts]
    order = np.argsort(start_scores)

    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.50, 0.30, 0.20]

    cum = 0.0
    for rank_pos, rank in enumerate(order):
        if rank_pos >= len(fracs):
            break
        cum += fracs[rank_pos]
        sub_dead = t_start + cum * (remaining / sum(fracs[:len(order)]))
        sub_dead = min(sub_dead, deadline - 280.0)

        if time.monotonic() > sub_dead - 10.0:
            continue

        sq = starts[rank].copy()

        # For non-best starts, blend with current best
        if _exact_score(sq) > best_score + 0.001:
            n_sq = sq.size
            best_resampled = _normalize(_resample(best_seq, n_sq))
            sq = _normalize(0.5 * sq + 0.5 * best_resampled)

        run_dead = sub_dead
        # Run mirror Nesterov
        sc, seq = _mirror_nesterov(sq, run_dead, tol=1e-5, eta0=0.5,
                                   verbose=True, weight_mode="uniform")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After start {}: {:.8f}".format(rank_pos + 1, best_score))

        # Try with inverse weighting
        if time.monotonic() < deadline - 300:
            sc2, seq2 = _mirror_nesterov(seq, min(deadline - 280, time.monotonic() + 60),
                                         tol=1e-6, eta0=0.3,
                                         verbose=False, weight_mode="inverse")
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = seq2.copy()
                print("  Inverse weight improved: {:.8f}".format(best_score))

    # Phase 4: Intensive refinement on the best
    # Use multiple passes with decreasing tolerance
    refine_dead = deadline - 100.0
    if time.monotonic() < refine_dead - 60:
        print("\nPhase 4: Intensive refinement")
        for tol_val in [1e-5, 1e-6, 1e-7]:
            if time.monotonic() > refine_dead - 30:
                break
            chunk_dead = min(refine_dead, time.monotonic() + 60)
            sc, seq = _mirror_nesterov(best_seq, chunk_dead, tol=tol_val,
                                       eta0=0.3, verbose=False, weight_mode="uniform")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("  tol={:.0e}: {:.8f}".format(tol_val, sc))

            # Also try inverse
            if time.monotonic() < refine_dead - 30:
                chunk_dead2 = min(refine_dead, time.monotonic() + 30)
                sc, seq = _mirror_nesterov(best_seq, chunk_dead2, tol=tol_val,
                                           eta0=0.2, verbose=False, weight_mode="inverse")
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print("  tol={:.0e} inv: {:.8f}".format(tol_val, sc))

    # Phase 5: Perturbation restarts
    if time.monotonic() < deadline - 50:
        print("\nPhase 5: Perturbation restarts")
        n_perturb = 4
        perturb_budget = deadline - 50 - time.monotonic()
        chunk = perturb_budget / n_perturb

        for pi in range(n_perturb):
            chunk_dead = time.monotonic() + chunk
            if time.monotonic() > chunk_dead - 5:
                break

            # Various perturbation strategies
            n = best_seq.size
            mode = pi % 4
            if mode
