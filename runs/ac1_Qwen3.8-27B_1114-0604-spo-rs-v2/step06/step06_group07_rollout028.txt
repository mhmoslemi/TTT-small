"""
Optimized step function construction using LP-based direction finding combined
with spectral shaping, peak-directed descent, and multi-scale L-BFGS refinement.
Key novelty: LP-based iteration that finds optimal blending directions by
solving linear programs constraining the cross-convolution at dominant peaks.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_constraint_matrix(f, top_k, n):
    """Build sparse K x n constraint matrix for the LP."""
    K = len(top_k)
    A_rows = []
    A_cols = []
    A_vals = []
    for i in range(K):
        k = int(top_k[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        if length > 0:
            A_rows.extend([i] * length)
            A_cols.extend(range(j_start, j_end + 1))
            A_vals.extend(f[j_start:j_end + 1].tolist())
    return csr_matrix((A_vals, (A_rows, A_cols)), shape=(K, n))


def solve_lp_direction(f, conv_f, M, K):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for top K peaks."""
    if not HAS_CVXPY:
        return None
    n = len(f)
    if K >= len(conv_f):
        top_k = np.arange(len(conv_f))
    else:
        top_k = np.argpartition(conv_f, -K)[-K:]
    top_k = top_k[np.argsort(conv_f[top_k])[::-1]]

    A = build_constraint_matrix(f, top_k, n)
    K_actual = len(top_k)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    solvers = [cp.ECOS, cp.SCS]
    if cp.GLPK is not None:
        solvers.append(cp.GLPK)

    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None:
                g_val = np.maximum(np.array(g.value).ravel(), 0.0)
                if g_val.sum() > 1e-12:
                    return g_val
        except Exception:
            continue
    return None

    """LP-based iteration loop to reduce the objective."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    S = float(f.sum())
    if S < 1e-12:
        return f

    best_f = f.copy()

        M = float(c.max())

        g = g / g_sum * S

def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.85):

            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Gradient via cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_x = x[::-1]
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def local_search(a, n_iters=2000, n_perturb=20, rng=None):
    """Simple local search: randomly perturb a few coordinates."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    temp = 0.01

    for it in range(n_iters):
        idx = rng.integers(0, n, size=n_perturb)
        a_new = a.copy()
        for i in idx:
            a_new[i] = max(0.0, a_new[i] * (1.0 + temp * (rng.standard_normal() * 2.0 - 1.0)))
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            temp = max(0.001, temp * 0.999)

    return best_a


def resize_sequence(a, new_n):
    """Resize a sequence to new_n using linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    a_resized = np.interp(new_x, old_x, a)
    return np.maximum(a_resized, 0.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function combining LP-based iteration, spectral shaping,
    peak-directed descent, and L-BFGS refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}, CVXPY available: {HAS_CVXPY}")
    print(f"Parent available: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ============================================================
    # Phase 1: LP-based iteration (main novelty)
    # ============================================================
    phase1_end = t_start + 0.35 * (total_budget - 30.0)
    print(f"\n=== Phase 1: LP-based Iteration (deadline t={phase1_end - t_start:.0f}s) ===")

    if HAS_CVXPY and parent is not None:
        f_lp = parent.copy()
        S = f_lp.sum()
        f_lp = f_lp / S

        # Multiple LP runs with different K values
        for K in [30, 50, 100, 200]:
            if time.monotonic() > phase1_end - 60:
                break
            print(f"  LP with K={K}")
            t_lp = min(120.0, phase1_end - time.monotonic() - 30)
            if t_lp < 30:
                break
            f_lp = lp_iterate(f_lp, n_iters=30, K=K, t_init=0.08, time_limit=t_lp)
            sc = compute_score(f_lp)
            update_best(sc, f_lp, f"LP K={K}")

        # Second round with smaller step
        if time.monotonic() < phase1_end - 60:
            print(f"  LP round 2 (smaller steps)")
            t_lp = min(120.0, phase1_end - time.monotonic() - 30)
            if t_lp > 30:
                f_lp = lp_iterate(f_lp, n_iters=20, K=80, t_init=0.03, time_limit=t_lp)
                sc = compute_score(f_lp)
                update_best(sc, f_lp, "LP-R2")

    # ============================================================
    # Phase 2: L-BFGS refinement (continuation method)
    # ============================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS Continuation (deadline t={phase2_end - t_start:.0f}s) ===")

    betas = [200, 500, 1000, 3000, 8000, 20000]
    for beta in betas:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  L-BFGS beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(6000, int(remaining * 20))
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS b={beta}")

    # ============================================================
    # Phase 3: Peak-directed descent with multiple peak counts
    # ============================================================
    phase3_end = t_start + 0.78 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak-Directed Descent (deadline t={phase3_end - t_start:.0f}s) ===")

    for n_peaks in [5, 10, 20, 50]:
        if time.monotonic() > phase3_end - 30:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        n_iters_pd = min(3000, int(remaining * 15))
        print(f"  Peak-directed n_peaks={n_peaks}, iters={n_iters_pd}")
        a_pd = peak_directed_descent(best_a, n_iters=n_iters_pd, n_peaks=n_peaks,
                                     step_init=0.01, decay=0.99995)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

    # ============================================================
    # Phase 4: Spectral shaping + local search
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Spectral Shaping + Local Search (deadline t={phase4_end - t_start:.0f}s) ===")

    # Geometric Fienup
    if time.monotonic() < phase4_end - 60:
        remaining = phase4_end - time.monotonic()
        n_fienup = min(300, int(remaining * 2))
        a_geo = fienup_geometric(best_a, n_iters=n_fienup, alpha_start=0.03, alpha_end=0.6)
        sc = compute_score(a_geo)
        update_best(sc, a_geo, "GEO")

    # Elliptical
    if time.monotonic() < phase4_end - 60:
        remaining = phase4_end - time.monotonic()
        n_ellip = min(300, int(remaining * 2))
        a_ell = fienup_ellip(best_a, n_iters=n_ellip, beta_shape=4.0)
        sc = compute_score(a_ell)
        update_best(sc, a_ell, "ELLI")

    # Local search
    if time.monotonic() < phase4_end - 30:
        remaining = phase4_end - time.monotonic()
        n_ls = min(5000, int(remaining * 30))
        a_ls = local_search(best_a, n_iters=n_ls, n_perturb=50, rng=rng)
        sc = compute_score(a_ls)
        update_best(sc, a_ls, "LOCAL")

    # ============================================================
    # Phase 5: Final polish - more LP + L-BFGS
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    if HAS_CVXPY and time.monotonic() < deadline - 120:
        remaining = deadline - time.monotonic() - 30
        if remaining > 60:
            print(f"  Final LP iterations, remaining={remaining:.0f}s")
            f_final = lp_iterate(best_a, n_iters=20, K=60, t_init=0.05,
                                 time_limit=remaining * 0.5)
            sc = compute_score(f_final)
            update_best(sc, f_final, "FINAL-LP")

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            print(f"  Final L-BFGS, remaining={remaining:.0f}s")
            sc, a_opt = run_lbfgs(best_a, 5000, maxiter=min(4000, int(remaining * 15)),
                                  maxfun=min(4000, int(remaining * 15)) * 15)
            update_best(sc, a_opt, "FINAL-LBFGS")

    # Peak-directed final
    if time.monotonic() < deadline - 30:
        a_pf = peak_directed_descent(best_a, n_iters=2000, n_peaks=15,
                                     step_init=0.005, decay=0.99999)
        sc = compute_score(a_pf)
        update_best(sc, a_pf, "FINAL-PEAK")

    # ============================================================
    # Try alternate lengths (truncation)
    # ============================================================
    if time.monotonic() < deadline - 60:
        print(f"\n=== Trying alternate lengths ===")
        for target_n in [len(best_a) // 2, int(len(best_a) * 0.75)]:
            if target_n < 512 or time.monotonic() > deadline - 40:
                break
            a_resized = resize_sequence(best_a, target_n)
            sc_r = compute_score(a_resized)
            print(f"  Resized to n={target_n}: score={sc_r:.8f}")
            if sc_r < best_score:
                # Quick L-BFGS on resized
                sc, a_opt = run_lbfgs(a_resized, 1000, maxiter=2000, maxfun=30000)
                update_best(sc, a_opt, f"RESIZE n={target_n}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
