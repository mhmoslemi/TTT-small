"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adaptive LSE-smoothed L-BFGS-B with backtracking
2. Nesterov accelerated subgradient with backtracking line search
3. Frequency-domain power shaping (PAPR-reduction inspired peak shaving)
4. Active-set direct gradient descent on max-conv
5. Multi-start with diverse perturbations and length exploration

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """Exact subgradient using top-k active peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size) / a.size
    return a / S


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size
    if maxfun is None:
        maxfun = 300

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(maxfun // 2, 30),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _nesterov_backtrack(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """Nesterov subgradient with backtracking line search."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-14:
        return best_score, best_a

    eta = 0.1 / max(g_norm, 1e-10)
    no_improve_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break

        # Backtracking line search
        step = eta
        best_x = a.copy()
        best_f = _exact_score(a)

        for bt in range(20):
            x_try = y_norm - step * grad_y
            x_try = _project_simplex(x_try)
            f_try = _exact_score(x_try)

            if np.isfinite(f_try) and f_try < best_f * (1.0 - 1e-5):
                best_x = x_try
                best_f = f_try
                break
            elif np.isfinite(f_try) and f_try < best_f:
                best_x = x_try
                best_f = f_try
            step *= 0.5

        v = best_x - a
        a = best_x
        t = t_new

        if best_f < best_score:
            best_score = best_f
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count % 30 == 0:
                eta *= 0.6

        # Periodic eta refresh if improving well
        if no_improve_count == 0 and it % 100 == 0:
            _, g_now = _subgrad_exact(a, k_active, tol)
            gn = np.linalg.norm(g_now)
            if gn > 1e-14:
                eta = max(eta * 1.1, 0.5 / gn)

    return best_score, best_a


def _freq_shaping(a, n_iters=50, deadline=None):
    """
    Frequency-domain power shaping: reduce peak of IFFT(|A|^2) by
    iteratively de-emphasizing frequency bins that contribute to the peak lag.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        c = _fast_self_conv(a)
        m_peak = int(np.argmax(c))
        cmax = float(c[m_peak])
        c_mean = float(c.mean())

        if cmax <= c_mean * 1.01:
            break

        # Get power spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2

        # Contribution of each freq bin to the peak lag
        # IFFT[P][m_peak] = (1/N) * sum_k P[k] * exp(2*pi*i*k*m_peak/N)
        k_freqs = np.arange(len(P))
        phase = 2.0 * np.pi * k_freqs * m_peak / N
        contrib_real = P * np.cos(phase)
        contrib_imag = P * np.sin(phase)

        # We want to reduce the real part contribution to the peak
        # Reduce power where cos(phase) > 0 (constructive interference)
        # But we can't make P negative, so we scale down
        # Use a soft threshold: reduce P where it contributes most to peak
        thresh = np.percentile(P, 90)
        mask = (contrib_real > 0) & (P > thresh)

        # Scale factor: reduce by a small amount
        alpha = 0.05
        P_new = P.copy()
        P_new[mask] *= (1.0 - alpha)

        # Compensate: add back to non-peak-contributing bins
        total_removed = P[mask].sum() - P_new[mask].sum()
        inv_mask = ~mask
        if inv_mask.sum() > 0:
            P_new[inv_mask] += total_removed / inv_mask.sum()

        # Recover a from modified power spectrum (keep phase)
        A_new = A.copy()
        A_new[np.arange(len(A)) <= len(P_new)] = np.sqrt(P_new) * (A[:len(P_new)] / np.maximum(np.abs(A[:len(P_new)])), 1e-30))

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = _project_simplex(a_new)

        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce alpha gradually
            alpha *= 0.7

    return best_score, best_a


def _peak_shave(a, n_iters=100, deadline=None):
    """
    Direct peak shaving: at each step, compute gradient of the maximum
    convolution lag and take a projected gradient step.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    # Initial step size
    _, g = _subgrad_exact(a, k_active=1, tol=0.0)
    gn = np.linalg.norm(g)
    step = 0.01 / max(gn, 1e-10)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        # Use single active peak for focused descent
        score, grad = _subgrad_exact(a, k_active=1, tol=0.0)

        # Backtracking
        s = step
        improved = False
        for _ in range(15):
            a_try = _project_simplex(a - s * grad)
            sc_try = _exact_score(a_try)
            if np.isfinite(sc_try) and sc_try < score * (1 - 1e-6):
                a = a_try
                step = min(s * 1.5, 0.5 / max(np.linalg.norm(grad), 1e-10))
                improved = True
                break
            s *= 0.5

        if not improved:
            step *= 0.3

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def _perturb(a, rng, strength=None):
    """Generate diverse perturbations."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.4))
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = a * np.exp(strength * 0.4 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        K = min(51, max(7, n // 200))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        sig = max(1.0, c / 3.0)
        kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.2 * noise)
    elif mode == 4:
        # Anti-symmetric perturbation: boost one half, reduce other
        t = float(rng.uniform(0.1, 0.3))
        q = a.copy()
        half = n // 2
        q[:half] *= (1.0 + t)
        q[half:] *= (1.0 - t * half / (n - half))
    elif mode == 5:
        # Frequency-domain noise
        Nf = _next_pow2(n)
        A = np.fft.rfft(a, Nf)
        phase_noise = rng.standard_normal(len(A)) * 0.01
        A = A * np.exp(1j * phase_noise)
        q = np.fft.irfft(A, Nf)[:n]
    else:
        # Blend with a structured sequence
        x = np.linspace(0, 1, n)
        structured = 1.0 + 0.5 * np.cos(2 * np.pi * float(rng.integers(1, 5)) * x)
        t = float(rng.uniform(0.1, 0.3))
        q = (1 - t) * a + t * structured / structured.mean()

    return _project_simplex(q)


def _hybrid_optimize(a0, deadline, label=""):

            best_a = a2.copy()
            print("[{}] LSE b={:.0f}: {:.8f}".format(label, beta, best_score))

                best_a = a2.copy()

        sc, a2 = _peak_shave(a, n_iters=200, deadline=p4_end - 5.0)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] PeakShave: {:.8f}".format(label, best_score))
        a = _normalize(a2)

        # Final Nesterov with very tight tol
        rem = p4_end - time.monotonic()
        if rem > 10.0:
            sc, a2 = _nesterov_backtrack(a, n_iters=int(max(100, rem * 35)),
                                          deadline=p4_end, k_active=1, tol=1e-8)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Final Nest: {:.8f}".format(label, best_score))
            a = _normalize(a2)

        # One more LSE with very high beta
        rem = p4_end - time.monotonic()
        if rem > 15.0:
            for beta in [200000, 500000, 1000000]:
                rem = p4_end - time.monotonic()
                if rem < 10.0:
                    break
                maxfun = int(max(20, min(200, rem * 3)))
                sc, a2 = _lbfgs_lse(a, beta, p4_end - 1.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("[{}] Final LSE b={:.0f}: {:.8f}".format(label, beta, best_score))
                a = _normalize(a2)

    return best_score, best_a

            parent_len = 4096
        lengths = [parent_len]
        for L in [8192, 12288, 6144, 10240, 16384]:
            if L != parent_len and L not in lengths:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288, 6144]

    lengths = [L for L in lengths if 2048 <= L <= 16384][:5]
    print("Lengths: {}".format(lengths))

    # Generate initial candidates
    all_scores = []
    all_seqs = []

    for n in lengths[:2]:  # Only evaluate top 2 lengths for speed
        # Parent resampled
        if parent is not None:
            pr = _normalize(_resample(parent, n))
            all_seqs.append(pr)
            all_scores.append(_exact_score(pr))
            # Reversed
            pr_rev = _normalize(pr[::-1])
            all_seqs.append(pr_rev)
            all_scores.append(_exact_score(pr_rev))

        # Structured inits
        x = np.linspace(0.0, 1.0, n)
        inits = [
            np.ones(n),
            1.0 + 0.5 * np.abs(2.0 * x - 1.0),
            1.0 + 0.8 * (1.0 - (2.0 * x - 1.0) ** 2),
            1.0 + 0.5 * np.cos(2.0 * np.pi * x),
            1.0 + 0.5 * np.cos(4.0 * np.pi * x),
            np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x)) + 0.01,
            # Anti-symmetric emphasis
            np.exp(-2.0 * x) + 0.01,
            np.exp(-2.0 * (1.0 - x)) + 0.01,
            # Bimodal
            np.exp(-(((x - 0.3) / 0.15) ** 2)) + np.exp(-(((x - 0.7) / 0.15) ** 2)) + 0.01,
        ]
        for init in inits:
            init = _normalize(init)
            all_seqs.append(init)
            all_scores.append(_exact_score(init))

    # Sort and select
    all_scores = np.array(all_scores)
    order = np.argsort(all_scores)
    print("Best init: {:.8f} (len={})".format(all_scores[order[0]], all_seqs[order[0]].size))

    # Select diverse starts (different lengths if possible)
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 3:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Main optimization: allocate time to top starts
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))
    n_starts = len(starts_sorted)

    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.50, 0.30, 0.20]

    main_end = deadline - 500.0
    main_budget = max(150.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 30.0:
            continue

        sq = starts[si].copy()

        # For secondary starts, blend with current best
        if rank > 0:
            if len(best_seq) != len(sq):
                best_r = _resample(best_seq, len(sq))
            else:
                best_r = best_seq.copy()
            t_blend = 0.3 + 0.2 * rank
            sq = _normalize((1.0 - t_blend) * _normalize(sq) + t_blend * _normalize(best_r))

        sc, seq = _hybrid_optimize(sq, sub_dead, "main{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After main{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Perturbation restarts
    n_perturb = 4
    perturb_start = time.monotonic()
    perturb_end = deadline - 300.0
    if perturb_end > perturb_start + 60.0:
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 5.0:
                break

            p = _perturb(best_seq, rng, strength=float(rng.uniform(0.05, 0.25)))
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Final intensive polishing
    final_dead = deadline - 15.0
    if time.monotonic() < final_dead - 30.0:
        # High beta LSE
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            for beta in [200000, 500000, 1000000, 2000000]:
                rem = final_dead - time.monotonic()
                if rem < 12.0:
                    break
                maxfun = int(max(20, min(300, rem * 4)))
                sc, a2 = _lbfgs_lse(best_seq, beta, final_dead - 3.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final LSE b={:.0f}: {:.8f}".format(beta, best_score))
                best_seq = _normalize(a2)

        # Nesterov k=1, very tight
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            sc, a2 = _nesterov_backtrack(best_seq, n_iters=int(max(200, rem * 30)),
                                          deadline=final_dead, k_active=1, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nest k=1: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Peak shave
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            sc, a2 = _peak_shave(best_seq, n_iters=int(max(50, rem * 20)),
                                  deadline=final_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final PeakShave: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Final Nesterov k=2
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            sc, a2 = _nesterov_backtrack(best_seq, n_iters=int(max(50, rem * 25)),
                                          deadline=final_dead, k_active=2, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nest k=2: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
