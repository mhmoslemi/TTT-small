import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def pressure_fixed_point(a0, n_iters=5000, alpha=0.1, p_power=5.0, time_limit=None):
    """
    Multiplicative weight / pressure-based fixed point iteration.
    At optimality, the 'pressure' P_i = sum_k w_k * a_{k-i} should be constant
    for all i with a_i > 0 (KKT-like condition).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute weights: emphasize peaks
        c_norm = c / (c_max + 1e-30)
        w = c_norm ** p_power
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones_like(c) / len(c)
        else:
            w = w / w_sum

        # Compute pressure: P_i = 2 * sum_k w_k * a_{k-i} = 2 * (w * a_rev)[i]
        a_rev = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        WA = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(a_rev, M2_fft)
        pressure = np.fft.irfft(WA * RA, M2_fft)[n - 1:2 * n - 1]
        pressure = np.maximum(pressure, 1e-15)

        # Multiplicative update: a_i <- a_i / pressure_i^alpha
        a = a / (pressure ** alpha)
        a = np.maximum(a, 1e-15)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def nesterov_subgradient(a0, n_iters=10000, n_peaks=50, temp=0.01,
                          step_init=0.02, decay=0.99995, time_limit=None):
    """
    Nesterov-accelerated subgradient method for minimizing max(conv(a))/(sum a)^2.
    Uses softmax-weighted peaks for a smooth subgradient approximation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and (time.monotonic() - t0) > time_limit:
            break

        # Nesterov lookahead
        a_look = a - v
        a_look = np.maximum(a_look, 1e-15)

        c = fast_conv(a_look)
        c_max = float(c.max())
        S_look = float(a_look.sum())

        # Softmax weights over peaks
        temp_actual = temp * c_max + 1e-30
        exponents = (c - c_max) / temp_actual
        exponents = np.maximum(exponents, -50)  # clip for numerical stability
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        # Weighted gradient of C: sum_k w_k * grad_Ck
        # grad_Ck[j] = 2 * a_{k-j} (if valid)
        # So weighted grad[j] = 2 * sum_k w_k * a_{k-j} = 2 * (w * a_rev)[j]
        a_rev = a_look[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        WA = np.fft.rfft(w, M2_fft)
        RA = np.fft.rfft(a_rev, M2_fft)
        weighted_grad_C = np.fft.irfft(WA * RA, M2_fft)[n - 1:2 * n - 1]
        weighted_grad_C = 2.0 * weighted_grad_C

        # Gradient of full objective: 2n * (weighted_grad_C * S - 2*L*a) / S^3
        L = float(np.dot(w, c))  # weighted "LSE" approximation of max
        S2 = S_look * S_look
        S3 = S2 * S_look
        grad = (2.0 * n) * (weighted_grad_C * S_look - 2.0 * L * a_look) / S3

        # Nesterov update
        v = 0.9 * v + step * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= decay

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def soft_active_set_lbfgs(a0, n_iters_inner=3000, max_outer=20, time_limit=None):

        S3 = S2 * S_local
        grad = (2.0 * n) * (g * S_local - 2.0 * L) / S3

    for outer in range(max_outer):

        if S < 1e-12:

            best_a = a.copy()
            if (outer + 1) % 5 == 0:

def direct_peak_reduction(a0, n_rounds=100, top_k=30, time_limit=None):
    """
    Directly reduce top peaks by finding the contributing elements
    and reducing them proportionally to their contribution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if (time_limit is not None and (time.monotonic() - t0) > time_limit):
            break

        c = fast_conv(a)
        c_max = float(c.max())

        # Find top-k peaks
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For each peak, compute contribution of each a_j
        reduction = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner_idx = k - np.arange(j_start, j_end + 1)
            partner = a[partner_idx]
            contrib = seg * partner
            # Add to reduction proportional to contribution
            reduction[j_start:j_end + 1] += contrib

        # Apply reduction
        max_red = reduction.max()
        if max_red > 1e-15:
            reduction = reduction / max_red
            a = a * (1.0 - 0.3 * reduction)

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        else:

    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.3 * np.cos(2.0 * np.pi * x), 0.01))

        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, np.maximum(parent, 0)).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting optimization with n={n}")
    print(f"Initial best: {best_score:.8f}")

    remaining_budget = lambda: deadline - time.monotonic()

    # ==================================================================
    # Phase 1: Pressure fixed point (30% of time)
    # ==================================================================
    phase1_frac = 0.30
    print(f"\n=== Phase 1: Pressure Fixed Point ===")

    pf_time = phase1_frac * (total_budget - 50.0) * 0.5
    a_pf = pressure_fixed_point(best_a, n_iters=10000, alpha=0.05, p_power=8.0, time_limit=pf_time)
    sc = compute_score(a_pf)
    update_best(sc, a_pf)

    if time.monotonic() < t_start + phase1_frac * (total_budget - 50.0):
        pf_time2 = (t_start + phase1_frac * (total_budget - 50.0)) - time.monotonic()
        a_pf2 = pressure_fixed_point(best_a, n_iters=10000, alpha=0.02, p_power=20.0, time_limit=max(pf_time2, 10))
        sc = compute_score(a_pf2)
        update_best(sc, a_pf2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE-LBFGS with beta schedule (30% of time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 50.0)
    print(f"\n=== Phase 2: LSE-LBFGS ===")

    betas = [5000, 20000, 50000, 100000, 200000, 500000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        t_beta = time.monotonic()
        time_left = max(phase2_end - time.monotonic(), 30)
        maxfun = min(30000, int(time_left * 200))
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=maxfun // 3, maxfun=maxfun)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Nesterov subgradient (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 50.0)
    print(f"\n=== Phase 3: Nesterov Subgradient ===")

    time_left = max(phase3_end - time.monotonic(), 30)
    n_iters_sg = int(time_left * 150)  # estimate iterations
    a_sg = nesterov_subgradient(best_a, n_iters=n_iters_sg, n_peaks=50,
                                 temp=0.005, step_init=0.01, decay=0.99998,
                                 time_limit=time_left)
    sc = compute_score(a_sg)
    update_best(sc, a_sg)

    # Also try with different temperature
    if time.monotonic() < phase3_end:
        time_left2 = max(phase3_end - time.monotonic(), 10)
        a_sg2 = nesterov_subgradient(best_a, n_iters=int(time_left2 * 150),
                                      n_peaks=20, temp=0.001, step_init=0.005,
                                      decay=0.99999, time_limit=time_left2)
        sc = compute_score(a_sg2)
        update_best(sc, a_sg2)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Soft active set + peak reduction (10% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 50.0)
    print(f"\n=== Phase 4: Active Set + Peak Reduction ===")

    time_left = max(phase4_end - time.monotonic(), 20)
    a_sas = soft_active_set_lbfgs(best_a, n_iters_inner=2000, max_outer=10, time_limit=time_left * 0.6)
    sc = compute_score(a_sas)
    update_best(sc, a_sas)

    if time.monotonic() < phase4_end:
        time_left2 = max(phase4_end - time.monotonic(), 10)
        a_ptr = direct_peak_reduction(best_a, n_rounds=200, top_k=50, time_limit=time_left2)
        sc = compute_score(a_ptr)
        update_best(sc, a_ptr)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish - interleaved methods (10% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    polish_rounds = 8
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # Short pressure iteration
        t_lim = min(30, max(deadline - time.monotonic(), 5))
        a_pf_final = pressure_fixed_point(best_a, n_iters=int(t_lim * 50),
                                           alpha=0.01, p_power=30.0, time_limit=t_lim)
        sc = compute_score(a_pf_final)
        update_best(sc, a_pf_final)

        # LSE with very high beta
        if time.monotonic() > deadline:
            break
        beta_final = 500000 + 200000 * pr
        t_lim2 = max(deadline - time.monotonic() - 10, 5)
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta_final,),
                          maxiter=2000, maxfun=min(5000, int(t_lim2 * 100)))
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Nesterov
        if time.monotonic() > deadline:
            break
        t_lim3 = max(deadline - time.monotonic() - 5, 5)
        a_nm = nesterov_subgradient(best_a, n_iters=int(t_lim3 * 200),
                                     n_peaks=30, temp=0.001, step_init=0.003,
                                     decay=0.99999, time_limit=t_lim3)
        sc = compute_score(a_nm)
        update_best(sc, a_nm)

        print(f"  Polish round {pr}: best = {best_score:.8f}")

    # ==================================================================
    # Try different n values if time permits
    # ==================================================================
    if time.monotonic() < deadline - 120:
        print(f"\n=== Phase 6: Try different n ===")
        for n_try in [n * 2, n // 2, n + 1000]:
            if time.monotonic() > deadline - 60:
                break
            if n_try < 100 or n_try > 50000:
                continue
            # Resample parent to new n
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()

            t_lim = max(deadline - time.monotonic() - 60, 30)

            # Quick LSE
            a_opt = run_lbfgs(a_try, lse_score_grad, (50000,), maxiter=2000, maxfun=5000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

            # Quick pressure
            if time.monotonic() < deadline - 30:
                a_pf = pressure_fixed_point(a_opt, n_iters=int(t_lim * 30),
                                             alpha=0.03, p_power=10.0,
                                             time_limit=t_lim * 0.5)
                sc = compute_score(a_pf)
                update_best(sc, a_pf)

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
