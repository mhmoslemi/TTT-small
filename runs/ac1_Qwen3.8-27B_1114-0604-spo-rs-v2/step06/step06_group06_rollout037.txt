def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_active_weights_conv(conv, top_k=10, eps_frac=0.01):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.05 * conv.size))
    if active.sum() > max_count:
        active = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active[top_idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_update(a, conv, top_k, eps_frac, eta):
    w = compute_active_weights_conv(conv, top_k=top_k, eps_frac=eps_frac)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def variance_reduction_update(a, conv, top_m, eta):
    d = np.sort(conv)[::-1][:top_m]
    mean_top = float(np.mean(d))
    excess = np.maximum(conv - mean_top, 0.0)
    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def lse_score_grad(a, beta):
    """LSE-smoothed objective and gradient (a assumed normalized, sum=1)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    # Gradient of 2*n*LSE with respect to a (with sum(a)=S constraint handled by normalizing)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxfun=2000):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def spectral_init(n, rng, iterations=200):
    """Gerchberg-Saxton: find non-negative sequence with flat power spectrum."""
    x = np.linspace(0.0, 1.0, n)
    # Start with raised cosine taper (zero at boundaries)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    N = next_power_of_two(n)
    for _ in range(iterations):
        A = np.fft.rfft(a, N)
        mags = np.abs(A)
        # Target: smooth spectrum (slightly tilted to reduce DC)
        freqs = np.arange(len(mags))
        target = 1.0 / (1.0 + 0.1 * freqs / max(1, len(mags)))
        # Blend current and target
        alpha = 0.5
        new_mags = alpha * mags + (1.0 - alpha) * target * mags.max()
        phases = np.angle(A)
        A_new = new_mags * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new /= ss
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new
    return a

        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.6))
    mode = int(rng.integers(0, 12))
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.5))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = compute_active_weights_conv(conv, top_k=10, eps_frac=0.02)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.5 * noise)
    elif mode == 5:
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        nb = int(rng.integers(4, 30))
        block = 1.0 + strength * 0.5 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 9:
        # Spectral perturbation
        Nf = next_power_of_two(n)
        A = np.fft.rfft(p, Nf)
        mags = np.abs(A)
        phases = np.angle(A)
        mags *= np.exp(0.1 * strength * rng.standard_normal(len(mags)))
        A_new = mags * np.exp(1j * phases)
        q = np.fft.irfft(A_new, Nf)[:n]
    elif mode == 10:
        # Large perturbation: blend with spectral init
        sp = spectral_init(n, rng, iterations=50)
        t = float(rng.uniform(0.2, 0.6))
        q = (1.0 - t) * p + t * sp
    else:
        q = p * (0.7 + 0.6 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def active_set_optimize(a, deadline, verbose=True, tag=""):
    """Active-set exponentiated gradient with Armijo-style line search."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    top_k = 5
    eps_frac = 0.05
    eta = 0.3
    stagnant = 0
    max_stagnant = 50

    for phase in range(30):
        if time.monotonic() > deadline:
            break
        phase_dead = min(deadline, t0 + (deadline - t0) * (phase + 1) / 30.0 + 10.0)

        for it in range(300):
            if time.monotonic() > phase_dead - 3.0:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))

            # Armijo-style: try decreasing step sizes
            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            # Try a range of step sizes centered on current eta
            etas = [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0, eta * 5.0]
            # Also try very small steps for fine adjustment
            etas = [max(0.001, e) for e in etas]

            for eta_try in etas:
                new_a = exp_grad_update(a, conv, top_k, eps_frac, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            # Variance reduction
            for tm_factor in [0.5, 1.0, 2.0]:
                tm = max(3, int(min(50, top_k * tm_factor + 5)))
                vr_a = variance_reduction_update(a, conv, tm, best_eta * 0.5)
                vr_a = normalize_a(vr_a)
                vr_sc = exact_score(vr_a)
                if vr_sc < best_sc_iter:
                    best_sc_iter = vr_sc
                    best_a_iter = vr_a

            if best_sc_iter < best_sc - 1e-11:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter.copy()
                eta = best_eta
                stagnant = 0
                if verbose and it % 100 == 0:
                    print("  [{} p{} i{}] score={:.9f} top_k={} eps={:.3f} eta={:.3f}".format(
                        tag, phase, it, best_sc, top_k, eps_frac, eta))
            else:
                stagnant += 1
                eta *= 0.8
                if stagnant >= max_stagnant:
                    break

        # Adapt parameters
        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.5) + 2, 150)
            eps_frac = min(eps_frac * 1.4, 0.2)
            eta = max(0.05, eta * 0.5)
            stagnant = 0
            if verbose:
                print("  [{}] expand: top_k={}, eps={:.3f}".format(tag, top_k, eps_frac))
        else:
            top_k = max(3, int(top_k * 0.93))
            eps_frac = max(0.003, eps_frac * 0.93)

    return float(best_sc), best_a


def subgrad_fine_polish(a, deadline, verbose=False, tag=""):
    """Fine-grained subgradient: systematically sweep (top_k, eta, eps)."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    conv = fast_self_conv(a)

    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 75, 100, 150]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0, 3.0]
    eps_fracs = [0.0, 0.003, 0.005, 0.01, 0.02, 0.05]

    improved = True
    round_num = 0
    while improved and time.monotonic() < deadline and round_num < 12:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)
        for tk in top_ks:
            if time.monotonic() > deadline:
                break
            for ef in eps_fracs:
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    new_a = exp_grad_update(best_a, conv, tk, ef, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = new_a
                        improved = True
                        if verbose:
                            print("  [{} sg r{}] top_k={} eps={:.3f} eta={:.2f} -> {:.9f}".format(
                                tag, round_num, tk, ef, eta, best_sc))
        # Variance reduction
        for tm in [5, 10, 20, 30, 50]:
            if time.monotonic() > deadline:
                break
            for eta in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                if time.monotonic() > deadline:
                    break
                new_a = variance_reduction_update(best_a, conv, tm, eta)
                new_a = normalize_a(new_a)
                sc = exact_score(new_a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = new_a
                    improved = True

    return float(best_sc), best_a


def lse_lbfgs_refine(a, deadline, verbose=True, tag=""):
    """Extended LSE-LBFGS with adaptive beta annealing."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Beta schedule: start low (smooth), end high (sharp)
    betas = [100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]

    for beta in betas:
        if time.monotonic() > deadline - 10:
            break
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        # Scale maxfun with remaining time
        mf = max(100, int(rem * 2.0))
        mf = min(mf, 5000)
        sc, a = run_lbfgs(a, float(beta), deadline, maxfun=mf)
        a = normalize_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print("  [{}] LSE beta={}: {:.9f}".format(tag, beta, best_sc))
        # Also try from a slightly perturbed version
        if time.monotonic() < deadline - 20:
            a2 = perturb(a, np.random.default_rng(int(beta)), strength=0.05)
            sc2, a2 = run_lbfgs(a2, float(beta), deadline, maxfun=mf // 2)
            a2 = normalize_a(a2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.copy()
                a = a2.copy()
                if verbose:
                    print("  [{}] LSE beta={} (pert): {:.9f}".format(tag, beta, best_sc))

    return float(best_sc), best_a


def coordinate_descent_polish(a, deadline, verbose=False, tag=""):
    """
    Coordinate descent on the LSE objective.
    For each coordinate, optimize a[i] while fixing others.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Use LSE with moderate beta for smoothness
    beta = 5000.0

    n_sweeps = 0
    while time.monotonic() < deadline and n_sweeps < 5:
        n_sweeps += 1
        improved = False
        for i in range(0, n, max(1, n // 500)):
            if time.monotonic() > deadline:
                break
            # Try different values of a[i]
            orig = a[i]
            # Sample log-spaced values around current
            candidates = [orig * 0.5, orig * 0.7, orig * 0.85, orig * 1.15, orig * 1.3, orig * 2.0, 0.0]
            best_val = orig
            best_local_sc = exact_score(a)
            for val in candidates:
                a[i] = max(0.0, val)
                a_norm = normalize_a(a)
                sc = exact_score(a_norm)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_val = max(0.0, val)
            a[i] = best_val
            a = normalize_a(a)
            sc = exact_score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
                print("Loaded parent: n={}, score={:.9f}".format(
                    p.size, exact_score(p)))
    except Exception:
        parent = None

    # Choose primary n
    n = 10240
    if parent is not None:
        n = int(parent.size)
        if n < 2048:
            n = 8192
        elif n > 20000:
            n = 16384

    a0 = None
    best_sc = np.inf
    best_a = None

    # Generate initial candidates and pick the best
    if parent is not None:
        a0 = resample_sequence(parent, n)
        a0 = normalize_a(a0)
        sc0 = exact_score(a0)
        best_a = a0.copy()
        best_sc = sc0
        print("Initial from parent: n={}, score={:.9f}".format(n, best_sc))
    else:
        inits = generate_inits(n, rng, None, cap=20)
        for init in inits:
            sc = exact_score(init)
            if sc < best_sc:
                best_sc = sc
                best_a = init.copy()
        a0 = best_a.copy()
        print("No parent. Best init: n={}, score={:.9f}".format(n, best_sc))

    a = a0.copy()

    # Phase 1: Active-set optimization (40% of time)
    p1_dead = t_start + 0.40 * total_budget
    sc, a = active_set_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 1: {:.9f}".format(best_sc))

    # Phase 2: LSE-LBFGS refinement (30% of time)
    p2_dead = t_start + 0.70 * total_budget
    a = normalize_a(best_a)
    sc, a = lse_lbfgs_refine(a, deadline=p2_dead, verbose=True, tag="p2")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 2 (LSE-LBFGS): {:.9f}".format(best_sc))

    # Phase 3: Active-set again on improved solution (10% of time)
    p3_dead = t_start + 0.80 * total_budget
    a = normalize_a(best_a)
    sc, a = active_set_optimize(a, deadline=p3_dead, verbose=True, tag="p3")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 3: {:.9f}".format(best_sc))

    # Phase 4: Subgradient fine polish (5% of time)
    p4_dead = t_start + 0.85 * total_budget
    a = normalize_a(best_a)
    sc, a = subgrad_fine_polish(a, deadline=p4_dead, verbose=True, tag="p4")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 4: {:.9f}".format(best_sc))

    # Phase 5: Diversification restarts (7% of time)
    p5_dead = t_start + 0.92 * total_budget
    n_restarts = 5
    restart_rng = np.random.default_rng(seed + 1000)
    while time.monotonic() < p5_dead - 20 and n_restarts > 0:
        rem = p5_dead - 20.0 - time.monotonic()
        if rem < 30:
            break
        n_restarts -= 1
        strength = float(restart_rng.uniform(0.1, 0.4))
        p = perturb(best_a, restart_rng, strength=strength)
        chunk_dead = time.monotonic() + min(rem * 0.6, 120.0)
        sc, seq = active_set_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(n_restarts))
        if sc < best_sc:
            # Quick LSE polish on the improved restart
            if time.monotonic() < p5_dead - 15:
                sc2, seq2 = lse_lbfgs_refine(seq, deadline=p5_dead - 5, verbose=False, tag="rp{}".format(n_restarts))
                if sc2 < sc:
                    sc = sc2
                    seq = seq2
            best_sc = sc
            best_a = seq.copy()
            print("Restart {}: {:.9f}".format(n_restarts, best_sc))

    # Phase 6: Final LSE-LBFGS polish (remaining time)
    if time.monotonic() < deadline - 10:
        a = normalize_a(best_a)
        sc, a = lse_lbfgs_refine(a, deadline=deadline - 5, verbose=True, tag="final")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("After final polish: {:.9f}".format(best_sc))

    # Phase 7: Try one more subgradient sweep if time
    if time.monotonic() < deadline - 5:
        a = normalize_a(best_a)
        sc, a = subgrad_fine_polish(a, deadline=deadline - 2, verbose=False, tag="finalsg")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final cleanup
    print("Best internal: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
