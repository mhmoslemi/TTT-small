```python
def peak_pair_optimize(a, K_lags=5, K_pairs_per_lag=20, n_iters=100):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = a.sum()
        M = c.max()
        sc = 2*n*M/S**2
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top K lags
        top_lags = np.argsort(c)[::-1][:K_lags]
        
        # For each top lag, find top contributing pairs
        positions = set()
        for k in top_lags:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k - js]
            top_local = np.argsort(pairs)[::-1][:K_pairs_per_lag]
            for t in top_local:
                positions.add(int(js[t]))
                positions.add(int(k - js[t]))
        
        # Now optimize: reduce a[i] for i in positions
        # Use a simple approach: reduce proportionally to contribution
        pos_array = np.array(list(positions))
```

```python
    """Exact score: 2n * max(conv) / (sum)^2."""

def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * 1.0 / S3)

    score = (2.0 * n * L) / S2
    return score, grad

def stochastic_lse_anneal(a0, rng, total_time, n_passes=3):
    """
    Stochastic LSE annealing: interleave L-BFGS optimization with
    random perturbations. Beta schedule is fine-grained in the critical range.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    # Fine-grained beta schedule with emphasis on critical range
    betas = [10, 20, 40, 80, 150, 300, 500, 800, 1200, 2000,
             3500, 5000, 7000, 10000, 15000, 20000, 30000, 50000, 80000]

    pert_scale = 0.02  # Initial perturbation scale

    for beta in betas:
        if time.monotonic() - t_start > total_time:
            break

        remaining = total_time - (time.monotonic() - t_start)
        # Allocate time: more for mid-range betas
        beta_weight = 1.0 / (1.0 + 0.2 * abs(np.log10(beta) - 3.5))
        maxiter = max(200, int(remaining * 8 * beta_weight / (len(betas) - betas.index(beta) + 1)))
        maxiter = min(maxiter, 8000)

        # Multiple sub-passes at this beta with perturbation
        n_subpasses = min(n_passes, max(1, int(maxiter / 500)))
        for sp in range(n_subpasses):
            if time.monotonic() - t_start > total_time - 10:
                break

            sc, a_opt = run_lbfgs(a, beta, maxiter=maxiter // max(1, n_subpasses),
                                   maxfun=(maxiter // max(1, n_subpasses)) * 20)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()

            a = normalize(a_opt)

            # Stochastic perturbation (decreasing over time)
            if sp < n_subpasses - 1:
                progress = (time.monotonic() - t_start) / total_time
                current_pert = pert_scale * (1.0 - 0.7 * progress)
                n_pert = max(10, n // 100)
                pert_idx = rng.integers(0, n, size=n_pert)
                a[pert_idx] += current_pert * rng.standard_normal(n_pert) * a[pert_idx].mean()
                a = normalize(a)

    return best_a, best_sc

            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def interpolate_sequence(a, new_n):
    """Interpolate to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a) + 1)  # Include endpoint
    x_new = np.linspace(0, 1, new_n + 1)
    # Use the sequence values + implicit 0 at boundaries
    a_padded = np.concatenate([a, [0]])
    a_padded = np.concatenate([[0], a_padded])
    x_old = np.linspace(0, 1, len(a_padded))
    a_new = np.interp(x_new, x_old, a_padded)
    return a_new


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer combining novel spectral methods, stochastic LSE
    annealing, peak-pair surgery, and adaptive subgradient.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and len(parent) > 0:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "PARENT")
    else:
        best_a = np.ones(8192) / 8192.0
        best_score = compute_score(best_a)
        print(f"No parent. Starting uniform n=8192, score={best_score:.8f}")

    if best_a is None:
        best_a = np.ones(8192) / 8192.0
        best_score = compute_score(best_a)

    n0 = len(best_a)
    print(f"Starting: n={n0}, score={best_score:.8f}")

    # ==================================================================
    # Phase 1: Spectral equalization (10% time)
    # Novel: peak-aware spectral shaping
    # ==================================================================
    phase1_end = t_start + 0.10 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Spectral Equalization (budget {phase1_end - t_start:.0f}s) ===")

    # Peak-aware spectral
    if time.monotonic() < phase1_end - 30:
        a1, sc1 = peak_aware_spectral(best_a, n_iters=300, strength=0.12)
        update(sc1, a1, "SPEC-PEAK")

    # Top-peak spectral
    if time.monotonic() < phase1_end - 20:
        a2, sc2 = top_peak_spectral(best_a, n_iters=200)
        update(sc2, a2, "SPEC-TOP")

    # Standard flatten
    if time.monotonic() < phase1_end - 10:
        a3, sc3 = spectral_flatten(best_a, n_iters=200)
        update(sc3, a3, "SPEC-FLAT")

    # Combine best spectral result with LSE
    if time.monotonic() < phase1_end - 5:
        sc4, a4 = run_lbfgs(best_a, 500, maxiter=2000, maxfun=40000)
        update(sc4, a4, "SPEC+LSE")

    # ==================================================================
    # Phase 2: Stochastic LSE Annealing (40% time)
    # Fine-grained beta schedule with perturbations
    # ==================================================================
    phase2_end = t_start + 0.50 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Stochastic LSE Annealing (budget {phase2_end - t_start:.0f}s) ===")

    t2_start = time.monotonic()
    time_budget_2 = phase2_end - 10 - t2_start
    a2, sc2 = stochastic_lse_anneal(best_a, rng, total_time=time_budget_2, n_passes=2)
    update(sc2, a2, "STOCH-LSE")

    # Additional pass from the best so far
    if time.monotonic() < phase2_end - 20:
        remaining = phase2_end - 20 - time.monotonic()
        if remaining > 30:
            a2b, sc2b = stochastic_lse_anneal(best_a, rng, total_time=remaining, n_passes=3)
            update(sc2b, a2b, "STOCH-LSE-2")

    # ==================================================================
    # Phase 3: Peak-Pair Surgery (15% time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Peak-Pair Surgery (budget {phase3_end - t_start:.0f}s) ===")

    # Adaptive subgradient
    if time.monotonic() < phase3_end - 30:
        remaining_iters = int((phase3_end - 30 - time.monotonic()) * 1500)
        remaining_iters = max(1000, min(remaining_iters, 10000))
        a3 = adaptive_subgrad(best_a, n_iters=remaining_iters, top_k=8)
        sc3 = compute_score(a3)
        update(sc3, a3, "SUBGRAD")

    # Peak-pair surgery
    if time.monotonic() < phase3_end - 20:
        remaining_iters = int((phase3_end - 20 - time.monotonic()) * 2000)
        remaining_iters = max(500, min(remaining_iters, 8000))
        a3b = peak_pair_surgery(best_a, n_iters=remaining_iters, n_top_lags=5, n_top_pairs=20)
        sc3b = compute_score(a3b)
        update(sc3b, a3b, "SURGERY")

    # Follow up with LSE on surgical result
    if time.monotonic() < phase3_end - 10:
        sc3c, a3c = run_lbfgs(best_a, 20000, maxiter=3000, maxfun=60000)
        update(sc3c, a3c, "SURGERY+LSE")

    # ==================================================================
    # Phase 4: Multi-restart exploration (15% time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Multi-Restart (budget {phase4_end - t_start:.0f}s) ===")

    n_restarts = 10
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 30:
            break
        remaining = phase4_end - 30 - time.monotonic()
        if remaining < 30:
            break

        # Diverse perturbation strategies
        if r % 4 == 0:
            # Multi-frequency perturbation
            n = len(best_a)
            x = np.arange(n, dtype=np.float64) / n
            pert = np.zeros(n)
            for m in range(1, 8):
                freq = rng.integers(1, 30)
                phase = 2.0 * np.pi * rng.uniform(0, 1)
                amp = 0.015 / m
                pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
            a_pert = normalize(best_a * (1.0 + pert))
        elif r % 4 == 1:
            # Gaussian + clip
            a_pert = normalize(best_a + 0.01 * rng.standard_normal(len(best_a)))
        elif r % 4 == 2:
            # Peak-targeted: boost non-peak areas
            a_pert = best_a.copy()
            c = fast_conv(a_pert)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - len(a_pert) + 1)
            j_end = min(k_peak, len(a_pert) - 1)
            peak_mask = np.ones(len(a_pert), dtype=bool)
            peak_mask[j_start:j_end + 1] = False
            non_peak_idx = np.where(peak_mask)[0]
            n_add = max(50, len(a_pert) // 15)
            if len(non_peak_idx) > n_add:
                add_idx = rng.choice(non_peak_idx, n_add, replace=False)
                delta = 0.003 / len(a_pert)
                a_pert[add_idx] += delta
            a_pert = normalize(a_pert)
        else:
            # Spectral perturbation: random phase shift
            N_fft = 1 << (2 * len(best_a) - 1).bit_length()
            A = np.fft.rfft(best_a, N_fft)
            phase_noise = 0.05 * rng.standard_normal(len(A))
            A_new = A * np.exp(1j * phase_noise)
            a_temp = np.fft.irfft(A_new, N_fft)[:len(best_a)]
            a_pert = normalize(a_temp)

        # Optimize the perturbed version
        beta = 1000 + r * 2000
        maxiter = max(500, int(remaining * 5 / (n_restarts - r)))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_opt, f"RESTART-{r}")

        # If improved, do a quick spectral pass
        if sc < best_score and time.monotonic() < phase4_end - 15:
            a_sp, sc_sp = peak_aware_spectral(a_opt, n_iters=100, strength=0.05)
            update(sc_sp, a_sp, f"RESTART-{r}-SPEC")

    # ==================================================================
    # Phase 5: Larger-n exploration (5% time)
    # ==================================================================
    phase5_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 5: Larger-n Exploration ===")

    current_n = len(best_a)
    target_ns = []
    for new_n in [12000, 15000, 20000]:
        if new_n > current_n and time.monotonic() < phase5_end - 40:
            target_ns.append(new_n)

    for new_n in target_ns:
        if time.monotonic() > phase5_end - 40:
            break
        remaining = phase5_end - 40 - time.monotonic()
        if remaining < 40:
            break

        a_large = interpolate_sequence(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  Interpolated to n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        if time.monotonic() < phase5_end - 20:
            maxiter = max(500, int((phase5_end - 20 - time.monotonic()) * 4))
            sc2, a2 = run_lbfgs(a_large, 5000, maxiter=maxiter, maxfun=maxiter * 12)
            update(sc2, a2, f"INTERP+LSE n={new_n}")

    # ==================================================================
    # Phase 6: Final polish (10% time)
    # ==================================================================
    print(f"\n=== Phase 6: Final Polish ===")

    # Very high beta LSE
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        maxiter = max(2000, int(remaining * 6))
        sc, a_fin = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 10)
        update(sc, a_fin, "FIN-LSE-50k")

    if time.monotonic() < deadline - 25:
        maxiter = max(1000, int((deadline - 25 - time.monotonic()) * 5))
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 8)
        update(sc, a_fin, "FIN-LSE-100k")

    # Final subgradient
    if time.monotonic() < deadline - 15:
        remaining_iters = max(500, int((deadline - 15 - time.monotonic()) * 2000))
        a_ps = adaptive_subgrad(best_a, n_iters=min(remaining_iters, 5000), top_k=5)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FIN-SUBGRAD")

    # Final coordinate descent
    if time.monotonic() < deadline - 8:
        a_cd, sc_cd = coord_descent_fine(best_a, n_sweeps=5, n_top=30)
        update(sc_cd, a_cd, "FIN-CD")

    # Final LSE polish on the very best
    if time.monotonic() < deadline - 3:
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=2000, maxfun=20000)
        update(sc, a_fin, "FIN-FINAL")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
