```
def simplex_project(v, target=1.0):
    v=np.asarray(v,float).ravel()
    n=len(v)
    if n==0: return v
    if target <=0: return np.zeros(n)
    u=np.sort(v)[::-1]
    if u.size==0: return np.zeros(n)
    cvu=np.cumsum(u)
    rho=np.nonzero(u * (np.arange(1,n+1)) > (cvu - target))[0]
    if rho.size==0:
       return np.full(n, target/n)
    r=rho[-1]
    theta=(cvu[r]-target)/(r+1.0)
    out=np.maximum(v-theta,0.0)
    s=out.sum()
    if s<=1e-300:
       out=np.full(n,target/n)
    else:
       out=out*target/s
    return out
```

```
def exp_normalize(logv):

    if logv.size==0: return logv
    m=float(np.max(logv))
    if not np.isfinite(m): return np.full(logv.shape,1.0/logv.size)
```

```
def lse_score_and_grad(x,beta):
    x=np.asarray(x,float).ravel()
    x=np.maximum(x,0.0)
    n=x.size
    S=float(x.sum())
    if S < 1e-30:
       return 1e12, np.zeros(n)
    c=self_conv(x)
    mx=float(c.max())
    z=beta*(c-mx)
    zmax=float(z.max())
    ez=np.exp(z-zmax)
    esum=float(ez.sum())
    if esum <= 0 or not np.isfinite(esum):
        p=np.ones(c.size)/c.size
    else:
        p=ez/esum
    logsum = zmax + np.log(esum) if esum>0 else 0.0
    L = mx + logsum/beta
    S2=S*S
    score=2.0*n*L/S2
    corr=cross_corr(x,p)
    grad_L=2.0*corr
    S3=S2*S
    grad=2.0*n*(grad_L/S2 - 2.0*L/S3)
    if not np.all(np.isfinite(grad)):
       grad=np.zeros(n)
    if not np.isfinite(score): score=1e12
    return float(score), grad
```

```python
"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a))/(sum(a)**2)
via a blend of:
  * inverse-square-root flat-convolution seeding,
  * zero-sum multiplicative-weights best-response alternating updates,
  * exact peak subgradient / simplex-projection local search,
  * mirror-descent peak reduction,
  * spectral Newton-style peak peeling,
  * L-BFGS polish on an LSE-smoothed objective.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d

_LBFGS_BETA = 100.0
_LBFGS_DEADLINE = 0.0


def _next_pow2(n):
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def self_conv(a):
    n = int(a.size)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    M = 2 * n - 1
    L = _next_pow2(M)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:M]


def linear_conv(a, b):
    la = int(a.size)
    lb = int(b.size)
    if la <= 0 or lb <= 0:
        return np.zeros(0, dtype=np.float64)
    M = la + lb - 1
    L = _next_pow2(M)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:M]


def cross_corr(a, b):
    """
    Returns res[i] = sum_j a[j] * b[j + i], i = 0..len(a)-1.
    """
    na = int(a.size)
    nb = int(b.size)
    if na <= 0:
        return np.zeros(0, dtype=np.float64)
    if nb <= 0:
        return np.zeros(na, dtype=np.float64)
    L = _next_pow2(na + nb - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(np.conj(A) * B, L)[:na]


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if not np.isfinite(S) or S < 1e-30:
        return np.inf
    c = self_conv(a)
    if c.size == 0:
        return np.inf
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (S * S))


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-30:
        return np.ones(max(1, a.size), dtype=np.float64) / max(1, a.size)
    return a / S


def resample_seq(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return normalize_seq(a)
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, n)
    b = np.interp(new, old, a)
    return normalize_seq(b)


def simplex_project(v, target=1.0):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    u = np.sort(v)[::-1]
    cvu = np.cumsum(u)
    rho = np.nonzero(u * (np.arange(1, n + 1, dtype=np.float64)) > (cvu - target))[0]
    if rho.size == 0:
        return np.full(n, target / n, dtype=np.float64)
    r = int(rho[-1])
    theta = (cvu[r] - target) / float(r + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.full(n, target / n, dtype=np.float64)
    out *= target / s
    return out


def exp_normalize(logv):
    logv = np.asarray(logv, dtype=np.float64)
    n = logv.size
    if n == 0:
        return logv
    m = float(np.max(logv))
    if not np.isfinite(m):
        return np.full(n, 1.0 / n, dtype=np.float64)
    v = np.exp(logv - m)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s


def softmax_c(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.zeros(0, dtype=np.float64), 0.0
    cmax = float(c.max())
    if beta <= 0:
        return np.ones(c.size, dtype=np.float64) / float(c.size), cmax
    z = beta * (c - cmax)
    p = exp_normalize(z)
    return p, cmax


def lse_score_and_grad(x, beta):
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.maximum(x, 0.0)
    n = x.size
    if n == 0:
        return 1e12, np.zeros(0, dtype=np.float64)
    S = float(x.sum())
    if not np.isfinite(S) or S < 1e-30:
        return 1e12, np.zeros(n, dtype=np.float64)

    c = self_conv(x)
    mx = float(c.max())
    if beta <= 0 or not np.isfinite(beta):
        beta = 1.0
    z = beta * (c - mx)
    zmax = float(z.max())
    ez = np.exp(z - zmax)
    esum = float(ez.sum())
    if not np.isfinite(esum) or esum <= 1e-300:
        p = np.ones(c.size, dtype=np.float64) / max(1, c.size)
        logsum = 0.0
    else:
        p = ez / esum
        logsum = zmax + float(np.log(esum))

    L = mx + logsum / beta
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2

    corr = cross_corr(x, p)
    grad_L = 2.0 * corr
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if not np.isfinite(score):
        score = 1e12
    return float(score), grad


def _lbfgs_fun_grad(x):
    return lse_score_and_grad(x, _LBFGS_BETA)


def _lbfgs_cb(k):
    if time.monotonic() > _LBFGS_DEADLINE:
        raise RuntimeError("LBFGS deadline")


def run_lbfgs(a, beta, deadline, maxiter=300, maxfun=3000):
    global _LBFGS_BETA, _LBFGS_DEADLINE

    a0 = np.asarray(a, dtype=np.float64).ravel()
    a0 = np.clip(a0, 0.0, 1000.0)
    a0 = normalize_seq(a0)
    n = a0.size
    if n == 0:
        return np.inf, a0

    old_beta = _LBFGS_BETA
    _LBFGS_BETA = float(beta)
    _LBFGS_DEADLINE = float(max(0.0, deadline))

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _lbfgs_fun_grad,
            a0.copy(),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            callback=_lbfgs_cb,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "ftol": 1e-15,
                "gtol": 1e-12,
                "maxls": 40,
            },
        )
        x = res.x
    except Exception:
        x = a0
    finally:
        _LBFGS_BETA = old_beta

    x = np.clip(np.asarray(x, dtype=np.float64).ravel(), 0.0, 1000.0)
    x = normalize_seq(x)
    sc = exact_score(x)
    return sc, x


def invsqrt_sequence(n):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    return c / s


def power_sequence(n, alpha, taper=0.0):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-alpha)
    if taper != 0.0:
        t = np.linspace(0.0, 1.0, n)
        a *= np.power(1.0 - 0.85 * t + 1e-3, taper)
    return normalize_seq(a)


def tapered_inv_sqrt_sequence(n, gamma):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    i = np.arange(1, n + 1, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (np.sqrt(i) * np.power(1.0 - 0.85 * t + 1e-3, gamma))
    return normalize_seq(a)


def sym_inv_sqrt_sequence(n):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    u = np.minimum(idx, float(n - 1) - idx)
    a = 1.0 / np.sqrt(u + 1.0)
    return normalize_seq(a)


def thue_sequence(n, low=0.25, high=0.75):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    tm = np.zeros(n, dtype=np.int8)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    a = np.where(tm == 1, high, low).astype(np.float64)
    return normalize_seq(a)


def build_inits(n, rng, parent):
    inits = []
    if n <= 0:
        return inits

    if parent is not None:
        inits.append(resample_seq(parent, n))

    base = invsqrt_sequence(n)
    inits.append(base.copy())
    inits.append(base[::-1].copy())
    inits.append(power_sequence(n, 0.45))
    inits.append(power_sequence(n, 0.55))
    inits.append(power_sequence(n, 0.50, taper=1.0))
    inits.append(tapered_inv_sqrt_sequence(n, 1.0))
    inits.append(sym_inv_sqrt_sequence(n))

    x = np.linspace(0.0, 1.0, n)
    inits.append(normalize_seq(0.5 + 0.5 * np.cos(2.0 * np.pi * x)))
    inits.append(normalize_seq(0.5 - 0.5 * np.cos(2.0 * np.pi * x)))
    inits.append(normalize_seq(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(7.0 * np.pi * x)))

    for _ in range(4):
        w = 0.35 * rng.standard_normal(n)
        w = uniform_filter1d(w, size=max(3, n // 60))
        a = base * (1.0 + 0.45 * w)
        inits.append(normalize_seq(a))

    inits.append(thue_sequence(n, 0.20, 0.80))

    out = []
    for a in inits:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == n and np.all(np.isfinite(a)) and a.sum() > 1e-30:
            out.append(normalize_seq(a))
    return out


def mirror_desc(a, beta, n_iter, eta, deadline=None):
    x = normalize_seq(a)
    n = x.size
    if n == 0:
        return x, np.inf

    c = self_conv(x)
    best_sc = 2.0 * n * float(c.max())
    best_x = x.copy()
    stale = 0

    for _ in range(int(max(0, n_iter))):
        if deadline is not None and time.monotonic() > deadline:
            break
        p, cmax = softmax_c(c, beta)
        if cmax <= 1e-300:
            break
        h = cross_corr(x, p)
        maxh = float(h.max())
        if not np.isfinite(maxh) or maxh <= 1e-15:
            break
        logx = np.log(x + 1e-300) - eta * (h / maxh)
        x_new = exp_normalize(logx)
        c2 = self_conv(x_new)
        sc2 = 2.0 * n * float(c2.max())
        if np.isfinite(sc2) and sc2 < best_sc:
            best_sc = sc2
            best_x = x_new.copy()
            stale = 0
        else:
            stale += 1
        x = x_new
        c = c2
        if stale > 80:
            break

    return best_x, best_sc


def active_peak_pg(a, K, beta, n_iter, deadline=None):
    x = normalize_seq(a)
    n = x.size
    if n == 0:
        return x, np.inf

    m = 2 * n - 1
    c = self_conv(x)
    best_sc = 2.0 * n * float(c.max())
    best_x = x.copy()
    stale = 0

    for _ in range(int(max(0, n_iter))):
        if deadline is not None and time.monotonic() > deadline:
            break

        if beta is None or beta <= 0:
            if K >= m:
                p = np.ones(m, dtype=np.float64) / float(m)
            elif K <= 0:
                p = np.ones(m, dtype=np.float64) / float(m)
            else:
                p = np.zeros(m, dtype=np.float64)
                idx = np.argpartition(c, -K)[-K:]
                p[idx] = 1.0 / float(K)
        else:
            p, _ = softmax_c(c, beta)

        grad = 2.0 * cross_corr(x, p)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm <= 1e-15:
            break

        maxc = float(c.max())
        step_base = min(10.0, max(0.01, 0.5 * maxc / (gnorm * gnorm + 1e-30)))
        accepted = False

        for mult in (1.0, 0.5, 0.2, 0.05, 0.02, 2.0):
            step = step_base * mult
            if step <= 0:
                continue
            x_try = simplex_project(x - step * grad)
            c2 = self_conv(x_try)
            sc2 = 2.0 * n * float(c2.max())
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_x = x_try.copy()
                x = x_try
                c = c2
                accepted = True
                break

        if not accepted:
            stale += 1
            if stale > 60:
                break

    return best_x, best_sc


def spectral_peel(a, n_iter, red, damp, deadline=None):
    x = normalize_seq(a)
    n = x.size
    if n == 0:
        return x, np.inf

    M = 2 * n - 1
    L = _next_pow2(M)
    c = self_conv(x)
    best_sc = 2.0 * n * float(c.max())
    best_x = x.copy()
    stale = 0

    for _ in range(int(max(0, n_iter))):
        if deadline is not None and time.monotonic() > deadline:
            break

        maxc = float(c.max())
        if maxc <= 1e-300:
            break
        thr = maxc * (1.0 - red)
        excess = np.maximum(c - thr, 0.0)
        if float(excess.max()) < 1e-15:
            break

        target = c - red * excess
        target = np.maximum(target, 0.0)

        A = np.fft.rfft(x, L)
        C = A * A
        Ct = np.fft.rfft(target, L)
        D = np.zeros_like(A)
        den = 2.0 * A
        mask = np.abs(den) > 1e-14
        D[mask] = (Ct - C)[mask] / den[mask]

        accepted = False
        for d in (damp, 0.5 * damp, 0.2 * damp, 0.05 * damp):
            if d <= 0:
                continue
            Anew = A + d * D
            xnew = np.fft.irfft(Anew, L)[:n]
            xnew = np.maximum(xnew, 0.0)
            xnew = normalize_seq(xnew)
            c2 = self_conv(xnew)
            sc2 = 2.0 * n * float(c2.max())
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_x = xnew.copy()
                x = xnew
                c = c2
                accepted = True
                break

        if not accepted:
            stale += 1
            if stale > 40:
                break

    return best_x, best_sc


def mw_best_response(a, iters, eta, r0=None, q0=None, deadline=None):
    n = int(a.size)
    m = 2 * n - 1
    if n == 0:
        return np.zeros(0, dtype=np.float64), np.zeros(0, dtype=np.float64), np.zeros(0, dtype=np.float64)

    a_pay = normalize_seq(a) * float(n)

    if r0 is None or r0.size != n:
        r = np.ones(n, dtype=np.float64) / float(n)
    else:
        r = normalize_seq(r0)
    if q0 is None or q0.size != m:
        q = np.ones(m, dtype=np.float64) / float(m)
    else:
        q = normalize_seq(q0)

    log_r = np.log(np.maximum(r, 1e-300))
    log_q = np.log(np.maximum(q, 1e-300))

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(n + m - 1)
    A1 = np.fft.rfft(a_pay, L1)
    A2 = np.fft.rfft(a_pay, L2)

    burn = max(1, int(iters) // 10)
    r_acc = np.zeros(n, dtype=np.float64)
    count = 0

    for it in range(int(max(0, iters))):
        if deadline is not None and time.monotonic() > deadline:
            break

        RB = np.fft.rfft(r, L1)
        c = np.fft.irfft(A1 * RB, L1)[:m]
        QB = np.fft.rfft(q, L2)
        h = np.fft.irfft(np.conj(A2) * QB, L2)[:n]

        c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
        h = np.maximum(np.asarray(h, dtype=np.float64), 0.0)

        scale = max(1.0, float(c.max()), float(h.max()))
        if not np.isfinite(scale) or scale <= 0:
            break
        c = c / scale
        h = h / scale

        log_r = log_r - eta * h
        r = exp_normalize(log_r)

        log_q = log_q + eta * c
        q = exp_normalize(log_q)

        log_r = np.log(np.maximum(r, 1e-300))
        log_q = np.log(np.maximum(q, 1e-300))

        if it >= burn:
            r_acc += r
            count += 1

    if count > 0:
        g = r_acc / float(count)
    else:
        g = r
    g = normalize_seq(g)
    return g, r, q


def mix_line_search(a, g):
    a = normalize_seq(a)
    g = normalize_seq(g)
    best_a = a.copy()
    best_sc = exact_score(a)
    best_t = 0.0

    for t in (0.0, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
        x = (1.0 - t) * a + t * g
        x = normalize_seq(x)
        sc = exact_score(x)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = x.copy()
            best_t = float(t)

    step = 0.2
    for _ in range(5):
        improved = False
        for tt in (best_t - step, best_t + step):
            if 0.0 <= tt <= 1.0:
                x = (1.0 - tt) * a + tt * g
                x = normalize_seq(x)
                sc = exact_score(x)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = x.copy()
                    best_t = float(tt)
                    improved = True
        if not improved:
            step *= 0.5
            if step < 0.005:
                break

    return best_a, best_sc


def random_perturb(a, rng, scale=0.1):
    a = normalize_seq(a)
    x = a + scale * a * rng.standard_normal(a.shape)
    x = np.maximum(x, 1e-12)
    return normalize_seq(x)


def insert_pool(pool, sc, a, limit=10):
    sc = float(sc)
    if not np.isfinite(sc):
        return
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-30:
        return
    a = a / S

    if len(pool) < int(limit):
        pool.append((sc, a.copy()))
        return

    worst_i = 0
    worst_sc = pool[0][0]
    for i in range(1, len(pool)):
        if pool[i][0] > worst_sc:
            worst_sc = pool[i][0]
            worst_i = i
    if sc < worst_sc:
        pool[worst_i] = (sc, a.copy())


def pool_best(pool):
    best_sc = np.inf
    best_a = None
    for sc, a in pool:
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a
    return best_sc, best_a


def top_pool(pool, k):
    out = []
    k = int(k)
    if k <= 0:
        return out
    for item in pool:
        if len(out) < k:
            out.append(item)
        else:
            worst_i = 0
            worst_sc = out[0][0]
            for i in range(1, len(out)):
                if out[i][0] > worst_sc:
                    worst_sc = out[i][0]
                    worst_i = i
            if item[0] < worst_sc:
                out[worst_i] = item
    return out


def propose_candidate(budget_s=2400):
    t0 = time.monotonic()
    total = float(max(60.0, budget_s))
    work = max(30.0, total - 40.0)
    deadline = t0 + work
    rng = np.random.default_rng(20240528)

    pool = []

    parent = None
    try:
        hg = globals().get("height_sequence_1")
        if hg is not None:
            p = np.asarray(hg, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            if p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-12:
                p = p / float(p.sum())
                parent = p
    except Exception:
        parent = None

    if parent is not None:
        ps = exact_score(parent)
        insert_pool(pool, ps, parent, limit=12)
        print(f"parent len={parent.size} score={ps:.10f}")

    base_len = parent.size if parent is not None else 6962

    length_set = []
    for n in (
        base_len,
        6962,
        7168,
        7424,
        7680,
        8192,
        int(base_len * 1.10),
        int(base_len * 1.25),
    ):
        n = int(n)
        if 512 <= n <= 12000 and n not in length_set:
            length_set.append(n)
    lengths = sorted(length_set)[:6]
    print("Lengths:", lengths)

    # Phase 0: seed and cheap global moves.
    p0_deadline = t0 + 0.07 * work
    print("Phase 0: seeding")
    for n in lengths:
        if time.monotonic() > p0_deadline:
            break
        inits = build_inits(n, rng, parent)
        n_best_a = None
        n_best_sc = np.inf
        for a0 in inits:
            sc = exact_score(a0)
            insert_pool(pool, sc, a0, limit=12)
            if np.isfinite(sc) and sc < n_best_sc:
                n_best_sc = sc
                n_best_a = a0
        if n_best_a is not None:
            a1, sc1 = mirror_desc(n_best_a, beta=1000, n_iter=150, eta=1.0, deadline=p0_deadline)
            insert_pool(pool, sc1, a1, limit=12)
            a2, sc2 = spectral_peel(n_best_a, n_iter=45, red=0.40, damp=0.25, deadline=p0_deadline)
            insert_pool(pool, sc2, a2, limit=12)

    best_sc, best_a = pool_best(pool)
    print(f"After phase 0: best={best_sc:.10f}")

    # Phase 1: exact peak-driven local search on the best seeds.
    p1_deadline = t0 + 0.28 * work
    print("Phase 1: peak local search")
    for _, a0 in top_pool(pool, 3):
        if time.monotonic() > p1_deadline:
            break
        a1, s1 = active_peak_pg(a0, K=4, beta=None, n_iter=250, deadline=p1_deadline)
        insert_pool(pool, s1, a1, limit=12)
        a2, s2 = mirror_desc(a1, beta=5000, n_iter=250, eta=1.0, deadline=p1_deadline)
        insert_pool(pool, s2, a2, limit=12)
        a3, s3 = mirror_desc(a2, beta=12000, n_iter=250, eta=3.0, deadline=p1_deadline)
        insert_pool(pool, s3, a3, limit=12)
        a4, s4 = spectral_peel(a3, n_iter=70, red=0.35, damp=0.25, deadline=p1_deadline)
        insert_pool(pool, s4, a4, limit=12)

    best_sc, best_a = pool_best(pool)
    print(f"After phase 1: best={best_sc:.10f}")

    # Phase 2: zero-sum multiplicative-weights best-response alternating.
    p2_deadline = t0 + 0.62 * work
    print("Phase 2: MW best-response alternating")
    best_sc, best_a = pool_best(pool)
    current = normalize_seq(best_a)
    current_sc = exact_score(current)
    r_warm = None
    q_warm = None
    no_improve = 0
    round_idx = 0

    while time.monotonic() < p2_deadline:
        ps, pa = pool_best(pool)
        if np.isfinite(ps) and ps < current_sc - 1e-12:
            current = normalize_seq(pa)
            current_sc = exact_score(current)

        eta = 0.20 + 0.15 * round_idx
        if eta > 2.0:
            eta = 0.5 + float(rng.random()) * 1.5

        if current.size < 8500:
            iters = 1200
        else:
            iters = 800
        slab_deadline = min(p2_deadline, time.monotonic() + 45.0)

        r0 = r_warm if (r_warm is not None and r_warm.size == current.size) else None
        q0 = q_warm if (q_warm is not None and q_warm.size == 2 * current.size - 1) else None
        g, r_final, q_final = mw_best_response(
            current,
            iters=iters,
            eta=eta,
            r0=r0,
            q0=q0,
            deadline=slab_deadline,
        )
        new, new_sc = mix_line_search(current, g)
        insert_pool(pool, new_sc, new, limit=12)

        if np.isfinite(new_sc) and new_sc < current_sc - 1e-12:
            current = new.copy()
            current_sc = new_sc
            if r_final.size == current.size:
                r_warm = r_final.copy()
            if q_final.size == 2 * current.size - 1:
                q_warm = q_final.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3 and len(pool) > 1:
                items = top_pool(pool, min(3, len(pool)))
                cand = items[1][1] if len(items) > 1 else current
                current = random_perturb(normalize_seq(cand), rng, scale=0.10)
                current_sc = exact_score(current)
                no_improve = 1
            if no_improve >= 6:
                break

        round_idx += 1

    best_sc, best_a = pool_best(pool)
    print(f"After phase 2: best={best_sc:.10f}")

    # Phase 3: LSE/L-BFGS polishing.
    p3_deadline = t0 + 0.82 * work
    print("Phase 3: L-BFGS LSE polishing")
    best_sc, best_a = pool_best(pool)
    a_work = normalize_seq(best_a)
    for beta in (300, 800, 2000, 5000, 10000):
        if time.monotonic() > p3_deadline - 60:
            break
        if a_work.size < 8000:
            maxiter = 450
            maxfun = 7000
        else:
            maxiter = 300
            maxfun = 4500
        sc, a_opt = run_lbfgs(a_work, beta=beta, deadline=p3_deadline - 30,
                              maxiter=maxiter, maxfun=maxfun)
        if np.isfinite(sc):
            insert_pool(pool, sc, a_opt, limit=12)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt
                a_work = normalize_seq(a_opt)
            else:
                a_work = normalize_seq(best_a)
        else:
            break

    best_sc, best_a = pool_best(pool)
    print(f"After phase 3: best={best_sc:.10f}")

    # Phase 4: final stabbing at the remaining peak.
    p4_deadline = deadline - 30
    print("Phase 4: final polish")
    best_sc, best_a = pool_best(pool)

    a1, s1 = active_peak_pg(best_a, K=2, beta=None, n_iter=250, deadline=p4_deadline)
    insert_pool(pool, s1, a1, limit=12)
    a2, s2 = active_peak_pg(a1, K=16, beta=2000, n_iter=180, deadline=p4_deadline)
    insert_pool(pool, s2, a2, limit=12)
    a3, s3 = mirror_desc(a2, beta=20000, n_iter=250, eta=0.6, deadline=p4_deadline)
    insert_pool(pool, s3, a3, limit=12)
    a4, s4 = spectral_peel(a3, n_iter=70, red=0.25, damp=0.20, deadline=p4_deadline)
    insert_pool(pool, s4, a4, limit=12)

    if time.monotonic() < p4_deadline - 25:
        best_sc, best_a = pool_best(pool)
        g, _, _ = mw_best_response(best_a, iters=1000, eta=1.0, r0=None, q0=None,
                                   deadline=p4_deadline - 15)
        new, new_sc = mix_line_search(best_a, g)
        insert_pool(pool, new_sc, new, limit=12)

    best_sc, best_a = pool_best(pool)
    if best_a is None or not np.all(np.isfinite(best_a)) or best_a.sum() <= 1e-12:
        best_a = invsqrt_sequence(6962)
        best_sc = exact_score(best_a)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if len(out) == 0 or sum(out) < 0.01:
        out = [0.5] * 64

    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = np.inf

    if not np.isfinite(verified):
        for fb in (parent, invsqrt_sequence(max(1, len(out)))):
            if fb is None:
                continue
            cand = [float(max(0.0, min(1000.0, x))) for x in normalize_seq(fb)]
            if len(cand) == 0 or sum(cand) < 0.01:
                continue
            try:
                v2 = float(evaluate_sequence(cand))
                if np.isfinite(v2):
                    out = cand
                    verified = v2
                    break
            except Exception:
                pass

    print(f"FINAL internal best={best_sc:.10f} verified={verified:.10f} len={len(out)}")
    return out
```
