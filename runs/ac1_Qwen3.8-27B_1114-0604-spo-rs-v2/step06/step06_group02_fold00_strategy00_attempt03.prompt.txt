def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

        active_peaks = np.where(c >= 0.99 * c_max)[0]

                continue
            js = np.arange(j_start, j_end + 1)

            total = contribs.sum()
            if total < 1e-15:
                continue

            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global[:50]:  # Further limit
                a_old = a[j]
                if a_old < 1e-12:
                    continue
                # Try several multipliers
                best_val = a_old
                best_local = sc
                for f in [0.6, 0.75, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2]:
                    cv = a_old * f
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < best_local:
                        best_local = sc_tmp
                        best_val = cv

def alternating_projection(a, n_iters=2000, decay=0.999):
    """
    Alternating projections between:
    - Time domain: non-negativity + normalization
    - Frequency domain: spectral flattening with adaptive strength
    With "repair" smoothing in time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    strength = 0.6

    for it in range(n_iters):
        # Frequency domain step: flatten spectrum
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_max = P.max()
        P_min = P[P > 1e-30].min() if (P > 1e-30).any() else 1e-30

        # Adaptive flattening: stronger for high peaks, preserve low values
        threshold = P_max * (0.5 + 0.3 * (it / n_iters))
        P_mod = P.copy()
        above = P > threshold
        if above.any():
            P_mod[above] = threshold * (1.0 + 0.5 * (P[above] - threshold) / (P_max - threshold + 1e-30))
        P_mod = np.maximum(P_mod, 1e-20)

        scale = np.sqrt(P_mod / (P + 1e-30))
        A_new = A * scale ** strength
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Time domain step: non-negativity + smoothing + normalization
        a_new = np.maximum(a_new, 0.0)
        # Light Gaussian smoothing to repair artifacts
        if n > 8:
            kernel = np.exp(-0.5 * (np.arange(-3, 4) / 2.0) ** 2)
            kernel /= kernel.sum()
            a_smooth = np.convolve(a_new, kernel, mode='same')
            a_new = 0.7 * a_new + 0.3 * a_smooth
            a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Reduce strength if not improving
            strength *= 0.995

        strength = max(strength, 0.05)

def try_different_length(parent, target_n, rng):
    """Interpolate parent to a different length and re-optimize briefly."""
    if len(parent) == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a

def symmetry_exploit(a):
    """
    If the sequence is close to symmetric, enforce symmetry and optimize
    only the first half. This can help escape local minima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    # Check symmetry
    a_sym = 0.5 * (a + a[::-1])
    sym_error = np.linalg.norm(a - a_sym) / (np.linalg.norm(a) + 1e-30)

    if sym_error < 0.3:  # Close enough to symmetric
        # Enforce symmetry and optimize the half
        half = (n + 1) // 2
        x = a_sym[:half].copy()
        x = np.maximum(x, 0.0)
        x = x / x.sum()

        best_x = x.copy()
        best_sc = compute_score(_expand_symmetric(x, n))

        # Gradient descent on x
        for it in range(2000):
            a_full = _expand_symmetric(x, n)
            S = a_full.sum()
            if S < 1e-12:
                break
            a_full = a_full / S
            c = fast_conv(a_full)
            c_max = c.max()
            sc = 2.0 * n * c_max / (S * S)

            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()

            # LSE gradient for the full sequence
            beta = 500
            scaled = beta * (c - c_max)
            scaled = np.clip(scaled, -500, 500)
            exp_s = np.exp(scaled)
            p = exp_s / exp_s.sum()
            L = c_max + np.log(exp_s.sum()) / beta

            # Gradient w.r.t. full a
            S2 = S * S
            S3 = S2 * S
            a_rev = a_full[::-1]
            M_fft = 1 << (2 * n - 1).bit_length()
            PA = np.fft.rfft(p, M_fft)
            AA = np.fft.rfft(a_rev, M_fft)
            h = np.fft.irfft(AA * PA, M_fft)[:2 * n - 1][n - 1:2 * n - 1]
            grad_full = 2.0 * n * (h / S2 - 2.0 * L / S3)

            # Project gradient onto symmetric subspace
            grad_sym = 0.5 * (grad_full[:half] + grad_full[:half][::-1])

            # Step
            step = 0.01 / (np.linalg.norm(grad_sym) + 1e-10)
            x_new = x - step * grad_sym
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new

        best_a = _expand_symmetric(best_x, n)
        best_a = best_a / best_a.sum()
        return best_a
    else:
        return a


def _expand_symmetric(x, n):
    """Expand a symmetric half to full sequence."""
    half = (n + 1) // 2
    a = np.zeros(n)
    a[:half] = x
    a[half:] = x[:n - half][::-1]
    return a
