"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key new ideas:
1. LP-based reweighting: solve LP to find optimal descent direction
2. Mirror descent (multiplicative weights) on the simplex
3. GS with Wiener-optimal target
4. Adaptive LSE-LBFGS with restart
"""

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

def lp_reweight(a, M, peak_indices, max_time=30):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for k in peak_indices, g >= 0"""
    n = len(a)
    # Build constraint matrix
    # For each peak index k, constraint: sum_i a_{k-i} * g_i <= M
    # where a_{k-i} is valid (0 <= k-i < n)
    
    # Use dense matrix for simplicity (n=6962, K peaks)
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # g_j is multiplied by a_{k-j}
            A_mat[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    
    g_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [A_mat @ g_var <= np.full(K, M)]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try solvers in order of preference
        for solver in [cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    return g_var.value
            except Exception:
                continue
        prob.solve()
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return g_var.value
    except Exception:
        pass
    
    return None

"""
Hybrid optimizer for minimizing 2n*max(self_conv(a))/sum(a)^2.
Novel approaches: LP-based reweighting, multiplicative weights (mirror descent),
Wiener-filtered GS targets, and adaptive multi-restart LSE-LBFGS.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score computation."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    shifted = scaled - max_s
    exp_s = np.exp(shifted)
    log_sum = max_s + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def gs_wiener_target(a, n_iters=300, alpha_init=0.5, alpha_final=0.05):
    """GS with Wiener-filtered target for optimal spectral shaping."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c[:2 * n - 1].mean())

        # Wiener target: balance between flat and current shape
        # Target power spectrum should be uniform (white noise property)
        target_conv = np.zeros(M)
        target_conv[:2 * n - 1] = c_mean * np.ones(2 * n - 1)
        # Add a small peak at center to respect c_0 >= sum(a^2)
        c0_min = float(np.dot(a, a))
        target_conv[0] = max(target_conv[0], c0_min * 1.01)
        target_conv[n - 1] = max(target_conv[n - 1], c0_min * 1.01)

        T = np.fft.fft(target_conv)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def multiplicative_weights(a, n_iters=5000, eta_init=0.02, eta_decay=0.9995):
    """Mirror descent with entropy regularization (multiplicative weights)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find top peaks for subgradient
        n_peaks = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute subgradient (average of partner sequences at peaks)
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partner = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += partner
        sg /= len(top_idx)

        # Multiplicative update: a_i <- a_i * exp(-eta * sg_i / S)
        # This is mirror descent with KL divergence
        update = np.exp(-eta * sg / S)
        a_new = a * update
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        eta *= eta_decay

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def lp_reweight_direction(a, M_val, peak_indices):
    """
    Solve LP dual: min M*sum(y) s.t. sum_k y_k * a_{k-i} >= 1 for all i, y >= 0.
    This gives a certificate that the peak cannot be reduced further in a certain direction.
    We use the dual variables to construct a descent direction.
    
    Returns a direction g that reduces the peak convolution.
    """
    n = len(a)
    K = len(peak_indices)
    if K == 0:
        return None

    # Build the constraint matrix for the dual
    # Dual constraint i: sum_{k in peaks} y_k * a_{k-i} >= 1
    # This is a K-variable LP with n constraints
    # Use scipy.optimize.linprog

    # A_ub @ y <= b_ub form: -sum_k y_k * a_{k-i} <= -1 for each i
    # Build sparse-like rows
    A_rows = []
    for i in range(n):
        row = np.zeros(K)
        count = 0
        for idx, k in enumerate(peak_indices):
            k = int(k)
            j = k - i
            if 0 <= j < n:
                row[idx] = -a[j]
                count += 1
        if count > 0:
            A_rows.append(row)

    if not A_rows:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.full(len(A_rows), -1.0)

    # Objective: minimize M * sum(y), i.e., c = M * ones(K)
    c_obj = M_val * np.ones(K)

    from scipy.optimize import linprog
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * K,
                      method='highs')
        if res.success:
            y_opt = res.x
            # Complementary slackness: g_i > 0 only where dual constraint is tight
            # g_i = sum_k y_k * a_{k-i} - 1 (slack)
            # Actually, use the primal solution via sensitivity
            # For now, use a heuristic direction based on the dual
            # The dual tells us which constraints are binding
            slacks = A_ub @ y_opt + 1.0  # = -(dual_lhs) + rhs, should be <= 0
            # g direction: emphasize positions where the constraint is tight
            g = np.zeros(n)
            for idx, k in enumerate(peak_indices):
                k = int(k)
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                if j_end >= j_start:
                    g[j_start:j_end + 1] += y_opt[idx] * a[k - j_end:k - j_start + 1][::-1]
            g = np.maximum(g, 0.0)
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return None


def active_set_lp_direction(a, M_val, n_peaks=100, n_iters=50):
    """
    Active set approach: iteratively add peak constraints and solve reduced LP.
    Returns a direction g that should reduce the peak.
    """
    n = len(a)
    c = fast_conv(a)

    # Start with top peaks
    peak_idx = list(np.argsort(c)[::-1][:n_peaks])

    for it in range(n_iters):
        g = lp_reweight_direction(a, M_val, peak_idx)
        if g is None:
            break

        # Try the direction: a_new = (1-t)*a + t*S*g
        S = float(a.sum())
        best_t = 0.0
        best_sc = compute_score(a)

        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = (1 - t) * a + t * S * g
            a_try = np.maximum(a_try, 0.0)
            if a_try.sum() < 1e-12:
                continue
            a_try = a_try / a_try.sum() * S
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            # Add next peak
            if len(peak_idx) < 2 * n - 1:
                all_idx = np.argsort(c)[::-1]
                new_peak = [i for i in all_idx if i not in peak_idx]
                if new_peak:
                    peak_idx.append(int(new_peak[0]))
                    c = fast_conv(a)
            else:
                break
        else:
            S_save = float(a.sum())
            a = (1 - best_t) * a + best_t * S_save * g
            a = np.maximum(a, 0.0)
            break

    return a if it > 0 and best_t > 0 else None


def peak_reducing_step(a, n_rounds=200, top_k=30):
    """Directly reduce top peaks by adjusting contributing elements with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:top_k]

        # For each peak, compute the "partner" and reduce top contributors
        reduction = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            # Proportional reduction
            reduction[j_start:j_end + 1] += contrib

        # Apply reduction: a_i *= (1 - lambda * reduction_i / max_reduction)
        max_red = float(reduction.max())
        if max_red < 1e-20:
            break

        # Try multiple step sizes
        best_t = best_sc
        best_a_new = a.copy()
        for lam in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            scale = 1.0 - lam * reduction / max_red
            scale = np.maximum(scale, 0.0)
            a_new = a * scale
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            sc = compute_score(a_new)
            if sc < best_t:
                best_t = sc
                best_a_new = a_new.copy()

        if best_t < best_sc:
            best_sc = best_t
            best_a = best_a_new.copy()
            a = best_a_new.copy()

    return best_a


def adaptive_lbfgs_schedule(a, maxiter=8000, maxfun=30000):
    """LSE-LBFGS with adaptive increasing beta."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    if a.sum() < 1e-6:
        a[:] = 1.0 / n

    best_a = a.copy()
    best_sc = compute_score(a)

    # Start with moderate beta, increase over time
    beta_schedule = [500, 2000, 10000, 50000, 200000, 500000]
    for beta in beta_schedule:
        a_opt = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a = a_opt.copy()
        else:
            # Small improvement or none - try from best with perturbation
            a = best_a.copy()

    return best_a


def random_perturb(a0, rng, strength=0.03):
    """Add Gaussian perturbation and renormalize."""
    a = a0 + rng.standard_normal(len(a0)) * strength
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a0.copy()
    return a / S


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) * np.cos(2.0 * np.pi * x), 0.01))

    # Fourier-based
    for _ in range(3):
        n_harm = rng.integers(3, 12)
        weights = rng.standard_normal(n_harm)
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        sig = np.zeros(n)
        for i in range(n_harm):
            sig += weights[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
        inits.append(np.maximum(0.5 + 0.3 * sig, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses LP reweighting, multiplicative weights, GS, and adaptive LSE-LBFGS.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None and abs(len(parent) - n) < 200:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    print(f"Starting with n={n}")
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f}")

    # Also check uniform
    reg_score = compute_score(np.ones(n) * 0.5)
    print(f"Uniform score: {reg_score:.8f}")
    if reg_score < best_score:
        best_score = reg_score
        best_a = np.ones(n) * 0.5

    print(f"Initial best: {best_score:.8f}")

    # ==========================
    # Phase 1: GS warmup (8%)
    # ==========================
    phase1_end = t_start + 0.08 * (total_budget - 30.0)
    print(f"\n=== Phase 1: GS warmup ===")

    inits = build_inits(n, rng, parent if parent is not None and len(parent) == n else None)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / max(a0.sum(), 1e-12)

        a1 = gs_flat_target(a0, n_iters=200, alpha_init=0.4, alpha_final=0.05)
        sc = compute_score(a1)
        update_best(sc, a1)

        if time.monotonic() < phase1_end:
            a2 = gs_wiener_target(a1, n_iters=200, alpha_init=0.3, alpha_final=0.03)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==========================
    # Phase 2: Multiplicative Weights (15%)
    # ==========================
    phase2_end = t_start + 0.23 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Multiplicative Weights ===")

    a_mw = multiplicative_weights(best_a, n_iters=10000, eta_init=0.01, eta_decay=0.9998)
    sc = compute_score(a_mw)
    update_best(sc, a_mw)

    # Also try from perturbed
    if time.monotonic() < phase2_end:
        a_pert = random_perturb(best_a, rng, strength=0.05)
        a_mw2 = multiplicative_weights(a_pert, n_iters=5000, eta_init=0.02, eta_decay=0.9995)
        sc2 = compute_score(a_mw2)
        update_best(sc2, a_mw2)

    # GS after MW
    if time.monotonic() < phase2_end:
        a_gs = gs_wiener_target(best_a, n_iters=300, alpha_init=0.3, alpha_final=0.02)
        sc3 = compute_score(a_gs)
        update_best(sc3, a_gs)

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==========================
    # Phase 3: Adaptive LSE-LBFGS (30%)
    # ==========================
    phase3_end = t_start + 0.53 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Adaptive LSE-LBFGS ===")

    a_adapt = adaptive_lbfgs_schedule(best_a, maxiter=5000, maxfun=20000)
    sc = compute_score(a_adapt)
    update_best(sc, a_adapt)

    # Continue with more aggressive betas
    if time.monotonic() < phase3_end:
        for beta in [100000, 300000, 1000000, 3000000]:
            if time.monotonic() > phase3_end:
                break
            a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=10000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

    # Try from perturbed versions
    if time.monotonic() < phase3_end:
        for _ in range(3):
            if time.monotonic() > phase3_end:
                break
            a_pert = random_perturb(best_a, rng, strength=0.02)
            a_opt = run_lbfgs(a_pert, 100000, maxiter=3000, maxfun=10000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==========================
    # Phase 4: Peak reduction + GS refinement (20%)
    # ==========================
    phase4_end = t_start + 0.73 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Peak reduction + GS ===")

    # Peak reducing steps
    a_peak = peak_reducing_step(best_a, n_rounds=200, top_k=20)
    sc = compute_score(a_peak)
    update_best(sc, a_peak)

    # GS on the result
    if time.monotonic() < phase4_end:
        a_gs = gs_wiener_target(a_peak, n_iters=300, alpha_init=0.2, alpha_final=0.02)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)

    # More peak reduction
    if time.monotonic() < phase4_end:
        a_peak2 = peak_reducing_step(best_a, n_rounds=300, top_k=50)
        sc = compute_score(a_peak2)
        update_best(sc, a_peak2)

    # MW on the refined version
    if time.monotonic() < phase4_end:
        a_mw3 = multiplicative_weights(best_a, n_iters=5000, eta_init=0.005, eta_decay=0.9999)
        sc = compute_score(a_mw3)
        update_best(sc, a_mw3)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==========================
    # Phase 5: Final polish with interleaved methods (29%)
    # ==========================
    print(f"\n=== Phase 5: Final polish ===")

    polish_rounds = 8
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # LSE fine-tune with very high beta
        beta_f = 500000 + pr * 500000
        a_opt = run_lbfgs(best_a, beta_f, maxiter=3000, maxfun=10000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Quick GS
        if time.monotonic() > deadline:
            break
        a_gs = gs_flat_target(best_a, n_iters=100, alpha_init=0.1, alpha_final=0.01)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)

        # Quick MW
        if time.monotonic() > deadline:
            break
        a_mw = multiplicative_weights(best_a, n_iters=2000, eta_init=0.003, eta_decay=0.9999)
        sc = compute_score(a_mw)
        update_best(sc, a_mw)

        # Peak reduction
        if time.monotonic() > deadline:
            break
        a_pk = peak_reducing_step(best_a, n_rounds=100, top_k=20)
        sc = compute_score(a_pk)
        update_best(sc, a_pk)

        print(f"  Polish {pr}: best = {best_score:.8f}")

    # ==========================
    # Final answer
    # ==========================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
