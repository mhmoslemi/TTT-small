"""
LP-based and Fienup hybrid optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key innovation: Linear Programming to find optimal descent directions for the
non-smooth max-autocorrelation objective, combined with spectral methods.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

    3. Take a step with line search
    4. Update active set

        # Variables: delta[0..n-1]

        # Constraints: for each active lag k:
        #   sum_j a[k-j] * delta[j] <= (M - c[k]) / 2 + epsilon
        #   (ensures c[k] doesn't increase above M)
        # Equality: sum(delta) = 0
        # Bounds: delta[i] >= -a[i] (ensures a[i] + delta[i] >= 0)
        
        # Objective vector
        c_obj = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        for j in range(j_start, j_end + 1):
            c_obj[j] = a[k_star - j]

        # since n_active is limited)
        n_act = len(active_lags)
        A_ub = np.zeros((n_act, n))
        b_ub = np.zeros(n_act)

            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                A_ub[idx, j] = a[k - j]
            b_ub[idx] = max(0, (M - c[k])) / 2 + 1e-12

        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = [(-a[i], 0.5 * max(a)) for i in range(n)]
        
        # Solve LP
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 5.0})
        except Exception:
            break
        
        if not res.success:
            break
        
        delta = res.x
        delta_step = step_scale * delta
        
        # Line search
        best_t = 0.0
        best_local_sc = sc
        for t in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_try = a + t * delta_step
            a_try = normalize(a_try)
            sc_try = compute_score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.03, 0.05]:
                a_try = a + t * delta_step
                a_try = normalize(a_try)
                sc_try = compute_score(a_try)
                if sc_try < best_local_sc:
                    best_local_sc = sc_try
                    best_t = t
        
        if best_t == 0.0:
            break
        
        a = a + best_t * delta_step
        a = normalize(a)
        
        if time.monotonic() > time.time():  # safety (should never trigger)
            break
    
    return best_a

27. **Another important consideration: the Fienup approach.**

The Fienup/Gerchberg-Saxton algorithm iterates between:
- Time domain: enforce a >= 0 (and optionally sum = 1)
- Frequency domain: enforce |A(f)| = constant (flat spectrum)

The key parameter is how aggressively we flatten the spectrum. Too aggressive → oscillations in time domain → clipping to 0 → injection of harmonics. Too gentle → slow convergence.

A better approach: instead of exactly flattening, use a "soft" target:
P_target = P_mean * (1 + gamma * (P - P_mean) / P_std)
where gamma controls the degree of flattening (gamma=0 means no change, gamma→∞ means full flattening).

Or use a "water-filling" approach:
P_new = max(P * alpha, P_mean * beta)
where alpha < 1 reduces peaks and beta > 1 boosts valleys.
