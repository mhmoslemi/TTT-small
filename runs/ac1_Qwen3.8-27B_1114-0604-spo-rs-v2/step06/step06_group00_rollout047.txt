Optimized step function search combining top-K peak gradient with line search,
high-beta LSE annealing, LP-based blending, and multi-scale exploration.
Targets score < 1.5027 by precisely attacking the convolutional peak structure.
"""

import numpy as np
from scipy.optimize import minimize
import time
import scipy.sparse as sp


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

    Minimize top-K max of convolution using exact line search.
    Targets the K largest peaks simultaneously with proper gradient.

    best_sc = compute_score(a)

        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

            best_a = a.copy()

        grad_f /= K

        # Full score gradient (scale-invariant part)
        f_val = float(np.sum(c[top_idx])) / K
        S2 = S * S
        S3 = S2 * S
        grad = (grad_f / S2) - 2.0 * f_val * np.ones(n) / S3

        # Remove mean (since score is scale-invariant, only relative changes matter)
        grad -= grad.mean()

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Line search: try several step sizes
        best_alpha = 0.0
        best_local_sc = sc
        alphas = [max_step, max_step * 0.5, max_step * 0.25, max_step * 0.1,
                  max_step * 0.05, max_step * 0.02, max_step * 0.01]
        for alpha in alphas:
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_t = a_trial.sum()
            if S_t < 1e-12:
                continue
            a_trial = a_trial / S_t
            sc_t = compute_score(a_trial)
            if sc_t < best_local_sc:
                best_local_sc = sc_t
                best_alpha = alpha
                best_local_a = a_trial.copy()

        if best_alpha > 0:
            a = best_local_a
        else:
            # No improvement found, try smaller steps
            a_trial = a - max_step * 0.005 * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_t = a_trial.sum()
            if S_t < 1e-12:
                break
            a_trial = a_trial / S_t
            sc_t = compute_score(a_trial)
            if sc_t < sc:
                a = a_trial
            else:
                # Stuck, perturb slightly
                a = a + 0.001 * np.random.standard_normal(n)
                a = normalize(a)

    return best_a


def topk_grad_aggressive(a, K, n_iters):
    """
    More aggressive top-K gradient: also increases non-contributors.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top K peaks
        top_idx = np.argsort(c)[::-1][:K]

        # Compute which positions contribute to the top peaks
        peak_contrib = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if hi >= lo:
                s = k - hi
                e = min(k + 1, n)
                peak_contrib[lo:hi + 1] += a[s:e][::-1]

        # Gradient: reduce peak contributors, boost non-contributors
        grad = np.zeros(n)
        top_count = float(len([k for k in top_idx]))
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if hi >= lo:
                s = k - hi
                e = min(k + 1, n)
                grad[lo:hi + 1] += 2.0 * a[s:e][::-1]
        grad /= top_count

        # Boost positions with low peak contribution
        low_contrib = peak_contrib < np.percentile(peak_contrib, 30)
        grad[low_contrib] -= 0.5 * (grad.mean() * top_count / max(1, low_contrib.sum()))

        S2 = S * S
        S3 = S2 * S
        f_val = float(np.sum(c[top_idx])) / top_count
        grad_score = (grad / S2) - 2.0 * f_val / S3
        grad_score -= grad_score.mean()

        gn = np.linalg.norm(grad_score)
        if gn < 1e-15:
            break

        a_trial = a - 0.01 * grad_score / gn * 0.01
        a_trial = np.maximum(a_trial, 0.0)
        S_t = a_trial.sum()
        if S_t < 1e-12:
            break
        a_trial = a_trial / S_t
        sc_t = compute_score(a_trial)
        if sc_t < sc:
            a = a_trial
        else:
            a = a - 0.003 * grad_score / gn * 0.01
            a = normalize(a)

    return best_a
