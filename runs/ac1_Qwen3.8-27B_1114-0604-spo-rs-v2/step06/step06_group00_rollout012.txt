"""
Spectral-flattening optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses LSE-annealing with L-BFGS, Fienup spectral iterations adapted for linear convolution,
iterative peak shaving via subgradient, and multi-restart with structured perturbations.
Key insight: for linear conv of non-negative a with itself, minimize peak-to-mean ratio
of the convolution to approach the theoretical lower bound 2n/(2n-1) ~ 1.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    L = (1/beta) * log(sum(exp(beta * c_k))) approximates max(c).
    score = 2n * L / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    # dL/da_j = 2 * sum_k p_k * a_{k-j} (for valid indices)
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * a / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fienup_linear(a, n_iters=400):
    """
    Fienup adapted for linear convolution flattening.
    Targets making b_k = (a*a)_k as flat as possible.
    In frequency domain: B(f) = A(f)^2 should be concentrated at DC.
    We iteratively: compute spectrum, shape |A| to concentrate energy at DC,
    reconstruct, project to non-negative.
    """
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        # Power spectrum |A|^2 corresponds to circular conv (with zero-padding = linear conv)
        P = A.real ** 2 + A.imag ** 2
        P_total = float(P.sum())
        if P_total < 1e-30:
            break

        # Target: concentrate power at low frequencies
        # Use exponential decay: target[f] ~ exp(-alpha * f^2 / (2*sigma^2))
        freqs = np.arange(len(P))
        alpha = 2.0 + 8.0 * progress  # Increase concentration over time
        sigma = n / (3.0 + 2.0 * progress)
        target = np.exp(-alpha * freqs**2 / (2.0 * sigma**2))
        target = target / target.sum() * P_total

        # Blend
        blend = 0.3 + 0.5 * progress
        P_new = (1.0 - blend) * P + blend * target
        P_new = np.maximum(P_new, 1e-30)

        # Update amplitude, preserve phase
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a


def fienup_quantile(a, n_iters=400):
    """Fienup with quantile-based spectral clipping to reduce peaks."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Clip high-frequency power
        q = 0.60 + 0.35 * progress
        thresh = np.quantile(P, q)
        P_clipped = P.copy()
        above = P_clipped > thresh
        P_clipped[above] = thresh

        # Also boost low-frequency power
        P_low_mask = P < np.median(P) * 0.1
        P_clipped[P_low_mask] *= 1.5 + progress

        ratio = np.sqrt(np.maximum(P_clipped, 1e-30) / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a


def peak_shave(a, n_iters=5000, top_k=10, step_init=0.005, decay=0.9998):
    """
    Subgradient descent targeting top-K convexolution peaks simultaneously.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = (c[k] / M_val) ** 2  # Square weighting for focus
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # dc_k/da_j = 2*a_{k-j} for valid j
                js = np.arange(j_start, j_end + 1)
                contrib = 2.0 * a[js * 0 + (k - js)]  # a[k-j] for j in range
                # Actually: dc_k/da_j involves a_{k-j} (and a_{j-k} for symmetric)
                # For linear conv: b_k = sum_i a_i a_{k-i}, db_k/da_j = a_{k-j} + a_{j-k} (if valid)
                # But since we're convolving a with itself: db_k/da_j = a_{k-j} (from i=j term) + a_{j-k} (from i=k-j term)
                # For real a: both terms exist. But in our formula, k-j and j-k are different.
                # Let's just use: db_k/da_j = 2*a_{abs(k-j)} is wrong.
                # Correct: b_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * a_{k-i}
                # db_k/da_j: a_j appears when i=j (giving a_{k-j}) and when k-i=j i.e. i=k-j (giving a_j*1)
                # So db_k/da_j = a_{k-j}[0<=k-j<n] + a_{k-j}[0<=k-j<n] = 2*a_{k-j} if 0<=k-j<n
                # Wait: when i=j, term is a_j * a_{k-j}, derivative wrt a_j is a_{k-j}
                # When i=k-j, term is a_{k-j} * a_j, derivative wrt a_j is a_{k-j}
                # So db_k/da_j = 2*a_{k-j} if 0 <= k-j < n, else 0 (or a_j if k=2j, but that's included)
                valid = (k - js >= 0) & (k - js < n)
                contrib = np.zeros(len(js))
                contrib[valid] = 2.0 * a[k - js[valid]]
                sg[js] += w * contrib

        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step_polyak = (sc - 1.0) / (gn * gn + 1e-30) if sc > 1.0 else step
        step = min(step, step_polyak, 0.02)
        step = max(step, 1e-10)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coordinate_descent(a, n_sweeps=8, n_top=50):
    """Coordinate descent: reduce top contributors to the peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        # Contribution to peak: a[i] * a[k_star - i]
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.8]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def perturb_structured(a, rng, scale=0.02):
    """Structured perturbation: low-frequency sinusoidal modulation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 8):
        freq = rng.integers(1, 15)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / (m + 1)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_additive(a, rng, scale=0.015):
    """Additive Gaussian perturbation."""
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)


def perturb_peak_targeted(a, rng, scale=0.003):
    """Add mass away from peak contributors."""
    n = len(a)
    c = fast_conv(a)
    k_peak = int(np.argmax(c))
    j_start = max(0, k_peak - n + 1)
    j_end = min(k_peak, n - 1)
    peak_mask = np.ones(n, dtype=bool)
    peak_mask[j_start:j_end + 1] = False
    add_idx = np.where(peak_mask)[0]
    n_add = max(20, n // 30)
    if len(add_idx) > n_add:
        add_idx = rng.choice(add_idx, n_add, replace=False)
    a_new = a.copy()
    a_new[add_idx] += scale
    return normalize(a_new)


def interpolate_to(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

        weight = 1.0 / (1.0 + 0.4 * (i - mid)**2 / mid)
        maxiter = int(min(8000, remaining * 15 * weight))
        maxiter = max(maxiter, 400)
        print(f"  LSE beta={beta:>6}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 25)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE b={beta}")
        a_work = a_opt.copy()

    # Second refinement pass from best
    if time.monotonic() < phase2_end - 30:
        print("  LSE refinement...")
        for beta in [10000, 30000, 80000]:
            if time.monotonic() > phase2_end - 15:
                break
            remaining = phase2_end - 15 - time.monotonic()
            if remaining < 10:
                break
            maxiter = max(500, int(remaining * 10))
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
            print(f"    refine beta={beta}: score={sc:.8f}")
            update(sc, a_opt, f"LSE-ref b={beta}")

    # ==================================================================
    # Phase 3: Peak Shaving (12% time)
    # ==================================================================
    phase3_end = t_start + 0.57 * (budget_s - 35.0)
    print(f"\n=== Phase 3: Peak Shaving (budget {phase3_end - t_start:.0f}s) ===")

    for top_k in [5, 10, 20, 40]:
        if time.monotonic() > phase3_end - 20:
            break
        remaining_iters = int((phase3_end - 20 - time.monotonic()) * 2500)
        remaining_iters = max(1000, min(remaining_iters, 8000))
        a_ps = peak_shave(best_a, n_iters=remaining_iters, top_k=top_k)
        sc = compute_score(a_ps)
        update(sc, a_ps, f"PSAVE k={top_k}")

        # LSE follow-up on shaved result
        if sc < best_score and time.monotonic() < phase3_end - 15:
            sc2, a2 = run_lbfgs(a_ps, 15000, maxiter=2000, maxfun=50000)
            update(sc2, a2, f"PSAVE+LSE k={top_k}")

    # ==================================================================
    # Phase 4: Coordinate Descent (5% time)
    # ==================================================================
    phase4_end = t_start + 0.62 * (budget_s - 35.0)
    print(f"\n=== Phase 4: Coordinate Descent ===")

    a_cd = coordinate_descent(best_a, n_sweeps=12, n_top=40)
    sc = compute_score(a_cd)
    update(sc, a_cd, "COORD")

    if time.monotonic() < phase4_end - 10:
        sc2, a2 = run_lbfgs(a_cd, 20000, maxiter=2000, maxfun=50000)
        update(sc2, a2, "COORD+LSE")

    # ==================================================================
    # Phase 5: Multi-Restart (15% time)
    # ==================================================================
    phase5_end = t_start + 0.77 * (budget_s - 35.0)
    print(f"\n=== Phase 5: Multi-Restart ===")

    n_restarts = 20
    for r in range(n_restarts):
        if time.monotonic() > phase5_end - 20:
            break
        remaining = phase5_end - 20 - time.monotonic()
        if remaining < 15:
            break

        # Vary perturbation type
        r_type = r % 4
        if r_type == 0:
            a_pert = perturb_structured(best_a, rng, scale=0.01 + 0.004 * (r % 5))
        elif r_type == 1:
            a_pert = perturb_additive(best_a, rng, scale=0.005 + 0.003 * (r % 5))
        elif r_type == 2:
            a_pert = perturb_peak_targeted(best_a, rng, scale=0.002 + 0.001 * (r % 3))
        else:
            # Bigger jump: blend with random
            a_rand = np.maximum(0.5 + 0.5 * rng.standard_normal(len(best_a)), 0)
            a_pert = normalize(0.8 * best_a + 0.2 * a_rand)

        maxiter = max(400, int(remaining * 7))
        beta = 300 + r * 500
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 6: Larger-n Exploration (7% time)
    # ==================================================================
    phase6_end = t_start + 0.84 * (budget_s - 35.0)
    print(f"\n=== Phase 6: Larger-n Exploration ===")

    current_n = len(best_a)
    target_ns = []
    if current_n < 12000:
        target_ns.append(12000)
    if current_n < 15000:
        target_ns.append(15000)
    if current_n < 20000:
        target_ns.append(20000)

    for new_n in target_ns:
        if time.monotonic() > phase6_end - 30:
            break
        remaining = phase6_end - 30 - time.monotonic()
        if remaining < 30:
            break

        a_large = interpolate_to(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  Interp n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        if time.monotonic() < phase6_end - 15:
            maxiter = max(400, int((phase6_end - 15 - time.monotonic()) * 5))
            sc2, a2 = run_lbfgs(a_large, 5000, maxiter=maxiter, maxfun=maxiter * 20)
            update(sc2, a2, f"INTERP+LSE n={new_n}")

    # ==================================================================
    # Phase 7: Final Polish (14% time)
    # ==================================================================
    print(f"\n=== Phase 7: Final Polish ===")

    # Very high beta LSE
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        maxiter = max(1000, int(remaining * 8))
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_final, "FINAL-LSE-50k")

    if time.monotonic() < deadline - 25:
        remaining = deadline - 25 - time.monotonic()
        maxiter = max(500, int(remaining * 6))
        sc, a_final = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-100k")

    # Final peak shave
    if time.monotonic() < deadline - 15:
        remaining_iters = max(1000, int((deadline - 15 - time.monotonic()) * 4000))
        a_ps = peak_shave(best_a, n_iters=min(remaining_iters, 5000), top_k=5,
                          step_init=0.002, decay=0.9999)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd = coordinate_descent(best_a, n_sweeps=5, n_top=30)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # One more LSE pass at the very end
    if time.monotonic() < deadline - 3:
        sc, a_final = run_lbfgs(best_a, 200000, maxiter=1000, maxfun=20000)
        update(sc, a_final, "LAST-LSE")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 100

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
