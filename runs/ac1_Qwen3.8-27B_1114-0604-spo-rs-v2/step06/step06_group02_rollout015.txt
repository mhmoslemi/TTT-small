"""
Multi-stage search for nonnegative sequences minimizing
    2*n*max(conv(a,a) / sum(a)^2).

The program combines:
  - parent interpolation and asymmetric seed generation,
  - a Newton-like linear correction toward a flat convolution profile,
  - smooth-max (log-sum-exp) gradient L-BFGS-B with corrected gradients,
  - mirror/exponential simplex descent with active-set and smooth priors,
  - exact coordinate descent on single variables,
  - an optional linear-programming direction for moderate sizes.

All methods are conservative: a candidate is kept only when its exact score
improves the current best.
"""

def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _conv_square(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 1
    L = 2 * n - 1
    M = _next_pow2(L)
    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:L]
    c = np.maximum(c, 0.0)
    return c, M


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=np.float64), 0.0, 1
    S = float(a.sum())
    if S < 1e-15:
        return np.inf, np.zeros(0, dtype=np.float64), 0.0, 1
    c, M = _conv_square(a)
    maxc = float(c.max()) if c.size else 0.0
    score = 2.0 * n * maxc / (S * S)
    return float(score), c, maxc, M


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / S


def _percentile_fast(x, pct):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return 0.0
    k = int(x.size * float(pct))
    if k < 0:
        k = 0
    if k >= x.size:
        k = x.size - 1
    return float(np.partition(x, k)[k])


def _p_smooth(c, cmax, K):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    p = np.zeros(L, dtype=np.float64)
    if L == 0:
        return p
    if cmax <= 0.0:
        p[np.argmax(c)] = 1.0
        return p
    if K <= 0.0:
        return _p_active(c, cmax, 0.1)

    c_ref = _percentile_fast(c, 0.25)
    gap = cmax - c_ref
    if gap <= 1e-30:
        gap = max(1e-18, 1e-10 * cmax)
    eps = max(gap * float(K), 1e-12 * cmax)
    scaled = (c - cmax) / eps
    scaled = np.clip(scaled, -50.0, 0.0)
    e = np.exp(scaled)
    s = float(e.sum())
    if s <= 0.0:
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return p


def _p_active(c, cmax, frac):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    p = np.zeros(L, dtype=np.float64)
    if L == 0:
        return p
    if cmax <= 0.0:
        p[np.argmax(c)] = 1.0
        return p
    frac = max(1e-8, float(frac))
    c_ref = _percentile_fast(c, 0.50)
    gap = cmax - c_ref
    if gap <= 1e-30:
        gap = max(1e-18, 1e-10 * cmax)
    thresh = cmax - frac * gap
    active = c >= thresh
    if not np.any(active):
        p[np.argmax(c)] = 1.0
        return p
    p[active] = 1.0
    p /= p.sum()
    return p


def _correlate(a, p, M, n):
    A = np.fft.rfft(a, M)
    P = np.fft.rfft(p, M)
    q = np.fft.irfft(A * np.conj(P), M)[:n]
    return np.maximum(q, 0.0)


def _lse_value_grad(a, K):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if n == 0 or S < 1e-15:
        return 1e100, np.zeros(0, dtype=np.float64)

    b = a / S
    c, M = _conv_square(b)
    cmax = float(c.max()) if c.size else 0.0
    if c.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)

    p = _p_smooth(c, cmax, K)
    q = _correlate(b, p, M, n)
    c_weight = float(np.dot(p, c))

    # Smooth value (log-sum-exp of scaled peaks).
    c_ref = _percentile_fast(c, 0.25)
    gap = cmax - c_ref
    if gap <= 1e-30:
        gap = max(1e-18, 1e-10 * cmax)
    eps = max(gap * float(K), 1e-12 * cmax)
    scaled = np.clip((c - cmax) / eps, -50.0, 0.0)
    e = np.exp(scaled)
    lse = cmax + eps * float(np.log(e.sum()))
    value = 2.0 * n * lse

    # Euclidean gradient of the scale-invariant normalized objective.
    grad = (4.0 * n * (q - c_weight)) / S
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(value), grad


def _lse_value(a, K):
    global _CACHE_KEY, _CACHE_A, _CACHE_G
    val, g = _lse_value_grad(a, K)
    _CACHE_KEY = K
    _CACHE_A = np.asarray(a, dtype=np.float64).copy()
    _CACHE_G = g.copy()
    return val


def _lse_grad(a, K):

    if _CACHE_KEY == K and _CACHE_A is not None and _CACHE_G is not None:

        if a0.shape == _CACHE_A.shape and float(np.max(np.abs(a0 - _CACHE_A))) <= 1e-12:
            return _CACHE_G
    return _lse_value_grad(a, K)[1]

            d = g / scale

            improved = False
            for mult in (1.0, 0.35, 0.1, 0.03):
                h = eta * mult
                if h < eta_min:
                    break
                z = np.exp(-np.clip(h * d, -50.0, 50.0))
                new_a = cur * z
                Snew = float(new_a.sum())
                if Snew < 1e-12:
                    continue
                new_a = new_a / Snew
                sc_new, _, _, _ = _score_exact(new_a)
                if np.isfinite(sc_new) and sc_new < sc - 1e-14:
                    cur = new_a
                    sc = sc_new
                    improved = True
                    if sc < best_sc - 1e-15:
                        best_sc = float(sc)
                        best_a = cur.copy()
                        print(f"      mirror stage={stage:+.4g}: score={best_sc:.10f}")
                    if mult > 0.9:
                        eta = min(eta * 1.25, eta_max)
                    elif mult > 0.25:
                        eta = min(eta * 1.10, eta_max)
                    else:
                        eta = max(eta * 0.85, eta_min)
                    break

            if not improved:
                stall += 1
                eta = max(eta * 0.55, eta_min)
                if stall >= 35 or eta <= eta_min * 1.01:
                    break

    final_sc, _, _, _ = _score_exact(best_a)
    return best_a, float(final_sc)


def _newton_flat_phase(a, deadline, max_rounds=6):
    a = _normalize(a)
    best_a = a.copy()
    best_sc, c, cmax, M = _score_exact(a)
    n = best_a.size
    L = 2 * n - 1
    if L <= 0:
        return best_a, best_sc

    target = np.full(L, 1.0 / float(L), dtype=np.float64)
    print(f"    Newton-flat start: n={n}, score={best_sc:.10f}")

    for rnd in range(max_rounds):
        if time.monotonic() >= deadline:
            break
        cur = best_a.copy()
        c, M = _conv_square(cur)
        R = 0.5 * (target - c)
        R -= R.mean()

        F_P = np.fft.rfft(cur, M)
        F_R = np.fft.rfft(R, M)
        denom = np.abs(F_P) ** 2
        max_denom = float(np.max(denom)) if denom.size else 1.0
        if max_denom <= 1e-30:
            break

        floor = 1e-8 * max_denom
        improved = False

        for lam_factor in (0.0, 1e-5, 1e-3):
            den = np.maximum(denom, floor) + lam_factor * max_denom
            F_d = np.conj(F_P) * F_R / den
            d = np.fft.irfft(F_d, M)[:n]
            d = np.asarray(d, dtype=np.float64)
            d = d - d.mean()
            max_d = float(np.max(np.abs(d)))
            if max_d < 1e-18:
                continue
            d = d / max_d
            amp = float(np.mean(cur))

            for f in (0.01, 0.05, 0.2, 0.8, 2.5, 8.0):
                for sign in (1.0, -1.0):
                    new_a = cur + sign * f * amp * d
                    new_a = np.maximum(new_a, 0.0)
                    Snew = float(new_a.sum())
                    if Snew < 1e-12:
                        continue
                    new_a = new_a / Snew
                    sc, _, _, _ = _score_exact(new_a)
                    if np.isfinite(sc) and sc < best_sc - 1e-13:
                        best_sc = float(sc)
                        best_a = new_a.copy()
                        improved = True
                        print(f"      Newton-flat round={rnd}, lam={lam_factor:.1e}, f={f:.2f}, sign={sign:+.0f}: score={best_sc:.10f}")
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    final_sc, _, _, _ = _score_exact(best_a)
    return best_a, float(final_sc)


def _build_toeplitz_structure(n):
    L = 2 * n - 1
    rows = []
    cols = []
    for k in range(L):
        i0 = max(0, k - n + 1)
        i1 = min(k, n - 1)
        if i1 < i0:
            continue
        idx = np.arange(i0, i1 + 1, dtype=np.int64)
        rows.extend([k] * idx.size)
        cols.extend(idx.tolist())
    return np.asarray(rows, dtype=np.int64), np.asarray(cols, dtype=np.int64)


def _lp_polish(a, deadline, max_iters=8):
    a = _normalize(a)
    best_a = a.copy()
    best_sc, c, cmax, M = _score_exact(a)
    n = best_a.size
    if n < 64 or n > 1200:
        return best_a, best_sc

    rows, cols = _build_toeplitz_structure(n)
    L = 2 * n - 1
    obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1.0)] * n
    print(f"    LP polish start: n={n}, score={best_sc:.10f}")

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        cur = best_a.copy()
        c, M = _conv_square(cur)
        maxc = float(c.max())
        if maxc <= 0.0:
            break

        data = cur[cols]
        A = sp.csr_matrix((data, rows, cols), shape=(L, n))
        b_ub = np.full(L, maxc, dtype=np.float64)

        try:
            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                res = linprog(
                    obj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                )
        except Exception:
            break

        if res.status != 0 or res.x is None:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        Sg = float(g.sum())
        if Sg <= 1.0 + 1e-7:
            break
        g = g / Sg

        cur_sc, _, _, _ = _score_exact(cur)
        improved = False
        for t in (0.03, 0.08, 0.16, 0.3, 0.5):
            new_a = (1.0 - t) * cur + t * g
            new_a = np.maximum(new_a, 0.0)
            Snew = float(new_a.sum())
            if Snew < 1e-12:
                continue
            new_a = new_a / Snew
            sc, _, _, _ = _score_exact(new_a)
            if np.isfinite(sc) and sc < cur_sc - 1e-13:
                cur = new_a
                cur_sc = float(sc)
                improved = True
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = new_a.copy()
                    print(f"      LP iter={it}, t={t:.2f}: score={best_sc:.10f}")
                break

        if not improved:
            break

    final_sc, _, _, _ = _score_exact(best_a)
    return best_a, float(final_sc)


def _coordinate_polish(a, deadline, max_sweeps=6):
    a = _normalize(a)
    best_a = a.copy()
    best_sc, c, cmax, M = _score_exact(a)
    n = best_a.size
    if n < 8 or time.monotonic() >= deadline:
        return best_a, best_sc

    factors = np.array(
        [0.0, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0, 1.3, 1.7, 2.4],
        dtype=np.float64,
    )
    L = 2 * n - 1
    print(f"    coordinate polish start: n={n}, score={best_sc:.10f}")

    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline:
            break
        cur = best_a.copy()
        c, M = _conv_square(cur)
        cmax = float(c.max())
        sc = 2.0 * n * cmax

        c_ref = _percentile_fast(c, 0.80)
        gap = cmax - c_ref
        if gap <= 1e-30:
            gap = max(1e-18, 1e-10 * cmax)
        thresh = cmax - max(0.05 * gap, 1e-12 * cmax)
        active = np.where(c >= thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
        else:
            active = active[np.argsort(c[active])[::-1][:min(active.size, 80)]]

        important = np.zeros(n, dtype=np.float64)
        for kk in active:
            kk = int(kk)
            i0 = max(0, kk - n + 1)
            i1 = min(kk, n - 1)
            if i1 < i0:
                continue
            js = np.arange(i0, i1 + 1, dtype=np.int64)
            partners = cur[kk - js]
            important[js] += partners

        top_count = int(max(30, min(220, n // 12)))
        top = np.argsort(important)[::-1][:top_count]
        if n > 12:
            rand = _RNG.integers(0, n, size=min(24, n))
            selected = np.unique(np.concatenate((top.astype(np.int64), rand)))
        else:
            selected = top.astype(np.int64)
        order = _RNG.permutation(selected)

        for jj in order:
            if time.monotonic() >= deadline:
                break
            j = int(jj)
            y0 = float(cur[j])
            mean_a = 1.0 / float(n)
            base = max(y0, mean_a)

            cand_y = base * factors
            cand_y = np.concatenate((cand_y, y0 * np.array([0.0, 0.5, 2.0], dtype=np.float64)))
            cand_y = cand_y[cand_y >= 0.0]
            if cand_y.size == 0:
                continue
            cand_y = np.unique(cand_y)
            if cand_y.size > 14:
                idxs = np.linspace(0, cand_y.size - 1, 14, dtype=int)
                cand_y = cand_y[idxs]

            if j > 0:
                left_max = float(np.max(c[:j]))
            else:
                left_max = 0.0
            if j + n < L:
                right_max = float(np.max(c[j + n:]))
            else:
                right_max = 0.0
            outside = max(left_max, right_max)

            oldc = c[j:j + n]
            dy = cand_y[:, None] - y0
            inc = 2.0 * cur[None, :] * dy
            inc[:, j] += dy[:, 0] * dy[:, 0]
            new_vals = oldc[None, :] + inc

            S_new = 1.0 - y0 + cand_y
            max_in = np.max(new_vals, axis=1)
            max_new = np.maximum(outside, max_in)

            valid = S_new > 1e-12
            if not np.any(valid):
                continue
            scores = np.full(cand_y.size, np.inf, dtype=np.float64)
            scores[valid] = 2.0 * n * max_new[valid] / (S_new[valid] * S_new[valid])
            bi = int(np.argmin(scores))
            sc_best = float(scores[bi])

            if np.isfinite(sc_best) and sc_best < sc - 1e-14:
                yb = float(cand_y[bi])
                c[j:j + n] = new_vals[bi]
                S = float(S_new[bi])
                if S > 1e-12:
                    cur = cur / S
                    c = c / (S * S)
                sc = float(2.0 * n * c.max())
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = cur.copy()
                    if sweep % 2 == 0:
                        print(f"      CD sweep={sweep}, j={j}: score={best_sc:.10f}")

    final_sc, _, _, _ = _score_exact(best_a)
    return best_a, float(final_sc)


def _interp_seq(parent, n_target):
    parent = np.asarray(parent, dtype=np.float64).ravel()
    parent = np.clip(parent[parent > -1.0], 0.0, 1000.0)
    N = parent.size
    if N == 0:
        return np.full(n_target, 1.0, dtype=np.float64)
    if N == int(n_target):
        return parent.copy()
    x0 = np.linspace(0.0, 1.0, N)
    x1 = np.linspace(0.0, 1.0, int(n_target))
    return np.interp(x1, x0, parent).astype(np.float64)


def _smooth_seq(a, sigma_frac=0.008):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 16:
        return a
    sigma = max(1.0, n * sigma_frac)
    rad = int(max(2, sigma * 3.5))
    x = np.arange(-rad, rad + 1, dtype=np.float64)
    k = np.exp(-0.5 * (x / sigma) ** 2)
    k /= k.sum()
    out = np.convolve(a, k, mode="same")
    return np.maximum(out, 0.0)


def _ramp_seq(n):
    x = np.linspace(0.0, 1.0, int(n))
    return 1.0 + 0.6 * (x - 0.5)


def _random_seq(n, kind=0):
    n = int(n)
    if kind == 0:
        a = np.abs(_RNG.standard_normal(n)) + 0.25
    elif kind == 1:
        x = np.linspace(0.0, 1.0, n)
        center = _RNG.uniform(0.15, 0.85)
        width = _RNG.uniform(0.05, 0.20)
        a = np.exp(-((x - center) / width) ** 2) + 0.08
    else:
        f = np.fft.rfft(_RNG.standard_normal(n))
        M = f.size
        keep = max(2, int(0.08 * M))
        f[keep:] *= 0.05
        a = np.fft.irfft(f, n)
        a = np.abs(a) + 0.2
    return np.maximum(a, 0.0)


def _seed_key(item):
    return item[0]


def _add_seed(seeds, name, seq):
    seq = _normalize(seq)
    if seq.size < 32:
        return
    sc, _, _, _ = _score_exact(seq)
    if np.isfinite(sc):
        seeds.append((float(sc), name, seq))


def _select_candidates(seeds):
    if not seeds:
        return []
    seeds = sorted(seeds, key=_seed_key)
    best_score = float(seeds[0][0])
    cands = [seeds[0][2]]

    if len(seeds) > 1:
        cands.append(seeds[1][2])

    max_len = 0
    for q in cands:
        if q.size > max_len:
            max_len = q.size

    # If there is a properly longer candidate close to the best, keep it.
    for sc, name, seq in seeds:
        if len(cands) >= 3:
            break
        if seq.size > 1.25 * max_len and sc <= best_score * 1.004 + 1e-7:
            already = False
            for q in cands:
                if q.size == seq.size:
                    already = True
                    break
            if not already:
                cands.append(seq)
                max_len = seq.size

    return cands[:3]


def _optimize_seed(a, deadline, label=""):
    start = time.monotonic()
    budget = max(10.0, deadline - start)
    a = _normalize(a)
    best_sc, _, _, _ = _score_exact(a)
    print(f"  [{label}] initial score={best_sc:.10f}")

    # Newton-like flat-profile corrections.
    end = start + 0.22 * budget
    if time.monotonic() < end:
        a, best_sc = _newton_flat_phase(a, min(deadline, end), max_rounds=5)

    # Mirror / active-set descent.
    end = start + 0.52 * budget
    if time.monotonic() < end:
        a, best_sc = _mirror_polish(a, min(deadline, end))

    # Optional LP direction if moderate size.
    if a.size <= 1200:
        end_lp = start + 0.65 * budget
        if time.monotonic() < end_lp:
            a, best_sc = _lp_polish(a, min(deadline, end_lp), max_iters=5)

    # L-BFGS smooth-max polish.
    end = start + 0.86 * budget
    if time.monotonic() < end:
        a, best_sc = _lbfgs_schedule(a, min(deadline, end))

    # Coordinate descent polish.
    if time.monotonic() < deadline:
        a, best_sc = _coordinate_polish(a, deadline, max_sweeps=4)

    # A final short focus if time remains.
    now = time.monotonic()
    if now < deadline - 20.0:
        a, best_sc = _lbfgs_schedule(a, deadline, K_list=[2.0, 0.5, 0.1, 0.02])
        if time.monotonic() < deadline - 5.0:
            a, best_sc = _coordinate_polish(a, deadline, max_sweeps=2)

    return a, best_sc


def _final_polish(a, deadline):
    start = time.monotonic()
    budget = max(10.0, deadline - start)
    a = _normalize(a)
    best_sc, _, _, _ = _score_exact(a)
    print(f"  [final] initial score={best_sc:.10f}")

    end = start + 0.25 * budget
    if time.monotonic() < end:
        a, best_sc = _newton_flat_phase(a, min(deadline, end), max_rounds=4)

    end = start + 0.80 * budget
    if time.monotonic() < end:
        a, best_sc = _lbfgs_schedule(
            a,
            min(deadline, end),
            K_list=[20.0, 5.0, 1.0, 0.2, 0.05, 0.012, 0.004],
        )

    if time.monotonic() < deadline:
        a, best_sc = _coordinate_polish(a, deadline, max_sweeps=3)

    now = time.monotonic()
    if now < deadline - 10.0:
        a, best_sc = _lbfgs_schedule(a, deadline, K_list=[0.5, 0.1, 0.02])
        if time.monotonic() < deadline - 5.0:
            a, best_sc = _coordinate_polish(a, deadline, max_sweeps=2)

    return a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _RNG
    _RNG = np.random.default_rng(int(seed))

    t0 = time.monotonic()
    total_budget = float(budget_s)
    # Keep a safety margin so the function returns before hard termination.
    deadline = t0 + max(60.0, total_budget - 150.0)

    parent = None
    try:
        hp = globals().get("height_sequence_1")
        if hp is not None:
            p = np.asarray(hp, dtype=np.float64).ravel()
            p = p[np.isfinite(p)]
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 32 and fp_sum := float(p.sum()) > 0.0:
                parent = p / fp_sum
    except Exception:
        parent = None

    seeds = []

    if parent is not None:
        _add_seed(seeds, "parent", parent)
        _add_seed(seeds, "parent_smooth", _smooth_seq(parent, 0.004))
        _add_seed(seeds, "parent_smooth2", _smooth_seq(parent, 0.012))

        N = parent.size

        # Small asymmetric perturbations of the best existing shape.
        x = np.linspace(0.0, 1.0, N)
        for amp in (0.05, 0.12, 0.25):
            for freq in (1, 2, 3):
                mod = parent * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 0.7 * freq))
                _add_seed(seeds, f"mod_{amp}_{freq}", mod)

        # Length variants.
        targets = []
        if N >= 1024:
            targets.append(N // 2)
        if N < 4096:
            targets.append(N * 2)
        if N < 2048:
            targets.append(N * 4)
        for tt in (512, 1024, 2048, 4096, 8192):
            targets.append(tt)

        seen = set()
        for tt in targets:
            tt = int(tt)
            if tt in seen:
                continue
            if tt < 256 or tt > 8192:
                continue
            if tt == N:
                seen.add(tt)
                continue
            seen.add(tt)
            _add_seed(seeds, f"len_{tt}", _interp_seq(parent, tt))

        # A few random basins with comparable length.
        for kind in range(3):
            _add_seed(seeds, f"rand_{kind}", _random_seq(N, kind))
    else:
        for n in (1024, 2048, 4096):
            _add_seed(seeds, f"uniform_{n}", np.ones(n))
            _add_seed(seeds, f"ramp_{n}", _ramp_seq(n))
            for kind in range(3):
                _add_seed(seeds, f"rand_{n}_{kind}", _random_seq(n, kind))

    if not seeds:
        a = np.ones(1024, dtype=np.float64)
        a, sc = _optimize_seed(a, deadline, "fallback")
        out = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(out) < 0.01:
            out = [1.0] * 64
        verified = float(evaluate_sequence(out))
        print(f"FALLBACK final verified={verified:.10f}")
        return out

    candidates = _select_candidates(seeds)
    if not candidates:
        candidates = [seeds[0][2]]

    global_best = candidates[0]
    global_score, _, _, _ = _score_exact(global_best)
    print(f"Initial global best: n={global_best.size}, score={global_score:.10f}")

    elapsed_total = deadline - t0
    for ci, seq in enumerate(candidates):
        now = time.monotonic()
        if ci == 0:
            phase_end = t0 + 0.55 * elapsed_total
        elif ci == 1:
            phase_end = t0 + 0.82 * elapsed_total
        else:
            phase_end = deadline - 45.0

        if now >= phase_end - 25.0:
            continue

        try:
            seq, sc = _optimize_seed(seq, phase_end, f"candidate{ci}")
        except Exception:
            print(f"  candidate{ci} exception, skipping")
            continue

        seq = _normalize(seq)
        sc, _, _, _ = _score_exact(seq)
        if np.isfinite(sc) and sc < global_score - 1e-13:
            global_score = float(sc)
            global_best = seq.copy()
            print(f"  NEW GLOBAL after candidate{ci}: {global_score:.10f} (n={global_best.size})")

    # Final polish of the absolute best.
    if time.monotonic() < deadline - 40.0:
        try:
            global_best, global_score = _final_polish(global_best, deadline)
        except Exception:
            pass

    out_arr = np.asarray(global_best, dtype=np.float64).ravel()
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    if out_arr.size < 2 or float(out_arr.sum()) < 0.01:
        out_arr = np.ones(64, dtype=np.float64)

    out = [float(x) for x in out_arr]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t0
    print("=" * 70)
    print(f"FINAL internal={global_score:.10f}, verified={verified:.10f}, n={len(out)}, t={elapsed:.1f}/{total_budget:.1f}s")
    print("=" * 70)
    return out
