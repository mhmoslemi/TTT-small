def project_simplex(v):
    v = v.astype(float, copy=False)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    inds = np.arange(1, len(v)+1, dtype=float)
    cond = u * inds - (css - 1.0) > 0
    if not np.any(cond): return np.ones_like(v)/len(v) # fallback
    rho = np.nonzero(cond)[0][-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s=out.sum(); if s<=0: out=np.ones_like(v)/len(v)
    else: out = out/s
    return out
```

```
def project_simplex(v):
    n = v.size
    if n == 0: return v.copy()
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    inds = np.arange(1, n+1, dtype=np.float64)
    cond = u * inds - (css - 1.0) > 0.0
    if not np.any(cond):
        return np.full(n, 1.0/n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-12:
        return np.full(n, 1.0/n)
    out /= s  # numerical ensure sum exactly 1
    return out
```

```python
"""
Hybrid search for non-negative sequences minimizing
    2*n*max(np.convolve(a, a)) / (sum(a)**2).

The program combines:
  1. Multi-peak projected gradient descent on the simplex using smooth softmax
     weights over the convolution peaks.
  2. Active-set linear programming directions (Frank-Wolfe-like updates) built
     from the currently most offending convolution coefficients.
  3. LSE-smoothed L-BFGS-B polishing on the scale-invariant raw objective.
  4. Coordinate peak-targeted line searches and a small random finisher.

All helper functions are top-level, no closures/lambdas are used.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog


def pow2_ge(m):
    """Smallest power of two >= m (m integer, m >= 1)."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_conv(a):
    """Linear self-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    M = pow2_ge(m)
    A = np.fft.rfft(a, M)
    out = np.fft.irfft(A * A, M)
    return out[:m]


def conv_grad(a, w):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.
    w has length 2*len(a)-1.
    Returns vector g where g_i = 2 * sum_j a_j * w_{i+j}.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    L = w.size
    if n == 0 or L == 0:
        return np.empty(0, dtype=np.float64)
    m = n + L - 1
    M = pow2_ge(m)
    ra = np.ascontiguousarray(a[::-1])
    RA = np.fft.rfft(ra, M)
    W = np.fft.rfft(w, M)
    out = np.fft.irfft(RA * W, M)
    return out[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, 1.0 / n)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    inds = np.arange(1, n + 1, dtype=np.float64)
    cond = u * inds - (css - 1.0) > 0.0
    if not np.any(cond):
        return np.full(n, 1.0 / n)

    rho = int(np.nonzero(cond)[0][-1])
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        return np.full(n, 1.0 / n)
    out /= s
    return out


def normalize_nonneg(v):
    """Radial projection to non-negative, sum 1."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    v[np.nonzero(v < 0.0)] = 0.0
    s = float(v.sum())
    n = v.size
    if s <= 1e-15 or n == 0:
        return np.full(n, 1.0 / max(1, n))
    v /= s
    return v


def exact_score(a):
    """Exact internal score, equivalent to the displayed evaluator for valid a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    y = project_simplex(a)
    c = fft_conv(y)
    return float(2.0 * y.size * float(np.max(c)))


def make_weights(c, beta, uniform_mix=0.15):
    """
    Smooth softmax weights over all convolution lags.
    A small uniform component prevents ignoring the next-highest peaks.
    """
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return np.empty(0, dtype=np.float64)
    vals = beta * c
    m = float(np.max(vals))
    exps = np.exp(vals - m)
    s = float(np.sum(exps))
    if s <= 1e-300 or not np.isfinite(s):
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = exps / s
    if uniform_mix > 0.0:
        p = (1.0 - uniform_mix) * p + (uniform_mix / L)
    s = float(p.sum())
    if s <= 0.0:
        p = np.full(L, 1.0 / L)
    else:
        p /= s
    return p


def weighted_pg_refine(a, time_limit, beta, start_step=0.10, max_step=0.40,
                       max_trials=6, stagnation_limit=800, print_tag=""):
    """
    Projected gradient descent on the simplex for a smoothed multi-peak
    objective. The direction is the exact gradient of a weighted sum of the
    convolution coefficients.
    """
    if time_limit <= 0.0:
        y = project_simplex(np.maximum(a, 0.0))
        return y, exact_score(y)

    y = project_simplex(np.maximum(a, 0.0))
    n = y.size
    c = fft_conv(y)
    best = y.copy()
    best_sc = float(2.0 * n * float(np.max(c)))

    t0 = time.monotonic()
    it = 0
    step = start_step
    stall = 0

    while time.monotonic() - t0 < time_limit and it < 500000:
        p = make_weights(c, beta)
        J = float(np.dot(p, c))
        g = 2.0 * conv_grad(y, p)

        accepted = False
        trial = step
        tol = 1e-12 * max(1.0, abs(J))

        for _ in range(max_trials):
            cand = project_simplex(y - trial * g)
            cc = fft_conv(cand)
            Jc = float(np.dot(p, cc))
            sc = float(2.0 * n * float(np.max(cc)))

            if (Jc < J - tol and sc < 1.25 * best_sc) or sc < best_sc - 1e-12:
                y = cand
                c = cc
                accepted = True
                if sc < best_sc:
                    best_sc = sc
                    best = cand.copy()
                    stall = 0
                else:
                    stall += 1
                break

            trial *= 0.5

        if accepted:
            step = min(step * 1.10, max_step)
        else:
            step = max(step * 0.35, 1e-5)
            stall += 1
            if step <= 1.0001e-5:
                break

        if stall > stagnation_limit:
            break

        it += 1
        if print_tag and (it % 1000 == 0 or time.monotonic() - t0 > 60.0):
            print(f"{print_tag} beta={beta:.0f} t={time.monotonic()-t0:.1f}s best={best_sc:.8f}")

    return best, best_sc


def lse_objective_raw(x, beta):
    """
    Scale-invariant smoothed objective for scipy.optimize.minimize.
    Max of convolution is approximated by log-sum-exp.
    A tiny regularization fixes the scale to sum=1.
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return 1e10, np.empty(0, dtype=np.float64)

    x = np.maximum(x, 0.0)
    S = float(x.sum())
    if S <= 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = fft_conv(x)
    vals = beta * c
    m = float(np.max(vals))
    exps = np.exp(vals - m)
    z = float(np.sum(exps))
    if z <= 1e-300 or not np.isfinite(z):
        L = m / beta
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        L = m / beta + float(np.log(z)) / beta
        p = exps / z

    g = 2.0 * conv_grad(x, p)
    K = 2.0 * n
    S2 = S * S
    S3 = S2 * S

    score = K * L / S2

    # gentle scale anchoring
    reg = 1e-6 * (S - 1.0) * (S - 1.0)
    grad_reg = 2e-6 * (S - 1.0)

    grad = K * (g * S - 2.0 * L) / S3
    grad += grad_reg

    return float(score + reg), grad


def run_lbfgs_lse(a0, beta, maxiter=300, maxfun=2000):
    """Polish with L-BFGS-B on the smoothed raw objective."""
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    if n == 0:
        return a0
    a0 = np.maximum(a0, 0.0)
    if a0.sum() <= 1e-12:
        a0 = np.ones(n, dtype=np.float64) * 0.5

    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_raw,
            a0,
            method="L-BFGS-B",
            jac=True,
            args=(beta,),
            bounds=bounds,
            options={
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 30
            }
        )
        out = np.asarray(res.x, dtype=np.float64).ravel()
        out = np.clip(out, 0.0, 1000.0)
        if out.sum() <= 1e-12:
            return a0
        return out
    except Exception:
        return a0


def build_conv_matrix(active_rows, f, n):
    """
    Build dense matrix A such that A g gives selected lags of f*g.
    Used by the active-set LP direction.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    active_rows = np.asarray(active_rows, dtype=np.int64).ravel()
    P = active_rows.size
    A = np.zeros((P, n), dtype=np.float64)
    for r in range(P):
        k = int(active_rows[r])
        s = max(0, k - n + 1)
        e = min(k, n - 1)
        if e >= s:
            js = np.arange(s, e + 1, dtype=np.int64)
            A[r, s:e + 1] = f[k - js]
    return A


def lp_direction(f, active_rows, M, upper=1.0, time_limit=12.0):
    """
    Solve: max sum(g)  s.t. (f*g)_k <= M for active rows, 0 <= g <= upper.
    Returns a normalized direction or None.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    active_rows = np.asarray(active_rows, dtype=np.int64).ravel()
    n = f.size
    P = active_rows.size
    if n == 0 or P == 0:
        return None
    if time_limit <= 0.0:
        return None

    A = build_conv_matrix(active_rows, f, n)
    b = np.full(P, float(M), dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, float(upper))] * n

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={
                "time_limit": float(max(0.5, time_limit)),
                "presolve": True,
                "primal_feasibility_tolerance": 1e-9,
                "dual_feasibility_tolerance": 1e-9
            }
        )
        if res is None or getattr(res, "x", None) is None:
            return None
        g = np.asarray(res.x, dtype=np.float64).ravel()
        if g.size != n:
            return None
        if not np.all(np.isfinite(g)):
            return None
        g = np.clip(g, 0.0, float(upper))
        if g.sum() <= 1e-12:
            return None
        return g
    except Exception:
        return None


def lp_refine(a, time_limit, print_tag=""):
    """
    Active-set LP-based exploration. Each round selects the most offending
    convolution lags, solves a small active-set LP over non-negative directions,
    and tries compressed steps toward the resulting direction.
    """
    if time_limit <= 0.0:
        y = project_simplex(np.maximum(a, 0.0))
        return y, exact_score(y)

    best = project_simplex(np.maximum(a, 0.0))
    best_sc = exact_score(best)
    n = best.size

    P_list = (60, 120, 240)
    margin_list = (1e-5, 0.02, 0.08, 0.18)
    t_list = (0.02, 0.05, 0.10, 0.18, 0.30)

    t0 = time.monotonic()
    rnd = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and rnd < 160 and no_improve < 12:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20.0:
            break

        y = best.copy()
        c = fft_conv(y)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0:
            break

        P = int(P_list[rnd % len(P_list)])
        margin = float(margin_list[rnd % len(margin_list)])
        active = np.argsort(c)[::-1][:P]
        M_lp = M * (1.0 + margin) + 1e-15

        g = lp_direction(
            y,
            active,
            M_lp,
            upper=1.0,
            time_limit=max(2.0, min(18.0, rem - 10.0))
        )

        improved = False
        if g is not None:
            g = project_simplex(g)
            for t in t_list:
                cand = (1.0 - t) * y + t * g
                cand = project_simplex(np.maximum(cand, 0.0))
                sc = exact_score(cand)
                if sc < best_sc - 1e-12:
                    best = cand
                    best_sc = sc
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        rnd += 1
        if print_tag and rnd % 4 == 0:
            print(f"{print_tag} lp round={rnd} best={best_sc:.8f} no_improve={no_improve}")

    return best, best_sc


def peak_coord_polish(a, time_limit, top_k=5, print_tag=""):
    """
    Local coordinate line searches on the largest contributing items for each
    of the strongest convolution peaks.
    """
    if time_limit <= 0.0:
        y = project_simplex(np.maximum(a, 0.0))
        return y, exact_score(y)

    best = project_simplex(np.maximum(a, 0.0))
    best_sc = exact_score(best)
    n = best.size
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit and rounds < 500:
        c = fft_conv(best)
        sc = float(2.0 * n * float(np.max(c)))
        if sc < best_sc:
            best_sc = sc

        idx = np.argsort(c)[::-1][:top_k]
        improved = False

        for k_obj in idx:
            k = int(k_obj)
            s = max(0, k - n + 1)
            e = min(k, n - 1)
            if e < s:
                continue

            seg = best[s:e + 1]
            partner = best[k - e:k - s + 1][::-1]
            contrib = seg * partner
            top_local = np.argsort(contrib)[::-1][:min(4, len(contrib))]

            for li in top_local:
                j = s + int(li)
                if best[j] <= 1e-12:
                    continue

                for factor in (0.90, 0.75, 0.60):
                    cand = best.copy()
                    cand[j] *= factor
                    cand = project_simplex(np.maximum(cand, 0.0))
                    sc2 = exact_score(cand)
                    if sc2 < best_sc - 1e-12:
                        best = cand
                        best_sc = sc2
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

        rounds += 1
        if print_tag and rounds % 20 == 0:
            print(f"{peak_coord_polish.__name__} t={time.monotonic()-t0:.1f}s best={best_sc:.8f}")

    return best, best_sc


def random_finish(a, time_limit, rng, print_tag=""):
    """Short random multiplicative coordinate perturbations with exact checks."""
    if time_limit <= 0.0 or rng is None:
        y = project_simplex(np.maximum(a, 0.0))
        return y, exact_score(y)

    best = project_simplex(np.maximum(a, 0.0))
    best_sc = exact_score(best)
    n = best.size
    t0 = time.monotonic()
    strength = 0.05

    while time.monotonic() - t0 < time_limit:
        m = int(rng.integers(2, 16))
        idx = rng.integers(0, n, size=m)
        factors = rng.uniform(0.55, 1.65, size=m)

        cand = best.copy()
        cand[idx] *= factors
        cand = project_simplex(np.maximum(cand, 0.0))
        sc = exact_score(cand)

        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            strength = min(0.12, strength * 1.15)
        else:
            strength *= 0.92
            if strength < 0.005:
                break

    if print_tag:
        print(f"{print_tag} random finish best={best_sc:.8f}")

    return best, best_sc


def build_inits(n, rng, parent):
    """Construct diverse feasible nonnegative starting points."""
    inits = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    if parent is not None and parent.size == n:
        inits.append(np.maximum(parent, 0.0))

    inits.append(np.ones(n, dtype=np.float64) * 0.5)
    inits.append(0.1 + np.exp(-3.0 * x))
    inits.append(0.1 + np.exp(-3.0 * (1.0 - x)))
    inits.append(0.05 + 0.95 * np.power(np.sin(np.pi * x), 2))
    inits.append(0.1 + 0.45 * np.cos(2.0 * np.pi * x) + 0.30 * np.cos(4.0 * np.pi * x))
    inits.append(0.1 + 0.35 * np.cos(6.0 * np.pi * x) + 0.25 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.25 * rng.standard_normal(n), 0.01, 10.0))

    for _ in range(6):
        sig = np.zeros(n, dtype=np.float64)
        nh = int(rng.integers(3, 13))
        for h in range(1, nh + 1):
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = float(rng.standard_normal())
            sig += amp * np.sin(float(h) * 2.0 * np.pi * x + ph)
        init = 0.45 + 0.25 * sig
        inits.append(np.maximum(init, 0.01))

    cleaned = []
    for v in inits:
        v = np.asarray(v, dtype=np.float64).ravel()
        if v.size != n or not np.all(np.isfinite(v)):
            continue
        v = np.maximum(v, 0.0)
        if v.sum() <= 1e-12:
            v = np.ones(n, dtype=np.float64) * 0.5
        cleaned.append(v)
    return cleaned


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search entry point. Returns a list of nonnegative floats.
    """
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 45.0

    rng = np.random.default_rng(seed)

    # Try to load a previous construction if available.
    parent = None
    try:
        gs = globals()
        if "height_sequence_1" in gs:
            p = np.asarray(gs["height_sequence_1"], dtype=np.float64).ravel()
            if p.size >= 16:
                p = np.clip(p, 0.0, 1000.0)
                if np.all(np.isfinite(p)) and p.sum() > 0.0:
                    parent = p
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        a0 = parent.copy()
    else:
        n = 6962
        a0 = np.ones(n, dtype=np.float64) * 0.5

    best = project_simplex(np.maximum(a0, 0.0))
    best_sc = exact_score(best)

    print(f"initial n={n} score={best_sc:.8f} time={time.monotonic()-t_start:.1f}s")

    # If there is no parent, use a few diverse inits and quick PG to get competitive.
    if parent is None:
        inits = build_inits(n, rng, None)
        for ii, init in enumerate(inits):
            if time.monotonic() > deadline - 60.0:
                break
            init = normalize_nonneg(init)
            cand, sc = weighted_pg_refine(
                init,
                time_limit=min(8.0, max(1.0, deadline - time.monotonic())),
                beta=80000.0,
                stagnation_limit=120,
                print_tag=f"init{ii}"
            )
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = cand
                print(f"init{ii} improved: {sc:.8f}")

    # Primary multi-peak projected gradient blocks.
    beta_blocks = (
        (20000.0, 55.0),
        (50000.0, 60.0),
        (100000.0, 70.0),
        (200000.0, 70.0),
        (500000.0, 60.0),
    )
    for bi, (beta, blimit) in enumerate(beta_blocks):
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        cand, sc = weighted_pg_refine(
            best,
            time_limit=min(blimit, rem - 30.0),
            beta=beta,
            print_tag=f"PG{bi}"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"PG{bi} beta={beta:.0f} new best={best_sc:.8f}")

    # Active-set LP exploration.
    rem = deadline - time.monotonic()
    if rem > 150.0:
        cand, sc = lp_refine(
            best,
            time_limit=min(700.0, rem - 30.0),
            print_tag="LP"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"LP new best={best_sc:.8f}")

    # LSE L-BFGS polishing.
    for bi, (beta, maxfun) in enumerate(((120000, 1800), (300000, 1200), (700000, 900))):
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break
        cand = run_lbfgs_lse(best, beta=beta, maxiter=320, maxfun=maxfun)
        cand = project_simplex(np.maximum(cand, 0.0))
        sc = exact_score(cand)
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"LBFGS{bi} beta={beta:.0f} new best={sc:.8f}")

    # Second high-beta PG block after polishing.
    rem = deadline - time.monotonic()
    if rem > 90.0:
        cand, sc = weighted_pg_refine(
            best,
            time_limit=min(120.0, rem - 30.0),
            beta=1000000.0,
            print_tag="PG-final"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"PG-final new best={sc:.8f}")

    # Coordinate peak polish.
    rem = deadline - time.monotonic()
    if rem > 70.0:
        cand, sc = peak_coord_polish(
            best,
            time_limit=min(120.0, rem - 30.0),
            top_k=6,
            print_tag="COORD"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"COORD new best={sc:.8f}")

    # Random finisher.
    rem = deadline - time.monotonic()
    if rem > 40.0:
        cand, sc = random_finish(
            best,
            time_limit=min(70.0, rem - 20.0),
            rng=rng,
            print_tag="RAND"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
            print(f"RAND new best={sc:.8f}")

    # Tiny final PG if any time remains.
    rem = deadline - time.monotonic()
    if rem > 20.0:
        cand, sc = weighted_pg_refine(
            best,
            time_limit=max(5.0, rem - 12.0),
            beta=2000000.0,
            stagnation_limit=300,
            print_tag="PG-tiny"
        )
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc

    out_arr = np.asarray(best, dtype=np.float64).ravel()
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    if out_arr.size == 0 or out_arr.sum() < 0.01:
        out_arr = np.ones(n, dtype=np.float64) * 0.5

    out_list = [float(x) for x in out_arr]

    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = np.inf

    print("=" * 60)
    print(f"FINAL internal={best_sc:.8f}")
    print(f"FINAL verified={verified:.8f}")
    print(f"Length={len(out_list)} Time={time.monotonic()-t_start:.1f}s / {budget_s}s")
    print("=" * 60)

    return out_list
