"""
Nesterov-accelerated LSE optimization with adaptive beta, 
exact subgradient mirror descent, and multi-start strategy
for minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient via FFT
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def nesterov_lse(a0, beta, max_iter=50000, lr_init=0.01, deadline=None):
    """Nesterov accelerated gradient descent on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    
    y = a.copy()  # Nesterov momentum variable
    old_y = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    momentum = 0.9
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        # Nesterov step
        if it > 0:
            new_momentum = (1 + momentum) / 2  # standard Nesterov
            y = a + (momentum - new_momentum) * (a - old_y)
        old_y = a.copy()
        
        # Gradient at y (projected to non-negative)
        y_pos = np.maximum(y, 0.0)
        S_y = y_pos.sum()
        if S_y < 1e-12:
            break
        y_norm = y_pos / S_y
        
        sc, grad = lse_value_and_grad(y_norm, beta)
        
        # Gradient step
        a_new = y_norm - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = y_norm
        else:
            a_new = a_new / S_new
        
        a = a_new
        
        sc_check = compute_score(a)
        if sc_check < best_sc:
            best_sc = sc_check
            best_a = a.copy()
        
        # Adaptive learning rate
        if it % 1000 == 999 and it > 0:
            lr *= 0.95
        momentum = min(momentum + 0.0001 * it, 0.95)
        
        lr = max(lr, 1e-6)
    
    return best_sc, best_a


def subgrad_mirror_descent(a0, max_iter=20000, deadline=None):
    """Mirror descent (multiplicative updates) with exact max subgradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.05
    
    for it in range(max_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max conv
        # Find all argmax positions
        threshold = c_max - 1e-10
        peak_positions = np.where(c >= threshold)[0]
        
        # Subgradient: for each a_i, sum over peak positions k of a_{k-i}
        grad = np.zeros(n)
        for k in peak_positions:
            k = int(k)
            # c_k = sum_j a_j * a_{k-j} (with 0 <= j < n, 0 <= k-j < n)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            # dc_k/da_i = a_{k-i} for valid i
            for i in range(j_start, j_end + 1):
                grad[i] += a[k - i]
        
        # Gradient of 2n*max_c/S^2
        S2 = S * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * c_max / (S2 * S))
        
        # Mirror descent (multiplicative) update
        grad_norm = np.linalg.norm(grad_score)
        if grad_norm > 1e-15:
            step = eta * grad_score / grad_norm
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                eta = min(eta * 1.005, 0.3)
            else:
                eta *= 0.9
        eta = max(eta, 0.001)
    
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        S = x.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        x = x / S
        return lse_value_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_enhanced(a, n_iters=500, deadline=None):
    """Enhanced Fienup with adaptive spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Adaptive: more aggressive early
        alpha = 0.7 * (1.0 - progress) ** 2 + 0.03
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_beta_lbfgs(a, deadline, betas, iters_per_beta=10000):
    """Run L-BFGS with a schedule of increasing betas."""
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        maxiter = min(iters_per_beta, int(remaining * 10))
        maxiter = max(maxiter, 1000)
        
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 30)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    L-BFGS beta={beta}: {sc:.8f}")
    
    return best_a, best_sc


def perturbation_restart(a, rng, n_restarts=8, deadline=None):
    """Multiple perturbation strategies followed by quick optimization."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for r in range(n_restarts):
        if deadline and time.monotonic() > deadline:
            break
        
        if r == 0:
            # Gaussian noise
            a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        elif r == 1:
            # Peak-targeted noise
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            a_pert = a.copy()
            noise = rng.standard_normal(n) * 0.03
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            noise[j_start:j_end + 1] *= 3.0
            a_pert = a * (1.0 + noise)
        elif r == 2:
            # Low-freq modulation
            x = np.arange(n) / n
            freq = rng.integers(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif r == 3:
            # High-freq modulation
            x = np.arange(n) / n
            freq = rng.integers(n // 4, n // 2)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.03 * np.cos(2 * np.pi * freq * x + phase))
        elif r == 4:
            # Block swap
            a_pert = a.copy()
            b1 = rng.integers(0, n // 2)
            b2 = rng.integers(n // 2, n)
            sz = max(1, n // 32)
            a_pert[b1:b1+sz], a_pert[b2:b2+sz] = a_pert[b2:b2+sz].copy(), a_pert[b1:b1+sz].copy()
        elif r == 5:
            # Reciprocal perturbation
            a_pert = a * (1.0 + 0.02 / (a + 0.02))
        elif r == 6:
            # Smooth noise (low-pass filtered)
            noise = rng.standard_normal(n)
            # Simple box blur
            kernel = np.ones(8) / 8
            noise = np.convolve(noise, kernel, mode='same')
            a_pert = a * (1.0 + 0.04 * noise / (noise.std() + 1e-10))
        else:
            # Sidon-inspired: add structured pattern
            a_pert = a.copy()
            p = rng.integers(2, 100, size=n // 50)
            for pi in p:
                for i in range(0, n, pi):
                    a_pert[i] *= 1.0 + 0.05 * (1 if (i // pi) % 2 == 0 else -1)

        if S < 1e-12:

        a_pert = a_pert / S

        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Restart {r}: {sc:.8f}")

def coordinate_descent_polish(a, n_sweeps=10, deadline=None):
    """Coordinate descent: try adjusting individual elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = c.max()
        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Focus on top contributing elements to the peak
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Try adjusting the top 50 elements contributing to the peak
        contribs = a[j_start:j_end + 1] * np.maximum(a[k_star - j_start:k_star - j_start + (j_end - j_start + 1)][::-1], 0)
        # Actually: c[k] = sum_{j} a[j] * a[k-j]
        # For k = k_star: term j contributes a[j]*a[k_star - j]
        contribs = np.zeros(j_end - j_start + 1)
        for idx, j in enumerate(range(j_start, j_end + 1)):
            other = k_star - j
            if 0 <= other < n:

"""
Nesterov-accelerated LSE optimization + exact subgradient mirror descent +
multi-beta L-BFGS + perturbation restarts + coordinate descent polish.
Targets minimizing 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def _lbfgs_obj(x, beta):
    n = len(x)
    x = np.maximum(x, 0.0)
    S = x.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    x = x / S
    return lse_value_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        _lbfgs_obj, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        print(f"  [FIENUP] NEW BEST: {sc:.8f}")

    # Phase 2: Nesterov ACG (15%)
    p2_end = t_start + 0.20 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Nesterov ACG ===")
    a_ng, sc_ng = nesterov_lse_phase(best_a, p2_end, betas=[100, 500, 1500], lr_init=0.005)
    if sc_ng < best_score:
        best_score = sc_ng
        best_a = np.clip(a_ng.copy(), 0.0, 1000.0)
        print(f"  [NESTEROV] NEW BEST: {sc_ng:.8f}")

    # Phase 3: Multi-beta L-BFGS (30%)
    p3_end = t_start + 0.50 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Multi-beta L-BFGS ===")
    betas3 = [500, 1500, 4000, 10000, 30000, 80000]
    a_lb, sc_lb = multi_beta_lbfgs(best_a, p3_end, betas3, iters_per_beta=15000)
    if sc_lb < best_score:
        best_score = sc_lb
        best_a = np.clip(a_lb.copy(), 0.0, 1000.0)
        print(f"  [LBFGS] NEW BEST: {sc_lb:.8f}")

    # Phase 4: Subgradient mirror descent (15%)
    p4_end = t_start + 0.65 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Mirror Descent ===")
    sc_md, a_md = subgrad_mirror_descent(best_a, max_iter=40000, deadline=p4_end)
    if sc_md < best_score:
        best_score = sc_md
        best_a = np.clip(a_md.copy(), 0.0, 1000.0)
        print(f"  [MIRROR] NEW BEST: {sc_md:.8f}")

    # Phase 5: Very high-beta L-BFGS (15%)
    p5_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 5: High-beta L-BFGS ===")
    betas5 = [50000, 150000, 400000]
    a_lb2, sc_lb2 = multi_beta_lbfgs(best_a, p5_end, betas5, iters_per_beta=18000)
    if sc_lb2 < best_score:
        best_score = sc_lb2
        best_a = np.clip(a_lb2.copy(), 0.0, 1000.0)
        print(f"  [LBFGS-HIGH] NEW BEST: {sc_lb2:.8f}")

    # Phase 6: Perturbation restarts (10%)
    p6_end = t_start + 0.90 * (total_budget - 60.0)
    print(f"\n=== Phase 6: Restarts ===")
    a_sr, sc_sr = perturbation_restart(best_a, rng, n_restarts=6, deadline=p6_end)
    if sc_sr < best_score:
        best_score = sc_sr
        best_a = np.clip(a_sr.copy(), 0.0, 1000.0)
        print(f"  [RESTART] NEW BEST: {sc_sr:.8f}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final polish ===")
    if time.monotonic() < deadline - 40:
        a_cd = coordinate_descent_polish(best_a, n_sweeps=8, deadline=deadline - 40)
        sc = compute_score(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cd.copy(), 0.0, 1000.0)
            print(f"  [CD] NEW BEST: {sc:.8f}")

    if time.monotonic() < deadline - 25:
        remaining = max(10.0, deadline - 25 - time.monotonic())
        maxiter = min(15000, max(1000, int(remaining * 12)))
        sc, a_fin = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 20)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_fin.copy(), 0.0, 1000.0)
            print(f"  [FINAL-LBFGS] NEW BEST: {sc:.8f}")

    if time.monotonic() < deadline - 10:
        a_cd2 = coordinate_descent_polish(best_a, n_sweeps=5, deadline=deadline - 10)
        sc = compute_score(a_cd2)
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cd2.copy(), 0.0, 1000.0)
            print(f"  [FINAL-CD] NEW BEST: {sc:.8f}")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
