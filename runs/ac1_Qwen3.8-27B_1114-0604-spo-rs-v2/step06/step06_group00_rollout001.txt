import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def fienup_improved(a, n_iters=500):
    """
    Fienup with adaptive target spectrum for linear autocorrelation flattening.
    Uses zero-padding to N >= 2n-1 so circular = linear autocorrelation.
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()  # Power of 2 >= 2n-1
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        prog = it / max(1, n_iters)
        A = np.fft.fft(a, N)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum (uniform power)
        P_mean = float(P.mean())
        
        # Adaptive alpha: start gentle, get aggressive
        alpha = 0.1 + 0.6 * (prog ** 0.5)
        
        # Blend towards flat
        P_new = (1 - alpha) * P + alpha * P_mean
        
        # Also apply a gentle quantile clip to prevent extreme modes
        q = 0.8 + 0.15 * prog
        thresh = np.quantile(P_new, q) * (1.2 - 0.4 * prog)
        P_new = np.minimum(P_new, thresh)
        P_new = np.maximum(P_new, P_mean * 0.05)
        
        # Enforce conjugate symmetry
        P_new = (P_new + P_new[np.arange(N)[::-1] % N]) / 2
        P_new = np.real(P_new)
        
        # Update
        P_old = np.abs(A) ** 2
        ratio = np.sqrt(P_new / np.maximum(P_old, 1e-30))
        A_new = A * ratio
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    
    p = exp_s / exp_s.sum()
    
    # grad_L[j] = 2 * sum_k p[k] * a[k-j] (circular-ish, but linear conv)
    # = 2 * (p * a_rev)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    M3 = (2*n-1) + n - 1
    M3_fft = 1 << (M3-1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    
    score = (2.0 * n * L) / S2
    return score, grad

def peak_shave(a, n_iters=3000, top_k=10):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.005
    lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S*S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if sc - lb < 1e-8:
            break
        
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w * a[j_start:j_end+1][::-1]
        
        S2 = S*S
        S3 = S2*S
        avg_w = 1.0
        grad = (2.0*n) * (sg/S2 - 2.0*M_val*avg_w/S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_p = (sc - lb) / (gn*gn + 1e-30)
        step = min(step, step_p, 0.02)
        step = max(step, 1e-10)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s
        step *= 0.9998
    
    return best_a
