import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """FFT-based linear convolution."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute 2n*max(conv)/sum²."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_obj_grad(a, beta, M_fft):
    """LSE objective and gradient."""
    n = len(a)
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2*n-1]
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clamp for numerical stability
    scaled = np.maximum(scaled, -100)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return c_max, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (with correction term)
    # Compute conv(p, flip(a)):
    rev_a = np.zeros(M_fft)  # or use a reversed
    rev_a[:n] = a[::-1]
    p_ext = np.zeros(M_fft)
    p_ext[:len(c)] = p
    G = np.fft.rfft(p_ext, M_fft) * np.fft.rfft(rev_a, M_fft)
    conv_g = np.fft.irfft(G, M_fft)
    # g_cross[i] = conv_g[i + n - 1] = sum_k p[k] * a[k-i]
    g_cross = conv_g[n-1:2*n-1]
    
    # Correction: for k=2i, subtract p[2i]*a[i] (the double-counting)
    k_vals = 2 * np.arange(n)
    valid = k_vals < 2*n-1
    correction = np.zeros(n)
    correction[valid] = p[k_vals[valid]] * a[valid]
    
    # Full gradient of L w.r.t. a
    g_L = 2.0 * g_cross - correction
    
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    # f = 2n*L/S², df/da = 2n*(g_L/S² - 2L*S'*S⁻³) where S'=sum(da)=each component
    # df/da_i = 2n*(g_L[i]/S² - 2*L*(1/S³))
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    
    obj = 2.0 * n * L / S2
    return obj, grad

def fista_lse(a0, beta, n_iters, M_fft, lr=None):
    """FISTA on LSE objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    a = a / S
    
    M = 2 * n - 1
    best_a = a.copy()
    best_obj = compute_score(a)
    
    # Estimate Lipschitz constant
    if lr is None:
        # Use backtracking
        # L ~ 2n * 2 * max(a) * beta (rough upper bound)
        # Actually L depends on the problem structure
        # Let's use adaptive backtracking
        lr = 1.0 / (2.0 * n * 2.0 * float(a.max()) * beta + 1.0)
        lr = min(lr, 1.0)
    
    y = a.copy()
    t = 1.0
    
    for it in range(n_iters):
        obj_y, grad_y = lse_obj_grad(y, beta, M_fft)
        
        # Gradient step
        y_new = y - lr * grad_y
        y_new = np.maximum(y_new, 0.0)
        
        # (We skip the simplex projection for sum since scale-invariant)
        
        # Normalize to prevent drift
        S_new = float(y_new.sum())
        if S_new < 1e-12:
            break
        y_new = y_new / S_new
        
        # Nesterov extrapolation for current iterate
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x_new = y_new
        y = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        t = t_new
        
        # Track best
        sc = compute_score(x_new)
        if sc < best_obj:
            best_obj = sc
            best_a = x_new.copy()
        
        a = x_new  # for reference
    
    return best_a

def lbfgs_lse(a0, beta, M_fft, maxiter, maxfun):
    """L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1e6).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0/n
    a0 = a0 / a0.sum()
    
    def fun_and_grad(x):
        obj, grad = lse_obj_grad(x, beta, M_fft)
        return obj, grad
    
    res = minimize(
        fun_and_grad, a0,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1e6)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1e6)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

def peak_subgrad(a, n_peaks, n_iters, M_fft, step_init=0.01):
    """Peak-directed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0  # theoretical lower bound
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[js:je+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.02)
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.99999
    
    return best_a

def simplex_project(z, S=1.0):
    """Project z onto {a >= 0, sum(a) = S}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0]

        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)
