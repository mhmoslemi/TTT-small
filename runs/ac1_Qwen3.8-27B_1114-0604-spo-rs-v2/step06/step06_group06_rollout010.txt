def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Compute cross-correlation: result[j] = sum_i a[i]*w[j-i] for valid indices (n outputs)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = max(n, m)
    N = next_power_of_two(M + M - 1)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M + M - 1]
    return full[n - 1:n - 1 + n]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

def get_active_indices(conv, top_k=10, threshold_frac=0.02):
    """Get indices of near-maximal convolution positions."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.array([len(conv) // 2])
    threshold = dmax * (1.0 - threshold_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) < top_k:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        active = np.unique(np.concatenate([active, idx]))
    if len(active) > top_k * 4:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        active = idx
    return np.sort(active)


def exp_grad_step(a, conv, active_idx, eta):
    """Exponentiated gradient step with concentrated weights on active positions."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    active_vals = conv[active_idx]
    # Weight proportional to how close to max
    dmax = float(np.max(conv))
    weights = np.maximum(active_vals / dmax - (1.0 - 0.05), 0.01)
    w[active_idx] = weights
    ws = w.sum()
    if ws < 1e-300:
        w[active_idx] = 1.0
        ws = len(active_idx)
    w = w / ws
    h = cross_corr(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-30:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def lse_smoothed(alpha, beta):
    """Log-sum-exp smoothed version of max for a vector alpha."""
    amax = float(np.max(alpha))
    if not np.isfinite(amax) or amax <= 0:
        return 0.0, np.zeros(len(alpha))
    z = beta * (alpha - amax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return amax, np.full(len(alpha), 1.0 / len(alpha))
    w = e / s
    val = amax + np.log(s) / beta
    return val, w


def lse_objective_grad(a, beta):
    """Compute LSE-smoothed objective value and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    scale = n / (S * S)
    d = scale * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    w = e / s
    L = dmax + np.log(s) / beta
    # Gradient: d(2n*L)/da_j
    # 2n * [scale * sum_k w_k * d(conv_k)/da_j / S^2 ... ]
    # conv_k = sum_i a_i a_{k-i}, d(conv_k)/da_j = 2*a_{k-j} (approx)

    h = cross_corr(a, w)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

def edge_taper_optimize(a, deadline, verbose=False):
    """Try multiplying by various window functions and keep the best."""
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    x = np.linspace(0.0, 1.0, n)

    windows = {}
    # Sine windows
    for p in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0]:
        w = (np.sin(np.pi * x) ** p).astype(np.float64)
        w[0] = max(w[0], 0.01)
        w[-1] = max(w[-1], 0.01)
        windows["sin_{}".format(p)] = w
    # Cosine taper (only at edges)
    for edge_frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
        m = int(n * edge_frac)
        w = np.ones(n)
        if m > 1:
            taper = (1.0 - np.cos(np.pi * np.arange(m) / m)) / 2.0
            w[:m] = taper
            w[n - m:] = taper[::-1]
        windows["cos_{:.2f}".format(edge_frac)] = w
    # Exponential taper
    for rate in [2.0, 5.0, 10.0, 20.0]:
        w = np.minimum(np.exp(-rate * x), np.exp(-rate * (1.0 - x)))
        w = w / max(w.max(), 1e-10)
        w = np.clip(w, 0.01, 1.0)
        windows["exp_{:.0f}".format(rate)] = w
    # Power taper
    for p in [2.0, 4.0, 8.0]:
        w = (1.0 - np.abs(2.0 * x - 1.0) ** p)
        w = np.clip(w, 0.01, 1.0)
        windows["pow_{:.0f}".format(p)] = w

    for name, w in windows.items():
        if time.monotonic() > deadline:
            break
        new_a = normalize_a(a * w)
        sc = exact_score(new_a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = new_a.copy()
            a = new_a.copy()
            if verbose:
                print("  edge_{}: {:.9f}".format(name, best_sc))

    return float(best_sc), best_a


def fine_polish(a, deadline, verbose=False):
    """Very fine polishing with tiny steps and many parameter combos."""
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]
    eps_fracs = [0.0, 0.002, 0.005, 0.01, 0.02, 0.05]

    for rnd in range(20):
        if time.monotonic() > deadline:
            break
        improved = False
        conv = fast_self_conv(best_a)

        for tk in top_ks:
            if time.monotonic() > deadline:
                break
            active = get_active_indices(conv, top_k=tk, threshold_frac=0.005)
            for ef in eps_fracs:
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    new_a = exp_grad_step(best_a, conv, active, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = new_a.copy()
                        conv = fast_self_conv(best_a)
                        improved = True
                        if verbose:
                            print("  [polish r{}] tk={} eta={:.3f} -> {:.9f}".format(
                                rnd, tk, eta, best_sc))
        if not improved:
            break

    return float(best_sc), best_a


def perturb_sequence(a, rng, strength=0.1):
    """Perturb a sequence for diversification."""
    n = len(a)
    p = normalize_a(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.02, 0.1))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        conv = fast_self_conv(p)
        active = get_active_indices(conv, top_k=15, threshold_frac=0.02)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0
        h = cross_corr(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 3:
        # Smooth perturbation
        k = min(101, max(7, n // 100))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(2.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.5 * noise)
    elif mode == 4:
        # Block perturbation
        nb = int(rng.integers(8, 32))
        block = 1.0 + strength * 0.2 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    else:
        q = p * (0.9 + 0.2 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main optimization function. Returns best sequence found within budget."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 10.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        if n < 512:
            n = 4096
        elif n > 20000:
            n = 16384
        a = normalize_a(parent)
    else:
        n = 10240
        a = np.ones(n) / n
    print("Starting: n={}, initial score={:.9f}".format(n, exact_score(a)))

    # Phase 1: Active-set optimization (~40% of time)
    p1_dead = t_start + 0.40 * total_budget
    print("Phase 1: Active-set optimization (budget {:.0f}s)".format(p1_dead - t_start))
    sc, a = active_set_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    best_sc = sc
    best_a = a.copy()
    print("After Phase 1: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 2: LP direction finding (~20% of time)
    p2_dead = t_start + 0.60 * total_budget
    print("Phase 2: LP direction (budget {:.0f}s)".format(p2_dead - t_start))
    sc, a = lp_direction_optimize(a, deadline=p2_dead, verbose=True, tag="p2")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    a = best_a
    print("After Phase 2: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 3: Edge taper optimization (~5% of time)
    p3_dead = t_start + 0.65 * total_budget
    print("Phase 3: Edge taper (budget {:.0f}s)".format(p3_dead - t_start))
    sc, a = edge_taper_optimize(a, deadline=p3_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    a = best_a
    print("After Phase 3: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 4: LSE-LBFGS refinement (~15% of time)
    p4_dead = t_start + 0.80 * total_budget
    print("Phase 4: LSE-LBFGS (budget {:.0f}s)".format(p4_dead - t_start))
    sc, a = lse_lbfgs_refine(a, deadline=p4_dead, verbose=True, tag="p4")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    a = best_a
    print("After Phase 4: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 5: Active-set again on improved solution (~10% of time)
    p5_dead = t_start + 0.90 * total_budget
    print("Phase 5: Active-set round 2 (budget {:.0f}s)".format(p5_dead - t_start))
    sc, a = active_set_optimize(a, deadline=p5_dead, verbose=True, tag="p5")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    a = best_a
    print("After Phase 5: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 6: Diversification restarts (~5% of time)
    p6_dead = t_start + 0.95 * total_budget
    print("Phase 6: Restarts (budget {:.0f}s)".format(p6_dead - t_start))
    n_restarts = 4
    while time.monotonic() < p6_dead - 30 and n_restarts > 0:
        rem = p6_dead - 30.0 - time.monotonic()
        if rem < 30:
            break
        n_restarts -= 1
        p = perturb_sequence(best_a, rng, strength=float(rng.uniform(0.03, 0.15)))
        chunk_dead = time.monotonic() + rem * 0.75
        sc, seq = active_set_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(n_restarts))
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print("  Restart {}:{:.9f}".format(n_restarts, best_sc))
    print("After Phase 6: {:.9f} (elapsed {:.1f}s)".format(best_sc, time.monotonic() - t_start))

    # Phase 7: Final fine polish
    print("Phase 7: Final polish")
    sc, a = fine_polish(best_a, deadline=deadline - 3, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # Final edge taper try
    sc, a = edge_taper_optimize(best_a, deadline=deadline - 1, verbose=False)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # Final micro-polish
    if time.monotonic() < deadline - 2:
        sc, a = fine_polish(best_a, deadline=deadline - 1, verbose=False)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    print("Best internal: {:.9f}".format(best_sc))

    # Output preparation
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
        S = 1024.0
    out = [float(x) for x in out_arr]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
