Minimax saddle-point method for minimizing 2n*max(self_conv(a))/sum(a)^2.
Uses alternating mirror descent:
  - Primal (a): multiplicative update a <- a * exp(-eta*(corr_pair(a,w) - mean))
  - Dual (w): softmax update w <- softmax(beta * self_conv(a))
With increasing beta (annealing) and adaptive step sizes.
"""

import numpy as np
import time

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def softmax_stable(x, beta):
    x = x - np.max(x)
    x = np.clip(beta * x, -500, 500)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def primal_mirror_step(a, w, eta):
    """Multiplicative update: a_new = a * exp(-eta * (corr_pair(a,w) - reference))"""
    h = corr_pair(a.astype(np.float64), w.astype(np.float64))
    # Use the mean weighted by a as reference (like the Lagrange multiplier)
    S = a.sum()
    if S < 1e-12:
        return a
    ref = float(np.dot(a, h)) / S
    log_ratio = -eta * (h - ref)
    log_ratio = np.clip(log_ratio, -20, 20)
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 0.0)
    return normalize_a(a_new)

def dual_softmax_update(b, beta):
    """Update dual weights from convolution values."""
    return softmax_stable(b, beta)

def dual_topk_update(b, k):
    """Update dual weights: uniform over top-k."""
    m = len(b)
    if k >= m:
        return np.ones(m) / m
    idx = np.argpartition(b, m - k)[m - k:]
    w = np.zeros(m)
    w[idx] = 1.0 / k
    return w

def run_saddle_point(a_init, n_outer, n_inner, beta_start, beta_end, eta_start, eta_end, 
                     use_softmax=True, k_topk=5, deadline=None, verbose=True):
    """Main saddle point iteration."""
    a = normalize_a(a_init.copy())
    n = len(a)
    b = fast_self_conv(a)
    beta = beta_start
    
    best_a = a.copy()
    best_score = exact_score(a)
    
    total_iters = n_outer
    for outer in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal beta and eta
        frac = outer / max(1, total_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start + (eta_end - eta_start) * frac
        
        # Primal: inner iterations of mirror descent
        for inner in range(n_inner):
            if deadline is not None and time.monotonic() > deadline:
                break
            if use_softmax:
                w = dual_softmax_update(b, beta)
            else:
                w = dual_topk_update(b, k_topk)
            a_new = primal_mirror_step(a, w, eta)
            sc = exact_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            b = fast_self_conv(a)
        
        # Update dual
        if use_softmax:
            w = dual_softmax_update(b, beta)
        else:
            w = dual_topk_update(b, k_topk)
        
        if verbose and outer % 20 == 0:
            print(f"  outer={outer}, score={best_score:.8f}, beta={beta:.1f}")
    
    return best_score, best_a

def perturb(a, rng, strength=0.1):
