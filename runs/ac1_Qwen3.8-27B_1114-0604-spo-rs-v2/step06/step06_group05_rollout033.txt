                best_a = a.copy()

    return best_a


def multi_beta_lbfgs(a0, betas, deadline, maxfun_each=30000, maxiter_each=5000):
    """Run LSE-LBFGS with increasing beta values (annealing)."""
    best_a = a0.copy()
    best_sc = compute_score(a0)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        a_opt = run_lbfgs(best_a, beta, maxiter=maxiter_each, maxfun=maxfun_each, deadline=deadline)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    LSE-beta={beta}: score={sc:.8f}")

    return best_a


def diverse_perturb(a0, rng, n_samples=5, strength=0.02):
    """Generate diverse perturbations of a0."""
    results = []
    n = len(a0)
    for i in range(n_samples):
        a = a0.copy()
        # Different perturbation styles
        style = i % 4
        if style == 0:
            a += rng.standard_normal(n) * strength
        elif style == 1:
            # Low-frequency perturbation
            freq_idx = rng.integers(1, n // 4)
            a += strength * np.cos(2 * np.pi * freq_idx * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        elif style == 2:
            # Sparse perturbation
            idx = rng.choice(n, size=n // 10, replace=False)
            a[idx] += rng.standard_normal(len(idx)) * strength * 3
        else:
            # Scale perturbation on random subset
            idx = rng.choice(n, size=n // 5, replace=False)
            a[idx] *= (1 + rng.standard_normal(len(idx)) * strength * 2)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            continue
        a = a / S
        results.append(a)
    return results


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(self_conv)/(sum)^2.
    Multi-phase: LSE-annealing LBFGS, spectral reshaping, peak rebalancing,
    Frank-Wolfe, simulated annealing, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Try multiple n values
    n_candidates = [6962, 8192, 10000, 12000]
    if parent is not None:
        n_candidates.insert(0, len(parent))

    best_overall_score = np.inf
    best_overall_a = None

    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        # Skip if this n takes too long (rough estimate: n^1.5 * 10 should be < remaining time)
        remaining = deadline - time.monotonic()
        if n > 15000 and remaining < 600:
            continue

        print(f"\n{'='*60}")
        print(f"Optimizing with n={n} (time left: {remaining:.0f}s)")
        print(f"{'='*60}")

        # Initialize
        if parent is not None:
            if len(parent) == n:
                a = parent.copy()
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, parent).copy()
        else:
            a = np.ones(n) * 0.5

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            a = np.ones(n) / n
        a = a / S

        best_a = a.copy()
        best_sc = compute_score(a)
        print(f"  Initial score: {best_sc:.8f}")

        def update_best(sc, candidate_a):
            nonlocal best_a, best_sc
            if sc < best_sc:
                best_a = candidate_a.copy()
                best_sc = sc
                return True
            return False

        # Phase 1: Quick LSE warmup (10%)
        phase1_dl = t_start + 0.10 * (total - 60.0)
        print(f"  Phase 1: LSE warmup (until +{phase1_dl - t_start:.0f}s)")
        a = run_lbfgs(a, 500, maxiter=2000, maxfun=10000, deadline=phase1_dl)
        a = np.maximum(a, 0.0)
        S = a.sum()
        a = a / max(S, 1e-12)
        update_best(compute_score(a), a)
        print(f"  After Phase 1: {best_sc:.8f}")

        # Phase 2: Beta annealing (40%)
        phase2_dl = t_start + 0.50 * (total - 60.0)
        print(f"  Phase 2: Beta annealing (until +{phase2_dl - t_start:.0f}s)")
        betas = [1000, 3000, 10000, 30000, 100000, 300000, 1000000, 3000000]
        a = multi_beta_lbfgs(best_a, betas, phase2_dl, maxfun_each=40000, maxiter_each=6000)
        update_best(compute_score(a), a)
        print(f"  After Phase 2: {best_sc:.8f}")

        # Phase 3: Spectral reshape + local refinement (20%)
        phase3_dl = t_start + 0.70 * (total - 60.0)
        print(f"  Phase 3: Spectral + local (until +{phase3_dl - t_start:.0f}s)")

        # Spectral reshape
        a_spec = spectral_reshape(best_a, n_iters=150, strength=0.3, seed=seed)
        sc_spec = compute_score(a_spec)
        update_best(sc_spec, a_spec)
        print(f"    Spectral: {best_sc:.8f}")

        # LSE after spectral
        if time.monotonic() < phase3_dl:
            a_after = run_lbfgs(best_a, 200000, maxiter=3000, maxfun=15000, deadline=phase3_dl)
            update_best(compute_score(a_after), a_after)

        # Peak rebalance
        if time.monotonic() < phase3_dl:
            a_pr = peak_rebalance(best_a, n_rounds=150, top_k=30, seed=seed + 1)
            update_best(compute_score(a_pr), a_pr)
            print(f"    After peak rebalance: {best_sc:.8f}")

        # LSE after peak rebalance
        if time.monotonic() < phase3_dl:
            a_after2 = run_lbfgs(best_a, 500000, maxiter=3000, maxfun=15000, deadline=phase3_dl)
            update_best(compute_score(a_after2), a_after2)

        # Phase 4: Frank-Wolfe + Simulated Annealing (15%)
        phase4_dl = t_start + 0.85 * (total - 60.0)
        print(f"  Phase 4: FW + SA (until +{phase4_dl - t_start:.0f}s)")

        # Frank-Wolfe
        if time.monotonic() < phase4_dl:
            a_fw = frank_wolfe(best_a, n_iters=2000, n_peaks=3)
            update_best(compute_score(a_fw), a_fw)
            print(f"    After FW: {best_sc:.8f}")

        # LSE after FW
        if time.monotonic() < phase4_dl:
            a_after3 = run_lbfgs(best_a, 1000000, maxiter=3000, maxfun=10000, deadline=phase4_dl)
            update_best(compute_score(a_after3), a_after3)

        # Simulated annealing
        if time.monotonic() < phase4_dl:
            a_sa = sim_anneal(best_a, n_iters=2000, top_k=10, t_init=0.03, t_final=0.001, seed=seed + 2)
            update_best(compute_score(a_sa), a_sa)
            print(f"    After SA: {best_sc:.8f}")

        # Phase 5: Diverse restarts + final polish (15%)
        print(f"  Phase 5: Diverse restarts + polish (until +{deadline - t_start:.0f}s)")

        # Try diverse perturbations
        if time.monotonic() < deadline:
            perturbations = diverse_perturb(best_a, rng, n_samples=4, strength=0.015)
            for i, a_p in enumerate(perturbations):
                if time.monotonic() > deadline:
                    break
                # Quick spectral + LSE on perturbation
                a_sp = spectral_reshape(a_p, n_iters=50, strength=0.2, seed=seed + 100 + i)
                update_best(compute_score(a_sp), a_sp)
                if time.monotonic() > deadline:
                    break
                a_opt = run_lbfgs(best_a, 500000, maxiter=2000, maxfun=8000, deadline=deadline)
                update_best(compute_score(a_opt), a_opt)
                print(f"    Restart {i}: {best_sc:.8f}")

        # Final very-high-beta LSE
        if time.monotonic() < deadline - 30:
            for beta_final in [5000000, 10000000]:
                if time.monotonic() > deadline - 30:
                    break
                a_fin = run_lbfgs(best_a, beta_final, maxiter=2000, maxfun=8000, deadline=deadline - 30)
                update_best(compute_score(a_fin), a_fin)
                print(f"    Final LSE beta={beta_final}: {best_sc:.8f}")

        # Update overall best
        if best_sc < best_overall_score:
            best_overall_score = best_sc
            best_overall_a = best_a.copy()
            print(f"\n  *** NEW OVERALL BEST: {best_sc:.8f} (n={n}) ***")
        else:
            print(f"  n={n} best: {best_sc:.8f} (overall: {best_overall_score:.8f})")

    # Final output
    if best_overall_a is None:
        out = [0.5] * 6962
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_overall_a]

    if sum(out) < 0.01:
        out = [0.5] * 6962

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_overall_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'='*60}")
    return out
