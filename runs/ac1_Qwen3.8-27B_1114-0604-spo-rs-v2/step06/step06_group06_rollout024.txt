def compute_active_weights_v2(conv, top_k=10, eps_frac=0.02, temperature=0.1):

        # Compute weights (blend of active set and CVaR)
        w_active = compute_active_weights_v2(conv, top_k=top_k, eps_frac=eps_frac, temperature=0.1)
        w_cvar = cvar_weight(conv, cvar_k, beta=cvar_beta)
        w = 0.6 * w_active + 0.4 * w_cvar
        
        # Try multiple step sizes
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            hmax = 1e-15
        
        best_a_iter = a
        best_sc_iter = sc
        best_eta = eta
        
        for eta_try in [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0]:
            if eta_try < 0.001:
                continue
            # Exponentiated gradient step with momentum
            v = momentum * v - eta_try * h / hmax
            log_a_new = log_a + v
            log_a_new -= float(np.max(log_a_new))
            new_a = np.exp(log_a_new)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_iter:
                best_sc_iter = new_sc
                best_a_iter = new_a
                best_eta = eta_try
        
        if best_sc_iter < best_sc - 1e-10:
            best_sc = best_sc_iter
            best_a = best_a_iter
            a = best_a_iter
            eta = best_eta
            stagnant = 0
            # Update log_a and v
            log_a = np.log(a + 1e-300)
            log_a -= float(np.max(log_a))
            if verbose and it % 100 == 0:
                print("  [{} i{}] score={:.9f} top_k={} eps={:.3f} eta={:.3f}".format(
                    tag, it, best_sc, top_k, eps_frac, eta))
        else:
            stagnant += 1
            # Without momentum, just keep v small
            v = v * 0.5
            if stagnant >= max_stagnant:
                break
    
    return float(best_sc), best_a

def lse_score_and_grad(a, beta):
    """LSE-smoothed objective and gradient for the normalized problem."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    # d_k = n * c_k / S^2 (so that 2 * max(d) = our normalized objective / 2)
    # Actually score = 2*n*max(c)/S^2, so let's define f = 2*n*max(c)/S^2
    # LSE approximation: f ≈ 2*n/S^2 * (max(c) + (1/beta)*log(sum(exp(beta*c))))
    # = 2*n/S^2 * LSE_beta(c)
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = cmax + np.log(sum_e) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient: d(score)/d(a[i]) = 2*n/S^2 * d(L)/d(a[i]) - 4*n*L/S^3 * S_i
    # where S_i = d(S)/d(a[i]) = 1
    # d(L)/d(a[i]) = w_k * d(c_k)/d(a[i]) = sum_k w_k * 2*a[k-i] (as computed above)
    # = 2 * corr_pair(a, w)[i]
    h = corr_pair(a, w)
    dL_da = 2.0 * h
    grad = 2.0 * n * dL_da / (S * S) - 4.0 * n * L / (S * S)  # since S_i = 1 for all i
    # Wait, that's not right. Let me redo.
    # score = 2*n*L(a) / S(a)^2
    # d(score)/d(a[i]) = 2*n * d(L)/d(a[i]) / S^2 - 2*n*L * d(1/S^2)/d(a[i])
    # d(1/S^2)/d(a[i]) = -2/S^3 * dS/d(a[i]) = -2/S^3
    # So: grad_i = 2*n * dL_da[i] / S^2 + 4*n*L / S^3
    # Hmm wait: d(1/S^2)/dS = -2/S^3, and dS/d(a[i]) = 1
    # So d(2nL/S^2)/d(a[i]) = 2n * (dL_da[i] * S^2 - L * 2S) / S^4
    # = 2n * (dL_da[i] * S - 2L) / S^3
    grad = 2.0 * n * (dL_da * S - 2.0 * L) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(score), grad

def run_lbfgs_v2(a, beta, deadline, maxfun=500):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        x = np.clip(x, 0.0, None)
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(np.clip(res.x, 0, None)) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def generate_inits(n, rng, parent, cap=30):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        # Add small floor to ensure positivity
        arr = arr + 0.001
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Scaled versions
        for s in [0.5, 1.0, 2.0]:
            add(pr * s)

    add(np.ones(n))
    
    # Various shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    
    # Power tapers (asymmetric)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        add(x ** p)
        add((1.0 - x) ** p)
        add(x ** p + (1.0 - x) ** p)
    
    # Exponential
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * np.abs(x - 0.5)))
    
    # Cosine modulations
    for f in (1, 2, 3, 4, 5, 7):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.sin(np.pi * f * x))
    
    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2, 0.3):
        for center in (0.15, 0.25, 0.35, 0.4):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center)) / width) ** 2)))
    
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    
    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits
