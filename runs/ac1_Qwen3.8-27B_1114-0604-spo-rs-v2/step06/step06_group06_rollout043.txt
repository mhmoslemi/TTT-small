"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using progressive LSE-LBFGS,
Nesterov-accelerated mirror descent with adaptive active set,
and coordinate descent polishing. Improvements over prior: agressive
beta schedule, Nesterov momentum, coordinate descent, better inits.

import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def fast_cross_conv(a, w):
    """Compute g[i] = sum_k w[k] * a[k-i] for valid indices, i.e. the
    cross-correlation of w and a truncated to n values."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    LSE = cmax + np.log(sum_e) / beta
    h = fast_cross_conv(a, w)
    grad_LSE = (2.0 / beta) * h
    grad = (2.0 * n / (S * S)) * (grad_LSE - 2.0 * LSE)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e10, 1e10)
    score = 2.0 * n * LSE / (S * S)
    return float(score), grad


def run_lbfgs(a, beta, deadline, maxfun=1000, verbose=False, tag=""):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    call_count = [0]

    def fun(x):
        call_count[0] += 1
        x = np.clip(x, 0.0, None)
        return lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    maxiter = max(20, maxfun // 25)
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": maxiter,
        "maxfun": int(maxfun),
        "maxls": 30,
        "disp": False,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    if verbose:
        print("  [{}] beta={:.0f} funs={} score={:.9f}".format(tag, beta, call_count[0], sc))
    return float(sc), a_out


def compute_active_weights(conv, top_k=10, eps_frac=0.01):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.05 * conv.size))
    if active.sum() > max_count:
        active = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active[top_idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def nesterov_mirror_descent(a, deadline, verbose=True, tag=""):
    """Nesterov-accelerated mirror descent with active set weighting."""
    n = len(a)
    a = normalize_a(a).copy()
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    # Nesterov variables
    y = a.copy()  # extrapolated point
    beta_mom = 1.0  # momentum coefficient (updated each iter)
    it = 0
    eta = 0.15
    stagnant = 0
    max_stagnant = 50

    top_k = 5
    eps_frac = 0.03

    while time.monotonic() < deadline - 5.0:
        it += 1
        # Nesterov extrapolation
        t = float(it)
        beta_mom = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (1.0 + beta_mom) / (beta_mom + 1.0)  # simplified
        # Standard Nesterov: y = x + ((t-1)/t) * (x - x_prev)
        # We use a simpler scheme: y = x + momentum * (x - x_prev)
        y = a + 0.5 * (a - best_a) if it > 1 else a.copy()
        y = np.clip(y, 1e-30, None)

        conv_y = fast_self_conv(y)
        w = compute_active_weights(conv_y, top_k=top_k, eps_frac=eps_frac)
        h = fast_cross_conv(y, w)
        hmax = max(float(np.max(h)), 1e-15)

        # Mirror descent step in log-space
        log_a = np.log(a + 1e-300)
        log_a -= eta * h / hmax
        log_a -= float(np.max(log_a))
        new_a = np.exp(log_a)
        new_a = normalize_a(new_a)
        new_sc = exact_score(new_a)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = new_a.copy()
            a = new_a
            stagnant = 0
            if verbose and it % 200 == 0:
                print("  [{}] iter={} score={:.9f} eta={:.3f}".format(tag, it, best_sc, eta))
        else:
            stagnant += 1
            if stagnant % 20 == 0:
                eta *= 0.8
                if stagnant > max_stagnant:
                    # Adapt active set
                    top_k = min(int(top_k * 1.4) + 1, 80)
                    eps_frac = min(eps_frac * 1.2, 0.12)
                    eta = max(0.03, eta * 0.7)
                    stagnant = 0

        # Periodically try variance reduction
        if it % 50 == 0:
            d_sorted = np.sort(conv_y)[::-1][:20]
            mean_top = float(np.mean(d_sorted))
            excess = np.maximum(conv_y - mean_top, 0.0)
            total_excess = float(np.sum(excess))
            if total_excess > 1e-15:
                w_vr = excess / total_excess
                h_vr = fast_cross_conv(y, w_vr)
                hmax_vr = max(float(np.max(h_vr)), 1e-15)
                log_a_vr = np.log(a + 1e-300)
                log_a_vr -= 0.08 * h_vr / hmax_vr
                log_a_vr -= float(np.max(log_a_vr))
                vr_a = normalize_a(np.exp(log_a_vr))
                vr_sc = exact_score(vr_a)
                if vr_sc < best_sc - 1e-12:
                    best_sc = vr_sc
                    best_a = vr_a.copy()
                    a = vr_a

    return float(best_sc), best_a


def coordinate_descent(a, deadline, verbose=True, tag=""):
    """Coordinate descent: optimize small blocks of coordinates at a time."""
    n = len(a)
    a = normalize_a(a).copy()
    best_a = a.copy()
    best_sc = exact_score(a)

    block_sizes = [1, 2, 3, 4, 5]
    iterations = 0

    while time.monotonic() < deadline - 5.0:
        iterations += 1
        improved = False
        # Try different block sizes
        for bs in block_sizes:
            if time.monotonic() > deadline - 5.0:
                break
            # Pick random starting position
            for trial in range(max(1, n // (bs * 50))):
                if time.monotonic() > deadline - 10.0:
                    break
                start = int(np.random.randint(0, max(1, n - bs)))
                idx = list(range(start, min(start + bs, n)))

                # Try several multipliers for the block
                orig_vals = a[idx].copy()
                S_rest = float(a.sum()) - float(orig_vals.sum())

                for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0, 0.0]:
                    new_a = a.copy()
                    new_vals = orig_vals * mult
                    new_a[idx] = new_vals
                    new_a = np.clip(new_a, 0.0, None)
                    S_new = float(new_a.sum())
                    if S_new < 1e-12:
                        continue
                    new_a = new_a / S_new
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = new_a.copy()
                        a = new_a
                        improved = True

        if not improved and iterations > 20:
            break

        if verbose and iterations % 10 == 0:
            print("  [{}] cd iter={} score={:.9f}".format(tag, iterations, best_sc))

    return float(best_sc), best_a


def progressive_lse_lbfgs(a, deadline, verbose=True, tag=""):
    """Progressive LSE-LBFGS: start at low beta, increase gradually."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Adaptive beta schedule
    betas = [100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]

    for beta in betas:
        if time.monotonic() > deadline - 10.0:
            break
        rem = deadline - 10.0 - time.monotonic()
        if rem < 15:
            break
        # Scale maxfun with remaining time
        maxfun = max(100, int(rem * 0.5 / 0.02))
        maxfun = min(maxfun, 3000)

        sc, a_new = run_lbfgs(best_a, beta, deadline, maxfun=maxfun,
                               verbose=verbose, tag=tag)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # If no improvement, try with more iterations at same beta
            if rem > 30:
                maxfun2 = max(200, int(rem * 0.3 / 0.02))
                sc2, a2 = run_lbfgs(best_a, beta, deadline, maxfun=maxfun2,
                                     verbose=verbose, tag=tag + "-retry")
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a2.copy()

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=30):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            # Resample
            x_old = np.linspace(0.0, 1.0, arr.size)
            arr = np.interp(x, x_old, arr)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        if float(arr.sum()) < 1e-12:
            arr = np.ones(n)
        inits.append(normalize_a(arr))

    if parent is not None:
        pr = np.interp(x, np.linspace(0, 1, parent.size), parent)
        add(pr)
        add(pr[::-1])
        # Perturbed parent
        add(pr * (1.0 + 0.1 * rng.standard_normal(n)))
        add(pr[::-1] * (1.0 + 0.1 * rng.standard_normal(n)))

    add(np.ones(n))
    # Linear ramps
    add(x + 0.01)
    add(1.0 - x + 0.01)
    # Quadratic
    add((x * (1.0 - x)) + 0.01)
    add((2.0 * x - 1.0) ** 2 + 0.01)
    # Cubic ramp
    add(x**3 + 0.01)
    add((1.0 - x)**3 + 0.01)
    # Power ramps
    for p in (0.3, 0.5, 2.0, 3.0, 5.0):
        add(x**p + 0.01)
        add((1.0 - x)**p + 0.01)
    # Cosine shapes
    for f in (1, 2, 3, 4, 5):
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * f * x))
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * f * x + np.pi / 3))
    # Sine absolute
    for f in (1, 2, 3, 4):
        add(0.5 + 0.5 * np.abs(np.sin(np.pi * f * x)))
    # Exponential
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    # Bimodal
    for width in (0.05, 0.1, 0.15, 0.2):
        for center in (0.2, 0.3, 0.4):
            add(np.exp(-(((x - center) / width) ** 2)) +
                np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Random smooth
    Nf = next_power_of_two(2 * n)
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= float(noise.mean())
        add(1.0 + 0.5 * noise)

    return inits


def resample_to_length(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(a.size)
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a)
