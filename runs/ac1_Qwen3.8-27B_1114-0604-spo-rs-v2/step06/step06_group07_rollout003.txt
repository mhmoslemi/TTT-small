import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

                           temperature=0.5, use_momentum=True, momentum=0.9):

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()

    if S < 1e-12:

        S = 1.0

    best_a = a.copy()
    best_sc = compute_score(a)
    g_ema = np.zeros(n)  # Exponentially averaged subgradient
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        step = step_start + (step_end - step_start) * progress
        temp = temperature * (0.5 + progress)
        
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        M_val = float(c.max())

            best_sc = sc
            best_a = a.copy()

        if K >= len(c):

        else:
            top_idx = np.argpartition(c, -K)[-K:]

        w = np.exp(c_shifted)

        g = np.zeros(n)
        for k, wk in zip(top_idx, w):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                rev = a[i_start:i_end + 1][::-1]
                # Wait: for i in [i_start, i_end], a[k-i]
                # i_start = max(0,k-n+1), i_end = min(k,n-1)
                # k-i_start = min(k, n-1), k-i_end = max(0, k-n+1)

                # This is a[max(0,k-n+1):min(k,n-1)+1][::-1]

        if use_momentum:
            g_ema = momentum * g_ema + (1 - momentum) * g
            g_use = g_ema
        else:

def subgradient_descent(a, n_iters=20000, K=10, step_start=0.005, step_end=0.0005,
                        decay=0.99999):
    """Additive subgradient descent with top-K peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_start
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        step_curr = step_start + (step_end - step_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        M_val = float(c.max())
        sc = 2.0 * n * M_val / S / S
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-K
        if K >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_end >= i_start:
                g[i_start:i_end + 1] += 2.0 * a[i_start:i_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_use = min(step_curr, max((sc - 1.001) / (gn * gn + 1e-30) * n, 1e-6))
        step_use = min(step_use, 0.01)
        
        a -= step_use * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a
