"""
Step function minimization using adaptive spectral shaping, Adam-based
subgradient descent, high-precision L-BFGS, and coordinate descent polish.
Key innovations: (1) adaptive alpha in spectral shaping based on spectral flatness,
(2) Adam optimizer for per-coordinate adaptive learning rates in time-domain descent,
(3) multi-scale L-BFGS with increasing beta, (4) coordinate descent with 1D line search.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_flatness(amp):
    """Compute spectral flatness (geometric mean / arithmetic mean)."""
    amp = np.maximum(amp, 1e-20)
    log_amp = np.log(amp)
    geo_mean = np.exp(np.mean(log_amp))
    arith_mean = np.mean(amp)
    return geo_mean / arith_mean


def fienup_adaptive(a, n_iters=500, use_flatness_adapt=True):
    """
    Adaptive geometric spectral shaping where alpha is modulated by
    the current spectral flatness. When the spectrum is already flat,
    use a smaller alpha to avoid destroying the non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        # Adaptive alpha based on flatness
        if use_flatness_adapt:
            # Consider only non-DC bins for flatness
            if len(amp) > 1:
                flat = spectral_flatness(amp[1:])
            else:
                flat = 1.0
            # Map flatness [0,1] to alpha [0.05, 0.8]
            alpha = 0.05 + 0.75 * (1.0 - flat)
        else:
            alpha = 0.1 + 0.7 * progress

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Soft non-negativity: use a smooth ReLU approximation
        # a_new = a_new * sigmoid(k*a_new) for smoother gate
        k = 20.0 + 20.0 * progress  # increase sharpness over time
        gate = 1.0 / (1.0 + np.exp(-k * np.clip(a_new, -50, 50)))
        a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def adam_subgradient_descent(a, n_iters=5000, n_peaks=10, lr=0.01,
                              beta1=0.9, beta2=0.999, eps=1e-8,
                              backtracking=True):
    """
    Adam-optimized subgradient descent targeting top-K peaks of the convolution.
    Uses backtracking line search for robustness.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for it in range(n_iters):
        # Compute convolution and gradient
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_curr = float(a.sum())
        M_val = float(c.max())
        sc_curr = 2.0 * n * M_val / (S_curr * S_curr)

        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S_curr * S_curr
        S3 = S2 * S_curr
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam step
        step_dir = -m_hat / (np.sqrt(v_hat) + eps)
        step_size = lr

        if backtracking:
            # Backtracking line search
            alpha_bt = 1.0
            for _ in range(20):
                a_trial = a + alpha_bt * step_size * step_dir
                a_trial = np.maximum(a_trial, 0.0)
                S_trial = float(a_trial.sum())
                if S_trial < 1e-12:
                    alpha_bt *= 0.5
                    continue
                a_trial = a_trial / S_trial
                sc_trial = compute_score(a_trial)
                if sc_trial <= sc_curr + 1e-10:
                    break
                alpha_bt *= 0.5
            step_size *= alpha_bt

        a = a + step_size * step_dir
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Decay learning rate
        lr *= 0.9999

    return best_a


def lse_objective_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent_polish(a, n_passes=10, n_samples=200):
    """
    Coordinate descent: for a random subset of coordinates, find the optimal
    1D step that minimizes the objective.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        # Sample random coordinates
        idx = rng.choice(n, size=min(n_samples, n), replace=False)
        
        for i in idx:
            # Compute current convolution
            c = fast_conv(a)
            M_val = float(c.max())
            S_curr = float(a.sum())
            sc_curr = 2.0 * n * M_val / (S_curr * S_curr)

            # Try different values for a[i]
            # The current value is a[i]. Try scaling it.
            best_val = a[i]
            best_local_sc = sc_curr
            
            # Try a range of multipliers
            if a[i] > 1e-10:
                multipliers = np.array([0.5, 0.7, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.3, 1.5])
            else:
                multipliers = np.array([0.0, 0.001, 0.01, 0.05, 0.1, 0.5])
                multipliers = multipliers * (S_curr / n)  # scale to reasonable level

            for mult in multipliers:
                a[i] = max(0.0, a[i] * mult) if a[i] > 1e-10 else mult
                S_trial = float(a.sum())
                if S_trial < 1e-12:
                    continue
                # Quick score computation
                c_trial = fast_conv(a)
                M_trial = float(c_trial.max())
                sc_trial = 2.0 * n * M_trial / (S_trial * S_trial)
                if sc_trial < best_local_sc:
                    best_local_sc = sc_trial
                    best_val = a[i]
            
            a[i] = best_val
        
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shaving_spectral(a, n_iters=200, n_peaks=5):
    """
    Instead of uniform spectral flattening, specifically notch out the
    frequency components that contribute most to the time-domain peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # For each peak, identify which frequency bins contribute most
        # by looking at the cross-correlation between the peak pattern and the spectrum
        amp_new = amp.copy()
        
        # Simple approach: reduce amplitude in bins where |A| is large
        # (these contribute most to time-domain peaks)
        threshold = np.percentile(amp[1:], 80) if len(amp) > 1 else amp[0]
        reduction = 0.95 + 0.03 * (it / n_iters)  # gradually less aggressive
        amp_new[amp > threshold] *= reduction

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.maximum(0.2 + 0.6 * np.cos(4.0 * np.pi * (x - 0.25)) +
                            0.6 * np.cos(4.0 * np.pi * (x - 0.75)), 0.01))
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    
    # Random with structure
    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits

def fast_conv(a):
    """Fast linear convolution of a with itself via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Fast linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def weighted_subgradient_descent(a, n_iters=5000, n_peaks=20, lr=0.01,
                                  decay_rate=0.5, momentum_bias=0.9,
                                  variance_bias=0.999, eps=1e-8):
    """
    Adam-optimized subgradient descent with exponential peak weighting.
    Higher peaks get more weight in the gradient, creating a focused
    but smooth optimization landscape.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_curr = float(a.sum())
        M_val = float(c.max())
        sc_curr = 2.0 * n * M_val / (S_curr * S_curr)

        # Select top peaks and weight them exponentially
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(top_idx))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_sorted = top_idx[np.argsort(c[top_idx])[::-1]]
            # Exponential weighting: first peak gets weight 1, second gets decay_rate, etc.
            positions = np.arange(len(top_sorted))
            weights = decay_rate ** positions
            top_idx = top_sorted

        # Compute weighted subgradient
        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]

        S2 = S_curr * S_curr
        S3 = S2 * S_curr
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Adam update
        t += 1
        m = momentum_bias * m + (1.0 - momentum_bias) * grad
        v = variance_bias * v + (1.0 - variance_bias) * grad * grad
        m_hat = m / (1.0 - momentum_bias ** t)
        v_hat = v / (1.0 - variance_bias ** t)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Periodic score check (every 50 iterations to save time)
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        # Adaptive learning rate decay
        lr *= 0.99995

    return best_a


def backtracking_subgradient(a, n_iters=3000, n_peaks=10, lr_init=0.02):
    """
    Subgradient descent with backtracking line search.
    More robust than Adam for non-smooth objectives.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_curr = float(a.sum())
        M_val = float(c.max())
        sc_curr = 2.0 * n * M_val / (S_curr * S_curr)

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S_curr * S_curr
        S3 = S2 * S_curr
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Backtracking line search
        alpha = 1.0
        for _ in range(10):
            a_trial = a - alpha * lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                alpha *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial <= sc_curr + 1e-8:
                break
            alpha *= 0.5

        a = a - alpha * lr * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            lr *= 0.95

    return best_a

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_passes=5, n_samples=500, seed=42):
    """
    Coordinate descent: for sampled coordinates, try multiple 1D steps
    and keep the best one.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(seed)

    for p in range(n_passes):
        idx = rng.choice(n, size=min(n_samples, n), replace=False)

        for i in idx:
            c = fast_conv(a)
            M_val = float(c.max())
            S_curr = float(a.sum())
            sc_curr = 2.0 * n * M_val / (S_curr * S_curr)

            best_val = a[i]
            best_local = sc_curr

            # Try scaling the current value
            if a[i] > 1e-12:
                multipliers = np.array([0.5, 0.75, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5])
            else:
                avg_val = S_curr / n
                multipliers = np.array([0.0, 0.1 * avg_val, 0.5 * avg_val, avg_val, 2.0 * avg_val])

            for mult in multipliers:
                old_val = a[i]
                if a[i] > 1e-12:
                    a[i] = max(0.0, a[i] * mult)
                else:
                    a[i] = max(0.0, mult)
                S_trial = float(a.sum())
                if S_trial < 1e-12:
                    a[i] = old_val
                    continue
                c_trial = fast_conv(a)
                sc_trial = 2.0 * n * float(np.max(c_trial)) / (S_trial * S_trial)
                if sc_trial < best_local:
                    best_local = sc_trial
                    best_val = a[i]
                else:
                    a[i] = old_val

            a[i] = best_val

        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        M_val = float(c.max())

        # Select top peaks with exponential weighting
        if n_peaks >= len(c):
            top_idx = np.argsort(c)[::-1]
            n_sel = len(top_idx)
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
            n_sel = len(top_idx)

        positions = np.arange(n_sel, dtype=np.float64)
        weights = decay_rate ** positions

        # Compute weighted subgradient
        sg = np.zeros(n)
        for k, w in zip(top_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * w * seg[::-1]

        S2 = S_curr * S_curr
        S3 = S2 * S_curr
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Adam update
        t += 1
        m = momentum_bias * m + (1.0 - momentum_bias) * grad
        v = variance_bias * v + (1.0 - variance_bias) * grad * grad
        m_hat = m / (1.0 - momentum_bias ** t)
        v_hat = v / (1.0 - variance_bias ** t)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Periodic score evaluation
        if it % 100 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        lr *= 0.99998

    return best_a
