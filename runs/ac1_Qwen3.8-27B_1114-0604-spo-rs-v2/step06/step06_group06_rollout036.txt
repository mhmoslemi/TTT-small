import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:2 * n - 1]
    return np.maximum(out, 0.0)  # Small negative values from numerical error


def score(a):
    """Compute the objective value."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

    a: non-negative sequence with sum=1
    threshold_frac: target peak as fraction of current peak (e.g., 0.9 means reduce to 90%)

    Returns: new sequence (non-negative, sum=1)

    P = A * A  # rfft of self-convolution (only positive freqs)
    # Full FFT for proper handling
    # Actually, let's use full complex FFT for clarity
    Af = np.fft.fft(a, N)
    Pf = Af * Af
    r = np.fft.ifft(Pf, N).real[:2 * n - 1]
    r = np.maximum(r, 0.0)
    
    rmax = float(np.max(r))
    if rmax < 1e-15:

    r_target = np.minimum(r, target)

    r_target_full[:2 * n - 1] = r_target
