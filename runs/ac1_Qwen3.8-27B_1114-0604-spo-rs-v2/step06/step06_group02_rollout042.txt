    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def make_exponential(n, r, rng):
    a = r ** np.arange(n)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S

def make_ramp(n, direction=1, rng=None):
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    a = np.maximum(a, 0)
    return a / a.sum()

def make_cosine(n, center_frac=0.25, width_frac=0.5, rng=None):
    x = np.arange(n) / n
    center = center_frac
    width = width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S

def make_bimodal(n, rng):
    # Two bumps at different positions
    x = np.arange(n) / n
    c1, c2 = 0.1, 0.6
    w1, w2 = 0.15, 0.2
    a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.5 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()

def projected_gradient_step(a, beta, lr, deadline=None):
    """Single projected gradient step on the LSE-smoothed objective."""
    n = len(a)
    score, grad = lse_objective_and_grad(a, beta)
    
    # Project gradient onto the tangent space of sum(a)=1
    # The constraint is sum(a) = 1, so the normal is (1,1,...,1)
    # Projected grad = grad - (mean(grad)) * 1
    grad_proj = grad - np.mean(grad)
    
    a_new = a - lr * grad_proj
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a, score
    a_new = a_new / S
    return a_new, score

def multi_length_search(best_a, deadline, rng, lengths):

    best_sc = compute_score(best_a)
    best_result = best_a.copy()

            break
        if n == len(best_a):
            continue
        
        # Interpolate
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, best_a)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:

def stochastic_refine(a, rng, n_trials=30, deadline=None):
    """
    Stochastic local search with diverse perturbation strategies.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for t in range(n_trials):
        if deadline and time.monotonic() > deadline:
            break
        
        strategy = t % 5
        if strategy == 0:
            # Gaussian noise
            a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        elif strategy == 1:
            # Improve the worst peak contributors
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            # Scale down top contributors to the max peak
            js = np.arange(j_start, j_end + 1)
            partners = k_star - js
            contribs = a_pert[js] * a_pert[partners]
            top_idx = np.argsort(contribs)[::-1][:10]
            for idx in top_idx:
                a_pert[js[idx]] *= 0.8
        elif strategy == 2:
            # Frequency-domain perturbation
            M_fft = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, M_fft)
            # Add small random phase perturbation
            phase = 0.02 * rng.standard_normal(len(A))
            A_pert = A * np.exp(1j * phase)
            a_pert = np.fft.irfft(A_pert, M_fft)[:n]
        elif strategy == 3:
            # Block swap
            block_size = max(1, n // 16)
            b1 = int(rng.integers(0, n - 2 * block_size))
            b2 = int(rng.integers(b1 + block_size, n - block_size))
            a_pert = a.copy()
            a_pert[b1:b1+block_size], a_pert[b2:b2+block_size] = \
                a_pert[b2:b2+block_size].copy(), a_pert[b1:b1+block_size].copy()
        else:
            # Multiplicative scaling of a segment
            seg_len = max(1, n // 8)
            start = int(rng.integers(0, n - seg_len))
            scale = 1.0 + 0.1 * rng.standard_normal()
            a_pert = a.copy()
            a_pert[start:start+seg_len] *= scale
        
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        
        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Stochastic trial {t}: {sc:.8f}")
    
    return best_a, best_sc

`score = 2n * max(c) / S^2`

If we normalize `S = 1`, then `score = 2n * max(c)`. And `sum(c) = S^2 = 1`.

"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = n + (2 * n - 1) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def make_exponential(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_ramp(n, direction=1):
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def make_gaussian(n, center_frac=0.3, width_frac=0.3):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center_frac) / width_frac) ** 2)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def nesterov_projected(a, beta, n_iters=5000, lr_init=0.01, deadline=None):
    """Nesterov-accelerated projected gradient on LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    y = a.copy()
    momentum = 0.9
    lr = lr_init
    
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        if it % 500 == 0 and it > 0:
            momentum = max(0.5, momentum * 0.999)
            lr *= 0.95
        
        _, grad = lse_obj_grad(y, beta)
        grad_proj = grad - np.mean(grad)
        
        t_new = (it + 1 + np.sqrt(1 + 4 * it * it)) / 2
        beta_nest = (it - 1 + momentum) / t_new if it > 0 else 0
        
        y_new = y - lr * grad_proj + beta_nest * (y - a)
        y_new = np.maximum(y_new, 0.0)
        S_new = y_new.sum()
        if S_new < 1e-12:
            break
        y_new = y_new / S_new
        a = y_new
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc
    return best_a


def peak_tracking_cd(a, n_sweeps=100, deadline=None):
    """
    Coordinate descent that tracks which peaks are active and focuses
    optimization on reducing them. Uses multiplicative updates for
    non-negativity preservation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Identify active peaks (top peaks)
        top_k = min(10, len(c))
        active_idx = np.argsort(c)[::-1][:top_k]
        
        # For each active peak, find and reduce top contributors
        for k in active_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            
            js = np.arange(j_start, j_end + 1)
            partners = k - js
            contribs = a[js] * a[partners]
            total = contribs.sum()
            if total < 1e-15:
                continue
            
            # Get top contributors
            n_top = min(15, len(contribs))
            order = np.argsort(contribs)[::-1][:n_top]
            
            for idx in order:
                j = js[idx]
                a_j = a[j]
                if a_j < 1e-12:
                    continue
                # Try reducing a_j
                for f in [0.85, 0.9, 0.95]:
                    a[j] = a_j * f
                    S_tmp = float(a.sum())
                    if S_tmp < 1e-12:
                        a[j] = a_j
                        break
                    a_n = a / S_tmp
                    c_tmp = fast_conv(a_n)
                    sc_tmp = 2.0 * n * float(c_tmp.max()) / (S_tmp * S_tmp)
                    if sc_tmp < sc:
                        sc = sc_tmp
                        a = a_n
                    else:
                        a[j] = a_j
                        break
        
        # Also try increasing elements that don't contribute to active peaks
        # (to Spread energy)
        if sweep % 3 == 0:
            c = fast_conv(a)
            c_max = float(c.max())
            # Find elements that contribute least to active peaks
            low_contrib = np.zeros(n)
            for k in active_idx[:3]:
                k = int(k)
                j_start = max(0, k - n + 1)
                j_end = min(k, n - 1)
                js = np.arange(j_start, j_end + 1)
                partners = k - js
                low_contrib[js] += a[js] * a[partners] / (c[k] + 1e-30)
            
            # Increase elements with low contribution
            if (low_contrib < 0.1 * (low_contrib.max() + 1e-30)).any():
                mask = low_contrib < 0.05 * (low_contrib.max() + 1e-30)
                if mask.sum() > n // 10:
                    a[mask] *= 1.02
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S
    
    return best_a


def mulitple_peak_subgrad_lbfgs(a, n_peaks=5, beta=1000, maxiter=5000):
    """
    L-BFGS on a modified objective that heavily weights the top n_peaks.
    This creates a sharper focus on the binding constraints.
    """
    n = len(a)
    
    def focused_obj_grad(a_inner):
        c = fast_conv(a_inner)
        S = float(a_inner.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        c_top = c[top_idx]
        c_max = c_top[0]
        # Soft weights over top peaks
        scaled = beta * (c_top - c_max)
        p_top = np.exp(np.clip(scaled, -500, 500))
        p_top = p_top / p_top.sum()
        
        # Effective "max" is the weighted average of top peaks
        L = float(np.dot(p_top, c_top))
        score = 2.0 * n * L / (S * S)
        
        # Build full p vector
        p = np.zeros(len(c))
        p[top_idx] = p_top
        
        # Gradient
        a_rev = a_inner[::-1]
        M2 = n + len(c) - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        PA = np.fft.rfft(p, M2_fft)
        AA = np.fft.rfft(a_rev, M2_fft)
        h_full = np.fft.irfft(AA * PA, M2_fft)[:M2]
        h = h_full[n - 1:2 * n - 1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
        return score, grad
    
    a0 = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        focused_obj_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxiter * 20}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def adaptive_lbfgs_schedule(a, deadline, betas=None):
    """Adaptive L-BFGS with increasing beta and time-aware iteration allocation."""

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for i, beta in enumerate(betas):

            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        # Higher beta gets proportionally more iterations
        weight = (i + 1) ** 1.5
        total_weight = sum((j + 1) ** 1.5 for j in range(len(betas)))
        time_alloc = remaining * weight / (total_weight - sum((j+1)**1.5 for j in range(i)))
        time_alloc = max(5.0, min(time_alloc, remaining * 0.3))
        
        maxiter = int(min(30000, time_alloc * 8 + 500))
        maxiter = max(maxiter, 2000)
        
        print(f"    L-BFGS beta={beta:>8d}, maxiter={maxiter:>6d} (rem={remaining:.0f}s)")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)

        print(f"      score={sc:.10f} ({dt:.1f}s)")
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
    
    return best_a, best_sc

def stochastic_refine(a, rng, n_trials=50, deadline=None):
    """Stochastic local search with diverse perturbation + quick L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for t in range(n_trials):
        if deadline and time.monotonic() > deadline:
            break
        
        strategy = t % 6
        if strategy == 0:
            a_pert = a * (1.0 + 0.03 * rng.standard_normal(n))
        elif strategy == 1:
            # Peak-targeted: reduce contributors to max peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            js = np.arange(j_start, j_end + 1)
            partners = k_star - js
            contribs = a_pert[js] * a_pert[partners]
            top_order = np.argsort(contribs)[::-1][:10]
            for idx in top_order:
                a_pert[js[idx]] *= 0.85
        elif strategy == 2:
            # Frequency perturbation
            M_fft = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, M_fft)
            phase = 0.01 * rng.standard_normal(len(A))
            A_pert = A * np.exp(1j * phase)
            a_pert = np.fft.irfft(A_pert, M_fft)[:n]
        elif strategy == 3:
            # Segment scaling
            seg = max(1, n // 10)
            start = int(rng.integers(0, n - seg))
            a_pert = a.copy()
            a_pert[start:start + seg] *= (1.0 + 0.05 * rng.standard_normal())
        elif strategy == 4:
            # Swap segments
            seg = max(1, n // 16)
            b1 = int(rng.integers(0, n - 2 * seg))
            b2 = int(rng.integers(b1 + seg, n - seg))
            a_pert = a.copy()
            tmp = a_pert[b1:b1+seg].copy()
            a_pert[b1:b1+seg] = a_pert[b2:b2+seg]
            a_pert[b2:b2+seg] = tmp
        else:
            # Log-uniform perturbation
            a_pert = a * np.exp(0.02 * rng.standard_normal(n))
        
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:

        a_pert = a_pert / S

    update(sc_opt, a_opt, "main-lbfgs")
    
    # ============================================================
    # Phase 3: Focused peak optimization (15%)
    # ============================================================
    p3_end = t_start + 0.58 * (total_budget - 45.0)
    print(f"\n=== Phase 3: Focused peaks (until {p3_end - t_start:.0f}s) ===")
    
    a_fp, sc_fp = focused_peak_optimization(best_a, p3_end)
    update(sc_fp, a_fp, "focused")
    
    # ============================================================
    # Phase 4: Peak-tracking coordinate descent (17%)
    # ============================================================
    p4_end = t_start + 0.75 * (total_budget - 45.0)
    print(f"\n=== Phase 4: Peak-tracking CD (until {p4_end - t_start:.0f}s) ===")
    
    a_cd = peak_tracking_cd(best_a, n_sweeps=80, deadline=p4_end)
    sc_cd = compute_score(a_cd)
    update(sc_cd, a_cd, "peak-cd")
    
    # L-BFGS polish after CD
    if time.monotonic() < p4_end - 10:
        sc2, a2 = run_lbfgs(a_cd, 50000, maxiter=8000, maxfun=160000)
        update(sc2, a2, "cd+lbfgs-polish")
    
    # ============================================================
    # Phase 5: Stochastic refinement (12%)
    # ============================================================
    p5_end = t_start + 0.87 * (total_budget - 45.0)
    print(f"\n=== Phase 5: Stochastic (until {p5_end - t_start:.0f}s) ===")
    
    a_st, sc_st = stochastic_refine(best_a, rng, n_trials=50, deadline=p5_end)
    update(sc_st, a_st, "stochastic")
    
    # ============================================================
    # Phase 6: Multi-length (5%)
    # ============================================================
    p6_end = t_start + 0.92 * (total_budget - 45.0)
    print(f"\n=== Phase 6: Multi-length (until {p6_end - t_start:.0f}s) ===")
    
    a_ml, sc_ml = multi_length_try(best_a, rng, p6_end)
    update(sc_ml, a_ml, "multi-length")
    
    # ============================================================
    # Phase 7: Final ultra-high-beta polish (8%)
    # ============================================================
    print(f"\n=== Phase 7: Final polish ===")
    
    if time.monotonic() < deadline - 20:
        rem = deadline - 20 - time.monotonic()
        maxiter = int(min(20000, rem * 6))
        maxiter = max(maxiter, 3000)
        sc, a_fin = run_lbfgs(best_a, 1000000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_fin, "final-1M")
    
    if time.monotonic() < deadline - 15:
        sc, a_fin = run_lbfgs(best_a, 5000000, maxiter=5000, maxfun=100000)
        update(sc, a_fin, "final-5M")
    
    # Final CD
    if time.monotonic() < deadline - 8:
        a_cd2 = peak_tracking_cd(best_a, n_sweeps=30, deadline=deadline - 5)
        sc_cd2 = compute_score(a_cd2)
        update(sc_cd2, a_cd2, "final-cd")
    
    # One more L-BFGS round after final CD
    if time.monotonic() < deadline - 5:
        sc, a_fin = run_lbfgs(best_a, 2000000, maxiter=3000, maxfun=60000)
        update(sc, a_fin, "very-final")
    
    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
