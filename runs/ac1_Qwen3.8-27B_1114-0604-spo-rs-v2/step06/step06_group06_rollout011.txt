"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE-smoothed L-BFGS optimization with exponentiated gradient
polishing, convolution variance equalization, multi-restart diversification,
and targeted bottleneck reduction. Key improvements over prior approach:
- LSE optimization as primary driver with fine-grained beta annealing
- Two-stage gradient: LSE smooth + hard active set
- Convolution plateau equalization
- Frequency-domain structured diversification
- Adaptive step-size with backtracking

import numpy as np
import time
import math
from scipy.optimize import minimize


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr_fft(a, w):
    """Compute cross-correlation g[i] = sum_k w[k] * a[k-i] for i in [0,n-1].
    a has length n, w has length 2n-1.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    M = 2 * n - 1
    N = next_pow2(n + M)  # need n + M - 1 length result
    A_rev = np.fft.rfft(np.concatenate([a[::-1], np.zeros(N - n)]), N)
    W = np.fft.rfft(np.concatenate([w, np.zeros(N - M)]), N)
    full = np.fft.irfft(A_rev * W, N)[:n + M - 1]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_objective_and_grad(x, beta):
    """Compute LSE-smoothed objective and gradient."""
    n = int(len(x))
    S = float(np.sum(x))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    conv = fast_self_conv(x)
    conv_max = float(np.max(conv))
    z = beta * (conv - conv_max)
    z = np.clip(z, -600.0, 600.0)
    ez = np.exp(z)
    sum_ez = float(np.sum(ez))
    if sum_ez <= 0 or not np.isfinite(sum_ez):
        return 1e10, np.zeros(n)
    w = ez / sum_ez
    L = conv_max + float(np.log(sum_ez)) / beta

    # Gradient of LSE w.r.t. x: dL/dx[i] = 2 * cross_corr(x, w)[i]
    cc = cross_corr_fft(x, w)
    dL = 2.0 * cc

    # Full: F = 2*n*L / S^2
    # dF/dx = 2*n*dL/S^2 - 4*n*L/(S^3)
    grad = 2.0 * n * dL / (S * S) - 4.0 * n * L / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    F = float(2.0 * n * L / (S * S))
    return F, grad


def run_lse_lbfgs(a, beta, deadline, maxfun=400):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    if time.monotonic() > deadline - 3.0:
        return exact_score(a), a

    def fun(x):
        return lse_objective_and_grad(x, beta)

    bounds = [(0.0, 10000.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(20, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def soft_active_weights(conv, top_k, eps_frac):
    """Compute soft weights for active set."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.03 * conv.size))
    if active.sum() > max_count:
        active2 = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active2[top_idx] = True
        active = active2
    w = active.astype(np.float64)
    # Apply smooth weighting: positions closer to max get more weight
    w *= (0.5 + 0.5 * (conv / dmax))
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_step(a, conv, w, eta):
    """One exponentiated gradient step with given weights w over conv positions."""
    cc = cross_corr_fft(a, w)
    ccmax = float(np.max(cc))
    if ccmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * cc / ccmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def variance_equalize_step(a, conv, top_m, eta):
    """Equalize top-m convolution values."""
    d = np.sort(conv)[::-1][:top_m]
    mean_top = float(np.mean(d))
    excess = np.maximum(conv - mean_top, 0.0)
    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    return exp_grad_step(a, conv, w, eta)


def active_set_optimize(a, deadline, verbose=True, tag=""):
    """Main active-set exponentiated gradient optimizer."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    top_k = 8
    eps_frac = 0.03
    eta = 0.2
    stagnant = 0
    max_stagnant = 25
    best_ever_sc = best_sc

    for phase in range(30):
        if time.monotonic() > deadline - 2.0:
            break

        for it in range(150):
            if time.monotonic() > deadline - 2.0:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))  # sum(a)=1

            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            etas = [eta * 0.3, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0]
            for eta_try in etas:
                if eta_try < 0.005:
                    continue
                w = soft_active_weights(conv, top_k, eps_frac)
                new_a = exp_grad_step(a, conv, w, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter - 1e-13:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            # Variance equalization
            for tm in [10, 25, 50]:
                vr_a = variance_equalize_step(a, conv, tm, best_eta * 0.4)
                vr_a = normalize_a(vr_a)
                vr_sc = exact_score(vr_a)
                if vr_sc < best_sc_iter - 1e-13:
                    best_sc_iter = vr_sc
                    best_a_iter = vr_a

            if best_sc_iter < best_sc - 1e-12:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter
                eta = best_eta * 1.05
                eta = min(eta, 2.0)
                stagnant = 0
                if best_sc < best_ever_sc:
                    best_ever_sc = best_sc
                    if verbose and (it % 20 == 0):
                        print("  [{} p{} i{}] score={:.10f} top_k={} eta={:.3f}".format(
                            tag, phase, it, best_sc, top_k, eta))
            else:
                stagnant += 1
                eta *= 0.8
                if stagnant >= max_stagnant:
                    break

        # Adapt parameters
        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.4) + 1, 80)
            eps_frac = min(eps_frac * 1.4, 0.12)
            eta = max(0.05, eta * 0.6)
            stagnant = 0
            if verbose:
                print("  [{}] expand: top_k={}, eps={:.3f}".format(tag, top_k, eps_frac))
        else:
            top_k = max(3, int(top_k * 0.92))
            eps_frac = max(0.005, eps_frac * 0.93)

    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 12))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.03 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = soft_active_weights(conv, 10, 0.02)
        cc = cross_corr_fft(p, w)
        ccmax = max(float(np.max(cc)), 1e-15)
        q = p * np.exp(-strength * cc / ccmax)
    elif mode == 4:
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.4 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.4 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        # Frequency-domain perturbation
        N = next_pow2(n)
        A = np.fft.rfft(p, N)
        mag = np.abs(A)
        # Perturb the phase smoothly
        dphase = rng.standard_normal(len(A)) * strength * 0.2
        # Smooth the phase perturbation
        if len(dphase) > 3:
            dphase_filt = np.convolve(dphase, np.ones(5)/5, mode='same')
        else:
            dphase_filt = dphase
        A_new = mag * np.exp(1j * (np.angle(A) + dphase_filt))
        A_new[0] = np.abs(A_new[0])  # keep DC real and positive
        q = np.fft.irfft(A_new, N)[:n]
    elif mode == 8:
        # Gaussian smoothing
        k = int(rng.integers(7, max(9, n // 150)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 5
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 9:
        # Targeted: increase where conv is low
        conv = fast_self_conv(p)
        conv_mean = float(np.mean(conv[:n]))
        low_mask = conv[:n] < conv_mean * 0.7
        q = p.copy()
        q[low_mask] *= (1.0 + strength * 0.5)
    elif mode == 10:
        # Block perturbation
        nb = int(rng.integers(6, 30))
        block = 1.0 + strength * 0.2 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    else:

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE-smoothed L-BFGS optimization with exponentiated gradient
polishing, convolution variance equalization, multi-restart diversification,
and targeted bottleneck reduction.

import numpy as np
import time
import math
from scipy.optimize import minimize


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr_fft(a, w):
    """Compute g[i] = sum_k w[k]*a[k-i] for i in [0,n-1].
    a: length n, w: length 2n-1.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    M = 2 * n - 1
    N = next_pow2(n + M)
    A_rev = np.fft.rfft(np.concatenate([a[::-1], np.zeros(N - n)]), N)
    W = np.fft.rfft(np.concatenate([w, np.zeros(N - M)]), N)
    full = np.fft.irfft(A_rev * W, N)[:n + M - 1]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_objective_and_grad(x, beta):
    n = int(len(x))
    S = float(np.sum(x))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    conv = fast_self_conv(x)
    conv_max = float(np.max(conv))
    z = beta * (conv - conv_max)
    z = np.clip(z, -600.0, 600.0)
    ez = np.exp(z)
    sum_ez = float(np.sum(ez))
    if sum_ez <= 0 or not np.isfinite(sum_ez):
        return 1e10, np.zeros(n)
    w = ez / sum_ez
    L = conv_max + float(np.log(sum_ez)) / beta

    cc = cross_corr_fft(x, w)
    dL = 2.0 * cc

    grad = 2.0 * n * dL / (S * S) - 4.0 * n * L / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    F = float(2.0 * n * L / (S * S))
    return F, grad


def run_lse_lbfgs(a, beta, deadline, maxfun=400):
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    if time.monotonic() > deadline - 3.0:
        return exact_score(a), a

    def fun(x):
        return lse_objective_and_grad(x, beta)

    bounds = [(0.0, 10000.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(20, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def soft_active_weights(conv, top_k, eps_frac):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.03 * conv.size))
    if active.sum() > max_count:
        active2 = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active2[top_idx] = True
        active = active2
    w = active.astype(np.float64)
    w *= (0.5 + 0.5 * (conv / dmax))
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_step(a, conv, w, eta):
    cc = cross_corr_fft(a, w)
    ccmax = float(np.max(cc))
    if ccmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * cc / ccmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def variance_equalize_step(a, conv, top_m, eta):

    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    return exp_grad_step(a, conv, w, eta)
