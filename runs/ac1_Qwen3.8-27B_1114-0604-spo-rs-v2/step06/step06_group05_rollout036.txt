        # The score is scale-invariant, so this is fine.
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, None)] * n,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, None)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    return a_out

def iterative_reweighting_pg(a0, n_outer=200, n_inner=5000, p_init=1.0, p_final=3.0, step_init=0.01):
    """
    Iterative reweighting with projected gradient inner loop.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for outer in range(n_outer):
        # Compute weights
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break
        
        # Power parameter: increase over time for more aggressive peak targeting
        p = p_init + (p_final - p_init) * (outer / n_outer)
        c_norm = c / c_max
        w = c_norm ** p
        w = 0.05 * (1.0 / (2*n - 1)) + 0.95 * w  # small uniform component
        w = w / w.sum()
        
        # Inner loop: projected gradient on F(a) = sum_k w[k]*c[k]
        step = step_init
        for inner in range(n_inner):
            grad = weighted_gradient(a, w)
            
            # Normalize gradient for step size
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                break
            
            # Step with backtracking
            a_trial = a - step * grad / gn  # normalize grad to unit norm
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                step *= 0.5
                continue
            a_trial = a_trial / S_trial
            
            # Check if objective improved
            F_trial = float(np.dot(w, fast_conv(a_trial)))
            F_curr = float(np.dot(w, fast_conv(a)))
            
            if F_trial < F_curr:
                a = a_trial
                step = min(step * 1.05, 0.1)  # increase step slightly
            else:
                step *= 0.8
            
            # Check score improvement
            if inner % 100 == 0:
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        
        # Update best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if outer % 20 == 0:
                print(f"    IR-PG outer={outer}: score={best_sc:.8f}")
    
    return best_a

def gs_fienup(a0, target_flat=True, n_iters=500, alpha=0.5):
    """Gerchberg-Saxton / Fienup error reduction."""
    n = len(a0)
    M = 1 << (2*n - 2).bit_length()
    
    if target_flat:
        b_target = np.zeros(M)
        b_target[:2*n-1] = 1.0 / (2*n - 1)
    else:
        b_target = np.ones(M) / M
    
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        new_mag = A_mag ** (1 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1 - alpha) * A_phase + alpha * T_phase * 0.5
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def multi_peak_subgrad(a0, n_iters=10000, n_peaks=50, step_init=0.02):
    """Subgradient method targeting multiple peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Top peaks with weights based on proximity to max
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:n_peaks]
        
        # Weights: exponential decay from max
        w_peaks = np.exp(-5.0 * (c_max - c[top_idx]) / c_max)
        w_peaks = w_peaks / w_peaks.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # grad contribution: 2 * a[k-j] for j in [j_start, j_end]
                partner = a[k - j_end:k - j_start + 1][::-1]
                grad[j_start:j_end+1] += 2.0 * w_peaks[i] * partner
        
        S2 = S * S
        S3 = S2 * S
        # Gradient of the full score
        c_w = float(np.dot(w_peaks, c[top_idx]))
        grad_full = (2.0 * n) * (grad / S2 - 2.0 * c_w / S3)
        
        a_new = a - step * grad_full
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        step *= 0.9999
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a
