"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2.
Novel approaches: peak-pair damping, direct subgradient with FFT,
adaptive L-BFGS with very high LSE beta, multi-length exploration,
and spectral equalization with non-negativity via exponential reparameterization.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation with FFT
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,

        a_out = np.clip(res.x, 0.0, 1000.0)

    return exact_sc, a_out

                continue
            js = np.arange(j_start, j_end + 1)
            # Product a[j] * a[kk-j] for each j
            partner = kk - js
            valid = (partner >= 0) & (partner < n)
            if not valid.any():
                continue
            products = a[js[valid]] * a[partner[valid]]
            total = products.sum()
            if total < 1e-15:
                continue
            # Damping proportional to product share
            frac = products / total
            # Only damp the larger member slightly
            for idx in range(len(js[valid])):
                j = js[valid][idx]
                p = partner[valid][idx]
                if a[j] > a[p]:
                    damp[j] += weight * frac[idx]
                else:
                    damp[p] += weight * frac[idx]

        if damp.max() < 1e-15:
            break

        # Multiplicative update
        a_new = a * np.exp(-eta * damp / max(damp.max(), 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        S_new = a_new.sum()
        a = a_new / S_new

        eta = max(eta * 0.9995, 0.002)

    return best_a


def subgradient_exact(a, n_iters=30000, lr_init=0.005, momentum=0.95):
    """
    Subgradient descent on the exact max, using FFT for gradient computation.
    Step is projected onto {a>=0, sum(a)=1}.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12).copy()
    a = a / a.sum()
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    lr = lr_init

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Active set: peaks within 0.1% of max
        threshold = c_max * 0.999
        active = c >= threshold
        n_act = int(active.sum())
        if n_act == 0:
            break

        # Compute subgradient: average of dc[k]/da[j] over active k
        # d c[k] / d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j!=k-j)
        # Use FFT: grad = IFFT(FFT(active) * FFT(a)) type computation
        # Actually: sum_k active[k] * a[k-j] = correlation
        # Let's compute via direct loop over active peaks (usually few)
        active_idx = np.where(active)[0]
        if len(active_idx) > 50:
            # For many active, use FFT
            w = active.astype(np.float64)
            w /= w.sum()
            W = np.fft.rfft(w, M_fft)
            # sum_k w[k]*a[k-j] for j=0..n-1
            # This is: for each j, sum_k w[k]*a[k-j] = (w conv a)[j] shifted
            # Actually: (w * a)[m] = sum_k w[k]*a[m-k]
            # We want sum_k w[k]*a[k-j] = sum_k a[k-j]*w[k]
            # Let m = k-j => k = m+j: sum_m a[m]*w[m+j] = correlation of a and w at lag j
            # corr[j] = sum_m a[m]*w[m+j]
            # In FFT: IFFT(conj(A_full) * W)[j] ... 
            # Let's just do: np.fft.irfft(A * W, M_fft) gives conv(a,w)
            # conv(a,w)[j] = sum_i a[i]*w[j-i]
            # We want sum_k w[k]*a[k-j] = sum_i a[i]*w[i+j] where i=k-j
            # = sum_i a[i]*w[i+j] which is NOT directly a standard conv/corr
            # Let's use: w_shifted[j] such that w_shifted[m] = w[m+j]
            # This is getting complicated. Let's just loop if n_act is small.
            active_idx = active_idx[:50]  # Limit

        grad = np.zeros(n)
        inv_nact = 1.0 / len(active_idx)
        for kk in active_idx:
            kk = int(kk)
            j_start = max(0, kk - n + 1)
            j_end = min(kk, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            kj = kk - js
            # d c[kk]/d a[j] = a[kk-j] + a[j-kk] (if valid and j!=kk-j)
            contrib = a[kj].copy()  # a[kk-j] for valid range
            jk = js - kk
            valid_jk = (jk >= 0) & (jk < n) & (js != kj)
            if valid_jk.any():
                contrib = contrib.copy()
                contrib[valid_jk] += a[jk[valid_jk]]
            grad[js] += contrib

        grad *= inv_nact

        # Momentum step
        v = momentum * v + lr * grad
        a_new = a + 0.01 * v
        a_new = np.maximum(a_new, 1e-12)
        S_new = a_new.sum()
        a = a_new / S_new

        if it % 1000 == 0:
            lr *= 0.9

    return best_a


def spectral_equalize(a, n_iters=3000, alpha_init=0.5):
    """
    Novel: Shape the spectrum so that IFFT(|A|^2) is as flat as possible.
    Target: |A(f)|^2 proportional to 1/(1 + gamma * |I_target(f)|)
    where I_target is the current autocorrelation shape.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, M_fft)[:2*n-1]
        c_max = float(c.max())
        c_mean = float(c.mean())

        # Target: flatten the autocorrelation
        # In freq domain, we want |A|^2 such that IFFT(|A|^2) is flat
        # This means |A|^2 should be flat! (since IFFT of constant is delta)
        # Wait no - IFFT of constant is delta at 0. We want IFFT to be uniform.
        # Uniform in time => delta in freq. So |A|^2 should be concentrated at DC.
        # That contradicts flatness... 
        # Actually: we want c[k] = const for all k. 
        # c = IFFT(|A|^2), so |A|^2 = FFT(c) = FFT(const) = const * delta_0
        # So |A|^2 should be a single spike at DC! That means a is all-DC (constant).
        # But constant gives c[0] = n*const^2 which is the max. Score = 2.
        # So perfect flatness is impossible with non-negativity (and n>1).
        
        # Instead, let's target: c[k] <= c_max * (0.5 + 0.5 * (1 - |k - (n-1)|/(n-1)))
        # A "tent" shape that's flat at center and tapers to edges
        k_vals = np.arange(2*n-1)
        center = n - 1
        tent = 1.0 - 0.5 * np.abs(k_vals - center) / max(center, 1)
        tent = np.maximum(tent, 0.1)

        # Current P in "time domain" is c. We want to modify P (freq) to shape c.
        # Simple approach: blend current P with target P
        # Target P: such that IFFT(target_P) is proportional to tent
        target_c = tent * (float(c.sum()) / float(tent.sum()))
        target_P = np.abs(np.fft.rfft(target_c, M_fft)) ** 2
        target_P = target_P / target_P.sum() * P.sum()

        # Blend in freq domain
        alpha_cur = alpha * (1.0 - it / n_iters) + 0.05
        P_new = (1 - alpha_cur) * P + alpha_cur * target_P
        P_new = np.maximum(P_new, 1e-20)

        # Recover a with same phase
        phase = A / (np.abs(A) + 1e-20)
        A_new = phase * np.sqrt(P_new)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 1e-12)
        S_new = a_new.sum()
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def exponential_reparam_lbfgs(a0, beta, maxiter=15000):
    """
    Optimize in log-space: a_j = exp(x_j) / sum(exp(x)).
    This automatically ensures non-negativity and normalization.
    """
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-10)
    S = a0.sum()
    a0 = a0 / S
    x0 = np.log(a0)

    def obj(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        c = fast_conv(a)
        S_a = a.sum()  # = 1 by construction
        c_max = float(c.max())
        return 2.0 * n * c_max / (S_a * S_a)

    def obj_lse(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        c = fast_conv(a)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        L = c_max + float(np.log(exp_s.sum())) / beta
        p = exp_s / exp_s.sum()
        S_a = a.sum()
        score = 2.0 * n * L / (S_a * S_a)
        # Gradient: need dx = da/a (since a = exp(x)/sum(exp(x)), da_j/dx_i = a_j*(delta_ij - a_i))
        # Simplification: for large n, da_j/dx_i ≈ a_j*delta_ij (when a_i is small)
        # More precisely: d score / d x_i = sum_j (d score/d a_j) * (a_j*(delta_ji - a_i))
        # ≈ (d score/d a_i) * a_i - a_i * sum_j (d score/d a_j) * a_j
        # For the second term, it's a constant shift, so effectively we use a_j * grad_a[j]
        a_rev = a[::-1]
        M2 = 2 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        PA = np.fft.rfft(p, M2_fft)
        AA = np.fft.rfft(a_rev, M2_fft)
        h_full = np.fft.irfft(AA * PA, M2_fft)[:M2]
        h = h_full[n-1:2*n-1]
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad_a = 2.0 * n * (h / S2 - 2.0 * L / S3)
        # Transform to x-space (approximate, ignoring the mean shift)
        grad_x = grad_a * a
        return score, grad_x

    try:
        res = minimize(
            obj_lse, x0, method='L-BFGS-B', jac=True,
            options={'ftol': 1e-20, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxiter * 30}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    sc = compute_score(a_out)
    return sc, a_out


def multi_scale_optimize(a_init, deadline):
    """
    Optimize at multiple scales: start small, upscale, refine.
    """
    n0 = len(a_init)
    best_a = a_init.copy()
    best_sc = compute_score(a_init)
    elapsed = lambda: time.monotonic() - t_global_start

    # Try a few target lengths
    target_lengths = []
    for nl in [1024, 2048, 4096, 8192, 16384]:
        if nl != n0 and nl <= 16384:
            target_lengths.append(nl)
    # Also include current length
    if n0 not in target_lengths:
        target_lengths.append(n0)
    target_lengths.sort()

    for n_t in target_lengths:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 120:
            break

        # Interpolate to target length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_t)
        a_t = np.interp(x_new, x_old, best_a)
        a_t = np.maximum(a_t, 0.0)
        S = a_t.sum()
        if S > 1e-12:
            a_t = a_t / S

        # L-BFGS with escalating beta
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        betas = [2000, 10000, 50000]
        for beta in betas:
            if time.monotonic() > deadline - 60:
                break
            maxiter = int(min(15000, (deadline - time.monotonic() - 30) * 8))
            if maxiter < 500:
                break
            sc, a_opt = run_lbfgs(a_t, beta, maxiter=maxiter, maxfun=maxiter * 25)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                print(f"    multi-scale n={n_t}, beta={beta}: score={sc:.8f}")
            a_t = a_opt

        # Peak pair damping
        if time.monotonic() < deadline - 30:
            n_iter = int(min(5000, (deadline - time.monotonic() - 10) * 30))
            if n_iter > 500:
                a_d = peak_pair_damping(a_t, n_iters=n_iter)
                sc = compute_score(a_d)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_d.copy()
                    print(f"    multi-scale n={n_t}, damped: score={sc:.8f}")
                a_t = a_d

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase: multi-scale L-BFGS, peak-pair damping,
    spectral equalization, subgradient, and final high-beta polish.
    """
    global t_global_start
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_global_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096)

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_global_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent. Starting from uniform n=4096.")
        a_init = np.ones(4096)
        update(compute_score(a_init), a_init, "uniform-init")

    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Quick L-BFGS on parent length (15% of time)
    # ==================================================================
    p1_end = t_global_start + 0.15 * (total_budget - 60.0)
    print(f"\n=== Phase 1: L-BFGS on parent (until t={p1_end - t_global_start:.0f}s) ===")

    betas_p1 = [5000, 20000, 50000]
    for beta in betas_p1:
        if time.monotonic() > p1_end:
            break
        remaining = p1_end - time.monotonic()
        maxiter = int(min(10000, remaining * 12))
        if maxiter < 500:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, f"L-BFGS-b{beta}")

    # ==================================================================
    # Phase 2: Peak pair damping (10%)
    # ==================================================================
    p2_end = t_global_start + 0.25 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Peak pair damping (until t={p2_end - t_global_start:.0f}s) ===")

    if time.monotonic() < p2_end:
        n_iter = int((p2_end - time.monotonic()) * 50)
        n_iter = max(2000, min(8000, n_iter))
        a_d = peak_pair_damping(best_a, n_iters=n_iter, eta_init=0.03)
        sc = compute_score(a_d)
        update(sc, a_d, "peak-damping")
        # Follow up with L-BFGS
        if sc < best_score + 0.01 and time.monotonic() < p2_end - 10:
            sc2, a2 = run_lbfgs(a_d, 50000, maxiter=8000, maxfun=150000)
            update(sc2, a2, "damping+LBFGS")

    # ==================================================================
    # Phase 3: Multi-scale optimization (30%)
    # ==================================================================
    p3_end = t_global_start + 0.55 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Multi-scale (until t={p3_end - t_global_start:.0f}s) ===")

    a_ms, sc_ms = multi_scale_optimize(best_a, p3_end)
    update(sc_ms, a_ms, "multi-scale")

    # ==================================================================
    # Phase 4: Spectral equalization + subgradient (15%)
    # ==================================================================
    p4_end = t_global_start + 0.70 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Spectral + Subgradient (until t={p4_end - t_global_start:.0f}s) ===")

    if time.monotonic() < p4_end:
        n_it_spec = int(min(3000, (p4_end - time.monotonic()) * 15))
        if n_it_spec > 500:
            a_sp = spectral_equalize(best_a, n_iters=n_it_spec, alpha_init=0.4)
            sc = compute_score(a_sp)
            update(sc, a_sp, "spectral-eq")
            if time.monotonic() < p4_end - 20:
                sc2, a2 = run_lbfgs(a_sp, 30000, maxiter=6000, maxfun=100000)
                update(sc2, a2, "spectral+LBFGS")

    if time.monotonic() < p4_end:
        n_it_sg = int(min(15000, (p4_end - time.monotonic()) * 40))
        if n_it_sg > 2000:
            a_sg = subgradient_exact(best_a, n_iters=n_it_sg, lr_init=0.003)
            sc = compute_score(a_sg)
            update(sc, a_sg, "subgradient")

    # ==================================================================
    # Phase 5: Exponential reparam L-BFGS (10%)
    # ==================================================================
    p5_end = t_global_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Exp-reparam L-BFGS (until t={p5_end - t_global_start:.0f}s) ===")

    if time.monotonic() < p5_end:
        for beta in [10000, 50000]:
            if time.monotonic() > p5_end:
                break
            remaining = p5_end - time.monotonic()
            maxiter = int(min(12000, remaining * 10))
            if maxiter < 500:
                break
            sc, a_opt = exponential_reparam_lbfgs(best_a, beta, maxiter=maxiter)
            update(sc, a_opt, f"exp-reparam-b{beta}")

    # ==================================================================
    # Phase 6: Final aggressive optimization (15%)
    # ==================================================================
    print(f"\n=== Phase 6: Final aggressive polish ===")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = int(min(20000, remaining * 10))
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_fin, "FINAL-LBFGS-100k")

    # Ultra-high beta
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        maxiter = int(min(15000, remaining * 10))
        sc, a_fin = run_lbfgs(best_a, 300000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_fin, "FINAL-LBFGS-300k")

    # One more peak damping + LBFGS combo
    if time.monotonic() < deadline - 20:
        n_iter = int(min(5000, (deadline - 20 - time.monotonic()) * 40))
        if n_iter > 1000:
            a_d = peak_pair_damping(best_a, n_iters=n_iter, eta_init=0.01)
            sc = compute_score(a_d)
            update(sc, a_d, "FINAL-DAMP")
            if time.monotonic() < deadline - 5:
                sc2, a2 = run_lbfgs(a_d, 200000, maxiter=5000, maxfun=80000)
                update(sc2, a2, "FINAL-DAMP+LBFGS")

    # ==================================================================
    # Final verification and output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_global_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
