        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Smooth non-negativity enforcement
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Momentum
        diff = a_new - a
        momentum = beta_mom * momentum + (1.0 - beta_mom) * diff
        a = a + momentum
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def soft_peak_descent(a, n_iters=3000, temperature=0.01, step_init=0.02, decay=0.99995):
    """
    Subgradient descent with soft-max peak weighting.
    Instead of targeting top-k peaks, weights all peaks by softmax(c/T).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Soft-max weighting of peaks
        T = temperature * (1.0 + 0.5 * it / n_iters)  # slowly increase temperature
        c_scaled = c / max(T, 1e-15)
        c_scaled = np.clip(c_scaled - c_scaled.max(), -50, 50)
        w = np.exp(c_scaled)
        w = w / w.sum()
        
        # Subgradient: sum_k w_k * dc_k/dx
        # dc_k/dx_i ≈ 2*x[k-i] (for valid k-i)
        # Vectorized: subgrad[i] = 2 * sum_k w[k] * x[k-i] = 2 * correlation(a, w)[i]
        M2 = n + (2 * n - 1) - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        X = np.fft.rfft(a, M2_fft)
        W = np.fft.rfft(w, M2_fft)
        corr = np.fft.irfft(np.conj(X) * W, M2_fft)
        sg = 2.0 * corr[:n]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Skewed (more mass at left)
    inits.append(np.maximum(0.1 + 0.9 * (1.0 - x), 0.01))
    
    # Skewed (more mass at right)
    inits.append(np.maximum(0.1 + 0.9 * x, 0.01))

    # Edge-enhanced
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))

    # Gaussian at center
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)

    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))

    # Random smooth
    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    # Quartic
    inits.append(np.maximum(0.1 + 0.8 * (x * (1-x)) ** 0.5, 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        # Slightly perturbed parent
        p = parent.astype(np.float64).copy()
        p *= (1.0 + 0.05 * rng.standard_normal(n))
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-12:
            p = p / p.sum()
        inits.append(p)

    return inits
