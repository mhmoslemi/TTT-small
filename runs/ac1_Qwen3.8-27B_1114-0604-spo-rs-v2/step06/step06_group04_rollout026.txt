1. FFT-based exact subgradient with Nesterov acceleration and BB step size
2. LSE-smoothed L-BFGS-B for initial optimization
3. Peak-shaving phase targeting the top-k active lags
4. Multi-length, multi-start search with diverse structured initializations

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    """Compute full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _subgradient(a, k_active=5, tol=1e-6):
    """
    Compute score and subgradient of 2*n*max(c)/S^2 at point a.
    Uses top-k peaks within tol of maximum.
    
    Key: ∂c[m]/∂a[j] = 2*a[m-j] (with a[k]=0 if k outside [0,n-1])
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    
    # Find active set
    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    
    # Subgradient: g[j] = (2/S^2) * sum_{m in active} a[m-j] * (n/S^2) ... 
    # Actually: ∂(2n*max(c)/S^2)/∂a[j]
    # = 2n * (1/|A|) * sum_{m in A} ∂c[m]/∂a[j] / S^2 - 2n*max(c)*2*S_j / S^3
    # where S_j = ∂S/∂a[j] = 1
    # ∂c[m]/∂a[j] = 2*a[m-j] (with boundary handling)
    
    # R[j] = (1/|A|) * sum_{m in A} a[m-j]
    # For each active m: a[m-j] is valid when 0 <= m-j < n, i.e., m-n+1 <= j <= m
    R = np.zeros(n)
    for m in active:
        # a[m-j] for j = 0..n-1: valid when 0 <= m-j < n
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        R[lo:hi+1] += a[m - hi:m - lo + 1][::-1]
    R /= len(active)
    
    T = float(np.mean(c[active]))  # average value at active lags
    
    grad = (2.0 * n / (S * S)) * 2.0 * R - (2.0 * n * T) * 2.0 / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad
