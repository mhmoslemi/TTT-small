import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def fienup_enhanced(a, n_iters=500, alpha_start=0.1, alpha_end=0.85,

        a = np.ones(n) / n
        S = 1.0

        alpha = alpha_start + (alpha_end - alpha_start) * progress

    Greedy LP-inspired step: find directions that reduce the peak
    while maintaining sum and non-negativity.
    """
    n = len(a)
    c = np.fft.irfft(np.fft.rfft(a, M_fft) ** 2, M_fft)[:2*n-1]
    S = float(a.sum())
    M_val = float(c.max())
    
    # Find active peaks (within 1% of max)
    threshold = M_val * 0.99
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        return a
    
    # For each position j, compute the "cost" of increasing a[j]
    # Cost = max over active k of (d conv_k / d a[j])
    # d conv_k / d a[j] = a[k-j] (if valid) + a[j-(k)] ... = 2*a[k-j] (if k != 2j)
    # Actually: conv_k = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] + a[j] (if k-j != j)
    # More precisely: d conv_k / d a[j] = sum over the two ways j can appear
    # If k-j is in [0,n-1]: contributes a[k-j]
    # If k-j == j (i.e., k = 2j): only one term, so a[j]
    # If k-j != j: two terms, so 2*a[k-j] (wait, no)
    # conv_k = sum_i a[i] * a[k-i]
    # d/d a[j]: the term where i=j gives a[k-j]
    #         the term where k-i=j, i.e., i=k-j, gives a[j]
    # If j != k-j (i.e., 2j != k): both terms exist, total = a[k-j] + a[j]
    # If 2j == k: only one term (i=j=k-j), total = a[j]
    # So: d conv_k / d a[j] = a[k-j] + a[j] if 0 <= k-j < n and k-j != j
    #                               = a[j] if k-j == j
    #                               = a[k-j] if 0 <= k-j < n and j not in sum (can't happen since we're differentiating w.r.t. a[j])
    
    # Simpler: just use the convolution
    # d conv_k / d a[j] for "linearized" = a[k-j] (treating one copy as fixed)
    # For the full derivative: 2 * a[k-j] (when k-j != j) or a[j] (when k-j == j)
    # For simplicity in the LP, use a[k-j] as the sensitivity (one-sided)
    
    # Greedy: find positions where we can increase without violating active constraints
    # For position j, the constraint is: for each active k, sum_i a[i]*(a[k-i] + delta_i) <= M
    # Linearized: for each active k, sum_i delta_i * a[k-i] <= 0 (for the active ones)
    # If we only change position j: delta_j * a[k-j] <= 0 for each active k
    # This means: for each active k, a[k-j] <= 0 (impossible since a >= 0)
    # OR delta_j <= 0 (decrease)
    
    # So the LP says: to maintain the peak, we can only DECREASE elements or
    # find combinations of increases and decreases that cancel out.
    
    # Better approach: find the combination that maximizes sum(delta)
    # subject to sum(delta) * a[k-j] <= 0 for active k (linearized)
    
    # This is equivalent to: find delta >= -a (maintain non-neg)
    # such that for active k: sum_j delta_j * a[k-j] <= 0
    # maximize sum(delta_j)
    
    # Greedy approximation:
    # 1. Compute "sensitivity profile" for each position j:
    #    s_j = max over active k of a[k-j] (the worst-case sensitivity)
    # 2. Positions with low s_j are "safe" to increase
    # 3. Positions with high s_j should be decreased
    
    # Compute sensitivity
    s = np.zeros(n)
    for k in active[:n_max]:  # Limit computation
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            s[js:je+1] = np.maximum(s[js:je+1], a[js:je+1])  # a[k-j] for j in [js, je]
            # Wait, a[k-j] for j in [js, je] means we need a[k-js], a[k-(js+1)], ..., a[k-je]
            # = a[k-je : k-js+1][::-1]

    # For position j, the sensitivity to increasing a[j] at peak k is:
    # d conv_k / d a[j] = a[k-j] (if 0 <= k-j < n)
    # (using one-sided derivative for the LP)
    
    # So s_j = max over active k of a[k-j] (where 0 <= k-j < n)
    s = np.zeros(n)
    for k in active[:n_max]:
        k = int(k)
        # k-j ranges from k-(n-1) to k, intersected with [0, n-1]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if hi >= lo:
            # For j in [0, n-1], k-j is in [lo, hi] when j is in [k-hi, k-lo]
            j_lo = max(0, k - hi)
            j_hi = min(n - 1, k - lo)
            if j_hi >= j_lo:
                vals = a[lo:hi+1][::-1]  # a[k-j] for j from j_lo to j_hi
                s[j_lo:j_hi+1] = np.maximum(s[j_lo:j_hi+1], vals)

    # Positions with s_j > 0: increasing them by delta increases the peak by delta * s_j
    # To maintain peak: sum of increases weighted by s must be offset by decreases
    
    # Greedy: increase positions with s_j = 0 (or very small), decrease positions with large s_j
    # Net effect: transfer mass from "expensive" positions to "cheap" positions
    
    max_s = s.max()
    if max_s < 1e-15:
        return a
    
    # Target: move mass from high-s to low-s positions
    # Amount to move: small fraction to avoid large perturbations
    eps = 0.01
    move = eps * a
    
    # Positions to decrease (high sensitivity)
    dec_mask = s > 0.7 * max_s
    # Positions to increase (low sensitivity)
    inc_mask = s < 0.3 * max_s
    
    total_dec = move[dec_mask].sum()
    if total_dec > 1e-15 and inc_mask.sum() > 0:
        # Distribute the moved mass to low-sensitivity positions
        inc_weights = np.ones(inc_mask.sum())
        a_new = a.copy()
        a_new[dec_mask] -= move[dec_mask]
        a_new[inc_mask] += total_dec * inc_weights / inc_mask.sum()
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a = a_new / S_new
    
    return a

Given current a (normalized to sum=1), with M = max(conv(a,a)):

- Then set new_a = (1-t)*a + t*b / sum(b) for some t in (0,1].

The LP: max sum(b) s.t. (A @ b)_k <= M for all k, b >= 0
where A is the "cross-convolution matrix" with A[k,j] = a[k-j] (for valid indices).

A is a (2n-1) x n matrix. For n~7000, this is a 13999 x 7000 LP, which is too large to solve directly.

But we can use the dual: min M * 1^T y s.t. A^T y >= 1, y >= 0
where y has 2n-1 components.

Even the dual is too large for direct LP solvers.

Alternative: use the **column generation** approach. Start with a small set of "active" constraints, solve the restricted dual, find a violated constraint, add it, repeat.

Or: use a **greedy** approximation. At each step:
1. Find the position j that gives the smallest "cost" of increasing b[j] (i.e., the position where all (A @ e_j)_k are small)
2. Increase b[j] as much as possible without violating any constraint
3. Repeat

The "cost" of increasing b[j] by 1 is the vector A[:,j] (the j-th column of A), which is just the "cross-correlation" of a with a unit impulse at position j, which is a shifted copy of a (reversed).

So the cost is: for each k, the value a[k-j] (if valid). The "worst case" is max_k a[k-j] = max(a) (since k-j ranges over all valid indices as k varies).

Let me reconsider. The constraint is: for each k, sum_j b[j] * a[k-j] <= M.
If we increase b[j] by delta, the constraint k is violated if delta * a[k-j] > M - sum_j' b[j'] * a[k-j'].

So the "slack" for constraint k is: slack_k = M - sum_j b[j] * a[k-j].
The maximum increase in b[j] is: min over k of (slack_k / a[k-j]) (for a[k-j] > 0).

This is a classic "bottleneck" calculation. For each j, the max increase is limited by the tightest constraint that involves j.

A greedy algorithm:
1. Initialize b = 0
2. While we can improve:
   a. For each j, compute max_increase[j] = min over k with a[k-j] > 0 of (slack_k / a[k-j])
   b. Pick j* = argmax max_increase[j]
   c. b[j*] += max_increase[j*] (or a fraction of it)
   d. Update slacks

This is O(n^2) per iteration, which is too slow for n~7000.
