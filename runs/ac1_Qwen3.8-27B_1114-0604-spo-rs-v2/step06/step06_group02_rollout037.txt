def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

    L = c_max + log_sum / beta

    # Define q_k = p_k (length 2n-1) and a (length n)
    # h_j = sum_{k=j}^{j+n-1} p_k * a_{k-j} = (p * a)[j] (linear convolution, offset)
    # Wait: (p*a)[j] = sum_k p_k * a_{j-k}... no
    # (p*a)[m] = sum_k p_k * a_{m-k}
    # So h_j = sum_k p_k * a_{k-j} = sum_l p_{l+j} * a_l (letting l=k-j)
    # = (a * p_rev)[j]...

    # h_j = sum_k p_k * a_{k-j} for k-j in [0, n-1]
    # = sum_{l=0}^{n-1} a_l * p_{l+j}
    # This is the cross-correlation of a and p at lag j
    # = ifft(conj(FFT(a)) * FFT(p)) ... but need proper padding

    # A_fft = rfft(a, M_fft), P_fft = rfft(p, M_fft)
    # cross_corr = ifft(conj(A_fft) * P_fft) -- this gives sum_k a_k * p_{k-j} (circular)
    # = h_{-j} (circular cross-correlation)

    # h_j = sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} p_k * a_{k-j}

    # For j in [0, n-1]: h_j = sum_{l=0}^{n-1} a_l * p_{l+j}

    # cross[j] = sum_{k=0}^{n-1} a_k * p_{k+j} (for k+j < M_fft, which holds)
    # = h_j ✓
    
    S2 = S * S
    S3 = S2 * S
    # Gradient of score w.r.t. a_j:
    # score = 2n * L / S^2
    # d(score)/d(a_j) = 2n * (dL/da_j * S^2 - L * 2S * da_j_sum) / S^4
    # where da_j_sum = 1 (since S = sum a)
    # dL/da_j = (1/beta) * sum_k p_k * (dc_k/da_j) = (1/beta) * h_j * beta = h_j
    # Wait: L = c_max + (1/beta) * log(sum exp(beta*(c_k - c_max)))

    # = 2n * (h_j/S^2 - 2L/S^3)
    
    grad = 2.0 * n * (cross / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=50000, maxfun=500000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return a_out

def upsample(a, factor=2):
    """Upsample by factor using interpolation."""
    n_old = len(a)
    n_new = n_old * factor
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n_new) / n_new
    a = a / S
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    S_new = a_new.sum()
    if S_new > 1e-12:
        a_new = a_new / S_new
    return a_new

def lp_refine(f, n_max_constraints=500, M_scale=1.0):
    """
    LP-based refinement step.
    Find g ≥ 0 maximizing sum(g) subject to (f*g)[k] ≤ M for top constraints.
    Returns the LP solution (rescaled to match sum(f)).
    """
    n = len(f)
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-12:
        return f.copy()
    f_norm = f / S  # Normalize so sum = 1
    
    c = fast_conv(f_norm)
    M = float(c.max()) * M_scale
    
    # Select top constraints
    n_cons = min(n_max_constraints, 2 * n - 1)
    active_k = np.argsort(c)[::-1][:n_cons]
    
    # Build constraint matrix (sparse for efficiency)
    # A[i, j] = f_norm[k_i - j] if 0 <= k_i - j < n
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_k):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f_norm[k - j])
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(n_cons, n)).tocsc()
    
    # LP: max sum(g) s.t. A@g ≤ M*1, g ≥ 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(n_cons, M),
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        S_g = g.sum()
        if S_g > 1e-12:
            g = g / S_g
            return g
    return f_norm.copy()

def coordinate_descent_polish(a, n_passes=5, top_k=50):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for passer in range(n_passes):
        c = fast_conv(a)
        c_max = c.max()
        # Find peaks above 95% of max
        threshold = 0.95 * c_max
        peak_indices = np.where(c >= threshold)[0]
        
        for k in peak_indices[:30]:  # Limit to 30 peaks
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            
            # Find top contributing pairs
            top_idx = np.argsort(contribs)[::-1][:top_k]
            top_js = js[top_idx]
            
            for j in top_js[:20]:
                a_orig = a[j]
                if a_orig < 1e-12:
                    continue
                best_val = a_orig
                best_local_sc = best_sc
                for frac in [0.5, 0.65, 0.75, 0.85, 0.9, 1.1]:
                    a[j] = a_orig * frac
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_orig
                        continue
                    a_norm = a / S_tmp
                    sc_tmp = compute_score(a_norm)
                    if sc_tmp < best_local_sc:
                        best_local_sc = sc_tmp
                        best_val = a_orig * frac
                a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
