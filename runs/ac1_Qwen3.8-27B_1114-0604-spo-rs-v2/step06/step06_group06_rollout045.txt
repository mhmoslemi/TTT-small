At the optimum, the KKT conditions say that the gradient of the active constraint (the max convolution value) is a linear combination of the constraint gradients. Specifically:

This is a strong condition! It means that the cross-correlation of the active-set weights with the sequence should be approximately flat.

4. Check if the max is indeed at A; if not, update A and repeat

"""
Optimize non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using GS preprocessing, LSE-LBFGS with adaptive beta annealing, active-set
projected gradient, and diverse restarts at multiple sequence lengths.
"""

import numpy as np
import time


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def corr_pair(a, w):
    """h[i] = sum_k w[k]*a[k-i] for valid indices (cross-correlation)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1][:n]


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient for f(a)=2n*max(conv)/S^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return 0.0, np.zeros(n)
    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = cmax + np.log(sum_e) / beta
    obj = float(2.0 * n * L / (S * S))
    h = corr_pair(a, w)
    grad = 4.0 * n * (h * S - L) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return obj, grad


def gs_preprocess(a, n_iter=300, verbose=False):
    """Gerchberg-Saxe to flatten self-convolution."""
    a = normalize_a(a)
    n = len(a)
    M = 2 * n - 1
    N = next_power_of_two(M)
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = 1.0 / np.sqrt(N)
        alpha = 0.4
        target_mag = alpha * target + (1.0 - alpha) * mag
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.clip(a_new, 0.0, None)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
    return a


def lse_lbfgs(a, beta, deadline, maxfun=800):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_obj_grad(x, beta)

    bounds = [(0.0, 1.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def active_set_gradient_step(a, conv, top_k, eta):
    """One projected gradient step using top-k active set."""
    n = len(a)
    threshold = np.sort(conv)[::-1][min(top_k, len(conv) - 1)]
    active = conv >= threshold
    if active.sum() == 0:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        active[idx] = True
    w = active.astype(np.float64)
    w /= w.sum()
    h = corr_pair(a, w)
    hmax = max(float(np.max(h)), 1e-15)
    log_a = np.log(a + 1e-300) - eta * h / hmax
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    return normalize_a(a_new)


def projected_gradient_descent(a, deadline, max_iter=5000):
    """Project

ed gradient descent with adaptive step size and active set."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    eta = 0.5
    top_k = 20
    stagnant = 0
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        conv = fast_self_conv(a)
        sc = float(2.0 * n * np.max(conv))

        # Try a few step sizes and active set sizes
        improved = False
        for tk in [top_k, top_k // 2, top_k * 2, 5, 50]:
            tk = max(1, min(tk, 200))
            for e in [eta * 0.5, eta, eta * 2.0]:
                if e < 0.01 or e > 10:
                    continue
                new_a = active_set_gradient_step(a, conv, tk, e)
                new_sc = float(2.0 * n * np.max(fast_self_conv(new_a)))
                if new_sc < best_sc - 1e-12:
                    best_sc = new_sc
                    best_a = new_a
                    a = new_a
                    eta = e
                    top_k = tk
                    improved = True
                    stagnant = 0
                    break
            if improved:
                break

        if not improved:
            stagnant += 1
            eta *= 0.8
            top_k = max(3, top_k + 5)
            if stagnant > 100:
                break

    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def perturb(a, rng, strength=0.1):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        conv = fast_self_conv(p)
        top_k = min(20, n)
        threshold = np.sort(conv)[::-1][top_k - 1]
        active = (conv >= threshold).astype(np.float64)
        active /= active.sum()
        h = corr_pair(p, active)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * 2 * h / hmax)
    elif mode == 3:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.3 * (2.0 * x - 1.0) ** 2)
    elif mode == 4:
        k = min(63, max(7, n // 300))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    else:
        q = p * (0.9 + 0.2 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def full_optimize(a, deadline, verbose=True, tag=""):
    """Full optimization pipeline: GS -> LSE annealing -> active-set -> LSE again."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Step 1: GS preprocessing
    gs_dead = min(deadline, time.monotonic() + 30.0)
    if time.monotonic() < gs_dead - 5:
        a_gs = gs_preprocess(a, n_iter=200)
        sc_gs = exact_score(a_gs)
        if sc_gs < best_sc:
            best_sc = sc_gs
            best_a = a_gs.copy()
            if verbose:
                print("  [{}] GS: {:.9f}".format(tag, best_sc))

    # Step 2: LSE annealing
    betas = [200, 500, 1000, 2000, 5000, 10000, 25000, 50000, 100000, 200000, 500000]
    lse_dead = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    for beta in betas:
        if time.monotonic() > lse_dead - 10:
            break
        rem = lse_dead - time.monotonic()
        if rem < 10:
            break
        maxfun = max(100, int(rem * 0.6 / 0.08))
        maxfun = min(maxfun, 2000)
        a = normalize_a(best_a)
        sc, a = lse_lbfgs(a, beta, lse_dead, maxfun=maxfun)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print("  [{}] LSE beta={}: {:.9f}".format(tag, beta, best_sc))

    # Step 3: Active-set projected gradient
    pg_dead = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    if time.monotonic() < pg_dead - 10:
        a = normalize_a(best_a)
        sc, a = projected_gradient_descent(a, pg_dead)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print("  [{}] PG: {:.9f}".format(tag, best_sc))

    # Step 4: Final LSE with very high beta
    if time.monotonic() < deadline - 20:
        a = normalize_a(best_a)
        for beta in [500000, 1000000, 2000000]:
            if time.monotonic() > deadline - 15:
                break
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            maxfun = max(50, int(rem * 0.7 / 0.1))
            maxfun = min(maxfun, 1000)
            sc, a = lse_lbfgs(a, beta, deadline - 15, maxfun=maxfun)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                if verbose:
                    print("  [{}] Final LSE beta={}: {:.9f}".format(tag, beta, best_sc))

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    # Try multiple lengths
    lengths = [16384, 20480, 12288]
    if parent is not None:
        pn = int(parent.size)
        lengths.insert(0, pn)
        # Deduplicate
        lengths = list(dict.fromkeys(lengths))

    global_best_sc = np.inf
    global_best_a = None

    # Allocate time across lengths
    n_lengths = len(lengths)
    time_per_length = (deadline - t_start - 60.0) / n_lengths

    for li, n in enumerate(lengths):
        phase_dead = t_start + 60.0 + (li + 1) * time_per_length
        if time.monotonic() > phase_dead - 30:
            break
        if phase_dead > deadline - 30:
            phase_dead = deadline - 30

        print("=== n={} deadline={:.1f}s ===".format(n, phase_dead - t_start))

        # Get initial sequence
        if parent is not None:
            a0 = resample_sequence(parent, n)
        else:
            x = np.linspace(0.0, 1.0, n)
            a0 = 1.0 + 0.5 * np.cos(2.0 * np.pi * x) + 0.3 * np.cos(4.0 * np.pi * x)
            a0 = np.clip(a0, 0.1, None)

        a0 = normalize_a(a0)
        sc0 = exact_score(a0)
        print("  Init: {:.9f}".format(sc0))

        # Also try GS on a few different inits
        inits = [a0]
        # U-shape
        x = np.linspace(0.0, 1.0, n)
        u = 0.5 + 1.0 * (2.0 * x - 1.0) ** 2
        inits.append(normalize_a(u))
        # V-shape
        v = 1.5 - 1.0 * (2.0 * x - 1.0) ** 2
        inits.append(normalize_a(v))
        # Flat
        inits.append(np.ones(n) / n)

        best_init_sc = np.inf
        best_init_a = a0
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init_a = init
        print("  Best init: {:.9f}".format(best_init_sc))

        # Run full optimization
        sc, a = full_optimize(best_init_a, phase_dead, verbose=True, tag="n{}".format(n))
        if sc < global_best_sc:
            global_best_sc = sc
            global_best_a = a.copy()
            print("  New global best: {:.9f} (n={})".format(sc, n))

        # Try a restart with perturbation
        if time.monotonic() < phase_dead - 40:
            p = perturb(best_init_a, rng, strength=0.15)
            sc2, a2 = full_optimize(p, phase_dead, verbose=False, tag="rst")
            if sc2 < global_best_sc:
                global_best_sc = sc2
                global_best_a = a2.copy()
                print("  Restart improved: {:.9f}".format(sc2))

    # Final polish on the global best
    if global_best_a is not None and time.monotonic() < deadline - 30:
        a_final = normalize_a(global_best_a)
        sc_f, a_f = full_optimize(a_final, deadline - 10, verbose=True, tag="final")
        if sc_f < global_best_sc:
            global_best_sc = sc_f
            global_best_a = a_f.copy()

    # Output
    print("Best final: {:.9f} (n={})".format(global_best_sc, len(global_best_a)))
    out_arr = np.asarray(global_best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    out_arr = out_arr / S
    out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = global_best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        global_best_sc, verified, len(out), time.monotonic() - t_start))
    return out
