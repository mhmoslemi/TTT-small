**Peak Surgery**: Suppose the peak of the convolution is at position `k*`. The peak value is `c_{k*} = sum_i a_i a_{k*-i}`. To reduce this peak:
- Option 1: Reduce the elements `a_i` that contribute most to the peak.
- Option 2: Redistribute mass from the contributing elements to positions that don't contribute to the peak.
- Option 3: Use a local optimization to find a small perturbation `δa` that reduces `c_{k*}` while keeping other `c_k` values bounded.

Option 3 is the most principled. We can solve:
minimize `c_{k*}(a + δa)`
subject to `c_k(a + δa) <= c_{k*}(a)` for all `k ≠ k*` (don't increase any other peak)
           `||δa||_2 <= ε` (small perturbation)
           `a + δa >= 0`
           `sum(a + δa) = sum(a)`

This is again a QCQP, but with a small trust region. For small `ε`, we can linearize:
minimize `2 sum_i (a_{k*-i}) δa_i`
subject to `2 sum_i (a_{k-i}) δa_i <= c_{k*}(a) - c_k(a)` for all `k ≠ k*`

           `a_i + δa_i >= 0`
           `sum(δa) = 0`

        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def proportional_peak_reduction(a, n_rounds=200, n_peaks=20, decay=0.998):
    """Reduce peaks proportionally to their contribution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.1

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # For each peak, reduce contributing elements proportionally
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner

            if contrib.max() < 1e-15:
                continue

            # Proportional reduction: reduce elements in proportion to their contribution
            # a_i *= (1 - step * contrib_i / contrib_max)
            norm_contrib = contrib / contrib.max()
            a[j_start:j_end + 1] *= (1.0 - step * norm_contrib)

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        step *= decay

    return best_a


def nesterov_lse(a0, beta, n_iters=20000, lr_init=0.01, lr_decay=0.99995, momentum=0.9):
    """Nesterov momentum gradient descent on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    patience = 0
    patience_limit = 3000

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)

        # Nesterov lookahead
        a_look = a - momentum * v
        sc_look, grad_look = lse_score_grad(a_look, beta)
        v = momentum * v + lr * grad_look

        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= lr_decay

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 200
                if patience > patience_limit:
                    break

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def subgradient_adaptive(a0, n_iters=10000, n_peaks=30, step_init=0.02, decay=0.99998):
    """Adaptive subgradient method with multiple peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        if 2.0 * n * c_max / (S * S) < best_sc:
            best_sc = compute_score(a)
            best_a = a.copy()

        # Top peaks weighted by excess
        top_idx = np.argsort(c)[::-1][:n_peaks]
        excess = np.maximum(c[top_idx] - c_max * 0.95, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break

        # Weighted subgradient
        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            w = excess[i] / total_excess
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                partner = a[k - j_end:k - j_start + 1][::-1]
                # Gradient of c_k w.r.t. a is 2*a_{k-i} for each i
                sg[j_start:j_end + 1] += w * 2.0 * partner

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def anneal_perturb_optimize(a0, rng, n_temp=20, temp_init=0.05, temp_decay=0.85, lbfgs_maxfun=5000, lbfgs_beta=10000):
    """Simulated annealing: perturb, optimize, accept/reject."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    temp = temp_init

    for t in range(n_temp):
        # Random perturbation
        strength = temp * 0.5
        perturbation = rng.standard_normal(n) * strength
        a_pert = a + perturbation
        a_pert = np.maximum(a_pert, 0.0)
        S_pert = a_pert.sum()
        if S_pert < 1e-12:
            temp *= temp_decay
            continue
        a_pert = a_pert / S_pert

        # Quick optimize
        a_opt = run_lbfgs(a_pert, maxiter=2000, maxfun=lbfgs_maxfun, beta=lbfgs_beta)
        a_opt = np.maximum(a_opt, 0.0)
        S_opt = a_opt.sum()
        if S_opt < 1e-12:
            temp *= temp_decay
            continue
        a_opt = a_opt / S_opt
        sc_opt = compute_score(a_opt)

        # Accept/reject
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            temp *= 0.95  # slow cooling on improvement
        elif rng.random() < np.exp(-(sc_opt - best_sc) / max(temp * 0.01, 1e-10)):
            a = a_opt.copy()
            temp *= temp_decay
        else:
            temp *= temp_decay

    return best_a


def iterative_reweighting(a0, n_outer=30, n_inner=2000, maxfun_per_inner=3000):
    """Iterative reweighting: re-weight peak positions and solve weighted subproblem."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for outer in range(n_outer):
        c = fast_conv(a)
        c_max = float(c.max())

        p = 1.0 + 2.0 * outer / max(1, n_outer)
        c_norm = c / c_max
        w = c_norm ** p
        w = 0.1 * (1.0 / (2 * n - 1)) + 0.9 * w
        w = w / w.sum()

        a = run_lbfgs_weighted(a, w, maxiter=n_inner, maxfun=maxfun_per_inner)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def weighted_score_grad(a, w):
    """Weighted convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    w_num = float(np.dot(w, c))
    score = 2.0 * n * w_num / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    w_full = np.zeros(M2_fft)
    w_full[:2 * n - 1] = w
    W = np.fft.rfft(w_full, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * w_num) / S3
    return score, grad


def run_lbfgs_weighted(a0, w, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on weighted objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = weighted_score_grad(x, w)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.3 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.sin(np.pi * x) * np.cos(2 * np.pi * x), 0.01))

    # Fourier-based
    for _ in range(3):
        freqs = np.fft.rfftfreq(n)
        n_harmonics = rng.integers(3, 12)
        weights = rng.standard_normal(n_harmonics)
        phases = rng.uniform(0, 2 * np.pi, n_harmonics)
        sig = np.zeros(n)
        for i in range(n_harmonics):
            sig += weights[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
        init = np.maximum(0.5 + 0.3 * sig, 0.01)
        inits.append(init)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: GS warmup, LSE-LBFGS, peak reduction, Nesterov, annealing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 45.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Choose n: use parent length or try slightly larger
    n = 6962
    if parent is not None:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting optimization with n={n}")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: GS warmup on diverse initializations (10% of time)
    # ==================================================================
    phase1_end = t_start + 0.10 * (total_budget - 45.0)
    print(f"\n=== Phase 1: GS warmup (deadline +{(phase1_end - t_start):.0f}s) ===")

    inits = build_inits(n, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / max(a0.sum(), 1e-12)

        a1 = gs_iterative_target(a0, n_iters=250, alpha_init=0.4, alpha_final=0.05)
        sc = compute_score(a1)
        update_best(sc, a1)

        if time.monotonic() < phase1_end:
            a2 = gs_flat_target(a1, n_iters=150, alpha_init=0.3, alpha_final=0.03)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE-LBFGS with beta sweep (30% of time)
    # ==================================================================
    phase2_end = t_start + 0.40 * (total_budget - 45.0)
    print(f"\n=== Phase 2: LSE-LBFGS (deadline +{(phase2_end - t_start):.0f}s) ===")

    betas = [500, 2000, 5000, 10000, 30000, 80000, 200000]
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        t_beta = time.monotonic()
        a_opt = run_lbfgs(best_a, maxiter=5000, maxfun=20000, beta=beta)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Peak reduction + Iterative Reweighting (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (total_budget - 45.0)
    print(f"\n=== Phase 3: Peak reduction + IR (deadline +{(phase3_end - t_start):.0f}s) ===")

    # Proportional peak reduction
    a_ptr = proportional_peak_reduction(best_a, n_rounds=300, n_peaks=30, decay=0.997)
    sc = compute_score(a_ptr)
    update_best(sc, a_ptr)

    # LSE refinement after peak reduction
    if time.monotonic() < phase3_end:
        a_opt = run_lbfgs(best_a, maxiter=4000, maxfun=15000, beta=10000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

    # Iterative reweighting
    if time.monotonic() < phase3_end:
        a_ir = iterative_reweighting(best_a, n_outer=20, n_inner=1500, maxfun_per_inner=2500)
        sc = compute_score(a_ir)
        update_best(sc, a_ir)

    # Subgradient
    if time.monotonic() < phase3_end:
        a_sg = subgradient_adaptive(best_a, n_iters=5000, n_peaks=30, step_init=0.015)
        sc = compute_score(a_sg)
        update_best(sc, a_sg)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Nesterov + Annealing (20% of time)
    # ==================================================================
    phase4_end = t_start + 0.85 * (total_budget - 45.0)
    print(f"\n=== Phase 4: Nesterov + Annealing (deadline +{(phase4_end - t_start):.0f}s) ===")

    # Nesterov with multiple betas
    for beta in [5000, 20000, 50000]:
        if time.monotonic() > phase4_end:
            break
        t_nm = time.monotonic()
        a_nm = nesterov_lse(best_a, beta=beta, n_iters=15000, lr_init=0.005, momentum=0.9)
        sc = compute_score(a_nm)
        update_best(sc, a_nm)
        print(f"  Nesterov beta={beta}: score={sc:.8f} ({time.monotonic()-t_nm:.1f}s)")

    # Annealing
    if time.monotonic() < phase4_end:
        a_ann = anneal_perturb_optimize(best_a, rng, n_temp=10, temp_init=0.03,
                                         temp_decay=0.8, lbfgs_maxfun=5000, lbfgs_beta=10000)
        sc = compute_score(a_ann)
        update_best(sc, a_ann)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish (15% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish (deadline +{(deadline - t_start):.0f}s) ===")

    polish_rounds = 6
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # GS refinement
        a_gs = gs_iterative_target(best_a, n_iters=150, alpha_init=0.25, alpha_final=0.02)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)

        # LSE fine-tune with increasing beta
        if time.monotonic() > deadline:
            break
        beta_final = 50000 + 50000 * pr
        a_opt = run_lbfgs(best_a, maxiter=4000, maxfun=15000, beta=beta_final)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Quick peak reduction
        if time.monotonic() > deadline:
            break
        a_ptr = proportional_peak_reduction(best_a, n_rounds=100, n_peaks=20, decay=0.998)
        sc = compute_score(a_ptr)
        update_best(sc, a_ptr)

        # Very high beta LSE
        if time.monotonic() > deadline:
            break
        a_opt2 = run_lbfgs(best_a, maxiter=3000, maxfun=10000, beta=200000 + 100000 * pr)
        sc = compute_score(a_opt2)
        update_best(sc, a_opt2)

        print(f"  Polish round {pr}: best = {best_score:.8f}")

    # ==================================================================
    # Final: one more round of the best technique
    # ==================================================================
    if time.monotonic() < deadline:
        a_final = run_lbfgs(best_a, maxiter=5000, maxfun=20000, beta=500000)
        sc = compute_score(a_final)
        update_best(sc, a_final)

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
