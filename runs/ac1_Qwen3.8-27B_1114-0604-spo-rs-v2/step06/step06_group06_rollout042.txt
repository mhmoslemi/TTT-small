def run_lse_lbfgs(a, betas, deadline, max_iter_per_beta=200, verbose=False, tag=""):
    """Run LSE-LBFGS with a schedule of beta values (decreasing)."""
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        
        a = normalize_a(best_a)
        
        def fun_and_grad(x):
            x = np.clip(x, 0.0, None)
            S = float(x.sum())
            if S < 1e-10:
                return 1e10, np.zeros_like(x)
            a_local = x / S
            return lse_objective_and_grad(a_local, beta)
        
        bounds = [(0.0, 10.0)] * n
        maxfun = max(50, min(max_iter_per_beta, int(rem * 0.8 / 0.05)))
        maxfun = min(maxfun, 1000)
        opts = {
            "ftol": 1e-14,
            "gtol": 1e-8,
            "maxiter": max(20, maxfun // 3),
            "maxfun": maxfun,
            "maxls": 30,
            "iprint": -1,
        }
        
        try:
            res = minimize(fun_and_grad, a, method="L-BFGS-B", jac=True,
                           bounds=bounds, options=opts)
            if res.x is not None:
                a_out = normalize_a(res.x)
                sc = exact_score(a_out)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_out
                    if verbose:
                        print("  [{} beta={:.0f}] score={:.9f}".format(tag, beta, best_sc))
        except Exception as e:
            if verbose:
                print("  [{} beta={:.0f}] error: {}".format(tag, beta, str(e)[:50]))
    
    return float(best_sc), best_a


def subgradient_step(a, conv, n_active=10, eta=0.1):
    """Single subgradient step: reduce the top-n_active convolution peaks."""
    n = len(a)
    dmax = float(np.max(conv))
    if dmax <= 0:
        return a.copy()
    
    # Find positions with convolution close to max
    threshold = dmax * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = [int(np.argmax(conv))]
    if len(active) > n_active:
        # Keep only the top n_active
        vals = conv[active]
        top_idx = np.argsort(vals)[-n_active:]
        active = active[top_idx]
    
    # Subgradient direction: average of the gradients at active positions
    # grad_k[i] = 2*a[k-i] for 0<=k-i<=n-1
    # So the average subgradient is: 2*mean of a shifted for each active k
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = int(k) - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)
    
    # Projected gradient step on the simplex {a>=0, sum=1}
    # Use multiplicative (exponentiated) update for simplicity
    log_a = np.log(a + 1e-30)
    log_a -= eta * grad / max(float(np.max(grad)), 1e-15)
    log_a -= float(np.max(log_a))
    new_a = np.exp(log_a)
    return normalize_a(new_a)


def subgradient_polish(a, n_steps=100, deadline=None, verbose=False, tag=""):
    """Multiple subgradient steps with adaptive eta."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    eta = 0.15
    stagnant = 0
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = fast_self_conv(a)
        sc = exact_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            eta *= 0.9
        
        if stagnant > 30:
            break
        
        a = subgradient_step(a, conv, n_active=5, eta=eta)
        a = normalize_a(a)
    
    if verbose:
        print("  [{} sg_polish] final={:.9f}".format(tag, best_sc))
    return float(best_sc), best_a


def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n_new)) / float(max(1, n_new))
    if m == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, a).astype(np.float64)
