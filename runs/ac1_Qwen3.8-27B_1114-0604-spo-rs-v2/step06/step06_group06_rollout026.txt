"""

using iterative LP-based bilinear optimization (alternating minimization),

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def partner_vector(a, k):
    """Compute P_k @ a, which is the 'partner' vector for the k-th convolution element."""
    n = len(a)
    b = np.zeros(n)
    # (a*a)_k = sum_j a_j * a_{k-j} where 0<=k-j<n and 0<=j<n
    # So j ranges from max(0, k-(n-1)) to min(k, n-1)
    # b_j = a_{k-j} for valid j, 0 otherwise
    j_min = max(0, k - (n - 1))
    j_max = min(k, n - 1)
    for j in range(j_min, j_max + 1):
        b[j] = a[k - j]
    return b


def partner_vector_fast(a, k):
    """Fast version using array operations."""
    n = len(a)
    j_min = max(0, k - (n - 1))
    j_max = min(k, n - 1)
    b = np.zeros(n)
    if j_min <= j_max:
        b[j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    return b


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def solve_lp_peak_shave(a, b_vecs, max_iter=100, tol=1e-10):
    """
    Solve: minimize t subject to dot(a_new, b_k) <= t for all k, sum(a_new)=1, a_new >= 0
    where b_vecs is a list of partner vectors (fixed).
    Returns the optimal a_new and t.
    """
    n = len(a)
    K = len(b_vecs)
    
    # LP: minimize t
    # subject to: B @ a <= t (K constraints, B is K x n)
    #             sum(a) = 1
    #             a >= 0
    # Variables: [a_0, ..., a_{n-1}, t] (n+1 variables)
    # Objective: [0, ..., 0, 1]
    # Inequality: [B, -I_K] @ [a; t] <= 0  →  B@a - t <= 0
    # Equality: [1, ..., 1, 0] @ [a; t] = 1
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: B @ a - t <= 0
    B = np.zeros((K, n))
    for i, bv in enumerate(b_vecs):
        B[i] = bv
    A_ub = np.hstack([B, -np.eye(K)])
    b_ub = np.zeros(K)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_i >= 0, t unbounded below (or t >= 0 for safety)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            a_new = res.x[:n]
            t_new = res.x[n]
            a_new = np.clip(a_new, 0, None)
            S = a_new.sum()
            if S < 1e-12:
                return a, np.max([np.dot(a, bv) for bv in b_vecs])
            a_new = a_new / S
            return a_new, t_new
        else:
            return a, None
    except Exception:
        return a, None


def lp_iterative_optimize(a, deadline, top_k_init=5, max_inner_iter=50, 
                           damping=0.5, verbose=True, tag=""):

    Iterative LP-based optimization:
    1. Compute active set (top-K peaks)
    2. Compute partner vectors
    3. Solve LP to minimize max dot(a, b_k)
    4. Damped update
    5. Repeat
    """
    n = len(a)
    a = normalize_a(a).copy()
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()
    
    stagnant = 0
    max_stagnant = 10
    
    for outer in range(500):
        if time.monotonic() > deadline - 5:
            break
        
        conv = fast_self_conv(a)
        sc = 2.0 * n * float(np.max(conv))  # sum(a) = 1
        
        # Identify active set: top peaks
        # Use a threshold-based approach
        cmax = float(np.max(conv))
        threshold = cmax * 0.95
        active_mask = conv >= threshold
        # Ensure at least top_k_init
        if active_mask.sum() < top_k_init:
            idx = np.argpartition(conv, -top_k_init)[-top_k_init:]
            active_mask[idx] = True
        # Cap the active set size
        max_active = min(15, n)
        if active_mask.sum() > max_active:
            idx = np.argpartition(conv, -max_active)[-max_active:]
            active_mask = np.zeros(conv.size, dtype=bool)
            active_mask[idx] = True
        
        active_idx = np.where(active_mask)[0]
        
        # Compute partner vectors for active set
        b_vecs = [partner_vector_fast(a, k) for k in active_idx]
        
        # Solve LP
        a_lp, t_lp = solve_lp_peak_shave(a, b_vecs)
        
        # Damped update
        if a_lp is not None and t_lp is not None:
            a_new = (1.0 - damping) * a + damping * a_lp
            a_new = normalize_a(a_new)
        else:
            a_new = a
        
        new_sc = exact_score(a_new)
        
        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            if verbose and outer % 20 == 0:
                print("  [{} lp o{}] score={:.9f} active={} t_lp={:.6f}".format(
                    tag, outer, best_sc, len(active_idx), t_lp if t_lp else -1))

            stagnant += 1

                top_k_init = min(top_k_init + 2, 20)

                    print("  [{}] adjust: top_k={}, damping={:.2f}".format(tag, top_k_init, damping))

def adam_expg(a, conv, active_idx, lr, beta1, beta2, m, v, t, verbose=False):
    """
    Adam exponentiated gradient step targeting active peaks.
    """
    n = len(a)
    # Compute the "gradient" in log-space: for each active peak k,
    # the partner vector b_k tells us which elements to reduce
    h = np.zeros(n)
    for k in active_idx:
        bv = partner_vector_fast(a, k)
        h += bv
    
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a, m, v
    
    h = h / hmax
    
    # Adam update in log-space
    m = beta1 * m + (1 - beta1) * h
    v = beta2 * v + (1 - beta2) * h * h
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    log_a = np.log(a + 1e-300)
    log_a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    log_a -= float(np.max(log_a))
    a_new = np.exp(log_a)
    
    return a_new, m, v

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_asymmetric_inits(n, rng, cap=30):
    """Generate initializations with controlled asymmetry."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        p = normalize_a(arr)
        inits.append(p)
    
    # Asymmetric exponential (mass on left)
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x) + 0.01)
        add(np.exp(-rate * (1.0 - x)) + 0.01)
    
    # Asymmetric power
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(x ** p + 0.01)
        add((1.0 - x) ** p + 0.01)
    
    # Skewed Gaussian
    for mu in (0.3, 0.35, 0.4, 0.45):
        for sig in (0.1, 0.15, 0.2, 0.3):
            add(np.exp(-((x - mu) / sig) ** 2) + 0.01)
    
    # Piecewise linear (ramp)
    for break_pt in (0.3, 0.4, 0.5, 0.6):
        ramp = np.maximum(0, 1 - np.abs(x - break_pt) / max(break_pt, 1-break_pt))
        add(ramp + 0.01)
    
    # Two-sided with different rates
    for r1 in (1, 2, 3, 5):
        for r2 in (1, 2, 3, 5):
            if r1 != r2:
                add(np.exp(-r1 * x) + np.exp(-r2 * (1-x)) + 0.01)
    
    # Sine-based asymmetric
    for f in (1, 2, 3):
        for phase in (-0.5, -0.2, 0.2, 0.5):
            add(0.5 + 0.5 * np.sin(2 * np.pi * f * x + phase * np.pi) + 0.01)
    
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    
    # Uniform
    add(np.ones(n))
    
    # Random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Asymmetric shift: boost one side
        side = rng.choice([0, 1])
        x = np.linspace(0, 1, n)
        if side == 0:
            q = p * np.exp(strength * 0.5 * (1 - 2*x))
        else:
            q = p * np.exp(strength * 0.5 * (2*x - 1))
    elif mode == 4:
        # Smooth noise
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        # Block perturbation
        nb = int(rng.integers(3, 15))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))
