    # Using FFT: this is the inverse FFT of P(ω) * conj(B(ω)) * exp(iωj)
    # Or equivalently, it's the j-th element of the cross-correlation.
    
    # Simplest: just do conv(p, b_reversed) and index appropriately.
    # conv(p, b_rev)[k] = sum_i p[i] * b_rev[k-i] = sum_i p[i] * b[n-1-k+i]
    # At k = n-1+j: sum_i p[i] * b[j+i]... that's not quite right either.
    
    # OK let me just use the formula from the previous code which worked:
    # grad_L = 2 * conv(p, a_reversed)[n-1 : 2n-1]
    
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1  # length of conv of p (len 2n-1) and a_rev (len n)
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    
    score = (2.0 * n * L) / S2
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxiter * 25, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return compute_score(a0), a0

def fienup_power(a, n_iters=500, alpha_start=0.9, alpha_end=0.3):
    """Fienup with power-based spectral shaping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_max = P.max()
        P_min = P.min()
        
        # Power-based shaping: compress dynamic range
        # P_new = P^alpha (with alpha < 1, this compresses)
        # Normalize to keep the same total power
        P_norm = P / P_max
        P_shaped = P_max * (P_norm ** alpha)
        
        # Ensure non-negative and not too small
        P_shaped = np.maximum(P_shaped, 1e-30)
        
        ratio = np.sqrt(P_shaped / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def fienup_clipped(a, n_iters=500):
    """Fienup with adaptive spectral clipping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        
        # Progressive clipping: start gentle, get more aggressive
        q = 0.85 - 0.35 * progress  # quantile threshold (0.85 -> 0.50)
        thresh = np.quantile(P, q)
        
        P_new = P.copy()
        P_new[P_new > thresh] = thresh
        P_new = np.maximum(P_new, 1e-30)
        
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def fienup_combined(a, n_iters=500):
    """Combined Fienup: alternate power and clipping modes."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_max = P.max()
        P_mean = P.mean()
        
        if it % 3 == 0:
            # Power mode
            alpha = 0.85 - 0.4 * progress
            P_norm = P / P_max
            P_new = P_max * (P_norm ** alpha)
        elif it % 3 == 1:
            # Clipping mode
            q = 0.80 - 0.30 * progress
            thresh = np.quantile(P, q)
            P_new = np.minimum(P, thresh)
        else:
            # Gentle flattening toward mean
            alpha = 0.1 + 0.4 * progress
            P_new = (1 - alpha) * P + alpha * P_mean
        
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_shave(a, n_iters=5000, top_k=10, step_init=0.005, decay=0.9995):
    """Subgradient peak shaving on top-k autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Lower bound check
        if sc - 2.0 * n / (2.0 * n - 1) < 1e-6:
            break
        
        top_idx = np.argsort(c)[::-1][:top_k]
        
        sg = np.zeros(n)
        total_w = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_w += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w * a[j_start:j_end+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        avg_w = total_w / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_polyak = (sc - 2.0 * n / (2.0 * n - 1)) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.02)
        step = max(step, 1e-10)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

def interpolate_seq(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
