def lp_active_set(a, K=200, method='HiGHS'):
    """
    Solve approximate LP using only K most binding constraints.
    min M' s.t. (a*x)[k] <= M' for k in top_K, sum(x)=1, x>=0
    """
    n = len(a)
    # Compute current convolution
    ax = self_conv(a)  # This is a*a, but we need a*x for the LP
    # Actually, for the LP we need the matrix A where (A*x)[k] = sum_i a[i]*x[k-i]
    # The "binding" constraints are those k where a*a is largest
    
    conv_aa = self_conv(a)
    M = conv_aa.max()
    
    # Top K constraints
    top_k = np.argsort(conv_aa)[-K:]
    
    # Build LP
    # Variables: x_0,...,x_{n-1}, M'
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))

        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, j] = a[i]
        A_ub[idx, -1] = -1.0
    
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='HiGHS')
    
    if res.status != 0:
        return a / a.sum()  # fallback
    
    x = res.x[:n]
    return x

6. Progressive constraint activation in LP for efficiency

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

            break

        # FFT
        A = np.fft.rfft(a, M_fft)
        R = A.real ** 2 + A.imag ** 2  # Power spectrum |A|^2

        # Autocorrelation
        r = np.fft.irfft(R, M_fft)[:2 * n - 1]

        # Target: flat autocorrelation
        target = 1.0 / (2 * n - 1)
        excess = np.maximum(r - target, 0.0)

        if excess.max() < 1e-14:
            break

        # Remove excess in frequency domain
        excess_pad = np.zeros(M_fft)
        excess_pad[:len(excess)] = excess
        R_excess = np.fft.rfft(excess_pad).real

        # Reduce power spectrum where excess has energy
        step = alpha * 0.5
        R_new = R - step * R_excess
        R_new = np.maximum(R_new, 0.0)  # Power spectrum must be non-negative

        # Update a: preserve phase, update magnitude
        A_new = np.sqrt(R_new) * (A / (np.abs(A) + 1e-15))
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        Sn = a_new.sum()
        if Sn < 1e-12:
            break
        a_new = a_new / Sn

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def lse_value_and_grad(a, beta):
    """
    Compute LSE-smoothed max of self-convolution and its gradient.
    Returns (lse_value, gradient) where gradient is d/d a of the LSE.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    R = A.real ** 2 + A.imag ** 2
    r = np.fft.irfft(R, M_fft)[:2 * n - 1]

    r_max = float(r.max())
    z = beta * (r - r_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(r)) / len(r)
    else:
        p = exp_z / exp_sum

    LSE = r_max + float(z.max() - z_shifted.max()) / beta + np.log(exp_sum) / beta

    # Gradient: dLSE/da[j] = sum_k p[k] * d(conv_k)/da[j] = 2 * sum_k p[k] * a[k-j]
    # This is 2 * cross_correlation(p, a)[j]
    P = np.fft.rfft(p, M_fft)
    # Cross-correlation: (p ★ a)[j] = ifft(conj(P) * A)[j]
    corr = np.fft.irfft(np.conj(P) * A, M_fft)[:n]
    grad = 2.0 * corr

    return LSE, grad


def lse_score_and_grad_normalized(a, beta):
    """
    LSE-smoothed score (with normalization) and gradient.
    Score = 2n * LSE / S^2 where S = sum(a).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    LSE, grad_LSE = lse_value_and_grad(a, beta)
    score = 2.0 * n * LSE / (S * S)

    # Gradient of 2n*LSE/S^2:
    # = 2n * (grad_LSE * S^2 - LSE * 2S) / S^4
    # = 2n * (grad_LSE / S^2 - 2*LSE/(S*S*S))
    S2 = S * S
    grad = 2.0 * n * (grad_LSE / S2 - 2.0 * LSE / (S2 * S))
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=100000, gtol=1e-12):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad_normalized(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_active_set(a, K=300):
    """
    Solve approximate LP using K most binding constraints.
    min M' s.t. (a*x)[k] <= M' for k in top_K, sum(x)=1, x>=0
    Returns optimal x.
    """
    n = len(a)
    conv_aa = self_conv(a)
    # Get top K lags (where a*a is largest)
    if K >= len(conv_aa):
        top_k = np.arange(len(conv_aa))
    else:
        top_k = np.argsort(conv_aa)[-K:]
    top_k = np.sort(top_k)

    # Build constraint matrix (sparse-friendly, but K is small so dense is fine)
    A_ub = np.zeros((len(top_k), n + 1))
    for idx, k in enumerate(top_k):
        # (a*x)[k] = sum_i a[i]*x[k-i]
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # index of x
            A_ub[idx, j] = a[i]
        A_ub[idx, -1] = -1.0  # coefficient of M'

    b_ub = np.zeros(len(top_k))

    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M'

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='HiGHS',
                  options={'presolve': True, 'time_limit': 30.0})

    if res.status != 0:
        return None

    x = res.x[:n]
    x = np.maximum(x, 0.0)
    Sx = x.sum()
    if Sx < 1e-12:
        return None
    return x / Sx


def lp_alternating_active_set(a, max_iters=200, K_schedule=None, time_limit=None,
                               line_search=True):
    """
    LP alternating method using active-set LP (HiGHS).
    Iteratively: compute a*a, solve LP for x, update a.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    if K_schedule is None:
        K_schedule = [100, 200, 300, 500, 800]

    x_warm = None
    no_improve = 0

    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        # Determine K for this iteration
        progress = it / max(1, max_iters)
        K_idx = min(int(progress * len(K_schedule)), len(K_schedule) - 1)
        K = K_schedule[K_idx]

        # Solve LP
        x = lp_active_set(a, K=K)
        if x is None:
            # Fallback: use current a
            x_warm = a.copy()
            continue
        x_warm = x.copy()

        # Line search on mixing parameter
        if line_search:
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            best_t = 0.3
            best_t_sc = best_sc
            for t in t_candidates:
                a_try = (1.0 - t) * a + t * x
                a_try = np.maximum(a_try, 0.0)
                St = a_try.sum()
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
            t = best_t
        else:
            t = 0.5

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                # Try larger step or restart
                a = x_warm.copy()
                no_improve = 0
                if no_improve > 50:
                    break

        a = a_new

        if it % 20 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {it}: score={sc:.8f}, best={best_sc:.8f}, K={K}, t={elapsed:.0f}s")

    return best_a, best_sc


def subgradient_descent(a, n_steps=5000, time_limit=None):
    """
    Direct subgradient descent on the true max objective.
    Uses all top peaks with exponential weighting for robustness.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    # Adaptive beta for subgradient
    beta = 50.0

    for it in range(n_steps):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        # Compute convolution
        A = np.fft.rfft(a, M_fft)
        R = A.real ** 2 + A.imag ** 2
        r = np.fft.irfft(R, M_fft)[:2 * n - 1]

        # Soft-max weights (to handle multiple peaks)
        r_max = r.max()
        z = beta * (r - r_max)
        z_shifted = z - z.max()
        p = np.exp(z_shifted)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
        else:
            p = np.zeros_like(r)
            p[int(np.argmax(r))] = 1.0

        # Gradient: 2 * cross_correlation(p, a)
        P = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(P) * A, M_fft)[:n]
        grad = 2.0 * corr

        # Project gradient (remove mean for sum constraint)
        grad = grad - grad.mean()

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = min(0.1, 0.5 / (beta * (r_max + 1e-10) * grad_norm + 1e-10))

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        # Gradually increase beta (sharpen the peak focus)
        if it % 100 == 0:
            beta *= 1.2

    return best_a


def peak_elimination(a, n_steps=1000, time_limit=None):
    """
    Target peak lags for elimination via local adjustments.
    At each step, identify the worst peak and adjust nearby elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_steps):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        r = self_conv(a)
        k_star = int(np.argmax(r))
        M_val = r[k_star]

        # Subgradient at the peak:
        # d(r[k_star])/da[j] = 2*a[k_star - j] if 0 <= k_star-j < n
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Normalize and step
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-14:
            break
        direction = direction / d_norm

        # Try several step sizes
        improved = False
        for step_mult in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            # We want to DECREASE r[k_star], so move AGAINST the gradient
            # But we also need to not increase other lags too much
            # Use a small step in the negative subgradient direction
            a_try = a - step_mult * direction
            a_try = np.maximum(a_try, 0.0)
            St = a_try.sum()
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 100 == 0:
            # Random perturbation to escape
            noise = 0.01 * np.random.default_rng(it).standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.maximum(parent.astype(np.float64).copy(), 0.0))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.0))

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    smooth = uniform_filter1d(rng.standard_normal(n), size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 100.0))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x + 0.5))
    inits.append(np.clip(0.3 + 0.3 * x + 0.2 * np.cos(6.0 * np.pi * x), 0.01, 100.0))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 100.0))

    for init in inits:
        s = init.sum()
        if s > 1e-12:
            inits[-1] = init / s  # This only normalizes the last one, fix below

    # Properly normalize all
    normalized = []
    for init in inits:
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-12:
            normalized.append(init / s)
    return normalized


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: spectral flattening, active-set LP alternating, LSE L-BFGS,
    subgradient descent, and peak elimination.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine main length
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 6962

    # ==================================================================
    # Phase 1: Spectral flattening for coarse optimization (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Spectral Flattening ---")

    a_work = None
    if parent is not None:
        a_work = parent.copy()
    else:
        a_work = np.ones(n_main) * 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n_main) / n_main)

    a_work = np.maximum(a_work, 0.0)
    a_work = a_work / a_work.sum()

    # Run spectral flattening
    a_spec = spectral_flattening(a_work, n_iters=2000, alpha=0.8,
                                  time_limit=phase1_deadline)
    sc = compute_score(a_spec)
    report(sc, a_spec, "spectral")

    # Try a few different initializations
    inits = build_initializations(n_main, rng, parent)
    for i, a0 in enumerate(inits[:4]):
        if time.monotonic() > phase1_deadline:
            break
        a_try = spectral_flattening(a0, n_iters=1000, alpha=0.7,
                                     time_limit=phase1_deadline - 5)
        sc = compute_score(a_try)
        report(sc, a_try, f"spec init{i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Active-set LP alternating (45% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Active-Set LP Alternating ---")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    # Progressive LP: start with small K, increase
    K_schedule = [100, 200, 300, 500, 700]
    n_outer = 300
    a_out, sc_out = lp_alternating_active_set(
        a_work, max_iters=n_outer, K_schedule=K_schedule,
        time_limit=phase2_deadline, line_search=True
    )
    report(sc_out, a_out, "LP active set")

    # If we have time, run again from the improved point
    if time.monotonic() < phase2_deadline - 120:
        print("  Second LP round with higher K...")
        K_schedule2 = [300, 500, 700, 1000]
        a_out2, sc_out2 = lp_alternating_active_set(
            a_out, max_iters=200, K_schedule=K_schedule2,
            time_limit=phase2_deadline - 10, line_search=True
        )
        report(sc_out2, a_out2, "LP round 2")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: LSE + L-BFGS with progressive beta (25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: LSE + L-BFGS ---")

    a_norm = best_a.copy()
    S = a_norm.sum()
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [50, 100, 200, 400, 800, 1600, 3200, 6400]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 30:
            break
        elapsed_budget = phase3_deadline - time.monotonic()
        if elapsed_budget < 30:
            break
        print(f"  L-BFGS beta={beta} (budget={elapsed_budget:.0f}s)...", end=" ", flush=True)
        t0 = time.monotonic()
        maxit = min(5000, int(elapsed_budget * 200))
        maxfun = min(50000, int(elapsed_budget * 2000))
        sc, a_opt = run_lbfgs_lse(a_norm, beta, maxiter=maxit, maxfun=maxfun,
                                   gtol=1e-12)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = a_norm.sum()
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient + Peak elimination + Final polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 4: Subgradient & Final Polish ---")

    # Subgradient descent
    a_sub = subgradient_descent(best_a, n_steps=5000,
                                 time_limit=t_start + 0.92 * (float(budget_s) - 30.0))
    sc = compute_score(a_sub)
    report(sc, a_sub, "subgradient")

    # Peak elimination
    a_peak = peak_elimination(best_a, n_steps=2000,
                               time_limit=t_start + 0.96 * (float(budget_s) - 30.0))
    sc = compute_score(a_peak)
    report(sc, a_peak, "peak elim")

    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 30:
        a_final = best_a.copy()
        S = a_final.sum()
        if S > 0:
            a_final = a_final / S
        sc, a_opt = run_lbfgs_lse(a_final, 8000, maxiter=3000, maxfun=30000,
                                   gtol=1e-13)
        report(sc, a_opt, "final LBFGS")

    # One more subgradient round on the best
    if time.monotonic() < deadline - 10:
        a_final = subgradient_descent(best_a, n_steps=2000, time_limit=deadline - 5)
        sc = compute_score(a_final)
        report(sc, a_final, "final subgrad")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
