"""
Optimization of non-negative step function sequences to minimize 2n*max(self-conv)/(sum)^2.
Uses Adam optimizer on LSE-smoothed objective with adaptive beta scheduling,
combined with L-BFGS polish and coordinate descent refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def adam_lse_optimizer(a0, beta_schedule, n_iters, lr_init=0.01, lr_final=0.001,
                       beta1=0.9, beta2=0.999, eps=1e-8, grad_clip=10.0,
                       n_print=5000):
    """Adam optimizer on LSE-smoothed objective with adaptive beta."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        # Get current beta from schedule
        progress = it / max(1, n_iters - 1)
        beta = beta_schedule(progress)

        sc, grad = lse_score_grad(a, beta)

        # Gradient clipping
        gnorm = float(np.linalg.norm(grad))
        if gnorm > grad_clip:
            grad = grad * (grad_clip / gnorm)

        # Learning rate schedule (cosine decay)
        lr = lr_final + 0.5 * (lr_init - lr_final) * (1.0 + np.cos(np.pi * progress))

        # Adam update
        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip and renormalize
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        if it % n_print == 0 or it == n_iters - 1:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_a


def adam_lse_with_restarts(a0, rng, budget_frac, n_restart=5):
    """Run Adam LSE with multiple restarts from perturbed versions."""
    n = len(a0)
    best_a = a0.copy()
    best_sc = compute_score(a0)

    for r in range(n_restart):
        # Geometric beta schedule: from 50 to 50000
        beta_start = 50.0
        beta_end = 50000.0

        def beta_schedule(progress, bs=beta_start, be=beta_end):
            return bs * (be / bs) ** progress

        # Number of iterations based on budget fraction
        n_iters = int(budget_frac * 50000 / n_restart)

        # Perturb for restart
        if r == 0:
            a_cur = a0.copy()
        else:
            a_cur = a0.copy()
            strength = 0.02 * (r + 1)
            a_cur += rng.standard_normal(n) * strength
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = a_cur / a_cur.sum()

        a_opt = adam_lse_optimizer(a_cur, beta_schedule, n_iters,
                                    lr_init=0.05, lr_final=0.001,
                                    beta1=0.9, beta2=0.999,
                                    grad_clip=5.0, n_print=5000)
        sc = compute_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Adam restart {r}: score={sc:.8f}, beta_end={beta_end}")

    return best_a

def fista_lse(a0, beta, n_iters=10000, lr_init=0.01, momentum=0.9):

def gs_flat_target(a, n_iters=300, alpha_init=0.5, alpha_final=0.1):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_inits(n, rng, parent):

        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * x), 0.01))

    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 5.0))
    inits.append(np.maximum(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2, 0.01))

    # Fourier-based
    for _ in range(3):

        for i in range(n_harmonics):
            sig += weights[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
        inits.append(np.maximum(0.5 + sig, 0.01))

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Adam + LSE (main optimization, 40% of time)
    # ==================================================================
    phase2_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Adam + LSE ===")

    # Multiple beta schedules
    schedules = [
        (50, 10000),    # moderate
        (100, 30000),   # aggressive
        (200, 50000),   # very aggressive
        (50, 30000),    # broad
    ]

    for sched_idx, (b_start, b_end) in enumerate(schedules):
        if time.monotonic() > phase2_end:
            break

        def make_beta_schedule(bs, be):
            def bs_fn(progress):
                return bs * (be / bs) ** progress
            return bs_fn

        beta_fn = make_beta_schedule(b_start, b_end)
        n_iters = 20000
        a_opt = adam_lse_optimizer(best_a, beta_fn, n_iters,
                                    lr_init=0.03, lr_final=0.001,
                                    beta1=0.9, beta2=0.999,
                                    grad_clip=5.0, n_print=5000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Adam schedule {sched_idx} (beta {b_start}->{b_end}): {sc:.8f}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: FISTA + LSE (15% of time)
    # ==================================================================
    phase3_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 3: FISTA + LSE ===")

    for beta_val in [5000, 20000, 50000]:
        if time.monotonic() > phase3_end:
            break
        a_opt = fista_lse(best_a, beta=beta_val, n_iters=15000, lr_init=0.005)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  FISTA beta={beta_val}: {sc:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS polish (20% of time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 4: L-BFGS polish ===")

    for beta_val in [10000, 30000, 80000, 150000]:
        if time.monotonic() > phase4_end:
            break
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta_val,),
                          maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta_val}: {sc:.8f}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Coordinate descent + final Adam (15% of time)
    # ==================================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Coordinate descent + final polish ===")

    # Coordinate descent (use only a subset of coordinates for speed)
    n_cd_coords = min(n, 2000)
    a_cd = coordinate_descent_polish(best_a, n_passes=20, n_coords_per_pass=n_cd_coords)
    sc = compute_score(a_cd)
    update_best(sc, a_cd)
    print(f"  Coord descent: {sc:.8f}")

    # Final Adam with high beta
    if time.monotonic() < phase5_end:
        def final_beta(progress):
            return 50000.0 * (100000.0 / 50000.0) ** progress

        a_opt = adam_lse_optimizer(best_a, final_beta, 10000,
                                    lr_init=0.01, lr_final=0.0005,
                                    beta1=0.9, beta2=0.999,
                                    grad_clip=3.0, n_print=2000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Final Adam: {sc:.8f}")

    # One more L-BFGS at very high beta
    if time.monotonic() < phase5_end:
        a_opt = run_lbfgs(best_a, lse_score_grad, (200000,),
                          maxiter=3000, maxfun=15000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Final LBFGS: {sc:.8f}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
