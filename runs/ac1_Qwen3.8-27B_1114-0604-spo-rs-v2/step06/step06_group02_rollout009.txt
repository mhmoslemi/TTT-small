"""
Optimized FISTA + Spectral Shaping for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations: accelerated proximal gradient (FISTA) with log-sum-exp smoothing,
adaptive spectral targets, multi-length exploration, and structured initializations.

import numpy as np
from scipy.optimize import minimize
import time


def _fft_size(m):
    """Next power of 2 >= m."""
    return 1 << (m - 1).bit_length()


def _convolve(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    M = 2 * n - 1
    N = _fft_size(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def _compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = _convolve(a)
    return float(2.0 * n * np.max(c) / (S * S))


def _lse_objective_and_grad(a, beta):
    """
    Log-sum-exp smoothed objective and its gradient.
    F(a) = 2n * LSE_beta(a*a) / sum(a)^2
    Returns (F, grad) where grad has same shape as a.
    """
    n = len(a)
    c = _convolve(a)  # length 2n-1
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta  # smooth approximation of max

    # Weights q_k proportional to exp(beta * (c_k - c_max))
    q = exp_s / exp_s.sum()  # length 2n-1

    # Gradient of LSE w.r.t. a_j = sum_k q_k * a_{k-j} = cross-correlation
    h = np.convolve(q, a[::-1])[n - 1: n - 1 + n]  # length n

    # Full gradient of F = 2n * L / S^2
    # dF/da_j = 2n * (dL/da_j * S^2 - L * 2S * 1) / S^4
    #         = 2n * (h_j / S^2 - 2L / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

    F = 2.0 * n * L / S2
    return F, grad


def _fista(a0, beta, n_iters, eta, deadline):
    """
    FISTA (accelerated proximal gradient) for minimizing F(a) with a >= 0.
    Normalizes after each step.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    S = a.sum()
    a = a / S

    best_a = a.copy()
    best_sc = _compute_score(a)

    z = a.copy()
    t = 1.0
    N_fft = _fft_size(3 * n - 1)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # FISTA momentum
        y = z + ((t - 1) / (t + 1)) * (z - a) if t > 1 else z.copy()

        # Gradient evaluation at y
        F_y, grad_y = _lse_objective_and_grad(y, beta)

        # Step
        a_new = y - eta * grad_y

        # Project onto a >= 0
        a_new = np.maximum(a_new, 0.0)

        # Normalize
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = z.copy()
            S_new = 1.0
        a_new = a_new / S_new

        # FISTA update
        z = a.copy()
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        a = a_new

        # Track best
        sc = _compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _spectral_shaping(a, n_iters, target_type='flat', deadline=None):
    """
    Fienup-style spectral shaping with non-trivial targets.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    N_fft = _fft_size(2 * n - 1)
    best_a = a.copy()
    best_sc = _compute_score(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break

        progress = it / max(1, n_iters)
        alpha = 0.6 * (1.0 - progress) ** 2 + 0.05

        A = np.fft.rfft(a, N_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target spectrum shape
        if target_type == 'flat':
            P_target = np.full_like(P, P_mean)
        elif target_type == 'raised_cos':
            freqs = np.fft.rfftfreq(N_fft, 1.0 / N_fft)
            shape = 0.5 + 0.5 * np.cos(2.0 * np.pi * freqs * n / N_fft)
            P_target = P_mean * shape / float(shape.mean())
        elif target_type == 'gentle_peak':
            freqs = np.fft.rfftfreq(N_fft, 1.0 / N_fft)
            shape = 1.0 + 0.15 * np.exp(-4.0 * (freqs * n) ** 2)
            P_target = P_mean * shape / float(shape.mean())
        else:
            P_target = np.full_like(P, P_mean)

        # Interpolate
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        # Update
        scale = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * scale ** alpha
        a_new = np.fft.irfft(A_new, N_fft)[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = _compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def _lbfgs_optimize(a0, beta, maxiter=20000, maxfun=500000, deadline=None):
    """L-BFGS-B optimization with log-sum-exp smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S

    bounds = [(0.0, 1000.0)] * n

    def fun(x):
        if deadline and time.monotonic() > deadline:
            return 1e10
        return _lse_objective_and_grad(x, beta)[0]

    def jac(x):
        if deadline and time.monotonic() > deadline:
            return 1e10, np.zeros(n)
        return _lse_objective_and_grad(x, beta)

    try:
        res = minimize(
            fun, a0, jac=jac,
            method='L-BFGS-B', bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                     'iprint': -1}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()

    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S

    sc = _compute_score(a_out)
    return sc, a_out


def _make_symmetric(a):
    """Symmetrize a sequence."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def _structured_init(n, rng, variant):
    """Create structured initial sequences."""
    if variant == 0:
        # Uniform
        return np.ones(n) / n
    elif variant == 1:
        # Raised cosine
        x = np.arange(n) / (n - 1)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif variant == 2:
        # Top-hat with smooth edges
        w = 0.6  # width fraction
        x = np.arange(n) / (n - 1)
        a = np.ones(n)
        edge = 0.05
        a[:int(n * (0.5 - w / 2 - edge))] = 0.1
        a[int(n * (0.5 + w / 2 + edge)):] = 0.1
        # Smooth edges
        for i in range(int(n * edge)):
            frac = i / (n * edge)
            idx1 = int(n * (0.5 - w / 2 - edge)) + i
            if 0 <= idx1 < n:
                a[idx1] = 0.1 + 0.9 * frac
            idx2 = int(n * (0.5 + w / 2 + edge)) - i
            if 0 <= idx2 < n:
                a[idx2] = 0.1 + 0.9 * frac
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif variant == 3:
        # Binary-like (Pseudo-random)
        a = rng.random(n)
        threshold = np.median(a)
        a = np.where(a > threshold, 1.0, 0.3)
        return a / a.sum()
    elif variant == 4:
        # Exponential decay (from center)
        center = (n - 1) / 2.0
        x = np.abs(np.arange(n) - center) / center
        a = np.exp(-1.0 * x ** 1.5)
        return a / a.sum()
    elif variant == 5:
        # Sinc-squared (spreading in frequency)
        center = (n - 1) / 2.0
        x = (np.arange(n) - center) / (n * 0.1)
        a = (np.sinc(x)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        # Random smooth
        a = rng.random(n)
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / 5.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def _interpolate_sequence(a, target_n):
    """Interpolate a sequence to a target length."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0.0)
    S = new_a.sum()
    if S > 1e-12:
        new_a = new_a / S
    return new_a


def _refined_fista(a0, beta_schedule, iters_per_beta, deadline):
    """
    FISTA with a schedule of increasing beta values.
    This is the main workhorse optimization.
    """
    n = len(a0)
    best_a = np.maximum(a0.copy(), 0.0)
    S = best_a.sum()
    if S < 1e-12:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / S
    best_sc = _compute_score(best_a)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Adaptive step size
        # For F = 2n * LSE / S^2, the Lipschitz constant is roughly 2n * beta * max_conv / S^2
        c = _convolve(best_a)
        c_max = float(np.max(c))
        S_val = float(np.sum(best_a))
        # Conservative eta
        eta = 1.0 / (4.0 * n * beta * max(c_max / (S_val * S_val), 1e-10) + 1e-10)
        eta = min(eta, 0.1)

        # Also try a fixed conservative step
        eta_fixed = 1.0 / (2.0 * n * beta + 1.0)

        for sub in range(2):
            if time.monotonic() > deadline:
                break
            eta_use = eta if sub == 0 else eta_fixed
            a_new, sc_new = _fista(best_a, beta, iters_per_beta, eta_use, deadline)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()

    return best_a, best_sc


def _coordinate_descent(a0, deadline, n_sweeps=5):
    """
    Local coordinate descent: for each element, try shifting it to reduce the peak.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _compute_score(a)

    N_fft = _fft_size(2 * n - 1)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = _convolve(a)
        c_max = float(np.max(c))
        S = float(a.sum())

        # Find the peak index
        k_star = int(np.argmax(c))

        # Concentrate updates on elements contributing to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        active = np.arange(j_start, j_end + 1)

        if len(active) > 50:
            # Top 50 contributors
            contribs = a[active] * a[k_star - active]
            top_idx = np.argsort(contribs)[::-1][:50]
            active = active[top_idx]

        for j in active:
            if time.monotonic() > deadline:
                break
            j = int(j)
            old_val = a[j]
            best_local = best_sc
            best_v = old_val

            for f in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                a[j] = old_val * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = old_val
                    continue
                a_norm = a / S_tmp
                sc_tmp = _compute_score(a_norm)
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_v = old_val * f

            a[j] = best_v

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = _compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _stochastic_restarts(a0, rng, n_restarts, deadline):
    """Try random perturbations + quick FISTA to escape local minima."""
    n = len(a0)
    best_a = a0.copy()
    best_sc = _compute_score(a0)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        strategy = r % 5
        if strategy == 0:
            # Gaussian perturbation
            a_pert = a0 * (1.0 + 0.05 * rng.standard_normal(n))
        elif strategy == 1:
            # Peak-targeted perturbation
            c = _convolve(a0)
            k_star = int(np.argmax(c))
            a_pert = a0.copy()
            j_start = max(0, k_star - n // 4)
            j_end = min(n - 1, k_star + n // 4)
            a_pert[j_start:j_end + 1] *= (1.0 + 0.1 * rng.standard_normal(j_end - j_start + 1))
        elif strategy == 2:
            # Frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, max(2, n // 100))
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a0 * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Block perturbation
            block_size = max(1, n // 20)
            block = rng.integers(0, max(1, n - block_size))
            a_pert = a0.copy()
            a_pert[block:block + block_size] *= (1.0 + 0.15 * rng.standard_normal(block_size))
        else:
            # Peak inversion: reduce peak-contributing elements
            c = _convolve(a0)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a0.copy()
            a_pert[j_start:j_end + 1] *= 0.8

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick FISTA
        a_opt, sc_opt = _fista(a_pert, 500, 500, 0.01, deadline)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    1. Load parent / generate diverse initializations
    2. Spectral shaping for each candidate
    3. FISTA with beta schedule (main workhorse)
    4. L-BFGS polish
    5. Coordinate descent
    6. Stochastic restarts
    7. Final high-beta refinement
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0  # 30s buffer for final verification

    best_score = np.inf
    best_a = np.ones(2048) / 2048.0

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # ---- Load parent ----
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                sc_p = _compute_score(parent)
                print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
                update(sc_p, parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    print(f"Initial best: {best_score:.10f}")

    # ---- Define lengths to try ----
    target_lengths = [2048, 4096, 8192]
    # Weight more time to feasible lengths
    lengths_to_try = []
    for blen in target_lengths:
        if parent is not None and abs(blen - len(parent)) < len(parent) * 0.5:
            lengths_to_try.append((blen, 1.5))  # boost
        else:
            lengths_to_try.append((blen, 1.0))

    total_weight = sum(w for _, w in lengths_to_try)
    time_per_length = {}
    for blen, w in lengths_to_try:
        time_per_length[blen] = (total - 120.0) * w / total_weight  # 120s reserve

    # ---- Main optimization loop over lengths ----
    for idx, (blen, w) in enumerate(lengths_to_try):
        phase_deadline = t_start + 30.0 + sum(time_per_length[l] for l in target_lengths[:idx+1] if l <= blen)
        # Simpler: just track global deadline
        phase_deadline = min(phase_deadline, deadline)

        if time.monotonic() > deadline - 60:
            break

        print(f"\n{'='*50}")
        print(f"Processing length n={blen} (time budget ~{time_per_length[blen]:.0f}s)")
        print(f"{'='*50}")

        # Get starting points for this length
        candidates = []

        # From parent (interpolated)
        if parent is not None:
            a_parent = _interpolate_sequence(parent, blen)
            a_parent = _make_symmetric(a_parent)
            S = a_parent.sum()
            if S > 1e-12:
                a_parent = a_parent / S
            candidates.append(('parent', a_parent))

        # Structured inits
        for v in range(7):
            a_init = _structured_init(blen, rng, v)
            a_init = _make_symmetric(a_init)
            S = a_init.sum()
            if S > 1e-12:
                a_init = a_init / S
            candidates.append((f'struct-{v}', a_init))

        # From current best (interpolated)
        if len(best_a) != blen:
            a_best = _interpolate_sequence(best_a, blen)
            a_best = _make_symmetric(a_best)
            S = a_best.sum()
            if S > 1e-12:
                a_best = a_best / S
            candidates.append(('best_interp', a_best))

        # Limit to top candidates by initial score (keep diverse set)
        scored = []
        for name, a_c in candidates:
            sc_c = _compute_score(a_c)
            scored.append((sc_c, name, a_c))
        scored.sort(key=lambda x: x[0])
        candidates = [(name, a) for _, name, a in scored[:6]]  # Keep top 6

        # Small time for spectral shaping
        t_spec_end = min(time.monotonic() + 0.08 * time_per_length[blen], phase_deadline)
        t_fista_start = t_spec_end

        for name, a_c in candidates:
            if time.monotonic() > t_spec_end:
                break

            # Phase A: Spectral shaping (short)
            a_spec = _spectral_shaping(a_c, n_iters=100, target_type='gentle_peak', deadline=t_spec_end)
            sc_spec = _compute_score(a_spec)

            # Phase B: FISTA with beta schedule
            if time.monotonic() < phase_deadline - 10:
                beta_sched = [50, 100, 200, 500, 1000, 2000, 5000]
                # Allocate iterations based on remaining time
                remaining_time = phase_deadline - time.monotonic()
                n_iters_total = int(remaining_time * 30)  # ~30 iterations/sec estimate
                n_iters_total = max(n_iters_total, 500)
                iters_per_beta = max(30, n_iters_total // len(beta_sched))

                a_fista, sc_fista = _refined_fista(a_spec, beta_sched, iters_per_beta, phase_deadline)
                sc_after = sc_fista
                a_after = a_fista
            else:
                sc_after = sc_spec
                a_after = a_spec

            update(sc_after, a_after, f"{name}/fista")

            # Phase C: L-BFGS polish (if time)
            if time.monotonic() < phase_deadline - 5:
                remaining = phase_deadline - 5 - time.monotonic()
                if remaining > 15:
                    maxit = int(min(10000, remaining * 8))
                    sc_lb, a_lb = _lbfgs_optimize(a_after, 2000, maxiter=maxit,
                                                   maxfun=maxit * 20, deadline=phase_deadline - 5)
                    update(sc_lb, a_lb, f"{name}/lbfgs")

    # ---- Global refinement phases ----
    print(f"\n{'='*50}")
    print(f"Global refinement (remaining: {deadline - time.monotonic():.0f}s)")
    print(f"{'='*50}")

    n_best = len(best_a)

    # High-beta FISTA
    t_hi_end = deadline - 60.0
    if time.monotonic() < t_hi_end:
        print("\nHigh-beta FISTA...")
        beta_hi = [5000, 10000, 20000, 50000, 100000]
        remaining = t_hi_end - time.monotonic()
        iters_per = max(50, int(remaining * 20 / len(beta_hi)))
        a_hi, sc_hi = _refined_fista(best_a, beta_hi, iters_per, t_hi_end)
        update(sc_hi, a_hi, "HI-BETA-FISTA")

    # L-BFGS with very high beta
    if time.monotonic() < deadline - 30:
        print("\nFinal L-BFGS polish...")
        remaining = deadline - 30 - time.monotonic()
        for beta_final in [50000, 200000]:
            if time.monotonic() > deadline - 30:
                break
            rem = deadline - 30 - time.monotonic()
            if rem < 10:
                break
            maxit = int(min(8000, rem * 5))
            sc_f, a_f = _lbfgs_optimize(best_a, beta_final, maxiter=maxit,
                                         maxfun=maxit * 15, deadline=deadline - 30)
            update(sc_f, a_f, f"FINAL-LBFGS-{beta_final}")

    # Coordinate descent
    if time.monotonic() < deadline - 20:
        print("\nCoordinate descent...")
        a_cd, sc_cd = _coordinate_descent(best_a, deadline - 20, n_sweeps=3)
        update(sc_cd, a_cd, "COORD-DESCENT")

    # Stochastic restarts (last resort)
    if time.monotonic() < deadline - 15:
        print("\nStochastic restarts...")
        a_sr, sc_sr = _stochastic_restarts(best_a, rng, n_restarts=5, deadline=deadline - 15)
        update(sc_sr, a_sr, "STOCHASTIC")

    # ---- Final verification and output ----
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
