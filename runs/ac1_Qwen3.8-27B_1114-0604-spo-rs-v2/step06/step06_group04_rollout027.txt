import numpy as np

from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf

    return float(2.0 * n * c.max() / (S * S))

def _normalize(a):

    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())

def _resample(a, n):

    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)

def _softmax_grad_objective(x, beta):
    """
    Compute the LSE-smoothed objective and gradient in softmax parameterization.
    x is the log-parameter, a = softmax(x) (normalized to sum=1).
    """
    n = x.size
    # Softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute autocorrelation
    c = _fast_self_conv(a)
    d = n * c  # Since sum(a) = 1, S^2 = 1
    
    # LSE of d
    dmax = d.max()
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = e.sum()
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    
    # Gradient w.r.t. a: same as before
    # R[i] = sum_j a[j] * w[i+j]
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3*n-2]
    R = conv_full[n-1:2*n-1]
    
    grad_a = n * R - n * T  # Since S=1
    
    # Chain rule: da/dx = a * I - a * a^T = a[:,None] - np.outer(a,a)
    # grad_x = J^T @ grad_a where J = a(I - 1a^T)
    # grad_x[i] = a[i] * grad_a[i] - a[i] * sum_j a[j] * grad_a[j]
    dot = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot)
    
    return score, grad_x

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Power-mean smooth approximation with Adam optimizer
2. Multi-scale p-increase strategy
3. Diverse multi-start with structural initializations
4. Perturbation restarts with quick re-optimization
5. Final Nesterov subgradient polish

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _power_mean_value_grad(a, p):
    """
    Compute the p-th power mean approximation of 2n*max(c)/S^2 and its gradient.
    a is assumed to have sum=1 (on the simplex).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    a_norm = a / S  # Ensure sum = 1
    
    c = _fast_self_conv(a_norm)
    # Power mean
    c_clipped = np.maximum(c, 1e-30)
    c_pow = c_clipped ** (p - 1)
    M_p = (c_clipped ** p).sum() ** (1.0 / p)
    
    if not np.isfinite(M_p) or M_p < 1e-30:
        return 1e10, np.zeros(n)
    
    score = 2.0 * n * M_p  # Since S = 1
    
    # Gradient of M_p w.r.t. a
    # dM_p/da[i] = M_p^(p-1) * sum_k c[k]^(p-1) * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] (with a[m]=0 for m outside [0,n-1])
    # So dM_p/da[i] = 2 * M_p^(p-1) * sum_k c[k]^(p-1) * a[k-i]
    # = 2 * M_p^(p-1) * R[i] where R[i] = sum_j c[j+i]^(p-1) * a[j]
    
    w = c_pow  # w[k] = c[k]^(p-1), length 2n-1
    
    # R[i] = conv(a_rev, w)[i + n - 1]
    a_rev = a_norm[::-1]
    N2 = _next_pow2(3 * n)
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3*n-2]
    R = conv_full[n-1:2*n-1]
    
    # dM_p/da[i] = 2 * M_p^(p-1) * R[i]
    # df/da[i] = 2n * dM_p/da[i] - 2n*M_p * 2/S (chain rule for normalization)
    # Since we're on the simplex (S=1), the normalization gradient is:
    # df/da[i] = 2n * 2 * M_p^(p-1) * R[i] - 2n * M_p * 2
    # But we need to account for the fact that we're optimizing on the simplex.
    # The effective gradient (for projection method) is just:
    grad = 2.0 * n * 2.0 * (M_p ** (p - 1)) * R
    # Subtract mean to make it "tangent" to the simplex
    grad = grad - np.mean(grad)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    return float(score), grad


def _adam_optimize(a0, p_schedule, iters_per_p, lr_schedule=None, deadline=None):
    """
    Adam optimizer for the power-mean approximation.
    p_schedule: list of p values to use (increasing).
    iters_per_p: number of iterations for each p value.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    if lr_schedule is None:
        lr_schedule = [2e-3] * len(p_schedule)
    
    for pi, p in enumerate(p_schedule):
        if deadline is not None and time.monotonic() > float(deadline) - 5.0:
            break
        lr = lr_schedule[min(pi, len(lr_schedule)-1)]
        
        for it in range(iters_per_p):
            if deadline is not None and time.monotonic() > float(deadline) - 3.0:
                break
            
            t += 1
            score, grad = _power_mean_value_grad(a, p)
            if not np.isfinite(score):
                break
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = _project_simplex(a_new)
            
            # Check improvement
            exact_sc = _exact_score(a_new)
            if np.isfinite(exact_sc) and exact_sc < best_score:
                best_score = exact_sc
                best_a = a_new.copy()
            
            a = a_new
            # Gradual learning rate decay within each p
            lr *= 0.9995
    
    return best_score, best_a


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """Nesterov accelerated subgradient method."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    
    score0, _ = _subgrad_exact(a, k_active, tol)
    # Estimate gradient norm
    _, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    
    eta = 0.03 / g_norm
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy
        
        score_y, grad_y = _subgrad_exact(y, k_active, tol)
        if not np.isfinite(score_y):
            break
        
        step = eta / (1.0 + 0.05 * it)
        x_new = y - step * grad_y
        x_new = _project_simplex(x_new)
        
        v = x_new - a
        a = x_new
        t = t_new
        
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if it > 0 and it % 100 == 0 and not (sc < best_score + 1e-10):
            eta *= 0.5
    
    return best_score, best_a


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """Compute exact subgradient of 2n*max(c)/S^2 at point a (sum a = 1)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    a_norm = a / S
    c = _fast_self_conv(a_norm)
    dmax = float(c.max())
    score = 2.0 * n * dmax
    
    threshold = dmax - tol * dmax
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > k_active:
        top_idx = np.argsort(c[active])[-k_active:]
        active = active[top_idx]
    
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    
    N2 = _next_pow2(3 * n)
    a_rev = a_norm[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3*n-2]
    R = conv_full[n-1:2*n-1]
    
    T = float(w_active.dot(c))
    grad = 2.0 * n * R - 2.0 * n * T
    grad = grad - np.mean(grad)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.5 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Low-frequency smooth perturbation
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** 2.0)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    else:
        # Block perturbation: add/remove mass from random blocks
        q = a.copy()
        n_blocks = int(rng.integers(2, 6))
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            delta = rng.uniform(-0.1, 0.1)
            q[start:end] += delta
        q = np.clip(q, 0.0, None)
    
    return _normalize(q)


def _generate_inits(n, rng, parent, cap=25):
    """Generate diverse initial sequences."""
    inits = []
    
    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))
    
    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
    
    x = np.linspace(0.0, 1.0, n)
    
    _add(np.ones(n))
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 1.0]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
    for rate in [1.0, 2.0, 4.0, 8.0]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1.0 - x)) + 0.01)
    for w in [0.1, 0.15, 0.25]:
        for ctr in [0.2, 0.3, 0.35]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))
    
    return inits


def _quick_optimize(a, deadline):
    """Quick optimization for a candidate."""
    a = _normalize(a)
    p_schedule = [4, 16, 64]
    iters = max(50, int((deadline - time.monotonic()) * 20))
    sc, seq = _adam_optimize(a, p_schedule, iters // len(p_schedule), deadline=deadline)
    rem = deadline - time.monotonic()
    if rem > 10.0:
        sc2, seq2 = _nesterov_optimize(seq, n_iters=int(rem * 15), deadline=deadline, k_active=3, tol=1e-7)
        if sc2 < sc:
            sc, seq = sc2, seq2
    return sc, seq
