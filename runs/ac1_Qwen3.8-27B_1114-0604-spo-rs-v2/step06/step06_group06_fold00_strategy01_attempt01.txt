    def normalize_sequence(seq, method):
        n_seq = len(seq)
        S = sum(seq)
        clipped = np.clip(seq, 0.0, 1000.0)
        if method == 'clip':
            max_val = clipped.max()
            if max_val > 1000:
                scale = 1000 / max_val
                seq = [min(1000, x) for x in clipped * scale]
            return seq
        elif method == 'smooth':
            if S < 0.01:
                scale = np.sqrt(2*n_seq)/0.5
                return [0.5 * scale] * n_seq
            else:
                seq = [max(0, x) for x in seq]
                return seq
