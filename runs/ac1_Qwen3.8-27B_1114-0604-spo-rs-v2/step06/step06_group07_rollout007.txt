    return 2.0 * n * float(np.max(c)) / (S * S)

    """Build the sparse matrix A such that (A @ b)_k = (a*b)_k for linear convolution."""
    n = len(a)
    M = 2 * n - 1
    rows = []
    cols = []
    vals = []

        j_end = min(k, n - 1)

            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(M, n))
    return A

Iterative LP-based optimization for minimum-peak non-negative autocorrelation.
Combines the SOTA LP approach with spectral shaping and L-BFGS polish.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def precompute_conv_structure(n):
    """Precompute the (row, col) indices for the convolution constraint matrix."""
    M = 2 * n - 1
    rows_list = []
    cols_list = []
    for j in range(n):
        num = min(n, M - j)  # always n for valid j
        rows_list.append(np.full(num, j, dtype=np.int32))
        cols_list.append(np.arange(j, j + num, dtype=np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def fill_conv_values(a, rows, cols):
    """Fill in the values a[k-j] for the precomputed structure."""
    n = len(a)
    vals = np.empty(len(rows), dtype=np.float64)
    # For column j, the values are a[0], a[1], ..., a[n-1]
    # Position in the array: column j starts at position j*n
    for j in range(n):
        start = j * n
        end = start + n
        vals[start:end] = a
    return vals


def solve_lp_conv(a, M_val, rows, cols, time_limit=None):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M_val, b >= 0."""
    n = len(a)
    vals = fill_conv_values(a, rows, cols)
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c_obj = np.full(n, -1.0)
    b_ub = np.full(2*n-1, M_val)
    bounds = [(0, None)] * n
    
    options = {}
    if time_limit:
        options['time_limit'] = time_limit
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if res.status == 0:
        return res.x
    return None
