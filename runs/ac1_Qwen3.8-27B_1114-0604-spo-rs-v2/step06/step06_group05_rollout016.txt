        if z.size != self.n-1 or not np.all(np.isfinite(z)):
            return 1e100, np.zeros(self.n-1)
        n=self.n; M=self.M

            z=np.clip(z, -700, 700)
        e_z=np.exp(z - m)
        e0=np.exp(-m)

        if Z <= 0 or not np.isfinite(Z):

            p=np.empty(n)

        exps=np.exp(scaled)
        ssum=float(exps.sum())
        if ssum <=0 or not finite:

            L=bmax

            w=exps/ssum

"""
Non-negative step-function heights minimizing 2*n*max(a*a)/(sum(a))^2.

Approach:
  1. Work on the simplex (probability sequences, sum = 1) using a softmax
     parameterization to keep entries non-negative and remove scale invariance.
  2. Use log-sum-exp smoothing of the max convolution coefficient and polish
     with L-BFGS-B in the softmax log-ratio variables.
  3. Use a Fourier square-root IFTA warm start: force the Fourier-square spectrum
     toward the flat linear-convolution target while enforcing non-negativity and
     finite support in the time domain.
  4. Add exact multi-peak Adam / subgradient polishing in the simplex.
  5. Try the parent length (if present) and, if time remains, a larger length.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _npow2(L):
    L = int(L)
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _interp(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64)
    if a.size == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a)


def _convolution_fft(a, M):
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)


def _score_norm(p, M):
    p = _normalize(p)
    n = len(p)
    c = _convolution_fft(p, M)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def _score_any(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-15:
        return np.inf
    n = a.size
    M = _npow2(2 * n - 1)
    c = _convolution_fft(a, M)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx / (S * S)


def _central_binomial(n):
    c = np.ones(int(n), dtype=np.float64)
    for k in range(1, int(n)):
        c[k] = c[k - 1] * (4.0 * k - 2.0) / (4.0 * k)
    return c


def _make_flat_target_transform(n, M):
    b = np.zeros(int(M), dtype=np.float64)
    b[:2 * n - 1] = 1.0 / float(2 * n - 1)
    return np.fft.rfft(b, M)


def _central_conv_weighted(w, p, M, n):
    # Returns h_i = 2 * sum_k w_k p_{k-i}, i = 0..n-1.
    rev = p[::-1]
    r = np.fft.irfft(np.fft.rfft(rev, M) * np.fft.rfft(w, M), M)
    return 2.0 * r[n - 1:2 * n - 1]


class _SoftmaxLSE(object):
    """
    Smoothed max-convolution score and gradient in log-ratio softmax variables.

    Variables z have length n-1. The probability is
        p_0 = 1 / Z,
        p_i = exp(z_i) / Z, i=1..n-1.
    Objective: 2*n*LogSumExp(beta * b) / beta, b = p * p.
    """

    def __init__(self, n, beta, M):
        self.n = int(n)
        self.beta = float(beta)
        self.M = int(M)
        self.wlen = 2 * self.n - 1
        self.two_n = 2.0 * float(self.n)

    def __call__(self, z):
        n = self.n
        M = self.M
        z = np.asarray(z, dtype=np.float64)
        if z.size != n - 1 or not np.all(np.isfinite(z)):
            return 1e100, np.zeros(n - 1, dtype=np.float64)

        # Stable softmax with fixed reference p0.
        m = max(0.0, float(z.max()))
        if m > 700.0:
            z = np.clip(z, -700.0, 700.0)
            m = float(z.max())
        ez = np.exp(z - m)
        e0 = np.exp(-m)
            Z = ez.sum() + e0
        if not np.isfinite(Z) or Z <= 0.0:
            p = np.ones(n, dtype=np.float64) / float(n)
        else:
            p = np.empty(n, dtype=np.float64)
            p[0] = e0 / Z
            p[1:] = ez / Z
        ps = float(p.sum())
        if not np.isfinite(ps) or ps <= 0.0:
            p = np.ones(n, dtype=np.float64) / float(n)
        else:
            p = p / ps

        A = np.fft.rfft(p, M)
        B = A * A
        b = np.fft.irfft(B, M)[:self.wlen]
        b = np.maximum(b, 0.0)
        bmax = float(b.max())
        if not np.isfinite(bmax) or bmax <= 0.0:
            return 1e100, np.zeros(n - 1, dtype=np.float64)

        scaled = self.beta * (b - bmax)
        exps = np.exp(scaled)
        ssum = float(exps.sum())
        if not np.isfinite(ssum) or ssum <= 0.0:
            w = np.full(self.wlen, 1.0 / self.wlen, dtype=np.float64)
            L = bmax
        else:
            w = exps / ssum
            L = bmax + float(np.log(ssum)) / self.beta

        h = _central_conv_weighted(w, p, M, n)
        T = 2.0 * float(np.dot(w, b))
        grad = self.two_n * (h[1:] - T * p[1:])
        score = self.two_n * L

        if not np.isfinite(score):
            score = 1e100
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n - 1, dtype=np.float64)
        return float(score), grad


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    n = z.size + 1
    m = max(0.0, float(z.max()))
    if m > 700.0:
        z = np.clip(z, -700.0, 700.0)
        m = float(z.max())
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    Z = ez.sum() + e0
    if not np.isfinite(Z) or Z <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    p = np.empty(n, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = ez / Z
    return _normalize(p)


def _make_z(p, bound=25.0):
    p = _normalize(p)
    # Keep the fixed p0 from becoming pathologically tiny.
    if p[0] < 1e-12:
        p[0] = 1e-12
        p = _normalize(p)
    z = np.log(p[1:] / p[0])
    z = np.clip(z, -bound, bound)
    return z


def _make_bounds(m, bound=25.0):
    return [(-bound, bound)] * int(m)


def _lse_maxfun(n):
    n = int(n)
    if n <= 5120:
        return 12000
    if n <= 10240:
        return 8000
    if n <= 20480:
        return 4000
    if n <= 40960:
        return 2500
    return 1500


def _run_lse_z(p, beta, M, maxfun, bound=25.0):
    n = len(p)
    z = _make_z(p, bound=bound)
    bounds = _make_bounds(n - 1, bound=bound)
    problem = _SoftmaxLSE(n, beta, M)
    try:
        res = minimize(
            problem,
            z,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": maxfun,
                "maxfun": maxfun,
                "ftol": 1e-16,
                "gtol": 1e-10,
                "maxcor": 50,
                "maxls": 20,
            },
        )
        if hasattr(res, "x") and np.all(np.isfinite(res.x)):
            z = res.x
    except Exception:
        pass
    return _softmax_from_z(z)


def _top_peak_adam(p, M, deadline, n_steps=300, npeaks=64, lr=0.02, bound=25.0):
    p = _normalize(p)
    n = len(p)
    z = _make_z(p, bound=bound)
    mv = np.zeros(n - 1, dtype=np.float64)
    vv = np.zeros(n - 1, dtype=np.float64)
    t = 1

    best_p = p.copy()
    best_sc = _score_norm(p, M)
    stale = 0

    for step in range(int(n_steps)):
        if deadline is not None and time.monotonic() > deadline:
            break

        p = _softmax_from_z(z)
        c = _convolution_fft(p, M)[:2 * n - 1]
        c = np.maximum(c, 0.0)

        if step % 20 == 0:
            sc = 2.0 * n * float(c.max())
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_p = p.copy()
                stale = 0
            else:
                stale += 1
                if stale > 8:
                    break

        L = 2 * n - 1
        k = int(min(int(npeaks), L))
        if k <= 0:
            break
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, L - k)[-k:]
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0 / float(k)

        h = _central_conv_weighted(w, p, M, n)
        T = 2.0 * float(np.dot(w, c))
        grad = (2.0 * n) * (h[1:] - T * p[1:])
        if not np.all(np.isfinite(grad)):
            break

        mv = 0.9 * mv + 0.1 * grad
        vv = 0.999 * vv + 0.001 * grad * grad
        m_hat = mv / (1.0 - 0.9 ** t)
        v_hat = vv / (1.0 - 0.999 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = np.clip(z, -bound, bound)
        t += 1

    p_final = _softmax_from_z(z)
    sc_final = _score_norm(p_final, M)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_p = p_final.copy()
    return best_p


def _spectral_rfft_init(n, M, Bt, a_init, iters=500, alpha0=0.7, alpha1=0.1, deadline=None):
    a = _normalize(a_init)
    n = int(n)
    A = np.fft.rfft(a, M)
    c = _convolution_fft(a, M)[:2 * n - 1]
    best_sc = 2.0 * n * float(np.max(np.maximum(c, 0.0)))
    if not np.isfinite(best_sc):
        best_sc = 1e100
    best_a = a.copy()

    for it in range(int(iters)):
        if deadline is not None and time.monotonic() > deadline:
            break

        alpha = alpha0 + (alpha1 - alpha0) * (float(it) / max(1.0, float(iters - 1)))
        B = A * A
        Bnew = (1.0 - alpha) * B + alpha * Bt

        Sroot = np.sqrt(Bnew.astype(np.complex128))
        if Sroot.size == 0:
            break

        v0 = float(np.real(Bnew[0]))
        Sroot[0] = np.complex128(np.sqrt(v0) if v0 > 0.0 else 0.0, 0.0)

        if Sroot.size > 1:
            dot = np.real(Sroot * np.conj(A))
            mask = dot < 0.0
            mask[0] = False
            mask[-1] = False
            Sroot[mask] *= -1.0

        last = Sroot.size - 1
        lv = float(np.real(Bnew[last]))
        Sroot[last] = np.complex128(np.sqrt(lv) if lv > 0.0 else 0.0, 0.0)

        d = 0.25 + 0.75 * alpha
        Anew = (1.0 - d) * A + d * Sroot
        Anew[last] = np.complex128(float(np.real(Anew[last])), 0.0)

        a_new = np.fft.irfft(Anew, M)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if not np.isfinite(s) or s <= 1e-12:
            break
        a_new = a_new / s

        A = np.fft.rfft(a_new, M)
        c = _convolution_fft(a_new, M)[:2 * n - 1]
        sc = 2.0 * n * float(np.max(np.maximum(c, 0.0)))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        if (it + 1) % 100 == 0:
            print(f"      spectral init it={it+1}, score={best_sc:.8f}")

    return best_a


def _random_phase_p(n, M, Bt, rng):
    mag = np.sqrt(np.abs(Bt)).astype(np.float64)
    ph = rng.random(mag.size)
    A = mag * np.exp(1j * ph)
    if mag.size > 0:
        A[0] = mag[0]
        A[-1] = mag[-1]
    a = np.fft.irfft(A, M)[:n].real
    return _normalize(a)


def _make_initials(n, rng, parent_interp):
    n = int(n)
    x = np.arange(n, dtype=np.float64) / float(n)
    two_pi = 2.0 * np.pi
    inits = []

    inits.append(np.ones(n, dtype=np.float64))

    if parent_interp is not None:
        inits.append(_normalize(parent_interp))
        inits.append(_normalize(0.7 * parent_interp + 0.3 * np.ones(n)))

    # One-sided binomial/square-root style starting point.
    inits.append(_central_binomial(n))

    # Low-order sinusoidal shapes, including the phase that reduces the center.
    for ph in (0.0, 0.25, 0.5, 0.75):
        shape = 1.0 + 0.75 * np.sin(two_pi * x + ph)
        inits.append(np.maximum(shape, 0.02))

    for ph in (0.1, 0.4, 0.7):
        shape = 1.0 + 0.45 * np.cos(two_pi * x + ph) + 0.25 * np.sin(4.0 * np.pi * x + 2.0 * ph)
        inits.append(np.maximum(shape, 0.02))

    # Random low-frequency positive shapes.
    for _ in range(3):
        shape = np.ones(n, dtype=np.float64)
        for m in range(1, 9):
            amp = 0.12 * float(rng.normal())
            ph = float(rng.uniform(0.0, two_pi))
            shape = shape + amp * np.cos(two_pi * (m * x + ph))
        inits.append(np.maximum(shape, 0.02))

    # Smooth positive noise.
    inits.append(np.abs(rng.normal(size=n)) + 0.05)

    normalized = []
    for a in inits:
        a = _normalize(a)
        if a.size == n and np.all(np.isfinite(a)):
            normalized.append(a)
    return normalized


def _select_top(pairs, k):
    if not pairs:
        return []
    scores = [float(p[0]) for p in pairs]
    order = np.argsort(scores)
    k = max(0, int(min(int(k), len(order))))
    if k == 0:
        return []
    return [pairs[int(i)] for i in order[:k]]


def _default_betas(n, num=4):
    n = int(n)
    base = max(2000.0, float(n) / 2.0)
    mults = [1.0, 4.0, 16.0, 64.0, 256.0]
    out = []
    for i in range(max(1, min(int(num), len(mults)))):
        out.append(base * mults[i])
    return out


def _refine_sequence(a, M, deadline, betas, bound=25.0):
    a = _normalize(a)
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a, M)
    maxfun = _lse_maxfun(n)

    print(f"    refine start n={n}, score={best_sc:.8f}")

    for beta in betas:
        if deadline is not None and time.monotonic() > deadline:
            break

        before = best_sc
        p = _run_lse_z(best, beta, M, maxfun=maxfun, bound=bound)
        sc = _score_norm(p, M)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best = p
            best_sc = sc
        print(f"      LBFGS beta={int(beta)} score={best_sc:.10f}")

        if best_sc < before + 1e-13 and deadline is not None and time.monotonic() < deadline:
            p2 = _top_peak_adam(best, M, deadline, n_steps=200, npeaks=64, lr=0.02, bound=bound)
            sc2 = _score_norm(p2, M)
            if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
                best = p2
                best_sc = sc2
                print(f"      top-peak after LBFGS beta={int(beta)} score={best_sc:.10f}")

    for npk in (16, 64, 256):
        if deadline is not None and time.monotonic() > deadline:
            break
        p2 = _top_peak_adam(best, M, deadline, n_steps=350, npeaks=npk, lr=0.01, bound=bound)
        sc2 = _score_norm(p2, M)
        if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
            best = p2
            best_sc = sc2
        print(f"      final top-peak k={npk} score={best_sc:.10f}")

    return best


def _optimize_size(N, M, Bt, parent_interp, deadline, seed):

        return a, _score_norm(a, M)

    top_inits = _select_top(scored, 8)
    best_a, best_sc = top_inits[0]

    print(f"    best initial score={best_sc:.10f}")

    candidate_pairs = []
    for sc, a in top_inits:
        candidate_pairs.append((float(sc), a))

    # Spectral warm starts for the best initializations.
    for sc, a in top_inits:
        if time.monotonic() > deadline:
            break
        a1 = _spectral_rfft_init(N, M, Bt, a, iters=500, alpha0=0.7, alpha1=0.1, deadline=deadline)
        sc1 = _score_norm(a1, M)
        if np.isfinite(sc1):

        if sc1 < best_sc:
            best_a = a1
            best_sc = sc1
            print(f"    new best after spectral: {best_sc:.10f}")

    top_candidates = _select_top(candidate_pairs, 5)
    alloc_fracs = [0.50, 0.25, 0.15, 0.07, 0.03]
    betas = _default_betas(N, num=5 if N <= 10000 else 4)

    for idx, _ in enumerate(top_candidates):
        if time.monotonic() > deadline:

        remaining = max(0.0, float(deadline) - time.monotonic())
        if remaining < 45.0:

        frac = alloc_fracs[idx] if idx < len(alloc_fracs) else 0.03
        alloc = remaining * frac
        if idx == len(top_candidates) - 1:
            cand_deadline = deadline
        else:
            cand_deadline = min(float(deadline), time.monotonic() + alloc)

        sc0, a0 = top_candidates[idx]

        a_opt = _refine_sequence(a0, M, cand_deadline, betas, bound=25.0)
        sc_opt = _score_norm(a_opt, M)
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt

    return best_a, float(best_sc)


def propose_candidate(seed=12345, budget_s=2400.0):
    t0 = time.monotonic()
    total = float(budget_s)
    working_deadline = t0 + total - 150.0

    print("=" * 70)
    print(f"propose_candidate budget={total:.0f}s, working_deadline in {working_deadline - t0:.0f}s")

    best_a = None
    best_sc = np.inf

    parent = globals().get("height_sequence_1", None)
    parent_a = None
    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64).ravel()
            if arr.size >= 64:
                arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.sum() > 0.01:
                    parent_a = arr / arr.sum()
                    if parent_a.size > 50000:
                        parent_a = _interp(parent_a, 20000)
                    psc = _score_any(parent_a)
                    print(f"parent length={parent_a.size}, parent score={psc:.10f}")
                    if np.isfinite(psc):
                        best_a = parent_a.copy()
                        best_sc = float(psc)
        except Exception as e:
            print("parent load failed:", repr(e))
            parent_a = None

    # Choose the primary size. Prefer the parent size if it is in a reasonable range.
    if parent_a is not None and 512 <= parent_a.size <= 30000:
        N1 = int(parent_a.size)
    else:
        N1 = 16384

    N2 = 0
    if N1 < 16384:
        N2 = 16384

    if N2 > 0:
        deadline1 = t0 + 0.68 * (total - 150.0)
    else:
        deadline1 = working_deadline

    def _update_global(a, sc):
        nonlocal best_a, best_sc
        if a is not None and np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  GLOBAL BEST: {best_sc:.10f} N={best_a.size}")

    # Primary size.
    M1 = _npow2(3 * N1 - 2)
    Bt1 = _make_flat_target_transform(N1, M1)
    parent_interp1 = None
    if parent_a is not None:
        parent_interp1 = _interp(parent_a, N1)
    a1, sc1 = _optimize_size(N1, M1, Bt1, parent_interp1, deadline1, seed=int(seed) + 1)
    _update_global(a1, sc1)

    # Secondary larger size if time remains and parent was smaller.
    if N2 > 0 and time.monotonic() < working_deadline - 60.0:
        M2 = _npow2(3 * N2 - 2)
        Bt2 = _make_flat_target_transform(N2, M2)
        if parent_a is not None:
            pt2 = _interp(parent_a, N2)
        elif best_a is not None:
            pt2 = _interp(best_a, N2)
        else:
            pt2 = None
        a2, sc2 = _optimize_size(N2, M2, Bt2, pt2, working_deadline, seed=int(seed) + 2)
        _update_global(a2, sc2)

    # If nothing was found, use a safe fallback.
    if best_a is None or best_a.size == 0:
        n_fallback = parent_a.size if parent_a is not None else max(N1, 1024)
        best_a = np.ones(n_fallback, dtype=np.float64) / float(n_fallback)
        try:
            best_sc = _score_any(best_a)
        except Exception:
            best_sc = np.inf

    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if not np.isfinite(s) or s < 0.01:
        out = np.ones(max(N1, 1024), dtype=np.float64)
        s = float(out.sum())
    out = out / s

    # Optional lightweight verification for small output only.
    try:
        if out.size <= 8192 and time.monotonic() < working_deadline + 60.0:
            test_list = [float(x) for x in out]
            if sum(test_list) >= 0.01:
                verified = float(evaluate_sequence(test_list))
                print(f"light verification evaluate_sequence={verified:.10f}")
    except Exception as e:
        print("verification skipped:", repr(e))

    print("=" * 70)
    print(f"FINAL internal score={best_sc:.10f}, length={out.size}, time={time.monotonic() - t0:.1f}s")
    print("=" * 70)

    return [float(max(0.0, min(1000.0, x))) for x in out]
