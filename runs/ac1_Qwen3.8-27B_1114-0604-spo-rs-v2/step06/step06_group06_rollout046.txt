```python
import cvxpy as cp

def lp_step(a_old, active_lags):
    n = len(a_old)
    c = fast_self_conv(a_old)
    
    a_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        a_var >= 0,
        cp.sum(a_var) == 1
    ]
    
    for k in active_lags:
        grad = compute_grad(a_old, k)
        g_k = c[k]
        rhs = a_old @ grad - g_k
        constraints.append(grad @ a_var - t_var <= rhs)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None
    
    if a_var.value is None:
        return None
    
    a_new = np.array(a_var.value).flatten()
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return None
    a_new = a_new / s
    return a_new
```

One more idea: **Dantzig-Wolfe decomposition**. The problem can be viewed as a semi-infinite program:
minimize t
subject to: a >= 0, sum(a) = 1, a^T C_k a <= t for all k in {0, ..., 2n-2}

The "master problem" is the LP with a finite subset of constraints (the active set), and the "subproblem" is to find a violated constraint (a lag k where a^T C_k a > t at the current solution).

The algorithm:
1. Solve the master problem (LP with current active set) to get a*.
2. Check if a* satisfies all constraints: compute c = conv(a*, a*), find max c[k].
3. If max c[k] <= t* (the LP objective), then a* is feasible for the original problem (approximately).
4. Otherwise, add the lag k* = argmax c[k] to the active set and go to step 1.

This is the cutting-plane method for semi-infinite programs! It's guaranteed to converge (in principle) to the optimal solution of the relaxed problem.

But in practice, the convergence might be slow because we're adding one constraint at a time. To speed it up, I can add multiple violated constraints at each iteration.

```python
def cutting_plane_optimize(a_init, deadline, K_max=200):
    n = len(a_init)
    a = normalize_a(a_init)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    active_lags = []
    
    it = 0
    while time.monotonic() < deadline and it < 500:
        it += 1
        
        # Add violated constraints
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        
        # Find lags where c[k] > current_t (violation)
        # Use the best score so far as a proxy for t
        t_current = cmax  # since sum(a) = 1
        violated = np.where(c > t_current * 0.99)[0]
        
        # Add top violated lags to active set
        if len(violated) > 0:
            # Sort by violation amount
            sort_idx = np.argsort(c[violated])[::-1]
            new_lags = violated[sort_idx]
            for lag in new_lags:
                if lag not in active_lags:
                    active_lags.append(lag)
        
        # Cap the active set size
        if len(active_lags) > K_max:
            # Keep the most violating lags
            c_vals = [c[k] for k in active_lags]
            order = np.argsort(c_vals)[::-1][:K_max]
            active_lags = [active_lags[i] for i in order]
        
        # Solve LP
        a_lp = lp_step(a, active_lags)
        if a_lp is None:
            # LP infeasible or solver failed
            # Try with a smaller active set
            active_lags = active_lags[:max(10, len(active_lags)//2)]
            a_lp = lp_step(a, active_lags)
            if a_lp is None:
                break
        
        # Check improvement
        sc_lp = exact_score(a_lp)
        if sc_lp < best_sc - 1e-10:
            # Linear search
            best_alpha = 1.0
            best_sc_alpha = sc_lp
            for alpha in [0.5, 0.3, 0.1]:
                a_try = (1 - alpha) * a + alpha * a_lp
                a_try = normalize_a(a_try)
                sc_try = exact_score(a_try)
                if sc_try < best_sc_alpha:
                    best_sc_alpha = sc_try
                    best_alpha = alpha
            a = normalize_a((1 - best_alpha) * a + best_alpha * a_lp)
            sc = exact_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            print("  CP it{}: sc={:.9f} |active|={}".format(it, best_sc, len(active_lags)))
        else:
            # No improvement, try with more active lags or break
            if len(active_lags) >= K_max:
                break
            # Expand active set
            c = fast_self_conv(a)
            threshold = cmax * 0.9
            more_violated = np.where(c >= threshold)[0]
            for lag in more_violated:
                if lag not in active_lags:
                    active_lags.append(lag)
    
    return best_sc, best_a
```

```python
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.05 * conv.size))
    if active.sum() > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active2 = np.zeros(conv.size, dtype=bool)
        active2[idx] = True
        active = active2
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_update(a, conv, top_k, eps_frac, eta):
    w = compute_active_weights_conv(conv, top_k=top_k, eps_frac=eps_frac)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def variance_reduction_update(a, conv, top_m, eta):
    d = np.sort(conv)[::-1][:top_m]
    mean_top = float(np.mean(d))
    excess = np.maximum(conv - mean_top, 0.0)
    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def solve_lp_step(a_old, active_lags):
    """Solve one cutting-plane LP. Returns (a_new, t_lp) or (None, None)."""
    if not HAS_CVXPY:
        return None, None

    n = len(a_old)
    c = fast_self_conv(a_old)

    a_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [a_var >= 0, cp.sum(a_var) == 1]

    for k in active_lags:
        Ca = Ck_a(a_old, int(k))
        g_k = float(c[int(k)])
        constraints.append(2.0 * cp.dot(Ca, a_var) - t_var <= g_k)

    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)

    solvers = []
    if hasattr(cp, 'CLARABEL'):
        solvers.append((cp.CLARABEL, {}))
    if hasattr(cp, 'SCS'):
        solvers.append((cp.SCS, {'max_iters': 50000, 'eps': 1e-8}))
    if hasattr(cp, 'ECOS'):
        solvers.append((cp.ECOS, {}))
    if hasattr(cp, 'GLPK'):
        solvers.append((cp.GLPK, {}))

    for solver, skw in solvers:
        try:
            prob.solve(solver=solver, **skw)
            if a_var.value is not None:
                a_new = np.array(a_var.value).flatten().astype(np.float64)
                a_new = np.clip(a_new, 0.0, None)
                s = float(a_new.sum())
                if s < 1e-10:
                    continue
                a_new = a_new / s
                t_val = float(t_var.value) if t_var.value is not None else np.inf
                return a_new, t_val
        except Exception:
            continue

    return None, None


def cutting_plane_optimize(a_init, deadline, verbose=True, tag="CP"):
    """Main cutting-plane LP optimizer with adaptive active set."""
    a = normalize_a(a_init)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    active_lags = []
    stagnant = 0
    max_stagnant = 15

    for it in range(2000):
        if time.monotonic() > deadline - 5.0:
            break

        c = fast_self_conv(a)
        cmax = float(np.max(c))
        sc = float(2.0 * len(a) * cmax)

        # Select active lags: those near the peak
        threshold = cmax * 0.95
        active_mask = c >= threshold
        min_count = max(5, min(50, len(c) // 100))
        if active_mask.sum() < min_count:
            idx = np.argpartition(c, -min_count)[-min_count:]
            active_mask[idx] = True

        # Also include top-K by value
        K = min(30, len(c))
        top_idx = np.argpartition(c, -K)[-K:]
        active_mask[top_idx] = True

        # Cap total active set
        max_active = 120
        if active_mask.sum() > max_active:
            vals = c[active_mask]
            order = np.argsort(vals)[::-1][:max_active]
            active_idx_all = np.where(active_mask)[0]
            keep = active_idx_all[order]
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[keep] = True

        active_lags = list(np.where(active_mask)[0])

        # Solve LP
        a_lp, t_lp = solve_lp_step(a, active_lags)

        if a_lp is None:
            # Try with smaller active set
            if len(active_lags) > 20:
                topK = np.argpartition(c, -20)[-20:]
                a_lp, t_lp = solve_lp_step(a, list(topK))
            if a_lp is None:
                stagnant += 1
                if stagnant >= max_stagnant:
                    if verbose:
                        print("  [{} it{}] LP failed, breaking".format(tag, it))
                    break
                continue

        # Check actual objective (linearization may be optimistic)
        sc_lp = exact_score(a_lp)

        if sc_lp < best_sc - 1e-10:
            # Line search
            best_alpha = 1.0
            best_sc_alpha = sc_lp
            for alpha in [0.75, 0.5, 0.3, 0.15]:
                a_try = (1.0 - alpha) * a + alpha * a_lp
                a_try = normalize_a(a_try)
                sc_try = exact_score(a_try)
                if sc_try < best_sc_alpha:
                    best_sc_alpha = sc_try
                    best_alpha = alpha

            a = normalize_a((1.0 - best_alpha) * a + best_alpha * a_lp)
            sc_now = exact_score(a)
            if sc_now < best_sc:
                best_sc = sc_now
                best_a = a.copy()
                stagnant = 0
                if verbose and (it % 10 == 0 or best_sc < 1.51):
                    print("  [{} it{}] sc={:.9f} |A|={} alpha={:.2f}".format(
                        tag, it, best_sc, len(active_lags), best_alpha))
        else:
            stagnant += 1
            if stagnant >= max_stagnant:
                if verbose:
                    print("  [{} it{}] stagnant, sc={:.9f}".format(tag, it, best_sc))
                break

    return float(best_sc), best_a


def active_set_eg_optimize(a_init, deadline, verbose=True, tag="EG"):
    """Exponentiated gradient with adaptive active set (from prior work)."""
    a = normalize_a(a_init)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    top_k = 5
    eps_frac = 0.05
    eta = 0.3
    stagnant = 0
    max_stagnant = 30

    for phase in range(20):
        if time.monotonic() > deadline - 5.0:
            break
        phase_dead = min(deadline - 5.0, t0 + (deadline - t0 - 5.0) * (phase + 1) / 20.0)

        for it in range(200):
            if time.monotonic() > phase_dead:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))

            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            etas = [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0]
            for eta_try in etas:
                if eta_try < 0.001:
                    continue
                new_a = exp_grad_update(a, conv, top_k, eps_frac, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            vr_a = variance_reduction_update(a, conv, min(20, top_k + 10), best_eta * 0.5)
            vr_a = normalize_a(vr_a)
            vr_sc = exact_score(vr_a)
            if vr_sc < best_sc_iter:
                best_sc_iter = vr_sc
                best_a_iter = vr_a

            if best_sc_iter < best_sc - 1e-10:
                best_sc = best_sc_iter
                best_a = best_a_iter.copy()
                a = best_a_iter
                eta = best_eta
                stagnant = 0
                if verbose and it % 50 == 0:
                    print("  [{} p{} i{}] sc={:.9f} tk={} e={:.3f} eta={:.3f}".format(
                        tag, phase, it, best_sc, top_k, eps_frac, eta))
            else:
                stagnant += 1
                eta *= 0.85
                if stagnant >= max_stagnant:
                    break

        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.5) + 2, 100)
            eps_frac = min(eps_frac * 1.3, 0.15)
            eta = max(0.1, eta * 0.7)
            stagnant = 0
        else:
            top_k = max(3, int(top_k * 0.95))
            eps_frac = max(0.005, eps_frac * 0.95)

    return float(best_sc), best_a


def lse_score_and_grad(a, beta):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxfun=500):
    if not HAS_SCIPY:
        return exact_score(a), a
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun_and_grad(x):
        val, g = lse_score_and_grad(x, beta)
        return val, g

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun_and_grad, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent, cap=25):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    for alpha in (0.3, 0.5, 0.8, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    for f in (1, 2, 3):
        for alpha in (0.5, 1.0, 1.5):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(x - 0.5)) + 0.01)
    for width in (0.05, 0.1, 0.15, 0.2):
        for center in (0.2, 0.3, 0.4):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center) / width) ** 2))) + 0.01)
    for p in (0.5, 1.0, 2.0, 3.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = compute_active_weights_conv(conv, top_k=10, eps_frac=0.02)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def subgrad_fine_polish(a, deadline, verbose=False, tag=""):
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 75, 100]
    etas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0]
    eps_fracs = [0.0, 0.005, 0.01, 0.02, 0.05]

    improved = True
    round_num = 0
    while improved and time.monotonic() < deadline - 3.0 and round_num < 8:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)
        for tk in top_ks:
            if time.monotonic() > deadline - 3.0:
                break
            for ef in eps_fracs:
                for eta in etas:
                    if time.monotonic() > deadline - 3.0:
                        break
                    new_a = exp_grad_update(best_a, conv, tk, ef, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = new_a
                        improved = True
                        if verbose:
                            print("  [{} r{}] tk={} ef={:.3f} eta={:.2f} -> {:.9f}".format(
                                tag, round_num, tk, ef, eta, best_sc))
        for tm in [5, 10, 20, 50, 100]:
            if time.monotonic() > deadline - 3.0:
                break
            for eta in [0.05, 0.1, 0.2, 0.3, 0.5]:
                if time.monotonic() > deadline - 3.0:
                    break
                new_a = variance_reduction_update(best_a, conv, tm, eta)
                new_a = normalize_a(new_a)
                sc = exact_score(new_a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = new_a
                    improved = True

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main entry point. Orchestrates LP, EG, and LSE-LBFGS phases."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent if available
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    n = 10240
    if parent is not None:
        n = int(parent.size)
        if n < 1024:
            n = 8192
        elif n > 20000:
            n = 16384
        a0 = resample_sequence(parent, n)
    else:
        inits = generate_inits(n, rng, None, cap=20)
        best_init_sc = np.inf
        a0 = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a0 = init
        print("No parent. Best init score: {:.9f}".format(best_init_sc))

    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score(a)
    print("Initial: n={}, score={:.9f}".format(n, best_sc))

    # Phase 1: Cutting-plane LP optimization (~45% of time)
    p1_dead = t_start + 0.45 * (total_budget - 30.0)
    sc, a = cutting_plane_optimize(a, deadline=p1_dead, verbose=True, tag="LP1")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 1 (LP): {:.9f}".format(best_sc))

    # Phase 2: EG polishing on the LP result (~20% of time)
    p2_dead = t_start + 0.62 * (total_budget - 30.0)
    sc, a = active_set_eg_optimize(a, deadline=p2_dead, verbose=True, tag="EG")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 2 (EG): {:.9f}".format(best_sc))

    # Phase 3: LP again on the EG-improved solution (~15% of time)
    p3_dead = t_start + 0.75 * (total_budget - 30.0)
    sc, a = cutting_plane_optimize(a, deadline=p3_dead, verbose=True, tag="LP2")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 3 (LP2): {:.9f}".format(best_sc))

    # Phase 4: LSE-LBFGS refinement (~10% of time)
    p4_dead = t_start + 0.85 * (total_budget - 30.0)
    a = normalize_a(best_a)
    for beta in [2000, 5000, 10000, 20000, 50000]:
        if time.monotonic() > p4_dead - 10:
            break
        rem = p4_dead - time.monotonic()
        if rem < 20:
            break
        mf = max(64, int(rem * 0.7 / 0.15))
        mf = min(mf, 2000)
        sc, a = run_lbfgs(a, beta, p4_dead, mf)
        a = normalize_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  Phase 4 beta={}: {:.9f}".format(beta, best_sc))

    # Phase 5: Fine subgradient polish (~5% of time)
    p5_dead = t_start + 0.92 * (total_budget - 30.0)
    sc, a = subgrad_fine_polish(best_a, deadline=p5_dead, verbose=False, tag="polish")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 5 (polish): {:.9f}".format(best_sc))

    # Phase 6: Diversification restarts (~5% of time)
    p6_dead = deadline - 10.0
    n_restarts = 3
    restart_num = 0
    while time.monotonic() < p6_dead - 30 and n_restarts > 0:
        rem = p6_dead - 30.0 - time.monotonic()
        if rem < 40:
            break
        n_restarts -= 1
        restart_num += 1
        p = perturb(best_a, rng, strength=float(rng.uniform(0.03, 0.15)))
        chunk_dead = time.monotonic() + rem * 0.7

        # Try LP on perturbed
        sc_r, seq_r = cutting_plane_optimize(p, deadline=chunk_dead, verbose=False, tag="rL{}".format(restart_num))
        if sc_r > best_sc - 1e-6:
            # EG as fallback
            sc_r, seq_r = active_set_eg_optimize(p, deadline=chunk_dead, verbose=False, tag="rE{}".format(restart_num))
        if sc_r < best_sc:
            best_sc = sc_r
            best_a = seq_r.copy()
            print("Restart {} ({}): {:.9f}".format(restart_num, "LP" if seq_r is not None else "EG", best_sc))

    # Phase 7: Final polish
    if time.monotonic() < deadline - 5:
        sc, a = subgrad_fine_polish(best_a, deadline=deadline - 3, verbose=False, tag="final")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final cleanup
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
```
