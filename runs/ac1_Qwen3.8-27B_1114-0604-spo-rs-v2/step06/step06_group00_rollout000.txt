```python
def solve_lp_direction(a, M):
    """Solve LP to find direction g that keeps conv(a,g) <= M."""
    n = len(a)
    # A @ g <= M * ones(2n-1)
    # A[k,j] = a[k-j] if 0 <= k-j < n
    # Plus g >= 0 (implicit)
    # Plus sum(g) = 1
    
    # Build sparse A
    # For each k (0 to 2n-2), the non-zero entries are at j where 0 <= k-j < n
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    
    # Actually, let's use the scipy.sparse approach
    # A is (2n-1) x n
    # Row k has nonzeros at columns max(0,k-n+1) to min(k,n-1)
    
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(g) (to maximize sum(g))
    c = -np.ones(n)
    
    # Inequality: A @ g <= M
    b = np.full(2*n-1, M)
    
    # Equality: sum(g) = 1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```
