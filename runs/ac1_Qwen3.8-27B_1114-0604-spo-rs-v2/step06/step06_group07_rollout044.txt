"""
Optimized step function construction using:
1. Fienup spectral shaping for initialization
2. LSE-smoothed L-BFGS for main optimization  
3. Subgradient descent with simplex projection
4. Peak-directed coordinate descent
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta):
    """Compute LSE objective and gradient."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    C = float(c.max())
    scaled = beta * (c - C)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    L = C + np.log(exp_sum) / beta
    
    # Gradient of L w.r.t. a
    # grad_L[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # = 2 * conv(p, a_rev)[i + n - 1]
    a_rev = a[::-1]
    M2 = M + n - 1  # = 3n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)
    grad_L = 2.0 * conv_pa[n-1:2*n-1]
    
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n / (S**3)) * (S * grad_L - 2.0 * L)
    
    return score, grad

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fienup_geometric(a, n_iters=400, alpha_start=0.05, alpha_end=0.85):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def subgradient_descent(a, n_iters=5000, n_peaks=10, use_line_search=True):
    """Subgradient descent with simplex projection and optional line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S  # Work on the simplex
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Polyak averaging
    avg_a = np.zeros(n)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        M_val = float(c.max())
        sc = 2.0 * n * M_val  # since sum(a) = 1
        
        avg_a += a
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        if n_peaks >= len(c):
            peak_mask = np.ones(len(c), dtype=bool)
        else:
            peak_mask = np.zeros(len(c), dtype=bool)
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            peak_mask[top_idx] = True
        
        # Compute subgradient of max (approximated by top peaks)
        sg = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[js:je+1][::-1]
        
        # Normalize subgradient
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break
        
        sg = sg / gn  # Unit subgradient
        
        # Step size: use 1/sqrt(t) schedule
        step = 1.0 / np.sqrt(1.0 + it)
        step = min(step, 0.5)
        
        a_new = a - step * sg
        a_new = project_simplex(a_new)
        
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    # Return the averaged solution or the best
    avg_a /= (it + 1)
    avg_a = project_simplex(avg_a)
    sc_avg = compute_score(avg_a)
    
    if sc_avg < best_sc:
        return avg_a
    return best_a

    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

    )

    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

"""
Optimized step function minimization using:
1. Fienup spectral shaping for initialization
2. LSE-smoothed L-BFGS with high beta values for main optimization
3. Subgradient descent with simplex projection
4. Coordinate descent polish
Key improvements: higher LSE betas, proper simplex projection,
better time allocation, and coordinate descent refinement.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value exactly."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_grad(a, beta):
    """LSE-smoothed objective with gradient."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    C = float(c.max())
    scaled = beta * (c - C)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    p = exp_scaled / exp_sum
    L = C + float(np.log(exp_sum)) / beta

    # Gradient of L: dL/da[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * conv(p, a_rev)[i + n - 1]
    a_rev = a[::-1]
    M2 = M + n - 1  # 3n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)
    grad_L = 2.0 * conv_pa[n - 1:2 * n - 1]

    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n / (S ** 3)) * (S * grad_L - 2.0 * L)

    return score, grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fienup_geometric(a, n_iters=400, alpha_start=0.05, alpha_end=0.85,
                     use_soft_clip=True):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        if use_soft_clip:
            gate = 1.0 / (1.0 + np.exp(-30.0 * np.clip(a_new, -50, 50)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_dual_constraint(a, n_iters=400, alpha=0.5, n_peaks=10):
    """
    Fienup that flattens the actual auto-convolution spectrum.
    Uses the FFT of the convolution directly to identify and reduce peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]

        # Find peaks in c and try to reduce them via spectral modification
        # The autocorrelation c[k] = sum_j a[j]*a[k-j]
        # In frequency domain: C(f) = |A(f)|^2 (for circular conv)
        # For linear conv, we approximate by modifying the spectrum

        # Use the peak locations to guide spectral shaping
        if n_peaks < M:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(M)

        # Compute the "peak energy ratio"
        total_energy = float(np.sum(c ** 2))
        peak_energy = float(np.sum(c[top_idx] ** 2))
        ratio = peak_energy / max(total_energy, 1e-30)

        # Adaptive alpha: stronger shaping when peaks are more concentrated
        adapt_alpha = alpha * (0.5 + ratio)
        adapt_alpha = min(adapt_alpha, 0.95)

        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** adapt_alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_descent(a, n_iters=5000, n_peaks=10, step_init=0.1, decay=0.9999):
    """Subgradient descent with simplex projection."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    # Polyak-Ruppert averaging
    avg_a = np.zeros(n)
    count = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        M_val = float(c.max())
        sc = 2.0 * n * M_val  # sum(a) = 1

        avg_a += a
        count += 1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        if n_peaks >= len(c):
            peak_idx = np.arange(len(c))
        else:
            peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            # Only keep those close to max
            thresh = c.max() * 0.9
            peak_idx = peak_idx[c[peak_idx] >= thresh]
            if len(peak_idx) == 0:
                peak_idx = np.array([np.argmax(c)])

        # Subgradient
        sg = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break

        # Step size
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)

        a_new = a - step * sg
        a_new = project_simplex(a_new)

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        step *= decay

    # Check averaged solution
    if count > 0:
        avg_a = project_simplex(avg_a / count)
        sc_avg = compute_score(avg_a)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = avg_a.copy()

    return best_a


def coordinate_descent(a, n_rounds=3, beta=5000):
    """
    Coordinate descent: optimize each element a[i] in turn
    by minimizing the LSE objective in 1D.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    for rnd in range(n_rounds):
        # Random permutation for diversity
        order = np.random.default_rng(42 + rnd).permutation(n)

        for i in order:
            # Current state
            old_val = a[i]
            if old_val < 1e-15:
                # Try a small positive value
                a[i] = 0.01
                sc = compute_score(a)
                if sc > compute_score(np.concatenate([a[:i], [old_val], a[i+1:]])):
                    a[i] = old_val
                continue

            # Minimize over a[i] in [0, 10*old_val]
            def obj_1d(x):
                a[i] = x
                res, _ = lse_objective_grad(a, beta)
                a[i] = old_val
                return res

            # Actually, let's be more efficient: compute the effect of changing a[i]
            # on the convolution directly
            # c[k] changes for k in [i, i+n-1] (where a[k-i] is valid)
            # c_new[k] = c[k] + (x - old_val) * (2*a[k-i] - old_val) for affected k
            # (removing old_val's contribution and adding x's)

            # Simpler: just use the LSE gradient to determine direction
            _, grad = lse_objective_grad(a, beta)
            gi = grad[i]

            if gi > 0:
                # Should decrease a[i]
                new_val = max(0.0, old_val - old_val * 0.1)
            elif gi < 0:
                # Should increase a[i]
                new_val = old_val * 1.1
            else:
                continue

            # Try the new value
            a[i] = new_val
            sc_new = compute_score(a)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
            else:
                a[i] = old_val

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lse_objective_grad_fixed_sum(a, beta, S_target):
    """LSE objective with sum constraint via penalty (alternative)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    C = float(c.max())
    scaled = beta * (c - C)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    p = exp_scaled / exp_sum
    L = C + float(np.log(exp_sum)) / beta

    # Full objective: 2n*L/S^2 + penalty*(S - S_target)^2
    score = 2.0 * n * L / (S * S)

    a_rev = a[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)
    grad_L = 2.0 * conv_pa[n - 1:2 * n - 1]

    grad = (2.0 * n / (S ** 3)) * (S * grad_L - 2.0 * L)

    return score, grad


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Fienup spectral shaping (25% time)
    Phase 2: LSE-LBFGS with increasing beta (45% time)
    Phase 3: Subgradient descent (15% time)
    Phase 4: Final polish with coordinate descent and high-beta LSE (15% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    safety_margin = 40.0
    deadline = t_start + total_budget - safety_margin

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        n_work = len(parent)
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
    else:
        n_work = 8192
        best_a = np.ones(n_work) * 0.5
        update_best(compute_score(best_a), best_a, "UNIFORM")

    print(f"n={n_work}, parent={'yes' if parent is not None else 'no'}, "
          f"deadline in {deadline - t_start:.0f}s")

    # ============================================================
    # Phase 1: Fienup spectral shaping (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - safety_margin)
    print(f"\n=== Phase 1: Fienup (until t={phase1_end - t_start:.0f}s) ===")

    # From parent with various parameters
    if parent is not None:
        for alpha_end in [0.4, 0.6, 0.8]:
            if time.monotonic() > phase1_end:
                break
            a_f = fienup_geometric(parent, n_iters=200,
                                   alpha_start=0.02, alpha_end=alpha_end)
            update_best(compute_score(a_f), a_f, f"FIENUP-P p={alpha_end}")

        # Dual constraint Fienup
        if time.monotonic() < phase1_end:
            a_f = fienup_dual_constraint(parent, n_iters=200, alpha=0.4, n_peaks=20)
            update_best(compute_score(a_f), a_f, "FIENUP-DUAL")

    # From other initializations
    x = np.arange(n_work, dtype=np.float64) / n_work
    inits = [
        ("uniform", np.ones(n_work) * 0.5),
        ("edge", np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01)),
        ("cosine", np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01)),
        ("gauss", np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05),
    ]
    for name, a0 in inits:
        if time.monotonic() > phase1_end:
            break
        a_f = fienup_geometric(a0, n_iters=150, alpha_start=0.05, alpha_end=0.7)
        update_best(compute_score(a_f), a_f, f"FIENUP-{name}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS main optimization (45% time)
    # ============================================================
    phase2_end = t_start + 0.70 * (total_budget - safety_margin)
    print(f"\n=== Phase 2: LSE-LBFGS (until t={phase2_end - t_start:.0f}s) ===")

    # Increasingly aggressive beta values
    beta_schedule = [500, 2000, 10000, 50000, 100000, 200000, 500000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 45:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")

        # Scale iterations with available time
        maxiter = min(4000, int(remaining * 15))
        maxfun = maxiter * 20
        if maxiter < 100:
            break

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")

        # If improved, also try a slightly perturbed version
        if improved and time.monotonic() < phase2_end - 30:
            a_pert = best_a * (1.0 + 0.005 * rng.standard_normal(n_work))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = a_pert / a_pert.sum()
            sc2, a_opt2 = run_lbfgs(a_pert, beta, maxiter=max(200, maxiter // 3),
                                    maxfun=max(1000, maxfun // 3))
            update_best(sc2, a_opt2, f"LBFGS-PERT b={beta}")

    # Also try from parent at high beta
    if parent is not None and time.monotonic() < phase2_end - 30:
        for beta in [50000, 100000]:
            if time.monotonic() > phase2_end - 30:
                break
            remaining = phase2_end - time.monotonic()
            if remaining < 30:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, f"LBFGS-PARENT b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient descent (15% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - safety_margin)
    print(f"\n=== Phase 3: Subgradient (until t={phase3_end - t_start:.0f}s) ===")

    for n_peaks in [5, 10, 30, 100]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        n_iters = min(3000, int(remaining * 3))
        if n_iters < 100:
            break
        print(f"  n_peaks={n_peaks}, iters={n_iters}")

        # Normalize to simplex for subgradient
        a_norm = best_a.copy()
        a_norm = a_norm / a_norm.sum()
        a_sg = subgradient_descent(a_norm, n_iters=n_iters, n_peaks=n_peaks,
                                   step_init=0.05, decay=0.99995)
        # Convert back (score is scale-invariant)
        update_best(compute_score(a_sg), a_sg, f"SG p={n_peaks}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish (until t={deadline - t_start:.0f}s) ===")

    # Very high beta LSE
    for beta in [200000, 500000, 1000000]:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        print(f"  FINAL LBFGS beta={beta}, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=min(3000, int(remaining * 10)),
                              maxfun=min(30000, int(remaining * 100)))
        update_best(sc, a_opt, f"FIN b={beta}")

    # Coordinate descent polish
    if time.monotonic() < deadline - 20:
        print(f"  Coordinate descent...")
        a_cd = coordinate_descent(best_a, n_rounds=2, beta=50000)
        update_best(compute_score(a_cd), a_cd, "COORD")

    # More Fienup from current best (conservative)
    if time.monotonic() < deadline - 20:
        a_f = fienup_geometric(best_a, n_iters=100, alpha_start=0.01, alpha_end=0.2)
        update_best(compute_score(a_f), a_f, "FIN-FIENUP")

    # Perturbation restarts
    if time.monotonic() < deadline - 15:
        for trial in range(4):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.005 * (trial + 1)
            a_p = best_a * (1.0 + noise * rng.standard_normal(n_work))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                break
            a_p = a_p / S
            sc = compute_score(a_p)
            update_best(sc, a_p, f"PERT {trial}")

            if time.monotonic() < deadline - 20:
                sc2, a_opt = run_lbfgs(a_p, 100000, maxiter=2000, maxfun=30000)
                update_best(sc2, a_opt, f"PERT-LB {trial}")

    # ============================================================
    # Final LSE-LBFGS squeeze with very high beta
    # ============================================================
    if time.monotonic() < deadline - 5:
        sc_final, a_final = run_lbfgs(best_a, 1000000, maxiter=2000, maxfun=30000)
        update_best(sc_final, a_final, "FINAL-SQUEEZE")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
