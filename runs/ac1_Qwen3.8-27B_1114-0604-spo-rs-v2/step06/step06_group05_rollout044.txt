"""
Spectral optimization with log-parameterization for minimizing
2n * max(self-convolution) / (sum)^2.
Uses LSE-smoothed L-BFGS in log-space, subgradient peak reduction,
and adaptive multi-scale refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a: 2 * (p * a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def lse_score_grad_b(b, beta):
    """LSE-smoothed score and gradient w.r.t. b (where a = exp(b))."""
    # Numerical stability: shift b so max is 0
    b_shift = b - b.max()
    a = np.exp(b_shift)
    score, grad_a = lse_score_grad_a(a, beta)
    grad_b = grad_a * a
    return score, grad_b


def run_lbfgs_b(b0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on b parameterization."""
    b0 = b0.copy()
    
    def wrapped(x):
        return lse_score_grad_b(x, beta)
    
    res = minimize(
        wrapped, b0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return res.x


def a_from_b(b):
    """Convert b-space to a-space with normalization."""
    b_shift = b - b.max()
    a = np.exp(b_shift)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def b_from_a(a):
    """Convert a-space to b-space."""
    a_safe = np.maximum(a, 1e-10)
    return np.log(a_safe)


def subgradient_peak_reduction(a0, n_iters=5000, top_k=50, lr_init=0.001, lr_decay=0.9999):
    """Subgradient method on the true max (averaged over top-k peaks)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-k peaks
        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Subgradient: average over top peaks
        # d(max conv)/da_j = average of 2*a_{k-j} over selected peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += 2.0 * a[k-j_end:k-j_start+1][::-1]
        sg /= len(top_idx)
        
        # Full gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        a -= lr * grad
        a = np.maximum(a, 1e-10)
        S = a.sum()
        a = a / S
        lr *= lr_decay
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a

        a = a_from_b(b)
        sc = compute_score(a)

            best_score = sc
            best_b = b.copy()
    
    return best_b


def peak_directive_moves(a0, n_rounds=100, top_k=20, reduction=0.9, grow=1.05):
    """Direct peak reduction: reduce elements contributing to peaks, grow others."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Top peaks
        peak_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Find elements that contribute most to peaks
        reduce_mask = np.zeros(n, dtype=bool)
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end+1]
                partner = a[k-j_end:k-j_start+1][::-1]
                contrib = seg * partner
                # Mark top contributors
                thresh = np.max(contrib) * 0.5
                local_mask = contrib > thresh
                reduce_mask[j_start:j_end+1] |= local_mask
        
        # Reduce peak contributors, slightly grow non-contributors
        a[reduce_mask] *= reduction
        a[~reduce_mask] *= grow
        
        a = np.maximum(a, 1e-10)
        S = a.sum()
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

The LSE-smoothed approximation of max(c) is:
`L = max(c) + (1/β) * log(sum_k exp(β * (c_k - max(c))))`

The score is:
`score = 2n * L / S²` where `S = sum(a)`

The gradient of `L` with respect to `a_j`:
`dL/da_j = sum_k p_k * (dc_k/da_j)`

where `p_k = exp(β*(c_k - max(c))) / sum_m exp(β*(c_m - max(c)))`

And `dc_k/da_j = 2 * a_{k-j}` (when valid, as derived earlier).

So `dL/da_j = 2 * sum_k p_k * a_{k-j}` (over valid k)

This is `2 * (p ⊛ a)[j]` where `⊛` denotes a specific cross-correlation.

"""

Key innovations:
1. Log-parameterization (a=exp(b)) eliminates non-negativity constraint entirely

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def lse_score_grad_b(b, beta):
    """LSE-smoothed score and gradient w.r.t. b where a=exp(b)."""
    b_shift = b - b.max()
    a = np.exp(b_shift)
    a = a / max(a.sum(), 1e-30)
    score, grad_a = lse_score_grad_a_raw(a, beta)
    grad_b = grad_a * a
    return score, grad_b


def lse_score_grad_a_raw(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (no normalization assumption)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs_b(b0, beta, maxiter=5000, maxfun=15000):
    """Run L-BFGS-B on b parameterization."""
    b0 = b0.astype(np.float64).copy()

    def wrapped(x):
        return lse_score_grad_b(x, beta)

    res = minimize(
        wrapped, b0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return res.x


def a_from_b(b):
    """Convert b-space to normalized a-space."""
    b_shift = b - b.max()
    a = np.exp(b_shift)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def b_from_a(a):
    """Convert a-space to b-space."""
    a_safe = np.maximum(a, 1e-15)
    return np.log(a_safe)


def subgradient_peak_reduction(a0, n_iters=10000, top_k=50, lr_init=0.001, lr_decay=0.99995):
    """Subgradient method on true max, averaged over top-k peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_k >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partner = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * partner
        sg /= len(top_idx)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        a -= lr * grad
        a = np.maximum(a, 1e-10)
        S = a.sum()
        if S < 1e-10:
            break
        a = a / S
        lr *= lr_decay

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def coordinate_descent(a0, n_passes=5, step_frac=0.05):
    """Coordinate descent: try adjusting each element up and down."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)

    for p in range(n_passes):
        changed = False
        for j in range(n):
            if a[j] < 1e-8:
                continue
            step = a[j] * step_frac
            for factor in [1.0 + step_frac, 1.0 - step_frac]:
                a[j] *= factor
                a = np.maximum(a, 1e-10)
                S_new = a.sum()
                a = a / S_new
                c_new = fast_conv(a)
                c_max_new = float(c_new.max())
                score_new = 2.0 * n * c_max_new / (S_new * S_new)
                if score_new < best_sc - 1e-10:
                    best_sc = score_new
                    best_a = a.copy()
                    score = score_new
                    c = c_new
                    c_max = c_max_new
                    changed = True
                else:
                    a[j] /= factor
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
        if not changed:
            break

    return best_a


def build_inits(n, rng, parent):

    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(np.linspace(0, 1, n), x_old, parent))

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x))

    for _ in range(4):
        M = rng.integers(5, 20)
        freqs = rng.integers(1, M + 1, size=M)
        amps = rng.standard_normal(M) * 0.15
        phases = rng.uniform(0, 2.0 * np.pi, M)
        sig = np.zeros(n)
        for i in range(M):
            sig += amps[i] * np.cos(2.0 * np.pi * freqs[i] * x + phases[i])
        inits.append(np.maximum(0.5 + sig, 0.01))

    inits.append(np.maximum(0.3 + 0.4 * np.abs(np.sin(np.pi * x)), 0.01))
    inits.append(0.5 + 0.2 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))

    result = []
    for init in inits:
        init = np.asarray(init, dtype=np.float64).ravel()
        if len(init) > n:
            init = init[:n]
        elif len(init) < n:
            init = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(init)), init)
        init = np.maximum(init, 0.01)
        result.append(init)

    return result


def annealed_lbfgs(a0, betas, maxiter_per_beta=5000, maxfun_per_beta=15000):
    """Run LSE-LBFGS with annealed beta schedule in b-space."""

    best_a = a0.copy()
    best_sc = compute_score(a0)

    for beta in betas:
        b = run_lbfgs_b(b, beta, maxiter=maxiter_per_beta, maxfun=maxfun_per_beta)

        sc = compute_score(a_out)
        if sc < best_sc:
            best_sc = sc
            best_a = a_out.copy()

    return best_a

        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.01, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
        if a.sum() < 0.01:
            return False
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            best_a = np.interp(np.linspace(0, 1, n), x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting: n={n}, initial best={best_score:.8f}")
    print(f"Time budget: {budget_s - 45:.0f}s")

    # Phase 1: Annealed LSE-LBFGS from best starting point
    phase1_end = t_start + 0.40 * (budget_s - 45.0)
    print(f"\n=== Phase 1: Annealed LSE-LBFGS (budget +{phase1_end - t_start:.0f}s) ===")

    betas_coarse = [50, 200, 500, 2000, 5000, 20000, 50000, 200000, 500000]
    t0 = time.monotonic()
    a_opt = annealed_lbfgs(best_a, betas_coarse, maxiter_per_beta=5000, maxfun_per_beta=15000)
    sc = compute_score(a_opt)
    update_best(sc, a_opt)
    print(f"  Main anneal: score={sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # Try from a few other starting points
    inits = build_inits(n, rng, parent if len(parent) == n else None)
    for i, a0 in enumerate(inits[1:], 1):
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_opt2 = annealed_lbfgs(a0, betas_coarse[:6], maxiter_per_beta=3000, maxfun_per_beta=10000)
        sc2 = compute_score(a_opt2)
        update_best(sc2, a_opt2)
        print(f"  Init {i}: score={sc2:.8f} ({time.monotonic() - t0:.1f}s)")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: Subgradient refinement
    phase2_end = t_start + 0.60 * (budget_s - 45.0)
    print(f"\n=== Phase 2: Subgradient (budget +{phase2_end - t_start:.0f}s) ===")

    for top_k in [10, 30, 50, 100, 200]:
        if time.monotonic() > phase2_end:
            break
        t0 = time.monotonic()
        a_sg = subgradient_peak_reduction(best_a, n_iters=10000, top_k=top_k,
                                          lr_init=0.001, lr_decay=0.99995)
        sc = compute_score(a_sg)
        update_best(sc, a_sg)
        print(f"  Subgrad k={top_k}: score={sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # LSE refinement after subgradient
    if time.monotonic() < phase2_end:
        t0 = time.monotonic()
        b_cur = b_from_a(best_a)
        for beta in [50000, 200000, 1000000]:
            if time.monotonic() > phase2_end:
                break
            b_cur = run_lbfgs_b(b_cur, beta, maxiter=5000, maxfun=15000)
            a_out = a_from_b(b_cur)
            sc = compute_score(a_out)
            update_best(sc, a_out)
        print(f"  LSE after subgrad: ({time.monotonic() - t0:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Coordinate descent + targeted moves
    phase3_end = t_start + 0.80 * (budget_s - 45.0)
    print(f"\n=== Phase 3: Coordinate descent (budget +{phase3_end - t_start:.0f}s) ===")

    t0 = time.monotonic()
    a_cd = coordinate_descent(best_a, n_passes=3, step_frac=0.03)
    sc = compute_score(a_cd)
    update_best(sc, a_cd)
    print(f"  CoordDescent: score={sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # Interleave short LSE and subgradient
    b_cur = b_from_a(best_a)
    for rnd in range(50):
        if time.monotonic() > phase3_end:
            break
        beta = 100000 + 50000 * rnd
        b_cur = run_lbfgs_b(b_cur, beta, maxiter=2000, maxfun=5000)
        a_out = a_from_b(b_cur)
        sc = compute_score(a_out)
        update_best(sc, a_out)

        if rnd % 10 == 0:
            a_sg = subgradient_peak_reduction(best_a, n_iters=3000, top_k=50,
                                              lr_init=0.0005, lr_decay=0.99999)
            sc2 = compute_score(a_sg)
            update_best(sc2, a_sg)
            b_cur = b_from_a(best_a)

        if rnd % 20 == 0:
            print(f"  Round {rnd}: best = {best_score:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Final high-beta polishing
    print(f"\n=== Phase 4: Final polish (deadline +{deadline - t_start:.0f}s) ===")

    b_cur = b_from_a(best_a)
    for beta in [500000, 2000000, 5000000, 20000000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        b_cur = run_lbfgs_b(b_cur, beta, maxiter=5000, maxfun=15000)
        a_out = a_from_b(b_cur)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  Beta={beta}: score={sc:.8f} ({time.monotonic() - t0:.1f}s)")

    # Final subgradient push
    if time.monotonic() < deadline - 10:
        a_sg = subgradient_peak_reduction(best_a, n_iters=8000, top_k=30,
                                          lr_init=0.0003, lr_decay=0.99999)
        sc = compute_score(a_sg)
        update_best(sc, a_sg)

    # One more coordinate descent pass
    if time.monotonic() < deadline - 5:
        a_cd = coordinate_descent(best_a, n_passes=2, step_frac=0.02)
        sc = compute_score(a_cd)
        update_best(sc, a_cd)

    # Verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
