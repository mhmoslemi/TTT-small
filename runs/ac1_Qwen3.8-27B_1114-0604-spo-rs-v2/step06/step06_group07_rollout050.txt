```python
from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) → minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_j f[k-j] * b[j] for valid j
    # Matrix A: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            if 0 <= k-j < n:
                A[k, j] = f[k-j]
    
    b_vec = np.full(2*n-1, M)
    
    # Non-negativity: b[j] >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    return res.x
```

```python
def fast_conv(a):

    S2 = S * S
    S3 = S2 * S
```
```python
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
```
