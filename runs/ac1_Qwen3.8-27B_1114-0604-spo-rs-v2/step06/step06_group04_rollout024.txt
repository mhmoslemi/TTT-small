def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _peak_diagnostics(a, percentile=90):
    """Analyze the convolution peak structure."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fast_self_conv(a)
    cmax = float(c.max())
    if cmax < 1e-15:
        return cmax, 0.0, [], 0
    # Spread: ratio of max to median
    c_sorted = np.sort(c)
    median_c = float(c_sorted[len(c_sorted)//2])
    spread = cmax / max(median_c, 1e-15)
    # Find peaks above percentile
    thresh = np.percentile(c, percentile)
    active_peaks = np.where(c >= thresh)[0]
    return cmax, spread, active_peaks.tolist(), len(active_peaks)


def _adaptive_beta(c, base_beta=5000):
    """Adapt LSE beta based on convolution spread."""
    cmax = float(c.max())
    if cmax < 1e-15:
        return base_beta
    # If peaks are well-separated, use higher beta
    c_sorted = np.sort(c)[::-1]
    top5 = c_sorted[:min(5, len(c_sorted))]
    rest_mean = float(c_sorted[min(5, len(c_sorted)):].mean()) if len(c_sorted) > 5 else top5[-1]
    separation = float(top5[-1]) / max(rest_mean, 1e-15)
    # More separation -> lower beta (smoother); less separation -> higher beta (sharper)
    if separation > 3.0:
        return int(base_beta * 0.5)
    elif separation > 1.5:
        return int(base_beta)
    else:
        return int(base_beta * 2.0)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    # Find active set
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    # Compute weights
    d_active = d[active]
    if weighting == 'uniform':
        weights = np.ones(len(active)) / len(active)
    elif weighting == 'linear':
        # Higher weight for larger peaks
        diffs = d_active - d_active.min() + 1e-15
        weights = diffs / diffs.sum()
    elif weighting == 'quadratic':
        diffs = d_active - d_active.min() + 1e-15
        weights = diffs**2 / (diffs**2).sum()
    elif weighting == 'inverse_gap':
        # Higher weight for peaks closer to the max (smaller gap)
        gaps = dmax - d_active + 1e-15
        weights = (1.0 / gaps)
        weights /= weights.sum()
    else:
        weights = np.ones(len(active)) / len(active)

    w_active = np.zeros(2 * n - 1)
    w_active[active] = weights

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_multi_scale(a, scales=None, tol=1e-6):
    """
    Multi-scale subgradient: combine gradients from different active set sizes
    to get a more robust descent direction.
    """
    if scales is None:
        scales = [1, 3, 7]
    
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    # Combine gradients from different scales with geometric weighting
    combined = np.zeros(n)
    total_w = 0.0
    for idx, k in enumerate(scales):
        sc_k, g_k = _subgrad_weighted(a, k_active=k, tol=tol, weighting='linear')
        w = 1.0 / (1.0 + idx * 0.5)  # Decreasing weight for larger sets
        combined += w * g_k
        total_w += w

    if total_w > 0:
        combined /= total_w

    combined = np.nan_to_num(combined, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), combined


def _nesterov_optimize_v2(a0, n_iters=2000, deadline=None, k_active=5, tol=1e-7,
                           weighting='linear', use_multi_scale=False, momentum_decay=1.0):
    """
    Enhanced Nesterov accelerated subgradient method.
    Uses importance-weighted active sets and optional multi-scale gradients.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    # Initial step size
    if use_multi_scale:
        score0, g0 = _subgrad_multi_scale(a, scales=[1, k_active, max(2*k_active, 10)], tol=tol)
    else:
        score0, g0 = _subgrad_weighted(a, k_active=k_active, tol=tol, weighting=weighting)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-14:
        return best_score, best_a

    # Larger initial step, slower decay
    eta = 0.1 / g_norm
    best_score_hist = [best_score]
    no_improve_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new * momentum_decay

        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        if use_multi_scale:
            score_y, grad_y = _subgrad_multi_scale(y_norm, scales=[1, k_active, max(2*k_active, 10)], tol=tol)
        else:
            score_y, grad_y = _subgrad_weighted(y_norm, k_active=k_active, tol=tol, weighting=weighting)

        if not np.isfinite(score_y):
            break

        # Polyak-inspired adaptive step
        step = eta / (1.0 + 0.05 * it)
        # Additional scaling based on gradient norm
        gn = np.linalg.norm(grad_y)
        if gn > 0:
            step = min(step, 0.5 / gn)

        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t = t_new

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive: reduce step if stuck
        if no_improve_count > 80:
            eta *= 0.5
            no_improve_count = 0
            # Occasionally try a small random kick
            if no_improve_count % 3 == 0:
                kick = 0.01 * np.random.standard_normal(n)
                a = _normalize(np.clip(a + kick, 0, None))

        # Periodic exact re-evaluation
        if it % 200 == 0 and it > 0:
            sc_check = _exact_score(a)
            if sc_check < best_score:
                best_score = sc_check
                best_a = a.copy()

            # Reset momentum if we've been stuck
            if sc_check > best_score + 1e-10:
                v = np.zeros(n)
                t = 1.0

    return best_score, best_a


def _lbfgs_lse_v2(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on the LSE-smoothed objective with better settings."""
    a = _normalize(a0)
    n = a.size

    if maxfun is None:
        maxfun = 300

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(maxfun, 100),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _peak_targeted_perturb(a, rng, strength=0.1):
    """
    Perturbation that specifically targets the peaks in the convolution.
    Reduces mass at positions contributing to the max peaks,
    increases mass at positions contributing to sub-max peaks.
    """
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    c = _fast_self_conv(a)
    cmax = float(c.max())
    if cmax < 1e-15:
        return a

    # Find the top peak
    peak_idx = int(np.argmax(c))
    
    # Compute sensitivity: how much each a[i] contributes to the peak
    # d(conv[peak])/da[i] = a[peak-i] + a[i] (if 2i != peak) or 2*a[i] (if 2i == peak)
    sens = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            if i == j:
                sens[i] = 2.0 * a[i]
            else:
                sens[i] = a[j] + a[i]
    
    # Find sub-max peaks (within 10% of max)
    thresh = cmax * 0.85
    sub_peaks = np.where((c >= thresh) & (c < cmax * 0.99))[0]
    
    # Weight: reduce high-sensitivity positions, increase low-sensitivity ones
    sens_norm = sens / (sens.max() + 1e-15)
    # Perturbation: reduce where sensitivity is high, increase where low
    pert = -strength * sens_norm
    # Add some randomness to avoid cycling
    pert += strength * 0.3 * rng.standard_normal(n)
    
    # Also nudge toward sub-peaks (increase mass there)
    if len(sub_peaks) > 0:
        for sp in sub_peaks[:5]:
            for i in range(n):
                j = sp - i
                if 0 <= j < n:
                    pert[i] += strength * 0.05 / (1.0 + abs(i - sp // 2))
    
    q = a * np.exp(pert)
    return _normalize(np.clip(q, 0.0, None))


def _structured_perturb(a, rng, mode=None):
    """Diverse structured perturbations."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if mode is None:
        mode = int(rng.integers(0, 8))

    if mode == 0:
        # Log-normal multiplicative
        q = a * np.exp(0.15 * rng.standard_normal(n))
    elif mode == 1:
        # Blend with slight reshaping
        t = float(rng.uniform(0.05, 0.2))
        x = np.linspace(0, 1, n)
        shape = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * x)
        q = (1-t) * a + t * _normalize(shape)
    elif mode == 2:
        # Fourier-domain perturbation
        Nf = _next_pow2(n)
        spec = np.fft.rfft(a, Nf)
        freqs = np.arange(len(spec))
        # Perturb mid-frequencies
        mask = (freqs > n//20) & (freqs < n//3)
        spec[mask] *= np.exp(0.1 * rng.standard_normal(mask.sum()))
        q = np.fft.irfft(spec, Nf)[:n]
    elif mode == 3:
        # Local block shuffle
        q = a.copy()
        n_blocks = int(rng.integers(3, 8))
        block_size = n // n_blocks
        for _ in range(int(rng.integers(1, 3))):
            i, j = int(rng.integers(0, n_blocks)), int(rng.integers(0, n_blocks))
            if i != j:
                si, sj = i*block_size, j*block_size
                q[si:si+block_size], q[sj:sj+block_size] = q[sj:sj+block_size].copy(), q[si:si+block_size].copy()
                # Blend to avoid discontinuity
                blend = np.linspace(1, 0, block_size)
                q[si:si+block_size] = (1-blend)*q[si:si+block_size] + blend*q[si+block_size-1:si+2*block_size-1][:block_size] if si+2*block_size-1 <= n else q[si:si+block_size]
                q[sj:sj+block_size] = q[sj:sj+block_size]
    elif mode == 4:
        # Peak-targeted
        q = _peak_targeted_perturb(a, rng, strength=0.1)
    elif mode == 5:
        # Symmetry breaking
        q = a.copy()
        t = float(rng.uniform(0.05, 0.15))
        asym = np.linspace(-t, t, n)
        q = a * (1.0 + asym)
    elif mode == 6:
        # Sparse modification
        q = a.copy()
        n_mod = int(rng.integers(n//50, n//10))
        idx = rng.integers(0, n, size=n_mod)
        factors = np.exp(0.3 * rng.standard_normal(n_mod))
        q[idx] *= factors
    else:
        # Smooth noise
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        kern = np.exp(-((np.arange(K) - c)**2) / (2.0*(c/2)**2))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode='same')
        noise -= noise.mean()
        q = a * np.exp(0.1 * noise)

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize_v2(a, deadline, use_multi_scale=False):
    """Quick optimization with adaptive method selection."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)
    n = a.size

    # Get peak diagnostics
    c = _fast_self_conv(a)
    beta = _adaptive_beta(c, base_beta=3000)

    # Phase 1: LSE-LBFGS
    rem = deadline - time.monotonic()
    if rem > 20.0:
        maxfun = int(max(30, min(400, rem * 8)))
        sc, a2 = _lbfgs_lse_v2(a, beta, deadline, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    # Phase 2: Subgradient
    rem = deadline - time.monotonic()
    if rem > 15.0:
        n_it = int(max(100, rem * 20))
        sc, a2 = _nesterov_optimize_v2(a, n_iters=n_it, deadline=deadline,
                                        k_active=5, tol=1e-7, weighting='linear')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    # Phase 3: Multi-scale subgradient
    rem = deadline - time.monotonic()
    if rem > 10.0:
        n_it = int(max(50, rem * 15))
        sc, a2 = _nesterov_optimize_v2(a, n_iters=n_it, deadline=deadline,
                                        k_active=7, tol=5e-7, use_multi_scale=True)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()

    return best_score, best_a


def _diagnostic_print(a, label=""):
    """Print diagnostic information about the current sequence."""
    cmax, spread, peaks, n_peaks = _peak_diagnostics(a, percentile=95)
    n = a.size
    score = _exact_score(a)
    print("[{}] n={} score={:.10f} cmax={:.6e} spread={:.2f} peaks>95th={}".format(
        label, n, score, cmax, spread, n_peaks))


def _intensive_polish(a, deadline):
    """
    Intensive polishing: alternate between very high beta LSE and tight subgradient.
    Progressive tightening of tolerance.
    """
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)
    t_start = time.monotonic()
    total = max(5.0, float(deadline - t_start))

    # Progressive params
    configs = [
        {'beta': 200000, 'k_act': 3, 'tol': 1e-8, 'frac': 0.2},
        {'beta': 500000, 'k_act': 5, 'tol': 1e-9, 'frac': 0.2},
        {'beta': 1000000, 'k_act': 3, 'tol': 1e-9, 'frac': 0.2},
        {'beta': 2000000, 'k_act': 2, 'tol': 1e-10, 'frac': 0.2},
        {'beta': None, 'k_act': 3, 'tol': 1e-10, 'frac': 0.2},  # Nesterov only
    ]

    cum = 0.0
    for cfg in configs:
        cum += cfg['frac']
        sub_dead = t_start + cum * total
        sub_dead = min(sub_dead, float(deadline - 2.0))

        if time.monotonic() > sub_dead - 3.0:
            continue

        # LSE phase
        if cfg['beta'] is not None:
            rem = sub_dead - time.monotonic()
            if rem > 5.0:
                maxfun = int(max(20, min(300, rem * 5)))
                sc, a2 = _lbfgs_lse_v2(a, cfg['beta'], sub_dead - 1.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("  Polish LSE beta={}: {:.10f}".format(cfg['beta'], best_score))
                a = _normalize(a2)

        # Nesterov phase
        rem = sub_dead - time.monotonic()
        if rem > 5.0:
            n_it = int(max(100, rem * 25))
            sc, a2 = _nesterov_optimize_v2(a, n_iters=n_it, deadline=sub_dead,
                                            k_active=cfg['k_act'], tol=cfg['tol'],
                                            weighting='inverse_gap')
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("  Polish Nesterov k={} tol={:.0e}: {:.10f}".format(
                    cfg['k_act'], cfg['tol'], best_score))
            a = _normalize(a2)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns list of non-negative floats."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 128:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    if parent is not None:
        print("Parent score: {:.10f}".format(_exact_score(_normalize(parent))))

    # Determine primary length
    if parent is not None:
        primary_n = min(max(parent.size, 4096), 16384)
    else:
        primary_n = 10240

    # Phase 1: Initialize and quick-optimize multiple candidates (200s)
    print("\n=== Phase 1: Initialization ===")
    phase1_end = t_start + 200.0

    candidates = []

    # Primary candidate from parent
    if parent is not None:
        a0 = _normalize(_resample(parent, primary_n))
        candidates.append(a0)
    else:
        a0 = np.ones(primary_n) / primary_n
        candidates.append(a0)

    # Additional candidates: slight variations
    for i in range(3):
        if parent is not None:
            a_var = _structured_perturb(_resample(parent, primary_n), rng, mode=i)
        else:
            x = np.linspace(0, 1, primary_n)
            a_var = _normalize(1.0 + 0.3 * i * np.cos(2*np.pi*(i+1)*x))
        candidates.append(a_var)

    # Quick-optimize each
    best_score = np.inf
    best_seq = candidates[0].copy()

    for i, cand in enumerate(candidates):
        chunk_dead = t_start + (i + 1) * (phase1_end - t_start) / len(candidates)
        sc, seq = _quick_optimize_v2(cand, chunk_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            _diagnostic_print(best_seq, "init{}".format(i))
        print("Init {}: score={:.10f}".format(i, sc))

    print("After Phase 1: best={:.10f}".format(best_score))

    # Phase 2: Main intensive optimization (1400s)
    print("\n=== Phase 2: Main Optimization ===")
    phase2_end = t_start + 1600.0
    a = _normalize(best_seq)
    n = a.size

    # Adaptive LSE sweep
    print("Starting LSE sweep...")
    lse_betas = [1000, 5000, 20000, 50000, 100000, 200000]
    lse_end = t_start + 800.0
    chunk_size = (lse_end - t_start) / len(lse_betas)

    for i, beta in enumerate(lse_betas):
        chunk_dead = t_start + (i + 1) * chunk_size
        chunk_dead = min(chunk_dead, lse_end)
        if time.monotonic() > chunk_dead - 5.0:
            break

        sc, a2 = _lbfgs_lse_v2(a, beta, chunk_dead, maxfun=300)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("LSE beta={}: {:.10f}".format(beta, best_score))
        a = _normalize(a2)

        # Quick Nesterov after each LSE
        rem = chunk_dead - time.monotonic()
        if rem > 10.0:
            sc2, a2 = _nesterov_optimize_v2(a, n_iters=int(max(100, rem*15)),
                                             deadline=chunk_dead, k_active=5,
                                             tol=1e-7, weighting='linear')
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
                print("Nesterov after LSE beta={}: {:.10f}".format(beta, best_score))
            a = _normalize(a2)

    _diagnostic_print(best_seq, "after LSE sweep")

    # Subgradient intensive phase
    print("Starting subgradient phase...")
    subgrad_end = phase2_end - 50.0
    a = _normalize(best_seq)

    # Multiple passes with different settings
    sg_configs = [
        {'k_act': 3, 'tol': 1e-7, 'wt': 'linear', 'multi': False},
        {'k_act': 5, 'tol': 1e-7, 'wt': 'inverse_gap', 'multi': False},
        {'k_act': 7, 'tol': 5e-7, 'wt': 'quadratic', 'multi': True},
        {'k_act': 10, 'tol': 1e-8, 'wt': 'linear', 'multi': True},
        {'k_act': 3, 'tol': 1e-9, 'wt': 'inverse_gap', 'multi': False},
    ]

    for i, cfg in enumerate(sg_configs):
        chunk_dead = t_start + (i + 1) * (subgrad_end - max(t_start+800, time.monotonic())) / len(sg_configs)
        chunk_dead = min(chunk_dead, subgrad_end)
        if time.monotonic() > chunk_dead - 5.0:
            continue

        n_it = int(max(200, (chunk_dead - time.monotonic()) * 20))
        sc, a2 = _nesterov_optimize_v2(a, n_iters=n_it, deadline=chunk_dead,
                                        k_active=cfg['k_act'], tol=cfg['tol'],
                                        weighting=cfg['wt'], use_multi_scale=cfg['multi'])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("SG k={} tol={:.0e} wt={} multi={}: {:.10f}".format(
                cfg['k_act'], cfg['tol'], cfg['wt'], cfg['multi'], best_score))
        a = _normalize(a2)

    _diagnostic_print(best_seq, "after subgradient")

    # Phase 3: Perturbation restarts (400s)
    print("\n=== Phase 3: Perturbation Restarts ===")
    phase3_end = t_start + 2000.0
    n_restarts = 6
    restart_budget = max(30.0, phase3_end - time.monotonic()) / n_restarts

    for r in range(n_restarts):
        chunk_dead = time.monotonic() + restart_budget
        chunk_dead = min(chunk_dead, phase3_end)
        if time.monotonic() > chunk_dead - 10.0:
            break

        # Mix of perturbation types, weighted toward peak-targeted
        if r % 3 == 0:
            p = _peak_targeted_perturb(best_seq, rng, strength=0.08)
        elif r % 3 == 1:
            p = _structured_perturb(best_seq, rng, mode=2)  # Fourier
        else:
            p = _structured_perturb(best_seq, rng, mode=5)  # Symmetry breaking

        sc, seq = _quick_optimize_v2(p, chunk_dead, use_multi_scale=(r % 2 == 0))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Restart {}: {:.10f}".format(r, best_score))
        else:
            print("Restart {}: {:.10f} (no improvement)".format(r, sc))

    _diagnostic_print(best_seq, "after restarts")

    # Phase 4: Final intensive polish (remaining time)
    print("\n=== Phase 4: Final Polish ===")
    final_dead = deadline - 5.0
    if time.monotonic() < final_dead - 30.0:
        sc, a2 = _intensive_polish(best_seq, final_dead)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()

    # One last Nesterov with k=1
    rem = final_dead - time.monotonic()
    if rem > 15.0:
        n_it = int(max(100, rem * 30))
        sc, a2 = _nesterov_optimize_v2(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=1, tol=1e-10, weighting='uniform')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Final k=1: {:.10f}".format(best_score))
        best_seq = _normalize(a2)

    # Final very high beta LSE
    rem = final_dead - time.monotonic()
    if rem > 10.0:
        sc, a2 = _lbfgs_lse_v2(best_seq, 5000000, final_dead, maxfun=100)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Final LSE: {:.10f}".format(best_score))
        best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    _diagnostic_print(best_seq, "FINAL")
    print("FINAL: internal={:.10f} verified={:.10f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
