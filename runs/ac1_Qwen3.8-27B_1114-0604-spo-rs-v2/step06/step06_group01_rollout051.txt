"""
Progressive LSE-LBFGS optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Multi-phase: quick init, progressive beta L-BFGS, multi-start perturbation,
peak shooting via exact subgradient, and final high-beta polish.
Builds on parent sequence with careful time management over 2400s budget.

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_corr(a, p):
    """Cross-correlation: C[m] = sum_l a[l]*p[l+m] for m=0..n-1."""
    n = len(a)
    n_p = len(p)
    M = n + n_p - 1
    M_fft = 1 << (M - 1).bit_length()
    RA = np.fft.rfft(a, M_fft)
    RP = np.fft.rfft(p, M_fft)
    return np.fft.irfft(np.conj(RA) * RP, M_fft)[:n]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_objective_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        log_sum = 0.0
        p = np.ones(len(c)) / len(c)
    else:
        log_sum = z.max() - z_shifted.max() + np.log(exp_sum)
        p = exp_z / exp_sum
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    corr = cross_corr(a, p)
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, gtol):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_objective_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                 'factr': 1e8}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = float(a_out.sum())
    if S_out < 1e-12:
        a_out = np.ones(n) / n
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            eta *= eta_decay
            if eta < 1e-6:
                break

    return best_a


def mirror_descent_shooting(a, n_steps=200, top_k=3):
    """Mirror descent (multiplicative) on the exact subgradient at top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.1

    for it in range(n_steps):
        c = self_conv(a)
        S = float(a.sum())
        idx = np.argsort(c)[::-1][:top_k]
        weights = c[idx]
        weights = weights / weights.sum()

        # Subgradient of max(conv)
        sg = np.zeros(n)
        for w, k in zip(weights, idx):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            sg[lo:hi+1] += w * 2.0 * a[k - np.arange(lo, hi + 1)]

        # Mirror descent: a[m] *= exp(-eta * sg[m] / (2*mean(sg)))
        sg_mean = float(sg.mean()) + 1e-30
        log_update = -eta * sg / (2.0 * sg_mean)
        log_update -= log_update.max()  # prevent overflow
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            eta *= 0.9

    return best_a


def error_reduction(a, n_iters=50, alpha=0.7):
    """Frequency-domain Newton-like error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        conv = np.fft.irfft(A * A, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def multi_start_refine(a, rng, n_starts=8, beta=500, maxiter=3000):
    """Try multiple random perturbations of a and re-optimize each."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for i in range(n_starts):
        # Different perturbation strategies
        scale = rng.uniform(0.01, 0.15)
        if i % 3 == 0:
            # Smooth perturbation
            noise = rng.standard_normal(n)
            kernel = np.ones(max(3, n // 100)) / max(3, n // 100)
            noise = np.convolve(noise, kernel, mode='same')
            a_try = a + scale * noise
        elif i % 3 == 1:
            # Block perturbation
            a_try = a.copy()
            n_blocks = rng.integers(3, 15)
            for _ in range(n_blocks):
                start = rng.integers(0, n)
                length = rng.integers(n // 50, n // 10)
                shift = rng.uniform(0, scale)
                a_try[start:start+length] *= (1.0 + shift)
        else:
            # Sinusoidal perturbation
            freq = rng.uniform(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            a_try = a * (1.0 + scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase))

        a_try = np.maximum(a_try, 0.0)
        S = float(a_try.sum())
        if S < 1e-12:
            continue
        a_try = a_try / S

        sc, a_opt, _ = run_lbfgs(a_try, beta, maxiter=maxiter, maxfun=10000, gtol=1e-12)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


def build_initialization(n, parent, rng):
    """Generate a good initialization of size n."""
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) > 10:
        if len(parent) == n:
            a = parent.astype(np.float64).copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, parent.astype(np.float64))
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        return a

    # Diverse initializations if no parent
    candidates = [
        np.ones(n) * 0.5,
        0.5 + 0.4 * np.cos(2.0 * np.pi * x),
        0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2,
        np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
        0.2 + 0.8 * (2.0 * x - 1.0) ** 2,
        0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x),
    ]
    a = candidates[rng.integers(0, len(candidates))]
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S > 1e-12:
        a = a / S
    return a


def propose_candidate(budget_s=2400):
    """
    Multi-phase optimizer minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phases: init, progressive LSE-LBFGS, multi-start, peak shooting, final polish.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is None:
        try:
            p = np.array(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
        except Exception:
            parent = None

    best_score = np.inf
    best_a = np.ones(64) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")

    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # ==================================================================
    # Phase 1: Quick initialization (3% time)
    # ==================================================================
    phase1_deadline = t_start + 0.03 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Initialization ---")

    if parent is not None:
        a_init = parent.copy()
        S = float(a_init.sum())
        if S > 0:
            a_init = a_init / S
        a_er = error_reduction(a_init, n_iters=30, alpha=0.7)
        sc = compute_score(a_er)
        report(sc, a_er, "init ER")

        # Also try from a fresh random init at same length
        n_parent = len(parent)
        a_rand = build_initialization(n_parent, None, rng)
        a_er2 = error_reduction(a_rand, n_iters=20, alpha=0.6)
        sc2 = compute_score(a_er2)
        report(sc2, a_er2, "init rand ER")
    else:
        n_init = 6962
        a_init = build_initialization(n_init, None, rng)
        a_er = error_reduction(a_init, n_iters=40, alpha=0.7)
        sc = compute_score(a_er)
        report(sc, a_er, "init")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Progressive LSE + L-BFGS (55% time)
    # ==================================================================
    phase2_deadline = t_start + 0.58 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: Progressive LSE-LBFGS ---")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S
    n_main = len(a_work)

    # Progressive beta schedule - more aggressive than before
    beta_schedule = [
        (30, 5000, 1e-12),
        (60, 8000, 1e-13),
        (120, 10000, 1e-13),
        (250, 12000, 1e-14),
        (500, 15000, 1e-14),
        (1000, 15000, 1e-15),
        (2000, 20000, 1e-15),
        (4000, 20000, 1e-15),
        (8000, 25000, 1e-16),
        (15000, 25000, 1e-16),
        (25000, 30000, 1e-16),
        (50000, 30000, 1e-16),
    ]

    for bi, (beta_val, maxiter, gtol) in enumerate(beta_schedule):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break

        # Scale maxiter based on remaining time
        est_time_per_iter = remaining / 10.0
        maxit = min(maxiter, max(500, int(est_time_per_iter)))

        print(f"  LBFGS beta={beta_val}, maxiter={maxit}, remaining={remaining:.0f}s", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(a_work, beta_val, maxiter=maxit, maxfun=maxit * 2, gtol=gtol)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, nfev={res.nfev}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LSE b={beta_val}")
        a_work = a_opt.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Multi-start perturbation (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.73 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Multi-Start Refinement ---")

    if time.monotonic() < phase3_deadline - 60:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

        n_starts = 10
        beta_ms = 2000
        a_ms, sc_ms = multi_start_refine(a_norm, rng, n_starts=n_starts,
                                         beta=beta_ms, maxiter=5000)
        report(sc_ms, a_ms, "multi-start")

        # Also try a few at medium beta for diversity
        if time.monotonic() < phase3_deadline - 60:
            a_ms2, sc_ms2 = multi_start_refine(a_norm, rng, n_starts=5,
                                               beta=500, maxiter=3000)
            report(sc_ms2, a_ms2, "multi-start b=500")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak shooting (12% time)
    # ==================================================================
    phase4_deadline = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Peak Shooting ---")

    if time.monotonic() < phase4_deadline - 30:
        a_peak = peak_shooting(best_a, n_steps=300, top_k=3)
        sc = compute_score(a_peak)
        report(sc, a_peak, "peak shooting")

        a_md = mirror_descent_shooting(best_a, n_steps=300, top_k=5)
        sc_md = compute_score(a_md)
        report(sc_md, a_md, "mirror descent")

        # If either helped, re-run L-BFGS on the improved solution
        if best_score < compute_score(best_a):
            pass  # already updated

        # Quick L-BFGS polish after peak shooting
        if time.monotonic() < phase4_deadline - 30:
            a_pol = best_a.copy()
            S = float(a_pol.sum())
            if S > 0:
                a_pol = a_pol / S
            sc, a_opt, _ = run_lbfgs(a_pol, 5000, maxiter=10000, maxfun=20000,
                                     gtol=1e-15)
            report(sc, a_opt, "polish after peak")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final high-beta polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    a_final = best_a.copy()
    S = float(a_final.sum())
    if S > 0:
        a_final = a_final / S

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 120:
        print("  Final LBFGS beta=50000...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(a_final, 50000, maxiter=20000, maxfun=40000,
                                 gtol=1e-16)
        print(f"score={sc:.8f}, time={time.monotonic()-t0:.1f}s")
        report(sc, a_opt, "final LBFGS 50k")
        a_final = a_opt.copy()

    # One more peak shooting pass
    if time.monotonic() < deadline - 60:
        a_ps = peak_shooting(a_final, n_steps=200, top_k=2)
        sc = compute_score(a_ps)
        report(sc, a_ps, "final peak")

    # Final mirror descent
    if time.monotonic() < deadline - 30:
        a_md_final = mirror_descent_shooting(best_a, n_steps=100, top_k=2)
        sc = compute_score(a_md_final)
        report(sc, a_md_final, "final MD")

    # ==================================================================
    # Try a different length if time permits
    # ==================================================================
    if time.monotonic() < deadline - 200 and parent is not None:
        n_alt = len(parent) + 128  # slightly longer
        if n_alt != len(best_a):
            print(f"\n  Trying alt length n={n_alt}")
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            S = float(a_alt.sum())
            if S > 0:
                a_alt = a_alt / S
            sc, a_opt, _ = run_lbfgs(a_alt, 2000, maxiter=5000, maxfun=15000,
                                     gtol=1e-14)
            report(sc, a_opt, f"alt n={n_alt}")

    # Slightly shorter
    if time.monotonic() < deadline - 100 and parent is not None:
        n_alt2 = len(parent) - 64
        if n_alt2 > 100 and n_alt2 != len(best_a):
            print(f"  Trying alt length n={n_alt2}")
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt2)
            a_alt2 = np.interp(x_new, x_old, best_a)
            a_alt2 = np.maximum(a_alt2, 0.0)
            S = float(a_alt2.sum())
            if S > 0:
                a_alt2 = a_alt2 / S
            sc, a_opt, _ = run_lbfgs(a_alt2, 2000, maxiter=3000, maxfun=10000,
                                     gtol=1e-14)
            report(sc, a_opt, f"alt n={n_alt2}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
