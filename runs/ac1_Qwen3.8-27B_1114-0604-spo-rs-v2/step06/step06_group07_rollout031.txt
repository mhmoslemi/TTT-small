"""
Optimized step function construction using adaptive subgradient methods,
Fienup iterations with corrected spectral targets, Nesterov acceleration,
and multi-scale L-BFGS refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_geometric(a, n_iters=400, alpha_start=0.1, alpha_end=0.9):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def nesterov_subgradient(a0, n_iters=5000, n_peaks=10, step_init=0.01, momentum_init=0.9):
    """Nesterov-accelerated subgradient descent with projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    v = np.zeros(n)  # momentum
    a_yst = a.copy()  # for Nesterov lookahead
    momentum = momentum_init
    step = step_init
    
    for it in range(n_iters):
        # Nesterov lookahead
        a_nes = a + momentum * (a - a_yst)
        a_nes = np.maximum(a_nes, 0.0)
        S_nes = float(a_nes.sum())
        if S_nes < 1e-12:
            break
        a_nes = a_nes / S_nes
        
        # Compute subgradient at a_nes
        A = np.fft.rfft(a_nes, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_val = float(a_nes.sum())
        M_val = float(c.max())
        
        # Find top peaks

            weights_fwd = c[top_idx]

                weights = weights / w_sum

def lse_objective_grad(x, beta, n_fft_cache=None):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_directed_descent(a, n_iters=3000, n_peaks=5, step_init=0.02, decay=0.9999):

    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:

        else:

            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        S = a.sum()
        if S < 1e-12:

        a = a / S
        step *= decay

def fast_conv(a):

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)
    a_yst = a.copy()
    momentum = 0.9
    step = step_init
    no_improve = 0
    patience = 500
    for it in range(n_iters):
        a_nes = a + momentum * (a - a_yst)
        a_nes = np.maximum(a_nes, 0.0)
        S_nes = float(a_nes.sum())
        if S_nes < 1e-12:
            break
        a_nes = a_nes / S_nes
        A = np.fft.rfft(a_nes, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_val = float(a_nes.sum())
        M_val = float(c.max())
        target = S_val * S_val / (2.0 * n - 1.0)
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c)) / len(c)
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            excess = np.maximum(c[top_idx] - target, 0.0)
            w_sum = float(excess.sum())
            if w_sum < 1e-30:
                weights = np.zeros(len(c))
                weights[top_idx] = 1.0 / n_peaks
            else:
                weights = np.zeros(len(c))
                weights[top_idx] = excess / w_sum
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = weights[k]
            if w < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += w * a_nes[js:je + 1][::-1]
        S2 = S_val * S_val
        S3 = S2 * S_val
        grad = (2.0 * n) * (2.0 * sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        sc_curr = 2.0 * n * M_val / S2
        step = max((sc_curr - sc_lb) / (gn * gn + 1e-30), 1e-12)
        step = min(step, 0.025)
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a_yst = a.copy()
        a = a_new
        momentum = min(0.995, momentum + 5e-7)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break
    return best_a


def multiplicative_update(a0, n_iters=2000, alpha=0.5):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    conv_fft = 1 << (4 * n - 3).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    patience = 400
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S_val = float(a.sum())
        target = S_val * S_val / (2.0 * n - 1.0)
        excess = np.maximum(c - target, 0.0)
        total_excess = float(excess.sum())
        if total_excess < 1e-15:
            break
        lam = excess / total_excess
        AR = np.fft.rfft(a[::-1], conv_fft)
        LM = np.fft.rfft(lam, conv_fft)
        conv_res = np.fft.irfft(LM * AR, conv_fft)
        contribution = conv_res[n - 1:2 * n - 1]
        eps = 1e-12 * float(np.max(contribution))
        ratio = 1.0 / np.maximum(contribution, eps)
        rmean = float(ratio.mean())
        if rmean > 0:
            ratio = ratio / rmean
        a *= np.power(ratio, alpha)
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break
    return best_a


def lse_objective_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_directed_descent(a, n_iters=3000, n_peaks=5, step_init=0.015, decay=0.9999):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    no_improve = 0
    patience = 500
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.025)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    return best_a


def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        n_work = len(parent)
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
    else:
        n_work = 8192
        best_a = np.ones(n_work) * 0.5
        best_score = compute_score(best_a)

    print(f"Working length: {n_work}, Parent: {parent is not None}")

    # Phase 1: Nesterov subgradient (25% time)
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Nesterov Subgradient ===")
    for n_peaks in [5, 10, 30]:
        if time.monotonic() > phase1_end:
            break
        print(f"  n_peaks={n_peaks}")
        a_r = nesterov_subgradient(best_a, n_iters=4000, n_peaks=n_peaks, step_init=0.012)
        sc = compute_score(a_r)
        update_best(sc, a_r, f"NEST m={n_peaks}")
    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: Multiplicative updates (15% time)
    phase2_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Multiplicative Updates ===")
    for alpha in [0.3, 0.7, 1.2]:
        if time.monotonic() > phase2_end:
            break
        print(f"  alpha={alpha}")
        a_mu = multiplicative_update(best_a, n_iters=2000, alpha=alpha)
        sc = compute_score(a_mu)
        update_best(sc, a_mu, f"MULT a={alpha}")
    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Fienup (10% time)
    phase3_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Fienup ===")
    a_f = fienup_geometric(best_a, n_iters=400, alpha_start=0.05, alpha_end=0.8)
    sc = compute_score(a_f)
    update_best(sc, a_f, "FIENUP")
    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Adaptive LSE L-BFGS annealing (30% time)
    phase4_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 4: LSE L-BFGS Annealing ===")
    betas = [50, 200, 800, 3000, 10000, 30000]
    for beta in betas:
        if time.monotonic() > phase4_end - 60:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 90:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(8000, int(remaining * 15))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 25)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")
    print(f"After Phase 4: best = {best_score:.8f}")

    # Phase 5: Peak-directed refinement (10% time)
    phase5_end = deadline - 60.0
    print(f"\n=== Phase 5: Peak-Directed Refinement ===")
    for n_peaks in [5, 15, 40]:
        if time.monotonic() > phase5_end:
            break
        a_pd = peak_directed_descent(best_a, n_iters=2500, n_peaks=n_peaks,
                                     step_init=0.008, decay=0.99995)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")
    print(f"After Phase 5: best = {best_score:.8f}")

    # Phase 6: Final combined polish
    if time.monotonic() < deadline - 30:
        print("\n=== Phase 6: Final Polish ===")
        a_fin = nesterov_subgradient(best_a, n_iters=2000, n_peaks=10, step_init=0.005)
        sc = compute_score(a_fin)
        update_best(sc, a_fin, "FINAL-NEST")

    if time.monotonic() < deadline - 30:
        a_fin2 = multiplicative_update(best_a, n_iters=1000, alpha=0.5)
        sc = compute_score(a_fin2)
        update_best(sc, a_fin2, "FINAL-MULT")

    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            maxiter = min(3000, int(remaining * 10))
            sc, a_opt = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 20)
            update_best(sc, a_opt, "FINAL-LBFGS")

    # Perturbation restarts
    if time.monotonic() < deadline - 15:
        for trial in range(2):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.015 * (1 + trial)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S = a_p.sum()
            if S < 1e-12:
                continue
            a_p = a_p / S
            a_r = peak_directed_descent(a_p, n_iters=500, n_peaks=10,
                                        step_init=0.01, decay=0.9999)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
