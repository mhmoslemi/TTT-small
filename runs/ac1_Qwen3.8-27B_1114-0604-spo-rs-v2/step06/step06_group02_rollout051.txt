"""
Corrected-gradient LSE-LBFGS optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key fix: the gradient's convolution term was halved in prior code (missing factor of 2
from dc_k/da_m = 2*a_{k-m}). This multi-beta schedule with corrected gradients,
combined with subgradient polishing and multi-start, targets scores below 1.5027.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed score and CORRECTED gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # h_m = sum_k p_k * a_{k-m}  (cross-correlation)
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    # CORRECTED: df/da_m = 2n*(2h_m/S^2 - 2L/S^3) = 4n/S^2*(h_m - L/S)
    grad = 4.0 * n / S2 * (h - L / S)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 80}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    sc = compute_score(a_out)
    return sc, a_out


def subgradient_polish(a, n_iters=5000, tol=1e-10):
    """Projected subgradient with Polyak step on the exact max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        c_max = float(np.max(c))
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient: indices where c is near max
        active = np.where(c >= 0.999 * c_max)[0]
        # Compute subgradient direction
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            grad[j_start:j_end + 1] += 2.0 * a[k - (j_start:j_end + 1)][::-1]
        # Normalize: projected subgradient with decreasing step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        step = 2.0 * (sc - 1.01) / (g_norm * g_norm + 1e-15)
        step = min(step, 0.1 / np.sqrt(it + 1))
        a_new = a - step * grad / S
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        if not np.all(np.isfinite(a_new)):
            break
        a = a_new

        if it % 500 == 0:
            sc_now = compute_score(a)
            if sc_now < best_sc:
                best_sc = sc_now
                best_a = a.copy()

    return best_a


def fienup_quick(a, n_iters=300):
    """Quick Fienup for potential improvement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.6 * (1.0 - progress) ** 2 + 0.02
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_descent_polish(a, n_sweeps=30):
    """Line-search coordinate descent on individual elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        improved = False
        # Randomize order for diversity
        order = np.arange(n)
        for j in order:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            # Try several multipliers
            best_val = a_old
            best_local = best_sc
            for f in [0.5, 0.75, 0.85, 0.92, 0.96, 1.0, 1.04, 1.1, 1.25, 1.5]:
                a[j] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                sc_tmp = compute_score(a_norm)
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_val = a_old * f
            if best_val != a_old:
                a[j] = best_val
                improved = True

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if not improved and sweep > 5:
            break

    return best_a


def try_lengths(a_best, scores, deadline):
    """Try different sequence lengths via interpolation."""
    results = []
    current_n = len(a_best)
    candidates = []
    for mult in [0.5, 0.75, 2.0]:
        target_n = int(round(current_n * mult))
        if target_n >= 64 and target_n <= 4096:
            candidates.append(target_n)
    # Also try some absolute sizes
    for sz in [1024, 2048, 4096]:
        if sz not in candidates and abs(sz - current_n) > current_n * 0.3:
            candidates.append(sz)

    for target_n in candidates[:3]:
        if time.monotonic() > deadline:
            break
        if len(a_best) == target_n:
            a_try = a_best.copy()
        else:
            x_old = np.linspace(0, 1, len(a_best))
            x_new = np.linspace(0, 1, target_n)
            a_try = np.interp(x_new, x_old, a_best)
            a_try = np.maximum(a_try, 0.0)
            S = a_try.sum()
            if S < 1e-12:
                continue
            a_try = a_try / S
        sc = compute_score(a_try)
        # Quick L-BFGS refinement
        if time.monotonic() < deadline - 60:
            sc, a_opt = run_lbfgs(a_try, 2000, maxiter=5000, maxfun=80000)
        results.append((sc, a_opt))
        print(f"    Length {target_n}: {sc:.8f}")

    return results
