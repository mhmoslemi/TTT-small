import time
import math
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)

def cross_corr(a, w):
    """Compute h[i] = sum_j w[j] * a[i+j] (with zero-padding for out-of bounds).
    This gives the gradient contribution when w weights the convolution lags."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    # We want h[i] = sum_j w[j] * a[i+j] for i = 0..n-1
    # This is the cross-correlation of a with w (reversed)
    Ar = a[::-1]
    A = np.fft.rfft(Ar, N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n-1:2*n-1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:

def top_k_weights(conv, k, frac=0.0):
    """Return weights concentrated on top-k convolution positions."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    if frac > 0:
        threshold = dmax * (1.0 - frac)
        active = conv >= threshold
        if active.sum() == 0:
            active[np.argmax(conv)] = True
    else:
        idx = np.argpartition(conv, -k)[-k:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    w = active.astype(np.float64)
    s = float(w.sum())
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def lse_smooth_score(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    # Scale: f = 2n * max(c) / S^2
    # LSE approximation: f ≈ 2n * (max(c) + log(sum(exp(beta*(c-max(c)))))/beta) / S^2
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e  # softmax weights over convolution positions
    LSE = cmax + np.log(sum_e) / beta
    
    # Gradient of LSE w.r.t. a:
    # dLSE/da = sum_k w[k] * d(conv(a,a)_k)/da
    # d(conv(a,a)_k)/da[i] = 2 * sum_j a[j] * delta(i+j-k or i-j+k)
    # = 2 * (a * delta_k)(i) where the convolution is appropriate
    # More precisely: conv_k = sum_i a[i]*a[i+k]
    # d(conv_k)/da[i] = a[i+k] + a[i-k] (with bounds)
    # So dLSE/da[i] = 2 * sum_k w[k] * (a[i+k] + a[i-k])
    # = 4 * cross_corr(a, w) [symmetric case]
    
    # Actually let me be more careful:
    # conv(a,a)[m] = sum_i a[i] * a[i+m] for m >= 0
    # For m < 0: conv(a,a)[m] = sum_i a[i] * a[i+m] = sum_i a[i-m] * a[i] = conv(a,a)[-m]
    # So conv is symmetric. The full convolution has 2n-1 entries indexed -n+1 to n-1.
    # Let's index them 0 to 2n-2 where index m corresponds to lag m-(n-1).
    # conv_full[m] = sum_i a[i] * a[i + m - (n-1)]
    # d(conv_full[m])/da[i] = a[i + m - n + 1] + a[i - m + n - 1] (with bounds checking)

    dLSE_da = 2.0 * h  # gradient of LSE w.r.t. a
    
    # Now f = 2n * LSE / S^2

    grad = 2.0 * n * (dLSE_da * S - 2.0 * LSE) / (S ** 3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    score = 2.0 * n * LSE / (S * S)

def momentum_exp_grad(a, conv, weights, eta, momentum=0.9):

    h = cross_corr(a, weights)
    hmax = float(np.max(np.abs(h)))

    log_a = np.log(a + 1e-300)

def direct_subgrad_step(a, eta, top_k=5):

    n = len(a)
    S = float(np.sum(a))
    c = fast_self_conv(a)

    top_mean = float(np.mean(top_vals))
    
    # Weights: proportional to excess over mean

    w[top_idx] = np.maximum(top_vals - top_mean, 0.0)

    if wsum < 1e-15:

    else:
        w /= wsum
    
    # Subgradient direction
    h = cross_corr(a, w)

    # df/da[i] = 2n*(2*h[i]*S - 2*cmax) / S^3 (approximately)

    # For step: a_new = a - eta * grad, projected

    # f = 2n*cmax, df/da = 2n*(2h - 2*cmax) [since S=1, dS/da=1]

    log_a -= float(np.max(log_a))

def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(n_new) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x_new, x_old, a).astype(np.float64)
    return result

def momentum_optimize(a, deadline, verbose=True):
    """
    Momentum-based exponentiated gradient optimization.
    Uses adaptive step size and active set management.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    eta = 0.2
    top_k = 5
    frac = 0.02
    momentum = 0.75
    v = np.zeros(n)  # momentum buffer (in log-space)
    
    stagnant = 0
    max_stagnant = 25
    
    it = 0
    while time.monotonic() < deadline:
        conv = fast_self_conv(a)
        sc = 2.0 * n * float(np.max(conv))  # S=1
        
        # Compute weights
        w = top_k_weights(conv, top_k, frac)
        h = cross_corr(a, w)
        hmax = float(np.max(np.abs(h)))
        
        if hmax < 1e-15:
            break
        
        # Momentum step in log-space
        v = momentum * v - eta * h / hmax
        log_a = np.log(a + 1e-300) + v
        log_a -= float(np.max(log_a))
        new_a = np.exp(log_a)
        new_a = normalize_a(new_a)
        
        # Backtracking line search
        alpha = 1.0
        new_sc = exact_score(new_a)
        for _ in range(10):
            if new_sc < sc - 1e-12:
                break
            alpha *= 0.5
            v = momentum * v - alpha * eta * h / hmax
            log_a = np.log(a + 1e-300) + v
            log_a -= float(np.max(log_a))
            new_a = np.exp(log_a)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if alpha < 0.01:
                break
        
        if new_sc < best_sc - 1e-12:
            a = new_a
            best_a = a.copy()
            best_sc = new_sc
            stagnant = 0
            if verbose and it % 100 == 0:
                print("  [mom it{}] score={:.9f} eta={:.3f}".format(it, best_sc, eta))
        else:
            stagnant += 1
            eta *= 0.8
            momentum = min(0.9, momentum * 1.02)
            if stagnant >= max_stagnant:
                # Diversify
                top_k = min(top_k * 2, 50)
                frac = min(frac * 1.5, 0.1)
                eta = max(0.05, eta * 0.5)
                v *= 0.5
                stagnant = 0
        it += 1
    
    return float(best_sc), best_a


def lse_refine(a, deadline, verbose=True):
    """Adaptive LSE-LBFGS refinement with geometric beta schedule."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        
        # Run L-BFGS for a few iterations
        maxiter = max(30, min(200, int((deadline - time.monotonic()) * 0.3 / 0.1)))
        a_new = run_lbfgs_lse(a, beta, deadline, maxiter)
        sc = exact_score(a_new)

            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print("  [LSE beta={}] score={:.9f}".format(beta, best_sc))
        else:

def perturb_diverse(a, rng, strength=0.15):
    """Diverse perturbation for restarts."""
    a = normalize_a(a)
    n = len(a)
    mode = int(rng.integers(0, 8))
    
    if mode == 0:
        # Gaussian multiplicative
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Mix with uniform
        t = float(rng.uniform(0.1, 0.4))
        q = (1-t)*a + t*(np.ones(n)/n)
    elif mode == 2:
        # Mix with reversed
        t = float(rng.uniform(0.1, 0.3))
        q = (1-t)*a + t*a[::-1]
    elif mode == 3:
        # Smooth noise
        k = min(21, max(5, n//500))
        if k % 2 == 0: k += 1
        c = k // 2
        sig = max(1.0, c/2.0)
        kern = np.exp(-((np.arange(k)-c)**2)/(2.0*sig*sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, "same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.5 * noise)
    elif mode == 4:
        # Block perturbation
        nb = int(rng.integers(4, 16))
        block = 1.0 + strength * 0.5 * (rng.random(nb)*2-1)
        per = max(1, n//nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n-arr.size, block[-1])])
        q = a * arr[:n]
    elif mode == 5:
        # Shape perturbation
        x = np.linspace(0, 1, n)
        alpha = float(rng.uniform(-0.5, 0.5))
        q = a * (1.0 + alpha*(2*x-1)**2)
    elif mode == 6:
        # Frequency-domain perturbation
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        dphase = float(rng.uniform(0.1, 0.5)) * rng.standard_normal(len(A))
        dphase[0] = 0
        A_new = mag * np.exp(1j*(phase + dphase))
        q = np.fft.irfft(A_new, N)[:n]
    else:
        # anti-peak: reduce where it contributes most to peaks
        conv = fast_self_conv(a)
        k = 10
        top_idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[top_idx] = 1.0/k
        h = cross_corr(a, w)
        q = a * np.exp(-strength * 2.0 * h / max(float(np.max(h)), 1e-15))
    
    return normalize_a(np.clip(q, 0.0, None))

"""

Combines momentum exponentiated gradient with Anderson acceleration,

import numpy as np
import time
import math
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Compute g[j] = sum_l a[l] * w[l+j] for j=0..n-1.
    This equals conv(reversed_a, w)[n-1 : 2n-1]."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return np.nan_to_num(full[n - 1:2 * n - 1], nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def top_k_weights(conv, k, frac=0.0):
    """Return normalized weights concentrated on top-k or near-max positions."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    if frac > 0:
        threshold = dmax * (1.0 - frac)
        active = conv >= threshold
        count = int(active.sum())
        if count == 0:
            active[np.argmax(conv)] = True
            count = 1
        if count < k:
            idx = np.argpartition(conv, -k)[-k:]
            active[idx] = True
        max_count = max(k, int(0.05 * conv.size))
        if count > max_count:
            idx = np.argpartition(conv, -max_count)[-max_count:]
            active = np.zeros(conv.size, dtype=bool)
            active[idx] = True
    else:
        k = min(k, conv.size)
        idx = np.argpartition(conv, -k)[-k:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    w = active.astype(np.float64)
    s = float(w.sum())
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def lse_smooth_score(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    LSE = cmax + np.log(sum_e) / beta
    h = cross_corr(a, w)
    dLSE_da = 2.0 * h
    grad = 2.0 * n * (dLSE_da * S - 2.0 * LSE) / (S ** 3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    score = 2.0 * n * LSE / (S * S)
    if not np.isfinite(score):
        score = 1e10
    return float(score), grad


def run_lbfgs_lse(a, beta, deadline, maxiter=200):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a)
    if time.monotonic() > deadline - 3:
        return a.copy()

    def fun_and_grad(x):
        x = np.clip(x, 1e-12, None)
        return lse_smooth_score(x, beta)

    bounds = [(1e-10, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-12,
        "maxiter": maxiter,
        "maxfun": maxiter * 3,
        "maxls": 30,
    }
    try:
        res = minimize(fun_and_grad, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        a_out = res.x if res.x is not None else a
    except Exception:
        a_out = a
    a_out = np.clip(a_out, 0.0, None)
    return normalize_a(a_out)


def anderson_exp_grad(a, deadline, verbose=True):
    """
    Exponentiated gradient with Anderson acceleration.
    Maintains a history of iterates and mixing coefficients.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    m = 5  # Anderson memory
    history = [a.copy()]
    F_history = [a.copy()]
    eta = 0.3
    top_k = 5
    frac = 0.02
    stagnant = 0
    max_stagnant = 40

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        conv = fast_self_conv(a)
        sc = 2.0 * n * float(np.max(conv))

        w = top_k_weights(conv, top_k, frac)
        h = cross_corr(a, w)
        hmax = float(np.max(np.abs(h)))
        if hmax < 1e-15:
            break

        # Fixed-point map: T(a) = exp(log(a) - eta*h/hmax) normalized
        log_a = np.log(a + 1e-300) - eta * h / hmax
        log_a -= float(np.max(log_a))
        T_a = np.exp(log_a)
        T_a = normalize_a(T_a)

        # Anderson update
        if len(history) >= 2:
            # Compute residuals
            r_now = T_a - a
            r_prev = F_history[-1] - history[-1]
            # Simple 2-step Anderson
            dr = r_now - r_prev
            da = T_a - history[-1]
            denom = float(np.dot(dr, dr))
            if denom > 1e-15:
                alpha = float(np.dot(r_now, dr)) / (1.0 + denom)
                alpha = np.clip(alpha, -0.5, 1.0)
                a_new = (1.0 - alpha) * T_a + alpha * history[-1]
                F_new = (1.0 - alpha) * T_a + alpha * F_history[-1]
            else:
                a_new = T_a
                F_new = T_a
        else:
            a_new = T_a
            F_new = T_a

        a_new = normalize_a(np.clip(a_new, 0.0, None))
        new_sc = exact_score(a_new)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if verbose and it % 200 == 0:
                print("  [andr it{}] score={:.9f}".format(it, best_sc))
        else:
            a = a_new.copy()
            stagnant += 1
            if stagnant % 5 == 0:
                eta *= 0.85
            if stagnant >= max_stagnant:
                top_k = min(top_k * 2, 50)
                frac = min(frac * 1.5, 0.1)
                eta = max(0.05, eta * 0.5)
                stagnant = 0
                # Reset history
                history = [a.copy()]
                F_history = [a.copy()]

        # Update history
        history.append(a_new.copy())
        F_history.append(F_new.copy())
        if len(history) > m:
            history.pop(0)
            F_history.pop(0)

    return float(best_sc), best_a

            log_a_t -= float(np.max(log_a_t))

        if new_sc < best_sc - 1e-12:

            a = new_a.copy()

                    improved = True
                    if verbose:
                        print("  [sg r{} tk={} eta={}] {:.9f}".format(rounds, tk, eta, best_sc))

    return float(best_sc), best_a


def lse_adaptive_refine(a, deadline, verbose=True):
    """Adaptive LSE-LBFGS with geometric beta schedule."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    betas = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 16000, 32000, 64000, 128000, 256000]

    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        rem = deadline - time.monotonic()
        if rem < 10:
            break

        # Determine iterations based on remaining time
        est_time_per_iter = 0.05 * (n / 8000.0)
        maxiter = max(20, min(500, int(rem * 0.4 / est_time_per_iter)))

        a_new = run_lbfgs_lse(a, beta, deadline, maxiter)
        sc = exact_score(a_new)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            if verbose:
                print("  [LSE beta={:>6}] {:.9f}".format(beta, best_sc))
        else:
            # Still use new iterate for next beta (smoother landscape)
            a = a_new
            if verbose:
                print("  [LSE beta={:>6}] no improvement ({:.9f})".format(beta, sc))

    return float(best_sc), best_a


def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(n_new) / n_new
    if m == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)


def perturb_diverse(a, rng, strength=0.15):
    """Generate diverse perturbations for restarts."""
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        t = float(rng.uniform(0.1, 0.3))
        q = (1 - t) * a + t * a[::-1]
    elif mode == 3:
        k = min(31, max(5, n // 400))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, "same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.5 * noise / max(1e-10, np.std(noise)))
    elif mode == 4:
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.5 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = a * arr[:n]
    elif mode == 5:
        x = np.linspace(0, 1, n)
        alpha = float(rng.uniform(-0.5, 0.5))
        q = a * (1.0 + alpha * (2 * x - 1) ** 2)
    elif mode == 6:
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        dphase = float(rng.uniform(0.2, 0.8)) * rng.standard_normal(len(A))
        dphase[0] = 0
        A_new = mag * np.exp(1j * (phase + dphase))
        q = np.fft.irfft(A_new, N)[:n]
        q = np.clip(q, 0.0, None)
    elif mode == 7:
        # Anti-peak
        conv = fast_self_conv(a)
        k = 10
        top_idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[top_idx] = 1.0 / k
        h = cross_corr(a, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = a * np.exp(-strength * 2.0 * h / hmax)
    elif mode == 8:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, "same")
    else:
        # Frequency amplitude perturbation
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        mag = np.abs(A) * (1.0 + strength * 0.3 * rng.standard_normal(len(A)))
        mag = np.clip(mag, 0.0, None)
        phase = np.angle(A)
        q = np.fft.irfft(mag * np.exp(1j * phase), N)[:n]

    return normalize_a(np.clip(q, 0.0, None))


def generate_inits(n, rng):
    """Generate diverse initial sequences for exploration."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=1.0, posinf=1.0, neginf=1.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        arr = np.clip(arr, 0.0, None)
        s = arr.sum()
        if s > 1e-10:
            inits.append(arr / s)

    add(np.ones(n))
    for alpha in [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]:
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for alpha in [0.3, 0.6, 1.0]:
        add(np.maximum(0.01, 1.0 - alpha * (2.0 * x - 1.0) ** 2))
    for f in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.7, 1.0, 1.5, 2.0]:
            add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
    for f in [1, 2, 3, 4]:
        for amp in [0.5, 1.0, 2.0, 3.0]:
            add(1.0 + amp * np.sin(np.pi * f * x) ** 2)
    for rate in [1, 2, 4, 8, 16]:
        add(np.exp(-rate * x) + np.exp(-rate * (1 - x)) + 0.01)
    for w in [0.05, 0.1, 0.15, 0.2, 0.3]:
        for c1 in [0.2, 0.25, 0.3, 0.35, 0.4]:
            c2 = 1.0 - c1
            add(np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2) + 0.01)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        add(np.abs(2.0 * x - 1.0) ** p + 0.01)
    Nf = next_power_of_two(2 * n)
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        alpha_exp = float(rng.uniform(2.0, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** alpha_exp)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        sig = float(rng.uniform(0.3, 1.5))
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        std = np.std(noise)
        if std > 1e-10:
            noise = sig * noise / std
        add(1.0 + noise)
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: multi-phase optimization with Anderson acceleration,
    adaptive LSE-LBFGS, subgradient polishing, and diversification restarts."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    # Choose length
    if parent is not None:
        n_parent = int(parent.size)
        # Keep parent length if reasonable, else adjust
        if 4096 <= n_parent <= 32768:
            n = n_parent
        elif n_parent < 4096:
            n = 8192
        else:
            n = 16384
        a0 = resample_sequence(parent, n)
    else:
        n = 16384
        inits = generate_inits(n, rng)
        best_init_sc = np.inf
        a0 = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a0 = init
        print("No parent. Best init: {:.9f}".format(best_init_sc))

    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score(a)
    print("Start: n={}, score={:.9f}".format(n, best_sc))

    # Phase 1: Anderson-accelerated exponentiated gradient (~25% of time)
    p1_dead = t_start + 0.25 * total_budget
    sc, a = anderson_exp_grad(a, deadline=p1_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("Phase 1 (Anderson): {:.9f}".format(best_sc))

    # Phase 2: Momentum exponentiated gradient (~15% of time)
    p2_dead = t_start + 0.40 * total_budget
    sc, a = momentum_exp_grad(a, deadline=p2_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("Phase 2 (Momentum): {:.9f}".format(best_sc))

    # Phase 3: Adaptive LSE-LBFGS refinement (~30% of time)
    p3_dead = t_start + 0.70 * total_budget
    sc, a = lse_adaptive_refine(best_a, deadline=p3_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("Phase 3 (LSE-LBFGS): {:.9f}".format(best_sc))

    # Phase 4: Subgradient polish (~10% of time)
    p4_dead = t_start + 0.80 * total_budget
    sc, a = subgrad_polish(best_a, deadline=p4_dead, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("Phase 4 (Subgrad): {:.9f}".format(best_sc))

    # Phase 5: High-beta LSE finishing (~10% of time)
    p5_dead = t_start + 0.90 * total_budget
    for beta in [100000, 200000, 500000, 1000000]:
        if time.monotonic() > p5_dead - 5:
            break
        rem = p5_dead - time.monotonic()
        if rem < 10:
            break
        maxiter = max(50, int(rem * 0.7 / 0.08))
        a_new = run_lbfgs_lse(best_a, beta, p5_dead, maxiter)
        sc = exact_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            print("  Phase 5 beta={}: {:.9f}".format(beta, best_sc))

    # Phase 6: Diversification restarts (~9% of time)
    p6_dead = deadline - 10.0
    n_restarts = 6
    restart_budget = max(0.0, p6_dead - time.monotonic())
    per_restart = restart_budget / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > p6_dead - 10:
            break
        rem = p6_dead - 10 - time.monotonic()
        if rem < 30:
            break

        # Create perturbation
        strength = float(rng.uniform(0.03, 0.2)) * (1.0 + 0.3 * r / n_restarts)
        p = perturb_diverse(best_a, rng, strength=strength)

        # Quick Anderson optimization
        chunk1 = time.monotonic() + rem * 0.35
        sc_r, a_r = anderson_exp_grad(p, deadline=chunk1, verbose=False)

        # LSE refinement
        chunk2 = time.monotonic() + rem * 0.65
        if time.monotonic() < chunk2:
            for beta in [500, 2000, 10000, 50000]:
                if time.monotonic() > chunk2 - 3:
                    break
                mi = max(20, int((chunk2 - time.monotonic()) * 0.5 / 0.06))
                a_r = run_lbfgs_lse(a_r, beta, chunk2, mi)
            sc_r = exact_score(a_r)

        # Final subgrad
        if time.monotonic() < p6_dead - 15:
            sc_r2, a_r2 = subgrad_polish(a_r, deadline=p6_dead - 15, verbose=False)
            if sc_r2 < sc_r:
                sc_r = sc_r2
                a_r = a_r2

        if sc_r < best_sc - 1e-12:
            best_sc = sc_r
            best_a = a_r.copy()
            print("Restart {}: {:.9f}".format(r + 1, best_sc))

    # Phase 7: Final polish
    if time.monotonic() < deadline - 5:
        sc, a = subgrad_polish(best_a, deadline=deadline - 3, verbose=False)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final high-beta squeeze
    if time.monotonic() < deadline - 3:
        for beta in [500000, 1000000]:
            if time.monotonic() > deadline - 3:
                break
            a_new = run_lbfgs_lse(best_a, beta, deadline, maxiter=100)
            sc = exact_score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()

    # Output
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
