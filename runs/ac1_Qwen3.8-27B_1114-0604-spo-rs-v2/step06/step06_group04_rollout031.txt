Minimize 2*n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
import time

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S

def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)

def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

def _subgrad_exact(a, k_active=3, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam optimizer with aggressive progressive LSE smoothing (many cheap iterations)
2. Exponentiated gradient (mirror descent) variants for diversity
3. Periodic Nesterov subgradient polishing with adaptive active set
4. Final L-BFGS-B polish with very high beta
5. Multi-length exploration (10240, 14336, 16384)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (4.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

def _nesterov_quick(a0, n_iters=200, deadline=None, k_active=3, tol=1e-7):

    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0

    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.04 / g_norm

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.03 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        a = x_new
        t = t_new
        sc = _exact_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

def _lbfgs_final(a0, deadline, maxfun=300):
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, 2000000)
        return val, grad

    opts = {"ftol": 1e-15, "gtol": 1e-12, "maxiter": maxfun, "maxfun": maxfun * 3, "maxls": 30}
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=[(0.0, None)] * n, options=opts)
        a_out = _normalize(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 10.0:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 5))
    if mode == 0:
        q = a * np.exp(strength * 0.4 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.25 * noise)
    else:
        q = a * np.exp(0.03 * rng.standard_normal(n))
    return _normalize(np.clip(q, 0.0, None))


def _full_pipeline(a0, deadline, label=""):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t0 = time.monotonic()
    total = max(10.0, deadline - t0)

    # Phase 1: Adam with low-mid beta (40% time)
    d1 = t0 + 0.40 * total
    if time.monotonic() < d1 - 5:
        beta_sched = [50, 200, 1000, 5000]
        sc, seq = _adam_optimize(a, d1, beta_sched, lr_base=0.005, nesterov_every=300, nesterov_iters=30, label=label + "-P1")
        if sc < best_score:
            best_score = sc
            best_a = seq.copy()
        a = _normalize(seq)

    # Phase 2: Adam with high beta (35% time)
    d2 = t0 + 0.75 * total
    if time.monotonic() < d2 - 5:
        beta_sched = [20000, 100000, 500000]
        sc, seq = _adam_optimize(a, d2, beta_sched, lr_base=0.002, nesterov_every=400, nesterov_iters=50, label=label + "-P2")
        if sc < best_score:
            best_score = sc
            best_a = seq.copy()
        a = _normalize(seq)

    # Phase 3: Nesterov intensive (15% time)
    d3 = t0 + 0.90 * total
    if time.monotonic() < d3 - 5:
        for k_act in [1, 2, 3, 5]:
            rem = d3 - time.monotonic()
            if rem < 10:
                break
            n_it = int(max(100, rem * 15))
            sc, seq = _nesterov_quick(a, n_iters=n_it, deadline=d3, k_active=k_act, tol=1e-8)
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
                print("[{}] Nesterov k={}: {:.8f}".format(label, k_act, best_score))
            a = _normalize(seq)

    # Phase 4: L-BFGS final + Nesterov (10% time)
    d4 = deadline
    if time.monotonic() < d4 - 15:
        rem = d4 - time.monotonic()
        if rem > 20:
            mf = int(max(50, min(400, rem * 3)))
            sc, seq = _lbfgs_final(a, d4 - 10, maxfun=mf)
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
                print("[{}] L-BFGS final: {:.8f}".format(label, best_score))
            a = _normalize(seq)
        rem = d4 - time.monotonic()
        if rem > 10:
            sc, seq = _nesterov_quick(a, n_iters=int(max(50, rem * 20)), deadline=d4, k_active=2, tol=1e-9)
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
                print("[{}] Final Nesterov: {:.8f}".format(label, best_score))

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        pl = min(parent.size, 16384)
        if pl < 4096:
            pl = 10240
        lengths = [pl, 14336, 16384]
    else:
        lengths = [10240, 14336, 16384]

    # Remove duplicates, keep sorted by closeness to parent
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen and 4096 <= L <= 20480:
            unique_lengths.append(L)
            seen.add(L)
    lengths = unique_lengths[:3]
    print("Lengths: {}".format(lengths))

    # Time allocation: 60% to first (parent) length, 25% to second, 15% to third
    if len(lengths) == 1:
        fracs = [1.0]
    elif len(lengths) == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.60, 0.25, 0.15]

    best_score = float(np.inf)
    best_seq = None

    cum = 0.0
    for i, n in enumerate(lengths):
        cum += fracs[i]
        sub_dead = t_start + cum * (deadline - t_start)
        sub_dead = min(sub_dead, deadline - 10.0)

        if time.monotonic() > sub_dead - 10.0:
            continue

        # Create initial sequence
        if parent is not None and parent.size >= 64:
            a0 = _resample(parent, n)
        else:
            a0 = np.ones(n) / n
        a0 = _normalize(a0)

        initial_score = _exact_score(a0)
        print("Length {} initial: {:.8f}".format(n, initial_score))

        # Run optimization pipeline
        sc, seq = _full_pipeline(a0, sub_dead, "L{}".format(n))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After L{}: {:.8f}".format(n, best_score))

        # If time remains and this is the best so far, do perturbation restarts
        rem = deadline - 10.0 - time.monotonic()
        if np.isfinite(sc) and sc <= best_score and rem > 60.0:
            for pi in range(3):
                pres = time.monotonic()
                if pres > deadline - 30.0:
                    break
                p = _perturb(best_seq, rng, strength=float(rng.uniform(0.03, 0.12)))
                p_dead = min(pres + rem / (3 - pi), deadline - 20.0)
                sc2, seq2 = _full_pipeline(p, p_dead, "L{}-pert{}".format(n, pi))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = seq2.copy()
                    print("Perturb {} L{}: {:.8f}".format(pi, n, best_score))

    # Final safety: if we have no result, use uniform
    if best_seq is None:
        n0 = lengths[0] if lengths else 10240
        best_seq = np.ones(n0) / n0
        best_score = _exact_score(best_seq)

    # Final verification and output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
        S = float(out_arr.sum())
    out_arr = out_arr / S
    out = [float(x) for x in out_arr]

    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
