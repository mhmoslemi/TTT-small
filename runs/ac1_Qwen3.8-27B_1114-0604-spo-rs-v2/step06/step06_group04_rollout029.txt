def _next_pow2(m):
    m = int(max(1, m))

def _fast_self_conv(a):

    out = np.fft.irfft(A * A, N)[:M]

    # Simpler: create a vector v where v[k] = 1 (length 2n-1), then grad = convolve(a, v) truncated
    # Actually: grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) where w[k] = 1 for the specific k
    
    # For a single k:
    # grad[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    
    grad_c = np.zeros(n)
    # First term: a[k-i] for i where 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        # grad_c[i] += a[k-i] for i in [lo1, hi1]
        # a[k-i] for i=lo1,...,hi1 is a[k-lo1], a[k-lo1-1], ..., a[k-hi1]
        # = a[min(n-1,k)] : a[max(0,k-n+1)]-1 (reversed slice)
        idx = np.arange(lo1, hi1 + 1)
        grad_c[lo1:hi1+1] += a[k - lo1 : k - hi1 : -1]  # reversed
    
    # Second term: a[i-k] for i where 0 <= i-k <= n-1, i.e., k <= i <= k+n-1
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad_c[lo2:hi2+1] += a[lo2 - k : hi2 - k + 1]
    
    # Full gradient of n * (a*a)_k / S^2:
    # = n/S^2 * grad_c - 2n * (a*a)_k / S^3 * a
    T = c[k]  # (a*a)_k
    grad = (n / (S * S)) * grad_c - (2.0 * n * T) / (S * S * S) * a
    return grad

def _subgrad_multi_peak(a, peaks, weights):
    """Compute weighted subgradient at multiple peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)
    
    c = _fast_self_conv(a)
    
    # Use FFT-based approach for multiple peaks
    w = np.zeros(2 * n - 1)
    for k, wt in zip(peaks, weights):
        w[k] = wt
    
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    T = float(w.dot(c))
    grad = (n / (S * S)) * R - (2.0 * n * T) / (S * S * S) * a
    return np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

def _project_simplex(v):
    """Euclidean projection onto {x : sum(x)=1, x>=0}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _exp_grad_step(a, grad, eta):
    """Exponentiated gradient step on the simplex."""
    # a * exp(-eta * grad), then normalize
    log_a = np.log(np.maximum(a, 1e-30))
    new_log = log_a - eta * grad
    # Shift for numerical stability
    new_log -= new_log.max()
    new_a = np.exp(new_log)
    S = new_a.sum()
    if S < 1e-15:
        return a.copy()
    return new_a / S

def _lse_lbfgs(a, beta, maxiter=200, maxfun=500):
    """Run L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a)
    n = a.size
    
    def fun(x):
        x = np.clip(x, 0.0, None)
        S = x.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad
    
    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": maxiter,
        "maxfun": maxfun,
        "maxls": 50,
    }
    
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        a_out = _normalize(res.x) if res.x is not None else a.copy()
    except Exception:
        a_out = a.copy()
    
    return a_out

def _nesterov_opt(a, n_iters, deadline, k_active=3):
    """Nesterov accelerated subgradient method."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    c = _fast_self_conv(a)
    S = float(a.sum())
    d = n * c / (S * S)
    dmax = d.max()
    tol = 0.01 * dmax
    
    active = np.where(d >= dmax - tol)[0]
    if len(active) > k_active:
        active = active[np.argsort(d[active])[-k_active:]]
    
    w_active = np.ones(len(active)) / len(active)
    grad = _subgrad_multi_peak(a, active, w_active)
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-12:
        return best_score, best_a
    
    eta = 0.1 / g_norm
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_m = (t - 1.0) / t_new
        y = a + beta_m * v
        y = np.clip(y, 0.0, None)
        Sy = y.sum()
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy
        
        # Recompute active set periodically
        if it % 20 == 0:
            c = _fast_self_conv(y)
            S_curr = float(y.sum())
            d_curr = n * c / (S_curr * S_curr)
            dmax_curr = d_curr.max()
            tol_curr = 0.01 * dmax_curr
            active = np.where(d_curr >= dmax_curr - tol_curr)[0]
            if len(active) > k_active:
                active = active[np.argsort(d_curr[active])[-k_active:]]
            w_active = np.ones(len(active)) / len(active)
        
        grad = _subgrad_multi_peak(y, active, w_active)
        step = eta / (1.0 + 0.05 * it)
        x_new = y - step * grad
        x_new = _project_simplex(x_new)
        
        v = x_new - a
        a = x_new
        t = t_new
        
        sc = _exact_score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            eta *= 0.5
            no_improve = 0
    
    return best_score, best_a

def _exp_grad_opt(a, n_iters, deadline):
    """Exponentiated gradient optimization with adaptive step size."""
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    eta = 5.0  # initial step size
    t = 0.0
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _fast_self_conv(a)
        S = float(a.sum())
        d = n * c / (S * S)
        dmax = d.max()
        score = 2.0 * dmax
        
        # LSE weights for the subgradient
        beta = max(50.0, 1000.0 * (1.0 + it / float(n_iters)))
        z = beta * (d - dmax)
        z = np.clip(z, -300, 300)
        e = np.exp(z)
        w = e / e.sum()
        
        # Compute subgradient with LSE weights
        N2 = _next_pow2(3 * n)
        a_rev = a[::-1]
        AR = np.fft.rfft(a_rev, N2)
        WR = np.fft.rfft(w, N2)
        conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
        R = conv_full[n - 1:2 * n - 1]
        T = float(w.dot(c))
        grad = (n / (S * S)) * R - (2.0 * n * T) / (S * S * S) * a
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        
        # Adaptive step: target is to move a fraction of the range
        step = eta / g_norm
        a_new = _exp_grad_step(a, grad, step)
        
        sc_new = _exact_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            eta *= 0.7
            # Try a smaller step
            if eta > 0.01:
                a_try = _exp_grad_step(a, grad, eta / g_norm)
                sc_try = _exact_score(a_try)
                if sc_try < _exact_score(a):
                    a = a_try
        
        t += 1
        # Slowly increase eta to allow exploration
        if it % 100 == 0 and it > 0:
            eta = min(eta * 1.1, 20.0)
    
    return best_score, best_a

def _perturb(a, rng, strength=0.3):
    """Generate a perturbation of a."""
    a = _normalize(a)
    n = a.size
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Smooth noise
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * (c/2.5)**2))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.2 * noise)
    elif mode == 3:
        # Localized perturbation
        center = int(rng.integers(n//4, 3*n//4))
        width = int(n * float(rng.uniform(0.05, 0.2)))
        idx = np.arange(n)
        bump = np.exp(-((idx - center)**2) / (2*width**2))
        q = a * (1.0 + strength * 0.5 * (bump - bump.mean()))
    else:
        # Multiplicative shift
        shift = float(rng.uniform(0.1, 0.5))
        x = np.linspace(0, 1, n)
        q = a * (1.0 + shift * (x - 0.5))
    
    q = np.clip(q, 0.0, None)
    S = q.sum()
    if S < 1e-12:
        return a.copy()
    return q / S
