def fienup_geometric(a, n_iters=400, alpha_start=0.1, alpha_end=0.9):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for L-BFGS."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum + 1e-30)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # g[i] = 2 * sum_k p[k] * a[k-i]
    rev_a = x[::-1]
    N2 = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N2)
    RA = np.fft.rfft(rev_a, N2)
    conv_res = np.fft.irfft(P * RA, N2)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def spectral_nulling(a, n_iters=200, n_nulls=5):
    """
    Identify spectral bins that contribute most to the time-domain peak
    and suppress them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_peak = int(np.argmax(c))
        
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # The peak at lag k_peak is influenced by spectral components.
        # b_k = inverse_fft(|A|^2)_k
        # To reduce b_{k_peak}, we want to reduce |A|^2 at frequencies
        # that constructively interfere at that lag.
        # b_k = sum_m |A_m|^2 * exp(2πi*m*k/M) ... (for circular conv)
        # For linear conv it's more complex, but as an approximation:
        # The contribution of bin m to b_k is |A_m|^2 * exp(-2πi*m*k/M_fft)
        # (for the circular part of the linear conv)
        
        # Compute contribution of each bin to the peak
        m_vals = np.arange(len(A))
        contrib = amp ** 2 * np.cos(2 * np.pi * m_vals * k_peak / M_fft)
        
        # Suppress bins with high positive contribution
        top_bins = np.argsort(contrib)[-n_nulls:]
        for bin_idx in top_bins:
            amp[bin_idx] *= 0.95
        
        A_new = amp * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

4. The gradient is clean: `db_{k*}/da_i = 2*a_{k*-i}`

For the center lag (k* = n-1): `b_{n-1} = sum_j a_j a_{n-1-j}`
Gradient: `db_{n-1}/da_i = 2*a_{n-1-i}` (the reversed sequence!)

So gradient descent on the center peak is: `a_i -= step * 2*a_{n-1-i}`
This means: elements that are large in the "mirror" position get reduced.
If the sequence is symmetric (a_i = a_{n-1-i}), then all elements get reduced equally, which just reduces S. The normalization then brings them back. So for symmetric sequences, this alone doesn't help!

This is a crucial insight. The peak descent only helps when the sequence is **asymmetric**. For symmetric sequences, the gradient is proportional to the sequence itself (just reversed = same), and after normalization, there's no change.

So the algorithm needs to:
1. Create asymmetry (e.g., through spectral shaping or random perturbation)
2. Use peak descent to exploit the asymmetry
3. Repeat

"""
Step function minimization using clean gradient peak descent, LSE-smoothed
multi-peak optimization, Fienup spectral shaping, and L-BFGS-B polish.
Key insight: gradient of b_k w.r.t. a_i = 2*a_{k-i}, giving a simple
reversed/shifted descent direction. LSE smoothing enables robust multi-peak
handling with L-BFGS for fast convergence near the optimum.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_weighted_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    L = c_max + log(sum(exp(beta*(c_k - c_max))))/beta
    f = 2*n*L/S^2
    grad_i = 2*n*(g_i/S^2 - 2*L/S^3) where g_i = 2*sum_k p_k*a_{k-i}
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    s_max = float(scaled.max())
    exp_scaled = np.exp(scaled - s_max)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + (s_max + np.log(exp_sum)) / beta
    p = exp_scaled / exp_sum

    # g[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * conv(p, a[::-1])[i + n - 1]
    rev_a = a[::-1]
    N2 = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N2)
    RA = np.fft.rfft(rev_a, N2)
    conv_res = np.fft.irfft(P * RA, N2)
    g = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def peak_descent(a, n_iters=5000, use_lse=False, beta=500.0,
                 step_init=0.05, backtracking=True):
    """
    Peak-directed descent using exact or LSE-weighted gradient.
    Uses backtracking line search for robust convergence.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    stall_count = 0

    for it in range(n_iters):
        if use_lse:
            sc_smooth, grad = lse_weighted_grad(a, beta)
        else:
            c = fast_conv(a)
            k_peak = int(np.argmax(c))
            S_curr = float(a.sum())
            b_k = c[k_peak]
            # grad_i = 2*n*(2*a_{k-i}/S^2 - 2*b_k/S^3)
            S2 = S_curr * S_curr
            S3 = S2 * S_curr
            g = np.zeros(n)
            i_start = max(0, k_peak - n + 1)
            i_end = min(k_peak, n - 1)
            if i_end >= i_start:
                indices = np.arange(i_start, i_end + 1)
                a_idx = k_peak - indices
                g[indices] = 2.0 * a[a_idx]
            grad = (2.0 * n) * (g / S2 - 2.0 * b_k / S3)
            sc_smooth = 2.0 * n * b_k / S2

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-14:
            break

        # Backtracking line search
        a_new = None
        sc_new = None
        s = step
        for _ in range(25):
            trial = a - s * grad
            trial = np.maximum(trial, 0.0)
            S_t = float(trial.sum())
            if S_t < 1e-12:
                s *= 0.4
                continue
            trial = trial / S_t
            sc_t = compute_score(trial)
            if sc_t < sc_smooth - 1e-14:
                a_new = trial
                sc_new = sc_t
                break
            s *= 0.5

        if a_new is not None:
            # Expand step if we took full step
            if s == step:
                step = min(step * 1.3, 0.2)
            else:
                step = min(step, s * 1.5)
            a = a_new
            stall_count = 0
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            step *= 0.5
            stall_count += 1
            if stall_count > 100:
                break

    return best_a


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.8):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_ellip(a, n_iters=200, beta_shape=3.0):
    """Elliptical spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        beta = beta_shape * (0.5 + progress)

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        ratio = amp / C
        amp_new = amp / np.sqrt(1.0 + beta * (ratio - 1.0) ** 2)

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for L-BFGS."""
    return lse_weighted_grad(x, beta)


def run_lbfgs(a0, beta, timeout_s=300, maxiter=5000):
    """Run L-BFGS-B on the LSE-smoothed objective with timeout."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    t0 = time.monotonic()

    def callback(xk):
        return time.monotonic() - t0 > timeout_s

    bounds = [(0.0, 1000.0)] * n
    maxfun = maxiter * 20
    try:
        res = minimize(
            lse_obj_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30},
            callback=callback
        )
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def alternating_descent(a, n_iters=2000, beta_start=50.0, beta_end=5000.0,
                        step_init=0.03):
    """
    Alternating between LSE-smoothed gradient steps with increasing beta.
    Simulates annealing: start with many peaks considered, focus on dominant.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        beta = beta_start * (beta_end / beta_start) ** progress

        sc_smooth, grad = lse_weighted_grad(a, beta)
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-14:
            break

        # Backtracking
        a_new = None
        s = step
        for _ in range(20):
            trial = a - s * grad
            trial = np.maximum(trial, 0.0)
            S_t = float(trial.sum())
            if S_t < 1e-12:
                s *= 0.5
                continue
            trial = trial / S_t
            sc_t = compute_score(trial)
            if sc_t < sc_smooth:
                a_new = trial
                break
            s *= 0.5

        if a_new is not None:
            if s == step:
                step = min(step * 1.2, 0.15)
            else:
                step = max(step * 0.8, s)
            a = a_new
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            step *= 0.5

    return best_a
