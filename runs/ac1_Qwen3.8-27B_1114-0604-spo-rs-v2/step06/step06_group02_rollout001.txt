from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_s = np.exp(scaled)
    e_sum = float(exp_s.sum())
    if e_sum < 1e-300:
        p = np.zeros(len(c)); p[int(np.argmax(c))] = 1.0
        L = c_max
    else:
        L = c_max + float(np.log(e_sum)) / beta
        p = exp_s / e_sum
    score = 2.0 * n * L / (S * S)
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)

    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)

        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

    try:
        res = minimize(
            lse_obj_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,

        a_out = a0
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out

        sc = 2.0 * n * c_max / (S * S)

        M2 = 2 * n - 1

        S3 = S2 * S

            lr = min(lr * 1.005, 0.1)
        else:
