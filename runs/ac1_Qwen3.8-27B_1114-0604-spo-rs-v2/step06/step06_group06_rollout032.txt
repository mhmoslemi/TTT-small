def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def conv_with_a(d, a):
    """Compute conv(d, a) where d and a have the same length n.
    Returns the first 2n-1 elements of the convolution."""
    n = len(a)
    N = next_power_of_two(2 * n - 1)
    D = np.fft.rfft(d, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(D * A, N)[:2*n-1]

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

def qp_active_set_step(a, M_target, active_indices, max_iter=50):
    """
    Find a direction d that reduces the active convolution values below M_target.
    Uses projected gradient on the QP: min ||d||^2 s.t. conv(d,a)[k] <= rhs[k] for k in active,
    a[i]+d[i] >= 0, sum(d) = 0.
    
    Simplified: use subgradient of the max of the active constraints.
    """
    n = len(a)
    # Active constraints: conv(d,a)[k] <= rhs[k]
    # where rhs[k] = (M_target - conv(a,a)[k]) / 2
    c = fast_self_conv(a)
    rhs = (M_target - c[active_indices]) / 2.0  # These should be negative (we want to reduce)
    
    # Subgradient method on the QP
    # The QP is: min 0.5*d'd s.t. A_d @ d <= b, d >= -a, 1'd = 0
    # where A_d[k,i] = a[k-i] (valid) + a[i-k] (valid) ... actually conv(d,a)[k] = sum_i d[i]*a[k-i]
    # So the constraint matrix row k is: [a[k], a[k-1], ..., a[0], 0, ..., a[k-n+1], ..., a[k-(n-1)]]
    
    # For subgradient: at current d=0, the violated constraints are those with rhs < 0
    # (which is all active ones since M_target < max(c))
    # Subgradient of constraint k w.r.t. d is: [a[k-i] for valid i] (the k-th row)
    
    # Let's just use a simple approach: d = -gradient of (sum of violated constraints)
    # projected onto {sum(d)=0, d>=-a}
    
    # Sum of active constraints at d: sum_k conv(d,a)[k] = sum_k sum_i d[i]*a[k-i]
    # = sum_i d[i] * sum_k a[k-i] = sum_i d[i] * (sum of a shifted by i)
    
    # For large n, sum_k a[k-i] ≈ sum(a) = 1 (for i in the middle)
    # So the subgradient is approximately uniform!
    
    # This means the subgradient direction is approximately -1 (uniform decrease),
    # which after projection (sum=0) gives 0. Not useful.
    
    # Better: use weighted subgradient focusing on the most violated constraints.
    # The most violated are those with the largest c[k] (closest to max).
    
    # Let's use: d_i = -sum_k w[k] * a[k-i] (the cross-correlation of w with a)
    # where w[k] = 1 for k in active set, 0 otherwise.
    # Then project onto {sum(d)=0, d >= -a/t} for some t.
    
    w_active = np.zeros(2*n-1)
    w_active[active_indices] = 1.0
    
    # d[i] = -sum_k w_active[k] * a[k-i] (with valid indices)
    # = -conv(w_active, a)[i] (first n elements)
    N = next_power_of_two(2*n-1+n-1)
    WA = np.fft.rfft(w_active, N)
    A = np.fft.rfft(a, N)
    d = -np.fft.irfft(WA * A, N)[:n]
    
    # Project: sum(d) = 0, d >= -a*0.5 (limit step size)
    d = d - d.mean()
    d = np.maximum(d, -0.5 * a)
    # Re-project sum
    d = d - d.mean()
    
    return d

    best_sc = exact_score(a)
    
    if betas is None:
        betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        
        # Estimate time per function evaluation
        t0 = time.monotonic()
        f0, g0 = lse_objective_and_grad(a, beta)
        t_eval = time.monotonic() - t0
        maxfun = max(20, int(rem * 0.6 / max(t_eval, 0.01)))
        maxfun = min(maxfun, 5000)
        
        def fun(x):
            return lse_objective_and_grad(x, beta)
        
        bounds = [(0.0, 10.0)] * n
        opts = {
            "ftol": 1e-16,
            "gtol": 1e-12,
            "maxiter": max(20, maxfun // 3),
            "maxfun": maxfun,
            "maxls": 30,
        }
        
        try:
            res = minimize(fun, a, method="L-BFGS-B", jac=True,
                           bounds=bounds, options=opts)
            a_new = normalize_a(res.x) if res.x is not None else a
        except Exception:
            a_new = a
        
        sc = exact_score(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print("    LSE beta={}: {:.9f} ({} fun evals)".format(beta, best_sc, res.nfev))
    
    return float(best_sc), best_a

def active_set_optimize(a, deadline, verbose=True, tag=""):
    """
    Main active-set optimizer using iterative QP/gradient steps.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()
    
    top_k = 20
    stagnant = 0
    
    while time.monotonic() < deadline - 5:
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        sc = 2.0 * n * cmax  # since sum(a)=1
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        
        # Find active set
        threshold = cmax * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) < top_k:
            idx = np.argpartition(c, -top_k)[-top_k:]
            active = idx
        active = np.sort(active[:min(len(active), top_k)])
        
        # Try to reduce the max by moving mass away from active positions
        # Direction: d[i] = -sum_{k in active} a[k-i] (push down active conv values)
        w_active = np.zeros(2*n-1)
        w_active[active] = 1.0 / len(active)
        
        N = next_power_of_two(3*n-2)
        WA = np.fft.rfft(w_active, N)
        A = np.fft.rfft(a, N)
        d = -np.fft.irfft(WA * A, N)[:n]
        
        # Also try the "raise the valley" direction
        # d2[i] = sum_{k NOT in active} a[k-i] * (1 - c[k]/cmax)
        w_valley = np.zeros(2*n-1)
        w_valley[~np.isin(np.arange(2*n-1), active)] = 1.0 - c/cmax
        w_valley[active] = 0
        if w_valley.sum() > 0:
            w_valley /= w_valley.sum()
            WV = np.fft.rfft(w_valley, N)
            d2 = np.fft.irfft(WV * A, N)[:n]
        else:
            d2 = np.zeros(n)
        
        # Try different step sizes and direction combinations
        best_step_sc = sc
        best_step_a = a
        
        for alpha in [0.5, 1.0, 1.5]:
            for mix in [1.0, 0.8, 0.5, 0.3]:
                d_combo = mix * d + (1 - mix) * d2
                d_combo = d_combo - d_combo.mean()
                
                for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
                    a_try = a + alpha * t * d_combo
                    a_try = np.clip(a_try, 0, None)
                    S = a_try.sum()
                    if S < 1e-10:
                        continue
                    a_try = a_try / S
                    sc_try = exact_score(a_try)
                    if sc_try < best_step_sc:
                        best_step_sc = sc_try
                        best_step_a = a_try.copy()
        
        if best_step_sc < sc - 1e-10:
            a = best_step_a
        else:
            stagnant += 1
            if stagnant > 30:
                # Diversify: random perturbation
                a = normalize_a(a * np.exp(0.1 * np.random.standard_normal(n)))
                stagnant = 0
    
    # Final check
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return float(best_sc), best_a

def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n_new)) / float(max(1, n_new))
    if m == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, np.maximum(a, 0)).astype(np.float64)

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel().copy()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        S = arr.sum()
        if S > 1e-10:
            inits.append(arr / S)
    
    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
    
    # Uniform
    add(np.ones(n))
    
    # V-shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        add(1.0 - alpha * (2.0*x - 1.0)**2 + 0.1)
    
    # U-shapes
    for alpha in [0.5, 1.0, 2.0]:
        add(1.0 + alpha * (2.0*x - 1.0)**2)
    
    # Ripples
    for f in [1, 2, 3, 5]:
        for alpha in [0.3, 0.7, 1.5]:
            add(1.0 + alpha * np.cos(2*np.pi*f*x) * 0.5 + 0.5)
    
    # Exponential
    for rate in [1, 2, 4, 8]:
        add(np.exp(-rate*x) + np.exp(-rate*(1-x)) + 0.01)
    
    # Random smooth
    for _ in range(5):
        Nf = next_power_of_two(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** rng.uniform(1.5, 4.0))
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j*phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + 0.5*noise)
    
    return inits

def spectral_perturb(a, rng, strength=0.3):
    """Perturb the sequence by randomizing the phase of its spectrum."""
    n = len(a)
    N = next_power_of_two(n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    # Randomize phase but keep DC and conjugate symmetry
    new_phase = np.angle(A) + strength * rng.standard_normal(len(A))
    A_new = mag * np.exp(1j * new_phase)
    # Enforce conjugate symmetry (real signal)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    S = a_new.sum()
    if S > 1e-10:
        a_new = a_new / S
    else:
        a_new = a.copy()
    return a_new
