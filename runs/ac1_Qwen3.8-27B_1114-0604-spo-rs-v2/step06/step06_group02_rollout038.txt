LP-based active set optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linearized Chebyshev approximation with periodic L-BFGS polishing.
"""

from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

    else:

    return exact_sc, a_out

def solve_lp_direction(a, c, M, active_set, margin=1e-8):
    """
    Solve the LP to find a direction d that minimizes the linearized max
    of inactive peaks while keeping active peaks from increasing.
    Returns the direction d (not including the step size).
    """
    n = len(a)
    num_peaks = 2 * n - 1
    
    # Select which peaks to include in constraints
    # Include all active peaks + top 200 inactive peaks
    c_sorted_idx = np.argsort(c)[::-1]
    active_set = set(active_set)
    
    # Inactive peaks: top 200 that are not in active set
    inactive = [k for k in c_sorted_idx if k not in active_set][:200]
    
    # Build the constraint matrix
    # Variables: d[0..n-1], t
    # Total variables: n + 1
    
    # Inequality constraints:
    # For each peak k in (active + inactive): 2*conv(a,d)[k] - (t if inactive else 0) <= rhs[k]
    # For each j: -d[j] <= a[j]
    # Equality: sum d = 0
    
    peaks = list(active_set) + inactive
    num_ineq = len(peaks) + n  # peak constraints + non-negativity
    
    # Build A_ub (num_ineq x (n+1))
    # This is sparse, so let's build it as COO and convert
    
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    offset = 0
    for k in peaks:
        k = int(k)
        # conv(a, d)[k] = sum_j a[j]*d[k-j] for valid j
        # d[k-j] is variable at index (k-j) if 0 <= k-j < n
        # So the coefficient of d[j'] (where j' = k-j) is a[j]
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            j_prime = k - j  # index into d
            rows.append(offset)
            cols.append(j_prime)
            vals.append(2.0 * a[j])
        
        if k not in active_set:
            # Inactive: subtract t
            rows.append(offset)
            cols.append(n)  # t variable
            vals.append(-1.0)
            b_ub.append(-c[k])
        else:
            # Active: no t term
            b_ub.append(margin - c[k])
        offset += 1
    
    # Non-negativity: -d[j] <= a[j]
    for j in range(n):
        rows.append(offset)
        cols.append(j)
        vals.append(-1.0)
        b_ub.append(a[j])
        offset += 1
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_ineq, n + 1))
    
    # Equality: sum d[j] = 0
    A_eq = csr_matrix([np.array([1.0]*n + [0.0])], shape=(1, n+1))
    b_eq = [0.0]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: t >= 0 (or t >= c_min to make it meaningful)
    bounds = [(None, None)] * n + [(0.0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x[:n]
    else:
        return None

def fienup_adaptive(a, n_iters=500, deadline=None):

    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()
    eta = 0.1

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: score={sc:.8f}")

        # Pick a random peak from top 10
        top_idx = np.argsort(c)[::-1][:10]
        k_star = int(top_idx[rng.integers(0, len(top_idx))])

        # Subgradient: dC[k*]/da[j]
        d = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        for j in range(j_start, j_end + 1):
            kp = k_star - j
            if 0 <= kp < n:
                d[j] += a[kp]
            km = j - k_star
            if 0 <= km < n and j != kp:
                d[j] += a[km]

        # Normalize and apply
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            continue
        d = d / d_norm

        # Adaptive step size with backtracking
        best_alpha = 0.0
        best_sc_ls = sc
        for alpha in [eta, eta * 0.5, eta * 0.25, eta * 0.1]:
            a_try = a - alpha * d
            a_try = np.maximum(a_try, 1e-15)
            S_try = float(a_try.sum())
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < best_sc_ls:
                best_sc_ls = sc_try
                best_alpha = alpha

        if best_alpha > 0:
            a = a - best_alpha * d
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.7

        eta = max(eta, 0.005)

    return best_a


def pairwise_redistribution(a, n_sweeps=50, deadline=None, verbose=True):
    """Try pairwise mass redistribution to reduce peak convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    c = fast_conv(a)
    c_max = float(c.max())

    # Focus on elements that contribute to the top peak
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    active_range = list(range(j_start, j_end + 1))

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and sweep % 10 == 0:
                print(f"    PW sweep {sweep}: score={sc:.8f}")

        c_max = float(c.max())
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        active_range = list(range(j_start, j_end + 1))

        if not active_range:
            break

        # Try redistributing between pairs in the active range
        improved = False
        for idx_i in range(0, len(active_range), max(1, len(active_range) // 200)):
            if deadline and time.monotonic() > deadline:
                break
            i = active_range[idx_i]
            if a[i] < 1e-10:
                continue
            # Try moving mass from i to a nearby element
            for di in [-3, -2, -1, 1, 2, 3, -10, 10, -50, 50]:
                j = i + di
                if j < 0 or j >= n:
                    continue
                # Try several fractions of mass transfer
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if a[j] + delta > 1000 or a[i] - delta < 0:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    sc_try = compute_score(a)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta

        if not improved:
            break

    return best_a


def spectral_lp_interleave(a, deadline, budget_frac=1.0, verbose=True):
    """Interleave spectral (Fienup) and LP steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    phase = 0
    fienup_iter = 0
    lp_stall = 0

    while time.monotonic() < deadline:
        if phase % 3 == 0:
            # Fienup step
            A = np.fft.rfft(a, M_fft)
            P = A.real ** 2 + A.imag ** 2
            P_mean = float(P.mean())
            alpha = 0.3
            P_blend = (1 - alpha) * P + alpha * P_mean
            P_blend = np.maximum(P_blend, 1e-20)
            A_new = A * np.sqrt(P_blend / (P + 1e-30))
            a_new = np.fft.irfft(A_new, M_fft)[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new > 1e-12:
                a_new = a_new / S_new
                sc = compute_score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    a = a_new
                    lp_stall = 0
                a = a_new
            fienup_iter += 1
        elif phase % 3 == 1:
            # LP step
            c = fast_conv(a)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                lp_stall = 0
            else:
                lp_stall += 1

            M = float(c.max())
            active_set = set(int(k) for k in np.where(c >= 0.998 * M)[0])
            d = solve_lp_direction(a, c, M, active_set, max_inactive=150, margin=1e-8)

            if d is not None:
                best_alpha = 0.0
                best_sc_ls = sc
                for alpha in [0.3, 0.15, 0.075, 0.0375]:
                    a_try = a + alpha * d
                    a_try = np.maximum(a_try, 0.0)
                    S_try = float(a_try.sum())
                    if S_try < 1e-12:
                        continue
                    a_try = a_try / S_try
                    sc_try = compute_score(a_try)
                    if sc_try < best_sc_ls:
                        best_sc_ls = sc_try
                        best_alpha = alpha
                if best_alpha > 0:
                    a = a + best_alpha * d
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    lp_stall = 0
            else:
                lp_stall += 5

            if lp_stall > 20:
                # Reset with perturbation
                a = a * (1.0 + 0.01 * np.random.standard_normal(n))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                lp_stall = 0
        else:
            # Quick L-BFGS step
            sc, a_opt = run_lbfgs(a, 10000, maxiter=500, maxfun=10000)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                a = a_opt

        phase += 1

        if verbose and phase % 50 == 0:
            print(f"    SPL iter {phase}: best={best_sc:.8f} (t={time.monotonic():.0f})")

    return best_a
