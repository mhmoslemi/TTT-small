def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    """Compute the exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad_direct(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (unconstrained, for use with bounds)."""
    n = len(a)

    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)

    # Shift for numerical stability
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = c_max + float(np.log(exp_s.sum())) / beta
    L = log_sum

def run_lbfgs_exp(a0, beta, maxiter=5000, maxfun=30000):
    """Run L-BFGS-B on exponential parameterization."""

        sc, g = lse_score_grad_exp(b, beta)
        return sc, g
    
    res = minimize(
        wrapped, b0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    b_out = res.x
    a_out = np.exp(b_out - b_out.max())
    a_out = a_out / a_out.sum()
    return a_out

def peak_reduction(a0, n_rounds=50, top_k=20, reduction_rate=0.3):
    """Directly reduce top peaks by adjusting contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top-k peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            
            # Reduce the top-contributing elements
            n_reduce = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            
            for li in top_local:
                j = j_start + li
                a[j] *= (1.0 - reduction_rate)
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def build_inits(n, rng, parent):

        A[:M//2] = A[:M//2]

        for k in range(M // 2 + 1):
            A_real[k] = amp * np.exp(1j * rng.uniform(0, 2 * np.pi))
            if k < M - k and k > 0:
                A_real[M - k] = np.conj(A_real[k])
        A_real[0] = n
        if M % 2 == 0:
            A_real[M//2] = amp  # Nyquist is real
        a_trial = np.fft.ifft(A_real, M).real[:n]
        a_trial = np.maximum(a_trial, 0.0)
        S = a_trial.sum()
        if S > 1e-6:
            inits.append(a_trial / S)
    
    # Chirp-based (quadratic phase)

        A = np.zeros(M, dtype=complex)

def annealed_gs(a, n_iters=1000):
    """GS with simulated annealing: start with aggressive changes, cool down."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    current_sc = best_sc
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        # Annealing schedule: start hot, cool down
        T = 1.0 * (1.0 - progress) ** 2 + 0.01
        relax = 0.8 * T + 0.02
        
        A = np.fft.fft(a, M)
        A_mag = np.abs(A)
        
        # Target: flat amplitude
        target_mag = np.full(M, A_mag[1:].mean())
        target_mag[0] = A_mag[0]
        
        # Add noise proportional to temperature
        noise = rng.standard_normal(M) * T * 0.01 if hasattr(annealed_gs, 'rng') else 0
        target_mag = np.maximum(target_mag + noise, 1e-10)
        
        new_mag = A_mag ** (1.0 - relax) * target_mag ** relax
        new_phase = np.angle(A)
        A_new = new_mag * np.exp(1j * new_phase)
        
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc_new = compute_score(a_new)
        
        # Metropolis acceptance
        if sc_new < current_sc or np.random.random() < np.exp(-(sc_new - current_sc) / (T * 0.01)):
            a = a_new
            current_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
        # else: keep a unchanged
    
    return best_a
