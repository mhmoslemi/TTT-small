    grad_L_x = grad_L_b - b * (grad_L_b @ b)

    return score, grad_L_x

    """Minimize the single peak value using gradient descent with restarts."""
    b = normalize(b.copy())
    best_b = b.copy()
    best_sc = compute_score(b)
    
    step = 0.01
    decay = 0.999
    
    for it in range(n_iters):
        c = fast_conv(b)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        S = float(b.sum())
        sc = 2.0 * len(b) * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_b = b.copy()
        
        # Gradient of (b*b)_{k_star} w.r.t. b
        grad = np.zeros(len(b))
        for j in range(len(b)):
            other = k_star - j
            if 0 <= other < len(b):
                if other == j:
                    grad[j] = b[j]
                else:
                    grad[j] = 2.0 * b[other]
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Step
        b_new = b - step * grad / gn
        b_new = np.maximum(b_new, 0.0)
        b_new = b_new / b_new.sum()
        
        # Check if improvement
        sc_new = compute_score(b_new)
        if sc_new < sc:
            b = b_new
        else:
            step *= 0.9
        
        step *= decay
    
    return best_b

Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses: improved Fienup spectral flattening, extended LSE annealing with
Nesterov acceleration, focused subgradient peak shaving, and multi-scale
n exploration. Targets score below 1.503.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fienup_improved(a, n_iters=1000, mode='adaptive'):
    """Improved Fienup with adaptive spectral flattening."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if mode == 'adaptive':
            # Gradually increase flattening
            alpha = 0.05 + 0.9 * progress
            P_target = np.full_like(P, P_mean)
            P_new = (1.0 - alpha) * P + alpha * P_target
        elif mode == 'quantile':
            q = 0.60 + 0.35 * progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        elif mode == 'log':
            # Log-flattening: reduce dynamic range
            P_log = np.log(np.maximum(P, 1e-30))
            P_log_mean = P_log.mean()
            alpha = 0.1 + 0.5 * progress
            P_log_new = (1.0 - alpha) * P_log + alpha * P_log_mean
            P_new = np.exp(P_log_new)
        else:
            P_new = P

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_lse(a0, beta, n_iters=10000, alpha_init=0.005):
    """Nesterov-accelerated LSE descent with simplex projection."""
    n = len(a0)
    x = normalize(a0.copy())
    v = np.zeros(n)  # velocity
    best_a = x.copy()
    best_sc = compute_score(x)
    alpha = alpha_init

    for it in range(n_iters):
        t = it + 1
        # Nesterov lookahead
        momentum = (t - 1.0) / (t + 2.0)
        y = x + momentum * v

        # Compute gradient at y
        y_clipped = np.maximum(y, 0.0)
        s_y = y_clipped.sum()
        if s_y < 1e-15:
            break
        y_norm = y_clipped / s_y

        sc, grad = lse_score_and_grad(y_norm, beta)

        # Project gradient onto tangent space of simplex
        # (remove component in direction of ones)
        grad_mean = grad.mean()
        grad_tan = grad - grad_mean

        # Update
        v = momentum * v - alpha * grad_tan
        x = x + v

        # Project onto probability simplex (clip + normalize)
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            break
        x = x / s

        # Adaptive step size
        sc_x = compute_score(x)
        if sc_x < best_sc:
            best_sc = sc_x
            best_a = x.copy()
        else:
            alpha *= 0.95

        # Decay step size
        alpha *= 0.9999

    return best_a
