"""
Hybrid optimization for minimizing 2n*max(self_conv)/(sum)^2.
Methods: GS (multiple variants), LSE-LBFGS, ADMM, peak-targeted CG, multi-scale.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return 2.0 * n * L / S2, grad


def grad_conv_weighted(a, w):
    """Gradient of sum_k w_k * conv_k(a) w.r.t. a. Result is length n."""
    n = len(a)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    return 2.0 * conv_res


def gs_iterate(a, target, M, alpha):
    """Single GS iteration: interpolate between current spectrum and target spectrum."""
    A = np.fft.rfft(a, M)
    A_mag = np.maximum(np.abs(A), 1e-30)
    A_phase = np.angle(A)
    
    T = np.fft.rfft(target, M)
    T_mag = np.maximum(np.abs(T), 1e-30)
    T_phase = np.angle(T)
    
    log_mag = (1.0 - alpha) * np.log(A_mag) + alpha * np.log(T_mag)
    phase = (1.0 - alpha) * A_phase + alpha * T_phase
    
    A_new = np.exp(log_mag) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M)[:len(a)]
    return a_new


def gs_optimize(a0, n_iters, target_type='flat', alpha_init=0.5, alpha_final=0.05, clip='hard'):
    """GS optimization with various targets."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    # Build target
    if target_type == 'flat':
        target = np.zeros(M)
        target[:2*n-1] = 1.0 / (2*n-1)
    elif target_type == 'triangle':
        # Triangle-aware: slightly higher at center to account for overlap
        target = np.zeros(M)
        for k in range(2*n-1):
            overlap = min(k+1, n, 2*n-1-k)
            target[k] = 1.0 / overlap  # normalize by overlap count
        target /= target.sum()
    elif target_type == 'inverse_sqrt':
        target = np.zeros(M)
        for k in range(2*n-1):
            overlap = min(k+1, n, 2*n-1-k)
            target[k] = 1.0 / np.sqrt(overlap)
        target /= target.sum()
    
    T = np.fft.rfft(target, M)
    T_mag = np.maximum(np.abs(T), 1e-30)
    T_phase = np.angle(T)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-30)
        A_phase = np.angle(A)
        
        log_mag = (1.0 - alpha) * np.log(A_mag) + alpha * np.log(T_mag)
        phase = (1.0 - alpha) * A_phase + alpha * T_phase
        
        A_new = np.exp(log_mag) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        
        if clip == 'hard':
            a_new = np.maximum(a_new, 0.0)
        elif clip == 'soft':
            a_new = a_new * np.exp(-np.minimum(a_new, 0.0) * 5.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def admm_optimize(a0, n_iters, rho_init=1.0, rho_max=1e6, rho_growth=1.5, 
                  a_sub_iters=100, a_sub_lr=0.01):
    """ADMM for min t s.t. conv_k <= t, sum(a)=1, a>=0."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    # Initialize dual variables
    c = fast_conv(a)
    t = float(c.max())
    z = np.minimum(c, t).astype(np.float64)  # z_k = min(conv_k, t)
    
    rho = rho_init
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # a-update: minimize rho/2 * ||z - conv(a,a)||^2 s.t. sum(a)=1, a>=0
        # Use projected gradient descent
        a_sub = a.copy()
        lr = a_sub_lr / (1.0 + it * 0.01)
        for sub in range(a_sub_iters):
            c_sub = fast_conv(a_sub)
            diff = z - c_sub  # length 2n-1
            # Gradient of rho/2 * ||diff||^2 w.r.t. a
            # = -rho * gradient of conv(a,a) dotted with diff
            # = -rho * 2 * cross_corr(diff, a)
            grad = rho * 2.0 * grad_conv_weighted(a_sub, diff)
            # Also need to maintain sum(a) = 1
            # Projected gradient: subtract mean of grad to maintain sum
            grad_centered = grad - grad.mean()
            a_sub -= lr * grad_centered
            a_sub = np.maximum(a_sub, 0.0)
            S_sub = a_sub.sum()
            if S_sub < 1e-12:
                break
            a_sub = a_sub / S_sub
        
        a = a_sub
        
        # z-update: z_k = min(conv_k(a), t)
        c = fast_conv(a)
        z = np.minimum(c, t).astype(np.float64)
        
        # t-update: t = max(z)
        t = float(z.max())
        
        # Update rho
        # rho = min(rho_growth * rho, rho_max)
        if it % 10 == 0:
            rho = min(rho * 1.2, rho_max)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        return obj_func(x, *obj_params)
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def peak_reduction(a0, n_rounds=50, top_k=20, reduce_frac=0.1):
    """Reduce top peaks by scaling down contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end+1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            n_reduce = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            for li in top_local:
                j = j_start + li
                a[j] *= (1.0 - reduce_frac)
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def subgrad_optimize(a0, n_iters=5000, n_peaks=30, step_init=0.03):
    """Subgradient method targeting multiple peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end+1]
                sg[j_start:j_end+1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        a -= min(step, 0.05) * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
    
    return best_a
