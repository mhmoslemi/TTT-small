    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.asarray(a, dtype=np.float64).copy()
    S = a.sum()
    a = a / S
    
    for it in range(max_iter):
        # Compute power spectrum
        F = np.fft.fft(a, N)
        P = np.abs(F)**2
        
        # Flatten: P_new = P^alpha * P_mean^(1-alpha)
        P_mean = P.mean()
        P_flat = P**alpha * P_mean**(1-alpha)
        
        # Update magnitude, keep phase
        phase = np.angle(F)
        F_new = np.sqrt(P_flat) * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(F_new).real[:len(a)]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
    
    return a

def flat_spectrum_iteration(a, N, relax=0.7, max_iter=5000):
    """
    Iterate between flat-spectrum constraint and non-negativity + support constraint.
    a: non-negative sequence of length n
    N: FFT size (>= 2n-1)
    relax: relaxation parameter (0 < relax < 1) for the spectral update
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S  # normalize to sum 1
    
    a_padded = np.zeros(N, dtype=np.float64)
    a_padded[:n] = a
    
    for it in range(max_iter):
        # Forward FFT
        F = np.fft.fft(a_padded)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flat magnitude
        target_mag = S / np.sqrt(N)  # since sum |F|^2 = N * sum |a|^2 ... 
        # Actually, by Parseval: sum_k |F[k]|^2 = N * sum_k |a_padded[k]|^2
        # We want |F[k]| = constant = c for all k
        # Then sum_k |F[k]|^2 = N * c^2 = N * sum |a_padded|^2
        # So c = sqrt(sum |a_padded|^2)
        c = np.sqrt(np.sum(a_padded**2))
        
        # Relaxation: new_mag = mag^relax * c^(1-relax)
        # This gradually flattens the spectrum
        new_mag = mag**relax * c**(1-relax)
        # Avoid zeros
        new_mag = np.maximum(new_mag, 1e-30)
        
        F_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_full = np.fft.ifft(F_new).real
        
        # Apply time-domain constraints:
        # 1. Support: a_padded[i] = 0 for i >= n
        # 2. Non-negativity: a_padded[i] >= 0
        
        a_new = a_full[:n].copy()
        a_new = np.maximum(a_new, 0)
        
        # Renormalize to sum 1
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        a_padded = np.zeros(N)
        a_padded[:n] = a
    
    return a

"""
Combined spectral-flattening (Gerchberg-Saxton) and subgradient optimization
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
New components vs. previous: spectral iteration for initialization and
diversification, momentum subgradient, and improved LSE-LBFGS tuning.
"""

import numpy as np
import time
import math

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def compute_active_weights_conv(conv, top_k=10, eps_frac=0.01):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.1 * conv.size))
    if active.sum() > max_count:
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[top_idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s

def exp_grad_update(a, conv, top_k, eps_frac, eta):
    w = compute_active_weights_conv(conv, top_k=top_k, eps_frac=eps_frac)
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)

def spectral_flat_iteration(a, N, relax, max_iter, tol=1e-12, verbose=False):
    """
    Gerchberg-Saxton style iteration for flat power spectrum.
    Alternates between:
    - Spectral domain: move |FFT(a_padded)| toward flat
    - Time domain: project to non-negative, support [0, n-1]
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    F = np.fft.fft(a_padded)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Target flat magnitude
    c_target = np.sqrt(np.sum(a_padded**2))
    # Actually, by Parseval: sum |F|^2 = N * sum |a_padded|^2
    # If |F| is flat = c, then N*c^2 = N * sum(a_padded^2), so c = sqrt(sum(a_padded^2))
    
    for it in range(max_iter):
        # Update magnitude with relaxation
        # new_mag = mag^relax * c_target^(1-relax)
        # This is a geometric interpolation between current and target
        log_new_mag = relax * np.log(mag + 1e-300) + (1.0 - relax) * np.log(c_target + 1e-300)
        new_mag = np.exp(log_new_mag)
        
        F_new = new_mag * np.exp(1j * phase)
        a_full = np.fft.ifft(F_new).real
        
        a_new = a_full[:n].copy()
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Check convergence
        a_padded[:n] = a_new
        a_padded[n:] = 0.0
        
        F = np.fft.fft(a_padded)
        mag_new = np.abs(F)
        
        if np.max(np.abs(mag_new - mag)) < tol * np.max(mag):
            break
        
        mag = mag_new
        phase = np.angle(F)
        c_target = np.sqrt(np.sum(a_padded**2))
        
        if verbose and it % 500 == 0:
            sc = exact_score(a_new)
            print("    Spectral iter {}: score={:.9f}".format(it, sc))
    
    return a_new

def spectral_flat_iteration_v2(a, N, relax, max_iter, verbose=False):
    """
    Alternative spectral iteration using error-reduction approach.
    Instead of geometric interpolation, use additive error reduction.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    for it in range(max_iter):
        F = np.fft.fft(a_padded)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target flat magnitude
        c_target = np.sqrt(np.sum(a_padded**2))
        
        # Error: how far from flat
        error = mag - c_target
        
        # Reduce error: new_mag = mag - relax * error = (1-relax)*mag + relax*c_target
        new_mag = (1.0 - relax) * mag + relax * c_target
        new_mag = np.maximum(new_mag, 1e-30)
        
        F_new = new_mag * np.exp(1j * phase)
        a_full = np.fft.ifft(F_new).real
        
        a_new = a_full[:n].copy()
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        a_padded[:n] = a
        a_padded[n:] = 0.0
    
    return a

def lse_score_grad(a, beta):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad

def run_lbfgs(a, beta, deadline, maxfun=500):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a
    
    def fun(x):
        return lse_score_grad(x, beta)
    
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def active_set_optimize(a, deadline, verbose=True, tag=""):
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()
    
    top_k = 5
    eps_frac = 0.05
    eta = 0.3
    stagnant = 0
    max_stagnant = 30
    
    for phase in range(25):
        if time.monotonic() > deadline:
            break
        phase_dead = min(deadline, t0 + (deadline - t0) * (phase + 1) / 25.0 + 5.0)
        
        for it in range(200):
            if time.monotonic() > phase_dead - 2.0:
                break
            
            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))
            
            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta
            
            etas = [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0]
            for eta_try in etas:
                if eta_try < 0.001:
                    continue
                new_a = exp_grad_update(a, conv, top_k, eps_frac, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try
            
            if best_sc_iter < best_sc - 1e-10:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter
                eta = best_eta
                stagnant = 0
                if verbose and it % 50 == 0:
                    print("  [{} p{} i{}] score={:.9f} top_k={} eps={:.3f} eta={:.3f}".format(
                        tag, phase, it, best_sc, top_k, eps_frac, eta))
            else:
                stagnant += 1
                eta *= 0.85
                if stagnant >= max_stagnant:
                    break
        
        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.5) + 2, 150)
            eps_frac = min(eps_frac * 1.3, 0.2)
            eta = max(0.1, eta * 0.7)
            stagnant = 0
            if verbose:
                print("  [{}] expand: top_k={}, eps={:.3f}".format(tag, top_k, eps_frac))
        else:
            top_k = max(3, int(top_k * 0.95))
            eps_frac = max(0.005, eps_frac * 0.95)
    
    return float(best_sc), best_a

def subgrad_fine_polish(a, deadline, verbose=False, tag=""):
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 75, 100, 150]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0]
    eps_fracs = [0.0, 0.002, 0.005, 0.01, 0.02, 0.05]
    
    improved = True
    round_num = 0
    while improved and time.monotonic() < deadline and round_num < 10:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)
        for tk in top_ks:
            if time.monotonic() > deadline:
                break
            for ef in eps_fracs:
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    new_a = exp_grad_update(best_a, conv, tk, ef, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = new_a
                        improved = True
                        if verbose:
                            print("  [{} sg r{}] top_k={} eps={:.3f} eta={:.2f} -> {:.9f}".format(
                                tag, round_num, tk, ef, eta, best_sc))
    
    return float(best_sc), best_a

def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 10))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = compute_active_weights_conv(conv, top_k=10, eps_frac=0.02)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))

    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses combined spectral-flattening (GS iteration) and subgradient optimization.
    """
    t_start = time.monotonic()

    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    parent = None

            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)

    except Exception:
