def _lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    d = _self_conv(a)
    d = (float(n) * d) / (S * S)
    dmax = float(np.max(d))
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se
    L = dmax + np.log(se) / beta
    LDS = float(np.dot(w, d))
    h = _corr(a, w)
    score = 2.0 * L + 1e-3 * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0e-3 * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _run_lbfgs(a0, beta, deadline, maxfun):
    a = _normalize(a0)
    n = len(a)
    if time.monotonic() > deadline - 5:
        return _score(a), a

    def fun(x):
        return _lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-11, "maxiter": max(30, maxfun // 2),
            "maxfun": max(32, maxfun), "maxls": 40}
    try:

    sc = _score(a_out)

def _subgrad_step(a, w_active, eta):
    """Take one subgradient step with given weights on active lags."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    h = _corr(a, w_active)
    d = _self_conv(a)
    d_norm = (float(n) * d) / (S * S)
    LDS = float(np.dot(w_active, d_norm))
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, None)
    s2 = float(a_new.sum())
    if s2 < 1e-15:
        return a
    a_new = a_new * (S / s2)
    return a_new


def _active_weights(d, k, mode="top"):
    m = len(d)
    if m <= 0:
        return np.array([])
    if k <= 0 or k >= m:
        return np.ones(m) / m
    if mode == "top":
        idx = np.argpartition(d, m - k)[m - k:]
        w = np.zeros(m)
        w[idx] = 1.0 / k
        return w
    elif mode == "random":
        idx = np.random.choice(m, size=min(k, m), replace=False)
        w = np.zeros(m)
        w[idx] = 1.0 / k
        return w
    else:  # "pressure" - weight by squared deviation from mean
        d_mean = float(d.mean())
        d_var = np.maximum(d - d_mean, 0)
        d_var = d_var ** 2
        if d_var.sum() < 1e-30:
            return _active_weights(d, k, "top")
        probs = d_var / d_var.sum()
        idx = np.random.choice(m, size=min(k, m), replace=False, p=probs)
        w = np.zeros(m)
        w[idx] = 1.0 / k
        return w

            # Decay eta
            eta_base *= 0.7
            if eta_base < 1e-4:
                break

    return float(best_sc), best_a


def _mirror_descent_phase(a, deadline, max_iters=150):
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        d = _self_conv(a)
        S = float(a.sum())
        d_norm = (float(n) * d) / (S * S)
        improved = False
        for k in (1, 2, 3, 5, 8):
            if time.monotonic() > deadline:
                break
            w = _active_weights(d_norm, k, "top")
            h = _corr(a, w)
            h = np.maximum(h, 1e-15)
            for t in (0.02, 0.05, 0.1, 0.15, 0.25):
                log_a = np.log(np.maximum(a, 1e-300))
                log_h = np.log(h)
                log_new = (1 - t) * log_a + t * log_h
                a_try = _normalize(np.exp(log_new - np.max(log_new)))
                sc = _score(a_try)
                if np.isfinite(sc) and sc < best_sc - 1e-11:
                    best_sc = sc
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved and it > 5:
            break
    return float(best_sc), best_a


def _generate_inits(n, rng, parent, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = _resample(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-15:
            arr = np.ones(n)
        inits.append(_normalize(arr))

    if parent is not None and len(parent) > 0:
        add(_resample(parent, n))
        add(_resample(parent[::-1], n))

    # V-shapes
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Cosine
    for f in (1, 2, 3):
        for alpha in (0.3, 0.7, 1.2, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.cos(np.pi * x))

    # Bimodal peaks
    for w in (0.08, 0.12, 0.18, 0.25):
        add(np.exp(-(((x - 0.15) / w) ** 2)) + np.exp(-(((x - 0.85) / w) ** 2)) + 0.1)
        add(np.exp(-(((x - 0.3) / w) ** 2)) + np.exp(-(((x - 0.7) / w) ** 2)) + 0.1)

    # Exponential decay
    for r in (1.5, 3.0, 5.0, 8.0):
        add(np.exp(-r * x) + np.exp(-r * (1 - x)) + 0.05)

    # Sine
    for f in (1, 2, 3, 4, 6, 8):
        add(1.0 + 0.6 * np.abs(np.sin(np.pi * f * x)))

    # W-shaped (multi-peak)
    for npeaks in (3, 4, 5):
        centers = np.linspace(0.1, 0.9, npeaks)
        w = 0.15
        sig = sum(np.exp(-(((x - c) / w) ** 2)) for c in centers)
        add(sig + 0.05)

    # Random smooth (Fourier)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** rng.uniform(1.0, 3.5))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        add(1.0 + 0.8 * (noise - noise.mean() + 1.0))

    return inits


def _perturb(a, rng, strength=0.15):
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.05, 0.25)
        q = (1 - t) * a + t * (1.0 / n)
    elif mode == 2:
        q = a * (1.0 + strength * 0.5 * np.abs(2.0 * np.linspace(0, 1, n) - 1.0))
    elif mode == 3:
        # Smoothed noise
        logp = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        k = min(21, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * (c / 2.0) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = np.exp(logp + strength * 0.3 * noise - (logp + strength * 0.3 * noise).max())
    elif mode == 4:
        # Targeted reduction at peak-contributing positions
        d = _self_conv(a)
        dmax = float(np.max(d))
        if dmax > 1e-15:
            w = _active_weights(d, 3, "top")
            h = _corr(a, w)
            hmax = max(float(np.max(h)), 1e-15)
            q = a * np.exp(-strength * 0.5 * h / hmax)
        else:
            q = a.copy()
    else:
        q = a[::-1].copy() * np.exp(0.05 * rng.standard_normal(n))
    return _normalize(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {}, len={}".format(parent is not None, parent.size if parent is not None else 0))

    # Choose primary length
    n_prim = 10240
    if parent is not None:
        n_prim = parent.size
        n_prim = max(4096, min(n_prim, 20480))
        # Round to nice number
        if n_prim > 0:
            n_prim = int(round(n_prim / 512.0)) * 512
            n_prim = max(4096, min(n_prim, 20480))

    print("Using n = {}" .format(n_prim))

    # Initial point from parent
    if parent is not None:
        a0 = _normalize(_resample(parent, n_prim))
    else:
        x = np.linspace(0, 1, n_prim)
        a0 = _normalize(1.0 + 1.5 * np.abs(2.0 * x - 1.0))

    best_sc = _score(a0)
    best_seq = a0.copy()
    print("Initial score: {:.8f}".format(best_sc))

    # === Phase 1: LSE continuation (25% time) ===
    t1_end = t_start + 0.25 * total_budget
    betas = [500, 2000, 10000, 50000, 200000, 1000000]
    cum_frac = 0.0
    total_frac = len(betas)
    a_cur = a0.copy()
    for i, beta in enumerate(betas):
        stage_end = t_start + ((i + 1) / total_frac) * (t1_end - t_start)
        stage_end = min(stage_end, deadline - 5)
        while time.monotonic() < stage_end - 3:
            rem = stage_end - time.monotonic()
            if rem < 10:
                break
            est_t = 0.02 * max(0.5, n_prim / 8192.0)
            mf = int(min(2000, max(50, 0.6 * rem / est_t)))
            sc, a2 = _run_lbfgs(a_cur, beta, stage_end, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
            a_cur = _normalize(a2)
            if time.monotonic() > stage_end:
                break
        print("Phase 1 beta={:.0f}: best={:.8f}".format(beta, best_sc))

    # === Phase 2: Active-set subgradient (30% time) ===
    t2_end = t_start + 0.55 * total_budget
    a_work = _normalize(best_seq)
    sc, seq = _active_set_polish(a_work, t2_end, max_iters=500)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
    print("Phase 2 (subgrad): best={:.8f}".format(best_sc))

    # === Phase 3: Mirror descent (15% time) ===
    t3_end = t_start + 0.70 * total_budget
    sc, seq = _mirror_descent_phase(_normalize(best_seq), t3_end, max_iters=200)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
    print("Phase 3 (mirror): best={:.8f}".format(best_sc))

    # === Phase 4: High-beta LSE polish (10% time) ===
    t4_end = t_start + 0.80 * total_budget
    a_cur = _normalize(best_seq)
    for beta in (500000, 2000000, 5000000):
        if time.monotonic() > t4_end - 10:
            break
        rem = t4_end - time.monotonic()
        if rem < 15:
            break
        est_t = 0.02 * max(0.5, n_prim / 8192.0)
        mf = int(min(1500, max(50, 0.5 * rem / est_t)))
        sc, a2 = _run_lbfgs(a_cur, beta, t4_end, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
        a_cur = _normalize(a2)
        print("Phase 4 beta={:.0f}: best={:.8f}".format(beta, best_sc))

    # === Phase 5: Multi-length exploration (10% time) ===
    t5_end = t_start + 0.90 * total_budget
    other_ns = [8192, 12288, 16384]
    other_ns = [x for x in other_ns if x != n_prim]
    for n_alt in other_ns[:2]:
        if time.monotonic() > t5_end - 20:
            break
        a_alt = _normalize(_resample(best_seq, n_alt))
        for beta in (10000, 100000):
            if time.monotonic() > t5_end - 15:
                break
            rem = t5_end - time.monotonic()
            if rem < 10:
                break
            est_t = 0.02 * max(0.5, n_alt / 8192.0)
            mf = int(min(800, max(40, 0.5 * rem / est_t)))
            sc, a2 = _run_lbfgs(a_alt, beta, t5_end, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Phase 5 n={}: best={:.8f}".format(n_alt, best_sc))
            a_alt = _normalize(a2)
        # Quick subgradient on alt
        if time.monotonic() < t5_end - 10:
            sc, seq = _active_set_polish(a_alt, t5_end - 5, max_iters=50)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("Phase 5 subgrad n={}: best={:.8f}".format(n_alt, best_sc))

    # === Phase 6: Perturbation restarts (7% time) ===
    t6_end = t_start + 0.97 * total_budget
    n_perturb = 5
    for i in range(n_perturb):
        if time.monotonic() > t6_end - 20:
            break
        chunk_end = t6_end - (n_perturb - 1 - i) * 15.0
        p = _perturb(best_seq, rng, strength=rng.uniform(0.05, 0.3))
        # Quick LSE
        if time.monotonic() < chunk_end - 5:
            sc, a2 = _run_lbfgs(p, 100000, chunk_end, 200)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
            p = _normalize(a2)
        # Quick subgradient
        if time.monotonic() < chunk_end - 3:
            sc, seq = _active_set_polish(p, chunk_end, max_iters=30)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
        print("Perturb {}: best={:.8f}".format(i, best_sc))

    # === Phase 7: Final polish (3% time) ===
    if time.monotonic() < deadline - 5:
        sc, seq = _active_set_polish(_normalize(best_seq), deadline - 3, max_iters=30)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
        if time.monotonic() < deadline - 2:
            sc, seq = _mirror_descent_phase(_normalize(best_seq), deadline - 1, max_iters=20)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()

    # Final output preparation
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} t={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
