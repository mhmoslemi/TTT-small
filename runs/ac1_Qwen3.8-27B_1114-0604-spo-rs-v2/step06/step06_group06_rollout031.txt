    """
    Solve LP to find best descent direction.
    
    Minimize t
    Subject to:
      c[k] + 2 * (a*d)[k] <= t  for k in active positions
      sum(d) = 0
      d >= -a
    """

    d = cp.Variable(n)
    t = cp.Variable()

    # Only constrain at active positions
    active_idx = np.where(active_mask)[0]
    # (a*d)[k] = sum_j a[j] * d[k-j]
    # This is the cross-correlation of a and d at lag k
    # We can express this as a linear function of d
    
    # (a*d)[k] = sum_j a[j] * d[k-j]
    # For a fixed k, this is a linear form in d with coefficients a[j] where j is in valid range
    
    # Actually, let me think about this differently.
    # (a*d)[k] = sum_j a[j] * d[k-j]
    # Let l = k-j, so j = k-l
    # (a*d)[k] = sum_l a[k-l] * d[l]
    # This is the dot product of d with a reversed and shifted version.

    # For each active k, the coefficient of d[l] in (a*d)[k] is a[k-l] if 0 <= k-l < n, else 0.

    for idx, k in enumerate(active_idx):

        l_min = max(0, k - (n-1))
        l_max = min(k, n-1)
        A_mat[idx, l_min:l_max+1] = a[k-l_max:k-l_min:-1]

    c_active = c[active_idx]
    constraints.append(c_active + 2 * A_mat @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if d.value is not None:
        return d.value, t.value
    return None, None

"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LP-based direction finding, multiplicative weights, and LSE-LBFGS.
"""

import numpy as np
import time
import math
import cvxpy as cp
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_conv(a, d):
    """Compute linear cross-convolution a * d (same as np.convolve(a, d))."""
    n = len(a)
    m = len(d)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    D = np.fft.rfft(d, N)
    out = np.fft.irfft(A * D, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def_grade(a, active_positions):
    """
    Compute the subgradient of max(c[k]) for k in active_positions, w.r.t. a.
    Returns a vector g such that g[i] = dc[k]/da[i] for the active k.
    
    Actually, for the subgradient of max, we use a distribution over active points.
    Here we use uniform weight over active positions.
    """
    n = len(a)
    g = np.zeros(n)
    for k in active_positions:
        # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # First term: a[k-i] for i in [max(0,k-n+1), min(k,n-1)]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            g[i_start:i_end+1] += a[k - i_end:k - i_start:-1]
        # Second term: a[i-k] for i in [k, min(k+n-1, n-1)]
        i_start2 = k
        i_end2 = min(k + n - 1, n - 1)
        if i_start2 <= i_end2:
            g[i_start2:i_end2+1] += a[i_start2 - k:i_end2 - k + 1]
    return g

"""
Hybrid LP + multiplicative weights + LSE-LBFGS optimizer for minimum

import numpy as np

import math
from scipy.optimize import minimize, linprog


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def weighted_conv(a, w):
    """Compute 2 * sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    This is 2 * the gradient contribution of the active set w to the peak.
    Result has length n.
    """
    n = len(a)
    lw = len(w)
    if n == 0 or lw == 0:
        return np.zeros(n)
    a_rev = a[::-1]
    M = lw + n - 1
    N = next_power_of_two(M)
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a_rev, N)
    conv_full = np.fft.irfft(W * AR, N)[:M]
    # result[i] = conv_full[i + n - 1] for i = 0, ..., n-1
    result = conv_full[n-1:n-1+n]
    return 2.0 * result


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

    S = float(np.sum(a))

    c = fast_self_conv(a)

def get_active_weights(c, top_k=20, eps_frac=0.02):
    """Get weights for active positions (where c is near max)."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.ones(c.size) / c.size, None
    threshold = cmax * (1.0 - eps_frac)
    active = c >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(c, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, min(50, int(0.02 * c.size)))
    if active.sum() > max_count:
        top_idx = np.argpartition(c, -max_count)[-max_count:]
        active = np.zeros(c.size, dtype=bool)
        active[top_idx] = True
    # Weight proportional to excess over threshold
    w = np.zeros(c.size)
    w[active] = (c[active] - threshold * 0.5)  # some weighting
    sw = w.sum()
    if sw < 1e-15:
        w[active] = 1.0
        sw = active.sum()
    w /= sw

    """Multiplicative weights update using active set weights w."""

    gmax = float(np.max(g))

    log_a = np.log(a + 1e-300)
    log_a -= eta * g / gmax
    log_a -= float(np.max(log_a))
