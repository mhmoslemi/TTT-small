def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    """Compute full self-convolution using FFT."""
    n = a.size
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    """Compute the exact evaluation score."""
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size) / a.size
    return a / S

# Simplex projection (O(n) water-filling)
def _project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

OK this is getting complicated. Let me just carefully derive and code the gradient.

Let me define:
- a: length n, sum = 1, a >= 0
- c = conv(a, a): length 2n-1, c[k] = sum_i a[i]*a[k-i]
- score = 2*n*max(c) (since S=1)

For LSE smoothing:
- beta: smoothing parameter
- L = max(c) + (1/beta)*log(sum_k exp(beta*(c_k - max(c))))
- smoothed_score = 2*n*L

Gradient of L w.r.t. a[i]:
- w_k = exp(beta*(c_k - max(c))) / sum_j exp(beta*(c_j - max(c)))
- dL/da[i] = sum_k w_k * dc_k/da[i]
- dc_k/da[i] = 2*a[k-i] if max(0,k-n+1) <= i <= min(n-1,k), else 0

So: dL/da[i] = 2 * sum_k w_k * a[k-i] * I(max(0,k-n+1) <= i <= min(n-1,k))
            = 2 * sum_{k: i <= k <= i+n-1} w_k * a[k-i]  (as shown above)

Let R[i] = sum_{j=0}^{n-1} a[j] * w[i+j]  (where w has length 2n-1, indexed 0 to 2n-2)
Then: dL/da[i] = 2 * R[i]

And: d(smoothed_score)/da[i] = 2*n * 2 * R[i] = 4*n * R[i]

Let me verify with a simple example:
n=2, a=(0.5, 0.5)
c = [0.25, 0.5, 0.25], max(c) = 0.5, score = 2*2*0.5 = 2
If beta → ∞, w = [0, 1, 0] (only peak at k=1)
R[0] = a[0]*w[0+0] + a[1]*w[0+1] = 0.5*0 + 0.5*1 = 0.5
R[1] = a[0]*w[1+0] + a[1]*w[1+1] = 0.5*1 + 0.5*0 = 0.5
d(score)/da[0] = 4*2*0.5 = 4
d(score)/da[1] = 4*2*0.5 = 4

"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    S = float(w.sum())
    if S < 1e-15:
        return np.ones(n) / n
    return w / S


def _lse_value_grad(a, beta):
    """LSE-smoothed 2*max(n*c) and gradient, a on simplex."""
    n = a.size
    c = _fast_self_conv(a)
    d = n * c
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    # Gradient: dL/da[i] = 2 * R[i] where R[i] = sum_j a[j]*w[i+j]
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = 2.0 * n * R
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_full_active(a, tol_frac=1e-6, max_active=None):
    """
    Compute exact subgradient of max(c) at a using ALL active lags.
    Returns (max_c, gradient) where gradient is of max(c) w.r.t. a.
    a is on the simplex (sum=1).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fast_self_conv(a)
    dmax = float(c.max())

    # Find ALL active lags (within tol_frac of max)
    threshold = dmax - max(tol_frac * dmax, 1e-18)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

    if max_active is not None and len(active) > max_active:
        # Use top max_active
        scores = c[active]
        top_idx = np.argsort(scores)[-max_active:]
        active = active[top_idx]

    k_act = len(active)
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / k_act

    # Compute R[i] = sum_j a[j] * w_active[i+j] via FFT
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    # Subgradient of max(c): g[i] = 2*R[i]
    grad = 2.0 * R
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return dmax, grad


def _nesterov_subgrad(a0, n_iters, deadline, tol_frac=1e-6, max_active=None, verbose=False):
    """
    Nesterov accelerated projected subgradient for minimizing max(conv(a,a))
    on the simplex. a is kept on the simplex.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_c = float(_fast_self_conv(a).max())

    t = 1.0
    y = a.copy()
    prev_a = a.copy()

    # Initial step
    _, g0 = _subgrad_full_active(a, tol_frac, max_active)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-14:
        return best_c, best_a

    # Adaptive step size
    # Lipschitz constant estimate: L ~ 4n (for max(c) on the simplex)
    L_est = 4.0 * n
    gap_init = best_c - (1.0 / n)  # lower bound: max(c) >= 1/n is false, use 0
    gap_init = max(best_c - 0.0, 1e-6)
    alpha = np.sqrt(2.0) * np.sqrt(gap_init / L_est)
    eta = alpha

    # Polyak-like upper bound
    c_upper = best_c
    c_lower = 0.0

    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * (a - prev_a)
        prev_a = a.copy()

        # Ensure y is on simplex for evaluation
        y = _project_simplex(y)

        # Evaluate subgradient at y
        c_y, grad_y = _subgrad_full_active(y, tol_frac, max_active)
        if not np.isfinite(c_y):
            break

        # Update bounds
        if c_y < c_upper:
            c_upper = c_y
        c_lower = max(c_lower, 0.0)

        # Step size: combine 1/sqrt(t) with Polyak
        eta_t = alpha / np.sqrt(t + 1.0)
        # Polyak step
        gap = c_y - c_lower
        g_norm_sq = float(np.dot(grad_y, grad_y))
        if g_norm_sq > 1e-20:
            eta_polyak = gap * np.sqrt(n) / (n * g_norm_sq) * 2.0
            eta_t = max(eta_t, min(eta_polyak, eta * 3.0))
        eta_t = min(eta_t, eta * 5.0)

        # Gradient step + project
        x_new = y - eta_t * grad_y
        x_new = _project_simplex(x_new)

        a = x_new
        t = t_new

        # Track best
        c_a = float(_fast_self_conv(a).max())
        if c_a < best_c:
            best_c = c_a
            best_a = a.copy()
            if c_lower < 0:
                c_lower = best_c * 0.9999

        # Adaptive alpha: if no improvement in 200 iters, reduce
        if it > 0 and it % 200 == 0 and not (c_a < best_c + 1e-12):
            alpha *= 0.8

    return best_c, best_a
