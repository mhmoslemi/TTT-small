"""
Optimized subgradient + high-beta L-BFGS approach for minimizing
2n * max(conv(a,a)) / sum(a)^2 with non-negative constraint.
"""

import numpy as np
from scipy.optimize import minimize
import time

# --- Precomputed FFT sizes ---
# Will be set at runtime based on sequence length

def fft_size(n):
    M = 2 * n - 1
    return 1 << (M - 1).bit_length()

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = fft_size(n)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()  # Numerical safety

def subgradient_max(a, c, top_k=5):
    """Compute subgradient of max(c) using top-k peaks."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * 0.999  # Near-max peaks
    active = np.where(c >= threshold)[0]
    active = active[:top_k]  # Limit to top_k
    g = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            g[j] += a[k - j]
            if k == 2 * j:
                g[j] += a[j]
    g /= len(active)
    return g

    """
    Nesterov-like momentum subgradient method for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.

        return np.ones(n) / n

    alpha = alpha_init

        c = fast_conv(a_y)
        M_val = c.max()
        if M_val < best_M:
            best_M = M_val
            best_a = a_y.copy()
        
        g = subgradient_max(a_y, c, top_k=top_k)
        
        # Project gradient onto tangent space of simplex (remove mean)
        g_tan = g - g.mean()
        
        # Step
        a_new = a_y - alpha * g_tan
        a_new = project_simplex(a_new)

        # Actually for Nesterov: v = (momentum * (1 + alpha) / (1 + momentum * alpha)) * (a_new - a)
        # Simplified: v = momentum * (a_new - a)
        v = momentum * (a_new - a)
        a = a_new

    return best_a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient via FFT
    a_rev = a[::-1]
    Mf = fft_size(n)
    PA = np.fft.rfft(p, Mf)
    AA = np.fft.rfft(a_rev, Mf)
    h_full = np.fft.irfft(AA * PA, Mf)[:2 * n - 1]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

"""
Multi-strategy optimizer: subgradient momentum on exact max, adaptive multi-beta L-BFGS,
golden-section coordinate descent, Fienup spectral shaping, and stochastic restarts.
Targets score = 2n * max(conv(a,a)) / sum(a)^2 with a >= 0.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _fft_size(n):
    M = 2 * n - 1
    return 1 << (M - 1).bit_length()


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = _fft_size(n)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def subgradient_of_max(a, c, top_k=5):
    n = len(a)
    c_max = c.max()
    active = np.where(c >= c_max * 0.9995)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active = active[:top_k]
    g = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            g[j] += a[k - j]
            if k == 2 * j:
                g[j] += a[j]
    g /= float(len(active))
    return g


def run_subgradient_momentum(a, n_iters=3000, top_k=5, alpha_init=0.003, momentum=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_M = fast_conv(a).max()
    alpha = alpha_init
    v = np.zeros(n)

    for it in range(n_iters):
        a_y = project_simplex(a + v)
        c = fast_conv(a_y)
        M_val = float(c.max())
        if M_val < best_M:
            best_M = M_val
            best_a = a_y.copy()

        g = subgradient_of_max(a_y, c, top_k=top_k)
        g_tan = g - g.mean()
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-15:
            break

        a_new = a_y - (alpha / g_norm) * g_tan
        a_new = project_simplex(a_new)

        v = momentum * (a_new - a)
        a = a_new
        alpha = alpha_init / (1.0 + 0.005 * it)

    return best_a


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    Mf = _fft_size(n)
    PA = np.fft.rfft(p, Mf)
    AA = np.fft.rfft(a_rev, Mf)
    h_full = np.fft.irfft(AA * PA, Mf)[:2 * n - 1]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent_golden(a, n_sweeps=30):

    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a_j = a[j]
            if a_j < 1e-8:
                lo, hi = 0.0, max(0.1, 2.0 * a_j + 1e-3)
            else:
                lo, hi = 0.0, 3.0 * a_j
            best_val = a_j
            best_local = best_sc
            for _ in range(20):
                mid = (hi - lo) / (1.0 + np.sqrt(5.0))
                x1 = hi - mid
                x2 = lo + mid
                a[j] = x1
                t1 = a.sum()
                if t1 < 1e-12:
                    s1 = 1e10
                else:
                    s1 = compute_score(a / t1)
                a[j] = x2
                t2 = a.sum()
                if t2 < 1e-12:
                    s2 = 1e10
                else:
                    s2 = compute_score(a / t2)
                if s1 < s2:
                    hi = x2
                else:
                    lo = x1
                if s1 < best_local:
                    best_local = s1
                    best_val = x1
                if s2 < best_local:
                    best_local = s2
                    best_val = x2
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

def fienup_flat(a, n_iters=500):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    Mf = _fft_size(n)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, Mf)
        P = A.real ** 2 + A.imag ** 2
        alpha = 0.6 * (1.0 - progress) ** 2 + 0.02
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

            j_e = min(k_star, n - 1)
            a_p = best_a.copy()
            noise = 0.015 * rng.standard_normal(n)
            noise[j_s:j_e + 1] *= 3.0
            a_p = best_a * (1.0 + noise)
        elif strategy == 2:
            freq = rng.integers(1, 6)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) / n
            a_p = best_a * (1.0 + 0.025 * np.cos(2 * np.pi * freq * x + phase))
        else:
            # Block perturbation
            block_size = max(1, n // 15)
            block = rng.integers(0, n - block_size)
            a_p = best_a.copy()
            a_p[block:block + block_size] *= (1.0 + 0.08 * rng.standard_normal(block_size))

        a_p = project_simplex(a_p)
        sc, a_opt = run_lbfgs(a_p, 50000, maxiter=4000, maxfun=80000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  Restart {r}: {best_sc:.10f}")

    # === Phase 5: Final ultra-high-beta polish + CD ===
    print(f"\n=== Phase 5: Final polish ===")
    for beta in [5000000, 20000000]:
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - 15 - time.monotonic()
        if remaining < 10:
            break
        maxiter = int(max(2000, remaining * 5))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 15)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"  UltraBeta {beta}: {best_sc:.10f}")

    # Final tiny CD
    if time.monotonic() < deadline - 5:
        a_cd = coord_descent_golden(best_a, n_sweeps=10)
        sc_cd = compute_score(a_cd)
        if sc_cd < best_sc:
            best_sc = sc_cd
            best_a = a_cd.copy()
            print(f"  Final CD: {best_sc:.10f}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_sc:.10f}, verified={verified:.10f}")
    print(f"Length={len(out)}, Time={elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
