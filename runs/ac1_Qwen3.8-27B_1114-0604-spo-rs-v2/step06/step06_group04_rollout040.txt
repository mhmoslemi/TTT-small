def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, rel_tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - rel_tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, deadline, maxfun=None):

    n = a.size

        maxfun = 500
    bounds = [(0.0, None)] * n

        "gtol": 1e-10,

def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, rel_tol=1e-6):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    
    sc0, g0 = _subgrad_exact(a, k_active, rel_tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    
    eta = 0.1 / max(g_norm, 1e-10)
    no_improve = 0
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        
        score_y, grad_y = _subgrad_exact(y_norm, k_active, rel_tol)
        if not np.isfinite(score_y):
            break
        
        step = eta / (1.0 + 0.05 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        
        v = x_new - a
        a = x_new
        t = t_new

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 30:
                eta *= 0.5
                no_improve = 0
    
    return best_score, best_a


def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.02, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=5, rel_tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.5 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth noise
        K = min(31, max(5, n // 200))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.5)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.5 * noise)
    else:
        # High-frequency perturbation
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        # Emphasize high frequencies
        mask = np.ones(len(freqs))
        low_cutoff = int(n * 0.1 / Nf * len(freqs))
        mask[:low_cutoff] = 0.1
        power = mask / (np.abs(freqs) + 0.01)
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    
    return _normalize(np.clip(q, 0.0, None))

def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)

import numpy as np

from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, rel_tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax * (1.0 - rel_tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    a = _normalize(a0)
    n = a.size
    if maxfun is None:
        maxfun = 500

    def _fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(_fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = _normalize(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, rel_tol=1e-6):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    sc0, g0 = _subgrad_exact(a, k_active, rel_tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.08 / max(g_norm, 1e-10)
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_exact(y_norm, k_active, rel_tol)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.03 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 40:
                eta *= 0.6
                no_improve = 0
    return best_score, best_a


def _mirror_descent(a0, n_iters=3000, deadline=None, k_active=3, rel_tol=1e-6):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    sc0, g0 = _subgrad_exact(a, k_active, rel_tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 1.0 / max(g_norm, 1e-10)
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        _, g = _subgrad_exact(a, k_active, rel_tol)
        step = eta / (1.0 + 0.02 * it)
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                eta *= 0.5
                no_improve = 0
        a = a_new
    return best_score, best_a


def _peak_balance(a0, deadline, k_peaks=5):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    step = 0.03
    no_improve = 0
    for it in range(30000):
        if time.monotonic() > float(deadline) - 2.0:
            break
        c = _fast_self_conv(a)
        S = float(a.sum())
        d = n * c / (S * S)
        dmax = float(d.max())
        top_k = np.argsort(d)[-k_peaks:]
        peak_vals = d[top_k]
        mean_d = float(d.mean())
        excess = peak_vals - mean_d
        if np.all(excess < 1e-12):
            break
        w_peaks = np.zeros(2 * n - 1)
        for p, exc in zip(top_k, excess):
            if exc > 0:
                w_peaks[int(p)] = exc
        N2 = _next_pow2(3 * n)
        a_rev = a[::-1]
        AR = np.fft.rfft(a_rev, N2)
        WR = np.fft.rfft(w_peaks, N2)
        conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
        R = conv_full[n - 1:2 * n - 1]
        T = float(w_peaks.dot(c))
        grad = (n / (S * S)) * R - (n * T) / (S * S * S)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        a_trial = a - step * grad / g_norm
        a_trial = np.clip(a_trial, 0.0, None)
        S_trial = float(a_trial.sum())
        if S_trial < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial / S_trial
        sc_trial = _exact_score(a_trial)
        if np.isfinite(sc_trial) and sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            step = min(step * 1.01, 0.3)
        else:
            no_improve += 1
            if no_improve > 15:
                step *= 0.7
                no_improve = 0
            if step < 1e-8:
                break
    return best_score, best_a


def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.03, 0.25))
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.01, 0.1))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=5, rel_tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.5 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        K = min(31, max(5, n // 200))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.5)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.5 * noise)
    elif mode == 4:
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        mask = np.ones(len(freqs))
        low_cut = max(1, int(0.05 * n * len(freqs) / Nf))
        mask[:low_cut] = 0.1
        power = mask / (np.abs(freqs) + 0.01)
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    else:
        q = a.copy()
        n_swaps = int(rng.integers(2, max(3, n // 200)))
        for _ in range(n_swaps):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = a * np.exp(0.03 * rng.standard_normal(n))
    return _normalize(np.clip(q, 0.0, None))

        n_it = int(max(100, rem * 20))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                     k_active=3, rel_tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)
    return best_score, best_a


def _main_optimize(a0, deadline, label=""):
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    remaining = max(20.0, float(deadline) - t0)
    print("[{}] start: {:.8f} n={}".format(label, best_score, a.size))

    # Stage 1: Progressive LSE (35% of time)
    s1_end = t0 + 0.35 * remaining
    betas_s1 = [200, 1000, 5000, 20000, 100000]
    for beta in betas_s1:
        rem = s1_end - time.monotonic()
        if rem < 15.0:
            break
        maxfun = int(max(50, min(2000, rem * 6)))
        sc, a2 = _lbfgs_lse(a, beta, s1_end, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Stage 2: Peak balance (15%)
    s2_end = t0 + 0.50 * remaining
    if time.monotonic() < s2_end - 10.0:
        sc, a2 = _peak_balance(a, s2_end, k_peaks=5)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] PeakBalance: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    # Stage 3: Mirror descent (15%)
    s3_end = t0 + 0.65 * remaining
    if time.monotonic() < s3_end - 10.0:
        n_it = int(max(300, (s3_end - time.monotonic()) * 25))
        sc, a2 = _mirror_descent(a, n_iters=n_it, deadline=s3_end, k_active=3)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] MirrorDescent: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    # Stage 4: Nesterov with cycling k_active (35%)
    k_cycle = [5, 3, 2, 1, 2, 3, 1]
    for ki, k_act in enumerate(k_cycle):
        rem = float(deadline) - time.monotonic()
        if rem < 15.0:
            break
        n_it = int(max(200, rem * 12 / (len(k_cycle) - ki)))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                     k_active=k_act, rel_tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] Nesterov k={} (iter {}): {:.8f}".format(label, k_act, ki, best_score))
        a = _normalize(a2)

    # Final: Ultra-high beta LSE
    rem = float(deadline) - time.monotonic()
    if rem > 30.0:
        for beta in [500000, 2000000]:
            rem = float(deadline) - time.monotonic()
            if rem < 20.0:
                break
            maxfun = int(max(50, min(1000, rem * 4)))
            sc, a2 = _lbfgs_lse(a, beta, float(deadline) - 5.0, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] FinalLSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
            a = _normalize(a2)

    # Final Nesterov polish
    rem = float(deadline) - time.monotonic()
    if rem > 20.0:
        n_it = int(max(300, rem * 25))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                     k_active=1, rel_tol=1e-9)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] FinalNesterov k=1: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    rem = float(deadline) - time.monotonic()
    if rem > 15.0:
        n_it = int(max(200, rem * 25))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                     k_active=2, rel_tol=1e-9)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] FinalNesterov k=2: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-stage approach:
    - Progressive LSE-LBFGS for smooth descent
    - Peak balancing for targeted peak reduction
    - Mirror descent for non-smooth exploration
    - Nesterov with cycling k_active for fine polishing
    - Extensive perturbation restarts for global exploration
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    if parent is not None:
        best_score = _exact_score(parent)
        best_seq = _normalize(parent)
        print("Parent score: {:.8f}".format(best_score))
    else:
        best_score = np.inf
        best_seq = np.ones(8192) / 8192

    # Phase 1: Main optimization on parent (60% of time)
    main_end = t_start + 0.60 * (deadline - t_start)
    if parent is not None:
        sc, seq = _main_optimize(best_seq, main_end, "main")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After main: {:.8f}".format(best_score))
    else:
        sc, seq = _main_optimize(best_seq, main_end, "noinit")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Phase 2: Perturbation restarts (30% of time)
    perturb_start = max(main_end, time.monotonic())
    perturb_end = t_start + 0.90 * (deadline - t_start)
    n_perturb = 8
    if perturb_end > perturb_start + 60.0:
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 5.0:
                break
            # Vary perturbation strength
            if pi < 3:
                p = _perturb(best_seq, rng, strength=0.08)
            elif pi < 6:
                p = _perturb(best_seq, rng, strength=0.15)
            else:
                p = _perturb(best_seq, rng, strength=0.25)
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 3: Final intensive polish (10% of time)
    final_dead = deadline - 10.0
    if time.monotonic() < final_dead - 30.0:
        # High beta LSE sweep
        for beta in [1000000, 3000000, 10000000]:
            rem = final_dead - time.monotonic()
            if rem < 20.0:
                break
            maxfun = int(max(50, min(800, rem * 4)))
            sc, a2 = _lbfgs_lse(best_seq, beta, final_dead - 5.0, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("FinalLSE beta={:.0f}: {:.8f}".format(beta, best_score))
            best_seq = _normalize(a2)

        # Peak balance with more peaks
        rem = final_dead - time.monotonic()
        if rem > 30.0:
            sc, a2 = _peak_balance(best_seq, final_dead - 5.0, k_peaks=7)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("FinalPeakBalance: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Ultra-tight Nesterov k=1
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            n_it = int(max(300, rem * 25))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=1, rel_tol=1e-10)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("FinalNest k=1: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Nesterov k=2
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            n_it = int(max(200, rem * 25))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=2, rel_tol=1e-10)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("FinalNest k=2: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # One more mirror descent pass
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            n_it = int(max(200, rem * 20))
            sc, a2 = _mirror_descent(best_seq, n_iters=n_it, deadline=final_dead,
                                     k_active=2, rel_tol=1e-8)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("FinalMirror: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
