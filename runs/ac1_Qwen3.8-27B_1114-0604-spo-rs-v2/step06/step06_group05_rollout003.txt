import time


def fast_conv(a):

    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())

    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    S2 = S * S
    S3 = S2 * S

"""
Spectral Fienup + Adaptive LSE + Active-set gradient + Peak targeting
to minimize score(a) = 2n * max(conv(a,a)) / sum(a)^2

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def active_peak_score_grad(a, top_p):
    """Score and gradient using top-p peaks with weighted sum."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    
    # Top-p peaks
    sorted_idx = np.argsort(c)[::-1][:top_p]
    c_top = c[sorted_idx]
    
    # Use uniform weighting over top-p
    w_full = np.zeros(2 * n - 1)
    w_full[sorted_idx] = 1.0 / top_p
    
    w_num = float(np.dot(w_full, c))
    score = 2.0 * n * w_num / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def spectral_fienup(a0, n_iters=500, relax=0.3):
    """Iterative amplitude projection in frequency domain."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-12)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Target: flat amplitude spectrum
    # |A_target(k)| = constant for all k
    # The flat amplitude is sqrt(M * mean(|A|²)) = sqrt(M * sum(a²)/M * M/M) ...
    # Actually: sum_k |A(k)|² = M * sum_j a_j² (Parseval)
    # Flat: |A(k)| = sqrt(M * sum(a²)/M) = sqrt(sum(a²)) for all k
    a_energy = float(np.sum(a**2))
    flat_amp = np.sqrt(a_energy)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Relax toward flat amplitude
        new_amp = amp ** (1 - relax) * flat_amp ** relax
        # Or: new_amp = (1-relax)*amp + relax*flat_amp
        
        A_new = new_amp * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Update flat amplitude target
        a_energy = float(np.sum(a_new**2))
        flat_amp = np.sqrt(a_energy)
    
    return best_a


def gradient_descent_active(a0, n_iters=5000, top_p=5, lr_init=0.02, decay=0.9999):
    """Gradient descent using active-set (top-p) objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-12)
    
    v = np.zeros(n)
    momentum = 0.85
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        sc, grad = active_peak_score_grad(a, top_p)
        
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = a_new = a / max(a.sum(), 1e-12)
            break
        a_new = a_new / S_new
        a = a_new
        lr *= decay
        
        if it % 20 == 0 or it < 3:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def peak_reduction(a0, n_rounds=50, top_k=20, reduction=0.9):
    """Directly reduce top peaks by scaling down their contributors."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / max(a.sum(), 1e-12)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            
            n_reduce = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            
            for li in top_local:
                j = j_start + li
                a[j] *= reduction
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def random_perturb(a0, rng, strength=0.05):
