```
            # But if k-j = j (i.e., k=2j), these are the same term, so change = 2*delta*a[j]
            # Actually the formula delta*(a[k-j] + a[j-k]) already handles this correctly
            # because when k=2j, a[k-j] = a[j] and a[j-k] = a[j], giving 2*delta*a[j]
            
            if 0 <= idx1 < n:
                c[k] += delta * a[idx1]
            if 0 <= idx2 < n:
                c[k] += delta * a[idx2]

        S += delta
        new_score = 2*n*max(c) / (S*S)

            score = new_score

        else:

            # undo
            for k in range(2*n-1):

                if 0 <= idx2 < n:
                    c[k] -= delta * a[idx2]
            a[j] -= delta
            S -= delta
```

Wait, there's a subtlety. When I update `c[k]`, I'm using the OLD values of `a` (before the change to `a[j]`). So the update formula `c[k] += delta * (a[k-j] + a[j-k])` is correct using the old `a`. But when I undo, I need to use the same old `a` values.

Actually, the issue is simpler: the change in `c[k]` is `delta * (a_old[k-j] + a_old[j-k])`, where `a_old` is the value before the step. Since only `a[j]` changes, and we're computing the change based on the old values, the formula is:

`delta_c[k] = delta * (a_old[k-j] + a_old[j-k])`

where `a_old` is the sequence before the change. This is correct regardless of whether `k-j` or `j-k` equals `j` (the index being changed), because we're using the OLD value.

`c[k] = sum_i a[i] * a[k-i]`

The terms involving `a[j]` are:

Great, the formula is correct.

Now, for efficiency, note that `delta_c[k] = delta * (a[k-j] + a[j-k])` is a "shifted" version of `a`. Specifically:
- `a[k-j]` for `k` in `[j, j+n-1]` is just `a[0], a[1], ..., a[n-1]`
- `a[j-k]` for `k` in `[0, j]` is `a[j], a[j-1], ..., a[0]`

So the update is:
- For `k = j` to `j+n-1`: `c[k] += delta * a[k-j]`
- For `k = 0` to `j`: `c[k] += delta * a[j-k]`

This is just adding a scaled, shifted copy of `a` to `c`. In numpy, this can be done with slice assignments:

This is `O(n)` and uses numpy's vectorized operations, so it's very fast.

```python
"""
Hybrid optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses fast O(n) local search with incremental convolution updates,
frequency-domain spectrum flattening, and L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def fast_local_search(a, n_steps=500000, time_limit=None, rng=None, 
                       initial_step=None, verbose=True):
    """
    Fast local search using O(n) incremental convolution updates.
    a: non-negative sequence (will be normalized to sum 1)
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S
    S = 1.0
    
    c = self_conv(a)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    best_a = a.copy()
    best_score = score
    best_c_max = c_max
    
    if initial_step is None:
        initial_step = 0.5 / n  # half the mean value
    
    step = initial_step
    k_star = int(np.argmax(c))
    
    t_start = time.monotonic()
    accepted = 0
    rejected = 0
    
    for it in range(n_steps):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        if verbose and it % 100000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FLS iter {it}: score={score:.8f}, step={step:.2e}, "
                  f"accept_rate={accepted/(accepted+rejected+1e-10):.3f}, t={elapsed:.0f}s")
        
        # Choose j: biased towards positions that affect k_star
        # Random uniform is a good default
        j = rng.integers(0, n)
        
        # Determine direction based on effect on k_star
        idx1 = k_star - j
        idx2 = j - k_star
        influence = 0.0
        if 0 <= idx1 < n:
            influence += a[idx1]
        if 0 <= idx2 < n:
            influence += a[idx2]
        
        if influence > 0:
            direction = -1  # decrease a[j] to reduce c[k_star]
        elif influence < 0:
            direction = 1  # increase a[j] (though this is unusual for non-negative)
        else:
            direction = 1 if a[j] < 1.0/n else -1  # move towards mean
        
        delta = direction * step
        
        # Bound check
        new_aj = a[j] + delta
        if new_aj < 0:
            delta = -a[j] * 0.5  # reduce by half
            new_aj = a[j] + delta
            if new_aj < 0:
                continue  # skip this move
        
        # Compute c update (using OLD a values)
        # c[j:j+n] += delta * a
        # c[:j+1] += delta * a[j::-1]
        
        save1 = c[j:j+n].copy()  # need to save for potential undo
        save2 = c[:j+1].copy()
        
        c[j:j+n] += delta * a
        c[:j+1] += delta * a[j::-1]
        
        # Update score
        new_c_max = float(c.max())
        new_S = S + delta
        if new_S < 1e-12:
            # undo
            c[j:j+n] = save1
            c[:j+1] = save2
            continue
        new_score = 2.0 * n * new_c_max / (new_S * new_S)
        
        if new_score < score:
            # Accept
            a[j] = new_aj
            S = new_S
            score = new_score
            c_max = new_c_max
            k_star = int(np.argmax(c))
            accepted += 1
            # Adapt step size: increase slightly
            step = min(step * 1.01, 0.05 / n)
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_c_max = new_c_max
        else:
            # Reject
            c[j:j+n] = save1
            c[:j+1] = save2
            rejected += 1
            # Adapt step size: decrease
            step = max(step * 0.99, 1e-8 / n)
        
        # Periodic recompute to prevent drift
        if it % 50000 == 0 and it > 0:
            c = self_conv(a)
            c_max = float(c.max())
            score = 2.0 * n * c_max / (S * S)
            k_star = int(np.argmax(c))
    
    if verbose:
        print(f"    FLS done: best_score={best_score:.8f}, "
              f"accepts={accepted}, rejects={rejected}")
    
    return best_a, best_score
```

```python
def spectrum_flatten(a, n_iters=100, target_level=None, verbose=True):
    """Flatten the power spectrum of a while maintaining non-negativity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        power = A * np.conj(A)  # |A|^2, shape (M_fft//2 + 1,)
        # Real signal: power is symmetric, but rfft only gives positive freqs
        
        # Target: flat power spectrum
        # The total power is sum(|A|^2) / M_fft = S^2 = 1 (by Parseval, with appropriate normalization)
        # Actually, for rfft: sum_{k=0}^{M_fft/2} |A_k|^2 (with appropriate weights) = M_fft * S^2 / M_fft = S^2
        
        # The mean power per frequency bin (accounting for the DC and Nyquist) is:
        # For a real signal of length M_fft: E[|A_k|^2] = M_fft * S^2 / M_fft = S^2
        # But the DC bin (k=0) has |A_0|^2 = S^2 exactly.
        # And the other bins should have mean power = S^2 on average.
        
        # Actually, let me think more carefully.
        # a has length n, padded to M_fft for the FFT.
        # sum(a) = 1, so A[0] = 1.
        # By Parseval: sum_{k=0}^{M_fft-1} |A_k|^2 = M_fft * sum(a^2)
        # For rfft: |A_0|^2 + 2*sum_{k=1}^{M_fft/2-1} |A_k|^2 + |A_{M_fft/2}|^2 = M_fft * sum(a^2)
        # (assuming M_fft is even)
        
        mean_power = M_fft * float(np.sum(a*a)) / M_fft  # = sum(a^2)
        # This is the mean power per bin (approximately)
        
        # Actually, I want the convolution to be flat, which means |A|^2 should be flat.
        # The convolution is ifft(|A|^2). For it to be flat (all values = 1/(2n-1)),
        # we need |A|^2 to be a constant.
        # sum(conv) = 1, and conv has 2n-1 values, so flat conv = 1/(2n-1) per value.
        # ifft(constant) = delta at 0. That's not right.
        
        # Let me reconsider. The circular autocorrelation of the zero-padded signal is:
        # r[m] = sum_k a[k] * a[k+m mod M]  (circular)
        # For zero-padded signal (length M >= 2n-1), the circular autocorrelation equals
        # the linear autocorrelation for |m| < n.
        
        # The DFT relationship: r = ifft(|A|^2) (with appropriate scaling)
        # For flat r (r[m] = 1/M for all m), we need |A|^2 = constant.
        # But r[0] = sum(a^2) which is >= 1/n > 1/M (for M = 2n-1 > n).
        # So we can't have perfectly flat r. The best we can do is minimize the peak.
        
        # Strategy: pull |A|^2 towards its mean value.
        # |A|^2_target[k] = alpha * |A|^2[k] + (1-alpha) * mean(|A|^2[k])
        # where mean is computed over all frequency bins.
        
        if target_level is None:
            # Mean power (excluding DC which is fixed at S^2 = 1)
            n_bins = len(power)
            total_power = float(np.sum(power))  # This includes DC
            # For rfft of real signal: total circular energy = MFFT * sum(a^2)
            # But we want to flatten the non-DC part
            mean_non_dc = (total_power - float(power[0])) / (n_bins - 1)
        else:
            mean_non_dc = target_level
        
        # Adjust power spectrum (keep phase)
        phase = A / np.abs(A + 1e-30)
        new_power = power.copy()
        # Pull towards mean (only for non-DC bins)
        for k in range(1, n_bins):
            new_power[k] = 0.7 * power[k] + 0.3 * mean_non_dc
        # Keep DC fixed
        new_power[0] = power[0]
        
        # Reconstruct: A_new = phase * sqrt(new_power)
        A_new = phase * np.sqrt(new_power)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

```python
def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

    S = 1.0

            total_window += 1

                    step = min(step * 1.05, 0.5 * mean_a)

                    step = max(step * 0.95, 1e-6 * mean_a)
                accepted = 0
                rejected = 0
                total_window = 0
```

```python
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fast_local_search(a, time_limit, rng, verbose=True):
    """
    Fast local search using O(n) incremental convolution updates.
    Single-element moves with adaptive step size targeting the current peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S
    S = 1.0

    c = self_conv(a)
    c_max = float(c.max())
    score = 2.0 * n * c_max
    best_a = a.copy()
    best_score = score

    mean_a = 1.0 / n
    step = 0.15 * mean_a
    k_star = int(np.argmax(c))

    t_start = time.monotonic()
    accepted = 0
    rejected = 0

    for it in range(100000000):
        if time.monotonic() > time_limit:
            break
        if verbose and it > 0 and it % 500000 == 0:
            elapsed = time.monotonic() - t_start
            ar = accepted / (accepted + rejected + 1e-10)
            print(f"    FLS {it//1000000}M: score={score:.8f}, step={step:.1e}, "
                  f"ar={ar:.3f}, t={elapsed:.0f}s")

        j = rng.integers(0, n)

        idx1 = k_star - j
        idx2 = j - k_star
        influence = 0.0
        if 0 <= idx1 < n:
            influence += a[idx1]
        if 0 <= idx2 < n:
            influence += a[idx2]

        if influence > 1e-15:
            direction = -1
        elif influence < -1e-15:
            direction = 1
        else:
            direction = -1 if a[j] > mean_a else 1

        delta = direction * step
        new_aj = a[j] + delta
        if new_aj < 0:
            delta = -a[j] * 0.5
            new_aj = a[j] + delta
            if new_aj < 0:
                continue

        c[j:j + n] += delta * a
        c[:j + 1] += delta * a[j::-1]

        new_S = S + delta
        if new_S < 1e-12:
            c[j:j + n] -= delta * a
            c[:j + 1] -= delta * a[j::-1]
            continue

        new_c_max = float(c.max())
        new_score = 2.0 * n * new_c_max / (new_S * new_S)

        if new_score < score:
            a[j] = new_aj
            S = new_S
            score = new_score
            k_star = int(np.argmax(c))
            accepted += 1
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            c[j:j + n] -= delta * a
            c[:j + 1] -= delta * a[j::-1]
            rejected += 1

        if (accepted + rejected) % 2000 == 0:
            ar = accepted / (accepted + rejected)
            if ar > 0.65:
                step = min(step * 1.04, 0.5 * mean_a)
            elif ar < 0.25:
                step = max(step * 0.95, 1e-7 * mean_a)
            accepted = 0
            rejected = 0

        if (it + 1) % 50000 == 0:
            c = self_conv(a)
            S = float(a.sum())
            c_max = float(c.max())
            score = 2.0 * n * c_max / (S * S)
            k_star = int(np.argmax(c))
            if score < best_score:
                best_score = score
                best_a = a.copy()

    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    FLS done: best={best_score:.8f}, t={elapsed:.0f}s")
    return best_a, best_score


def pairwise_local_search(a, time_limit, rng, verbose=True):
    """
    Pairwise moves: transfer mass between two positions, keeping sum constant.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S
    S = 1.0

    c = self_conv(a)
    c_max = float(c.max())
    score = 2.0 * n * c_max
    best_a = a.copy()
    best_score = score

    mean_a = 1.0 / n
    step = 0.1 * mean_a
    k_star = int(np.argmax(c))

    t_start = time.monotonic()
    accepted = 0
    rejected = 0

    for it in range(100000000):
        if time.monotonic() > time_limit:
            break
        if verbose and it > 0 and it % 500000 == 0:
            elapsed = time.monotonic() - t_start
            ar = accepted / (accepted + rejected + 1e-10)
            print(f"    PLS {it//1000000}M: score={score:.8f}, step={step:.1e}, "
                  f"ar={ar:.3f}, t={elapsed:.0f}s")

        j1 = rng.integers(0, n)
        j2 = rng.integers(0, n)
        if j1 == j2:
            continue

        inf1 = 0.0
        t1a = k_star - j1
        t1b = j1 - k_star
        if 0 <= t1a < n:
            inf1 += a[t1a]
        if 0 <= t1b < n:
            inf1 += a[t1b]

        inf2 = 0.0
        t2a = k_star - j2
        t2b = j2 - k_star
        if 0 <= t2a < n:
            inf2 += a[t2a]
        if 0 <= t2b < n:
            inf2 += a[t2b]

        if inf1 < inf2:
            j_inc, j_dec = j1, j2
        else:
            j_inc, j_dec = j2, j1

        d = step
        if a[j_dec] < d:
            d = a[j_dec] * 0.8
            if d < 1e-14:
                continue

        c[j_inc:j_inc + n] += d * a
        c[:j_inc + 1] += d * a[j_inc::-1]
        c[j_dec:j_dec + n] -= d * a
        c[:j_dec + 1] -= d * a[j_dec::-1]

        new_c_max = float(c.max())
        new_score = 2.0 * n * new_c_max

        if new_score < score:
            a[j_inc] += d
            a[j_dec] -= d
            score = new_score
            k_star = int(np.argmax(c))
            accepted += 1
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            c[j_inc:j_inc + n] -= d * a
            c[:j_inc + 1] -= d * a[j_inc::-1]
            c[j_dec:j_dec + n] += d * a
            c[:j_dec + 1] += d * a[j_dec::-1]
            rejected += 1

        if (accepted + rejected) % 2000 == 0:
            ar = accepted / (accepted + rejected)
            if ar > 0.6:
                step = min(step * 1.03, 0.4 * mean_a)
            elif ar < 0.25:
                step = max(step * 0.96, 1e-7 * mean_a)
            accepted = 0
            rejected = 0

        if (it + 1) % 50000 == 0:
            c = self_conv(a)
            c_max = float(c.max())
            score = 2.0 * n * c_max
            k_star = int(np.argmax(c))
            if score < best_score:
                best_score = score
                best_a = a.copy()

    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    PLS done: best={best_score:.8f}, t={elapsed:.0f}s")
    return best_a, best_score


def spectrum_flatten(a, n_iters=300, mix_rate=0.25, verbose=True):
    """
    Gerchberg-Saxton style: alternate between flattening |A|^2 in frequency
    domain and enforcing non-negativity in time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if verbose and it % 50 == 0:
            print(f"    SF iter {it}: score={best_score:.8f}")

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        n_bins = len(A)
        total_power = float(np.sum(mag * mag))
        non_dc_power = total_power - float(mag[0] ** 2)
        mean_non_dc = non_dc_power / max(1, n_bins - 1)

        new_power = mag * mag
        for k in range(1, n_bins):
            new_power[k] = (1.0 - mix_rate) * new_power[k] + mix_rate * mean_non_dc
        new_power[0] = mag[0] ** 2

        new_mag = np.sqrt(new_power)
        A_new = phase * new_mag
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def weighted_peak_search(a, time_limit, rng, verbose=True):
    """
    Local search that targets multiple peaks simultaneously using weighted
    subgradient. More effective than targeting only the single max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    a = a / S
    S = 1.0

    c = self_conv(a)
    c_max = float(c.max())
    score = 2.0 * n * c_max
    best_a = a.copy()
    best_score = score

    mean_c = 1.0 / (2 * n - 1)
    mean_a = 1.0 / n
    step = 0.1 * mean_a
    k_star = int(np.argmax(c))

    # Compute weighted potential: positions above threshold get weight
    threshold = mean_c + 0.3 * (c_max - mean_c)
    weights = np.maximum(c - threshold, 0.0)
    w_sum = float(weights.sum())
    if w_sum < 1e-15:
        weights = np.zeros_like(c)
        weights[k_star] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum

    t_start = time.monotonic()
    accepted = 0
    rejected = 0

    for it in range(100000000):
        if time.monotonic() > time_limit:
            break
        if verbose and it > 0 and it % 500000 == 0:
            elapsed = time.monotonic() - t_start
            ar = accepted / (accepted + rejected + 1e-10)
            print(f"    WPS {it//1000000}M: score={score:.8f}, step={step:.1e}, "
                  f"ar={ar:.3f}, t={elapsed:.0f}s")

        j = rng.integers(0, n)

        # Compute weighted influence on c
        # dc[k]/da[j] = a[k-j] + a[j-k] (for valid indices)
        # Weighted: sum_k weights[k] * dc[k]/da[j]
        # This equals: sum_k weights[k] * (a[k-j] + a[j-k])
        # = sum_i weights[i+j] * a[i] + sum_i weights[j-i] * a[i] (with bounds)
        # For efficiency, just compute the unweighted influence to k_star
        # and use it as a proxy (faster)
        idx1 = k_star - j
        idx2 = j - k_star
        influence = 0.0
        if 0 <= idx1 < n:
            influence += a[idx1]
        if 0 <= idx2 < n:
            influence += a[idx2]

        if influence > 1e-15:
            direction = -1
        elif influence < -1e-15:
            direction = 1
        else:
            direction = -1 if a[j] > mean_a else 1

        delta = direction * step
        new_aj = a[j] + delta
        if new_aj < 0:
            delta = -a[j] * 0.5
            new_aj = a[j] + delta
            if new_aj < 0:
                continue

        c[j:j + n] += delta * a
        c[:j + 1] += delta * a[j::-1]

        new_S = S + delta
        if new_S < 1e-12:
            c[j:j + n] -= delta * a
            c[:j + 1] -= delta * a[j::-1]
            continue

        new_c_max = float(c.max())
        new_score = 2.0 * n * new_c_max / (new_S * new_S)

        if new_score < score:
            a[j] = new_aj
            S = new_S
            score = new_score
            k_star = int(np.argmax(c))
            accepted += 1
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            c[j:j + n] -= delta * a
            c[:j + 1] -= delta * a[j::-1]
            rejected += 1

        if (accepted + rejected) % 2000 == 0:
            ar = accepted / (accepted + rejected)
            if ar > 0.65:
                step = min(step * 1.04, 0.5 * mean_a)
            elif ar < 0.25:
                step = max(step * 0.95, 1e-7 * mean_a)
            accepted = 0
            rejected = 0

        # Update weights periodically
        if (it + 1) % 20000 == 0:
            c = self_conv(a)
            S = float(a.sum())
            c_max = float(c.max())
            score = 2.0 * n * c_max / (S * S)
            k_star = int(np.argmax(c))
            if score < best_score:
                best_score = score
                best_a = a.copy()
            mean_c = 1.0 / (2 * n - 1)
            threshold = mean_c + 0.3 * (c_max - mean_c)
            weights = np.maximum(c - threshold, 0.0)
            w_sum = float(weights.sum())
            if w_sum > 1e-15:
                weights /= w_sum

    elapsed = time.monotonic() - t_start
    if verbose:
        print(f"    WPS done: best={best_score:.8f}, t={elapsed:.0f}s")
    return best_a, best_score


def perturb_sequence(a, rng, scale=0.05):
    """Apply a smooth random perturbation to escape local minima."""
    n = len(a)
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 20))
    a_pert = a + scale * smooth * a
    a_pert = np.maximum(a_pert, 0.0)
    S = float(a_pert.sum())
    if S < 1e-12:
        return a.copy()
    return a_pert / S


def build_initialization(n, rng, parent=None):
    """Generate a diverse initialization of length n."""
    if parent is not None:
        if len(parent) == n:
            return parent.astype(np.float64).copy()
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        return np.interp(x_new, x_old, parent).copy()

    x = np.arange(n, dtype=np.float64) / n
    choices = [
        np.ones(n) * 0.5,
        0.5 + 0.3 * np.cos(2.0 * np.pi * x),
        np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0),
        0.3 + 0.4 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x),
        np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
    ]
    a = choices[rng.integers(0, len(choices))]
    return np.maximum(a, 0.01)


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: fast O(n) local search, pairwise search, spectrum flattening,
    perturbation/restart, and L-BFGS polish.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")

    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        try_update(ps, parent, "parent")

    if parent is not None:
        n_main = len(parent)
        a_main = parent.astype(np.float64).copy()
    else:
        n_main = 6962
        a_main = build_initialization(n_main, rng)

    # ==================================================================
    # Phase 1: Fast local search (35% time)
    # ==================================================================
    t_phase1 = 0.35 * (float(budget_s) - 30.0)
    phase1_deadline = t_start + t_phase1
    print(f"\n--- Phase 1: Fast Local Search (budget={t_phase1:.0f}s) ---")

    a_work = a_main.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Sub-phase 1a: Single-move FLS
    t_sub1 = phase1_deadline - 0.4 * t_phase1
    print(f"  1a: Single-move FLS ({t_sub1 - t_start:.0f}s budget)")
    a_out, sc_out = fast_local_search(a_work, t_sub1, rng, verbose=True)
    try_update(sc_out, a_out, "FLS-1a")
    a_work = a_out.copy()

    # Sub-phase 1b: Pairwise search
    print(f"  1b: Pairwise search ({phase1_deadline - time.monotonic():.0f}s budget)")
    a_out, sc_out = pairwise_local_search(a_work, phase1_deadline - 5, rng, verbose=True)
    try_update(sc_out, a_out, "PLS-1b")
    a_work = a_out.copy()

    # Sub-phase 1c: Another round of FLS
    if time.monotonic() < phase1_deadline - 30:
        print(f"  1c: FLS round 2 ({phase1_deadline - time.monotonic():.0f}s budget)")
        a_out, sc_out = fast_local_search(a_work, phase1_deadline - 5, rng, verbose=True)
        try_update(sc_out, a_out, "FLS-1c")
        a_work = a_out.copy()

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Perturbation + re-optimize (25% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Perturbation & Re-optimize ---")

    n_perturbations = 4
    pert_time = max(30.0, (phase2_deadline - time.monotonic()) / n_perturbations - 10)

    for p in range(n_perturbations):
        if time.monotonic() > phase2_deadline - 30:
            break
        print(f"  Perturbation {p+1}/{n_perturbations} ({pert_time:.0f}s each)")

        # Different perturbation scales
        scales = [0.03, 0.08, 0.15, 0.25]
        a_pert = perturb_sequence(best_a, rng, scale=scales[p % len(scales)])

        # Quick FLS on perturbed
        a_out, sc_out = fast_local_search(a_pert, time.monotonic() + pert_time, rng,
                                          verbose=False)
        try_update(sc_out, a_out, f"PERT-{p+1}")
        if sc_out < best_score:
            a_work = a_out.copy()

    # Also try a completely fresh initialization
    if time.monotonic() < phase2_deadline - 60:
        print(f"  Fresh init + FLS")
        a_fresh = build_initialization(n_main, rng, parent=best_a)
        a_out, sc_out = fast_local_search(a_fresh, phase2_deadline - 10, rng,
                                          verbose=False)
        try_update(sc_out, a_out, "FRESH")

    # Weighted peak search on current best
    if time.monotonic() < phase2_deadline - 30:
        print(f"  Weighted peak search")
        a_out, sc_out = weighted_peak_search(best_a, phase2_deadline - 5, rng,
                                             verbose=False)
        try_update(sc_out, a_out, "WPS")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Spectrum flattening (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: Spectrum Flattening ---")

    if time.monotonic() < phase3_deadline - 30:
        n_sf_iters = max(100, int((phase3_deadline - time.monotonic()) / 0.001))
        n_sf_iters = min(n_sf_iters, 500)
        print(f"  Spectrum flatten ({n_sf_iters} iters, mix=0.25)")
        a_sf, sc_sf = spectrum_flatten(best_a, n_iters=n_sf_iters, mix_rate=0.25,
                                       verbose=True)
        try_update(sc_sf, a_sf, "SPECTRUM")

        # Polish with FLS after spectrum flatten
        if time.monotonic() < phase3_deadline - 20:
            a_out, sc_out = fast_local_search(a_sf, phase3_deadline - 5, rng,
                                              verbose=False)
            try_update(sc_out, a_out, "SF+FLS")

        # Try different mixing rates
        if time.monotonic() < phase3_deadline - 30:
            a_sf2, sc_sf2 = spectrum_flatten(best_a, n_iters=100, mix_rate=0.5,
                                             verbose=False)
            try_update(sc_sf2, a_sf2, "SPECTRUM-m0.5")
            if time.monotonic() < phase3_deadline - 15:
                a_out, sc_out = fast_local_search(a_sf2, phase3_deadline - 5, rng,
                                                  verbose=False)
                try_update(sc_out, a_out, "SF2+FLS")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS polish (12% time)
    # ==================================================================
    phase4_deadline = t_start + 0.92 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 4: L-BFGS Polish ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [200, 500, 1000, 2000, 4000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=3000, maxfun=100000,
                                  gtol=1e-13)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, t={elapsed:.1f}s")
        try_update(sc, a_opt, f"LBFGS-b{beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final FLS polish (8% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    if time.monotonic() < deadline - 20:
        final_budget = deadline - time.monotonic() - 20
        print(f"  Final FLS ({final_budget:.0f}s)")
        a_out, sc_out = fast_local_search(best_a, deadline - 20, rng, verbose=True)
        try_update(sc_out, a_out, "FINAL-FLS")

        # One more pairwise on the result
        if time.monotonic() < deadline - 15:
            a_out, sc_out = pairwise_local_search(best_a, deadline - 10, rng,
                                                  verbose=False)
            try_update(sc_out, a_out, "FINAL-PLS")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
```
