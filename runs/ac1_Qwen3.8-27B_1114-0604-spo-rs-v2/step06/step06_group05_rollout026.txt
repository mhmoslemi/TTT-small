Fourier-smoothed restarts, and periodic block perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def topk_softmax_grad(a, K, beta):
    """Compute score and gradient using top-K softmax approximation to max."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    # Find top-K indices
    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        # Use partition for efficiency
        top_idx = np.argpartition(c, -K)[-K:]
    
    c_top = c[top_idx]
    c_max = float(c_top.max())
    
    if beta * (c_max - c_top.min()) > 700:  # prevent overflow
        beta = 700.0 / max(c_max - c_top.min(), 1e-30)
    
    scaled = beta * (c_top - c_max)
    exp_s = np.exp(scaled)
    p_top = exp_s / exp_s.sum()
    
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Build full p vector (sparse: only non-zero at top_idx)
    p = np.zeros(len(c))
    p[top_idx] = p_top
    
    # Gradient of L w.r.t. a:
    # dL/da_i = (1/beta) * sum_k p_k * beta * dc_k/da_i = sum_k p_k * 2*a[k-i]
    # = 2 * (rev_a convolved with p) at position i + n - 1
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res
    
    # d(score)/da_i = 2n * (g_L_i * S - 2 * L) / S^3
    S3 = S * S * S
    grad = (2.0 * n) * (g_L * S - 2.0 * L) / S3
    
    return score, grad

For a problem like this (minimizing the max of many functions), L-BFGS might not be the best because the problem is essentially a "max of linear functions" (in the lifted space). A **cutting plane** or **bundle method** might be more appropriate.

But implementing a bundle method from scratch is complex. Let me try a simpler approach:

**Method: Iterative peak reduction with restart**

1. Compute `c = conv(a,a)`.
2. Find the global maximum `c_max` and its location `k_max`.
3. The elements that contribute to `c[k_max]` are `a[j] * a[k_max - j]` for valid j.
4. To reduce `c[k_max]`, we should reduce the elements `a[j]` that are part of large products.
5. Specifically, reduce `a[j]` proportional to `a[k_max - j]` (the "partner" element).
6. But we also need to maintain the sum, so we increase other elements.

This is essentially a "water-filling" approach: reduce the peak contributors and increase the valley contributors.

Let me formalize:
- `c[k_max] = sum_j a[j] * a[k_max - j]`
- To reduce this, decrease `a[j]` for j where `a[j] * a[k_max-j]` is large.
- The "damage" of reducing `a[j]` by `delta_j` is `delta_j * a[k_max-j]` (approximately, for small delta).
- We want to choose `delta_j` to minimize the new max of the convolution.

This is a one-step problem that's hard to solve exactly, but we can use a greedy approach:
- Identify the top contributing pairs `(j, k_max-j)`.
- Reduce both elements (or the larger one) slightly.
- Check if the max decreased. If yes, keep the change. If no, revert.

This is essentially a **coordinate descent** on the non-negativity-simplex, where each "coordinate" is a pair of elements.

"""
Optimization using top-K peak softmax gradient with adaptive step sizes,
Fourier-smoothed restarts, block perturbations, and high-beta LSE polishing.
Implements focused subgradient descent on the simplex with Barzilai-Borwein
step sizes and periodic diversification to escape local minima.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_softmax_score_grad(a, K, beta):
    """Score and gradient using top-K softmax approximation to max."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -K)[-K:]

    c_top = c[top_idx]
    c_max = float(c_top.max())
    c_min = float(c_top.min())

    range_topk = c_max - c_min
    if range_topk > 1e-30:
        eff_beta = min(beta, 500.0 / range_topk)
    else:
        eff_beta = beta

    scaled = eff_beta * (c_top - c_max)
    exp_s = np.exp(scaled)
    p_top = exp_s / exp_s.sum()
    L = c_max + float(np.log(exp_s.sum())) / eff_beta
    score = 2.0 * n * L / (S * S)

    p = np.zeros(len(c))
    p[top_idx] = p_top

    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S3 = S * S * S
    grad = (2.0 * n) * (g_L * S - 2.0 * L) / S3
    return score, grad


def lse_score_grad(a, beta):
    """Full LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    range_c = c_max - float(c.min())
    if range_c > 1e-30:
        eff_beta = min(beta, 500.0 / range_c)
    else:
        eff_beta = beta
    scaled = eff_beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / eff_beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S3 = S * S * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def focused_descent(a0, K_list, beta, n_iters, use_bb=True, step_init=0.003):
    """Focused gradient descent cycling through K values with BB step sizes."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    a_prev = a.copy()
    g_prev = None
    step = step_init
    stall_count = 0

    for it in range(n_iters):
        K = K_list[it % len(K_list)]
        sc, grad = topk_softmax_score_grad(a, K, beta)

        grad_proj = grad - grad.mean()

        if use_bb and g_prev is not None:
            s_vec = a - a_prev
            y_vec = grad_proj - g_prev
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            if yy > 1e-30:
                if sy > 1e-15:
                    step = sy / yy
                else:
                    ss = float(np.dot(s_vec, s_vec))
                    step = ss / max(abs(sy), 1e-30)
                step = float(np.clip(step, 1e-10, 0.3))

        alpha = step
        success = False
        best_trial_a = a.copy()
        best_trial_sc = sc

        for ls in range(25):
            a_try = a - alpha * grad_proj
            a_try = np.maximum(a_try, 0.0)
            st = float(a_try.sum())
            if st < 1e-12:
                alpha *= 0.5
                continue
            a_try = a_try / st
            sc_try = compute_score(a_try)
            if sc_try < best_trial_sc:
                best_trial_sc = sc_try
                best_trial_a = a_try.copy()
            if sc_try < sc * (1.0 - 1e-14):
                success = True
                a = a_try
                sc = sc_try
                break
            alpha *= 0.5

        if not success:
            if best_trial_sc < sc:
                a = best_trial_a
                sc = best_trial_sc
            stall_count += 1
            if stall_count > 200:
                break
        else:
            stall_count = 0

        a_prev = a.copy()
        g_prev = grad_proj.copy()

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=30000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(wrapped, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-15,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def fourier_perturb(a0, rng, num_harmonics=20):
    """Perturb in Fourier domain: random phase on low frequencies."""
    n = len(a0)
    a = a0.copy()
    A = np.fft.rfft(a)
    magnitudes = np.abs(A)
    phases = np.angle(A)
    for k in range(1, min(num_harmonics, len(A))):
        phases[k] += rng.standard_normal() * 0.05
    A_new = magnitudes * np.exp(1j * phases)
    a_new = np.fft.irfft(A_new, n)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a0.copy()
    return a_new / S


def block_shuffle_perturb(a0, rng, n_blocks=10):
    """Shuffle blocks of the sequence."""
    n = len(a0)
    block_size = n // n_blocks
    blocks = []
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        blocks.append(a0[start:end])
    perm = rng.permutation(n_blocks)
    a_new = np.concatenate([blocks[i] for i in perm])
    return a_new


def gaussian_smooth(a0, sigma):
    """Apply Gaussian smoothing in the spatial domain."""
    n = len(a0)
    x = np.arange(n)
    kernel_width = int(6 * sigma)
    if kernel_width < 1:
        return a0.copy()
    kx = np.arange(-kernel_width, kernel_width + 1)
    kernel = np.exp(-0.5 * (kx / sigma) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a0, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    S = a_smooth.sum()
    if S < 1e-12:
        return a0.copy()
    return a_smooth / S


def peak_pair_reduction(a0, n_rounds=200, n_peaks=10, reduce_frac=0.1):
    """Directly reduce top peak-contributing pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            top_local = np.argsort(contrib)[::-1][:5]
            for li in top_local:
                j = j_start + li
                a[j] *= (1.0 - reduce_frac)

        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def cyclic_peak_descent(a0, n_iters=5000, n_peaks=5, step_init=0.01, step_decay=0.99995):
    """Cyclically target each of the top peaks in turn."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        k_target = top_idx[it % n_peaks]
        k_target = int(k_target)

        j_start = max(0, k_target - n + 1)
        j_end = min(k_target, n - 1)
        if j_end < j_start:
            continue
        seg = a[j_start:j_end + 1]
        partner = a[k_target - j_end:k_target - j_start + 1][::-1]
        contrib = seg * partner
        contrib_norm = contrib / (contrib.max() + 1e-30)

        S = float(a.sum())
        S2 = S * S
        S3 = S2 * S
        c_max = float(c.max())

        sg = np.zeros(n)
        sg[j_start:j_end + 1] += 2.0 * partner

        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        grad_proj = grad - grad.mean()
        grad_proj[j_start:j_end + 1] -= contrib_norm * 0.5

        a = a - step * grad_proj
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= step_decay

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a

        a_opt4 = focused_descent(best_a, [1, 3, 5], 10000.0, n_iters=10000, step_init=0.003)
        sc4 = compute_score(a_opt4)
        update_best(sc4, a_opt4)

    print(f"After Phase 4: best = {best_score:.8f}")

    # Phase 5: Final polish - alternate strategies (15% of time)
    print(f"\n=== Phase 5: Final polish (deadline +{deadline - t_start:.0f}s) ===")

    polish_rounds = 8
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # Focused descent with very high beta
        a_p = focused_descent(best_a, [1, 2, 3], 50000.0, n_iters=5000, step_init=0.002)
        sc = compute_score(a_p)
        update_best(sc, a_p)

        if time.monotonic() > deadline:
            break

        # LSE polish
        beta_polish = 100000 + 50000 * pr
        a_l2 = run_lbfgs_lse(best_a, beta_polish, maxiter=3000, maxfun=15000)
        sc2 = compute_score(a_l2)
        update_best(sc2, a_l2)

        if time.monotonic() > deadline:
            break

        # Small perturbation + focused descent
        a_pert = fourier_perturb(best_a, rng, num_harmonics=5)
        a_p2 = focused_descent(a_pert, [1, 3, 5, 10], 20000.0, n_iters=3000, step_init=0.002)
        sc3 = compute_score(a_p2)
        update_best(sc3, a_p2)

        print(f"  Polish round {pr}: best = {best_score:.8f}")

    # Final high-precision polish if time remains
    if time.monotonic() < deadline - 60:
        print(f"\n=== Final precision polish ===")
        a_final = focused_descent(best_a, [1], 100000.0, n_iters=5000, step_init=0.001)
        sc = compute_score(a_final)
        update_best(sc, a_final)

        a_final2 = run_lbfgs_lse(best_a, 500000, maxiter=5000, maxfun=30000)
        sc2 = compute_score(a_final2)
        update_best(sc2, a_final2)

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
