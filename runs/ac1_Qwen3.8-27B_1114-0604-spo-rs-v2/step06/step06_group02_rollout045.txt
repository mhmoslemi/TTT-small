"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE-LBFGS, LP-based direction finding, peak equalization,
and diverse restarts with improved scheduling.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000, time_limit=None, t_start=None):
    """Run L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    
    def objective(x):
        return lse_score_and_grad(x, beta)[0]
    
    def objective_grad(x):
        return lse_score_and_grad(x, beta)
    
    bounds = [(0.0, 1000.0)] * n
    opts = {'ftol': 1e-20, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    
    if time_limit is not None and t_start is not None:
        # Add time check via maxiter estimation
        pass
    
    res = minimize(
        objective_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds, options=opts
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_flat(a, n_iters=300, alpha_start=0.9, alpha_end=0.05):
    """Error reduction (Fienup) targeting flat spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        alpha = alpha_start * (1 - t) + alpha_end * t
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_target = P_mean  # flat target
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def peak_equalization_lp(a, M_target=None, max_iters=50):
    """
    LP-based peak equalization.
    At each step, solve an LP to find a direction that reduces the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(max_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find active peaks (top fraction)
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # LP: maximize sum(b) subject to:
        #   b >= 0
        #   conv(a, b)[k] <= c_max for k in active
        #   sum(b) <= 2 * S (to keep it reasonable)
        #
        # conv(a, b)[k] = sum_j a[j] * b[k-j]
        # This is a linear constraint in b.
        
        # Build the constraint matrix
        # For each active peak k: sum_j a[j] * b[k-j] <= c_max
        # The coefficient of b[m] in constraint k is: a[k-m] (if 0 <= k-m < n)
        
        m = len(active)
        A_ub = np.zeros((m, n))
        for i, k in enumerate(active):
            k = int(k)
            for j in range(n):
                b_idx = k - j
                if 0 <= b_idx < n:
                    A_ub[i, b_idx] += a[j]
        
        # This is O(n * len(active)) which could be slow for large n
        # Use FFT for efficiency: conv(a, b) for variable b
        # Actually, let's just use the matrix form for moderate sizes
        
        b_ub = np.full(m, c_max)
        
        # Also add: sum(b) <= 2*S as an extra constraint
        A_ub_extra = np.ones((1, n))
        b_ub_extra = np.array([2.0 * S])
        
        A_ub = np.vstack([A_ub, A_ub_extra])
        b_ub = np.concatenate([b_ub, b_ub_extra])
        
        # Objective: maximize sum(b) => minimize -sum(b)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        S_b = float(b.sum())
        if S_b < 1e-12:
            break
        
        # Update: a_new = (1-t)*a + t*b/S_b * S (normalize b to have same sum)
        t = 0.3  # mixing parameter
        b_norm = b / S_b * S
        a_new = (1 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        else:
            # Reduce mixing if it doesn't help
            t *= 0.7
            if t < 0.01:
                break
    
    return best_a


def peak_equalization_direct(a, n_iters=2000, top_k=5):
    """
    Direct peak equalization using gradient of the max.
    Focus on the top-k peaks and try to equalize them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        k_top = np.argsort(c)[::-1][:top_k]
        c_top = c[k_top]
        
        # Compute combined gradient from top peaks (equal weight)
        # grad = sum_k (1/|S|) * 2 * shift(a, k)
        grad = np.zeros(n)
        for k in k_top:
            k = int(k)
            # dc_k/da[m] = 2*a[k-m] (if 0 <= k-m < n)
            # This is 2 * a shifted by k
            valid_m = range(max(0, k - n + 1), min(k, n - 1) + 1)
            # Actually: dc_k/da[m] = 2*a[k-m] for 0 <= k-m < n, i.e., 0 <= m <= k (if k < n) 
            # or k-n+1 <= m <= n-1 (if k >= n)
            m_start = max(0, k - (n - 1))
            m_end = min(k, n - 1)
            if m_end < m_start:
                continue
            grad[m_start:m_end+1] += 2.0 * a[k - m_start:k - m_start + (m_end - m_start + 1)][::-1]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size
        eta = 0.1 / (grad_norm / S + 1e-10)
        
        # Update
        a_new = a - eta * grad / S
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        # Check if it helped
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.01:
            # Backtrack
            eta *= 0.5
            a_new = a - eta * grad / S
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
        
        a = a_new
    
    return best_a


def coordinate_descent(a, n_sweeps=10, top_peaks=5):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = c.max()
        
        # Focus on elements that contribute most to the top peaks
        top_k = np.argsort(c)[::-1][:top_peaks]
        affected = set()
        for k in top_k:
            k = int(k)
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            affected.update(range(m_start, m_end + 1))
        
        # Sort affected indices by contribution to peaks
        if len(affected) > 200:
            # Only optimize the most impactful ones
            contribs = np.zeros(n)
            for k in top_k:
                k = int(k)
                m_start = max(0, k - n + 1)
                m_end = min(k, n - 1)
                for m in range(m_start, m_end + 1):
                    contribs[m] += c[k] * a[m] * (a[k-m] if 0 <= k-m < n else 0)
            top_indices = np.argsort(contribs)[::-1][:200]
            indices = top_indices
        else:
            indices = np.array(list(affected))
        
        for m in indices:
            m = int(m)
            if a[m] < 1e-12:
                continue
            a_old = a[m]
            best_val = a_old
            best_local = best_sc
            
            # Try several multipliers
            for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[m] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[m] = a_old
                    continue
                a_tmp = a / S_tmp
                sc_tmp = compute_score(a_tmp)
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_val = a_old * f
            
            a[m] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def mulberry32(seed):

def diverse_restart(a, rng, strategy):
    """Create a diverse restart from the current best."""
    n = len(a)
    if strategy == 0:
        # Gaussian noise

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: adaptive LSE-beta with convergence detection,
gradient-flow method for high-beta regime, LP-based direction finding,

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    bounds = [(0.0, None)] * n
    opts = {'ftol': 1e-18, 'gtol': 1e-12, 'maxiter': maxiter,
            'maxfun': maxfun, 'maxls': 50}
    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds, options=opts)
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 500:
            beta *= 1.5
            eta = max(eta * 0.5, 0.001)
            stall_count = 0
            if eta < 0.0005:
                break

        # Compute LSE gradient for smoothness
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta

        a_rev = a[::-1]
        PA = np.fft.rfft(p, M_fft)
        AA_rev = np.fft.rfft(a_rev, M_fft)
        h_full = np.fft.irfft(AA_rev * PA, M_fft)[:2 * n - 1]
        h = h_full[n - 1:2 * n - 1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

        # Precondition: scale gradient by 1/a to get multiplicative update direction
        # This helps with non-negativity and scale invariance
        a_safe = np.maximum(a, 1e-10)
        grad_mul = grad / a_safe

        # Step with backtracking
        for attempt in range(10):
            a_new = a * np.exp(-eta * grad_mul)
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                eta *= 0.5
                continue
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new <= sc * 1.001:
                a = a_new
                eta = min(eta * 1.02, 0.5)
                break
            else:
                eta *= 0.5
        else:
            if eta < 1e-6:
                break

    return best_a


def fienup_flat(a, n_iters=200):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        alpha = 0.8 * (1 - t) ** 2 + 0.05
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lp_direction(a, max_peaks=10):
    """
    LP-based direction finding. Solve max sum(b) s.t. b>=0, conv(a,b)_k <= M for top peaks.
    Returns the direction b (normalized).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    S = float(a.sum())

    # Get top peaks
    top_idx = np.argsort(c)[::-1][:max_peaks]

    # Build constraint matrix: A[i, m] = a[k_i - m] for valid m
    A_ub = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_end >= m_start:
            A_ub[i, m_start:m_end + 1] = a[k - m_start:k - m_start + (m_end - m_start + 1)][::-1]

    b_ub = np.full(len(top_idx), c_max)

    # Add constraint: sum(b) <= 2*S
    A_ub = np.vstack([A_ub, np.ones((1, n))])
    b_ub = np.append(b_ub, 2.0 * S)

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if not res.success:
            return None
        b = res.x
        S_b = float(b.sum())
        if S_b < 1e-12:
            return None
        return b / S_b
    except Exception:
        return None


def coordinate_descent(a, n_sweeps=8, top_peaks=5, max_elems=300):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        c_max = c.max()
        top_k = np.argsort(c)[::-1][:top_peaks]

        affected = set()
        for k in top_k:
            k = int(k)
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            affected.update(range(m_start, m_end + 1))

        if len(affected) > max_elems:
            contribs = np.zeros(n)
            for k in top_k:
                k = int(k)
                m_start = max(0, k - n + 1)
                m_end = min(k, n - 1)
                for m in range(m_start, m_end + 1):
                    km = k - m
                    if 0 <= km < n:
                        contribs[m] += a[m] * a[km]
            top_indices = np.argsort(contribs)[::-1][:max_elems]
        else:
            top_indices = np.array(list(affected))

        for m in top_indices:
            m = int(m)
            if a[m] < 1e-12:
                continue
            a_old = a[m]
            best_val = a_old
            best_local = best_sc
            for f in [0.6, 0.8, 0.9, 0.95, 1.0, 1.05, 1.1, 1.2, 1.4]:
                a[m] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[m] = a_old
                    continue
                a_tmp = a / S_tmp
                c_tmp = fast_conv(a_tmp)
                sc_tmp = 2.0 * n * float(c_tmp.max()) / (S_tmp * S_tmp)
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_val = a_old * f
            a[m] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def diverse_perturb(a, rng, strategy):
    n = len(a)
    if strategy == 0:
        a_new = a * (1.0 + 0.04 * rng.standard_normal(n))
    elif strategy == 1:
        x = np.arange(n) / n
        freq = rng.integers(2, 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.1)
        a_new = a * (1.0 + amp * np.cos(2 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = rng.integers(4, 12)
        block_size = max(1, n // n_blocks)
        a_new = a.copy()
        for b_idx in range(n_blocks):
            s = rng.uniform(0.6, 1.4)
            a_new[b_idx * block_size:(b_idx + 1) * block_size] *= s
    elif strategy == 3:
        frac = rng.uniform(0.05, 0.2)
        n_rand = max(1, int(frac * n))
        idx = rng.choice(n, n_rand, replace=False)
        a_new = a.copy()
        a_new[idx] *= rng.uniform(0.3, 2.5, n_rand)
    else:
        A = np.fft.rfft(a)
        A *= (1.0 + 0.02 * rng.standard_normal(len(A)))
        a_new = np.fft.irfft(A, n)

    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a.copy()
    return a_new / S


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: adaptive LSE-LBFGS with convergence-driven beta scheduling,
    gradient-flow for high-beta regime, LP direction finding, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 45.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) / 2048.0

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent available, starting from uniform")

    print(f"Initial best: {best_score:.10f}")
    n = len(best_a)

    # =========================================================================
    # Phase 1: Fienup + low-beta L-BFGS warmup (8% time)
    # =========================================================================
    p1_end = t_start + 0.08 * (total_budget - 45.0)
    print(f"\n=== Phase 1: Warmup ===")

    a_f = fienup_flat(best_a, n_iters=150)
    sc = compute_score(a_f)
    update(sc, a_f, "FIENUP")

    if time.monotonic() < p1_end:
        sc, a_opt = run_lbfgs(best_a, 100, maxiter=5000, maxfun=100000)
        update(sc, a_opt, "LBFGS-100")

    # =========================================================================
    # Phase 2: Adaptive LSE + L-BFGS (42% time) - main workhorse
    # =========================================================================
    p2_end = t_start + 0.50 * (total_budget - 45.0)
    print(f"\n=== Phase 2: Adaptive LSE + L-BFGS ===")

    # Adaptive beta schedule: increase beta only when making progress
    beta = 300
    prev_score = best_score
    stagnant_betas = 0

    while time.monotonic() < p2_end:
        remaining = p2_end - time.monotonic()
        if remaining < 30:
            break

        # Allocate iterations based on remaining time and beta level
        if beta < 1000:
            maxiter = int(min(20000, remaining * 12 + 2000))
        elif beta < 10000:
            maxiter = int(min(25000, remaining * 8 + 3000))
        else:
            maxiter = int(min(30000, remaining * 6 + 5000))
        maxiter = max(maxiter, 3000)

        print(f"  L-BFGS beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s, score={best_score:.8f})")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        dt = time.monotonic() - t0
        print(f"    -> score={sc:.10f} ({dt:.1f}s)")

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            stagnant_betas = 0
            # Increase beta moderately
            beta = int(beta * 1.8)
        else:
            stagnant_betas += 1
            if stagnant_betas >= 2:
                beta = int(beta * 2.5)
                stagnant_betas = 0
            # If no improvement, try gradient flow instead
            if beta > 50000 and remaining > 60:
                t0 = time.monotonic()
                a_gf = gradient_flow(best_a, n_iters=min(3000, int(remaining * 3)))
                sc_gf = compute_score(a_gf)
                print(f"    GF: score={sc_gf:.10f} ({time.monotonic()-t0:.1f}s)")
                if sc_gf < best_score:
                    best_score = sc_gf
                    best_a = a_gf.copy()
                    beta = int(beta * 1.5)
                else:
                    beta = int(beta * 2)

        if beta > 500000:
            break

    # =========================================================================
    # Phase 3: LP direction + gradient flow (15% time)
    # =========================================================================
    p3_end = t_start + 0.65 * (total_budget - 45.0)
    print(f"\n=== Phase 3: LP + Gradient Flow ===")

    if time.monotonic() < p3_end:
        # Try LP direction
        if n <= 3000:
            b_dir = lp_direction(best_a, max_peaks=8)
            if b_dir is not None:
                for t_mix in [0.1, 0.2, 0.3, 0.5]:
                    a_mix = (1 - t_mix) * best_a + t_mix * b_dir
                    a_mix = np.maximum(a_mix, 0.0)
                    S_mix = a_mix.sum()
                    if S_mix < 1e-12:
                        continue
                    a_mix = a_mix / S_mix
                    sc = compute_score(a_mix)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_mix.copy()
                        print(f"    LP mix t={t_mix}: {sc:.10f}")
                # Polish after LP
                if time.monotonic() < p3_end:
                    sc, a_opt = run_lbfgs(best_a, 10000, maxiter=10000, maxfun=200000)
                    update(sc, a_opt, "LP+POLISH")

        # Gradient flow
        if time.monotonic() < p3_end:
            a_gf = gradient_flow(best_a, n_iters=min(5000, int((p3_end - time.monotonic()) * 4)))
            sc = compute_score(a_gf)
            update(sc, a_gf, "GRAD-FLOW")

    # =========================================================================
    # Phase 4: Diverse restarts (20% time)
    # =========================================================================
    p4_end = t_start + 0.85 * (total_budget - 45.0)
    print(f"\n=== Phase 4: Diverse Restarts ===")

    restart_count = 0
    while time.monotonic() < p4_end and restart_count < 30:
        strategy = restart_count % 5
        a_pert = diverse_perturb(best_a, rng, strategy)

        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=6000, maxfun=120000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Restart {restart_count} (strat={strategy}): {sc:.10f}")

            # Further polish if time allows
            if time.monotonic() < p4_end - 30:
                sc2, a2 = run_lbfgs(a_opt, 20000, maxiter=12000, maxfun=240000)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a2.copy()
                    print(f"    Polished: {sc2:.10f}")
        restart_count += 1

    # =========================================================================
    # Phase 5: Final high-precision polish (10% time)
    # =========================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 90:
        remaining = deadline - 90 - time.monotonic()
        maxiter = int(min(20000, remaining * 10))
        sc, a_fin = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_fin, "FINAL-LBFGS-200k")

    # Gradient flow with very high internal beta
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        a_gf = gradient_flow(best_a, n_iters=int(min(4000, remaining * 5)))
        sc = compute_score(a_gf)
        update(sc, a_gf, "FINAL-GF")

    # Coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd = coordinate_descent(best_a, n_sweeps=10, top_peaks=5, max_elems=400)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # One more L-BFGS at extreme beta
    if time.monotonic() < deadline - 15:
        sc, a_fin = run_lbfgs(best_a, 1000000, maxiter=5000, maxfun=100000)
        update(sc, a_fin, "FINAL-LBFGS-1M")

    # Final CD
    if time.monotonic() < deadline - 5:
        a_cd = coordinate_descent(best_a, n_sweeps=5, top_peaks=3, max_elems=200)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD-2")

    # =========================================================================
    # Output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
