"""
Optimized step function search using improved LP-based alternating,
frequency-domain Newton steps, and adaptive FISTA with exact Lipschitz constants.
Aims to beat 1.506067 by reducing the peak self-convolution more effectively.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def compute_lipschitz(a, beta):
    """Compute tight Lipschitz constant for the smoothed LP objective."""
    n = len(a)
    # ||B||_2 = max_omega |A(e^{i omega})|
    n_fft = 8 * n
    A = np.fft.rfft(a, n_fft)
    max_abs_A = float(np.max(np.abs(A)))
    L = beta * max_abs_A * max_abs_A
    return max(L, 1e-12)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-12):

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

        return lse_score_and_grad(x, beta)

        obj_and_grad, a0,

    )
    a_out = np.clip(res.x, 0.0, 1000.0)

    return exact_sc, a_out, res


def fista_lp_solve(a, M, n_iters, beta, x_init=None, use_backtracking=True):
    """
    Solve smoothed LP dual via FISTA with backtracking line search.
    Minimize: (1/beta) * log(sum exp(beta*(Bx - M))) + indicator(x>=0)
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()

    # Precompute FFT of a
    FA = np.fft.rfft(a, M_fft)

    # Compute initial Lipschitz constant
    L = compute_lipschitz(a, beta)
    step = 1.0 / L

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        # Gradient: (B^T p)[j]
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr  # gradient of (1/beta)*log(sum exp(beta*(Bx-M))) w.r.t. x is B^T p

        # Projection step (FISTA)
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        # Backtracking: every 100 iterations, recompute L
        if use_backtracking and it % 200 == 199:
            L_new = compute_lipschitz(a, beta)
            if L_new < L * 0.9:
                L = L_new
                step = 1.0 / L

    return x

"""
Improved optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: tighter Lipschitz constants via FFT for faster FISTA,
adaptive M-tightening in LP alternation, multi-scale restarts for local minimum escape,
frequency-domain Newton acceleration, and progressive L-BFGS with beta annealing.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def compute_lipschitz_tight(a, beta):
    """Compute tight Lipschitz constant: beta * max_omega |A(e^{i omega})|^2."""
    n = len(a)
    n_fft = max(8 * n, 4096)
    A = np.fft.rfft(a, n_fft)
    max_abs_A = float(np.max(np.abs(A)))
    return max(beta * max_abs_A * max_abs_A, 1e-12)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-12):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fista_lp_solve(a, M_val, n_iters, beta, x_init=None):
    """FISTA for smoothed LP: min (1/beta)*log(sum exp(beta*(Bx - M))) s.t. x>=0."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)

    L = compute_lipschitz_tight(a, beta)
    step = 1.0 / L

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M_val)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

        if it % 300 == 299:
            L_new = compute_lipschitz_tight(a, beta)
            if L_new < L * 0.95:
                L = L_new
                step = 1.0 / L

    return x

def multi_scale_restart(a, rng, n_iters_coarse=25):

                + 0.1 * np.cos(6 * np.pi * x + 1.3))
    return inits


def error_reduction_quick(a, n_iters=40, alpha=0.75):
    """Quick error reduction for diverse initial exploration."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.4 + 0.4 * min(1.0, it / 10.0)
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(budget_s=2400):
    """
    Multi-phase optimizer with tight Lipschitz FISTA, adaptive LP,
    frequency Newton, multi-scale restarts, and progressive L-BFGS.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [6962, 8192, 10240]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}")

    # Phase 1: Diverse initialization + quick error reduction (8% time)
    phase1_end = t_start + 0.08 * total_budget
    print(f"\n--- Phase 1: Diverse Init + Quick ER ---")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, int(remaining / 5.0))
        inits = inits[:min(len(inits), max_inits)]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_er = error_reduction_quick(a0, n_iters=40, alpha=0.75)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: LP alternating with adaptive M (45% time)
    phase2_end = t_start + 0.53 * total_budget
    print(f"\n--- Phase 2: LP Alternating (Improved) ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP configs: (fista_iters, beta, m_shrink, max_outer)
    lp_configs = [
        (8000, 100.0, 0.997, 25),
        (12000, 200.0, 0.995, 25),
        (16000, 400.0, 0.993, 20),
        (20000, 700.0, 0.990, 15),
        (25000, 1200.0, 0.985, 15),
    ]

    x_warm = None
    for ci, (fi, beta_val, m_shr, max_out) in enumerate(lp_configs):
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 45:
            break
        print(f"  LP round {ci}: fista={fi}, beta={beta_val}, shrink={m_shr}, "
              f"remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating_improved(
            a_work, max_outer=max_out, fista_iters=fi,
            time_limit=phase2_end - 20, beta=beta_val, m_shrink=m_shr
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{ci}")
        a_work = a_out.copy()

        # Every 2 rounds, try freq Newton + multi-scale
        if ci % 2 == 1:
            a_fn = freq_newton(a_work, n_iters=30)
            sc_fn = compute_score(a_fn)
            report(sc_fn, a_fn, f"FreqNewton r{ci}")
            if sc_fn < compute_score(a_work):
                a_work = a_fn

            a_ms = multi_scale_restart(a_work, rng)
            sc_ms = compute_score(a_ms)
            report(sc_ms, a_ms, f"MultiScale r{ci}")
            if sc_ms < compute_score(a_work):
                a_work = a_ms

    # Try alternate lengths
    for n_alt in lengths:
        if n_alt == n_main:
            continue
        if time.monotonic() > phase2_end - 60:
            break
        print(f"  LP alt length n={n_alt}")
        a_test = np.interp(np.linspace(0, 1, n_alt),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        St = float(a_test.sum())
        if St > 1e-12:
            a_test = a_test / St
        a_out, sc_out, _ = lp_alternating_improved(
            a_test, max_outer=15, fista_iters=12000,
            time_limit=phase2_end - 15, beta=300.0, m_shrink=0.995
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: L-BFGS with progressive beta (25% time)
    phase3_end = t_start + 0.78 * total_budget
    print(f"\n--- Phase 3: L-BFGS Progressive ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        maxit = 8000 if beta < 400 else 5000
        sc, a_opt, res = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=150000, gtol=1e-13)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, t={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Peak search + final polish (17% time)
    print(f"\n--- Phase 4: Peak Search & Final Polish ---")

    # Peak-directed subgradient search
    if time.monotonic() < deadline - 120:
        a_peak = peak_subgradient_search(best_a, n_steps=300, rng=rng)
        sc = compute_score(a_peak)
        report(sc, a_peak, "peak search")

    # Freq Newton one more time
    if time.monotonic() < deadline - 100:
        a_fn = freq_newton(best_a, n_iters=50)
        sc = compute_score(a_fn)
        report(sc, a_fn, "final freq newton")

    # Multi-scale restart
    if time.monotonic() < deadline - 80:
        a_ms = multi_scale_restart(best_a, rng, n_iters_coarse=40)
        sc = compute_score(a_ms)
        report(sc, a_ms, "final multi-scale")

    # Final LP polish
    if time.monotonic() < deadline - 100:
        print("  Final LP polish...")
        a_in = best_a.copy()
        S = float(a_in.sum())
        if S > 0:
            a_in = a_in / S
        a_final, sc_final, _ = lp_alternating_improved(
            a_in, max_outer=15, fista_iters=20000,
            time_limit=deadline - 60, beta=800.0, m_shrink=0.990
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # Final L-BFGS
    if time.monotonic() < deadline - 40:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 10000, maxiter=10000, maxfun=300000, gtol=1e-14)
        report(sc, a_opt, "final LBFGS b=10000")

    # One more peak search
    if time.monotonic() < deadline - 20:
        a_final = peak_subgradient_search(best_a, n_steps=150, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final peak")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
