Optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. GS-Fienup warmup with diverse initializations
2. Adam optimizer with LSE-smoothed objective and beta annealing
3. Periodic perturbation to escape local minima

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    exp_s = np.exp(scaled - max_s)
    Z = float(exp_s.sum())
    p = exp_s / Z
    L = c_max + (max_s + np.log(Z)) / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    M_fft = 1 << (3 * n - 2).bit_length()
    P = np.fft.fft(p, M_fft)
    A = np.fft.fft(a, M_fft)
    q_full = np.fft.ifft(P * np.conj(A), M_fft).real
    q = q_full[:n]
    
    dL = 2.0 * q
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL * S - 2.0 * L) / S3
    
    return score, grad

def adam_lse(a0, beta_schedule, n_steps_per_beta, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective with beta annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

                    best_a = a.copy()

def gs_fienup(a0, n_iters=500, alpha=0.5):

    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    best_sc = compute_score(a)

    # sum(c) = S^2, so C = S^2/(2n-1)
    # |A(m)|^2 = S^2 * M / (2n-1)
    # |A(m)| = S * sqrt(M/(2n-1))

        A = np.fft.fft(a, M)
        A_mag = np.abs(A)
        A_phase = np.angle(A)

        a_new = np.fft.ifft(new_A, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:

        a = a_new

"""
Optimization of step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Strategy: GS-Fienup warmup + Adam optimizer with LSE-smoothed objective and
progressive beta annealing, combined with perturbation-based diversification
and multi-phase schedule for robust local minimum escape.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for the objective."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    exp_s = np.exp(scaled - max_s)
    Z = float(exp_s.sum())
    p = exp_s / Z
    L = c_max + (max_s + np.log(Z)) / beta

    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da_i = 2 * sum_k p_k * a_{k-i}
    # Compute q[i] = sum_k p_k * a_{k-i} via cross-correlation FFT
    M_fft = 1 << (3 * n - 2).bit_length()
    P = np.fft.fft(p, M_fft)
    A = np.fft.fft(a, M_fft)
    q_full = np.fft.ifft(P * np.conj(A), M_fft).real
    q = q_full[:n]

    dL = 2.0 * q
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL * S - 2.0 * L) / S3

    return score, grad


def adam_lse(a0, beta_schedule, steps_per_beta, lr_init, lr_decay_per_beta=1.0,
             beta1=0.9, beta2=0.999, eps=1e-8, check_interval=2000, time_limit=None, t_start=None):
    """Adam optimizer on LSE-smoothed objective with beta annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init

    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in beta_schedule:
        for step in range(steps_per_beta):
            if time_limit is not None and t_start is not None:
                if time.monotonic() > t_start + time_limit:
                    return best_a

            sc, grad = lse_score_grad(a, beta)
            t += 1

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step_size = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step_size
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                a = best_a.copy()
                break

            if check_interval > 0 and step % check_interval == 0:
                sc_check = compute_score(a)
                if sc_check < best_sc:
                    best_sc = sc_check
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_a = a.copy()

    return best_a


def gs_fienup(a0, n_iters=500, target_type="flat"):
    """GS-Fienup with specified magnitude spectrum target."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    if target_type == "flat":
        target_mag = np.sqrt(M) / (2 * n - 1)
        target_mag = np.ones(M) * np.sqrt(M / (2 * n - 1))
    elif target_type == "raised_cosine":
        m = np.arange(M)
        target_mag = np.ones(M) * np.sqrt(1.0 / (2 * n - 1))
        # Slightly boost low frequencies to help non-negativity
        wc = 0.3
        for k in range(M):
            dist = min(k, M - k) / (M / 2.0)
            target_mag[k] *= (1.0 + wc * (1.0 - dist))
    else:
        target_mag = np.ones(M) * np.sqrt(M / (2 * n - 1))

    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_phase = np.angle(A)
        new_A = target_mag * np.exp(1j * A_phase)
        a_new = np.fft.ifft(new_A, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 10 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def gs_adaptive(a0, n_iters=400):
    """GS with adaptive target that down-weights current peaks."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_min = float(c[:2 * n - 1].min())

        # Target: emphasize flattening the peaks
        t_frac = it / n_iters
        flatness = 0.5 + 0.5 * t_frac

        # Compute target magnitude from current spectrum but flattened
        A = np.fft.fft(a, M)
        A_mag = np.abs(A)
        A_phase = np.angle(A)

        # Blend between current magnitude and flat
        flat_mag = np.ones(M) * np.mean(A_mag)
        new_mag = (1.0 - flatness) * A_mag + flatness * flat_mag
        new_mag = np.maximum(new_mag, 1e-10)

        new_A = new_mag * np.exp(1j * A_phase)
        a_new = np.fft.ifft(new_A, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 10 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)
    inits.append(np.maximum(0.5 * (1.0 + np.sin(2.0 * np.pi * x * 7.0)), 0.01))

    for _ in range(8):
        n_harmonics = int(rng.integers(3, 25))
        weights = rng.standard_normal(n_harmonics)
        phases = rng.uniform(0, 2.0 * np.pi, n_harmonics)
        sig = np.zeros(n)
        for i in range(n_harmonics):
            sig += weights[i] * np.cos(2.0 * np.pi * (i + 1) * x + phases[i])
        init = np.maximum(0.5 + 0.25 * sig, 0.01)
        inits.append(init)

    return inits


def peak_reduction(a0, n_rounds=100, n_peaks=20, reduce_factor=0.9):
    """Directly reduce top peaks by scaling contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            n_reduce = min(8, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            for li in top_local:
                j = j_start + li
                a[j] *= reduce_factor

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/S^2.
    Uses GS warmup + Adam/LSE annealing + perturbation diversification.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return False
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting optimization: n={n}, initial={best_score:.8f}")

    # ================================================================
    # Phase 1: GS warmup from diverse initializations (first 8%)
    # ================================================================
    t1 = t_start + 0.08 * (total - 40.0)
    print(f"\n=== Phase 1: GS warmup (until t={t1 - t_start:.0f}s) ===")

    inits = build_inits(n, rng, parent if len(parent) == n else None)
    for i, a0 in enumerate(inits):
        if time.monotonic() > t1:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / max(a0.sum(), 1e-12)

        a_gs = gs_fienup(a0, n_iters=300)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)

        if time.monotonic() < t1 and sc < best_score + 0.5:
            a_gs2 = gs_adaptive(a_gs, n_iters=200)
            sc2 = compute_score(a_gs2)
            update_best(sc2, a_gs2)

        # Quick Adam from GS result
        if time.monotonic() < t1:
            a_adam = adam_lse(a_gs, [500, 2000], 5000, lr=0.01,
                              time_limit=30, t_start=time.monotonic())
            sc3 = compute_score(a_adam)
            update_best(sc3, a_adam)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Main Adam+LSE with progressive beta annealing (45%)
    # ================================================================
    t2 = t_start + 0.53 * (total - 40.0)
    print(f"\n=== Phase 2: Adam+LSE annealing (until t={t2 - t_start:.0f}s) ===")

    # Progressive beta schedule with decreasing learning rate
    # Key insight: start smooth (low beta) to find good basin, then sharpen
    beta_schedule = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
    steps_per_beta = 15000
    lr_start = 0.02

    a_work = best_a.copy()
    for idx, beta in enumerate(beta_schedule):
        if time.monotonic() > t2:
            break
        lr_here = lr_start * (0.7 ** idx)
        t_beta = time.monotonic()
        a_work = adam_lse(a_work, [beta], steps_per_beta, lr=lr_here,
                          check_interval=3000,
                          time_limit=total - 40.0 - (time.monotonic() - t_start),
                          t_start=t_start)
        sc = compute_score(a_work)
        update_best(sc, a_work)
        print(f"  beta={beta:>6d}, lr={lr_here:.4f}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: High-beta refinement + perturbation (27%)
    # ================================================================
    t3 = t_start + 0.80 * (total - 40.0)
    print(f"\n=== Phase 3: High-beta + perturbation (until t={t3 - t_start:.0f}s) ===")

    # First, push with high beta from current best
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > t3:
            break
        a_work = adam_lse(best_a, [beta], 10000, lr=0.005,
                          check_interval=2000,
                          time_limit=total - 40.0 - (time.monotonic() - t_start),
                          t_start=t_start)
        sc = compute_score(a_work)
        update_best(sc, a_work)
        print(f"  High-beta {beta}: score={sc:.8f}")

    # Perturbation trials
    pert_trial = 0
    while time.monotonic() < t3:
        pert_trial += 1
        strength = 0.01 + 0.04 * rng.random()
        a_pert = best_a + rng.standard_normal(n) * strength
        a_pert = np.maximum(a_pert, 0.0)
        S_p = a_pert.sum()
        if S_p < 1e-12:
            continue
        a_pert = a_pert / S_p

        # GS on perturbed
        a_gs_p = gs_fienup(a_pert, n_iters=200)
        sc_gs = compute_score(a_gs_p)
        update_best(sc_gs, a_gs_p)

        # Adam on perturbed (or GS result if better)
        a_start = a_gs_p if sc_gs < compute_score(a_pert) else a_pert
        a_opt_p = adam_lse(a_start, [1000, 5000, 20000], 8000, lr=0.005,
                           check_interval=2000,
                           time_limit=60, t_start=time.monotonic())
        sc_opt = compute_score(a_opt_p)
        update_best(sc_opt, a_opt_p)
        print(f"  Perturbation {pert_trial} (s={strength:.3f}): best={best_score:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish - alternate GS, Adam, peak reduction (20%)
    # ================================================================
    print(f"\n=== Phase 4: Final polish (until t={deadline - t_start:.0f}s) ===")

    polish_round = 0
    while time.monotonic() < deadline:
        polish_round += 1

        # Very high beta Adam
        beta_polish = 200000 + polish_round * 50000
        a_opt = adam_lse(best_a, [beta_polish], 8000, lr=0.002,
                         check_interval=2000,
                         time_limit=30, t_start=time.monotonic())
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Peak reduction
        if time.monotonic() < deadline:
            a_peak = peak_reduction(best_a, n_rounds=50, n_peaks=15, reduce_factor=0.92)
            sc_peak = compute_score(a_peak)
            update_best(sc_peak, a_peak)

        # Quick GS
        if time.monotonic() < deadline:
            a_gs = gs_fienup(best_a, n_iters=150)
            sc_gs = compute_score(a_gs)
            update_best(sc_gs, a_gs)

        # Small perturbation + Adam
        if time.monotonic() < deadline:
            a_small_pert = best_a + rng.standard_normal(n) * 0.005
            a_small_pert = np.maximum(a_small_pert, 0.0)
            a_small_pert = a_small_pert / max(a_small_pert.sum(), 1e-12)
            a_final = adam_lse(a_small_pert, [50000, 200000], 5000, lr=0.003,
                               check_interval=1000,
                               time_limit=20, t_start=time.monotonic())
            sc_final = compute_score(a_final)
            update_best(sc_final, a_final)

        print(f"  Polish {polish_round}: best={best_score:.8f}")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
