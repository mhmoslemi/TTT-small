```python
"""
Search for non-negative coefficients minimizing
    2*n*max(np.convolve(a,a)) / (sum(a)**2).

This program uses several complementary ideas:
1. Multi-length multi-start search (parent, 16384/10240 lengths).
2. Exponentiated-gradient active-set updates with both non-negative
   active weights and signed variance/softmax weights, plus line search.
3. A sparse linear-programming "LP direction" on the most violated
   convolution constraints, used as a large exploratory move.
4. LSE-smoothed L-BFGS-B polishing of the best candidate.

The output is the best finite non-negative list found before the budget.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


TINY = 1e-300
TARGET_EARLY_STOP = 1.5010


def next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def first_key(item):
    return item[0]


def resample_vector(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, None)
    n = int(n)
    m = int(arr.size)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return arr.astype(np.float64, copy=True)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, arr).astype(np.float64)
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:M]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c = np.maximum(c, 0.0)
    return c


def make_base(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    M = 2 * n - 1
    N = next_pow2(M)
    base = np.fft.rfft(a[::-1], N)
    return base, N


def corr_from_base(base, w, n):
    A, N = base
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)
    return full[n - 1:2 * n - 1]


def corr_active(a, w):
    base, N = make_base(a)
    return corr_from_base(base, w, int(a.size))


def score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    S = float(a.sum())
    if S < 1e-12:
        return float("inf")
    if abs(S - 1.0) > 1e-8:
        a = a / S
    c = self_conv(a)
    if c.size == 0:
        return float("inf")
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * float(n) * cmax)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return float("inf")
    c = self_conv(a)
    cmax = float(np.max(c)) if c.size else 0.0
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * float(n) * cmax / (S * S))


def top_weight(c, k=1):
    T = int(c.size)
    if T == 0:
        return np.ones(1, dtype=np.float64)
    k = max(1, min(T, int(k)))
    if k == T:
        return np.ones(T, dtype=np.float64) / float(T)
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(T, dtype=np.float64)
    w[idx] = 1.0
    return w


def make_weights(c, cmax, mode, params):
    c = np.asarray(c, dtype=np.float64)
    T = int(c.size)
    if T == 0:
        return None
    cmax = max(float(cmax), 1e-300)

    if mode == 0:  # non-negative softmax over convolution values
        tau = float(params)
        z = c / cmax
        val = np.exp(tau * np.clip(z - 1.0, -50.0, 0.0))
        s = float(val.sum())
        if s <= 1e-300:
            return top_weight(c, 1)
        return val / s

    if mode == 1:  # top-k
        return top_weight(c, int(params))

    if mode == 2:  # threshold active set
        eps = float(params)
        thr = cmax * (1.0 - eps)
        w = (c >= thr).astype(np.float64)
        if w.sum() < 1.0:
            w = top_weight(c, 1)
        else:
            w /= w.sum()
        return w

    if mode == 3:  # excess over target, power
        target = float(params[0])
        power = float(params[1])
        ex = np.maximum(c - target * cmax, 0.0)
        ex = np.power(ex, power)
        s = float(ex.sum())
        if s <= 1e-300:
            return top_weight(c, 8)
        return ex / s

    if mode == 4:  # signed variance: reduce variance of all coefficients
        mean = float(c.mean())
        std = float(c.std())
        if std < 1e-15:
            return None
        w = (c - mean) / std
        w = np.clip(w, -5.0, 5.0)
        m = float(np.max(np.abs(w)))
        if m <= 1e-15:
            return None
        return w / m

    if mode == 5:  # signed zero-mean softmax
        tau = float(params)
        z = c / cmax
        val = np.exp(tau * np.clip(z - 1.0, -50.0, 0.0))
        s = float(val.sum())
        if s <= 1e-300:
            return None
        p = val / s
        w = p - 1.0 / float(T)
        m = float(np.max(np.abs(w)))
        if m <= 1e-15:
            return None
        return w / m

    if mode == 6:  # centered signed normalized values
        z = c / cmax
        low = float(np.min(z))
        high = float(np.max(z))
        span = high - low
        if span <= 1e-15:
            return None
        w = (z - low) / span - 0.5
        m = float(np.max(np.abs(w)))
        if m <= 1e-15:
            return None
        return w / m

    return top_weight(c, 1)


def update_exp(a, h, eta):
    a = np.asarray(a, dtype=np.float64).ravel()
    h = np.asarray(h, dtype=np.float64).ravel()
    if a.size == 0:
        return a.copy()
    hmax = float(np.max(np.abs(h)))
    if hmax <= 1e-300 or eta <= 0.0:
        return a.copy()
    s = np.clip(h / hmax, -100.0, 100.0)
    exponent = np.clip(-eta * s, -700.0, 700.0)
    loga = np.log(np.maximum(a, 1e-300)) + exponent
    loga = loga - float(np.max(loga))
    out = np.exp(np.clip(loga, -745.0, 700.0))
    return normalize_a(out)


def inv_direction(a, h):
    a = np.asarray(a, dtype=np.float64).ravel()
    h = np.asarray(h, dtype=np.float64).ravel()
    if a.size == 0:
        return a.copy()
    hmax = float(np.max(np.abs(h)))
    if hmax <= 1e-300:
        return a.copy()
    s = np.clip(h / hmax, -50.0, 50.0)
    den = 50.0 + s
    den = np.maximum(den, 1e-3)
    out = 1.0 / den
    return normalize_a(np.clip(out, 0.0, None))


def perturb(a, rng, strength=None):
    a = normalize_a(a)
    n = int(a.size)
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.30))
    mode = int(rng.integers(0, 8))
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.25))
        q = (1.0 - t) * a + t * (1.0 / float(n))
    elif mode == 2:
        q = a[::-1]
    elif mode == 3:
        q = a * (1.0 + 0.35 * strength * np.sin(2.0 * np.pi * x))
    elif mode == 4:
        q = a * (1.0 + 0.5 * strength * (2.0 * x - 1.0) ** 2)
    elif mode == 5:
        q = a * np.exp(0.10 * rng.standard_normal(n))
        t = float(rng.uniform(0.02, 0.10))
        q = (1.0 - t) * q + t * (1.0 / float(n))
    elif mode == 6:
        sh = int(rng.integers(1, n))
        q = np.roll(a, sh)
    else:
        q = a * (1.0 + strength * rng.random(n))
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if s < 1e-12:
        return normalize_a(a)
    return q / s

    best_a = a.copy()
    best_sc = score_normalized(a)
    if not np.isfinite(best_sc):
        best_a = np.ones(n, dtype=np.float64) / float(n)
        best_sc = 2.0

    eta_pos = 0.25
    eta_signed = 0.12
    stagnant = 0
    it = 0
    perturb_count = 0
    cycle = strategy_cycle()
    L = len(cycle)
    t0 = time.monotonic()

    while time.monotonic() < deadline - 2.0:
        c = self_conv(a)
        if c.size == 0:
            break
        cmax = float(np.max(c))
        sc = float(2.0 * n * cmax)

        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
            if best_sc < TARGET_EARLY_STOP:
                break

        base, N = make_base(a)
        committed = False
        idx = it % L
        strategies = [
            cycle[idx % L],
            cycle[(idx + 1) % L],
            cycle[(idx + 2) % L],
        ]
        if it % 4 == 0:
            strategies.append(cycle[(idx + 3) % L])

        for mode, params in strategies:
            if committed:
                break
            w = make_weights(c, cmax, mode, params)
            if w is None:
                continue
            h = corr_from_base(base, w, n)
            hmax = float(np.max(np.abs(h)))
            if hmax <= 1e-300:
                continue

            signed = mode in (4, 5, 6)
            eta0 = eta_signed if signed else eta_pos
            etas = [0.5 * eta0, eta0, 2.0 * eta0]
            if eta0 * 4.0 <= 50.0:
                etas.append(4.0 * eta0)

            for eta_try in etas:
                if eta_try < 1e-5:
                    continue
                cand = update_exp(a, h, eta_try)
                csc = score_normalized(cand)
                if np.isfinite(csc) and csc < best_sc - 1e-13:
                    best_sc = csc
                    best_a = cand.copy()
                    a = cand.copy()
                    if signed:
                        eta_signed = min(50.0, max(0.005, eta_try))
                    else:
                        eta_pos = min(50.0, max(0.005, eta_try))
                    committed = True
                    stagnant = 0
                    if verbose and (it % 10 == 0):
                        print("  [{} eg it{}] mode={} eta={:.4f} score={:.10f}".format(
                            tag, it, mode, eta_try, best_sc))
                    break

        if not committed:
            # Inverse-LP-inspired direction, then stagnation handling.
            w = make_weights(c, cmax, 4, 1.0)
            if w is not None:
                h = corr_from_base(base, w, n)
                dirv = inv_direction(a, h)
                for alpha in (0.25, 0.55, 0.85):
                    cand = normalize_a((1.0 - alpha) * a + alpha * dirv)
                    csc = score_normalized(cand)
                    if np.isfinite(csc) and csc < best_sc - 1e-13:
                        best_sc = csc
                        best_a = cand.copy()
                        a = cand.copy()
                        committed = True
                        stagnant = 0
                        break

            if not committed:
                stagnant += 1
                eta_pos = max(0.01, eta_pos * 0.90)
                eta_signed = max(0.005, eta_signed * 0.90)

                if stagnant == 20 or (stagnant > 20 and stagnant % 20 == 0):
                    if perturb_count < 8:
                        a = perturb(best_a, rng, strength=float(rng.uniform(0.08, 0.30)))
                        perturb_count += 1
                        stagnant = 0
                        if verbose:
                            print("  [{} eg perturbed {} -> exploring".format(tag, perturb_count))

                if stagnant >= 300:
                    break

        it += 1
        if verbose and it % 40 == 0:
            print("  [{} eg] it={} stag={} pos_eta={:.4f} sig_eta={:.4f} best={:.10f}".format(
                tag, it, stagnant, eta_pos, eta_signed, best_sc))

    return best_sc, best_a


def lp_direction(a, c, cmax, K, time_limit):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        n = int(a.size)
        T = int(c.size)
        if n == 0 or T == 0 or cmax <= 0.0 or time_limit < 5.0:
            return None

        K = max(30, min(700, int(K)))

        top = np.argpartition(c, -K)[-K:]
        sample = np.linspace(0, T - 1, max(3, K // 3)).astype(int)
        sel = np.unique(np.concatenate((top, sample)))
        if sel.size > 800:
            sel = sel[:800]

        row_parts = []
        col_parts = []
        data_parts = []

        for idx, t in enumerate(sel):
            t = int(t)
            low = max(0, t - (n - 1))
            high = min(n - 1, t)
            if low > high:
                continue
            length = high - low + 1
            data_parts.append(a[t - high:t - low + 1][::-1].astype(np.float64))
            col_parts.append(np.arange(low, high + 1, dtype=np.int64))
            row_parts.append(np.full(length, idx, dtype=np.int64))

        if not data_parts:
            return None

        data = np.concatenate(data_parts)
        rows = np.concatenate(row_parts)
        cols = np.concatenate(col_parts)

        A = csr_matrix((data, (rows, cols)), shape=(len(sel), n))
        b_ub = np.full(len(sel), float(cmax), dtype=np.float64)
        obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 10.0)] * n

        opts = {"presolve": True}
        tl = int(max(1, min(120, float(time_limit))))
        opts["time_limit"] = tl

        try:
            res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=opts)
        except Exception:
            opts.pop("time_limit", None)
            res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=opts)

        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g = np.maximum(g, 0.0)
            if g.sum() > 1e-15:
                return normalize_a(g)
    except Exception:
        pass
    return None


def lse_fun(x, beta):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(x.size)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.maximum(x, 0.0)
    S = float(x.sum())
    if S < 1e-12 or n == 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = self_conv(x)
    if c.size == 0:
        return 1e10, np.zeros(n, dtype=np.float64)
    cmax = float(np.max(c))
    if cmax <= 1e-300:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = float(beta) * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    ez = np.exp(z)
    ezsum = float(ez.sum())
    if ezsum <= 1e-300 or not np.isfinite(ezsum):
        return 1e10, np.zeros(n, dtype=np.float64)

    L = cmax + float(np.log(ezsum)) / float(beta)
    u = ez / ezsum

    h = corr_active(x, u)
    g = 2.0 * h
    grad = (2.0 * float(n) / (S * S)) * (g - 2.0 * L * x / S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    value = float(2.0 * n * L / (S * S))
    if not np.isfinite(value):
        return 1e10, np.zeros(n, dtype=np.float64)
    return value, grad


def lse_lbfgs_polish(a, deadline, tag="", verbose=True):
    a = normalize_a(a)
    n = int(a.size)
    if time.monotonic() > deadline - 30.0:
        return score_normalized(a), a

    best_a = a.copy()
    best_sc = score_normalized(a)

    c = self_conv(a)
    cmax = float(np.max(c)) if c.size else 1e-6
    cmax = max(cmax, 1e-12)
    scale = max(1.0, 1.0 / (cmax * 0.05))
    betas = [0.02 * scale, 0.10 * scale, 0.40 * scale, 1.00 * scale, 2.50 * scale, 6.00 * scale]
    bounds = [(0.0, 100.0)] * n

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 15.0:
            break
        beta = float(beta)
        maxfun = 80
        if bi >= 3:
            maxfun = 110
        maxfun = max(30, min(maxfun, 160))

        try:
            res = minimize(
                lse_fun,
                a.copy(),
                method="L-BFGS-B",
                jac=True,
                args=(beta,),
                bounds=bounds,
                options={
                    "ftol": 1e-12,
                    "gtol": 1e-6,
                    "maxiter": 50,
                    "maxfun": int(maxfun),
                    "maxls": 30,
                },
            )
            a_out = res.x if res.x is not None else a
            a_out = normalize_a(a_out)
            sc = score_normalized(a_out)
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_out.copy()
                a = a_out.copy()
                if verbose:
                    print("  [{} LSE beta={:.3g}] score={:.10f}".format(tag, beta, best_sc))
        except Exception:
            pass

    return float(best_sc), best_a


def refine_candidate(n, parent, deadline, rng, tag, full_inits=True):
    n = int(n)
    if n <= 0:
        n = 1024
    t0 = time.monotonic()
    if full_inits:
        inits = generate_inits(n, rng, parent=parent, cap=55)
    else:
        inits = generate_inits(n, rng, parent=parent, cap=12)

    scored = []
    for ini in inits:
        ini = normalize_a(ini)
        sc = score_normalized(ini)
        if np.isfinite(sc):
            scored.append((sc, ini))

    if not scored:
        uni = np.ones(n, dtype=np.float64) / float(n)
        sc0 = score_normalized(uni)
        if not np.isfinite(sc0):
            sc0 = 2.0
        scored.append((sc0, uni))

    scored.sort(key=first_key)
    best_sc, best_a = scored[0]
    if verbose_like_print := True:
        pass

    print("[{}] n={} initial={:.10f}".format(tag, n, best_sc))

    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 180.0:
        eg_dead = time.monotonic() + 0.62 * remaining
        eg_dead = min(eg_dead, deadline - 120.0)
    else:
        eg_dead = deadline
    sc, a = eg_refine(best_a, eg_dead, rng, tag=tag, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # LP directions, if time remains
    if time.monotonic() < deadline - 80.0:
        for K in (120, 320):
            if time.monotonic() > deadline - 60.0:
                break
            a_norm = normalize_a(best_a)
            c = self_conv(a_norm)
            cmax = float(np.max(c)) if c.size else 0.0
            if cmax <= 0.0:
                break
            rem_for_lp = max(5.0, deadline - 70.0 - time.monotonic())
            g = lp_direction(a_norm, c, cmax, K, rem_for_lp)
            if g is None:
                continue
            for alpha in (0.25, 0.55, 0.85, 1.00):
                cand = normalize_a((1.0 - alpha) * a_norm + alpha * g)
                csc = score_normalized(cand)
                if np.isfinite(csc) and csc < best_sc - 1e-13:
                    best_sc = csc
                    best_a = cand.copy()
                    a_norm = cand.copy()
                    print("[{}] LP K={} alpha={:.2f} -> {:.10f}".format(tag, K, alpha, best_sc))

    # Final LSE-BFGS polish within this block
    if time.monotonic() < deadline - 25.0:
        sc2, a2 = lse_lbfgs_polish(best_a, deadline, tag=tag + "-polish", verbose=True)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()

    print("[{}] n={} best={:.10f} time={:.1f}s".format(
        tag, n, best_sc, time.monotonic() - t0))
    return float(best_sc), best_a


def load_parent_sequence():
    try:
        h = globals().get("height_sequence_1", None)
    except Exception:
        return None
    if h is None:
        return None
    try:
        p = np.asarray(h, dtype=np.float64).ravel()
    except Exception:
        return None
    p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
    p = np.clip(p, 0.0, None)
    if p.size < 16:
        return None
    if float(p.sum()) <= 0.01:
        return None
    return p


def choose_primary_length(parent):
    if parent is not None:
        L = int(parent.size)
        if 4096 <= L <= 16384:
            return L
    return 10240


def choose_lengths(primary_n):
    lengths = [int(primary_n)]
    for cand in (16384, 10240):
        if cand not in lengths:
            lengths.append(cand)
    return lengths[:2]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 35.0
    rng = np.random.default_rng(int(seed))

    parent = load_parent_sequence()
    primary_n = choose_primary_length(parent)
    lengths = choose_lengths(primary_n)

    best_sc = float("inf")
    best_seq = None
    global_parent = parent

    if parent is not None:
        psc = exact_score(parent)
        if np.isfinite(psc):
            best_sc = psc
            best_seq = normalize_a(parent).copy()
        print("Parent score: {:.10f}".format(best_sc if np.isfinite(best_sc) else 2.0))

    if global_parent is not None and best_seq is not None:
        global_parent = resample_vector(global_parent, primary_n)

    total_window = max(1.0, deadline - t_start)
    if len(lengths) >= 2:
        fracs = [0.55, 0.80, 1.0]
    else:
        fracs = [0.75, 1.0]

    # First block: new length with full init suite.
    for bi, n in enumerate(lengths):
        end_frac = fracs[bi] if bi < len(fracs) else fracs[-1]
        block_dead = t_start + end_frac * total_window
        if time.monotonic() > block_dead - 30.0:
            continue

        parent_for_block = best_seq if best_seq is not None else global_parent
        if parent_for_block is not None:
            parent_for_block = resample_vector(parent_for_block, n)
        else:
            parent_for_block = global_parent

        tag = "block{}_n{}".format(bi, n)
        full_inits = (bi == 0)
        sc, seq = refine_candidate(n, parent_for_block, block_dead, rng, tag=tag, full_inits=full_inits)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()

    # Final polish block on the current best sequence/length.
    if time.monotonic() < deadline - 20.0 and best_seq is not None:
        n_best = int(best_seq.size)
        if n_best <= 0:
            n_best = 1024
        if n_best > 16384:
            best_seq = resample_vector(best_seq, 16384)
            n_best = 16384
        parent_final = resample_vector(best_seq, n_best)
        tag = "final_n{}".format(n_best)
        sc, seq = refine_candidate(
            n_best,
            parent_final,
            deadline,
            rng,
            tag=tag,
            full_inits=False,
        )
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
    elif best_seq is None:
        best_seq = np.ones(1024, dtype=np.float64) / 1024.0

    # Final cleanup/validation.
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out_list = [float(v) for v in out_arr]

    internal = exact_score(out_arr)
    verified = None
    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = internal

    print("DONE internal={:.10f} verified={} n={} time={:.1f}s".format(
        internal,
        "{:.10f}".format(verified) if verified is not None else "n/a",
        len(out_list),
        time.monotonic() - t_start,
    ))

    return out_list
```
