Hmm, (a*a)[k] = sum_i a[i]*a[k-i] is quadratic in a. So the problem is NOT convex in a.

But it IS convex in the "outer" variable if we view it as: max_k <a, S_k a> where S_k is the "shift" matrix. Wait no, that's still quadratic.

OK so the problem is non-convex. The LP best-response approach is a heuristic that works well in practice.

Let me also consider: what if I use the LP only for a subset of constraints (the active set)?

**Reduced LP approach:**
1. Compute c = a*a, find the top K indices where c is maximum
2. Solve LP: max sum g s.t. (a*g)[k] ≤ M for k in active set, g ≥ 0
3. Update a

This is much faster because we have only K constraints instead of 2n-1.

With K = 100, the LP has 100 constraints and n variables. Very fast.

But the update might not respect the inactive constraints. We can handle this by:
- After the update, check if any inactive constraint is violated
- If so, add it to the active set (cutting plane / separation oracle approach)

This is essentially the cutting plane method applied to the outer problem.

Let me combined approaches:
1. Start with the parent sequence
2. Use the LP best-response with active set for the main iteration
3. Periodically do a "full check" and add violated constraints
4. Polish with the LSE/Nesterov approach at the end

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based best response iteration (fictitious play style)
2. Active set management for constraint reduction
3. LSE smoothing + Nesterov subgradient for final polishing
4. Multi-start with perturbation restarts
"""

import numpy as np
import time
from scipy.optimize import minimize

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    """Fast self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    """Compute the exact score 2*n*max(c)/S^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    """Resample sequence to length n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """Exact subgradient with top-k active peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """Nesterov accelerated subgradient method."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t_mom = 1.0
    y = a.copy()
    v = np.zeros(n)

    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a

    eta = 0.05 / g_norm

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta_mom = (t_mom - 1.0) / t_new

        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break

        step = eta / (1.0 + 0.1 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t_mom = t_new

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if it > 0 and it % 50 == 0:
            if not (sc < best_score + 1e-10):
                eta *= 0.7

    return best_score, best_a


def _lbfgs_lse(a0, beta, maxfun=500):
    """L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = _normalize(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _solve_lp_best_response(a, M, active_indices=None, max_active=200):
    """
    Solve the LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0.
    Returns the optimal g (normalized to sum=1).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    # Scale M to account for normalization (a is normalized to sum=1)
    # The constraint is (a*g)[k] <= M where M = max(a*a)
    
    if active_indices is None:
        c = _fast_self_conv(a)
        active_indices = np.arange(len(c))
    
    active_indices = np.asarray(active_indices).astype(int)
    active_indices = active_indices[(active_indices >= 0) & (active_indices <= 2*n-2)]
    
    if len(active_indices) == 0:
        return _normalize(a)
    
    if len(active_indices) > max_active:
        c = _fast_self_conv(a)
        idx_sorted = np.argsort(c[active_indices])[::-1]
        active_indices = active_indices[idx_sorted[:max_active]]
    
    # Build constraint matrix A_con, b_con
    # For each active lag k: sum_i a[i]*g[i+k-valid] <= M
    # We need to express (a*g)[k] as a linear function of g
    # (a*g)[k] = sum_{i: valid} a[i] * g[k-i]
    # where valid means 0 <= i < n and 0 <= k-i < n
    
    # For a given k, the contributing indices of g are:
    # g[j] where j = k - i, so i = k - j
    # Valid: 0 <= k-j < n and 0 <= j < n
    # i.e., max(0, k-n+1) <= j <= min(k, n-1)
    
    # Build the matrix implicitly
    # A_con has shape (num_active, n)
    # A_con[r, j] = a[k-r - j] if 0 <= k-r-j < n, else 0
    # Wait, let me be more careful.
    
    # (a * g)[k] = sum_{j=0}^{n-1} a[k-j] * g[j] (with a[invalid] = 0)
    # So A_con[r, j] = a[k_r - j] if 0 <= k_r - j < n, else 0
    
    # This is a "-Toeplitz-like" matrix. For efficiency, let's build it.
    num_constraints = len(active_indices)
    
    # For n up to 4096 and num_constraints up to 200, we can build the matrix
    # Each row has at most n non-zeros (actually at most min(n, k+1, 2n-1-k) non-zeros)
    
    # Use sparse representation
    from scipy.sparse import csr_matrix
    
    # Build row by row
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(r)
                cols.append(j)
                vals.append(a[i])
    
    A_con = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    b_con = np.full(num_constraints, M, dtype=np.float64)
    
    # Solve LP
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            objective = cp.Maximize(cp.sum(g_var))
            constraints = [A_con @ g_var <= b_con]
            prob = cp.Problem(objective, constraints)
            
            # Try different solvers
            for solver in [cp.GLOP, cp.ECOS, cp.PDLP, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        break
                except Exception:
                    continue
            
            if prob.status in ["optimal", "optimal_inaccurate"] and g_var.value is not None:
                g_sol = np.asarray(g_var.value, dtype=np.float64).ravel()
                g_sol = np.clip(g_sol, 0, None)
                Sg = float(g_sol.sum())
                if Sg > 1e-12:
                    return g_sol / Sg
                else:
                    return _normalize(a)
            else:
                return _normalize(a)
        except Exception:
            pass
    
    # Fallback: use scipy linprog
    try:
        from scipy.optimize import linprog
        c_obj = -np.ones(n)  # minimize -sum(g)
        A_ub = A_con
        b_ub = b_con
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if res.success and res.x is not None:
            g_sol = np.clip(res.x, 0, None)
            Sg = float(g_sol.sum())
            if Sg > 1e-12:
                return g_sol / Sg
    except Exception:
        pass
    
    return _normalize(a)


def _lp_iteration(a, deadline, max_active=200, step_size=None, verbose=True):
    """
    Perform LP-based best response iterations.
    Returns (best_score, best_a, num_iterations).
    """
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    
    if step_size is None:
        step_size = 0.1
    
    num_iters = 0
    stagnant = 0
    
    while time.monotonic() < float(deadline) - 5.0:
        c = _fast_self_conv(a)
        M = float(c.max())
        
        # Find active indices
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) > max_active:
            active = np.argsort(c[active])[::-1][:max_active]
        
        # Solve LP
        g = _solve_lp_best_response(a, M, active_indices=active, max_active=max_active)
        
        # Check if g is different from a
        if np.linalg.norm(g - a) < 1e-10:
            stagnant += 1
            if stagnant > 5:
                break
        else:
            stagnant = 0
        
        # Update
        if step_size is not None:
            t = step_size
        else:
            t = 0.1
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = _exact_score(a_new)
        num_iters += 1
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            if verbose and (best_score - 1.5) < 0.01:
                print("LP iter {}: score={:.8f}".format(num_iters, best_score))
        elif sc > best_score * 1.01:
            # Revert if it got much worse
            t = t * 0.5
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            S = float(a_new.sum())
            if S < 1e-12:
                break
            a_new = a_new / S
            sc = _exact_score(a_new)
        
        a = a_new
        
        # Adaptive step size
        if num_iters % 20 == 0:
            if sc < best_score + 1e-9:
                step_size = min(0.3, step_size * 1.1)
            else:
                step_size = max(0.01, step_size * 0.8)
    
    return best_score, best_a, num_iters


def _multiplicative_update(a, n_iters=100, deadline=None, gamma=1.0):

        # Which is: convolve(a, delta_k) reversed... 
        
        # Simpler: use the fact that the subgradient of max(c) at a is:
        # grad[i] = (1/|A|) * sum_{k in A} (a[k-i] + a[i+(2n-2-k)]) ... this is getting too complex
        # Let me just use the FFT approach from _subgrad_exact
        
        score_cur, grad = _subgrad_exact(a, k_active=min(len(active), 10), tol=0.001)
        
        if not np.isfinite(score_cur) or np.linalg.norm(grad) < 1e-12:
            break
        
        # Multiplicative update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
        eta = gamma / (np.linalg.norm(grad) * np.sqrt(n))
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.clip(a_new, 0, None)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_score, best_a


def _perturb(a, rng, strength=0.3):
    """Perturb a sequence."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    mode = int(rng.integers(0, 4))
    
    if mode == 0:
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Anti-symmetric perturbation around center
        center = n / 2.0
        x = np.linspace(0, 1, n)
        anti = 1.0 + strength * 0.2 * np.sin(np.pi * (x - 0.5))
        q = a * anti
    else:
        # Random smooth
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        kern = np.exp(-((np.arange(K) - K//2)**2) / (2.0*(K//4+1)**2))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.2 * noise)
    
    return _normalize(np.clip(q, 0.0, None))


def _generate_inits(n, rng, parent=None, cap=15):
    """Generate diverse initial sequences."""
    inits = []
    
    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))
    
    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
    
    x = np.linspace(0.0, 1.0, n)
    _add(np.ones(n))
    
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    
    for f in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.5, 1.0]:
            _add(np.maximum(0.01, 1.0 + amp * np.cos(2.0 * np.pi * f * x)))
            _add(np.maximum(0.01, 1.0 + amp * np.sin(np.pi * f * x)))
    
    # Asymmetric profiles (important for beating the symmetric lower bound)
    for rate in [1.0, 2.0, 4.0]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1-x)) + 0.01)
    
    for _ in range(5):
        _add(np.maximum(0.01, 1.0 + 0.5 * rng.standard_normal(n) * np.exp(-2.0 * (x-0.5)**2 / 0.05)))
    
    return inits
