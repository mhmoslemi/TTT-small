def fast_conv(a):

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_sum
    # d c_k / d x_i: c_k = sum_j x[j]*x[k-j]
    # d c_k / d x_i = sum_j [delta(j,i)*x[k-j] + x[j]*delta(k-j,i)]
    # = x[k-i] + x[k-i] (when k-i != i) or x[k-i] (when k-i == i)
    # More simply: d c_k / d x_i = x[k-i] if 0<=k-i<n, times 2 except when 2i=k
    # Using convolution: the gradient of L w.r.t. x is 2 * conv(p, x_reversed)
    # where conv is the appropriate convolution
    
    # Actually, let's use: g_i = 2 * sum_k p_k * dc_k/dx_i
    # dc_k/dx_i = x[k-i] (if 0<=k-i<n) + x[k-i] (if 0<=k-i<n and k-i != i)
    # Wait no: c_k = sum_j x[j] x[k-j], so dc_k/dx_i:
    #   From the first factor: delta(j,i) -> x[k-i] (valid if 0<=k-i<n)
    #   From the second factor: delta(k-j,i) i.e. j=k-i -> x[k-i] (valid if 0<=k-i<n)
    #   These are the same term! So dc_k/dx_i = 2*x[k-i] if 0<=k-i<n and i!=k-i
    #   Or dc_k/dx_i = x[i] if i==k-i (i.e., k=2i)
    #   
    # For the sum: sum_k p_k * dc_k/dx_i = 2*sum_k p_k*x[k-i] - sum_k p_k*x[i]*delta(k,2i)
    # = 2*sum_k p_k*x[k-i] - p[2i]*x[i] (if 2i < M)
    #
    # The sum_k p_k * x[k-i] is a cross-correlation, which can be computed via FFT.
    
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    
    # conv_res[i] = sum_k p[k] * x[k-i] (for valid k)
    g_L = 2.0 * conv_res
    
    # Subtract the diagonal correction: g_L[i] -= p[2i] * x[i] for 2i < M
    for i in range(n):
        if 2 * i < M:
            g_L[i] -= p[2 * i] * x[i]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def multiplicative_descent(a, n_iters=5000, eta_start=0.01, eta_end=0.001, n_peaks=10):
    """
    Multiplicative gradient descent: a *= exp(-eta * grad)
    This naturally maintains non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        eta = eta_start * (eta_end / eta_start) ** progress
        
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if M_val <= sc_lb / (2.0 * n) * S * S + 1e-15:
            break
        
        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += a[js:je + 1][::-1]
        
        # dc/dx_i for the max: 2 * sum over active peaks of x[k-i]
        # Normalized gradient
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (2.0 * sg / S2 - 2.0 * M_val / S3)
        
        # Multiplicative update
        # a *= exp(-eta * grad)
        update = np.exp(-eta * grad)
        a = a * update
        a = np.clip(a, 1e-15, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

"""
Enhanced step function optimization using multi-resolution LSE optimization,
peak-targeted adaptive gradient descent with backtracking line search,
multiplicative updates, and coordinate descent fine-tuning.
Key innovations: (1) very high-beta LSE for precise peak targeting,
(2) backtracking line search for adaptive step sizing,
(3) peak-elimination coordinate descent, (4) multi-restart strategy.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_conv_and_score(a, M_fft):
    """Compute convolution, score, and peak in one pass."""
    n = len(a)
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
    S = float(a.sum())
    M_val = float(c.max())
    sc = 2.0 * n * M_val / (S * S)
    return c, S, M_val, sc


def lse_obj_grad(x, beta, M_fft=None, M2_fft=None):
    """LSE-smoothed objective with gradient for L-BFGS."""
    n = len(x)
    M = 2 * n - 1
    if M_fft is None:
        M_fft = 1 << (M - 1).bit_length()
    if M2_fft is None:
        M2_fft = 1 << (3 * n - 2 - 1).bit_length()

    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum

    # Gradient of L w.r.t. x
    rev_x = x[::-1]
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]

    g_L = 2.0 * conv_res
    # Diagonal correction: subtract p[2i]*x[i] where 2i < M
    idx = np.arange(n)
    valid = 2 * idx < M
    g_L[valid] -= p[2 * idx[valid]] * x[valid]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective with precomputed FFT sizes."""
    n = len(a0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    M2_fft = 1 << (3 * n - 3 - 1).bit_length()

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta, M_fft, M2_fft),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out


def peak_directed_linesearch(a, n_iters=5000, n_peaks=8, M_fft=None):
    """
    Peak-directed gradient descent with backtracking line search.
    Directly targets the top peaks using subgradient information.
    """
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    step = 0.01
    stall_count = 0

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
            step = min(step * 1.05, 0.03)
        else:
            stall_count += 1
            if stall_count % 20 == 0:
                step *= 0.7

        if sc - sc_lb < 1e-12:
            break

        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        # Compute subgradient for the peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (2.0 * sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(15):
            a_trial = a - alpha * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                alpha *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial < sc:
                a = a_trial
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try multiplicative update as fallback
            update = np.exp(-0.005 * grad)
            a = a * update
            a = np.clip(a, 0.0, 1000.0)
            S = float(a.sum())
            if S < 1e-12:
                break
            a = a / S

        step = max(step * 0.95, 1e-6)

    return best_a


def multiplicative_weight_descent(a, n_iters=5000, eta_start=0.02, eta_end=0.002,
                                  n_peaks=10, M_fft=None):
    """
    Multiplicative weight update: a *= exp(-eta * grad).
    Naturally maintains non-negativity.
    """
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        eta = eta_start * (eta_end / eta_start) ** progress

        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - sc_lb < 1e-12:
            break

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (2.0 * sg / S2 - 2.0 * M_val / S3)

        # Multiplicative update with clipping
        log_update = -eta * grad
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.clip(a, 1e-15, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def coordinate_descent_peaks(a, n_rounds=200, n_peaks=5, M_fft=None):
    """
    Coordinate descent: for each peak, try adjusting individual elements
    to reduce the peak value.
    """
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find the worst peak
        k_peak = int(np.argmax(c))

        # Elements contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(k_peak, n - 1)
        if je < js:
            break

        # For each contributing element, try small adjustments
        elements = list(range(js, je + 1))
        # Sort by contribution (largest first)
        contribs = [(j, a[j] * a[k_peak - j]) for j in elements]
        contribs.sort(key=lambda x: -x[1])

        improved_this_round = False
        for j, _ in contribs[:20]:  # Try top 20 contributors
            partner = k_peak - j
            # Try reducing a[j]
            for factor in [0.95, 0.9, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[j] *= factor
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                S_t = float(a_trial.sum())
                if S_t < 1e-12:
                    continue
                a_trial = a_trial / S_t
                sc_t = compute_score(a_trial)
                if sc_t < sc:
                    a = a_trial
                    sc = sc_t
                    improved_this_round = True
                    break
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        if not improved_this_round:
            # Try increasing some elements (to increase S without increasing peak much)
            # Find elements that don't contribute to the peak
            non_peak = [i for i in range(n) if i < js or i > je]
            if non_peak:
                # Try increasing a random non-peak element
                idx = non_peak[np.random.randint(len(non_peak))]
                a_trial = a.copy()
                a_trial[idx] *= 1.05
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                S_t = float(a_trial.sum())
                if S_t > 1e-12:
                    a_trial = a_trial / S_t
                    sc_t = compute_score(a_trial)
                    if sc_t < sc:
                        a = a_trial
                        sc = sc_t
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()

    return best_a


def fienup_geometric(a, n_iters=400, alpha_start=0.03, alpha_end=0.6, M_fft=None):
    """Geometric amplitude equalization Fienup iteration."""
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

                je = min(k, n - 1)
                if je >= js:
                    sg[js:je + 1] += a[js:je + 1][::-1]

            S2v = S2 * S2
            S3v = S2v * S2
            grad = (2.0 * n) * (2.0 * sg / S2v - 2.0 * M_val / S3v)
            gn = float(np.linalg.norm(grad))
            if gn > 1e-15:
                step = min(0.008, max((sc_curr - sc_lb) / (gn * gn + 1e-30), 1e-8))
                a -= step * grad
                a = np.maximum(a, 0.0)
                S = float(a.sum())
                if S < 1e-12:
                    break
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Hybrid spectral + peak-directed (25% time)
    Phase 2: High-beta LSE L-BFGS (40% time)
    Phase 3: Peak-directed line search + coordinate descent (20% time)
    Phase 4: Final polish with multiple strategies (15% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    M_fft = 1 << (2 * n_work - 1).bit_length()

    print(f"Working length: {n_work}")
    print(f"Parent available: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        print(f"  Parent score: {sc:.10f}")

    # ============================================================
    # Phase 1: Hybrid optimization (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Hybrid Spectral+Peak (deadline t={phase1_end - t_start:.0f}s) ===")

    # Start from parent (or uniform if no parent)
    a_init = parent.copy() if parent is not None else np.ones(n_work) * 0.5
    a_init = np.clip(a_init, 0.0, 1000.0)
    S = float(a_init.sum())
    if S > 1e-12:
        a_init = a_init / S

    # Run hybrid
    a_h = hybrid_optimize(a_init, n_spectral=200, n_peak=3000, M_fft=M_fft)
    sc = compute_score(a_h)
    update_best(sc, a_h, "HYBRID-1")

    # Try geometric Fienup from parent
    if time.monotonic() < phase1_end:
        a_g = fienup_geometric(a_init, n_iters=300, alpha_start=0.02, alpha_end=0.5, M_fft=M_fft)
        sc = compute_score(a_g)
        update_best(sc, a_g, "GEO-1")

    # Try multiplicative descent
    if time.monotonic() < phase1_end:
        a_m = multiplicative_weight_descent(a_init, n_iters=3000, eta_start=0.015,
                                            eta_end=0.003, n_peaks=8, M_fft=M_fft)
        sc = compute_score(a_m)
        update_best(sc, a_m, "MULT-1")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: High-beta LSE L-BFGS (40% time)
    # ============================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS LSE (deadline t={phase2_end - t_start:.0f}s) ===")

    # Multi-beta schedule: start moderate, go very high
    beta_schedule = [500, 2000, 5000, 15000, 50000, 100000, 200000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  Running L-BFGS with beta={beta}, remaining={remaining:.0f}s")

        # Estimate how many iterations we can afford
        # Each LSE eval takes roughly O(n log n) ~ a few ms for n~7000
        # L-BFGS does ~10-20 evals per iteration
        maxiter = min(5000, int(remaining * 15))
        maxfun = maxiter * 20

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        improved = update_best(sc, a_opt, f"LBFGS b={beta}")

        if not improved:
            # Try from parent as well for lower betas
            if parent is not None and beta <= 5000:
                sc2, a_opt2 = run_lbfgs(parent, beta, maxiter=maxiter//2, maxfun=maxfun//2)
                update_best(sc2, a_opt2, f"LBFGS-P b={beta}")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Peak-directed line search + coordinate descent (20% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak Line Search + Coord Descent (deadline t={phase3_end - t_start:.0f}s) ===")

    # Peak-directed with line search
    if time.monotonic() < phase3_end:
        a_pl = peak_directed_linesearch(best_a, n_iters=5000, n_peaks=10, M_fft=M_fft)
        sc = compute_score(a_pl)
        update_best(sc, a_pl, "PEAK-LINE")

    # Coordinate descent on peaks
    if time.monotonic() < phase3_end:
        a_cd = coordinate_descent_peaks(best_a, n_rounds=100, n_peaks=5, M_fft=M_fft)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD")

    # More peak-directed with different parameters
    if time.monotonic() < phase3_end:
        a_pl2 = peak_directed_linesearch(best_a, n_iters=3000, n_peaks=20, M_fft=M_fft)
        sc = compute_score(a_pl2)
        update_best(sc, a_pl2, "PEAK-LINE-20")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Final L-BFGS with highest beta
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        maxiter = min(3000, int(remaining * 10))
        sc, a_opt = run_lbfgs(best_a, 150000, maxiter=maxiter, maxfun=maxiter * 15)
        update_best(sc, a_opt, "FINAL-LBFGS-150k")

    # Final multiplicative descent
    if time.monotonic() < deadline - 60:
        a_fm = multiplicative_weight_descent(best_a, n_iters=2000, eta_start=0.005,
                                             eta_end=0.001, n_peaks=5, M_fft=M_fft)
        sc = compute_score(a_fm)
        update_best(sc, a_fm, "FINAL-MULT")

    # Final coordinate descent
    if time.monotonic() < deadline - 30:
        a_fc = coordinate_descent_peaks(best_a, n_rounds=50, n_peaks=3, M_fft=M_fft)
        sc = compute_score(a_fc)
        update_best(sc, a_fc, "FINAL-COORD")

    # Perturbation restarts: targeted perturbations at peak-contributing positions
    if time.monotonic() < deadline - 20:
        conv_best = fast_conv(best_a)
        peak_pos = int(np.argmax(conv_best))
        # Elements contributing to the peak
        js = max(0, peak_pos - n_work + 1)
        je = min(peak_pos, n_work - 1)

        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_p = best_a.copy()
            # Reduce elements at the peak position more aggressively
            pert_size = min(30, je - js + 1)
            pert_idx = np.random.choice(range(js, je + 1), pert_size, replace=False)
            a_p[pert_idx] *= (1.0 - 0.02 * (1 + trial * 0.5))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = float(a_p.sum())
            if S_p < 1e-12:
                continue
            a_p = a_p / S_p

            # Quick optimization from perturbed state
            a_r = peak_directed_linesearch(a_p, n_iters=1000, n_peaks=8, M_fft=M_fft)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    # Very final: one more LSE pass with the absolute highest beta
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 500000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "FINAL-LBFGS-500k")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
