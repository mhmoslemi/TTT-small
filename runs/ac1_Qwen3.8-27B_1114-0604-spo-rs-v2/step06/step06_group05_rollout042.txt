Hybrid LSE-LBFGS + Peak-Targeted LP optimization for minimizing
the normalized maximum self-convolution of a non-negative sequence.

from scipy.optimize import minimize
import cvxpy as cp
import time

    n = len(a)

    A = np.fft.rfft(a, M)

    max_scaled = float(scaled.max())

    p = exp_s / exp_s.sum()

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3

"""
Hybrid LSE-LBFGS + Peak-Targeted LP optimization for minimizing
2n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Key innovation: Alternates between smooth LSE gradient optimization and
direct LP-based peak reduction (linearizing the max at the current point),
with adaptive line search for guaranteed progress.
"""

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def _lbfgs_func(x, obj_grad, params):
    """Top-level wrapper for L-BFGS (avoids closure)."""
    return obj_grad(x, params)


def run_lbfgs(a0, obj_grad, params, maxiter=3000, maxfun=5000):
    """Run L-BFGS-B optimization with a top-level objective wrapper."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            _lbfgs_func, a0,
            args=(obj_grad, params),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def solve_peak_lp(a0, n_peaks):
    """Solve LP: minimize M s.t. linearized top-peak constraints, sum(a)=1, a>=0.
    The linearization is at a0: c_k(a) ≈ sum_j a0[j]*a[k-j]."""
    n = len(a0)
    c = fast_conv(a0)
    top_k = np.argsort(c)[::-1][:n_peaks]

    a_var = cp.Variable(n)
    M_var = cp.Variable()

    constraints = [a_var >= 0, cp.sum(a_var) == 1]

    for k_idx in range(len(top_k)):
        k = int(top_k[k_idx])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        a_idx = k - j_range
        terms = a0[j_range] * a_var[a_idx]
        constraints.append(cp.sum(terms) <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=20000)
        except Exception:
            return None

    if a_var.value is None:
        return None

    a_new = np.maximum(np.array(a_var.value).ravel(), 0.0)
    S = a_new.sum()
    if S < 1e-6:
        return None
    a_new = a_new / S
    return a_new


def line_search_blend(a_old, a_new, n_steps=7):
    """Find best convex combination: a = (1-t)*a_old + t*a_new."""
    best_t = 0.0
    best_sc = compute_score(a_old)
    for i in range(1, n_steps + 1):
        t = float(i) / float(n_steps)
        a_blend = (1.0 - t) * a_old + t * a_new
        a_blend = np.maximum(a_blend, 0.0)
        sb = float(a_blend.sum())
        if sb < 1e-6:
            continue
        a_blend = a_blend / sb
        sc = compute_score(a_blend)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        return a_old.copy()
    a_out = (1.0 - best_t) * a_old + best_t * a_new
    a_out = np.maximum(a_out, 0.0)
    sb = float(a_out.sum())
    if sb > 1e-6:
        a_out = a_out / sb
    return a_out


def subgradient_peak_step(a0, n_peaks=20, n_iters=200, step_init=0.02, decay=0.9995):
    """Subgradient descent targeting top peaks, with normalization."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_peaks]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partner = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * partner

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def gs_flat_target(a, n_iters=300, alpha_init=0.5, alpha_final=0.05):
    """GS-Fienup with flat convolution target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a
