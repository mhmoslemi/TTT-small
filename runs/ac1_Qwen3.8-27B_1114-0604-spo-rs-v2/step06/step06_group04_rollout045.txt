def solve_coord_x(c, a, i, n, lo=0.0, hi=None):
    # Minimize max(c + 2*(x-ai)*a_shifted + (x^2-ai^2)*e_2i) over x in [lo, hi]
    ai = a[i]
    if hi is None:
        hi = ai * 10 + 1.0
    
    # a_shifted: position k gets a[k-i] if valid
    a_shifted = np.zeros(2*n-1)
    k_start = max(0, i)
    k_end = min(2*n-1, 2*i+1)  # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
    valid_k = np.arange(k_start, k_end)
    valid_j = valid_k - i  # these are 0 to n-1
    # Exclude j = i (i.e., k = 2i)
    mask = valid_j != i
    a_shifted[valid_k[mask]] = a[valid_j[mask]]
    
    # e_2i
    pos_2i = 2*i
    has_sq = (pos_2i < 2*n - 1)
    
    def obj(x):
        c_new = c + 2*(x - ai)*a_shifted
        if has_sq:
            c_new = c_new.copy()
            c_new[pos_2i] += x*x - ai*ai
        return c_new.max()
    
    def subgrad(x):
        c_new = c + 2*(x - ai)*a_shifted
        if has_sq:
            c_new = c_new.copy()
            c_new[pos_2i] += x*x - ai*ai
        k_star = np.argmax(c_new)
        sg = 2*a_shifted[k_star]
        if has_sq and k_star == pos_2i:
            sg += 2*(x - ai)
        return sg
    
    # Bisection
    for _ in range(50):
        mid = (lo + hi) / 2
        s = subgrad(mid)
        if s > 0:
            hi = mid
        elif s < 0:
            lo = mid
        else:
            break
    return (lo + hi) / 2

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _normalize(a):

    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())

        return np.ones(a.size, dtype=np.float64) / a.size

# LSE value and gradient
def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

- f'(x) (where it exists) = max of the derivatives of the active functions

Key observation: at the optimum, the maximum is achieved by at least two "branches" (otherwise we could move x to reduce the max). The branches are:
- The quadratic: derivative = 2x
- A linear function: derivative = m_j
- The constant B: derivative = 0 (any x works)

- Case 1: quadratic = linear_j, and 2x = m_j → x = m_j/2, and x^2 + A = m_j*x + b_j
- Case 2: quadratic = B, and 2x <= 0 → x = 0 (and x^2 + A = B → A = B, which means c_{2i} - ai^2 = B)
- Case 3: linear_j = B, and m_j = 0 (or the max is achieved by multiple lines all with slope >= 0 and the constant)
- Case 4: x = 0 is optimal (boundary)
