import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.zeros_like(a)
    return a / S


def topk_gradient(c, a, K, threshold_frac=0.5):
    """
    Compute gradient of top-K sum of c = conv(a,a) w.r.t. a.
    Uses all peaks above threshold_frac * max(c).
    Returns (gradient, num_peaks_used, score_approx).
    """
    n = len(a)
    c_max = c.max()
    threshold = threshold_frac * c_max
    significant = np.where(c >= threshold)[0]
    K_use = min(K, len(significant))
    if K_use == 0:
        return np.zeros(n), 0, c_max
    top_idx = significant[np.argsort(c[significant])[::-1]][:K_use]
    
    M = 2 * n - 1
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
    
    grad *= 2.0 / K_use
    return grad, K_use, c_max


def topk_descent(a, n_iters=5000, K=20, threshold_frac=0.7, step_init=0.01,
                 backtracking_rho=0.5, max_bt=20):
    """
    Top-K subgradient descent with backtracking line search.
    Minimizes the sum of top-K convolution peaks.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - 2.0 * n / (2.0 * n - 1.0) < 1e-9:
            break

        grad, K_used, _ = topk_gradient(c, a, K, threshold_frac)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Backtracking line search
        a_old = a.copy()
        sc_old = sc
        found = False
        for bt in range(max_bt):
            s = step * (backtracking_rho ** bt)
            a_trial = a_old - s * grad
            a_trial = np.maximum(a_trial, 0.0)
            S_new = a_trial.sum()
            if S_new < 1e-12:
                continue
            a_trial = a_trial / S_new
            sc_trial = compute_score(a_trial)
            if sc_trial < sc_old - 1e-12:
                a = a_trial
                found = True
                break
        if not found:
            step *= backtracking_rho
            if step < 1e-10:
                break
            a = a_old - step * grad
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            step = min(step * 1.1, 0.05)

        if it % 500 == 0:
            print(f"    topK iter {it}: score={sc:.8f}, step={step:.6f}, K_used={K_used}")

    return best_a, best_sc


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def spectral_shape_selfconv(a, n_iters=200, target='triangular'):
    """
    Spectral shaping that targets the self-convolution directly.
    Modifies the spectrum to make b = a*a flatter.
    """
    n = len(a)
    a = normalize(a.copy())
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Target: triangular fading (matches the "ideal" linear convolution shape)
    x = np.linspace(0, 1, M)
    if target == 'triangular':
        target_time = np.abs(2 * x - 1)
    else:
        target_time = np.ones(M)

    target_time = target_time / target_time.sum()

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = 0.1 + 0.6 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_floor = 1e-30

        # Compute current b and compare to target
        b = fast_conv(a)
        b_norm = b / b.sum()

        # In frequency domain: want B = |A|^2 to be "shaped" so that b matches target
        # B_target = FFT(target_time)
        B_target = np.fft.rfft(target_time, M_fft)
        # We want |A|^2 proportional to B_target, but B_target may be negative
        # Use: |A|^2_new = (1-alpha)*|A|^2 + alpha * max(B_target, 0) * scale
        B_pos = np.maximum(B_target.real, 0.0)
        scale = P.sum() / max(B_pos.sum(), 1e-30)
        P_new = (1.0 - alpha) * P + alpha * B_pos * scale
        P_new = np.maximum(P_new, P_floor)

        ratio = np.sqrt(P_new / np.maximum(P, P_floor))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

            freqs = np.fft.rfftfreq(M_fft) * M_fft

def coord_descent_fine(a, n_sweeps=5, n_top=30):
    """Coordinate descent: reduce top contributors to the peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.8, 0.95]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def make_ramp(n, direction=1):
    """Create a linear ramp sequence (non-palindromic)."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def make_gaussian_bump(n, sigma_frac=0.3, center_frac=0.4):
    """Gaussian bump offset from center (non-palindromic)."""
    x = np.arange(n, dtype=np.float64) / n
    center = center_frac
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_smooth_random(n, rng, n_freqs=10):
    """Smooth random sequence using low-frequency sinusoids."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for m in range(n_freqs):
        freq = rng.integers(1, 15)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = rng.uniform(0.1, 0.5) / (1.0 + 0.3 * m)
        a += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def make_half_cosine(n, shift=0.0):
    """Half-cosine window with a shift to break symmetry."""
    x = np.arange(n, dtype=np.float64) / n
    x_shifted = (x - shift) % 1.0
    a = np.maximum(np.cos(np.pi * (2 * x_shifted - 1)), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s) ***")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent)
    else:
        best_a = make_ramp(8192)
        best_score = compute_score(best_a)
        print(f"No parent, using ramp n=8192, score={best_score:.8f}")

    n_parent = len(best_a) if best_a is not None else 8192

    # ============================================================
    # Phase 1: Optimize from parent (40% time)
    # ============================================================
    p1_end = t_start + 0.40 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Optimize parent (budget {p1_end - t_start:.0f}s) ===")
    sc1, a1 = multi_stage_optimize(best_a, p1_end - t_start, rng, tag="parent")
    update(sc1, a1)

    # ============================================================
    # Phase 2: Diverse starting points (30% time)
    # ============================================================
    p2_end = t_start + 0.70 * (budget_s - 30.0)
    print(f"\n=== Phase 2: Diverse starts (budget {p2_end - t_start:.0f}s) ===")

    n_current = len(best_a)
    diverse_starts = []

    # Ramp (non-palindromic)
    diverse_starts.append(("ramp_up", make_ramp(n_current, 1)))
    diverse_starts.append(("ramp_down", make_ramp(n_current, -1)))
    # Gaussian bumps at different positions
    diverse_starts.append(("gauss_0.3", make_gaussian_bump(n_current, 0.3, 0.3)))
    diverse_starts.append(("gauss_0.5", make_gaussian_bump(n_current, 0.25, 0.5)))
    diverse_starts.append(("gauss_0.7", make_gaussian_bump(n_current, 0.3, 0.7)))
    # Smooth random
    diverse_starts.append(("smooth_rand1", make_smooth_random(n_current, rng, 10)))
    diverse_starts.append(("smooth_rand2", make_smooth_random(n_current, rng, 5)))
    # Half-cosine with shift
    diverse_starts.append(("halfcos_0.1", make_half_cosine(n_current, 0.1)))
    diverse_starts.append(("halfcos_0.3", make_half_cosine(n_current, 0.3)))
    # Blend: 50% parent + 50% ramp
    blend = 0.5 * normalize(best_a) + 0.5 * make_ramp(n_current, 1)
    diverse_starts.append(("blend_parent_ramp", normalize(blend)))

    # Allocate time per diverse start
    sub_budget = (p2_end - t_start) / len(diverse_starts)
    sub_budget = max(sub_budget, 30.0)  # minimum 30s each

    for i, (name, a_start) in enumerate(diverse_starts):
        if time.monotonic() > p2_end - 15:
            break
        actual_budget = min(sub_budget, p2_end - 15 - time.monotonic())
        if actual_budget < 20:
            break
        sc_d, a_d = multi_stage_optimize(a_start, actual_budget, rng, tag=f"div:{name}")
        update(sc_d, a_d)

    # ============================================================
    # Phase 3: Different n values (15% time)
    # ============================================================
    p3_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Different n (budget {p3_end - t_start:.0f}s) ===")

    n_targets = []
    if n_current < 6000:
        n_targets.append(6000)
    if n_current < 12000:
        n_targets.append(12000)
    if n_current < 16000:
        n_targets.append(16000)
    if n_current > 8000:
        n_targets.append(8000)

    for new_n in n_targets[:3]:  # Try at most 3 different n
        if time.monotonic() > p3_end - 30:
            break
        actual_budget = min(60.0, p3_end - 30 - time.monotonic())
        if actual_budget < 20:
            break

        a_interp = interpolate_to_n(best_a, new_n)
        a_interp = normalize(a_interp)
        sc_i = compute_score(a_interp)
        print(f"  Interpolated to n={new_n}: score={sc_i:.8f}")
        update(sc_i, a_interp)

        # Quick optimization on interpolated
        if time.monotonic() < p3_end - 15:
            sc_n, a_n = multi_stage_optimize(a_interp, actual_budget, rng, tag=f"n={new_n}")
            update(sc_n, a_n)

    # ============================================================
    # Phase 4: Final intensive polish (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")
    p4_end = deadline

    # Top-K with more iterations
    if time.monotonic() < p4_end - 60:
        remaining = p4_end - 60 - time.monotonic()
        iters = max(1000, int(remaining * 2000))
        a_k, sc_k = topk_descent(best_a, n_iters=min(iters, 15000), K=10,
                                  threshold_frac=0.5, step_init=0.005)
        update(sc_k, a_k)

    # High-beta LSE
    if time.monotonic() < p4_end - 40:
        remaining = p4_end - 40 - time.monotonic()
        maxiter = max(1000, int(remaining * 8))
        sc_l, a_l = run_lbfgs(best_a, 30000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc_l, a_l)

    # Coordinate descent
    if time.monotonic() < p4_end - 15:
        a_c, sc_c = coord_descent_fine(best_a, n_sweeps=10, n_top=40)
        update(sc_c, a_c)

    # One more LSE after coord descent
    if time.monotonic() < p4_end - 5:
        sc_l2, a_l2 = run_lbfgs(best_a, 50000, maxiter=2000, maxfun=30000)
        update(sc_l2, a_l2)

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
