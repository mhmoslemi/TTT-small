def fast_conv(a):

def adam_optimize(a0, beta, top_k, n_iters, lr_init, deadline, best_score, best_a):
    """Adam optimizer with top-k LSE gradient and projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a0.copy(), best_score, best_a
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    lr_decay = 0.99995
    best_sc = compute_score(a)
    if best_sc < best_score:
        best_score = best_sc
        best_a = a.copy()
    no_improve = 0
    patience = 300

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        sc, grad = topk_lse_score_grad(a, beta, top_k)

        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project: clip to non-negative, renormalize
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a.copy()
            S_new = S
        a_new = a_new / S_new
        a = a_new
        lr *= lr_decay

        # Check improvement
        if it % 50 == 0 or it < 10:
            sc_check = compute_score(a)
            if sc_check < best_sc - 1e-10:
                best_sc = sc_check
                best_a = a.copy()
                no_improve = 0
                if best_sc < best_score:
                    best_score = best_sc
                    print(f"    Adam: new best {best_sc:.8f} at iter {it}, beta={beta}")
            else:
                no_improve += 50

        if no_improve > patience:
            # Perturb and reset
            rng_local = np.random.default_rng(it)
            a = a + 0.01 * rng_local.standard_normal(n)
            a = np.maximum(a, 0.0)
            S = a.sum()
            a = a / S
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0

    return a, best_score, best_a


def peak_deflation(a0, n_steps, step_init, deadline):
    """Directly reduce top peaks using subgradient steps."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peak
        k_star = int(np.argmax(c))

        # Subgradient of c[k_star] w.r.t. a
        # dc_k/da[j] = a[k-j] + a[j-k] (with a[.] = 0 outside [0,n-1])
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            # a[k_star - j] for j in [j_start, j_end]
            idx = np.arange(j_start, j_end + 1)
            partners = k_star - idx
            valid = (partners >= 0) & (partners < n)
            sg[idx[valid]] += a[partners[valid]]
            # a[j - k_star] for j in [j_start, j_end]
            partners2 = idx - k_star
            valid2 = (partners2 >= 0) & (partners2 < n)
            sg[idx[valid2]] += a[partners2[valid2]]

        # Subtract the mean gradient component (for the 1/S^2 term)
        # Full gradient of score: (2n/S^2) * sg - (4n*c_max/S^3) * ones * S
        # Simplified: just use sg direction
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:

        a -= step * sg / sg_norm
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        step *= 0.999

    return best_a


def block_swap_search(a0, n_swaps, block_sizes, deadline):
    """Local search using block swaps."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a0.copy()
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)

    for it in range(n_swaps):
        if time.monotonic() > deadline:
            break

        # Pick a random block size and position
        bs = int(block_sizes[rng.integers(len(block_sizes))])
        if bs >= n:
            bs = n // 2
        pos1 = int(rng.integers(0, max(1, n - bs)))
        pos2 = int(rng.integers(0, max(1, n - bs)))
        if pos1 == pos2:
            continue

        # Try swapping blocks
        a_new = a.copy()
        block1 = a_new[pos1:pos1 + bs].copy()
        block2 = a_new[pos2:pos2 + bs].copy()
        a_new[pos1:pos1 + bs] = block2
        a_new[pos2:pos2 + bs] = block1

        # Also try a small perturbation variant
        a_pert = a_new + 0.005 * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()

        sc1 = compute_score(a_new)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a_new.copy()
            a = a_new
        else:
            sc2 = compute_score(a_pert)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_pert.copy()
                a = a_pert

        if it % 500 == 0 and it > 0:
            print(f"    Block swap iter {it}: best={best_sc:.8f}")

    return best_a


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=30000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def gs_iteration(a, M, T_mag, T_phase, alpha, n):
    """Single GS iteration."""
    A = np.fft.fft(a, M)
    A_mag = np.maximum(np.abs(A), 1e-20)
    A_phase = np.angle(A)

    new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
    new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

    A_new = new_mag * np.exp(1j * new_phase)
    a_new = np.fft.ifft(A_new, M).real[:n]
    a_new = np.maximum(a_new, 0.0)
    S = float(a_new.sum())
    if S < 1e-12:
        return a
    return a_new / S

        a_work, best_score, best_a = adam_optimize(
            a_work, beta, top_k, n_iters=50000,
            lr_init=0.02, deadline=phase2_end,
            best_score=best_score, best_a=best_a
        )
        a_work = best_a.copy()

        # Peak deflation after Adam
        if time.monotonic() < phase2_end:
            a_defl = peak_deflation(a_work, n_steps=5000, step_init=0.02,
                                    deadline=phase2_end)
            sc = compute_score(a_defl)
            update_best(sc, a_defl)
            if sc < compute_score(a_work):
                a_work = a_defl

        print(f"  After Adam+deflation (beta={beta}): best={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Block swap local search (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Block Swap Local Search ===")

    block_sizes = [1, 2, 3, 5, 8, 13, 21, 34]
    a_swapped = block_swap_search(best_a, n_swaps=50000, block_sizes=block_sizes,
                                   deadline=phase3_end)
    sc = compute_score(a_swapped)
    update_best(sc, a_swapped)

    # Also try smaller perturbations after swap
    if time.monotonic() < phase3_end:
        for strength in [0.01, 0.005, 0.002]:
            if time.monotonic() > phase3_end:
                break
            a_pert = best_a + strength * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            a_ds, best_score, best_a = adam_optimize(
                a_pert, 50000, 5, n_iters=20000,
                lr_init=0.01, deadline=phase3_end,
                best_score=best_score, best_a=best_a
            )

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: GS + LSE alternating (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 4: GS + LSE Alternating ===")

    for round_i in range(10):
        if time.monotonic() > phase4_end:
            break

        # GS step
        a_gs = gs_optimize(best_a, n_iters=300, deadline=phase4_end)
        sc = compute_score(a_gs)
        update_best(sc, a_gs)

        # LSE-LBFGS step
        if time.monotonic() > phase4_end:
            break
        beta = 20000 + round_i * 10000
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=3000, maxfun=10000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Peak deflation
        if time.monotonic() > phase4_end:
            break
        a_defl = peak_deflation(best_a, n_steps=3000, step_init=0.01,
                                deadline=phase4_end)
        sc = compute_score(a_defl)
        update_best(sc, a_defl)

        print(f"  Round {round_i}: best={best_score:.8f}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish - high-beta LSE + fine peak deflation (15% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    for beta in [50000, 100000, 200000, 500000]:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Final LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final peak deflation with small steps
    if time.monotonic() < deadline:
        a_defl = peak_deflation(best_a, n_steps=10000, step_init=0.005,
                                deadline=deadline)
        sc = compute_score(a_defl)
        update_best(sc, a_defl)

    # One more Adam pass with very high beta
    if time.monotonic() < deadline:
        a_final, best_score, best_a = adam_optimize(
            best_a, 100000, 3, n_iters=30000,
            lr_init=0.005, deadline=deadline,
            best_score=best_score, best_a=best_a
        )

    print(f"After Phase 5: best = {best_score:.8f}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
