"""
Optimized step function minimization using LSE-Adam with adaptive beta,
peak-directed coordinate descent, and multi-length exploration.
Improves upon previous Fienup+L-BFGS approach with more robust gradient-based optimization.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(a, beta):
    """LSE-smoothed objective and exact gradient (a normalized to sum=1)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Work with normalized a (sum = 1)
    # Objective: 2n * LSE_beta(c)
    c_max = float(c.max())
    # Clip for numerical stability
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -100, 0)  # all <= 0
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    val = 2.0 * n * L  # since S=1, objective = 2n * L
    
    # p_k
    p = exp_scaled / exp_sum
    p_max = p.max()
    
    # Gradient of L w.r.t. a_i:
    # dL/da_i = 2 * sum_k p[k] * a[k-i] (valid indices)
    # = 2 * conv(rev_a, p)[n-1+i]
    rev_a = a[::-1]
    M3 = 3 * n - 2
    M3_fft = 1 << (M3 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M3_fft)
    P = np.fft.rfft(p, M3_fft)
    conv_res = np.fft.irfft(RA * P, M3_fft)[:M3]
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    # Since we're working in normalized space (S=1), the gradient of 2n*L is just 2n * g_L
    # But we also need to account for the constraint sum(a)=1.
    # Using Lagrange multiplier: grad = 2n*g_L - lambda * ones
    # At optimum, sum of a * grad = 0 (tangential to constraint)
    # lambda = sum(a * 2n*g_L) / n = 2*sum(a*g_L)
    # So effective grad = 2n*g_L - 2*sum(a*g_L)*ones
    # Actually for projected gradient, we just project: subtract mean weighted by a
    # Simpler: just use the raw gradient and let the normalization handle it
    
    grad = 2.0 * n * g_L
    return val, grad

def adam_lse_optimize(a0, beta, n_iters, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective with non-negativity constraint."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a0
    a = a / S  # normalize to sum=1
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for t in range(n_iters):
        val, grad = lse_objective_and_grad(a, beta)
        
        # LSE value is a lower bound... no, upper bound on the max
        # val = 2n * LSE_max(c) >= 2n * max(c) = true objective (for S=1)
        # So val >= true_score. Use compute_score for tracking.
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_shave(a, n_peaks=10, n_iters=500, lr=0.01):
    """Directly reduce the top peaks of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        M_val = float(c.max())
        sc = 2.0 * n * M_val  # S=1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        # Subgradient: for active peaks, gradient of max is average of their gradients
        # grad_i = (1/|idx|) * sum_{k in idx} 2*a[k-i]
        sg = np.zeros(n)
        for k in idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
        sg /= len(idx)
        
        # Projected gradient step (stay on sum=1 simplex)
        a -= lr * sg
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay learning rate
        lr *= 0.999
    
    return best_a

def fienup_variant(a, n_iters=300, alpha=0.5):
    """Simple Fienup spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def lbfgs_lse(a0, beta, maxiter=5000):
    """L-BFGS on LSE objective."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        return a0
    a0 = a0 / S
    
    def obj(x):
        val, grad = lse_objective_and_grad(x, beta)
        # Project gradient to tangent space of {sum=1, a>=0}
        # Simple: subtract mean
        grad = grad - np.mean(grad)
        return val, grad
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True,
                   options={'ftol': 1e-16, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxiter * 20})
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S < 1e-6:
        return a0
    return a_out / S
