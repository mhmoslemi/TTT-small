```
def _project_simplex(v):
    v=np.asarray(v,float)
    if v.size==1: return np.array([1.0])
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    rho=np.arange(1, v.size+1)
    cond=(u-css/rho)>0
    i=np.where(cond)[0][-1]
    theta=css[i]/(i+1)
    w=np.maximum(v-theta,0)
    s=w.sum()
    if s<=0: w=np.ones(v.size)/v.size
    else: w/=s
    return w
```

```
def _project_simplex(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    if v.size==0: return np.empty(0)
    if v.size==1: return np.array([1.0])
    u=np.sort(v)[::-1]
    css=np.cumsum(u)
    rho=np.arange(1,v.size+1,dtype=np.float64)
    cond=(u - css/rho) > 0
    if not np.any(cond):
       return np.ones(v.size)/v.size
    i=np.where(cond)[0][-1]
    theta=css[i]/(i+1)
    w=np.maximum(v-theta,0.0)
    s=w.sum()
    if s<=0: return np.ones(v.size)/v.size
    return w/s
```

```python
"""
Search for non-negative coefficient sequences minimizing

    2*n*max(conv(a,a)) / sum(a)**2

using a blend of:
  * FFT-based exact active scoring,
  * alternating fixed-weight smoothed convolution minimization,
  * true log-sum-exp smoothing of the maximum with correct softmax gradients,
  * active-set least-squares weight balancing,
  * multi-start / perturb-and-resume heuristics.

The code is deliberately self-contained and returns a finite list of
non-negative floats after the allotted wall-clock time.
"""

import numpy as np
import time
from scipy.optimize import minimize


class _TimeLimit(Exception):
    pass


_LAST_Y = [None]
_LAST_Y_COUNT = [0]


def _current_time():
    return time.monotonic()


def _check_time(deadline):
    if deadline is not None and _current_time() >= float(deadline):
        raise _TimeLimit()


def _save_y(y):
    _LAST_Y_COUNT[0] += 1
    if _LAST_Y_COUNT[0] % 10 == 0:
        _LAST_Y[0] = np.asarray(y, dtype=np.float64).copy()


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    a = a / s
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 0.01:
        a = a * (1.0 / max(s, 1e-12))
    return a / float(a.sum())


def _resample(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(max(1, m))
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / m
    if a.size == m:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, m)
    return np.interp(x1, x0, a).astype(np.float64)


def _softmax(y):
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.empty(0, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.0, posinf=50.0, neginf=-50.0)
    y = y - float(np.max(y))
    exp = np.exp(y)
    s = float(exp.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(y.size, dtype=np.float64) / y.size
    a = exp / s
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / y.size
    return a / s


def _fft_self_conv(a, N=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    if N is None:
        N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:M]
    c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    cs = float(c.sum())
    if cs > 1e-30:
        c = c / cs
    return c


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = _normalize(a)
    c = _fft_self_conv(a)
    return float(2.0 * n * float(c.max()))


def _conv_w(a, w, N, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, N)
    Wr = np.fft.rfft(w, N)
    full = np.fft.irfft(Ar * Wr, N)
    out = full[n - 1:2 * n - 1]
    return np.maximum(np.asarray(out, dtype=np.float64), 0.0)


def _prepare_w_fft(w, n):
    w = np.asarray(w, dtype=np.float64).ravel()
    expected = 2 * n - 1
    if w.size != expected:
        w = _resample(w, expected)
    w = np.maximum(w, 0.0)
    s = float(w.sum())
    if s <= 1e-30:
        w = np.ones(expected, dtype=np.float64) / expected
    else:
        w = w / s
    N = _next_pow2(3 * n)
    return np.fft.rfft(w, N), N, w


def _fixed_w_value(a, Wfft, N, n):
    R = _conv_w(a, np.fft.irfft(Wfft, N), N, n)
    L = float(np.dot(R, a))
    return float(2.0 * n * max(0.0, L))


def _fixed_w_fun(y, Wfft, N, n, deadline):
    _check_time(deadline)
    a = _softmax(y)
    R = _conv_w(a, np.fft.irfft(Wfft, N), N, n)
    L = float(np.dot(R, a))
    if not np.isfinite(L) or L < 0.0:
        L = abs(L) if np.isfinite(L) else 0.0
    val = float(2.0 * n * L)
    grad = float(4.0 * n) * a * (R - L)
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)) or not np.isfinite(val):
        return 1e12, np.zeros_like(y, dtype=np.float64)
    _save_y(y)
    return val, grad


def _lse_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    beta = float(beta)
    m = c.size
    if m == 0:
        return np.empty(0, dtype=np.float64)
    cm = float(c.max())
    if beta <= 1e-12 or cm <= 0.0:
        return np.ones(m, dtype=np.float64) / m
    z = beta * (c - cm)
    z = np.maximum(z, -35.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0
        return w
    return e / s


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - css / rho) > 0.0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    i = int(np.where(cond)[0][-1])
    theta = css[i] / float(i + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-30:
        return np.ones(n, dtype=np.float64) / n
    return w / s


def _active_ls_weights(c, a, K=32, max_iter=40):
    c = np.asarray(c, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    K = int(min(K, c.size, 64))
    if K <= 0:
        return np.ones(c.size, dtype=np.float64) / c.size
    if K == 1:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w

    idx = np.argsort(c)[-K:]
    M = float(c.max())

    B = np.zeros((n, K), dtype=np.float64)
    for mm in range(K):
        off = int(idx[mm])
        if off < n:
            B[:n - off, mm] = a[off:]

    C = M * a
    w = np.ones(K, dtype=np.float64) / K

    # Estimate a Lipschitz constant from B^T B.
    try:
        BTB = B.T @ B
        eig = np.linalg.eigvalsh(BTB)
        L = float(max(1e-12, eig[-1]))
    except Exception:
        L = 1e-3
    step = min(1e9, 1.0 / max(1e-12, 2.0 * L))

    f = None
    for _ in range(int(max_iter)):
        Bw = B @ w
        r = Bw - C
        f_new = 0.5 * float(np.dot(r, r))
        if f is None:
            f = f_new
        grad = B.T @ r
        w_new = _project_simplex(w - step * grad)
        r_new = (B @ w_new) - C
        f_candidate = 0.5 * float(np.dot(r_new, r_new))

        if f_candidate <= f + 1e-14:
            if float(np.max(np.abs(w_new - w))) < 1e-7:
                w = w_new
                f = f_candidate
                break
            w = w_new
            f = f_candidate
            step = min(1e9, step * 1.15)
        else:
            step *= 0.5
            if step < 1e-14:
                break

    wfull = np.zeros(c.size, dtype=np.float64)
    wfull[idx] = w
    s = float(wfull.sum())
    if s <= 1e-30:
        return np.ones(c.size, dtype=np.float64) / c.size
    return wfull / s


def _lse_val_grad_y(y, beta, Nw, Nc, deadline):
    _check_time(deadline)
    a = _softmax(y)
    n = a.size
    c = _fft_self_conv(a, Nc)
    cm = float(c.max())
    beta = float(beta)

    if beta <= 1e-12 or cm <= 0.0:
        L = cm
        w = np.ones(c.size, dtype=np.float64) / c.size
    else:
        z = beta * (c - cm)
        z = np.maximum(z, -35.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            L = cm
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
            L = cm + float(np.log(s)) / beta

    R = _conv_w(a, w, Nw, n)
    val = float(2.0 * n * L)
    grad = float(4.0 * n) * a * (R - L)
    grad = grad - float(np.mean(grad))
    if not np.all(np.isfinite(grad)) or not np.isfinite(val):
        return 1e12, np.zeros_like(y, dtype=np.float64)
    _save_y(y)
    return val, grad


def _lbfgs_run(func, y0, args, deadline, maxfun):
    y = np.asarray(y0, dtype=np.float64).ravel().copy()
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.0, posinf=50.0, neginf=-50.0)
    y = y - float(np.mean(y))
    _LAST_Y[0] = y.copy()
    _LAST_Y_COUNT[0] = 0

    maxfun = int(max(30, min(5000, maxfun)))
    opts = {
        "maxfun": maxfun,
        "maxiter": maxfun,
        "maxls": 20,
        "disp": False,
        "ftol": 1e-13,
        "gtol": 1e-9,
    }

    try:
        res = minimize(func, y, jac=True, args=args, method="L-BFGS-B", options=opts)
        if getattr(res, "x", None) is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if np.all(np.isfinite(x)):
                y = x
    except _TimeLimit:
        pass
    except Exception:
        pass

    if _LAST_Y[0] is not None:
        saved = np.asarray(_LAST_Y[0], dtype=np.float64).ravel()
        if saved.size == y.size and np.all(np.isfinite(saved)):
            y = saved

    y = y - float(np.mean(y))
    return y


def _fixed_w_minimize(a, Wfft, N, n, deadline):

    a = np.maximum(a, 1e-300)

    y = np.log(a)
    rem = max(0.0, float(deadline) - _current_time())
    if rem < 5.0:

    maxfun = int(max(40, min(3500, rem * 50.0)))
    y = _lbfgs_run(_fixed_w_fun, y, (Wfft, N, n, deadline), deadline, maxfun)
    return _softmax(y)


def _lse_minimize(a, beta, deadline):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    a = np.maximum(a, 1e-300)
    a = a / float(a.sum())
    y = np.log(a)

def _smooth_noise(n, rng):
    N = _next_pow2(n)
    freqs = np.fft.rfftfreq(N)
    slope = float(rng.uniform(1.5, 3.0))
    power = 1.0 / ((np.abs(freqs) * n + 1.0) ** slope)
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=power.size)
    spec = power * np.exp(1j * phase)
    noise = np.fft.irfft(spec, N)[:n]
    noise = noise - float(noise.mean())
    std = float(np.std(noise))
    if std > 1e-12:
        noise = noise / std
    else:
        noise = rng.standard_normal(n)
        noise = noise - float(noise.mean())
    return noise.astype(np.float64)


def _perturb_start(a, rng, strength=0.6):
    a = _normalize(a)
    noise = _smooth_noise(a.size, rng)
    q = a * np.exp(0.35 * float(strength) * noise)
    return _normalize(q)


def _add_init(inits, arr, cap=24):
    if len(inits) >= int(cap):
        return
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, None)
    if float(arr.sum()) > 1e-12:
        inits.append(_normalize(arr))


def _structured_inits(n, rng, parent=None):
    inits = []
    n = int(max(64, n))

    _add_init(inits, np.ones(n, dtype=np.float64))

    if parent is not None:
        p = _resample(parent, n)
        _add_init(inits, p)
        for t in (0.05, 0.15, 0.30):
            u = np.ones(n, dtype=np.float64)
            _add_init(inits, (1.0 - t) * _normalize(p) + (t / n) * u)
        for fr in (0.01, 0.025, 0.06):
            sigma = max(1.0, fr * n)
            half = int(min(n - 1, max(3, 6 * sigma + 1)))
            if half % 2 == 0:
                half += 1
            r = np.arange(-half // 2, half // 2 + 1, dtype=np.float64)
            kern = np.exp(-0.5 * (r / sigma) ** 2)
            kern = kern / float(kern.sum())
            Nk = _next_pow2(n + half)
            A = np.fft.rfft(p, Nk)
            K = np.fft.rfft(kern, Nk)
            sm = np.fft.irfft(A * K, Nk)[:n]
            _add_init(inits, sm)

    x = np.linspace(0.0, 1.0, n)
    for p in (-1.2, -0.6, -0.25, 0.0, 0.25, 0.6, 1.2):
        _add_init(inits, np.ones(n) + 0.35 * p * x)
        _add_init(inits, (np.maximum(x, 1e-3) ** p) + 0.02)
        _add_init(inits, (np.maximum(1.0 - x, 1e-3) ** p) + 0.02)

    for f in (1, 2, 3, 5):
        for ph in (0.0, 1.0, 2.1):
            _add_init(inits, 1.0 + 0.45 * np.sin(2.0 * np.pi * f * x + ph))
            _add_init(inits, 1.0 + 0.35 * (np.cos(np.pi * f * x)))

    for _ in range(6):
        amp = float(rng.uniform(0.35, 1.35))
        noise = _smooth_noise(n, rng)
        _add_init(inits, np.maximum(1.0 + amp * noise, 0.0))

    for _ in range(2):
        _add_init(inits, 0.05 + 0.95 * rng.random(n))

    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / n)
    return inits


def _choose_next_a(a, a2, score_before, sc2, best_score, q_before=None, q_after=None):
    a = _normalize(a)
    a2 = _normalize(a2)
    if sc2 < best_score - 1e-12:
        return _normalize(0.85 * a2 + 0.15 * a)
    if sc2 < score_before - 1e-12:
        return _normalize(0.75 * a2 + 0.25 * a)
    if q_before is not None and q_after is not None:
        if q_after < q_before - 1e-10 and sc2 < score_before + 0.0025:
            return _normalize(0.50 * a2 + 0.50 * a)
        if q_after < q_before - 1e-10:
            return _normalize(0.15 * a2 + 0.85 * a)
    if sc2 < score_before + 0.0007:
        return _normalize(0.05 * a2 + 0.95 * a)
    return a

        c = _fft_self_conv(a)
        score_before = float(2.0 * n * c.max())
        if score_before < best_score - 1e-12:
            best_score = score_before
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 5 and int(rng.integers(0, 3)) == 0:
            a = _perturb_start(a, rng, strength=float(rng.uniform(0.55, 1.0)))
            scp = _exact_score(a)
            if scp < best_score - 1e-12:
                best_score = scp
                best_a = a.copy()
                no_improve = 0
            else:
                a = _normalize(0.8 * a + 0.2 * best_a)
            no_improve = 0
            continue

        if mode.startswith("f"):
            wlse = _lse_weights(c, beta)
            if beta >= 20.0 and n <= 20000:
                try:
                    wlso = _active_ls_weights(c, a, K=40, max_iter=40)
                    alpha = 0.20 + 0.50 * min(1.0, beta / 1000.0)
                    w = alpha * wlso + (1.0 - alpha) * wlse
                    w = np.maximum(w, 0.0)
                    w = w / float(w.sum())
                except Exception:
                    w = wlse
            else:
                w = wlse
            Wfft, Nw, _ = _prepare_w_fft(w, n)
            q_before = _fixed_w_value(a, Wfft, Nw, n)
            a2 = _fixed_w_minimize(a, Wfft, Nw, n, stage_end - 2.0)
            q_after = _fixed_w_value(a2, Wfft, Nw, n)
        else:
            a2 = _lse_minimize(a, beta, stage_end - 2.0)
            q_before = None
            q_after = None

        a2 = _normalize(a2)
        sc2 = _exact_score(a2)

        if sc2 < best_score - 1e-12:
            best_score = sc2
            best_a = a2.copy()
            no_improve = 0

        a = _choose_next_a(a, a2, score_before, sc2, best_score, q_before, q_after)

        if _current_time() > stage_end - 2.0:
            break

    # Final exact-best selection between current path and stored best.
    sc_cur = _exact_score(a)
    if sc_cur < best_score:
        best_score = sc_cur
        best_a = a.copy()

    # Reversal mixtures are cheap and can occasionally reduce boundary artifacts.
    rev = best_a[::-1].copy()
    for t in (0.20, 0.40, 0.60, 0.80):
        cand = _normalize((1.0 - t) * best_a + t * rev)
        sc = _exact_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    print("[{}] end n={} best={:.10f}".format(label, best_a.size, best_score))
    return best_score, best_a


def _load_parent():
    try:
        h = globals().get("height_sequence_1", None)
        if h is None:
            return None
        a = np.asarray(h, dtype=np.float64).ravel()
        a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) < 1e-12:
            return None
        return a
    except Exception:
        return None


def _to_output_list(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if a.size == 0 or s < 0.01:
        return [0.5] * 64
    a = a / s
    a = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in a]


def _safe_eval_list(lst):
    try:
        val = evaluate_sequence(lst)
        if val is None:
            return np.inf
        return float(val)
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = _current_time()
    total_budget = max(60.0, float(budget_s))
    deadline_search = t_start + total_budget - 120.0
    rng = np.random.default_rng(int(seed))

    parent = _load_parent()
    if parent is not None and 2048 <= parent.size <= 16384:
        n = int(parent.size)
    else:
        n = 10240

    print("Using length n={} (parent available: {})".format(n, parent is not None))

    inits = _structured_inits(n, rng, parent)
    scored = []
    for arr in inits:
        arr = _normalize(_resample(arr, n))
        sc = _exact_score(arr)
        if np.isfinite(sc):
            scored.append((sc, arr))

    if not scored:
        scored.append((2.0, np.ones(n, dtype=np.float64) / n))

    scored.sort(key=lambda item: item[0])

    best_score, best_seq = scored[0]
    print("Best initial score = {:.10f}".format(best_score))

    # Select up to three diverse starts; ensure parent (if present) is included.
    starts = [scored[0][1].copy()]
    for sc, arr in scored[1:]:
        if len(starts) >= 3:
            break
        starts.append(arr.copy())

    if parent is not None:
        p = _normalize(_resample(parent, n))
        if not any(p.size == best_seq.size and np.allclose(p, s) for s in starts):
            if len(starts) >= 3:
                starts[-1] = p
            else:
                starts.append(p)

    main_deadline = min(deadline_search, t_start + max(300.0, 0.78 * total_budget))
    main_total = max(0.0, main_deadline - t_start)
    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.72, 0.28]
    else:
        fracs = [0.60, 0.28, 0.12]

    cum = 0.0
    for rank, st in enumerate(starts):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_total
        sub_dead = min(sub_dead, main_deadline - 5.0)
        if _current_time() > sub_dead - 45.0:
            continue

        cand = st.copy()
        if rank > 0 and best_seq.size == cand.size:
            # For secondary trajectories, mix with current global best but preserve diversity.
            blend = 0.25 + 0.10 * rank
            cand = _normalize((1.0 - blend) * cand + blend * _normalize(best_seq))
        else:
            cand = _normalize(cand)

        sc, seq = _optimize_start(cand, sub_dead, rng, label="traj{}".format(rank + 1))
        if np.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = seq.copy()
                print("After traj{}: best={:.10f}".format(rank + 1, best_score))
            else:
                # Keep the better of path and global best for resampling.
                if exact_global := _exact_score(_resample(best_seq, cand.size)):
                    if exact_global < _exact_score(cand):
                        pass

    # Final intensive polishing from the best sequence.
    final_dead = min(deadline_search, t_start + total_budget - 105.0)
    if final_dead - _current_time() > 150.0:
        final_schedule = [
            (30000.0, 2.0, "lse"),
            (100000.0, 3.0, "lse"),
            (300000.0, 4.0, "lse"),
            (800000.0, 4.0, "lse"),
        ]
        sc, seq = _optimize_start(
            best_seq,
            final_dead,
            rng,
            label="final_polish",
            schedule=final_schedule,
        )
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = seq.copy()
            print("After final polish: best={:.10f}".format(best_score))

    # If time remains, try resampling to nearby lengths and do one short polish.
    extra_end = min(deadline_search, t_start + total_budget - 80.0)
    if extra_end - _current_time() > 180.0:
        for n2 in (8192, 12288):
            if n2 == best_seq.size or _current_time() > extra_end - 90.0:
                continue
            cand = _resample(best_seq, n2)
            sc2, seq2 = _optimize_start(
                cand,
                extra_end - 30.0,
                rng,
                label="resample_{}".format(n2),
                schedule=[(20000.0, 2.0, "lse"),
                          (100000.0, 3.0, "lse"),
                          (400000.0, 4.0, "lse")],
            )
            if np.isfinite(sc2) and sc2 < best_score - 1e-12:
                best_score = sc2
                best_seq = seq2.copy()
                print("After resample {}: best={:.10f}".format(n2, best_score))

    best_seq = _normalize(best_seq)
    out = _to_output_list(best_seq)

    # Final safeguard: verify with the provided evaluator if time allows.
    if _current_time() < t_start + total_budget - 20.0:
        verified = _safe_eval_list(out)
        if not np.isfinite(verified):
            out = _to_output_list(np.ones(best_seq.size, dtype=np.float64))
        else:
            print("Verified final score = {:.10f}".format(verified))

    S = sum(out)
    if not out or S < 0.01:
        out = [0.5] * 64

    final_score = _exact_score(np.asarray(out, dtype=np.float64))
    print(
        "FINAL internal={:.10f} len={} time={:.1f}s".format(
            final_score, len(out), _current_time() - t_start
        )
    )
    return out
```
