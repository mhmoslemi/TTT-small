def generate_slepian(n_freqs, n_sample, decay_factor=0.995):

    # Typically uses Chebyshev polynomials of second kind (T_nu) and prolate polynomials (P_nu)
    # Here, we approximate the Slepian sequence generation by optimizing in frequency domain
    
    # Define the eigenvector equation: satisfies P_nu,Slepian(theta_k) ~ 0 for |theta_k|> bandwidth
    # We'll set an artificial bandwidth limit (pi) and generate T_nu as the prolate shape
    # Bandwidth factor N determines the decay rate (N larger means tighter bandlimit)
    N = 1000  # large eigenvalue parameter (controls 'bandwidth')
    
    # We'll generate a sequence optimized to minimize:
    # score = (2 * n_sample * max_conv) / (sum_sq²)
    #
    # In frequency domain, note:
    #   Convolution (self-convolve) corresponds to square of the Fourier transform of a (with padding)
    #   So, if we limit the frequency domain, the convolution has less high peaks
    
    # Define frequency response: 
    #   a_{k} = (magnitude[l]) * exp(1j * phase[l]) with careful optimization
    #   Such that inverse laplace converts it to positive sequence

    omega = np.linspace(0, np.pi, n_freqs, endpoint=True)
    T = np.sin((n_freqs+1)*omega)**2 / ((n_freqs+1)*omega**2 - np.pi**2)  # Unnormalized prolate sequence
    
    # Smooth T-shape using Gaussian filtering to handle edge effects
    T_shape = np.maximum(gaussian_smooth(T), 1e-16)
    
    # Now, combine these within a norm constraint
    U = np.exp(1j * np.random.uniform(-np.pi, np.pi, n_freqs))  # random phase (optional part)
    norm_factor = np.sqrt(n_sample)  # for normalization
    
    # Convert back to time domain and normalize
    real_signal = np.fft.irfft(np.fft.rfft(U) * np.fft.rfft(T_shape), n_sample)[:n_freqs]
    real_signal = real_signal[:n_sample]  # trim to needed length
    real_signal[:, np.abs(real_signal).argsort()[n_sample//2]] += 1  # regularize to be strictly positive
    
    # Normalize to have positive sum
    if np.sum(real_signal) < 1e-16:
        real_signal[:]=1/n_sample  # fallback
    else:
        real_signal = real_signal / np.sum(real_signal)
        
    return real_signal
