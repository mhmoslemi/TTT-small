def lse_grad(a, beta):
    """LSE-smoothed score gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:

    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

def fast_conv(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

        return np.inf

def softplus(x):
    return np.log1p(np.exp(np.clip(x, -50, 50)))

def softplus_grad(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))

def lse_score_grad_direct(a, beta):
    """LSE-smoothed score and gradient for direct (non-negative) parameterization."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)

    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3

"""
Optimized step function search combining:
1. Softplus reparameterization (eliminates non-negativity projection)
2. Spectral magnitude damping (fast initial progress via frequency-domain projection)
3. LSE-LBFGS with adaptive beta schedule (smooth continuation to L-infinity)
4. Direct peak reduction (surgical reduction of convolution peaks)
5. Multi-scale spectral iteration with adaptive damping

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def softplus(x):
    """Numerically stable softplus."""
    return np.log1p(np.exp(np.clip(x, -50, 50)))


def softplus_grad(x):
    """Sigmoid (derivative of softplus)."""
    return 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))


def sp_to_a(b):
    """Convert softplus parameters to sequence (normalized to sum=1)."""
    a = softplus(b)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(b)) / len(b)
    return a / S


def sp_init(a):
    """Initialize softplus parameters from a sequence."""
    a = np.maximum(a, 1e-10)
    return np.log(np.expm1(np.clip(a, 1e-10, 1e10)))


def lse_score_grad_direct(a, beta):
    """LSE-smoothed score and gradient for direct (non-negative) parameterization."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def lse_score_grad_softplus(b, beta):
    """LSE-smoothed score and gradient for softplus parameterization."""
    a = sp_to_a(b)
    sig = softplus_grad(b)
    score, grad_a = lse_score_grad_direct(a, beta)
    grad_b = grad_a * sig
    return score, grad_b

    """
    Spectral magnitude damping: reduce |A[m]| for m>0 to flatten convolution.
    Higher frequencies get more damping (controlled by freq_power).
    """
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    freq_damp_shape[0] = 0.0

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):

        gn = float(np.linalg.norm(score_grad))
        if gn < 1e-15:
            break

        a -= lr * score_grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        lr *= decay

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def nesterov_momentum(a0, beta, n_iters=10000, lr_init=0.003, momentum=0.9):
    """Nesterov momentum on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        a_look = np.maximum(a - momentum * v, 0.0)
        S_look = a_look.sum()
        if S_look > 1e-12:
            a_look = a_look / S_look
            sc_l, grad_l = lse_score_grad_direct(a_look, beta)
        else:
            sc_l, grad_l = lse_score_grad_direct(a, beta)

        v = momentum * v + lr * grad_l
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= 0.99995

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def iterative_reweighting(a0, n_outer=30, n_inner=2000, maxfun_per_inner=3000):
    """
    Iterative reweighting: weight peaks by current convolution structure,
    solve weighted subproblem, repeat.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for outer in range(n_outer):
        c = fast_conv(a)
        c_max = float(c.max())

        p = 1.0 + 3.0 * outer / max(1, n_outer)
        c_norm = c / max(c_max, 1e-15)
        w = c_norm ** p
        w = 0.05 * (1.0 / (2 * n - 1)) + 0.95 * w
        w = w / w.sum()

        # Solve weighted subproblem using LSE-like approach
        # Objective: sum_k w_k * c_k / S^2
        # Use L-BFGS on this
        a = run_lbfgs_weighted(a, w, maxiter=n_inner, maxfun=maxfun_per_inner)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def weighted_score_grad(a, w):
    """Weighted convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    w_num = float(np.dot(w, c))
    score = 2.0 * n * w_num / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * w_num) / S3
    return score, grad


def run_lbfgs_weighted(a0, w, maxiter=2000, maxfun=5000):
    """Run L-BFGS-B on weighted objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def fun(x):
        return weighted_score_grad(x, w)

    res = minimize(
        fun, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    return np.clip(res.x, 0.0, 1000.0)


def build_inits(n, rng, parent=None):
    """Build diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    inits.append(np.maximum(1 - 2 * np.abs(x - 0.5), 0.01))
    inits.append(np.exp(-5 * (x - 0.5) ** 2) + 0.05)
    inits.append(np.maximum(0.5 + 0.25 * np.cos(4 * np.pi * x) + 0.15 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(6 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(10 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1 - 2 * x) ** 2)
    inits.append(np.maximum(0.5 + 0.2 * np.sin(2 * np.pi * 3 * x) + 0.15 * np.cos(2 * np.pi * 7 * x), 0.01))

    # Fourier-based random smooth
    for _ in range(4):
        n_harm = rng.integers(3, 12)
        freqs = rng.integers(1, 20, size=n_harm)
        weights = rng.standard_normal(n_harm) * 0.15
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        sig = np.zeros(n)
        for i in range(n_harm):
            sig += weights[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        inits.append(np.maximum(0.5 + sig, 0.01))

    return [np.maximum(s, 0.0) / max(s.sum(), 1e-12) for s in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.

    Multi-phase strategy:
    Phase 1 (15%): Spectral magnitude damping for fast initial progress
    Phase 2 (40%): LSE-LBFGS with increasing beta (direct + softplus)
    Phase 3 (25%): Iterative reweighting + peak reduction + subgradient
    Phase 4 (20%): Final polish with high-beta LSE + spectral + momentum
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None and abs(len(parent) - n) < 100:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return False
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_a = np.maximum(best_a, 0.0)
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting: n={n}, initial best={best_score:.8f}")

    # ============================================================
    # Phase 1: Spectral warmup (15%)
    # ============================================================
    p1_end = t_start + 0.15 * (budget_s - 30)
    print(f"\n=== Phase 1: Spectral warmup (until t={p1_end - t_start:.0f}s) ===")

    inits = build_inits(n, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > p1_end:
            break
        a1 = spectral_magnitude_damp(a0, n_iters=800, alpha_start=0.25, alpha_end=0.02, freq_power=0.5)
        sc = compute_score(a1)
        update_best(sc, a1)

        if time.monotonic() < p1_end:
            a2 = spectral_magnitude_v2(a1, n_iters=500)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS with adaptive beta (40%)
    # ============================================================
    p2_end = t_start + 0.55 * (budget_s - 30)
    print(f"\n=== Phase 2: LSE-LBFGS (until t={p2_end - t_start:.0f}s) ===")

    betas_direct = [200, 500, 1000, 2000, 5000, 10000, 20000, 40000]
    for beta in betas_direct:
        if time.monotonic() > p2_end:
            break
        t_beta = time.monotonic()
        a_opt = run_lbfgs_direct(best_a, beta, maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LSE-direct beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    # Softplus parameterization (unconstrained - can escape local minima differently)
    if time.monotonic() < p2_end:
        b0 = sp_init(best_a)
        for beta_sp in [500, 2000, 5000, 10000, 20000]:
            if time.monotonic() > p2_end:
                break
            t_sp = time.monotonic()
            b_opt = run_lbfgs_softplus(b0, beta_sp, maxiter=3000, maxfun=10000)
            a_out = sp_to_a(b_opt)
            sc = compute_score(a_out)
            update_best(sc, a_out)
            print(f"  LSE-softplus beta={beta_sp}: score={sc:.8f} ({time.monotonic()-t_sp:.1f}s)")
            if sc < best_score:
                b0 = b_opt.copy()

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Iterative reweighting + Peak reduction (25%)
    # ============================================================
    p3_end = t_start + 0.80 * (budget_s - 30)
    print(f"\n=== Phase 3: Reweighting + Peak reduction (until t={p3_end - t_start:.0f}s) ===")

    # Iterative reweighting
    a_ir = iterative_reweighting(best_a, n_outer=20, n_inner=1500, maxfun_per_inner=2500)
    sc = compute_score(a_ir)
    update_best(sc, a_ir)

    # Direct peak reduction
    if time.monotonic() < p3_end:
        a_pr = direct_peak_reduction(best_a, n_rounds=60, top_k=40, reduce_frac=0.07)
        sc = compute_score(a_pr)
        update_best(sc, a_pr)

    # Subgradient with different peak counts
    if time.monotonic() < p3_end:
        for n_top in [20, 50, 100]:
            if time.monotonic() > p3_end:
                break
            a_sg = subgradient_descent(best_a, n_iters=3000, n_top=n_top, lr_init=0.004)
            sc = compute_score(a_sg)
            update_best(sc, a_sg)

    # LSE after peak reduction
    if time.monotonic() < p3_end:
        a_opt = run_lbfgs_direct(best_a, 30000, maxiter=4000, maxfun=12000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (20%)
    # ============================================================
    print(f"\n=== Phase 4: Final polish (until deadline) ===")

    polish_count = 0
    while time.monotonic() < deadline:
        # Quick spectral refinement
        a_sp = spectral_magnitude_v2(best_a, n_iters=300)
        sc = compute_score(a_sp)
        update_best(sc, a_sp)

        # High-beta LSE
        if time.monotonic() > deadline:
            break
        beta_final = 50000 + 25000 * polish_count
        a_opt = run_lbfgs_direct(best_a, beta_final, maxiter=4000, maxfun=12000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Nesterov momentum fine-tune
        if time.monotonic() > deadline:
            break
        a_nm = nesterov_momentum(best_a, beta=20000, n_iters=5000, lr_init=0.002)
        sc = compute_score(a_nm)
        update_best(sc, a_nm)

        # Softplus polish
        if time.monotonic() > deadline:
            break
        b_pol = sp_init(best_a)
        b_opt = run_lbfgs_softplus(b_pol, 50000, maxiter=2000, maxfun=6000)
        a_out = sp_to_a(b_opt)
        sc = compute_score(a_out)
        update_best(sc, a_out)

        polish_count += 1
        elapsed = time.monotonic() - t_start
        print(f"  Polish {polish_count}: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Final: try a few more peak reductions at the very end
    if time.monotonic() < deadline:
        a_final = direct_peak_reduction(best_a, n_rounds=30, top_k=30, reduce_frac=0.05)
        sc = compute_score(a_final)
        update_best(sc, a_final)

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
