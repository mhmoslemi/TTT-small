Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2 with n~10000.
Uses: Adam LSE optimization, asymmetry-preserving spectral methods,
exact peak shaving via gradient structure, and diverse initialization.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses Adam LSE optimization (fast many-step), asymmetry-preserving spectral methods,
exact peak shaving via gradient structure, and diverse asymmetric initialization.
Key innovation: Adam-based LSE allows 10-50x more iterations than L-BFGS,
and explicit asymmetry breaking avoids symmetric local minima.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad


def adam_lse(a0, beta, n_iters, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
             patience=2000, verbose=False):
    n = len(a0)
    a = normalize(a0.copy()).astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    best_sc = compute_score(a)
    best_a = a.copy()
    best_t = 0
    no_improve = 0
    lr_current = lr

    for t in range(1, n_iters + 1):
        sc, grad = lse_score_and_grad(a, beta)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            best_t = t
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            lr_current *= 0.5
            if lr_current < 1e-8:
                break
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            if verbose:
                print(f"    Adam restart at t={t}, lr={lr_current:.2e}")

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a = a - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

    return best_sc, best_a


def lbfgs_lse(a0, beta, maxiter=3000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, None)
        if a_out.sum() < 1e-6:
            a_out = a0
        a_out = normalize(a_out)
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fienup_asymmetric(a, n_iters=300, mode='flat'):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        if mode == 'flat':
            alpha = 0.1 + 0.6 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean
        elif mode == 'quantile':
            q = 0.70 + 0.25 * progress
            thresh = np.quantile(P, q)
            P_new = np.minimum(P, thresh)
        elif mode == 'asym_flat':
            # Flatten but boost the low-frequency end to preserve asymmetry
            alpha = 0.1 + 0.5 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean
            # Boost DC and low frequencies (preserves "ramp-like" structure)
            n_low = max(1, M_fft // 4)
            P_new[:n_low] *= (1.0 + 0.3 * (1.0 - progress))
        else:
            P_new = P.copy()

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave_gradient(a, n_iters=5000, top_k=5, step_init=0.01):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peak lags
        top_idx = np.argsort(c)[::-1][:top_k]

        # Build subgradient targeting all top peaks
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                # d(c[k])/da[j] = 2*a[k-j] for valid j
                js = np.arange(j_lo, j_hi + 1)
                contrib = 2.0 * a[k - js]
                sg[-js] += w * contrib

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step
        lower = 2.0 * n / (2.0 * n - 1)
        step_polyak = max(1e-10, (sc - lower) / (gn * gn + 1e-10))
        step_use = min(step, step_polyak, 0.02)

        a = a - step_use * grad
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        step *= 0.9995

    return best_a


def peak_shave_direct(a, n_iters=3000):
    """Directly reduce contributors to the peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi <= j_lo:
            break

        js = np.arange(j_lo, j_hi + 1)
        # Contribution of pair (j, k_star - j) to c[k_star]
        pairs = a[js] * a[k_star - js]
        total = pairs.sum()
        if total < 1e-15:
            break

        # Find top contributors
        n_reduce = max(3, n // 100)
        top_local = np.argsort(pairs)[::-1][:n_reduce]
        reduce_idx = j_lo + top_local

        # Find "safe" positions (don't contribute to peak)
        peak_set = set(js.tolist())
        non_peak = np.array([i for i in range(n) if i not in peak_set])
        if len(non_peak) < 5:
            return best_a
        n_add = min(len(non_peak), n_reduce * 3)
        add_idx = non_peak[np.random.choice(len(non_peak), n_add, replace=False)]

        delta = 0.0005 / n
        for j in reduce_idx:
            if a[j] > delta * 3:
                a[j] -= delta
        for j in add_idx:
            a[j] += delta

        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a


def coord_descent(a, n_sweeps=8, n_top=30):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi <= j_lo:
            break

        js = np.arange(j_lo, j_hi + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_lo + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.85, 0.95]:

                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()

def make_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def make_exponential(n, rate=0.5):
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    return a / a.sum()


def make_bimodal(n):
    a = np.zeros(n)
    a[:n // 4] = np.linspace(1, 0.3, n // 4)
    a[3 * n // 4:] = np.linspace(0.3, 0.1, n // 4)
    return a / a.sum()


def make_random_asymmetric(n, rng):
    # Smooth random with bias toward one side
    a = rng.random(n)
    a = a * np.linspace(0.3, 1.0, n)
    # Smooth
    kernel = np.ones(10) / 10
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    return a / a.sum()


def make_zadoff_real(n):
    # Real part of a Zadoff-Chu-like sequence, clipped to non-negative
    u = 3
    root = 2 * u + 1
    a = np.zeros(n)
    for i in range(n):
        phase = np.pi * u * i * (2 * i + 1) / root
        a[i] = np.cos(phase)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n) * 0.1
        a[:n // 2] = 0.2
    return a / a.sum()


def interpolate_sequence(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def perturb_smooth(a, rng, scale=0.02):
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 8):
        freq = rng.integers(1, 30)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_additive(a, rng, scale=0.01):
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)

    print(f"\n=== Phase 2: Adam LSE (budget {phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas_adam = [15, 30, 60, 120, 300, 700, 1500, 4000, 10000, 25000]

    for i, beta in enumerate(betas_adam):
        if time.monotonic() > phase2_end - 15:
            break
        remaining = phase2_end - 15 - time.monotonic()
        if remaining < 10:
            break

        # Learning rate: larger for low beta, smaller for high beta
        lr = 0.008 / (1.0 + 0.05 * beta / 100.0)
        n_iters = int(remaining * 800)
        n_iters = max(500, min(n_iters, 50000))

        print(f"  Adam beta={beta:>6}, lr={lr:.4f}, iters={n_iters}")
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_work, beta, n_iters, lr=lr, patience=3000)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"ADAM b={beta}")
        a_work = a_opt.copy()

        # Occasionally do a Fienup break to escape local minima
        if i % 3 == 2 and time.monotonic() < phase2_end - 20:
            a_f = fienup_asymmetric(a_work, n_iters=100, mode='asym_flat')
            sc_f = compute_score(a_f)
            if sc_f < compute_score(a_work):
                update(sc_f, a_f, f"FIENUP break b={beta}")
                a_work = a_f

    # Second Adam pass from best
    if time.monotonic() < phase2_end - 20:
        print("  Adam second pass...")
        for beta in [5000, 15000, 40000]:
            if time.monotonic() > phase2_end - 10:
                break
            remaining = phase2_end - 10 - time.monotonic()
            if remaining < 10:
                break
            lr = 0.002 / (1.0 + 0.05 * beta / 100.0)
            n_iters = max(500, int(remaining * 600))
            sc, a_opt = adam_lse(best_a, beta, n_iters, lr=lr, patience=2000)
            print(f"    pass2 beta={beta}: score={sc:.8f}")
            update(sc, a_opt, f"ADAM2 b={beta}")
            if sc < best_score:
                a_work = a_opt

    # ============================================================
    # Phase 3: L-BFGS polish (12% time)
    # ============================================================
    phase3_end = t_start + 0.58 * (budget_s - 30)
    print(f"\n=== Phase 3: L-BFGS Polish ===")

    for beta in [3000, 8000, 20000]:
        if time.monotonic() > phase3_end - 10:
            break
        remaining = phase3_end - 10 - time.monotonic()
        if remaining < 10:
            break
        maxiter = max(500, int(remaining * 10))
        sc, a_opt = lbfgs_lse(best_a, beta, maxiter=maxiter)
        print(f"  L-BFGS beta={beta}: score={sc:.8f}")
        update(sc, a_opt, f"LBFGS b={beta}")

    # ============================================================
    # Phase 4: Peak shaving + spectral alternation (20% time)
    # ============================================================
    phase4_end = t_start + 0.78 * (budget_s - 30)
    print(f"\n=== Phase 4: Peak Shave + Spectral (budget {phase4_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    n_cycles = 15
    for cyc in range(n_cycles):
        if time.monotonic() > phase4_end - 10:
            break
        remaining = phase4_end - 10 - time.monotonic()
        if remaining < 5:
            break

        # Peak shaving
        n_iters_ps = max(500, int(remaining * 50))
        top_k = 3 + (cyc % 5)
        a_ps = peak_shave_gradient(a_work, n_iters=min(n_iters_ps, 5000), top_k=top_k)
        sc_ps = compute_score(a_ps)
        update(sc_ps, a_ps, f"PSAVE cyc={cyc}")
        if sc_ps < compute_score(a_work):
            a_work = a_ps

        # Spectral break
        if time.monotonic() < phase4_end - 5:
            mode = 'asym_flat' if cyc % 2 == 0 else 'quantile'
            a_f = fienup_asymmetric(a_work, n_iters=80, mode=mode)
            sc_f = compute_score(a_f)
            update(sc_f, a_f, f"F{mode[:4]} cyc={cyc}")
            if sc_f < compute_score(a_work):
                a_work = a_f

        # Quick Adam follow-up
        if time.monotonic() < phase4_end - 5:
            beta = 2000 + cyc * 2000
            lr = 0.003
            n_iters_a = max(300, int((phase4_end - 5 - time.monotonic()) * 400))
            sc_a, a_a = adam_lse(a_work, beta, n_iters=min(n_iters_a, 3000), lr=lr, patience=1000)
            update(sc_a, a_a, f"ADAM cyc={cyc}")
            if sc_a < compute_score(a_work):
                a_work = a_a

    # ============================================================
    # Phase 5: Multi-restart with diverse perturbations (8% time)
    # ============================================================
    phase5_end = t_start + 0.86 * (budget_s - 30)
    print(f"\n=== Phase 5: Multi-Restart ===")

    for r in range(10):
        if time.monotonic() > phase5_end - 10:
            break
        remaining = phase5_end - 10 - time.monotonic()
        if remaining < 10:
            break

        if r % 3 == 0:
            a_pert = perturb_smooth(best_a, rng, scale=0.01 + 0.005 * (r % 4))
        elif r % 3 == 1:
            a_pert = perturb_additive(best_a, rng, scale=0.005 + 0.003 * (r % 4))
        else:
            # Asymmetric perturbation: boost one side
            a_pert = best_a.copy()
            half = n_base // 2
            a_pert[:half] *= (1.0 + 0.02 * (r % 5))
            a_pert[half:] *= (1.0 - 0.01 * (r % 5))
            a_pert = normalize(a_pert)

        beta = 3000 + r * 3000
        lr = 0.004
        n_iters = max(500, int(remaining * 400))
        sc, a_opt = adam_lse(a_pert, beta, n_iters=min(n_iters, 5000), lr=lr, patience=1500)
        update(sc, a_opt, f"RESTART {r}")

    # ============================================================
    # Phase 6: Final polish (6% time)
    # ============================================================
    print(f"\n=== Phase 6: Final Polish ===")

    a_work = best_a.copy()

    # Coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd = coord_descent(a_work, n_sweeps=6, n_top=25)
        sc_cd = compute_score(a_cd)
        update(sc_cd, a_cd, "COORD")
        if sc_cd < best_score:
            a_work = a_cd

    # Direct peak shave
    if time.monotonic() < deadline - 20:
        a_ds = peak_shave_direct(a_work, n_iters=2000)
        sc_ds = compute_score(a_ds)
        update(sc_ds, a_ds, "DIRECT")
        if sc_ds < best_score:
            a_work = a_ds

    # Very high beta Adam
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        n_iters = max(500, int(remaining * 800))
        sc, a_opt = adam_lse(a_work, 50000, n_iters=min(n_iters, 10000), lr=0.001, patience=3000)
        update(sc, a_opt, "FINAL-ADAM")
        if sc < best_score:
            a_work = a_opt

    # Final L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = lbfgs_lse(a_work, 30000, maxiter=2000)
        update(sc, a_opt, "FINAL-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
