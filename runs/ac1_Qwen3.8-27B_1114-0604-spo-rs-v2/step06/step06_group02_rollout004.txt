"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (via multiplicative weights), adaptive LSE-L-BFGS,
multi-scale initialization, and targeted peak shaving.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def mwu_lp_direction(a, n_iters=500, eta=0.1):
    """
    Approximate the LP direction max sum(g) s.t. (a*g)_k <= M, g >= 0
    using multiplicative weight updates.
    Returns g normalized to sum 1.
    """
    n = len(a)
    M = float(fast_conv(a).max())
    g = np.ones(n) / n
    M_fft = 1 << (2 * n - 1).bit_length()
    
    A_fft = np.fft.rfft(a, M_fft)
    
    for it in range(n_iters):
        G_fft = np.fft.rfft(g, M_fft)
        h = np.fft.irfft(A_fft * G_fft, M_fft)[:2 * n - 1]
        k_star = int(np.argmax(h))
        h_max = h[k_star]
        if h_max < 1e-15:
            break
        
        # MWU update: increase g[j] proportionally to a[k*-j]
        # (a*g)_{k*} = sum_j a[k*-j] * g[j]
        # Violation ratio = a[k*-j] * g[j] / (M / h_max) -- but simpler:
        # Just boost g[j] where a[k*-j] is large
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi < j_lo:
            break
        contrib = a[j_lo:j_hi + 1]
        boost = 1.0 + eta * contrib / (M + 1e-30)
        g[j_lo:j_hi + 1] *= boost
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    
    return g

def fienup_flat(a, n_iters=300):

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.7 * (1.0 - progress) ** 1.5 + 0.03
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Novel: LP-guided direction finding via HiGHS, adaptive LSE-LBFGS with very high beta,
multi-scale initialization, targeted peak-lag subgradient, and diverse stochastic restarts.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def solve_lp_direction(a, M, timeout=10):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g normalized to sum 1, or None if LP fails.
    """
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    # (a*g)_k = sum_j A[k,j] * g[j]
    # A is a banded matrix (convolution structure)
    # For efficiency, use scipy sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            v = a[k - j]
            if v > 1e-18:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # c = -1 (maximize sum = minimize -sum)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            S = g.sum()
            if S > 1e-12:
                return g / S
    except Exception:
        pass
    return None


def mwu_direction(a, n_iters=300, eta=0.08):
    """
    Multiplicative weight approximation of LP direction.
    Faster than full LP for large n.
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    g = np.ones(n) / n
    M_fft = 1 << (2 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    
    for it in range(n_iters):
        G_fft = np.fft.rfft(g, M_fft)
        h = np.fft.irfft(A_fft * G_fft, M_fft)[:2 * n - 1]
        k_star = int(np.argmax(h))
        h_max = h[k_star]
        if h_max < 1e-15:
            break
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        if j_hi < j_lo:
            break
        contrib = a[j_lo:j_hi + 1]
        boost = 1.0 + eta * contrib / (M + 1e-30)
        g[j_lo:j_hi + 1] *= boost
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    return g


def fienup_flat(a, n_iters=300):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.6 * (1.0 - progress) ** 1.5 + 0.02
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

    p4_end = t_start + 0.78 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Stochastic restarts (t < {p4_end - t_start:.0f}s) ===")
    
    a_sr, sc_sr = stochastic_restart(best_a, rng, n_restarts=15, deadline=p4_end)
    update(sc_sr, a_sr, "STOCHASTIC")

    # ============================================================
    # Phase 5: Multi-length exploration (10% time)
    # ============================================================
    p5_end = t_start + 0.88 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Multi-length (t < {p5_end - t_start:.0f}s) ===")
    
    for target_n in [1024, 4096, 8192]:
        if time.monotonic() > p5_end - 30:
            break
        if target_n == len(best_a):
            continue
        a_trial = try_length(best_a, target_n, p5_end)
        sc_trial = compute_score(a_trial)
        print(f"  n={target_n} initial score: {sc_trial:.8f}")
        
        if target_n <= 2048:
            # Can use full LP
            a_opt, sc_opt = lp_guided_optimize(a_trial, p5_end - 10, use_full_lp=(target_n <= 3000), n_lp_rounds=15)
        else:
            maxiter = int(min(8000, (p5_end - 10 - time.monotonic()) * 5))
            if maxiter < 1000:
                continue
            sc_opt, a_opt = run_lbfgs(a_trial, 10000, maxiter=max(2000, maxiter), maxfun=maxiter * 12)
        
        update(sc_opt, a_opt, f"LEN-{target_n}")

    # ============================================================
    # Phase 6: Final polish (last 5% time)
    # ============================================================
    print(f"\n=== Phase 6: Final polish ===")
    
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = int(min(20000, remaining * 10))
        sc, a_fin = run_lbfgs(best_a, 500000, maxiter=max(3000, maxiter), maxfun=maxiter * 10)
        update(sc, a_fin, "FINAL-500k")

    if time.monotonic() < deadline - 20:
        maxiter = int(min(10000, (deadline - 20 - time.monotonic()) * 8))
        if maxiter > 2000:
            sc, a_fin = run_lbfgs(best_a, 2000000, maxiter=maxiter, maxfun=maxiter * 8)
            update(sc, a_fin, "FINAL-2M")

    # Final LP-guided round if time permits
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        if remaining > 60:
            n_cur = len(best_a)
            use_lp_final = n_cur <= 2500
            a_fin, sc_fin = lp_guided_optimize(best_a, deadline - 10, use_full_lp=use_lp_final, n_lp_rounds=10)
            update(sc_fin, a_fin, "FINAL-LP")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
