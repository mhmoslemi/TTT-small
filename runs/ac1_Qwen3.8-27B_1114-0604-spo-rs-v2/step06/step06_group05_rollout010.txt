def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    # Work with normalized convolution
    cn = c / S2
    cn_max = float(cn.max())
    scaled = beta * (cn - cn_max)
    scaled_shifted = scaled - float(scaled.max())
    exp_s = np.exp(scaled_shifted)
    log_sum_exp = float(np.log(exp_s.sum()))
    LSE = cn_max + log_sum_exp / beta
    score = 2.0 * n * LSE
    # Probability distribution over convolution positions
    p = exp_s / exp_s.sum()
    # Gradient: df/db_i = 4n/S^2 * sum_k p_k * b_{k-i} (valid indices)
    # = 4n/S^2 * conv(rev_a, p)[i + n - 1]
    rev_a = a[::-1]
    M2 = 1 << (3 * n - 3).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(RA * P, M2)[n - 1:2 * n - 1]
    grad = (4.0 * n / S2) * conv_res
    return score, grad

    """Run L-BFGS-B on LSE objective."""
    n = len(a0)

    def wrapped(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxiter * 10, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)

def gs_spectral(a, n_iters=500, alpha=0.5):
    """GS-style spectral amplitude iteration."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        # Target: flatten c toward mean
        c_mean = float(c[:n].mean())  # mean of central part
        target = np.zeros(M)
        target[:2*n-1] = c / c_max * (1 - alpha) + (c_mean / c_max) * alpha
        target = np.maximum(target, 1e-10)
        
        A = np.fft.fft(a, M)
        A_abs = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        # Target spectrum: sqrt(target) in magnitude
        T = np.fft.fft(target, M)
        T_abs = np.maximum(np.abs(T), 1e-20)
        
        # Blend
        new_abs = A_abs ** (1 - alpha) * T_abs ** alpha
        A_new = new_abs * np.exp(1j * A_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Adaptive LSE-beta-continuation optimization with spectral amplitude iteration
and peak-targeted refinement for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations: correct gradient with S-dependence, aggressive beta annealing,
spectral GS, and multi-stage polish with re-optimization from perturbed states.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and full gradient including S-dependence."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    S3 = S2 * S
    cn = c / S2
    cn_max = float(cn.max())
    scaled = beta * (cn - cn_max)
    scaled_shifted = scaled - float(scaled.max())
    exp_s = np.exp(scaled_shifted)
    log_sum_exp = float(np.log(exp_s.sum()))
    LSE = cn_max + log_sum_exp / beta
    score = 2.0 * n * LSE
    p = exp_s / exp_s.sum()
    # Gradient: 4n/S^2 * conv rela[i] - 4n/S^3 * C_p
    # conv rela[i] = sum_k p_k * a_{k-i} = conv(rev_a, p)[i + n - 1]
    rev_a = a[::-1]
    M2 = 1 << (3 * n - 3).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(RA * P, M2)[n - 1:2 * n - 1]
    C_p = float(np.dot(p, c))
    grad = (4.0 * n / S2) * conv_res - (4.0 * n / S3) * C_p
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        return lse_score_grad(x, beta)

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)


def gs_spectral(a, n_iters=300, alpha_init=0.5, alpha_final=0.1):
    """GS-style spectral amplitude iteration to flatten convolution."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c[:2 * n - 1].mean())
        target = np.zeros(M)
        target[:2 * n - 1] = np.maximum(c / c_max * (1 - alpha) + (c_mean / c_max) * alpha, 1e-10)
        A = np.fft.fft(a, M)
        A_abs = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        T = np.fft.fft(target, M)
        T_abs = np.maximum(np.abs(T), 1e-20)
        new_abs = A_abs ** (1 - alpha) * T_abs ** alpha
        A_new = new_abs * np.exp(1j * A_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def peak_reduce(a, n_rounds=200, top_k=20, reduce_frac=0.08):
    """Reduce top peaks by modifying contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]
        for k in top_idx[:8]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            n_red = min(8, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_red]
            for li in top_local:
                j = j_start + li
                a[j] *= (1.0 - reduce_frac)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def projected_gradient(a0, beta, n_iters=5000, lr_init=0.01, momentum=0.9):
    """Nesterov momentum gradient descent with projection onto a>=0."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        if it % 100 == 0:
            sc, grad = lse_score_grad(a, beta)
        else:
            sc, grad = lse_score_grad(a, beta)
        a_look = a - momentum * lr * v
        a_look = np.maximum(a_look, 0.0)
        _, grad_look = lse_score_grad(a_look, beta)
        v = momentum * v + grad_look
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= 0.99995
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def build_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.01, 1000.0) / max(parent.sum(), 1e-12))
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01) / n)
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01) / n)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x), 0.01) / n)
    inits.append(np.clip(0.5 + 0.15 * rng.standard_normal(n), 0.01, 5.0) / n)
    # Periodic structure inits
    for p in [7, 13, 31]:
        sig = np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * x * p / n * n), 0.01)
        inits.append(sig / n)
    # Random Fourier
    for _ in range(3):
        freqs = rng.integers(2, 20, size=5)
        weights = rng.standard_normal(5) * 0.2
        phases = rng.uniform(0, 2 * np.pi, 5)
        sig = np.ones(n) / n
        for f, w, ph in zip(freqs, weights, phases):
            sig += w * np.cos(2 * np.pi * f * x + ph)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())
    return inits

    # === Phase 5: Final ultra-high-beta polish ===
    print(f"\n=== Phase 5: Final polish ===")
    while time.monotonic() < deadline:
        for beta in [500000, 1000000, 2000000]:
            if time.monotonic() > deadline:
                break
            t0 = time.monotonic()
            a = run_lbfgs(best_a, beta, maxiter=3000, maxfun=30000)
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
            sc = compute_score(a)
            update_best(sc, a)
            print(f"  Final beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s}")
        # Quick peak reduce between runs
        if time.monotonic() < deadline - 120:
            a_pr = peak_reduce(best_a, n_rounds=50, top_k=15, reduce_frac=0.03)
            sc = compute_score(a_pr)
            update_best(sc, a_pr)

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
