def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score_LSE = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    # Softmax weights
    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a:
    # dL/da_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * conv(p, a_reversed)[n-1+j]
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1  # length of linear conv of p (len 2n-1) and a_rev (len n)
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    # Full gradient of score = 2n * L / S^2
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * 1.0 / S3)

    score = (2.0 * n * L) / S2
    return score, grad

def fienup_flat(a, n_iters=500):
    """Fienup: iteratively flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Gradually increase flattening strength
        alpha = 0.15 + 0.7 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        # Update amplitude, preserve phase
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def peak_shave_subgrad(a, n_iters=5000, top_k=8, step_init=0.008, decay=0.9997):
    """
    Subgradient descent targeting top-K autocorrelation peaks simultaneously.
    WithPolyak step size.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-7:
            break

        # Find top-K peaks and build subgradient
        top_idx = np.argsort(c)[::-1][:top_k]

        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with cap
        step_polyak = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.03)
        step = max(step, 1e-10)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def peak_shave_focused(a, n_iters=3000):
    """
    Focused peak shaving: add mass to off-peak positions and remove from peak contributors.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Identify contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        # Contribution of each pair to the peak
        js = np.arange(j_start, j_end + 1)
        pairs = a[js] * a[k_star - js]
        total_pair = pairs.sum()
        if total_pair < 1e-15:
            break

        # Reduce top contributors, increase non-contributors
        n_reduce = max(5, n // 50)
        top_contrib = np.argsort(pairs)[::-1][:n_reduce]
        reduce_idx = j_start + top_contrib

        # Find positions that don't contribute to the peak
        peak_set = set(js.tolist())
        non_peak = np.array([i for i in range(n) if i not in peak_set])
        if len(non_peak) < 10:
            return best_a
        n_add = min(len(non_peak), n_reduce * 2)
        add_candidates = non_peak[rng_choice(non_peak, n_add)]

        delta = 0.001 / n  # Small transfer amount
        for j in reduce_idx:
            if a[j] > delta * 2:
                a[j] -= delta
        for j in add_candidates:
            a[j] += delta

        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a


def rng_choice(arr, k):
    """Randomly select k elements without replacement."""
    if k >= len(arr):
        return arr
    idx = np.random.choice(len(arr), k, replace=False)
    return arr[idx]


def coord_descent(a, n_sweeps=10, n_top=40):
    """
    Coordinate descent: reduce top contributors to the peak lag.
    Uses multiplicative factors with line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue

            # Try reducing
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using cubic-ish interpolation."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def perturb_rng(a, rng, scale=0.02):
    """Additive Gaussian perturbation."""
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)


def perturb_mult(a, rng, scale=0.03):
    """Multiplicative perturbation with low-frequency structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 6):
        freq = rng.integers(1, 20)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)

    if time.monotonic() < deadline - 5:
        a_cd = coord_descent(best_a, n_sweeps=5, n_top=20)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]

    print(f"{'=' * 60}")
    return out
