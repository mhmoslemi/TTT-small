    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for t in range(int(max_iters)):

        if not np.isfinite(score):

def active_set_gradient_step(a, k_active, eta):
    """One gradient step on the exact max using top-k active lags."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    
    # Get top-k active lags
    m = d.size
    if k_active >= m:
        w = np.ones(m) / m
    else:
        idx = np.argpartition(d, m - k_active)[m - k_active:]
        w = np.zeros(m)
        w[idx] = 1.0 / k_active
    
    h = corr_pair(a, w)
    LDS = float(np.dot(w, d))
    grad = (2.0 * float(len(a)) / (S * S)) * h - (4.0 / S) * LDS
    
    return a - eta * grad

def soft_active_set(d, temperature=0.01):
    """Soft active set: weights proportional to exp(d/temperature)."""
    dmax = float(np.max(d))
    z = (d - dmax) / max(temperature, 1e-10)
    w = np.exp(z)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.zeros(d.size)
        w[np.argmax(d)] = 1.0
    else:
        w = w / wsum
    return w

def generate_diverse_inits(n, parent, rng, cap=20):

    inits = []

        add(pr)
        add(pr[::-1])

    for amp in (0.05, 0.1, 0.2, 0.3, 0.5):
        for freq in (1, 2, 3, 5, 8):
            add(1.0 + amp * np.sin(2.0 * np.pi * freq * x))
            add(1.0 + amp * np.cos(2.0 * np.pi * freq * x))
    
    # L-shapes (monotonically decreasing)
    for rate in (0.5, 1.0, 2.0, 5.0):
        add(np.exp(-rate * x))
        add(1.0 + rate * (1.0 - x))

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of: diverse spectral initializations, LSE optimization with
increasing beta (L-BFGS-B + Adam), soft-active-set gradient descent with
backtracking, mirror descent / power iteration, spectral shaping, and
multi-length exploration. Key insight: near-uniform sequences with small
structured perturbations minimize the peak self-convolution; the optimal
perturbation has a specific spectral profile that flattens the autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    return a * (target / S)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    return np.maximum(d, 0.0), S


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def lse_grad(a, beta):
    """Compute LSE-approximated objective and its gradient.
    Score = 2 * LSE_beta(d) where d = n*conv(a,a)/S^2.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    LSE = dmax + np.log(sum_e) / beta
    h = corr_pair(a, w)
    LDS = float(np.dot(w, d))

    # f = 2*LSE(d), df/da[i] = 2*(n/S^2)*h[i] - 4*LDS/S
    score = 2.0 * LSE
    grad = (2.0 * float(n) / (S * S)) * h - (4.0 / S) * LDS

    # Add penalty for S != 1
    score += 1e-4 * (S - 1.0) ** 2
    grad += 2e-4 * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs_lse(a0, beta, deadline, maxfun=500):
    """Run L-BFGS-B on LSE objective."""
    a = normalize_a(a0.copy())

    def fun(x):
        s, g = lse_grad(x, beta)
        # Re-normalize in objective? No, let bounds handle it.
        return s, g

    bounds = [(0.0, 10.0)] * len(a)
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds,
                       options={"ftol": 1e-15, "gtol": 1e-12,
                                "maxiter": int(maxfun), "maxfun": int(maxfun), "maxls": 50})
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    return exact_score_a(a_out), a_out


def adam_lse(a0, beta, max_iters, lr=0.05, deadline=None):
    """Adam optimizer for LSE objective."""
    a = normalize_a(a0.copy())
    n = len(a)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    b1, b2, eps = 0.85, 0.999, 1e-8

    for t in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        score, grad = lse_grad(a, beta)
        if not np.isfinite(score):
            break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad ** 2
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))

def soft_active_set(d, temperature):
    """Soft active set weights."""
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.zeros(d.size)
    z = (d - dmax) / max(temperature, 1e-10)
    z = np.clip(z, -30.0, 0.0)

    ws = float(w.sum())
    if ws < 1e-15:

        w[int(np.argmax(d))] = 1.0

def exact_grad_at_active(a, w):
    """Exact gradient of 2*max approximation using weight vector w."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)
    d, S = d_from_a(a)
    h = corr_pair(a, w)
    LDS = float(np.dot(w, d))
    return (2.0 * float(n) / (S * S)) * h - (4.0 / S) * LDS


def backtracking_step(a, grad, obj_fn, max_iter=40):
    """Backtracking line search."""
    f0 = obj_fn(a)
    if not np.isfinite(f0):
        return a, f0
    gscale = float(np.max(np.abs(grad)))
    if gscale < 1e-15:
        return a, f0
    step = 1.0 / gscale
    for _ in range(max_iter):
        a_new = normalize_a(np.clip(a - step * grad, 0.0, None))
        f_new = obj_fn(a_new)
        if np.isfinite(f_new) and f_new < f0 - 1e-10:
            return a_new, f_new
        step *= 0.5
        if step < 1e-12:
            break
    return a, f0

def mirror_descent_polish(a, max_iters=100, deadline=None):
    """Mirror descent / power iteration polish."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        d, S = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 0:

            if deadline is not None and time.monotonic() > float(deadline):
                break

            if k >= m:

            else:

                w = np.zeros(m)
                w[idx] = 1.0 / k

            h = corr_pair(a, w)
            h = np.maximum(h, 1e-15)

            for t_p in (0.05, 0.1, 0.2, 0.4):
                if deadline is not None and time.monotonic() > float(deadline):

                a_new = np.exp(log_new - np.max(log_new))

                sc = exact_score_a(a_new)

                    best_a = a_new.copy()

def spectral_shaping(a, num_steps=20, deadline=None):
    """Iteratively shape the spectrum to flatten the autocorrelation."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    N = next_power_of_two(2 * n - 1)

    for step in range(int(num_steps)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        A = np.fft.rfft(a, N)
        # Compute current autocorrelation in freq domain
        R = np.abs(A) ** 2
        # We want IDFT(R) to be flat.
        # Gradient: reduce components of R that contribute to the peak.
        r = np.fft.irfft(R, N)
        r_peak = r[n - 1]  # center of linear conv
        # Actually the peak could be anywhere in the first 2n-1 elements
        r_lin = r[:2*n-1]
        peak_idx = int(np.argmax(r_lin))

        # Compute which frequency components contribute most to the peak
        # r_lin[peak_idx] = (1/N) * sum_k R[k] * exp(2*pi*i*k*peak_idx/N)
        # To reduce this, we want to reduce R[k] where the phase aligns with the peak.

        # Simple approach: reduce high-frequency components (they contribute to sharp peaks)
        freqs = np.arange(len(A))
        # Damp high frequencies
        damping = np.exp(-freqs / (len(A) * 0.1))
        A_new = A * (1.0 + 0.05 * (damping - 1.0))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_a(a_new)
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new

    return best_sc, best_a


def generate_inits(n, parent, rng):
    """Generate diverse initial candidates."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 1e-3
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-12:
            arr = np.ones(n)
        inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    # Near-uniform with small sinusoidal perturbations
    for amp in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        for freq in (1, 2, 3, 4, 6, 10):
            add(1.0 + amp * np.sin(2.0 * np.pi * freq * x))
            if len(inits) > 30:
                break
        if len(inits) > 30:
            break

    # L-shapes
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x))
    # V-shapes
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Random smooth (spectral)
    Nf = next_power_of_two(n)
    for _ in range(10):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(0.5, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= float(noise.mean())
        std = float(np.std(noise))
        if std > 1e-12:
            noise /= std
        add(1.0 + float(rng.uniform(0.1, 0.6)) * noise)

    # Multi-sinusoid
    for _ in range(8):
        ncomp = int(rng.integers(2, 6))
        sig = np.zeros(n)
        for _c in range(ncomp):
            f = int(rng.integers(1, 15))
            a_m = float(rng.uniform(0.05, 0.4))
            ph = float(rng.uniform(0, 2 * np.pi))
            sig += a_m * np.sin(2.0 * np.pi * f * x + ph)
        add(1.0 + sig)

    # Bimodal / structured
    for width in (0.1, 0.15, 0.25):
        for c1, c2 in ((0.3, 0.7), (0.2, 0.8)):
            add(np.exp(-(((x - c1) / width) ** 2)) + np.exp(-(((x - c2) / width) ** 2)))

    return inits[:40]


def perturb_sequence(a, rng, strength=0.1):
    """Perturb for exploration."""
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        logp = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        k = max(3, min(31, int(n) // 100))
        if k % 2 == 0:
            k += 1
        if 3 <= k < n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = np.exp(logp + strength * 0.3 * noise - logp.max())
    elif mode == 2:
        t = float(rng.uniform(0.01, 0.1))
        q = (1.0 - t) * a + t * (np.ones(n) / n)
    elif mode == 3:
        # Spectral perturbation
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        freqs = np.arange(len(A), dtype=np.float64)
        mask = np.exp(-freqs / (len(A) * 0.2))
        noise = rng.standard_normal(len(A)) * mask
        noise[0] = 0.0
        A_new = A + strength * 0.05 * noise
        q = np.fft.irfft(A_new, N)[:n]
    elif mode == 4:
        # Smooth
        k = int(rng.integers(5, max(7, n // 50)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode="same")
    else:
        q = a * (1.0 + strength * 0.1 * np.sin(2.0 * np.pi * rng.uniform(1, 10) * np.linspace(0, 1, n)))

    q = np.nan_to_num(q, nan=0.0, posinf=1e6, neginf=0.0)
    return normalize_a(np.clip(q, 0.0, None))


def optimize_candidate(a_start, deadline, rng):
    """Full optimization pipeline for a single candidate."""
    a = normalize_a(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    # Phase 1: LSE with moderate beta (L-BFGS-B)
    for beta in (500.0, 2000.0, 10000.0):
        if time.monotonic() > deadline:
            break
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            break
        maxfun = int(min(800, max(100, rem * 3.0)))
        sc, a2 = run_lbfgs_lse(a, beta, deadline, maxfun)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    # Phase 2: LSE with high beta (Adam)
    for beta in (50000.0, 200000.0, 1000000.0):
        if time.monotonic() > deadline:
            break
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            break
        max_it = int(min(500, max(50, rem * 10.0)))
        sc, a2 = adam_lse(a, beta, max_it, lr=0.02, deadline=deadline)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = normalize_a(a2)

    # Phase 3: Active-set gradient polish
    rem = float(deadline - time.monotonic())
    if rem > 5.0:
        sc, a2 = active_set_polish(a, max_iters=200, deadline=deadline)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = a2

    # Phase 4: Mirror descent
    rem = float(deadline - time.monotonic())
    if rem > 3.0:
        sc, a2 = mirror_descent_polish(a, max_iters=100, deadline=deadline)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = min(max(parent.size, 2048), 16384)

    # Generate diverse initializations
    inits = generate_inits(n_primary, parent, rng)
    print("Generated {} initializations".format(len(inits)))

    best_sc = np.inf
    best_seq = None

    # Evaluate all inits quickly
    scored_inits = []
    for i, init in enumerate(inits):
        sc = exact_score_a(init)
        scored_inits.append((sc, init))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = init.copy()

    scored_inits.sort(key=lambda x: x[0])
    print("Best init score: {:.8f}".format(best_sc))

    # Allocate time to top candidates
    # Top 3 get full optimization, next 5 get partial, rest get minimal
    n_full = min(3, len(scored_inits))
    n_partial = min(5, len(scored_inits) - n_full)
    t_per_full = 0.55 * total_budget / max(1, n_full)
    t_per_partial = 0.15 * total_budget / max(1, n_partial)
    t_per_minimal = 0.05 * total_budget / max(1, len(scored_inits) - n_full - n_partial)

    # Optimize top candidates (full)
    t_alloc = t_start + 0.05 * total_budget
    for i in range(n_full):
        if time.monotonic() > deadline - 30.0:
            break
        chunk_deadline = min(t_alloc + (i + 1) * t_per_full, deadline - 20.0)
        sc, seq = optimize_candidate(scored_inits[i][1], chunk_deadline, rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Full cand {}: {:.8f}".format(i, best_sc))

    # Optimize next candidates (partial - just LSE + active set)
    t_alloc = t_start + 0.60 * total_budget
    for i in range(n_full, n_full + n_partial):
        if time.monotonic() > deadline - 20.0:
            break
        chunk_deadline = min(t_alloc + (i - n_full + 1) * t_per_partial, deadline - 15.0)
        a = normalize_a(scored_inits[i][1])
        # Quick LSE
        for beta in (5000.0, 50000.0):
            if time.monotonic() > chunk_deadline - 2.0:
                break
            rem = float(chunk_deadline - time.monotonic())
            if rem < 3.0:
                break
            maxfun = int(min(300, max(50, rem * 5.0)))
            sc, a2 = run_lbfgs_lse(a, beta, chunk_deadline, maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Partial cand {}: {:.8f}".format(i, best_sc))
            a = normalize_a(a2)
        # Quick active set
        if time.monotonic() < chunk_deadline - 2.0:
            sc, a2 = active_set_polish(a, max_iters=50, deadline=chunk_deadline)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()

    # Multi-length exploration
    t_len = t_start + 0.80 * total_budget
    other_lengths = [8192, 12288, 16384]
    if n_primary in other_lengths:
        other_lengths.remove(n_primary)

    for n_other in other_lengths[:2]:
        if time.monotonic() > t_len or time.monotonic() > deadline - 15.0:
            break
        a_other = normalize_a(resample_sequence(best_seq, n_other))
        sc_0 = exact_score_a(a_other)
        # Quick optimization
        for beta in (10000.0, 100000.0, 500000.0):
            if time.monotonic() > min(t_len, deadline - 10.0) - 3.0:
                break
            rem = float(min(t_len, deadline - 10.0) - time.monotonic())
            if rem < 5.0:
                break
            maxfun = int(min(400, max(80, rem * 3.0)))
            sc, a2 = run_lbfgs_lse(a_other, beta, min(t_len, deadline - 10.0), maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Length {}: {:.8f}".format(n_other, best_sc))
            a_other = normalize_a(a2)
        if time.monotonic() < min(t_len, deadline - 10.0) - 5.0:
            sc, a2 = active_set_polish(a_other, max_iters=80, deadline=min(t_len, deadline - 10.0))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Length {} polish: {:.8f}".format(n_other, best_sc))

    # Perturbation restarts
    t_pert = deadline - 10.0
    n_pert = 3
    for pi in range(n_pert):
        if time.monotonic() > t_pert:
            break
        chunk = min(t_start + (0.85 + 0.05 * (pi + 1)) * total_budget, t_pert)
        p = perturb_sequence(best_seq, rng, strength=float(rng.uniform(0.03, 0.15)))
        for beta in (50000.0, 500000.0):
            if time.monotonic() > chunk:
                break
            rem = float(chunk - time.monotonic())
            if rem < 3.0:
                break
            maxfun = int(min(300, max(50, rem * 5.0)))
            sc, a2 = run_lbfgs_lse(p, beta, chunk, maxfun)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("Perturb {}: {:.8f}".format(pi, best_sc))
            p = normalize_a(a2)
        if time.monotonic() < chunk:
            sc, a2 = active_set_polish(p, max_iters=50, deadline=chunk)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()

    # Final polish
    if time.monotonic() < deadline - 5.0:
        sc, a2 = active_set_polish(best_seq, max_iters=50, deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
    if time.monotonic() < deadline - 3.0:
        sc, a2 = mirror_descent_polish(best_seq, max_iters=30, deadline=deadline - 3.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
