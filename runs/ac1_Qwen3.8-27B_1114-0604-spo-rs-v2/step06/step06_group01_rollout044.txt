```python
"""
Multi-phase optimizer combining exact LP alternating (via HiGHS), spectral
initialization, progressive L-BFGS with LSE smoothing, and peak-specific
polishing. Key improvements: exact LP solver replacing FISTA approximation,
reduced-constraint LP for speed, adaptive line search, and multi-start
diversity.
"""

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.sparse import csr_matrix
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_lp_reduced(a, M, K=300):
    """
    Solve reduced LP: max sum(b) s.t. conv(a,b)[k] <= M for top-K lags, b >= 0.
    Returns optimized b or None.
    """
    n = len(a)
    c = self_conv(a)
    if K >= 2 * n - 1:
        top_k = np.arange(2 * n - 1)
    else:
        top_k = np.argsort(c)[-K:]
    top_k = np.sort(top_k)

    rows = []
    cols = []
    vals = []
    row_map = {}
    for i, k in enumerate(top_k):
        row_map[int(k)] = i
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(row_map[int(k)])
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return None

    C = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))

    c_obj = -np.ones(n)
    b_ub = np.full(len(top_k), M)
    bounds = [(0.0, None)] * n

    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success and result.x is not None:
        b = np.maximum(result.x, 0.0)
        if b.sum() > 1e-12:
            return b
    return None


def lp_alternating_exact(a, max_outer_iters=30, K_constraints=300,
                         time_limit=None, beta_lp=None):
    """
    LP-based alternating with exact HiGHS solver and adaptive line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())

        b = solve_lp_reduced(a, M, K=K_constraints)
        if b is None:
            break
        Sb = float(b.sum())
        if Sb < 1e-12:
            break
        b = b / Sb

        # Adaptive line search using minimize_scalar
        def obj_of_t(t):
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                return 1e10
            a_try = a_try / St
            return compute_score(a_try)

        res_ls = minimize_scalar(obj_of_t, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 0.01})
        t_opt = res_ls.x if res_ls.success else 0.3

        a_new = (1.0 - t_opt) * a + t_opt * b
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc * (1.0 - 1e-8):
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                break

        a = a_new

    return best_a, best_sc


def error_reduction(a, n_iters=80, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def spectral_initialization(n, rng, target_shape='raised_cosine'):
    """Generate a non-negative sequence via iterative FFT projection."""
    a = np.zeros(n)
    if target_shape == 'raised_cosine':
        x = np.arange(n) / n
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif target_shape == 'gaussian':
        x = np.arange(n) / n
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif target_shape == 'random':
        a = np.abs(rng.standard_normal(n))
    else:
        a = np.ones(n) * 0.5

    a = a / a.sum()
    M = 1 << (2 * n - 1).bit_length()

    for it in range(50):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude (but with a taper to avoid edge effects)
        target_mag = np.ones(M)
        # Apply a window to the target to reduce edge effects
        w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(M) / M))
        target_mag = np.sqrt(mag) * np.ones(M)  # keep current magnitude as target
        # Actually, let's use a "flattened" target
        mag_mean = float(mag.mean())
        target_mag = np.ones(M) * mag_mean
        # Blend: 50% flat + 50% current
        target_mag = 0.3 * np.ones(M) * mag_mean + 0.7 * mag
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        a = a_new

    return a


def peak_local_search(a, n_steps=300, rng=None):
    """Targeted local search reducing the convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        direction *= 2.0

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.05, max(0.001, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3))
        improved = False
        for mult in [0.2, 0.4, 0.6, 1.0, 1.5, 2.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 10 == 0:
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            s_tmp = a.sum()
            if s_tmp > 1e-12:
                a = a / s_tmp

    return best_a


def subgradient_projection(a, n_steps=200, step_init=0.01, rng=None):
    """Subgradient method with projection onto {a>=0, sum(a)=1}."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        grad = np.zeros(n)
        if j_end >= j_start:
            grad[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        grad *= 2.0

        a_raw = a - step * grad
        # Project onto {a >= 0, sum(a) = 1} using bisection
        # Find tau such that sum(max(0, a_raw - tau)) = 1
        lo = float(a_raw.min()) - 1.0
        hi = float(a_raw.max()) + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            projected = np.maximum(a_raw - mid, 0.0)
            s_proj = float(projected.sum())
            if s_proj > 1.0:
                lo = mid
            else:
                hi = mid
        tau = (lo + hi) / 2.0
        a_new = np.maximum(a_raw - tau, 0.0)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.9

        # Occasional noise
        if it % 20 == 0:
            a_new = a_new * (1.0 + 0.005 * rng.standard_normal(n))
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

        a = a_new

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)

    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x + 0.5))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))

    # Spectral initializations
    inits.append(spectral_initialization(n, rng, 'raised_cosine'))
    inits.append(spectral_initialization(n, rng, 'gaussian'))

    return inits


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: spectral init, exact LP alternating, progressive L-BFGS,
    peak-specific polishing, and subgradient projection.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [6962, 8192, 10240]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Multi-start initialization + quick L-BFGS (20% time)
    # ==================================================================
    phase1_deadline = t_start + 0.20 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Multi-Start Init + Quick L-BFGS ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = build_initializations(n, rng, parent)
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = np.maximum(a0, 0.0)
            s0 = float(a0.sum())
            if s0 < 1e-12:
                continue
            a0 = a0 / s0

            # Quick error reduction
            a_er = error_reduction(a0, n_iters=40, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

            # Quick L-BFGS with low beta
            if time.monotonic() < phase1_deadline - 20:
                sc2, a_opt, _ = run_lbfgs(a_er, 100, maxiter=2000, maxfun=50000)
                report(sc2, a_opt, f"LBFGS-low n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Exact LP Alternating (40% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: Exact LP Alternating ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP with increasing constraint count
    lp_configs = [
        (200, 20),   # K=200, 20 outer iters
        (300, 20),
        (500, 15),
        (800, 10),
    ]

    for ci, (K_val, max_outer) in enumerate(lp_configs):
        if time.monotonic() > phase2_deadline - 60:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP config {ci}: K={K_val}, max_outer={max_outer}, "
              f"remaining={remaining:.0f}s")

        a_out, sc_out = lp_alternating_exact(
            a_work, max_outer_iters=max_outer, K_constraints=K_val,
            time_limit=phase2_deadline - 30
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP K={K_val}")
        a_work = a_out.copy()

    # Try alternate length
    for n_alt in lengths:
        if n_alt == n_main:
            continue
        if time.monotonic() > phase2_deadline - 60:
            break
        print(f"  LP alt length n={n_alt}")
        a_test = np.interp(np.linspace(0, 1, n_alt),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        s_t = float(a_test.sum())
        if s_t > 1e-12:
            a_test = a_test / s_t
        a_out, sc_out = lp_alternating_exact(
            a_test, max_outer_iters=15, K_constraints=300,
            time_limit=phase2_deadline - 30
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n_alt}")
        n_main = len(best_a)

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Progressive L-BFGS (25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: Progressive L-BFGS ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [100, 200, 500, 1000, 2000, 4000, 8000, 15000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 30:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        maxit = 6000 if beta < 1000 else 4000
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=200000,
                                  gtol=1e-13)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak polish + subgradient + final (15% time)
    # ==================================================================
    print(f"\n--- Phase 4: Peak Polish & Final ---")

    # Peak-specific local search
    a_peak = peak_local_search(best_a, n_steps=300, rng=rng)
    sc = compute_score(a_peak)
    report(sc, a_peak, "peak search")

    # Subgradient projection
    a_sg = subgradient_projection(best_a, n_steps=200, step_init=0.01, rng=rng)
    sc = compute_score(a_sg)
    report(sc, a_sg, "subgradient")

    # Another LP round on the best
    if time.monotonic() < deadline - 120:
        print("  Final LP polish...")
        a_lp = best_a.copy()
        S = float(a_lp.sum())
        if S > 0:
            a_lp = a_lp / S
        a_out, sc_out = lp_alternating_exact(
            a_lp, max_outer_iters=10, K_constraints=500,
            time_limit=deadline - 60
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 40:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 20000, maxiter=8000, maxfun=400000,
                                  gtol=1e-14)
        report(sc, a_opt, "final LBFGS b=20000")

    # Another peak search
    if time.monotonic() < deadline - 20:
        a_final = peak_local_search(best_a, n_steps=150, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final peak")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
```
