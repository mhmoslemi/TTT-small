def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def power_sum_obj_grad(a, p):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    cp_sum = float(cp.sum())
    f_p = cp_sum ** (1.0 / p)
    score = 2.0 * n * f_p / (S * S)
    w = cp ** (1.0 - 1.0 / p)
    w = w / w.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(w, M2_fft)

    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * f_p / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,

    )
    a_out = np.clip(res.x, 0.0, 1000.0)

    if S < 1e-6:
        a_out = a0.copy()
    else:

def fienup_multivariant(a, n_iters=500, alpha_init=0.8, alpha_end=0.05,
                        target='flat', exponent=1.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    freqs = np.fft.rfftfreq(M_fft)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init * (1.0 - progress) ** 1.5 + alpha_end
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        if target == 'flat':
            P_target = np.full_like(P, P_mean)
        elif target == 'dc_boost':
            norm_f = np.abs(freqs) * 2 * n
            shape = 1.0 + 0.5 * np.exp(-norm_f / (n * 0.5))
            P_target = P_mean * shape / float(shape.mean())
        else:
            norm_f = np.abs(freqs) * 2 * n
            shape = 1.0 - 0.3 * np.exp(-norm_f / (n * 0.3))
            shape = np.maximum(shape, 0.1)
            P_target = P_mean * shape / float(shape.mean())
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)
        ratio = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * ratio ** (alpha * exponent)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def stochastic_restarts(a, rng, n_restarts=15, deadline=None):
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    for r in range(n_restarts):
        if deadline is not None and time.monotonic() > deadline:
            break
        strategy = r % 5
        if strategy == 0:
            a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        elif strategy == 1:
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            noise = 0.03 * rng.standard_normal(n)
            if j_end > j_start:
                noise[j_start:j_end + 1] += 0.05 * rng.standard_normal(j_end - j_start + 1)
            a_pert = a * (1.0 + noise)
        elif strategy == 2:
            x = np.arange(n) / n
            freq = rng.integers(1, 12)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 3:
            block_size = max(1, n // 16)
            block = rng.integers(0, max(1, n - block_size))
            a_pert = a.copy()
            a_pert[block:block + block_size] *= (1.0 + 0.15 * rng.standard_normal(block_size))
        else:
            mask = rng.random(n) < 0.3
            a_pert = a.copy()
            a_pert[mask] *= (0.5 + rng.random(int(mask.sum())))
        a_pert = np.maximum(a_pert, 0.0)
        S_p = a_pert.sum()
        if S_p < 1e-12:
            continue
        a_pert = a_pert / S_p
        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=4000, maxfun=80000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Restart {r} (strat={strategy}): {sc:.8f}")
    return best_a, best_sc


def adapt_length(parent, target_n):
    n_old = len(parent)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)

    S = a.sum()
    if S > 1e-12:

    return a

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
        n_primary = len(parent)
    else:
        print("No parent, starting fresh with n=2048")
        n_primary = 2048
        best_a = np.ones(n_primary) / n_primary

    print(f"Initial best: {best_score:.8f}, n={n_primary}")

    # Phase 1: Quick Fienup variants to improve starting point
    p1_end = t_start + 0.08 * (total_budget - 60.0)
    print(f"\n=== Phase 1: Fienup warmup ===")
    a_f1 = fienup_multivariant(best_a, n_iters=300, target='flat')
    sc = compute_score(a_f1)
    update(sc, a_f1, "FIENUP-flat")

    if time.monotonic() < p1_end:
        a_f2 = fienup_multivariant(best_a, n_iters=200, target='dc_boost')
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-dc")

    if time.monotonic() < p1_end:
        a_f3 = fienup_multivariant(best_a, n_iters=200, target='dc_dip')
        sc = compute_score(a_f3)
        update(sc, a_f3, "FIENUP-dip")

    # Phase 2: L-BFGS with LSE, adaptive beta (main workhorse)
    p2_end = t_start + 0.35 * (total_budget - 60.0)
    print(f"\n=== Phase 2: L-BFGS LSE schedule ===")
    betas = [50, 100, 200, 500, 1000, 3000, 8000, 20000, 50000, 100000]
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        frac = (i + 1) / len(betas)
        maxiter = int(min(25000, 2000 + remaining * 15 * frac))
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        dt = time.monotonic() - t0
        print(f"    LSE beta={beta:>6d}: score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE-b{beta}")

    # Phase 3: Power-sum L-BFGS (alternative smooth approximation)
    p3_end = t_start + 0.50 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Power-sum L-BFGS ===")
    for p_val in [4, 8, 16, 32, 64]:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(10000, 1000 + remaining * 10))
        t0 = time.monotonic()
        sc, a_opt = run_power_lbfgs(best_a, p_val, maxiter=maxiter, maxfun=maxiter * 15)
        dt = time.monotonic() - t0
        print(f"    Power p={p_val:>3d}: score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"PWR-p{p_val}")

    # Phase 4: Peak-directed multiplicative updates
    p4_end = t_start + 0.65 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Peak-directed updates ===")
    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        n_iters = int(min(5000, max(500, remaining * 50)))
        a_pd, sc_pd = peak_directed_update(best_a, n_iters=n_iters, n_top=5, eta_init=0.02)
        update(sc_pd, a_pd, "PEAK-DIRECTED")

    if time.monotonic() < p4_end:
        a_mw, sc_mw = multiplicative_weight_update(best_a, n_iters=3000, top_k=3, eta_init=0.02)
        update(sc_mw, a_mw, "MULTI-WEIGHT")

    # Phase 5: Variance minimization
    p5_end = t_start + 0.75 * (total_budget - 60.0)
    print(f"\n=== Phase 5: Variance minimization ===")
    if time.monotonic() < p5_end:
        remaining = p5_end - time.monotonic()
        n_steps = int(min(4000, max(500, remaining * 30)))
        a_vm, sc_vm = variance_minimization_gradient(best_a, n_steps=n_steps)
        update(sc_vm, a_vm, "VAR-MIN")

    # Phase 6: Fienup again (potentially from improved starting point)
    p6_end = t_start + 0.82 * (total_budget - 60.0)
    print(f"\n=== Phase 6: Fienup refinement ===")
    if time.monotonic() < p6_end:
        a_fr = fienup_multivariant(best_a, n_iters=400, target='flat', alpha_init=0.4)
        sc = compute_score(a_fr)
        update(sc, a_fr, "FIENUP-refined")

    # Phase 7: Coordinate descent polish
    p7_end = t_start + 0.88 * (total_budget - 60.0)
    print(f"\n=== Phase 7: Coordinate descent ===")
    if time.monotonic() < p7_end:
        a_cd = coordinate_descent_peaks(best_a, n_sweeps=10, n_peaks=8)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD-DESCT")

    # Phase 8: Stochastic restarts
    p8_end = t_start + 0.94 * (total_budget - 60.0)
    print(f"\n=== Phase 8: Stochastic restarts ===")
    a_sr, sc_sr = stochastic_restarts(best_a, rng, n_restarts=12, deadline=p8_end)
    update(sc_sr, a_sr, "STOCH-RESTART")

    # Phase 9: Final high-beta L-BFGS polish
    print(f"\n=== Phase 9: Final polish ===")
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = int(min(15000, max(2000, remaining * 12)))
        sc, a_fin = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_fin, "FINAL-LSE-100k")

    if time.monotonic() < deadline - 15:
        sc, a_fin = run_lbfgs(best_a, 300000, maxiter=5000, maxfun=100000)
        update(sc, a_fin, "FINAL-LSE-300k")

    if time.monotonic() < deadline - 5:
        a_cd2 = coordinate_descent_peaks(best_a, n_sweeps=5, n_peaks=5)
        sc = compute_score(a_cd2)
        update(sc, a_cd2, "FINAL-CD")

    # Try a different length if time permits
    if time.monotonic() < deadline - 10 and n_primary != 4096:
        print(f"\n=== Trying n=4096 ===")
        a_4096 = adapt_length(best_a, 4096)
        sc = compute_score(a_4096)
        print(f"  Interpolated to 4096: {sc:.8f}")
        if sc < best_score * 1.001:
            a_opt_4096, sc_opt = run_lbfgs(a_4096, 5000, maxiter=5000, maxfun=100000)
            update(sc_opt, a_opt_4096, "N4096-LBFGS")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
