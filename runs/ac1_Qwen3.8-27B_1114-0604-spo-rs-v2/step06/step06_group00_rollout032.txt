"""
Improved optimizer for minimizing 2n*max(conv(a,a))/S^2.

1. Adaptive LSE with progress-based beta scheduling

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    score = 2n * LSE_beta(conv(a,a)) / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da_i = (2/beta) * (a_rev * p)[i + n - 1]
    a_rev = a[::-1]
    M_conv = n + (2 * n - 1) - 1  # = 3n - 2
    M_fft = 1 << (M_conv - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    R = np.fft.rfft(a_rev, M_fft)
    conv_res = np.fft.irfft(P * R, M_fft)[:M_conv]
    grad_L = (2.0 / beta) * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def lbfgs_optimize(a0, beta, maxiter=3000, maxfun=60000, ftol=1e-16, gtol=1e-12):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': ftol, 'gtol': gtol,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        S = a_out.sum()
        if S < 1e-6:
            a_out = a0
        else:
            a_out = a_out / S
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def adaptive_lse_anneal(a0, deadline, min_time_per_beta=5):
    """

    Stop when time is up or no progress for consecutive betas.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_sc = compute_score(a)
    best_a = a.copy()
    
    beta = 20
    no_progress = 0
    max_no_progress = 4
    
    while time.monotonic() < deadline - 30 and no_progress < max_no_progress:
        remaining = deadline - 30 - time.monotonic()
        if remaining < min_time_per_beta:
            break
        
        # Scale maxiter based on remaining time and beta
        time_per_iter = max(0.005, 0.001 + beta * 0.0001)
        maxiter = max(200, int(min(remaining * 0.8 / time_per_iter, 8000)))
        maxfun = maxiter * 30
        
        t0 = time.monotonic()
        sc, a_new = lbfgs_optimize(a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_progress = 0
            print(f"    LSE beta={beta:>6}: score={sc:.8f} (improved, {dt:.1f}s)")
        else:
            no_progress += 1
            print(f"    LSE beta={beta:>6}: score={sc:.8f} (no improvement, {dt:.1f}s)")
        
        # Update a for next iteration (use the new result even if not better)
        a = normalize(a_new)
        
        # Increase beta adaptively
        if no_progress == 0:
            beta = int(beta * 1.5)  # Moderate increase when making progress
        else:
            beta = int(beta * 2.0)  # Aggressive increase when stuck

def fienup_enhanced(a, n_iters=500, mode='adaptive'):
    """
    Enhanced Fienup with adaptive flattening strength and multiple strategies.
    """
    n = len(a)
    a = normalize(a.copy())

    for it in range(n_iters):
        if time.monotonic() > getattr(fienup_enhanced, '_deadline', 1e18):
            break
            
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        P_mean = float(P.mean())
        P_std = float(P.std())
        
        if mode == 'adaptive':
            # Adaptively choose the flattening strategy
            if progress < 0.3:
                # Early: gentle flattening
                alpha = 0.1 + 0.3 * progress
                P_new = (1 - alpha) * P + alpha * P_mean
            elif progress < 0.7:
                # Mid: quantile clipping
                q = 0.8 + 0.15 * progress
                thresh = np.quantile(P, q)
                P_new = P.copy()
                P_new[P_new > thresh] = thresh
            else:
                # Late: aggressive flattening with boost of low frequencies
                alpha = 0.4 + 0.3 * (progress - 0.7) / 0.3
                P_new = (1 - alpha) * P + alpha * P_mean
                P_low = P_new < P_mean * 0.3
                P_new[P_low] = P_mean * 0.5
        
        elif mode == 'quantile':
            q = 0.7 + 0.25 * progress
            thresh = np.quantile(P, q) * (1.0 + 0.3 * (1 - progress))
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        
        elif mode == 'weighted':
            # Weighted flattening: give more weight to high frequencies
            w = 1.0 + 0.5 * np.linspace(0, 1, len(P))
            P_mean_w = float(np.sum(P * w) / np.sum(w))
            alpha = 0.2 + 0.5 * progress
            P_new = (1 - alpha) * P + alpha * P_mean_w * (w / w.mean())
        
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def subgradient_peak_shave(a, n_iters=3000, top_k=5, momentum=0.9):
    """
    Subgradient descent on the max with momentum and adaptive step size.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Running average for momentum
    v = np.zeros(n)
    step = 0.005
    
    for it in range(n_iters):
        if time.monotonic() > getattr(subgradient_peak_shave, '_deadline', 1e18):
            break
            
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"    SubGrad it={it}: score={sc:.8f}")
        
        # Compute subgradient (average over top-K peaks)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # d(conv_k)/da_i = 2 * a_{k-i} for valid i
                for i in range(j_start, j_end + 1):
                    sg[i] += 2.0 * a[k - i]
        
        # Normalize by number of peaks
        sg /= top_k
        
        # Scale by score components
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        grad_norm = np.linalg.norm(v)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step size (Polyak-style)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        step_polyak = (sc - lower_bound) / (grad_norm ** 2 + 1e-30)
        step_effective = min(step, step_polyak * 0.5, 0.02)
        step_effective = max(step_effective, 1e-10)
        
        a = a - step_effective * v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay step
        step *= 0.9995

def coordinate_polish(a, n_sweeps=5):

def interpolate_to(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def diverse_perturb(a, rng, mode=0):
    """Generate diverse perturbations."""
    n = len(a)
    if mode == 0:
        # Gaussian noise
        a_new = a + 0.01 * rng.standard_normal(n)
        return normalize(a_new)
    elif mode == 1:
        # Multiplicative with low-frequency structure
        x = np.arange(n, dtype=np.float64) / n
        pert = np.zeros(n)
        for m in range(1, 8):
            freq = rng.integers(1, 15)
            phase = 2.0 * np.pi * rng.uniform(0, 1)
            amp = 0.02 / m
            pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
        return normalize(a * (1.0 + pert))
    elif mode == 2:
        # Peak-targeted: boost non-peak areas
        c = fast_conv(a)
        k_peak = int(np.argmax(c))

        j_end = min(k_peak, n - 1)
        a_new = a.copy()
        peak_mask = np.ones(n, dtype=bool)
        peak_mask[j_start:j_end+1] = False
        n_boost = max(50, n // 25)
        candidates = np.where(peak_mask)[0]
        if len(candidates) > n_boost:
            selected = rng.choice(candidates, n_boost, replace=False)
        else:
            selected = candidates
        delta = 0.003 / n
        a_new[selected] += delta
        return normalize(a_new)
    elif mode == 3:
        # Spectral perturbation
        M_fft = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, M_fft)
        # Random phase perturbation
        phase_pert = 0.05 * rng.standard_normal(len(A))
        A_new = A * np.exp(1j * phase_pert)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        return normalize(a_new)
    else:
        # Mixed: combine two types
        a1 = a + 0.005 * rng.standard_normal(n)
        a1 = np.maximum(a1, 0)
        x = np.arange(n, dtype=np.float64) / n
        freq = rng.integers(2, 10)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        a1 *= (1.0 + 0.015 * np.cos(2 * np.pi * freq * x + phase))
        return normalize(a1)
