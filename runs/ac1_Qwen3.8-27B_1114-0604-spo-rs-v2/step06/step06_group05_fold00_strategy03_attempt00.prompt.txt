def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def weighted_score_grad(a, w):
    """Weighted convolution score and gradient.
    Objective: sum_k w_k * conv_k(a) / (sum(a))^2

    # = 2 * (rev_a * w)[i+n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * w_num) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

def iterative_reweighting(a0, n_outer=50, n_inner=3000, maxfun_per_inner=5000):
    """
    Iterative reweighting: at each outer iteration, compute weights from current
    peaks, then solve the weighted subproblem using L-BFGS.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for outer in range(n_outer):
        # Compute current convolution
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Compute weights: emphasize regions near the max
        # Use a power law: w_k = (c_k / c_max)^p
        # Higher p -> more aggressive peak targeting
        p = 1.0 + 2.0 * outer / n_outer
        c_norm = c / c_max
        w = c_norm ** p
        # Also add small uniform component to avoid degenerate weights
        w = 0.1 * (1.0 / (2 * n - 1)) + 0.9 * w
        w = w / w.sum()
        
        # Solve weighted subproblem
        a = run_lbfgs(a, weighted_score_grad, (w,), maxiter=n_inner, maxfun=maxfun_per_inner)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if (outer + 1) % 10 == 0:
                print(f"    IR outer={outer}: score={best_sc:.8f}")
    
    return best_a


def nesterov_momentum(a0, beta, n_iters=20000, lr_init=0.01, lr_decay=0.9999, momentum=0.9):
    """Nesterov momentum gradient descent on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        
        # Nesterov lookahead
        a_look = a - momentum * v
        if it % 50 == 0 or it < 5:
            sc_look, grad_look = lse_score_grad(a_look, beta)
            v = momentum * v + lr * grad_look
        else:
            v = momentum * v + lr * grad
        
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        lr *= lr_decay
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def peak_targeted_refinement(a0, n_rounds=50, top_k=20):
    """Directly reduce top peaks by adjusting contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top-k peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        # For each peak, find the elements that contribute most
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            # Contribution of a_j to c_k is a_j * a_{k-j}
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            
            # Reduce the top-contributing elements
            n_reduce = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            
            for li in top_local:
                j = j_start + li
                a[j] *= 0.85  # Reduce by 15%
        
        # Renormalize
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def random_perturb(a0, rng, strength=0.05):
    """Add random perturbation and renormalize."""
    a = a0.copy()
    perturbation = rng.standard_normal(len(a)) * strength
    a += perturbation
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a0.copy()
    return a / S


def build_diverse_inits(n, rng, parent):

    inits = []

    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.3 * np.cos(2.0 * np.pi * x), 0.01))

        n_harmonics = rng.integers(3, 15)

def subgradient_multi_peak(a0, n_iters=5000, n_peaks=30, step_init=0.03, decay=0.9999):
    """Subgradient method targeting multiple peaks simultaneously."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a
