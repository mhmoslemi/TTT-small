"""
Optimizes non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: LSE-continuation for smooth initial descent, then Nesterov-accelerated
projected subgradient with adaptive active-set, diverse multi-restart global search,
and multi-length exploration. Uses FFT for all gradient/convolution computations.

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def fft_self_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    out = np.fft.irfft(A * A, size)[:2 * n - 1]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def fft_corr_pair(a, w):
    """Compute aligned cross-correlation: result[r] = sum_j w[j] * a[j-r] (valid terms)."""
    n = len(a)
    m = len(w)
    M = n + m - 1
    N = next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return np.nan_to_num(full[n - 1:2 * n - 1], nan=0.0, posinf=0.0, neginf=0.0)


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / S)


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    full = fft_self_conv(a)
    cmax = float(np.max(full))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def active_weights(d, k):
    m = len(d)
    if m <= 0:
        return np.array([], dtype=np.float64)
    k = min(k, m)
    if k <= 0:
        return np.ones(m) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def subgradient_score(a, k_active=5):
    """Return (score, gradient) using top-k active lags."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    full = fft_self_conv(a)
    cmax = float(np.max(full))
    score = float(2.0 * n * cmax / (S * S))
    w = active_weights(full, k_active)
    h = fft_corr_pair(a, w)
    # Gradient of f = 2*n*cmax/S^2:
    # df/da[r] = (2*n/S^2)*(2/m_active)*h[r] - (4*n*cmax/S^3)
    grad = (4.0 * n / (S * S)) * h
    grad -= (4.0 * n * cmax / (S * S * S))
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return score, grad


def lse_score_grad(a, beta):
    """Smooth LSE approximation of max for gradient-based optimization."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    full = fft_self_conv(a)
    dmax = float(np.max(full))
    if dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (full - dmax)
    z = np.clip(z, -60.0, 60.0)
    e = np.exp(z)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.exp(z - z.max())
        se = float(e.sum())
        if se <= 0:
            return 1e10, np.zeros(n)
    w = e / se
    L = dmax + np.log(se) / beta
    # Score with LSE: 2*n*L/S^2
    score = 2.0 * n * L / (S * S)
    # Gradient
    h = fft_corr_pair(a, w)
    LDS = float(np.dot(w, full))
    grad = (4.0 * n * h) / (S * S)
    grad -= (4.0 * n * LDS) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return score, grad


def fun_lse(a, beta):
    s, g = lse_score_grad(a, float(beta))
    return s, g


def run_lbfgs_lse(a0, beta, maxfun=500):
    a = normalize(a0)
    n = len(a)
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-15, "gtol": 1e-12, "maxiter": max(50, maxfun // 2),
            "maxfun": int(maxfun), "maxls": 40}
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = normalize(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def nesterov_subgrad(a_start, max_iters=500, k_list=(3, 5, 8, 12, 20),
                     deadline=None, log_interval=20):
    """Nesterov-accelerated projected subgradient method."""
    a = normalize(a_start)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    v = np.zeros(n)  # momentum
    gamma = 2.0 / (2.0 + 1.0)  # Nesterov parameter
    eta = 0.5 / (np.sqrt(n) + 1.0)  # initial step size
    eta_min = 1e-8
    eta_decay = 0.995

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        if it % log_interval == 0:
            print("  NS iter {}: score={:.8f} eta={:.2e}".format(it, best_sc, eta))
        # Try multiple active set sizes
        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > deadline:
                break
            # Nesterov lookahead
            y = a + gamma * v
            sc_y, grad_y = subgradient_score(y, k)
            if not np.isfinite(sc_y):
                continue
            # Gradient step
            a_trial = y - eta * grad_y
            a_trial = normalize(np.maximum(a_trial, 0.0))
            sc_trial = exact_score(a_trial)
            if np.isfinite(sc_trial) and sc_trial < best_sc:
                best_sc = sc_trial
                best_a = a_trial.copy()
                v = a_trial - a
                a = a_trial.copy()
                improved = True
                break
        if not improved:
            # Shrink step size
            eta *= eta_decay
            v *= 0.9
            if eta < eta_min:
                break
    return float(best_sc), best_a


def lse_trajectory(a_start, deadline, betas=None):
    """LSE continuation: gradually increase beta."""
    if betas is None:
        betas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    a = normalize(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0))
    t_per = total / len(betas)

    for i, beta in enumerate(betas):
        sub_dead = t0 + (i + 1) * t_per - 2.0
        sub_dead = min(sub_dead, float(deadline - 3.0))
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est_time = 0.02 * max(1.0, n / 8192.0)
            mf = int(min(800, max(64, (rem * 0.8) / est_time)))
            sc, a2 = run_lbfgs_lse(a, beta, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  LSE beta={}: score={:.8f}".format(beta, sc))
            a = normalize(a2)
    return float(best_sc), best_a


def perturb(a, rng, mode=None):
    """Generate a perturbed version of a."""
    p = normalize(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if mode is None:
        mode = int(rng.integers(0, 10))

    if mode == 0:
        # Gaussian multiplicative
        strength = float(rng.uniform(0.1, 0.6))
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * p + t * (1.0 / n)
    elif mode == 2:
        # Reverse + noise
        q = p[::-1]
        q = q * np.exp(0.15 * rng.standard_normal(n))
    elif mode == 3:
        # Anti-correlation push
        full = fft_self_conv(p)
        w = active_weights(full, int(rng.integers(2, 10)))
        h = fft_corr_pair(p, w)
        hmax = max(1e-15, float(np.max(np.abs(h))))
        strength = float(rng.uniform(0.1, 0.4))
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        # Smoothed log-space noise
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(21, max(3, int(n) // 250))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp += 0.2 * noise
        q = np.exp(logp - logp.max())
    elif mode == 5:
        # V-shape modulation
        x = np.linspace(0, 1, n)
        vshape = 1.0 + float(rng.uniform(0.1, 0.5)) * np.abs(2 * x - 1)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(7, max(9, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Frequency perturbation
        x = np.linspace(0, 1, n)
        f = float(rng.uniform(2, 12))
        amp = float(rng.uniform(0.05, 0.3))
        q = p * (1.0 + amp * np.cos(2 * np.pi * f * x))
    elif mode == 8:
        # Targeted: reduce peak contributors
        full = fft_self_conv(p)
        dmax = float(np.max(full))
        if dmax > 1e-12:
            w = active_weights(full, 3)
            h = fft_corr_pair(p, w)
            hmax = max(1e-15, float(np.max(h)))
            q = p * np.exp(-0.3 * h / hmax)
        else:
            q = p.copy()
    else:
        # Crossover-like: replace random segment with mirrored version
        seg_len = int(rng.integers(n // 10, n // 3))
        start = int(rng.integers(0, n - seg_len))
        q = p.copy()
        q[start:start + seg_len] = p[start + seg_len - 1:start - 1:-1]
        q *= np.exp(0.05 * rng.standard_normal(n))

    return normalize(np.clip(q, 0.0, None))


def resample(a, n_target):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n_target) / n_target
    if m == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent, cap=20):
    """Generate diverse initializations."""
    inits = []

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        inits.append(normalize(arr))

    if parent is not None and len(parent) > 0:
        add(resample(parent, n))
        add(resample(parent, n)[::-1])

    x = np.linspace(0, 1, n)
    add(np.ones(n) / n)

    for alpha in (0.3, 0.5, 0.8, 1.2, 2.0):
        add(1.0 + alpha * np.abs(2 * x - 1))

    for f in (1, 2, 3, 5, 7):
        add(1.0 + 0.5 * np.cos(2 * np.pi * f * x))
        add(1.0 + 0.5 * np.cos(np.pi * f * x))

    for w in (0.15, 0.25, 0.35):
        add(np.exp(-(((x - 0.2) / w) ** 2)) + np.exp(-(((x - 0.8) / w) ** 2)))

    add(np.exp(-2.0 * x) + np.exp(-2.0 * (1 - x)))
    add(np.ones(n) + 0.3 * np.sin(np.pi * x))

    # Random smooth
    Nf = next_pow2(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + 0.5 * noise)

    return inits


def polish_active_set(a, deadline, max_cycles=30):
    """Focused polishing: identify active set, optimize to make active lags equal."""
    a = normalize(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    for cyc in range(max_cycles):
        if deadline is not None and time.monotonic() > deadline:
            break
        full = fft_self_conv(a)
        cmax = float(np.max(full))
        if cmax < 1e-15:
            break
        # Active set: lags within 99% of max
        threshold = 0.99 * cmax
        active_idx = np.where(full >= threshold)[0]
        k_active = len(active_idx)
        if k_active == 0:
            k_active = 5
        k_active = max(1, min(k_active, 30))

        w = active_weights(full, k_active)
        h = fft_corr_pair(a, w)

        # Check KKT: h should be ~constant at optimum
        h_mean = float(np.mean(h))
        h_var = float(np.var(h))
        if h_var < 1e-10:
            break

        # Gradient step to make h more uniform
        S = float(np.sum(a))
        grad = (4.0 * n / (S * S)) * (h - h_mean)
        eta = 0.1 / (np.sqrt(n) + 1)
        a_trial = normalize(np.maximum(a - eta * grad, 0.0))
        sc = exact_score(a_trial)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Smaller step
            eta *= 0.3
            a_trial = normalize(np.maximum(a - eta * grad, 0.0))
            sc = exact_score(a_trial)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial
            else:
                break

    return float(best_sc), best_a

        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = min(max(parent.size, 2048), 16384)
        # Round to multiple of 128 for FFT efficiency
        n_primary = int(round(n_primary / 128.0) * 128)
        n_primary = max(2048, min(n_primary, 16384))

    # Initial sequence from parent
    if parent is not None:
        a0 = normalize(resample(parent, n_primary))
    else:
        a0 = np.ones(n_primary, dtype=np.float64) / n_primary
        x = np.linspace(0, 1, n_primary)
        a0 = normalize(1.0 + 0.8 * np.abs(2 * x - 1))

    best_sc = exact_score(a0)
    best_seq = a0.copy()
    print("Initial: score={:.8f} n={}".format(best_sc, n_primary))

    # === Phase 1: LSE trajectory (30% time) ===
    t1_dead = t_start + 0.30 * total_budget
    print("Phase 1: LSE trajectory (deadline {:.0f}s)".format(t1_dead - t_start))
    sc, seq = lse_trajectory(a0, t1_dead,
                             betas=[50, 200, 1000, 5000, 20000, 100000, 500000])
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 1 best: {:.8f}".format(best_sc))

    # === Phase 2: Nesterov subgradient (25% time) ===
    t2_dead = t_start + 0.55 * total_budget
    print("Phase 2: Nesterov subgradient (deadline {:.0f}s)".format(t2_dead - t_start))
    sc, seq = nesterov_subgrad(best_seq, max_iters=2000,
                               k_list=(3, 5, 8, 12, 20),
                               deadline=t2_dead)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 2 best: {:.8f}".format(best_sc))

    # === Phase 3: Active set polishing (10% time) ===
    t3_dead = t_start + 0.65 * total_budget
    print("Phase 3: Active set polish (deadline {:.0f}s)".format(t3_dead - t_start))
    sc, seq = polish_active_set(best_seq, t3_dead, max_cycles=100)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 3 best: {:.8f}".format(best_sc))

    # === Phase 4: Multi-restart exploration (25% time) ===
    t4_dead = t_start + 0.90 * total_budget
    print("Phase 4: Multi-restart (deadline {:.0f}s)".format(t4_dead - t_start))

    # Generate diverse inits
    inits = generate_inits(n_primary, rng, best_seq, cap=16)

    t_per_restart = (t4_dead - time.monotonic()) / max(1, len(inits))
    for ii, init in enumerate(inits):
        if time.monotonic() > t4_dead - 10.0:
            break
        restart_dead = min(time.monotonic() + t_per_restart * 0.9 - 2.0,
                           t4_dead - 5.0)
        if time.monotonic() > restart_dead:
            continue

        # Quick LSE
        if time.monotonic() < restart_dead - 5.0:
            sc_r, seq_r = lse_trajectory(init, restart_dead,
                                         betas=[200, 2000, 20000, 200000])
            if np.isfinite(sc_r) and sc_r < best_sc:
                best_sc = sc_r
                best_seq = seq_r.copy()
                print("  Restart {} LSE: {:.8f}".format(ii, best_sc))
                print("  Restart {} best: {:.8f}".format(ii, best_sc))
            a_cur = normalize(seq_r)
        else:
            a_cur = init

        # Quick Nesterov
        if time.monotonic() < restart_dead - 3.0:
            sc_r, seq_r = nesterov_subgrad(a_cur, max_iters=300,
                                           k_list=(3, 5, 8),
                                           deadline=restart_dead)
            if np.isfinite(sc_r) and sc_r < best_sc:
                best_sc = sc_r
                best_seq = seq_r.copy()
                print("  Restart {} NS: {:.8f}".format(ii, best_sc))

        # Quick active set polish
        if time.monotonic() < restart_dead - 1.0:
            sc_r, seq_r = polish_active_set(a_cur, restart_dead, max_cycles=20)
            if np.isfinite(sc_r) and sc_r < best_sc:
                best_sc = sc_r
                best_seq = seq_r.copy()
                print("  Restart {} AS: {:.8f}".format(ii, best_sc))

    # === Phase 5: Perturb-reoptimize cycles (8% time) ===
    t5_dead = t_start + 0.98 * total_budget
    print("Phase 5: Perturb cycles (deadline {:.0f}s)".format(t5_dead - t_start))
    n_perturb = 6
    t_per_pert = (t5_dead - time.monotonic()) / max(1, n_perturb)
    for pp in range(n_perturb):
        if time.monotonic() > t5_dead - 5.0:
            break
        p_dead = min(time.monotonic() + t_per_pert - 3.0, t5_dead - 3.0)
        p = perturb(best_seq, rng)
        # LSE
        if time.monotonic() < p_dead - 5.0:
            sc_p, seq_p = lse_trajectory(p, p_dead, betas=[5000, 50000, 500000])
            if np.isfinite(sc_p) and sc_p < best_sc:
                best_sc = sc_p
                best_seq = seq_p.copy()
                print("  Perturb {}: {:.8f}".format(pp, best_sc))
            a_cur = normalize(seq_p)
        else:
            a_cur = p
        # Nesterov
        if time.monotonic() < p_dead - 3.0:
            sc_p, seq_p = nesterov_subgrad(a_cur, max_iters=200,
                                           k_list=(3, 5, 8, 12),
                                           deadline=p_dead)
            if np.isfinite(sc_p) and sc_p < best_sc:
                best_sc = sc_p
                best_seq = seq_p.copy()
                print("  Perturb {} NS: {:.8f}".format(pp, best_sc))

    # === Phase 6: Multi-length exploration ===
    if time.monotonic() < deadline - 30.0:
        other_n = [8192, 12288, 16384]
        other_n = [x for x in other_n if x != n_primary][:2]
        for n_o in other_n:
            if time.monotonic() > deadline - 20.0:
                break
            a_o = normalize(resample(best_seq, n_o))
            sc_o = exact_score(a_o)
            if sc_o < best_sc:
                print("  Length {}: {:.8f} (before opt)".format(n_o, sc_o))
            # Quick optimize
            rem = deadline - 25.0 - time.monotonic()
            if rem > 10.0:
                sc_o, seq_o = lse_trajectory(a_o, deadline - 10.0,
                                             betas=[1000, 10000, 100000])
                if np.isfinite(sc_o) and sc_o < best_sc:
                    best_sc = sc_o
                    best_seq = seq_o.copy()
                    n_primary = n_o
                    print("  Length {}: {:.8f}".format(n_o, best_sc))

    # === Phase 7: Final polish ===
    if time.monotonic() < deadline - 10.0:
        sc, seq = polish_active_set(best_seq, deadline - 5.0, max_cycles=50)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
        sc, seq = nesterov_subgrad(best_seq, max_iters=200,
                                   k_list=(1, 2, 3, 5),
                                   deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
