"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2 using aggressive
L-BFGS with escalating beta, peak-directed coordinate descent, and restarts.
Key insight: optimal sequences are asymmetric, allowing center convolution
values to be below the sum-of-squares bound.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da_j = cross_correlation(p, a) at position j
    # = sum_m p[m] * a[m-j] (valid indices)
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B optimization with LSE smoothing at given beta."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coord_descent(a, n_sweeps=20, top_k=5, deadline=None):

    if S < 1e-12:

    a = a / S
    best_a = a.copy()

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:

        sc = compute_score(a)

        if S > 1e-12:

def multiplicative_update(a, n_iters=2000, beta=100, eta_init=0.1, deadline=None):
    """
    Multiplicative weight update for peak reduction.
    a_j *= exp(-eta * grad_j / ||grad||)
    This preserves non-negativity automatically.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute LSE gradient
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta

        a_rev = a[::-1]
        M2_fft = M_fft
        PA = np.fft.rfft(p, M2_fft)
        AA_rev = np.fft.rfft(a_rev, M2_fft)
        h = np.fft.irfft(AA_rev * PA, M2_fft)[:2 * n - 1][n - 1:2 * n - 1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step = eta * grad / grad_norm
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 1e-15)
            S_new = float(a_new.sum())
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                eta = min(eta * 1.02, 0.5)
            else:
                eta *= 0.85
        eta = max(eta, 0.005)

    return best_a


def randomized_restart(a, rng, n_trials=20, deadline=None):
    """Try random perturbations followed by quick L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(n_trials):
        if deadline and time.monotonic() > deadline:
            break

        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Gaussian noise in multiplicative space
            noise = rng.standard_normal(n) * 0.05
            a_pert = a * np.exp(noise)
        elif r < 0.5:
            # Targeted: boost elements that contribute to the worst peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            mask = np.zeros(n, dtype=bool)
            mask[j_start:j_end + 1] = True
            noise = rng.standard_normal(n) * 0.03
            noise[mask] *= 2.0
            a_pert = a * np.exp(noise)
        elif r < 0.7:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.1 * np.cos(2 * np.pi * freq * x + phase)
            a_pert = a * mod
        elif r < 0.85:
            # Block perturbation
            block = max(1, n // 10)
            start = rng.integers(0, n - block)
            a_pert = a.copy()
            a_pert[start:start + block] *= (1.0 + 0.15 * rng.standard_normal(block))
        else:
            # Shift-like perturbation
            a_pert = a.copy()
            a_pert[:-1] += 0.02 * a_pert[1:]
            a_pert[1:] -= 0.02 * a_pert[:-2] if n > 2 else 0

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        # Quick L-BFGS with moderate beta
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Restart {t}: improved to {sc:.8f}")

    return best_a, best_sc

Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive L-BFGS with escalating beta (up to 10M), peak-directed
coordinate descent, multiplicative updates, and random restarts.
Key insight: optimal sequences are asymmetric, allowing center convolution
values below the sum-of-squares bound that applies to symmetric sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coord_descent(a, n_sweeps=30, top_k=3, deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:top_k]

        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            total = contribs.sum()
            if total < 1e-15:
                continue

            order = np.argsort(contribs)[::-1]
            for idx in order[:15]:
                j = js[idx]
                a_old = a[j]
                if a_old < 1e-12:
                    continue
                best_val = a_old
                best_local = sc
                for f in [0.75, 0.85, 0.92, 0.97, 1.03, 1.08, 1.15, 1.25]:
                    a[j] = a_old * f
                    sc_tmp = compute_score(a)
                    if sc_tmp < best_local:
                        best_local = sc_tmp
                        best_val = a_old * f
                a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a


def multiplicative_update(a, n_iters=3000, beta=200, eta_init=0.08, deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta

        a_rev = a[::-1]
        PA = np.fft.rfft(p, M_fft)
        AA_rev = np.fft.rfft(a_rev, M_fft)
        h = np.fft.irfft(AA_rev * PA, M_fft)[:2 * n - 1][n - 1:2 * n - 1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            step = eta * grad / grad_norm
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 1e-15)
            S_new = float(a_new.sum())
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                eta = min(eta * 1.02, 0.5)
            else:
                eta *= 0.85
        eta = max(eta, 0.005)

    return best_a


def randomized_restart(a, rng, n_trials=25, deadline=None):
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for t in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        r = rng.random()
        if r < 0.3:
            noise = rng.standard_normal(n) * 0.04
            a_pert = a * np.exp(noise)
        elif r < 0.5:
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            noise = rng.standard_normal(n) * 0.03
            noise[j_start:j_end + 1] *= 2.5
            a_pert = a * np.exp(noise)
        elif r < 0.7:
            x = np.arange(n) / n
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase)
            a_pert = a * mod
        elif r < 0.85:
            block = max(1, n // 8)
            start = int(rng.integers(0, max(1, n - block)))
            a_pert = a.copy()
            a_pert[start:start + block] *= (1.0 + 0.12 * rng.standard_normal(block))
        else:
            a_pert = a.copy()
            if n > 2:
                a_pert[:-1] += 0.015 * a_pert[1:]
                a_pert[1:] -= 0.015 * a_pert[:-2]

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Restart {t}: improved to {sc:.8f}")

    return best_a, best_sc


def try_length_change(a, target_n, rng):
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    else:
        a_new = np.ones(target_n) / target_n
    return a_new
