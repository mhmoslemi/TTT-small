Let me also add a key new idea: **iteratively tightening M**. 

The algorithm from the SoTA paper is:
1. Start with f, compute M = max(conv(f,f))
2. Solve LP: max sum(b) s.t. conv(f, b) <= M, b >= 0
3. Set g = b / sum(b) * sum(f)

5. Recompute M (should decrease)
6. Repeat

Let me also think about what happens at convergence. At the LP optimum, some constraints are tight (b is at an extreme point). The tight constraints correspond to lags k where conv(f, b)[k] = M. These are the "bottleneck" lags.

An alternative to the full LP: I can use the KKT conditions to derive a simpler update. At the optimum:

1. Given b, find active lags: K = {k : conv(f,b)[k] >= M - eps}
2. Given active lags K, solve the reduced LP (only K constraints)

"""
Optimizer for minimizing 2n*max(conv(a,a))/S^2 using:
- Diverse initialization (parent, periodic, Gaussian)
- Fienup spectral flattening
- LP projection via scipy HiGHS (key novelty: iterative M-tightening from SoTA paper)
- LSE annealing with L-BFGS-B
- Subgradient peak shaving
- Multi-restart with structured perturbations
- n-growth with interpolation

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv_same(a):
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def conv_ab(a, b):
    M = len(a) + len(b) - 1
    N = 1 << (M - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv_same(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a / S


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv_same(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    sh = np.clip(beta * (c - c_max), -600, 600)
    e = np.exp(sh)
    L = c_max + np.log(e.sum()) / beta
    p = e / e.sum()
    a_rev = a[::-1]
    M3 = 3 * n - 2
    N3 = 1 << (M3 - 1).bit_length()
    conv_res = np.fft.irfft(np.fft.rfft(p, N3) * np.fft.rfft(a_rev, N3), N3)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2, S3 = S * S, S * S * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(lse_score_grad, a0, args=(beta,), method='L-BFGS-B',
                       jac=True, bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40})
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fienup_flat(a, n_iters=200):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a, best_sc = a.copy(), compute_score(a)
    for it in range(n_iters):
        prog = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = P.mean()
        alpha = 0.2 + 0.6 * prog
        P_new = (1 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a


def lp_project_iteration(f, M_override=None, n_lp=3072, t_blend=0.5):
    """
    One LP projection step from the SoTA paper:
    Solve: max sum(b) s.t. conv(f_lp, b) <= M, b >= 0
    Then blend: f_new = (1-t)*f + t*normalize(b)
    """
    n = len(f)
    f_norm = f / f.sum()

    if M_override is None:
        c_self = fast_conv_same(f_norm)
        M = float(c_self.max())
    else:
        M = M_override

    # Interpolate to n_lp
    if n_lp < n:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_lp)
        f_lp = np.interp(x_new, x_old, f_norm)
        f_lp = np.maximum(f_lp, 0.0)
        f_lp /= f_lp.sum()
    elif n_lp > n:
        x_new = np.linspace(0, 1, n_lp)
        x_old = np.linspace(0, 1, n)
        f_lp = np.interp(x_new, x_old, f_norm)
        f_lp = np.maximum(f_lp, 0.0)
        f_lp /= f_lp.sum()
    else:
        f_lp = f_norm.copy()

    n_l = len(f_lp)

    # Build convolution matrix A (2*n_l-1) x n_l
    n_rows = 2 * n_l - 1
    A_mat = np.zeros((n_rows, n_l), dtype=np.float64)
    for m in range(n_l):
        end = min(m + n_l, n_rows)
        A_mat[m:end, m] = f_lp[:end - m]

    # LP: max 1^T b  s.t. A_mat @ b <= M_lp, b >= 0
    # Scale M: conv(f_lp, b) where f_lp has sum=1, so peaks are ~1/n_l
    # Use M directly since f_lp is normalized
    M_lp = M * (n / max(n_l, 1))  # approximate scaling
    # Actually, let's just use the natural scale
    c_check = conv_ab(f_lp, np.ones(n_l) / n_l)
    M_lp = float(np.max(c_check)) * 1.0  # use current peak as target

    c_obj = -np.ones(n_l)
    b_ub = np.full(n_rows, M_lp)
    bounds = [(0.0, None)] * n_l

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 30})
        if res.success and res.x is not None:
            b_lp = np.clip(res.x, 0, None)
            b_sum = b_lp.sum()
            if b_sum > 1e-10:
                b_lp = b_lp / b_sum
                # Interpolate back
                if n_l != n:
                    b_out = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_l), b_lp)
                else:
                    b_out = b_lp.copy()
                b_out = np.maximum(b_out, 0.0)
                Sn = b_out.sum()
                if Sn > 1e-12:
                    b_out = b_out / Sn
                f_new = (1 - t_blend) * f_norm + t_blend * b_out
                f_new = np.maximum(f_new, 0.0)
                Sn2 = f_new.sum()
                if Sn2 > 1e-12:
                    f_new = f_new / Sn2
                return f_new, M_lp
    except Exception:
        pass
    return f, M_lp


def lp_anneal(f, n_iters=100, n_lp=3072):
    """Iteratively tighten M via LP projection."""
    n = len(f)
    f_cur = normalize(f.copy())
    best_a = f_cur.copy()
    best_sc = compute_score(f_cur)
    M_cur = float(fast_conv_same(f_cur).max())

    for it in range(n_iters):
        f_new, M_new = lp_project_iteration(f_cur, M_override=None, n_lp=n_lp,
                                            t_blend=max(0.2, 0.8 - 0.005 * it))
        sc_new = compute_score(f_new)
        if sc_new < best_sc:
            best_sc, best_a = sc_new, f_new.copy()
        if M_new >= M_cur * 0.999:
            # No progress, reduce target
            M_cur *= 0.9995
        else:
            M_cur = M_new
        f_cur = f_new
    return best_a


def subgrad_shave(a, n_iters=3000, top_k=10, step_init=0.006, decay=0.9998):
    n = len(a)
    a = normalize(a.copy())
    best_a, best_sc = a.copy(), compute_score(a)
    step = step_init
    lb = 2.0 * n / (2.0 * n - 1)
    for it in range(n_iters):
        c = fast_conv_same(a)
        S = float(a.sum())
        Mv = float(c.max())
        sc = 2.0 * n * Mv / (S * S)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        if sc - lb < 1e-8:
            break
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        tw = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (Mv + 1e-30)
            tw += w
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += w * a[je:js - 1: -1]
        S2, S3 = S * S, S * S * S
        aw = tw / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mv * aw / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step_p = (sc - lb) / (gn * gn + 1e-30)
        step = min(step, step_p, 0.02)
        step = max(step, 1e-10)
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    return best_a


def coord_descent(a, n_sweeps=8, n_top=25):
    n = len(a)
    a = normalize(a.copy())
    best_a, best_sc = a.copy(), compute_score(a)
    for sw in range(n_sweeps):
        c = fast_conv_same(a)
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je <= js:
            break
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        cnt = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:cnt]
        top_global = js + top_local
        for j in top_global:
            old = a[j]
            if old < 1e-14:
                continue
            bv, bl = old, best_sc
            for fr in [0.4, 0.6, 0.75, 0.85]:
                a[j] = old * fr
                an = normalize(a)
                sc = compute_score(an)
                if sc < bl:
                    bl, bv = sc, old * fr
            a[j] = bv
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a


def interpolate_seq(a, new_n):
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_periodic(n, period, amp, freq):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / period)
    return normalize(a)


def perturb_struct(a, rng, scale=0.015):
    n = len(a)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for m in range(1, 5):
        freq = rng.integers(2, 30)
        ph = 2 * np.pi * rng.uniform(0, 1)
        pert += (scale / m) * np.cos(2 * np.pi * freq * x + ph)
    return normalize(a * (1.0 + pert))

        if time.monotonic() > p5_end - 15:
            break
        rem = p5_end - 15 - time.monotonic()
        if rem < 15:
            break
        if r % 4 == 0:
            a_p = perturb_struct(best_a, rng, scale=0.01 + 0.004 * (r % 4))
        elif r % 4 == 1:
            a_p = best_a + 0.008 * rng.standard_normal(len(best_a))
            a_p = normalize(a_p)
        elif r % 4 == 2:
            a_p = make_periodic(len(best_a), len(best_a) // rng.integers(2, 8),
                                0.2, rng.integers(1, 5))
            a_p = 0.7 * best_a + 0.3 * a_p
            a_p = normalize(a_p)
        else:
            # Focused: boost non-peak contributors
            c = fast_conv_same(best_a)
            k_pk = int(np.argmax(c))
            js, je = max(0, k_pk - n + 1), min(k_pk, n - 1)
            mask = np.ones(n, dtype=bool)
            mask[js:je + 1] = False
            add_idx = np.where(mask)[0]
            if len(add_idx) > 50:
                sel = rng.choice(add_idx, min(100, len(add_idx)), replace=False)
                a_p = best_a.copy()
                a_p[sel] += 0.003 / n
                a_p = normalize(a_p)
            else:
                a_p = perturb_struct(best_a, rng, 0.01)

        beta = 3000 + r * 2000
        mx = max(400, int(rem * 5))
        sc, a_opt = run_lbfgs(a_p, beta, maxiter=mx, maxfun=mx * 15)
        update(sc, a_opt, f"RESTART {r}")

    # Phase 6: n-growth (7% time)
    p6_end = deadline - 60
    print(f"\n=== Phase 6: n-growth ===")

    cur_n = len(best_a)
    targets = []
    if cur_n < 12000:
        targets.append(12000)
    if cur_n < 16000:
        targets.append(16000)

    for tn in targets:
        if time.monotonic() > p6_end - 40:
            break
        a_big = interpolate_seq(best_a, tn)
        a_big = normalize(a_big)
        sc = compute_score(a_big)
        print(f"  Interp n={tn}: {sc:.8f}")
        update(sc, a_big, f"INTERP n={tn}")
        if time.monotonic() < p6_end - 20:
            sc2, a2 = run_lbfgs(a_big, 5000, maxiter=1500, maxfun=30000)
            update(sc2, a2, f"INTERP+LSE n={tn}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final polish ===")
    if time.monotonic() < deadline - 30:
        rem = deadline - 30 - time.monotonic()
        sc, af = run_lbfgs(best_a, 80000, maxiter=max(500, int(rem * 6)),
                           maxfun=max(10000, int(rem * 100)))
        update(sc, af, "FINAL-LSE-80k")

    if time.monotonic() < deadline - 15:
        n_it = max(300, int((deadline - 15 - time.monotonic()) * 2000))
        a_ps = subgrad_shave(best_a, n_iters=min(n_it, 2000), top_k=3, step_init=0.002)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    if time.monotonic() < deadline - 5:
        a_cd = coord_descent(best_a, n_sweeps=3, n_top=15)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    el = time.monotonic() - t0
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {el:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
