2. Conjugate gradient with Polak-Ribiere restarts for faster convergence

import numpy as np

def fast_conv(a):

def profile_grad_descent(a, n_steps=100000, gamma=1.5, use_cg=True, verbose=False):
    """
    Profile-gradient descent with optional conjugate gradient acceleration.
    Uses a smooth surrogate: sum of (excess)^gamma over all lags.
    The gradient is computed via FFT-based cross-correlation.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1)
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()

    # Conjugate gradient state
    d = None
    g_dot_d_prev = 0.0

    for it in range(n_steps):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if verbose and it % 20000 == 0:
            print(f"      PG iter {it}: score={sc:.8f}, step_size tracking")

        if sc - lb < 1e-9:
            break

        # Compute excess profile
        mean_c = S * S / (2 * n - 1)
        excess = np.maximum(c - mean_c, 0.0)
        w = excess ** gamma
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum  # Normalize weights

        # Compute gradient: dE/da[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        a_rev = a[::-1]
        WA = np.fft.rfft(w, M3_fft)
        RA = np.fft.rfft(a_rev, M3_fft)
        conv_res = np.fft.irfft(WA * RA, M3_fft)[:M3]
        grad = 2.0 * conv_res[n - 1:2 * n - 1]

        # Subtract mean to preserve sum (projection onto sum=0 subspace)
        grad -= np.mean(grad)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        if use_cg:
            # Polak-Ribiere conjugate gradient with restarts
            if d is None:
                d = -grad
                g_dot_d = -gn * gn
            else:
                g_dot_d = float(np.dot(grad, d))
                # Check for restart (negative curvature)
                if g_dot_d > 0:
                    beta_cg = 0.0  # Restart
                else:
                    beta_cg = g_dot_d / (g_dot_d_prev + 1e-30)
                    beta_cg = max(0.0, min(beta_cg, 1.0))
                d = -grad + beta_cg * d
                g_dot_d_prev = g_dot_d
        else:
            d = -grad
            g_dot_d = -gn * gn

        # Step size via backtracking line search
        step = 0.05
        a_trial = a - step * d / (gn + 1e-30)
        a_trial = np.maximum(a_trial, 0.0)
        S_trial = a_trial.sum()
        if S_trial < 1e-12:
            step *= 0.5
            a_trial = a - step * d / (gn + 1e-30)
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                break
        # Just normalize and accept (simpler than full line search for speed)
        a = a_trial / S_trial

        # Periodic LSE refinement checkpoint
        if it % 50000 == 49999 and it > 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_a

        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current autocorrelation to find peaks
        c = fast_conv(a)
        peak_idx = int(np.argmax(c))

        # Target: make autocorrelation flatter
        # In frequency domain, autocorrelation = IFFT(|A|^2)
        # To reduce a specific peak, we need to adjust phases
        # Use gradient of peak w.r.t. phase
        # dc[peak]/dphase[k] = 2 * Re(A * conj(dA/dphase[k]) ... )
        # Simpler: use a heuristic phase rotation at peaks

        # Modify magnitude slightly to flatten
        P = mag ** 2
        P_mean = float(P.mean())
        P_flat = (1.0 - 0.3 * progress) * P + 0.3 * progress * P_mean
        P_flat = np.maximum(P_flat, 1e-30)
        mag_new = np.sqrt(P_flat)

        # Slight phase adjustment: rotate phases at high-frequency components
        # to reduce constructive interference at the peak
        n_half = len(phase)
        if n_half > 2:
            phase_adjust = np.zeros(n_half)
            for k in range(2, n_half):
                # Small rotation proportional to frequency
                phase_adjust[k] = 0.01 * progress * np.sin(2.0 * np.pi * k / n_half)
            phase = phase + phase_adjust

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat(a, n_iters=300):
    """Standard magnitude-based Fienup."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.1 + 0.6 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_aggressive(a, n_iters=400):
    """More aggressive spectral flattening with quantile clipping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Aggressive clipping of spectral peaks
        q = 0.85 - 0.3 * progress  # Tighter over time
        thresh = np.quantile(P, q)
        P_new = np.minimum(P, thresh)

        # Also boost the floor
        P_floor = np.quantile(P, 0.1) * 0.5
        P_new = np.maximum(P_new, P_floor)

        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coord_descent(a, n_sweeps=10, n_top=30):
    """Coordinate descent on top contributors to peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.75, 0.85]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_peak_coord(a, n_sweeps=5, n_lags=5, n_top=15):
    """Coordinate descent targeting multiple peak lags simultaneously."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top lags
        top_lags = np.argsort(c)[::-1][:n_lags]

        for k_star in top_lags:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k_star - js]
            top_count = min(n_top, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old = a[j]
                if a_old < 1e-14:
                    continue
                best_val = a_old
                best_local = best_sc
                for f in [0.3, 0.5, 0.7]:
                    a[j] = a_old * f
                    a_n = normalize(a)
                    sc = compute_score(a_n)
                    if sc < best_local:
                        best_local = sc
                        best_val = a_old * f
                a[j] = best_val
                a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def perturb_mult(a, rng, scale=0.03, n_harmonics=10):
    """Multiplicative perturbation with low-frequency structure."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, n_harmonics):
        freq = rng.integers(1, 40)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / (m ** 0.7)
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_additive(a, rng, scale=0.02):
    """Additive Gaussian perturbation."""
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)


def perturb_structured(a, rng):
    """Perturb by adding mass to random sub-intervals."""
    n = len(a)
    a_new = a.copy()
    n_blocks = rng.integers(3, 10)
    for _ in range(n_blocks):
        start = rng.integers(0, n - 100)
        length = rng.integers(50, 500)
        delta = rng.uniform(-0.001, 0.001) / n
        a_new[start:start + length] += delta
    return normalize(a_new)


def two_scale_optimize(a, budget_s):
    """
    Two-scale approach: downsample, optimize, upsample, refine.
    This can escape local minima by exploring a coarser landscape.
    """
    n = len(a)
    t_start = time.monotonic()

    # Downsample by factor 4
    ds_factor = 4
    n_coarse = n // ds_factor
    if n_coarse < 256:
        return a

    # Average blocks
    a_coarse = np.array([a[i * ds_factor:(i + 1) * ds_factor].mean()
                         for i in range(n_coarse)])
    a_coarse = normalize(a_coarse)
    print(f"    Coarse n={n_coarse}, score={compute_score(a_coarse):.8f}")

    # Optimize at coarse scale (fast)
    t0 = time.monotonic()
    a_coarse_opt = profile_grad_descent(a_coarse, n_steps=20000, gamma=1.5)
    sc_c = compute_score(a_coarse_opt)
    print(f"    Coarse after PG: score={sc_c:.8f} ({time.monotonic()-t0:.1f}s)")

    # Upsample
    a_up = np.repeat(a_coarse_opt, ds_factor)
    # Blend with original
    t0 = time.monotonic()
    for blend in [0.3, 0.5, 0.7, 1.0]:
        a_mixed = (1 - blend) * normalize(a) + blend * normalize(a_up)
        a_mixed = normalize(a_mixed)
        sc = compute_score(a_mixed)
        print(f"    Blend {blend}: score={sc:.8f}")
        if time.monotonic() - t_start > budget_s * 0.8:
            break
    return a_mixed


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer combining:
    1. Profile gradient descent (new) - smooth surrogate on excess profile
    2. LSE annealing - proven technique
    3. Phase-domain Fienup (new)
    4. Multi-peak coordinate descent
    5. Two-scale exploration
    6. Multi-restart with structured perturbations
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and len(parent) >= 1000:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
        n = len(parent)
    else:
        n = 10000
        # Try structured initializations
        candidates = []
        candidates.append(np.ones(n) / n)
        # Tilted window
        x = np.arange(n) / (n - 1)
        candidates.append(np.maximum(0.3 + 0.7 * x, 0.01))
        # Quadratic
        candidates.append((x ** 2 + 0.1))
        # Sinusoidal modulation
        candidates.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * x))
        candidates.append(1.0 + 0.5 * np.cos(2 * np.pi * 2 * x + 0.5))

        for i, c in enumerate(candidates):
            c = normalize(c)
            sc = compute_score(c)
            print(f"  Init {i} (n={n}): score={sc:.8f}")
            update(sc, c, f"init-{i}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_score = compute_score(best_a)

    print(f"Starting best: {best_score:.8f} (n={n})")

    # ==================================================================
    # Phase 1: Profile Gradient Descent (25% time)
    # This is the main new contribution - different gradient landscape
    # ==================================================================
    phase1_end = t_start + 0.25 * (budget_s - 35.0)
    print(f"\n=== Phase 1: Profile Gradient ({phase1_end - t_start:.0f}s budget) ===")

    a_work = best_a.copy()

    # Multiple gamma values for the excess profile
    for gamma in [1.0, 2.0, 3.0]:
        if time.monotonic() > phase1_end - 30:
            break
        remaining = phase1_end - 30 - time.monotonic()
        if remaining < 30:
            break
        n_steps = int(min(80000, remaining * 80))
        print(f"  PGD gamma={gamma}, steps={n_steps}")
        t0 = time.monotonic()
        a_pg = profile_grad_descent(a_work, n_steps=n_steps, gamma=gamma, use_cg=True)
        sc = compute_score(a_pg)
        print(f"    PGD gamma={gamma} done: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")
        update(sc, a_pg, f"PGD g={gamma}")
        if sc < best_score:
            a_work = a_pg.copy()

    # ==================================================================
    # Phase 2: LSE Annealing (30% time)
    # ==================================================================
    phase2_end = t_start + 0.55 * (budget_s - 35.0)
    print(f"\n=== Phase 2: LSE Annealing ({phase2_end - phase1_end:.0f}s budget) ===")

    a_work = best_a.copy()

    # Beta schedule: start moderate, go high
    betas = [200, 500, 1500, 4000, 10000, 25000, 50000, 100000]
    for i, beta in enumerate(betas):
        if time.monotonic() > phase2_end - 20:
            break
        remaining = phase2_end - 20 - time.monotonic()
        if remaining < 20:
            break
        # Allocate more time to mid-range betas
        mid_weight = 1.0 / (1.0 + 0.5 * abs(i - 3))
        maxiter = max(1000, int(remaining * 10 * mid_weight))
        maxfun = maxiter * 20
        print(f"  LSE beta={beta:>6}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE b={beta}")
        a_work = a_opt.copy()

    # ==================================================================
    # Phase 3: Hybrid PGD+LSE (15% time)
    # Alternate between profile gradient and LSE to escape local minima
    # ==================================================================
    phase3_end = t_start + 0.70 * (budget_s - 35.0)
    print(f"\n=== Phase 3: Hybrid PGD+LSE ===")

    a_work = best_a.copy()
    n_cycles = 8
    for cyc in range(n_cycles):
        if time.monotonic() > phase3_end - 20:
            break
        remaining = phase3_end - 20 - time.monotonic()
        if remaining < 20:
            break
        half_time = remaining * 0.7
        if half_time < 15:
            break

        # Profile gradient step
        n_steps = int(min(30000, half_time * 50))
        a_pg = profile_grad_descent_v2(a_work, n_steps=n_steps, top_k=15 + cyc * 5, gamma=1.5)
        sc = compute_score(a_pg)
        update(sc, a_pg, f"PG2 cyc{cyc}")
        a_work = a_pg.copy()

        # LSE polish
        if time.monotonic() < phase3_end - 15:
            beta = 3000 + cyc * 5000
            maxiter = max(500, int((phase3_end - 15 - time.monotonic()) * 5))
            sc2, a2 = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 15)
            update(sc2, a2, f"LSE cyc{cyc}")
            a_work = a2.copy()

    # ==================================================================
    # Phase 4: Fienup variants (8% time)
    # ==================================================================
    phase4_end = t_start + 0.78 * (budget_s - 35.0)
    print(f"\n=== Phase 4: Fienup ===")

    if time.monotonic() < phase4_end - 10:
        a_f1 = fienup_flat(best_a, n_iters=200)
        update(compute_score(a_f1), a_f1, "FIENUP-flat")

    if time.monotonic() < phase4_end - 10:
        a_f2 = fienup_aggressive(best_a, n_iters=300)
        update(compute_score(a_f2), a_f2, "FIENUP-agg")

    if time.monotonic() < phase4_end - 10:
        a_f3 = phase_fienup(best_a, n_iters=300)
        update(compute_score(a_f3), a_f3, "PHASE-FIENUP")

    # ==================================================================
    # Phase 5: Multi-peak coordinate descent (7% time)
    # ==================================================================
    phase5_end = t_start + 0.85 * (budget_s - 35.0)
    print(f"\n=== Phase 5: Multi-peak CD ===")

    if time.monotonic() < phase5_end - 10:
        a_mc = multi_peak_coord(best_a, n_sweeps=8, n_lags=4, n_top=20)
        sc = compute_score(a_mc)
        update(sc, a_mc, "MULTI-CD")
        if time.monotonic() < phase5_end - 5:
            sc2, a2 = run_lbfgs(a_mc, 20000, maxiter=2000, maxfun=40000)
            update(sc2, a2, "MULTI-CD+LSE")

    # Single-peak CD
    if time.monotonic() < phase5_end - 5:
        a_cd = coord_descent(best_a, n_sweeps=8, n_top=25)
        sc = compute_score(a_cd)
        update(sc, a_cd, "CD")

    # ==================================================================
    # Phase 6: Multi-restart escape (7% time)
    # ==================================================================
    phase6_end = t_start + 0.92 * (budget_s - 35.0)
    print(f"\n=== Phase 6: Multi-Restart ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > phase6_end - 20:
            break
        remaining = phase6_end - 20 - time.monotonic()
        if remaining < 15:
            break

        # Different perturbation strategies
        if r % 4 == 0:
            a_pert = perturb_mult(best_a, rng, scale=0.015 + 0.005 * (r % 4), n_harmonics=12)
        elif r % 4 == 1:
            a_pert = perturb_additive(best_a, rng, scale=0.008 + 0.003 * (r % 4))
        elif r % 4 == 2:
            a_pert = perturb_structured(best_a, rng)
        else:
            # Asymmetric perturbation: tilt the sequence
            n = len(best_a)
            x = np.arange(n, dtype=np.float64) / n
            tilt = 1.0 + 0.05 * (2 * x - 1) * (r % 3 + 1)
            a_pert = normalize(best_a * tilt)

        # Short LSE
        beta = 2000 + r * 3000
        maxiter = max(500, int(remaining * 4))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_opt, f"RESTART {r}")

    # ==================================================================
    # Phase 7: Final polish (3% time)
    # ==================================================================
    print(f"\n=== Phase 7: Final Polish ===")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = max(1000, int(remaining * 6))
        sc, a_final = run_lbfgs(best_a, 80000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-80k")

    if time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        if remaining > 5:
            a_pg_final = profile_grad_descent(best_a, n_steps=int(remaining * 4000), gamma=2.0)
            sc = compute_score(a_pg_final)
            update(sc, a_pg_final, "FINAL-PGD")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd_final = coord_descent(best_a, n_sweeps=5, n_top=20)
        sc = compute_score(a_cd_final)
        update(sc, a_cd_final, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
