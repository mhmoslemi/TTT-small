"""
Fine-tuned optimization of non-negative step function minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE-LBFGS, iterative reweighting, GS, and LP-based direction finding.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:2*n-1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0*n*float(c.max())/(S*S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta*(c - c_max)
    max_s = float(scaled.max())
    e = np.exp(scaled - max_s)
    log_sum = max_s + float(np.log(e.sum()))
    L = c_max + log_sum/beta
    score = 2.0*n*L/(S*S)
    p = e/e.sum()
    # grad of L w.r.t. a: 2*(rev_a * p)[i+n-1] for i=0..n-1
    rev_a = a[::-1]
    L2 = 3*n-2
    M2 = 1 << (L2-1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    cv = np.fft.irfft(RA*P, M2)[n-1:2*n-1]
    g = 2.0*cv
    S2 = S*S
    S3 = S2*S
    grad = (2.0*n)*(g*S - 2.0*L)/S3
    return score, grad

def run_lbfgs(a0, func, maxiter=10000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.copy(), 0.0, 1000.0)
    if a0.sum() < 1e-8:
        a0[:] = 1.0/n
    bounds = [(0.0, 1000.0)]*n
    def wrapped(x):
        s, g = func(x)
        return s, g
    res = minimize(wrapped, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    return np.clip(res.x, 0.0, 1000.0)

def gs_flat(a0, n_iters=500, alpha_init=0.5, alpha_final=0.05):
    n = len(a0)
    M = 1 << (2*n-2).bit_length()
    bt = np.zeros(M)
    bt[:2*n-1] = 1.0/(2*n-1)
    T = np.fft.fft(bt)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_ph = np.angle(T)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n)/n
    a /= S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        alpha = alpha_init + (alpha_final - alpha_init)*(it/n_iters)
        A = np.fft.fft(a, M)
        Am = np.maximum(np.abs(A), 1e-20)
        Ap = np.angle(A)
        nm = Am**(1.0-alpha) * T_mag**(alpha*0.5)
        np_ = (1.0-alpha)*Ap + alpha*T_ph*0.5
        An = nm*np.exp(1j*np_)
        an = np.fft.ifft(An, M).real[:n]
        an = np.maximum(an, 0.0)
        Sn = an.sum()
        if Sn < 1e-12:
            break
        an /= Sn
        a = an
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def weighted_score_grad(a, w):

    S3 = S2*S

    n = len(a)
    # (a*g)_k = sum_j a_j * g_{k-j} for appropriate j
    # This is a Toeplitz matrix times g
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A[k, j] = a[j] * ... wait
    # (a*g)_k = sum_{j: 0<=k-j<n} a[k-j] * g[j]
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[k-j] * g[j]
    # So A[k,j] = a[k-j] if 0<=k-j<n and 0<=j<n, else 0
    
    # For LP: maximize c^T g = 1^T g
    # subject to A g <= M_limit * 1, g >= 0
    
    # Build A as a scipy sparse matrix for efficiency
    from scipy import sparse

    # This is a convolution/Toeplitz matrix
    # First column: a[n-1], a[n-2], ..., a[0], 0, 0, ..., 0 (for k=n-1,...,0, n, ..., 2n-2)
    # Actually let me just build it properly

    # So row k has nonzeros at j = max(0,k-n+1) to min(k, n-1)
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A_sparse = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: min -1^T g s.t. A g <= M_limit, g >= 0
    c_obj = -np.ones(n)
    A_ub = A_sparse
    b_ub = np.full(2*n-1, M_limit)
    bounds = [(0, None)]*n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:

    else:
        return None, None

from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:2*n-1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0*n*float(c.max())/(S*S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta*(c - c_max)
    max_s = float(scaled.max())
    e = np.exp(scaled - max_s)
    log_sum = max_s + float(np.log(e.sum()))
    L = c_max + log_sum/beta
    score = 2.0*n*L/(S*S)
    p = e/e.sum()
    rev_a = a[::-1]
    L2 = 3*n-2
    M2 = 1 << (L2-1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    cv = np.fft.irfft(RA*P, M2)[n-1:2*n-1]
    g = 2.0*cv
    S2 = S*S
    S3 = S2*S
    grad = (2.0*n)*(g*S - 2.0*L)/S3
    return score, grad

    return np.clip(res.x, 0.0, 1000.0)

"""
Optimized step function search using MWMU, LSE-LBFGS, iterative reweighting,
GS, and FISTA to minimize 2n * max(conv(a,a)) / (sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    e = np.exp(scaled - max_s)
    log_sum = max_s + float(np.log(e.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = e / e.sum()
    rev_a = a[::-1]
    L2 = 3 * n - 2
    M2 = 1 << (L2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    cv = np.fft.irfft(RA * P, M2)[n - 1:2 * n - 1]
    g = 2.0 * cv
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=5000, maxfun=20000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-8:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_score_grad, a0, args=(beta,), method='L-BFGS-B',
                   jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    return np.clip(res.x, 0.0, 1000.0)


def weighted_score_grad(a, w):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    wn = float(w @ c)
    score = 2.0 * n * wn / (S * S)
    rev_a = a[::-1]
    L2 = 3 * n - 2
    M2 = 1 << (L2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    W = np.fft.rfft(w, M2)
    cv = np.fft.irfft(RA * W, M2)[n - 1:2 * n - 1]
    g = 2.0 * cv
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * wn) / S3
    return score, grad
