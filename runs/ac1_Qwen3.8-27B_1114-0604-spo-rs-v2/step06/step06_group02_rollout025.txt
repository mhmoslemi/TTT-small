"""
LP-guided optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding combined with L-BFGS and coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient computation
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    if S < 1e-6:

    else:

        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

        M = float(c.max())

        # Blend

        a_new = np.maximum(a_new, 0)

        if S_new < 1e-12:

            best_a = a_new.copy()
            a = a_new
            # Keep t or increase slightly

            # Reduce t

    return best_a

"""
LP-guided multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations: LP-based direction finding for principled mass redistribution,
adaptive L-BFGS with log-sum-exp smoothing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def solve_lp_direction(a, M, n_max_constraints=None):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, convolve(a, g)[k] <= M for all k.
    Returns normalized g (sum=1) or None on failure.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-12:
        return None
    a_norm = a / S

    if n_max_constraints is not None and 2 * n - 1 > n_max_constraints:
        k_indices = np.unique(np.linspace(0, 2 * n - 2, n_max_constraints, dtype=int))
    else:
        k_indices = np.arange(2 * n - 1)

    n_cons = len(k_indices)
    A_ub = np.zeros((n_cons, n))
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            A_ub[idx, js] = a_norm[k - js]

    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 120})
    except Exception:
        return None

    if res.status == 0:
        g = np.maximum(res.x, 0)
        S_g = g.sum()
        if S_g > 1e-12:
            return g / S_g
    return None

                continue

            # Check if this element contributes to any significant peak
            k_star = int(np.argmax(c))
            if 0 <= k_star - j < n:
                contrib = a[k_star - j]
            else:
                contrib = 0.0

            if contrib < 0.01 * np.max(a):
                continue

            # Try reducing a[j]
            best_local = best_sc
            best_val = a_j
            for factor in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9]:
                a[j] = a_j * factor
                S_trial = a.sum()
                if S_trial < 1e-12:
                    a[j] = a_j
                    continue
                c_trial = fast_conv(a)
                sc_trial = 2.0 * n * float(np.max(c_trial)) / (S_trial * S_trial)
                if sc_trial < best_local:
                    best_local = sc_trial
                    best_val = a_j * factor

            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def spectral_shaping(a, n_iters=200, target_width=0.3, deadline=None):
    """
    Fienup variant targeting a specific spectral shape with a raised-cosine window.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    freqs = np.fft.rfftfreq(M_fft)
    norm_f = np.abs(freqs) * 2.0 * n
    shape = np.exp(-0.5 * (norm_f / (n * target_width)) ** 2)
    shape = shape / np.max(shape)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        P_mean = float(P.mean())
        P_target = P_mean * shape / np.mean(shape)

        alpha = 0.5 * (1.0 - progress) + 0.05
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    LP-guided multi-strategy optimizer combining:
    - Fienup spectral iterations for fast initial improvement
    - LP-based direction finding for principled mass redistribution
    - Adaptive L-BFGS with log-sum-exp for smooth max-convolution minimization
    - Coordinate descent for fine-grained peak reduction
    - Multi-length exploration to avoid local minima
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # --- Initialize ---
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def _update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None and len(parent) >= 128:
        n = len(parent)
        a_init = parent.astype(np.float64).copy()
        a_init = np.maximum(a_init, 0)
        S = a_init.sum()
        if S > 1e-12:
            a_init = a_init / S
        best_a = a_init.copy()
        best_score = compute_score(best_a)
        print(f"Parent loaded: n={n}, score={best_score:.8f}")
    else:
        # Generate initialization
        for n_try in [2048, 4096]:
            x = np.arange(n_try)
            # Raised cosine
            a_r = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n_try))
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            sc = compute_score(a_r)
            if sc < best_score:
                best_score = sc
                best_a = a_r.copy()
        print(f"Generated init: n={len(best_a)}, score={best_score:.8f}")

    n = len(best_a)

    # === Phase 1: Fienup + Spectral shaping (10%) ===
    p1_end = t_start + 0.10 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup + Spectral (n={n}) ===")

    a_f = fienup_refined(best_a, n_iters=400, deadline=p1_end)
    sc = compute_score(a_f)
    _update(sc, a_f, "FIENUP")

    if time.monotonic() < p1_end:
        a_ss = spectral_shaping(best_a, n_iters=200, target_width=0.3, deadline=p1_end)
        sc = compute_score(a_ss)
        _update(sc, a_ss, "SPECTRAL")

    # === Phase 2: LP-guided iterations (20%) ===
    p2_end = t_start + 0.30 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LP-guided (n={n}) ===")

    if n <= 4096:
        a_lp = lp_iteration_sequence(best_a, n_rounds=10, deadline=p2_end)
        sc = compute_score(a_lp)
        _update(sc, a_lp, "LP-ITER")
    else:
        print("  Skipping LP (sequence too long)")

    # === Phase 3: L-BFGS with adaptive beta (35%) ===
    p3_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Adaptive L-BFGS (n={n}) ===")

    betas = [50, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for beta in betas:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 15:
            break
        # Scale maxiter based on remaining time and n
        maxiter = int(min(15000, remaining * 3 + 2000))
        maxiter = max(maxiter, 1000)
        maxfun = maxiter * 15

        t0 = time.monotonic()
        try:
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        except Exception as e:
            print(f"    L-BFGS beta={beta} failed: {e}")
            continue
        dt = time.monotonic() - t0
        print(f"  L-BFGS beta={beta}: score={sc:.8f} ({dt:.1f}s)")
        if sc < best_score:
            _update(sc, a_opt, f"LBFGS-{beta}")

    # === Phase 4: Coordinate descent polish (15%) ===
    p4_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Coordinate descent (n={n}) ===")

    n_cd_sweeps = 3 if n <= 4096 else 2
    a_cd = coordinate_descent(best_a, n_sweeps=n_cd_sweeps, deadline=p4_end)
    sc = compute_score(a_cd)
    _update(sc, a_cd, "COORD")

    if time.monotonic() < p4_end - 20:
        sc, a_opt = run_lbfgs(a_cd, 50000, maxiter=5000, maxfun=100000)
        _update(sc, a_opt, "CD+LBFGS")

    # === Phase 5: Multi-length exploration (10%) ===
    p5_end = deadline - 60.0
    print(f"\n=== Phase 5: Multi-length exploration ===")

    target_lengths = [4096, 8192]
    for n_try in target_lengths:
        if n_try == n:
            continue
        if time.monotonic() > p5_end:
            break

        print(f"  Trying n={n_try}...")
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.0)
        S = a_interp.sum()
        if S < 1e-12:
            continue
        a_interp = a_interp / S

        sc_init = compute_score(a_interp)
        print(f"    Initial n={n_try}: score={sc_init:.8f}")

        # Fienup on interpolated
        a_f2 = fienup_refined(a_interp, n_iters=200, deadline=p5_end - 30)
        sc = compute_score(a_f2)
        if sc < best_score:
            _update(sc, a_f2, f"LEN-{n_try}-FIENUP")
        else:
            # Don't update best_a, just use as starting point for L-BFGS
            if time.monotonic() < p5_end - 60:
                sc2, a_opt2 = run_lbfgs(a_f2, 1000, maxiter=5000, maxfun=100000)
                if sc2 < best_score:
                    _update(sc2, a_opt2, f"LEN-{n_try}-LBFGS")

    # === Phase 6: Stochastic restarts + final polish (5%) ===
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 6: Stochastic + Final polish ===")
        for r in range(10):
            if time.monotonic() > deadline - 60:
                break

            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(len(best_a)))
            a_pert = np.maximum(a_pert, 0.0)
            S = a_pert.sum()
            if S < 1e-12:
                continue
            a_pert = a_pert / S

            sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=3000, maxfun=60000)
            if sc < best_score:
                _update(sc, a_opt, f"STOCH-{r}")

        # Very high beta final polish
        if time.monotonic() < deadline - 20:
            sc, a_opt = run_lbfgs(best_a, 200000, maxiter=5000, maxfun=100000)
            _update(sc, a_opt, "FINAL-HIGH-BETA")

    # === Final verification ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
