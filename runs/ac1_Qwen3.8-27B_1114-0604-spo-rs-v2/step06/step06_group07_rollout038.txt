Step function minimization using adaptive LSE-smoothed L-BFGS with 
subgradient refinement and multi-phase optimization.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective value: 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective and gradient.
    Objective: 2n * LSE_beta(conv(a,a)) / (sum(a))^2
    where LSE_beta(x) = (1/beta) * log(sum(exp(beta * x)))

    n = len(a)
    n_fft = 2 * n - 1

        a_out = x0.copy()
    else:
        a_out = a_out / S  # Re-normalize
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def subgradient_descent(a0, n_iters=5000, step_init=0.001, momentum=0.9):
    """
    Subgradient descent on the exact max of conv(a,a).
    At each step, find the argmax lag, compute the subgradient, and take a step.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    n_fft = 2 * n - 1
    M_fft = 1 << (n_fft - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # Momentum
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:n_fft]
        S = float(a.sum())
        M_val = float(c.max())
        k_max = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max_conv / S^2 * 2n at the argmax
        # d max_conv / da_i where max is at lag k_max
        # c_k = sum_j a[j]*a[k-j]
        # dc_k/da_i = 2*a[k-i] if 2i != k, a[i] if 2i == k
        sg = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(k_max, n - 1)
        if je >= js:
            # For i in [js, je], a[k_max - i] is valid
            sg[js:je+1] = 2.0 * a[js:je+1][::-1]
            # Correct the self-term
            if 0 <= k_max and 2 * (js + (k_max - js) // 2) == k_max:
                i_self = k_max // 2
                if 2 * i_self == k_max and js <= i_self <= je:
                    sg[i_self] = a[i_self]  # Only one term
        
        # Actually, let me be more careful:
        # c_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * a[k-j]
        # dc_k/da_i = a[k-i] (if 0<=k-i<n and we're differentiating w.r.t. the j=i factor)
        #            + a[k-i] (if 0<=k-i<n and we're differentiating w.r.t. the (k-j)=i factor, i.e., j=k-i)
        # If i != k-i: both terms present, dc_k/da_i = 2*a[k-i]
        # If i == k-i (k=2i): only one term, dc_k/da_i = a[i]
        # Condition for i to appear: 0 <= i < n AND 0 <= k-i < n
        # i.e., max(0, k-n+1) <= i <= min(k, n-1)
        
        # Let me recompute more carefully
        sg = np.zeros(n)
        for i in range(max(0, k_max - n + 1), min(k_max, n - 1) + 1):
            if k_max - i == i:  # self-term
                sg[i] = a[i]
            else:
                sg[i] = 2.0 * a[k_max - i]
        
        # Gradient of 2n * M / S^2
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Normalized gradient for step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step
        step = min(step, max((sc - sc_lb) * 0.5 / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.01)
        
        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a_new / S_new  # Renormalize
        
        step *= 0.9999
    
    return best_a

def fienup_quick(a, n_iters=100, alpha=0.5):
    """Quick Fienup spectral shaping for initialization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    n_fft = 2 * n - 1
    M_fft = 1 << (n_fft - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        
        # Target: mean amplitude (excluding DC)
        if len(amp) > 1:
            C = np.mean(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_new = amp * (C / np.maximum(amp, 1e-20)) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return a

def gradient_check(a, beta, i_check=0):
    """Verify gradient computation numerically."""
    eps = 1e-7
    sc0, grad = lse_obj_grad(a, beta)
    a_plus = a.copy()
    a_plus[i_check] += eps
    a_minus = a.copy()
    a_minus[i_check] -= eps
    sc_plus, _ = lse_obj_grad(a_plus, beta)
    sc_minus, _ = lse_obj_grad(a_minus, beta)
    num_grad = (sc_plus - sc_minus) / (2 * eps)
    return grad[i_check], num_grad

"""
Step function minimization using adaptive LSE-smoothed L-BFGS with
exact subgradient refinement and top-K peak-weighted descent.
Key innovations: (1) correct vectorized gradient for LSE objective,
(2) top-K weighted subgradient for final refinement avoiding LSE
approximation error, (3) geometric beta annealing with warm-starting,
(4) momentum-based exact subgradient with adaptive step size.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective and its gradient.
    f(a) = 2n * LSE_beta(conv(a,a)) / (sum(a))^2
    LSE_beta(x) = (1/beta) * log(sum_k exp(beta * x_k))
    """
    n = len(a)
    n_fft = 2 * n - 1
    M_fft = 1 << (n_fft - 1).bit_length()

    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:n_fft]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    # Probability weights
    p = exp_scaled / exp_sum

    # Gradient of L w.r.t. a:
    # dL/da_i = sum_k p[k] * dc_k/da_i
    # dc_k/da_i = 2*a[k-i] if k != 2i, a[i] if k == 2i (for valid indices)
    # Equivalently: sum_k p[k]*a[k-i] * 2 - p[2i]*a[i] (correction for self-term)
    # g[i] = sum_k p[k] * a[k-i] via cross-correlation

    # Compute g[i] = sum_k p[k] * a[k-i] using FFT
    # g = conv(p, a_flipped)[n-1 : 2n-1]
    n2 = n_fft + n - 1
    M2 = 1 << (n2 - 1).bit_length()
    h = a[::-1]
    P = np.fft.rfft(p, M2)
    H = np.fft.rfft(h, M2)
    conv_ph = np.fft.irfft(P * H, M2)
    g = conv_ph[n - 1:2 * n - 1]

    # Self-term correction: when 2i == k, the product a[i]*a[i] is counted once,
    # but 2*g[i] counts it twice. Subtract p[2i]*a[i] for valid 2i.
    indices = np.arange(n)
    two_i = 2 * indices
    valid = two_i < n_fft
    correction = np.zeros(n)
    correction[valid] = p[two_i[valid]] * a[indices[valid]]

    g_L = 2.0 * g - correction

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs_lse(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    x0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = x0.sum()
    if S0 < 1e-6:
        x0[:] = 1.0 / n
        S0 = 1.0
    x0 = x0 / S0

    bounds = [(0.0, 1000.0)] * n

    def fun_local(x):
        val, _ = lse_obj_grad(x, beta)
        return val

    def jac_local(x):
        _, g = lse_obj_grad(x, beta)
        return g

    res = minimize(
        fun_local, x0, method='L-BFGS-B', jac=jac_local,
        bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )

    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = x0.copy()
    else:
        a_out = a_out / S

    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def topk_subgradient_step(a, n_fft, M_fft, k_top, power):
    """
    Compute a weighted subgradient using top-k peaks.
    Returns (gradient, k_max, c_max, score).
    """
    n = len(a)
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:n_fft]
    S = float(a.sum())
    M_val = float(c.max())
    sc = 2.0 * n * M_val / (S * S)

    # Find top-k peaks
    if k_top >= n_fft:
        top_idx = np.arange(n_fft)
    else:
        top_idx = np.argpartition(c, -k_top)[-k_top:]

    # Weight by (c[k] / c_max)^power
    ratio = c[top_idx] / M_val
    weights = ratio ** power
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones_like(top_idx) / len(top_idx)
    else:
        weights = weights / w_sum

    # Compute weighted subgradient
    sg = np.zeros(n)
    indices = np.arange(n)
    for ki, w in zip(top_idx, weights):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if je >= js:
            seg_idx = np.arange(js, je + 1)
            k_minus_i = ki - seg_idx
            self_mask = (k_minus_i == seg_idx)
            contrib = 2.0 * a[k_minus_i]
            contrib[self_mask] = a[seg_idx[self_mask]]
            sg[js:je + 1] += w * contrib

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
    return grad, M_val, sc


def subgradient_descent(a0, n_iters, step_init, momentum, k_top, power):
    """
    Momentum-based subgradient descent with top-k peak weighting.
    """
    n = len(a0)
    n_fft = 2 * n - 1
    M_fft = 1 << (n_fft - 1).bit_length()

    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    patience = 0
    patience_limit = 500

    for it in range(n_iters):
        grad, M_val, sc = topk_subgradient_step(a, n_fft, M_fft, k_top, power)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                break

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        gap = max(sc - sc_lb, 1e-10)
        step = min(step, gap * 0.5 / (gn * gn + 1e-30))
        step = min(step, 0.005)
        # Decay
        step *= 0.99995

        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a_new / S_new

        # Periodic check
        if it % 200 == 0:
            pass  # Could add logging here

    return best_a


def fienup_quick(a, n_iters, alpha):
    """Quick Fienup spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    n_fft = 2 * n - 1
    M_fft = 1 << (n_fft - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.mean(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return a
