"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using iterative active-set exponentiated
gradient, multi-precision p-norm L-BFGS refinement, subgradient polish, and
structured restart diversification.
"""

import numpy as np
import time
import math


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Fast linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Correlation: (a * w)[n-1 : 2n-1] which gives the gradient-like array."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    """Normalize to sum=1, clip to [0, inf]."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    """Compute exact score = 2n*max(conv)/sum^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_weights(conv, top_k=10, eps_frac=0.01):
    """Compute active set weights from convolution peaks."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.08 * conv.size))
    if active.sum() > max_count:
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[top_idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def eg_update(a, conv, weights, eta):
    """One exponentiated gradient step."""
    h = corr_pair(a, weights)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(np.clip(a, 1e-300, None))
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)

    # This is getting complicated. Let me use a different approach.
    
    # Use LSE approximation instead:
    beta = float(p)  # repurpose p as beta for LSE
    z = beta * (c_scaled - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = cmax + np.log(sum_e) / beta
    
    # dL/d(a[j]) = sum_k w[k] * d(c_scaled[k])/d(a[j])
    # c_scaled[k] = n * conv[k] / S^2
    # d(conv[k])/d(a[j]): for linear conv, this is a "correlated" quantity
    # sum_k w[k] * d(conv[k])/d(a[j]) = (w * a)[j] + (w * reverse(a))[j] ... 
    # Actually: sum_k w[k] * d(conv[k])/d(a[j]) = corr_pair(a, w)[j] * 2 (approximately)
    # More precisely: d(conv[k])/d(a[j]) = a[k-j] (if valid) + a[k-j] (if valid, same thing)
    # sum_k w[k] * a[k-j] (for valid k,j pairs) = (convolve... ) = corr_pair(w, a)[j]
    
    hw = corr_pair(a, w)  # This gives sum_i a[i]*w[j-i] for the "overlapping" part
    # Actually corr_pair(a, w) as defined gives the cross-correlation
    # Let me verify: corr_pair reverses a, then convolves with w, takes middle part
    # This should give sum_k w[k] * d(conv[k])/d(a[j]) / 2 (factor of 2 from the double count)
    
    dL_da = (float(n) / (S * S)) * 2.0 * hw  # d(c_scaled)/d(a[j]) summed over k with weight w[k]
    
    # Full gradient of score = 2*n*L/(S^2):
    # d/d(a[j]) [2*n*L/(S^2)] = 2*n/(S^2) * dL/da[j] - 4*n*L*S*a[j]/S^4
    # = 2*n/(S^2) * dL/da[j] - 4*n*L*a[j]/S^3
    # But dL/da[j] already includes the 1/S^2 factor from c_scaled
    # So: grad[j] = 2*n * (n/(S^2)) * 2*hw[j] / (S^2) - 4*n*L*a[j]/S^3
    # Hmm, let me redo this more carefully.
    
    # score = 2*n*max(c)/(S^2) ≈ 2*n*L/(S^2) where L = LSE approx of max(c)
    # d(score)/d(a[j]) = 2*n * [cmax' * S^2 - cmax * 2*S*a[j]] / S^4  (where cmax' is dL/da)
    # = 2*n * [cmax' * S - 2*a[j]*L] / S^3
    
    # cmax' = dL/da[j] = sum_k w[k] * n/(S^2) * d(c[k])/d(a[j])
    #        = n/(S^2) * sum_k w[k] * d(c[k])/d(a[j])
    #        = n/(S^2) * 2*hw[j]  (from the correlation computation)
    
    cmax_prime = (float(n) / (S * S)) * 2.0 * hw
    
    grad = (2.0 * float(n) / (S ** 3)) * (cmax_prime * S - 2.0 * a * L)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    
    return float(2.0 * n * L / (S * S)), grad


def run_lbfgs_pnorm(a, p, deadline, maxfun=400):
    """Run L-BFGS-B on the LSE/P-norm smoothed objective."""
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a
    
    def fun(x):
        return pnorm_fun_grad(x, float(p))
    
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-12,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def subgrad_polish(a, deadline, verbose=False, tag=""):
    """Systematic subgradient polish over parameter grid."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 75, 100, 150]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0, 3.0]
    eps_fracs = [0.0, 0.003, 0.005, 0.01, 0.02, 0.05]
    
    improved = True
    round_num = 0
    while improved and time.monotonic() < deadline and round_num < 10:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)
        
        for tk in top_ks:
            if time.monotonic() > deadline:
                break
            for ef in eps_fracs:
                if time.monotonic() > deadline:
                    break
                w = compute_weights(conv, top_k=tk, eps_frac=ef)
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    new_a = eg_update(best_a, conv, w, eta)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = new_a
                        conv = fast_self_conv(best_a)
                        improved = True
        
        # Variance reduction
        for tm in [5, 10, 20, 50, 100]:
            if time.monotonic() > deadline:
                break
            d = np.sort(conv)[::-1][:tm]
            mean_top = float(np.mean(d))
            excess = np.maximum(conv - mean_top, 0.0)
            total_excess = float(np.sum(excess))
            if total_excess < 1e-15:
                continue
            w_vr = excess / total_excess
            for eta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
                if time.monotonic() > deadline:
                    break
                new_a = eg_update(best_a, conv, w_vr, eta)
                new_a = normalize_a(new_a)
                sc = exact_score(new_a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = new_a
                    conv = fast_self_conv(best_a)
                    improved = True
    
    return float(best_sc), best_a


def resample_sequence(a, n_new):
    """Resample sequence to new length."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n_new)) / float(max(1, n_new))
    if m == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, a).astype(np.float64)


def perturb_structured(a, rng, strength=0.1):
    """Structured perturbation targeting peaks."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    mode = int(rng.integers(0, 8))
    
    if mode == 0:
        # Log-space smooth noise
        k = min(31, max(5, n // 500))
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 1:
        # Peak-targeted: reduce elements contributing to highest peak
        conv = fast_self_conv(p)
        peak_at = int(np.argmax(conv))
        # Elements contributing to this peak
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_at - i
            if 0 <= j < n:
                contrib[i] += p[j]
            j2 = i + ... # other direction
        # Simplified: uniform pressure on peak-contributing elements
        w = compute_weights(conv, top_k=5, eps_frac=0.01)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 2:
        # Bump a random smooth region
        center = float(rng.uniform(0.1, 0.9))
        width = float(rng.uniform(0.05, 0.3))
        x = np.linspace(0.0, 1.0, n)
        bump = np.exp(-((x - center) / width) ** 2)
        q = p + strength * 0.1 * (bump - bump.mean())
    elif mode == 3:
        # Taper modification
        x = np.linspace(0.0, 1.0, n)
        taper = np.exp(-strength * 2.0 * x * (1 - x))
        q = p * taper
    elif mode == 4:
        # Frequency domain perturbation
        Nf = next_pow2(n)
        spec = np.fft.rfft(p, Nf)
        # Add small noise to high frequencies
        mask = np.ones(len(spec))
        mask[:len(spec)//8] = 0  # don't touch low freqs
        spec *= 1.0 + strength * 0.05 * rng.standard_normal(len(spec)) * mask
        q = np.fft.irfft(spec, Nf)[:n]
    elif mode == 5:
        # Power modification
        alpha = 1.0 + strength * 0.5 * (rng.random() * 2 - 1)
        q = np.power(np.clip(p, 1e-10, None), alpha)
    elif mode == 6:
        # Block permutation (swap two segments)
        i1 = int(rng.integers(0, n // 2))
        i2 = int(rng.integers(n // 2, n))
        length = int(rng.integers(n // 10, n // 4))
        q = p.copy()
        q[i1:i1+length], q[i2:i2+length] = q[i2:i2+length].copy(), q[i1:i1+length].copy()
    else:
        # Multiplicative random (smooth)
        k = max(5, n // 1000)
        smooth = np.convolve(rng.random(n), np.ones(k) / k, mode="same")
        q = p * (1.0 + strength * 0.2 * (smooth - 0.5))
    
    return normalize_a(np.clip(q, 0.0, None))
