    """
    Compute LSE-smoothed objective and its gradient.
    Objective: 2n * LSE_beta(C) / S^2
    where LSE_beta(C) = (1/beta) * log(sum exp(beta * C[k]))
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    # Scale C for numerical stability
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    # LSE in the scaled space
    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e  # softmax weights
    LSE = cmax + np.log(sum_e) / beta
    
    # Objective value
    obj = 2.0 * n * LSE / (S * S)
    
    # Gradient of LSE w.r.t. a: 2 * compute_grad(a, w)
    dlse_da = 2.0 * compute_grad(a, w)
    
    # Full gradient of obj = 2n * LSE / S^2
    # d(obj)/da[j] = 2n * [dlse_da[j] * S^2 - LSE * 2S] / S^4
    # = 2n * [dlse_da[j] * S - 2*LSE] / S^3
    grad = 2.0 * n * (dlse_da * S - 2.0 * LSE) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e10, 1e10)
    
    return float(obj), grad


def run_lbfgs(a, beta, deadline, maxfun=1000, maxiter=200):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a
    
    def fun(x):
        return lse_objective_and_grad(x, beta)
    
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-20,
        "gtol": 1e-12,
        "maxiter": max(20, maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception as e:
        print(f"  LBFGS error: {e}")
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def active_set_step(a, conv, top_k, eta):
    """
    One active-set gradient step: reduce the top-k highest convolution values.
    """
    n = len(a)
    # Select top-k positions
    if top_k >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    w = np.zeros(len(conv), dtype=np.float64)
    w[idx] = 1.0
    w /= w.sum()
    
    # Compute gradient direction
    grad = compute_grad(a, w)
    grad_max = float(np.max(np.abs(grad)))
    if grad_max < 1e-15:
        return a.copy()
    
    # Multiplicative update (stay in positive orthant)
    log_a = np.log(a + 1e-300)
    log_a -= eta * grad / grad_max
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def equalization_loop(a, deadline, verbose=True, tag=""):
    """
    Iteratively equalize the top convolution values.
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    top_ks = [1, 2, 3, 5, 8, 12, 20, 30, 50]
    etas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    
    stagnant = 0
    max_stagnant = 20
    
    while time.monotonic() < deadline - 5 and stagnant < max_stagnant:
        conv = fast_self_conv(a)
        cmax = float(np.max(conv))
        
        improved = False
        for tk in top_ks:
            for eta in etas:
                if time.monotonic() > deadline - 5:
                    break
                new_a = active_set_step(a, conv, tk, eta)
                new_a = normalize_a(new_a)
                sc = exact_score(new_a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = new_a
                    a = new_a
                    improved = True
                    conv = fast_self_conv(a)
                    if verbose:
                        print(f"  [{tag}] top_k={tk} eta={eta:.3f} -> {sc:.10f}")
        
        if not improved:
            stagnant += 1
        else:
            stagnant = 0
    
    return float(best_sc), best_a


def generate_antisymmetric_init(n, rng):
    """Generate a sequence with anti-symmetric structure."""
    x = np.linspace(0, 1, n)
    # Base: slightly above uniform
    base = np.ones(n) * 0.5
    
    # Anti-symmetric perturbation
    # a[i] = base[i] + delta * f(i) where f is anti-symmetric: f[i] = -f[n-1-i]
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        # Various anti-symmetric functions
        f = np.sin(np.pi * x)  # Anti-symmetric about x=0.5
        candidate = base + alpha * f / np.max(np.abs(f))
        candidate = np.clip(candidate, 0.01, None)
        yield normalize_a(candidate)
    
    # More complex anti-symmetric
    for freq in [1, 3, 5]:
        for alpha in [0.1, 0.3, 0.5]:
            f = np.sin(np.pi * freq * x)
            candidate = base + alpha * f
            candidate = np.clip(candidate, 0.01, None)
            yield normalize_a(candidate)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    
    def add(arr):
        if len(inits) >= 30:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            # Resample
            x_old = np.linspace(0, 1, arr.size)
            arr = np.interp(x, x_old, arr)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        arr = np.clip(arr, 0.0, None)
        inits.append(normalize_a(arr))
    
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size != n:
            x_old = np.linspace(0, 1, p.size)
            p = np.interp(x, x_old, p)
        add(p)
        add(p[::-1])
    
    # Uniform
    add(np.ones(n))
    
    # Anti-symmetric inits
    for init in generate_antisymmetric_init(n, rng):
        add(init)
    
    # V-shapes
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2 + 0.1)
    
    # U-shapes
    for alpha in (0.3, 0.5, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    
    # Cosine
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    
    return inits
