def next_power_of_two(m):

        # Adaptive beta: logarithmic schedule
        frac = min(1.0, t / total_iters_est)
        beta_lse = beta_lse_start * (beta_lse_end / beta_lse_start) ** frac
        
        sc, grad = lse_obj_grad(a, beta_lse)
        
        if not np.isfinite(sc):
            break
            
        # Compute BB step size if possible
        if prev_x is not None and prev_g is not None:
            s = a - prev_x
            y = grad - prev_g
            ss = float(np.dot(s, s))
            sy = float(np.dot(s, y))
            if ss > 1e-20 and sy > 1e-20:
                bb_lr = ss / sy
                bb_lr = max(1e-5, min(1.0, bb_lr))
        
        prev_x = a.copy()
        prev_g = grad.copy()
        
        # Adam update with BB scaling
        base_lr = 0.005 * math.exp(-frac * 1.5)
        lr = base_lr * (1.0 + 0.5 * (bb_lr / max(base_lr, 1e-10)))
        lr = min(lr, 0.1)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        direction = -m_hat / (np.sqrt(v_hat) + eps)
        
        # Projected gradient step (maintain non-negativity)
        a_new = a + lr * direction
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_a(a_new)
        
        new_sc = exact_score(a_new)
        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a_new.copy()
            a = a_new
        
        # Every 100 iters, try a spectral equalization step
        if it % 100 == 99:
            a_spec = spectral_equalize(a, alpha=0.95, n_steps=5)
            sc_spec = exact_score(a_spec)
            if sc_spec < best_sc:
                best_sc = sc_spec
                best_a = a_spec.copy()
                a = a_spec
        
        if it % 200 == 0:
            print("  [{} adam] iter={} beta_lse={:.0f} score={:.9f} best={:.9f}".format(
                tag, it, beta_lse, sc, best_sc))
    
    return float(best_sc), best_a


def bb_projected_gradient(a, deadline, tag=""):
    """Barzilai-Borwein projected gradient on LSE."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    beta_lse = 500.0
    prev_x = None
    prev_g = None
    lr = 0.005
    
    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        sc, grad = lse_obj_grad(a, beta_lse)
        if not np.isfinite(sc):
            break
        
        if prev_x is not None and prev_g is not None:
            s = a - prev_x
            y = grad - prev_g
            ss = float(np.dot(s, s))
            sy = float(np.dot(s, y))
            if ss > 1e-25:
                if sy > 1e-15:
                    lr_bb = ss / sy
                else:
                    sy2 = float(np.dot(y, y))
                    lr_bb = ss / max(sy2, 1e-20)
                lr = max(1e-6, min(lr_bb, 0.1))
        
        prev_x = a.copy()
        prev_g = grad.copy()
        
        # Nesterov look-ahead
        if it > 1:
            momentum = 0.9 + 0.05 * math.sin(it * 0.01)
            grad_nest = grad + momentum * (grad - prev_g)
        else:
            grad_nest = grad
        
        a_new = a - lr * grad_nest
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_a(a_new)
        
        new_sc = exact_score(a_new)
        if new_sc < best_sc - 1e-13:
            best_sc = new_sc
            best_a = a_new.copy()
            a = a_new
        
        # Slowly increase beta_lse
        if it % 200 == 199:
            beta_lse = min(beta_lse * 1.5, 50000.0)
    
    return float(best_sc), best_a


def run_lbfgs(a, beta, deadline, maxfun=1000):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_obj_grad(np.clip(x, 0.0, None), beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-12,
        "maxiter": max(50, maxfun // 3),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def subgrad_polish(a, deadline, tag=""):
    """Fine subgradient polish with multiple active set sizes."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    top_ks = [3, 5, 8, 12, 20, 35, 50, 80, 120, 200]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]
    
    for round_num in range(12):
        if time.monotonic() > deadline - 2.0:
            break
        conv = fast_self_conv(a)
        dmax = float(np.max(conv))
        if dmax <= 0:
            break
        improved = False
        for tk in top_ks:
            if time.monotonic() > deadline - 2.0:
                break
            # Active set: top tk positions
            idx = np.argpartition(conv, -tk)[-tk:]
            w = np.zeros(conv.size)
            w[idx] = 1.0 / tk
            
            # Compute h = cross-correlation of w and a
            N_fft = next_power_of_two(3 * n + 4)
            W = np.fft.rfft(w, N_fft)
            A = np.fft.rfft(a, N_fft)
            h = np.fft.irfft(W * np.conj(A), N_fft)[:n]
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-30:
                continue
            
            for eta in etas:
                new_a = a * np.exp(-eta * h / hmax)
                new_a = normalize_a(new_a)
                sc = exact_score(new_a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = new_a.copy()
                    a = new_a
                    improved = True
                    if round_num % 3 == 0:
                        print("  [{} sg r{}] tk={} eta={:.3f} -> {:.9f}".format(
                            tag, round_num, tk, eta, best_sc))
        if not improved:
            break
    
    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent, cap=30):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        if float(np.sum(p)) > 0.001:
            inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Slightly perturbed versions
        add(pr * np.exp(0.05 * rng.standard_normal(n)))
        add(pr * np.exp(0.1 * rng.standard_normal(n)))

    add(np.ones(n))
    
    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    # U-shapes  
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    # Cosine modulations
    for f in (1, 2, 3, 4, 6, 8):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    # Sine absolute
    for f in (1, 2, 3, 4):
        for alpha in (0.5, 1.0, 1.5, 2.0):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(x - 0.5) * 2) + 0.01)
    # Bimodal
    for width in (0.03, 0.05, 0.08, 0.12, 0.15):
        for center in (0.15, 0.2, 0.25, 0.3, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Power shapes
    for p in (0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Frechet-like
    for s in (1.5, 2.0, 3.0, 5.0):
        add(x ** (1.0/s) * (1.0-x) ** (1.0/s) + 0.01)
    # Random smooth (spectral)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    # Random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))
    # Sawtooth-like
    for period in (8, 16, 32, 64):
        saw = np.array([abs(((i * period // n) % period) / (period / 2) - 1) for i in range(n)])
        add(1.0 + 0.5 * saw)

    return inits


def perturb(a, rng, strength=0.1):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.03 * rng.standard_normal(n))
    elif mode == 3:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * np.cos(2.0 * np.pi * rng.integers(1, 20) * x))
    elif mode == 4:
        # Frequency-domain perturbation
        N_fft = next_power_of_two(2 * n)
        A = np.fft.rfft(p, N_fft)
        mag = np.abs(A) * np.exp(0.05 * rng.standard_normal(len(A)))
        phase = A / (mag + 1e-30)
        A_new = mag * phase
        q = np.fft.irfft(A_new, N_fft)[:n]
    elif mode == 5:
        # Block perturbation
        nb = int(rng.integers(4, 16))
        per = max(1, n // nb)
        blocks = 1.0 + strength * 0.2 * (rng.random(nb) * 2 - 1)
        arr = np.repeat(blocks, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, blocks[-1])])
        q = p * arr[:n]
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 500)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        q = p * (0.9 + 0.2 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    n = 10240
    if parent is not None:
        n = int(parent.size)
        if n < 1024:
            n = 8192
        elif n > 20000:
            n = 16384
        a0 = resample_sequence(parent, n)
    else:
        inits = generate_inits(n, rng, None, cap=15)
        best_init_sc = np.inf
        a0 = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a0 = init
        print("No parent. Best init score: {:.9f}".format(best_init_sc))

    best_a = normalize_a(a0).copy()
    best_sc = exact_score(best_a)
    print("Initial: n={}, score={:.9f}".format(n, best_sc))

    # Phase 1: Multi-start spectral equalization + quick Adam (25% time)
    p1_dead = t_start + 0.25 * total_budget
    candidates = []
    
    if parent is not None:
        base = normalize_a(resample_sequence(parent, n))
        candidates.append(("parent", base))
        candidates.append(("parent_rev", normalize_a(base[::-1])))
        candidates.append(("perturb1", perturb(base, rng, 0.05)))
        candidates.append(("perturb2", perturb(base, rng, 0.1)))
    else:
        inits = generate_inits(n, rng, None, cap=10)
        for i, init in enumerate(inits[:5]):
            candidates.append(("init{}".format(i), init))
    
    # Add some structured candidates
    x = np.linspace(0.0, 1.0, n)
    for alpha in [0.5, 1.0, 2.0]:
        candidates.append(("cos{}_1".format(alpha), normalize_a(1.0 + alpha * np.cos(2 * np.pi * 3 * x))))
    candidates.append(("bimodal", normalize_a(np.exp(-((x - 0.3) ** 2) / 0.01) + np.exp(-((x - 0.7) ** 2) / 0.01) + 0.01)))
    
    for cname, cand in candidates:
        if time.monotonic() > p1_dead - 30:
            break
        sc = exact_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            print("  Phase1 [{}]: {:.9f}".format(cname, best_sc))
        # Quick spectral equalization
        if time.monotonic() < p1_dead - 30:
            eq = spectral_equalize(cand, alpha=0.9, n_steps=10)
            sc_eq = exact_score(eq)
            if sc_eq < best_sc:
                best_sc = sc_eq
                best_a = eq.copy()
                print("  Phase1 [{}+eq]: {:.9f}".format(cname, best_sc))
    
    # Quick Adam on best
    if time.monotonic() < p1_dead - 10:
        sc, a = adam_lbfgs_hybrid(best_a, p1_dead - 5, tag="p1adam")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    print("After Phase 1: {:.9f}".format(best_sc))

    # Phase 2: Main Adam optimization (35% time)
    p2_dead = t_start + 0.60 * total_budget
    sc, a = adam_lbfgs_hybrid(best_a, p2_dead, tag="p2")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 2: {:.9f}".format(best_sc))

    # Phase 3: BB projected gradient (15% time)
    p3_dead = t_start + 0.75 * total_budget
    sc, a = bb_projected_gradient(best_a, p3_dead, tag="p3")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 3: {:.9f}".format(best_sc))

    # Phase 4: L-BFGS refinement (15% time)
    p4_dead = t_start + 0.90 * total_budget
    a = normalize_a(best_a)
    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000, 40000]:
        if time.monotonic() > p4_dead - 10:
            break
        rem = p4_dead - time.monotonic()
        if rem < 15:
            break
        mf = max(50, int(rem * 0.8 / 0.2))
        mf = min(mf, 1500)
        sc, a = run_lbfgs(a, beta, p4_dead, mf)
        a = normalize_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  Phase4 beta={}: {:.9f}".format(beta, best_sc))

    # Phase 5: Subgradient fine polish (7% time)
    p5_dead = t_start + 0.97 * total_budget
    sc, a = subgrad_polish(best_a, p5_dead, tag="p5")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 5: {:.9f}".format(best_sc))

    # Phase 6: Diversification restarts (3% time)
    p6_dead = deadline - 5.0
    if time.monotonic() < p6_dead - 40:
        n_restarts = 2
        while time.monotonic() < p6_dead - 30 and n_restarts > 0:
            rem = p6_dead - 30.0 - time.monotonic()
            if rem < 30:
                break
            n_restarts -= 1
            p = perturb(best_a, rng, strength=float(rng.uniform(0.03, 0.12)))
            chunk_dead = time.monotonic() + rem * 0.7
            sc, seq = adam_lbfgs_hybrid(p, chunk_dead, tag="r{}".format(n_restarts))
            if sc < best_sc:
                best_sc = sc
                best_a = seq.copy()
                print("Restart {}: {:.9f}".format(n_restarts, best_sc))

    # Final polish
    if time.monotonic() < deadline - 5:
        sc, a = subgrad_polish(best_a, deadline - 2, tag="final")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final output
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(min(n, 1024)) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
