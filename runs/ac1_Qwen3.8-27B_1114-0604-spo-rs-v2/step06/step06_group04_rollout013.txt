"""
Minimize 2*n*max(a*a)/(sum(a)^2) for nonnegative a.

This solver combines:
  1. length-preserving and upscale restarts from the previous best sequence,
  2. a fast smoothed explorer (Adam on exponential variables),
  3. an exact improving exponentiated-gradient line searcher with adaptive
     active-lag weights,
  4. local rescues: symmetrization, positive smoothing, Frank-Wolfe-style
     dual-lag-weight construction, and controlled random restarts.
"""

import numpy as np
import time
import math


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    np.maximum(c, 0.0, out=c)
    return c


def _score_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1e-15:
        return np.inf
    c = _fast_self_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * float(c.max()) / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / S


def _resample(a, n):
    n = int(n)
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    if a.size == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    S = float(a.sum())
    if S <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    a = a / S
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a).astype(np.float64)
    S = float(out.sum())
    if S <= 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return out / S


def _log_from_a(a):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    S = float(a.sum())
    if S <= 1e-15:
        z = np.zeros(n, dtype=np.float64)
    else:
        a = a / S
        z = np.log(a)
    z = np.nan_to_num(z, nan=-30.0, posinf=30.0, neginf=-30.0)
    z -= float(z.mean())
    z = np.clip(z, -50.0, 50.0)
    z -= float(z.mean())
    return z


def _softmax_log(logx):
    logx = np.asarray(logx, dtype=np.float64).ravel()
    n = logx.size
    if n == 0:
        return np.ones(0, dtype=np.float64), np.zeros(0, dtype=np.float64)
    logx = np.nan_to_num(logx, nan=-50.0, posinf=50.0, neginf=-50.0)
    logx = np.clip(logx, -60.0, 60.0)
    m = float(logx.max())
    y = logx - m
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        z = np.zeros(n, dtype=np.float64)
        return a, z
    a = e / s
    z = y - math.log(s)
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    z = np.clip(z, -50.0, 50.0)
    z -= float(z.mean())
    z = np.clip(z, -50.0, 50.0)
    return a, z


def _lag_weights(d, top_count):
    d = np.asarray(d, dtype=np.float64).ravel()
    L = d.size
    if L == 0:
        return np.zeros(0, dtype=np.float64)
    top_count = int(top_count)
    if top_count <= 1:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(d))] = 1.0
        return p
    if top_count >= L:
        min_top = float(d.min())
    else:
        k = L - top_count
        if k < 0:
            k = 0
        part = np.argpartition(d, k)
        min_top = float(d[part[k:]].min())
    maxd = float(d.max())
    gap = maxd - min_top
    if not np.isfinite(gap) or gap <= 1e-12:
        gap2 = maxd - float(d.min())
        if np.isfinite(gap2) and gap2 > 1e-12:
            gap = gap2
        else:
            gap = max(abs(maxd), 1e-12) * 0.01
    width = max(gap, max(abs(maxd), 1e-15) * 1e-9, 1e-15)
    z = (d - maxd) / width
    z = np.clip(z, -700.0, 0.0)
    p = np.exp(z)
    p += 1e-300
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.full(L, 1.0 / float(L), dtype=np.float64)
    else:
        p /= s
    return p


def _corr_flip(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    Lw = w.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    need = n + Lw - 1
    N = _next_pow2(need)
    ar = a[::-1]
    A = np.fft.rfft(ar, N)
    W = np.fft.rfft(w, N)
    v = np.fft.irfft(A * W, N)
    v = v[:need]
    R = np.zeros(n, dtype=np.float64)
    lo = n - 1
    hi = 2 * n - 1
    if lo < 0:
        lo = 0
    if hi > need:
        hi = need
    take = min(n, hi - lo)
    if take > 0:
        R[:take] = v[lo:lo + take]
    return R


def _structured_inits(n, rng, cap=8):
    out = []
    n = int(n)
    if n <= 0:
        return out
    x = np.linspace(0.0, 1.0, n)
    cands = []
    cands.append(np.ones(n, dtype=np.float64))

    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        cands.append(np.power(np.sin(np.pi * x), alpha) + 1e-7)

    for alpha in (0.5, 1.0, 2.0):
        cands.append(np.power(x * (1.0 - x), alpha) + 1e-7)

    for rate in (1.0, 2.0, 4.0, 8.0):
        cands.append(0.5 * (np.power(x, rate) + np.power(1.0 - x, rate)) + 0.01)

    Nf = _next_pow2(n)
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf)
        pwr = 1.0 / ((np.abs(freqs) * float(n) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        pwr[0] = 0.0
        ph = rng.uniform(-np.pi, np.pi, size=pwr.size)
        spec = pwr * np.exp(1j * ph)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= float(noise.mean())
        base = np.sin(np.pi * x) ** 2 + 0.2
        cands.append(base + 0.5 * noise)

    for arr in cands:
        if len(out) >= cap:
            break
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 1e-12, None)
        S = float(arr.sum())
        if S > 1e-15:
            out.append(arr / S)
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


def _fw_lag_weights(a, max_iters=60):
    a = _normalize(a)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    v = np.full(L, 1.0 / float(L), dtype=np.float64)
    bottom_count = max(4, min(32, n // 512))
    if bottom_count >= L:
        bottom_count = L
    max_iters = int(max(1, max_iters))
    for it in range(max_iters):
        H = _corr_flip(a, v)
        H = np.nan_to_num(H, nan=0.0, posinf=0.0, neginf=0.0)
        if H.size == 0:
            break
        if H.size < bottom_count:
            idx = np.arange(H.size)
        else:
            part = np.argpartition(H, bottom_count)
            idx = part[:bottom_count]
        hv = H[idx]
        hmaxv = float(np.max(hv))
        weights = np.maximum(hmaxv - hv, 0.0)
        ws = float(weights.sum())
        if ws <= 0.0 or not np.isfinite(ws):
            weights = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
        else:
            weights /= ws
        g = np.zeros(L, dtype=np.float64)
        for ii in range(idx.size):
            i = int(idx[ii])
            wt = float(weights[ii])
            if wt > 0.0:
                end = min(L, i + n)
                g[i:end] += wt * a
        if g.size == 0:
            break
        kmax = int(np.argmax(g))
        t = 2.0 / float(it + 2)
        v *= (1.0 - t)
        v[kmax] += t
        if (it + 1) % 10 == 0:
            s = float(v.sum())
            if s > 0.0:
                v /= s
    s = float(v.sum())
    if s > 0.0:
        v /= s
    return v


def _quick_adam(a0, deadline, seed=1, prefix="Adam"):
    a0 = _normalize(a0)
    n = int(a0.size)
    if n == 0:
        return 2.0, np.ones(1, dtype=np.float64)
    deadline = float(deadline)
    z = _log_from_a(a0)
    a, z = _softmax_log(z)
    c = _fast_self_conv(a)
    cur = float(2.0 * n * float(c.max()))
    best = cur if np.isfinite(cur) else np.inf
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    bias1 = 1.0
    bias2 = 1.0
    t = 0
    base_lr = 0.05
    no_impr = 0

    L = 2 * n - 1
    start_top = 0.25
    end_top = max(8.0 / float(L), 1e-6)

    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    opt_end = deadline
    last_print = t_start

    while time.monotonic() < opt_end:
        if time.monotonic() > opt_end - 2.0:
            break
        if time.monotonic() - last_print > 20.0:
            print("[{}] best={:.8f} cur={:.8f} eta={:.5f}".format(prefix, best, cur, base_lr))
            last_print = time.monotonic()

        c = _fast_self_conv(a)
        sc = float(2.0 * n * float(c.max()))
        if np.isfinite(sc) and sc < best:
            best = sc
            best_a = a.copy()
            no_impr = 0
            base_lr = min(0.18, base_lr * 1.015)
        else:
            no_impr += 1
            if no_impr >= 500:
                if np.isfinite(sc) and sc > best + 0.02:
                    z = _log_from_a(best_a)
                    a, z = _softmax_log(z)
                    c = _fast_self_conv(a)
                    sc = float(2.0 * n * float(c.max()))
                base_lr = max(0.005, base_lr * 0.6)
                no_impr = 0

        d = n * c
        elapsed = time.monotonic() - t_start
        total = opt_end - t_start
        prog = 0.0
        if total > 1e-9:
            prog = min(1.0, max(0.0, elapsed / total))
        if start_top > end_top:
            top_frac = max(end_top, start_top * (end_top / start_top) ** (1.0 - prog))
        else:
            top_frac = start_top
        top_count = int(round(L * top_frac))
        top_count = max(8, min(L, top_count))

        w = _lag_weights(d, top_count)
        R = _corr_flip(a, w)
        H = R - float(np.mean(R))
        hmax = float(np.max(np.abs(H)))
        if not np.isfinite(hmax) or hmax < 1e-12:
            z = z + 0.05 * rng.standard_normal(n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(z.mean())
            a, z = _softmax_log(z)
            continue
        H = H / hmax

        t += 1
        bias1 = b1 * bias1
        bias2 = b2 * bias2
        m = b1 * m + (1.0 - b1) * H
        v = b2 * v + (1.0 - b2) * (H * H)
        mhat = m / (1.0 - bias1)
        vhat = v / (1.0 - bias2)
        step = base_lr * mhat / (np.sqrt(vhat) + 1e-8)
        step = np.nan_to_num(step, nan=0.0, posinf=0.0, neginf=0.0)

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z -= float(z.mean())
        a, z = _softmax_log(z)

    return best, best_a

            if np.isfinite(sc_sym) and sc_sym < cur - 1e-12:
                a = sym
                z = _log_from_a(a)
                cur = sc_sym
                rescued = True
            if not rescued:
                for K in (3, 5, 7, 11, 15):
                    if K < 3 or K > n:
                        continue
                    kern = np.ones(K, dtype=np.float64)
                    kern /= float(K)
                    sm = np.convolve(a, kern, mode="same")
                    sm = _normalize(sm)
                    sc_sm = _score_seq(sm)
                    if np.isfinite(sc_sm) and sc_sm < cur - 1e-12:
                        a = sm
                        z = _log_from_a(a)
                        cur = sc_sm
                        rescued = True
                        break
                smooth_attempts += 1

        if rescued:
            if cur < best:
                best = cur
                best_a = a.copy()
            no_impr = 0
            base_eta = min(max_eta, max(base_eta, 0.5))
            top_frac = max(target_top, top_frac * 0.9)
            continue

        if no_impr >= 150 and fw_attempts < 6 and time.monotonic() < opt_end - 15.0:
            w_fw = _fw_lag_weights(a, 50)
            a_v = _normalize(_resample(w_fw, n))
            sc_v = _score_seq(a_v)
            if np.isfinite(sc_v) and sc_v < cur - 1e-12:
                a = a_v
                z = _log_from_a(a)
                cur = sc_v
                rescued = True
            else:
                Rfw = _corr_flip(a, w_fw)
                Hfw = Rfw - float(np.mean(Rfw))
                hmaxfw = float(np.max(np.abs(Hfw)))
                if np.isfinite(hmaxfw) and hmaxfw > 1e-12:
                    Hfw = Hfw / hmaxfw
                    eta = base_eta
                    for _ in range(3):
                        if eta < min_eta:
                            break
                        logc = z - eta * Hfw
                        a2, z2 = _softmax_log(logc)
                        c2 = _fast_self_conv(a2)
                        sc2 = float(2.0 * n * float(c2.max()))
                        if np.isfinite(sc2) and sc2 < cur - 1e-12:
                            a = a2
                            z = z2
                            cur = sc2
                            rescued = True
                            break
                        eta *= 0.5
            fw_attempts += 1

        if rescued:
            if cur < best:
                best = cur
                best_a = a.copy()
            no_impr = 0
            base_eta = min(max_eta, max(base_eta, 0.5))
            top_frac = max(target_top, 0.02)
            continue

        if no_impr >= 100:
            sig = 0.05 + 0.05 * min(3.0, math.log10(1.0 + no_impr / 20.0))
            z = z + sig * rng.standard_normal(n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(z.mean())
            a, z = _softmax_log(z)
            c = _fast_self_conv(a)
            sc = float(2.0 * n * float(c.max()))
            if np.isfinite(sc):
                cur = sc
                if sc < best:
                    best = sc
                    best_a = a.copy()
            no_impr = 0
            top_frac = max(target_top, 0.1)
            base_eta = 0.5
            smooth_attempts = 12
            fw_attempts = 6
            continue

        base_eta = max(min_eta, min(base_eta, eta) * 0.8)
        top_frac = min(0.6, top_frac * 1.07)

        if no_impr >= 300:
            z = _log_from_a(best_a)
            a, z = _softmax_log(z)
            c = _fast_self_conv(a)
            cur = float(2.0 * n * float(c.max()))
            if not np.isfinite(cur):
                cur = best
            no_impr = 100
            top_frac = max(target_top, 0.1)
            base_eta = 0.3

    return best, best_a


def _run_stage(a0, stage_dead, seed=1, prefix="Stage"):
    a0 = _normalize(a0)
    sc0 = _score_seq(a0)
    if not np.isfinite(sc0):
        sc0 = np.inf
    best_score = sc0
    best_a = a0.copy()
    now = time.monotonic()
    stage_dead = float(stage_dead)

    if np.isfinite(sc0) and sc0 > 1.75 and stage_dead - now > 150.0:
        adam_dead = now + 0.45 * (stage_dead - now)
        if adam_dead > now + 30.0:
            try:
                sc_adam, a_adam = _quick_adam(a0, adam_dead, int(seed), prefix + ".Adam")
                if np.isfinite(sc_adam) and sc_adam < best_score:
                    best_score = sc_adam
                    best_a = a_adam
            except Exception:
                pass

    try:
        sc_pol, a_pol = _polish_seq(best_a, stage_dead, int(seed + 17), prefix + ".Polish")
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol
    except Exception:
        pass

    if not np.isfinite(best_score):
        best_score = sc0 if np.isfinite(sc0) else 2.0
        best_a = a0.copy()
    return best_score, best_a


def _load_parent():
    try:
        h = globals().get("height_sequence_1", None)
        if h is None:
            return None
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 64 and float(p.sum()) > 1e-15:
            return _normalize(p)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = float(max(60.0, budget_s))
    deadline = t0 + budget - 60.0
    rng = np.random.default_rng(int(seed))

    parent = _load_parent()
    print("Parent loaded:", "yes" if parent is not None else "no",
          "len=", 0 if parent is None else int(parent.size))
    if parent is not None:
        try:
            print("Parent internal score: {:.8f}".format(_score_seq(parent)))
        except Exception:
            pass

    # Build candidate lengths.
    lengths = []
    if parent is not None:
        par_len = int(parent.size)
        if par_len < 2048:
            par_len = 8192
        elif par_len > 24576:
            par_len = 16384
        lengths.append(par_len)
        if par_len < 16384:
            lengths.append(16384)
        if par_len <= 16384:
            lengths.append(32768)
        else:
            lengths.append(16384)
    else:
        lengths = [16384, 32768]

    uniq = []
    for L in lengths:
        L = int(L)
        if 2048 <= L <= 32768 and L not in uniq:
            uniq.append(L)
    if not uniq:
        uniq = [16384]
    lengths = uniq[:3]
    print("Search lengths:", lengths)

    starts = []
    for n in lengths:
        inits = []
        if parent is not None:
            pr = _resample(parent, n)
            inits.append(pr)
            if n > 16:
                for K in (5, 7, 11):
                    kern = np.ones(K, dtype=np.float64)
                    kern /= float(K)
                    inits.append(_normalize(np.convolve(pr, kern, mode="same")))
            inits.append(_normalize(pr + 0.05 * rng.standard_normal(n)))

        cap = 5 if n > 16384 else 7
        inits.extend(_structured_inits(n, rng, cap=cap))

        for arr in inits:
            arr = _normalize(arr)
            if arr.size != n:
                continue
            sc = _score_seq(arr)
            if np.isfinite(sc):
                starts.append((sc, arr))

    if not starts:
        n_fallback = lengths[0]
        starts = [(2.0, np.ones(n_fallback, dtype=np.float64) / float(n_fallback))]

    # Sort starts by score without using lambdas.
    m = len(starts)
    scores = np.empty(m, dtype=np.float64)
    for i in range(m):
        scores[i] = starts[i][0]
    order = np.argsort(scores)

    chosen = []
    used_lengths = set()
    for idx in order:
        if len(chosen) >= 3:
            break
        sc, a = starts[idx]
        if a.size not in used_lengths:
            chosen.append(a)
            used_lengths.add(a.size)
    if len(chosen) < 2 and m > 1:
        for idx in order:
            if len(chosen) >= 2:
                break
            sc, a = starts[idx]
            chosen.append(a)

    best_score = np.inf
    best_arr = None
    for sc, a in starts:
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = a.copy()

    print("Initial best internal score: {:.8f} len={}".format(
        best_score, 0 if best_arr is None else int(best_arr.size)))

    search_deadline = deadline - 180.0
    if search_deadline < t0 + 300.0:
        search_deadline = min(deadline, t0 + 300.0)

    stage_start = time.monotonic()
    total_stage_time = max(120.0, search_deadline - stage_start)

    if len(chosen) == 1:
        fracs = [1.0]
    elif len(chosen) == 2:
        fracs = [0.55, 0.45]
    else:
        fracs = [0.50, 0.25, 0.25]

    cum = 0.0
    for i in range(len(chosen)):
        if i >= len(fracs):
            break
        cum += fracs[i]
        sub_dead = stage_start + cum * total_stage_time
        sub_dead = min(sub_dead, search_deadline)
        if time.monotonic() > sub_dead - 30.0:
            continue
        try:
            sc, arr = _run_stage(chosen[i], sub_dead, int(seed + 101 * (i + 1)),
                                 "S{}-n{}".format(i + 1, int(chosen[i].size)))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = arr
                print("After stage {}: {:.8f} len={}".format(i + 1, best_score, int(best_arr.size)))
        except Exception as e:
            print("Stage error: {}".format(e))

    # Final cross-length / polish phase.
    final_start = time.monotonic()
    final_dead = deadline - 25.0
    if best_arr is not None and final_dead > final_start + 40.0:
        if (np.isfinite(best_score) and best_score > 1.70 and
                best_arr.size < 24000 and final_dead > final_start + 180.0):
            try:
                a_big = _resample(best_arr, 32768)
                sc_big, arr_big = _run_stage(a_big, final_dead - 30.0, int(seed + 555),
                                             "FinalBig32k")
                if np.isfinite(sc_big) and sc_big < best_score:
                    best_score = sc_big
                    best_arr = arr_big
                    print("Final big length: {:.8f} len=32768".format(best_score))
            except Exception as e:
                print("Final big error: {}".format(e))

        if final_dead > time.monotonic() + 30.0:
            try:
                sc_now, arr_now = _run_stage(best_arr, final_dead, int(seed + 777),
                                             "FinalPolish")
                if np.isfinite(sc_now) and sc_now < best_score:
                    best_score = sc_now
                    best_arr = arr_now
                    print("Final polish: {:.8f} len={}".format(best_score, int(best_arr.size)))
            except Exception as e:
                print("Final polish error: {}".format(e))

    if best_arr is None:
        best_arr = np.ones(4096, dtype=np.float64) / 4096.0

    out_arr = _normalize(best_arr)
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.full(1024, 0.5, dtype=np.float64)
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float("nan")

    print("FINAL internal={:.8f} verified={} len={} elapsed={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t0))
    return out
