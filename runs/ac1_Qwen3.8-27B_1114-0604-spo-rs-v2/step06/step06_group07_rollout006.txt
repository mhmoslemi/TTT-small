"""
Optimized step function construction using enhanced spectral shaping,
momentum-based proximal gradient, and adaptive L-BFGS refinement.

import numpy as np
from scipy.optimize import minimize
import time


def fftsize(M):
    return 1 << (M - 1).bit_length()


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = fftsize(M)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    return a / S


def fienup_enhanced(a, n_iters=500, schedule='sinusoidal',
                    alpha_min=0.05, alpha_max=0.9,
                    sigmoid_beta=30.0, target='median'):
    """Enhanced Fienup with multiple alpha schedules."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    Mf = fftsize(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        if schedule == 'linear':
            alpha = alpha_min + (alpha_max - alpha_min) * t
        elif schedule == 'sinusoidal':
            alpha = 0.5 * (alpha_min + alpha_max) + 0.5 * (alpha_max - alpha_min) * np.sin(np.pi * t)
        elif schedule == 'anneal':
            alpha = alpha_max * (1 - t) + alpha_min * t
        elif schedule == 'plateau':
            if t < 0.3:
                alpha = alpha_min + (0.5 * (alpha_min + alpha_max) - alpha_min) * (t / 0.3)
            elif t < 0.7:
                alpha = 0.5 * (alpha_min + alpha_max)
            else:
                alpha = 0.5 * (alpha_min + alpha_max) + (alpha_max - 0.5 * (alpha_min + alpha_max)) * ((t - 0.7) / 0.3)
        else:
            alpha = alpha_min + (alpha_max - alpha_min) * t
        
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        
        if target == 'median':
            C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        elif target == 'mean':
            C = np.mean(amp[1:]) if len(amp) > 1 else amp[0]
        elif target == 'geomean':
            C = np.exp(np.mean(np.log(amp[1:] + 1e-30))) if len(amp) > 1 else amp[0]
        else:
            C = np.median(amp)
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp[0] = amp_safe[0]  # Keep DC
        amp_new = amp_safe * (C / amp_safe) ** alpha
        amp_new[0] = amp[0]  # Keep DC unchanged
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, Mf)[:n]
        
        # Soft non-negativity
        if sigmoid_beta > 0:
            gate = 1.0 / (1.0 + np.exp(-sigmoid_beta * np.clip(a_new, -100, 100)))
            a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        if a_new.sum() < 1e-12:
            break
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Optimized step function construction using enhanced spectral shaping with
multiple alpha schedules, momentum-based proximal subgradient descent,
Frank-Wolfe iterations, and adaptive L-BFGS with log-sum-exp smoothing.
Key improvements: (1) sinusoidal/plateau alpha schedules in Fienup,
(2) momentum proximal gradient with adaptive step, (3) FW vertex steps,
(4) multi-stage adaptive-beta L-BFGS, (5) anti-peak spectral targeting.

import numpy as np
from scipy.optimize import minimize
import time


def _fftsize(M):
    return 1 << (M - 1).bit_length()


def _fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = _fftsize(M)
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def _compute_score(a):
    n = len(a)
    c = _fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def _normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    return a / S


def _subgradient_conv(a, top_k=5):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a, using top peaks.
    Returns gradient vector (length n) for f(a) = max_k (a*a)_k.
    """
    n = len(a)
    Mf = _fftsize(2 * n - 1)
    A = np.fft.rfft(a, Mf)
    c = np.fft.irfft(A * A, Mf)[:2 * n - 1]
    
    k = min(top_k, len(c))
    if k < len(c):
        idx = np.argpartition(c, -k)[-k:]
    else:
        idx = np.arange(len(c))
    
    # Weight peaks by their value (larger peaks get more weight)
    vals = c[idx]
    weights = vals ** 2  # Emphasize larger peaks
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(idx)) / len(idx)
    else:
        weights = weights / wsum
    
    sg = np.zeros(n)
    for i, ki in enumerate(idx):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if je >= js:
            sg[js:je + 1] += weights[i] * 2.0 * a[js:je + 1][::-1]
    return sg


def _fienup_enhanced(a, n_iters=400, schedule='sinusoidal',
                     alpha_min=0.05, alpha_max=0.9,
                     sigmoid_beta=40.0, target='median',
                     dc_fixed=True):

    a = _normalize(a.astype(np.float64).copy())
    Mf = _fftsize(2 * n - 1)
    best_a = a.copy()
    best_sc = _compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        if schedule == 'linear':
            alpha = alpha_min + (alpha_max - alpha_min) * t
        elif schedule == 'sinusoidal':
            alpha = 0.5 * (alpha_min + alpha_max) + 0.5 * (alpha_max - alpha_min) * np.sin(np.pi * t)
        elif schedule == 'plateau':
            if t < 0.25:
                alpha = alpha_min + (0.5 * (alpha_min + alpha_max) - alpha_min) * (t / 0.25)
            elif t < 0.7:
                alpha = 0.5 * (alpha_min + alpha_max)
            else:
                alpha = 0.5 * (alpha_min + alpha_max) + (alpha_max - 0.5 * (alpha_min + alpha_max)) * ((t - 0.7) / 0.3)
        elif schedule == 'anneal':
            alpha = alpha_max - (alpha_max - alpha_min) * t
        elif schedule == 'oscillate':
            alpha = 0.5 * (alpha_min + alpha_max) + 0.4 * (alpha_max - alpha_min) * np.sin(3.0 * np.pi * t)
            alpha = max(alpha_min, min(alpha_max, alpha))
        else:
            alpha = alpha_min + (alpha_max - alpha_min) * t
        
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)

        if len(amp) > 1:

                C = np.median(amp[1:])
            elif target == 'geomean':
                C = np.exp(np.mean(np.log(np.maximum(amp[1:], 1e-30))))
            elif target == 'mean':
                C = np.mean(amp[1:])
            else:
                C = np.median(amp[1:])
        else:
            C = amp[0]

def _momentum_prox(a, n_iters=5000, momentum=0.9, step_init=0.01,
                   top_k=5, decay=0.99998):
    """Momentum-based proximal subgradient descent."""
    n = len(a)
    a = _normalize(a.astype(np.float64).copy())
    v = np.zeros(n)
    step = step_init
    best_a = a.copy()
    best_sc = _compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        sg = _subgradient_conv(a, top_k=top_k)
        S = float(a.sum())
        S2 = S * S
        M_val = float(_fast_conv(a).max())
        sc = 2.0 * n * M_val / S2
        
        # Full gradient of objective 2n*max_c / S^2
        # d(2n*M/S^2)/da = 2n*(sg/S^2 - 2*M*sg_sum... )
        # Actually: dM/da = sg, d(S^2)/da = 2*S*1
        # d(2n*M/S^2)/da = 2n*(sg/S^2 - M*2/S^3 * ones)
        # = 2n/S^2 * (sg - 2*M/S * ones)
        grad = (2.0 * n / S2) * (sg - 2.0 * M_val / S * np.ones(n))
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)
        
        if a_new.sum() < 1e-12:
            break
        a = a_new
        
        sc_new = _compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        else:
            # Rejected step: reduce momentum velocity
            v *= 0.5
        
        step *= decay
    
    return best_a


def _frank_wolfe(a, n_iters=5000, top_k=5):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    n = len(a)
    a = _normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = _compute_score(a)
    
    for it in range(n_iters):
        t_frac = (it + 1) / (it + 2.0)
        sg = _subgradient_conv(a, top_k=top_k)
        S = float(a.sum())
        S2 = S * S
        M_val = float(_fast_conv(a).max())
        sc = 2.0 * n * M_val / S2
        
        # Full gradient
        grad = (2.0 * n / S2) * (sg - 2.0 * M_val / S * np.ones(n))
        
        # FW: linearize, find argmin of <grad, b> over simplex
        i_star = int(np.argmin(grad))
        e = np.zeros(n)
        e[i_star] = 1.0
        
        a_new = (1.0 - t_frac) * a + t_frac * e
        a_new = _normalize(a_new)
        
        sc_new = _compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def _lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    M = 2 * n - 1
    Mf = _fftsize(M)
    A = np.fft.rfft(x, Mf)
    c = np.fft.irfft(A * A, Mf)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via convolution
    p = exp_s / exp_sum
    rev_x = x[::-1]
    M2 = 2 * n + n - 2  # enough for conv(p, rev_x)
    M2f = _fftsize(2 * n + M)
    RX = np.fft.rfft(rev_x, M2f)
    PP = np.fft.rfft(p, M2f)
    conv_res = np.fft.irfft(RX * PP, M2f)
    # conv_res[i + n - 1] = sum_k p[k] * x[k - (i+n-1) + n - 1] = sum_k p[k]*x[k-i]
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def _run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        _lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    a_out = _normalize(a_out)
    exact_sc = _compute_score(a_out)
    return exact_sc, a_out


def _anti_peak_fienup(a, n_iters=300, alpha=0.5, top_peaks=5):
    """Fienup variant that specifically targets time-domain peaks by
    adjusting the spectral components that contribute most to them."""
    n = len(a)
    a = _normalize(a.astype(np.float64).copy())
    Mf = _fftsize(2 * n - 1)
    M = 2 * n - 1
    best_a = a.copy()
    best_sc = _compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, Mf)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Find time-domain peaks
        c = _fast_conv(a)
        k_peaks = min(top_peaks, M)
        if k_peaks < M:
            peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        else:
            peak_idx = np.arange(M)
        peak_vals = c[peak_idx]
        w = peak_vals ** 2
        w = w / w.sum()
        
        # Compute which spectral bins contribute most to peaks
        # c_k = (1/Mf) * sum_j |A_j|^2 * exp(2*pi*i*j*k/Mf)
        # To reduce c at peak positions, reduce |A_j|^2 where exp(2*pi*i*j*k/Mf) is positive
        # Simple heuristic: weight amplification by how much each bin contributes to peaks
        # Contribution of bin j to peak k: |A_j|^2 * cos(2*pi*j*k/Mf)
        # We want to reduce bins that have positive cosine at peaks
        
        if len(amp) > 1:
            C = np.median(amp[1:])
            C = max(C, 1e-20)
            amp_safe = np.maximum(amp, 1e-30)
            
            # Compute average cosine weight for each bin
            cos_w = np.zeros(len(amp))
            for i, pk in enumerate(peak_idx):
                pk = int(pk)
                angles = 2.0 * np.pi * np.arange(len(amp)) * pk / Mf
                cos_w += w[i] * np.cos(angles)
            
            # Bins with high positive cos_w contribute to peaks; suppress them more
            # Bins with negative cos_w help reduce peaks; keep them
            weight = np.ones(len(amp))
            weight[1:] = 1.0 - 0.5 * np.clip(cos_w[1:], -1, 1)
            weight = np.maximum(weight, 0.3)
            
            amp_new = amp_safe.copy()
            for j in range(1, len(amp)):
                amp_new[j] = amp_safe[j] * (C / amp_safe[j]) ** (alpha * weight[j])
            
            A_new = amp_new * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, Mf)[:n]
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            
            if a_new.sum() > 1e-12:
                a = a_new
                sc = _compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
    
    return best_a


def _build_inits(n, rng, parent):
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    inits.append(np.maximum(0.2 + 0.6 * np.cos(4.0 * np.pi * x), 0.01))

        sig += am * np.cos(2.0 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    if parent is not None and len(parent) == n:
