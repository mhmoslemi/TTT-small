import numpy as np
import time
import math

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute the cross-correlation of size n: sum_k w[k] * conv[a,w][k+n-1]"""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_active_weights(conv, top_k=10, eps_frac=0.01):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.03 * conv.size))
    if active.sum() > max_count:
        active = np.zeros(conv.size, dtype=bool)
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active[top_idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_grad_step(a, conv, w, eta):
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def lse_grad(a, beta):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (c - cmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = cmax + np.log(sum_e) / beta
    h = corr_pair(a, w)
    grad = 2.0 * n * 2.0 * h
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def lse_lscd(a, beta, deadline, max_iter=200, lr=0.01):
    """LSE Smoothed Coordinate Descent (LSCD) for the LSE-smoothed objective."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    # Precompute gradient
    L, grad = lse_grad(a, beta)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return best_sc, best_a
    
    # Adaptive step size
    step = lr / max(grad_norm, 1e-10)
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break
        
        # Projected gradient step
        new_a = a - step * grad
        new_a = np.clip(new_a, 0.0, None)
        S = new_a.sum()
        if S < 1e-12:
            break
        new_a = new_a / S
        
        new_sc = exact_score(new_a)
        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = new_a.copy()
            a = new_a
            # Recompute gradient
            L, grad = lse_grad(a, beta)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = lr / max(grad_norm, 1e-10)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    
    return best_sc, best_a


def run_lbfgs(a, beta, deadline, maxfun=500):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun_and_grad(x):
        sc, g = lse_grad(x, beta)
        # Add constraint handling: project back to non-negative and normalize
        x_clean = np.clip(x, 0.0, None)
        S = x_clean.sum()
        if S < 1e-12:
            return 1e10, np.zeros(n)
        return sc, g

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(50, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun_and_grad, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def generate_inits(n, rng, parent, cap=30):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        # Asymmetric perturbation of parent
        asym = pr * (1.0 + 0.3 * (2.0 * x - 1.0))
        add(asym)
        asym2 = pr * (1.0 - 0.3 * (2.0 * x - 1.0))
        add(asym2)

    add(np.ones(n))
    # Strong asymmetric shapes
    for alpha in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-alpha * x) + 0.01)
        add(np.exp(-alpha * (1.0 - x)) + 0.01)
    # Linear ramp
    add(x + 0.01)
    add((1.0 - x) + 0.01)
    # Quadratic asymmetric
    for alpha in (1.0, 2.0, 3.0):
        add((1.0 - alpha * x ** 2) + 0.01)
        add((1.0 - alpha * (1.0 - x) ** 2) + 0.01)
    # V-shapes (high at center)
    for alpha in (0.3, 0.5, 1.0, 2.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    # U-shapes (high at ends)
    for alpha in (0.3, 0.5, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    # Cosine
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    # Sine
    for f in (1, 2, 3):
        for alpha in (0.5, 1.0):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    # Power shapes
    for p in (0.5, 1.0, 1.5, 2.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    # Random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))
    # Heavy-tailed asymmetric
    for _ in range(3):
        add(np.abs(rng.standard_t(3, size=n)) + 0.1)

    return inits


def perturb(a, rng, strength=None):

    elif mode == 7:
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 9:
        # Asymmetric exponential push
        rate = rng.uniform(0.5, 3.0)
        q = p * (1.0 + strength * 0.5 * np.exp(-rate * x))
    elif mode == 10:
        # Quadratic asymmetric
        q = p * (1.0 + strength * 0.3 * (x - 0.5))
    else:
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def active_set_optimize(a, deadline, verbose=False, tag=""):
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    top_k = 5
    eps_frac = 0.03
    eta = 0.3
    stagnant = 0
    max_stagnant = 25

    for phase in range(30):
        if time.monotonic() > deadline - 3:
            break
        phase_dead = min(deadline - 3, t0 + (deadline - t0 - 3) * (phase + 1) / 30.0 + 2.0)

        for it in range(300):
            if time.monotonic() > phase_dead - 1.0:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))

            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            w = compute_active_weights(conv, top_k=top_k, eps_frac=eps_frac)
            
            etas = [eta * 0.3, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0]
            for eta_try in etas:
                if eta_try < 0.001 or eta_try > 5.0:
                    continue
                new_a = exp_grad_step(a, conv, w, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter - 1e-12:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            # Variance reduction style
            d_sorted = np.sort(conv)[::-1]
            top_m = min(20, len(conv))
            mean_top = float(np.mean(d_sorted[:top_m]))
            excess = np.maximum(conv - mean_top, 0.0)
            total_excess = float(np.sum(excess))
            if total_excess > 1e-15:
                w_vr = excess / total_excess
                for eta_try in [eta * 0.3, eta * 0.5, eta]:
                    new_a = exp_grad_step(a, conv, w_vr, eta_try)
                    new_a = normalize_a(new_a)
                    new_sc = exact_score(new_a)
                    if new_sc < best_sc_iter - 1e-12:
                        best_sc_iter = new_sc
                        best_a_iter = new_a

            if best_sc_iter < best_sc - 1e-11:
                best_sc = best_sc_iter
                best_a = best_a_iter
                a = best_a_iter
                eta = best_eta
                stagnant = 0
                if verbose and it % 100 == 0:
                    print("  [{} p{} i{}] score={:.10f} top_k={} eps={:.3f} eta={:.3f}".format(
                        tag, phase, it, best_sc, top_k, eps_frac, eta))
            else:
                stagnant += 1
                eta *= 0.8
                if stagnant >= max_stagnant:
                    break

        if stagnant >= max_stagnant:
            top_k = min(int(top_k * 1.5) + 2, 80)
            eps_frac = min(eps_frac * 1.4, 0.12)
            eta = max(0.05, eta * 0.6)
            stagnant = 0
            if verbose:
                print("  [{}] expand: top_k={}, eps={:.3f}".format(tag, top_k, eps_frac))
        else:
            top_k = max(3, int(top_k * 0.93))
            eps_frac = max(0.005, eps_frac * 0.93)

    return float(best_sc), best_a


def coordinate_polish(a, deadline, verbose=False, tag=""):
    """Coordinate-wise polish: try small adjustments to individual elements."""
    n = len(a)
    a = normalize_a(a)
    best_sc = exact_score(a)
    conv = fast_self_conv(a)
    cmax = float(np.max(conv))
    
    # Focus on positions that can most reduce the max
    # The gradient tells us which direction helps
    w = compute_active_weights(conv, top_k=10, eps_frac=0.01)
    grad = corr_pair(a, w)
    
    # Sort positions by gradient magnitude (most impactful first)
    order = np.argsort(-np.abs(grad))
    
    # Try adjusting top positions
    num_try = min(n, 500)
    improved = True
    rounds = 0
    max_rounds = 5
    
    while improved and time.monotonic() < deadline - 5 and rounds < max_rounds:
        improved = False
        rounds += 1
        conv = fast_self_conv(best_a if 'best_a' in dir() else a)
        a_current = a.copy()
        
        for idx in order[:num_try]:
            if time.monotonic() > deadline - 5:
                break
            orig = a_current[idx]
            if orig < 1e-10:
                continue
            # Try decreasing
            for frac in [0.5, 0.7, 0.85]:
                trial = a_current.copy()
                trial[idx] = orig * frac
                S = trial.sum()
                trial = trial / S
                sc = exact_score(trial)
                if sc < best_sc - 1e-11:
                    best_sc = sc
                    a_current = trial
                    improved = True
                    break
            if time.monotonic() > deadline - 5:
                break
            # Try increasing
            for frac in [1.3, 1.7, 2.0]:
                trial = a_current.copy()
                trial[idx] = orig * frac
                S = trial.sum()
                trial = trial / S
                sc = exact_score(trial)
                if sc < best_sc - 1e-11:
                    best_sc = sc
                    a_current = trial
                    improved = True
                    break
            if time.monotonic() > deadline - 5:
                break
        
        if improved:
            a = a_current
            if verbose:
                print("  [{} coord r{}] score={:.10f}".format(tag, rounds, best_sc))
            # Recompute gradient order
            conv = fast_self_conv(a)
            w = compute_active_weights(conv, top_k=10, eps_frac=0.01)
            grad = corr_pair(a, w)
            order = np.argsort(-np.abs(grad))
    
    best_a = a
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    # Phase 1: Multi-length exploration (15% of time)
    p1_dead = t_start + 0.15 * total_budget
    n_candidates = [4096, 6144, 8192, 10240, 12288, 16384]
    best_sc_all = np.inf
    best_a_all = None
    
    print("Phase 1: Multi-length exploration")
    for n_try in n_candidates:
        if time.monotonic() > p1_dead - 10:
            break
        
        # Generate inits for this length
        inits = generate_inits(n_try, rng, parent, cap=10)
        
        # Quick evaluation of inits
        best_init = None
        best_init_sc = np.inf
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = init
        
        if best_init is None:
            continue
        
        # Brief optimization
        chunk_dead = time.monotonic() + min(30, (p1_dead - time.monotonic()) * 0.8)
        sc, a = active_set_optimize(best_init, deadline=chunk_dead, verbose=False, tag="n{}".format(n_try))
        
        if sc < best_sc_all:
            best_sc_all = sc
            best_a_all = a.copy()
            print("  n={}: score={:.10f}".format(n_try, sc))
    
    if best_a_all is not None:
        n = len(best_a_all)
        a = best_a_all.copy()
        best_sc = best_sc_all
        best_a = best_a_all.copy()
        print("Phase 1 best: n={}, score={:.10f}".format(n, best_sc))
    else:
        n = 10240
        if parent is not None:
            a = resample_sequence(parent, n)
        else:
            a = np.ones(n)
        a = normalize_a(a)
        best_sc = exact_score(a)
        best_a = a.copy()
        print("Phase 1 fallback: n={}, score={:.10f}".format(n, best_sc))

    # Phase 2: Main active-set optimization (35% of time)
    p2_dead = t_start + 0.50 * total_budget
    print("Phase 2: Main active-set optimization")
    sc, a = active_set_optimize(a, deadline=p2_dead, verbose=True, tag="main")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        n = len(best_a)
        print("After Phase 2: {:.10f} (n={})".format(best_sc, n))

    # Phase 3: LSE-LBFGS refinement (20% of time)
    p3_dead = t_start + 0.70 * total_budget
    print("Phase 3: LSE-LBFGS refinement")
    a = normalize_a(best_a)
    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000, 50000]:
        if time.monotonic() > p3_dead - 15:
            break
        rem = p3_dead - time.monotonic()
        if rem < 25:
            break
        mf = max(80, int(rem * 0.6 / 0.1))
        mf = min(mf, 1500)
        sc, a = run_lbfgs(a, beta, p3_dead, mf)
        a = normalize_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  LSE beta={}: {:.10f}".format(beta, best_sc))
    
    # Also try LSCD
    if time.monotonic() < p3_dead - 10:
        for beta in [500, 2000, 10000]:
            if time.monotonic() > p3_dead - 10:
                break
            sc, a = lse_lscd(best_a, beta, p3_dead, max_iter=100, lr=0.005)
            a = normalize_a(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print("  LSCD beta={}: {:.10f}".format(beta, best_sc))

    # Phase 4: Coordinate polish (10% of time)
    p4_dead = t_start + 0.80 * total_budget
    print("Phase 4: Coordinate polish")
    sc, a = coordinate_polish(best_a, deadline=p4_dead, verbose=True, tag="coord")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 4: {:.10f}".format(best_sc))

    # Phase 5: Second active-set pass (10% of time)
    p5_dead = t_start + 0.90 * total_budget
    print("Phase 5: Second active-set pass")
    sc, a = active_set_optimize(best_a, deadline=p5_dead, verbose=False, tag="p5")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 5: {:.10f}".format(best_sc))

    # Phase 6: Diversification restarts (10% of time)
    p6_dead = deadline - 10.0
    print("Phase 6: Diversification")
    n_restarts = 5
    while time.monotonic() < p6_dead - 20 and n_restarts > 0:
        rem = p6_dead - 20.0 - time.monotonic()
        if rem < 30:
            break
        n_restarts -= 1
        p = perturb(best_a, rng, strength=float(rng.uniform(0.03, 0.2)))
        chunk_dead = time.monotonic() + rem * 0.75
        sc, seq = active_set_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(n_restarts))
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print("Restart {}: {:.10f}".format(n_restarts, best_sc))

    # Final polish
    if time.monotonic() < deadline - 15:
        sc, a = coordinate_polish(best_a, deadline=deadline - 5, verbose=False, tag="fin")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final verification
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
