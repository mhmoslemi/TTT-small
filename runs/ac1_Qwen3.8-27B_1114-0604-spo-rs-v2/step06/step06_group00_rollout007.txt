"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key improvements: spectral Fienup with non-negativity projection,
correct LSE gradient with high-beta annealing, water-filling peak shaving,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da_m = 2 * sum_k p_k * a_{k-m}
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def fienup_improved(a, n_iters=400, mode="hybrid"):
    """Improved Fienup with stronger flattening and non-negativity handling."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A)**2
        P_mean = float(P.mean())

        if mode == "flat":
            alpha = 0.2 + 0.6 * progress
            P_new = (1.0 - alpha) * P + alpha * P_mean
        elif mode == "quantile":
            q = 0.65 + 0.30 * progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        elif mode == "clipboost":
            alpha = 0.15 + 0.4 * progress
            thresh_high = P_mean * (1.5 - 0.5 * progress)
            thresh_low = P_mean * (0.5 + 0.3 * progress)
            P_new = P.copy()
            P_new[P_new > thresh_high] = thresh_high
            P_new[P_new < thresh_low] = thresh_low
        else:
            # hybrid
            if it % 3 == 0:
                alpha = 0.2 + 0.5 * progress
                P_new = (1.0 - alpha) * P + alpha * P_mean
            elif it % 3 == 1:
                q = 0.70 + 0.2 * progress
                thresh = np.quantile(P, q)
                P_new = P.copy()
                P_new[P_new > thresh] = thresh
            else:
                P_new = P.copy()
                P_new[P_new > P_mean * 1.3] = P_mean * 1.3
                P_new[P_new < P_mean * 0.4] = P_mean * 0.4

        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def water_filling_shave(a, n_iters=4000, n_peaks=5, step_init=0.005, decay=0.9998):
    """
    Water-filling peak shaving: simultaneously reduce all peaks above a threshold.
    Uses the subgradient of max(0, c_k - threshold) for multiple k.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - 2.0 * n / (2.0 * n - 1.0) < 1e-7:
            break

        # Water-filling: find peaks above a running threshold
        # Threshold starts at max and tightens
        threshold = M_val * (0.85 + 0.1 * min(1.0, it / (n_iters * 0.3)))
        peak_mask = c >= threshold
        peak_idx = np.where(peak_mask)[0]
        if len(peak_idx) == 0:
            peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Build subgradient targeting all peaks
        sg = np.zeros(n)
        for k in peak_idx[:n_peaks * 3]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg_a = a[j_start:j_end + 1]
                seg_rev = a[k - j_end:k - j_start + 1][::-1]
                # d c_k / d a_j = 2 * a_{k-j} for each j
                sg[j_start:j_end + 1] += 2.0 * seg_rev

        S2 = S * S
        S3 = S2 * S
        # Score gradient (approximate for multi-peak)
        n_peek = len(peak_idx[:n_peaks * 3])
        avg_c = float(c[peak_idx[:n_peaks * 3]].mean())
        grad = (2.0 * n / n_peek) * (sg / S2 - 2.0 * avg_c / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step_polyak = (sc - 2.0 * n / (2.0 * n - 1.0)) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.02)
        step = max(step, 1e-10)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def spectral_interp(a, new_n):
    """Interpolate via Fourier domain for better quality."""
    n = len(a)
    if n == new_n:
        return a.copy()
    M_old = 1 << (2 * n - 1).bit_length()
    M_new = 1 << (2 * new_n - 1).bit_length()
    A = np.fft.rfft(a, M_old)
    # Resample in frequency domain
    n_bins_old = len(A)
    n_bins_new = 1 + M_new // 2
    freq_old = np.linspace(0, 1, n_bins_old, endpoint=False)
    freq_new = np.linspace(0, 1, min(n_bins_new, n_bins_old), endpoint=False)
    A_new = np.interp(freq_new, freq_old, A.real) + 1j * np.interp(freq_new, freq_old, A.imag)
    # Pad to full length
    A_full = np.zeros(n_bins_new, dtype=complex)
    A_full[:len(A_new)] = A_new
    a_new = np.fft.irfft(A_full, M_new)[:new_n]
    return normalize(a_new)


def perturb_structured(a, rng, scale=0.02):
    """Structured perturbation with low-frequency components."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 8):
        freq = rng.integers(1, 30)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = scale / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_gaussian(a, rng, scale=0.015):
    """Gaussian perturbation."""
    a_new = a + scale * rng.standard_normal(len(a))
    return normalize(a_new)


def perturb_antipeak(a, rng, n_add=100):
    """Add mass away from peak contributors."""
    n = len(a)
    a_new = a.copy()
    c = fast_conv(a_new)
    k_peak = int(np.argmax(c))
    j_start = max(0, k_peak - n + 1)
    j_end = min(k_peak, n - 1)
    peak_set = np.ones(n, dtype=bool)
    peak_set[j_start:j_end + 1] = False
    non_peak_idx = np.where(peak_set)[0]
    if len(non_peak_idx) > n_add:
        add_idx = rng.choice(non_peak_idx, n_add, replace=False)
    else:
        add_idx = non_peak_idx
    delta = 0.003 / n
    a_new[add_idx] += delta
    return normalize(a_new)


def directed_descent(a, n_iters=3000):
    """
    Directed coordinate descent: iteratively reduce the element that most
    contributes to the current peak, with binary search for optimal value.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        pairs = a[js] * a[k_star - js]
        top_i = int(np.argmax(pairs))
        j_opt = int(js[top_i])
        partner = k_star - j_opt

        # Binary search: find best multiplier for a[j_opt]
        a_old = a[j_opt]
        if a_old < 1e-15:
            continue

        best_f = 1.0
        best_local = sc
        for f in [0.2, 0.35, 0.5, 0.65, 0.8, 0.9]:
            a[j_opt] = a_old * f
            a_n = normalize(a)
            sc_local = compute_score(a_n)
            if sc_local < best_local:
                best_local = sc_local
                best_f = f
        a[j_opt] = a_old * best_f
        a = normalize(a)

    return best_a

    if time.monotonic() < p4_end - 10:
        remaining_iters = int(min(4000, max(500, (p4_end - 10 - time.monotonic()) * 2000)))
        a_dd = directed_descent(best_a, n_iters=remaining_iters)
        sc = compute_score(a_dd)
        update(sc, a_dd, "DIRDESC")

        # LSE polish
        if sc < best_score and time.monotonic() < p4_end - 8:
            sc2, a2 = run_lbfgs(a_dd, 30000, maxiter=2000, maxfun=40000)
            update(sc2, a2, "DIRDESC+LSE")

    # === Phase 5: Multi-Restart (15%) ===
    p5_end = t_start + 0.75 * effective_budget
    print(f"\n=== Phase 5: Multi-Restart ===")

    n_restarts = 12
    for r in range(n_restarts):
        if time.monotonic() > p5_end - 20:
            break
        remaining = p5_end - 20 - time.monotonic()
        if remaining < 25:
            break

        # Diverse perturbation strategies
        r_mod = r % 4
        if r_mod == 0:
            a_pert = perturb_structured(best_a, rng, scale=0.01 + 0.004 * (r % 6))
        elif r_mod == 1:
            a_pert = perturb_gaussian(best_a, rng, scale=0.01 + 0.003 * (r % 4))
        elif r_mod == 2:
            a_pert = perturb_antipeak(best_a, rng, n_add=max(50, len(best_a) // 15))
        else:
            # Blend two good forms
            a_pert = 0.5 * best_a + 0.5 * fienup_improved(best_a, n_iters=50, mode="flat")
            a_pert = normalize(a_pert)

        beta = 1000 + r * 500
        maxiter = max(500, int(remaining * 5))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 18)
        update(sc, a_opt, f"RESTART {r}")

        # If improvement, quick Fienup follow-up
        if sc < best_score and time.monotonic() < p5_end - 15:
            a_f = fienup_improved(a_opt, n_iters=80, mode="hybrid")
            sc2 = compute_score(a_f)
            update(sc2, a_f, f"RESTART+FIENUP {r}")

    # === Phase 6: n-Growth (12%) ===
    p6_end = t_start + 0.87 * effective_budget
    print(f"\n=== Phase 6: n-Growth ===")

    current_n = len(best_a)
    growth_factors = [1.25, 1.5, 2.0]
    for gf in growth_factors:
        if time.monotonic() > p6_end - 30:
            break
        new_n = int(current_n * gf)
        if new_n <= current_n or new_n > 50000:
            continue
        remaining = p6_end - 30 - time.monotonic()
        if remaining < 30:
            break

        a_large = spectral_interp(best_a, new_n)
        sc = compute_score(a_large)
        print(f"  Spectral interp to n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        # LSE on larger
        if time.monotonic() < p6_end - 15:
            maxiter = max(400, int(min(3000, (p6_end - 15 - time.monotonic()) * 5)))
            sc2, a2 = run_lbfgs(a_large, 5000, maxiter=maxiter, maxfun=maxiter * 18)
            update(sc2, a2, f"INTERP+LSE n={new_n}")

            if sc2 < best_score and time.monotonic() < p6_end - 10:
                a_f = fienup_improved(a2, n_iters=100, mode="flat")
                sc3 = compute_score(a_f)
                update(sc3, a_f, f"INTERP+FIENUP n={new_n}")
                if sc3 < sc2:
                    current_n = new_n

    # === Phase 7: Final Polish (8%) ===
    print(f"\n=== Phase 7: Final Polish ===")

    # Very high beta LSE
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = max(1000, int(remaining * 7))
        sc, a_final = run_lbfgs(best_a, 50000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-50k")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = max(500, int(remaining * 5))
        sc, a_final = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 12)
        update(sc, a_final, "FINAL-LSE-200k")

    # Final water-filing
    if time.monotonic() < deadline - 12:
        remaining_iters = max(500, min(3000, int((deadline - 12 - time.monotonic()) * 3000)))
        a_ws = water_filling_shave(best_a, n_iters=remaining_iters, n_peaks=3,
                                    step_init=0.002, decay=0.9999)
        sc = compute_score(a_ws)
        update(sc, a_ws, "FINAL-WF")

    # Final directed descent
    if time.monotonic() < deadline - 5:
        a_dd = directed_descent(best_a, n_iters=1500)
        sc = compute_score(a_dd)
        update(sc, a_dd, "FINAL-DD")

    # Final tiny LSE
    if time.monotonic() < deadline - 3:
        sc, a_final = run_lbfgs(best_a, 500000, maxiter=1000, maxfun=15000)
        update(sc, a_final, "FINAL-MEGA")

    # === Output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
