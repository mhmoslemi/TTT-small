def fienup_annealed(a, n_iters=300):
    """Fienup/alternating projection with annealed flattening."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        alpha = 0.7 * (1.0 - progress) ** 2 + 0.05
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coord_descent_polish(a, n_sweeps=30, deadline=None):
    """Coordinate descent: for each element, try several multipliers."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in multipliers:
                a[j] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_tmp = a / S_tmp
                sc = compute_score(a_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        if compute_score(a) < best_sc:
            best_sc = compute_score(a)
            best_a = a.copy()

    return best_a

    deadline = t_start + total_budget - 90.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform n=2048")

    print(f"Initial best: {best_score:.8f}")

    # ============================================================
    # Phase 1: Fienup + quick L-BFGS (10% of time)
    # ============================================================
    p1_end = t_start + 0.10 * (total_budget - 90.0)
    print(f"\n=== Phase 1: Init + Fienup (deadline {p1_end - t_start:.0f}s) ===")

    a_f = fienup_annealed(best_a, n_iters=200)
    update(compute_score(a_f), a_f, "FIENUP")

    if time.monotonic() < p1_end:
        sc, a_opt = run_lbfgs(best_a, 500, maxiter=5000, maxfun=100000)
        update(sc, a_opt, "LBFGS-500")

    # ============================================================
    # Phase 2: LP Iteration (50% of time) - MAIN PHASE
    # ============================================================
    p2_end = t_start + 0.60 * (total_budget - 90.0)
    print(f"\n=== Phase 2: LP Iteration (deadline {p2_end - t_start:.0f}s) ===")

    t_lp0 = time.monotonic()
    a_lp = lp_iteration(best_a, n_iters=500, deadline=p2_end - 30, verbose=True)
    sc_lp = compute_score(a_lp)
    update(sc_lp, a_lp, "LP-ITER")
    print(f"  LP phase took {time.monotonic() - t_lp0:.1f}s")

    # ============================================================
    # Phase 3: High-beta L-BFGS (20% of time)
    # ============================================================
    p3_end = t_start + 0.80 * (total_budget - 90.0)
    print(f"\n=== Phase 3: High-beta L-BFGS (deadline {p3_end - t_start:.0f}s) ===")

    betas = [2000, 10000, 50000, 100000, 200000]
    for beta in betas:
        if time.monotonic() > p3_end - 60:
            break
        remaining = p3_end - 60 - time.monotonic()
        if remaining < 30:
            break
        maxiter = int(min(15000, remaining * 10 + 2000))
        maxiter = max(maxiter, 3000)
        print(f"  L-BFGS beta={beta}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
