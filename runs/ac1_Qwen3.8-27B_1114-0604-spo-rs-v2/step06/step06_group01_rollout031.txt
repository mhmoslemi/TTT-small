"""
LP-driven optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses exact LP (via cvxpy) for direction finding, LSE-smoothed L-BFGS for polish,
and coordinate descent for fine-tuning. Multi-length exploration with time budget.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d
import scipy.sparse as sp
import cvxpy as cp
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_toeplitz_row(a, k):
    """Build row k of the Toeplitz matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_end >= j_start:
        row[j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
    return row


def lp_direction_reduced(a, n_top=50, solver='GLPK'):
    """
    Find direction b >= 0, sum(b)=1, minimizing max conv(a,b)_k 
    using a reduced constraint set (top n_top positions).
    """
    n = len(a)
    # Compute current cross-conv to identify active constraints
    # Use a as initial guess for active set
    c_aa = self_conv(a)
    top_indices = np.argsort(c_aa)[::-1][:n_top]
    
    # Build reduced constraint matrix
    rows = []
    for k in top_indices:
        rows.append(build_toeplitz_row(a, k))
    T_red = np.array(rows)  # shape (n_top, n)
    
    # Solve LP: min t s.t. T_red @ b <= t, 1^T b = 1, b >= 0
    b_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [T_red @ b_var <= t_var, cp.sum(b_var) == 1, b_var >= 0]
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            prob.solve(solver='CBC', verbose=False)
        except:
            prob.solve(solver='ECOS', verbose=False)
    
    if b_var.value is None:
        return None, None
    
    b_opt = np.maximum(b_var.value, 0.0)
    S = b_opt.sum()
    if S < 1e-12:
        return None, None
    b_opt = b_opt / S
    
    # Verify: check full conv(a, b_opt)
    c_ab = cross_conv(a, b_opt)
    actual_max = float(c_ab.max())
    
    return b_opt, actual_max

            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break
        
        a = a_new
    
    return best_a, best_sc


def coordinate_descent(a, n_steps=2000, rng=None, time_limit=None):
    """
    Coordinate descent: try pairwise transfers to reduce the peak.
    For each pair (i, j), find the optimal amount to transfer from i to j.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Precompute which positions are "hot" (contribute to peak)
    c = self_conv(a)
    k_star = int(np.argmax(c))
    
    for step in range(n_steps):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        # Bias towards positions that contribute to the peak
        i = rng.integers(n)
        # j is chosen to be "far" from i to reduce peak contribution
        # Specifically, if k_star - i is in range, we want to move mass AWAY from positions that pair with i at lag k_star
        j_candidates = list(rng.integers(n, size=20))
        
        best_pair_sc = best_sc
        best_delta = 0
        best_j = -1

            if j == i:
                continue

            # a[i] decreases, a[j] increases

            # We want this to be negative (reduce peak)
            a_ki = a[k_star - i] if 0 <= k_star - i < n else 0.0
            a_kj = a[k_star - j] if 0 <= k_star - j
