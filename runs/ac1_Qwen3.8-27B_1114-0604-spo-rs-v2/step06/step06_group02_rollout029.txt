"""
Novel optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: corrected gradient (factor-of-2 fix),
FISTA acceleration, frequency-domain basin hopping, multiscale refinement,
and exact subgradient polishing.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_f_and_grad(a, beta):
    """
    Correct LSE-smoothed objective and gradient.
    f(a) = 2n * L / S^2
    grad_j = 2n * (2*h_j / S^2 - 2*L / S^3) = 4n*(h_j/S^2 - L/S^3)
    where h_j = sum_i a_i * p_{j+i}, p = softmax(beta*(c - c_max))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # Compute h_j = sum_i a_i * p_{j+i} via FFT cross-correlation
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    # h_j = sum_i a_i * p_{j+i} = cross-correlation of a and p at lag j
    # = ifft(fft(a) * conj(fft(p))) for j=0,...,n-1
    corr = np.fft.irfft(A_fft * np.conj(P_fft), M2_fft)
    h = corr[:n]
    S2 = S * S
    S3 = S2 * S
    # Correct gradient: 4n * (h/S^2 - L/S^3)
    grad = 4.0 * n * (h / S2 - L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def run_lbfgs_correct(a0, beta, maxiter=20000, maxfun=400000):
    """L-BFGS-B with corrected gradient."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S * 1.0  # Keep on same scale, L-BFGS handles it
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_f_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def fista_lse(a0, beta, n_iters=5000, L_init=None):
    """
    FISTA (accelerated proximal gradient) for LSE-smoothed objective.
    Handles non-negativity constraint via projection.
    """
    n = len(a0)
    x = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = x.sum()
    if S < 1e-6:
        x = np.ones(n) / n
    xz = x.copy()
    t = 1.0

    # Estimate Lipschitz constant
    if L_init is None:
        _, g0 = lse_f_and_grad(x, beta)
        L = max(np.linalg.norm(g0) * 2, 1.0)
    else:
        L = L_init

    best_x = x.copy()
    best_sc = compute_score(x)

    for k in range(n_iters):
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - xz)
        y = np.maximum(y, 0.0)

        f_y, grad_y = lse_f_and_grad(y, beta)

        # Backtracking line search
        step = 1.0 / L
        for _ in range(20):
            x_new = np.maximum(y - step * grad_y, 0.0)
            f_new, _ = lse_f_and_grad(x_new, beta)
            # Quadratic model check
            model = f_y + grad_y @ (x_new - y) + 0.5 * L * np.sum((x_new - y) ** 2)
            if f_new <= model + 1e-12:
                break
            step *= 0.5
        else:
            x_new = np.maximum(y - step * grad_y, 0.0)

        xz = x
        x = x_new
        t = t_new

        sc = compute_score(x)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()

        # Adaptive L: if we used full step, try increasing L
        if step == 1.0 / L:
            L *= 0.9  # Try to be more aggressive
        # Cap L to avoid too-small steps
        L = max(L, 0.1)

    return best_sc, best_x


def exact_subgrad_step(a, lr=0.01, n_steps=500):
    """
    Subgradient method on the exact (non-smooth) max objective.
    Uses Nesterov acceleration.
    """
    n = len(a)
    x = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    S = x.sum()
    if S < 1e-6:
        return compute_score(x), x
    x = x / S
    xz = x.copy()
    t = 1.0
    best_x = x.copy()
    best_sc = compute_score(x)

    for k in range(n_steps):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - xz)
        y = np.maximum(y, 0.0)
        S_y = y.sum()
        if S_y < 1e-12:
            break
        y = y / S_y

        # Compute exact subgradient
        c = fast_conv(y)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        S_y = float(y.sum())
        # grad_j = 4n*(2*a_{k*-j}/S^2 - 2*M/S^3) for valid j, else -4nM/S^3
        grad = np.full(n, -4.0 * n * M_val / (S_y ** 3))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        for j in range(j_start, j_end + 1):
            grad[j] += 8.0 * n * y[k_star - j] / (S_y ** 2)

        # Adaptive step size
        lr_k = lr * (2.0 / (t_new + 3.0))
        x_new = np.maximum(y - lr_k * grad, 0.0)
        S_new = x_new.sum()
        if S_new < 1e-12:
            break
        x_new = x_new / S_new

        xz = x
        x = x_new
        t = t_new

        sc = compute_score(x)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()

    return best_sc, best_x


def fienup_corrected(a0, n_iters=300, target_flat=True):
    """
    Fienup/GER with corrected non-negativity handling.
    Alternates between:
    - Time: clip to non-negative, normalize
    - Frequency: flatten power spectrum
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if target_flat:
            # Annealed flattening
            alpha = 0.7 * (1.0 - progress) ** 2 + 0.05
            P_target = P_mean * np.ones_like(P)
            # Boost DC to help non-negativity
            P_target[0] *= (1.0 + 0.5 * (1.0 - progress))
        else:
            alpha = 0.3
            P_target = P  # No modification

        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * np.sqrt(P_blend / (P + 1e-30)) ** alpha
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def spectral_basin_hop(a0, rng, n_hops=20, deadline=None):
    """
    Basin hopping in frequency domain:
    1. Randomize phase of DFT
    2. Take inverse FFT, clip to non-negative
    3. Quick L-BFGS to polish
    """
    n = len(a0)
    best_a = a0.copy()
    best_sc = compute_score(a0)
    M_fft = 1 << (2 * n - 1).bit_length()

    for hop in range(n_hops):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a0, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Random phase perturbation
        strength = rng.uniform(0.1, 0.8)
        noise_phase = np.exp(1j * rng.standard_normal(len(phase)) * strength)
        A_new = mag * phase * noise_phase
        # Keep DC positive
        A_new[0] = np.abs(A_new[0])

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S

        # Quick L-BFGS polish
        sc, a_opt = run_lbfgs_correct(a_new, 2000, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Spectral hop {hop}: {sc:.8f}")

    return best_sc, best_a


def time_domain_basin_hop(a0, rng, n_hops=20, deadline=None):
    """
    Basin hopping with structured time-domain perturbations.
    """
    n = len(a0)
    best_a = a0.copy()
    best_sc = compute_score(a0)

    for hop in range(n_hops):
        if deadline is not None and time.monotonic() > deadline:
            break

        strategy = hop % 5
        if strategy == 0:
            # Gaussian noise
            a_pert = a0 * (1.0 + 0.02 * rng.standard_normal(n))
        elif strategy == 1:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a0 * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 2:
            # Window function modulation
            x = np.arange(n) / n
            w = 1.0 + 0.1 * rng.uniform(-1, 1) * np.cos(np.pi * x)
            a_pert = a0 * w
        elif strategy == 3:
            # Block perturbation
            block = max(1, n // 16)
            start = rng.integers(0, n - block)
            a_pert = a0.copy()
            a_pert[start:start + block] *= (1.0 + 0.15 * rng.standard_normal(block))
        else:
            # Swap perturbation
            a_pert = a0.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            tmp = a_pert[i].copy()
            a_pert[i] = a_pert[j]
            a_pert[j] = tmp

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S

        sc, a_opt = run_lbfgs_correct(a_pert, 3000, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Time-domain hop {hop} (strat={strategy}): {sc:.8f}")

    return best_sc, best_a


def multi_length_search(a0, candidate_lengths, deadline):
    """
    Try optimizing at different sequence lengths by interpolation.
    """
    n_orig = len(a0)
    best_a = a0.copy()
    best_sc = compute_score(a0)

    for target_n in candidate_lengths:
        if time.monotonic() > deadline:
            break
        if target_n == n_orig:
            continue
        if target_n < 64:
            continue

        # Interpolate
        x_old = np.linspace(0, 1, n_orig)
        x_new = np.linspace(0, 1, target_n)
        a_interp = np.interp(x_new, x_old, a0)
        a_interp = np.maximum(a_interp, 0.0)
        S = a_interp.sum()
        if S < 1e-12:
            continue
        a_interp = a_interp / S

        # Optimize
        sc, a_opt = run_lbfgs_correct(a_interp, 3000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Multi-length n={target_n}: {sc:.8f}")
        else:
            # Try FISTA too
            sc2, a_fista = fista_lse(a_interp, 2000, n_iters=3000)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_fista.copy()
                print(f"    Multi-length n={target_n} (FISTA): {sc2:.8f}")

    return best_sc, best_a


def adaptive_beta_schedule(a0, deadline):
    """
    L-BFGS with increasing beta schedule using corrected gradient.
    Allocates time adaptively based on remaining budget.
    """
    best_a = a0.copy()
    best_sc = compute_score(a0)

    # Start lower, go higher
    betas = [50, 100, 300, 800, 2000, 5000, 10000, 25000, 50000, 100000, 250000]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # More iterations for higher beta (harder problems)
        maxiter = int(min(30000, max(2000, remaining * 15 * (1 + i * 0.3))))
        maxfun = maxiter * 25

        print(f"    Corrected L-BFGS beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s)")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_correct(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"      score={sc:.8f} ({dt:.1f}s)")

        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_sc, best_a


def fista_with_annealing(a0, n_iters=5000, beta_init=500, beta_end=50000):
    """FISTA with annealed beta: start smooth, get sharper."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a = np.ones(n) / n

    x = a.copy()
    xz = x.copy()
    t = 1.0
    best_x = x.copy()
    best_sc = compute_score(x)
    L = 1.0

    n_beta_changes = 5
    iters_per_beta = n_iters // n_beta_changes

    for b_idx in range(n_beta_changes):
        beta = beta_init * (beta_end / beta_init) ** (b_idx / max(1, n_beta_changes - 1))
        for k in range(iters_per_beta):
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x + ((t - 1.0) / t_new) * (x - xz)
            y = np.maximum(y, 0.0)

            f_y, grad_y = lse_f_and_grad(y, beta)

            step = 1.0 / L
            for _ in range(15):
                x_new = np.maximum(y - step * grad_y, 0.0)
                f_new, _ = lse_f_and_grad(x_new, beta)
                model = f_y + grad_y @ (x_new - y) + 0.5 * L * np.sum((x_new - y) ** 2)
                if f_new <= model + 1e-12:
                    break
                step *= 0.5
            else:
                x_new = np.maximum(y - step * grad_y, 0.0)

            xz = x
            x = x_new
            t = t_new

            if k % 100 == 0:
                sc = compute_score(x)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x.copy()

        # Reset momentum at beta changes
        xz = x.copy()
        t = 1.0
        L = 1.0

    sc = compute_score(x)
    if sc < best_sc:
        best_sc = sc
        best_x = x.copy()

    return best_sc, best_x


def polish_exact(a0, n_steps=2000):
    """Final polish with exact subgradient + Nesterov."""
    n = len(a0)
    best_a, best_sc = exact_subgrad_step(a0, lr=0.5, n_steps=n_steps)
    return best_sc, best_a


def smoothed_fienup_extra(a0, n_iters=500):
    """
    Enhanced Fienup with adaptive target and repair.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target: flat with DC boost proportional to non-negativity violation
        neg_count = np.mean(A.real < 0) if n > 2 else 0
        dc_boost = 1.0 + 0.3 * (1.0 - progress)

        P_target = np.full_like(P, P_mean)
        P_target[0] *= dc_boost
        # Slight roll-off at high frequencies to help non-negativity
        freqs = np.arange(len(P)) / len(P)
        P_target *= (1.0 - 0.2 * freqs)

        alpha = 0.6 * (1.0 - progress) ** 1.5 + 0.03
        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        # Gaussian repair
        if n > 16:
            sigma = max(1.0, n / 50.0)
            kernel_size = int(6 * sigma) | 1
            kernel = np.exp(-0.5 * (np.arange(-kernel_size // 2, kernel_size // 2 + 1) / sigma) ** 2)
            kernel /= kernel.sum()
            a_smooth = np.convolve(a_new, kernel, mode='same')
            repair = 0.1 * (1.0 - progress)
            a_new = (1 - repair) * a_new + repair * a_smooth
            a_new = np.maximum(a_new, 0.0)

        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase with corrected gradient.
    Phases:
      1. Fienup initialization (5%)
      2. Corrected L-BFGS adaptive beta (35%)
      3. FISTA with annealing (15%)
      4. Basin hopping (25%)
      5. Multi-length + exact polish (20%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048)

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform n=2048")

    print(f"Initial best: {best_score:.8f}")
    n = len(best_a)

    # ============================================================
    # Phase 1: Fienup initialization (5% of time)
    # ============================================================
    p1_end = t_start + 0.05 * (total_budget - 60.0)
    print(f"\n=== Phase 1: Fienup (until t={p1_end - t_start:.0f}s) ===")

    a_f = fienup_corrected(best_a, n_iters=300, target_flat=True)
    sc = compute_score(a_f)
    update(sc, a_f, "FIENUP-flat")

    if time.monotonic() < p1_end:
        a_f2 = smoothed_fienup_extra(best_a, n_iters=400)
        sc2 = compute_score(a_f2)
        update(sc2, a_f2, "FIENUP-extra")

    # ============================================================
    # Phase 2: Corrected L-BFGS with adaptive beta (35%)
    # ============================================================
    p2_end = t_start + 0.40 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Corrected L-BFGS (until t={p2_end - t_start:.0f}s) ===")

    sc_lb, a_lb = adaptive_beta_schedule(best_a, p2_end)
    update(sc_lb, a_lb, "ADAPTIVE-LBFGS")

    # ============================================================
    # Phase 3: FISTA with annealing (15%)
    # ============================================================
    p3_end = t_start + 0.55 * (total_budget - 60.0)
    print(f"\n=== Phase 3: FISTA annealed (until t={p3_end - t_start:.0f}s) ===")

    if time.monotonic() < p3_end:
        remaining_iters = int((p3_end - time.monotonic()) * 30)  # estimate iterations
        remaining_iters = max(2000, min(10000, remaining_iters))
        sc_f, a_fista = fista_with_annealing(best_a, n_iters=remaining_iters,
                                              beta_init=200, beta_end=50000)
        update(sc_f, a_fista, "FISTA-annealed")

    # ============================================================
    # Phase 4: Basin hopping (25%)
    # ============================================================
    p4_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Basin hopping (until t={p4_end - t_start:.0f}s) ===")

    if time.monotonic() < p4_end:
        n_hops = int((p4_end - time.monotonic()) // 30)  # ~30s per hop
        n_hops = max(5, min(25, n_hops))
        sc_hop, a_hop = spectral_basin_hop(best_a, rng, n_hops=n_hops, deadline=p4_end)
        update(sc_hop, a_hop, "SPECTRAL-HOP")

    if time.monotonic() < p4_end:
        n_hops2 = int((p4_end - time.monotonic()) // 30)
        n_hops2 = max(3, min(15, n_hops2))
        sc_hop2, a_hop2 = time_domain_basin_hop(best_a, rng, n_hops=n_hops2, deadline=p4_end)
        update(sc_hop2, a_hop2, "TIME-HOP")

    # ============================================================
    # Phase 5: Multi-length + exact polish (20%)
    # ============================================================
    p5_end = deadline
    print(f"\n=== Phase 5: Multi-length + polish (until t={p5_end - t_start:.0f}s) ===")

    # Try a few different lengths
    if time.monotonic() < p5_end - 120:
        current_n = len(best_a)
        candidate_lengths = []
        for frac in [0.5, 1.0, 2.0]:
            target_n = int(current_n * frac)
            target_n = max(128, min(16384, target_n))
            target_n = 1 << (target_n - 1).bit_length()  # power of 2
            if target_n != current_n and target_n >= 128:
                candidate_lengths.append(target_n)
        # Deduplicate
        candidate_lengths = sorted(set(candidate_lengths))
        if candidate_lengths:
            sc_ml, a_ml = multi_length_search(best_a, candidate_lengths, p5_end - 60)
            update(sc_ml, a_ml, "MULTI-LEN")

    # Final L-BFGS polish with high beta
    if time.monotonic() < p5_end - 30:
        remaining = p5_end - 30 - time.monotonic()
        maxiter = int(min(20000, remaining * 15))
        if maxiter > 1000:
            sc, a_fin = run_lbfgs_correct(best_a, 200000, maxiter=maxiter,
                                           maxfun=maxiter * 20)
            update(sc, a_fin, "FINAL-LBFGS-200k")

    # Exact subgradient polish
    if time.monotonic() < p5_end - 10:
        n_steps = int((p5_end - 10 - time.monotonic()) * 50)
        n_steps = max(500, min(5000, n_steps))
        sc_ex, a_ex = polish_exact(best_a, n_steps=n_steps)
        update(sc_ex, a_ex, "EXACT-SUBGRAD")

    # Very final L-BFGS
    if time.monotonic() < p5_end - 5:
        sc, a_fin2 = run_lbfgs_correct(best_a, 500000, maxiter=5000, maxfun=100000)
        update(sc, a_fin2, "ULTIMATE-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
