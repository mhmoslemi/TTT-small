def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    for x in sequence:

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):

    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):

    x_new = np.linspace(0.0, 1.0, n)

def _lse_value_grad(a, beta):
    """LSE-smoothed 2*max(n*c/S^2) and its gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    # Gradient: dL/da[i] = 2n/S^2 * R[i] - 2n*T/S^3
    # R[i] = sum_j a[j] * w[i+j]  (w has length 2n-1)
    # R[i] = convolve(a_rev, w)[i+n-1]
    T = float(w.dot(c))

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """
    Compute exact subgradient of 2*n*max(c)/S^2 at point a.
    Uses top-k peaks within tol of the maximum.
    Returns (score, gradient_vector) where gradient is for the function 2*n*max(c)/S^2.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    # Find active set: peaks within tol*dmax of the maximum
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    # Limit to top k
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    # Average weight on active peaks
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    # R[i] = sum_k w_active[k] * a[k-i]  (a extended by 0)
    # Using the same convolution trick: R[i] = conv(a_rev, w_active)[i+n-1]
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """
    Nesterov accelerated subgradient method for minimizing 2*n*max(c)/S^2.
    a is maintained on the simplex (non-negative, sum=1).
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    # Nesterov parameters
    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    # Initial step size estimate
    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a

    # Scale step: we want the step to move ~1% of the range
    eta = 0.05 / g_norm

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * v

        # Evaluate subgradient at y
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break

        # Gradient step
        step = eta / (1.0 + 0.1 * it)
        x_new = y_norm - step * grad_y

        # Project to simplex (approximate: clip and normalize)
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        # Update momentum
        v = x_new - a
        a = x_new
        t = t_new

        # Track best
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive step size: if no improvement in last 50 iters, reduce eta
        if it > 0 and it % 50 == 0:
            if not (sc < best_score + 1e-10):
                eta *= 0.7

    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    if maxfun is None:
        maxfun = 500

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        # Chain rule for normalization: x = xn * S, so we optimize over unnormalized
        # Actually simpler: just optimize normalized directly with bounds
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out

def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        # Multiplicative log-normal noise
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.05, 0.25))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Subgradient-based push
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.3 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth noise (low frequency)
        K = min(31, max(5, n // 200))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.5)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 4:
        # V-shape modulation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)
        q = a * vshape
    else:
        # Coordinate swap perturbation
        q = a.copy()
        n_swaps = int(rng.integers(3, max(4, n // 100)))
        for _ in range(n_swaps):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = a * np.exp(0.05 * rng.standard_normal(n))

    return _normalize(np.clip(q, 0.0, None))

def _generate_structured_inits(n, rng, parent, cap=25):
    """Generate diverse initial sequences."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Uniform
    _add(np.ones(n))
    # V-shapes
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    # Inverted V
    for alpha in [0.3, 0.6, 1.0]:
        _add(1.0 + alpha * (1.0 - np.abs(2.0 * x - 1.0) ** 2))
    # Cosine
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 1.0]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
            _add(1.0 + amp * np.cos(np.pi * f * x))
    # Exponential
    for rate in [1.0, 2.0, 4.0, 8.0]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1.0 - x)) + 0.01)
        _add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    # Bimodal
    for w in [0.1, 0.15, 0.25, 0.4]:
        for ctr in [0.2, 0.3, 0.35, 0.4]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)
    # Sine absolute
    for f in [1, 2, 3, 4]:
        _add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    return inits
