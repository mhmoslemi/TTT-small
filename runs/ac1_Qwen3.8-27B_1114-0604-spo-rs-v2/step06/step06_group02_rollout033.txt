"""
Multi-phase optimizer with novel peak-equalization and Nesterov-accelerated
subgradient methods for minimizing 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

def grad_of_lag(a, k):
    """Gradient of c_k = sum_j a_j * a[k-j] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_end >= j_start:
        js = np.arange(j_start, j_end + 1)
        g[js] = a[k - js]
    return g

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via FFT
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

    )
    a_out = normalize(np.clip(res.x, 0.0, 1000.0))

            k = int(k)

            grad += wk * g

"""
Multi-phase optimizer combining Nesterov-accelerated subgradient, peak equalization,
multi-scale LSE-LBFGS, and peak-aware restarts for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: harder max approximation, smarter time allocation, diverse restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a0 = normalize(a0)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        a_out = normalize(np.clip(res.x, 0.0, 1000.0))
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def nesterov_subgradient(a, n_iters=5000, top_k=5, deadline=None):
    n = len(a)
    a = normalize(a.copy())
    v = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr_scale = 0.1

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break

        t_curr = 1.0 + np.sqrt(1.0 + 4.0 * (it + 1)) / 2.0
        t_prev = 1.0 + np.sqrt(1.0 + 4.0 * it) / 2.0 if it > 0 else 1.0
        momentum = (t_prev - 1.0) / t_curr
        y = a + momentum * (a - v)
        v = a.copy()
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy < 1e-12:
            y = a.copy()
            Sy = a.sum()

        c = fast_conv(y)
        c_max = float(c.max())
        S2 = Sy * Sy
        sc = 2.0 * n * c_max / S2

        k_eff = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_eff]
        top_vals = c[top_idx]
        w_scaled = 20.0 * (top_vals - top_vals[0])
        weights = np.exp(w_scaled)
        weights /= weights.sum()

        c_max_smooth = float(np.dot(weights, top_vals))
        S3 = S2 * Sy
        S2_inv = 1.0 / S2
        S3_inv = 1.0 / S3

        direction_mem = weights[0]
        for wi, ki in zip(weights[1:], top_idx[1:]):
            direction_mem += wi

        k0 = int(top_idx[0])
        j_lo = max(0, k0 - n + 1)
        j_hi = min(k0, n - 1)
        if j_hi >= j_lo:
            direction_mem *= 0.0
            direction_mem[:] = 0.0
            for wi, ki in zip(weights, top_idx):
                ki = int(ki)
                los = max(0, ki - n + 1)
                his = min(ki, n - 1)
                if his >= los:
                    js = np.arange(los, his + 1)
                    direction_mem[js] += wi * y[ki - js]

        grad = 2.0 * n * (direction_mem * S2_inv - 2.0 * c_max_smooth * S3_inv)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        lr = lr_scale / (gnorm * np.sqrt(1.0 + it * 0.001))
        lr = min(lr, 0.5)

        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0.0)
        Sn = a_new.sum()
        if Sn < 1e-12:
            break
        a_new = a_new / Sn

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return best_sc, best_a

        c_top_min = top_vals[-1]
        spread = c_top_max - c_top_min

        if spread < 1e-12 * c_max:
            a = a * (1.0 + 0.002 * rng.standard_normal(n))
            a = normalize(a)
            continue

        k_max = int(top_idx[0])
        k_min = int(top_idx[-1])
        S = float(a.sum())

        g_max = np.zeros(n)
        j_lo = max(0, k_max - n + 1)
        j_hi = min(k_max, n - 1)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            g_max[js] = a[k_max - js]

        g_min = np.zeros(n)
        j_lo2 = max(0, k_min - n + 1)

        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
