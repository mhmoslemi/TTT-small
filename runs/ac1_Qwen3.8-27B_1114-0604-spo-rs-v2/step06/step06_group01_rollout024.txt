import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def cross_corr(a, p):
    """Compute (B^T p)[j] = sum_i a[i]*p[i+j] via FFT for j=0..n-1."""
    n = len(a)
    n_c = len(p)
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)
    FP = np.fft.rfft(p, M_fft)
    corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
    return corr


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def top_k_lse(a, K, beta):
    """Compute LSE of top-K convolution values and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    # Find top K indices
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))

    c_top = c[top_idx]
    c_top_max = float(c_top.max())
    z = beta * (c_top - c_top_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_top_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)

    # Gradient: only top-K components contribute
    p_full = np.zeros(len(c))
    p_full[top_idx] = exp_z / exp_z.sum()

    # Gradient of L w.r.t. a
    grad_L = 2.0 * cross_corr(a, p_full)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

    n = len(a)
    c = self_conv(a)
    S = float(a.sum())

    c_max = float(c.max())

    grad_L = 2.0 * cross_corr(a, p)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)

def fista_lp(a, M, n_iters, beta, x_init=None, use_backtracking=True):
    """
    Solve: min (1/beta)*log(sum exp(beta*(conv(a,x)-M))) + (0) s.t. x >= 0

    """
    n = len(a)
    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()

    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
        Sx = x.sum()
        if Sx > 0:
            x = x / Sx
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    # Initial step size
    sum_a2 = float(np.sum(a * a))
    L = max(beta * n * sum_a2 * 4.0, 1.0)
    step = 1.0 / L

    # Precompute FFT of a
    FA = np.fft.rfft(a, M_fft)

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr * beta

        # Backtracking line search
        if use_backtracking:
            f_x = float(np.log(exp_sum) + z_max - np.log(1.0))  # approximate
            v = u - step * grad
            x_new = np.maximum(v, 0.0)
            # Check sufficient decrease
            c_new = cross_conv(a, x_new)
            z_new = beta * (c_new - M)
            z_new_max = float(z_new.max())
            exp_new = np.exp(z_new - z_new_max)
            f_new = float(np.log(exp_new.sum()) + z_new_max)
            f_old = float(np.log(exp_sum) + z_max)

            allowance = f_old + 0.5 * step * float(np.sum(grad * (x_new - u)))
            if f_new > allowance + 1e-10:
                for _ in range(20):
                    step *= 0.5
                    v = u - step * grad
                    x_new = np.maximum(v, 0.0)
                    c_new = cross_conv(a, x_new)
                    z_new = beta * (c_new - M)
                    z_new_max = float(z_new.max())
                    exp_new = np.exp(z_new - z_new_max)

    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    x_warm = None

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        # Adaptive beta: increase over iterations
        frac = outer / max(1, max_outer_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        c = self_conv(a)
        M = float(c.max())

        x = fista_lp(a, M, n_iters=fista_iters, beta=beta,
                     x_init=x_warm, use_backtracking=use_backtracking)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Adaptive line search for mixing parameter
        t_candidates = [0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 1.0]
        best_t = 0.3
        best_t_sc = best_sc
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                break
        a = a_new

    return best_a, best_sc, x_warm


def error_reduction(a, n_iters=80, alpha=0.7):
    """Newton-like error reduction in frequency domain to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.2 + 0.8 * (it / max(1, n_iters // 2))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving_topk(a, n_steps=300, K=5, beta=500.0, rng=None):
    """Targeted local search using top-K LSE to reduce the convolution peaks."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        score, grad = top_k_lse(a, K, beta)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        base_step = min(0.03, max(0.0005, 1.0 / (gn * gn * n * 0.1 + 1e-10)))
        improved = False
        for mult in [0.1, 0.25, 0.5, 1.0, 1.5, 2.5]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Try smaller K for more focused gradient
            if K > 1:
                K = max(1, K // 2)
            # Increase beta for sharper peak focusing
            beta = min(beta * 1.5, 10000.0)
            # Random perturbation to escape
            a = a * (1.0 + 0.005 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def lbfgs_run(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-13, use_topk=False, K=5):
    """Run L-BFGS-B on the smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        if use_topk:
            return top_k_lse(x, K, beta)
        else:
            return full_lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def multistage_error_reduction(a, stages=[(40, 0.5), (60, 0.65), (80, 0.8)]):
    """Multi-stage error reduction with increasing aggressiveness."""
    best_a = a.copy()
    best_sc = compute_score(a)
    for n_it, alpha in stages:
        a = error_reduction(best_a, n_iters=n_it, alpha=alpha)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(interpolate_sequence(parent, n))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 4, 0.01, 1000.0))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x + 0.5)
                + 0.1 * np.cos(8.0 * np.pi * x + 1.0))
    inits.append(np.clip(0.3 + 0.3 * x + 0.2 * np.cos(2.0 * np.pi * 3 * x), 0.01, 1000.0))
    # Exponential decay shapes
    inits.append(np.clip(np.exp(-2.0 * x) + 0.2, 0.01, 1000.0))
    inits.append(np.clip(0.5 * np.exp(-3.0 * x * (1-x)) + 0.3, 0.01, 1000.0))
    # Sine power
    inits.append(np.clip(0.5 + 0.3 * np.sin(np.pi * x) ** 2, 0.01, 1000.0))
    return inits


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: multi-scale initialization, improved LP alternating with
    backtracking FISTA, top-K peak shaving, and progressive L-BFGS polish.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        _report(ps, parent, "parent")

    # ==================================================================
    # Phase 1: Multi-scale initialization (12% time)
    # ==================================================================
    phase1_deadline = t_start + 0.12 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Multi-Scale Initialization ---")

    # Start at small n and progressively increase
    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))

    # Determine target length
    n_target = 6962
    if parent is not None:
        n_target = len(parent)
    if n_target not in target_lengths:
        target_lengths.append(n_target)

    # Multi-scale: 512 -> 2048 -> target
    scale_lengths = [512, 1024, 2048, n_target]
    scale_lengths = sorted(set(l for l in scale_lengths if l >= 256))

    a_work = None
    for n_scale in scale_lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break
        print(f"  Multi-scale n={n_scale}, remaining={remaining:.0f}s")

        if a_work is None:
            # Generate diverse inits at this scale
            inits = build_initializations(n_scale, rng, parent)
            best_at_scale = None
            best_sc_at_scale = np.inf
            for i, a0 in enumerate(inits[:6]):
                if time.monotonic() > phase1_deadline:
                    break
                a_er = multistage_error_reduction(a0,
                                                  stages=[(30, 0.5), (50, 0.7), (70, 0.85)])
                sc = compute_score(a_er)
                if sc < best_sc_at_scale:
                    best_sc_at_scale = sc
                    best_at_scale = a_er.copy()
                _report(sc, a_er, f"ER ms n={n_scale} i={i}")
            a_work = best_at_scale
        else:
            # Interpolate from previous scale
            a_interp = interpolate_sequence(a_work, n_scale)
            a_interp = np.maximum(a_interp, 0.0)
            S = float(a_interp.sum())
            if S > 0:
                a_interp = a_interp / S
            a_work = multistage_error_reduction(a_interp,
                                                stages=[(30, 0.5), (60, 0.7), (80, 0.85)])
            sc = compute_score(a_work)
            _report(sc, a_work, f"ER interp n={n_scale}")

    if a_work is not None:
        _report(compute_score(a_work), a_work, "multi-scale final")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP Alternating with backtracking (45% time)
    # ==================================================================
    phase2_deadline = t_start + 0.57 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LP Alternating (Improved) ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP with increasing FISTA iterations and beta
    configs = [
        (8000, 150, 800, 35),
        (12000, 300, 1500, 30),
        (16000, 500, 3000, 25),
        (20000, 800, 5000, 20),
        (25000, 1200, 8000, 15),
    ]

    x_warm = None
    for fi, (fista_it, beta_s, beta_e, max_outer) in enumerate(configs):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 45:
            break
        print(f"  LP round {fi}: fista={fista_it}, beta={beta_s}->{beta_e}, "
              f"outer={max_outer}, rem={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating_improved(
            a_work,
            max_outer_iters=max_outer,
            fista_iters=fista_it,
            time_limit=phase2_deadline - 20,
            beta_start=beta_s,
            beta_end=beta_e,
            use_backtracking=True
        )
        sc = compute_score(a_out)
        _report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Top-K Peak Shaving (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.77 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: Top-K Peak Shaving ---")

    a_shave = best_a.copy()
    S = float(a_shave.sum())
    if S > 0:
        a_shave = a_shave / S

    # Try different K values
    for K_try in [3, 5, 8, 1]:
        if time.monotonic() > phase3_deadline - 15:
            break
        print(f"  Top-K shaving K={K_try}...")
        a_out = peak_shaving_topk(a_shave, n_steps=200, K=K_try,
                                  beta=500.0, rng=rng)
        sc = compute_score(a_out)
        _report(sc, a_out, f"peak K={K_try}")
        if sc < best_score:
            a_shave = a_out.copy()

    # Second pass with higher beta
    if time.monotonic() < phase3_deadline - 15:
        print("  Top-K shaving K=2, beta=2000...")
        a_out = peak_shaving_topk(a_shave, n_steps=150, K=2,
                                  beta=2000.0, rng=rng)
        sc = compute_score(a_out)
        _report(sc, a_out, "peak K=2 hi")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS Polish (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.92 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 4: L-BFGS Polish ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [200, 500, 1000, 2000, 5000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        maxit = 5000 if beta < 1000 else 3000
        sc, a_opt = lbfgs_run(a_norm, beta, maxiter=maxit, maxfun=150000,
                              gtol=1e-14)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        _report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    # Also try top-K LSE with L-BFGS
    if time.monotonic() < phase4_deadline - 20:
        print("  L-BFGS top-K=3 beta=3000...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = lbfgs_run(a_norm, 3000, maxiter=4000, maxfun=100000,
                              gtol=1e-14, use_topk=True, K=3)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        _report(sc, a_opt, "LBFGS topK")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final mixed polish (8% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Quick LP pass on best
    if time.monotonic() < deadline - 60:
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 0:
            a_final_in = a_final_in / S
        a_final, sc_final, _ = lp_alternating_improved(
            a_final_in, max_outer_iters=8, fista_iters=15000,
            time_limit=deadline - 40, beta_start=500.0, beta_end=2000.0
        )
        sc = compute_score(a_final)
        _report(sc, a_final, "final LP")

    # Final peak shave
    if time.monotonic() < deadline - 20:
        a_final = peak_shaving_topk(best_a, n_steps=100, K=2,
                                    beta=5000.0, rng=rng)
        sc = compute_score(a_final)
        _report(sc, a_final, "final peak")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 10:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt = lbfgs_run(a_norm, 15000, maxiter=5000, maxfun=200000,
                              gtol=1e-15)
        _report(sc, a_opt, "final LBFGS b=15000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
