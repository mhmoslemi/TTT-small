    conv_wa = np.fft.irfft(Wft * Aft, N2)  # conv(w, a), length 3n-2
    wf = w[::-1]
    Wfft = np.fft.rfft(wf, N2)
    conv_awf = np.fft.irfft(Aft * Wfft, N2)  # conv(a, w_flipped), length 3n-2
    
    grad_conv = conv_wa[:n] + conv_awf[2*n-2 : 2*n-2+n][::-1]
    # Wait, I need conv_awf[2n-2-i] for i in [0, n-1]
    # That's conv_awf[2n-2], conv_awf[2n-3], ..., conv_awf[n-1]
    # = conv_awf[n-1 : 2n-1][::-1]
    grad_conv = conv_wa[:n] + conv_awf[n-1:2*n-1][::-1]
    
    grad = (float(n) * grad_conv) / (S * S)
    L_val = dmax + np.log(sum_e) / beta
    grad -= (2.0 * float(n) * L_val) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    
    score = 2.0 * L_val + PEN_SUM * (S - 1.0) ** 2
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix C where C@b = conv(a, b). C is (2n-1) x n."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            if a[i] > 1e-15:
                rows.append(k)
                cols.append(i)
                data.append(a[i])
    C = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return C


def lp_step(a, M, time_limit=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    if n > 8000:
        return None  # Too large for practical LP
    C = build_conv_matrix_sparse(a)
    nnz = C.nnz
    if nnz > 50_000_000:
        return None
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    try:
        res = linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            bounds=bounds, method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
    except Exception:
        return None
    if time.monotonic() - t0 > time_limit + 2.0:
        return None
    if res.status == 0 and res.x is not None:
        b = np.clip(res.x, 0, None)
        if b.sum() > 1e-10:
            return b
    elif res.status == 3:  # time limit
        # Try to salvage
        if hasattr(res, 'x') and res.x is not None:
            b = np.clip(res.x, 0, None)
            if b.sum() > 1e-10:
                return b
    return None


def lp_iteration(a, max_iters=50, deadline=None, n_lp=2048):
    """Iterative LP-based optimization at reduced length."""
    a_small = normalize_a(resample_sequence(a, n_lp))
    best_a = a_small.copy()
    best_sc = exact_score_a(a_small)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        sc = exact_score_a(a_small)
        if not np.isfinite(sc):
            break
        S = float(a_small.sum())
        c = fast_self_conv(a_small)
        M = float(np.max(c))
        if M < 1e-15:
            break
        
        b = lp_step(a_small, M, time_limit=8.0)
        if b is None:
            # Fallback: use subgradient step
            cmax_idx = np.argmax(c)
            w = np.zeros(2 * n_lp - 1)
            w[cmax_idx] = 1.0
            wr = w[::-1]
            wf = w[::-1]
            N2 = next_power_of_two(3 * n_lp)
            Aft = np.fft.rfft(a_small, N2)
            Wft = np.fft.rfft(w, N2)
            conv_wa = np.fft.irfft(Wft * Aft, N2)
            Wfft = np.fft.rfft(wf, N2)
            conv_awf = np.fft.irfft(Aft * Wfft, N2)
            grad_conv = conv_wa[:n_lp] + conv_awf[n_lp-1:2*n_lp-1][::-1]
            grad = (float(n_lp) * grad_conv) / (S * S) - (2.0 * float(n_lp) * c[cmax_idx]) / (S * S * S)
            grad = grad - (float(n_lp) * c[cmax_idx] * 2.0) / (S * S * S)
            # Simple update
            eta = 0.05 / (np.max(np.abs(grad)) + 1e-15)
            a_new = a_small - eta * grad
            a_small = normalize_a(np.maximum(a_new, 0))
            continue
        
        b = normalize_a(b)
        for t in (0.5, 0.3, 0.2, 0.1):
            a_trial = normalize_a((1 - t) * a_small + t * b)
            sc_trial = exact_score_a(a_trial)
            if np.isfinite(sc_trial) and sc_trial < sc:
                a_small = a_trial
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a_small.copy()
                break
    
    return float(best_sc), best_a


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def exact_subgrad_step(a, k_active, eta):
    d, S = _d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = _grad_conv(a, w)
    grad = (float(len(a)) * h) / (S * S)
    L_val = float(np.dot(w, d))
    grad -= (2.0 * float(len(a)) * L_val) / S
    return a - eta * grad


def _d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def _grad_conv(a, w):
    """Compute sum_k w[k]*(a[k-i]+a[i-k]) (valid) for all i using FFT."""
    n = len(a)
    N2 = next_power_of_two(3 * n)
    Aft = np.fft.rfft(a, N2)
    Wft = np.fft.rfft(w, N2)
    conv_wa = np.fft.irfft(Wft * Aft, N2)
    wf = w[::-1]
    Wfft = np.fft.rfft(wf, N2)
    conv_awf = np.fft.irfft(Aft * Wfft, N2)
    grad_conv = conv_wa[:n] + conv_awf[n-1:2*n-1][::-1]
    return np.nan_to_num(grad_conv, nan=0.0, posinf=0.0, neginf=0.0)


def subgrad_polish(a, max_iters=100, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    k_list = (1, 2, 3, 5, 8)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for k in k_list:
            for eta in (0.002, 0.005, 0.01, 0.02, 0.05):
                if deadline is not None and time.monotonic() > deadline:
                    break
                d, S = _d_from_a(a)
                if d.size == 0:
                    break
                dmax = float(np.max(d))
                if dmax <= 1e-15:
                    break
                w = active_weights_topk(d, k)
                h = _grad_conv(a, w)
                L_val = float(np.dot(w, d))
                grad = (float(n) * h) / (S * S) - (2.0 * float(n) * L_val) / S
                gmax = float(np.max(np.abs(grad)))
                if gmax < 1e-15:
                    continue
                step = eta / gmax
                q = normalize_a(np.maximum(a - step * grad, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q
        if not improved and it > 10:
            break
    return float(best_sc), best_a


def mirror_descent_step(a, k_active, t_power):
    d, S = _d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = _grad_conv(a, w)
    h = np.maximum(h, 1e-15)
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def mirror_descent_phase(a_start, max_iters=200, deadline=None, k_list=(1, 2, 3, 5, 8)):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        for k in k_list:
            if deadline is not None and time.monotonic() > deadline:
                break
            for t_p in (0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4):
                if deadline is not None and time.monotonic() > deadline:
                    break
                q = mirror_descent_step(a, k, t_p)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
    return float(best_sc), best_a


def spectral_shaping(a, n_iters=50, deadline=None):
    """Work in frequency domain to reduce peak of self-convolution."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    N = next_power_of_two(2 * n - 1)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N)
        A2 = A * A  # Spectrum of self-convolution
        conv = np.fft.irfft(A2, N)[:2*n-1]
        conv = np.maximum(conv, 0)
        cmax = float(np.max(conv))
        if cmax < 1e-15:
            break
        # Find active lags
        threshold = cmax * 0.95
        active = conv > threshold
        # In frequency domain, reduce power at frequencies contributing to active lags
        # The contribution of frequency f to conv[k] is |A[f]|^2 * exp(2j*pi*f*k/N)
        # To reduce conv at active lags, reduce |A[f]| at frequencies where
        # the phase aligns with those lags.
        # Simpler: multiply A by (1 - eps * |contribution to peak|)
        # Or: reduce A[f] where the energy concentrates at the peak lag
        k_peak = int(np.argmax(conv))
        # Phase of A[f]^2 at lag k_peak: exp(2j*pi*f*k_peak/N)
        # Reduce |A[f]| where this phase is near 1 (contributing to peak)
        f_vals = np.fft.rfftfreq(N, d=1.0/N) * N  # 0 to N/2
        phase = 2.0 * np.pi * f_vals * k_peak / N
        contribution = np.cos(phase)  # Real part of exp(2j*phase)
        # Reduce where contribution is positive
        reduction = 1.0 - 0.05 * np.maximum(contribution, 0)
        A_new = A * np.sqrt(np.maximum(reduction, 0.5))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = normalize_a(a_new)
        sc = exact_score_a(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
    
    return float(best_sc), best_a


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.4))
    mode = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    
    if mode == 0:
        q = p * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        d, S = _d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_topk(d, int(rng.integers(1, 6)))
                h = _grad_conv(p, w)
                hmax = float(np.max(np.abs(h))) + 1e-15
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        vshape = 1.0 + strength * 0.5 * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Sinusoidal perturbation
        freq = float(rng.uniform(1, 5))
        phase = float(rng.uniform(0, 2 * np.pi))
        q = p * (1.0 + strength * 0.3 * np.sin(2 * np.pi * freq * x + phase))
    elif mode == 8:
        # W-shape perturbation
        wshape = 1.0 + strength * 0.3 * np.abs(4.0 * x - 2.0) * np.abs(4.0 * x - 2.0)
        q = p * wshape
    else:
        # Targeted reduction at peak-contributing positions
        d, S = _d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_topk(d, 2)
                h = _grad_conv(p, w)
                hmax = float(np.max(h)) + 1e-15
                q = p * np.exp(-strength * 0.3 * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    
    return normalize_a(np.clip(q, 0.0, None))


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    return base * scale * max(1.0, np.log2(max(16, n)) / np.log2(8192.0))


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 5.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    mf = max(64, min(3000, mf))
    return mf


def optimize_with_lbfgs_trajectory(a_start, deadline, label=""):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    stages = [
        (500.0, 0.06),
        (2000.0, 0.08),
        (5000.0, 0.08),
        (20000.0, 0.08),
        (50000.0, 0.10),
        (100000.0, 0.12),
        (300000.0, 0.12),
        (500000.0, 0.12),
        (1000000.0, 0.12),
        (3000000.0, 0.14),
        (10000000.0, 0.18),
    ]
    
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))
        
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.85, len(a))
            if mf < 64:
                break
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)
            if time.monotonic() > sub_dead - 3.0:
                break
    
    return float(best_sc), best_a
