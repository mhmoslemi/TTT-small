def solve_lp(a, M, active_lags):
    n = len(a)
    # Maximize sum(b) = ones^T b
    # s.t. A b <= M * ones, b >= 0
    # linprog minimizes, so minimize -sum(b)

    # Build A matrix (sparse)
    from scipy.sparse import csr_matrix, coo_matrix
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(len(active_lags), n)).tocsr()
    d = np.full(len(active_lags), M)
    
    res = linprog(c, A_ub=A, b_ub=d, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

"""
LP-iteration + spectral + subgradient optimizer for minimizing 2n*max(conv(a,a))/S^2.
Key new idea: use the LP direction from the state-of-the-art, implemented efficiently
with reduced active sets. Combined with spectral shaping and adaptive subgradient.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def lse_smooth_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_s = np.exp(shifted)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M3 = (2*n-1) + n - 1
    M3_fft = 1 << (M3-1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    cr = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * cr[n-1:2*n-1]
    S2 = S*S
    S3 = S2*S
    grad = (2.0*n) * (grad_L/S2 - 2.0*L/S3)
    score = (2.0*n*L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(lse_smooth_and_grad, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fienup(a, n_iters=300, mode='flat'):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2*n-1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        if mode == 'flat':
            alpha = 0.15 + 0.7*progress
            P_new = (1-alpha)*P + alpha*P_mean
        elif mode == 'quantile':
            q = 0.70 + 0.25*progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        else:  # 'boost_low'
            alpha = 0.1 + 0.3*progress
            P_new = P.copy()
            low = P < P_mean*0.5
            P_new[low] = P[low] / (1.0 + alpha)
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def lp_direction(a, M, active_lags, n):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for k in active_lags, b >= 0."""
    c_obj = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        # conv(a,b)_k = sum_j a[k-j] * b[j] for j in [j_start, j_end]
        for j in range(j_start, j_end + 1):
            w = a[k - j]
            if w > 1e-15:
                rows.append(idx)
                cols.append(j)
                vals.append(w)
    if not rows:
        return None
    A_sparse = coo_matrix((vals, (rows, cols)), shape=(len(active_lags), n)).tocsc()
    d = np.full(len(active_lags), M)
    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=d,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            b_val = np.clip(res.x, 0, None)
            if b_val.sum() > 1e-12:
                return b_val
    except Exception:
        pass
    return None


def lp_iterate(a, n_iters=500, tol=0.005, max_active=200):
    """Iterate: solve LP for direction, line search, update."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    t = 0.1  # initial step size
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if it % 50 == 0:
            elapsed_info = ""
            print(f"    LP iter {it}: score={sc:.8f}, M/Mean={M_val/(S*S/(2*n-1)):.6f}")
        
        # Find active lags
        c_mean = float(c.sum()) / len(c)
        thresh = M_val - max(0.01 * c_mean, 1e-12)
        active_mask = c >= thresh
        active_lags = np.where(active_mask)[0]
        if len(active_lags) > max_active:
            # Keep only the worst ones
            active_lags = np.argsort(c)[::-1][:max_active]
        
        if len(active_lags) == 0:
            break
        
        # Solve LP
        g0 = lp_direction(a, M_val, active_lags, n)
        if g0 is None:
            break
        g0 = g0 / g0.sum()  # normalize
        
        # Line search: try different t values
        best_local_sc = sc
        best_local_a = a.copy()
        for t_try in [t*0.5, t, t*1.5, t*2, t*0.25]:
            t_try = min(t_try, 0.5)
            a_try = (1-t_try)*a + t_try*g0
            a_try = normalize(a_try)
            sc_try = compute_score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local_a = a_try.copy()
                best_t = t_try
        
        if best_local_sc >= sc:
            # No improvement, reduce step size
            t *= 0.7
        else:
            t = min(best_t * 1.1, 0.5)
            a = best_local_a
        
        if t < 1e-6:
            break
    
    return best_a
