"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: LP alternating, LSE+L-BFGS, heavy-ball subgradient with momentum,
and Griffin-Lim frequency-domain autocorrelation flattening.

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient via FFT cross-correlation
    n_c = 2 * n - 1
    M2_fft = 1 << (n + n_c - 2).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, gtol=1e-12, time_limit=None, t_start=None):
    """Run L-BFGS-B on LSE-smoothed objective with time check."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    # Check time within callback
    def obj_and_grad(x):
        if time_limit is not None and time.monotonic() > time_limit:
            return lse_score_and_grad(x, beta)
        return lse_score_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxiter * 50, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def fista_lp(a, M, n_iters, beta, x_init=None):
    """
    Solve: min FISTA of smooth proxy for max(conv(a,x)-M) s.t. x >= 0.
    Returns x (unnormalized).
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    
    # Precompute FFT of a
    FA = np.fft.rfft(a, M_fft)
    sum_a2 = float(np.sum(a * a))
    L = beta * n * sum_a2 * 2.0 + 1e-12
    step = 1.0 / L
    
    for it in range(n_iters):
        c = np.fft.irfft(FA * np.fft.rfft(x, M_fft), M_fft)[:n_c]
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum
        
        # Gradient: B^T p
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr  # d/dx of log-sum-exp(beta*(Bx - M))
        
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x

        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x_warm = x.copy()
        x = x / Sx
        
        # Line search on mixing

            no_improve = 0

def griffin_lim_flatten(a_init, n_iters, time_limit, flatten_power=0.7):
    """
    Griffin-Lim style iteration to flatten autocorrelation.
    In frequency domain, modify |A|^2 to be flatter, then project back to non-negative.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if time.monotonic() > time_limit:
            break
        
        A = np.fft.rfft(a, M)
        power = np.abs(A)**2
        
        # Current autocorrelation
        c = np.fft.irfft(A * A, M)[:2*n-1]
        c_flat = float(c.mean())
        c_max = float(c.max())
        
        if c_max - c_flat < 1e-15:
            break
        
        # Target: make power spectrum flatter
        # The ideal autocorrelation is flat (all equal to c_flat)
        # Target power spectrum: uniform (all equal to total_energy / M)
        total_energy = float(power.sum())
        target_power = np.ones_like(power) * total_energy / len(power)
        
        # Blend: new_power = (1-p)*current + p*target
        p = flatten_power * (0.5 + 0.5 * (it / max(1, n_iters)))
        new_power = (1.0 - p) * power + p * target_power
        
        # New magnitude
        new_mag = np.sqrt(np.maximum(new_power, 0.0))
        
        # Preserve phase
        mag_old = np.maximum(np.abs(A), 1e-30)
        phase = A / mag_old
        
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            if it % 50 == 0:
                print(f"    GL it={it}: {sc:.8f}")
        a = a_new
    
    return best_a, best_sc


def heavy_ball_subgrad(a_init, n_iters, time_limit, momentum=0.85):
    """Heavy ball method on the subgradient of the max-convolution objective."""
    n = len(a_init)
    x = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    S = float(x.sum())
    if S < 1e-12:
        x = np.ones(n) / n
        S = 1.0
    x = x / S
    
    v = np.zeros(n)
    best_a = x.copy()
    best_sc = compute_score(x)
    step = 0.01
    
    for it in range(n_iters):
        if time.monotonic() > time_limit:
            break
        
        c = self_conv(x)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        S2 = S * S
        
        # Subgradient of max(c) w.r.t. x[j]
        # d c[k_star] / d x[j] = x[j+k_star] + x[j-k_star] (for valid indices)
        # For k_star = 0: d/dx[j] = 2*x[j]
        grad_c = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            if k_star == 0:
                grad_c[j_start:j_end+1] = 2.0 * x[j_start:j_end+1]
            else:
                # For j in [j_start, j_end]:
                # d c[k_star]/dx[j] = x[j+k_star] (if j+k_star < n) + x[j-k_star] (if j-k_star >= 0)
                for j in range(j_start, j_end + 1):
                    val = 0.0
                    if j + k_star < n:
                        val += x[j + k_star]
                    if j - k_star >= 0:
                        val += x[j - k_star]
                    grad_c[j] = val
        
        # Gradient of 2n*M/S^2
        grad_f = 2.0 * n * (grad_c / S2 - 2.0 * M_val / (S2 * S))
        
        # Heavy ball update
        v = momentum * v - step * grad_f
        x_new = np.maximum(x + v, 0.0)
        S_new = float(x_new.sum())
        if S_new < 1e-12:
            v = np.zeros(n)
            continue
        x_new = x_new / S_new
        
        # Adaptive step
        sc_new = compute_score(x_new)
        if sc_new > best_sc * 1.01:
            step *= 0.8
            v = 0.8 * v
        elif sc_new < best_sc:
            step = min(step * 1.1, 0.05)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = x_new.copy()
        else:
            step *= 0.95
        
        x = x_new
    
    return best_a, best_sc

            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x + 0.3))
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    
    for a in inits:
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S > 0:
            a = a / S
    return inits
