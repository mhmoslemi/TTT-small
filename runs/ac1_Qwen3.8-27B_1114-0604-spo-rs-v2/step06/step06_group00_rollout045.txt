"""
LP-guided subgradient optimizer with spectral methods for minimizing
2n * max(conv(a,a)) / (sum(a))^2.
Key innovations over previous approach:
1. O(n) LP dual computation provides globally-informed subgradient weights
2. Direct "peak surgery" targeting top contributors
3. Adaptive multi-resolution optimization
4. LP-primal recovery for informed update directions
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.zeros_like(a)
    return a / S


def compute_lp_dual(a):
    """
    Compute LP dual using O(n) greedy sliding window algorithm.
    The dual is: min M * sum(lam) s.t. sum(lam[j:j+n]) >= 1/a[j], lam >= 0
    """
    n = len(a)
    d = 1.0 / np.maximum(a, 1e-15)
    lam = np.zeros(2 * n - 1)

    window_sum = 0.0
    for j in range(n):
        deficit = d[j] - window_sum
        if deficit > 0:
            pos = j + n - 1
            lam[pos] += deficit
            window_sum += deficit

        # Slide window
        window_sum -= lam[j]
        if j + n < 2 * n - 1:
            window_sum += lam[j + n]

    return lam


def compute_lp_weights(a):
    """
    Compute normalized LP dual weights for subgradient.
    Returns (weights, total_lambda) where weights sum to 1.
    """
    n = len(a)
    lam = compute_lp_dual(a)
    total = float(lam.sum())
    if total < 1e-15:
        # Fallback: uniform over active lags
        c = fast_conv(a)
        M = float(c.max())
        active = c >= M * 0.999
        weights = np.zeros(2 * n - 1)
        weights[active] = 1.0 / active.sum()
        return weights, 0.0
    return lam / total, total


def lp_guided_subgrad(a, step_size, use_momentum=True, velocity=None):
    """
    One step of LP-weighted subgradient descent on the score function.
    Returns (new_a, velocity, score).
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    S = float(a.sum())
    score = 2.0 * n * M / (S * S)

    # Compute LP weights
    w, total_lam = compute_lp_weights(a)

    # Compute gradient: grad_M[j] = 2 * (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    grad_M = 2.0 * fast_conv_cross(w, a_rev)[n - 1:2 * n - 1]

    # Full score gradient
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * grad_M - (4.0 * n * M / S3)

    # Momentum
    if use_momentum and velocity is not None:
        velocity = 0.9 * velocity - step_size * grad
    else:
        velocity = -step_size * grad

    a_new = a + velocity
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        a_new = a
    else:
        a_new = a_new / S_new

    return a_new, velocity, score


def lp_guided_optimize(a, n_iters, step_init=0.01, decay=0.9999, min_step=1e-6):
    """
    Run LP-guided subgradient optimization with adaptive step size.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    velocity = np.zeros(n)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        a, velocity, score = lp_guided_subgrad(a, step, use_momentum=True, velocity=velocity)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 100 == 0:
            actual = compute_score(a)
            if actual < best_score:
                best_score = actual
                best_a = a.copy()

        # Adaptive step size (Polyak-like)
        gap = best_score - lower_bound
        if gap > 1e-8:
            grad_norm = np.linalg.norm(velocity) / max(step, 1e-10)
            if grad_norm > 1e-10:
                step_polyak = gap / (grad_norm * grad_norm + 1e-10)
                step = min(step * 1.01, max(step_polyak, min_step))
            else:
                step = min(step * 0.95, 0.05)
        step = max(step, min_step)
        step *= decay

    return best_a, best_score


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def fienup_direct(a, n_iters=300):
    """
    Direct peak-reduction spectral method.
    Instead of flattening, target reducing specific time-domain peaks.
    """
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        # Get current autocorrelation peak locations
        c = fast_conv(a)
        c_max = float(c.max())
        # Find top peaks
        top_k = 5
        peak_idxs = np.argsort(c)[::-1][:top_k]

        # Modify spectrum: reduce energy that contributes to peaks
        # The contribution of frequency m to autocorrelation lag k is |A[m]|^2 * exp(i*2*pi*m*k/M)
        # We want to reduce |A[m]|^2 where it contributes to peak lags

        alpha = 0.1 + 0.4 * progress
        P_new = P.copy()
        for k in peak_idxs:
            # Reduce spectral components that align with lag k
            freqs = np.arange(len(P))
            phase_factor = np.cos(2.0 * np.pi * freqs * k / M_fft)
            # Reduce where phase_factor is positive (constructive at this lag)
            mask = phase_factor > 0.5
            P_new[mask] *= (1.0 - alpha * 0.3)

        P_new = np.maximum(P_new, 1e-30)
        # Also apply mild flattening
        P_new = (1.0 - alpha * 0.2) * P_new + alpha * 0.2 * P_mean

        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def peak_surgery(a, n_iters=2000):
    """
    Direct peak surgery: identify top contributors to each peak and
    redistribute mass to non-contributing positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]

        for k_star in top_peaks:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            js = np.arange(j_start, j_end + 1)
            pairs = a[js] * a[k_star - js]
            total = pairs.sum()
            if total < 1e-15:
                continue

            # Find top contributors
            n_reduce = max(3, n // 200)
            top_local = np.argsort(pairs)[::-1][:n_reduce]
            reduce_idx = j_start + top_local

            # Find safe positions (not contributing to any top peak)
            peak_positions = set()
            for kp in top_peaks:
                kp = int(kp)
                js2 = range(max(0, kp - n + 1), min(kp, n - 1) + 1)
                peak_positions.update(js2)

            safe = [i for i in range(n) if i not in peak_positions and a[i] < 0.1 * S / n]
            if len(safe) < 5:
                safe = [i for i in range(n) if i not in peak_positions]
            if len(safe) < 5:
                safe = list(range(n))

            # Transfer mass
            delta = 0.002 / n
            for j in reduce_idx:
                if a[j] > 2 * delta:
                    a[j] -= delta
            n_add = min(len(safe), n_reduce * 2)
            add_idx = rng.choice(safe, n_add, replace=False)
            for j in add_idx:
                a[j] += delta

        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a, best_sc

def perturb_structured(a, rng, strength=0.02):
    """Add structured perturbation (low-frequency modulations)."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n)
    for m in range(1, 5):
        freq = rng.integers(1, 15)
        phase = 2.0 * np.pi * rng.uniform(0, 1)
        amp = strength / m
        pert += amp * np.cos(2.0 * np.pi * freq * x + phase)
    a_new = a * (1.0 + pert)
    return normalize(a_new)


def perturb_asymmetric(a, rng, strength=0.01):
    """Add asymmetric perturbation to break symmetry."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    # Linear gradient perturbation (breaks symmetry)
    grad_pert = strength * (2.0 * x - 1.0) * rng.uniform(0.5, 1.5)
    # Random high-frequency
    hf = strength * 0.5 * rng.standard_normal(n)
    a_new = a * (1.0 + grad_pert) + hf
    return normalize(a_new)


def interpolate_sequence(a, new_n):
    """Interpolate to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            peak_mask = np.ones(n, dtype=bool)
            peak_mask[j_start:j_end + 1] = False
            trough_idx = np.where(peak_mask)[0]
            n_add = max(100, n // 30)
            if len(trough_idx) > n_add:
                trough_idx = rng.choice(trough_idx, n_add, replace=False)
            delta = 0.001 / n
            a_pert[trough_idx] += delta
            a_pert = normalize(a_pert)
        else:
            # Random multiplicative
            a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            a_pert = normalize(a_pert)

        # Quick LP + LSE optimization
        maxiter_lp = max(500, int(remaining * 4))
        a_opt, sc_opt = lp_guided_optimize(a_pert, n_iters=maxiter_lp, step_init=0.015)
        update(sc_opt, a_opt, f"RESTART-LP {r}")

        if time.monotonic() < phase4_end - 10:
            beta = 500 + r * 300
            maxiter_lse = max(500, int((phase4_end - 10 - time.monotonic()) * 4))
            sc2, a2 = run_lbfgs(a_opt, beta, maxiter=maxiter_lse, maxfun=maxiter_lse * 15)
            update(sc2, a2, f"RESTART {r}")

    # ==================================================================
    # Phase 5: N-growth exploration (5% time)
    # ==================================================================
    phase5_end = t_start + 0.92 * (budget_s - 30.0)
    print(f"\n=== Phase 5: N-growth ===")

    current_n = len(best_a)
    target_ns = []
    if current_n < 12000:
        target_ns.append(12000)
    if current_n < 15000:
        target_ns.append(15000)

    for new_n in target_ns:
        if time.monotonic() > phase5_end - 30:
            break
        remaining = phase5_end - 30 - time.monotonic()
        if remaining < 30:
            break

        a_large = interpolate_sequence(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  Interpolated to n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        # Quick optimization on larger seq
        if time.monotonic() < phase5_end - 15:
            n_iters = max(500, int(remaining * 3))
            a_opt, sc_opt = lp_guided_optimize(a_large, n_iters=n_iters, step_init=0.01)
            update(sc_opt, a_opt, f"INTERP+LP n={new_n}")

    # ==================================================================
    # Phase 6: Final polish (8% time)
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")

    # High-beta LSE
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        for beta in [5000, 15000, 40000]:
            if time.monotonic() > deadline - 20:
                break
            maxiter = max(500, int((deadline - 20 - time.monotonic()) * 4))
            sc, a_fin = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 15)
            update(sc, a_fin, f"FINAL-LSE b={beta}")

    # Final LP pass
    if time.monotonic() < deadline - 30:
        remaining_iters = max(500, int((deadline - 30 - time.monotonic()) * 5))
        a_fin, sc_fin = lp_guided_optimize(best_a, n_iters=remaining_iters, step_init=0.005)
        update(sc_fin, a_fin, "FINAL-LP")

    # Final peak surgery
    if time.monotonic() < deadline - 15:
        remaining_iters = max(300, int((deadline - 15 - time.monotonic()) * 1500))
        a_ps, sc_ps = peak_surgery(best_a, n_iters=min(remaining_iters, 2000))
        update(sc_ps, a_ps, "FINAL-PSUR")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
