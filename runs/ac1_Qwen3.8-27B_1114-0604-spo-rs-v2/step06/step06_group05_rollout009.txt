```python
    S = float(a.sum())

    score = 2.0 * n * Q / (S2 * S2)
```

```python
"""
Optimized step function: L2-smoothed gradient descent + adaptive LSE-LBFGS + peak coordinate descent.
Key improvements: L2 pre-optimization for smooth basin finding, adaptive beta scheduling,
coordinate-wise peak reduction, and multi-restart strategy.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr_ac(a_rev, c, n):
    """Compute (a_rev * c)[n-1 : 2n-1]."""
    M2 = 1 << (3 * n - 2).bit_length()
    AR = np.fft.rfft(a_rev, M2)
    C = np.fft.rfft(c, M2)
    return np.fft.irfft(AR * C, M2)[n - 1:2 * n - 1]


def l2_score_grad(a):
    """Score = 2n * sum(c²) / S⁴, with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    Q = float(np.dot(c, c))
    score = 2.0 * n * Q / (S2 * S2)
    a_rev = a[::-1]
    g = cross_corr_ac(a_rev, c, n)
    S5 = S * S2 * S2
    grad = 4.0 / S5 * (S * g - Q)
    return score, grad


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_s = np.exp(scaled - max_scaled)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 1 << (3 * n - 2).bit_length()
    AR = np.fft.rfft(a_rev, M2)
    P = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(AR * P, M2)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def l4_score_grad(a):
    """Score = 2n * sum(c⁴) / S⁸, with gradient. More sensitive to peaks."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    S4 = S2 * S2
    R = float(np.sum(c ** 4))
    score = 2.0 * n * R / (S4 * S4)
    # dR/da_i = 4 * sum_k c_k³ * dc_k/da_i = 8 * sum_k c_k³ * a_{k-i}
    c3 = c ** 3
    a_rev = a[::-1]
    M2 = 1 << (3 * n - 2).bit_length()
    AR = np.fft.rfft(a_rev, M2)
    C3 = np.fft.rfft(c3, M2)
    g = np.fft.irfft(AR * C3, M2)[n - 1:2 * n - 1]
    dR = 8.0 * g
    S8 = S4 * S4
    S9 = S8 * S
    grad = (2.0 * n) * (dR * S - 4.0 * R) / S9
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def projected_grad_descent(a0, obj_func, obj_params, n_iters=50000, lr_init=0.1, lr_decay=0.9999):
    """Projected gradient descent with backtracking line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        sc, grad = obj_func(a, *obj_params)
        
        # Gradient step
        v = lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a
            break
        a_new = a_new / S_new
        
        sc_new, _ = obj_func(a_new, *obj_params)
        if sc_new < sc:
            a = a_new
        else:
            lr *= 0.5
        
        lr = min(lr * lr_decay, 1.0)
        
        if (it + 1) % 1000 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
            print(f"    PGD it={it}: score={exact_sc:.8f}, lr={lr:.6f}")

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_a = a.copy()
    return best_a


def peak_coordinate_descent(a0, n_rounds=100, n_top=50, shrink=0.9):
    """Directly reduce top peaks by shrinking contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_top]

        max_reduction = 0
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner

            # Find the element that contributes most and shrink it
            if contrib.max() > 0:
                top_idx = np.argmax(contrib)
                j = j_start + top_idx
                a[j] *= shrink
                max_reduction = contrib[top_idx]

        # Renormalize
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        elif sc > best_sc * 1.01:
            # Revert if we made it worse
            a = best_a.copy()
            break

    return best_a


def gs_spectral_optimize(a0, n_iters=500, target='flat'):
    """GS iteration optimizing the spectral shape."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        if target == 'flat':
            b_target = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            # Target: current c but with peaks suppressed
            c_norm = c / c_max
            target_level = 0.5
            b_target = np.minimum(c_norm, target_level) / np.minimum(c_norm, target_level).sum()

        b_target_ext = np.zeros(M)
        b_target_ext[:2 * n - 1] = b_target

        T = np.fft.fft(b_target_ext)
        # We want A such that A² ≈ T, so A ≈ sqrt(T)
        # But T might be complex, so take sqrt of magnitude and keep phase
        T_mag = np.abs(T)
        T_phase = np.angle(T)
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        # Interpolate: A_new ≈ A * (|T|/|A|²)^(1/2) * exp(i*(phase_T - 2*phase_A)/2)
        # Simplified: blend magnitudes and phases
        alpha = 0.3 * (1.0 - it / n_iters) + 0.05
        new_mag = A_mag * (T_mag / (A_mag * A_mag + 1e-20)) ** (alpha * 0.5)
        new_phase = A_phase + alpha * 0.5 * (T_phase - 2.0 * A_phase)

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if (it + 1) % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def multi_scale_peak_target(a0, n_iters=10000):
    """Adaptive multi-scale peak targeting with LSE."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Start with moderate beta, increase over time
    beta_start = 500
    beta_end = 100000

    for it in range(n_iters):
        progress = it / n_iters
        beta = beta_start * (beta_end / beta_start) ** progress

        sc, grad = lse_score_grad(a, beta)

        # Gradient step with adaptive lr
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Polyak-like step size
        c = fast_conv(a)
        c_max = float(c.max())
        current = compute_score(a)
        target = best_sc * 0.999  # aim for slight improvement
        gap = current - target
        if gap > 0:
            lr = gap / (gn * gn + 1e-10)
        else:
            lr = 1.0 / (gn + 1e-10)
        lr = np.clip(lr, 1e-6, 0.5)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        a = a_new

        if (it + 1) % 500 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
                print(f"    MSP it={it}: score={best_sc:.8f}, beta={beta:.0f}")

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_a = a.copy()
```

```python
"""
Optimized step function search: L2/L4-smoothed gradient descent + adaptive LSE + peak coordinate descent.
Key innovations: smooth multi-order pre-optimization, Nesterov momentum, adaptive Polyak step,
and aggressive peak-targeted coordinate descent for final improvement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _cross_corr_rev(a, c, n):
    """Compute cross-correlation: result[i] = sum_j a[j]*c[i+j] via FFT."""
    a_rev = a[::-1]
    M2 = 1 << (3 * n - 2).bit_length()
    AR = np.fft.rfft(a_rev, M2)
    C = np.fft.rfft(c, M2)
    return np.fft.irfft(AR * C, M2)[n - 1:2 * n - 1]


def l2_score_grad(a):
    """Score = 2n*sum(c^2)/S^4 with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    Q = float(np.dot(c, c))
    score = 2.0 * n * Q / (S2 * S2)
    g = _cross_corr_rev(a, c, n)
    grad = 4.0 / (S2 * S2 * S) * (S * g - Q)
    return score, grad


def l4_score_grad(a):
    """Score = 2n*sum(c^4)/S^8 with gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    S2 = S * S
    S4 = S2 * S2
    R = float(np.sum(c ** 4))
    score = 2.0 * n * R / (S4 * S4)
    c3 = c ** 3
    g = _cross_corr_rev(a, c3, n)
    dR = 8.0 * g
    grad = (2.0 * n) * (dR * S - 4.0 * R) / (S4 * S4 * S)
    return score, grad


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_s = float(scaled.max())
    exp_s = np.exp(scaled - max_s)
    log_sum = max_s + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    g = 2.0 * _cross_corr_rev(a, p, n)
    S3 = S * S * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(wrapped, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def nesterov_optimize(a0, obj_func, obj_params, n_iters=50000, lr_init=0.05,
                      momentum=0.9, lr_decay=0.99995):
    """Nesterov momentum gradient descent with projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        if it % 100 == 0:
            sc, grad = obj_func(a, *obj_params)
        else:
            a_look = a - momentum * v
            sc, grad = obj_func(a_look, *obj_params)

        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a
            break
        a_new = a_new / S_new
        a = a_new
        lr *= lr_decay

        if (it + 1) % 2000 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
                print(f"    Nesterov it={it}: score={best_sc:.8f}")

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_a = a.copy()
    return best_a


def adam_optimize(a0, obj_func, obj_params, n_iters=100000, lr_init=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        t += 1
        sc, grad = obj_func(a, *obj_params)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = a
            break
        a_new = a_new / S_new
        a = a_new

        if (it + 1) % 5000 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
                print(f"    Adam it={it}: score={best_sc:.8f}")

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_a = a.copy()
    return best_a


def peak_coord_descent(a0, n_rounds=200, n_top=100, shrink=0.92):
    """Aggressive peak-targeted coordinate descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    stall = 0

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        peak_indices = np.argsort(c)[::-1][:n_top]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            if contrib.max() > 0:
                top_idx = np.argmax(contrib)
                a[j_start + top_idx] *= shrink

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def polyak_grad_descent(a0, obj_func, obj_params, n_iters=20000):
    """Polyak step-size gradient descent (adaptive for smooth objectives)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    best_smooth = obj_func(a, *obj_params)[0]

    for it in range(n_iters):
        sc, grad = obj_func(a, *obj_params)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-16:
            break
        gap = max(sc - best_smooth, 1e-10)
        lr = gap / (gn * gn + 1e-12)
        lr = np.clip(lr, 1e-8, 1.0)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc_new, _ = obj_func(a, *obj_params)
        if sc_new < best_smooth:
            best_smooth = sc_new

        if (it + 1) % 5000 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
                print(f"    Polyak it={it}: score={best_sc:.8f}")

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_a = a.copy()
    return best_a


def gs_spectral(a0, n_iters=400, alpha_init=0.4, alpha_final=0.03):
    """GS-Fienup with flat target."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_init + (alpha_final - alpha_init) * (it / n_iters)
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        new_mag = A_mag ** (1 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1 - alpha) * A_phase + alpha * T_phase * 0.5
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if (it + 1) % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def random_perturb(a0, rng, strength=0.05):

    a = np.maximum(a, 0.0)

    if S < 1e-12:
        return a0.copy()

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
