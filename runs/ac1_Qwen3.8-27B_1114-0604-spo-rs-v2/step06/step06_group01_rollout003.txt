    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    n = len(a)

    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_value_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z -= z.max()
    exp_z = np.exp(z)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_z / exp_sum
    L_val = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L_val / (S * S)
    
    # Gradient of L = LSE_beta(conv(a,a)) w.r.t. a
    # grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j in [0, n-1])
    # = 2 * cross_correlation(a, p)[j]
    n_c = len(c)
    N_fft = 1 << (n + n_c - 2).bit_length()
    RA = np.fft.rfft(np.concatenate([a, np.zeros(N_fft - n)]), N_fft)
    RP = np.fft.rfft(np.concatenate([p, np.zeros(N_fft - n_c)]), N_fft)
    corr = np.fft.irfft(np.conj(RA) * RP, N_fft)[:n]
    grad_L = 2.0 * corr
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad
