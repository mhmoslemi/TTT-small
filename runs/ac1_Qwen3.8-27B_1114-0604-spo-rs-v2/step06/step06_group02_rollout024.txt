"""
Optimized search for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-start L-BFGS with LSE approximation, spectral escape moves, and adaptive
beta continuation. Key improvements: diverse asymmetric initializations, softmax
reparameterization for escape, and very high-beta final polish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_grad(a, beta):
    """Smooth approximation of the objective using log-sum-exp."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(np.sum(exp_scaled)))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()
    # Gradient computation
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-22, 'gtol': 1e-18,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        S = a_out.sum()
        if S < 1e-6:
            a_out = a0
        else:
            a_out = a_out / S
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def continuation_lbfgs(a0, deadline, betas=None):
    """Run L-BFGS with adaptive beta continuation."""
    if betas is None:
        betas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    best_a = a0.copy()
    best_sc = compute_score(a0)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # Scale maxiter based on remaining time and beta
        time_per_iter = 0.05 * (1 + np.log10(beta))
        maxiter = max(1000, int(remaining / time_per_iter * 0.7))
        maxiter = min(maxiter, 50000)
        maxfun = maxiter * 20

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        else:
            # No improvement; try a smaller step from best_a
            best_a = best_a.copy()

    return best_a, best_sc


def spectral_escape(a, rng, n_attempts=5):
    """
    Try to escape local minima using spectral analysis of the correlation structure.
    Identify the peak lag and perturb in a direction that reduces it.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for attempt in range(n_attempts):
        c = fast_conv(a)
        k_peak = int(np.argmax(c))
        c_max = float(c.max())

        # Find indices contributing most to the peak
        j_start = max(0, k_peak - n + 1)
        j_end = min(k_peak, n - 1)
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_peak - js]
        total = contribs.sum()

        if total < 1e-15:
            continue

        # Strategy: reduce the top contributors, increase others
        top_frac = 0.3
        n_top = max(1, int(top_frac * len(contribs)))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_global = j_start + top_local

        # Perturbation: reduce top contributors, increase bottom
        bottom_local = np.argsort(contribs)[:n_top]
        bottom_global = j_start + bottom_local

        epsilon = 0.05 * (1 + 0.5 * attempt)
        a_new = a.copy()
        a_new[top_global] *= (1 - epsilon)
        a_new[bottom_global] *= (1 + epsilon)

        # Also add some structured noise
        noise_scale = 0.02 * (1 + attempt * 0.3)
        a_new += noise_scale * rng.standard_normal(n)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new

        # Quick L-BFGS polish
        sc, a_opt = run_lbfgs(a_new, 2000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


def geometric_init(n, r):
    """Create a geometric (exponential) initialization."""
    j = np.arange(n)
    a = r ** j
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def bimodal_init(n, center_frac=0.3, width=0.15):
    """Create a bimodal initialization (two bumps)."""
    x = np.arange(n) / n
    sigma = width
    a = np.exp(-0.5 * ((x - center_frac) / sigma) ** 2) + np.exp(-0.5 * ((x - (1 - center_frac)) / sigma) ** 2)
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def smooth_random_init(n, rng, n_bumps=3):
    """Create a random smooth non-negative initialization."""
    centers = rng.uniform(0, 1, n_bumps)
    widths = rng.uniform(0.02, 0.15, n_bumps)
    amps = rng.uniform(0.5, 2.0, n_bumps)
    x = np.arange(n) / n
    a = np.zeros(n)
    for c, w, amp in zip(centers, widths, amps):
        a += amp * np.exp(-0.5 * ((x - c) / w) ** 2)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def cosine_modulated_init(n, freq, amplitude):
    """Initialize with cosine modulation (encourages asymmetry with structure)."""
    x = np.arange(n) / n
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.0)
    # Add asymmetry
    a *= np.exp(-0.3 * x)
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def fienup_quick(a, n_iters=200):
    """Quick Fienup iterations for initialization improvement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.5 * (1.0 - it / n_iters) + 0.05
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_directed_descent(a, n_sweeps=30):
    """
    Coordinate descent that directly targets the peak lag.
    For the peak lag k*, adjust elements to reduce conv(a,a)[k*].
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_peak = int(np.argmax(c))
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # For peak lag k_peak, the coupling is:
        # c[k_peak] = sum_j a[j] * a[k_peak - j] for valid j
        # To reduce c[k_peak], we want to reduce products a[j]*a[k_peak-j]
        # Strategy: for each pair (j, k_peak-j), try to reduce the larger one

        j_start = max(0, k_peak - n + 1)
        j_end = min(k_peak, n - 1)

        # Try reducing top contributors
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_peak - js]
        n_adjust = min(30, len(contribs))
        top_local = np.argsort(contribs)[::-1][:n_adjust]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-12:
                continue
            # Try reducing a[j]
            best_val = a_old
            best_local_sc = sc
            for f in [0.5, 0.7, 0.8, 0.9]:
                a[j] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                # Quick score evaluation
                c_tmp = fast_conv(a_norm)
                sc_tmp = 2.0 * n * float(c_tmp.max()) / (S_tmp * S_tmp)
                if sc_tmp < best_local_sc:
                    best_local_sc = sc_tmp
                    best_val = a_old * f
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a


def multi_scale_optimize(a, deadline):
    """
    Optimize at multiple scales: coarse-to-fine.
    First optimize a downsampled version, then interpolate back.
    """
    n = len(a)
    if n <= 256:
        return continuation_lbfgs(a, deadline, betas=[100, 1000, 10000, 100000])

    # Downsample by factor 4
    factor = 4
    n_coarse = n // factor
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a[i * factor:(i + 1) * factor].sum()
    a_coarse = a_coarse / a_coarse.sum()

    # Optimize coarse
    a_coarse_opt, sc_coarse = continuation_lbfgs(
        a_coarse, min(deadline, time.monotonic() + 120),
        betas=[50, 500, 5000, 50000]
    )

    # Interpolate back
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n)
    a_interp = np.interp(x_fine, x_coarse, a_coarse_opt)
    a_interp = np.maximum(a_interp, 0.0)
    S = a_interp.sum()
    if S > 1e-12:
        a_interp = a_interp / S

    # Blend with original
    a_blend = 0.5 * a + 0.5 * a_interp
    a_blend = a_blend / a_blend.sum()

    # Fine optimization
    return continuation_lbfgs(a_blend, deadline,
                               betas=[100, 1000, 10000, 100000, 1000000])


def softmax_escape(a, rng, n_restarts=3):
    """
    Escape using softmax reparameterization.
    Map a to log-space, perturb, and map back.
    """
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for r in range(n_restarts):
        # Map to log space
        a_safe = np.maximum(a, 1e-10)
        x = np.log(a_safe)
        # Add noise in log space
        noise = rng.standard_normal(n) * (0.5 + 0.3 * r)
        x_new = x + noise
        # Map back with softmax-like normalization
        x_new = x_new - x_new.max()
        a_new = np.exp(x_new)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S

        # Quick optimize
        sc, a_opt = run_lbfgs(a_new, 5000, maxiter=8000, maxfun=150000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc
