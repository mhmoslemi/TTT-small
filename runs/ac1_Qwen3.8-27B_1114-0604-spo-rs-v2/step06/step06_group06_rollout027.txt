def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Cross-correlation: result[j] = sum_k w[k] * a[k-j] (valid indices)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_flattening_weights(conv, temperature=1.0, floor_frac=0.1):
    """
    Compute weights for flattening: emphasize all above-average positions,
    not just the top-k. Uses a soft threshold.
    """
    cmax = float(np.max(conv))
    cmean = float(np.mean(conv))
    if cmax <= cmean:
        return np.ones(conv.size) / conv.size
    spread = cmax - cmean
    if spread < 1e-15:
        return np.ones(conv.size) / conv.size
    # Soft weights: exponential of (conv - cmean)/temperature
    z = (conv - cmean) / (temperature * spread)
    z = np.clip(z, -10.0, 10.0)
    w = np.exp(z)
    # Floor: ensure at least floor_frac of positions get some weight
    w = np.maximum(w, floor_frac)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def compute_topk_weights(conv, top_k=10, eps_frac=0.01):
    """Compute active weights: uniform over top-k or positions near max."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.05 * conv.size))
    if active.sum() > max_count:
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active2 = np.zeros(conv.size, dtype=bool)
        active2[top_idx] = True
        active = active2
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def exp_update_with_momentum(a, h, velocity, eta, beta_mom):
    """Exponentiated gradient update with momentum in the gradient."""
    hmax = float(np.max(h))
    if hmax < 1e-30:
        return a.copy(), velocity
    velocity = beta_mom * velocity + (1.0 - beta_mom) * h / hmax
    log_a = np.log(a + 1e-300)
    log_a -= eta * velocity
    log_a -= float(np.max(log_a))
    new_a = np.exp(log_a)
    return new_a, velocity

def lse_score_grad(a, beta):
    """LSE-smoothed objective: 2n * LSE_beta(c) where c = conv(a,a)/S^2 * n."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    # Gradient of 2n * sum(c)/S^2 smoothed:
    # d(2n*L)/da[j] = 2n * [2n/S^2 * h[j] - 4n*a[j]*wdotd/S^3]... 
    # More carefully: L ≈ max(d) + log-sum-exp correction
    # d[j] = n*c[j]/S^2
    # dL/da[j] ≈ (n/S^2) * sum_k w[k] * dc[k]/da[j] - 2n*c̄/S^3 * a[j]... 
    # Simplified: grad ≈ 2n * (n/S^2) * h[j] - 4n*a[j]*(n/S^2)*wdotd/S
    # = 2n^2 * h[j]/S^2 - 4n^2 * a[j] * wdotd / S^3
    grad = (2.0 * n * n * h) / (S * S) - (4.0 * n * n * a * wdotd) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxfun=500):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-11,
        "maxiter": max(30, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.full(int(n), 1.0 / max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    # U-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 2.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    # Cosine
    for f in (1, 2, 3, 4, 5, 7, 11):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    # Sine absolute
    for f in (1, 2, 3, 5):
        for alpha in (0.5, 1.0, 1.5, 2.0):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    # Exponential tapers
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(x - 0.5)) + 0.01)
    # Bimodal
    for width in (0.03, 0.05, 0.08, 0.1, 0.15, 0.2):
        for center in (0.15, 0.2, 0.25, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Triangular
    for shift in (0.0, 0.05, 0.1, -0.05, -0.1):
        add(np.maximum(0.0, 1.0 - np.abs(2.0 * x + shift - 1.0)) + 0.01)
    # Power shapes
    for p in (0.3, 0.5, 1.0, 2.0, 3.0, 5.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Riesz-product-like (product of cosines)
    for (f1, f2, a1, a2) in [(3, 7, 0.5, 0.3), (5, 11, 0.4, 0.4), (2, 13, 0.6, 0.3),
                              (4, 9, 0.5, 0.5), (6, 17, 0.3, 0.4)]:
        add((1.0 + a1 * np.cos(2.0 * np.pi * f1 * x)) * (1.0 + a2 * np.cos(2.0 * np.pi * f2 * x)) + 0.01)
    # Random smooth (spectral)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    # Random
    for _ in range(4):
        add(0.05 + 0.95 * rng.random(n))
    # Log-uniform (biased toward ends)
    for skew in (0.5, 1.0, 2.0, 3.0):
        add(x ** skew * (1.0 - x) ** skew + 0.01)

    return inits


def perturb(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 12))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = compute_flattening_weights(conv, temperature=0.2)
        h = corr_pair(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        # Smooth noise in log space
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 7:
        nb = int(rng.integers(4, 30))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    elif mode == 8:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    elif mode == 9:
        # Oscillatory perturbation
        x = np.linspace(0.0, 1.0, n)
        f = float(rng.uniform(5, 50))
        q = p * (1.0 + strength * 0.3 * np.cos(2.0 * np.pi * f * x))
    elif mode == 10:
        # Center boost/reduce
        x = np.linspace(0.0, 1.0, n)
        w = float(rng.uniform(0.1, 0.5))
        center = float(rng.uniform(0.3, 0.7))
        q = p * (1.0 + strength * np.exp(-((x - center) / w) ** 2))
    else:
        q = p * (0.8 + 0.4 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def subgrad_fine_polish(a, deadline, verbose=False, tag=""):
    """Fine-grained subgradient polish with adaptive strategy."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Adaptive parameter sweep
    improved = True
    round_num = 0
    max_rounds = 6

    top_ks = [1, 2, 3, 4, 5, 7, 10, 15, 20, 30, 50, 75, 100]
    etas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0, 3.0]
    temperatures = [0.05, 0.1, 0.2, 0.3, 0.5, 1.0]

    while improved and time.monotonic() < deadline and round_num < max_rounds:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)

        # Try top-k based
        for tk in top_ks:
            if time.monotonic() > deadline:
                break
            for ef in [0.0, 0.005, 0.01, 0.02, 0.05]:
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    w = compute_topk_weights(conv, top_k=tk, eps_frac=ef)
                    h = corr_pair(best_a, w)
                    hmax = float(np.max(h))
                    if hmax < 1e-15:
                        continue
                    log_a = np.log(best_a + 1e-300)
                    log_a -= eta * h / hmax
                    log_a -= float(np.max(log_a))
                    new_a = normalize_a(np.exp(log_a))
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = new_a
                        improved = True
                        conv = fast_self_conv(best_a)

        # Try flattening-based
        for temp in temperatures:
            if time.monotonic() > deadline:
                break
            for eta in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8]:
                if time.monotonic() > deadline:
                    break
                w = compute_flattening_weights(conv, temperature=temp)
                h = corr_pair(best_a, w)
                hmax = float(np.max(h))
                if hmax < 1e-15:
                    continue
                log_a = np.log(best_a + 1e-300)
                log_a -= eta * h / hmax
                log_a -= float(np.max(log_a))
                new_a = normalize_a(np.exp(log_a))
                sc = exact_score(new_a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = new_a
                    improved = True
                    conv = fast_self_conv(best_a)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
                print("Loaded parent: n={}, sum={:.4f}".format(p.size, float(np.sum(p))))
    except Exception:
        parent = None

    # Choose n: try 12288 (larger than parent's 10240 for better bound)
    n = 12288
    if parent is not None:
        if parent.size >= 1024:
            # Resample parent to our chosen n
            a0 = resample_sequence(parent, n)
        else:
            a0 = resample_sequence(parent, n)
    else:
        inits = generate_inits(n, rng, None, cap=25)
        best_init_sc = np.inf
        a0 = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a0 = init
        print("No parent. Best init score: {:.9f}".format(best_init_sc))

    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score(a)
    print("Initial: n={}, score={:.9f}".format(n, best_sc))

    # Phase 1: Main flattening + peak optimization (~45% of time)
    p1_dead = t_start + 0.45 * total_budget
    sc, a = flatten_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 1: {:.9f}".format(best_sc))

    # Phase 2: LSE-LBFGS refinement (~20% of time)
    p2_dead = t_start + 0.65 * total_budget
    a = normalize_a(best_a)
    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000]:
        if time.monotonic() > p2_dead - 10:
            break
        rem = p2_dead - time.monotonic()
        if rem < 20:
            break
        mf = max(64, int(rem * 0.6 / 0.1))
        mf = min(mf, 1500)
        sc, a = run_lbfgs(a, beta, p2_dead, mf)
        a = normalize_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("  Phase 2 beta={}: {:.9f}".format(beta, best_sc))

    # Phase 3: Fine subgradient polish (~15% of time)
    p3_dead = t_start + 0.80 * total_budget
    sc, a = subgrad_fine_polish(best_a, deadline=p3_dead, verbose=True, tag="p3")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 3: {:.9f}".format(best_sc))

    # Phase 4: Another flatten-optimize round on improved solution (~10% of time)
    p4_dead = t_start + 0.90 * total_budget
    sc, a = flatten_optimize(best_a, deadline=p4_dead, verbose=True, tag="p4")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 4: {:.9f}".format(best_sc))

    # Phase 5: Diversification restarts (~5% of time)
    p5_dead = deadline - 15.0
    n_restarts = 4
    restart_count = 0
    while time.monotonic() < p5_dead - 60 and restart_count < n_restarts:
        rem = p5_dead - 60.0 - time.monotonic()
        if rem < 80:
            break
        restart_count += 1
        strength = float(rng.uniform(0.03, 0.2))
        p = perturb(best_a, rng, strength=strength)
        # Quick optimize
        chunk_dead = time.monotonic() + rem * 0.6
        sc, seq = flatten_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(restart_count))
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print("Restart {}: {:.9f}".format(restart_count, best_sc))
        else:
            print("Restart {}: no improvement (best_so_far={:.9f}, got={:.9f})".format(
                restart_count, best_sc, sc))

    # Phase 6: Final polish
    if time.monotonic() < deadline - 5:
        sc, a = subgrad_fine_polish(best_a, deadline=deadline - 3, verbose=False, tag="final")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("Final polish: {:.9f}".format(best_sc))

    # Final cleanup
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
