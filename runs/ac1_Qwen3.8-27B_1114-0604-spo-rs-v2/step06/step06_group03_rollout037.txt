Actually, one key improvement I can make: instead of the LSE approximation, I can use a **direct subgradient method** with a smart step size rule. The subgradient of $\max_k f_k(a)$ at point $a$ is $f'_k(a)$ where $k$ is an argmax. If multiple $k$ achieve the max, any convex combination of their gradients is a valid subgradient.

The use subgradient method:
$$a^{t+1} = \text{proj}_{\{a \geq 0, \sum a = 1\}}(a^t - \eta_t g^t)$$

where $g^t$ is the subgradient at $a^t$. The step size $\eta_t = c / \sqrt{t}$ gives $O(1/\sqrt{T})$ convergence, but in practice we can do better with a backtracking or Polyak step size.

For the projection onto the simplex ($a \geq 0$, $\sum a = 1$), there's a well-known $O(n \log n)$ algorithm.

But actually, the mirror descent (multiplicative weights) approach is better for this problem because:
1. It automatically maintains $a \geq 0$
2. The "geometry" of the simplex is better matched by the entropy regularizer
3. The convergence is faster in practice

The mirror descent update for $\min f(a)$ s.t. $a \in \Delta$ (simplex) is:
$$a^{t+1}_j = \frac{a^t_j \exp(-\eta_t g^t_j)}{\sum_k a^t_k \exp(-\eta_t g^t_k)}$$

This is equivalent to the multiplicative weights update.

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous: longer sequences (16384), Nesterov-accelerated

import numpy as np
import time
from scipy.optimize import minimize


PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a, left=0.0, right=0.0).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z = z - z.max()
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def active_weights_soft(d, temperature):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    dmax = float(np.max(d))
    z = (d - dmax) / max(temperature, 1e-10)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if s < 1e-15:
        w = np.zeros(m)
        w[int(np.argmax(d))] = 1.0
    else:
        w = w / s
    return w


def mirror_step_accelerated(a, a_prev, k_active, eta, momentum):
    """Mirror descent step with Nesterov-style momentum in log-space."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)
    S = float(a.sum())
    grad = (2.0 * float(len(a)) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= 2.0 * LDS / S
    # Log-space update with momentum
    log_a = np.log(np.maximum(a, 1e-15))
    if a_prev is not None and momentum > 0:
        log_prev = np.log(np.maximum(a_prev, 1e-15))
        log_a = log_a + momentum * (log_a - log_prev)
    log_step = log_a - eta * grad
    log_step = log_step - float(np.max(log_step))
    a_new = np.exp(log_step)
    return normalize_a(a_new)


def mirror_descent_accelerated(a_start, max_iters=500, deadline=None,
                                k_list=(1, 2, 3, 5, 8),
                                eta_start=0.1, eta_min=0.001,
                                momentum_start=0.5, momentum_end=0.9):

    a = normalize_a(a_start)
    a_prev = a.copy()

        for k in k_list:

                best_sc = sc
                best_a = q.copy()
                a_prev = a.copy()
                a = q.copy()

                stagnant = 0
            elif np.isfinite(sc) and sc < exact_score_a(a):
                a_prev = a.copy()
                a = q.copy()
                stagnant = 0
        if not improved:
            stagnant += 1
            eta *= 0.9
            if stagnant > 30:
                break
        if it > 0 and it % 50 == 0:
            eta = max(eta_min, eta * 0.95)
    return float(best_sc), best_a

def generate_inits(n, rng, parent, cap=20):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.02
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
        add(1.0 + alpha * (2.0 * x - 1.0) ** 4)

    # Cosine shapes
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
        add(1.0 + alpha * np.cos(np.pi * x))

    # Half-cosine (nonsymmetric)
    for alpha in (1.0, 2.0, 3.0, 5.0):
        add(np.cos(np.pi * x / 2.0) ** alpha + 0.1)
        add(np.cos(np.pi * (1.0 - x) / 2.0) ** alpha + 0.1)
        add(np.cos(np.pi * x / 2.0) ** alpha * (1.0 + 0.3 * np.cos(2 * np.pi * x)))

    # Power shapes (nonsymmetric)
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(x ** p + 0.1)
        add((1.0 - x) ** p + 0.1)
        add(x ** p + (1.0 - x) ** p + 0.1)

    # Bimodal
    for width in (0.1, 0.15, 0.2, 0.3):
        for pos in ((0.25, 0.75), (0.15, 0.85), (0.3, 0.7)):
            add(np.exp(-(((x - pos[0]) / width) ** 2)) + np.exp(-(((x - pos[1]) / width) ** 2)))

    # Exponential
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x) + 0.1)
        add(np.exp(-rate * (1.0 - x)) + 0.1)

    # Sine
    for f in (1, 2, 3, 4):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.7 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)

    return inits


def perturb(a, rng, strength=None):

    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:

                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 3:
        # Smoothed noise in log-space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.5 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 4:
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * np.abs(2.0 * x - 1.0)

    elif mode == 5:
        k = int(rng.integers(5, max(7, n // 100)))
        if k % 2 == 0:
            k += 1

        c = k // 2
        sig = max(1.0, c / 2.0)

        kern = kern / float(kern.sum())

    elif mode == 6:
        d, S = d_from_a(p)
        if d.size > 0:

            if dmax > 1e-12:
                w = active_weights_soft(d, dmax * 0.1)

    elif mode == 7:

    else:

        q = p * np.exp(strength * 0.3 * (rng.standard_normal(1) * np.ones(n)))

    return normalize_a(np.clip(q, 0.0, None))


def optimize_lse_trajectory(a_start, deadline, label=""):
    """Run LSE optimization with increasing beta trajectory."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    stages = [
        (100.0, 0.05),
        (300.0, 0.07),
        (1000.0, 0.10),
        (3000.0, 0.12),
        (10000.0, 0.12),
        (30000.0, 0.13),
        (80000.0, 0.13),
        (200000.0, 0.13),
        (500000.0, 0.10),
        (1000000.0, 0.05),
    ]

    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    cum = 0.0

    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est = 0.02 * max(0.5, len(a) / 8192.0) * max(1.0, np.log2(max(16, len(a))) / np.log2(8192.0))
            mf = int(min(3000, max(64, (0.7 * rem) / max(est, 1e-4))))
            if mf < 64:

            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)

    return float(best_sc), best_a


def spectrum_init(n, rng, num_modes=20):
    """Generate diverse spectral initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    Nf = 1 << (n - 1).bit_length()

    for _ in range(num_modes):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(0.8, 3.0))
        power = 1.0 / ((np.abs(freqs) + 0.5) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - float(sig.mean())
        amp = float(rng.uniform(0.2, 1.5))
        arr = 1.0 + amp * sig
        # Optionally add structure
        if rng.random() < 0.3:
            arr *= (1.0 + 0.5 * np.abs(2.0 * x - 1.0))
        if rng.random() < 0.3:
            arr *= (1.0 + 0.3 * np.cos(2 * np.pi * x))
        inits.append(normalize_a(np.maximum(arr, 0.0)))
    return inits
