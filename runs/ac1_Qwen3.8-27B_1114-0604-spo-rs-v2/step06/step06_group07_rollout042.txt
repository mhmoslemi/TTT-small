            best_a = a.copy()

"""
Step function optimization using block coordinate descent with L-BFGS,
subgradient descent with water-filling projection, and spectral shaping.
Key innovation: block-wise L-BFGS on LSE-smoothed objective enables
efficient local optimization, combined with global subgradient steps.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv_fft(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score_fast(a):
    """Compute the objective value efficiently."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad_full(a, beta):
    """LSE-smoothed objective and gradient for full L-BFGS."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_vals = np.exp(scaled)
    exp_sum = float(exp_vals.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via convolution
    p = exp_vals / exp_sum
    # dL/da_i = 2 * sum_k p_k * (dc_k/da_i)
    # dc_k/da_i = a[k-i] (with factor 2 correction)
    # Use: grad[i] = 2 * sum_k p[k] * a[k-i] (approximately, ignoring boundary effects)
    # More precisely, use convolution
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)[n-1:2*n-1]
    g_L = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

def subgradient_step(a, n_peaks=20, step_scale=0.01):
    """One subgradient step with water-filling projection."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    M_val = float(c.max())
    
    if S < 1e-12:
        return a
    
    # Find top peaks
    if n_peaks >= len(c):
        top_idx = np.arange(len(c))
        weights = np.ones(len(c)) / len(c)
    else:
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        top_vals = c[top_idx]
        # Weight by proximity to max
        weights = top_vals / top_vals.sum()
    
    # Compute subgradient: (1/|K|) * sum_{k in K} grad c_k
    # grad c_k at a: for each i, (grad c_k)_i = 2*a[k-i] (approximately)
    # More precisely: (grad c_k)_i = a[k-i] + a[i] if k-i != i, else a[i]
    # Simplified: (grad c_k)_i ≈ 2 * a[k mod n] with boundary handling
    # Use convolution: sg = conv(p, a_reversed) indexed appropriately
    
    p = np.zeros(M)
    p[top_idx] = weights
    
    # sg[i] = sum_k p[k] * 2 * a[k-i] (for valid k-i)
    # = 2 * (p * a_rev)[i + n - 1] (with proper indexing)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    PP = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(PP * RA, M2_fft)[n-1:2*n-1]
    sg = 2.0 * conv_res
    
    # Full gradient of score
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
    
    # Step size
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a
    
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    sc = 2.0 * n * M_val / S2
    max_step = step_scale
    step = min(max_step, (sc - sc_lb) / (gn * gn + 1e-30))
    step = max(step, 1e-8)
    
    a_new = a - step * grad
    # Water-filling projection onto {a >= 0, sum a = S}
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a
    a_new = a_new * (S / S_new)
    
    return a_new

    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2*n-1).bit_length()
    best_a = a.copy()

$c_k = \sum_j a_j a_{k-j}$ (sum over $j$ such that $0 \leq j < n$ and $0 \leq k-j < n$)

$\frac{\partial c_k}{\partial a_i}$:
- When $j = i$ in the sum: contributes $a_{k-i}$ (if $0 \leq k-i < n$)
- When $k-j = i$ in the sum, i.e., $j = k-i$: contributes $a_i$ (if $0 \leq k-i < n$ and $k-i \neq i$)
- When $j = i$ and $k-j = i$ simultaneously, i.e., $k = 2i$: contributes $a_i$ (only once, not twice)

So: $\frac{\partial c_k}{\partial a_i} = a_{k-i} \cdot \mathbf{1}[0 \leq k-i < n] + a_i \cdot \mathbf{1}[0 \leq k-i < n, k-i \neq i]$

$= a_{k-i} \cdot \mathbf{1}[0 \leq k-i < n] + a_{k-i} \cdot \mathbf{1}[0 \leq k-i < n] \cdot \mathbf{1}[k \neq 2i]$

$= a_{k-i} \cdot \mathbf{1}[0 \leq k-i < n] \cdot (1 + \mathbf{1}[k \neq 2i])$

$= a_{k-i} \cdot \mathbf{1}[0 \leq k-i < n] \cdot (2 - \mathbf{1}[k = 2i])$

For the LSE gradient:
$\frac{\partial L}{\partial a_i} = \sum_k p_k \frac{\partial c_k}{\partial a_i} = \sum_k p_k a_{k-i} (2 - \mathbf{1}[k=2i]) \mathbf{1}[0 \leq k-i < n]$

$\approx 2 \sum_k p_k a_{k-i} \mathbf{1}[0 \leq k-i < n]$ (ignoring the diagonal correction)

The approximation is: $\frac{\partial L}{\partial a_i} \approx 2 (p * a_{rev})[i]$ where the convolution is proper.

$(p * a_{rev})[m] = \sum_k p_k \cdot a_{rev}[m-k] = \sum_k p_k \cdot a_{n-1-(m-k)} = \sum_k p_k \cdot a_{n-1-m+k}$

def subgradient_step(a, n_peaks=20, step_scale=0.01):
    """One subgradient step with projection."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return a
    M_val = float(c.max())

    if n_peaks >= len(c):
        top_idx = np.arange(len(c))
        w = np.ones(len(c)) / len(c)
    else:
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        tv = c[top_idx]
        w = tv / tv.sum()

    p = np.zeros(M)
    p[top_idx] = w

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2f = 1 << (M2 - 1).bit_length()
    PP = np.fft.rfft(p, M2f)
    RA = np.fft.rfft(rev_a, M2f)
    conv_res = np.fft.irfft(PP * RA, M2f)[n - 1:2 * n - 1]
    sg = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a

    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    sc = 2.0 * n * M_val / S2
    step = min(step_scale, (sc - sc_lb) / (gn * gn + 1e-30))
    step = max(step, 1e-8)

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    S_new = float(a_new.sum())
    if S_new < 1e-12:
        return a
    a_new = a_new * (S / S_new)
    return a_new


def fienup_shape(a, n_iters=200, alpha_start=0.1, alpha_end=0.7):
    """Spectral shaping to flatten amplitude spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        return lse_obj_full(x, None, beta, n, M_fft, M)

    def _jac(x):
        return lse_grad_full(x, None, beta, n, M_fft, M)

    try:
        res = minimize(_fun, a0, jac=_jac, method='L-BFGS-B', bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    except Exception:
        res = None
        a_out = a0.copy()

    if res is not None:
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
    else:
        a_out = a0.copy()

    return compute_score(a_out), a_out


def run_block_lbfgs(a0, beta_schedule, n_blocks, block_size, t_budget):
    """Block coordinate descent with L-BFGS and progressive beta."""
    n = len(a0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    n_full_blocks = n // block_size
    actual_block_size = block_size

    improved = 0
    patience = 200
    no_improve_count = 0

    for bi in range(n_blocks):
        if time.monotonic() - t0 > t_budget:
            break

        # Progressive beta
        progress = bi / max(1, n_blocks)
        idx = min(int(progress * len(beta_schedule)), len(beta_schedule) - 1)
        beta = beta_schedule[idx]

        # Select block: cyclic with random offset
        if n_full_blocks > 0:
            block_idx = (bi + rng.integers(0, max(1, n_full_blocks // 10))) % n_full_blocks
        else:
            block_idx = 0
        start = block_idx * actual_block_size
        end = min(start + actual_block_size, n)
        bs = end - start

        if bs < 5:
            continue

        a_block = a[start:end].copy()
        base = a.copy()
        base[start:end] = 0.0  # Zero out block in base

        bounds = [(0.0, 1000.0)] * bs
        maxit = max(10, min(50, int(t_budget / max(1, n_blocks) * 10)))

        try:
            res = minimize(lse_block_obj, a_block,
                           args=(base, start, end, beta, n, M_fft, M),
                           jac=lse_block_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxit, 'maxfun': maxit * 10,
                                    'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 20})
            a[start:end] = np.clip(res.x, 0.0, 1000.0)
        except Exception:
            pass

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            improved += 1
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > patience and bi % 50 == 0:
            # Inject a subgradient step to escape
            a = subgradient_step(a, n_peaks=30, step_scale=0.02)

    return best_a, best_sc


def run_subgradient_loop(a0, n_iters, t_budget, n_peaks_schedule, step_scale):
    """Iterated subgradient descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > t_budget:
            break

        progress = it / max(1, n_iters)
        if len(n_peaks_schedule) > 1:
            idx = min(int(progress * len(n_peaks_schedule)), len(n_peaks_schedule) - 1)
            n_pk = n_peaks_schedule[idx]
        else:
            n_pk = n_peaks_schedule[0]

        a = subgradient_step(a, n_peaks=n_pk, step_scale=step_scale)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)

    # Multi-frequency random
    freqs = rng.integers(1, 15, size=7)
    phases = rng.uniform(0, 2 * np.pi, size=7)
    amps = rng.uniform(0.05, 0.3, size=7)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2.0 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    # Bimodal
    inits.append(np.maximum(0.1 + 0.7 * np.exp(-8.0 * (x - 0.3) ** 2) +
                            0.7 * np.exp(-8.0 * (x - 0.7) ** 2), 0.01))

    # Triangular
    inits.append(np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: multi-phase optimization combining spectral shaping,
    block coordinate L-BFGS, subgradient descent, and full L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 40.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length - try parent length and a few alternatives
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}, parent available: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
        a_work = parent.copy()
    else:
        a_work = np.ones(n_work) * 0.5

    # ============================================================
    # Phase 1: Spectral shaping initialization (15% time)
    # ============================================================
    phase1_end = t_start + 0.15 * (total_budget - 40.0)
    print(f"\n=== Phase 1: Spectral Shaping ===")

    inits = build_diverse_inits(n_work, rng, parent)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a_s = fienup_shape(a0, n_iters=150, alpha_start=0.05, alpha_end=0.6)
        sc = compute_score(a_s)
        update_best(sc, a_s, f"SHAPE i={i}")

    # Also try from parent with different alphas
    if parent is not None:
        if time.monotonic() < phase1_end:
            a_s = fienup_shape(parent, n_iters=100, alpha_start=0.02, alpha_end=0.3)
            sc = compute_score(a_s)
            update_best(sc, a_s, "SHAPE-PARENT")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Subgradient descent (15% time)
    # ============================================================
    phase2_end = t_start + 0.30 * (total_budget - 40.0)
    print(f"\n=== Phase 2: Subgradient Descent ===")

    n_peaks_sched = [10, 20, 50, 100, 200]
    a_sg, sc_sg = run_subgradient_loop(best_a, n_iters=5000,
                                       t_budget=phase2_end - time.monotonic(),
                                       n_peaks_schedule=n_peaks_sched,
                                       step_scale=0.015)
    update_best(sc_sg, a_sg, "SUBGRAD")

    # Interleave with spectral refresh
    if time.monotonic() < phase2_end:
        a_mix = 0.5 * a_sg + 0.5 * best_a
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()
        a_mix = fienup_shape(a_mix, n_iters=50, alpha_start=0.1, alpha_end=0.4)
        sc = compute_score(a_mix)
        update_best(sc, a_mix, "MIX-REFRESH")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Block coordinate L-BFGS (40% time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 40.0)
    print(f"\n=== Phase 3: Block L-BFGS ===")

    remaining3 = phase3_end - time.monotonic()
    print(f"  Time available: {remaining3:.0f}s")

    # Progressive beta schedule: high to low
    beta_schedule = [5000, 2000, 1000, 500, 200, 100, 50, 30, 20, 15, 10]

    # Multiple passes with different block sizes
    block_configs = [
        (64, 3000),   # (block_size, n_blocks)
        (128, 2000),
        (256, 1500),
    ]

    for bsize, n_blks in block_configs:
        if time.monotonic() > phase3_end:
            break
        rem = phase3_end - time.monotonic()
        if rem < 60:
            break
        print(f"  Block size={bsize}, n_blocks={n_blks}, remaining={rem:.0f}s")
        a_blk, sc_blk = run_block_lbfgs(best_a, beta_schedule, n_blks, bsize, rem)
        update_best(sc_blk, a_blk, f"BLOCK b={bsize}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Full L-BFGS polish (20% time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 40.0)
    print(f"\n=== Phase 4: Full L-BFGS ===")

    betas_full = [500, 200, 100, 50]
    for beta in betas_full:
        if time.monotonic() > phase4_end:
            break
        rem = phase4_end - time.monotonic()
        if rem < 60:
            break
        print(f"  beta={beta}, remaining={rem:.0f}s")
        maxit = min(3000, int(rem * 5))
        sc, a_opt = run_full_lbfgs(best_a, beta, maxiter=maxit,
                                   maxfun=maxit * 20, t_budget=rem)
        update_best(sc, a_opt, f"LBFGS b={beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final refinement (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Refinement ===")

    # Final block L-BFGS with very low beta
    if time.monotonic() < deadline - 60:
        rem = deadline - time.monotonic() - 30
        beta_final = [20, 10, 5, 3]
        a_fin, sc_fin = run_block_lbfgs(best_a, beta_final, 1000, 128, rem)
        update_best(sc_fin, a_fin, "FINAL-BLOCK")

    # Final subgradient with small steps
    if time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic() - 15
        a_fin2, sc_fin2 = run_subgradient_loop(best_a, n_iters=2000,
                                               t_budget=rem,
                                               n_peaks_schedule=[50, 100],
                                               step_scale=0.005)
        update_best(sc_fin2, a_fin2, "FINAL-SG")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(2):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.01 * (1 + trial)
            a_p = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = a_p.sum()
            if S_p < 1e-12:
                break
            a_p = a_p / S_p
            a_r = fienup_shape(a_p, n_iters=30, alpha_start=0.05, alpha_end=0.3)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART {trial}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
