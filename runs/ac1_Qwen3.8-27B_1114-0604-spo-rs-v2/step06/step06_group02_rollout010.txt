```python
"""
Focused optimizer: multi-stage L-BFGS with LSE smoothing, 
peak-targeted coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

# LSE-smoothed objective and gradient
def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    # Gradient via FFT
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out
```

```python
"""
Corrected multi-stage optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key fix: corrected gradient (missing factor of 2 on h term).
Adds: multi-length exploration, tuned LSE beta schedule, peak-targeted CD finishing.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad_corrected(a, beta):
    """LSE-smoothed objective with CORRECTED gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()

    # h[m] = sum_{r=0}^{n-1} a[r] * p[m+r]
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    # CORRECTED: grad[m] = 4n*h[m]/S^2 - 4n*L/S^3
    grad = 4.0 * n * (h / S2 - L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad_corrected, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def peak_targeted_cd(a, n_sweeps=30, deadline=None):
    """Coordinate descent targeting top peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_max = c.max()
        # Find peaks above 99.5% of max
        active_peaks = np.where(c >= 0.995 * c_max)[0]
        # Limit to top 8 peaks
        if len(active_peaks) > 8:
            peaks_c = c[active_peaks]
            top_order = np.argsort(peaks_c)[::-1][:8]
            active_peaks = active_peaks[top_order]

        for k in active_peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            total = contribs.sum()
            if total < 1e-15:
                continue

            # Target top 20 contributing elements
            top_count = min(20, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old = a[j]
                if a_old < 1e-12:
                    continue
                best_val = a_old
                best_local = sc
                for f in [0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.3]:
                    cv = a_old * f
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old
                        continue
                    a_tmp = a / S_tmp
                    sc_tmp = compute_score(a_tmp)
                    if sc_tmp < best_local:
                        best_local = sc_tmp
                        best_val = cv
                a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S

    return best_a


def fienup_reflect(a, n_iters=300, alpha_init=0.7):
    """Gerchberg-Saxton / Fienup with reflection for non-negativity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init * (1.0 - progress) ** 1.5 + 0.05

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target: flat spectrum
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-25)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Reflect into non-negative orthant
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_line_search(a, n_iters=5000, deadline=None):
    """Custom subgradient method with backtracking line search for exact max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()
    step = 0.1

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        S2 = S * S
        S3 = S2 * S
        c_max = float(c.max())
        sc = 2.0 * n * c_max / S2

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Soft max: weight peaks by exp(beta*(c - c_max)) with moderate beta
        beta_scene = 50.0
        scaled = beta_scene * (c - c_max)
        scaled = np.clip(scaled, -100, 0)
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute gradient using soft weights (like LSE but we track exact max for score)
        a_rev = a[::-1]
        M2 = 2 * n - 1
        PA = np.fft.rfft(w, M_fft)
        AA_rev = np.fft.rfft(a_rev, M_fft)
        h_full = np.fft.irfft(AA_rev * PA, M_fft)[:M2]
        h = h_full[n - 1:2 * n - 1]

        # LSE value for gradient (not for score)
        L_soft = c_max + float(np.log(w.sum())) / beta_scene
        grad = 4.0 * n * (h / S2 - L_soft / S3)

        # Backtracking line search
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad_dir = -grad / grad_norm

        a_candidate = a
        improved = False
        for ls in range(15):
            a_try = a + step * grad_dir
            a_try = np.maximum(a_try, 0.0)
            S_try = a_try.sum()
            if S_try < 1e-12:
                step *= 0.5
                continue
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < sc:
                a = a_try
                a_candidate = a_try
                improved = True
                break
            step *= 0.7
        if not improved:
            step *= 0.5
        step = min(step, 0.5)

    return best_a


def try_length(parent, target_n, rng):
    """Resample parent to target length."""
    if parent is None or len(parent) < 8:
        return np.ones(target_n) / target_n
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a

    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
