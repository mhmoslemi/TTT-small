def fast_conv(a):
    """Compute full convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B with log-sum-exp smoothing, multiplicative weight updates for peak
reduction, coordinate descent polish, and multi-start with diverse initializations.
Exploits the non-convex structure by alternating between smooth approximations
and direct peak-targeting updates.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    L = (1/beta)*log(sum exp(beta*c_k)) approximates max(c).
    Score = 2n*L/S^2.
    Gradient uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta

    p = exp_scaled / exp_scaled.sum()

    # h_j = sum_m a[m] * p[m+j] via FFT cross-correlation
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (2.0 * h / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B on normalized non-negative sequence."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

    bounds = [(0.0, 100.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    a_out = a_out / S
    sc = compute_score(a_out)
    return sc, a_out

    """

        c_max = float(c.max())

        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()

        M2_fft = M_fft
        PA = np.fft.rfft(p, M2_fft)

            a_new = np.maximum(a_new, 1e-15)

def coordinate_descent_polish(a, n_sweeps=30, time_budget=120.0):
    """
    Cyclic coordinate descent: for each element, try a local search
    to minimize the score while keeping others fixed.
    Uses golden-section search in a small window around current value.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_budget:
            break

        for j in range(n):
            if time.monotonic() - t_start > time_budget:
                break
            a_old = a[j]
            if a_old < 1e-12:
                # Try setting to small positive values
                for f in [0.001, 0.01, 0.05]:
                    a[j] = f
                    S_tmp = a.sum()
                    a_tmp = a / S_tmp
                    sc = compute_score(a_tmp)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_tmp.copy()
                a[j] = a_old
                continue

            # Try a range of multipliers
            best_val = a_old
            best_local = best_sc
            for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                a[j] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_tmp = a / S_tmp
                sc = compute_score(a_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f

            if best_val != a_old:
                a[j] = best_val
                S = a.sum()
                a = a / S
                if best_local < best_sc:
                    best_sc = best_local
                    best_a = a.copy()
            else:
                a[j] = a_old

        # Occasionally re-full-normalize
        if (sweep + 1) % 5 == 0:
            a = a / a.sum()

    return best_a


def fienup_iterations(a, n_iters=300, alpha_start=0.7, alpha_end=0.05):
    """
    Fienup/Gerchberg-Saxton: alternate between time-domain (non-negativity)
    and frequency-domain (shape targeting) projections.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    for it in range(n_iters):
        progress = it / n_iters
        alpha = alpha_start * (1 - progress) + alpha_end * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target: slightly non-flat (peaked at DC to help non-negativity)
        P_target = P_mean * (1.0 + 0.3 * np.exp(-np.arange(len(P)) * 0.5 / max(n, 1)))
        P_target = P_target / P_target.mean() * P_mean

        P_blend = (1 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-20)

        ratio = np.sqrt(P_blend / (P + 1e-30)) ** alpha
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def make_uniform(n):
    return np.ones(n) / n


def make_edge_weighted(n, edge_frac=0.3):
    """More mass at edges, less in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 + edge_frac * (2 * x - 1) ** 2
    a = np.maximum(a, 0.1)
    return a / a.sum()


def make_random(rng, n):
    a = np.abs(rng.standard_normal(n))
    a = a + 0.1
    return a / a.sum()


def make_bimodal(n, center_frac=0.25):
    """Two bumps away from center."""
    x = np.linspace(0, 1, n)
    c1 = center_frac
    c2 = 1.0 - center_frac
    sigma = 0.08
    a = np.exp(-((x - c1) / sigma) ** 2) + np.exp(-((x - c2) / sigma) ** 2) + 0.05
    return a / a.sum()


def make_triangular(n):
    """Triangular shape (higher in middle)."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2 * x - 1)
    a = a + 0.1
    return a / a.sum()


def make_antisymmetric(n, eps=0.1):
    """Slightly anti-symmetric: a_j = 1 + eps*(j - (n-1)/2)/(n/2)."""
    x = np.arange(n).astype(float)
    a = 1.0 + eps * (2 * x / (n - 1) - 1) if n > 1 else np.array([1.0])
    a = np.maximum(a, 0.01)
    return a / a.sum()

    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                sc_p = compute_score(parent)
                print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
                update(sc_p, parent, "parent")
                parent = parent / parent.sum()
    except Exception as e:
        print(f"Parent loading issue: {e}")

    if best_a is None:
        n_init = 512
        best_a = make_uniform(n_init)
        best_score = compute_score(best_a)
        print(f"Initial (uniform n={n_init}): {best_score:.8f}")

    print(f"Starting best: {best_score:.8f}")

    n = len(best_a)

    # ==================================================================
    # Phase 1: Quick Fienup + L-BFGS on parent (15% time)
    # ==================================================================
    p1_end = t_start + 0.15 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup + quick L-BFGS (until t={p1_end - t_start:.0f}s) ===")

    a_f = fienup_iterations(best_a, n_iters=200)
    update(compute_score(a_f), a_f, "FIENUP-1")

    sc, a_lb = run_lbfgs(best_a, beta=500, maxiter=5000, maxfun=100000)
    update(sc, a_lb, "LBFGS-500")

    sc, a_lb = run_lbfgs(a_lb, beta=2000, maxiter=8000, maxfun=200000)
    update(sc, a_lb, "LBFGS-2k")

    # ==================================================================
    # Phase 2: Adaptive L-BFGS schedule (25% time)
    # ==================================================================
    p2_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Adaptive L-BFGS (until t={p2_end - t_start:.0f}s) ===")

    current = best_a.copy()
    betas = [5000, 10000, 30000, 80000, 200000]
    for i, beta in enumerate(betas):
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 20:
            break
        maxiter = int(min(15000, remaining * 15 + 2000))
        maxiter = max(maxiter, 3000)
        print(f"  Beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(current, beta, maxiter=maxiter, maxfun=maxiter * 25)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            current = a_opt.copy()
            print(f"    NEW BEST via LBFGS-{beta}")

    # ==================================================================
    # Phase 3: Multiplicative peak reduction (15% time)
    # ==================================================================
    p3_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Multiplicative peak reduction (until t={p3_end - t_start:.0f}s) ===")

    if time.monotonic() < p3_end:
        n_iter_mp = int(min(3000, (p3_end - time.monotonic()) * 8))
        a_mp = multiplicative_peak_reduce(best_a, n_iters=n_iter_mp,
                                          beta_smooth=200.0, eta_init=0.04)
        sc = compute_score(a_mp)
        update(sc, a_mp, "MULTI-PEAK")
        # Follow up with L-BFGS
        if sc < best_score + 0.01:
            sc2, a2 = run_lbfgs(a_mp, beta=10000, maxiter=5000, maxfun=100000)
            update(sc2, a2, "MULTI+LBFGS")

    # ==================================================================
    # Phase 4: Coordinate descent (15% time)
    # ==================================================================
    p4_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Coordinate descent (until t={p4_end - t_start:.0f}s) ===")

    remaining_cd = max(10, p4_end - time.monotonic())
    n_sweeps_cd = int(min(40, remaining_cd * 2))
    a_cd = coordinate_descent_polish(best_a, n_sweeps=n_sweeps_cd,
                                     time_budget=remaining_cd * 0.8)
    sc = compute_score(a_cd)
    update(sc, a_cd, "COORD-DESC")

    if time.monotonic() < p4_end:
        sc2, a2 = run_lbfgs(a_cd, beta=50000, maxiter=8000, maxfun=200000)
        update(sc2, a2, "CD+LBFGS-50k")

    # ==================================================================
    # Phase 5: Multi-restart with diverse inits (20% time)
    # ==================================================================
    p5_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 5: Multi-restart (until t={p5_end - t_start:.0f}s) ===")

    inits = []
    # Perturbation of best
    inits.append(best_a * (1 + 0.03 * rng.standard_normal(n)))
    # Edge-weighted
    inits.append(make_edge_weighted(n, 0.4))
    # Bimodal
    inits.append(make_bimodal(n, 0.2))
    inits.append(make_bimodal(n, 0.35))
    # Antisymmetric
    inits.append(make_antisymmetric(n, 0.15))
    inits.append(make_antisymmetric(n, 0.25))
    # Random
    inits.append(make_random(rng, n))
    # Triangular (inverted)
    x = np.linspace(0, 1, n)
    inits.append((1.0 + 0.5 * x) / (1.0 + 0.5 * x).sum())
    inits.append((1.0 + 0.5 * (1 - x)) / (1.0 + 0.5 * (1 - x)).sum())
    # Strong edge emphasis
    for ef in [0.5, 0.7, 1.0]:
        inits.append(make_edge_weighted(n, ef))

    # Shuffle
    rng.shuffle(inits)

    n_restarts_done = 0
    for idx, init in enumerate(inits):
        if time.monotonic() > p5_end:
            break
        remaining = p5_end - time.monotonic()
        if remaining < 15:
            break

        a_init = np.maximum(init, 0.0)
        S = a_init.sum()
        if S < 1e-6:
            continue
        a_init = a_init / S

        # Quick L-BFGS from this init
        maxiter_r = int(min(8000, remaining * 10))
        maxiter_r = max(maxiter_r, 2000)
        t0 = time.monotonic()
        sc_r, a_r = run_lbfgs(a_init, beta=3000, maxiter=maxiter_r, maxfun=maxiter_r * 20)
        dt_r = time.monotonic() - t0
        n_restarts_done += 1
        if sc_r < best_score:
            best_score = sc_r
            best_a = a_r.copy()
            print(f"  Restart {idx}: {sc_r:.8f} (t={dt_r:.1f}s) *** NEW BEST ***")
        else:
            print(f"  Restart {idx}: {sc_r:.8f} (t={dt_r:.1f}s)")

        # If improved, follow up with higher beta
        if sc_r < best_score + 0.005 and time.monotonic() < p5_end - 20:
            sc_r2, a_r2 = run_lbfgs(a_r, beta=30000, maxiter=5000, maxfun=100000)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_a = a_r2.copy()
                print(f"  Restart {idx} beta=30k: {sc_r2:.8f} *** NEW BEST ***")

    # ==================================================================
    # Phase 6: Try different lengths (10% time)
    # ==================================================================
    p6_end = deadline
    print(f"\n=== Phase 6: Length exploration (until t={p6_end - t_start:.0f}s) ===")

    best_n = len(best_a)
    # Try smaller and larger lengths
    length_candidates = []
    for ln in [256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]:
        if ln != best_n:
            length_candidates.append(ln)

    # Sort by proximity to best_n
    length_candidates.sort(key=lambda x: abs(x - best_n))

    for ln in length_candidates[:4]:
        if time.monotonic() > p6_end:
            break
        remaining = p6_end - time.monotonic()
        if remaining < 30:
            break

        print(f"  Trying length {ln} (remaining={remaining:.0f}s)")
        # Interpolate best_a to this length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, ln)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp = a_interp / a_interp.sum()

        # Quick optimization
        maxiter_l = int(min(6000, remaining * 5))
        sc_l, a_l = run_lbfgs(a_interp, beta=5000, maxiter=maxiter_l, maxfun=maxiter_l * 15)
        print(f"    Length {ln}: score={sc_l:.8f}")
        if sc_l < best_score:
            best_score = sc_l
            best_a = a_l.copy()
            print(f"    *** NEW BEST at length {ln} ***")

    # ==================================================================
    # Phase 7: Final polish (remaining time)
    # ==================================================================
    print(f"\n=== Phase 7: Final polish ===")

    if time.monotonic() < deadline - 10:
        remaining = deadline - 10 - time.monotonic()
        maxiter_f = int(min(10000, remaining * 10))
        print(f"  Final L-BFGS beta=100k, maxiter={maxiter_f}")
        sc_f, a_f = run_lbfgs(best_a, beta=100000, maxiter=maxiter_f, maxfun=maxiter_f * 20)
        update(sc_f, a_f, "FINAL-100k")

    if time.monotonic() < deadline - 10:
        sc_f2, a_f2 = run_lbfgs(best_a, beta=500000, maxiter=5000, maxfun=100000)
        update(sc_f2, a_f2, "FINAL-500k")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd_final = coordinate_descent_polish(best_a, n_sweeps=10,
                                               time_budget=max(5, deadline - 5 - time.monotonic()))
        sc_cd = compute_score(a_cd_final)
        update(sc_cd, a_cd_final, "FINAL-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out_a = np.maximum(best_a, 0.0)
    out_a = np.minimum(out_a, 1000.0)
    # Ensure sum >= 0.01
    S_out = out_a.sum()
    if S_out < 0.01:
        out_a = out_a * (1.0 / S_out if S_out > 1e-15 else 100.0)

    out = [float(x) for x in out_a]

    # Verify
    verified = float(evaluate_sequence(out))
    final_sc = min(best_score, verified) if verified != np.inf else best_score
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
