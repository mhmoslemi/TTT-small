"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Nesterov momentum subgradient descent with exact max
2. LP-based active set refinement (solving small QPs)
3. Block coordinate descent with FFT
4. Multi-length exploration and perturbation restarts
Different from prior approach: uses momentum, block methods, and spectral deflation.
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _self_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _cross_conv(a, b):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    cmax = float(np.max(c))
    return 2.0 * n * cmax / (S * S)


def _normalize(a):
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S


def _resample(a, n_target):
    m = len(a)
    if m == n_target:
        return a.copy()
    if m == 0:
        return np.ones(n_target) / n_target
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_target)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _subgrad_max_conv(a, top_k=3):
    """Compute subgradient of score = 2*n*max(conv)/S^2 w.r.t. a.
    Returns (score, gradient) where gradient points in direction to decrease score.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _self_conv(a)
    cmax = float(np.max(c))
    d = c * n / (S * S)  # d[k] = n*c[k]/S^2, score = 2*max(d)
    dmax = float(np.max(d))

    # Find top_k active lags
    idx = np.argpartition(d, -top_k)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        # d[k] = n/S^2 * sum_i a[i]*a[i+k] (valid range)
        # dd[k]/da[j] = n/S^2 * (a[j-k] + a[j+k]) - 2/S * d[k]
        # where a[.] = 0 for out-of-range
        if k == 0:
            shifted = 2.0 * a
        else:
            shifted = np.zeros(n)
            # a[j-k] for j in [k, n): a[j-k]
            shifted[k:] += a[:n - k]
            # a[j+k] for j in [0, n-k): a[j+k]
            shifted[:n - k] += a[k:]
        grad += shifted
    grad *= n / (S * S) / float(top_k)
    # Subtract the -2/S * d_max * (1/S) term from the S^2 denominator
    # Actually: d[k] = n*c[k]/S^2, dd/da[j] = n*(a[j-k]+a[j+k])/S^2 - 2*n*c[k]/S^3
    # = n*(a[j-k]+a[j+k])/S^2 - 2*d[k]/S
    grad -= (2.0 / S) * dmax / float(top_k)
    # Final score gradient: d(score)/da = 2 * dd/da (since score = 2*max(d))
    grad *= 2.0 / float(top_k)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return 2.0 * dmax, grad


def _momentum_descent(a_init, max_iters, t_deadline, top_k=3, momentum=0.9):
    """Nesterov momentum subgradient descent."""
    a = _normalize(a_init)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    v = np.zeros(n)
    eta = 1.0 / (np.linalg.norm(a) * n ** 0.5 + 1e-10)
    eta = min(eta, 0.1)

    for it in range(int(max_iters)):
        if time.monotonic() > t_deadline:
            break
        # Nesterov lookahead
        a_look = _normalize(a - momentum * v)
        sc, grad = _subgrad_max_conv(a_look, top_k)
        # Step
        v = momentum * v + grad
        a_new = _normalize(a - eta * v)
        sc_new = _score(a_new)
        if np.isfinite(sc_new) and sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new
        # Adaptive step size
        if it % 5 == 4:
            if sc_new < sc * 0.999:
                eta *= 1.2
            elif sc_new > sc * 1.001:
                eta *= 0.7
            eta = max(1e-8, min(eta, 1.0))
        if it % 100 == 0:
            # Periodic restart of momentum
            v = v * 0.5
    return best_sc, best_a


def _lp_active_set_refine(a_init, max_iters, t_deadline, k_active=5):
    """Refine using LP: fix a, find b maximizing sum(b) s.t. conv(a,b)[k]<=M for active k.
    Then blend a and b."""
    a = _normalize(a_init)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(int(max_iters)):
        if time.monotonic() > t_deadline:
            break
        c = _self_conv(a)
        d = c * n / (float(np.sum(a)) ** 2)
        dmax = float(np.max(d))

        # Get active set (top-k lags)
        idx = np.argpartition(d, -k_active)[-k_active:]

        # Build LP: max sum(b) s.t. A_active @ b <= M_i, b >= 0
        # A_active[k, j] = a[j-k] + a[j+k] (the cross-conv kernel for lag k)
        # conv(a,b)[k] = sum_j a[j]*b[j+k] = sum_j a[j-k]*b[j]
        # So conv(a,b)[k] = (a_reversed_conv_kernel) @ b
        # Specifically: conv(a,b)[k] = sum_j b[j] * a[j-k] (where j-k in [0,n-1])
        # = sum_j b[j] * a_rev[j + (n-1-k)] ... let me just compute directly

        M = dmax * float(n) / n  # M = cmax = dmax * S^2 / n, but we normalized S=1
        # Actually with S=1: d[k] = n*c[k], so c[k] = d[k]/n, M = cmax = dmax/n
        M = dmax  # in d-space, constraint is d_conv(a,b)[k] <= dmax

        # conv(a,b) with S_a = 1: d_conv[k] = n * conv(a,b)[k]
        # conv(a,b)[k] = sum_j a[j]*b[j+k] (proper indexing)
        # = sum_j b[j] * a[j-k]
        # So the matrix row for constraint k: row[j] = a[j-k] if j-k in [0,n-1] else 0
        #           row[j] += a[j+k] if j+k in [0,n-1] else 0  (no! conv is one-directional)
        # Actually np.convolve(a,b)[k] = sum_i a[i]*b[k-i] for valid i
        # = sum_j b[j] * a[k-j] for valid j
        # So row_k of the matrix: entry j = a[k-j] if 0<=k-j<n else 0

        # For the LP to be tractable, use only top 3-5 constraints
        A_rows = []
        b_vec = []
        for k in idx:
            k = int(k)
            row = np.zeros(n)
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    row[j] = a[i]
            A_rows.append(row)
            b_vec.append(M)

        A_mat = np.array(A_rows)
        b_vec = np.array(b_vec)

        # Solve: max 1^T b s.t. A_mat @ b <= b_vec, b >= 0
        # Using scipy: minimize(-1^T b, s.t. A_mat @ b <= b_vec, b>=0)
        try:
            res = minimize(
                lambda x: -np.sum(x),
                np.ones(n) / n,
                method='L-BFGS-B',
                bounds=[(0, 10)] * n,
                options={'maxiter': 200, 'maxfun': 500}
            )
            b_sol = np.maximum(res.x, 0)
        except Exception:
            b_sol = a.copy()

        # Blend
        for t_blend in (0.1, 0.3, 0.5, 0.7):
            q = _normalize((1 - t_blend) * a + t_blend * b_sol)
            sc_q = _score(q)
            if np.isfinite(sc_q) and sc_q < best_sc:
                best_sc = sc_q
                best_a = q.copy()
                a = q
                break
        else:
            # No improvement, try smaller blend
            a = _normalize(0.95 * a + 0.05 * b_sol)
            sc_a = _score(a)
            if np.isfinite(sc_a) and sc_a < best_sc:
                best_sc = sc_a
                best_a = a.copy()

    return best_sc, best_a

def _spectral_deflation(a_init, max_iters, t_deadline):
    """Work in frequency domain: try to flatten |FFT(a)|^2 to reduce IFFT peaks."""
    a = _normalize(a_init)
    n = len(a)
    N = _next_pow2(2 * n - 1)

        pmax = float(np.max(power))
        if pmax < 1e-30:
            break
        # Deflate: reduce power at peaks, increase at valleys
        # a_new = IFFT(A * (power/pmax)^alpha) with alpha > 1 flattens
        # But we need to keep a non-negative and real
        for alpha in (0.3, 0.5, 0.8, 1.2, 1.5):

            A_mod = A * np.power(power / pmax, alpha - 1.0) * (1.0 / (np.power(power / pmax, alpha - 1.0).sum() + 1e-30) * power.sum())
            # Actually simpler: blend with uniform spectrum
            t_blend = 0.05 * (alpha - 0.3)
            A_unif = np.zeros_like(A)
            A_unif[0] = 1.0  # DC component for mean
            A_new = (1 - t_blend) * A + t_blend * A_unif * np.mean(np.abs(A))

def _lse_optimize(a_init, beta, t_deadline):
    """L-BFGS-B on log-sum-exp smoothed objective."""
    a = _normalize(a_init)
    n = len(a)

    def objective(x):
        x = np.maximum(x, 0)
        S = float(np.sum(x))
        if S < 1e-8:
            return 1e10, np.zeros(n)
        c = _self_conv(x)
        d = c * n / (S * S)
        dmax = float(np.max(d))
        z = beta * (d - dmax)
        z = np.clip(z, -50, 50)
        e = np.exp(z)
        sum_e = float(np.sum(e))
        if sum_e < 1e-30:
            return 1e10, np.zeros(n)
        w = e / sum_e
        L = dmax + np.log(sum_e) / beta
        score = 2.0 * L
        # Gradient
        h = _cross_conv(x, w)
        LDS = float(np.dot(w, d))
        grad = (4.0 * n * h) / (S * S)
        grad -= (4.0 * LDS) / S
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        return score, grad

    try:
        rem = max(1.0, t_deadline - time.monotonic())
        maxfun = int(min(5000, max(200, rem * 800)))
        res = minimize(
            objective, a, jac=True, method='L-BFGS-B',
            bounds=[(0, 10)] * n,
            options={'maxiter': maxfun // 2, 'maxfun': maxfun, 'ftol': 1e-14, 'gtol': 1e-11}
        )
        a_out = _normalize(res.x)
    except Exception:
        a_out = a

    sc = _score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _score(a)
    return sc, a_out


def _perturb(a, rng, strength=0.2):
    """Perturb the sequence for restarts."""
    n = len(a)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.05, 0.2)
        q = (1 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        q = a[::-1].copy()
    elif mode == 3:
        # Smooth
        k = int(rng.integers(5, max(7, min(31, n // 500))))
        if k % 2 == 0:
            k += 1
        if k < 3:
            k = 3
        if k > n:
            k = n if n % 2 == 1 else n - 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    elif mode == 4:
        # Frequency perturbation
        N = _next_pow2(2 * n - 1)
        A = np.fft.rfft(a, N)
        freqs = np.abs(np.fft.rfftfreq(N) * N)
        mask = freqs < n * 0.1
        A[mask] *= np.exp(strength * 0.3 * rng.standard_normal(mask.sum()))
        q = np.fft.irfft(A, N)[:n]
    else:
        # Targeted: reduce peak contributors
        c = _self_conv(a)
        k_max = int(np.argmax(c))
        q = a.copy()
        # Identify elements contributing most to lag k_max
        if k_max == 0:
            contrib = a.copy()
        else:
            contrib = np.zeros(n)
            contrib[k_max:] += a[:n - k_max]
            contrib[:n - k_max] += a[k_max:]
        contrib_norm = contrib / (np.max(contrib) + 1e-15)
        q = a * np.exp(-strength * 0.5 * contrib_norm)

    return _normalize(np.maximum(q, 0))

            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 32:
                parent = p
    except Exception:
        pass

    print("Parent: {} (len={})".format(parent is not None, parent.size if parent is not None else 0))

    # Primary length
    n = 10240
    if parent is not None:
        n = parent.size
        if n > 16384:
            n = 16384
        if n < 4096:
            n = 4096

    a = _init_from_parent(parent, n)
    best_sc = _score(a)
    best_seq = a.copy()
    print("Initial: score={:.8f} n={}".format(best_sc, n))

    # === Phase 1: LSE optimization with increasing beta (30% time) ===
    t1 = t0 + 0.30 * budget_s
    betas = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    for beta in betas:
        if time.monotonic() > t1 - 10:
            break
        rem = t1 - time.monotonic()
        if rem < 10:
            break
        sc, a2 = _lse_optimize(a, beta, t1)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  LSE b={}: {:.8f}".format(beta, best_sc))
        a = a2

    # === Phase 2: Momentum subgradient descent (20% time) ===
    t2 = t0 + 0.50 * budget_s
    sc, a2 = _momentum_descent(best_seq, max_iters=5000, t_deadline=t2, top_k=3, momentum=0.85)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = a2.copy()
        print("Momentum: {:.8f}".format(best_sc))

    # === Phase 3: LP active set refinement (15% time) ===
    t3 = t0 + 0.65 * budget_s
    sc, a2 = _lp_active_set_refine(best_seq, max_iters=200, t_deadline=t3, k_active=3)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = a2.copy()
        print("LP refine: {:.8f}".format(best_sc))

    # === Phase 4: Spectral deflation (10% time) ===
    t4 = t0 + 0.75 * budget_s
    sc, a2 = _spectral_deflation(best_seq, max_iters=200, t_deadline=t4)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = a2.copy()
        print("Spectral: {:.8f}".format(best_sc))

    # === Phase 5: Block coordinate descent (10% time) ===
    t5 = t0 + 0.85 * budget_s
    sc, a2 = _block_coordinate_descent(best_seq, max_iters=2000, t_deadline=t5, block_size=max(8, n // 128))
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = a2.copy()
        print("Block CD: {:.8f}".format(best_sc))

    # === Phase 6: More LSE + momentum (5% time) ===
    t6 = t0 + 0.90 * budget_s
    for beta in [500000, 2000000]:
        if time.monotonic() > t6 - 5:
            break
        sc, a2 = _lse_optimize(best_seq, beta, t6)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  LSE2 b={}: {:.8f}".format(beta, best_sc))
        a = a2
    sc, a2 = _momentum_descent(best_seq, max_iters=2000, t_deadline=t6, top_k=5, momentum=0.8)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = a2.copy()
        print("Momentum2: {:.8f}".format(best_sc))

    # === Phase 7: Multi-length exploration (3% time) ===
    t7 = t0 + 0.93 * budget_s
    for n_alt in [8192, 12288, 16384]:
        if time.monotonic() > t7 - 20:
            break
        if n_alt == n:
            continue
        a_alt = _normalize(_resample(best_seq, n_alt))
        sc_alt = _score(a_alt)
        if np.isfinite(sc_alt) and sc_alt < best_sc:
            best_sc = sc_alt
            best_seq = a_alt.copy()
            print("  n={}: {:.8f}".format(n_alt, best_sc))
        # Quick optimize
        for beta in [10000, 100000]:
            if time.monotonic() > t7 - 15:
                break
            sc, a2 = _lse_optimize(a_alt, beta, t7)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  n={} b={}: {:.8f}".format(n_alt, beta, best_sc))
            a_alt = a2
        sc, a2 = _momentum_descent(a_alt, 500, t7, top_k=3)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  n={} mom: {:.8f}".format(n_alt, best_sc))

    # === Phase 8: Perturbation restarts (2% time) ===
    t8 = deadline
    for restart in range(5):
        if time.monotonic() > t8 - 20:
            break
        p = _perturb(best_seq, rng, strength=rng.uniform(0.05, 0.2))
        chunk_end = t8 - 15.0
        sc, a2 = _lse_optimize(p, 50000, chunk_end)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  Perturb {}: {:.8f}".format(restart, best_sc))
        sc, a2 = _momentum_descent(p, 500, chunk_end, top_k=3)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  Perturb {} mom: {:.8f}".format(restart, best_sc))

    # Final cleanup
    out_arr = best_seq.astype(np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} t={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t0))
    return out
