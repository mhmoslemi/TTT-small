def lp_direction(a, M, active_peaks):
    """
    Find the LP direction: maximize sum(b) s.t. (a*b)_k <= M for k in active_peaks, b >= 0.
    Uses the dual (m variables, n constraints) for efficiency.
    """
    m = len(active_peaks)
    n = len(a)
    
    # Dual: minimize M * sum(y) s.t. sum_i a[k_i - j] * y_i >= 1 for all j, y >= 0
    # For m=1: y_1 = 1/min(a), b_j ∝ a[k_1 - j]
    # For m>1: solve small LP
    
    if m == 1:
        k = active_peaks[0]
        # b_j ∝ a[k - j] for j where 0 <= k-j <= n-1
        idx = np.arange(0, n)
        valid = (k - idx >= 0) & (k - idx < n)
        b = np.zeros(n)
        b[valid] = a[k - idx[valid]]
        return b / b.sum()
    
    # For m > 1, use scipy linprog on the dual
    # Variables: y_1, ..., y_m (non-negative)
    #Objective: minimize M * sum(y)
    # Constraints: for each j, sum_i a[k_i - j] * y_i >= 1
    # i.e., -sum_i a[k_i - j] * y_i <= -1
    
    # Build constraint matrix (n x m)
    A = np.zeros((n, m))
    for i, k in enumerate(active_peaks):
        idx = np.arange(0, n)
        valid = (k - idx >= 0) & (k - idx < n)
        A[valid, i] = a[k - idx[valid]]
    
    # Minimize c^T y s.t. A^T y >= 1, y >= 0
    # => Minimize c^T y s.t. -A^T y <= -1, y >= 0
    c = M * np.ones(m)
    A_ub = -A.T  # shape (n, m)
    b_ub = -np.ones(n)
    bounds = [(0, None)] * m
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        y = result.x
        # Recover primal b from complementary slackness
        # Or just use b_j = sum_i y_i * a[k_i - j] (this gives the optimal b)
        b = np.zeros(n)
        for i, k in enumerate(active_peaks):
            idx = np.arange(0, n)
            valid = (k - idx >= 0) & (k - idx < n)
            b[valid] += y[i] * a[k - idx[valid]]
        b = np.maximum(b, 0)
        S = b.sum()
        if S > 1e-12:
            b /= S
        else:
            b = a.copy()
        return b
    
    # Fallback: average of individual peak directions
    b = np.zeros(n)
    for k in active_peaks:
        idx = np.arange(0, n)
        valid = (k - idx >= 0) & (k - idx < n)
        b[valid] += a[k - idx[valid]]
    b = np.maximum(b, 0)
    S = b.sum()
    if S > 1e-12:
        b /= S
    return b

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Symmetric sequence optimization
2. LP dual-based direction finding (cutting-plane)
3. Mirror descent (multiplicative) with adaptive step
4. LSE smoothing + L-BFGS-B
5. Nesterov accelerated exact subgradient
6. Multi-length search and perturbation restarts
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(m):
    m = max(1, int(m))
    p = 1
    while p < m:
        p <<= 1
    return p


def _self_conv(a):
    """Compute linear self-convolution a*a using FFT."""
    n = np.size(a)
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _cross_conv(a, b):
    """Compute linear cross-convolution a*b using FFT."""
    n = np.size(a)
    m = np.size(b)
    M = n + m - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.maximum(out, 0.0)


def _score(a):
    """Compute 2*n*max(c)/S^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _norm1(a):
    """Normalize to sum 1, non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _symm(a):
    """Symmetrize: a_i = a_{n-1-i}."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1:
        return a.copy()
    return (a + a[::-1]) * 0.5


def _resample(a, n):
    """Resample sequence to length n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == n:
        return a.copy()
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    xo = np.linspace(0, 1, m)
    xn = np.linspace(0, 1, n)
    return _norm1(np.interp(xn, xo, a).astype(np.float64))
