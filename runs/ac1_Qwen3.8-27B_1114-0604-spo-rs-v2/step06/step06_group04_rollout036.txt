Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- FFT-based self-convolution and gradient
- LSE-smoothed objective with exponential beta scheduling
- Nesterov accelerated subgradient with adaptive active set
- Peak-shaving projection heuristic
- Multi-scale asymmetric initialization
- Aggressive final polishing with very high beta
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    """LSE-smoothed 2n*max(c)/S^2 and its gradient for unnormalized a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    # R[j] = sum_k w[k] * a[k-j] for valid k-j
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """Exact subgradient of 2n*max(c)/S^2 at point a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_normalized(a, k_active=3, tol=1e-6):
    """Subgradient of 2n*max(c) for normalized a (sum=1)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fast_self_conv(a)
    cmax = float(c.max())
    score = 2.0 * n * cmax

    threshold = cmax - tol * cmax
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > k_active:
        top_idx = np.argsort(c[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = 2.0 * n * R
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """Nesterov accelerated subgradient on the simplex."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = 2.0 * n * _fast_self_conv(a).max()

    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    score0, g0 = _subgrad_normalized(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-14:
        return best_score, best_a

    eta = 0.02 / max(g_norm, 1e-10)
    no_improve_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy

        score_y, grad_y = _subgrad_normalized(y, k_active, tol)
        if not np.isfinite(score_y):
            break

        step = eta / (1.0 + 0.05 * it)
        x_new = y - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t = t_new

        sc = 2.0 * n * _fast_self_conv(a).max()
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count % 100 == 0:
                eta *= 0.8

        if no_improve_count > 500 and eta < 1e-12:
            break

    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on the LSE-smoothed objective (unnormalized)."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    n = a.size

    if maxfun is None:
        maxfun = 400

    def fun(x):
        x = np.clip(x, 1e-15, None)
        val, grad = _lse_value_grad(x, beta)
        return val, np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    bounds = [(1e-15, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": max(maxfun // 2, 100),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(fun, a.copy(), method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _peak_shave(a, deadline, n_steps=50):
    """
    Peak shaving: iteratively reduce the highest peak by moving mass
    from positions that contribute to it to positions that don't.
    """
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = 2.0 * n * _fast_self_conv(a).max()

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        c = _fast_self_conv(a)
        cmax = c.max()
        max_idx = np.argmax(c)

        # Compute subgradient of c[max_idx]
        j_valid = np.arange(max(0, max_idx - n + 1), min(n, max_idx + 1))
        if len(j_valid) == 0:
            break
        # grad c[k] / da[j] = 2*a[k-j] for valid j
        idx_in_a = max_idx - j_valid
        grad_k = np.zeros(n)
        grad_k[j_valid] = 2.0 * a[idx_in_a]

        # We want to decrease c[max_idx], so move in -grad_k direction
        # But project to simplex: find the best alpha
        g_norm = np.linalg.norm(grad_k)
        if g_norm < 1e-14:
            break

        # Try a few step sizes
        best_local = best_score
        best_local_a = best_a
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_new = a - alpha * grad_k / g_norm
            a_new = np.clip(a_new, 0.0, None)
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            sc = 2.0 * n * _fast_self_conv(a_new).max()
            if sc < best_local - 1e-10:
                best_local = sc
                best_local_a = a_new.copy()

        if np.isfinite(best_local) and best_local < best_score:
            best_score = best_local
            best_a = best_local_a
            a = best_local_a
        else:
            break

    return best_score, best_a


def _nesterov_lbfgs_hybrid(a0, deadline, label=""):
    """
    Hybrid optimization with exponential beta scheduling and
    interleaved Nesterov + peak shaving.
    """
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total_time = max(20.0, float(deadline - t0 - 15.0))

    print("[{}] start: score={:.8f} n={}".format(label, best_score, a.size))

    # Exponential beta schedule: from 100 to 5e6
    n_phases = 12
    betas = np.exp(np.linspace(np.log(100), np.log(5e6), n_phases))
    time_per_phase = total_time / (n_phases * 3)  # LSE + Nesterov + shave per phase

    for phase in range(n_phases):
        beta = betas[phase]
        phase_dead = t0 + (phase + 1) / n_phases * total_time
        phase_dead = min(phase_dead, float(deadline - 10.0))

        if time.monotonic() > phase_dead - 10.0:
            continue

        # LSE step
        rem = phase_dead - time.monotonic()
        if rem > 15.0:
            maxfun = int(max(50, min(800, rem * 8)))
            sc, a2 = _lbfgs_lse(a, beta, phase_dead - 5.0, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                if phase % 2 == 0:
                    print("[{}] phase {} beta={:.0f}: {:.8f}".format(
                        label, phase, beta, best_score))
            a = _normalize(a2)

        # Nesterov step
        rem = phase_dead - time.monotonic()
        if rem > 10.0:
            k_act = max(1, min(5, int(5 - phase * 0.3)))
            tol_val = max(1e-9, 1e-5 * np.exp(-0.2 * phase))
            n_it = int(max(100, rem * 20))
            sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=phase_dead,
                                        k_active=k_act, tol=tol_val)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] phase {} nest k={} tol={:.0e}: {:.8f}".format(
                    label, phase, k_act, tol_val, best_score))
            a = _normalize(a2)

        # Peak shaving
        rem = phase_dead - time.monotonic()
        if rem > 8.0:
            sc, a2 = _peak_shave(a, phase_dead, n_steps=30)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] phase {} shave: {:.8f}".format(
                    label, phase, best_score))
            a = _normalize(a2)

    return best_score, best_a


def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 8))

    if mode == 0:
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=5, tol=1e-5)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-14:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth low-freq noise
        K = min(65, max(5, n // 100))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.2 * noise)
    elif mode == 4:
        # Vary the asymmetry tilt
        x = np.linspace(0.0, 1.0, n)
        tilt = 1.0 + strength * (2.0 * x - 1.0) * float(rng.uniform(0.5, 2.0))
        q = a * tilt
    elif mode == 5:
        # Quadratic tilt
        x = np.linspace(-1.0, 1.0, n)
        q = a * (1.0 + strength * 0.5 * x ** 2)
    elif mode == 6:
        # Shift the mass distribution
        shift = int(rng.integers(-n // 10, n // 10 + 1))
        q = np.roll(a, shift)
        q = np.clip(q, 0.0, None)
        q = (0.9 * q + 0.1 * a)
    else:
        # Blend with a different profile
        x = np.linspace(0.0, 1.0, n)
        profiles = [
            np.exp(-2.0 * x) + 0.01,
            np.exp(-2.0 * (1.0 - x)) + 0.01,
            1.0 + x,
            1.0 - x * 0.5,
            np.sin(np.pi * x) ** 2 + 0.01,
        ]
        p = profiles[int(rng.integers(0, len(profiles)))]
        t = float(rng.uniform(0.1, 0.3))
        q = (1.0 - t) * a + t * _normalize(p)

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize(a, deadline):
    """Quick 2-phase optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    rem = deadline - time.monotonic()
    if rem > 15.0:
        for beta in [1000, 20000, 200000]:
            rem = deadline - time.monotonic()
            if rem < 10.0:
                break
            maxfun = int(max(30, min(300, rem * 10)))
            sc, a2 = _lbfgs_lse(a, beta, deadline, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
            a = _normalize(a2)

    rem = deadline - time.monotonic()
    if rem > 10.0:
        n_it = int(max(100, rem * 20))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                    k_active=3, tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    rem = deadline - time.monotonic()
    if rem > 8.0:
        sc, a2 = _peak_shave(a, deadline, n_steps=30)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()

    return best_score, best_a


def _generate_inits(n, rng, parent, cap=30):
    """Generate diverse initial sequences, emphasizing asymmetry."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    x = np.linspace(0.0, 1.0, n)

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
        # Tilted versions of parent
        for t in [0.1, 0.2, 0.3, 0.5]:
            _add(pr * (1.0 + t * (2.0 * x - 1.0)))
            _add(pr * (1.0 - t * (2.0 * x - 1.0)))

    # Strongly asymmetric profiles
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        _add(np.exp(-rate * x) + 0.001)
        _add(np.exp(-rate * (1.0 - x)) + 0.001)

    # Polynomial tilts
    for p in [1, 2, 3]:
        for amp in [0.2, 0.5, 1.0, 2.0]:
            _add(1.0 + amp * (x ** p - 0.5))
            _add(1.0 + amp * ((1.0 - x) ** p - 0.5))

    # Sinusoidal modulations (asymmetric)
    for f in [1, 2, 3]:
        for amp in [0.3, 0.6, 1.0]:
            _add(1.0 + amp * np.sin(np.pi * f * x + np.pi / 4))
            _add(1.0 + amp * np.cos(np.pi * f * x))

    # Cosh profiles (asymmetric when offset)
    for center in [0.3, 0.4, 0.6, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            _add(np.cosh(2.0 * (x - center)) ** -2 + 0.001)

    # Bilinear (triangle, asymmetric)
    for peak in [0.3, 0.4, 0.5, 0.6, 0.7]:
        tri = np.minimum(x / peak, (1.0 - x) / (1.0 - peak) + 0.01)
        tri = np.clip(tri, 0.01, None)
        _add(tri)

    # S-curve (sigmoid)
    for k in [5, 10, 20]:
        for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
            _add(1.0 / (1.0 + np.exp(-k * (x - c))) + 0.01)

    # Power law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        _add(x ** p + 0.01)
        _add((1.0 - x) ** p + 0.01)

    # Random smooth (FFT-based)
    Nf = _next_pow2(n)
    for trial in range(8):
        freqs = np.fft.rfftfreq(Nf)
        alpha = float(rng.uniform(1.5, 4.0))
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** alpha)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        amp = float(rng.uniform(0.5, 2.0))
        _add(1.0 + amp * noise)

    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    # Uniform
    _add(np.ones(n))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Minimizes 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid LSE+Nesterov+peak_shaving with exponential beta schedule,
    multi-start from diverse asymmetric profiles, and aggressive final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Determine lengths
    if parent is not None:
        parent_len = min(parent.size, 16384)
        if parent_len < 2048:
            parent_len = 4096
        lengths = [parent_len]
        for L in [8192, 10240, 12288, 14336, 16384]:
            if L not in lengths:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288, 6144]

    lengths = list(dict.fromkeys(lengths))  # deduplicate preserving order
    lengths = [L for L in lengths if 2048 <= L <= 16384][:5]
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate and evaluate inits
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = _generate_inits(n, rng, parent, cap=25)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)
                all_lens.append(n)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)
        all_lens.append(n0)

    order = np.argsort(all_scores)
    print("Best init: {:.8f} (len={})".format(
        all_scores[order[0]], all_seqs[order[0]].size))
    print("Top 5 init scores:", ["{:.6f}".format(all_scores[i]) for i in order[:5]])

    # Select top starts from different lengths
    starts = []
    used_lens = set()
    for idx in order[:15]:
        if len(starts) >= 4:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    # Ensure at least 2 starts
    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main optimization
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))

    if len(starts_sorted) == 1:
        fracs = [1.0]
    elif len(starts_sorted) == 2:
        fracs = [0.70, 0.30]
    elif len(starts_sorted) == 3:
        fracs = [0.55, 0.25, 0.20]
    else:
        fracs = [0.50, 0.25, 0.15, 0.10]

    main_end = deadline - 500.0
    main_budget = max(120.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        if time.monotonic() > main_end - 30.0:
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 20.0:
            continue

        sq = starts[si].copy()

        # For secondary starts, blend with current best
        if rank > 0:
            if len(best_seq) != len(sq):
                best_resampled = _resample(best_seq, len(sq))
            else:
                best_resampled = best_seq.copy()
            t_blend = 0.3 + 0.15 * rank
            sq = _normalize((1.0 - t_blend) * _normalize(sq) + t_blend * _normalize(best_resampled))

        sc, seq = _nesterov_lbfgs_hybrid(sq, sub_dead, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)
    print("After main phase: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts
    n_perturb = 6
    perturb_start = time.monotonic()
    perturb_end = deadline - 350.0
    if perturb_end > perturb_start + 60.0:
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 5.0:
                break

            p = _perturb(best_seq, rng)
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 4: Final intensive optimization
    final_dead = deadline - 15.0
    if time.monotonic() < final_dead - 30.0:
        # Very high beta LSE
        rem = final_dead - time.monotonic()
        if rem > 30.0:
            for beta in [500000, 1000000, 3000000]:
                rem = final_dead - time.monotonic()
                if rem < 15.0:
                    break
                maxfun = int(max(50, min(600, rem * 6)))
                sc, a2 = _lbfgs_lse(best_seq, beta, final_dead - 5.0, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
                best_seq = _normalize(a2)

        # Nesterov with very tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            n_it = int(max(300, rem * 30))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=2, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov k=2: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Peak shaving
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            sc, a2 = _peak_shave(best_seq, final_dead, n_steps=100)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final shave: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # One more Nesterov with k=1
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            n_it = int(max(200, rem * 25))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=1, tol=1e-10)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov k=1: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # Final peak shave
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            sc, a2 = _peak_shave(best_seq, final_dead, n_steps=50)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final shave 2: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
