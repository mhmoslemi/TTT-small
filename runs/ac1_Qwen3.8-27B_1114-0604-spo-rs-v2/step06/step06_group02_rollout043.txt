"""
Refined optimizer: multi-start LSE-LBFGS + exact subgradient polish + targeted CD.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed objective and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_quick(a, n_iters=200):
    """Quick Fienup to get a good starting point."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        alpha = 0.6 * (1.0 - it / n_iters) + 0.05
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        
        A_new = A * np.sqrt(P_blend / (P + 1e-30)) ** alpha
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Refined optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: correct LSE gradient (factor-of-2 fix), multi-start with diverse
initializations, adaptive beta schedule, exact subgradient polish with backtracking,
and targeted coordinate descent on peak-contributing elements.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad_fixed(a, beta):
    """LSE-smoothed objective with CORRECT gradient (factor of 2 in first term)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    # Correct: dL/da[m] = 2*h[m], so grad = 2n*(2h/S^2 - 2L/S^3)
    grad = 2.0 * n * (2.0 * h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000, use_fixed_grad=True):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    func = lse_obj_grad_fixed if use_fixed_grad else lse_obj_grad_original
    res = minimize(
        func, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_quick(a, n_iters=200):
    """Quick Fienup for initialization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = 0.7 * (1.0 - it / n_iters) ** 1.5 + 0.05
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * np.sqrt(P_blend / (P + 1e-30)) ** alpha
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, max_iters=20000, deadline=None):
    """
    Projected subgradient method with exact subgradient and backtracking line search.
    Minimizes 2n*max(conv(a,a))/S^2 over {a>=0, sum(a)=1}.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    step_size = 0.05
    patience = 0
    PATIENCE_MAX = 2000

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > PATIENCE_MAX:
            step_size *= 0.5
            patience = 0
            if step_size < 1e-10:
                break

        # Find near-argmax indices
        tol = 1e-12 * c_max + 1e-15
        near_max = c >= (c_max - tol)
        if not near_max.any():
            near_max[int(np.argmax(c))] = True

        # Build probability vector on near-max indices
        p = np.zeros(2 * n - 1)
        idx = np.where(near_max)[0]
        weights = c[idx] - (c_max - tol)
        weights = np.maximum(weights, 1e-30)
        weights = weights / weights.sum()
        p[idx] = weights

        # Compute subgradient: grad[m] = 2n*(2*(p*a_rev)[m+n-1]/S^2 - 2*c_max/S^3)
        a_rev = a[::-1]
        PA = np.fft.rfft(p, M_fft)
        AA_rev = np.fft.rfft(a_rev, M_fft)
        h_full = np.fft.irfft(AA_rev * PA, M_fft)[:2 * n - 1]
        h = h_full[n - 1:2 * n - 1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (2.0 * h / S2 - 2.0 * c_max / S3)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        alpha_try = step_size
        accepted = False
        for _try in range(15):
            a_trial = a - alpha_try * grad / grad_norm
            # Project onto {x >= 0, sum(x) = 1}
            a_trial = np.maximum(a_trial, 0.0)
            S_trial = a_trial.sum()
            if S_trial < 1e-12:
                alpha_try *= 0.5
                continue
            a_trial = a_trial / S_trial
            sc_trial = compute_score(a_trial)
            if sc_trial < sc - 1e-15:
                a = a_trial
                accepted = True
                break
            alpha_try *= 0.5

        if accepted:
            step_size = min(step_size * 1.02, 0.5)
        else:
            step_size *= 0.7

    return best_a


def coordinate_descent(a, deadline=None, max_sweeps=100):
    """
    Coordinate descent targeting the top-contributing elements to the peak convolution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all near-peak indices
        tol = 0.999 * c_max
        peak_indices = np.where(c >= tol)[0][:5]

        for k_star in peak_indices:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            js = np.arange(j_start, j_end + 1)
            # Partner values
            partners_idx = k_star - js
            valid = (partners_idx >= 0) & (partners_idx < n)
            partners = np.zeros(len(js))
            partners[valid] = a[partners_idx[valid]]
            contribs = a[js] * partners

            # Sort by contribution descending
            order = np.argsort(contribs)[::-1]
            top_count = min(80, len(order))

            for ii in range(top_count):
                j = int(js[order[ii]])
                a_orig = a[j]
                if a_orig < 1e-14:
                    continue

                best_val = a_orig
                best_local = sc

                # Try a range of multipliers
                for frac in [0.4, 0.6, 0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5]:
                    a[j] = a_orig * frac
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_orig
                        continue
                    c_tmp = fast_conv(a / S_tmp)
                    sc_tmp = 2.0 * n * float(c_tmp.max())
                    if sc_tmp < best_local - 1e-15:
                        best_local = sc_tmp
                        best_val = a_orig * frac

                a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S

        # Early termination if no improvement
        current_sc = compute_score(a)
        if abs(current_sc - best_sc) < 1e-12 and sweep > 10:
            break

    return best_a


def multiplicative_update(a, n_iters=3000, deadline=None):

        scaled = np.clip(scaled, -500, 500)

        S2 = S * S

        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)

        grad_mean = grad.mean()
