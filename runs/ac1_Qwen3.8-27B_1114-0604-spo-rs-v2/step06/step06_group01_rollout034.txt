```python
def solve_lp_inner(a, M):
    """Solve max sum(x) s.t. conv(a,x) <= M, x >= 0 using cvxpy."""
    n = len(a)
    # Build constraint matrix B (2n-1) x n
    # B[k, j] = a[j] if 0 <= k-j <= n-1
    import cvxpy as cp
    import scipy.sparse as sp
    
    # Build B as sparse
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n - 1)
        ks = np.arange(k_start, k_end)
        rows.append(ks)
        cols.append(np.full(len(ks), j))
        vals.append(np.full(len(ks), a[j]))
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    B = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    x = cp.Variable(n)
    constraints = [B @ x <= M, x >= 0]
    objective = cp.Maximize(cp.sum(x))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.HIGHS)
    
    if x.value is not None:
        return np.maximum(x.value, 0)
    else:
        return np.ones(n) / n
```
