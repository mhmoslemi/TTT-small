Search for nonnegative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

Strategy:
- Work on the simplex (sum = 1) using log-softmax coordinates.
- Smooth the max-convolution objective by log-sum-exp and optimize with
  L-BFGS-B in log coordinates.
- Add a different cross-convolution stage: for fixed f, solve the convex
  problem min_g max_k (f*g)_k using the same smooth log-softmax/L-BFGS
  machinery, then mix f <- (1-t)f + t g.  This is inspired by the LP
  active-set idea and helps escape self-convolution local minima.
- Try several nearby lengths, local perturbations, and a final polished run.
"""

import numpy as np
import time
from scipy.optimize import minimize


class _Timeout(Exception):
    pass


_CURRENT_DEADLINE = 0.0
_LAST_FUN_VAL = np.inf
_CBEST_U = None
_CBEST_VAL = np.inf
_REG_LOG = 1e-13
_P_FLOOR = 1e-18
_MAX_LEN = 20480


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _clean_add(arr, out_list, cap, expected_len=None):
    if len(out_list) >= cap:
        return
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if expected_len is not None and arr.size != expected_len:
        return
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s <= 1e-15:
        return
    out_list.append(arr / s)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    n = int(max(1, n))
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / n
    if a.size == n:
        return _normalize(a)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    N = _next_pow2(m)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:m]
    return np.maximum(out, 0.0)


def _fast_cross_conv(f, g, f_fft=None):
    f = np.asarray(f, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    N = _next_pow2(m)
    if f_fft is None or len(f_fft) != N // 2 + 1:
        f_fft = np.fft.rfft(f, N)
    G = np.fft.rfft(g, N)
    out = np.fft.irfft(f_fft * G, N)[:m]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    a = a / s
    n = a.size
    if n == 0:
        return np.inf
    c = _fast_self_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())


def _cross_score(f, g, f_fft=None):
    f = _normalize(f)
    g = _normalize(g)
    q = _fast_cross_conv(f, g, f_fft)
    if q.size == 0:
        return np.inf
    return float(np.max(q))


def _to_prob(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-70.0)
    u = u - float(np.max(u))
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        p = np.exp(u)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, _P_FLOOR)
    s = float(p.sum())
    if s <= 1e-15:
        return np.ones(u.size, dtype=np.float64) / max(1, u.size)
    return p / s


def _to_log(p):
    p = _normalize(p)
    p = np.maximum(p, _P_FLOOR)
    u = np.log(p)
    u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-70.0)
    return _center_log(u)


def _center_log(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-70.0)
    u = u - float(np.mean(u))
    u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-70.0)
    return u


def _weighted_shift(a, w):
    """
    R_i = sum_k w_k a_{k-i}, with invalid indices omitted.
    a has length n, w has length 2*n-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    m = 2 * n - 1
    if w.size < m:
        tmp = np.zeros(m)
        tmp[:w.size] = w
        w = tmp
    if w.size > m:
        w = w[:m]

    nz = np.flatnonzero(w > 1e-12)
    fft_cutoff = max(24, int(3.0 * np.log2(max(16, n))))
    if nz.size <= fft_cutoff:
        R = np.zeros(n, dtype=np.float64)
        for idx in nz:
            k = int(idx)
            wk = float(w[k])
            if wk == 0.0:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo > hi:
                continue
            seg = a[k - hi:k - lo + 1]
            R[lo:hi + 1] += wk * seg[::-1]
        return np.maximum(R, 0.0)

    M = 3 * n - 1
    N = _next_pow2(M)
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(A * W, N)[:M]
    R = conv[n - 1:2 * n - 1]
    return np.maximum(R, 0.0)


def _lse_value_grad_self_u(x, beta):
    if time.monotonic() > _CURRENT_DEADLINE:
        raise _Timeout()

    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=-70.0)
    n = x.size
    beta = float(max(1.0, beta))

    p = _to_prob(x)
    c = _fast_self_conv(p)
    if c.size == 0:
        return 1e10, np.zeros(n)

    cmax = float(np.max(c))
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        z = np.clip(beta * (c - cmax), -700.0, 0.0)
        e = np.exp(z)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        L = cmax
    else:
        w = e / se
        L = cmax + float(np.log(se)) / beta

    val = 2.0 * float(n) * L

    R = _weighted_shift(p, w)
    meanR = float(np.dot(p, R))
    g = (4.0 * float(n)) * (p * (R - meanR))

    u = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    pen = 0.5 * _REG_LOG * float(np.dot(u, u))
    val += pen
    g += _REG_LOG * u

    val = float(np.nan_to_num(val, nan=1e10, posinf=1e10, neginf=-1e10))
    if not np.isfinite(val):
        return 1e10, np.zeros(n)

    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    _LAST_FUN_VAL = val
    return val, g


def _lse_value_grad_g_u(x, f, f_fft, beta):
    if time.monotonic() > _CURRENT_DEADLINE:
        raise _Timeout()

    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=-70.0)
    n = x.size
    beta = float(max(1.0, beta))
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size != n:
        return 1e10, np.zeros(n)

    g = _to_prob(x)
    q = _fast_cross_conv(f, g, f_fft)
    if q.size == 0:
        return 1e10, np.zeros(n)

    qmax = float(np.max(q))
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        z = np.clip(beta * (q - qmax), -700.0, 0.0)
        e = np.exp(z)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        w = np.zeros(q.size)
        w[int(np.argmax(q))] = 1.0
        L = qmax
    else:
        w = e / se
        L = qmax + float(np.log(se)) / beta

    val = float(n) * L

    R = _weighted_shift(f, w)
    meanR = float(np.dot(g, R))
    grad = (float(n)) * (g * (R - meanR))

    u = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    pen = 0.5 * _REG_LOG * float(np.dot(u, u))
    val += pen
    grad += _REG_LOG * u

    val = float(np.nan_to_num(val, nan=1e10, posinf=1e10, neginf=-1e10))
    if not np.isfinite(val):
        return 1e10, np.zeros(n)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    _LAST_FUN_VAL = val
    return val, grad


def _self_obj(x, beta):
    return _lse_value_grad_self_u(x, beta)


def _g_obj(x, f, f_fft, beta):
    return _lse_value_grad_g_u(x, f, f_fft, beta)


def _lbfgs_callback(x):
    global _CBEST_U, _CBEST_VAL, _LAST_FUN_VAL
    if np.isfinite(_LAST_FUN_VAL) and _LAST_FUN_VAL < float(_CBEST_VAL):
        _CBEST_VAL = float(_LAST_FUN_VAL)
        _CBEST_U = np.asarray(x, dtype=np.float64).copy()


def _run_lbfgs(obj_func, args, u0, sub_dead):
    global _CURRENT_DEADLINE, _LAST_FUN_VAL, _CBEST_U, _CBEST_VAL

    u0 = _center_log(np.asarray(u0, dtype=np.float64).ravel())
    _CURRENT_DEADLINE = float(sub_dead)
    _LAST_FUN_VAL = np.inf
    _CBEST_U = u0.copy()
    _CBEST_VAL = np.inf

    try:
        val, _g = obj_func(u0, *args)
        if np.isfinite(float(val)):
            _LAST_FUN_VAL = float(val)
            _CBEST_VAL = float(val)
            _CBEST_U = u0.copy()
    except _Timeout:
        return u0
    except Exception:
        return u0

    opts = {
        "maxiter": 100000,
        "maxfun": 800000,
        "ftol": 1e-12,
        "gtol": 1e-7,
        "maxls": 50,
        "disp": False,
    }

    try:
        minimize(obj_func, u0, jac=True, args=args,
                 method="L-BFGS-B", callback=_lbfgs_callback, options=opts)
    except _Timeout:
        pass
    except Exception:
        pass

    if _CBEST_U is not None and np.isfinite(float(_CBEST_VAL)):
        return _center_log(_CBEST_U)
    return u0


def _lbfgs_self_u(u0, beta, sub_dead):
    return _run_lbfgs(_self_obj, (float(beta),), u0, sub_dead)


def _lbfgs_g_u(u0, f, f_fft, beta, sub_dead):
    return _run_lbfgs(_g_obj, (f, f_fft, float(beta)), u0, sub_dead)


def _lbfgs_self_plan(u, betas, stage_dead):
    best_p = _to_prob(u).copy()
    best_score = _exact_score(best_p)
    n = len(u)

    for beta in betas:
        now = time.monotonic()
        if now > stage_dead - 12.0:
            break

        base = 18.0 + 0.0045 * float(n)
        if float(beta) > 50000.0:
            base *= 0.75
        dead = min(stage_dead, now + base)

        u = _lbfgs_self_u(u, beta, dead)
        p = _to_prob(u)
        sc = _exact_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("[self-lbfgs-beta-{}] {}".format(int(beta), "{:.8f}".format(best_score)))

    return u, best_p, float(best_score)


def _optimize_g_for_fixed_f(f, g0, stage_dead):
    f = _normalize(f)
    n = f.size
    f_fft = np.fft.rfft(f, _next_pow2(2 * n - 1))

    g = _normalize(g0)
    u = _to_log(g)
    best_g = g.copy()
    best_loss = _cross_score(f, best_g, f_fft)

    betas = [200.0, 2000.0, 20000.0, 100000.0, 500000.0]
    for beta in betas:
        now = time.monotonic()
        if now > stage_dead - 12.0:
            break

        base = 14.0 + 0.0035 * float(n)
        if beta > 100000.0:
            base *= 0.75
        dead = min(stage_dead, now + base)

        u = _lbfgs_g_u(u, f, f_fft, beta, dead)
        g = _to_prob(u)
        loss = _cross_score(f, g, f_fft)
        if np.isfinite(loss) and loss < best_loss * (1.0 - 1e-12):
            best_loss = float(loss)
            best_g = g.copy()
            print("[cross-g-beta-{}] cross_loss={:.8f}".format(int(beta), best_loss))

    return best_g


def _cross_stage(f0, g0, stage_dead):
    f = _normalize(f0)
    g = _normalize(g0)
    best_p = f.copy()
    best_score = _exact_score(f)
    n = len(f)
    now0 = time.monotonic()
    cycles = 2 if (stage_dead - now0) > 350.0 else 1

    for cyc in range(cycles):
        now = time.monotonic()
        if now > stage_dead - 80.0:
            break

        # allow most of the remaining cross-stage budget in alternating cycles
        cycle_dead = min(stage_dead, now + max(30.0, 0.42 * (stage_dead - now)))
        g = _optimize_g_for_fixed_f(f, g, cycle_dead)

        sc = _exact_score(g)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = g.copy()
            print("[cross-cycle-{}] g {}".format(cyc, "{:.8f}".format(best_score)))

        # Mixes into current f.
        f_score = _exact_score(f)
        for t in (0.50, 0.35, 0.25, 0.15, 0.08):
            mix = _normalize((1.0 - t) * f + t * g)
            sc = _exact_score(mix)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_p = mix.copy()
                print("[cross-cycle-{}] mix t={} {}".format(cyc, t, "{:.8f}".format(best_score)))

        # If g itself was much better than f, or a mix improved things,
        # use best as the new current f for next cycle.
        if best_score < f_score - 1e-12:
            f = best_p.copy()

    return best_p, float(best_score), g


def _full_optimize(p, deadline, label, do_cross=True, quick=False):
    t_start = time.monotonic()
    end = float(deadline) - 8.0
    p = _normalize(p)
    best_p = p.copy()
    best_score = _exact_score(p)
    print("[{}] start n={} score={:.8f}".format(label, p.size, best_score))

    if end <= t_start + 20.0:
        return best_p, float(best_score)

    total = max(1.0, end - t_start)

    if quick:
        low_betas = [100.0, 800.0]
        high_betas = [10000.0, 80000.0]
        use_cross = False
    else:
        low_betas = [20.0, 100.0, 500.0, 2000.0]
        high_betas = [8000.0, 40000.0, 150000.0, 700000.0]
        use_cross = bool(do_cross) and total > 350.0

    if use_cross:
        low_dead = t_start + 0.27 * total
        cross_dead = t_start + 0.52 * total
        high_dead = end
    else:
        low_dead = t_start + 0.42 * total
        cross_dead = low_dead
        high_dead = end

    u = _to_log(best_p)

    if t_start < low_dead - 20.0:
        u, p2, sc2 = _lbfgs_self_plan(u, low_betas, low_dead)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_p = p2.copy()
            print("[{}] after low self: {:.8f}".format(label, best_score))

    if use_cross and time.monotonic() < cross_dead - 70.0:
        u_best = _to_log(best_p)
        p_best = _to_prob(u_best).copy()
        p3, sc3, _g = _cross_stage(p_best, p_best, cross_dead)
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = float(sc3)
            best_p = p3.copy()
            print("[{}] after cross: {:.8f}".format(label, best_score))
        u = _to_log(best_p)

    if time.monotonic() < high_dead - 20.0:
        u, p4, sc4 = _lbfgs_self_plan(u, high_betas, high_dead)
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = float(sc4)
            best_p = p4.copy()
            print("[{}] after high self: {:.8f}".format(label, best_score))

    print("[{}] end score={:.8f} len={}".format(label, best_score, best_p.size))
    return best_p, float(best_score)


def _generate_candidates(n, parent, rng, cap=10):
    out = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None:
        pr = _resample(parent, n)
        _clean_add(pr, out, cap, n)
        _clean_add(pr[::-1], out, cap, n)

        # Small smooth multiplicatively perturbed parent.
        K = 15
        kc = K // 2
        kern = np.exp(-0.5 * ((np.arange(K) - kc) ** 2) / max(1.0, (kc * 0.7) ** 2))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.05, 0.25))
        _clean_add(pr * np.exp(amp * noise), out, cap, n)

    _clean_add(np.ones(n), out, cap, n)

    for a in (0.35, 0.80, 1.40, 2.20):
        _clean_add(1.0 + a * np.abs(2.0 * x - 1.0), out, cap, n)
        _clean_add(1.0 + a * (1.0 - np.abs(2.0 * x - 1.0)), out, cap, n)

    for f in (1.0, 2.0, 3.0):
        for amp in (0.25, 0.60, 1.10):
            _clean_add(1.0 + amp * np.cos(2.0 * np.pi * f * x), out, cap, n)
            _clean_add(1.0 + amp * np.cos(np.pi * f * x), out, cap, n)
            _clean.add if False else None
            _clean_add(1.0 + amp * np.sin(np.pi * f * x), out, cap, n)

    for a in (0.40, 0.90, 1.60):
        _clean_add(np.exp(-a * x) + np.exp(-a * (1.0 - x)) + 0.02, out, cap, n)

    for _ in range(3):
        K = 15
        kc = K // 2
        kern = np.exp(-0.5 * ((np.arange(K) - kc) ** 2) / max(1.0, (kc * 0.7) ** 2))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        std = float(np.std(noise))
        if std > 1e-12:
            noise = noise / std
        amp = float(rng.uniform(0.25, 1.40))
        _clean_add(1.0 + amp * noise, out, cap, n)

    for _ in range(2):
        _clean_add(0.25 + 0.75 * rng.random(n), out, cap, n)

    return out


def _make_starts(lengths, parent, rng):
    all_starts = []
    best_per_len = {}

    for n in lengths:
        cands = _generate_candidates(n, parent, rng, cap=12)
        scored = []
        for arr in cands:
            if arr.size != n:
                continue
            sc = _exact_score(arr)
            if np.isfinite(sc):
                scored.append((float(sc), arr))

        if scored:
            arrs = [x[0] for x in scored]
            vals = np.array([float(x[0]) for x in scored], dtype=np.float64)
            order = np.argsort(vals)
            scored = [scored[int(i)] for i in order]
            # keep the two best per length
            for sc, arr in scored[:2]:
                all_starts.append((sc, arr))
            best_per_len[n] = scored[0][1]

    starts = []

    # Prefer one distinct length per presumably promising length.
    if best_per_len:
        infos = []
        for n in lengths:
            if n in best_per_len:
                scores = np.array([_exact_score(best_per_len[n])])
                vals = scores
                infos.append((float(vals[0]), n))
        if infos:
            v = np.array([x[0] for x in infos], dtype=np.float64)
            order = np.argsort(v)
            for idx in order:
                n = infos[int(idx)][1]
                arr = best_per_len[n]
                if arr.size not in [s.size for s in starts]:
                    starts.append(arr.copy())
                if len(starts) >= 3:
                    break

    # Fill with top overall candidates not already represented by length.
    if all_starts:
        vals = np.array([x[0] for x in all_starts], dtype=np.float64)
        order = np.argsort(vals)
        for idx in order:
            if len(starts) >= 3:
                break
            arr = all_starts[int(idx)][1]
            duplicate = False
            for existing in starts:
                if existing.size == arr.size:
                    if np.allclose(existing, arr, rtol=0.0, atol=1e-14):
                        duplicate = True
                        break
            if not duplicate:
                starts.append(arr.copy())

    if not starts:
        starts.append(np.ones(max(lengths[0], 4096), dtype=np.float64) / max(lengths[0], 4096))

    if len(starts) > 1:
        sc = np.array([_exact_score(a) for a in starts], dtype=np.float64)
        order = np.argsort(sc)
        starts = [starts[int(i)].copy() for i in order if np.isfinite(sc[int(i)])][:3]

    return starts


def _parent_sequence():
    try:
        h = globals().get("height_sequence_1", None)
        if h is None:
            return None
        p = np.asarray(h, dtype=np.float64).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.clip(p, 0.0, None)
        if p.size >= 32 and float(p.sum()) > 1e-8:
            p = _normalize(p)
            if p.size > _MAX_LEN:
                p = _resample(p, _MAX_LEN)
            return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    horizon = float(max(120.0, budget_s))
    deadline = t0 + horizon - 30.0
    rng = np.random.default_rng(seed)

    parent = _parent_sequence()
    if parent is not None:
        psc = _exact_score(parent)
        print("Parent len={} score={:.8f}".format(parent.size, psc))
    else:
        print("No parent found.")

    if parent is not None:
        pl = int(min(max(parent.size, 4096), _MAX_LEN))
        cand_lens = [pl]
        for L in (pl - 2048, pl + 2048, pl + 4096, 8192, 10240, 12288, 16384, _MAX_LEN):
            L = int(max(4096, min(_MAX_LEN, L)))
            if L not in cand_lens:
                cand_lens.append(L)
    else:
        cand_lens = [8192, 10240, 12288]

    lengths = [int(L) for L in sorted(set(cand_lens)) if 4096 <= int(L) <= _MAX_LEN][:4]
    if not lengths:
        lengths = [8192]
    print("Lengths: {}".format(lengths))

    starts = _make_starts(lengths, parent, rng)
    if not starts:
        starts = [np.ones(lengths[0]) / lengths[0]]

    start_scores = np.array([_exact_score(a) for a in starts], dtype=np.float64)
    order = np.argsort(start_scores)
    starts = [starts[int(i)].copy() for i in order][:3]
    print("Initial best starts: {}".format(
        ["{:.8f}".format(start_scores[int(i)]) for i in order][:3]
    ))

    best_seq = starts[0].copy()
    best_score = float(start_scores[int(order[0])])

    if best_score >= np.inf or len(starts) == 0:
        best_seq = np.ones(4096) / 4096
        best_score = _exact_score(best_seq)

    # Main budget.  Reserve time for expansion + final.
    total_budget = max(1.0, deadline - t0)
    reserve_final = max(260.0, 0.22 * total_budget)
    reserve_expand = 260.0 if total_budget > 1200.0 else 0.0
    main_end = deadline - reserve_final - reserve_expand
    main_end = max(main_end, t0 + max(60.0, 0.35 * total_budget))
    main_end = min(main_end, deadline - 60.0)

    nstarts = len(starts)
    if nstarts == 1:
        fracs = [1.0]
    elif nstarts == 2:
        fracs = [0.68, 0.32]
    else:
        fracs = [0.55, 0.30, 0.15]

    main_dur = max(60.0, main_end - t0)
    cum = 0.0
    for i, st in enumerate(starts):
        if i >= len(fracs):
            break
        cum += float(fracs[i])
        sub_dead = t0 + cum * main_dur
        sub_dead = min(sub_dead, main_end)
        now = time.monotonic()
        if now > sub_dead - 80.0:
            break

        quick = (i > 1)
        do_cross = (i == 0) or (budget_s > 1800 and not quick)
        p2, sc2 = _full_optimize(
            st, sub_dead,
            label="main{}".format(i + 1),
            do_cross=do_cross,
            quick=quick
        )
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = float(sc2)
            best_seq = p2.copy()
            print("After main{}: {:.8f}".format(i + 1, best_score))

    # Try a larger length if time remains.
    expand_dead = deadline - reserve_final - 20.0
    if reserve_expand > 0 and time.monotonic() < expand_dead - 180.0:
        cur_len = int(best_seq.size)
        large_lens = []
        for L in (12288, 16384, _MAX_LEN):
            if L > cur_len and L <= _MAX_LEN:
                large_lens.append(L)
        for L in large_lens[:1]:
            if time.monotonic() > expand_dead - 120.0:
                break
            p_res = _resample(best_seq, L)
            # reverse is sometimes a useful basin-breaker
            inits = [p_res, p_res[::-1]]
            d = min(expand_dead, time.monotonic() + 320.0)
            for j, init in enumerate(inits):
                if time.monotonic() > d - 80.0:
                    break
                p3, sc3 = _full_optimize(
                    init, d,
                    label="large{}-{}".format(L, j),
                    do_cross=False,
                    quick=True
                )
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = float(sc3)
                    best_seq = p3.copy()
                    print("After large{}-{}: {:.8f}".format(L, j, best_score))

    # Final intensive polishing.
    final_dead = deadline - 8.0
    if time.monotonic() < final_dead - 70.0:
        p4, sc4 = _full_optimize(
            best_seq, final_dead,
            label="final",
            do_cross=True,
            quick=False
        )
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = float(sc4)
            best_seq = p4.copy()
            print("Final best: {:.8f}".format(best_score))

    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1e3, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    s = float(out_arr.sum())
    if s < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / s
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    # Optional sanity call for smaller outputs; larger outputs use the fast
    # internal FFT score to avoid spending too much time in a dense np.convolve.
    if time.monotonic() < deadline and out_arr.size <= 12288:
        try:
            verified = float(evaluate_sequence([float(v) for v in out_arr]))
            if np.isfinite(verified) and verified < best_score:
                best_score = verified
            print("FINAL internal={} verified_final_check={:.8f} len={} time={:.1f}s".format(
                "{:.8f}".format(best_score), best_score, out_arr.size, time.monotonic() - t0
            ))
        except Exception:
            print("FINAL internal={} verified_skipped len={} time={:.1f}s".format(
                "{:.8f}".format(best_score), out_arr.size, time.monotonic() - t0
            ))
    else:
        print("FINAL internal={} len={} time={:.1f}s".format(
            "{:.8f}".format(best_score), out_arr.size, time.monotonic() - t0
        ))

    out_list = []
    for v in out_arr:
        out_list.append(float(v))
    return out_list
