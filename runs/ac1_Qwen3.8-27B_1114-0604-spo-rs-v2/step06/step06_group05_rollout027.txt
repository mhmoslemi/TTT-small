import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score computation."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):

def run_lbfgs_timed(a0, obj_func, obj_params, maxiter, maxfun, deadline, bounds=None):
    """Run L-BFGS-B with time awareness."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    if bounds is None:
        bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        return obj_func(x, *obj_params)

    # Check time budget
    remaining_funs = min(maxfun, max(100, int((deadline - time.monotonic()) * 500)))
    remaining_iters = min(maxiter, max(10, int((deadline - time.monotonic()) * 200)))
    if remaining_funs < 20 or remaining_iters < 5:
        return a0

    try:
        res = minimize(
            wrapped, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': remaining_iters, 'maxfun': remaining_funs, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
        return a_out
    except Exception:
        return a0


def peak_chipping(a0, n_rounds=200, top_k=10, deadline=None):
    """Iteratively reduce top peaks by adjusting contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max / (S * S)

        if current_sc < best_sc:
            best_sc = current_sc
            best_a = a.copy()

        # Identify top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For each peak, compute and reduce top contributors
        reduction = 0.92 + 0.05 * (rnd / n_rounds)  # Start aggressive, become conservative

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner

            # Only reduce elements that contribute significantly
            threshold = c_max * 0.3
            significant = contrib > threshold / max(len(contrib), 1)
            if not np.any(significant):
                continue

            n_reduce = min(5, int(np.sum(significant)))
            if n_reduce == 0:
                continue
            top_local = np.argsort(contrib)[::-1][:n_reduce]

            for li in top_local:
                j = j_start + li
                a[j] *= reduction

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def subgradient_refine(a0, n_iters=5000, top_k=50, deadline=None):
    """Subgradient descent on the true (non-smooth) max objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step_init = 0.02
    step = step_init

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        c_max = float(c.max())

        # Top peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Subgradient of max(conv) w.r.t. a
        # d(conv_k)/da_i = 2*a_{k-i} if 0<=k-i<n
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partner = a[k - j_end:k - j_start + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * partner

        # Normalize subgradient
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break

        # Gradient of score = 2n * (sg*S - 2*c_max) / S^3
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.99995

        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def frequency_flatten(a0, n_iters=300, deadline=None):
    """Iteratively flatten the convolution in frequency domain."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()

            break

        c_max = float(c.max())
        c_mean = float(c.sum()) / (2 * n - 1)

        target_level = c_mean * 1.1 + c_max * 0.05
        # Create flattened target
        b_target = np.minimum(c, target_level).copy()
        b_target = np.maximum(b_target, c_mean * 0.5)

        # Frequency domain: modify |A| to make ifft(|A|^2) closer to b_target
        T = np.fft.rfft(b_target, M)
        T_mag = np.abs(T)
        T_phase = np.angle(T)

        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        # Blend magnitude
        alpha = 0.3 * (1.0 - it / n_iters) + 0.05
        new_mag = A_mag ** (1.0 - alpha) * np.maximum(T_mag, 1e-20) ** alpha
        new_phase = A_phase  # Keep phase

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def structured_perturb(a0, rng, strength=0.02):
    """Add structured perturbation (low-frequency bias)."""
    n = len(a0)
    a = a0.copy()
    # Low-frequency perturbation (smooth)
    x = np.arange(n) / n
    n_harm = rng.integers(2, 8)
    pert = np.zeros(n)
    for i in range(n_harm):
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.cos(2 * np.pi * freq * x + phase)
    pert /= max(np.abs(pert).max(), 1e-10)
    a += strength * pert
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a0.copy()
    return a / S

    # Subgradient refinement from current best

        a_sg = subgradient_refine(best_a, n_iters=5000, top_k=30, deadline=phase2_end)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, "subgrad")

    # LSE polish after peak-chipping
    if time.monotonic() < phase2_end:
        for beta in [50000, 100000, 200000]:
            if time.monotonic() > phase2_end:
                break
            a_opt = run_lbfgs_timed(best_a, lse_score_grad, (beta,),
                                     maxiter=5000, maxfun=20000,
                                     deadline=phase2_end)
            sc = compute_score(a_opt)
            update_best(sc, a_opt, f"polish_beta={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Multi-restart with structured perturbations (15% of time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Multi-restart (deadline +{phase3_end - t_start:.0f}s) ===")

    n_restarts = 8
    time_per_restart = (phase3_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase3_end:
            break
        r_deadline = min(time.monotonic() + time_per_restart, phase3_end)

        # Structured perturbation
        strength = 0.01 + 0.02 * r / n_restarts
        a_pert = structured_perturb(best_a, rng, strength=strength)

        # Quick LSE refinement
        beta = 50000 + 20000 * r
        maxfun = max(2000, int((r_deadline - time.monotonic()) * 200))
        a_opt = run_lbfgs_timed(a_pert, lse_score_grad, (beta,),
                                 maxiter=3000, maxfun=maxfun,
                                 deadline=r_deadline)
        sc = compute_score(a_opt)
        update_best(sc, a_opt, f"restart_{r}")

        # Peak chip the result
        if time.monotonic() < r_deadline:
            a_pc2 = peak_chipping(a_opt, n_rounds=50, top_k=10, deadline=r_deadline)
            sc2 = compute_score(a_pc2)
            update_best(sc2, a_pc2, f"restart_{r}_chip")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final high-precision polish (15% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final polish (deadline +{deadline - t_start:.0f}s) ===")

    # Very high beta LSE
    for beta in [500000, 1000000, 2000000]:
        if time.monotonic() > deadline:
            break
        a_opt = run_lbfgs_timed(best_a, lse_score_grad, (beta,),
                                 maxiter=5000, maxfun=30000,
                                 deadline=deadline)
        sc = compute_score(a_opt)
        update_best(sc, a_opt, f"final_beta={beta}")

    # Final subgradient pass
    if time.monotonic() < deadline:
        a_sg2 = subgradient_refine(best_a, n_iters=3000, top_k=20, deadline=deadline)
        sc = compute_score(a_sg2)
        update_best(sc, a_sg2, "final_sg")

    # Frequency flattening
    if time.monotonic() < deadline:
        a_ff = frequency_flatten(best_a, n_iters=200, deadline=deadline)
        sc = compute_score(a_ff)
        update_best(sc, a_ff, "freq_flat")

    # One more LSE pass
    if time.monotonic() < deadline:
        a_opt = run_lbfgs_timed(best_a, lse_score_grad, (1000000,),
                                 maxiter=5000, maxfun=20000,
                                 deadline=deadline)
        sc = compute_score(a_opt)
        update_best(sc, a_opt, "last_lbfgs")

    # Final peak chipping
    if time.monotonic() < deadline:
        a_pc_final = peak_chipping(best_a, n_rounds=100, top_k=10, deadline=deadline)
        sc = compute_score(a_pc_final)
        update_best(sc, a_pc_final, "final_chip")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
