```python
"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a multi-phase approach: LSE optimization with increasing beta,
active-set power iteration with adaptive k, targeted coordinate descent,
and multi-length exploration. Key improvements: soft active-set weighting,
adaptive step selection, and 1D sub-optimization on hot indices.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    mw = int(w.size)
    if n == 0 or mw == 0:
        return np.empty(n, dtype=np.float64)
    M = n + mw - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    a = a * (target / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta < 1.0:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -60.0, 60.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z = z - z.max()
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * float(np.dot(w, d))) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 8.0:
            return exact_score_a(a), a
    maxfun = max(32, int(maxfun))
    maxiter = max(30, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-11,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(np.maximum(res.x, 0.0))
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights_soft(d, top_frac, beta_w):
    """Soft weights based on how close each lag is to the max."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.ones(m) / m
    z = beta_w * (d - dmax) / dmax
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if s < 1e-15:
        idx = np.argmax(d)
        w = np.zeros(m)
        w[idx] = 1.0
    else:
        w = w / s
    return w


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def power_iter_step(a, w_lags, eta):
    """Multiplicative update: a_i <- a_i * exp(-eta * (g_i - mean(g)))"""
    n = len(a)
    g = corr_pair(a, w_lags)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g_centered = g - float(np.mean(g))
    max_gc = float(np.max(np.abs(g_centered)))
    if max_gc < 1e-15:
        return a.copy()
    log_update = -eta * g_centered / max_gc
    log_update = np.clip(log_update, -5.0, 5.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    return normalize_a(a_new)


def coord_descent_hot(a, deadline, max_indices=200):
    """Coordinate descent on the indices that contribute most to the top convolution lags."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(2000):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        d = fast_self_conv(a)
        cmax = float(np.max(d))
        if cmax < 1e-15:
            break

        # Find top 5 lags
        top_k = min(5, len(d))
        top_idx = np.argpartition(d, -top_k)[-top_k:]
        top_idx = sorted(top_idx.tolist())

        # For each top lag, find indices that contribute most
        hot_indices = set()
        for k in top_idx:
            k = int(k)
            # conv[k] = sum_i a[i]*a[k-i]
            # The contribution of a[i] to conv[k] is a[k-i] (if valid)
            # Find top contributors
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                contribs = np.zeros(n)
                for i in range(lo, hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                # Add top contributors from this lag
                top_contrib = np.argsort(contribs)[-min(20, hi - lo + 1):]
                for ci in top_contrib:
                    if contribs[ci] > 0.01 * cmax:
                        hot_indices.add(int(ci))

        if not hot_indices:
            break

        # Optimize a subset of hot indices
        indices = list(hot_indices)[:max_indices]
        np.random.shuffle(indices)

        improved = False
        for idx in indices:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            # 1D optimization for a[idx]
            # conv[k] changes linearly with a[idx] for k != 2*idx
            # and quadratically for k == 2*idx
            a_old = a[idx]
            if a_old < 1e-15:
                continue

            # Find the range of a[idx] to search
            a_min = 0.0
            a_max = 3.0 * a_old

            # Compute current conv
            d_cur = fast_self_conv(a)
            cmax_cur = float(np.max(d_cur))

            # Binary search for optimal a[idx]
            # Evaluate at a few points
            candidates = [0.0, 0.25 * a_old, 0.5 * a_old, 0.75 * a_old,
                          a_old, 1.25 * a_old, 1.5 * a_old, 2.0 * a_old]

            best_local_val = a_old
            best_local_sc = cmax_cur

            for trial_val in candidates:
                a[idx] = trial_val
                d_trial = fast_self_conv(a)
                trial_cmax = float(np.max(d_trial))
                if trial_cmax < best_local_sc:
                    best_local_sc = trial_cmax
                    best_local_val = trial_val
                if time.monotonic() > float(deadline):
                    break

            a[idx] = best_local_val

        # Check if overall score improved
        sc = exact_score_a(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            improved = True

        if not improved and it > 10:
            # Try with fresh random indices
            random_idx = np.random.choice(n, size=min(50, n), replace=False)
            for idx in random_idx[:20]:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                a_old = a[idx]
                if a_old < 1e-15:
                    continue
                for factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
                    a[idx] = a_old * factor
                    sc_trial = exact_score_a(a)
                    if np.isfinite(sc_trial) and sc_trial < best_sc:
                        best_sc = sc_trial
                        best_a = a.copy()
                        break
                    a[idx] = a_old

    return float(best_sc), best_a


def soft_power_iteration(a_start, max_rounds, deadline, k_list=None):
    """Active-set power iteration with soft lag weighting."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    if k_list is None:
        k_list = [1, 2, 3, 5, 8, 12, 20]

    for rnd in range(int(max_rounds)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        d = fast_self_conv(a)
        dmax = float(np.max(d))
        if dmax < 1e-15:
            break

        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break

            # Use soft weighting: weight each lag by exp(beta * (d[k] - dmax)/dmax)
            # with effective "top-k" cutoff
            beta_w = 50.0  # controls sharpness
            w = active_weights_soft(d, 1.0, beta_w)
            # Also try hard top-k
            w_topk = active_weights_topk(d, k)

            for w_use in [w, w_topk]:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                for eta in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                    if deadline is not None and time.monotonic() > float(deadline):
                        break
                    a_new = power_iter_step(a, w_use, eta)
                    sc = exact_score_a(a_new)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new.copy()
                        break  # Move to next eta if this one worked
                    elif np.isfinite(sc) and sc < exact_score_a(a):
                        a = a_new.copy()

        if best_sc < 1.505:
            # Tighter search near the optimum
            k_list_fine = [1, 2, 3, 5]
            for k in k_list_fine:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                w_topk = active_weights_topk(d, k)
                for eta in [0.001, 0.002, 0.005, 0.01]:
                    if deadline is not None and time.monotonic() > float(deadline):
                        break
                    a_new = power_iter_step(a, w_topk, eta)
                    sc = exact_score_a(a_new)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                        a = a_new.copy()

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=20):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for alpha in [0.3, 0.8, 1.5, 2.5]:
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Inverted shapes (peak in middle)
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        add(1.0 + alpha * np.ones(n))
        add(np.exp(-alpha * (2.0 * x - 1.0) ** 2))

    # Cosine
    for f in [1, 2, 3, 4]:
        for alpha in [0.3, 0.6, 1.0, 1.5]:
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Bimodal (peaks at edges)
    for width in [0.05, 0.1, 0.15, 0.2]:
        add(np.exp(-(((x - 0.0) / width) ** 2)) + np.exp(-(((x - 1.0) / width) ** 2)) + 0.01)

    # Exponential decay from edges
    for rate in [1.0, 2.0, 3.0, 5.0]:
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.05)

    # Sine-based
    for f in [1, 2, 3]:
        add(1.0 + 0.5 * np.sin(np.pi * f * x) ** 2)

    # Random smooth
    Nf = next_power_of_two(n)
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = p * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        d = fast_self_conv(p)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            w = active_weights_topk(d, int(rng.integers(1, 5)))
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Targeted: reduce hot spots
        d = fast_self_conv(p)
        if d.size > 0:
            top_idx = np.argpartition(d, -3)[-3:]
            # Reduce elements that contribute to these lags
            for k in top_idx:
                k = int(k)
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                if lo <= hi:
                    contrib = np.zeros(n)
                    for i in range(lo, hi + 1):
                        j = k - i
                        if 0 <= j < n and j != i:
                            contrib[i] += a[j] if False else p[j]
                    max_c = float(np.max(contrib))
                    if max_c > 1e-12:
                        q = p * np.exp(-strength * 0.3 * contrib / max_c)
                        break
            else:
                q = p.copy()
        else:
            q = p.copy()
    else:
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * 0.3 * np.abs(2.0 * x - 1.0)
        q = p * vshape

    return normalize_a(np.clip(q, 0.0, None))


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 5.0:
        return 32
    est = estimate_eval_time(n)
    mf = int((0.6 * t_left) / max(est, 1e-4))
    mf = max(32, min(2000, mf))
    return mf


def lse_trajectory(a_start, deadline, label=""):
    """Run LSE optimization with increasing beta."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    stages = [
        (100.0, 0.10),
        (500.0, 0.10),
        (2000.0, 0.10),
        (10000.0, 0.10),
        (50000.0, 0.10),
        (200000.0, 0.15),
        (1000000.0, 0.15),
        (5000000.0, 0.20),
    ]

    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 10.0))
    cum = 0.0

    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 15.0))

        while time.monotonic() < sub_dead - 5.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 10.0:
                break
            mf = estimate_maxfun(rem * 0.85, len(a))
            if mf < 32:
                break
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        parent is not None, parent.size if parent is not None else 0))

    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = parent.size
        if n_primary > 16384:
            n_primary = 16384
        if n_primary < 4096:
            n_primary = 8192

    # Initialize from parent
    a_start = normalize_a(resample_sequence(parent, n_primary)) if parent is not None else None
    if a_start is None:
        x = np.linspace(0.0, 1.0, n_primary)
        a_start = normalize_a(1.0 + 1.0 * np.abs(2.0 * x - 1.0))

    best_sc = exact_score_a(a_start)
    best_seq = a_start.copy()
    print("Initial: {:.8f} (n={})".format(best_sc, len(best_seq)))

    # === Phase 1: LSE optimization (35% of time) ===
    t1_end = t_start + 0.35 * total_budget
    sc, seq = lse_trajectory(a_start, t1_end, "p1")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 1 (LSE): {:.8f}".format(best_sc))

    # === Phase 2: Soft power iteration (25% of time) ===
    t2_end = t_start + 0.60 * total_budget
    sc, seq = soft_power_iteration(best_seq, max_rounds=200, deadline=t2_end,
                                   k_list=[1, 2, 3, 5, 8, 12, 20])
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 2 (power iter): {:.8f}".format(best_sc))

    # === Phase 3: Another LSE pass at very high beta (15% of time) ===
    t3_end = t_start + 0.75 * total_budget
    sc, seq = lse_trajectory(best_seq, t3_end, "p3")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 3 (LSE high-beta): {:.8f}".format(best_sc))

    # === Phase 4: Power iteration again (10% of time) ===
    t4_end = t_start + 0.85 * total_budget
    sc, seq = soft_power_iteration(best_seq, max_rounds=100, deadline=t4_end,
                                   k_list=[1, 2, 3, 5])
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 4 (power iter ref): {:.8f}".format(best_sc))

    # === Phase 5: Coordinate descent on hot indices (10% of time) ===
    t5_end = t_start + 0.95 * total_budget
    sc, seq = coord_descent_hot(best_seq, deadline=t5_end, max_indices=100)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("Phase 5 (coord descent): {:.8f}".format(best_sc))

    # === Phase 6: Multi-length exploration + perturbation restarts ===
    t6_end = deadline - 20.0
    if time.monotonic() < t6_end - 60.0:
        # Try a different length
        n_other = 16384 if n_primary != 16384 else 8192
        a_other = normalize_a(resample_sequence(best_seq, n_other))
        sc_o, seq_o = lse_trajectory(a_other, min(t6_end - 40.0, t_start + 0.3 * (t6_end - t_start)), "other")
        if np.isfinite(sc_o) and sc_o < best_sc:
            best_sc = sc_o
            best_seq = seq_o.copy()
            print("Phase 6 (n={}): {:.8f}".format(n_other, best_sc))

    if time.monotonic() < t6_end - 30.0:
        # Perturbation restart
        p = perturb_best(best_seq, rng, strength=0.1)
        sc_p, seq_p = soft_power_iteration(p, max_rounds=50, deadline=t6_end - 10.0,
                                           k_list=[1, 2, 3, 5])
        if np.isfinite(sc_p) and sc_p < best_sc:
            best_sc = sc_p
            best_seq = seq_p.copy()
            print("Phase 6b (perturb): {:.8f}".format(best_sc))

    # === Final: one more tiny power iteration + verify ===
    if time.monotonic() < deadline - 10.0:
        sc, seq = soft_power_iteration(best_seq, max_rounds=30, deadline=deadline - 10.0,
                                       k_list=[1, 2, 3])
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()

    # Final cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
```
