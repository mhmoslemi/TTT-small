from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    M = 2 * n - 1

    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def cross_corr(a, w):
    """Compute sum_k w[k] * a[k-j] for each j (valid correlation)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    m = len(w)
    M = n + m - 1
    N = next_power_of_two(M)
    # corr[j] = sum_k w[k] * a[k-j]
    # = (w * a_flipped)[j + n - 1] where a_flipped = a[::-1]
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n-1:n-1+n]

def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-15:
        return np.ones(a.size) / a.size
    return a / S

def score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * c.max() / (S * S)

def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -600.0, 600.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0:
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    
    # Gradient
    wc = float(w @ d)
    h = cross_corr(a, w)  # h[j] = sum_k w[k] * a[k-j]
    grad = (4.0 * n / (S * S)) * h - (4.0 * n / (S * S) * wc) * np.ones(n)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return 2.0 * n * L, grad

def active_weight(c, frac=0.01, min_k=3, max_frac=0.05):
    """Get weights for active-set: top few lags or those near max."""
    cmax = c.max()
    if cmax <= 0:
        return np.ones(len(c)) / len(c)
    thresh = cmax * (1.0 - frac)
    active = c >= thresh
    # Ensure at least min_k
    if active.sum() < min_k:
        idx = np.argpartition(c, -min_k)[-min_k:]
        active[idx] = True
    # Cap at max_frac
    max_k = max(min_k, int(max_frac * len(c)))
    if active.sum() > max_k:
        idx = np.argpartition(c, -max_k)[-max_k:]
        active2 = np.zeros(len(c), dtype=bool)
        active2[idx] = True
        active = active2
    w = active.astype(float)
    s = w.sum()
    return w / s if s > 0 else np.ones(len(c)) / len(c)

def subgrad_step(a, eta, top_frac=0.01, min_k=3):
    """One subgradient step of the active-set method."""
    c = fast_self_conv(a)
    w = active_weight(c, top_frac, min_k)
    h = cross_corr(a, w)
    hmax = h.max()
    if hmax < 1e-15:
        return a
    a_new = a * np.exp(-eta * h / hmax)
    return normalize(a_new)

def nesterov_subgrad(a, deadline, max_iter=5000, eta_init=0.3, tag=""):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = normalize(a).copy()
    y = a.copy()
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    
    momentum = 0.95
    eta = eta_init
    stagnant = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if it % 100 == 0 and time.monotonic() - t0 > 0.5:
            pass  # just a time check
        
        c = fast_self_conv(y)
        w = active_weight(c, 0.01, 3)
        h = cross_corr(y, w)
        hmax = h.max()
        if hmax < 1e-15:
            break
        
        # Subgradient step
        eta_eff = eta / math.sqrt(max(1, it))
        a_new = y * np.exp(-eta_eff * h / hmax)
        a_new = normalize(a_new)
        
        # Nesterov update
        t_k = (1 + math.sqrt(1 + 4 * momentum * momentum * (it+1) * (it+1))) / (2 * momentum) if momentum > 0 else it + 1
        # Actually, standard Nesterov: beta_k = (1 + sqrt(1+4k^2))/(2k)
        beta_k = (1 + math.sqrt(1 + 4.0 * (it + 1) ** 2)) / (2.0 * (it + 1))
        y_new = beta_k * a_new + (1 - beta_k) * y
        
        sc = score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                eta *= 0.9
                stagnant = 0
        
        a = a_new
        y = y_new
    
    return best_sc, best_a

def peak_shave(a, n_iter=100, deadline=None, tag=""):
    """Directly shave the top convolution peaks by modifying a."""
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_sc = score(a)
    
    for it in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break
        c = fast_self_conv(a)
        cmax = c.max()
        if cmax < 1e-15:
            break
        
        # Find the argmax lag
        k_star = int(c.argmax())
        
        if k_star == 0:
            # c[0] = sum a_j^2: reduce the largest a_j
            order = np.argsort(a)[::-1]
            # Reduce top 5%
            n_top = max(1, n // 20)
            top_idx = order[:n_top]
            # Increase bottom 5%
            bot_idx = order[-n_top:]
            delta = 0.01 * a[top_idx].mean()
            a[top_idx] -= delta
            a[bot_idx] += delta
        elif k_star < n:
            # c[k] = sum_j a[j+k]*a[j]
            pairs = a[k_star:n] * a[:n-k_star]
            pmax = pairs.max()
            if pmax < 1e-15:
                break
            # Find positions contributing most
            order = np.argsort(pairs)[::-1]
            n_top = max(1, min(20, len(order) // 10))
            top_pairs = order[:n_top]
            for j in top_pairs:
                # Reduce the larger of a[j] and a[j+k]
                if a[j+k_star] > a[j]:
                    a[j+k_star] *= 0.995
                else:
                    a[j] *= 0.995
            # Renormalize
        else:
            # k_star >= n: c[k] = sum_j a[j]*a[k-j] where j ranges from k-n+1 to n-1
            j_lo = k_star - n + 1
            j_hi = n - 1
            pairs = a[j_lo:j_hi+1] * a[k_star-j_hi:k_star-j_lo+1]
            pmax = pairs.max()
            if pmax < 1e-15:
                break
            order = np.argsort(pairs)[::-1]
            n_top = max(1, min(20, len(order) // 10))
            top_pairs = order[:n_top]
            for idx in top_pairs:
                j = j_lo + idx
                k_j = k_star - j
                if a[j] > a[k_j]:
                    a[j] *= 0.995
                else:
                    a[k_j] *= 0.995
        
        a = np.clip(a, 0.0, None)
        a = normalize(a)
        sc = score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def lse_lbfgs(a, beta, deadline, maxiter=500):
    """LSE-smoothed L-BFGS-B optimization."""
    n = len(a)
    a = normalize(a)
    if time.monotonic() > deadline - 5:
        return score(a), a
    
    def fun(x):
        x = np.clip(x, 0.0, None)
        return lse_objective_and_grad(x, beta)
    
    bounds = [(0.0, 10.0)] * n
    opts = {
        "maxiter": maxiter,
        "maxfun": maxiter * 3,
        "maxls": 50,
        "ftol": 1e-16,
        "gtol": 1e-12,
        "iprint": -1,
    }
    try:
        res = minimize(lambda x: lse_objective_and_grad(x, beta)[0], a,
                      method="L-BFGS-B", jac=lambda x: lse_objective_and_grad(x, beta)[1],
                      bounds=bounds, options=opts)
        a_out = res.x if res.x is not None else a
    except Exception:
        a_out = a
    
    a_out = normalize(np.clip(a_out, 0.0, None))
    sc = score(a_out)
    if not np.isfinite(sc):
        a_out = a
        sc = score(a)
    return sc, a_out
