```python
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(M, n))

def solve_lp_fienup(a, M_val):
    """
    Solve LP: maximize sum(g) subject to C @ g <= M_val, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    C = build_conv_matrix(a)
    M = 2 * n - 1
    
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M_val, g >= 0]
    prob = cp.Problem(objective, constraints)
    
    solver = 'ECOS'  # or 'SCS', 'CLARABEL'
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='SCS')
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.clip(g.value, 0, None)
    return g_val

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=50000, maxfun=1000000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def exact_line_search(a, g, M_val):
    """
    Find largest t in [0,1] such that max(conv(a_new, a_new)) <= M_val
    where a_new = (1-t)*a + t*g.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    # conv(a_new, a_new) = (1-t)^2 * ba + 2*(1-t)*t * bg + t^2 * gg
    ba = fast_conv(a)  # conv(a,a)
    bg = np.fft.irfft(np.fft.rfft(a, M_fft) * np.fft.rfft(g, M_fft), M_fft)[:2*n-1]  # conv(a,g)
    gg = fast_conv(g)  # conv(g,g)
    
    # For each k: (1-t)^2 * ba_k + 2*(1-t)*t * bg_k + t^2 * gg_k <= M_val
    # Expand: ba_k(1-2t+t^2) + 2t(1-t)*bg_k + t^2*gg_k <= M_val
    # = ba_k - 2t*ba_k + t^2*ba_k + 2t*bg_k - 2t^2*bg_k + t^2*gg_k <= M_val
    # = ba_k + t*(2*bg_k - 2*ba_k) + t^2*(ba_k - 2*bg_k + gg_k) <= M_val
    # Let A_k = ba_k - 2*bg_k + gg_k, B_k = 2*bg_k - 2*ba_k, C_k = ba_k - M_val
    # A_k * t^2 + B_k * t + C_k <= 0
    
    A = ba - 2 * bg + gg
    B = 2 * bg - 2 * ba
    C = ba - M_val
    
    # For each k, find the largest t in [0,1] satisfying A_k*t^2 + B_k*t + C_k <= 0
    t_max = 1.0
    for k in range(len(C)):
        if C[k] > 0:
            # Already violated at t=0? Shouldn't happen since ba_k <= M_val
            # But due to numerical errors, handle gracefully
            t_max = min(t_max, 0.0)
            continue
        if A[k] < 1e-15:
            # Linear: B_k * t + C_k <= 0
            if B[k] < 0:
                t_k = -C[k] / B[k]
                t_max = min(t_max, max(t_k, 0.0))
            # else: B_k >= 0, C_k <= 0, always satisfied for t >= 0
        else:
            # Quadratic: A_k * t^2 + B_k * t + C_k <= 0
            disc = B[k]**2 - 4 * A[k] * C[k]
            if disc < 0:
                # No real roots, check sign at t=0
                if C[k] <= 0:
                    continue  # Always negative (if A_k > 0) or need to check
                else:
                    t_max = 0.0
            else:
                sqrt_disc = np.sqrt(disc)
                if A[k] > 0:
                    t1 = (-B[k] - sqrt_disc) / (2 * A[k])
                    t2 = (-B[k] + sqrt_disc) / (2 * A[k])
                    # The quadratic is <= 0 between t1 and t2
                    t_max = min(t_max, t2)
                else:
                    t1 = (-B[k] - sqrt_disc) / (2 * A[k])
                    t2 = (-B[k] + sqrt_disc) / (2 * A[k])
                    # A_k < 0: quadratic is <= 0 outside [t1, t2]
                    # We want t >= 0, so t <= t1 or t >= t2
                    # Since at t=0 it's satisfied (C_k <= 0 and A_k < 0 means it goes to -inf)
                    # Actually if A_k < 0, the quadratic opens downward
                    # At t=0: C_k <= 0 (satisfied)
                    # It becomes > 0 for large t
                    # So the feasible region is [0, t_max_root] where t_max_root is the larger root
                    t_max = min(t_max, max(t2, t1))
    
    t_max = max(0.0, min(1.0, t_max))
    return t_max
```

I could solve this dual using a subgradient method or a projected gradient method. The gradient of the constraint w.r.t. `y` is just `C` (the convolution matrix), and the objective gradient is `M * 1`.

But this is also an LP, so I might as well use a proper LP solver.

Let me just use cvxpy with the primal LP and build the sparse matrix. Let me set a time limit and see.

Actually, you know what, let me try a completely different approach that avoids the LP entirely:

**Approach: Iterative Water-Filling (IWF)**

The idea is inspired by the water-filling algorithm in signal processing.

1. Start with `a` (non-negative, normalized)
2. Compute `b = conv(a, a)`, `M = max(b)`
3. Identify the "active set": lags where `b_k` is close to M (within some tolerance)
4. For each element `a_j`, compute its "contribution" to the active peaks:
   `contrib_j = sum_{k in active} b_k * a_j * a_{k-j} / (sum_i a_i * a_{k-i})^2`
   
   Hmm, this is getting complicated.

Let me try a simpler approach:

**Approach: Gradient of the smoothed max**

Use the LSE (log-sum-exp) approximation of the max:
`max_k b_k ≈ (1/β) log(sum_k exp(β * b_k))`

The gradient of this w.r.t. `a_j` is:
`∂/∂a_j [LSE] = (1/β) * sum_k exp(β b_k) / sum_m exp(β b_m) * ∂b_k/∂a_j`
`= sum_k p_k * ∂b_k/∂a_j`

where `p_k = exp(β b_k) / sum_m exp(β b_m)` is a probability distribution over lags.

And `∂b_k/∂a_j = a_{k-j} + a_{k-j}` (if we account for both i=j and j=i in the sum)... 

Wait, `b_k = sum_{i+j=k} a_i * a_j`. So `∂b_k/∂a_j = a_{k-j}` (if `0 <= k-j <= n-1`).

So the gradient of the LSE of `b` w.r.t. `a` is:
`∇_a LSE = (1/β) * sum_k p_k * (a shifted by k)` = `(1/β) * conv(p, a)` (with appropriate indexing)

More precisely: `(∇_a LSE)_j = (1/β) * sum_k p_k * a_{k-j}` (if valid) = `(1/β) * (p * a)_j` (cross-correlation).

This is what the previous program already uses. So the LSE approach is what L-BFGS is already doing.

The difference with the LP approach is:
- LSE: smooths the max, then does gradient descent. Effective but might be slow to converge for high β.
- LP: exactly solves the "linearized" problem at each step. Should converge faster.

I think the LP approach is the key innovation here. Let me implement it.

Let me also think about what happens at the boundary of the sequence. The first and last elements of `a` have fewer "neighbors" in the convolution, so they contribute less to the convolution. This means the peak is more likely to be in the middle of the convolution (around lag n-1).

To reduce the peak, we want to "spread out" the mass of `a` so that the convolution is flat. This means `a` should be as "spread out" as possible (uniform) while maintaining the constraint that the convolution doesn't exceed M.

The tension is:
- Uniform `a` gives a triangular convolution (peak at center)
- To flatten the convolution, we need to make `a` non-uniform (larger at edges, smaller in the middle, or vice versa)

Actually, for the convolution to be flat, `a` should be such that `a_i * a_{k-i}` is balanced for all pairs summing to k. This is a complex constraint.

```python
"""
LP-based Fienup optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions, combined with
exact line search, L-BFGS polishing, and coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import cvxpy as cp
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix(a):
    n = len(a)
    M = 2 * n - 1
    nnz = n * n  # exact count
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    
    idx = 0
    for k in range(M):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        cnt = j_end - j_start + 1
        if cnt > 0:
            rows[idx:idx+cnt] = k
            cols[idx:idx+cnt] = np.arange(j_start, j_end + 1)
            vals[idx:idx+cnt] = a[k - j_start : k - j_end + 1]
            idx += cnt
    
    C = sparse.csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(M, n))
    return C

def solve_lp_direction(a, M_val):
    n = len(a)
    C = build_conv_matrix(a)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M_val, g >= 0]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver='ECOS', verbose=False)
    except Exception:
        try:
            prob.solve(solver='SCS', verbose=False)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=50000, maxfun=1000000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1e6).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1e6)] * n
    try:
        res = minimize(
            lse_score_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0
    a_out = np.clip(res.x, 0.0, 1e6)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_lp_main(a, max_iters=500, time_limit=60.0):
    """
    LP-based Fienup iteration.
    At each step:
    1. Compute M = max(conv(a,a))
    2. Solve LP to find best direction g
    3. Line search + update
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    t_start = time.monotonic()
    stale = 0
    t = 0.3  # initial step size
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
            if stale > 50:
                # Try a smaller step
                t = max(0.01, t * 0.5)
                stale = 0
        
        # Solve LP
        g = solve_lp_direction(a, M_val)
        if g is None or g.sum() < 1e-12:
            # LP failed, try a simple direction
            g = a.copy()
        
        g = g / g.sum() * S  # normalize to same sum
        
        # Line search: find largest t in [0, 1] such that peak doesn't increase
        t_candidates = [t, 0.5, 0.3, 0.2, 0.1, 0.05]
        best_t = 0.01
        best_peak = float(fast_conv((1 - best_t) * a + best_t * g).max())
        
        for tc in sorted(t_candidates, reverse=True):
            a_trial = (1 - tc) * a + tc * g
            a_trial = np.maximum(a_trial, 0)  # ensure non-negative
            peak = float(fast_conv(a_trial).max())
            if peak <= M_val * 1.001:  # allow tiny numerical tolerance
                best_t = tc
                best_peak = peak
                break
            elif tc > best_t:
                best_t = tc
                best_peak = peak
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new * S  # maintain normalization
        
        a = a_new
        t = min(1.0, t * 1.05)  # gradually increase step size
        
    return best_a, best_sc

def coordinate_descent_polish(a, n_sweeps=50, time_limit=120.0):
    """Targeted coordinate descent on the peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    t_start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end <= j_start:
            continue
        
        # Optimize each element in the peak's "shadow"
        for j in range(j_start, j_end + 1):
            a_old = a[j]
            if a_old < 1e-15:
                continue
            
            # Try several multipliers
            best_val = a_old
            best_local = sc
            for f in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3]:
                a[j] = a_old * f
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                sc_tmp = 2.0 * n * float(fast_conv(a_norm).max()) / (1.0)  # sum=1 after norm
                # Wait, after normalization sum=1, so score = 2n * max_c
                # But we need to compute max_c of the normalized sequence
                # Actually compute_score does this
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_val = a_old * f
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    return best_a, best_sc

def interpolate_sequence(src, target_n):
    x_old = np.linspace(0, 1, len(src))
    x_new = np.linspace(0, 1, target_n)
    result = np.interp(x_new, x_old, src)
    result = np.maximum(result, 0)
    S = result.sum()
    if S > 1e-12:
        result = result / S
    return result
```
