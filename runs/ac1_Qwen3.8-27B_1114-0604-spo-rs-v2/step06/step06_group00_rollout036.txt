import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

def lp_update(a, top_k=50, solver='highs'):
    """
    LP-based update: find b >= 0 that maximizes sum(b)
    subject to (a * b)[k] <= M for all k in top_k peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M_val = float(c.max())
    
    # Get top_k peak lags
    top_lags = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix: A_eq @ b <= b_eq
    # (a * b)[k] = sum_j a[j] * b[k-j] for valid j
    # For each k in top_lags: sum_j a[j] * b[k-j] <= M_val
    # where b[index] = 0 if index < 0 or index >= n
    
    # The constraint matrix is A[i, j] = a[j] if (k_i - j) is in [0, n-1], else 0
    # i.e., A[i, j] = a[j] if k_i - n + 1 <= j <= k_i, else 0 (clamped to [0, n-1])
    
    # This matrix is very sparse and banded. For n=10000 and top_k=50,
    # it has 50 rows and 10000 columns, with at most n nonzeros per row.
    
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[j])
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(len(top_lags), n))
    b_eq = np.full(len(top_lags), M_val)
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)

    b = res.x
    S_b = float(b.sum())

    b = b / S_b  # normalize

New approach: LP-based direction finding + LSE annealing + multi-n exploration.

from scipy.optimize import minimize, linprog
import time
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

def lp_direction(a, top_k=100):
    """
    Find direction b >= 0 maximizing sum(b) subject to (a*b)[k] <= M for top_k peaks.
    Returns normalized b or None if infeasible.
    """
    n = len(a)
    c = fast_conv(a)
    M_val = float(c.max())
    
    top_lags = np.argsort(c)[::-1][:top_k]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(a[js].tolist())
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(len(top_lags), n))
    b_ub = np.full(len(top_lags), M_val)
    
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, 
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    b = np.maximum(res.x, 0)
    S_b = b.sum()
    if S_b < 1e-10:
        return None
    
    return b / S_b

    a = normalize(a.copy())

def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())

    shifted = np.clip(shifted, -600, 600)

    p = exp_shifted / exp_shifted.sum()

    M3_fft = 1 << (M3 - 1).bit_length()

    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0

def subgrad_peak_shave(a, n_iters=2000, top_k=10, step_init=0.005):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            sg[j_start:j_end+1] += 2.0 * a[j_start:j_end+1][::-1]
        
        sg /= top_k  # average
        
        # Normalize for the score gradient
        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * (1.0/top_k) / (S2 * S))
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a = a - step * grad / gn
        a = normalize(a)
        step *= 0.999
    
    return best_a

def fienup_flat(a, n_iters=200):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        
        alpha = 0.2 + 0.5 * progress
        P_new = (1 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        
        ratio = np.sqrt(P_new / P)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return (2.0 * n * L) / S2, grad

def run_lbfgs(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(lse_score_and_grad, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0

def subgrad_shave(a, n_iters, top_k, step_init, deadline):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            sg[j_start:j_end+1] += 2.0 * a[k - np.arange(j_start, j_end+1)]
        sg /= top_k
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / (S3 * top_k))
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a = a - (step / gn) * grad
        a = normalize(a)
        step *= 0.9995
    
    return best_sc, best_a

def fienup(a, n_iters, mode='flat'):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = P.mean()
        
        if mode == 'flat':
            alpha = 0.15 + 0.6 * progress
            P_new = (1 - alpha) * P + alpha * P_mean
        elif mode == 'quantile':
            q = 0.70 + 0.25 * progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        else:  # hybrid
            if it % 3 == 0:
                alpha = 0.2 + 0.4 * progress
                P_new = (1 - alpha) * P + alpha * P_mean
            elif it % 3 == 1:
                thresh = np.quantile(P, 0.75)
                P_new = np.minimum(P, thresh)
            else:
                P_new = P.copy()
                P_new[P_new < P_mean * 0.3] = P_mean * 0.3
        
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / P)
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a = a_new / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def coord_descent(a, n_sweeps, n_top, deadline):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.8]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def interpolate_seq(a, new_n):
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
