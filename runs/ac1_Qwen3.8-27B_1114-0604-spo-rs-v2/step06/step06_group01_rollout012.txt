```
def wiener_direction(a, residual, lam=1e-3, factor=1.0, sym=False):
   n=len(a); L=2*n-1; N=next_p2(L)
```

```python
"""
Optimizer for non-negative sequences minimizing
    2*n*max(conv(a,a))/(sum(a)^2).

The program starts from the provided height_sequence_1 when available,
then combines several complementary local/global heuristics:

1. Spectral / constructive starting points approximating flat convolution.
2. Gauss-Newton / Wiener flattening steps in the convolution domain.
3. Exact line searches on quadratic directions (peak and flattening directions).
4. L-BFGS refinement on an LSE-smoothed objective with a sum penalty.
5. Symmetric-search passes, since the problem is invariant under reversal.

The goal is to polish the current state-of-the-art sequence below the target
while remaining strictly within the time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

_LSE_BETA = 100.0
_LSE_PEN = 0.0


def next_p2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_sequence(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.shape, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def add_normalized_candidate(cands, a, max_count=24):
    if len(cands) >= max_count:
        return
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return
    if a.size < 2:
        return
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return
    cands.append(a / s)


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(max(1, n), dtype=np.float64) / max(1, n)
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(max(1, n), float(a[0]), dtype=np.float64)
    xo = np.linspace(0.0, 1.0, a.size)
    xn = np.linspace(0.0, 1.0, n)
    return np.interp(xn, xo, a)


def convolution_full_raw(a, b, N=None):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    M = a.size + b.size - 1
    if N is None:
        N = next_p2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:M]


def self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    c = convolution_full_raw(a, a)
    return np.maximum(c, 0.0)


def cross_corr(a, p):
    """
    Returns h[j] = sum_i a[i] p[i+j] for j=0..len(a)-1.
    This is the standard derivative term for convolution coefficients.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    M = n + p.size - 1
    N = next_p2(M)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    return np.fft.irfft(np.conj(A) * P, N)[:n]


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-12:
        return np.inf
    c = self_conv(a)
    m = float(c.max())
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return float(2.0 * a.size * m / (S * S))


def make_symmetric(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = 0.5 * (a + a[::-1])
    b = np.maximum(b, 0.0)
    return normalize_sequence(b, 1.0)


def is_symmetric_sequence(a, atol=1e-7):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 2:
        return True
    return float(np.max(np.abs(a - a[::-1]))) <= atol


def perturb_sequence(a, rng, scale=0.02, sym=False):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = a * (1.0 + scale * rng.standard_normal(a.size))
    b = np.maximum(b, 0.0)
    b = normalize_sequence(b, 1.0)
    if sym:
        b = make_symmetric(b)
    return b


def line_search_direction(a, q, delta, curr_sc, sym=False):
    """
    Exact line search along a signed direction delta from non-negative a.
    q must be convolution_full(a,a) (clamped to non-negative).
    Uses the exact quadratic expansion:
        conv(a+t*delta, a+t*delta) = q + 2t conv(a, delta) + t^2 conv(delta, delta)
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    q = np.asarray(q, dtype=np.float64).ravel()
    delta = np.asarray(delta, dtype=np.float64).ravel()

    n = a.size
    if delta.size != n or q.size != 2 * n - 1:
        return False, None, None

    delta = np.nan_to_num(delta, nan=0.0, posinf=0.0, neginf=0.0)
    if sym:
        delta = 0.5 * (delta + delta[::-1])

    dmin = float(delta.min())
    dmax = float(delta.max())
    if abs(dmin) < 1e-300 and abs(dmax) < 1e-300:
        return False, None, None

    S0 = float(a.sum())
    amin = float(a.min())

    safe = np.inf
    if dmin < -1e-300:
        safe = 0.9999 * (-amin / dmin)
        if not np.isfinite(safe) or safe <= 1e-14:
            safe = 1e-12

    b = convolution_full_raw(a, delta)
    dd = convolution_full_raw(delta, delta)
    L = q.size
    if len(b) > L:
        b = b[:L]
    if len(dd) > L:
        dd = dd[:L]

    sumd = float(delta.sum())

    base = np.array(
        [1e-4, 5e-4, 1e-3, 5e-3, 0.01, 0.02, 0.05, 0.1,
         0.2, 0.35, 0.5, 0.75, 1.0, 1.5, 2.0],
        dtype=np.float64
    )

    if np.isfinite(safe):
        base = np.minimum(base, safe)
        extras = np.array(
            [0.25 * safe, 0.5 * safe, 0.75 * safe, 0.9 * safe, 0.999 * safe],
            dtype=np.float64
        )
        base = np.concatenate((base, extras))
    base = base[np.isfinite(base) & (base > 0.0)]
    if base.size == 0:
        if np.isfinite(safe) and safe > 0.0:
            base = np.array([0.9 * safe, 0.999 * safe], dtype=np.float64)
        else:
            base = np.array([1e-8], dtype=np.float64)

    base = np.unique(np.round(base, 14))

    best_fast = curr_sc
    best_t = None

    for t in base:
        St = S0 + t * sumd
        if not np.isfinite(St) or St <= 1e-15:
            continue
        qt = q + 2.0 * t * b + (t * t) * dd
        m = float(qt.max())
        if not np.isfinite(m):
            continue
        if m < 0.0:
            m = 0.0
        sc = 2.0 * n * m / (St * St)
        if np.isfinite(sc) and sc < best_fast:
            best_fast = sc
            best_t = float(t)

    if best_t is None:
        return False, None, None

    a_try = a + best_t * delta
    if sym:
        a_try = 0.5 * (a_try + a_try[::-1])
    if float(a_try.min()) < -1e-9:
        a_try = np.maximum(a_try, 0.0)

    sc_full = compute_score(a_try)
    if np.isfinite(sc_full) and sc_full < curr_sc - 1e-14:
        a_new = normalize_sequence(a_try, 1.0)
        if sym:
            a_new = make_symmetric(a_new)
        return True, a_new, sc_full

    # Fallback: try smaller/clipped candidates and verify exactly.
    for t in (best_t * 0.5, best_t * 0.1, 1e-4, 1e-3):
        if not np.isfinite(t) or t <= 0.0:
            continue
        if np.isfinite(safe) and t > 0.999 * safe:
            t = max(0.0, 0.999 * safe)
        if t <= 0.0:
            continue
        a2 = a + t * delta
        if sym:
            a2 = 0.5 * (a2 + a2[::-1])
        a2 = np.maximum(a2, 0.0)
        a2 = normalize_sequence(a2, 1.0)
        if sym:
            a2 = make_symmetric(a2)
        sc2 = compute_score(a2)
        if np.isfinite(sc2) and sc2 < curr_sc - 1e-14:
            return True, a2, sc2

    return False, None, None


def wiener_direction(a, residual, lam=1e-3, factor=1.0, sym=False):
    """
    Gauss-Newton direction for matching convolution residual:
        min_d ||B_a d - residual||^2 + lam*||d||^2
    with B_a(delta) = conv(a, delta), solved in the Fourier domain.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    residual = np.asarray(residual, dtype=np.float64).ravel()
    n = a.size
    L = 2 * n - 1
    N = next_p2(L)

    A = np.fft.rfft(a, N)
    R = np.fft.rfft(residual, N)
    den = np.abs(A) ** 2 + max(float(lam), 1e-12)

    D = np.conj(A) * R / den
    D = np.nan_to_num(D, nan=0.0, posinf=0.0, neginf=0.0)
    d = np.fft.irfft(D, N)[:n]
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    d *= float(factor)

    if sym:
        d = 0.5 * (d + d[::-1])

    d -= float(d.mean())
    return d


def gn_polish(a, deadline, sym=False, mode="threshold",
              eta_start=0.9, eta_end=0.0,
              lam_start=1e-3, lam_end=1e-9, factor=1.0):
    """
    Iterative flattened-convolution polishing.

    mode='threshold': quantify how far convolution samples are above a moving
                      level and solve a Wiener correction.
    mode='flat':      try to make convolution as flat as possible (avg-q).
    """
    a = normalize_sequence(np.asarray(a, dtype=np.float64).ravel(), 1.0)
    if sym:
        a = make_symmetric(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    cur_a = a.copy()
    cur_sc = best_sc

    eta = float(eta_start)
    lam = float(lam_start)
    no_accept = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        q = self_conv(cur_a)
        L = q.size
        avg = 1.0 / L
        maxq = float(q.max())

        if not np.isfinite(maxq) or (maxq - avg) < 1e-15:
            break

        if mode == "flat":
            residual = avg - q
        else:
            level = avg + eta * (maxq - avg)
            if level < avg:
                level = avg
            excess = np.maximum(q - level, 0.0)
            essum = float(excess.sum())
            if essum < 1e-18:
                residual = avg - q
            else:
                residual = -excess + (essum / L)

        residual = np.nan_to_num(residual, nan=0.0, posinf=0.0, neginf=0.0)
        residual -= float(residual.mean())

        rnorm = float(np.linalg.norm(residual))
        if not np.isfinite(rnorm) or rnorm < 1e-18:
            break

        d = wiener_direction(cur_a, residual, lam=lam, factor=factor, sym=sym)
        if not np.all(np.isfinite(d)):
            break
        if float(np.linalg.norm(d)) < 1e-300:
            break

        ok, new_a, new_sc = line_search_direction(cur_a, q, d, cur_sc, sym=sym)

        if ok:
            cur_a = new_a
            cur_sc = new_sc
            no_accept = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = cur_a.copy()
            eta = max(eta_end, 0.997 * eta)
            lam = max(lam_end, 0.997 * lam)
        else:
            no_accept += 1
            lam = min(1e-1, lam * 1.5)
            if no_accept % 5 == 0:
                if no_accept < 25:
                    eta = max(eta_end, 0.92 * eta)
                elif eta < 0.999:
                    eta = min(0.999, eta + 0.05)
            if no_accept > 80:
                break

    return best_a, best_sc


def make_peak_weights(q, maxq, P, eta):
    q = np.asarray(q, dtype=np.float64)
    L = q.size
    avg = 1.0 / L
    weights = np.zeros(L, dtype=np.float64)

    if P <= 1:
        k = int(np.argmax(q))
        weights[k] = 1.0
        return weights

    if eta <= 1e-12:
        excess = np.maximum(q - avg, 0.0)
    else:
        level = avg + eta * (maxq - avg)
        excess = np.maximum(q - level, 0.0)

    excesssum = float(excess.sum())
    if excesssum > 1e-15:
        weights = excess.copy()

    # Add a small top-P component so that local peak constraints always enter.
    if P < L:
        order = np.argpartition(-q, P)[:P]
    else:
        order = np.argsort(-q)
    top = q[order]
    top_mean = float(top.mean()) if top.size else 0.0

    top_mask = np.zeros(L, dtype=np.float64)
    top_mask[order] = 1.0
    weights = weights + 0.1 * top_mask * top_mean

    total = float(weights.sum())
    if not np.isfinite(total) or total <= 1e-15:
        weights = np.zeros(L, dtype=np.float64)
        weights[int(np.argmax(q))] = 1.0
    else:
        weights /= total
    return weights


def peak_polish(a, deadline, sym=False):
    """
    Direct exact-direction descent on weighted convolution peaks.
    """
    a = normalize_sequence(np.asarray(a, dtype=np.float64).ravel(), 1.0)
    if sym:
        a = make_symmetric(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    configs = (
        (1, 1.0, 120),
        (3, 1.0, 80),
        (10, 0.7, 80),
        (30, 0.4, 80),
        (100, 0.15, 50),
        (300, 0.0, 30),
        (1, 0.0, 80),
    )

    improved_any = False

    while time.monotonic() < deadline:
        changed = False

        for P, eta, maxit in configs:
            if time.monotonic() >= deadline:
                break

            cur_a = best_a.copy()
            cur_sc = best_sc
            fails = 0

            for _ in range(maxit):
                if time.monotonic() >= deadline:
                    break

                q = self_conv(cur_a)
                maxq = float(q.max())
                L = q.size
                avg = 1.0 / L
                if not np.isfinite(maxq) or (maxq - avg) < 1e-15:
                    break

                weights = make_peak_weights(q, maxq, P, eta)
                h = cross_corr(cur_a, weights)
                h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
                delta = -(h - float(h.mean()))

                if sym:
                    delta = 0.5 * (delta + delta[::-1])

                if not np.all(np.isfinite(delta)):
                    break
                if float(np.linalg.norm(delta)) < 1e-300:
                    break

                ok, new_a, new_sc = line_search_direction(cur_a, q, delta, cur_sc, sym=sym)

                if ok:
                    cur_a = new_a
                    cur_sc = new_sc
                    fails = 0
                    changed = True
                    if cur_sc < best_sc:
                        best_sc = cur_sc
                        best_a = cur_a.copy()
                else:
                    fails += 1
                    if fails > 5:
                        break

            best_a = best_a.copy()

        if not changed:
            break

        improved_any = True

    return best_a, best_sc


def lse_score_and_grad(x, beta):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e10, np.zeros(n, dtype=np.float64)

    S = float(x.sum())
    if not np.isfinite(S) or S <= 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = max(1.0, float(beta))
    c = self_conv(x)
    cmax = float(c.max())
    if not np.isfinite(cmax):
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (c - cmax)
    m = float(z.max())
    exp_z = np.exp(z - m)
    exsum = float(exp_z.sum())
    if exsum <= 0.0 or not np.isfinite(exsum):
        return 1e10, np.zeros(n, dtype=np.float64)

    lse = cmax + (m + float(np.log(exsum))) / beta
    score = 2.0 * n * lse / (S * S)

    p = exp_z / exsum
    corr = cross_corr(x, p)
    corr = np.nan_to_num(corr, nan=0.0, posinf=0.0, neginf=0.0)

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (2.0 * corr / S2 - 2.0 * lse / S3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    return float(score), grad


def lse_objective_and_grad(x):
    x = np.asarray(x, dtype=np.float64)
    val, grad = lse_score_and_grad(x, _LSE_BETA)
    if not np.isfinite(val):
        return 1e10, np.zeros_like(x, dtype=np.float64)

    S = float(x.sum())
    pen = float(_LSE_PEN)
    if pen != 0.0:
        val = float(val + pen * (S - 1.0) * (S - 1.0))
        grad = grad + 2.0 * pen * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return val, grad


def lbfgs_polish(a, deadline, sym=False):
    global _LSE_BETA, _LSE_PEN

    a = normalize_sequence(np.asarray(a, dtype=np.float64).ravel(), 1.0)
    if sym:
        a = make_symmetric(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    betas = (30, 60, 120, 250, 500, 1000, 2000, 4000)

    for beta in betas:
        if time.monotonic() > deadline - 10:
            break

        x = best_a.astype(np.float64).copy()
        _LSE_BETA = float(beta)
        _LSE_PEN = 0.01

        remaining = max(1.0, deadline - time.monotonic())
        if n < 10000:
            maxfun = 2500
        else:
            maxfun = 900
        if remaining < 60.0:
            maxfun = 400
        maxiter = max(500, maxfun // 2)

        bounds = [(0.0, 10.0)] * n
        try:
            res = minimize(
                lse_objective_and_grad,
                x,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "maxfun": int(maxfun),
                    "ftol": 1e-15,
                    "gtol": 1e-10,
                    "maxls": 30,
                },
            )
            y = res.x
        except Exception:
            y = x

        y = np.asarray(y, dtype=np.float64).ravel()
        y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
        y = np.maximum(y, 0.0)
        y = normalize_sequence(y, 1.0)
        if sym:
            y = make_symmetric(y)

        sc = compute_score(y)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = y.copy()

    return best_a, best_sc


def spectral_candidate(n, shift=0):
    """
    Inverse-spectral construction: take square root of the flat rectangular
    convolution target. This can produce a very useful U-shaped initialization.
    """
    L = 2 * int(n) - 1
    N = next_p2(L)
    target = np.ones(L, dtype=np.float64) / L
    Q = np.fft.rfft(target, N)
    Q = np.nan_to_num(Q, nan=0.0, posinf=0.0, neginf=0.0)

    A = np.sqrt(Q.astype(np.complex128))
    if shift != 0 and A.size > 1:
        ks = np.arange(A.size, dtype=np.float64)
        phase = np.exp(-1j * 2.0 * np.pi * ks * float(shift) / float(N))
        phase[-1] = 1.0  # Nyquist component must remain real for rfft/irfft
        A *= phase
        A[-1] = A[-1].real

    a = np.fft.irfft(A, N)[:n]
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        return a
    return a / s


def build_initial_candidates(n, rng, parent=None, max_count=22):
    cands = []

    if parent is not None:
        p = resample_sequence(parent, n)
        add_normalized_candidate(cands, p, max_count)
        add_normalized_candidate(cands, make_symmetric(p), max_count)
        if rng is not None:
            add_normalized_candidate(cands, perturb_sequence(p, rng, scale=0.07), max_count)

    for sh in (0, max(1, n // 4), max(1, n // 2)):
        add_normalized_candidate(cands, spectral_candidate(n, shift=sh), max_count)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-6
    u = np.minimum(x, 1.0 - x)

    add_normalized_candidate(cands, np.ones(n, dtype=np.float64), max_count)
    add_normalized_candidate(cands, 0.5 + 0.4 * np.cos(2.0 * np.pi * x), max_count)
    add_normalized_candidate(cands, 0.2 + 0.8 * np.cos(np.pi * x) ** 2, max_count)
    add_normalized_candidate(cands, 0.5 + 0.5 * np.cos(4.0 * np.pi * x + 0.7), max_count)

    add_normalized_candidate(cands, 1.0 / (eps + x), max_count)
    add_normalized_candidate(cands, 1.0 / (eps + (1.0 - x)), max_count)
    add_normalized_candidate(cands, (1.0 / (eps + x)) * (1.0 / (eps + (1.0 - x))), max_count)
    add_normalized_candidate(cands, (1.0 / (eps + x)) ** 0.5 * (1.0 / (eps + (1.0 - x))) ** 0.5, max_count)
    add_normalized_candidate(cands, (1.0 / (eps + x)) ** 0.25 * (1.0 / (eps + (1.0 - x))) ** 0.25, max_count)

    add_normalized_candidate(cands, 1.0 / (eps + u), max_count)
    add_normalized_candidate(cands, 1.0 / (eps + u) ** 0.5, max_count)
    add_normalized_candidate(cands, 1.0 / (eps + u) ** 0.25, max_count)

    if rng is not None:
        phase = rng.uniform(0.0, 2.0 * np.pi)
        add_normalized_candidate(cands, 0.5 + 0.35 * np.cos(2.0 * np.pi * x + phase), max_count)

        noise = rng.standard_normal(n)
        N = next_p2(n)
        spec = np.fft.rfft(noise, N)
        hold = max(1, n // 50)
        spec[hold:] = 0.0
        noise = np.fft.irfft(spec, N)[:n]
        add_normalized_candidate(cands, 0.5 + 0.5 * np.tanh(noise), max_count)

        add_normalized_candidate(cands,
                                 0.3 + 0.7 * (np.abs(np.sin(np.pi * x + rng.uniform(-0.5, 0.5))) ** 2),
                                 max_count)

    return cands


def polish_candidate(a, deadline, sym=False, gn_frac=0.55):
    """
    Polynomial-style multi-stage polish:
      - threshold Gauss-Newton,
      - flat Gauss-Newton,
      - peak direction line searches.
    """
    t0 = time.monotonic()
    a = normalize_sequence(np.asarray(a, dtype=np.float64).ravel(), 1.0)
    if sym:
        a = make_symmetric(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    remaining = max(0.0, deadline - t0)
    if remaining <= 0.0:
        return best_a, best_sc

    gn_deadline = min(deadline, t0 + max(1.0, float(gn_frac) * remaining))

    try:
        a_try, sc_try = gn_polish(
            best_a,
            gn_deadline,
            sym=sym,
            mode="threshold",
            eta_start=0.9,
            eta_end=0.0,
            lam_start=1e-3,
            lam_end=1e-9,
            factor=1.0,
        )
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass

    if time.monotonic() < gn_deadline - 2.0:
        try:
            a_try, sc_try = gn_polish(
                best_a,
                gn_deadline,
                sym=sym,
                mode="flat",
                eta_start=0.0,
                eta_end=0.0,
                lam_start=1e-4,
                lam_end=1e-10,
                factor=0.5,
            )
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
        except Exception:
            pass

    if time.monotonic() < deadline - 2.0:
        try:
            a_try, sc_try = peak_polish(best_a, deadline, sym=sym)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
        except Exception:
            pass

    return best_a, best_sc


def propose_candidate(budget_s=2400.0):
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0
    T = max(1.0, deadline - t_start)

    rng = np.random.default_rng(20240528)

    best_a = None
    best_sc = np.inf

    # Load parent solution if available.
    parent_raw = globals().get("height_sequence_1", None)
    parent_arr = None
    if parent_raw is not None:
        try:
            p = np.asarray(parent_raw, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if p.size >= 2 and s > 1e-12:
                parent_arr = p / s
                sc_parent = compute_score(parent_arr)
                if np.isfinite(sc_parent):
                    best_a = parent_arr.copy()
                    best_sc = sc_parent
                    print(f"parent score = {sc_parent:.12f}, n={p.size}")
                    spc = make_symmetric(parent_arr)
                    ssc = compute_score(spc)
                    if np.isfinite(ssc) and ssc < best_sc:
                        best_sc = ssc
                        best_a = spc.copy()
                        print(f"symmetric parent score = {ssc:.12f}")
        except Exception:
            pass

    if best_a is None:
        n0 = 6962
        cands0 = build_initial_candidates(n0, rng, None)
        for a in cands0:
            sc = compute_score(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        if best_a is None:
            best_a = np.ones(64, dtype=np.float64) / 64.0
            best_sc = 2.0

    n_main = int(best_a.size)

    # Candidate lengths: parent length plus modestly larger lengths.
    lengths = [n_main]
    for n_try in (8192, 9216, 10240, 12288):
        if n_main <= n_try <= int(1.75 * n_main) + 1000:
            if n_try not in lengths:
                lengths.append(n_try)
    lengths = sorted(set(lengths))[:3]

    pool = []
    idx = 0
    for n_len in lengths:
        cands = build_initial_candidates(n_len, rng, parent_arr)
        for a in cands:
            sc = compute_score(a)
            pool.append((sc, idx, n_len, a))
            idx += 1
    pool.sort()

    # Time plan.
    t_quick = t_start + 0.10 * T
    t_main = t_start + 0.48 * T
    t_lbfs_nonsym = t_start + 0.62 * T
    t_sym = t_start + 0.82 * T
    t_lbfs_sym = t_start + 0.92 * T
    t_final = max(t_start + T * 0.98, deadline - 25.0)

    print(f"planned phases: quick={t_quick - t_start:.0f}s, main={t_main - t_start:.0f}s, "
          f"lbfgs={t_lbfs_nonsym - t_start:.0f}s, sym={t_sym - t_start:.0f}s, "
          f"lbfgs_sym={t_lbfs_sym - t_start:.0f}s, final={t_final - t_start:.0f}s")

    # Quick diversity pass.
    for item in pool[:6]:
        sc, _, n_len, a = item
        if time.monotonic() > t_quick - 25.0:
            break
        sub_deadline = min(t_quick - 10.0, time.monotonic() + 28.0)
        try:
            a_try, sc_try = polish_candidate(a, sub_deadline, sym=False, gn_frac=0.55)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                print(f"quick nonsym n={n_len} score={sc_try:.12f}")
        except Exception:
            pass

        if time.monotonic() < t_quick - 20.0 and sc < best_sc + 0.02:
            a_sym = make_symmetric(a)
            sub_deadline = min(t_quick - 10.0, time.monotonic() + 14.0)
            try:
                a_try, sc_try = polish_candidate(a_sym, sub_deadline, sym=True, gn_frac=0.50)
                if np.isfinite(sc_try) and sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    print(f"quick sym n={n_len} score={sc_try:.12f}")
            except Exception:
                pass

    print(f"after quick: {best_sc:.12f}, n={best_a.size}")

    # Main non-symmetric polish.
    try:
        a_try, sc_try = polish_candidate(best_a, t_main, sym=False, gn_frac=0.55)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass
    print(f"after main nonsym polish: {best_sc:.12f}")

    # L-BFGS refinement on non-symmetric branch.
    try:
        a_try, sc_try = lbfgs_polish(best_a, t_lbfs_nonsym, sym=False)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass
    print(f"after nonsym L-BFGS: {best_sc:.12f}")

    # Symmetric branch.
    try:
        a_sym = make_symmetric(best_a)
        sc_sym = compute_score(a_sym)
        if np.isfinite(sc_sym) and sc_sym < best_sc:
            best_sc = sc_sym
            best_a = a_sym.copy()

        a_try, sc_try = polish_candidate(a_sym, t_sym, sym=True, gn_frac=0.55)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass
    print(f"after sym polish: {best_sc:.12f}")

    # L-BFGS symmetric branch.
    try:
        a_sym = make_symmetric(best_a)
        a_try, sc_try = lbfgs_polish(a_sym, t_lbfs_sym, sym=True)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass
    print(f"after sym L-BFGS: {best_sc:.12f}")

    # Final exact-direction polishing on the best branch.
    try:
        sym_final = is_symmetric_sequence(best_a)
        a_try, sc_try = polish_candidate(best_a, t_final, sym=sym_final, gn_frac=0.50)
        if np.isfinite(sc_try) and sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    except Exception:
        pass
    print(f"after final polish: {best_sc:.12f}")

    # Build final output.
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    s_out = float(out.sum())
    if not np.isfinite(s_out) or s_out < 0.01:
        out = np.full(64, 0.5, dtype=np.float64)

    out_list = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(out_list))
        print(f"final verified score: {verified:.12f}, length={len(out_list)}, "
              f"time={time.monotonic() - t_start:.2f}s")
    except Exception:
        print("evaluate_sequence verification skipped; internal score =", best_sc)

    return out_list
```
