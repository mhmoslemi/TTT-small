def fast_conv(a):

    to the top autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        # Compute current autocorrelation to identify peaks
        c = np.fft.irfft(A * A, M_fft)[:M]
        r_max = float(c.max())
        
        # Target: flat spectrum (median of non-DC)
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        # Additional suppression: identify frequency bins that contribute
        # most to the peak autocorrelation values and suppress them further
        # (This is a heuristic - the contribution of each frequency bin to
        # a specific autocorrelation lag is not directly separable, but we
        # can use the fact that high-magnitude bins at certain frequencies
        # create constructive interference at specific lags.)
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def momentum_peak_descent(a, n_iters=5000, n_peaks=5, step_init=0.01,
                          momentum=0.9, decay=0.99995):
    """
    Peak-directed descent with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)  # momentum

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        top_idx = np.argpartition(c, -min(n_peaks, len(c)))[-min(n_peaks, len(c)):]

        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-12))
        step = min(step, 0.02)

        # Nesterov momentum
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= decay

    return best_a


def lse_objective(x, beta, M_fft, M):
    n = len(x)
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient
    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n - 1:2 * n - 1]
    g_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_adaptive(a0, beta_start=200, beta_end=50000, n_phases=5,
                       maxiter_per_phase=3000):
    """
    L-BFGS with adaptive (progressive) beta.
    Start with moderate beta and increase over phases.
    """
    n = len(a0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5
    a = a / a.sum()
    bounds = [(0.0, 1000.0)] * n

    best_a = a.copy()
    best_sc = compute_score(a)

    for ph in range(n_phases):
        progress = ph / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = max(50.0, min(50000.0, beta))

        res = minimize(
            lse_objective, a, args=(beta, M_fft, M),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter_per_phase,
                     'maxfun': maxiter_per_phase * 20, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        if a_new.sum() > 1e-6:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                a = a_new.copy()  # still use it as starting point for next phase

    return best_sc, best_a


def phase_diversify(a, n_trials=5, rng=None):
    """
    Try random phase perturbations in the frequency domain and
    re-apply non-negativity. This can escape local minima.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_trials):
        A = np.fft.rfft(a, M_fft)
        # Add random phase perturbation
        phase_noise = rng.standard_normal(len(A)) * 0.1
        A_perturbed = A * np.exp(1j * phase_noise)
        a_new = np.fft.irfft(A_perturbed, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            continue
        a_new = a_new / S
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_a


def build_initializations(n, rng, parent):

    if parent is not None and len(parent) == n:

    return inits

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_reshape(a, n_iters=300, alpha_start=0.05, alpha_end=0.85):
    """Fienup iteration with geometric amplitude reshaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def momentum_peak_descent(a, n_iters=50000, n_peaks=5, step_init=0.008,
                          momentum=0.85, decay=0.99999, reshape_every=5000):
    """
    Peak-directed descent with Nesterov momentum and periodic spectral reshaping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_k = min(n_peaks, M)
        top_idx = np.argpartition(c, -top_k)[-top_k:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-14))
        step = min(step, 0.015)

        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= decay

        # Periodic spectral reshaping to escape flat regions
        if reshape_every > 0 and (it + 1) % reshape_every == 0:
            A2 = np.fft.rfft(a, M_fft)
            amp2 = np.sqrt(A2.real ** 2 + A2.imag ** 2)

            S_r = a_r.sum()
            if S_r > 1e-12:
                a = a_r / S_r
                v = np.zeros(n)

    return best_a


def lse_objective_and_grad(x, beta, M_fft, M):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_sum
    rev_x = x[::-1]
    M2 = len(p) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_progressive(a0, beta_schedule, maxiter_per_phase=5000):
    """
    L-BFGS with progressive beta schedule.
    beta_schedule: list of beta values, one per phase.
    """
    n = len(a0)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5
    a = a / a.sum()
    bounds = [(0.0, 1000.0)] * n

    best_a = a.copy()
    best_sc = compute_score(a)

    for ph, beta in enumerate(beta_schedule):
        res = minimize(
            lse_objective_and_grad, a, args=(beta, M_fft, M),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-14,
                     'maxiter': maxiter_per_phase,
                     'maxfun': maxiter_per_phase * 25, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        if a_new.sum() > 1e-6:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new.copy()
        else:
            break

    return best_sc, best_a

                               step_init=0.005, momentum=0.85,
                               decay=0.999999, reshape_every=20000)
    sc = compute_score(a2)
    update_best(sc, a2, "PHASE2-MOM3")

    if time.monotonic() < p2_end - 60:
        a2b = momentum_peak_descent(best_a, n_iters=1000000, n_peaks=8,
                                    step_init=0.004, momentum=0.8,
                                    decay=0.999999, reshape_every=15000)
        sc = compute_score(a2b)
        update_best(sc, a2b, "PHASE2-MOM8")

    if time.monotonic() < p2_end - 30:
        a2c = momentum_peak_descent(best_a, n_iters=1000000, n_peaks=15,
                                    step_init=0.003, momentum=0.75,
                                    decay=0.999999, reshape_every=10000)
        sc = compute_score(a2c)
        update_best(sc, a2c, "PHASE2-MOM15")

    # Phase 3: Hybrid reshape + descent (20% time)
    p3_end = t_start + 0.65 * total_budget
    print(f"=== Phase 3: Hybrid (t < {p3_end - t_start:.0f}s) ===")

    a3 = hybrid_reshape_descent(best_a, n_iters=1000000, n_peaks=5,
                                step=0.004, reshape_every=3000,
                                reshape_alpha=0.2, momentum=0.8)
    sc = compute_score(a3)
    update_best(sc, a3, "PHASE3-HYBRID")

    if time.monotonic() < p3_end - 30:
        a3b = hybrid_reshape_descent(best_a, n_iters=1000000, n_peaks=3,
                                     step=0.003, reshape_every=5000,
                                     reshape_alpha=0.1, momentum=0.9)
        sc = compute_score(a3b)
        update_best(sc, a3b, "PHASE3-HYBRID2")

    # Phase 4: Progressive L-BFGS (25% time)
    p4_end = t_start + 0.90 * total_budget
    print(f"=== Phase 4: Progressive L-BFGS (t < {p4_end - t_start:.0f}s) ===")

    remaining4 = p4_end - time.monotonic()
    if remaining4 > 60:
        # Determine number of phases based on available time
        # Each L-BFGS phase takes roughly 30-120s depending on n and beta
        n_phases = min(6, max(2, int(remaining4 / 80)))
        # Progressive beta schedule
        if n_phases >= 5:
            beta_sched = [50, 200, 1000, 5000, 20000, 80000][:n_phases]
        elif n_phases >= 3:
            beta_sched = [100, 1000, 10000][:n_phases]
        else:
            beta_sched = [500, 5000][:n_phases]

        maxiter_per_phase = min(8000, max(500, int(remaining4 * 10 / n_phases)))
        print(f"  Beta schedule: {beta_sched}, maxiter/phase: {maxiter_per_phase}")

        sc, a4 = run_lbfgs_progressive(best_a, beta_sched,
                                        maxiter_per_phase=maxiter_per_phase)
        update_best(sc, a4, "PHASE4-LBFGS")

    # Phase 5: Final polish (remaining time)
    print(f"=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 30:
        a5 = momentum_peak_descent(best_a, n_iters=200000, n_peaks=3,
                                   step_init=0.002, momentum=0.9,
                                   decay=0.9999999, reshape_every=0)
        sc = compute_score(a5)
        update_best(sc, a5, "PHASE5-FINE")

    if time.monotonic() < deadline - 20:
        # One more L-BFGS with very high beta
        remaining5 = deadline - time.monotonic()
        if remaining5 > 20:
            sc, a5b = run_lbfgs_progressive(best_a, [50000],
                                            maxiter_per_phase=min(5000, int(remaining5 * 10)))
            update_best(sc, a5b, "PHASE5-LBFGS-HIGH")

    # Try slight perturbations and re-optimize
    if time.monotonic() < deadline - 15:
        for trial in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise_level = 0.005 * (trial + 1)
            a_p = best_a * (1.0 + noise_level * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            S_p = a_p.sum()
            if S_p < 1e-12:
                continue
            a_p = a_p / S_p
            a_r = momentum_peak_descent(a_p, n_iters=50000, n_peaks=5,
                                        step_init=0.003, momentum=0.85,
                                        decay=0.999999, reshape_every=0)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART-{trial}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
