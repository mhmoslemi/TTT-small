Uses: high-beta LSE annealing with L-BFGS, targeted peak reduction,
spectral shaping, multi-start, and n-scaling. Targets < 1.5027.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    # Gradient of L w.r.t. a
    a_rev = a[::-1]

    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    return score, grad

def targeted_peak_reduction(a, n_iters, top_k=10):
    """Directly reduce peak autocorrelation using subgradient + line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # d c_k / d a_j = a_{k-j} for j in [j_start, j_end]
                for j in range(j_start, j_end + 1):
                    sg[j] += a[k - j]
        
        # Normalize subgradient
        S2 = S * S
        grad = (2.0 * n / S2) * sg - (4.0 * n * M_val / (S2 * S))
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Line search
        best_alpha = 0.0
        best_f = sc
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            a_trial = a - alpha * grad / gn
            a_trial = normalize(a_trial)
            f_trial = compute_score(a_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        a = normalize(a - best_alpha * grad / gn)
    
    return best_a

"""
High-beta LSE annealing optimizer with targeted peak reduction for minimizing
2n*max(conv(a,a))/S^2. Key improvements over previous:
- Much higher LSE beta values (up to 1,000,000) for closer-to-true-max optimization
- Two-phase approach: shape finding (moderate beta) then extreme refinement (very high beta)
- Targeted subgradient descent with careful line search on active peaks
- Multi-start diversity and spectral shaping for escaping local minima
- Aggressive time management to use full 2400s budget

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score_LSE = 2n * LSE_beta(conv(a,a)) / S^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    # Softmax weights
    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a:
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 80}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def targeted_subgrad(a, n_iters, top_k=10, step_init=0.01):
    """
    Subgradient descent targeting top-K autocorrelation peaks simultaneously.
    With careful line search and Polyak step size.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks and build subgradient
        top_idx = np.argsort(c)[::-1][:top_k]

        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step with line search
        best_alpha = step
        best_f = sc
        for alpha in [step * 2.0, step, step * 0.5, step * 0.25, step * 0.1]:
            a_trial = a - alpha * grad / gn
            a_trial = normalize(a_trial)
            f_trial = compute_score(a_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_alpha = alpha

        if best_alpha <= 0:
            step *= 0.5
            continue

        a = normalize(a - best_alpha * grad / gn)
        # Decay step slowly
        step = min(step * 0.9995, best_alpha * 1.5)
        step = max(step, 1e-8)

    return best_a


def spectral_flatten(a, n_iters=300):
    """Fienup-style spectral flattening to reduce autocorrelation peak."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Gradually increase flattening
        alpha = 0.1 + 0.6 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)

        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def spectral_quantile(a, n_iters=300):
    """Spectral quantile clipping."""

        thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - progress))

        P_clipped = P.copy()
        above = P_clipped > thresh
        P_clipped[above] = thresh

        ratio = np.sqrt(P_clipped / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multi_beta_anneal(a0, betas, time_budget):
    """
    Anneal through a sequence of beta values, always continuing from the best point.
    More time allocated to higher (later) betas.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_sc = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic() - t0
        if elapsed > time_budget - 10:
            break
        remaining = time_budget - 10 - elapsed
        if remaining < 10:
            break

        # Give more iterations to later (higher) betas
        progress = i / max(1, len(betas) - 1)
        weight = 1.0 + 2.0 * progress
        n_remaining_betas = len(betas) - i
        maxiter = int(max(200, remaining * 15 * weight / n_remaining_betas))
        maxiter = min(maxiter, 10000)
        maxfunc = maxiter * 30

        sc, a_new = run_lbfgs(a, beta, maxiter, maxfunc)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = normalize(a_new)
        print(f"      beta={beta:>7}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    return best_sc, best_a


def coord_descent_peak(a, n_sweeps=5, n_top=20):
    """
    Coordinate descent: reduce top contributors to the peak lag.
    Uses multiplicative factors.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue

            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer targeting score < 1.5027.
    Phases:
    1. Spectral shaping for global mixing (5%)
    2. Moderate-beta LSE for shape finding (25%)
    3. High-beta LSE for refinement (25%)
    4. Targeted peak reduction (10%)
    5. Ultra-high-beta LSE polish (15%)
    6. Multi-start diversification (10%)
    7. Final micro-polish (10%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    total_budget = budget_s - 35.0

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            prev = best_score
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (prev={prev:.8f}, n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        best_a = np.ones(10000) / 10000.0
        print("No parent. Starting uniform n=10000.")

    if best_a is None:
        best_a = np.ones(10000)

    n = len(best_a)
    print(f"Initial best: {best_score:.8f}, n={n}")

    # ============================================================
    # Phase 1: Spectral Shaping (5% time)
    # ============================================================
    phase1_end = t_start + 0.05 * total_budget
    print(f"\n=== Phase 1: Spectral Shaping (until t={phase1_end - t_start:.0f}s) ===")

    a_f1 = spectral_flatten(best_a, n_iters=200)
    update(compute_score(a_f1), a_f1, "SPECT-FLAT")

    if time.monotonic() < phase1_end - 5:
        a_f2 = spectral_quantile(best_a, n_iters=150)
        update(compute_score(a_f2), a_f2, "SPECT-QUANT")

    # ============================================================
    # Phase 2: Moderate-Beta LSE (25% time) - Shape Finding
    # ============================================================
    phase2_end = t_start + 0.30 * total_budget
    print(f"\n=== Phase 2: Moderate LSE (until t={phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas_phase2 = [100, 300, 1000, 3000, 10000, 30000]
    multi_beta_anneal(a_work, betas_phase2, phase2_end - time.monotonic())
    sc = compute_score(a_work)
    update(sc, a_work, "PHASE2-END")

    # ============================================================
    # Phase 3: High-Beta LSE (25% time) - Refinement
    # ============================================================
    phase3_end = t_start + 0.55 * total_budget
    print(f"\n=== Phase 3: High LSE (until t={phase3_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas_phase3 = [50000, 100000, 200000]
    multi_beta_anneal(a_work, betas_phase3, phase3_end - time.monotonic())
    sc = compute_score(a_work)
    update(sc, a_work, "PHASE3-END")

    # ============================================================
    # Phase 4: Targeted Peak Reduction (10% time)
    # ============================================================
    phase4_end = t_start + 0.65 * total_budget
    print(f"\n=== Phase 4: Targeted Peak Reduction (until t={phase4_end - t_start:.0f}s) ===")

    for top_k in [5, 10, 20]:
        if time.monotonic() > phase4_end - 15:
            break
        remaining_s = phase4_end - 15 - time.monotonic()
        n_iters = int(max(300, remaining_s * 400))
        n_iters = min(n_iters, 5000)
        a_ps = targeted_subgrad(best_a, n_iters=n_iters, top_k=top_k, step_init=0.005)
        sc = compute_score(a_ps)
        update(sc, a_ps, f"SUBGRAD k={top_k}")

    # Coordinate descent polish
    if time.monotonic() < phase4_end - 10:
        a_cd = coord_descent_peak(best_a, n_sweeps=5, n_top=30)
        sc = compute_score(a_cd)
        update(sc, a_cd, "COORD")

    # ============================================================
    # Phase 5: Ultra-High-Beta LSE (15% time)
    # ============================================================
    phase5_end = t_start + 0.80 * total_budget
    print(f"\n=== Phase 5: Ultra-High Beta LSE (until t={phase5_end - t_start:.0f}s) ===")

    a_work = best_a.copy()
    betas_phase5 = [300000, 500000, 800000, 1200000]
    multi_beta_anneal(a_work, betas_phase5, phase5_end - time.monotonic())
    sc = compute_score(a_work)
    update(sc, a_work, "PHASE5-END")

    # ============================================================
    # Phase 6: Multi-Start Diversification (10% time)
    # ============================================================
    phase6_end = t_start + 0.90 * total_budget
    print(f"\n=== Phase 6: Multi-Start (until t={phase6_end - t_start:.0f}s) ===")

    n_restarts = 6
    for r in range(n_restarts):
        if time.monotonic() > phase6_end - 20:
            break
        remaining = phase6_end - 20 - time.monotonic()
        if remaining < 20:
            break

        # Diverse perturbations
        if r % 3 == 0:
            # Small Gaussian noise
            a_pert = best_a + 0.005 * rng.standard_normal(n)
            a_pert = normalize(a_pert)
        elif r % 3 == 1:
            # Spectral perturbation
            A = np.fft.rfft(best_a)
            P = np.abs(A)**2
            P_pert = P * (1 + 0.02 * rng.standard_normal(len(P)))
            P_pert = np.maximum(P_pert, 1e-20)
            A_new = A * np.sqrt(P_pert / P)
            a_pert = np.fft.irfft(A_new)[:n]
            a_pert = normalize(a_pert)
        else:
            # Peak-targeted redistribution
            c = fast_conv(best_a)
            k_peak = int(np.argmax(c))
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            a_pert = best_a.copy()
            # Reduce peak contributors, add to non-peak
            peak_region = a_pert[j_start:j_end+1]
            non_peak_mask = np.ones(n, dtype=bool)
            non_peak_mask[j_start:j_end+1] = False
            non_peak_idx = np.where(non_peak_mask)[0]
            n_transfer = max(100, n // 50)
            transfer_idx = rng.choice(non_peak_idx, min(n_transfer, len(non_peak_idx)), replace=False)
            delta = 0.001 * a_pert[j_start:j_end+1].sum() / len(transfer_idx)
            a_pert[j_start:j_end+1] -= delta / (j_end - j_start + 1)
            a_pert[transfer_idx] += delta
            a_pert = normalize(a_pert)

        beta = 100000 + r * 100000
        maxiter = max(500, int(remaining * 8))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter, maxiter * 20)
        update(sc, a_opt, f"RESTART {r} (b={beta})")

    # ============================================================
    # Phase 7: Final Micro-Polish (remaining ~10%)
    # ============================================================
    print(f"\n=== Phase 7: Final Polish ===")

    # Ultra-high beta single shot
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = max(1000, int(remaining * 15))
        sc, a_final = run_lbfgs(best_a, 2000000, maxiter, maxiter * 15)
        update(sc, a_final, "FINAL-ULTRA-2M")

    # Very high beta
    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = max(500, int(remaining * 12))
        sc, a_final = run_lbfgs(best_a, 5000000, maxiter, maxiter * 10)
        update(sc, a_final, "FINAL-ULTRA-5M")

    # Final subgradient
    if time.monotonic() < deadline - 15:
        remaining_iters = max(200, int((deadline - 15 - time.monotonic()) * 600))
        a_ps = targeted_subgrad(best_a, n_iters=min(remaining_iters, 3000),
                                 top_k=5, step_init=0.002)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-SUBGRAD")

    # Final coordinate descent
    if time.monotonic() < deadline - 8:
        a_cd = coord_descent_peak(best_a, n_sweeps=3, n_top=15)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-COORD")

    # Last resort: one more high-beta LSE
    if time.monotonic() < deadline - 3:
        maxiter = max(100, int((deadline - 3 - time.monotonic()) * 10))
        sc, a_final = run_lbfgs(best_a, 3000000, maxiter, maxiter * 8)
        update(sc, a_final, "ABSOLUTE-FINAL")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"Target: 1.5027, Achieved: {verified:.8f}, Margin: {1.5027 - verified:.6f}")
    print(f"{'=' * 60}")
    return out
