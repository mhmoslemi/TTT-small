Optimization of non-negative step function to minimize 2n*max(conv)/sum^2.
Strategy: Multi-phase spectral + gradient optimization with adaptive targets.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2*n-1]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    t = beta * (c - c_max)
    t_shifted = t - t.max()
    exp_t = np.exp(t_shifted)
    p = exp_t / exp_t.sum()
    LSE = c_max + (t_shifted.max() + np.log(exp_t.sum())) / beta

"""
Optimized step function search for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses multi-phase spectral (GS-Fienup with adaptive targets) and gradient (LSE) optimization.
Key improvements: adaptive anti-peak GS targets, higher LSE betas, phase-randomized GS for diversification,
energy redistibution in peak-targeted moves, and multiple sequence lengths.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    t = beta * (c - c_max)
    t_shifted = t - t.max()
    exp_t = np.exp(t_shifted)
    p = exp_t / exp_t.sum()
    LSE = c_max + (t_shifted.max() + np.log(exp_t.sum())) / beta
    score = 2.0 * n * LSE / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * cross
    S3 = S * S * S
    grad = (2.0 * n) * (g * S - 2.0 * LSE * a) / S3
    return score, grad


def gs_adaptive(a, n_iters=500, peak_penalty=0.5, alpha_init=0.4, alpha_final=0.03, phase_perturb=0.0):
    """GS-Fienup with adaptive anti-peak target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        # Compute adaptive target: downweight current peaks
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max > 0:
            c_norm = c / c_max
            target = np.maximum(1.0 - peak_penalty * c_norm, 0.01)
            # Normalize target
            target = target / target.sum() * (2 * n - 1)
        else:
            target = np.ones(2 * n - 1) / (2 * n - 1)
        
        b_target_full = np.zeros(M)
        b_target_full[:2 * n - 1] = target
        
        T = np.fft.fft(b_target_full)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        # Optional phase perturbation for diversification
        if phase_perturb > 0:
            T_phase = T_phase + phase_perturb * np.random.randn(M)
        
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** alpha
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def grad_descent(a0, beta, n_iters=20000, lr_init=0.005, momentum=0.9, decay=0.99998):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

        sc, grad = lse_score_grad(a, beta)
        vn = np.linalg.norm(v)
        if vn > 0:
            step = min(lr, 0.01 * a.sum() / vn)  # adaptive step
        else:
            step = lr
        
        v = momentum * v + step * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= decay
        
        if it % 50 == 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:

def peak_reduction_with_redistribution(a, n_rounds=30, top_k=15, reduce_frac=0.1):
    """Reduce top peaks and redistribute energy to low-convolution positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top peaks
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            
            # Reduce top contributors
            n_red = min(8, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_red]
            energy_removed = 0.0
            for li in top_local:
                j = j_start + li
                energy_removed += a[j] * reduce_frac
                a[j] *= (1.0 - reduce_frac)
            
            # Redistribute to elements that contribute to low-conv positions
            low_idx = np.argsort(c)[:n_red]
            for li in range(n_red):
                k_low = int(low_idx[li])
                j_start_l = max(0, k_low - n + 1)
                j_end_l = min(k_low, n - 1)
                if j_end_l >= j_start_l:
                    # Add energy to middle of the segment
                    mid = (j_start_l + j_end_l) // 2
                    a[mid] += energy_removed / n_red
        
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Optimized step function search: minimize 2n*max(conv(a,a))/sum(a)^2.
Multi-phase approach: spectral GS-Fienup with adaptive/anti-peak targets,
high-beta LSE gradient descent with momentum and line search,
peak-targeted redistribution, and periodic spectral diversification.
Key improvements: phase-randomized GS restarts to escape local minima,
exponential-beta LSE schedule, backtracking line search for robustness,
and interleaved spectral+gradient refinement cycles.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    t = beta * (c - c_max)
    t_shifted = t - t.max()
    exp_t = np.exp(t_shifted)
    p = exp_t / exp_t.sum()
    LSE = c_max + (t_shifted.max() + float(np.log(exp_t.sum()))) / beta
    score = 2.0 * n * LSE / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * cross
    S3 = S * S * S
    grad = (2.0 * n) * (g * S - 2.0 * LSE * a) / S3
    return score, grad


def gs_fienup(a, n_iters=400, mode="flat", peak_penalty=0.4,
              alpha_init=0.35, alpha_final=0.02, phase_perturb=0.0, rng=None):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        if mode == "flat":
            b_target = np.zeros(M)
            b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
        elif mode == "anti_peak":
            c = fast_conv(a)
            c_max = float(c.max())
            if c_max > 1e-15:
                c_norm = c / c_max
                target = np.maximum(1.0 - peak_penalty * c_norm, 0.01)
                b_target = np.zeros(M)
                b_target[:2 * n - 1] = target / target.sum() * (2 * n - 1)
            else:
                b_target = np.zeros(M)
                b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
        elif mode == "random":
            if rng is None:
                rng = np.random.default_rng(it)
            rand_target = rng.dirichlet(np.ones(2 * n - 1) * 0.5)
            b_target = np.zeros(M)
            b_target[:2 * n - 1] = rand_target
        else:
            b_target = np.zeros(M)
            b_target[:2 * n - 1] = 1.0 / (2 * n - 1)

        T = np.fft.fft(b_target)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)

        if phase_perturb > 0:
            if rng is None:
                rng = np.random.default_rng(42 + it)
            T_phase = T_phase + phase_perturb * rng.standard_normal(M)

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** alpha
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 100 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def grad_momentum(a0, beta, n_iters=10000, lr_init=0.003, momentum=0.9,
                  lr_decay=0.99997, use_line_search=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break

        v = momentum * v + lr * grad
        step_dir = -v / np.linalg.norm(v)

        if use_line_search:
            alpha = 1.0
            cur_sc = compute_score(a)
            for ls in range(10):
                a_try = a + alpha * step_dir
                a_try = np.maximum(a_try, 0.0)
                S_try = float(a_try.sum())
                if S_try < 1e-12:
                    alpha *= 0.5
                    continue
                a_try = a_try / S_try
                sc_try = compute_score(a_try)
                if sc_try < cur_sc:
                    a = a_try
                    break
                alpha *= 0.5
            else:
                a = np.maximum(a + 0.01 * step_dir, 0.0)
                a = a / max(a.sum(), 1e-12)
        else:
            a_new = a - v
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            a = a_new

        lr *= lr_decay

        if it % 100 == 0:
            sc_check = compute_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    return best_a


def run_lbfgs(a0, beta, maxiter=4000, maxfun=20000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n

    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            wrapped, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0.copy()
        return a_out
    except Exception:
        return a0


def peak_targeted_refine(a, n_rounds=40, top_k=20, reduce_frac=0.08, redistribute=True):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        peak_idx = np.argsort(c)[::-1][:top_k]

        for k in peak_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner

            n_red = min(6, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_red]
            energy_removed = 0.0
            for li in top_local:
                j = j_start + li
                energy_removed += a[j] * reduce_frac
                a[j] *= (1.0 - reduce_frac)

            if redistribute and energy_removed > 0:
                low_idx = np.argsort(c)[:n_red]
                for li in range(n_red):
                    k_low = int(low_idx[li])
                    js = max(0, k_low - n + 1)
                    je = min(k_low, n - 1)
                    if je >= js:
                        mid = (js + je) // 2
                        a[mid] += energy_removed / n_red

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        x_new = np.linspace(0, 1, alt_n)
        a_alt = np.interp(x_new, x_old, best_a)
        a_alt = np.maximum(a_alt, 0.0)
        a_alt = a_alt / a_alt.sum()

        a_alt = gs_fienup(a_alt, n_iters=200, mode="anti_peak", peak_penalty=0.4,
                          alpha_init=0.3, alpha_final=0.02, rng=rng)
        a_opt = run_lbfgs(a_alt, 100000, maxiter=2000, maxfun=8000)
        sc_alt = compute_score(a_opt)
        update_best(sc_alt, a_opt, f"alt_n{alt_n}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
