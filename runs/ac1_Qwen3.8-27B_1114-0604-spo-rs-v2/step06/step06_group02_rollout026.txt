        return np.inf

def solve_lp_given_fixed(fixed, M):
    """
    Solve: max sum(g) s.t. (fixed * g)[k] <= M for all k, g >= 0.
    Returns optimal g (normalized to sum=1) or None if infeasible.
    """
    n = len(fixed)
    # (fixed * g)[k] = sum_j fixed[j] * g[k-j] for valid j, k-j
    # For variable g[i], its coefficient in constraint k is fixed[k-i] (if 0 <= k-i < n)

    c_obj = -np.ones(n)

            if 0 <= j < n:
                A_rows[k, i] = fixed[j]

                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})

        alpha = alpha_init * (1 - progress) ** 2 + alpha_end
        
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        P_target = (1 - alpha) * P + alpha * P_mean
        P_target = np.maximum(P_target, 1e-20)
        
        A_new = A * (np.sqrt(P_target / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def upsample(parent, target_n):
    """Upsample a sequence to a different length."""
    if len(parent) == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def downsample(parent, target_n):
    """Downsample a sequence to a different length."""
    if len(parent) == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def stochastic_restart_lp(a, rng, n_restarts=5, deadline=None, beta=3000):
    """Stochastic perturbations followed by LP or L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for r in range(n_restarts):
        if deadline and time.monotonic() > deadline:
            break
        
        strategy = r % 4
        if strategy == 0:
            a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        elif strategy == 1:
            # Modulate with random sinusoid
            x = np.arange(n) / n
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 2:
            # Block perturbation
            block_size = max(1, n // 10)
            n_blocks = n // block_size
            for _ in range(min(3, n_blocks)):
                b = rng.integers(0, max(1, n_blocks))
                a_pert = a * (1.0 + 0.1 * rng.standard_normal(n))
                a_pert[b * block_size:(b + 1) * block_size] *= (1.0 + 0.15 * rng.standard_normal(block_size))
        else:
            # Peak-targeted: reduce elements contributing to max peak
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            a_pert[j_start:j_end + 1] *= (1.0 - 0.1 * rng.uniform(0, 1, j_end - j_start + 1))
            a_pert = np.maximum(a_pert, 0)
        
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        
        # Quick L-BFGS polish
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            if time.monotonic() < (deadline or np.inf):
                print(f"      Restart {r}: {sc:.8f}")
    
    return best_a, best_sc


def coordinate_descent_peak(a, n_sweeps=30):
    """Coordinate descent targeting the peak contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end <= j_start:
            continue
        
        # Try adjusting top-contributing elements
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        order = np.argsort(contribs)[::-1]
        
        for idx in order[:30]:
            j = j_start + int(idx)
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                val = a_old * f
                a[j] = val
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                sc_tmp = compute_score(a / S_tmp)
                if sc_tmp < best_local:
                    best_local = sc_tmp
                    best_val = val
            a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP-Fienup (novel) + L-BFGS polish + multi-scale + restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    effective_budget = total_budget - 30.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def update(sc, a, label=""):
        nonlocal best_score, best_a
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        # Create a reasonable starting point
        n_init = 2048
        a_init = np.ones(n_init) / n_init
        # Apply a raised cosine window for better initial shape
        x = np.arange(n_init) / (n_init - 1)
        window = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
        a_init = window / window.sum()
        sc_init = compute_score(a_init)
        print(f"Initialized: n={n_init}, score={sc_init:.8f}")
        update(sc_init, a_init, "init")
    
    # Phase 1: Quick Fienup to get a good starting point (5% time)
    p1_end = t_start + 0.05 * effective_budget
    print(f"\n=== Phase 1: Fienup warmup ===")
    if best_a is not None:
        a_f = fienup_fast(best_a, n_iters=300)
        update(compute_score(a_f), a_f, "FIENUP-warmup")
    
    # Phase 2: LP-Fienup (main novel method, 30% time)
    p2_end = t_start + 0.35 * effective_budget
    print(f"\n=== Phase 2: LP-Fienup (until t={p2_end - t_start:.0f}s) ===")
    if best_a is not None:
        n_lp = len(best_a)
        # Use active constraint selection for speed
        n_active = min(2 * n_lp - 1, max(50, n_lp // 2))
        a_lp, sc_lp = lp_fienup(best_a, n_iters=300, n_active=n_active,
                                 verbose=True, deadline=p2_end)
        update(sc_lp, a_lp, "LP-FIENUP")
    
    # Phase 3: L-BFGS polish with increasing beta (25% time)
    p3_end = t_start + 0.60 * effective_budget
    print(f"\n=== Phase 3: L-BFGS polish (until t={p3_end - t_start:.0f}s) ===")
    if best_a is not None:
        betas = [500, 2000, 5000, 10000, 30000, 80000, 200000]
        for i, beta in enumerate(betas):
            if time.monotonic() >= p3_end:
                break
            remaining = p3_end - time.monotonic()
            if remaining < 15:
                break
            maxiter = int(min(20000, remaining * 15 + 500))
            maxiter = max(maxiter, 2000)
            print(f"    L-BFGS beta={beta}, maxiter={maxiter}")
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 20)
            dt = time.monotonic() - t0
            print(f"      score={sc:.8f} ({dt:.1f}s)")
            update(sc, a_opt, f"LBFGS-b{beta}")
    
    # Phase 4: More LP iterations with all constraints (15% time)
    p4_end = t_start + 0.75 * effective_budget
    print(f"\n=== Phase 4: Full-constraint LP (until t={p4_end - t_start:.0f}s) ===")
    if best_a is not None and time.monotonic() < p4_end:
        n_lp = len(best_a)
        a_lp2, sc_lp2 = lp_fienup(best_a, n_iters=100, n_active=None,
                                   verbose=True, deadline=p4_end)
        update(sc_lp2, a_lp2, "LP-FULL")
        # Quick L-BFGS after LP
        if time.monotonic() < p4_end - 10:
            sc, a_opt = run_lbfgs(best_a, 50000, maxiter=10000, maxfun=200000)
            update(sc, a_opt, "LP+LBFGS")
    
    # Phase 5: Multi-length exploration (10% time)
    p5_end = t_start + 0.85 * effective_budget
    print(f"\n=== Phase 5: Multi-length (until t={p5_end - t_start:.0f}s) ===")
    if best_a is not None:
        current_n = len(best_a)
        target_ns = []
        if current_n <= 2048:
            target_ns = [current_n * 2]
        elif current_n <= 4096:
            target_ns = [current_n * 2, current_n // 2]
        else:
            target_ns = [current_n // 2]
        
        for tn in target_ns:
            if time.monotonic() >= p5_end:
                break
            if tn < 256 or tn > 8192:
                continue
            print(f"    Trying n={tn}")
            a_new_len = upsample(best_a, tn)
            # Quick optimization at new length
            sc, a_opt = run_lbfgs(a_new_len, 3000, maxiter=5000, maxfun=100000)
            update(sc, a_opt, f"len-{tn}")
    
    # Phase 6: Stochastic restarts (7% time)
    p6_end = t_start + 0.92 * effective_budget
    print(f"\n=== Phase 6: Stochastic restarts ===")
    if best_a is not None:
        a_sr, sc_sr = stochastic_restart_lp(best_a, rng, n_restarts=8,
                                            deadline=p6_end, beta=5000)
        update(sc_sr, a_sr, "STOCHASTIC")
    
    # Phase 7: Final polish (8% time)
    print(f"\n=== Phase 7: Final polish ===")
    if best_a is not None:
        # High-beta L-BFGS
        if time.monotonic() < deadline - 20:
            remaining = deadline - 20 - time.monotonic()
            maxiter = int(min(15000, remaining * 10))
            maxiter = max(maxiter, 1000)
            sc, a_fin = run_lbfgs(best_a, 100000, maxiter=maxiter, maxfun=maxiter * 15)
            update(sc, a_fin, "FINAL-HB")
        
        # Coordinate descent on peak
        if time.monotonic() < deadline - 15:
            a_cd = coordinate_descent_peak(best_a, n_sweeps=15)
            sc_cd = compute_score(a_cd)
            update(sc_cd, a_cd, "CD-PEAK")
            # L-BFGS after CD
            if time.monotonic() < deadline - 10:
                sc2, a2 = run_lbfgs(a_cd, 200000, maxiter=5000, maxfun=100000)
                update(sc2, a2, "CD+FIN")
        
        # One more LP step if time
        if time.monotonic() < deadline - 15 and len(best_a) <= 4096:
            a_lp3, sc_lp3 = lp_fienup(best_a, n_iters=20, n_active=None,
                                       verbose=False, deadline=deadline - 10)
            update(sc_lp3, a_lp3, "FINAL-LP")
            if sc_lp3 < best_score and time.monotonic() < deadline - 5:
                sc_final, a_final = run_lbfgs(best_a, 200000, maxiter=3000, maxfun=60000)
                update(sc_final, a_final, "FINAL-LP+LBFGS")
    
    # Output
    if best_a is None:
        out = [0.5] * 128
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    
    if sum(out) < 0.01:
        out = [0.5] * 128
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
