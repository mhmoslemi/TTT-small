def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AAr = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AAr * PA, M2_fft)[:M2]
    h = h_full[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, peak-targeted coordinate descent,
Fienup spectral shaping, multi-length exploration, and adaptive exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    L = c_max + float(np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AAr = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AAr * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def peak_cd(a, n_sweeps=100, deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_max = c.max()
        top_peaks = np.argsort(c)[::-1][:8]

        for k in top_peaks:
            k = int(k)
            if deadline and time.monotonic() > deadline:
                break
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            partners = k - js
            contribs = a[js] * a[partners]
            if contribs.sum() < 1e-15:
                continue

            top_count = min(25, len(contribs))
            order = np.argsort(contribs)[::-1][:top_count]

            for idx in order:
                j = int(js[idx])
                a_old = a[j]
                if a_old < 1e-10:
                    continue
                best_val = a_old
                best_local = sc
                for factor in [0.5, 0.7, 0.8, 0.9, 1.1, 1.25, 1.5]:
                    a[j] = a_old * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old
                        continue
                    c_tmp = fast_conv(a / S_tmp)
                    sc_tmp = 2.0 * n * c_tmp.max() / (S_tmp * S_tmp)
                    if sc_tmp < best_local:
                        best_local = sc_tmp
                        best_val = a_old * factor
                a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
    return best_a


def fienup_v2(a, n_iters=500, deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        progress = it / n_iters
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.7 * (1.0 - progress) ** 2 + 0.04
        P_blend = (1.0 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        ratio = np.sqrt(P_blend / (P + 1e-30))
        A_new = A * ratio ** alpha
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def interpolate_seq(a, target_n):
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0.0)


def multiplicative_weights(a, n_iters=2000, deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = 0.05
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        c_max = c.max()
        beta_t = 30.0 + 50.0 * (it / n_iters)
        shifted = beta_t * (c - c_max)
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta_t

        a_rev = a[::-1]
        M2_fft = M_fft
        PA = np.fft.rfft(p, M2_fft)
        AAr = np.fft.rfft(a_rev, M2_fft)
        h = np.fft.irfft(AAr * PA, M2_fft)[:2 * n - 1][n - 1:2 * n - 1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
        grad_n = np.linalg.norm(grad)
        if grad_n > 1e-15:
            step = eta * grad / grad_n
            a_new = a * np.exp(-step)
            a_new = np.maximum(a_new, 1e-15)
            S_new = a_new.sum()
            a_new = a_new / S_new
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                eta = min(eta * 1.02, 0.15)
            else:
                eta *= 0.85
        eta = max(eta, 0.002)
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    peak-targeted coordinate descent, Fienup spectral shaping,
    multiplicative weight updates, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) / 2048.0

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform n=2048")

    n_parent = len(best_a)
    print(f"Initial best: {best_score:.10f} (n={n_parent})")

    # Phase 1: Multi-length + Fienup (12% time)
    p1_end = t_start + 0.12 * (total - 30.0)
    print(f"\n=== Phase 1: Multi-start + Fienup ({p1_end - t_start:.0f}s) ===")

    lengths_to_try = [n_parent]
    if n_parent >= 512:
        lengths_to_try.append(n_parent // 2)
    lengths_to_try.append(min(8192, n_parent * 2))
    lengths_to_try.append(min(4096, int(n_parent * 1.5)))
    lengths_to_try = sorted(set(lengths_to_try))

    for n_try in lengths_to_try:
        if time.monotonic() > p1_end:
            break
        a_try = interpolate_seq(best_a, n_try)
        S = a_try.sum()
        if S > 1e-12:
            a_try = a_try / S
        sc, a_opt = run_lbfgs(a_try, 500, maxiter=4000, maxfun=80000)
        update(sc, a_opt, f"lbfgs-n{n_try}")

    if time.monotonic() < p1_end:
        a_f = fienup_v2(best_a, n_iters=200, deadline=p1_end)
        sc = compute_score(a_f)
        update(sc, a_f, "fienup")
        if sc < best_score:
            sc2, a2 = run_lbfgs(a_f, 1000, maxiter=5000, maxfun=100000)
            update(sc2, a2, "fienup+lbfgs")

    # Phase 2: L-BFGS beta sweep (38% time)
    p2_end = t_start + 0.50 * (total - 30.0)
    print(f"\n=== Phase 2: L-BFGS beta sweep ({p2_end - t_start:.0f}s) ===")

    betas = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]
    for beta in betas:
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 15:
            break
        maxiter = int(min(25000, max(2000, remaining * 12)))
        print(f"  beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s)", end=" ")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter * 25)
        dt = time.monotonic() - t0
        print(f"score={sc:.10f} ({dt:.1f}s)")
        update(sc, a_opt, f"lbfgs-b{beta}")

    # Phase 3: Multiplicative weights + Coord descent (20% time)
    p3_end = t_start + 0.70 * (total - 30.0)
    print(f"\n=== Phase 3: Multi-weight + Coord descent ({p3_end - t_start:.0f}s) ===")

    if time.monotonic() < p3_end:
        a_mw = multiplicative_weights(best_a, n_iters=2000, deadline=p3_end)
        sc = compute_score(a_mw)
        update(sc, a_mw, "multiplicative")
        if sc < best_score:
            sc2, a2 = run_lbfgs(a_mw, 20000, maxiter=8000, maxfun=160000)
            update(sc2, a2, "mw+polish")

    if time.monotonic() < p3_end:
        a_cd = peak_cd(best_a, n_sweeps=80, deadline=p3_end)
        sc = compute_score(a_cd)
        update(sc, a_cd, "peak-cd")
        if sc < best_score:
            sc2, a2 = run_lbfgs(a_cd, 50000, maxiter=8000, maxfun=160000)
            update(sc2, a2, "cd+polish")

    # Phase 4: Exploration (18% time)
    p4_end = t_start + 0.88 * (total - 30.0)
    print(f"\n=== Phase 4: Exploration ({p4_end - t_start:.0f}s) ===")

    for r in range(30):
        if time.monotonic() > p4_end:
            break
        strategy = r % 5
        n = len(best_a)
        if strategy == 0:
            a_pert = best_a * (1.0 + 0.015 * rng.standard_normal(n))
        elif strategy == 1:
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_s = max(0, k_star - n + 1)
            j_e = min(k_star, n - 1)
            a_pert = best_a.copy()
            a_pert[j_s:j_e + 1] *= (1.0 + 0.04 * rng.standard_normal(j_e - j_s + 1))
        elif strategy == 2:
            x = np.arange(n) / n
            freq = int(rng.integers(1, 6))
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = best_a * (1.0 + 0.02 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 3:
            block = max(1, n // 40)
            start = int(rng.integers(0, max(1, n - block)))
            a_pert = best_a.copy()
            a_pert[start:start + block] *= (1.0 + 0.06 * rng.standard_normal(block))
        else:
            # Try a slightly different length + re-interpolate
            n_new = n + int(rng.integers(-n // 20, n // 20 + 1))
            n_new = max(128, min(8192, n_new))
            a_pert = interpolate_seq(best_a, n_new)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
            continue

        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=3000, maxfun=60000)
        if sc < best_score:
            update(sc, a_opt, f"explore-{r}")

    # Phase 5: Final high-beta polish (10% time)
    print(f"\n=== Phase 5: Final polish ===")

    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 500000, maxiter=8000, maxfun=160000)
        update(sc, a_opt, "final-500k")

    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 2000000, maxiter=5000, maxfun=100000)
        update(sc, a_opt, "final-2M")

    if time.monotonic() < deadline - 5:
        a_cd = peak_cd(best_a, n_sweeps=15, deadline=deadline)
        sc = compute_score(a_cd)
        update(sc, a_cd, "final-cd")

    if time.monotonic() < deadline - 2:
        sc, a_opt = run_lbfgs(best_a, 5000000, maxiter=3000, maxfun=60000)
        update(sc, a_opt, "final-5M")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total}s")
    print(f"{'=' * 60}")
    return out
