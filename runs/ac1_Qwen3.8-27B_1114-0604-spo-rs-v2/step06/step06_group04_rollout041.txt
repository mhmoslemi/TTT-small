```python
def _min1d(f, a, b, iters=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return 0.5 * (a + b)
```

```python
"""
```

```python
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} in exact manner."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.05 / g_norm
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * (a - y if it > 0 else np.zeros(n))
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.05 * it)
        x_new = y_norm - step * grad_y
        x_new = _simplex_project(x_new)
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if it > 0 and it % 100 == 0:
            if not (sc < best_score + 1e-10):
                eta *= 0.7
    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=None):
    a = _normalize(a0)
    n = a.size
    if maxfun is None:
        maxfun = 500

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, 1.5)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": max(maxfun // 2, 50),
            "maxfun": int(maxfun), "maxls": 40}
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, args=(),
                       bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a
    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _golden_search(f, a, b, iters=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return 0.5 * (a + b)


def _coord_descent_pass(a, S_conv, order, n_gs_iters=35):
    """One pass of coordinate descent. a has sum=1, S_conv is the full convolution."""
    n = a.size
    S_conv = S_conv.astype(np.float64).copy()
    a = a.astype(np.float64).copy()

    for i in order:
        ai = a[i]
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        slen = k_end - k_start + 1
        a_slice = a[:slen]

        B = S_conv[k_start:k_end + 1] - 2.0 * ai * a_slice
        C = 2.0 * a_slice.copy()

        has_quad = (2 * i >= k_start) and (2 * i <= k_end)
        S_quad = S_conv[2 * i] - ai * ai if has_quad else -1e30

        M_other = 0.0
        if k_start > 0:
            M_other = max(M_other, float(np.max(S_conv[:k_start])))
        if k_end < 2 * n - 2:
            M_other = max(M_other, float(np.max(S_conv[k_end + 1:])))

        S_total = 1.0

        def obj(t):
            lin_max = float(np.max(B + C * t))
            quad_val = S_quad + t * t
            val = max(lin_max, quad_val, M_other)
            denom = S_total - ai + t
            if denom < 1e-12:
                return 1e10
            return val / (denom * denom)

        t_hi = min(1.0, max(ai * 3.0, 1.0 / n * 5.0))
        t_lo = 0.0

        f0 = obj(0.0)
        fai = obj(ai)
        fhi = obj(t_hi)

        if f0 <= fai and f0 <= fhi:
            t_new = _golden_search(obj, t_lo, max(ai, t_hi * 0.1), n_gs_iters)
        elif fhi <= fai and fhi <= f0:
            t_new = _golden_search(obj, min(ai, t_hi * 0.1), t_hi, n_gs_iters)
        else:
            t_new = _golden_search(obj, t_lo, t_hi, n_gs_iters)

        t_new = float(np.clip(t_new, 0.0, 1.0 - 1e-10))
        delta = t_new - ai
        if abs(delta) > 1e-14:
            a[i] = t_new
            S_conv[k_start:k_end + 1] += 2.0 * delta * a_slice
            if has_quad:
                S_conv[2 * i] += delta * delta

    sc = _exact_score(a)
    return a, S_conv, sc


def _coord_descent_optimize(a0, n_passes=20, deadline=None, rng=None):
    """Run multiple passes of coordinate descent with random ordering."""
    a = _normalize(a0)
    n = a.size
    S_conv = _fast_self_conv(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    if rng is None:
        rng = np.random.default_rng(0)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > float(deadline) - 3.0:
            break
        order = rng.permutation(n)
        a, S_conv, sc = _coord_descent_pass(a, S_conv, order)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            if p > 0 and p % 5 == 0:
                conv_diff = abs(sc - best_score)
                if conv_diff < 1e-8:
                    break
    return best_score, best_a


def _anneal_coord_descent(a0, deadline, rng, n_anneal_rounds=5):
    """Simulated annealing: coordinate descent + perturb + coordinate descent."""
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    n = a.size
    chunk_time = max(5.0, (deadline - time.monotonic()) / (n_anneal_rounds + 1))

    for r in range(n_anneal_rounds):
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        sub_dead = time.monotonic() + chunk_time * 0.6
        sc, a_cd = _coord_descent_optimize(a, n_passes=5, deadline=sub_dead, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()

        # Perturb
        strength = 0.15 / (1.0 + 0.3 * r)
        n_perturb = max(3, n // 200)
        indices = rng.integers(0, n, size=n_perturb)
        a_pert = a_cd.copy()
        for idx in indices:
            a_pert[idx] *= (1.0 + strength * (rng.random() - 0.5))
        a_pert = _normalize(np.clip(a_pert, 0.0, None))

        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        sub_dead2 = min(deadline - 3.0, time.monotonic() + chunk_time * 0.3)
        sc2, a_cd2 = _coord_descent_optimize(a_pert, n_passes=3, deadline=sub_dead2, rng=rng)
        if sc2 < best_score:
            best_score = sc2
            best_a = a_cd2.copy()
            a = a_cd2.copy()
        else:
            # Accept with probability (Metropolis)
            delta = sc2 - best_score
            T = 0.001 * max(1.0, best_score)
            if rng.random() < np.exp(-delta / T):
                a = a_cd2.copy()
            else:
                a = a_cd.copy()

    # Final polish
    rem = deadline - time.monotonic()
    if rem > 10.0:
        sc, a_final = _coord_descent_optimize(best_a, n_passes=10, deadline=deadline, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()

    return best_score, best_a

        for beta in [200000, 1000000, 3000000]:
            if time.monotonic() > t0 + 0.5 * total_time:
                break
            maxfun = max(30, min(500, int((t0 + 0.5 * total_time - time.monotonic()) * 5)))
            sc, a2 = _lbfgs_lse(a, beta, t0 + 0.5 * total_time, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] LSE hi beta={:.0f}: {:.8f}".format(label, beta, best_score))
            a = _normalize(a2)

    # Stage 3: Nesterov
    if time.monotonic() < t0 + 0.75 * total_time:
        for k_act in [2, 3, 5]:
            if time.monotonic() > t0 + 0.75 * total_time:
                break
            sc, a2 = _nesterov_optimize(a, n_iters=2000, deadline=t0 + 0.75 * total_time,
                                        k_active=k_act, tol=1e-7)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Nesterov k={}: {:.8f}".format(label, k_act, best_score))
            a = _normalize(a2)

    # Stage 4: Coordinate descent
    rem = deadline - time.monotonic()
    if rem > 20.0:
        rng = np.random.default_rng(hash(label) % (2**31))
        sc, a2 = _coord_descent_optimize(best_a, n_passes=30, deadline=deadline, rng=rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] CoordDesc: {:.8f}".format(label, best_score))

    return best_score, best_a


def _generate_inits(n, rng, parent, cap=30):
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
        # Blended reversals
        for t in [0.3, 0.5, 0.7]:
            _add((1 - t) * pr + t * pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    _add(np.ones(n))
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0, 3.0]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for alpha in [0.3, 0.6, 1.0, 1.5]:
        _add(1.0 + alpha * (1.0 - np.abs(2.0 * x - 1.0) ** 2))
    for f in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.4, 0.7, 1.0]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
    for rate in [1.0, 2.0, 4.0, 6.0, 10.0]:
        _add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    for w in [0.08, 0.12, 0.15, 0.2, 0.3]:
        for ctr in [0.1, 0.15, 0.2, 0.25, 0.3]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w)) ** 2) +
                 np.exp(-(((x - (0.5 + ctr)) / w)) ** 2) + 0.01)
    # Flat-top with edges
    for edge_w in [0.05, 0.1, 0.15, 0.2]:
        mask = np.ones(n)
        edge = int(edge_w * n)
        for e in range(edge):
            mask[e] = 0.3 + 0.7 * e / max(1, edge)
            mask[-1 - e] = 0.3 + 0.7 * e / max(1, edge)
        _add(mask)
    # Sine power
    for p in [2, 4, 6]:
        _add(np.abs(np.sin(np.pi * x)) ** p + 0.01)
    # W-shape
    for depth in [0.3, 0.5, 0.7]:
        _add(1.0 - depth * np.exp(-((x - 0.5) / 0.1) ** 2))

    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.2, 1.0)) * noise)

    for _ in range(3):
        _add(0.05 + 0.95 * rng.random(n))

    return inits


def _quick_opt(a, deadline):
    """Quick multi-method optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    # LSE
    for beta in [2000, 20000, 200000]:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        maxfun = max(30, min(400, int(rem * 5)))
        sc, a2 = _lbfgs_lse(a, beta, deadline, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    # Nesterov
    rem = deadline - time.monotonic()
    if rem > 8.0:
        sc, a2 = _nesterov_optimize(a, n_iters=int(rem * 10), deadline=deadline,
                                    k_active=3, tol=1e-7)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    # Coord descent
    rem = deadline - time.monotonic()
    if rem > 8.0:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
        sc, a2 = _coord_descent_optimize(a, n_passes=int(rem * 0.8), deadline=deadline, rng=rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns list of non-negative floats."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths
    if parent is not None:
        parent_len = parent.size
        lengths = [parent_len]
        for L in [8192, 10240, 12288, 16384, 20480]:
            if abs(L - parent_len) > 200:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 16384, 12288]

    lengths = [L for L in lengths if 4096 <= L <= 20480][:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Initialization (10% of budget)
    init_end = t_start + 0.10 * (deadline - t_start)
    all_scores = []
    all_seqs = []

    for n in lengths:
        inits = _generate_inits(n, rng, parent, cap=25)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)

    order = np.argsort(all_scores)
    print("Best init: {:.8f} (len={})".format(all_scores[order[0]], all_seqs[order[0]].size))

    # Select top diverse starts (different lengths preferred)
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 3:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    while len(starts) < 2:
        base = starts[0].copy()
        noise = 0.1 * rng.standard_normal(base.size)
        starts.append(_normalize(base * np.exp(noise)))

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f} (len={})".format(best_score, best_seq.size))

    # Phase 2: Main optimization (50% of budget)
    main_end = t_start + 0.60 * (deadline - t_start)
    n_starts = len(starts)
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.50, 0.30, 0.20]

    starts_sorted = sorted(range(n_starts), key=lambda i: _exact_score(starts[i]))
    cum = 0.0
    main_budget = main_end - t_start

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)
        if time.monotonic() > sub_dead - 10.0:
            continue

        sq = starts[si].copy()
        if rank > 0 and len(best_seq) != len(sq):
            best_r = _resample(best_seq, len(sq))
            sq = _normalize(0.6 * sq + 0.4 * best_r)

        sc, seq = _nesterov_lbfgs_hybrid(sq, sub_dead, "main{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After main{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: Coordinate descent intensive polishing (25% of budget)
    cd_end = t_start + 0.85 * (deadline - t_start)
    if time.monotonic() < cd_end - 30.0:
        # Multiple rounds of coord descent with perturbations
        cd_dead = cd_end
        sc, a2 = _coord_descent_optimize(best_seq, n_passes=50, deadline=cd_dead - 60, rng=rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("CoordDesc intensive: {:.8f}".format(best_score))

        # Annealing
        rem = cd_end - time.monotonic()
        if rem > 30.0:
            sc, a2 = _anneal_coord_descent(best_seq, cd_end, rng, n_anneal_rounds=4)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Annealing: {:.8f}".format(best_score))

    # Phase 4: Final polish (remaining time)
    final_dead = deadline - 5.0
    if time.monotonic() < final_dead - 20.0:
        # Very high beta LSE
        for beta in [5000000, 10000000]:
            rem = final_dead - time.monotonic()
            if rem < 10.0:
                break
            maxfun = max(20, min(300, int(rem * 3)))
            sc, a2 = _lbfgs_lse(best_seq, beta, final_dead - 5, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
            best_seq = _normalize(a2)

        # Final coord descent
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            sc, a2 = _coord_descent_optimize(best_seq, n_passes=int(rem * 0.5),
                                             deadline=final_dead, rng=rng)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final CoordDesc: {:.8f}".format(best_score))

        # Final Nesterov with tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            sc, a2 = _nesterov_optimize(best_seq, n_iters=int(rem * 15),
                                        deadline=final_dead, k_active=2, tol=1e-9)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov: {:.8f}".format(best_score))

    # Try a different length if time remains
    rem = deadline - time.monotonic()
    if rem > 60.0:
        alt_n = best_seq.size + (2048 if best_seq.size < 14000 else -2048)
        if 4096 <= alt_n <= 20480:
            alt_seq = _resample(_normalize(best_seq), alt_n)
            sc, a2 = _quick_opt(alt_seq, deadline - 5.0)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Alt length {}: {:.8f}".format(alt_n, best_score))

    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
```
