def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient on the simplex."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, tol=1e-6):
    """
    Exact subgradient of 2*n*max(c)/S^2 on the simplex.
    Uses all peaks within relative tolerance of the max.
    Returns (score, gradient).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])

    # Cap active set to top 50 for numerical stability
    if len(active) > 50:
        top_idx = np.argsort(d[active])[-50:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _fista_optimize(a0, n_iters=3000, deadline=None, tol_active=1e-6, eta0=None):
    """
    FISTA (accelerated proximal gradient) with exact subgradient on the simplex.
    Uses backtracking for step size.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    # Step size: L is Lipschitz constant of gradient of active quadratic
    # L <= 4n (conservative), use 1/(4n) as initial
    if eta0 is None:
        eta = 1.0 / (4.0 * n)
    else:
        eta = float(eta0)

    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    no_improve = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v

        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy

        score_y, grad_y = _subgrad_exact(y, tol=tol_active)
        if not np.isfinite(score_y):
            break

        gn = np.linalg.norm(grad_y)
        if gn < 1e-14:
            break

        # Backtracking line search (simplified: halve eta if no improvement)
        step = eta
        x_new = y - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            break
        x_new = x_new / Sx

        sc = _exact_score(x_new)

        if not np.isfinite(sc) or sc > best_score * 1.1:
            # Reduce step size
            eta *= 0.5
            continue

        v = x_new - a
        a = x_new
        t = t_new

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            # Slightly increase step if we're improving well
            eta = min(eta * 1.02, 1.0 / (2.0 * n))
        else:
            no_improve += 1
            if no_improve > 100:
                eta *= 0.8
                no_improve = 0

    return best_score, best_a


def _lbfgs_lse(a0, beta, deadline, maxfun=300):
    """Run L-BFGS-B on the LSE-smoothed objective on the simplex."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 30),
        "maxfun": int(maxfun),
        "maxls": 30,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _multi_stage_optimize(a0, deadline, label=""):
    """
    Multi-stage optimization: progressive LSE + FISTA + final polish.
    """
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total_time = max(5.0, float(deadline - t0 - 5.0))

    print("  [{}] start: {:.8f} n={}".format(label, best_score, a.size))

    # Stage 1: LSE with moderate beta (30% time)
    stage1_end = t0 + 0.30 * total_time
    betas1 = [100, 500, 2000]
    cum = 0.0
    for beta in betas1:
        rem = stage1_end - time.monotonic()
        if rem < 5.0:
            break
        n = len(a)
        maxfun = int(max(20, min(500, rem * 8)))
        sc, a2 = _lbfgs_lse(a, beta, stage1_end - 2, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("  [{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Stage 2: LSE with high beta (30% time)
    stage2_end = t0 + 0.60 * total_time
    betas2 = [10000, 50000, 200000]
    for beta in betas2:
        rem = stage2_end - time.monotonic()
        if rem < 5.0:
            break
        maxfun = int(max(20, min(400, rem * 6)))
        sc, a2 = _lbfgs_lse(a, beta, stage2_end - 2, maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("  [{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
        a = _normalize(a2)

    # Stage 3: FISTA with exact subgradient (25% time)
    stage3_end = t0 + 0.85 * total_time
    rem = stage3_end - time.monotonic()
    if rem > 5.0:
        n_it = int(max(500, rem * 50))
        for tol_act in [1e-5, 1e-7, 1e-9]:
            rem = stage3_end - time.monotonic()
            if rem < 5.0:
                break
            sc, a2 = _fista_optimize(a, n_iters=int(max(200, rem * 40)),
                                      deadline=stage3_end, tol_active=tol_act)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("  [{}] FISTA tol={:.0e}: {:.8f}".format(label, tol_act, best_score))
            a = _normalize(a2)

    # Stage 4: Final polish (15% time)
    rem = deadline - time.monotonic()
    if rem > 10.0:
        # Very high beta LSE
        for beta in [500000, 2000000]:
            rem = deadline - time.monotonic()
            if rem < 5.0:
                break
            maxfun = int(max(20, min(300, rem * 5)))
            sc, a2 = _lbfgs_lse(a, beta, deadline - 2, maxfun)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("  [{}] Final LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
            a = _normalize(a2)

        # Final FISTA with very tight tolerance
        rem = deadline - time.monotonic()
        if rem > 5.0:
            sc, a2 = _fista_optimize(a, n_iters=int(max(100, rem * 60)),
                                      deadline=deadline, tol_active=1e-10)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("  [{}] Final FISTA: {:.8f}".format(label, best_score))
            a = _normalize(a2)

    return best_score, best_a


def _generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def _add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    # Uniform
    _add(np.ones(n))

    # Gaussian envelope
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
        _add(np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)) + 0.01)

    # Cosine modulations of uniform
    for f in [1, 2, 3, 4, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))

    # Bimodal
    for w in [0.08, 0.12, 0.15, 0.2]:
        for ctr in [0.15, 0.2, 0.25, 0.3]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) +
                 np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)

    # V-shape
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5]:
        _add(1.0 + alpha * np.abs(2.0 * x - 1.0))

    # Inverted parabola
    for alpha in [0.5, 1.0, 2.0]:
        _add(1.0 + alpha * (1.0 - (2.0 * x - 1.0) ** 2))

    # Sine absolute
    for f in [1, 2, 3]:
        _add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth (noise in frequency domain)
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.5, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        amp = float(rng.uniform(0.3, 1.2))
        if noise.std() > 1e-10:
            noise = noise / noise.std()
        _add(1.0 + amp * noise)

    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    # Smooth random (Gaussian filtered)
    for sigma in [0.02, 0.05, 0.1]:
        raw = 0.1 + 0.9 * rng.random(n)
        sm = gaussian_filter1d(raw, sigma=sigma * n)
        _add(sm)

    return inits


def _perturb(a, rng, strength=None):
    """Generate a perturbed version of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 5))

    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.5 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Low-frequency smooth noise
        K = min(61, max(11, n // 200))
        if K % 2 == 0:
            K -= 1
        sig = max(1.0, (K // 2) / 3.0)
        kern = np.exp(-((np.arange(K) - K // 2) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        if noise.std() > 1e-10:
            noise = noise / noise.std()
        q = a * np.exp(strength * 0.3 * noise)
    else:
        # Gaussian filter perturbation
        raw = rng.standard_normal(n)
        sm = gaussian_filter1d(raw, sigma=max(2, n // 500))
        sm = (sm - sm.mean()) / (sm.std() + 1e-10)
        q = a * np.exp(strength * 0.3 * sm)

    return _normalize(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns list of non-negative floats.
    Strategy: multi-length sweep with progressive LSE-LBFGS and FISTA optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 1e-6:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = parent.size
        if parent_len > 16384:
            parent_len = 16384
        lengths = sorted(set([parent_len, 8192, 10240, 12288, 14336, 16384]),
                         reverse=True)
    else:
        lengths = [10240, 12288, 8192, 14336, 16384]

    lengths = [L for L in lengths if 4096 <= L <= 20480][:6]
    print("Lengths to try: {}".format(lengths))

    # Global best
    best_score_global = float(np.inf)
    best_seq_global = None

    # Phase 1: Quick evaluation of initializations for each length
    candidates = []  # (score, sequence, length)

    for n in lengths:
        inits = _generate_inits(n, rng, parent)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc) and sc < best_score_global + 0.5:
                candidates.append((sc, arr, n))
                if sc < best_score_global:
                    best_score_global = sc
                    best_seq_global = arr.copy()

    candidates.sort(key=lambda x: x[0])
    print("Best init: {:.8f} (len={})".format(
        candidates[0][0], candidates[0][1].size if candidates else 0))

    # Phase 2: Select top candidates for full optimization
    # Pick diverse candidates (different lengths, different structures)
    selected = []
    used_lens = set()
    for sc, arr, n in candidates:
        if len(selected) >= 4:
            break
        if n not in used_lens:
            selected.append(arr.copy())
            used_lens.add(n)

    # Fill with best remaining if needed
    if len(selected) < 3:
        for sc, arr, n in candidates:
            if len(selected) >= 3:
                break
            if arr is not selected[0] and arr.size == selected[0].size:
                selected.append(arr.copy())

    if not selected:
        selected = [np.ones(10240) / 10240]

    # Time allocation for optimization
    n_sel = len(selected)
    main_end = deadline - 300.0
    main_budget = max(60.0, main_end - t_start)

    # Allocate more time to better candidates
    if n_sel == 1:
        fracs = [1.0]
    elif n_sel == 2:
        fracs = [0.65, 0.35]
    elif n_sel == 3:
        fracs = [0.50, 0.30, 0.20]
    else:
        fracs = [0.40, 0.25, 0.20, 0.15]

    cum = 0.0
    for i, arr in enumerate(selected):
        cum += fracs[i]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 15.0:
            continue

        # For non-primary starts, blend with current best
        if i > 0 and best_seq_global is not None:
            if len(best_seq_global) != len(arr):
                best_r = _resample(best_seq_global, len(arr))
            else:
                best_r = best_seq_global.copy()
            t_blend = 0.3 + 0.1 * i
            arr = _normalize((1.0 - t_blend) * _normalize(arr) +
                             t_blend * _normalize(best_r))

        sc, seq = _multi_stage_optimize(arr, sub_dead, "opt{}_n{}".format(i + 1, arr.size))
        if np.isfinite(sc) and sc < best_score_global:
            best_score_global = sc
            best_seq_global = seq.copy()
            print("After opt{}: {:.8f} (n={})".format(i + 1, best_score_global, seq.size))

    best_seq_global = _normalize(best_seq_global)
    print("After main phase: {:.8f}".format(best_score_global))

    # Phase 3: Perturbation restarts from best
    perturb_end = deadline - 150.0
    n_perturb = 4
    if perturb_end > time.monotonic() + 30.0:
        chunk = (perturb_end - time.monotonic()) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = time.monotonic() + chunk
            if time.monotonic() > perturb_end - 10.0:
                break

            p = _perturb(best_seq_global, rng, strength=0.1)
            sc, seq = _multi_stage_optimize(p, chunk_dead, "pert{}".format(pi + 1))
            if np.isfinite(sc) and sc < best_score_global:
                best_score_global = sc
                best_seq_global = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score_global))

    # Phase 4: Final intensive polish
    final_dead = deadline - 10.0
    if time.monotonic() < final_dead - 20.0:
        # Very high beta LSE
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            for beta in [500000, 2000000, 5000000]:
                rem = final_dead - time.monotonic()
                if rem < 5.0:
                    break
                maxfun = int(max(20, min(300, rem * 5)))
                sc, a2 = _lbfgs_lse(best_seq_global, beta, final_dead - 3, maxfun)
                if np.isfinite(sc) and sc < best_score_global:
                    best_score_global = sc
                    best_seq_global = a2.copy()
                    print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score_global))
                best_seq_global = _normalize(a2)

        # Final FISTA with very tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            for tol_act in [1e-9, 1e-11]:
                rem = final_dead - time.monotonic()
                if rem < 5.0:
                    break
                n_it = int(max(100, rem * 50))
                sc, a2 = _fista_optimize(best_seq_global, n_iters=n_it,
                                         deadline=final_dead, tol_active=tol_act)
                if np.isfinite(sc) and sc < best_score_global:
                    best_score_global = sc
                    best_seq_global = a2.copy()
                    print("Final FISTA tol={:.0e}: {:.8f}".format(tol_act, best_score_global))
                best_seq_global = _normalize(a2)

        # One more LSE pass
        rem = final_dead - time.monotonic()
        if rem > 5.0:
            sc, a2 = _lbfgs_lse(best_seq_global, 3000000, final_dead - 2,
                                 int(max(20, rem * 4)))
            if np.isfinite(sc) and sc < best_score_global:
                best_score_global = sc
                best_seq_global = a2.copy()
            best_seq_global = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq_global, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score_global

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score_global, verified, len(out), time.monotonic() - t_start))
    return out
