"""
Search for a non-negative discrete step function minimizing
    2*n*max(np.convolve(a,a)) / (sum(a)**2)

The optimizer is a hybrid:
  * a primal-dual mirror descent method that uses a temperature-controlled
    distribution over convolution lags to approximate the max operator;
  * log-domain Adam on a log-sum-exp smoothed objective with the correct
    scale-invariant gradient;
  * L-BFGS polish in a centered log-simplex parameterization;
  * periodic diversity restarts.

The final sequence is normalized to sum 1 and returned as a Python list.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    N = next_power_of_two(m)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:m]
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(out, 0.0)


def corr_pair_n(a, w):
    """
    h[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    for w of length 2n-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    m = int(w.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    ar = a[::-1]
    A = np.fft.rfft(ar, N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    start = n - 1
    out_len = min(n, max(0, M - start))
    h = full[start:start + out_len]
    if h.size < n:
        h = np.concatenate([h, np.zeros(n - h.size, dtype=np.float64)])
    return np.nan_to_num(h[:n], nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.empty(0, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
    arr = np.clip(arr, 0.0, None)
    s = float(np.sum(arr))
    if s < 1e-12:
        return np.full(arr.size, 1.0 / float(arr.size), dtype=np.float64)
    return arr / s


def score_from_conv(a, conv):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    s = float(np.sum(a))
    if n == 0 or s < 1e-12:
        return float("inf")
    if conv.size == 0:
        return float("inf")
    cm = float(np.max(conv))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float(2.0 * n * cm / (s * s))


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    c = fast_self_conv(a)
    if c.size == 0:
        return float("inf")
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float(2.0 * n * cm / (s * s))


def resample_sequence(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = int(n)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = int(arr.size)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if m == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    out = np.interp(x_new, x_old, arr)
    return np.nan_to_num(out, nan=0.0, posinf=1e6, neginf=0.0)


def softmax_from_theta(theta):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    n = int(theta.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    y = theta - float(np.mean(theta))
    y -= float(np.max(y))
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def log_centered(a):
    a = normalize_a(a)
    th = np.log(np.maximum(a, 1e-300))
    th = th - float(np.mean(th))
    return np.nan_to_num(th, nan=0.0, posinf=0.0, neginf=0.0)


def lse_score_and_grad(a, beta):
    """
    Returns (exact_max_score, smooth_lse_objective, grad)
    for F(a)=2*L where L is a log-sum-exp surrogate of max b/sum^2.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf"), float("inf"), np.empty(0, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        z = np.zeros(n, dtype=np.float64)
        return float("inf"), float("inf"), z

    c = fast_self_conv(a)
    if c.size != 2 * n - 1:
        c = fast_self_conv(np.resize(a, n))
    cm = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    exact = float(2.0 * n * cm / (s * s))

    d = (float(n) / (s * s)) * c
    dmax = float(np.max(d)) if d.size else 0.0
    if not np.isfinite(dmax) or dmax <= 0.0:
        return exact, max(0.0, 2.0 * dmax), np.zeros(n, dtype=np.float64)

    z = float(beta) * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        w = np.zeros_like(d)
        idx = np.where(d >= dmax - 1e-12)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(d))])
        w[idx] = 1.0 / float(idx.size)
        lse = dmax
    else:
        w = e / se
        lse = dmax + math.log(se) / float(beta)

    wdotd = float(np.dot(w, d))
    h = corr_pair_n(a, w)
    grad = (4.0 * n / (s * s)) * h - (4.0 / s) * wdotd
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return exact, float(2.0 * lse), grad


def log_lbfgs_fun(theta, beta):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    a = softmax_from_theta(theta)
    n = int(a.size)
    if n == 0:
        return 1e30, np.empty(0, dtype=np.float64)
    exact, val, grad = lse_score_and_grad(a, beta)
    if not np.isfinite(val):
        return 1e30, np.zeros(n, dtype=np.float64)
    grad = grad - float(np.mean(grad))
    grad_theta = a * grad
    grad_theta = grad_theta - float(np.mean(grad_theta))
    grad_theta = np.nan_to_num(grad_theta, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), grad_theta


def log_lbfgs_polish(a, deadline, tag="lbfgs"):
    a = normalize_a(a)
    n = int(a.size)
    if n == 0 or scipy_minimize is None:
        return exact_score(a), a

    best_a = a.copy()
    best_sc = exact_score(a)
    theta0 = log_centered(a)
    betas = (30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0, 30000.0)

    for beta in betas:
        rem = max(0.0, deadline - time.monotonic())
        if rem < 20.0:
            break
        maxfun = int(min(300, max(60, rem * 2.0)))
        maxiter = int(min(120, max(20, maxfun // 2)))
        opts = {
            "ftol": 1e-15,
            "gtol": 1e-10,
            "maxiter": maxiter,
            "maxfun": maxfun,
            "maxls": 20,
        }
        try:
            res = scipy_minimize(
                log_lbfgs_fun,
                theta0.copy(),
                jac=True,
                args=(float(beta),),
                method="L-BFGS-B",
                options=opts,
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == n and np.all(np.isfinite(x)):
                a_out = softmax_from_theta(x)
                sc = exact_score(a_out)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_out.copy()
                    theta0 = log_centered(best_a)
                    print("  [{}] beta={:.0f} new best {:.9f}".format(tag, beta, best_sc))

    return float(best_sc), best_a


def mirror_minimax(a, deadline, tag="mirror"):
    """
    Mirror descent primal-dual heuristic for min_a max_k (a*a)_k / sum(a)^2.
    Uses p_k = (b_k / max b)^q as the subgradient distribution.
    """
    a = normalize_a(a)
    n = int(a.size)
    if n == 0:
        return float("inf"), a

    best_a = a.copy()
    best_sc = exact_score(a)
    conv = fast_self_conv(a)
    current_sc = score_from_conv(a, conv)
    if current_sc < best_sc:
        best_sc = current_sc
        best_a = a.copy()

    q = 2.5
    lr = 0.35
    stagnant = 0
    it = 0

    print("  [{}] start incl. {:.9f}".format(tag, best_sc))

    while time.monotonic() < deadline:
        it += 1

        conv = fast_self_conv(a)
        current_sc = score_from_conv(a, conv)
        if not np.isfinite(current_sc):
            a = best_a.copy()
            current_sc = best_sc

        if current_sc < best_sc:
            best_sc = current_sc
            best_a = a.copy()

        dmax = float(np.max(conv)) if conv.size else 0.0
        if not np.isfinite(dmax) or dmax <= 0.0:
            break

        d = conv / dmax
        p = np.power(d, q)
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        ps = float(np.sum(p))
        if ps <= 1e-300 or not np.isfinite(ps):
            p = np.ones_like(conv) / float(conv.size)
        else:
            p = p / ps

        h = corr_pair_n(a, p)
        hc = h - float(np.mean(h))
        hscale = float(np.max(np.abs(hc)))
        if not np.isfinite(hscale) or hscale < 1e-14:
            stagnant += 1

        else:
            hnorm = hc / hscale
            best_trial = a
            best_trial_sc = current_sc
            best_fac = 1.0

            for fac in (1.0, 0.5, 2.0):
                arg = -lr * float(fac) * hnorm
                arg = np.clip(arg, -10.0, 10.0)
                at = a * np.exp(arg)
                at = normalize_a(at)
                ct = fast_self_conv(at)
                st = score_from_conv(at, ct)
                if np.isfinite(st) and st < best_trial_sc:
                    best_trial = at
                    best_trial_sc = st
                    best_fac = fac

            if best_trial_sc < current_sc - 1e-12:
                a = best_trial
                current_sc = best_trial_sc
                stagnant = 0

                if best_fac > 1.5:
                    lr = min(2.0, lr * 1.15)
                elif best_fac < 0.6:
                    lr = max(0.02, lr * 0.80)
                else:
                    lr = min(2.0, lr * 1.02)

                if current_sc < best_sc:
                    best_sc = current_sc
                    best_a = a.copy()
                    if tag and (it % 200 == 0 or current_sc < 1.51):
                        print("  [{}] it={} score={:.9f} q={:.1f} lr={:.3f}".format(
                            tag, it, best_sc, q, lr))
            else:
                stagnant += 1
                lr = max(0.02, lr * 0.72)

        if stagnant > 25:
            if q < 50.0:
                q *= 1.6
            elif q < 220.0:
                q *= 1.3
            else:
                q = 2.5
                lr = max(0.12, lr * 1.25)
            stagnant = 0
            a = normalize_a(0.80 * a + 0.20 * best_a)

        if it % 400 == 0:
            if time.monotonic() > deadline - 2.0:
                break

    if best_sc > current_sc:
        best_sc = current_sc
        best_a = a.copy()

    if tag:
        print("  [{}] finish best {:.9f}".format(tag, best_sc))
    return float(best_sc), best_a

                    m = np.zeros(n, dtype=np.float64)

def append_init(inits, cap, arr, n):
    if len(inits) >= int(cap):
        return
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return
    if arr.size != n:
        arr = resample_sequence(arr, n)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
    if arr.size:
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.01
    p = normalize_a(arr)
    if p.size == n and float(np.sum(p)) > 0.0:
        inits.append(p)

            append_init(inits, cap, par[::-1], n)

    # unite
    append_init(inits, cap, np.ones(n), n)

    # simple shapes
    for alpha in (0.2, 0.4, 0.7, 1.0, 1.5, 2.0):
        append_init(inits, cap, 1.0 - alpha * (2.0 * x - 1.0) ** 2, n)
        append_init(inits, cap, 1.0 + alpha * (2.0 * x - 1.0) ** 2, n)

    for f in (1, 2, 3, 4, 5, 6):
        for alpha in (0.25, 0.5, 0.75, 1.0, 1.5):
            append_init(inits, cap, 1.0 + alpha * np.cos(2.0 * np.pi * f * x), n)
            append_init(inits, cap, 1.0 + alpha * np.sin(2.0 * np.pi * f * x), n)

    # ramps, piecewise trends
    append_init(inits, cap, x, n)
    append_init(inits, cap, 1.0 - x, n)
    append_init(inits, cap, x * x, n)
    append_init(inits, cap, (1.0 - x) ** 2, n)
    append_init(inits, cap, np.minimum(x, 1.0 - x) + 0.05, n)
    append_init(inits, cap, np.maximum(x, 1.0 - x), n)

    # exponential tapers
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0):
        append_init(inits, cap, np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01, n)
        append_init(inits, cap, np.exp(-rate * np.abs(x - 0.5)) + 0.01, n)

    # bimodal
    for width in (0.05, 0.08, 0.12, 0.18):
        for c in (0.2, 0.3, 0.4):
            append_init(
                inits, cap,
                np.exp(-(((x - c) / width) ** 2)) +
                np.exp(-(((x - (1.0 - c)) / width) ** 2)) + 0.01,
                n,
            )

    # smooth random spectra
    N = next_power_of_two(max(1, n))
    freqs = np.fft.rfftfreq(N) * float(N)
    for _ in range(5):
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, N)[:n]
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.25, 1.2))
        append_init(inits, cap, 1.0 + amp * noise, n)

    # plain random
    for _ in range(4):
        append_init(inits, cap, 0.1 + 0.9 * rng.random(n), n)

    # anti-palindromic sine-inspired shapes
    for f in (1, 2, 3, 5, 7, 11):
        for alpha in (0.5, 0.8, 1.0):
            append_init(inits, cap, 1.0 + alpha * np.sin(2.0 * np.pi * f * x / n * n), n)
            append_init(inits, cap, 1.0 + alpha * np.sin(2.0 * np.pi * f * x + 0.7), n)

    return inits


def perturb(a, rng, strength=0.2):
    a = normalize_a(a)
    n = int(a.size)
    if n == 0:
        return a
    strength = abs(float(strength))
    mode = int(rng.integers(0, 9))

    if mode == 0:
        th = log_centered(a)
        th += strength * rng.standard_normal(n)
        th -= float(np.mean(th))
        return softmax_from_theta(th)

    if mode == 1:
        t = float(rng.uniform(0.05, 0.35))
        u = np.ones(n) / float(n)
        return normalize_a((1.0 - t) * a + t * u)

    if mode == 2:
        return normalize_a(a[::-1])

    if mode == 3:
        th = log_centered(a)
        k = min(31, max(5, n // 1000))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kk = np.arange(k) - c
        kern = np.exp(-(kk * kk) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise = noise - float(noise.mean())
        th += 0.5 * strength * noise
        th -= float(np.mean(th))
        return softmax_from_theta(th)

    if mode == 4:
        x = np.linspace(0.0, 1.0, n)
        mult = 1.0 - 0.5 * strength * (2.0 * x - 1.0) ** 2
        return normalize_a(a * mult)

    if mode == 5:
        x = np.linspace(0.0, 1.0, n)
        mult = 1.0 + 0.5 * strength * (2.0 * x - 1.0) ** 2
        return normalize_a(a * mult)

    if mode == 6:
        mult = np.exp(strength * rng.standard_normal(n))
        return normalize_a(a * mult)

    if mode == 7:
        x = np.linspace(0.0, 1.0, n)
        f = int(rng.integers(1, 8))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.2, 0.9))
        mult = 1.0 + amp * np.cos(2.0 * np.pi * f * x + ph)
        return normalize_a(a * np.clip(mult, 0.0, None))

    # blend with a random smooth shape
    th = log_centered(a)
    N = next_power_of_two(max(1, n))
    freqs = np.fft.rfftfreq(N) * float(N)
    power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(2.0, 5.0)))
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    spec[0] = 0.0
    noise = np.fft.irfft(spec, N)[:n]
    noise = noise - float(noise.mean())
    th += 0.5 * strength * noise
    th -= float(np.mean(th))
    return softmax_from_theta(th)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point. Searches for a non-negative sequence minimizing the score.
    """
    t_start = time.monotonic()
    budget = float(budget_s)
    deadline = t_start + max(60.0, budget - 30.0)
    rng = np.random.default_rng(int(seed))

    # Candidate length: slightly above previous SOTA but still on the 32768
    # FFT grid for 3n-2, keeping many iterations cheap.
    n = 10922

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, None)
            if p.size > 0 and float(np.sum(p)) > 0.0:
                parent = resample_sequence(p, n)
    except Exception:
        parent = None

    inits = generate_inits(n, rng, parent=parent, cap=50)
    if not inits:
        inits.append(np.ones(n) / float(n))

    best_a = None
    best_sc = float("inf")
    for init in inits:
        try:
            sc = exact_score(init)
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = normalize_a(init)

    if best_a is None:
        best_a = np.ones(n) / float(n)
        best_sc = exact_score(best_a)

    print("Initial best: n={} score={:.9f}".format(len(best_a), best_sc))

    # Phase schedule.  The fractions leave time for final restarts and I/O.
    total = max(1.0, deadline - t_start)
    p1_dead = t_start + 0.26 * total
    p2_dead = t_start + 0.55 * total
    p3_dead = t_start + 0.67 * total
    p4_dead = t_start + 0.82 * total
    p5_dead = t_start + 0.91 * total
    final_dead = deadline - 10.0

    # 1) Mirror/minimax: finds a better basin than pure LSE smoothers.
    sc, a = mirror_minimax(best_a, p1_dead, tag="m1")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # 2) Main log-domain Adam on LSE surrogate.
    sc, a = log_adam_lse_optimize(a, p2_dead, tag="adam1", rng=rng)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # 3) Mirror again, now that local minima changed.
    sc, a = mirror_minimax(a, p3_dead, tag="m2")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # 4) High-beta L-BFGS polish in log-simplex coordinates.
    sc, a = log_lbfgs_polish(a, p4_dead, tag="lbfgs")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # 5) Final local Adam.
    sc, a = log_adam_lse_optimize(a, p5_dead, tag="adam2", rng=rng)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    # 6) Diversification restarts.
    tries = 0
    while time.monotonic() < final_dead - 60 and tries < 8:
        rem = max(0.0, final_dead - 60.0 - time.monotonic())
        if rem < 90.0:
            break
        tries += 1

        strength = float(rng.uniform(0.05, 0.30)) + 0.01 * tries
        q = perturb(best_a, rng, strength=strength)

        m_dead = time.monotonic() + min(0.42 * rem, final_dead - time.monotonic() - 30.0)
        sc, aa = mirror_minimax(q, m_dead, tag="r{}m".format(tries))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = aa.copy()

        a_dead = time.monotonic() + min(0.30 * rem, final_dead - time.monotonic() - 20.0)
        if time.monotonic() < a_dead:
            sc, aa = log_adam_lse_optimize(a if not (aa is None) else best_a, a_dead, tag="r{}a".format(tries), rng=rng)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = aa.copy()

    # Final short mirror pass.
    if time.monotonic() < deadline - 15.0:
        sc, a = mirror_minimax(best_a, deadline - 5.0, tag="final")
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Produce valid output.
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1e6, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(np.sum(out))
    if s < 0.01:
        out = np.ones(n, dtype=np.float64) / float(n)
    else:
        out = out / s
    out = np.clip(out, 0.0, 1000.0)

    out_list = [float(x) for x in out]

    # Optional verification; avoid it near the wall or for overly large lists.
    verified = best_sc
    try:
        if len(out_list) <= 11500 and time.monotonic() < deadline - 3.0:
            verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = best_sc

    print("DONE internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out_list), time.monotonic() - t_start))

    return out_list
