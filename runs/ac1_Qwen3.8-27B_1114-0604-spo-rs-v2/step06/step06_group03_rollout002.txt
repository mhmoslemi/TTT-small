"""
Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses active-set LP, high-beta LSE, coordinate descent, and multi-scale optimization.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def convolve_shifted(a, w):
    """Compute the 'cross-correlation' needed for gradient: (a * w)[n-1:2n-1]"""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n-1:2*n-1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    n = int(n)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def lse_objective_and_grad(a, beta):
    """LSE approximation of the max-convolution objective."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    d = fast_self_conv(a) * n / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -100.0, 100.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e < 1e-300:
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(w @ d)
    
    h = convolve_shifted(a, w)
    score = 2.0 * L
    grad = (4.0 * n * h) / (S * S) - (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0)
    return float(score), grad


def lse_fun(a, beta):
    return lse_objective_and_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun=500):
    a = normalize_a(a0)
    n = len(a)
    rem = deadline - time.monotonic()
    if rem < 5.0:
        return exact_score(a), a
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-11, "maxiter": max(maxfun//2, 50), "maxfun": maxfun, "maxls": 40}
    try:
        res = minimize(lse_fun, a, method="L-BFGS-B", jac=True, args=(float(beta),), bounds=bounds, options=opts)
        if res.x is not None:
            a = normalize_a(res.x)
    except Exception:
        pass
    sc = exact_score(a)
    if not np.isfinite(sc) or sc > 1e6:
        a = normalize_a(a0)
        sc = exact_score(a)
    return float(sc), a


def active_set_lp(a, k_active, max_iter=20):
    """Solve active-set LP: max sum(b) s.t. b>=0, (a*b)_k <= M for top-k lags."""
    a = normalize_a(a, target=1.0)
    n = len(a)
    c = fast_self_conv(a)
    # Find top-k lags
    idx = np.argpartition(c, -k_active)[-k_active:] if k_active < len(c) else np.arange(len(c))
    M = float(c[idx].max())
    
    best_b = a.copy()
    best_score = exact_score(a)
    
    for it in range(max_iter):
        # Build constraint matrix for top-k lags
        # (a * b)_k = sum_i a_i * b_{k-i} for valid i, k-i
        # This is a linear constraint: A_k @ b <= M for k in active set
        A_list = []
        b_list = []
        for k in idx:
            row = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    row[i] = a[j]
            A_list.append(row)
            b_list.append(M)
        A_mat = np.array(A_list)
        b_vec = np.array(b_list)
        
        # Solve: max sum(b) s.t. A_mat @ b <= b_vec, b >= 0
        c_obj = -np.ones(n)  # minimize -sum(b)
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
        
        if res.success and res.x is not None:
            b_new = res.x
            S = float(b_new.sum())
            if S > 1e-6:
                b_new = b_new / S  # normalize
                sc = exact_score(b_new)
                if sc < best_score:
                    best_score = sc
                    best_b = b_new.copy()
                a = b_new
                # Update active set
                c = fast_self_conv(a)
                idx = np.argpartition(c, -k_active)[-k_active:] if k_active < len(c) else np.arange(len(c))
                M = float(c[idx].max())
    
    return float(best_score), best_b


def coordinate_descent(a, max_passes=10, deadline=None):
    """Optimize one element at a time to minimize the score."""
    a = normalize_a(a)
    n = len(a)
    best_sc = exact_score(a)
    best_a = a.copy()
    
    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Random order
        order = np.random.permutation(n)
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try different values for a[j]
            # The score as function of a[j] (with others fixed):
            # sum(a) = S_old - a[j] + a[j]_new
            # The max of conv changes at most linearly
            # Simple: try a few multiplicative factors
            for factor in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0):
                a[j] = old_val * factor
                sc = exact_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
            # Bisection for optimal a[j]
            lo, hi = 0.0, old_val * 3.0
            for _ in range(8):
                mid = (lo + hi) / 2
                a[j] = mid
                sc = exact_score(a)
                # Compare with best using lo and hi
                a[j] = lo
                sc_lo = exact_score(a)
                a[j] = hi
                sc_hi = exact_score(a)
                a[j] = mid
                if sc_lo <= sc and sc_lo <= sc_hi:
                    hi = mid
                else:
                    lo = mid
            a[j] = (lo + hi) / 2
            sc = exact_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True
        if not improved and p >= 2:
            break
    return float(best_sc), best_a


def mirror_descent(a, k_list, t_list, max_iters=100, deadline=None):
    """Mirror descent with power-iteration-inspired update."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        c = fast_self_conv(a)
        for k in k_list:
            if deadline is not None and time.monotonic() > deadline:
                break
            # top-k weights
            if k >= len(c):
                w = np.ones(len(c)) / len(c)
            else:
                idx = np.argpartition(c, -k)[-k:]
                w = np.zeros(len(c))
                w[idx] = 1.0
                w /= k
            h = convolve_shifted(a, w)
            h = np.maximum(h, 1e-15)
            for t_p in t_list:
                if deadline is not None and time.monotonic() > deadline:
                    break
                log_a = np.log(np.maximum(a, 1e-15))
                log_h = np.log(h)
                log_new = (1 - t_p) * log_a + t_p * log_h
                q = np.exp(log_new - log_new.max())
                q = normalize_a(q)
                sc = exact_score(q)
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
        if not improved and it > 10:
            break
    return float(best_sc), best_a


def subgradient_polish(a, deadline, k_list=(1, 2, 3, 5, 8), eta_list=(0.001, 0.005, 0.01, 0.02, 0.05, 0.1)):
    """Exact subgradient method."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    while time.monotonic() < deadline:
        improved = False
        c = fast_self_conv(a)
        for k in k_list:
            if time.monotonic() > deadline:
                break
            if k >= len(c):
                w = np.ones(len(c)) / len(c)
            else:
                idx = np.argpartition(c, -k)[-k:]
                w = np.zeros(len(c))
                w[idx] = 1.0
                w /= k
            h = convolve_shifted(a, w)
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-15:
                continue
            S = float(a.sum())
            LDS = float(w @ c)
            grad = (4.0 * len(a) * h) / (S * S) - (4.0 * LDS) / S
            for eta in eta_list:
                if time.monotonic() > deadline:
                    break
                q = np.maximum(a - eta * grad / hmax, 0.0)
                q = normalize_a(q)
                sc = exact_score(q)
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
        if not improved:
            break
    return float(best_sc), best_a


def generate_diverse_inits(n, parent, rng, n_inits=20):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    
    def add(arr):
        arr = np.nan_to_num(np.asarray(arr, dtype=np.float64).ravel(), nan=0.0, posinf=1e6, neginf=0.0)
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-12:
            arr = np.ones(n)
        inits.append(normalize_a(arr))
    
    if parent is not None:
        p = resample_sequence(parent, n)
        add(p)
        add(p[::-1])
        # Smoothed parent
        for sigma in (2, 5, 10, 20):
            k = 2*sigma + 1
            kern = np.exp(-0.5 * (np.arange(k) - sigma)**2 / sigma**2)
            kern /= kern.sum()
            add(np.convolve(p, kern, mode='same'))
    
    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2*x - 1))
    
    # U-shapes (inverted V)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 + alpha * (1 - np.abs(2*x - 1)))
        add(1.0 + alpha * np.sin(np.pi * x)**2)
    
    # Cosine
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.7, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2*np.pi*f*x))
            add(1.0 + alpha * np.cos(np.pi*x))
    
    # Bimodal
    for w in (0.08, 0.12, 0.15, 0.2, 0.25, 0.35):
        add(np.exp(-((x-0.1)/w)**2) + np.exp(-((x-0.9)/w)**2) + 0.1)
        add(np.exp(-((x-0.3)/w)**2) + np.exp(-((x-0.7)/w)**2) + 0.1)
        add(np.exp(-((x-0.5)/w)**2) + 0.05)
    
    # Power shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        add(np.abs(2*x - 1)**p + 0.1)
    
    # Random Fourier
    Nf = next_power_of_two(n)
    for _ in range(8):
        freqs = np.arange(Nf//2 + 1)
        decay = rng.uniform(1.0, 4.0)
        power = np.exp(-freqs / (Nf / (2*decay)))
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = np.zeros(Nf//2 + 1, dtype=complex)
        spec = power * np.exp(1j * phase)
        sig = np.fft.irfft(spec, Nf)[:n]
        add(1.0 + rng.uniform(0.3, 1.5) * sig - 0.5 * sig.min())
    
    # Random smooth
    for _ in range(5):
        sig = rng.standard_normal(n)
        k = max(5, n // 500)
        if k % 2 == 0: k += 1
        kern = np.exp(-0.5 * (np.arange(k) - k//2)**2 / max(1.0, (k//4)**2))
        kern /= kern.sum()
        sig = np.convolve(sig, kern, mode='same')
        add(1.0 + 0.5 * sig - 0.5*sig.min())
    
    return inits[:n_inits]


def perturb(a, rng, strength=0.15):
    """Perturb a sequence."""
    a = normalize_a(a)
    n = len(a)
    mode = int(rng.integers(0, 10))
    
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = rng.uniform(0.05, 0.3)
        q = (1-t)*a + t*(np.ones(n)/n)
    elif mode == 2:
        q = a * np.exp(-strength * fast_self_conv(a)[:n] / max(1e-15, fast_self_conv(a).max()))
    elif mode == 3:
        # Smoothing
        k = max(5, int(n * rng.uniform(0.001, 0.01)))
        if k % 2 == 0: k += 1
        sig = max(1.0, k//4)
        kern = np.exp(-0.5*(np.arange(k)-k//2)**2/sig**2)
        kern /= kern.sum()
        q = np.convolve(a, kern, mode='same')
    elif mode == 4:
        # Anti-smoothing (sharpen)
        q = a + strength * (a - np.roll(a, 1)) * (1 - np.roll(a, -1)) / max(1e-15, a.max())
    elif mode == 5:
        # V-shape modulation
        x = np.linspace(0, 1, n)
        q = a * (1 + strength * 0.5 * np.abs(2*x - 1))
    elif mode == 6:
        # Gaussian modulation
        x = np.linspace(0, 1, n)
        mu = rng.uniform(0.2, 0.8)
        sig = rng.uniform(0.05, 0.3)
        q = a * (1 + strength * np.exp(-((x-mu)/sig)**2))
    elif mode == 7:
        # Random blocks
        nb = int(rng.integers(2, 8))
        q = a.copy()
        for _ in range(nb):
            i = int(rng.integers(0, n-10))
            w = int(rng.integers(10, max(11, n//20)))
            q[i:i+w] *= rng.uniform(0.5, 1.5)
    elif mode == 8:
        # Log-normal perturbation
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
        # Smooth it
        k = max(5, n//2000)
        if k % 2 == 0: k += 1
        kern = np.ones(k)/k
        q = np.convolve(q, kern, mode='same')
    else:
        # Targeted: reduce at peak-contributing positions
        c = fast_self_conv(a)
        idx = np.argpartition(c, -3)[-3:]
        w = np.zeros(len(c))
        w[idx] = 1.0
        w /= 3
        h = convolve_shifted(a, w)
        q = a * np.exp(-strength * 0.5 * h / max(1e-15, h.max()))
    
    return normalize_a(np.clip(q, 0.0, None))


def optimize_trajectory(a_start, deadline):
    """Multi-stage LSE optimization with increasing beta."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    stages = [(100, 0.10), (300, 0.10), (1000, 0.10), (3000, 0.10),
              (10000, 0.12), (30000, 0.12), (100000, 0.13), (300000, 0.13),
              (1000000, 0.15), (3000000, 0.15)]
    
    t0 = time.monotonic()
    total = max(5.0, deadline - t0 - 5.0)
    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0: cum = 1.0
        sub_dead = min(t0 + cum * total, deadline - 3.0)
        
        while time.monotonic() < sub_dead - 2.0:
            rem = sub_dead - time.monotonic()
            if rem < 5.0: break
            n = len(a)
            est = max(0.02, 0.02 * max(0.5, n/8192) * max(1.0, np.log2(max(16,n))/np.log2(8192)))
            mf = max(64, min(2000, int(0.7*rem / est)))
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print(f"  LSE beta={beta:.0f}: {sc:.8f}")
            a = normalize_a(a2)
    
    return float(best_sc), best_a
