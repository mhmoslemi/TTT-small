Optimized step function construction using Nesterov-accelerated LSE optimization
with adaptive beta, spectral shaping initialization, and L-BFGS polish.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2n*max(conv)/sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_grad(a, beta):
    """
    Compute LSE-smoothed objective and its gradient.
    L(a) = c_max + log(sum exp(beta*(c_k - c_max)))/beta
    F(a) = 2n * L(a) / S(a)^2
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Probability weights p_k
    p = exp_scaled / exp_sum

    # Gradient of L w.r.t. a_i:
    # dL/da_i = 2 * sum_k p_k * a[k-i] - p[2i]*a[i]  (diagonal correction)
    # The first term is a convolution: conv(p, rev_a)[n-1+i]
    rev_a = a[::-1]
    M2 = M + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PP = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(RA * PP, M2_fft)
    g_L = 2.0 * conv_res[n - 1:n + n - 1]

    # Diagonal correction: subtract p[2i]*a[i] for i=0..n-1
    diag_idx = 2 * np.arange(n)
    diag_idx = diag_idx[diag_idx < M]
    g_L[diag_idx // 2] -= p[diag_idx] * a[diag_idx // 2]

    S2 = S * S
    S3 = S2 * S
    # dF/da_i = (2n/S^3) * (S * dL/da_i - 2L)
    grad = (2.0 * n / S3) * (S * g_L - 2.0 * L)
    return score, grad


def project_simplex_nonneg(a):
    """Project onto a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        n = len(a)
        a = np.ones(n) / n
        return a
    return a / S


def nesterov_lse(a0, beta_start, beta_end, n_iters, t_start, t_deadline,
                  max_step=0.05, momentum=True):
    """
    Nesterov-accelerated optimization of the LSE-smoothed objective.
    Beta increases adaptively from beta_start to beta_end.
    """
    n = len(a0)
    a = project_simplex_nonneg(a0.copy())
    y = a.copy()  # lookahead point
    best_a = a.copy()
    best_f = compute_score(a)
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for t in range(1, n_iters + 1):
        if time.monotonic() > t_deadline:
            break
        
        # Adaptive beta: exponential ramp
        progress = t / n_iters
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient at lookahead point y
        f_y, g = lse_value_grad(y, beta)

        if f_y < best_f:
            # Also check exact score periodically
            exact_f = compute_score(y)
            if exact_f < best_f:
                best_f = exact_f
                best_a = y.copy()

        # Compute step size
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        gap = max(f_y - f_lb, 1e-10)
        step = min(max_step, 2.0 * gap / (gn * gn + 1e-30))
        step = max(step, 1e-8)

        # Nesterov update
        x_new = y - step * g
        x_new = project_simplex_nonneg(x_new)

        if momentum:
            m = 2.0 / (t + 1.0)
            y = x_new + m * (x_new - y)
            y = project_simplex_nonneg(y)
        else:
            y = x_new

        # Periodic exact evaluation
        if t % 50 == 0:
            sc = compute_score(y)
            if sc < best_f:
                best_f = sc
                best_a = y.copy()

    # Final check
    sc = compute_score(y)
    if sc < best_f:
        best_f = sc
        best_a = y.copy()

    return best_a, best_f


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.85):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        return a

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        # Inverse shaping: reduce peaks more aggressively
        amp_safe = np.maximum(amp, 1e-20)

        # Where amp > C, ratio < 1, so ratio^gamma < 1 (reduction)

def peak_directed_descent(a, n_iters=2000, n_peaks=5, step_init=0.02, decay=0.99995):
    """Peak-directed coordinate descent with subgradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.03)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a

    diag_idx = 2 * np.arange(n)
    diag_idx = diag_idx[diag_idx < M]
    g_L[diag_idx // 2] -= p[diag_idx] * x[diag_idx // 2]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S3) * (S * g_L - 2.0 * L)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Edge-enhanced
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))

    # Inverse sinc-like
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))

    # Gaussian at center
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)

    # Random smooth
    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    # Hann window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))

    # Flat top
    inits.append(np.maximum(0.6 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: multi-phase optimizer combining spectral shaping,
    Nesterov LSE, peak descent, and L-BFGS polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}")
    print(f"Parent available: {parent is not None}")

    if parent is not None:
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")

    # ============================================================
    # Phase 1: Spectral shaping (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Spectral Shaping (until t={phase1_end - t_start:.0f}s) ===")

    start_a = parent if parent is not None else np.ones(n_work) * 0.5
    start_a = project_simplex_nonneg(start_a)

    # Geometric Fienup from parent
    a_g = fienup_geometric(start_a, n_iters=200, alpha_start=0.03, alpha_end=0.7)
    sc = compute_score(a_g)
    update_best(sc, a_g, "GEO")

    # Inverse Fienup
    if time.monotonic() < phase1_end:
        a_inv = fienup_inverse(start_a, n_iters=150, gamma=0.6)
        sc = compute_score(a_inv)
        update_best(sc, a_inv, "INV")

    # Try a few other initializations
    inits = build_initializations(n_work, rng, None)
    for i, a0 in enumerate(inits[:3]):
        if time.monotonic() > phase1_end:
            break
        a_t = fienup_geometric(a0, n_iters=150, alpha_start=0.05, alpha_end=0.8)
        sc = compute_score(a_t)
        update_best(sc, a_t, f"GEO-i{i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Nesterov LSE optimization (40% time)
    # ============================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Nesterov LSE (until t={phase2_end - t_start:.0f}s) ===")

    # Configuration: adaptive beta from smooth to sharp
    beta_configs = [
        (20, 500, 3000),      # Start very smooth, end moderately sharp
        (50, 2000, 5000),     # Medium range
        (200, 10000, 30000),  # Start sharper, end very sharp
    ]

    for bi, (bs, be, niters) in enumerate(beta_configs):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        # Scale iterations to available time (est ~2ms per iter)
        est_iters = int(remaining * 300)
        niters = min(niters, est_iters)
        if niters < 100:
            continue
        print(f"  Config {bi}: beta [{bs}->{be}], iters={niters}")

        a_n, sc_n = nesterov_lse(best_a, bs, be, niters,
                                  t_start, time.monotonic() + remaining)
        update_best(sc_n, a_n, f"NEST b={bi}")

    # Also try from the spectral shaping result
    if time.monotonic() < phase2_end - 60:
        a_n2, sc_n2 = nesterov_lse(a_g, 50, 5000, 2000,
                                    t_start, phase2_end - 60)
        update_best(sc_n2, a_n2, "NEST-GEO")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Peak-directed descent (15% time)
    # ============================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak Descent (until t={phase3_end - t_start:.0f}s) ===")

    for n_peaks in [3, 5, 10, 20]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        niters = min(2000, int(remaining * 200))
        if niters < 100:
            continue
        print(f"  n_peaks={n_peaks}, iters={niters}")
        a_pd = peak_directed_descent(best_a, n_iters=niters, n_peaks=n_peaks,
                                     step_init=0.01, decay=0.99998)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS polish (15% time)
    # ============================================================
    phase4_end = deadline - 30.0
    print(f"\n=== Phase 4: L-BFGS Polish (until t={phase4_end - t_start:.0f}s) ===")

    betas = [500, 2000, 8000, 20000]
    for beta in betas:
        if time.monotonic() > phase4_end:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(5000, int(remaining * 20))
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS b={beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final refinement (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Refinement ===")

    # One more Nesterov pass with high beta
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        niters = int(remaining * 300)
        if niters > 100:
            a_f, sc_f = nesterov_lse(best_a, 2000, 50000, niters,
                                      t_start, deadline - 60)
            update_best(sc_f, a_f, "FINAL-NEST")

    # Final peak descent
    if time.monotonic() < deadline - 30:
        a_fp = peak_directed_descent(best_a, n_iters=1000, n_peaks=8,
                                     step_init=0.005, decay=0.99999)
        sc = compute_score(a_fp)
        update_best(sc, a_fp, "FINAL-PEAK")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(2):
            if time.monotonic() > deadline - 15:
                break
            noise = 0.01 * (1 + trial * 0.5)
            a_pert = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            a_pert = project_simplex_nonneg(a_pert)
            a_r, sc_r = nesterov_lse(a_pert, 500, 10000, 500,
                                      t_start, deadline - 15)
            update_best(sc_r, a_r, f"RESTART {trial}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
