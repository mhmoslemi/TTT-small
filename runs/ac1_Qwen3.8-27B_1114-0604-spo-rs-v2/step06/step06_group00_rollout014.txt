```
def lse_score_grad_y(y, beta):
    a=softmax_positive(y)
    n=len(a)
    c=fast_conv(a) # uses A? 
    cmax=np.max(c)
    z=beta*(c-cmax); z=np.clip(z,-700,700)
    ez=np.exp(z); p=ez/ez.sum(); L=cmax + np.log(ez.sum())/beta
    # gradient L w.r.t a:
    M=next_pow2(3*n-2)
    A=np.fft.rfft(a,M); P=np.fft.rfft(p,M)
    corr=np.fft.irfft(P * np.conj(A), M)[:n]
    dL=2*corr
    # score = 2n L (S=1)
    score=2*n*L
    grad_a=2*n*(dL - 2*L)
    grad_y=a*grad_a
    return score, grad_y
```

```python
"""
Search for a non-negative sequence minimizing
  2*n*max(conv(a,a)) / (sum(a)^2)
using LSE-smoothed Adam in log-softmax coordinates, a fixed-convolution
deconvolution/blending stage, short L-BFGS polishing, coordinate shaving,
and modest length growth.
"""

import numpy as np
import time
from scipy.optimize import minimize

_FFT_CACHE = {}


def fft_len(n):
    """Next power of two length sufficient for self/cross convolutions."""
    n = int(n)
    if n <= 0:
        return 1
    if n in _FFT_CACHE:
        return _FFT_CACHE[n]
    m = 3 * n - 2
    size = 1 << (m - 1).bit_length()
    _FFT_CACHE[n] = size
    return size


def normalize_nonneg(a):
    """Clip to [0, inf], discard NaN/inf, normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1e100, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-300:
        return a
    return a / s


def fast_self_conv(a, M=None):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if M is None:
        M = fft_len(n)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Internal exact score matching evaluate_sequence for finite valid data."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    c = fast_self_conv(a)
    mc = float(np.max(c))
    return float(2.0 * len(a) * mc / (s * s))


def softmax_y(y):
    """Positive simplex vector from log-coordinates."""
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return y
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.0, posinf=50.0, neginf=-50.0)
    y = y - y.mean()
    m = float(y.max())
    e = np.exp(y - m)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(y.shape, 1.0 / y.size, dtype=np.float64)
    return e / s


def log_sequence(a):
    """Log-coordinates for a positive simplex vector."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-300)
    y = np.log(a)
    return y - y.mean()


def p_sequence(n):
    """
    Good analytic seed: first n coefficients of (1-z)^(-1/2), normalized.
    Its infinite square is flat; finite truncation gives score about pi/2.
    """
    n = int(max(1, n))
    p = np.empty(n, dtype=np.float64)
    p[0] = 1.0
    for i in range(1, n):
        p[i] = p[i - 1] * (2 * i - 1) / (2 * i)
    return normalize_nonneg(p)


def lse_self_info_y(y, beta):
    """
    LSE-smoothed self-convolution score in y-coordinates.
    Returns (lse_score, grad_y, exact_score).
    """
    a = softmax_y(y)
    n = len(a)
    M = fft_len(n)

    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:2 * n - 1]
    cmax = float(c.max())
    exact_score = 2.0 * n * cmax

    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 700.0)
    ez = np.exp(z)
    se = float(ez.sum())
    if se < 1e-300 or not np.isfinite(se):
        L = cmax
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        L = cmax + float(np.log(se)) / beta
        p = ez / se

    P = np.fft.rfft(p, M)
    corr = np.fft.irfft(P * np.conjugate(A), M)[:n]
    dL = 2.0 * corr

    # Objective is scale-invariant and we work on the simplex:
    # gradient projected to sum( da ) = 0.
    grad_a = 2.0 * n * (dL - 2.0 * L)
    grad_y = a * grad_a
    grad_y -= float(grad_y.mean())

    if not np.all(np.isfinite(grad_y)):
        grad_y = np.zeros_like(y)

    return float(2.0 * n * L), grad_y, float(exact_score)


def lse_self_y_objective(y, beta):
    """Objective + gradient wrapper for scipy L-BFGS-B."""
    val, grad, _ = lse_self_info_y(y, beta)
    if not np.isfinite(val):
        return 1e100, np.zeros_like(y)
    return val, grad

    Adam on LSE-smoothed self-convolution score.
    Each entry in betas gets an equal share of the remaining deadline.
    """
    a0 = normalize_nonneg(a0)
    n = len(a0)
    if n == 0:
        a0 = p_sequence(10000)
        n = len(a0)

    best_sc = compute_score(a0)
    if not np.isfinite(best_sc):
        a0 = p_sequence(n)
        best_sc = compute_score(a0)
    best_a = a0.copy()

    if deadline is None:
        deadline = time.monotonic() + 60.0
    if time.monotonic() >= deadline:
        return best_sc, best_a
    if len(betas) == 0:
        return best_sc, best_a

    if rng is None:
        rng = np.random.default_rng(12345)

    b1 = 0.90
    b2 = 0.99
    eps = 1e-8

    for idx, beta in enumerate(betas):
        now = time.monotonic()
        if now >= deadline:
            break
        remaining = deadline - now
        beta_end = min(deadline, now + max(1.0, remaining / float(len(betas) - idx)))

        y = log_sequence(best_a)
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        t = 0
        stag = 0
        prev_lse = float("inf")
        resets = 0

        while time.monotonic() < beta_end:
            if beta <= 0:
                break

            lse_sc, grad_y, ex_sc = lse_self_info_y(y, beta)
            if not np.isfinite(lse_sc) or not np.isfinite(ex_sc):
                break

            t += 1
            m = (1.0 - b1) * m + b1 * grad_y
            v = (1.0 - b2) * v + b2 * (grad_y * grad_y)

            corr1 = 1.0 - b1 ** t
            corr2 = 1.0 - b2 ** t
            mhat = m / corr1
            vhat = v / corr2

            lr = lr_base
            if beta >= 50000:
                lr *= 0.65
            if beta >= 200000:
                lr *= 0.55

            step = lr * mhat / (np.sqrt(vhat) + eps)
            y = y - step
            y -= float(y.mean())

            if not np.all(np.isfinite(y)):
                break

            if ex_sc < best_sc:
                best_sc = ex_sc
                best_a = softmax_y(y).copy()
                stag = 0
            else:
                if lse_sc < prev_lse - 1e-12:
                    prev_lse = lse_sc
                    if stag > 0:
                        stag = max(0, stag - 1)
                else:
                    prev_lse = lse_sc
                    stag += 1

            if stag > 3000:
                if resets < 5:
                    y = log_sequence(best_a) + 0.03 * rng.standard_normal(n)
                    y -= float(y.mean())
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 0
                    stag = 0
                    resets += 1
                else:
                    break

    return best_sc, best_a


def lbfgs_direct_y(a0, beta, maxfun=700, maxiter=300):
    """Short L-BFGS-B polish in y-coordinates on the LSE objective."""
    a0 = normalize_nonneg(a0)
    if len(a0) == 0:
        a0 = p_sequence(10000)
    sc0 = compute_score(a0)

    y0 = log_sequence(a0)
    try:
        res = minimize(
            lse_self_y_objective,
            y0,
            args=(beta,),
            method="L-BFGS-B",
            options={
                "maxiter": int(max(maxiter, 1)),
                "maxfun": int(max(maxfun, 1)),
                "ftol": 1e-18,
                "gtol": 1e-14,
                "disp": 0,
            },
        )
        a = softmax_y(res.x)
        sc = compute_score(a)
        if not np.isfinite(sc):
            return sc0, a0
        return sc, a
    except Exception:
        return sc0, a0


def deconv_info_pre(y, Ffft, M, beta):
    """
    LSE of f*g where f is fixed (Ffft supplied) and g=softmax(y).
    Returns (LSE_value, grad_y, hmax, g).
    """
    g = softmax_y(y)
    n = len(g)
    G = np.fft.rfft(g, M)
    h = np.fft.irfft(Ffft * G, M)[:2 * n - 1]
    hmax = float(h.max())

    z = beta * (h - hmax)
    z = np.clip(z, -700.0, 700.0)
    ez = np.exp(z)
    se = float(ez.sum())
    if se < 1e-300 or not np.isfinite(se):
        L = hmax
        p = np.zeros_like(h)
        p[int(np.argmax(h))] = 1.0
    else:
        L = hmax + float(np.log(se)) / beta
        p = ez / se

    P = np.fft.rfft(p, M)
    grad_g = np.fft.irfft(P * np.conjugate(Ffft), M)[:n]

    # Softmax gradient for a general objective on the simplex.
    scalar = float(np.dot(g, grad_g))
    grad_y = g * (grad_g - scalar)
    grad_y -= float(grad_y.mean())
    if not np.all(np.isfinite(grad_y)):
        grad_y = np.zeros_like(y)

    return float(L), grad_y, float(hmax), g


def run_adam_deconv(f, beta, deadline, lr_base=0.05, rng=None):
    """
    Optimize g >= 0, sum g = 1 to make f*g flat (LSE objective).
    Returns (best_g_by_cross, best_g_by_self_score).
    """
    f = normalize_nonneg(f)
    n = len(f)
    if n == 0:
        f = p_sequence(10000)
        n = len(f)

    M = fft_len(n)
    F = np.fft.rfft(f, M)

    if deadline is None:
        deadline = time.monotonic() + 60.0
    if time.monotonic() >= deadline:
        return f.copy(), f.copy()

    if rng is None:
        rng = np.random.default_rng(54321)

    y = log_sequence(f)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    stag = 0
    prev_L = float("inf")

    best_h = float("inf")
    best_g = f.copy()

    best_self = compute_score(f)
    best_self_g = f.copy()

    b1 = 0.90
    b2 = 0.99
    eps = 1e-8
    resets = 0
    it = 0

    while time.monotonic() < deadline:
        if beta <= 0:
            break
        L, grad_y, hmax, g = deconv_info_pre(y, F, M, beta)
        if not np.isfinite(L) or not np.isfinite(hmax):
            break

        it += 1
        t += 1
        m = (1.0 - b1) * m + b1 * grad_y
        v = (1.0 - b2) * v + b2 * (grad_y * grad_y)

        corr1 = 1.0 - b1 ** t
        corr2 = 1.0 - b2 ** t
        mhat = m / corr1
        vhat = v / corr2

        lr = lr_base
        if beta >= 20000:
            lr *= 0.6

        step = lr * mhat / (np.sqrt(vhat) + eps)
        y = y - step
        y -= float(y.mean())
        if not np.all(np.isfinite(y)):
            break

        if hmax < best_h:
            best_h = hmax
            best_g = g.copy()
            stag = 0
        else:
            if L < prev_L - 1e-12:
                prev_L = L
                if stag > 0:
                    stag = max(0, stag - 1)
            else:
                prev_L = L
                stag += 1

        if it % 200 == 0:
            sc = compute_score(g)
            if np.isfinite(sc) and sc < best_self:
                best_self = sc
                best_self_g = g.copy()

        if stag > 2000:
            if resets < 4:
                y = log_sequence(best_g) + 0.02 * rng.standard_normal(n)
                y -= float(y.mean())
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                stag = 0
                prev_L = float("inf")
                resets += 1
            else:
                break

    # In case we never evaluated self for the best cross-g, check it now.
    sc = compute_score(best_g)
    if np.isfinite(sc) and sc < best_self:
        best_self = sc
        best_self_g = best_g.copy()

    return best_g, best_self_g


def run_deconv_blend(a0, deadline, rng=None):
    """
    Alternate:
      1) find g that makes f*a nearly flat using deconvolution,
      2) blend f-(new) and keep the best self-convolution result.
    """
    f = normalize_nonneg(a0)
    if len(f) == 0:
        f = p_sequence(10000)

    best_sc = compute_score(f)
    best_a = f.copy()

    if deadline is None:
        deadline = time.monotonic() + 60.0
    if time.monotonic() >= deadline:
        return best_sc, best_a

    if rng is None:
        rng = np.random.default_rng(24680)

    cycles = 3
    betas = [3000, 10000, 40000]

    for cycle in range(cycles):
        now = time.monotonic()
        if now >= deadline:
            break
        remaining = deadline - now
        if remaining < 30:
            break
        period = remaining / float(cycles - cycle)
        c_end = min(deadline, now + period)

        beta = betas[cycle % len(betas)]
        mid_end = min(c_end, now + 0.45 * period)

        g1, g2 = run_adam_deconv(f, beta, mid_end, lr_base=0.05, rng=rng)

        candidates = [g1, g2]
        for t in (0.1, 0.3, 0.5, 0.8, 1.0):
            candidates.append(normalize_nonneg((1.0 - t) * f + t * g1))
            candidates.append(normalize_nonneg((1.0 - t) * f + t * g2))

        cycle_best_sc = compute_score(f)
        cycle_best_a = f.copy()

        for cand in candidates:
            cand = normalize_nonneg(cand)
            sc = compute_score(cand)
            if not np.isfinite(sc):
                continue
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                print(f"  deconv-blend best: score={sc:.8f}, n={len(cand)}")
            if sc < cycle_best_sc:
                cycle_best_sc = sc
                cycle_best_a = cand.copy()

        f = cycle_best_a

        if cycle_best_sc >= best_sc + 1e-12:
            # No progress in the last cycle; avoid wasting time.
            break

    return best_sc, best_a


def random_perturb_a(a, rng, scale=0.03):
    """Multiplicative perturbation in log-space."""
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0:
        return a
    y = log_sequence(a) + float(scale) * rng.standard_normal(n)
    y -= float(y.mean())
    return softmax_y(y)


def interpolate_sequence(a, new_n):
    """Linear interpolate a normalized sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    if a.size == int(new_n):
        return a.copy()
    old_n = a.size
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    return normalize_nonneg(np.interp(x_new, x_old, a))


def coordinate_shave(a, max_time=20.0):
    """
    Very targeted coordinate descent: reduce the largest grouped terms in the
    current peak coefficient. Each single-coordinate move is accepted only if
    the exact score improves.
    """
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0:
        return float("inf"), a

    best_sc = compute_score(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for _ in range(60):
        if time.monotonic() - t0 > max_time:
            break

        c = fast_self_conv(a)
        k = int(np.argmax(c))
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            break

        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        pairs = a[js] * a[k - js]
        top = np.argsort(pairs)[::-1][:12]

        improved = False
        base = a.copy()

        for idx in top:
            j = int(js[idx])
            old = float(a[j])
            if old < 1e-14:
                continue

            for mult in (0.95, 0.85, 0.70, 0.50):
                cand = base.copy()
                cand[j] = old * mult
                cand = normalize_nonneg(cand)
                sc = compute_score(cand)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = cand.copy()
                    base = best_a.copy()
                    j = int(np.argmin(np.abs(np.arange(n) - j)))  # keep same j
                    old = float(base[j])
                    improved = True
                    break

        if not improved:
            break

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage search:
      - multiple seeds (previous best, analytic p-seed, perturbations)
      - LSE Adam annealing in log-softmax coordinates
      - deconvolution/blending
      - L-BFGS polishing
      - high-beta Adam polish
      - small random restarts
      - optional modest length growth
      - final coordinate shaving
    """
    t_start = time.monotonic()
    usable = max(60.0, float(budget_s) - 35.0)
    deadline_global = t_start + usable

    rng = np.random.default_rng(int(seed) if np.isfinite(seed) else 42)

    # ------------------------------------------------------------------
    # Start point and length
    # ------------------------------------------------------------------
    parent = None
    try:
        h = height_sequence_1
        if h is not None:
            parent = np.asarray(h, dtype=np.float64).ravel()
            if parent.size > 0 and not np.all(np.isfinite(parent)):
                parent = None
            if parent is not None and parent.size > 20000:
                parent = interpolate_sequence(parent, 10000)
            if parent is not None and parent.size < 128:
                parent = None
    except Exception:
        parent = None

    target_n = 10000
    if parent is not None:
        n = int(parent.size)
    else:
        n = target_n

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    candidates = []

    if parent is not None:
        candidates.append(("parent", normalize_nonneg(parent)))

    pseed = p_sequence(n)
    candidates.append(("pseed", pseed))
    candidates.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))

    p_rev = pseed[::-1].copy()
    for lam in (0.1, 0.3, 0.6):
        candidates.append((f"p-mirror-{lam}", normalize_nonneg(((1 - lam) * pseed) + (lam * p_rev))))

    for scale in (0.04, 0.08, 0.14):
        candidates.append((f"p-rand-{scale}", random_perturb_a(pseed, rng, scale)))

    best_sc = float("inf")
    best_a = None
    for label, cand in candidates:
        cand = normalize_nonneg(cand)
        sc = compute_score(cand)
        print(f"init {label}: score={sc:.8f}, n={len(cand)}")
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = cand.copy()

    if best_a is None or not np.isfinite(best_sc):
        best_a = p_sequence(n)
        best_sc = compute_score(best_a)

    print(f"initial best: {best_sc:.8f}, n={len(best_a)}")

    # ------------------------------------------------------------------
    # Phase 1: direct LSE Adam, low-to-mid beta
    # ------------------------------------------------------------------
    phase1_end = min(deadline_global, t_start + 0.30 * usable)
    if time.monotonic() < phase1_end:
        end1 = min(phase1_end, time.monotonic() + 60.0)
        sc, a = run_adam_lse_y(best_a, [500, 2000], end1, lr_base=0.035, rng=rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"phase1 mid low: {best_sc:.8f}")

    if time.monotonic() < phase1_end:
        sc, a = run_adam_lse_y(best_a, [5000, 12000], phase1_end, lr_base=0.03, rng=rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"phase1 mid high: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2: deconvolution / blending
    # ------------------------------------------------------------------
    phase2_end = min(deadline_global, phase1_end + 0.15 * usable)
    if time.monotonic() < phase2_end:
        sc, a = run_deconv_blend(best_a, phase2_end, rng=rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"phase2 deconv: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3: L-BFGS polish at several beta values
    # ------------------------------------------------------------------
    phase3_end = min(deadline_global, phase2_end + 0.15 * usable)
    if time.monotonic() < phase3_end:
        for beta in (20000, 80000, 200000):
            if time.monotonic() >= phase3_end - 25:
                break
            maxfun = 800 if len(best_a) <= 12000 else 500
            sc, a = lbfgs_direct_y(best_a, beta, maxfun=maxfun, maxiter=250)
            if np.isfinite(sc) and sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"lbfgs beta={beta}: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4: high-beta direct Adam
    # ------------------------------------------------------------------
    phase4_end = min(deadline_global, phase3_end + 0.12 * usable)
    if time.monotonic() < phase4_end:
        sc, a = run_adam_lse_y(best_a, [50000, 150000, 400000], phase4_end, lr_base=0.022, rng=rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"phase4 high beta: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5: small random restarts from current best
    # ------------------------------------------------------------------
    phase5_end = min(deadline_global, phase4_end + 0.12 * usable)
    if time.monotonic() < phase5_end:
        for r in range(6):
            now = time.monotonic()
            if now >= phase5_end - 5:
                break
            sub_end = min(phase5_end, now + max(8.0, (phase5_end - now) / float(7 - r)))
            scale = 0.015 + 0.006 * r
            start = random_perturb_a(best_a, rng, scale=scale)
            sc, a = run_adam_lse_y(start, [8000, 40000, 120000], sub_end, lr_base=0.032, rng=rng)
            if np.isfinite(sc) and sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"restart {r}: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6: modest length growth, if promising
    # ------------------------------------------------------------------
    phase6_end = min(deadline_global, phase5_end + 0.08 * usable)
    if time.monotonic() < phase6_end and len(best_a) < 12000:
        for new_n in (11000, 12500):
            now = time.monotonic()
            if now >= phase6_end - 20:
                break
            big = interpolate_sequence(best_a, new_n)
            sc_big = compute_score(big)
            print(f"interpolated n={new_n}: {sc_big:.8f}")
            if np.isfinite(sc_big) and sc_big < best_sc:
                best_sc, best_a = sc_big, big.copy()
                print(f"accepted larger n={new_n}: {best_sc:.8f}")
                # quick Adam on the larger length
                end_sub = min(phase6_end, now + 35.0)
                sc, a = run_adam_lse_y(best_a, [8000, 40000], end_sub, lr_base=0.028, rng=rng)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc, best_a = sc, a.copy()
                    print(f"larger polish n={len(a)}: {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 7: final short coordinate shave + final LSE Adam
    # ------------------------------------------------------------------
    final_end = deadline_global
    if time.monotonic() < final_end - 20:
        sc, a = coordinate_shave(best_a, max_time=18.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"coord final: {best_sc:.8f}")

    if time.monotonic() < final_end - 5:
        sc, a = run_adam_lse_y(best_a, [200000, 600000], final_end, lr_base=0.012, rng=rng)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"final ultra-high Adam: {best_sc:.8f}")

    # If the larger-length result is only marginally better and the 10000
    # version is still competitive, prefer length 10000 for consistency.
    if len(best_a) != 10000:
        best_10000 = interpolate_sequence(best_a, 10000)
        sc10000 = compute_score(best_10000)
        if np.isfinite(sc10000) and sc10000 <= best_sc + 0.003 and sc10000 < best_sc + 1e-4:
            # Switch only if the penalty is tiny; otherwise keep the true best.
            if sc10000 < best_sc + 1e-6:
                best_sc, best_a = sc10000, best_10000.copy()

    # ------------------------------------------------------------------
    # Validate output
    # ------------------------------------------------------------------
    out = [float(max(0.0, min(1000.0, float(x)))) for x in best_a]
    if len(out) == 0 or sum(out) < 0.01:
        out = [0.001] * 10000

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float("inf")

    elapsed = time.monotonic() - t_start
    print("=" * 60)
    print(f"FINAL internal={best_sc:.8f}, verified={verified:.8f}, n={len(out)}, t={elapsed:.1f}s")
    print("=" * 60)

    return out
```
