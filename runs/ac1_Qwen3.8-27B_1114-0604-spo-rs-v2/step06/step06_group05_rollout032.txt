"""
Improved optimization for minimum peak-to-mean autocorrelation of non-negative sequences.
Key improvements over previous: adaptive LSE annealing schedule, spectral peak-shaping with
overrelaxation, QP-inspired peak-pair cancellation, and better time management.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def weighted_lse_score_grad(a, w, beta):
    """Weighted LSE score: emphasizes certain lags more."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    wc = w * c
    wc_max = float(wc.max())
    scaled = beta * (wc - wc_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = wc_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def spectral_annealing(a, n_iters=500, target='flat', alpha_init=0.4, alpha_final=0.05,
                       overrelax=1.0, power_target=0.5):
    """
    Improved spectral annealing (Gerchberg-Saxton variant).
    target: 'flat', 'gaussian', 'exponential'
    """
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a_norm = a / S

    best_a = a_norm.copy()
    best_sc = compute_score(a_norm)

    if target == 'flat':
        b_target = np.zeros(M)
        b_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    elif target == 'gaussian':
        k = np.arange(2 * n - 1)
        sigma = n * 0.5
        b_target = np.zeros(M)
        b_target[:2 * n - 1] = np.exp(-0.5 * ((k - (n - 1)) / sigma) ** 2)
        b_target = np.maximum(b_target, 1e-10)
    elif target == 'exponential':
        k = np.arange(2 * n - 1)
        decay = 1.0 / (n * 0.8)
        b_target = np.zeros(M)
        b_target[:2 * n - 1] = np.exp(-abs(k - (n - 1)) * decay)
        b_target = np.maximum(b_target, 1e-10)
    else:
        b_target = np.zeros(M)
        b_target[:2 * n - 1] = 1.0 / (2 * n - 1)

    T = np.fft.fft(b_target)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        alpha = alpha * overrelax

        A = np.fft.fft(a_norm, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        if power_target == 0.5:
            new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        else:
            new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5) * (A_mag / np.maximum(T_mag, 1e-20)) ** (alpha * (power_target - 0.5) * 0.1)
        new_phase = A_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a_norm = a_new

        sc = compute_score(a_norm)
        if sc < best_sc:
            best_sc = sc
            best_a = a_norm.copy()

    return best_a


def spectral_overrelax(a, n_iters=300, target='flat', alpha=0.3):
    """Spectral method with overrelaxation on the magnitude only."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    if target == 'flat':
        T_target = np.zeros(M)
        T_target[:2 * n - 1] = 1.0 / (2 * n - 1)
    else:
        T_target = np.zeros(M)
        T_target[:2 * n - 1] = 1.0 / (2 * n - 1)

    T_fft = np.fft.fft(T_target)
    T_mag = np.maximum(np.abs(T_fft), 1e-20)

    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.abs(A)
        A_phase = np.exp(1j * np.angle(A))

        # Over-relax: push magnitude toward target
        A_mag_new = A_mag + alpha * (T_mag - A_mag)
        A_mag_new = np.maximum(A_mag_new, 1e-20)

        A_new = A_mag_new * A_phase
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shaving_qp(a, n_rounds=100, n_peaks=5, step=0.1):
    """
    Peak shaving: for each top peak, find the elements that contribute most,
    and reduce them while increasing others to maintain the sum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution to c_k from each a_j
            seg = a[j_start:j_end + 1]
            partner_idx = k - np.arange(j_start, j_end + 1)
            partner = a[partner_idx]
            contrib = seg * partner

            # Reduce top contributors, increase bottom ones
            top_n = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_n]
            bot_local = np.argsort(contrib)[:top_n]

            reduction = contrib[top_local] * step
            total_reduction = reduction.sum()

            # Distribute increase to bottom contributors
            if total_reduction > 0:
                a[j_start + top_local] -= reduction
                if len(bot_local) > 0:
                    a[j_start + bot_local] += total_reduction / len(bot_local)

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_exact(a, n_iters=5000, step_init=0.02, decay=0.99995, n_avg=10):
    """
    Exact subgradient method on the max function, with gradient averaging.
    More stable than the previous multi-peak version.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find all lags within tolerance of max (for subgradient averaging)
        tol = 1e-6
        mask = c >= c_max - tol
        if mask.sum() == 0:
            mask = np.zeros_like(c, dtype=bool)
            mask[int(np.argmax(c))] = True
        active = np.where(mask)[0]

        # Subgradient: average of a_{k-i} over active peaks
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                partner_idx = k - np.arange(j_start, j_end + 1)
                sg[j_start:j_end + 1] += a[partner_idx]

        sg /= len(active)

        # Full gradient of score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg * S - c_max) / S3

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def adaptive_lse(a0, beta_schedule, maxiter_each=3000, maxfun_each=15000):
    """
    Adaptive LSE: run LSE with increasing beta values.
    Each stage starts from the previous result.
    """
    a = a0.copy()
    for beta in beta_schedule:
        a = run_lbfgs(a, lse_score_grad, (beta,), maxiter=maxiter_each, maxfun=maxfun_each)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return a


def random_perturb(a0, rng, strength=0.05):
    """Add random perturbation and renormalize."""
    a = a0.copy()
    perturbation = rng.standard_normal(len(a)) * strength
    a += perturbation
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a0.copy()
    return a / S


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.3 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)

    # Random Fourier-based
    for freq_range in [(2, 8), (5, 20), (1, 5), (3, 12)]:
        freqs = np.random.randint(freq_range[0], freq_range[1], size=5)
        weights = rng.standard_normal(5) * 0.3
        phases = rng.uniform(0, 2 * np.pi, 5)
        sig = np.zeros(n)
        for i in range(5):
            sig += weights[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        inits.append(np.maximum(0.5 + sig, 0.01))

    # Sparse-like initializations
    for _ in range(3):
        a = np.maximum(0.1 + 0.9 * rng.random(n), 0.01)
        inits.append(a)

    return inits


def nesterov_refinement(a0, beta, n_iters=15000, lr_init=0.008, lr_decay=0.99995, momentum=0.9):
    """Nesterov momentum on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)

        # Simple momentum (Nesterov lookahead every 100 steps)
        if it % 100 == 0:
            a_look = a - momentum * v
            a_look = np.maximum(a_look, 0.0)
            sl, gl = lse_score_grad(a_look, beta)
            v = momentum * v + lr * gl
        else:
            v = momentum * v + lr * grad

        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= lr_decay

        if it % 200 == 0:
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()

    sc_exact = compute_score(a)
    if sc_exact < best_sc:
        best_a = a.copy()

    return best_a


def coarse_to_fine(parent, target_n, rng):
    """Solve at coarser resolution then interpolate."""
    results = []
    for div in [8, 4, 2]:
        cn = target_n // div
        if cn < 64:
            continue
        if parent is not None:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, cn)
            a_init = np.interp(x_new, x_old, parent)
        else:
            a_init = np.ones(cn) * 0.5

        # Quick spectral + LSE at this resolution
        a1 = spectral_annealing(a_init, n_iters=200, target='flat', alpha_init=0.4, alpha_final=0.1)
        a2 = run_lbfgs(a1, lse_score_grad, (2000,), maxiter=3000, maxfun=10000)
        results.append(a2)

    # Interpolate the finest result to target_n
    if results:
        finest = results[-1]
        x_old = np.linspace(0, 1, len(finest))
        x_new = np.linspace(0, 1, target_n)
        return np.interp(x_new, x_old, finest)
    return np.ones(target_n) * 0.5

            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"Starting optimization with n={n}, budget={total_budget}s")
    print(f"Initial best: {best_score:.8f}")

    # ==================================================================
    # Phase 1: Spectral warmup with diverse targets (12% of time)
    # ==================================================================
    phase1_end = t_start + 0.12 * (total_budget - 60.0)
    print(f"\n=== Phase 1: Spectral warmup (deadline {phase1_end - t_start:.0f}s) ===")

    targets = ['flat', 'gaussian', 'exponential']
    inits = build_inits(n, rng, parent if len(parent) == n else None)

    for i, a0 in enumerate(inits[:8]):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        S0 = a0.sum()
        if S0 > 0:
            a0 = a0 / S0

        # Try different spectral targets
        for t in targets[:2]:
            if time.monotonic() > phase1_end:
                break
            a1 = spectral_annealing(a0, n_iters=200, target=t, alpha_init=0.4, alpha_final=0.08)
            sc = compute_score(a1)
            update_best(sc, a1)

        # Also try overrelaxation
        if time.monotonic() < phase1_end:
            a2 = spectral_overrelax(a0, n_iters=200, target='flat', alpha=0.3)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Adaptive LSE annealing (35% of time)
    # ==================================================================
    phase2_end = t_start + 0.47 * (total_budget - 60.0)
    print(f"\n=== Phase 2: Adaptive LSE (deadline {phase2_end - t_start:.0f}s) ===")

    # Simulated annealing on beta: start low, increase
    beta_schedule = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    current = best_a.copy()

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        t_beta = time.monotonic()
        # Scale maxiter based on time remaining
        time_per_beta = (phase2_end - t_beta) / max(1, len(beta_schedule) - beta_schedule.index(beta) - 1)
        maxiter = min(8000, max(1000, int(time_per_beta * 50)))
        maxfun = min(30000, max(5000, int(time_per_beta * 200)))

        current = run_lbfgs(current, lse_score_grad, (beta,), maxiter=maxiter, maxfun=maxfun)
        current = np.maximum(current, 0.0)
        S = current.sum()
        if S > 1e-12:
            current = current / S
        sc = compute_score(current)
        update_best(sc, current)
        print(f"  LSE beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Peak shaving + Subgradient (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.72 * (total_budget - 60.0)
    print(f"\n=== Phase 3: Peak shaving + Subgradient (deadline {phase3_end - t_start:.0f}s) ===")

    # Peak shaving with different parameters
    for n_peaks, step in [(3, 0.05), (5, 0.03), (10, 0.02), (5, 0.05)]:
        if time.monotonic() > phase3_end:
            break
        a_ps = peak_shaving_qp(best_a, n_rounds=80, n_peaks=n_peaks, step=step)
        sc = compute_score(a_ps)
        update_best(sc, a_ps)
        print(f"  PeakShave peaks={n_peaks} step={step}: score={sc:.8f}")

    # Subgradient with exact max
    if time.monotonic() < phase3_end:
        a_sg = subgradient_exact(best_a, n_iters=5000, step_init=0.015, decay=0.99995)
        sc = compute_score(a_sg)
        update_best(sc, a_sg)
        print(f"  Subgradient: score={sc:.8f}")

    # LSE refinement after shaving
    if time.monotonic() < phase3_end:
        for beta in [20000, 50000]:
            if time.monotonic() > phase3_end:
                break
            a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=5000, maxfun=20000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)
            print(f"  LSE post-shave beta={beta}: score={sc:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Nesterov + diverse restarts (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.87 * (total_budget - 60.0)
    print(f"\n=== Phase 4: Nesterov + restarts (deadline {phase4_end - t_start:.0f}s) ===")

    # Nesterov from current best
    if time.monotonic() < phase4_end:
        a_nm = nesterov_refinement(best_a, beta=10000, n_iters=10000, lr_init=0.005,
                                   lr_decay=0.99995, momentum=0.9)
        sc = compute_score(a_nm)
        update_best(sc, a_nm)
        print(f"  Nesterov: score={sc:.8f}")

    # Diversified restarts
    for trial in range(3):
        if time.monotonic() > phase4_end:
            break
        # Perturb current best
        a_pert = random_perturb(best_a, rng, strength=0.02 + 0.01 * trial)
        # Quick spectral + LSE
        a_sp = spectral_annealing(a_pert, n_iters=100, target='flat', alpha_init=0.3, alpha_final=0.05)
        a_ls = run_lbfgs(a_sp, lse_score_grad, (20000,), maxiter=3000, maxfun=10000)
        sc = compute_score(a_ls)
        update_best(sc, a_ls)
        print(f"  Restart {trial}: score={sc:.8f}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final high-beta polish (13% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final polish (deadline {deadline - t_start:.0f}s) ===")

    # Very high beta LSE (essentially exact max)
    for beta in [100000, 200000, 500000]:
        if time.monotonic() > deadline:
            break
        t_beta = time.monotonic()
        remaining = deadline - t_beta
        maxiter = min(10000, max(2000, int(remaining * 30)))
        maxfun = min(40000, max(10000, int(remaining * 100)))
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=maxiter, maxfun=maxfun)
        a_opt = np.maximum(a_opt, 0.0)
        S = a_opt.sum()
        if S > 1e-12:
            a_opt = a_opt / S
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  Final LSE beta={beta}: score={sc:.8f} ({time.monotonic()-t_beta:.1f}s)")

    # One more round of peak shaving + LSE
    if time.monotonic() < deadline:
        a_ps = peak_shaving_qp(best_a, n_rounds=50, n_peaks=3, step=0.02)
        a_fin = run_lbfgs(a_ps, lse_score_grad, (500000,), maxiter=5000, maxfun=20000)
        sc = compute_score(a_fin)
        update_best(sc, a_fin)
        print(f"  Final shave+LSE: score={sc:.8f}")

    # Final spectral touch-up
    if time.monotonic() < deadline:
        a_sp = spectral_annealing(best_a, n_iters=100, target='flat', alpha_init=0.1, alpha_final=0.02)
        a_fin2 = run_lbfgs(a_sp, lse_score_grad, (500000,), maxiter=3000, maxfun=10000)
        sc = compute_score(a_fin2)
        update_best(sc, a_fin2)
        print(f"  Final spectral+LSE: score={sc:.8f}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
