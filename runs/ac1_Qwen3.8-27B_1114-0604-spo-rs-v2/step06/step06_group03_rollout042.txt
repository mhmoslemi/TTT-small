def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_with_lags(a, lag_indices):
    """Compute conv(a,a) at specific lag indices (0-indexed in full conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    full = np.fft.irfft(A * A, N)[:M]
    lag_indices = np.clip(lag_indices, 0, M - 1)
    return full[lag_indices]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

    cmax = float(np.max(c))
    if cmax < 0.0:

    return float(2.0 * n * cmax / (S * S))

def run_lbfgs(a0, beta, time_limit_s, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if time_limit_s < 3.0:
        return exact_score_a(a), a
    maxfun = max(32, int(maxfun))
    maxiter = max(30, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    t0 = time.monotonic()
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_lags_topk(c, k):
    """Return indices of top-k values in convolution."""
    m = int(c.size)
    if m <= 0:
        return np.array([], dtype=np.int64)
    if k <= 0 or k >= m:
        return np.arange(m)
    idx = np.argpartition(c, m - k)[m - k:]
    return np.sort(idx)


def subgrad_step_active(a, k_active, eta):
    """Subgradient step targeting top-k active lags."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    lag_idx = active_lags_topk(c, k_active)
    w = np.zeros(c.size)
    w[lag_idx] = 1.0 / len(lag_idx)
    
    # Gradient using same method as LSE but with uniform weights on active set
    Nc = next_power_of_two(3 * n - 1)
    Ac = np.fft.rfft(a, Nc)
    Wc = np.fft.rfft(w, Nc)
    xc1 = np.fft.irfft(Wc * np.conj(Ac), Nc)
    xc2 = np.fft.irfft(Ac * np.conj(Wc), Nc)
    offset = n - 1
    cross1 = xc2[offset:offset + n]
    cross2 = xc1[offset:offset + n]
    
    d = (float(n) * c) / (S * S)
    LDS = float(np.dot(w, d))
    grad = (float(n) / (S * S)) * (cross1 + cross2) - 2.0 * LDS / S
    
    a_new = a - eta * grad
    return normalize_a(np.maximum(a_new, 0.0))


def mirror_descent_step(a, k_active, t_power):
    """Mirror descent: a_new ~ (crosscorr with active weights)^t * a^(1-t)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return a
    c = fast_self_conv(a)
    lag_idx = active_lags_topk(c, k_active)
    w = np.zeros(c.size)
    w[lag_idx] = 1.0 / len(lag_idx)
    
    Nc = next_power_of_two(3 * n - 1)
    Ac = np.fft.rfft(a, Nc)
    Wc = np.fft.rfft(w, Nc)
    xc1 = np.fft.irfft(Wc * np.conj(Ac), Nc)
    xc2 = np.fft.irfft(Ac * np.conj(Wc), Nc)
    offset = n - 1
    h = xc2[offset:offset + n] + xc1[offset:offset + n]
    h = np.maximum(h, 1e-15)
    
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def power_iteration_blend(a, k_active, blends=(0.1, 0.3, 0.5, 0.7, 0.9)):
    """Power iteration: replace a with cross-correlation with active set, blended."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-12:
        return a, exact_score_a(a)
    c = fast_self_conv(a)
    lag_idx = active_lags_topk(c, k_active)
    w = np.zeros(c.size)
    w[lag_idx] = 1.0 / len(lag_idx)
    
    Nc = next_power_of_two(3 * n - 1)
    Ac = np.fft.rfft(a, Nc)
    Wc = np.fft.rfft(w, Nc)
    xc1 = np.fft.irfft(Wc * np.conj(Ac), Nc)
    xc2 = np.fft.irfft(Ac * np.conj(Wc), Nc)
    offset = n - 1
    h = xc2[offset:offset + n] + xc1[offset:offset + n]
    h = np.maximum(h, 0.0)
    h = normalize_a(h)
    
    best_a = a.copy()
    best_sc = exact_score_a(a)
    for blend in blends:
        q = normalize_a(blend * h + (1.0 - blend) * a)
        sc = exact_score_a(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()
    return best_a, float(best_sc)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add_safe(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if np.sum(arr) < 1e-10:
            arr = np.ones(n)
        inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add_safe(pr)
        add_safe(pr[::-1])

    # Uniform
    add_safe(np.ones(n))

    # Geometric decay (key family - achieves ~1.841 asymptotically)
    for c in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        r = np.exp(-c / n)
        add_safe(r ** np.arange(n))
        add_safe((r ** np.arange(n))[::-1])

    # Double geometric (decay from center)
    for c in (1.0, 2.0, 3.0):
        center = np.arange(n) - n / 2.0
        add_safe(np.exp(-c * np.abs(center) / n))

    # V-shape
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0, 3.0):
        add_safe(1.0 + alpha * np.abs(2.0 * x - 1.0))
        add_safe(1.0 + alpha * np.abs(2.0 * x - 1.0) ** 2)
        add_safe(1.0 + alpha * np.abs(2.0 * x - 1.0) ** 0.5)

    # Cosine
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
            add_safe(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add_safe(1.0 + alpha * np.cos(np.pi * f * x))
            add_safe(1.0 + alpha * np.cos(2.0 * np.pi * f * (x - 0.25)))

    # Sine-squared
    for f in (1, 2, 3):
        for alpha in (0.5, 1.0, 2.0):
            add_safe(1.0 + alpha * np.sin(np.pi * f * x) ** 2)

    # Bimodal
    for width in (0.1, 0.15, 0.2, 0.3):
        for pos in ((0.25, 0.75), (0.2, 0.8), (0.3, 0.7)):
            add_safe(np.exp(-(((x - pos[0]) / width) ** 2)) + np.exp(-(((x - pos[1]) / width) ** 2)))

    # Edge concentration
    for width in (5, 10, 20, 50):
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        add_safe(np.exp(-d / float(width)))

    # Random smooth (Fourier)
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        p = float(rng.uniform(1.5, 4.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** p)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        amp = float(rng.uniform(0.3, 1.5))
        add_safe(1.0 + amp * noise)

    # Asymmetric polynomial
    for deg in (2, 3):
        for c1 in rng.uniform(-2, 2, size=4):
            add_safe((1.0 + c1 * x + 0.5 * c1**2 * x**2) ** 2 + 0.1)

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 10))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:

    elif mode == 3:
        # Reduce peak lags specifically
        c = fast_self_conv(p)
        lag_idx = active_lags_topk(c, int(rng.integers(1, 8)))
        w = np.zeros(c.size)
        w[lag_idx] = 1.0 / len(lag_idx)
        Nc = next_power_of_two(3 * n - 1)
        Ac = np.fft.rfft(p, Nc)
        Wc = np.fft.rfft(w, Nc)
        xc1 = np.fft.irfft(Wc * np.conj(Ac), Nc)
        xc2 = np.fft.irfft(Ac * np.conj(Wc), Nc)
        h = xc2[n-1:n-1+n] + xc1[n-1:n-1+n]
        h = np.maximum(h, 0)
        if h.max() > 1e-15:
            q = p * np.exp(-strength * h / h.max())
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed log-space noise
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        logp += strength * 0.3 * noise
