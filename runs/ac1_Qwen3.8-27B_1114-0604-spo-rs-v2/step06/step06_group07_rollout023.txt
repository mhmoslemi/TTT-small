def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(f, g):
    """Fast linear cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    F = np.fft.rfft(f, M_fft)
    G = np.fft.rfft(g, M_fft)
    return np.fft.irfft(F * G, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective with gradient."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # p_k = exp(beta*(c_k - c_max)) / sum(exp(...))
    p = exp_scaled / exp_sum

    # g_L = 2 * (p correlated with x)
    # (p * rev_x)[i + n - 1] = sum_k p[k] * x[i] ... need to be careful
    # Actually: conv(p, x)[i + n - 1] gives sum_k p[k]*x[n-1-(i+n-1-k)] = sum_k p[k]*x[k-i]
    # But we need sum_k p[k] * x[k-i] for the gradient of sum_k p[k]*c[k]
    # where c[k] = sum_j x[j]*x[k-j], dc[k]/dx[i] = 2*x[k-i] (or x[i] if 2i=k)
    # Simplification: use conv(p, rev_x)

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad

    """Run L-BFGS-B on the LSE-smoothed objective with time monitoring."""

def peak_gradient(a, n_peaks=None, threshold_frac=0.9):
    """
    Compute subgradient of max(conv(a,a)) targeting multiple peaks.
    Returns gradient of 2*n*max_conv/S^2 (approximately, using weighted peaks).
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    M_val = float(c.max())

    if threshold_frac is not None:
        threshold = threshold_frac * M_val
        peak_mask = c >= threshold
    else:
        if n_peaks is not None and n_peaks < M:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            peak_mask = np.zeros(M, dtype=bool)
            peak_mask[top_idx] = True
        else:
            peak_mask = np.ones(M, dtype=bool)

    # Weight peaks by how much they exceed threshold
    if threshold_frac is not None:
        weights = np.maximum(0, c - threshold) / M_val
    else:
        weights = np.ones(M) / peak_mask.sum()

    weights = weights / max(weights.sum(), 1e-30)

    # Compute weighted subgradient: sg[j] = 2 * sum_k weights[k] * a[k-j]
    # This is 2 * (weights correlated with a)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(weights, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    sg = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
    return grad, M_val, sc := 2.0 * n * M_val / (S * S)


def gradient_descent_line_search(a, n_iters=5000, threshold_frac=0.85,
                                  momentum_init=0.9, momentum_decay=1.0,
                                  max_step=0.05, time_limit=300):
    """
    Gradient descent with backtracking line search and momentum.
    Uses weighted peak subgradient targeting peaks above a threshold.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    momentum = np.zeros(n)
    mu = momentum_init
    t0 = time.monotonic()
    base_step = max_step

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc_curr = 2.0 * n * M_val / (S * S)

        if sc_curr < best_sc:
            best_sc = sc_curr
            best_a = a.copy()

        # Threshold for peak selection
        threshold = threshold_frac * M_val
        peak_mask = c >= threshold
        n_peaks_active = peak_mask.sum()
        if n_peaks_active == 0:
            peak_mask = np.zeros(len(c), dtype=bool)
            peak_mask[int(np.argmax(c))] = True

        # Weight peaks
        weights = np.zeros(len(c))
        weights[peak_mask] = (c[peak_mask] - threshold) / (M_val - threshold + 1e-30)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights[int(np.argmax(c))] = 1.0
            wsum = 1.0
        weights /= wsum

        # Compute subgradient via FFT
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        WF = np.fft.rfft(weights, M2_fft)
        conv_res = np.fft.irfft(RA * WF, M2_fft)[n - 1:2 * n - 1]
        sg = 2.0 * conv_res

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Momentum update
        momentum = mu * momentum - grad
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive base step
        gap = sc_curr - sc_lb
        base_step = min(max_step, max(gap / (gnorm * gnorm + 1e-30), 1e-10))

        # Backtracking line search
        step = base_step
        f_curr = sc_curr
        a_try = a - step * momentum
        a_try = np.maximum(a_try, 0.0)
        S_try = float(a_try.sum())
        if S_try < 1e-12:
            step *= 0.5
            a_try = a - step * momentum
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try < 1e-12:
                break
            a_try = a_try / S_try

        A_try = np.fft.rfft(a_try, M_fft)
        c_try = np.fft.irfft(A_try * A_try, M_fft)[:2 * n - 1]
        f_try = 2.0 * n * float(c_try.max()) / (S_try * S_try)

        c_armijo = 1e-4
        for _ls in range(20):
            if f_try <= f_curr - c_armijo * step * gnorm:
                break
            step *= 0.5
            a_try = a - step * momentum
            a_try = np.maximum(a_try, 0.0)
            S_try = float(a_try.sum())
            if S_try < 1e-12:
                break
            a_try = a_try / S_try
            A_try = np.fft.rfft(a_try, M_fft)
            c_try = np.fft.irfft(A_try * A_try, M_fft)[:2 * n - 1]
            f_try = 2.0 * n * float(c_try.max()) / (S_try * S_try)
        else:
            # Accept anyway if improvement
            if f_try < f_curr:

            continue

        a = a_try
        # No renormalization needed (line search already handles it)

def spectral_shaping_soft(a, n_iters=500, alpha_start=0.05, alpha_end=0.9,
                          soft_alpha=30.0, time_limit=200):
    """
    Spectral shaping with soft non-negativity enforcement.
    Uses sigmoid gating instead of hard clipping for smoother updates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        # Target: geometric mean of non-DC amplitudes (more robust than median)
        if len(amp) > 1:
            log_amps = np.log(np.maximum(amp[1:], 1e-30))
            C = float(np.exp(np.mean(log_amps)))
        else:
            C = amp[0]
        C = max(C, 1e-20)

        # Geometric shaping
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Soft non-negativity: use softplus-like gating
        # a_pos = a_new where a_new > 0, smoothly transitioned
        gate = 1.0 / (1.0 + np.exp(-soft_alpha * np.clip(a_new, -50, 50)))
        a_new = a_new * gate
        a_new = np.maximum(a_new, 0.0)

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coordinate_block_descent(a, n_iters=2000, block_size=8, time_limit=200):
    """
    Block coordinate descent: optimize groups of adjacent elements using
    quadratic approximation of the objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Process blocks
        n_blocks = n // block_size
        for b in range(n_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            start = b * block_size
            end = start + block_size
            if end > n:
                break

            # For this block, compute the contribution to the peak
            # The peak is at position k*. The block elements a[start:end]
            # contribute to c[k] for k in the relevant range.
            k_star = int(np.argmax(c))

            # Quadratic approximation: c[k_star] = sum_j a[j]*a[k_star-j]
            # For j in [start, end), the contribution is a[j]*a[k_star-j]
            # Write c[k_star] = (a_block)^T Q a_block + a_block^T r + const
            # where Q[j,j'] = ... (complex)
            # Simplification: just do a 1D line search in each coordinate

            for j in range(start, end):
                # Current value
                aj = a[j]
                if aj < 1e-15:
                    continue
                # Contribution of a[j] to c[k_star]: a[j] * a[k_star - j]
                k_j = k_star - j
                if 0 <= k_j < n and k_j != j:
                    # Reducing a[j] by delta reduces c[k_star] by 2*delta*a[k_j]
                    # But also affects other c[k] values
                    # Simple approach: try a small reduction
                    delta = 0.1 * aj
                    a[j] = aj - delta
                    # Check if peak decreased
                    A2 = np.fft.rfft(a, M_fft)
                    c2 = np.fft.irfft(A2 * A2, M_fft)[k_star - 1:k_star + 2]
                    if c2.max() > c[k_star] * 1.01:
                        a[j] = aj  # Revert
                    else:
                        a[j] = max(a[j], 0.0)

        # Renormalize periodically
        if it % 10 == 0:
            S = float(a.sum())
            if S > 0:
                a = a / S

    return best_a


def perturbation_restart(a, noise_level=0.03, n_iters_geometric=100, time_limit=100):
    """Perturb and re-optimize to escape local minima."""
    n = len(a)
    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    a_p = a * (1.0 + noise_level * rng.standard_normal(n))
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S < 1e-12:
        return a
    a_p = a_p / S

    # Quick geometric shaping
    a_r = spectral_shaping_soft(a_p, n_iters=n_iters_geometric,
                                 alpha_start=0.05, alpha_end=0.5,
                                 soft_alpha=30.0, time_limit=time_limit)
    return a_r


def lp_direction_iteration(a, M_target, n_max_iter=5000):
    """
    Approximate LP direction finding using coordinate ascent.
    Finds a direction g >= 0 that increases sum while keeping f*g <= M_target.
    Uses greedy coordinate selection with FFT-accelerated constraint evaluation.
    """
    n = len(a)
    f = a.copy()
    g = np.zeros(n)

    M_fft = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, M_fft)

    # Current cross-convolution f*g (initially zero)
    fg = np.zeros(2 * n - 1)

    # Precompute: for each coordinate j, the "column" of the constraint matrix
    # Column j of A is: A[k, j] = f[k-j] for valid k
    # The increment to fg when we increase g[j] by delta is: delta * f[reversed/padded]
    # Specifically, (f * delta*e_j)[k] = delta * f[k-j] for k in [j, j+n-1]

    # Greedy: at each step, find the j that allows the largest increase
    # while maintaining fg + delta*f_col_j <= M_target
    # delta_max[j] = min_k (M_target - fg[k]) / f[k-j] over k where f[k-j] > 0

    # For efficiency, we process in batches
    f_abs = np.abs(f)
    # Find coordinates with significant f values (skip near-zero f[j])
    active_js = np.where(f > 0.01 * f.max())[0]
    if len(active_js) == 0:
        return g

    step_count = 0
    t0 = time.monotonic()

    while step_count < n_max_iter:
        if time.monotonic() - t0 > 30:
            break

        # Compute max allowable increment for each active j
        # (cheap approximation: only check the current peak position)
        k_peak = int(np.argmax(fg))
        # For the peak constraint: fg[k_peak] + delta * f[k_peak - j] <= M_target
        # delta <= (M_target - fg[k_peak]) / f[k_peak - j]

        valid_weights = np.zeros(len(active_js))
        for ix, j in enumerate(active_js):
            k_j = k_peak - j
            if 0 <= k_j < n:
                fj = f[k_j]
                if fj > 1e-15:
                    slack = M_target - fg[k_peak]
                    valid_weights[ix] = min(slack / fj, 1.0) if slack > 0 else 0.0

        if valid_weights.max() < 1e-15:
            # Try all peak positions
            break

        # Pick the coordinate with the largest allowable step
        # Weight by how much it contributes to the sum increase
        best_ix = int(np.argmax(valid_weights))
        delta = valid_weights[best_ix]
        if delta < 1e-10:
            # Try a smaller set of peaks
            top_k = np.argpartition(fg, -5)[-5:]
            best_delta = 0
            best_j = -1
            for j in active_js[:200]:
                for k in top_k:
                    k_j = int(k) - j
                    if 0 <= k_j < n and f[k_j] > 1e-15:
                        d = (M_target - fg[k]) / f[k_j]
                        if d > best_delta:
                            best_delta = d
                            best_j = j
            if best_j < 0 or best_delta < 1e-10:
                break
            delta = min(best_delta, 0.5)
            j = best_j
        else:
            j = active_js[best_ix]

        # Update
        delta = min(delta, 0.5)
        g[j] += delta
        # Update fg: fg[k] += delta * f[k-j] for k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        f_seg = f[0:k_end - j]
        fg[k_start:k_end] += delta * f_seg

        step_count += 1

    return g


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function using multi-strategy optimization.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
            print(f"Parent loaded: length={len(parent)}, score={compute_score(parent):.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        n_work = len(parent)
        sc = compute_score(parent)
        update_best(sc, parent, "PARENT")
    else:
        n_work = 8192
        best_a = np.ones(n_work) * 0.5
        update_best(compute_score(best_a), best_a, "UNIFORM")

    print(f"Working length: {n_work}, Starting score: {best_score:.8f}")

    # ============================================================
    # Phase 1: L-BFGS with adaptive beta (40% time)
    # ============================================================
    phase1_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 1: L-BFGS (deadline t={phase1_end - t_start:.0f}s) ===")

    # Start with moderate beta and increase
    betas_seq = [200, 500, 1000, 2000, 5000, 10000]
    for beta in betas_seq:
        if time.monotonic() > phase1_end - 30:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 60:
            break
        print(f"  L-BFGS beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(8000, int(remaining * 25))
        maxfun = maxiter * 20
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun,
                               time_limit=max(60, remaining - 30))
        update_best(sc, a_opt, f"LBFGS b={beta}")

    # ============================================================
    # Phase 2: Gradient descent with line search (25% time)
    # ============================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Gradient Descent (deadline t={phase2_end - t_start:.0f}s) ===")

    # Different threshold fractions
    for thr in [0.80, 0.85, 0.90, 0.95]:
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        print(f"  GradDesc threshold={thr}, remaining={remaining:.0f}s")
        a_gd = gradient_descent_line_search(
            best_a, n_iters=5000, threshold_frac=thr,
            momentum_init=0.85, max_step=0.04,
            time_limit=min(remaining - 10, 300)
        )
        update_best(compute_score(a_gd), a_gd, f"GD thr={thr}")

    # ============================================================
    # Phase 3: Spectral shaping + hybrid (20% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Spectral Shaping (deadline t={phase3_end - t_start:.0f}s) ===")

    # Soft spectral shaping with different parameters
    for alpha_end_val in [0.5, 0.7, 0.9]:
        if time.monotonic() > phase3_end - 30:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        print(f"  Spectral alpha_end={alpha_end_val}")
        a_spec = spectral_shaping_soft(
            best_a, n_iters=400, alpha_start=0.03, alpha_end=alpha_end_val,
            soft_alpha=50.0, time_limit=min(remaining - 10, 200)
        )
        update_best(compute_score(a_spec), a_spec, f"SPEC ae={alpha_end_val}")

    # LP-inspired direction (simplified)
    if time.monotonic() < phase3_end - 30:
        print("  LP direction iteration")
        c = fast_conv(best_a)
        M_val = float(c.max())
        S = float(best_a.sum())
        a_norm = best_a / S
        g = lp_direction_iteration(a_norm, M_target=float(c.max()) / (S * S),
                                     n_max_iter=2000)
        if g.sum() > 0:
            # Blend
            for t_blend in [0.1, 0.3, 0.5]:
                a_blend = (1 - t_blend) * best_a + t_blend * g * S
                a_blend = np.maximum(a_blend, 0.0)
                S_b = float(a_blend.sum())
                if S_b > 0.01:
                    a_blend = a_blend / S_b * S
                    sc = compute_score(a_blend)
                    update_best(sc, a_blend, f"LP t={t_blend}")

    # ============================================================
    # Phase 4: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # L-BFGS with high beta from current best
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        print(f"  Final L-BFGS, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(best_a, 8000, maxiter=5000, maxfun=80000,
                               time_limit=max(30, int(remaining * 0.6)))
        update_best(sc, a_opt, "FINAL-LBFGS")

    # Gradient descent with very tight threshold
    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        a_gd2 = gradient_descent_line_search(
            best_a, n_iters=3000, threshold_frac=0.97,
            momentum_init=0.9, max_step=0.02,
            time_limit=max(20, int(remaining * 0.5))
        )
        update_best(compute_score(a_gd2), a_gd2, "FINAL-GD")

    # Short spectral refinement
    if time.monotonic() < deadline - 30:
        a_spec2 = spectral_shaping_soft(
            best_a, n_iters=150, alpha_start=0.02, alpha_end=0.4,
            soft_alpha=80.0, time_limit=60
        )
        update_best(compute_score(a_spec2), a_spec2, "FINAL-SPEC")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(2):
            if time.monotonic() > deadline - 10:
                break
            a_p = perturbation_restart(best_a, noise_level=0.02 * (1 + trial),
                                        n_iters_geometric=80, time_limit=60)
            update_best(compute_score(a_p), a_p, f"RESTART {trial}")

    # One more L-BFGS polish if time
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 15000, maxiter=3000, maxfun=50000,
                               time_limit=30)
        update_best(sc, a_opt, "ULTRA-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
