"""
Improved optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Subgradient descent with averaging and heavy-ball momentum (main workhorse)
2. Active-set LP refinement using scipy's HiGHS solver (exact LP for top-K constraints)
3. Diverse multi-start with U/W-shaped initializations
4. LSE-smoothed L-BFGS for final polish
5. Adaptive length selection

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def project_simplex(x):
    """Project x onto {y >= 0, sum(y) = 1}."""
    n = len(x)
    # Bisection on threshold
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(x - tau, 0.0)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient: (B^T p)[j] = sum_i a[i] * p[i+j]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def subgradient_descent(a_init, n_steps, rng, momentum=0.9, step_schedule='polyak'):
    """
    Subgradient descent with averaging for the non-smooth objective.
    Works on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Running average (Polyak averaging)
    a_avg = a.copy()
    avg_best_a = a.copy()
    avg_best_sc = best_sc
    
    v = np.zeros(n)  # momentum
    
    # Lower bound on score (theoretical min is 2n/(2n-1))
    f_lower = 2.0 * n / (2.0 * n - 1.0)
    
    for t in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        S2 = 1.0  # since a is normalized to sum=1
        
        # Subgradient of 2n * conv[k_star] w.r.t. a
        # d(conv[k_star])/da[j] = 2 * a[k_star - j] (with 0 padding)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        subgrad = np.zeros(n)
        if j_end >= j_start:
            # subgrad[j] = 2 * a[k_star - j] for j in [j_start, j_end]
            # a[k_star - j] for j in [j_start, j_end] means a at indices [k_star-j_start, ..., k_star-j_end]
            # = a[j_end, j_end-1, ..., j_start] = a[j_start:j_end+1][::-1]
            subgrad[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        grad = 2.0 * n * subgrad  # gradient of the objective (with S=1)
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        
        # Step size
        if step_schedule == 'polyak':
            gap = best_sc - f_lower
            if gap < 0.001:
                gap = 0.001
            gnorm2 = float(np.dot(grad, grad))
            if gnorm2 < 1e-30:
                step = 0.0
            else:
                step = 2.0 * gap / gnorm2
            step = min(step, 0.5)  # cap the step
        elif step_schedule == 'diminishing':
            step = 5.0 / np.sqrt(t + 1.0)
            step = min(step, 0.5)
        else:  # fixed
            step = 0.1
        
        # Update
        a_new = a - step * v
        a_new = project_simplex(a_new)
        
        # Check score
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Update average
        a_avg = (t * a_avg + a_new) / (t + 1)
        avg_sc = compute_score(a_avg)
        if avg_sc < avg_best_sc:
            avg_best_sc = avg_sc
            avg_best_a = a_avg.copy()
        
        a = a_new
    
    # Return the best of: best iterate, best average, final iterate
    candidates = [(best_sc, best_a), (avg_best_sc, avg_best_a)]
    candidates.sort(key=lambda x: x[0])
    return candidates[0][1], candidates[0][0]


def active_set_lp(a, K=50, time_limit=None):
    """
    Solve restricted LP: max sum(g) s.t. conv(a,g)[k_i] <= M for top-K positions k_i, g >= 0.
    Returns the normalized g.
    """
    n = len(a)
    c = self_conv(a)
    M = float(c.max())
    
    # Get top K positions
    top_k = np.argsort(c)[-K:]
    
    # Build constraint matrix A: K x n
    # A[i, j] = a[k_i - j] if 0 <= k_i - j <= n-1, else 0
    A = np.zeros((K, n))
    for i, k in enumerate(top_k):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]
    
    # LP: minimize -sum(g) s.t. A g <= M * 1, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    g = np.maximum(res.x, 0.0)
    Sg = float(g.sum())
    if Sg < 1e-12:
        return None
    return g / Sg


def active_set_lp_fast(a, K=50):
    """
    Faster active-set LP using efficient matrix construction.
    """
    n = len(a)
    c = self_conv(a)
    M = float(c.max())
    
    # Get top K positions
    top_k = np.argsort(c)[-K:]
    
    # Build constraint matrix efficiently
    A = np.zeros((K, n))
    for i, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # A[i, j] = a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start to k-j_end
            A[i, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    
    # LP: minimize -sum(g) s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    g = np.maximum(res.x, 0.0)
    Sg = float(g.sum())
    if Sg < 1e-12:
        return None
    return g / Sg

"""
Improved optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Subgradient descent with Polyak-Ruppert averaging and heavy-ball momentum
2. Active-set LP refinement using scipy HiGHS (exact LP on top-K binding constraints)
3. Diverse multi-start with structured initializations (U-shapes, modulated shapes)
4. LSE-smoothed L-BFGS for final polish
5. Adaptive length selection and time allocation
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def project_simplex(x):
    """Project x onto {y >= 0, sum(y) = 1} using bisection on threshold."""
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.maximum(x - tau, 0.0)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

    avg_best_a = a.copy()
    avg_best_sc = best_sc

    f_lower = 2.0 * n / (2.0 * n - 1.0)

        c = self_conv(a)
        k_star = int(np.argmax(c))
        q_val = float(c[k_star])

        # Subgradient: g_j = 4n * (a_{k*-j} - q_val) for S=1
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        grad = -4.0 * n * q_val * np.ones(n)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += 4.0 * n * a[j_start:j_end + 1][::-1]

        # Momentum (heavy ball)
        v = momentum * v + (1.0 - momentum) * grad

        # Step size
        if step_type == 'polyak':
            gap = best_sc - f_lower
            if gap < 0.002:
                gap = 0.002
            gnorm2 = float(np.dot(grad, grad))
            if gnorm2 < 1e-30:
                step = 0.0
            else:
                step = min(2.0 * gap / gnorm2, 0.3)
        elif step_type == 'diminishing':
            step = min(3.0 / np.sqrt(t + 1.0), 0.3)
        else:
            step = 0.05

        # Update with projection
        a_new = a - step * v
        a_new = project_simplex(a_new)

        # Track best iterate
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Update running average
        a_avg = (t * a_avg + a_new) / (t + 1.0)
        if t % 200 == 0:
            avg_sc = compute_score(a_avg)
            if avg_sc < avg_best_sc:
                avg_best_sc = avg_sc
                avg_best_a = a_avg.copy()

        a = a_new

    # Final check on average
    avg_sc = compute_score(a_avg)
    if avg_sc < avg_best_sc:
        avg_best_sc = avg_sc
        avg_best_a = a_avg.copy()

    # Return best of all candidates
    if avg_best_sc < best_sc:
        return avg_best_a, avg_best_sc
    return best_a, best_sc


def active_set_lp_solve(a, K=80):
    """
    Solve restricted LP: max sum(g) s.t. conv(a,g)[k_i] <= M for top-K positions, g >= 0.
    Returns normalized g or None on failure.
    """
    n = len(a)
    c = self_conv(a)
    M = float(c.max())

    top_k = np.argsort(c)[-K:]

    # Build constraint matrix A: K x n where A[i,j] = a[k_i - j] if valid
    A = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(top_k):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            idx_start = k - j_end
            idx_end = k - j_start
            A[i, j_start:j_end + 1] = a[idx_start:idx_end + 1]

    # LP: minimize -sum(g) s.t. A g <= M * 1, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    Sg = float(g.sum())
    if Sg < 1e-12:
        return None
    return g / Sg


def run_active_set_lp(a, max_iters=30, K=80, time_limit=None):
    """LP alternating with active-set (top-K constraints) using HiGHS."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        if no_improve >= 8:
            break

        t0 = time.monotonic()
        g = active_set_lp_solve(a, K=K)
        if g is None:
            break

        # Line search on mixing parameter
        best_t = 0.3
        best_t_sc = best_sc
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try a smaller step
            a_try = 0.95 * a + 0.05 * g
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St > 1e-12:
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0

    return best_a, best_sc


def run_lbfgs_polish(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    # U-shaped (high at edges)
    inits.append(np.abs(2.0 * x - 1.0) ** 1.5)
    # Strong U
    inits.append(np.abs(2.0 * x - 1.0) ** 0.7)
    # W-shaped
    inits.append(np.abs(4.0 * x - 2.0 - np.abs(2.0 * x - 1.0)) ** 0.8)
    # Cosine U
    inits.append(1.0 - 0.6 * np.cos(2.0 * np.pi * x) ** 2)
    # Modulated U: U-shape * fast oscillation
    inits.append(np.abs(2.0 * x - 1.0) ** 1.0 * (1.0 + 0.3 * np.cos(2.0 * np.pi * 7.0 * x)))
    # Two-peak
    inits.append(0.3 + 0.7 * (np.exp(-4.0 * ((x - 0.25) / 0.1) ** 2) +
                               np.exp(-4.0 * ((x - 0.75) / 0.1) ** 2)))
    # Multi-frequency
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x + 0.3)
                + 0.1 * np.cos(10.0 * np.pi * x + 1.0))
    # Triangular
    inits.append(1.0 - np.abs(2.0 * x - 1.0))
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Random smooth
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 20))
    inits.append(np.clip(0.5 + 0.4 * smooth, 0.01, 1.0))
    # Asymmetric: ramp with oscillation
    inits.append(np.clip(0.3 + 0.3 * x + 0.3 * np.cos(4.0 * np.pi * x), 0.01, 1.0))
    # Gaussian-shaped (peak in middle) - opposite of U
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2 / 0.1))

    result = []
    for a in inits:
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s > 1e-12:
            result.append(a / s)
        else:
            result.append(np.ones(n) / n)
    return result


def run_subgradient_full(a_init, total_seconds, rng):
    """
    Run subgradient descent with time budget.
    Uses adaptive step size and periodic evaluation.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    a_avg = a.copy()
    avg_best_a = a.copy()
    avg_best_sc = best_sc

    v = np.zeros(n)
    momentum = 0.95
    f_lower = 2.0 * n / (2.0 * n - 1.0)
    t_start = time.monotonic()
    t = 0

    while time.monotonic() - t_start < total_seconds:
        c = self_conv(a)
        k_star = int(np.argmax(c))
        q_val = float(c[k_star])

        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        grad = -4.0 * n * q_val * np.ones(n)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += 4.0 * n * a[j_start:j_end + 1][::-1]

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad

        # Adaptive step: Polyak with cap
        gap = best_sc - f_lower
        if gap < 0.001:
            gap = 0.001
        gnorm2 = float(np.dot(grad, grad))
        if gnorm2 < 1e-30:
            step = 0.0
        else:
            step = min(2.0 * gap / gnorm2, 0.2)

        # Update
        a_new = a - step * v
        a_new = project_simplex(a_new)

        # Evaluate (every few steps to save time)
        if t % 10 == 0:
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

        # Update average (cheap)
        a_avg = (t * a_avg + a_new) / (t + 1.0)
        if t % 500 == 0 and t > 0:
            avg_sc = compute_score(a_avg)
            if avg_sc < avg_best_sc:
                avg_best_sc = avg_sc
                avg_best_a = a_avg.copy()

        a = a_new
        t += 1

        # Periodic check on timing
        if t % 5000 == 0:
            elapsed = time.monotonic() - t_start
            remaining = total_seconds - elapsed
            if remaining < 5:
                break

    # Final evaluation
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    avg_sc = compute_score(a_avg)
    if avg_sc < avg_best_sc:
        avg_best_sc = avg_sc
        avg_best_a = a_avg.copy()

    if avg_best_sc < best_sc:
        return avg_best_a, avg_best_sc
    return best_a, best_sc


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: subgradient with averaging, active-set LP, L-BFGS polish.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(np.asarray(a, dtype=np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine lengths
    n_main = 6962
    if parent is not None:
        n_main = len(parent)
    lengths_to_try = [n_main]
    for n_try in [4096, 6962, 8192, 10240]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)
    lengths_to_try = sorted(set(lengths_to_try))[:4]
    print(f"Lengths to explore: {lengths_to_try}")

    # ==================================================================
    # Phase 1: Multi-start with subgradient (15% time)
    # ==================================================================
    phase1_time = 0.15 * (float(budget_s) - 30.0)
    phase1_deadline = t_start + phase1_time
    print(f"\n--- Phase 1: Multi-start Subgradient ({phase1_time:.0f}s budget) ---")

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")

        inits = build_inits(n, rng, parent)
        # Allocate time per init
        time_per_init = max(5.0, remaining / (len(inits) + 1))

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 5:
                break
            alloc = min(time_per_init, phase1_deadline - time.monotonic() - 2)
            if alloc < 3:
                break
            print(f"    init {i}, alloc={alloc:.0f}s", end=" ", flush=True)
            a_out, sc_out = run_subgradient(a0, int(alloc * 500), rng,
                                            momentum=0.95, step_type='polyak')
            sc = compute_score(a_out)
            print(f"score={sc:.8f}")
            report(sc, a_out, f"SG n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Extended subgradient with averaging (30% time)
    # ==================================================================
    phase2_time = 0.30 * (float(budget_s) - 30.0)
    phase2_deadline = t_start + 0.15 * (float(budget_s) - 30.0) + phase2_time
    print(f"\n--- Phase 2: Extended Subgradient ({phase2_time:.0f}s budget) ---")

    n_work = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Run for the full phase 2 time
    a_out, sc_out = run_subgradient_full(a_work, phase2_time - 10, rng)
    sc = compute_score(a_out)
    report(sc, a_out, "SG full")
    a_work = a_out.copy()

    # Also try from a different init at same length
    if time.monotonic() < phase2_deadline - 20:
        inits2 = build_inits(n_work, rng, parent)
        # Pick a few different ones
        for i in [0, 3, 7]:
            if i >= len(inits2) or time.monotonic() > phase2_deadline - 10:
                break
            alloc = phase2_deadline - time.monotonic() - 5
            alloc = max(5, alloc * 0.3)
            a_out2, sc_out2 = run_subgradient(inits2[i], int(alloc * 500), rng,
                                              momentum=0.9, step_type='diminishing')
            sc = compute_score(a_out2)
            report(sc, a_out2, f"SG2 n={n_work} i={i}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Active-set LP refinement (35% time)
    # ==================================================================
    phase3_time = 0.35 * (float(budget_s) - 30.0)
    phase3_deadline = phase2_deadline + phase3_time
    print(f"\n--- Phase 3: Active-Set LP ({phase3_time:.0f}s budget) ---")

    a_lp = best_a.copy()
    S = float(a_lp.sum())
    if S > 0:
        a_lp = a_lp / S

    # Progressive K: start small, increase
    K_schedule = [30, 50, 80, 120, 200]
    for K in K_schedule:
        if time.monotonic() > phase3_deadline - 30:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 20:
            break
        print(f"  K={K}, remaining={remaining:.0f}s")
        a_out, sc_out = run_active_set_lp(a_lp, max_iters=40, K=K,
                                           time_limit=phase3_deadline - 15)
        sc = compute_score(a_out)
        report(sc, a_out, f"LP K={K}")
        a_lp = a_out.copy()

    # Try different lengths with LP
    for n in lengths_to_try:
        if n == n_work or time.monotonic() > phase3_deadline - 40:
            continue
        print(f"  LP alt length n={n}")
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        St = float(a_test.sum())
        if St > 1e-12:
            a_test = a_test / St
        a_out, sc_out = run_active_set_lp(a_test, max_iters=20, K=50,
                                           time_limit=phase3_deadline - 20)
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS polish (15% time)
    # ==================================================================
    phase4_time = 0.15 * (float(budget_s) - 30.0)
    phase4_deadline = phase3_deadline + phase4_time
    print(f"\n--- Phase 4: L-BFGS Polish ({phase4_time:.0f}s budget) ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [100, 300, 800, 2000, 5000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs_polish(a_norm, beta, maxiter=3000, maxfun=80000,
                                           gtol=1e-13)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final LP + subgradient combined polish (5% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")
    final_deadline = deadline

    # Quick LP
    if time.monotonic() < final_deadline - 60:
        a_f = best_a.copy()
        S = float(a_f.sum())
        if S > 0:
            a_f = a_f / S
        a_out, sc_out = run_active_set_lp(a_f, max_iters=15, K=100,
                                           time_limit=final_deadline - 30)
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")
        if sc < best_score:
            best_a = a_out.copy()
            best_score = sc

    # Quick subgradient from best
    if time.monotonic() < final_deadline - 20:
        a_f = best_a.copy()
        S = float(a_f.sum())
        if S > 0:
            a_f = a_f / S
        alloc = final_deadline - time.monotonic() - 5
        a_out, sc_out = run_subgradient(a_f, int(alloc * 500), rng,
                                        momentum=0.9, step_type='polyak')
        sc = compute_score(a_out)
        report(sc, a_out, "final SG")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
