    # dL/da_i = 2n/S^2 * R[i] - 2n*T/S^3
    # where R[i] = sum_k w[k] * (db_k/da_i)
    # and T = sum_k w[k] * b[k]
    
    T = float(w.dot(b))
    
    # Compute R = cross-correlation of a and w
    # R[i] = sum_j a[j] * w[i+j] (for valid i+j)
    # Plus the "diagonal" correction: for each k, if k=2j, add w[k]*a[j]
    # Actually, from the derivative formula:
    # db_k/da_j = a[k-j] * 1[0<=k-j<=n-1] + a[j] * 1[k=2j]
    # R[j] = sum_k w[k] * (a[k-j]*1[0<=k-j<=n-1] + a[j]*1[k=2j])
    #       = sum_k w[k]*a[k-j]*1[0<=k-j<=n-1] + a[j]*w[2j]*1[2j < 2n-1]
    
    # First part: sum_k w[k]*a[k-j] where 0<=k-j<=n-1
    # = sum_i a[i] * w[i+j] where 0<=i<=n-1 (setting i=k-j, k=i+j)
    # = cross_correlation(a, w, lag=j)
    
    # Use FFT:
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3*n-2]
    R_cross = conv_full[n-1:2*n-1]  # R_cross[j] = sum_i a[i] * w[i+j]
    
    # Second part: a[j] * w[2j] for 2j < 2n-1, i.e., j < n (always true for j in [0,n-1])
    # 2j ranges from 0 to 2n-2, all valid
    R_diag = a * w[0:2*n:2]  # w[2j] for j=0,...,n-1
    
    R = R_cross + R_diag
    
    grad_a = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    
    # Chain rule for softmax: grad_x = a * (grad_a - dot(grad_a, a))
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    
    grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad_x

def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()
