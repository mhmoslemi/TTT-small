"""
Improved hybrid optimization for minimizing 2n*max(self_conv(a))/(sum(a))^2.
Uses MIQ phase retrieval, multi-scale LSE-LBFGS, active peak shaving,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def miq_phase_retrieval(a0, target_spec=None, n_iters=500):
    """
    Mixed In-phase and Quadrature (MIQ) phase retrieval.
    More powerful than plain GS.
    """
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    if target_spec is None:
        # Flat target
        target_spec = np.ones(M)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    gamma = 0.5  # MIQ parameter
    
    for it in range(n_iters):
        # Compute current spectrum
        A = np.fft.fft(a, M)
        
        # Target spectrum
        T = np.sqrt(target_spec)  # amplitude
        
        # MIQ: mix in-phase and quadrature components
        # A_new = A * T/|A| (ER) mixed with A * sqrt(T*conj(T)/|A|^2) ... 
        # Actually MIQ works as: 
        #   A_miq = gamma * A * (T / |A|) + (1-gamma) * A * sqrt(T/|A|) * exp(i*angle(T)/2)
        # Simplified: use geometric mean of ER and EE solutions
        
        A_mag = np.maximum(np.abs(A), 1e-20)
        
        # Error Reduction: A_ER = A * T / |A|
        A_ER = A * (T / A_mag)
        
        # Error Removal: A_EE = A * sqrt(|A|) / |A|^{3/4} ... 
        # Actually EE: project to |A|^2 = T^2
        # A_EE = A * (T / |A|)^{1/2}
        A_EE = A * np.sqrt(T / A_mag)
        
        # MIQ: average the phases
        A_miq = gamma * A_ER + (1 - gamma) * A_EE
        
        a_new = np.fft.ifft(A_miq, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
    
    return best_a


def riemann_phase_retrieval(a0, n_iters=500, alpha_init=0.5, alpha_final=0.1):
    """Riemann's EOS (Error Reduction + Overflow/Shift) method."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    
    # Flat target
    T_mag = np.ones(M)
    
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())

        return np.ones(n) / n

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)
        
        # Riemann: new_phase = (1-alpha)*old_phase + alpha*target_phase
        # target_phase for flat spectrum is 0

        new_mag = A_mag ** (1 - alpha) * T_mag ** alpha
        new_phase = (1 - alpha) * A_phase + alpha * target_phase
        
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        if it % 50 == 0:

            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_sc:

    return best_a


def peak_shaving(a0, n_rounds=100, n_peaks=20, step_decay=0.999):
    """
    Directly reduce top peaks using gradient information.
    For each peak k, the gradient of c_k w.r.t. a is:
    d(c_k)/d(a_i) = a_{k-i} (for valid i)
    So to reduce c_k, we should decrease a_i where a_{k-i} is large.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.1
    
    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find active peaks (within 5% of max)
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        active = active[:n_peaks]  # Cap at n_peaks
        
        if len(active) == 0:
            break
        
        # Compute combined gradient: direction that reduces all active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # d(c_k)/d(a_j) = a_{k-j}
                indices = j_start + np.arange(j_end - j_start + 1)
                partners = k - indices
                grad[indices] += a[partners]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Take step (two-sided: also try the positive direction)
        a_neg = a - step * grad
        a_neg = np.maximum(a_neg, 0.0)
        S_neg = a_neg.sum()
        if S_neg > 1e-12:
            a_neg = a_neg / S_neg
        
        a_pos = a + step * 0.5 * grad  # Smaller positive step
        a_pos = np.maximum(a_pos, 0.0)
        S_pos = a_pos.sum()
        if S_pos > 1e-12:
            a_pos = a_pos / S_pos
        
        sc_neg = compute_score(a_neg) if S_neg > 1e-12 else np.inf
        sc_pos = compute_score(a_pos) if S_pos > 1e-12 else np.inf
        sc_cur = compute_score(a)
        
        best_local = min(sc_neg, sc_pos, sc_cur)
        if best_local == sc_neg:
            a = a_neg
        elif best_local == sc_pos:
            a = a_pos
        # else keep a
        
        step *= step_decay
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check if we're stuck
        if rnd > 10 and abs(best_sc - sc) < 1e-10:
            # Add small noise to escape
            a += np.random.default_rng(rnd).standard_normal(n) * 0.01
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    
    return best_a


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=20000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def log_param_lbfgs(a0, beta, maxiter=5000, maxfun=20000):
    """
    Optimize in log-parameterization: a = exp(x).
    This ensures positivity and can escape zero boundaries.
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-10)
    x0 = np.log(a0)
    
    def log_obj(x):
        a = np.exp(np.clip(x, -30, 30))
        a = a / a.sum()  # normalize
        sc, grad_a = lse_score_grad(a, beta)
        # Chain rule: d(sc)/d(x_i) = d(sc)/d(a_i) * a_i
        grad_x = grad_a * a
        return sc, grad_x
    
    res = minimize(
        log_obj, x0,
        method='L-BFGS-B', jac=True,
        bounds=[(-10, 10)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.exp(np.clip(res.x, -30, 30))
    a_out = a_out / a_out.sum()
    return a_out


def diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Cosine variations
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2, 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(4 * np.pi * x) + 0.3 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(6 * np.pi * x), 0.01))
    
    # Gaussian bump
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.3)
    inits.append(np.maximum(0.1 + 0.9 * np.exp(-((x - center) / width) ** 2), 0.001))
    
    # Two bumps
    c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
    inits.append(np.maximum(0.1 + 0.8 * (np.exp(-((x - c1) / 0.1) ** 2) + np.exp(-((x - c2) / 0.1) ** 2)), 0.001))
    
    # Sparse random
    for _ in range(3):
        a = np.zeros(n)
        n_nonzero = rng.integers(n // 10, n // 3)
        idx = rng.choice(n, n_nonzero, replace=False)
        a[idx] = 1.0
        inits.append(a)
    
    # Random Fourier
    for k in range(4):
        n_harmonics = rng.integers(3, 20)
        freqs = rng.integers(1, n // 2, n_harmonics)
        amps = rng.standard_normal(n_harmonics)
        phases = rng.uniform(0, 2 * np.pi, n_harmonics)
        sig = np.zeros(n)
        for i in range(n_harmonics):
            sig += amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        inits.append(np.maximum(0.5 + 0.3 * sig, 0.01))
    
    # Block structure
    n_blocks = rng.integers(4, 16)
    block_size = n // n_blocks
    block_vals = rng.uniform(0.1, 1.0, n_blocks)
    a_block = np.repeat(block_vals, block_size + (1 if np.sum(np.ones(n_blocks) * block_size < n) else 0))[:n]
    inits.append(a_block)
    
    return inits

"""
Hybrid optimization for minimizing 2n*max(self_conv(a))/(sum(a))^2.
Combines adaptive counter-peak GS/MIQ phase retrieval, multi-scale LSE-LBFGS,
peak shaving with two-sided line search, crossover recombination, and
log-parameterization refinement. Uses diverse initializations including
structured (sinusoidal, sparse) and random starts, with adaptive n selection.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for gradient-based methods."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def gs_counter_peak(a0, n_iters=500, aggression=1.0, floor=0.1):
    """GS with adaptive counter-peak target: attenuate where current conv is high."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = max(float(c.max()), 1e-20)
        c_min = float(c.min())
        # Counter-peak target: low where conv is high
        c_norm = (c - c_min) / (c_max - c_min + 1e-20)
        target = floor + (1.0 - floor) * (1.0 - c_norm ** aggression)
        target = np.maximum(target, 1e-6)

        T = np.fft.fft(target, M)
        T_mag = np.maximum(np.abs(T), 1e-20)
        T_phase = np.angle(T)

        alpha = 0.3 * (1.0 - it / n_iters) + 0.05

        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def gs_flat(a0, n_iters=400, alpha_init=0.4, alpha_final=0.05):
    """Standard GS/Fienup with flat target."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()

    T_flat = np.fft.fft(np.ones(M))
    T_mag = np.maximum(np.abs(T_flat), 1e-20)
    T_phase = np.angle(T_flat)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = alpha_init + (alpha_final - alpha_init) * (it / max(1, n_iters - 1))
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * T_mag ** alpha
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.ifft(A_new, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def miq_retrieval(a0, n_iters=400, gamma=0.5):
    """MIQ (Mixed In-phase Quadrature) phase retrieval."""
    n = len(a0)
    M = 1 << (2 * n - 2).bit_length()
    T_mag = np.abs(np.fft.fft(np.ones(M)))
    T_mag = np.maximum(T_mag, 1e-20)

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)

        # ER: set magnitude to target
        A_ER = A * (T_mag / A_mag)
        # EE: project magnitude^2 to target^2
        A_EE = A * np.sqrt(T_mag / A_mag)

        A_mix = gamma * A_ER + (1 - gamma) * A_EE
        a_new = np.fft.ifft(A_mix, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def peak_shaving(a0, n_rounds=80, n_peaks=30, step_init=0.08, step_decay=0.998, rng=None):
    """Two-sided line search peak shaving with noise escape."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    stall_count = 0

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())

        # Active peaks: within 5% of max
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        active = active[:n_peaks]

        if len(active) == 0:
            break

        # Combined gradient to reduce all active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                partners = k - indices
                grad[indices] += a[partners]

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Two-sided line search
        a_neg = a - step * grad
        a_neg = np.maximum(a_neg, 0.0)
        sneg = a_neg.sum()
        sc_neg = compute_score(a_neg / sneg) if sneg > 1e-12 else np.inf

        a_pos = a + step * 0.3 * grad
        a_pos = np.maximum(a_pos, 0.0)
        spos = a_pos.sum()
        sc_pos = compute_score(a_pos / spos) if spos > 1e-12 else np.inf

        sc_cur = compute_score(a)
        candidates = [sc_neg, sc_cur, sc_pos]
        idx = int(np.argmin(candidates))
        if idx == 0:
            a = a_neg / sneg
        elif idx == 2:
            a = a_pos / spos
        else:
            stall_count += 1

        step *= step_decay

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Escape local minimum
        if stall_count > 20:
            noise_scale = 0.02 * (stall_count / 20.0)
            a = a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            stall_count = 0

    return best_a


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=20000):
    """Run L-BFGS-B with non-negativity bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        # Clip tiny negatives that may arise numerically
        g = np.where(np.abs(g) < 1e-30, 0.0, g)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def log_param_optimize(a0, beta, maxiter=3000, maxfun=10000):
    """Optimize in log-space to ensure positivity and escape zero boundaries."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-8)
    x0 = np.log(a0 / a0.sum())

    def log_obj(x):
        a = np.exp(np.clip(x, -25, 25))
        a = a / a.sum()
        sc, grad_a = lse_score_grad(a, beta)
        grad_x = grad_a * a
        return sc, grad_x

    try:
        res = minimize(
            log_obj, x0,
            method='L-BFGS-B', jac=True,
            bounds=[(-15, 15)] * n,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.exp(np.clip(res.x, -25, 25))
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a0 / a0.sum()


def crossover(a1, a2, rng, blend=0.5):
    """Blend two solutions and add noise."""
    n = len(a1)
    if len(a2) != n:
        x_old = np.linspace(0, 1, len(a2))
        x_new = np.linspace(0, 1, n)
        a2 = np.interp(x_new, x_old, a2)
    a = blend * a1 + (1 - blend) * a2
    a = a + rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return a / s


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2, 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * x), 0.01))

    # Gaussian bump
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.08, 0.25)
    inits.append(np.maximum(0.05 + 0.95 * np.exp(-((x - center) / width) ** 2 * 2), 0.001))

    # Two bumps
    c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
    inits.append(np.maximum(0.05 + 0.9 * (np.exp(-((x - c1) / 0.08) ** 2 * 2) +
                                           np.exp(-((x - c2) / 0.08) ** 2 * 2)), 0.001))

    # Sparse
    for _ in range(2):
        a = np.zeros(n)
        nnz = int(rng.integers(n // 8, n // 4))
        idx = rng.choice(n, nnz, replace=False)
        a[idx] = 1.0
        inits.append(a)

    # Random Fourier (more harmonics)
    for _ in range(3):
        n_harm = int(rng.integers(5, 25))
        freqs = rng.integers(1, n // 3, n_harm)
        amps = rng.standard_normal(n_harm) * 0.3
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        sig = np.zeros(n)
        for i in range(n_harm):
            sig += amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        inits.append(np.maximum(0.5 + sig, 0.01))

    # Structured: Chebyshev-like
    inits.append(np.maximum(0.1 + 0.9 * np.abs(np.cos(np.pi * x * 7)), 0.01))

    return inits

            break
        # Peak shaving
        a_ps = peak_shaving(best_a, n_rounds=40, n_peaks=30, step_init=0.06,
                           step_decay=0.998, rng=rng)
        sc = compute_score(a_ps)
        update_best(sc, a_ps, f"PS-{round}")

        # Counter-peak GS on the result
        if time.monotonic() < p3_end:
            a_cp = gs_counter_peak(best_a, n_iters=200,
                                   aggression=1.0 + 0.3 * round, floor=0.2)
            sc2 = compute_score(a_cp)
            update_best(sc2, a_cp, f"CPS-{round}")

        # Quick LSE
        if time.monotonic() < p3_end:
            beta = 10000 + 10000 * round
            a_l = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=2000, maxfun=6000)
            sc3 = compute_score(a_l)
            update_best(sc3, a_l, f"LSE-{round}-b{beta}")

        print(f"  Round {round}: best={best_score:.8f}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ===== Phase 4: Crossover recombination (13% time) =====
    p4_end = t_start + 0.70 * (total_budget - 60.0)
    print(f"\n[Phase 4] Crossover (deadline +{p4_end - t_start:.0f}s)")

    for c_round in range(6):
        if time.monotonic() > p4_end:
            break
        # Crossover top solutions
        if len(hall_of_fame) >= 2:
            s1, a1 = hall_of_fame[0]
            s2, a2 = hall_of_fame[min(1, len(hall_of_fame) - 1)]
            blend = 0.3 + 0.4 * rng.random()
            a_cross = crossover(a1, a2, rng, blend=blend)
        else:
            a_cross = best_a + rng.standard_normal(n) * 0.02
            a_cross = np.maximum(a_cross, 0.0)
            a_cross = a_cross / a_cross.sum()

        # Optimize the crossover
        sc_cs = gs_counter_peak(a_cross, n_iters=200, aggression=1.0, floor=0.2)
        scx = compute_score(sc_cs)
        update_best(scx, sc_cs, f"X-cross-{c_round}")

        if time.monotonic() < p4_end:
            a_opt = run_lbfgs(best_a, lse_score_grad, (30000,), maxiter=3000, maxfun=8000)
            sc_opt = compute_score(a_opt)
            update_best(sc_opt, a_opt, f"X-lse-{c_round}")

        print(f"  Crossover {c_round}: best={best_score:.8f}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ===== Phase 5: Log-param + high-beta LSE (13% time) =====
    p5_end = t_start + 0.83 * (total_budget - 60.0)
    print(f"\n[Phase 5] Log-param + high-beta (deadline +{p5_end - t_start:.0f}s)")

    # Log-parameterization (ensures strictly positive)
    for beta in [5000, 20000, 80000]:
        if time.monotonic() > p5_end:
            break
        t0 = time.monotonic()
        try:
            a_log = log_param_optimize(best_a, beta, maxiter=2000, maxfun=6000)
            sc = compute_score(a_log)
            update_best(sc, a_log, f"LogLSE-b{beta}")
            print(f"  LogLSE beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        except Exception as e:
            print(f"  LogLSE beta={beta} failed: {e}")

    # Very high beta LSE
    for beta in [200000, 500000, 1000000]:
        if time.monotonic() > p5_end:
            break
        t0 = time.monotonic()
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=3000, maxfun=8000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt, f"HBLSE-b{beta}")
        print(f"  HighBeta LSE beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"Phase 5 done: best={best_score:.8f}")

    # ===== Phase 6: Final interleaved polish (10% time) =====
    print(f"\n[Phase 6] Final polish (deadline +{deadline - t_start:.0f}s)")

    polish_rounds = 6
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # Quick GS counter-peak
        a_gs = gs_counter_peak(best_a, n_iters=100, aggression=0.8, floor=0.3)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"F-gs-{pr}")

        # Peak shaving
        if time.monotonic() > deadline:
            break
        a_ps = peak_shaving(best_a, n_rounds=20, n_peaks=50, step_init=0.03,
                           step_decay=0.999, rng=rng)
        sc2 = compute_score(a_ps)
        update_best(sc2, a_ps, f"F-ps-{pr}")

        # Final LSE with increasing beta
        if time.monotonic() > deadline:
            break
        beta_f = 100000 + 200000 * pr
        a_l = run_lbfgs(best_a, lse_score_grad, (beta_f,), maxiter=2000, maxfun=5000)
        sc3 = compute_score(a_l)
        update_best(sc3, a_l, f"F-lse-{pr}")

        print(f"  Polish {pr}: best={best_score:.8f}")

    # ===== Final verification =====
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
