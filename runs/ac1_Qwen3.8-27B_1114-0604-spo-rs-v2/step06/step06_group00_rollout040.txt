"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-k LSE annealing with L-BFGS, spectral A^2 shaping, adaptive subgradient
with backtracking line search, and multi-n exploration with interpolation.
Key improvements: stable top-k LSE at high beta, frequency-domain peak reduction,
and coordinated multi-phase time budgeting.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def topk_lse_score_and_grad(a, beta, k):
    """
    Top-K LSE smoothed score and gradient using FFT for the gradient.
    Only the top-k entries of conv(a,a) participate in the LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -k)[-k:]

    c_top = c[top_idx]
    c_max = float(c_top.max())
    shifted = beta * (c_top - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta

    p = exp_shifted / exp_sum

    # Build weight vector w of length 2n-1
    w = np.zeros(2 * n - 1)
    w[top_idx] = p

    # Gradient via FFT: dL/da[j] = sum_m w[m]*(a[m-j] + a[j-m])
    # = cross_correlation(w, a) + convolution(w, a), restricted to [0, n-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()

    W = np.fft.rfft(w, M3_fft)
    Aft = np.fft.rfft(a, M3_fft)

    # cross_correlation(w, a)[j] = sum_m w[m]*a[m-j] = ifft(W * conj(A))
    cross = np.fft.irfft(W * np.conj(Aft), M3_fft)
    # convolution(w, a)[j] = sum_m w[m]*a[j-m] = ifft(W * A)
    conv_wa = np.fft.irfft(W * Aft, M3_fft)

    # The cross-correlation result: cross[j] = sum_m w[m]*a[m-j] for j in [0, M3-1]
    # We need this for j in [0, n-1]
    # The convolution result: conv_wa[j] = sum_m w[m]*a[j-m] for j in [0, M3-1]
    # We need this for j in [0, n-1]
    grad_L = cross[:n] + conv_wa[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def full_lse_score_and_grad(a, beta):
    """Full LSE (all 2n-1 entries) score and gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    L = c_max + float(np.log(exp_sum)) / beta

    p = exp_shifted / exp_sum

    # p has length 2n-1, use FFT to compute gradient
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()

    P = np.fft.rfft(p, M3_fft)
    Aft = np.fft.rfft(a, M3_fft)

    cross = np.fft.irfft(P * np.conj(Aft), M3_fft)
    conv_pa = np.fft.irfft(P * Aft, M3_fft)

    grad_L = cross[:n] + conv_pa[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def lse_objective_wrapper(a, beta, k):
    """Wrapper for scipy minimize."""
    if k >= 2 * len(a) - 1:
        return full_lse_score_and_grad(a, beta)
    else:
        return topk_lse_score_and_grad(a, beta, k)


def run_lbfgs(a0, beta, k, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on top-k LSE objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta, k),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        sc = compute_score(a_out)
        return sc, a_out
    except Exception:
        return compute_score(a0), a0


def spectral_a2_shape(a, n_iters=200, strength=0.05):
    """
    Spectral approach: modify A to flatten A^2 (the autocorrelation spectrum).
    This directly targets making conv(a,a) flatter.
    """
    n = len(a)
    a = normalize(a.copy())
    L = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        C = A * A  # Autocorrelation spectrum
        c = np.fft.irfft(C, L)[:2 * n - 1]

        c_mean = float(c.sum()) / (2 * n - 1)
        c_max = float(c.max())
        if c_max < c_mean * 1.01:
            break

        # Find peaks and their frequency "fingerprint"
        excess = np.maximum(c - c_mean * 1.02, 0.0)
        E = np.fft.rfft(excess, L)

        # Reduce C by a fraction of E (to reduce peaks)
        alpha = strength * (1.0 - 0.5 * it / n_iters)
        C_new = C - alpha * E
        C_new = np.maximum(C_new.real, 0.0) + 0j
        # Ensure Hermitian symmetry is maintained (it is by construction since
        # excess is real, so E satisfies conjugate symmetry)

        # Recover A from C_new: A = sqrt(C_new)
        # Choose branch to maintain reality of a
        magnitude = np.sqrt(np.abs(C_new))
        phase = np.angle(C_new) / 2.0
        A_new = magnitude * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_backtrack(a, n_iters=5000, top_k=5, step_init=0.01):
    """
    Subgradient descent on the actual max with backtracking line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-9:
            break

        # Build subgradient from top-k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(2 * n - 1)
        for t in range(min(top_k, len(top_idx))):
            w[int(top_idx[t])] = 1.0 / top_k

        M3 = (2 * n - 1) + n - 1
        M3_fft = 1 << (M3 - 1).bit_length()
        W = np.fft.rfft(w, M3_fft)
        Aft = np.fft.rfft(a, M3_fft)
        cross = np.fft.irfft(W * np.conj(Aft), M3_fft)
        conv_wa = np.fft.irfft(W * Aft, M3_fft)
        sg = cross[:n] + conv_wa[:n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Backtracking line search
        best_trial = None
        best_trial_sc = sc
        for trial_step in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            a_trial = a - trial_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            St = a_trial.sum()
            if St < 1e-12:
                continue
            a_trial = a_trial / St
            sc_trial = compute_score(a_trial)
            if sc_trial < best_trial_sc:
                best_trial_sc = sc_trial
                best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.8
            if step < 1e-12:
                break

    return best_a


def fienup_spectral(a, n_iters=300):
    """
    Fienup-style iterative: flatten the power spectrum |A|^2 and also
    target A^2 flattening simultaneously.
    """
    n = len(a)
    a = normalize(a.copy())
    L = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, L)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())

        # Blend: flatten |A|^2 (standard Fienup)
        alpha = 0.1 + 0.6 * progress
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)

        # Also target: make |A^2| flatter
        C = A * A
        C_abs = np.abs(C)
        C_mean = float(C_abs.mean())
        beta_flat = 0.05 + 0.3 * progress
        C_abs_new = (1.0 - beta_flat) * C_abs + beta_flat * C_mean

        # Update: scale |A| from P_new, adjust phase of A^2 toward flat
        ratio_mag = np.sqrt(P_new / np.maximum(P, 1e-30))
        # For A^2 flattening: we want C_new = C * (C_abs_new / C_abs)^(some_fraction)
        # This changes both magnitude and phase of C, hence of A
        C_ratio = np.power(np.maximum(C_abs_new, 1e-30) / np.maximum(C_abs, 1e-30), 0.3)
        C_target = C * C_ratio
        # Recover A from C_target
        A_new_mag = np.sqrt(np.abs(C_target))
        A_new_phase = np.angle(C_target) / 2.0
        A_new = A_new_mag * np.exp(1j * A_new_phase)
        # Also apply the |A| flattening
        A_new = A_new * ratio_mag

        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def coord_descent_focused(a, n_sweeps=20, n_top=20):
    """
    Coordinate descent: reduce top contributors to the peak with line search.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue

            best_val = a_old
            best_local = best_sc
            for f in [0.2, 0.4, 0.6, 0.75, 0.85, 0.95]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def lse_anneal_schedule(a0, betas, ks, maxiter_per_beta=4000):
    """
    Anneal through beta values with corresponding top-k values.
    """
    n = len(a0)
    a = normalize(a0.copy())

    best_a = a.copy()

    for i, (beta, k) in enumerate(zip(betas, ks)):

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = normalize(a_new)

    return best_sc, best_a

        sc_p = compute_score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        best_a = np.ones(8192) / 8192.0
        update(compute_score(best_a), best_a, "uniform-init")
        print("No parent found. Starting uniform n=8192.")

    if best_a is None:
        best_a = np.ones(8192) / 8192.0

    print(f"Initial best: {best_score:.8f}")
    n_cur = len(best_a)

    # Phase 1: Spectral shaping (8% time)
    phase1_end = t_start + 0.08 * (budget_s - 20.0)
    print(f"\n=== Phase 1: Spectral Shaping (budget {phase1_end - t_start:.0f}s) ===")

    a_sp1 = spectral_a2_shape(best_a, n_iters=200, strength=0.03)
    update(compute_score(a_sp1), a_sp1, "SPECTRAL-A2")

    a_sp2 = fienup_spectral(best_a, n_iters=200)
    update(compute_score(a_sp2), a_sp2, "FIENUP-SPEC")

    # Phase 2: LSE Annealing (40% time)
    phase2_end = t_start + 0.48 * (budget_s - 20.0)
    print(f"\n=== Phase 2: LSE Annealing (budget {phase2_end - t_start:.0f}s) ===")

    a_work = best_a.copy()

    # Schedule: (beta, k) pairs - start coarse, end fine
    schedule = [
        (30, 50),
        (80, 30),
        (200, 20),
        (500, 10),
        (1000, 10),
        (2500, 8),
        (5000, 5),
        (10000, 5),
        (20000, 3),
        (40000, 3),
        (80000, 2),
        (150000, 2),
        (300000, 1),
    ]

    # Allocate time proportionally, more to mid-high range
    weights = [1, 1.5, 2, 2.5, 2.5, 3, 3, 3, 3, 3, 2.5, 2, 1.5]
    total_weight = sum(weights)

    for i, ((beta, k), w) in enumerate(zip(schedule, weights)):
        if time.monotonic() > phase2_end - 15:
            break
        allocated = (phase2_end - 15 - t_start) * (w / total_weight)
        remaining = min(allocated, phase2_end - 15 - time.monotonic())
        if remaining < 10:
            break
        maxiter = max(500, int(remaining * 15))
        print(f"  LSE beta={beta:>6}, k={k:>3}, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_work, beta, k, maxiter=maxiter, maxfun=maxiter * 20)
        dt = time.monotonic() - t0
        print(f"    score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE b={beta},k={k}")
        a_work = a_opt.copy()

    # Phase 3: Subgradient with backtracking (15% time)
    phase3_end = t_start + 0.63 * (budget_s - 20.0)
    print(f"\n=== Phase 3: Subgradient Backtracking (budget {phase3_end - t_start:.0f}s) ===")

    for top_k in [3, 5, 8]:
        if time.monotonic() > phase3_end - 15:
            break
        remaining_time = phase3_end - 15 - time.monotonic()
        if remaining_time < 10:
            break
        n_iters = int(remaining_time * 800)
        n_iters = max(500, min(n_iters, 5000))
        a_sg = subgradient_backtrack(best_a, n_iters=n_iters, top_k=top_k, step_init=0.005)
        sc = compute_score(a_sg)
        update(sc, a_sg, f"SUBGRAD k={top_k}")

        # Follow up with LSE
        if sc < best_score and time.monotonic() < phase3_end - 15:
            sc2, a2 = run_lbfgs(a_sg, 20000, 2, maxiter=2000, maxfun=40000)
            update(sc2, a2, f"SUBGRAD+LSE k={top_k}")

    # Phase 4: Coordinate descent (5% time)
    phase4_end = t_start + 0.68 * (budget_s - 20.0)
    print(f"\n=== Phase 4: Coordinate Descent ===")

    a_cd = coord_descent_focused(best_a, n_sweeps=15, n_top=15)
    sc = compute_score(a_cd)
    update(sc, a_cd, "COORD")

    if time.monotonic() < phase4_end - 10:
        sc2, a2 = run_lbfgs(a_cd, 30000, 2, maxiter=2000, maxfun=40000)
        update(sc2, a2, "COORD+LSE")

    # Phase 5: Multi-n exploration (15% time)
    phase5_end = t_start + 0.83 * (budget_s - 20.0)
    print(f"\n=== Phase 5: Multi-n Exploration ===")

    current_n = len(best_a)
    target_ns = []
    if current_n < 12000:
        target_ns.append(12000)
    if current_n < 16000:
        target_ns.append(16000)
    if current_n < 24000:
        target_ns.append(24000)

    for new_n in target_ns:
        if time.monotonic() > phase5_end - 20:
            break
        remaining = phase5_end - 20 - time.monotonic()
        if remaining < 20:
            break

        a_large = interpolate_sequence(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  Interpolated to n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"INTERP n={new_n}")

        # Optimize the larger sequence
        if time.monotonic() < phase5_end - 15:
            alloc = (phase5_end - 15 - time.monotonic())
            if alloc > 20:
                # Quick LSE on larger sequence
                maxiter = max(500, int(alloc * 8))
                sc2, a2 = run_lbfgs(a_large, 5000, 3, maxiter=maxiter, maxfun=maxiter * 15)
                update(sc2, a2, f"INTERP+LSE n={new_n}")

                # Spectral on larger
                if time.monotonic() < phase5_end - 10:
                    a_sp = spectral_a2_shape(a2, n_iters=100, strength=0.03)
                    sc3 = compute_score(a_sp)
                    update(sc3, a_sp, f"INTERP+SPEC n={new_n}")

    # Phase 6: Final polish (17% time)
    print(f"\n=== Phase 6: Final Polish ===")

    # Very high beta LSE
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        maxiter = max(1000, int(remaining * 10))
        sc, a_f = run_lbfgs(best_a, 100000, 1, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_f, "FINAL-LSE-100k")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = max(500, int(remaining * 8))
        sc, a_f = run_lbfgs(best_a, 200000, 1, maxiter=maxiter, maxfun=maxiter * 12)
        update(sc, a_f, "FINAL-LSE-200k")

    # Final subgradient with tiny steps
    if time.monotonic() < deadline - 15:
        remaining_time = deadline - 15 - time.monotonic()
        n_iters = max(300, int(remaining_time * 1000))
        a_sg = subgradient_backtrack(best_a, n_iters=n_iters, top_k=3, step_init=0.001)
        sc = compute_score(a_sg)
        update(sc, a_sg, "FINAL-SUBGRAD")

    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd = coord_descent_focused(best_a, n_sweeps=5, n_top=10)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-COORD")

    # One more LSE pass if time allows
    if time.monotonic() < deadline - 3:
        sc, a_f = run_lbfgs(best_a, 50000, 2, maxiter=1500, maxfun=20000)
        update(sc, a_f, "LAST-LSE")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
