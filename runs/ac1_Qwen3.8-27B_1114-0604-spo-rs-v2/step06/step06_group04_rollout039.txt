"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Spectral flattening (Gerchberg-Saxton) for global optimization
2. LP-based alternating minimization with constraint generation
3. Nesterov accelerated subgradient for local polishing
4. Multi-resolution and multi-start strategy
Key insight: optimal sequences have nearly flat power spectra and asymmetric structure.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _fast_cross_conv(a, b):
    n = a.size
    m = b.size
    M = n + m - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _spectral_flatten(a, n_iters=50, target_type='geom_mean', deadline=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Drives |FFT(a)| towards a flat target while maintaining a >= 0.
    """
    a = _normalize(a)
    n = a.size
    N = _next_pow2(2 * n)
    best_a = a.copy()
    best_score = _exact_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = A / np.where(mag > 1e-15, mag, 1.0)

        if target_type == 'geom_mean':
            log_mag = np.log(np.maximum(mag, 1e-15))
            target = np.exp(np.mean(log_mag))
            target = np.maximum(target, 0.05 * mag.max())
        elif target_type == 'soft_flat':
            mag_sorted = np.sort(mag)
            low = mag_sorted[int(0.1 * len(mag_sorted))]
            high = mag_sorted[int(0.9 * len(mag_sorted))]
            target = np.ones_like(mag) * np.sqrt(low * high)
            target = np.maximum(target, 0.1 * mag)
        else:
            target = np.ones_like(mag) * mag.mean()
            target = np.maximum(target, 0.1 * mag)

        # Blend old and new magnitude to avoid oscillation
        alpha = 0.5 + 0.3 * np.sin(np.pi * it / max(1, n_iters))
        new_mag = alpha * target + (1 - alpha) * mag
        # Preserve DC component
        new_mag[0] = mag[0]

        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = float(a_new.sum())
        if S < 1e-12:
            break
        a_new = a_new / S

        sc = _exact_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_score, best_a


def _subgrad_exact(a, k_active=3, tol=1e-6):
    """Compute exact subgradient of 2*n*max(c)/S^2 at point a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-6):
    """Nesterov accelerated subgradient method."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)

    t = 1.0
    y = a.copy()
    v = np.zeros(n)

    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a

    eta = 0.05 / g_norm
    no_improve_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v

        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy

        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break

        step = eta / (1.0 + 0.05 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx

        v = x_new - a
        a = x_new
        t = t_new

        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 100:
            eta *= 0.5
            no_improve_count = 0
            if eta < 1e-8:
                break

    return best_score, best_a


def _build_toeplitz_constraints(f, n, constraint_indices):
    """Build sparse constraint matrix for LP: (f*g)[k] for k in constraint_indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = list(range(j_start, j_end + 1))
            rows.extend([idx] * len(js))
            cols.extend(js)
            vals.extend(f[k - j] for j in js)
    A = csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))
    return A


def _lp_alternating_step(f, M, max_constraints=3000, max_iters=30, deadline=None):
    """
    LP step: fix f, find g >= 0 maximizing sum(g) 
    subject to (f*g)[k] <= M for all k.
    Uses constraint generation.
    """
    n = len(f)
    if n < 16:
        return None

    # Initial active set: sample constraints
    all_k = np.arange(2 * n - 1)
    if len(all_k) <= max_constraints:
        active = list(all_k)
    else:
        # Sample: edges, center, and uniform
        n_edge = max(10, n // 100)
        active = list(range(n_edge)) + list(range(2 * n - 1 - n_edge, 2 * n - 1))
        active.append(n - 1)
        # Add uniform samples
        uniform = np.linspace(0, 2 * n - 2, max_constraints // 2).astype(int)
        active.extend(uniform.tolist())
        active = list(set(active))

    g = np.ones(n) / n  # initial guess

    for iter_i in range(max_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 3.0:
            break

        # Build constraint matrix
        A = _build_toeplitz_constraints(f, n, active)
        b = np.full(len(active), M)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        Sg = float(g.sum())
        if Sg < 1e-10:
            break

        # Check violations using FFT
        conv = _fast_cross_conv(f, g)
        violations = conv - M

        max_viol = float(violations.max())
        if max_viol < 1e-6 * M:
            break  # Feasible

        # Add violated constraints to active set
        if len(active) < max_constraints:
            violated = np.where(violations > 0.005 * max(1e-10, max_viol))[0]
            if len(violated) > 0:
                active.extend(violated.tolist())
                active = list(set(active))
                if len(active) > max_constraints:
                    # Keep most violated
                    active_arr = np.array(active)
                    viol_arr = violations[active_arr]
                    keep = np.argsort(viol_arr)[-max_constraints:]
                    active = active_arr[keep].tolist()
        else:
            # Replace least violated with most violated
            active_arr = np.array(active)
            viol_arr = violations[active_arr]
            new_violated = np.where(violations > np.max(viol_arr))[0]
            if len(new_violated) > 0:
                to_remove = np.argsort(viol_arr)[:len(new_violated)]
                to_remove_set = set(active_arr[to_remove].tolist())
                active = [k for k in active if k not in to_remove_set]
                active.extend(new_violated.tolist())

    return g


def _nesterov_lbfgs_hybrid(a0, deadline, label=""):
    """Combined spectral + Nesterov optimization."""
    a = _normalize(a0)
    best_a = a.copy()
    best_score = _exact_score(a)
    t0 = time.monotonic()
    total_time = max(10.0, float(deadline) - t0 - 5.0)

    print("[{}] start: {:.8f} n={}".format(label, best_score, a.size))

    # Stage 1: Spectral flattening (20% of time)
    s1_dead = t0 + 0.20 * total_time
    if time.monotonic() < s1_dead - 5:
        sc, a2 = _spectral_flatten(a, n_iters=200, target_type='geom_mean', deadline=s1_dead)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] spectral: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    # Stage 2: More spectral with different target (20%)
    s2_dead = t0 + 0.40 * total_time
    if time.monotonic() < s2_dead - 5:
        sc, a2 = _spectral_flatten(a, n_iters=200, target_type='soft_flat', deadline=s2_dead)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("[{}] spectral2: {:.8f}".format(label, best_score))
        a = _normalize(a2)

    # Stage 3: Nesterov subgradient (35%)
    s3_dead = t0 + 0.75 * total_time
    if time.monotonic() < s3_dead - 5:
        for k_act in [3, 2, 1]:
            rem = s3_dead - time.monotonic()
            if rem < 10:
                break
            n_it = int(max(100, rem * 25 / (3 - k_act + 1)))
            sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=s3_dead,
                                        k_active=k_act, tol=1e-7)
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] nesterov k={}: {:.8f}".format(label, k_act, best_score))
            a = _normalize(a2)

    # Stage 4: LP-based alternating (20%)
    s4_dead = float(deadline) - 2.0
    if time.monotonic() < s4_dead - 10:
        a_work = best_a.copy()
        for lp_iter in range(10):
            rem = s4_dead - time.monotonic()
            if rem < 15:
                break
            c = _fast_self_conv(a_work)
            M = float(c.max())
            g = _lp_alternating_step(a_work, M, max_constraints=2000,
                                     max_iters=20, deadline=s4_dead - 5)
            if g is None:
                break
            Sg = float(g.sum())
            if Sg < 1e-10:
                break
            g = g / Sg
            # Blend
            t_blend = 0.3
            a_new = (1 - t_blend) * a_work + t_blend * g
            a_new = _normalize(a_new)
            sc = _exact_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print("[{}] LP iter {}: {:.8f}".format(label, lp_iter, best_score))
            a_work = a_new

    return best_score, best_a


def _generate_inits(n, rng, parent, cap=25):
    """Generate diverse initial sequences with emphasis on asymmetric structures."""
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Asymmetric ramps (key insight: asymmetry helps reduce center peak)
    for p in [1.5, 2.0, 3.0, 4.0, 6.0, 8.0]:
        _add(x ** p + 0.01)
        _add((1 - x) ** p + 0.01)

    # Asymmetric exponential
    for rate in [1.0, 2.0, 4.0, 6.0, 10.0]:
        _add(np.exp(-rate * x) + 0.01)
        _add(np.exp(-rate * (1 - x)) + 0.01)
        # Double exp (asymmetric)
        _add(np.exp(-rate * x) + 0.3 * np.exp(-rate * 2 * (1 - x)) + 0.01)

    # Bump (Gaussian) - various widths and positions
    for w in [0.1, 0.15, 0.2, 0.3, 0.4]:
        for ctr in [0.3, 0.4, 0.5, 0.6, 0.7]:
            _add(np.exp(-(((x - ctr) / w) ** 2)) + 0.01)

    # Bimodal asymmetric
    for w1 in [0.08, 0.12, 0.15]:
        for w2 in [0.08, 0.12, 0.15]:
            for c1 in [0.25, 0.35]:
                for c2 in [0.65, 0.75]:
                    _add(np.exp(-(((x - c1) / w1) ** 2)) +
                         np.exp(-(((x - c2) / w2) ** 2)) + 0.01)

    # Sine-based
    for f in [1, 2, 3, 4, 6]:
        _add(1.0 + 0.8 * np.sin(np.pi * f * x) + 1.0)
        _add(np.abs(np.sin(np.pi * f * x)) + 0.1)

    # Power law
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        _add(x ** p * (1 - x) ** p + 0.01)
        _add(x ** (2 * p) * (1 - x) ** p + 0.01)

    # Random smooth (spectral)
    Nf = _next_pow2(n)
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.3, 1.5)) * noise + 0.5)

    # Pure random (a few)
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))

    return inits


def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.4))
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = a * np.exp(strength * 0.4 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.3 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        K = min(51, max(7, n // 100))
        if K % 2 == 0:
            K -= 1
        c = K // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 4:
        # Asymmetric shift
        shift = int(rng.integers(1, max(2, n // 10)))
        q = np.roll(a, shift)
        q[:shift] *= 0.5
        q = np.clip(q, 0, None)
    elif mode == 5:
        # Mirror blend
        t = float(rng.uniform(0.3, 0.7))
        q = t * a + (1 - t) * a[::-1]
    else:
        # Frequency domain perturbation
        N = _next_pow2(2 * n)
        A = np.fft.rfft(a, N)
        delta = rng.standard_normal(len(A)) * 0.05
        A = A * (1 + delta)
        q = np.fft.irfft(A, N)[:n]

    return _normalize(np.clip(q, 0.0, None))


def _quick_optimize(a, deadline):
    """Quick multi-phase optimization for a candidate."""
    a = _normalize(a)
    best_a = a.copy()
    best_score = _exact_score(a)

    # Phase 1: Spectral
    rem = deadline - time.monotonic()
    if rem > 10:
        sc, a2 = _spectral_flatten(a, n_iters=min(100, int(rem * 10)), deadline=deadline - 5)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    # Phase 2: Nesterov
    rem = deadline - time.monotonic()
    if rem > 8:
        n_it = int(max(50, rem * 20))
        sc, a2 = _nesterov_optimize(a, n_iters=n_it, deadline=deadline,
                                    k_active=3, tol=1e-7)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
        a = _normalize(a2)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns list of non-negative floats."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        parent_len = min(parent.size, 16384)
        if parent_len < 4096:
            parent_len = 8192
        lengths = [parent_len]
        for L in [8192, 10240, 12288, 16384, 6144]:
            if L not in lengths:
                lengths.append(L)
    else:
        lengths = [10240, 8192, 12288]

    lengths = sorted(set(l for l in lengths if 4096 <= l <= 16384))[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Evaluate diverse initializations
    all_scores = []
    all_seqs = []

    for n in lengths[:2]:  # Focus on fewer lengths for deeper search
        inits = _generate_inits(n, rng, parent, cap=30)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)

    order = np.argsort(all_scores)
    print("Best init: {:.8f} (len={})".format(
        all_scores[order[0]], all_seqs[order[0]].size))
    for i in range(min(5, len(order))):
        print("  init {}: {:.8f} (len={})".format(i, all_scores[order[i]], all_seqs[order[i]].size))

    # Select top 3 starts from different lengths
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 3:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)

    while len(starts) < 2 and len(order) > len(starts):
        idx = order[len(starts)]
        if all_seqs[idx].size not in used_lens or len(starts) < 2:
            starts.append(all_seqs[idx].copy())

    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f}".format(best_score))

    # Phase 2: Main optimization on selected starts
    starts_sorted = sorted(range(len(starts)), key=lambda i: _exact_score(starts[i]))

    if len(starts_sorted) == 1:
        fracs = [1.0]
    elif len(starts_sorted) == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.50, 0.25, 0.25]

    main_end = deadline - 300.0
    main_budget = max(100.0, main_end - t_start)
    cum = 0.0

    for rank, si in enumerate(starts_sorted):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, main_end)

        if time.monotonic() > sub_dead - 15.0:
            continue

        sq = starts[si].copy()

        if rank > 0 and len(best_seq) != len(sq):
            best_resampled = _resample(best_seq, len(sq))
            t_blend = 0.3 + 0.2 * rank
            sq = _normalize((1.0 - t_blend) * _normalize(sq) + t_blend * best_resampled)

        sc, seq = _nesterov_lbfgs_hybrid(sq, sub_dead, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: Perturbation restarts
    perturb_start = time.monotonic()
    perturb_end = deadline - 150.0
    if perturb_end > perturb_start + 40.0:
        n_perturb = 4
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 5.0:
                break

            p = _perturb(best_seq, rng)
            sc, seq = _quick_optimize(p, chunk_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 4: Final intensive optimization
    final_dead = deadline - 10.0
    if time.monotonic() < final_dead - 20.0:
        # More spectral iterations
        rem = final_dead - time.monotonic()
        if rem > 15.0:
            sc, a2 = _spectral_flatten(best_seq, n_iters=200,
                                       target_type='geom_mean', deadline=final_dead - 5)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final spectral: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

        # LP-based final polishing
        rem = final_dead - time.monotonic()
        if rem > 20.0:
            a_work = best_seq.copy()
            for lp_iter in range(5):
                rem = final_dead - time.monotonic()
                if rem < 15:
                    break
                c = _fast_self_conv(a_work)
                M = float(c.max())
                g = _lp_alternating_step(a_work, M, max_constraints=1500,
                                         max_iters=15, deadline=final_dead - 5)
                if g is None:
                    break
                Sg = float(g.sum())
                if Sg < 1e-10:
                    break
                g = g / Sg
                a_new = 0.7 * a_work + 0.3 * g
                a_new = _normalize(a_new)
                sc = _exact_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new.copy()
                    print("Final LP {}: {:.8f}".format(lp_iter, best_score))
                a_work = a_new

        # Final Nesterov with tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            n_it = int(max(100, rem * 30))
            sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=2, tol=1e-9)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

            rem = final_dead - time.monotonic()
            if rem > 8.0:
                n_it = int(max(50, rem * 20))
                sc, a2 = _nesterov_optimize(best_seq, n_iters=n_it, deadline=final_dead,
                                            k_active=1, tol=1e-10)
                if sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final Nesterov k=1: {:.8f}".format(best_score))
                best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
