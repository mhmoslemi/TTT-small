    #   Since m is always in [0,n-1] (it's an index of a), the condition is just 0 <= k-m <= n-1.

    #   = sum_k 2*a[k-m]*p[k] (for 0 <= k-m <= n-1)
    #   = 2 * sum_{j=0}^{n-1} a[j] * p[j+m]  (substituting j = k-m)
    #   = 2 * (a reversed cross-correlated with p)

    # sum_j a[j] * p[j+m] = (a * p_flipped)[m + n - 1]... 
    # Or: = correlation of a and p at lag m = sum_j a[j+m] * p[j]

    ga_dot_a = float(np.dot(grad_a, a))
    grad_b = a * (grad_a - ga_dot_a)

"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovation: softmax-parametrized unconstrained L-BFGS optimization with
progressive LSE smoothing, combined with spectral error reduction for
initialization and peak-targeted subgradient descent for fine-tuning.
The softmax reparameterization removes box constraints, enabling more
efficient L-BFGS convergence.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def softmax_safe(x):
    """Numerically stable softmax."""
    x = x - x.max()
    # Clip to avoid overflow in edge cases
    x = np.clip(x, -700, 700)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s


def lse_obj_and_grad_b(b, beta):
    """
    LSE-smoothed objective and gradient w.r.t. b where a = softmax(b).
    Since sum(a)=1, score = 2n * LSE_beta(conv(a,a)).
    """
    n = len(b)
    a = softmax_safe(b)
    c = self_conv(a)
    c_max = float(c.max())
    n_c = len(c)
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        lse_val = c_max + n_c / beta
        p = np.ones(n_c) / n_c
    else:
        lse_val = c_max + (z.max() - z_shift.max() + np.log(exp_sum)) / beta
        p = exp_z / exp_sum
    score = 2.0 * n * lse_val

    # Gradient of LSE w.r.t. a: 2 * (B^T p)
    # (B^T p)[m] = 2 * sum_{j=0}^{n-1} a[j] * p[j+m]
    # Computed via FFT: conv(a[::-1], p)[m + n - 1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    FA_rev = np.fft.rfft(a[::-1], M2_fft)
    FP = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(FA_rev * FP, M2_fft)
    grad_a = 2.0 * corr[n - 1: 2 * n - 1]

    # Chain rule through softmax: grad_b = a * (grad_a - dot(grad_a, a))
    ga_da = float(np.dot(grad_a, a))
    grad_b = a * (grad_a - ga_da)

    return score, grad_b


def lse_obj_and_grad_direct(a, beta):
    """
    LSE-smoothed objective and gradient w.r.t. a (bounded version).
    Score = 2n * LSE_beta(conv(a,a)) / (sum(a))^2.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c = self_conv(a)
    c_max = float(c.max())
    n_c = len(c)
    z = beta * (c - c_max)
    z_shift = z - z.max()

        p = np.ones(n_c) / n_c
    else:
        lse_val = c_max + (z.max() - z_shift.max() + np.log(exp_sum)) / beta
        p = exp_z / exp_sum
    S2 = S * S
    score = 2.0 * n * lse_val / S2

    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    corr = np.fft.irfft(FA_rev * FP, M2_fft)
    grad_L = 2.0 * corr[n - 1: 2 * n - 1]

    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * lse_val / S3)

    return score, grad


def spectral_error_reduction(a, n_iters=80, alpha=0.6, damp=0.5):
    """
    Frequency-domain error reduction to flatten the convolution.
    Iteratively reduces peaks in conv(a,a) by modifying the spectrum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]

        c_mean = float(c.mean())
        c_max = float(c.max())
        if c_max - c_mean < 1e-16:
            break

        # Target: reduce values above threshold
        thresh = c_mean + (1.0 - alpha) * (c_max - c_mean)
        excess = np.maximum(c - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        # Modify spectrum to reduce excess
        c_mod = c.copy()
        c_mod[:2 * n - 1] -= damp * excess
        c_mod_pad = np.zeros(M_fft, dtype=np.float64)
        c_mod_pad[:2 * n - 1] = c_mod
        E = np.fft.rfft(c_mod_pad)
        E_orig = np.fft.rfft(np.concatenate([c, np.zeros(M_fft - len(c))]))

        delta_C = E - E_orig
        with np.errstate(divide='ignore', invalid='ignore'):
            delta_A = delta_C / (2.0 * A)
        delta_A[np.abs(A) < 1e-10] = 0.0

        A_new = A + delta_A
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def run_lbfgs_softmax(a_init, beta, maxiter=8000, maxfun=500000, gtol=1e-14):
    """Run L-BFGS-B on the softmax-parametrized LSE objective."""
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 1e-15)
    a_init = a_init / a_init.sum()
    b0 = np.log(a_init)
    b0 -= b0.mean()  # Center to reduce flat direction

    def obj_and_grad(b):
        return lse_obj_and_grad_b(b, beta)

    res = minimize(
        obj_and_grad, b0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 80}
    )
    a_out = softmax_safe(res.x)
    a_out = np.maximum(a_out, 0.0)
    S = float(a_out.sum())
    if S > 0:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def peak_subgradient_descent(a, n_steps=500, rng=None):
    """
    Targeted subgradient descent reducing the convolution peak.
    Uses the subgradient at the argmax of the convolution.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        # Subgradient of max(conv) at k_star:
        # d(conv[k_star])/da[j] = a[k_star - j] for valid j
        direction = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            direction[idx] = a[k_star - idx]
        direction *= 2.0

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size with backtracking
        base_step = min(0.02, max(0.0005, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.5))
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 2.0, 4.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 20 == 0:
            # Perturbation to escape
            scale = rng.uniform(0.01, 0.05)
            a = a + scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def lp_alternating_fista(a, M, n_iters=10000, beta=200.0, x_init=None):
    """
    FISTA solver for: min (1/beta)*log(sum exp(beta*(conv(a,x) - M))) s.t. x >= 0
    Returns x (unnormalized).
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    L = beta * 2.0 * n * sum_a2 + 1e-12
    step = 1.0 / L

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        # Gradient: (B^T p)[j] = sum_i a[i] * p[i+j]
        FA = np.fft.rfft(a, M_fft)
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=20, fista_iters=10000, time_limit=None,
                   line_search=True, beta=200.0):
    """LP-based alternating method with FISTA inner loop."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())
        x = lp_alternating_fista(a, M, n_iters=fista_iters, beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        if line_search:
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            best_t = 0.3
            best_t_sc = best_sc
            for t in t_candidates:
                a_try = (1.0 - t) * a + t * x
                a_try = np.maximum(a_try, 0.0)
                St = float(a_try.sum())
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
            a_new = a + best_t * (x - a)
        else:
            a_new = x

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 8:
                break

        a = a_new

    return best_a, best_sc, x_warm


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x + 0.5)
                + 0.1 * np.cos(8.0 * np.pi * x + 1.0))
    inits.append(np.clip(0.3 + 0.3 * x + 0.2 * np.cos(6.0 * np.pi * x), 0.01, 1000.0))
    # Additional diverse inits
    inits.append(np.clip(0.5 + 0.2 * np.sin(4.0 * np.pi * x) + 0.15 * np.cos(8.0 * np.pi * x + 0.3), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * np.exp(-2.0 * (x - 0.5) ** 2) + 0.2 * np.cos(2.0 * np.pi * 2 * x), 0.01, 1000.0))
    return inits


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: spectral error reduction init, softmax-LSE-LBFGS (new),
    LP alternating, peak subgradient, final polish.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [6962, 8192, 10240]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Initialization with spectral error reduction (12% time)
    # ==================================================================
    phase1_deadline = t_start + 0.12 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Spectral Error Reduction Init ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 20:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")
        inits = build_initializations(n, rng, parent)
        max_inits = max(3, int(remaining / 8.0))
        inits = inits[:min(max_inits, len(inits))]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 5:
                break
            a_er = spectral_error_reduction(a0, n_iters=60, alpha=0.6, damp=0.5)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    # Also try LP alternating briefly for init
    if time.monotonic() < phase1_deadline - 30:
        a_init_in = best_a.copy()
        S = float(a_init_in.sum())
        if S > 0:
            a_init_in = a_init_in / S
        a_lp, sc_lp, _ = lp_alternating(
            a_init_in, max_outer_iters=5, fista_iters=5000,
            time_limit=phase1_deadline - 10, line_search=True, beta=200.0
        )
        report(sc_lp, a_lp, "LP init")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE-LBFGS with softmax parametrization (45% time)
    # ==================================================================
    phase2_deadline = t_start + 0.57 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LSE-LBFGS (softmax parametrized) ---")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 40:
            break
        print(f"  Softmax L-BFGS beta={beta}, remaining={remaining:.0f}s...",
              end=" ", flush=True)
        t0 = time.monotonic()
        maxit = min(10000, int(remaining * 3))
        sc, a_opt, res = run_lbfgs_softmax(a_work, beta, maxiter=maxit,
                                            maxfun=30000, gtol=1e-14)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"SM-LBFGS b={beta}")
        a_work = a_opt.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    # Also run direct (bounded) L-BFGS at a few betas
    if time.monotonic() < phase2_deadline - 60:
        print(f"  Direct L-BFGS (bounded)...")
        for beta in [200, 500, 1000]:
            if time.monotonic() > phase2_deadline - 30:
                break
            a_d = best_a.copy()
            S = float(a_d.sum())
            if S > 0:
                a_d = a_d / S
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_direct(a_d, beta, maxiter=5000,
                                             maxfun=20000, gtol=1e-14)
            elapsed = time.monotonic() - t0
            print(f"    beta={beta}: score={sc:.8f}, time={elapsed:.1f}s")
            report(sc, a_opt, f"DIR-LBFGS b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: LP Alternating refinement (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.77 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: LP Alternating ---")

    a_lp_in = best_a.copy()
    S = float(a_lp_in.sum())
    if S > 0:
        a_lp_in = a_lp_in / S

    x_warm = None
    for fi, (fista_it, beta_val) in enumerate([(15000, 300), (20000, 500), (25000, 700)]):
        if time.monotonic() > phase3_deadline - 30:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP round {fi}: fista={fista_it}, beta={beta_val}, remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating(
            a_lp_in, max_outer_iters=15, fista_iters=fista_it,
            time_limit=phase3_deadline - 20, line_search=True, beta=beta_val
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_lp_in = a_out.copy()

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak subgradient + final polish (18% time)
    # ==================================================================
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Peak Descent & Final Polish ---")

    # Peak subgradient descent
    a_peak = peak_subgradient_descent(best_a, n_steps=400, rng=rng)
    sc = compute_score(a_peak)
    report(sc, a_peak, "peak descent")

    # Stochastic escape
    n = len(best_a)
    best_esc = best_a.copy()
    best_esc_sc = best_score
    for _ in range(20):
        if time.monotonic() > phase4_deadline - 60:
            break
        scale = rng.uniform(0.02, 0.15)
        a_try = best_a + scale * rng.standard_normal(n)
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            continue
        a_try = a_try / St
        sc_try = compute_score(a_try)
        if sc_try < best_esc_sc:
            best_esc_sc = sc_try
            best_esc = a_try.copy()
            best_a = a_try.copy()
            best_score = sc_try
            print(f"  Escape improved: {sc_try:.8f}")
            break

    # Final softmax L-BFGS polish at high beta
    if time.monotonic() < phase4_deadline - 50:
        print("  Final softmax L-BFGS polish (beta=10000)...")
        a_fin = best_a.copy()
        S = float(a_fin.sum())
        if S > 0:
            a_fin = a_fin / S
        sc, a_opt, _ = run_lbfgs_softmax(a_fin, 10000, maxiter=8000,
                                          maxfun=30000, gtol=1e-15)
        report(sc, a_opt, "final SM-LBFGS")

    # Final peak descent
    if time.monotonic() < phase4_deadline - 20:
        a_final = peak_subgradient_descent(best_a, n_steps=200, rng=rng)
        sc = compute_score(a_final)
        report(sc, a_final, "final peak")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
