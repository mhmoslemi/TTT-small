```python
"""
Multi-strategy optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses progressive LSE smoothing with Nesterov acceleration, multi-scale refinement,
diverse initializations including parabolic tapers, and LP-alternating polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_objective_and_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad
```

```python
def nesterov_lbfgs(a0, beta, maxiter=5000, time_limit=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_objective_and_grad(x, beta)
    
    t0 = time.monotonic()
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': 200000, 'maxls': 50}
    )
    elapsed = time.monotonic() - t0
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, elapsed
```

"""
Multi-scale progressive LSE optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses coarse-to-fine optimization, diverse 
initializations (parabolic tapers, raised cosines), progressive LSE smoothing
with L-BFGS, LP-alternating polishing, and adaptive peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    n_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    return np.fft.irfft(A * B, n_fft)[:n]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_max = float(z.max())
    exp_z = np.exp(z - z_max)
    exp_sum = float(exp_z.sum())
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_z / exp_sum
    log_sum = z_max + np.log(exp_sum)
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    # Gradient via cross-correlation
    n_fft = 1 << (2 * n - 2).bit_length()
    RA = np.fft.rfft(a, n_fft)
    P = np.fft.rfft(p, n_fft)
    corr = np.fft.irfft(np.conj(RA) * P, n_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, gtol=1e-12):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def error_reduction(a, n_iters=80, alpha=0.6):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-18:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-18:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-20)
        d[np.abs(A) < 1e-12] = 0.0

        damp = 0.2 + 0.8 * (it / max(1, n_iters // 2))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_local_search(a, n_steps=300, rng=None):
    """Subgradient descent targeting the convolution peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        direction = np.zeros(n)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        direction *= 2.0

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        base_step = min(0.03, max(0.001, (best_sc - 1.0) / (gn * gn + 1e-30) * 0.3))
        improved = False
        for mult in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
            step = base_step * mult
            a_try = a - step * grad
            a_try = np.clip(a_try, 0.0, 1000.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it % 8 == 0:
            a = a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a


def lp_fista_solve(a, M, n_iters=10000, beta=300.0, x_init=None):
    """FISTA for the smoothed LP subproblem."""
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.dot(a, a))
    L = beta * n * sum_a2 * 4.0 + 1e-10
    step = 1.0 / L

    n_c = 2 * n - 1
    M_fft = 1 << (2 * n - 2).bit_length()

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FA = np.fft.rfft(a, M_fft)
        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = beta * corr - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer=20, fista_iters=10000, time_limit=None,
                   beta=300.0, x_warm=None):
    """LP-based alternating optimization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())
        x = lp_fista_solve(a, M, n_iters=fista_iters, beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Line search on mixing
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t = 0.3
        best_t_sc = best_sc
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                break
        a = a_new

    return best_a, best_sc, x_warm


def interpolate(a, n_new):
    """Smoothly interpolate sequence to new length."""
    if len(a) == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xf = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            inits.append(interpolate(parent, n))

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Parabolic (zero at edges)
    inits.append(np.clip(1.0 - ((2.0 * xf - 1.0) ** 2), 0.0, 1000.0))
    # Raised cosine
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * xf)))
    # Sine-squared
    inits.append(np.sin(np.pi * xf) ** 2)
    # Power taper
    inits.append(np.clip(1.0 - (2.0 * xf - 1.0) ** 4, 0.0, 1000.0))
    # Smoothed random
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 100))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.0, 1000.0))
    # Gaussian-ish
    inits.append(np.exp(-2.0 * (2.0 * xf - 1.0) ** 2))
    # Multi-frequency
    inits.append(np.clip(0.5 + 0.2 * np.cos(2 * np.pi * xf) + 0.1 * np.cos(4 * np.pi * xf + 0.7), 0.0, 1000.0))

    return [np.clip(s, 0.0, 1000.0) for s in inits]


def progressive_lbfgs(a, n, beta_list, time_limit):
    """Run L-BFGS with progressively increasing beta."""
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S > 0:
        a = a / S

    best_sc = compute_score(a)
    best_a = a.copy()

    for beta in beta_list:
        if time.monotonic() > time_limit:
            break
        sc, a_opt = run_lbfgs(a, beta, maxiter=5000, maxfun=150000, gtol=1e-13)
        S = float(a_opt.sum())
        if S > 0:
            a_opt = a_opt / S
        sc_exact = compute_score(a_opt)
        if sc_exact < best_sc:
            best_sc = sc_exact
            best_a = a_opt.copy()
        a = a_opt

    return best_a, best_sc
