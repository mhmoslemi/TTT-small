```python
"""
Optimizer for minimizing
    F(a) = 2*n*max(conv(a,a)) / (sum(a))^2
over non-negative sequences.

Design:
- Work with probability-sum sequences (sum=1); the objective is scale invariant.
- Use FFT-based exact convolution and exact scores.
- Optimize in log-softmax variables to enforce positivity and sum=1.
- Smooth the non-smooth problem with:
    * LSE smoothed max,
    * soft hinge penalties centered at target max coefficients.
- Use a specialized spectral square-root iteration:
    H(rfft(a))^2 == rfft(conv(a,a)).
  Blend H toward a waterfilled/power-compressed target convolution, take a
  square root, invert, project to non-negative values, and line-search in
  coefficient space.  This directly shapes the coefficient spectrum of
  conv(a,a), which is the actual quantity being minimized.
- Try a few larger lengths to avoid local minima, then final polish.
"""

import numpy as np
import time
from scipy.optimize import minimize


def normalize_nonnegative(arr):
    """Clip to non-negative, remove non-finite entries, normalize sum to 1."""
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if (not np.isfinite(s)) or s < 1e-300:
        return None
    return a / s


def load_parent_sequence():
    """Load height_sequence_1 if it exists."""
    p = globals().get("height_sequence_1", None)
    if p is None:
        return None
    a = normalize_nonnegative(p)
    return a


def initial_sequence(n, seed):
    """Fallback initial sequence if no parent is available."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = (
        1.0
        + 0.75 * np.cos(np.pi * x)
        + 0.18 * np.cos(2.0 * np.pi * x)
        + 0.07 * np.cos(4.0 * np.pi * x)
        + 0.03 * rng.standard_normal(n)
    )
    a = normalize_nonnegative(a)
    if a is None:
        a = np.full(n, 1.0 / n)
    return a


def fast_conv(a):
    """Linear self-convolution by FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    m = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, m)
    return np.fft.irfft(A * A, m)[:L]


def score_from_a(a):
    """Internal exact score using FFT, with sum(a) assumed 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.inf
    if abs(s - 1.0) > 1e-14:
        a = a / s
    c = fast_conv(a)
    return float(2.0 * a.size * np.max(c))


def softmax_logits(u):
    """Stable softmax."""
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    u = np.clip(u, -45.0, 20.0)
    mu = float(np.max(u))
    with np.errstate(over="ignore", under="ignore"):
        e = np.exp(u - mu)
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def cross_corr_dense(a, weights):
    """
    Return g_j = sum_d a_d weights_{j+d}, for j=0..n-1.
    This is the core gradient primitive for sums of convolution coefficients.
    """
    a = np.asarray(a, dtype=np.float64)
    weights = np.asarray(weights, dtype=np.float64)
    n = a.size
    M = 3 * n - 2
    m = 1 << (M - 1).bit_length()
    W = np.fft.rfft(weights, m)
    AR = np.fft.rfft(a[::-1], m)
    conv = np.fft.irfft(W * AR, m)[:M]
    return conv[n - 1:2 * n - 1].copy()


def cross_corr_sparse(a, weights, indices):
    """Direct O(n*|indices|) version when only a few weights are active."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for s in np.asarray(indices, dtype=int).tolist():
        w = float(weights[s])
        if w == 0.0:
            continue
        js = max(0, int(s) - (n - 1))
        je = min(int(s), n - 1)
        if je >= js:
            g[js:je + 1] += w * a[js:je + 1][::-1]
    return g


def weighted_gradient_base(a, weights, sparse_threshold=16):
    """Choose dense FFT or direct sparse implementation."""
    nz = np.flatnonzero(weights > 1e-12)
    if nz.size == 0:
        return np.zeros(a.size, dtype=np.float64)
    if nz.size <= sparse_threshold:
        return cross_corr_sparse(a, weights, nz)
    return cross_corr_dense(a, weights)


def lse_log_objective(u, k):
    """
    Objective: 2n * LSE_{k*2n}(conv(a,a)) with a=softmax(u).
    Returns (value, gradient wrt u).
    """
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    a = softmax_logits(u)
    c = fast_conv(a)
    cmax = float(c.max())
    beta = 2.0 * n * float(k)

    # z = beta*(c - cmax) <= 0
    z = beta * (c - cmax)
    z = np.clip(z, -700.0, 0.0)

    with np.errstate(over="ignore", under="ignore"):
        ez = np.exp(z)
    sumexp = float(ez.sum())
    if (not np.isfinite(sumexp)) or sumexp <= 0.0:
        return 1e50, np.zeros(n, dtype=np.float64)

    L = cmax + float(np.log(sumexp)) / beta
    value = float(2.0 * n * L)

    p = ez / sumexp
    base = weighted_gradient_base(a, p)
    grad_a = 4.0 * n * base
    bias = float(np.dot(a, grad_a))
    grad_u = a * (grad_a - bias)

    if not np.all(np.isfinite(grad_u)):
        grad_u = np.zeros(n, dtype=np.float64)
    if not np.isfinite(value):
        return 1e50, grad_u
    return value, grad_u


def hinge_log_objective(u, M, k):
    """
    Objective:
        J = sum_k softplus(k*2n*(c_k-M)) / k
    with a=softmax(u).  Penalizes coefficients above M.
    Returns (J, gradient wrt u).
    """
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    a = softmax_logits(u)
    c = fast_conv(a)

    scale = 2.0 * n * float(k)
    z = scale * (c - float(M))
    z = np.clip(z, -700.0, 700.0)

    # softplus(z)
    sp = np.empty_like(z, dtype=np.float64)
    pos = z > 0.0
    with np.errstate(over="ignore", under="ignore"):
        sp[pos] = z[pos] + np.log1p(np.exp(-z[pos]))
        sp[~pos] = np.log1p(np.exp(z[~pos]))

    J = float(np.sum(sp) / float(k))
    if not np.isfinite(J):
        return 1e50, np.zeros(n, dtype=np.float64)

    # derivative dJ/dc_k = 2n * sigmoid(z_k)
    sig = np.empty_like(z, dtype=np.float64)
    with np.errstate(over="ignore", under="ignore"):
        sig[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
        ez = np.exp(z[~pos])
        sig[~pos] = ez / (1.0 + ez)

    base = weighted_gradient_base(a, sig)
    grad_a = 4.0 * n * base
    bias = float(np.dot(a, grad_a))
    grad_u = a * (grad_a - bias)

    if not np.all(np.isfinite(grad_u)):
        grad_u = np.zeros(n, dtype=np.float64)
    return J, grad_u


def run_lbfgs_objective(fun, u0, args, seconds, label,
                        lb=-45.0, ub=20.0,
                        maxfun_cap=8000, gtol=1e-10):
    """
    Run L-BFGS-B on a log-softmax objective for roughly `seconds` wall time.
    Returns (u_out, exact_score_of_a_out).
    """
    u0 = np.asarray(u0, dtype=np.float64).copy()
    n = u0.size
    if n == 0 or seconds <= 5.0:
        return u0, score_from_a(softmax_logits(u0))

    t0 = time.monotonic()
    try:
        f0, g0 = fun(u0, *args)
    except Exception:
        return u0, score_from_a(softmax_logits(u0))
    dt = time.monotonic() - t0
    if dt <= 1e-6 or not np.isfinite(dt):
        dt = 1e-3

    # Conservative; line search can take multiple func/grad evaluations.
    maxfun = int(min(maxfun_cap, max(200, 0.50 * seconds / dt)))
    if maxfun < 200:
        return u0, score_from_a(softmax_logits(u0))

    maxiter = max(50, maxfun // 4)

    bounds = [(lb, ub)] * n

    try:
        res = minimize(
            fun,
            u0,
            args=args,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1e-12,
                "gtol": gtol,
                "maxls": 30,
                "disp": False,
            },
        )
        u1 = np.asarray(res.x, dtype=np.float64).copy()
    except Exception:
        u1 = u0

    a1 = softmax_logits(u1)
    sc1 = score_from_a(a1)
    elapsed = time.monotonic() - t0
    print(f"    [{label}] score={sc1:.10f} t={elapsed:.1f}s")
    return u1, sc1


def waterfill_target(c, M):
    """
    Return non-negative t with sum(t)=1 and max(t) approximately min(M, feasible).
    This is the Euclidean projection of normalized c onto {0<=t<=M, sum(t)=1}.
    """
    c = np.asarray(c, dtype=np.float64).copy()
    L = c.size
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 0.0:
        return np.full(L, 1.0 / L, dtype=np.float64)
    c = c / s

    mean = 1.0 / L
    M = max(float(M), mean * (1.0 + 1e-12))

    cmax = float(c.max())
    if M >= cmax - 1e-15:
        return c.copy()

    # Since sum(c)=1 and M<cmax, adding lambda>=0 and clipping at M restores sum 1.
    lo = 0.0
    hi = max(HI if (HI := M - float(c.min())) is not None else 0.0, 1e-300)

    # Ensure hi is sufficient; if not, it is infeasible because M<mean, but we already guarded.
    # Binary search lambda such that sum(min(c+lambda, M)) = 1.
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        with np.errstate(over="ignore", under="ignore"):
            s_mid = float(np.sum(np.minimum(c + mid, M)))
        if s_mid < 1.0:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    t = np.minimum(c + lam, M)
    t = np.maximum(t, 0.0)
    ts = float(t.sum())
    if ts <= 0.0:
        return np.full(L, 1.0 / L, dtype=np.float64)
    return t / ts


def spectral_attack(a, stage_deadline, best_score, best_a, goal_score,
                    mode="water", alpha_init=0.10, max_iters=100000, seed=12345):
    """
    Spectral square-root iteration to shape conv(a,a) toward flat targets.
    """
    a = normalize_nonnegative(a)
    if a is None:
        return best_score, best_a

    n = a.size
    L = 2 * n - 1
    m = 1 << (L - 1).bit_length()
    lower_score = 2.0 * n / (2.0 * n - 1.0)

    if goal_score is None or goal_score < lower_score + 1e-9:
        goal_score = lower_score + 1e-7
    M_goal = float(goal_score) / (2.0 * n)

    rng = np.random.default_rng(seed)
    alpha = float(alpha_init)
    fail = 0
    it = 0
    start_time = time.monotonic()
    total_time = max(1e-6, float(stage_deadline) - start_time)

    while it < max_iters:
        it += 1
        if time.monotonic() >= float(stage_deadline):
            break

        A = np.fft.rfft(a, m)
        H = A * A
        c = np.fft.irfft(H, m)[:L]
        c = np.maximum(c, 0.0)
        cs = float(c.sum())
        if cs <= 0.0:
            break
        c = c / cs

        cmax = float(c.max())
        sc = float(2.0 * n * cmax)
        if not np.isfinite(sc):
            break

        if best_a is None or sc < best_score - 1e-9:
            if best_a is not None and sc > best_score + 1e-9:
                pass
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            if it <= 5 or it % 100 == 0:
                print(f"    [spectral {mode}] NEW BEST {best_score:.10f} n={n} it={it}")
            fail = 0

        if it == max_iters:
            break

        progress = min(1.0, (time.monotonic() - start_time) / total_time)
        # gradually move target max toward goal
        M_target = cmax - (cmax - M_goal) * (0.02 + 0.98 * progress)
        M_target = max(M_goal, min(cmax, M_target))

        if mode == "water":
            t = waterfill_target(c, M_target)
        elif mode == "power":
            p = 0.90 - 0.50 * min(1.0, it / 4000.0)
            floor = 1e-16 * max(1e-300, float(c.mean()))
            cp = c + floor
            with np.errstate(over="ignore", under="ignore"):
                tp = np.power(cp, p)
            tsum = float(tp.sum())
            if tsum <= 0.0 or not np.isfinite(tsum):
                t = c.copy()
            else:
                t = tp / tsum
        else:
            t = c.copy()

        T = np.fft.rfft(t, m)
        H_new = (1.0 - alpha) * H + alpha * T

        # Enforce DC/Nyquist realness.
        H_new[0] = float(np.real(H_new[0]))
        if m % 2 == 0:
            H_new[m // 2] = float(np.real(H_new[m // 2]))

        with np.errstate(invalid="ignore"):
            A_new = np.sqrt(H_new + 0j)

        if not np.all(np.isfinite(A_new.real)) or not np.all(np.isfinite(A_new.imag)):
            alpha = max(0.02, 0.7 * alpha)
            continue

        A_new[0] = complex(max(0.0, float(np.real(A_new[0][0]))), 0.0) if A_new[0].dtype != float else float(max(0.0, float(np.real(A_new[0]))))
        # The previous line is tricky; simpler robust assignment below:
        A_new[0] = float(np.sqrt(max(0.0, float(np.real(H_new[0])))))
        if m % 2 == 0:
            A_new[m // 2] = float(np.sqrt(max(0.0, float(np.real(H_new[m // 2])))))

        a_spec = np.fft.irfft(A_new, m)[:n]
        a_spec = normalize_nonnegative(a_spec)
        if a_spec is None:
            alpha = max(0.02, 0.7 * alpha)
            fail += 1
            continue

        blend = min(1.0, alpha + 0.25)
        trial = (1.0 - blend) * a + blend * a_spec
        trial = normalize_nonnegative(trial)
        if trial is None:
            trial = a

        sc_trial = score_from_a(trial)
        improved = (np.isfinite(sc_trial) and sc_trial < sc - 1e-12)
        # Occasionally allow slight worsening to escape local traps.
        if not improved and fail > 15 and sc_trial < sc + 0.0015 and rng.random() < 0.25:
            improved = True

        if improved:
            a = trial
            fail = 0
            alpha = min(0.45, alpha * 1.05)
        else:
            fail += 1
            if fail % 20 == 0:
                alpha = max(0.02, alpha * 0.72)
            if fail > 120:
                if best_a is not None:
                    a = normalize_nonnegative(best_a)
                    if a is None:
                        break
                else:
                    a = initial_sequence(n, seed + it)
                alpha = 0.05
                fail = 0

    if best_a is None:
        best_a = np.clip(a.copy(), 0.0, 1000.0)
    final_sc = score_from_a(best_a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, best_a


def is_better_sequence(sc, a, best_score, best_a):
    """Accept improvement; tie-break toward shorter sequences."""
    if a is None or (not np.isfinite(sc)):
        return False
    if best_a is None:
        return True
    if sc < best_score - 1e-9:
        return True
    if abs(sc - best_score) <= 1e-12 and len(a) < len(best_a):
        return True
    return False


def run_lse_polish(u0, stage_deadline, best_score, best_a,
                   k_list=None, max_sec=45.0):
    """Run a schedule of LSE-smoothed L-BFGS runs."""
    if u0 is None:
        return best_score, best_a, u0

    if k_list is None:
        k_list = [5, 10, 20, 40, 80, 160, 320, 640, 1280]

    u_work = np.asarray(u0, dtype=np.float64).copy()
    a_work = softmax_logits(u_work)
    sc_work = score_from_a(a_work)

    for idx, k in enumerate(k_list):
        now = time.monotonic()
        if now > float(stage_deadline) - 20.0:
            break
        remaining = float(stage_deadline) - now
        rest = len(k_list) - idx
        sec = max(10.0, min(max_sec, remaining / max(1, rest + 1)))

        u_new, sc_new = run_lbfgs_objective(
            lse_log_objective, u_work, (float(k),), sec, f"LSE k={int(k)}"
        )
        if not np.isfinite(sc_new):
            continue

        a_new = softmax_logits(u_new)
        sc_new = score_from_a(a_new)

        if is_better_sequence(sc_new, a_new, best_score, best_a):
            best_score = sc_new
            best_a = np.clip(a_new.copy(), 0.0, 1000.0)
            print(f"    [LSE k={int(k)}] UPDATED BEST {best_score:.10f} (n={len(a_new)})")

        # Keep the new parameter only if it is not catastrophic.
        if sc_new <= max(sc_work, best_score) + 0.02:
            u_work = u_new.copy()
            a_work = a_new.copy()
            sc_work = sc_new

    return best_score, best_a, u_work


def make_goal_scores(current_score, lower_score, target=1.5027):
    """
    Build a descending list of target F-scores to attack with hinge penalties.
    """
    goals = [
        current_score - 0.0005,
        current_score - 0.0015,
        current_score - 0.0030,
        current_score - 0.0060,
        target,
        target - 0.0010,
        target - 0.0030,
        target - 0.0060,
        target - 0.0100,
        1.4950,
        1.4920,
    ]
    filtered = []
    for g in goals:
        if np.isfinite(g) and g > lower_score + 1e-8:
            g = float(g)
            # avoid duplicates
            if not any(abs(g - x) < 1e-7 for x in filtered):
                filtered.append(g)
    filtered.sort(reverse=True)
    return filtered


def run_hinge_polish(u0, stage_deadline, best_score, best_a,
                     lower_score, target=1.5027,
                     k_list=(50, 200, 800), max_sec=35.0):
    """
    Run hinge-penalty L-BFGS for a sequence of target scores.
    """
    if u0 is None:
        return best_score, best_a, u0

    u_work = np.asarray(u0, dtype=np.float64).copy()
    a_work = softmax_logits(u_work)
    sc_work = score_from_a(a_work)
    n = u_work.size

    goals = make_goal_scores(sc_work, lower_score, target)

    for goal in goals:
        now = time.monotonic()
        if now > float(stage_deadline) - 25.0:
            break

        # If the target score is much looser than current working score, skip.
        if goal > sc_work + 5e-4:
            continue

        M = float(goal) / (2.0 * n)
        for k in k_list:
            now = time.monotonic()
            if now > float(stage_deadline) - 15.0:
                break
            remaining = float(stage_deadline) - now
            sec = max(8.0, min(max_sec, remaining / 3.0))

            u_new, sc_new = run_lbfgs_objective(
                hinge_log_objective,
                u_work,
                (M, float(k)),
                sec,
                f"HINGE target={goal:.6f} k={int(k)}"
            )
            if not np.isfinite(sc_new):
                continue

            a_new = softmax_logits(u_new)
            sc_new = score_from_a(a_new)

            if is_better_sequence(sc_new, a_new, best_score, best_a):
                best_score = sc_new
                best_a = np.clip(a_new.copy(), 0.0, 1000.0)
                print(f"    [HINGE target={goal:.6f}] UPDATED BEST {best_score:.10f} (n={len(a_new)})")

            if sc_new <= max(sc_work, best_score) + 0.02:
                u_work = u_new.copy()
                a_work = a_new.copy()
                sc_work = sc_new

    return best_score, best_a, u_work


def resample_to_length(a, new_n):
    """Linearly resample a positive-sum sequence to a new length."""
    a = normalize_nonnegative(a)
    if a is None:
        return None
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    if old_n < 2:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    b = normalize_nonnegative(b)
    if b is None:
        return normalize_nonnegative(np.exp(np.log(a + 1e-300)))
    return b


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function.  Must return the best non-negative sequence found.
    """
    t_start = time.monotonic()
    available = max(60.0, float(budget_s) - 35.0)
    deadline = t_start + available

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialize
    # ------------------------------------------------------------------
    parent = load_parent_sequence()
    if parent is not None and parent.size >= 16:
        best_a = parent.copy()
        best_score = score_from_a(best_a)
        print(f"Parent loaded: n={len(best_a)}, internal score={best_score:.10f}")
    else:
        best_a = initial_sequence(10000, seed)
        best_score = score_from_a(best_a)
        print(f"No usable parent. Initial n={len(best_a)}, internal score={best_score:.10f}")

    if best_a is None or (not np.isfinite(best_score)):
        best_a = np.full(10000, 1.0 / 10000.0, dtype=np.float64)
        best_score = score_from_a(best_a)

    best_a = np.clip(best_a, 0.0, 1000.0)

    def log_u_from_a(a):
        a = normalize_nonnegative(a)
        if a is None:
            a = np.full(10000, 1.0 / 10000.0)
        return np.clip(np.log(np.maximum(a, 1e-300)), -45.0, 20.0)

    u_work = log_u_from_a(best_a)
    a_work = softmax_logits(u_work)

    # ------------------------------------------------------------------
    # Stage 1: global spectral shaping (power-compressed target)
    # ------------------------------------------------------------------
    stage1_end = t_start + 0.09 * available
    lower0 = 2.0 * len(best_a) / (2.0 * len(best_a) - 1.0)
    goal0 = max(lower0 + 1e-7, min(1.500, best_score - 0.003))
    print(f"\n=== Stage 1: spectral power shaping (until t+{stage1_end-t_start:.1f}s) ===")
    best_score, best_a = spectral_attack(
        a_work,
        stage_deadline=stage1_end,
        best_score=best_score,
        best_a=best_a,
        goal_score=goal0,
        mode="power",
        alpha_init=0.12,
        seed=123,
    )
    a_work = best_a.copy()
    u_work = log_u_from_a(a_work)

    # ------------------------------------------------------------------
    # Stage 2: LSE L-BFGS polish
    # ------------------------------------------------------------------
    stage2_end = t_start + 0.29 * available
    print(f"\n=== Stage 2: LSE L-BFGS (until t+{stage2_end-t_start:.1f}s) ===")
    best_score, best_a, u_work = run_lse_polish(
        u_work,
        stage2_end,
        best_score,
        best_a,
        k_list=[10, 20, 40, 80, 160, 320, 640],
        max_sec=55.0,
    )
    a_work = softmax_logits(u_work)

    # ------------------------------------------------------------------
    # Stage 3: hinge-penalty L-BFGS
    # ------------------------------------------------------------------
    stage3_end = t_start + 0.54 * available
    lower0 = 2.0 * len(best_a) / (2.0 * len(best_a) - 1.0)
    print(f"\n=== Stage 3: hinge L-BFGS (until t+{stage3_end-t_start:.1f}s) ===")
    best_score, best_a, u_work = run_hinge_polish(
        u_work,
        stage3_end,
        best_score,
        best_a,
        lower_score=lower0,
        target=1.5027,
        k_list=(50, 200, 800),
        max_sec=45.0,
    )
    a_work = softmax_logits(u_work)

    # ------------------------------------------------------------------
    # Stage 4: fine spectral waterfill
    # ------------------------------------------------------------------
    stage4_end = t_start + 0.62 * available
    goal4 = max(lower0 + 1e-7, min(1.499, best_score - 0.002))
    print(f"\n=== Stage 4: spectral waterfill (until t+{stage4_end-t_start:.1f}s) ===")
    best_score, best_a = spectral_attack(
        a_work,
        stage_deadline=stage4_end,
        best_score=best_score,
        best_a=best_a,
        goal_score=goal4,
        mode="water",
        alpha_init=0.08,
        seed=987,
    )
    a_work = best_a.copy()
    u_work = log_u_from_a(a_work)

    # ------------------------------------------------------------------
    # Stage 5: larger-length exploration
    # ------------------------------------------------------------------
    stage5_end = t_start + 0.80 * available
    print(f"\n=== Stage 5: larger length search (until t+{stage5_end-t_start:.1f}s) ===")

    current_n = len(best_a)
    candidate_ns = []
    for cand in [10000, 12000, 15000, 16384]:
        if cand > current_n + 100 and cand <= 17000:
            candidate_ns.append(cand)
    candidate_ns = candidate_ns[:2]

    for jj, new_n in enumerate(candidate_ns):
        if time.monotonic() > stage5_end - 80.0:
            break

        a_big = resample_to_length(best_a, new_n)
        if a_big is None:
            continue
        sc_big = score_from_a(a_big)
        print(f"  Resampled to n={new_n}, score={sc_big:.10f}")

        if is_better_sequence(sc_big, a_big, best_score, best_a):
            best_score = sc_big
            best_a = np.clip(a_big.copy(), 0.0, 1000.0)

        u_big = log_u_from_a(a_big)

        # short spectral waterfill
        if time.monotonic() < stage5_end - 80.0:
            lower_n = 2.0 * new_n / (2.0 * new_n - 1.0)
            goal_big = max(lower_n + 1e-7, min(1.499, best_score - 0.002))
            tsub = min(stage5_end, time.monotonic() + 0.25 * (stage5_end - t_start) + 30.0)
            best_score, best_a = spectral_attack(
                a_big,
                stage_deadline=min(tsub, stage5_end - 60.0),
                best_score=best_score,
                best_a=best_a,
                goal_score=goal_big,
                mode="water",
                alpha_init=0.07,
                seed=555 + 17 * jj,
            )
            a_big = best_a.copy()
            u_big = log_u_from_a(a_big)

        # LSE polish
        if time.monotonic() < stage5_end - 50.0:
            tsub = min(stage5_end - 45.0, time.monotonic() + 120.0)
            best_score, best_a, u_big = run_lse_polish(
                u_big,
                tsub,
                best_score,
                best_a,
                k_list=[40, 160, 640],
                max_sec=45.0,
            )
            a_big = softmax_logits(u_big)

        # Hinge polish
        if time.monotonic() < stage5_end - 30.0:
            lower_n = 2.0 * len(u_big) / (2.0 * len(u_big) - 1.0)
            best_score, best_a, u_big = run_hinge_polish(
                u_big,
                stage5_end - 25.0,
                best_score,
                best_a,
                lower_score=lower_n,
                target=1.5027,
                k_list=(200, 800),
                max_sec=35.0,
            )

    # ------------------------------------------------------------------
    # Stage 6: final polish on the best sequence
    # ------------------------------------------------------------------
    print(f"\n=== Stage 6: final polish (until deadline) ===")
    a_final_base = best_a.copy()
    u_final = log_u_from_a(a_final_base)
    a_final = softmax_logits(u_final)
    sc_final = score_from_a(a_final)
    if is_better_sequence(sc_final, a_final, best_score, best_a):
        best_score = sc_final
        best_a = np.clip(a_final.copy(), 0.0, 1000.0)

    lower_final = 2.0 * len(best_a) / (2.0 * len(best_a) - 1.0)

    # final LSE high-beta
    if time.monotonic() < deadline - 60.0:
        tsub = min(deadline - 50.0, time.monotonic() + 300.0)
        best_score, best_a, u_final = run_lse_polish(
            u_final,
            tsub,
            best_score,
            best_a,
            k_list=[320, 1280, 4096],
            max_sec=45.0,
        )
        a_final = softmax_logits(u_final)

    # final hinge
    if time.monotonic() < deadline - 40.0:
        tsub = min(deadline - 35.0, time.monotonic() + 300.0)
        best_score, best_a, u_final = run_hinge_polish(
            u_final,
            tsub,
            best_score,
            best_a,
            lower_score=lower_final,
            target=1.5027,
            k_list=(300, 1200),
            max_sec=40.0,
        )
        a_final = softmax_logits(u_final)

    # final spectral water
    if time.monotonic() < deadline - 20.0:
        goal_final = max(lower_final + 1e-8, min(1.497, best_score - 0.003))
        best_score, best_a = spectral_attack(
            a_final,
            stage_deadline=min(deadline - 5.0, time.monotonic() + 180.0),
            best_score=best_score,
            best_a=best_a,
            goal_score=goal_final,
            mode="water",
            alpha_init=0.05,
            seed=3131,
        )

    # ------------------------------------------------------------------
    # Finalize output
    # ------------------------------------------------------------------
    best_a = normalize_nonnegative(best_a)
    if best_a is None:
        best_a = np.full(10000, 1.0 / 10000.0, dtype=np.float64)

    # The objective is scale-invariant; sum=1 is safe for the evaluator.
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    final_internal = score_from_a(np.asarray(out, dtype=np.float64) / sum(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*70}")
    print(f"FINAL internal score={final_internal:.10f}, length={len(out)}")
    print(f"Total time={elapsed:.1f}s / {budget_s:.0f}s")
    print(f"{'='*70}")

    # Optional verification if cheap enough.
    if len(out) <= 12000 and time.monotonic() < t_start + float(budget_s) - 8.0:
        try:
            verified = float(evaluate_sequence(out))
            print(f"Verified evaluate_sequence score={verified:.10f}")
        except Exception:
            pass

    return out
```
