def _exact_score(a):

        return np.inf

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())

def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a)

def _subgrad_topk(a, k, tol):
    """Subgradient of max c_k using top-k peaks within tol of max."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c = _fast_self_conv(a)
    cmax = c.max()
    score = 2.0 * n * cmax / (S * S)
    
    threshold = cmax - tol * cmax
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > k:
        top_idx = np.argsort(c[active])[-k:]
        active = active[top_idx]
    
    # Gradient of c_k at p: grad_k[j] = 2*p[k-j] if 0<=k-j<n else 0
    # Average over active set: grad[j] = (2/|A|) * sum_{k in A} p[k-j]
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 2.0 / len(active)
    
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    IR = np.fft.rfft(indicator, N2)
    conv_full = np.fft.irfft(AR * IR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    # Full gradient of 2*n*max(c)/S^2
    # = 2n/S^2 * R - 2n * (sum_k w_k * c_k) / S^3 * 1
    T = float(indicator.dot(c) / 2.0)  # (1/|A|) * sum c_k for active
    # Wait, indicator already has 2/|A|, so indicator.dot(c) = (2/|A|) * sum c_k
    # We need (1/|A|) * sum c_k = indicator.dot(c) / 2
    T = float(indicator.dot(c) / 2.0)
    
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

        indicator = np.zeros(2 * n - 1)
        indicator[active] = 1.0 / len(active)

        a_rev = a[::-1]
        AR = np.fft.rfft(a_rev, N2)
        IR = np.fft.rfft(indicator, N2)
        conv_full = np.fft.irfft(AR * IR, N2)[:3 * n - 2]
        d = conv_full[n - 1:2 * n - 1]

        if d_max < 1e-15:

        d = d / d_max

        candidates = []
        for g in [gamma * 0.5, gamma * 0.8, gamma, gamma * 1.2, gamma * 1.5]:

            scores.append(sc)

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return float("inf")
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _pressure_vector(a, active):
    """Compute d[i] = (1/|A|) * sum_{k in A} a[k-i] (a out of range = 0)."""
    n = a.size
    M = 2 * n - 1
    indicator = np.zeros(M)
    indicator[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    IR = np.fft.rfft(indicator, N2)
    conv_full = np.fft.irfft(AR * IR, N2)[:3 * n - 2]
    return conv_full[n - 1:2 * n - 1]


def _subgradient_weights(a, active):
    """Compute R[i] = sum_{k in active} 2*a[k-i] (unnormalized)."""
    n = a.size
    M = 2 * n - 1
    indicator = np.zeros(M)
    indicator[active] = 2.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    IR = np.fft.rfft(indicator, N2)
    conv_full = np.fft.irfft(AR * IR, N2)[:3 * n - 2]
    return conv_full[n - 1:2 * n - 1]


def _subgrad(a, k_active, tol_frac):
    """Exact subgradient of 2*n*max(c)/S^2 using top-k active peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return float("inf"), np.zeros(n)
    c = _fast_self_conv(a)
    cmax = float(c.max())
    score = 2.0 * n * cmax / (S * S)
    if cmax < 1e-15:
        return score, np.zeros(n)
    threshold = cmax * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > k_active:
        top_idx = np.argsort(c[active])[-k_active:]
        active = active[top_idx]
    R = _subgradient_weights(a, active)
    T = float(c[active].sum()) / len(active)
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _iaa_optimize(a0, deadline, max_iters=20000, verbose=False):

    a = _normalize(a0)
    n = a.size
    best_a = a.copy()

    gamma = 0.2
    no_improve = 0
    last_improve_iter = 0

    for it in range(max_iters):

            break
        c = _fast_self_conv(a)

        if cmax < 1e-15:
            break
        # Adaptive active set width
        if no_improve < 50:
            frac = 0.998
        elif no_improve < 150:
            frac = 0.995
        elif no_improve < 300:
            frac = 0.99
        else:
            frac = 0.98
        threshold = cmax * frac
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 50:
            top_idx = np.argsort(c[active])[-50:]
            active = active[top_idx]
        d = _pressure_vector(a, active)
        dmax = float(d.max())
        if dmax < 1e-15:
            break
        d = d / dmax
        # Line search
        best_g = gamma
        best_sc = best_score
        best_a_trial = a.copy()
        for g in [gamma * 0.4, gamma * 0.7, gamma, gamma * 1.4, gamma * 2.0]:
            a_trial = a * np.exp(-g * d)
            a_trial = _normalize(a_trial)
            sc = _exact_score(a_trial)

        a = best_a_trial
        # Periodic symmetrization

def _lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient for 2*n*max(c)/S^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400.0, 400.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, deadline, maxfun=500):
    """L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def _fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        p = x / S
        val, grad_p = _lse_obj_grad(p, beta)
        # Chain rule for normalization: x -> p = x/S
        # dp/dx_i = (1/S) * delta_{ij} - x_j/(S^2)
        # grad_x = grad_p/S - p * sum(grad_p)/S^2
        grad_x = grad_p / S - p * float(grad_p.sum()) / (S * S)
        grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)
        return val, grad_x

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": maxfun,
        "maxfun": maxfun * 3,
        "maxls": 50,
    }
    try:
        res = minimize(_fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        a_out = _normalize(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _nesterov_optimize(a0, deadline, max_iters=5000, k_active=3, tol_frac=1e-4):
    """Nesterov accelerated subgradient with adaptive step."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    # Initial step
    sc0, g0 = _subgrad(a, k_active, tol_frac)
    g_norm0 = np.linalg.norm(g0)
    if g_norm0 < 1e-14:
        return best_score, best_a
    # Polyak-like step size
    gap = max(sc0 - lower_bound, 1e-6)
    eta = gap / (g_norm0 * g_norm0 + 1e-12)
    eta = min(eta, 0.1 / (g_norm0 + 1e-12))
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_n = y / Sy
        score_y, grad_y = _subgrad(y_n, k_active, tol_frac)
        if not np.isfinite(score_y):
            break
        # Adaptive step: shrink if no improvement
        step = eta
        x_new = y_n - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Adaptive eta
        if no_improve == 20:
            eta *= 0.8
            no_improve = 0
        elif no_improve == 80:
            eta *= 0.6
            no_improve = 0
        elif no_improve > 200:
            break
    return best_score, best_a
