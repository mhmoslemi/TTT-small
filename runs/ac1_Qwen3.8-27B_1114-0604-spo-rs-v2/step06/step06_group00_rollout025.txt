```python
def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_with(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()

    # Gradient of L w.r.t. a
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad

    best_a = a.copy()
    best_sc = compute_score(a)

    v = np.zeros(n)  # second moment

        S = float(a.sum())

        S3 = S2 * S

                rows.append(i)
                cols.append(j)
                vals.append(a[k - j])

                    # Blend
                    t = 0.1  # small blending parameter
                    a_new = (1 - t) * a + t * g
                    a_new = normalize(a_new)
                    sc_new = compute_score(a_new)
                    if sc_new < sc:
                        a = a_new
        except Exception:
            pass
    
    return best_a
```

```python
"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/S^2.
Combines LP-based block coordinate descent, enhanced LSE annealing,
Adam subgradient with multi-peak targeting, spectral methods,
and diverse multi-restart.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time
import warnings
warnings.filterwarnings('ignore')


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_pair(a, b):
    """Linear convolution of a and b."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def normalize(a):
    """Clip to [0, inf) and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = np.clip(beta * (c - c_max), -600, 600)
    exp_s = np.exp(shifted)
    L = c_max + float(np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()
    # Gradient via FFT
    a_rev = a[::-1]
    M3 = (2 * n - 1) + (n - 1)
    M3f = 1 << (M3 - 1).bit_length()
    conv_r = np.fft.irfft(np.fft.rfft(p, M3f) * np.fft.rfft(a_rev, M3f), M3f)[:M3]
    grad_L = 2.0 * conv_r[n - 1:2 * n - 1]
    S2, S3 = S * S, S * S * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return 2.0 * n * L / S2, grad


def lbfgs_lse(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    try:
        res = minimize(lse_score_and_grad, a0, args=(beta,),
                       method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'ftol': 1e-20, 'gtol': 1e-16,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1.0)
        if a_out.sum() < 1e-8:
            return compute_score(a0), a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        if _deadline is not None and time.monotonic() > _deadline - 60:
            break

        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_active = np.argsort(c)[::-1][:n_constr]

        rows = np.empty(nc * n, dtype=np.int32)
        cols = np.empty(nc * n, dtype=np.int32)
        vals = np.empty(nc * n, dtype=np.float64)
        # Actually, let's build it more efficiently
        row_list = []
        col_list = []

        for i, k in enumerate(k_active):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            row_list.extend(np.full(len(js), i).tolist())
            col_list.extend(js.tolist())
            val_list.extend((a[k - js]).tolist())

        if not row_list:
            break
        D = csr_matrix((val_list, (row_list, col_list)), shape=(nc, n))

        # Solve LP
        try:
            res = linprog(c=-np.ones(n), A_ub=D, b_ub=np.full(nc, M_val),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': 30, 'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0.0)
                g_sum = float(g.sum())
                if g_sum > S * 1.001:
                    g = g * (S / g_sum)
                    a_new = (1.0 - blend) * a + blend * g
                    a_new = normalize(a_new)
                    sc_new = compute_score(a_new)
                    if sc_new < sc:
                        a = a_new
                        # Reduce blend as we converge
                        blend = max(0.05, blend * 0.95)
        except Exception:
            pass

    return best_a


def adam_multi_peak(a, n_iters=10000, top_k=10, lr=0.005,
                   beta1=0.9, beta2=0.999, time_limit=300):
    """
    Adam optimizer for subgradient of multi-peak LSE.
    Uses softmax weighting over top-K peaks.
    """
    global _deadline
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if _deadline is not None and time.monotonic() > _deadline - 30:
            break

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())

        # Softmax weights over top-K
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Temperature-controlled softmax
        temp = 1.0 / (50.0 + 5.0 * (1.0 - top_vals[0] / (c_max + 1e-15)))
        w = np.exp(top_vals * temp)
        w /= w.sum()

        # Weighted subgradient
        sg = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            # subgrad of conv(a,a)[k] w.r.t. a[m] is 2*a[k-m] (if valid)
            # Vectorized:
            m_vals = np.arange(n)
            idx = k - m_vals
            valid = (idx >= 0) & (idx < n)
            sg[valid] += wi * 2.0 * a[idx[valid]]

        # Objective gradient: 2n * (weighted_sg / S^2 - 2 * LSE / S^3)
        # Use the max as a proxy for LSE (hard max in the limit)
        L_approx = float(np.sum(w * top_vals))
        S2, S3 = S * S, S * S * S
        grad = 2.0 * n * (sg / S2 - 2.0 * L_approx / S3)

        # Adam
        t += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
        m_u = m_hat / (1 - beta1 ** t)
        v_u = v_hat / (1 - beta2 ** t)

        a = a - lr * m_u / (np.sqrt(v_u) + 1e-8)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = 2.0 * n * float(fast_conv(a).max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a
```

Let me also think about a **direct "peak push-down" approach**:

Instead of gradient-based methods, I can directly modify the sequence to reduce a specific peak:
- For peak at lag `k*`, the contribution is `sum_j a[j]*a[k*-j]`.
- To reduce this, I can reduce the products `a[j]*a[k*-j]`.
- The most effective way is to reduce the largest products.
- But I also need to maintain `sum(a) = 1` (or at least not decrease it too much).
- So I should transfer mass from the "peak contributors" to "non-peak contributors".

This is essentially what the coordinate descent in the previous code does, but I can make it more aggressive.

```python
def mass_transfer(a, n_iters=5000, n_reduce=50, n_add=100, time_limit=300):
    """
    Direct mass transfer: move mass from peak contributors to non-peak positions.
    """
    global _deadline
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if _deadline is not None and time.monotonic() > _deadline - 30:
            break

        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]

        # Reduce top contributors
        n_red = min(n_reduce, len(contribs))
        top_local = np.argsort(contribs)[::-1][:n_red]
        reduce_idx = j_start + top_local

        # Find non-peak positions
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[js] = True
        non_peak = np.where(~peak_mask)[0]

        if len(non_peak) < n_add:
            break

        # Transfer mass
        total_red = 0.0
        for j in reduce_idx:
            delta = min(a[j] * 0.01, 0.001 / n)
            if a[j] - delta > 0:
                a[j] -= delta
                total_red += delta

        if total_red < 1e-15:
            break

        # Distribute to non-peak (randomly selected)
        add_idx = np.random.choice(non_peak, min(n_add, len(non_peak)), replace=False)
        a[add_idx] += total_red / len(add_idx)

        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_a
```

Let me also think about one more novel approach: **using the structure of the optimal solution**.

From the KKT analysis earlier, at the optimum (with all `a_j > 0`), we need `a[k*-m] = const` for all `m`, which means `a` is constant. But we know constant gives objective 2, not 1.5. So the optimum must have some `a_j = 0` or multiple active constraints.

If multiple lags are at the maximum, the KKT conditions become:
`sum_l w_l * a[k_l - m] = const` for all `m`, where `w_l >= 0` are the KKT multipliers and `k_l` are the active lags.

This means `a` is in the "kernel" of a certain operator defined by the active lags. The active lags and their weights determine the shape of `a`.

This suggests that the optimal `a` has a specific structure determined by the set of active lags. If I can identify the right set of active lags, I can solve for `a` directly.

However, determining the right set of active lags is itself a combinatorial problem. The LP approach implicitly handles this by finding the optimal dual variables (which correspond to the active constraints).

Let me add one more approach: **solve the KKT conditions directly** for a given set of active lags.

For a given set of active lags `{k_1, ..., k_L}` and weights `{w_1, ..., w_L}` (summing to 1), the KKT condition is:
`sum_l w_l * a[k_l - m] = c` for all `m = 0, ..., n-1`

This is a system of `n` linear equations in `n` unknowns (`a[0], ..., a[n-1]`). I can solve it!

The matrix is: `B[m, j] = sum_l w_l * 1_{0 <= k_l - m < n and j = k_l - m}`... wait, that's not quite right.

Actually, the condition is: for each `m`, `sum_l w_l * a[k_l - m] = c` (where `a[i] = 0` for `i < 0` or `i >= n`).

This gives `n` equations:
`sum_l w_l * a[k_l - m] = c` for `m = 0, ..., n-1`

This is a linear system in the unknowns `a[0], ..., a[n-1]` and `c`. The matrix has a specific structure related to the active lags.

If I can solve this system and get `a >= 0`, then I've found a KKT point. The question is whether it's the global optimum.

```python
    #   (T a)[m] = sum_l w_l * a[k_l - m]
    
    # T is a linear operator. The condition is T a = c * 1 (vector of ones).
    # Combined with sum(a) = 1, we can solve for a.

    # = sum_l w_l * 1_{j = k_l - m, 0 <= k_l - m < n}

    # For a given (m, j), M[m, j] = w_l if there exists l with k_l = j + m.
    
    # The system is:
    #   M @ a = c * ones(n)
    #   ones(n)^T @ a = 1
    
    # This is (n+1) equations in (n+1) unknowns (a[0],...,a[n-1], c).
    # Augmented system:
    #   [M  -1*ones] [a]     [0]
    #   [ones^T  0  ] [c]  = [1]

    # [ones^T 0] [c] = [1]

    # M[m, j] = sum of w_l for l where k_l = j + m (and 0 <= k_l - m < n, j in [0,n))

    # So M@m has (M@m)[m] = sum_j M[m,j] * a[j] = sum_j sum_l w_l * 1_{j=k_l-m} * a[j]
```

```python
def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
```
