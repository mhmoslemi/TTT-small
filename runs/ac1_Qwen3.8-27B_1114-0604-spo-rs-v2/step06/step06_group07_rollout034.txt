3. Set $\theta = \frac{1}{\rho}(\sum_{j=1}^^\rho z_{(j)} - 1)$
4. Return $x_i = \max(z_i - \theta, 0)$

This is $O(n \log n)$ due to sorting. For our purposes, this is fine.

def fast_conv(a):

def proj_simplex(z, target=1.0):
    """Project z onto {x : x >= 0, sum(x) = target}."""
    n = len(z)
    z_sorted = np.sort(z)[::-1]
    cumsum = np.cumsum(z_sorted)
    rho = np.arange(1, n + 1)
    cond = z_sorted - (cumsum - target) / rho > 0
    rho_val = np.max(rho[cond])
    theta = (cumsum[rho_val - 1] - target) / rho_val
    return np.maximum(z - theta, 0.0)

    n = len(a0)

    if S < 1e-12:

    best_a = a.copy()
    best_sc = compute_score(a)
    
    for outer in range(n_outer):
        # Update weights
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        c_max = float(np.max(c))
        if c_max < 1e-20:
            break
        p = p_start + (p_end - p_start) * (outer / max(1, n_outer - 1))
        w = (c / c_max) ** p
        w = w / w.sum()  # normalize
        
        # Inner loop: projected (accelerated) gradient
        lr = lr_start * (lr_end / lr_start) ** (outer / max(1, n_outer - 1))
        x = a.copy()  # for Nesterov
        y = a.copy()  # for standard
        for inner in range(n_inner):

            # Normalized gradient (since we work on the simplex)
            # Actually, the gradient of sum_k w_k c_k with respect to a is 2 * cross
            # But we need to project onto the simplex

            y_new = proj_simplex(z)

                mu = 2.0 / (inner + 3.0)  # Nesterov momentum
                x = y_new + mu * (y_new - y)
                x = proj_simplex(x)  # Ensure feasibility
            else:
                x = y_new
            y = y_new
            
            a = y_new
        
        # Update best
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.85):

        A = np.fft.rfft(a, M_fft)

def peak_descent(a, n_iters=2000, n_peaks=5, step_init=0.01):
    """Peak-directed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (float(a.sum())**2)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[js:je+1][::-1]
        
        S2 = float(a.sum())**2
        S3 = S2 * float(a.sum())
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, 0.02)
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
    
    return best_a

import numpy as np

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2n * max(c) / S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def proj_simplex(z, target=1.0):
    """Project z onto {x : x >= 0, sum(x) = target}."""
    n = len(z)
    z_sorted = np.sort(z)[::-1]
    cumsum = np.cumsum(z_sorted)
    rho = np.arange(1, n + 1)
    cond = z_sorted - (cumsum - target) / rho > 0
    if not cond.any():
        return np.full(n, target / n)
    rho_val = np.max(rho[cond])
    theta = (cumsum[int(rho_val) - 1] - target) / rho_val
    return np.maximum(z - theta, 0.0)


def weighted_cross(a, w):
    """
    Compute cross[i] = sum_j a[j] * w[j+i] for i=0,...,n-1.
    a has length n, w has length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    w_rev = w[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, L)
    W_fft = np.fft.rfft(w_rev, L)
    conv_result = np.fft.irfft(A_fft * W_fft, L)
    # cross[i] = conv_result[2n-2-i]
    return conv_result[n - 1:2 * n - 1][::-1]


def irpg_additive(a0, n_outer=50, n_inner=200, p_start=2.0, p_end=50.0,
                  lr_start=0.005, lr_end=0.0005, use_nesterov=True,
                  verbose=False, deadline=None):
    """
    Iteratively Reweighted Projected Gradient (additive/Nesterov).
    At each outer step, reweight peaks and do inner projected gradient steps.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for outer in range(n_outer):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute autocorrelation and weights
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        c_max = float(np.max(c))
        if c_max < 1e-20:
            break
        t = outer / max(1, n_outer - 1)
        p = p_start + (p_end - p_start) * t
        w = (c / c_max) ** p
        w_sum = float(w.sum())
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Inner loop
        lr = lr_start * (lr_end / lr_start) ** t
        x = a.copy()  # Nesterov extrapolate
        y = a.copy()  # current point

        for inner in range(n_inner):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Gradient of sum_k w_k c_k
            cross = weighted_cross(y, w)
            grad = 2.0 * cross

            # Step
            z = y - lr * grad
            y_new = proj_simplex(z)

            if use_nesterov:
                mu = 2.0 / (inner + 3.0)
                x = y_new + mu * (y_new - y)
                x = proj_simplex(x)
            y = y_new
            a = y_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print(f"    IRPG outer={outer}, p={p:.1f}, score={sc:.8f}")

    return best_a


def irpg_mult(a0, n_iters=3000, p=10.0, eta_start=0.1, eta_end=0.01,
              verbose=False, deadline=None):
    """
    Iteratively Reweighted Multiplicative Weights (Mirror Descent with KL).
    Update: a_new[i] proportional to a[i] * exp(-eta * grad[i])
    This is more natural for the simplex constraint.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute weights
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        c_max = float(np.max(c))
        if c_max < 1e-20:
            break
        w = (c / c_max) ** p
        w = w / w.sum()

        # Gradient
        cross = weighted_cross(a, w)
        grad = 2.0 * cross

        # Normalize gradient for stable exponent
        g_max = float(np.max(grad))
        if g_max < 1e-20:
            break
        grad_norm = grad / g_max

        # Adaptive learning rate
        t = it / max(1, n_iters)
        eta = eta_start + (eta_end - eta_start) * t

        # Multiplicative update: a_new[i] prop to a[i] * exp(-eta * grad_norm[i])
        log_a = np.log(a)
        log_a_new = log_a - eta * grad_norm
        # Numerical stability
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        S_new = float(a_new.sum())
        if S_new < 1e-30:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            if verbose and it % 500 == 0:
                print(f"    IRPG-M it={it}, score={sc:.8f}")

        a = a_new

    return best_a


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.85,
                     deadline=None):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters))
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha

        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_ellip(a, n_iters=300, beta_shape=3.0, deadline=None):
    """Elliptical spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters)
        beta = beta_shape * (0.5 + progress)

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)

        ratio = amp / C
        amp_new = amp / np.sqrt(1.0 + beta * (ratio - 1.0) ** 2)

        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_objective(x, beta):
    """LSE-smoothed objective with gradient. x is the sequence (non-negative)."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: df/dx_i = 2n * (dL/dx_i / S^2 - 2L / S^3)
    # dL/dx_i = 2 * sum_k p_k * x_{k-i}
    p = exp_scaled / exp_sum
    cross = weighted_cross(x, p)
    g_L = 2.0 * cross

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_descent(a, n_iters=2000, n_peaks=5, step_init=0.01, deadline=None):
    """Peak-directed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, 0.02)
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999

    return best_a
