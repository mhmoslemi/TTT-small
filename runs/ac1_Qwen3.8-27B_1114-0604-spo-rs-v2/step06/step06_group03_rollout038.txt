"""
Multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combined approach: spectral GS for diverse restarts, LSE optimization,
mirror descent with adaptive active sets, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def exact_score_a(a):

    if S < 1e-12:

    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:

    return float(2.0 * n * cmax / (S * S))

def lse_score_grad(a, beta):

    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:

        return 1e10, np.zeros(n, dtype=np.float64)

def run_lbfgs(a0, beta, deadline, maxfun):

def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)

def active_weights_soft(d, temperature):
    """Soft active set based on temperature."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    dmax = float(np.max(d))
    z = (d - dmax) / max(temperature, 1e-10)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return active_weights_topk(d, 3)
    return e / s

def mirror_descent_step(a, w, t_power):
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)

def subgrad_step(a, w, eta):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:

        # Current autocorrelation
        c = np.fft.irfft(A * A, N)[:2*n-1]

        log_c = np.log(c)
        log_target = np.log(c_mean)
        # In frequency domain, we want to reduce the variance of |A|^2
        current_powers = np.abs(A) ** 2
        # Target powers: we want FFT(c_target) where c_target is flat
        c_target = np.zeros(N)
        c_target[:2*n-1] = c_mean
        target_powers = np.real(np.fft.rfft(c_target, N))
        target_powers = np.maximum(target_powers, 0)
        # Blend in log domain
        log_current = np.log(current_powers + 1e-30)
        log_target = np.log(target_powers + 1e-30)
        log_new = (1 - blend) * log_current + blend * log_target
        new_powers = np.exp(log_new)
        # Preserve DC
        new_powers[0] = current_powers[0]
        new_magnitudes = np.sqrt(new_powers)
        phases = np.angle(A)
        A_new = new_magnitudes * np.exp(1j * phases)
        a_curr = np.fft.irfft(A_new, N)[:n]
        a_curr = np.maximum(a_curr, 0)
        if a_curr.sum() < 1e-12:
            break
    return normalize_a(a_curr)

    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - float(np.max(log_new)))
    return normalize_a(a_new)


def subgrad_step(a, w, eta):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    h = corr_pair(a, w)
    d, S = d_from_a(a)
    LDS = float(np.dot(w, d))
    grad = (4.0 * n * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return normalize_a(np.maximum(a - eta * grad, 0.0))


def spectral_perturb(a, rng, n_gs_iters=10, mode=0):
    """Spectral domain perturbation for creating diverse restarts."""
    n = len(a)
    if n == 0:
        return a
    N = next_power_of_two(2 * n - 1)
    a_curr = a.copy()
    for it in range(n_gs_iters):
        A = np.fft.rfft(a_curr, N)
        powers = np.abs(A) ** 2
        dc = powers[0]
        if mode == 0:
            phases = np.angle(A)
            perturb = rng.uniform(-0.3, 0.3, size=len(phases))
            A_new = np.sqrt(powers) * np.exp(1j * (phases + perturb))
        elif mode == 1:
            c = np.fft.irfft(A * A, N)[:2 * n - 1]
            c = np.maximum(c, 1e-30)
            S = float(a_curr.sum())
            c_mean = S * S / (2 * n - 1)
            c_target = np.zeros(N)
            c_target[:2 * n - 1] = c_mean
            target_powers = np.real(np.fft.rfft(c_target, N))
            target_powers = np.maximum(target_powers, 0)
            blend = 0.25
            new_powers = (1.0 - blend) * powers + blend * target_powers
            new_powers[0] = dc
            new_mags = np.sqrt(np.maximum(new_powers, 0))
            phases = np.angle(A)
            A_new = new_mags * np.exp(1j * phases)
        elif mode == 2:
            freq_idx = np.arange(len(powers), dtype=np.float64)
            boost = 1.0 + 0.3 * (freq_idx / max(1, len(powers) - 1))
            new_powers = powers * boost
            new_powers[0] = dc
            new_mags = np.sqrt(np.maximum(new_powers, 0))
            phases = np.angle(A)
            A_new = new_mags * np.exp(1j * phases)
        else:
            new_powers = powers.copy()
            new_powers[1::2] *= 0.8
            new_powers[0] = dc
            new_mags = np.sqrt(np.maximum(new_powers, 0))
            phases = np.angle(A)
            A_new = new_mags * np.exp(1j * phases)
        a_curr = np.fft.irfft(A_new, N)[:n]
        a_curr = np.maximum(a_curr, 0.0)
        if a_curr.sum() < 1e-12:
            break
    return normalize_a(a_curr)


def time_adaptive_mirror_descent(a, deadline, max_iters=300):
    """Mirror descent with adaptive soft/hard active sets."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    k_list = (1, 2, 3, 5, 8, 12)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if d.size == 0 or S < 1e-12:
            break
        dmax = float(np.max(d))
        if dmax <= 0:
            break
        progress = float(it) / float(max(1, max_iters))
        temp = dmax * (0.3 * (1.0 - progress) + 0.01)

        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            a_before = a.copy()
            for use_soft in (False, True):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                if use_soft:
                    w = active_weights_soft(d, temp)
                else:
                    w = active_weights_topk(d, k)
                for t_p in (0.03, 0.08, 0.15):
                    if deadline is not None and time.monotonic() > float(deadline):
                        break
                    q = mirror_descent_step(a, w, t_p)
                    sc = exact_score_a(q)
                    if np.isfinite(sc) and sc < best_sc - 1e-10:
                        best_sc = sc
                        best_a = q.copy()
                        a = q.copy()
                        d, S = d_from_a(a)
                        dmax = float(np.max(d))
                        if dmax <= 0:
                            break

    return float(best_sc), best_a
