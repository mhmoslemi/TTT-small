import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    Compute LSE-smoothed objective and its gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient: df/da[i] = 2n * (h[i]/S^2 - 2L/S^3 * a[i])
    # where h[i] = 2 * cross_corr(p, a)[i] = 2 * conv(a_rev, p)[i + n - 1]
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2 + n - 1]
    h = 2.0 * h_full[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L * a / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)
        
        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def _active_set_deadline():
    """Global deadline for active set CD."""
    return getattr(_active_set_cd, '_deadline', time.monotonic() + 99999)


def interpolate_sequence(parent, target_n):
    """Interpolate a sequence to a different length."""
    if len(parent) == target_n:
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, target_n)
    a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    return a


def stochastic_restarts(a, rng, n_trials=8, beta=5000, maxiter=3000):
    """Random perturbations followed by quick L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for r in range(n_trials):
        # Different perturbation strategies
        if r % 4 == 0:
            a_pert = a * (1.0 + 0.05 * rng.standard_normal(n))
        elif r % 4 == 1:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = int(rng.integers(1, 8))
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.08 * np.cos(2 * np.pi * freq * x + phase))
        elif r % 4 == 2:
            # Block perturbation
            block = max(1, n // 16)
            start = int(rng.integers(0, n - block))
            a_pert = a.copy()
            a_pert[start:start + block] *= (1.0 + 0.15 * rng.standard_normal(block))
        else:
            # Mirror perturbation (breaks symmetry)
            a_pert = a.copy()
            half = n // 2
            a_pert[half:] *= (1.0 + 0.03 * rng.standard_normal(half))
        
        a_pert = np.maximum(a_pert, 0.0)
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 20)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Restart {r}: {sc:.8f}")
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase optimization:
    1. Fienup + initial L-BFGS for quick improvements
    2. Adaptive high-beta L-BFGS (main workhorse)
    3. Multiplicative peak shaving
    4. Active-set coordinate descent
    5. Stochastic restarts
    6. Final polish
    Tries multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 90.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(2048) * 0.5
    
    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.8f}")
        update(sc_p, parent, "parent")
    else:
        print("No parent, starting from uniform n=2048")
    
    print(f"Initial best: {best_score:.8f}")
    
    # Determine which lengths to try
    parent_n = len(parent) if parent is not None else 2048
    # Try parent length and one larger
    lengths_to_try = [parent_n]
    if parent_n <= 3000:
        lengths_to_try.append(min(parent_n * 2, 4096))
    if parent_n > 4096:
        lengths_to_try.append(4096)
    
    for trial_n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        print(f"\n--- Trying n={trial_n} ---")
        
        # Get initial sequence for this length
        if parent is not None:
            a_curr = interpolate_sequence(parent, trial_n)
        else:
            a_curr = np.ones(trial_n) / trial_n
        
        sc_curr = compute_score(a_curr)
        update(sc_curr, a_curr, f"init n={trial_n}")
        
        # Phase 1: Quick Fienup
        if time.monotonic() < deadline - 300:
            a_f = fienup_flat(a_curr, n_iters=200)
            sc = compute_score(a_f)
            update(sc, a_f, f"FIENUP n={trial_n}")
            if sc < sc_curr:
                a_curr = a_f
        
        # Phase 2: L-BFGS with increasing beta (main workhorse)
        # Time allocation: ~50% of total for this phase
        beta_schedule = [50, 200, 1000, 5000, 20000, 100000, 500000]
        
        for i, beta in enumerate(beta_schedule):
            if time.monotonic() > deadline - 120:
                break
            
            remaining = deadline - 90 - time.monotonic()
            if remaining < 15:
                break
            
            # Allocate more time to higher beta runs
            time_per_run = min(remaining / (len(beta_schedule) - i), 
                             0.12 * (total_budget - 90))
            maxiter = int(min(30000, max(2000, time_per_run * 12)))
            
            print(f"  L-BFGS beta={beta}, maxiter={maxiter} (remaining={remaining:.0f}s)")
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(a_curr, beta, maxiter=maxiter, maxfun=maxiter * 25)
            dt = time.monotonic() - t0
            print(f"    score={sc:.8f} ({dt:.1f}s)")
            
            if sc < best_score:
                update(sc, a_opt, f"LBFGS-{beta} n={trial_n}")
                a_curr = a_opt
            elif sc < compute_score(a_curr):
                # Keep the better of the two even if not global best
                a_curr = a_opt
        
        # Phase 3: Multiplicative peak shaving
        if time.monotonic() < deadline - 60:
            print(f"  Multiplicative peak shave (n={trial_n})")
            a_mp = multiplicative_peak_shave(a_curr, n_iters=500, beta=200)
            sc = compute_score(a_mp)
            update(sc, a_mp, f"MULTI-SHAVE n={trial_n}")
            if sc < compute_score(a_curr):
                a_curr = a_mp
        
        # Phase 4: L-BFGS again after shaving
        if time.monotonic() < deadline - 45:
            remaining = deadline - 90 - time.monotonic()
            if remaining > 10:
                maxiter = int(min(15000, remaining * 10))
                sc, a_opt = run_lbfgs(a_curr, 100000, maxiter=maxiter, maxfun=maxiter * 20)
                update(sc, a_opt, f"LBFGS-POSTSHAVE n={trial_n}")
                if sc < compute_score(a_curr):
                    a_curr = a_opt
    
    # Phase 5: Active-set coordinate descent on best
    if time.monotonic() < deadline - 30:
        print(f"\n--- Active-set CD ---")
        _active_set_cd._deadline = deadline - 30
        a_cd = active_set_cd(best_a, n_sweeps=50, top_frac=0.1)
        sc = compute_score(a_cd)
        update(sc, a_cd, "ACTIVE-SET-CD")
    
    # Phase 6: Stochastic restarts
    if time.monotonic() < deadline - 60:
        print(f"\n--- Stochastic restarts ---")
        a_sr, sc_sr = stochastic_restarts(best_a, rng, n_trials=6, beta=20000, maxiter=4000)
        update(sc_sr, a_sr, "STOCHASTIC")
    
    # Phase 7: Final L-BFGS polish
    if time.monotonic() < deadline - 20:
        print(f"\n--- Final polish ---")
        remaining = deadline - 20 - time.monotonic()
        if remaining > 5:
            maxiter = int(min(20000, remaining * 8))
            sc, a_fin = run_lbfgs(best_a, 200000, maxiter=maxiter, maxfun=maxiter * 15)
            update(sc, a_fin, "FINAL-LBFGS-200k")
    
    if time.monotonic() < deadline - 10:
        sc, a_fin = run_lbfgs(best_a, 1000000, maxiter=8000, maxfun=120000)
        update(sc, a_fin, "FINAL-LBFGS-1M")
    
    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        _active_set_cd._deadline = deadline - 2
        a_cd2 = active_set_cd(best_a, n_sweeps=15)
        sc = compute_score(a_cd2)
        update(sc, a_cd2, "FINAL-CD")
    
    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
