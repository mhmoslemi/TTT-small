The LP is:
maximize sum(b)
subject to: (a * b)[k] <= M for k = 0, ..., 2n-2
            b[j] >= 0

This is a "convolutional" LP. The constraint matrix is a banded Toeplitz matrix. For large n, we can exploit this structure.

A simple iterative approach:
1. Start with b = 1 (or b = a)
2. Compute c = a * b (full convolution)
3. Scale: b_new = b * M / max(c) (to satisfy the most violated constraint)
4. Actually this doesn't work directly because we need ALL constraints satisfied.

Better: projected subgradient / von Neumann alternating projections:
1. b = 1
2. Repeat:
   a. Compute c = a * b
   b. Find all violations: c[k] > M
   c. For each violated k, reduce b[j] for j that contribute to c[k]
   d. Project b onto b >= 0

This is O(n^2) per iteration in the worst case, but with FFT we can compute c in O(n log n). The reduction step is the expensive part.

Alternative: use the dual. The dual is:
minimize M * sum(mu)
subject to: (a * mu^T)[j] >= 1 for j = 0, ..., n-1  (where mu^T is the "time-reversed" mu)
            mu[k] >= 0

Here (a * mu^T)[j] = sum_k a[k] * mu[j-k] (with appropriate bounds).

Starting with mu = 0, we want to increase mu where the dual constraints are most violated. This is like a "gradient ascent" on the dual.

For a single iteration:
1. Compute d[j] = sum_k a[k] * mu[j-k] for all j (this is a*b-like computation)
2. Find j where d[j] < 1 (most violated)
3. Increase mu at positions that help satisfy d[j] >= 1
4. mu[k] += step * (contribution of mu[k] to d[j])

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """
    Compute the cross-correlation: result[j] = sum_k w[k] * a[k+j] 
    (shifted correlation, same as the gradient of the dot product w·(a*a) w.r.t. a)
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(n)
    M = n + m - 1
    N = next_pow2(M)
    # We want: result[j] = sum_k w[k] * a[k-j] for valid k
    # This is the same as: full correlation of a and w, centered at the n-1 position
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

    S = a.sum()

    c = fast_self_conv(a)
    return (n * c) / (S * S)

def run_lbfgs(a0, beta, maxfun, deadline):
    a = normalize_a(a0)
    n = len(a)
    bounds = [(0.0, 1e4)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": max(50, maxfun // 2),
        "maxfun": max(64, maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(beta,), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    
    # Check deadline
    if deadline is not None and time.monotonic() > deadline:
        a_out = a
    
    return exact_score(a_out), a_out


def resample_array(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, a)


def estimate_convert_time(n):
    """Estimate time for one convolution of length n."""
    return 0.01 * (n / 8192.0) * (np.log2(max(16, n)) / 13.0)

def lse_trajectory(a_start, deadline):
    """Run LSE optimization with increasing beta."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    
    stages = [
        (100.0, 0.12),
        (500.0, 0.12),
        (2000.0, 0.12),
        (10000.0, 0.13),
        (50000.0, 0.13),
        (200000.0, 0.14),
        (500000.0, 0.14),
        (2000000.0, 0.10),
    ]
    
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0))
    cum = 0.0
    
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + min(cum, 1.0) * total
        sub_dead = min(sub_dead, deadline)
        
        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 8.0:
                break
            est_t = estimate_convert_time(n)
            mf = max(64, min(2000, int(rem * 0.6 / max(est_t, 0.005))))
            if mf < 64:
                break
            
            sc, a2 = run_lbfgs(a, beta, mf, sub_dead)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)
    
    return best_sc, best_a


def power_iter_phase(a_start, deadline, k_list=None):
    """Active-set power iteration: a -> corr(a, w_active) to find fixed point."""
    if k_list is None:
        k_list = [1, 2, 3, 5, 8, 13, 21, 34, 55]
    
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)
    stagnation = 0
    
    while time.monotonic() < deadline - 5.0:
        improved = False
        d = d_from_a(a)
        if d.size == 0:
            break
        dmax = d.max()
        if dmax <= 0:
            break
        
        for k in k_list:
            if time.monotonic() > deadline - 5.0:
                break
            if k >= d.size:
                continue
            
            w = active_weights_topk(d, k)
            h = corr_pair(a, w)
            hmax = h.max()
            hmin = h.min()
            if hmax <= 0:
                continue
            
            # Try different update rules
            for rule in range(4):
                for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                    q = None
                    if rule == 0:
                        # Linear blend
                        h_norm = h / hmax
                        q = normalize_a((1.0 - t) * a + t * h_norm)
                    elif rule == 1:
                        # Geometric blend (in log space)
                        log_a = np.log(a + 1e-30)
                        log_h = np.log(np.maximum(h, 1e-30))
                        log_q = (1.0 - t) * log_a + t * (log_h - log_h.max())
                        q = normalize_a(np.exp(log_q))
                    elif rule == 2:
                        # Power iteration (full replacement with small noise)
                        q = normalize_a(np.maximum(h, 1e-15) + 1e-10 * np.random.default_rng(it).standard_normal(n))
                    elif rule == 3:
                        # Square root blend: a_new = sqrt(a * h)
                        q = normalize_a(np.sqrt(np.maximum(a, 0) * np.maximum(h, 0) + 1e-30))
                    
                    if q is None:
                        continue
                    sc = exact_score(q)
                    if np.isfinite(sc) and sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = q.copy()
                        a = q.copy()
                        improved = True
                        stagnation = 0
                        break
        
        if not improved:
            stagnation += 1
            if stagnation > 3:
                break
    
    return best_sc, best_a


def exact_subgrad_polish(a_start, deadline, k_list=(1, 2, 3, 5, 8)):
    """Polish using exact max subgradient."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for it in range(200):
        if time.monotonic() > deadline - 5.0:
            break
        improved = False
        d = d_from_a(a)
        if d.size == 0 or d.max() <= 0:
            break
        
        for k in k_list:
            if time.monotonic() > deadline - 5.0:
                break
            w = active_weights_topk(d, k)
            h = corr_pair(a, w)
            hmax = max(abs(h).max(), 1e-15)
            
            for eta in (0.001, 0.005, 0.01, 0.02, 0.05):
                # Gradient direction: reduce where h is large
                a_trial = normalize_a(np.maximum(a - eta * h / hmax, 0.0))
                sc = exact_score(a_trial)
                if np.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
        
        if not improved and it > 10:
            break
    
    return best_sc, best_a


def perturb_seq(a, rng, strength=0.2):
    """Generate a perturbation of the sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    mode = int(rng.integers(0, 6))
    
    if mode == 0:
        # Multiplicative noise
        q = a * np.exp(strength * 0.3 * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.1, 0.4))
        q = (1-t) * a + t * (1.0/n)
    elif mode == 2:
        # Reverse
        q = a[::-1].copy()
        q *= np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Smoothed noise in log space
        logp = np.log(a + 1e-30)
        noise = rng.standard_normal(n)
        k = max(3, n // 512 | 1)
        if k < 3: k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c)**2) / (2.0*sig*sig))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        q = np.exp(logp + 0.1 * strength * (noise - noise.mean()))
    elif mode == 4:
        # V-shape modulation
        x = np.linspace(0, 1, n)
        mod = 1.0 + 0.3 * strength * np.abs(2*x - 1)
        q = a * mod
    else:
        # Random smooth shape
        Nf = next_pow2(n)
        freqs = np.fft.rfftfreq(Nf) * Nf
        alpha = float(rng.uniform(1.5, 4.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** alpha)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        noise = np.fft.irfft(power * np.exp(1j*phase), Nf)[:n]
        q = a * (1.0 + 0.3 * strength * (noise - noise.mean()) / (noise.std() + 1e-15))
    
    return normalize_a(np.clip(q, 0, None))


def generate_structured_inits(n, rng, parent):
    """Generate diverse initial candidates."""
    inits = []
    x = np.linspace(0, 1, n)
    
    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_array(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = arr.min()
        if mn < 0:
            arr = arr - mn + 0.01
        inits.append(normalize_a(arr))
    
    if parent is not None and len(parent) >= 64:
        pr = resample_array(parent, n)
        add(pr)
        add(pr[::-1])
    
    add(np.ones(n))
    
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2*x - 1))
        add(1.0 + alpha * (2*x - 1)**2)
    
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2*np.pi*f*x))
    
    for width in (0.1, 0.2, 0.3):
        add(np.exp(-((x-0.5)/width)**2) + np.exp(-((x-0.25)/width)**2) + np.exp(-((x-0.75)/width)**2))
    
    # Spectral inits
    Nf = next_pow2(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        alpha = float(rng.uniform(1.5, 5.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** alpha)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        noise = np.fft.irfft(power * np.exp(1j*phase), Nf)[:n]
        add(1.0 + 0.5 * (noise - noise.mean()) / (noise.std() + 1e-15))
    
    return inits[:min(20, len(inits))]
