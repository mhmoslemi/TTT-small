import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def active_weights_topk(d, k, soft_width=0.0):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    if soft_width > 0.0:
        idx = np.argpartition(d, m - k)[m - k:]
        d_top = d[idx]
        dmin_top = float(d_top.min())
        w = np.zeros(m, dtype=np.float64)
        w[idx] = np.exp((d_top - dmin_top) / soft_width)
        return w / float(w.sum())
    else:
        idx = np.argpartition(d, m - k)[m - k:]
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0
        return w / float(k)


def exact_subgrad_step(a, k_active, eta):
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a
    grad = (4.0 * float(len(a)) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    return a - eta * grad


def mirror_descent_step(a, k_active, t_power, soft_width=0.0):
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active, soft_width)
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def lag_equalization(a, max_iters=200, k_list=(2, 3, 4, 5, 8, 12), deadline=None):
    """Iteratively equalize the top-k convolution lags by alternating
    between reducing the maximum lag and boosting the second-max lag."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if d.size == 0:
            break
        dmax = float(np.max(d))
        if dmax <= 0:
            break

        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            for eta in (0.002, 0.005, 0.01, 0.02, 0.05):
                q = exact_subgrad_step(a, k, eta)
                q = normalize_a(np.maximum(q, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q
        if not improved and it > 10:
            break

    return float(best_sc), best_a


def spectral_shaping(a, max_iters=50, deadline=None):
    """Work in Fourier domain: try to flatten |A(f)|^2 to reduce peak autocorrelation."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    N = next_power_of_two(2 * n - 1)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        # Try to make power more uniform (reduce peak)
        # Compute target: geometric mean as proxy for "flat" spectrum
        log_power = np.log(np.maximum(power, 1e-30))
        mean_lp = float(np.mean(log_power))
        # Damp peaks in spectrum
        for damp in (0.02, 0.05, 0.1, 0.2):
            new_log_power = log_power - damp * np.maximum(log_power - mean_lp, 0)
            new_power = np.exp(new_log_power)
            # Reconstruct a from modified power (keep phase)
            phase = np.angle(A)
            new_A = np.sqrt(new_power) * np.exp(1j * phase)
            new_a = np.fft.irfft(new_A, N)[:n]
            new_a = np.maximum(new_a, 0.0)
            new_a = normalize_a(new_a)
            sc = exact_score_a(new_a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new_a.copy()
                a = new_a.copy()
                break
    return float(best_sc), best_a


def generate_diverse_inits(n, rng, parent, cap=16):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    x = np.linspace(0.0, 1.0, n)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    # V-shapes (keys for this problem)
    for alpha in (0.2, 0.4, 0.6, 0.8, 1.2, 1.8, 2.5):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for p in (0.3, 0.5, 0.7, 1.5, 2.0, 3.0):
        add(1.0 + 1.0 * np.abs(2.0 * x - 1.0) ** p)

    # Power law at edges
    for gamma in (0.5, 1.0, 1.5, 2.0, 3.0):
        d = np.abs(2.0 * x - 1.0)
        d = np.maximum(d, 1e-6)
        add(d ** (-gamma) * 0.1 + 1.0)

    # Cosine / periodic
    for f in (1, 2, 3):
        for alpha in (0.3, 0.7, 1.2):
            add(1.0 + alpha * np.cos(np.pi * x))
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))

    # Bimodal
    for width in (0.08, 0.12, 0.18, 0.25):
        add(np.exp(-(((x - 0.0) / width) ** 2)) + np.exp(-(((x - 1.0) / width) ** 2)))
        add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)))

    # Exponential decay
    for rate in (2.0, 4.0, 8.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Sine-based
    for f in (1, 2, 3, 5):
        add(1.0 + 0.8 * np.abs(np.sin(np.pi * f * x)) ** 0.5)

    # Random smooth
    Nf = next_power_of_two(n)
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power_spec = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 3.0)))
        power_spec[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power_spec))
        spec = power_spec * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 0.8)) * noise)

    return inits


def perturb_best(a, rng, strength=None):

        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 300))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * 0.3 * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 300)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    elif mode == 7:
        # Targeted: reduce values at positions contributing most to peak
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_topk(d, 3)
                h = corr_pair(p, w)
                hmax = float(np.max(h) + 1e-15)
                q = p * np.exp(-strength * 0.4 * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 8:
        # Fourier domain perturbation
        N = next_power_of_two(2 * n - 1)
        A = np.fft.rfft(p, N)
        # Add small random phase perturbation
        phase_pert = rng.standard_normal(len(A)) * strength * 0.1
        A_new = A * np.exp(1j * phase_pert)
        q = np.fft.irfft(A_new, N)[:n]
    else:
        # Local random edits
        q = p.copy()
        n_edits = int(rng.integers(n // 50, n // 10))
        idx = rng.integers(0, n, size=n_edits)
        q[idx] *= np.exp(rng.standard_normal(n_edits) * strength * 0.5)

    return normalize_a(np.clip(q, 0.0, None))


def optimize_pipeline(a_start, deadline, label=""):
    """Full optimization pipeline for one starting sequence."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    # Phase A: LSE with increasing beta
    stages = [
        (200.0, 0.12),
        (500.0, 0.10),
        (1000.0, 0.10),
        (3000.0, 0.10),
        (8000.0, 0.10),
        (20000.0, 0.10),
        (50000.0, 0.10),
        (100000.0, 0.10),
        (300000.0, 0.08),
    ]

    t0 = time.monotonic()
    remaining = float(deadline - t0)
    if remaining < 10.0:
        return best_sc, best_a

    total_lse_time = 0.55 * remaining
    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 0.55:
            cum = 0.55
        sub_dead = t0 + cum * remaining
        sub_dead = min(sub_dead, deadline - 10.0)

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est_time = estimate_eval_time(n)
            mf = int(min(2000, max(64, (rem * 0.7) / max(est_time, 1e-4))))
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = normalize_a(a2)

    # Phase B: Lag equalization
    if time.monotonic() < deadline - 30.0:
        sc, seq = lag_equalization(best_a, max_iters=100,
                                   k_list=(2, 3, 4, 5, 8, 12),
                                   deadline=deadline - 30.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase C: Spectral shaping
    if time.monotonic() < deadline - 20.0:
        sc, seq = spectral_shaping(best_a, max_iters=30, deadline=deadline - 20.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase D: Mirror descent polishing
    if time.monotonic() < deadline - 15.0:
        a = normalize_a(best_a)
        k_list = (1, 2, 3, 5, 8)
        for _ in range(50):
            if time.monotonic() > deadline - 15.0:
                break
            improved = False
            for k in k_list:
                for t_p in (0.01, 0.03, 0.05, 0.1):
                    if time.monotonic() > deadline - 15.0:
                        break
                    q = mirror_descent_step(a, k, t_p)
                    sc = exact_score_a(q)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = q.copy()
                        a = q.copy()
                        improved = True
            if not improved:
                break

    return float(best_sc), best_a


def subgrad_polish_exact(a, max_iters=80, deadline=None, k_list=(1, 2, 3, 4, 5, 7, 10)):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            for eta in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = exact_subgrad_step(a, k, eta)
                q = normalize_a(np.maximum(q, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q
        if not improved and it > 10:
            break
    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent available: {}, length={}".format(
        parent is not None, parent.size if parent is not None else 0))

    # Primary length
    n_primary = 10240
    if parent is not None:
        n_primary = min(parent.size, 16384)
        if n_primary < 2048:
            n_primary = 2048
        # Round to nice number for FFT efficiency
        n_primary = 1024 * max(1, int(round(n_primary / 1024.0)))

    print("Using primary length: {}".format(n_primary))

    # Generate initializations
    inits = generate_diverse_inits(n_primary, rng, parent, cap=12)

    best_sc = np.inf
    best_seq = None

    # --- Phase 1: Optimize the best 3 initializations (55% time) ---
    t1_end = t_start + 0.55 * total_budget
    top_inits = inits[:3]
    time_per_init = (t1_end - t_start) / len(top_inits)

    for i, a0 in enumerate(top_inits):
        sub_dead = t_start + (i + 1) * time_per_init
        sc, seq = optimize_pipeline(a0, sub_dead, "init_{}".format(i))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 1 init {}: score={:.8f}".format(i, best_sc))

    if best_seq is None:
        best_seq = inits[0].copy()
        best_sc = exact_score_a(best_seq)

    # --- Phase 2: Optimize 2 more initializations (15% time) ---
    t2_end = t_start + 0.70 * total_budget
    more_inits = inits[3:5]
    time_per_init2 = (t2_end - t1_end) / max(1, len(more_inits))
    for i, a0 in enumerate(more_inits):
        sub_dead = t1_end + (i + 1) * time_per_init2
        sc, seq = optimize_pipeline(a0, sub_dead, "init_{}_b".format(i + 3))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 2 init {}: score={:.8f}".format(i + 3, best_sc))

    # --- Phase 3: Basin hopping from best (15% time) ---
    t3_end = t_start + 0.85 * total_budget
    n_restarts = 4
    time_per_restart = (t3_end - t2_end) / n_restarts

    for r in range(n_restarts):
        sub_dead = t2_end + (r + 1) * time_per_restart
        p = perturb_best(best_seq, rng, strength=float(rng.uniform(0.03, 0.15)))
        sc, seq = optimize_pipeline(p, sub_dead, "restart_{}".format(r))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 3 restart {}: score={:.8f}".format(r, best_sc))

    # --- Phase 4: Multi-length exploration (8% time) ---
    t4_end = t_start + 0.93 * total_budget
    other_lengths = []
    if n_primary != 8192:
        other_lengths.append(8192)
    if n_primary != 12288:
        other_lengths.append(12288)
    if n_primary != 16384:
        other_lengths.append(16384)

    for n_other in other_lengths[:2]:
        if time.monotonic() > t4_end - 20.0:
            break
        a_other = normalize_a(resample_sequence(best_seq, n_other))
        sc, seq = optimize_pipeline(a_other, t4_end - 10.0, "n={}".format(n_other))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 4 n={}: score={:.8f}".format(n_other, best_sc))

    # --- Phase 5: Final intensive polishing (7% time) ---
    t5_end = deadline
    a_final = normalize_a(best_seq)

    # High-beta LSE push
    for beta in (500000.0, 1000000.0, 2000000.0):
        if time.monotonic() > t5_end - 15.0:
            break
        rem = float(t5_end - time.monotonic())
        if rem < 10.0:
            break
        est_time = estimate_eval_time(len(a_final))
        mf = int(min(1000, max(100, (rem * 0.7) / max(est_time, 1e-4))))
        sc, seq = run_lbfgs(a_final, beta, t5_end - 10.0, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 5 high-beta {:.0f}: score={:.8f}".format(beta, best_sc))
        a_final = normalize_a(seq)

    # Subgradient polish
    if time.monotonic() < t5_end - 15.0:
        sc, seq = subgrad_polish_exact(best_seq, max_iters=40, deadline=t5_end - 10.0,
                                       k_list=(1, 2, 3, 4, 5, 7, 10))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Phase 5 subgrad: score={:.8f}".format(best_sc))

    # Final mirror descent
    if time.monotonic() < t5_end - 5.0:
        a = normalize_a(best_seq)
        for _ in range(30):
            if time.monotonic() > t5_end - 5.0:
                break
            improved = False
            for k in (1, 2, 3, 5):
                for t_p in (0.005, 0.01, 0.02, 0.05):
                    if time.monotonic() > t5_end - 5.0:
                        break
                    q = mirror_descent_step(a, k, t_p)
                    sc = exact_score_a(q)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_seq = q.copy()
                        a = q.copy()
                        improved = True
            if not improved:
                break

    # Final verification and cleanup
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
