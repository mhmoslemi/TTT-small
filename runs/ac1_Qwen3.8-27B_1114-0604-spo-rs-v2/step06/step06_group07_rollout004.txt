def fienup_geometric(a, n_iters=300, alpha_start=0.1, alpha_end=0.9):

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def lse_objective_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: g_L[i] = 2 * sum_k p[k] * d(c_k)/d(x_i)
    # d(c_k)/d(x_i) = x[k-i] (if 0<=k-i<n) + x[i] (if 0<=k-i<n)... 
    # Actually d(c_k)/d(x_i) = x[k-i] (from j=i term) + x[k-i] (from j=k-i term)
    # = 2*x[k-i] if k-i != i, else x[i]
    # More carefully: c_k = sum_j x_j x_{k-j}
    # d(c_k)/d(x_i) = x_{k-i} (j=i) + sum_j [k-j=i] x_j = x_{k-i} + x_{k-i} = 2*x_{k-i} (if k-i != i)
    # If k-i = i: d(c_k)/d(x_i) = x_i + x_i = 2*x_i (but it's the same term, so just x_i)
    # Actually for quadratic forms: c_k = x^T T_k x where T_k[i,j] = 1 if i+j=k (with symmetrization)
    # d(c_k)/d(x_i) = 2*(T_k x)_i (for symmetric T_k)
    # (T_k x)_i = sum_j T_k[i,j] x_j = x_{k-i} (if 0<=k-i<n)
    # So d(c_k)/d(x_i) = 2*x_{k-i} (if 0<=k-i<n), except when 2i=k where it's x_i
    # For the gradient computation using convolution:
    # sum_k p[k] * 2*x_{k-i} = 2 * (p * x_rev)[i + n - 1] ... let me just use the convolution approach
    
    # g_L[i] = 2 * sum_k p[k] * (d c_k / d x_i)
    # For most i, d c_k/d x_i = 2*x[k-i] (for valid k-i)
    # This equals 2 * conv(p, x_rev)[i + n - 1] approximately
    # But the k=2i correction... let me just use the FFT approach which handles it correctly.
    
    # Alternative: compute gradient via FFT
    # g_L = 2 * FFT^{-1}(2 * Re(FFT(p) * conj(FFT(x)))) ... no, that's not right either.
    
    # Let me just do it directly with the Toeplitz structure:
    # g_L[i] = 2 * sum_k p[k] * (x[k-i] + [k-i==i] * 0) ... 
    # Actually, the simplest correct formula:
    # g_L[i] = sum_k p[k] * 2 * x[k-i] (for 0<=k-i<n) 
    # minus the correction for k=2i where it should be x_i not 2*x_i
    # = 2 * sum_k p[k] * x[k-i] - sum_k p[k] * [k==2i] * x[i]
    # = 2 * conv(p, x_rev)[i+n-1] - p[2i] * x[i] (if 2i < M)
    
    x_rev = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(x_rev, M2_fft)
    PP = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * PP, M2_fft)[n-1:2*n-1]
    
    g_L = 2.0 * conv_res
    # Correction for k=2i:
    valid_2i = 2 * np.arange(n)
    valid_mask = valid_2i < M
    g_L[valid_mask] -= p[valid_2i[valid_mask]] * x[valid_mask]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    res = minimize(
        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True,
        bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
