Multi-phase optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses spectral shaping, LSE annealing, Nesterov-accelerated subgradient,
and multi-n exploration to push below 1.503.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad

def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):

def spectral_shape(a, n_iters=200, mode='flat'):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        
        if mode == 'flat':
            alpha = 0.1 + 0.6 * progress
            P_mean = float(P.mean())
            P_new = (1.0 - alpha) * P + alpha * P_mean
        elif mode == 'quantile':
            q = 0.6 + 0.35 * progress
            thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - progress))
            P_new = np.minimum(P, thresh)
        elif mode == 'targeted':
            # Target specific frequency bins that contribute to the peak
            c = fast_conv(a)
            k_peak = int(np.argmax(c))
            # The frequency that contributes to the peak at lag k_peak
            # is f = k_peak / (2n-1)
            # Reduce power near that frequency
            freq_idx = k_peak * (M_fft // 2) // (2 * n - 1)
            bandwidth = max(3, M_fft // (8 * n))
            lo = max(0, freq_idx - bandwidth)
            hi = min(len(P), freq_idx + bandwidth + 1)
            P_new = P.copy()
            P_new[lo:hi] *= (1.0 - 0.3 * (1.0 + progress))
            # Also flatten overall
            alpha = 0.1 + 0.3 * progress
            P_mean = float(P_new.mean())
            P_new = (1.0 - alpha) * P_new + alpha * P_mean
        else:
            P_new = P.copy()
        
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def nesterov_subgrad(a, n_iters=5000, top_k=10, lr=0.01, momentum=0.9):
    n = len(a)
    a = normalize(a.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look
        
        c = fast_conv(a_look)
        S = float(a_look.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = normalize(a_look.copy())
        
        # Subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w * a_look[j_start:j_end+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        avg_w = float(top_k) / top_k  # average weight = 1 roughly
        # Actually compute average weight
        total_w = 0.0
        for k in top_idx:
            k = int(k)
            total_w += c[k] / (M_val + 1e-30)
        avg_w = total_w / top_k
        
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        # Update with momentum
        v = momentum * v + lr * grad
        a = a - v
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay learning rate
        lr *= 0.9999
    
    return best_a

def coord_descent(a, n_sweeps=10, n_top=30):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.85]:
                a[j] = a_old * f
                sc = compute_score(normalize(a))
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
