"""
Multi-method optimizer for minimizing 2n*max(conv(a,a))/S^2.
Uses LSE annealing, LP direction finding, subgradient peak shaving,
coordinate descent, and multi-restart strategies.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time
import sys

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    
    p = exp_s / exp_s.sum()
    
    # Gradient: g_i = 2 * sum_k p_k * a_{k-i}
    aR = a[::-1]
    N = 3 * n
    N_fft = 1 << (N - 1).bit_length()
    P = np.fft.rfft(p, N_fft)
    AR = np.fft.rfft(aR, N_fft)
    conv = np.fft.irfft(P * AR, N_fft)
    g = 2.0 * conv[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n / S2) * g - (4.0 * n * L) / S3
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return compute_score(a0), a0

def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    # Build constraint matrix A (2n-1) x n where A[k,j] = a[k-j] if 0 <= k-j < n
    # This is a banded matrix
    # Use scipy linprog
    
    # For large n, building the full matrix is expensive.
    # Use sparse representation.
    from scipy.sparse import diags, csr_matrix
    
    # The constraint matrix has a Toeplitz-like structure.
    # Row k: [0, ..., 0, a[k], a[k-1], ..., a[0], ..., a[k-n+1], 0, ..., 0]
    # Specifically, A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    # So for row k, the non-zero entries are at columns j where 0 <= k-j < n,
    # i.e., max(0, k-n+1) <= j <= min(k, n-1).
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = range(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(list(js))
        ks = np.array(js)
        vals.extend(a[k - ks].tolist())
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: min -sum(b) s.t. A_sparse @ b <= M, b >= 0
    c_obj = -np.ones(n)
    b_rhs = np.full(2*n-1, M)
    
    try:
        res = linprog(
            c_obj, A_ub=A_sparse, b_ub=b_rhs,
            bounds=[(0, None)] * n,
            method='highs'
        )
        if res.success:
            b = res.x
            return b
        else:
            print(f"  LP failed: {res.message}")
            return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None

def fienup(a, n_iters=200, mode='flat'):
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_mean = float(P.mean())
        
        if mode == 'flat':
            alpha = 0.2 + 0.5 * progress
            P_new = (1 - alpha) * P + alpha * P_mean
        elif mode == 'quantile':
            q = 0.7 + 0.25 * progress
            thresh = np.quantile(P, q)
            P_new = P.copy()
            P_new[P_new > thresh] = thresh
        else:  # hybrid
            if it % 3 == 0:
                alpha = 0.2 + 0.4 * progress
                P_new = (1 - alpha) * P + alpha * P_mean
            elif it % 3 == 1:
                q = 0.75 + 0.15 * progress
                thresh = np.quantile(P, q)
                P_new = P.copy()
                P_new[P_new > thresh] = thresh
            else:
                P_new = P.copy()
                low = P_new < P_mean * 0.5
                P_new[low] = P_new[low] / 1.3
        
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def subgrad_peak_shave(a, n_iters=3000, top_k=10, step_init=0.005):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = 1.0 / top_k
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1)
            sg[js] += w * a[k - js] * 2  # 2*a_{k-i}
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n / S2) * sg - (4.0 * n * M_val) / S3
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        lb = 2.0 * n / (2.0 * n - 1.0)
        step_polyak = max(0, sc - lb) / (gn * gn + 1e-30)
        step = min(step_polyak, 0.02)
        step = max(step, 1e-10)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999
    
    return best_a

def coord_descent(a, n_sweeps=5, n_top=20):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        js = np.arange(j_start, j_end + 1)
        contribs = 2.0 * a[js] * a[k_star - js]
        
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.85, 0.95]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
