"""
Step function minimization using multi-stage optimization with LSE-smoothed
gradient descent, spectral shaping with phase randomization, and peak-directed
refinement. Targets sequences of length 8192-16384 for better asymptotic behavior.
"""
import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_loss_grad(a, beta):
    """
    Compute LSE-smoothed loss and gradient.
    Loss = 2*n * L / S^2 where L = logsumexp(beta * conv(a,a))
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum  # softmax weights
    
    # Gradient of L w.r.t. a: dL/da_i = 2*beta * sum_k p[k]*a[k-i]
    # This is 2*beta * cross_correlation(p, a)[i] for i=0..n-1
    M2 = 1 << (M + n - 1).bit_length()  # >= M + n - 1 = 3n-2
    P = np.fft.rfft(p, M2)
    Av = np.fft.rfft(a, M2)
    gL = 2.0 * beta * np.fft.irfft(P * np.conj(Av), M2)[:n]
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (gL / S2 - 2.0 * L / S3)
    
    return score, grad


def fienup_shaped(a, n_iters=300, alpha_start=0.1, alpha_end=0.9,
                  mode='geometric', phase_rand=0.0, rng=None):
    """
    Fienup-style spectral shaping.
    mode: 'geometric', 'power_law', 'elliptical'
    phase_rand: probability of randomizing phase at each step
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        
        if mode == 'geometric':
            amp_new = amp_safe * (C / amp_safe) ** alpha
        elif mode == 'power_law':
            amp_new = amp.copy()
            above = amp > C
            if above.any():
                amp_new[above] = amp[above] * (C / amp[above]) ** alpha
        elif mode == 'elliptical':
            ratio = amp / C
            amp_new = amp / np.sqrt(1.0 + 3.0 * (ratio - 1.0) ** 2)
        else:
            amp_new = amp_safe * (C / amp_safe) ** alpha
        
        # Phase randomization
        if phase_rand > 0 and rng is not None:
            rand_mask = rng.random(len(phase)) < phase_rand
            phase[rand_mask] = rng.uniform(-np.pi, np.pi, size=rand_mask.sum())
        
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def peak_descent(a, n_iters=5000, n_peaks=5, step_init=0.01, decay=0.9999):
    """Peak-directed subgradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step: ensure we don't overshoot the lower bound
        max_step = max((sc - sc_lb) * 0.5 / (gn * gn + 1e-30), 1e-12)
        step = min(step, max_step)
        step = min(step, 0.03)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_loss_and_grad(a, beta):
    """
    Compute LSE-smoothed loss and gradient.
    Loss = 2*n*L / S^2 where L = logsumexp(beta * conv(a,a))
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    exp_sum = float(exp_shifted.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    p = exp_shifted / exp_sum

    # Gradient of L w.r.t. a_i = 2*beta * sum_k p[k]*a[k-i]
    # This is 2*beta * cross_correlation(p, a)[i]
    M2 = 1 << (M + n - 1).bit_length()
    P_ft = np.fft.rfft(p, M2)
    A_ft = np.fft.rfft(a, M2)
    corr = np.fft.irfft(P_ft * np.conj(A_ft), M2)[:n]
    gL = 2.0 * beta * corr

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (gL / S2 - 2.0 * L / S3)

    return score, grad


def fienup_shaped(a, n_iters, alpha_start, alpha_end, mode, phase_rand, rng):
    """
    Fienup-style spectral shaping with optional phase randomization.
    Returns the best sequence found.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)

        if mode == 'geometric':
            amp_new = amp_safe * (C / amp_safe) ** alpha
        elif mode == 'power_law':
            amp_new = amp.copy()
            above = amp > C * 1.001
            if above.any():
                amp_new[above] = amp[above] * (C / amp[above]) ** alpha
        elif mode == 'elliptical':
            ratio = amp / C
            amp_new = amp / np.sqrt(1.0 + 2.0 * (ratio - 1.0) ** 2)
        else:
            amp_new = amp_safe * (C / amp_safe) ** alpha

        if phase_rand > 0.0 and rng is not None:
            rand_mask = rng.random(len(phase)) < phase_rand
            if rand_mask.any():
                phase[rand_mask] = rng.uniform(-np.pi, np.pi, size=int(rand_mask.sum()))

        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)

        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_descent(a, n_iters, n_peaks, step_init, decay):
    """Peak-directed subgradient descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        max_step = max((sc - sc_lb) * 0.3 / (gn * gn + 1e-30), 1e-12)
        step = min(step, max_step)
        step = min(step, 0.02)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def lbfgs_run(a0, beta, deadline):
    """Run L-BFGS-B on LSE-smoothed objective respecting deadline."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 0.5
        S0 = n * 0.5
    a0 = a0 / S0

    bounds = [(0.0, 1000.0)] * n
    remaining = deadline - time.monotonic()
    maxfun = max(500, int(remaining * 15))
    maxiter = max(100, maxfun // 5)

    res = minimize(
        lse_loss_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )

    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    return a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.3 + 0.5 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-2.0 * (2.0 * x - 1.0) ** 2) + 0.03)

    freqs = rng.integers(1, 6, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    sig = np.ones(n) * 0.3
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)

    if parent is not None:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    return inits

                sc = compute_score(a_fp3)
                try_update(sc, a_fp3, f"LBFGS-FINAL n={n_work}")

    # Perturbation restarts with remaining time
    if time.monotonic() < deadline - 15:
        print(f"\n  Perturbation restarts (remaining={deadline - time.monotonic():.0f}s)")
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.03 * (1 + trial * 0.5)
            noise = noise_scale * rng.standard_normal(len(best_a))
            a_pert = best_a * (1.0 + noise)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S_pert = a_pert.sum()
            if S_pert < 1e-12:
                continue
            a_pert = a_pert / S_pert

            # Quick refinement
            a_ref = fienup_shaped(a_pert, n_iters=50, alpha_start=0.05, alpha_end=0.5,
                                  mode='geometric', phase_rand=0.0, rng=rng)
            sc = compute_score(a_ref)
            try_update(sc, a_ref, f"PERT {trial}")
            if time.monotonic() < deadline - 10:
                a_pd = peak_descent(a_ref, n_iters=500, n_peaks=5,
                                    step_init=0.005, decay=0.99998)
                sc = compute_score(a_pd)
                try_update(sc, a_pd, f"PERT-PD {trial}")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
