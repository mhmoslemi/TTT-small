    # Actually, to maintain sum exactly, we should do:
    # a_new = a - eta * grad + eta * mean(grad) * ones  (so sum is preserved)

    eta = adapt_step(a, grad)
    a_new = a - eta * grad
    a_new = clip(a_new, 0, None)
    S = a_new.sum()
    if S > 1e-12: a_new /= S
    else: a_new = a
    a = a_new

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative weights (mirror descent) with Barzilai-Borwein adaptive step
2. LSE-smoothed L-BFGS-B for smooth convergence phases

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400.0, 400.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _soft_subgrad(a, k_active, tol):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - max(tol * dmax, 1e-10)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _soft_max_grad(a, k_soft, temp):
    """Compute gradient using a soft-max active set with exponential weighting."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    # Soft weights: exp(d/temp) on top-k_soft
    idx_all = np.argsort(d)[::-1][:k_soft]
    dd = d[idx_all]
    ww = np.exp((dd - dmax) / max(temp, 1e-12))
    ww = ww / ww.sum()
    w_active = np.zeros(2 * n - 1)
    w_active[idx_all] = ww
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _mirror_descent(a0, n_iters, deadline, k_active=5, tol=1e-6, use_soft=True):
    """Mirror descent (multiplicative weights) with BB adaptive step."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    a_prev = a.copy()
    score_prev, g_prev = _soft_max_grad(a, k_active, max(0.01, tol * 100)) if use_soft else _soft_subgrad(a, k_active, tol)
    if not np.isfinite(score_prev):
        return best_score, best_a
    gn = np.linalg.norm(g_prev)
    if gn < 1e-12:
        return best_score, best_a
    eta = 0.02 / gn
    eta_min = 1e-8
    eta_max = 0.5 / gn
    s_vec = np.zeros(n)
    y_vec = np.zeros(n)
    for it in range(n_iters):
        if time.monotonic() > float(deadline) - 2.0:
            break
        s_vec = a - a_prev
        y_vec = a + (0.0)  # placeholder
        # Compute new gradient
        if use_soft:
            score_cur, g_cur = _soft_max_grad(a, k_active, max(0.01, tol * 100))
        else:
            score_cur, g_cur = _soft_subgrad(a, k_active, tol)
        if not np.isfinite(score_cur):
            break
        # BB step size
        y_vec = g_cur - g_prev
        sy = float(np.dot(s_vec, y_vec))
        if sy > 1e-14:
            yy = float(np.dot(y_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if yy > 1e-14:
                eta_bb = sy / yy
            else:
                eta_bb = eta
            if ss > 1e-14:
                eta_bb2 = ss / sy
            else:
                eta_bb2 = eta
            eta = 0.5 * (eta_bb + eta_bb2)
        eta = float(np.clip(eta, eta_min, eta_max))
        # Multiplicative weight update
        eta_adj = eta * (1.0 + 0.01 * (it % 100 == 0 and 10 or 0))
        log_mult = -eta_adj * g_cur
        log_mult = np.clip(log_mult, -50.0, 50.0)
        mult = np.exp(log_mult)
        a_new = a * mult
        a_new = np.clip(a_new, 0.0, None)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / S_new
        # Track best
        sc = _exact_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a_prev = a
        g_prev = g_cur
        a = a_new
        # Decay eta slightly if no improvement
        if it > 0 and it % 100 == 0:
            if not (sc < best_score + 1e-12):
                eta *= 0.8
    return best_score, best_a


def _nesterov_soft(a0, n_iters, deadline, k_active=5, tol=1e-6, use_soft=True):
    """Nesterov accelerated subgradient with soft active set."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    score0, g0 = _soft_max_grad(y, k_active, max(0.01, tol * 100)) if use_soft else _soft_subgrad(y, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.03 / max(g_norm, 1e-10)
    no_improve = 0
    for it in range(n_iters):
        if time.monotonic() > float(deadline) - 1.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy
        score_y, grad_y = _soft_max_grad(y, k_active, max(0.01, tol * 100)) if use_soft else _soft_subgrad(y, k_active, tol)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.05 * it)
        x_new = y - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            eta *= 0.7
            no_improve = 0
    return best_score, best_a


def _lbfgs_lse(a0, beta, maxfun=500):
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": max(maxfun // 2, 50),
            "maxfun": int(maxfun), "maxls": 40}
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = _normalize(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _peak_pair_descent(a0, n_iters, deadline):
    """Direct peak-pair coordinate descent: reduce mass at peak-contributing coords."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    for it in range(n_iters):
        if time.monotonic() > float(deadline) - 1.0:
            break
        c = _fast_self_conv(a)
        # Get top 3 peaks
        top3 = np.argsort(c)[::-1][:3]
        # For each peak, compute contribution and push mass
        for p in top3:
            p = int(p)
            lo = max(0, p - (n - 1))
            hi = min(n - 1, p)
            if lo >= hi:
                continue
            # a[p-i] for i in [lo, hi]
            idx = np.arange(lo, hi + 1)
            partner = p - idx
            # partner must be in [0, n-1]
            valid = (partner >= 0) & (partner < n)
            idx_v = idx[valid]
            partner_v = partner[valid]
            if len(idx_v) == 0:
                continue
            contrib = a[idx_v] * a[partner_v]
            # Transfer small amount from larger to smaller in each pair
            # Vectorized: for each pair, move delta from max to min
            delta = 0.01 * a[idx_v]  # 1% of each coordinate
            # Reduce the one that's larger
            mask_a_larger = a[idx_v] > a[partner_v]
            # For pairs where a[i] > a[p-i]: reduce a[i], increase a[p-i]
            a[idx_v[mask_a_larger]] -= delta[mask_a_larger] * a[idx_v[mask_a_larger]] * 0.1
            a[partner_v[mask_a_larger]] += delta[mask_a_larger] * a[idx_v[mask_a_larger]] * 0.1
            # For pairs where a[i] < a[p-i]: reduce a[p-i], increase a[i]
            mask_b_larger = ~mask_a_larger
            a[idx_v[mask_b_larger]] += delta[mask_b_larger] * a[partner_v[mask_b_larger]] * 0.1
            a[partner_v[mask_b_larger]] -= delta[mask_b_larger] * a[partner_v[mask_b_larger]] * 0.1
        a = np.clip(a, 0.0, None)
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_score, best_a

def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _soft_subgrad(a, k_active=5, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        K = min(51, max(7, n // 100))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 3.0)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = rng.standard_normal(n)
            noise = np.convolve(noise, kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.4 * noise)
    elif mode == 4:
        x = np.linspace(0.0, 1.0, n)
        freq = int(rng.integers(1, 6))
        q = a * (1.0 + strength * 0.5 * np.cos(2.0 * np.pi * freq * x + rng.uniform(0, 2 * np.pi)))
    elif mode == 5:
        # Frequency domain perturbation
        N = _next_pow2(n)
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        ph = np.angle(A)
        ph += strength * 0.3 * rng.standard_normal(len(ph))
        A_new = mag * np.exp(1j * ph)
        q = np.fft.irfft(A_new, N)[:n]
        q = np.maximum(q, 0)
    elif mode == 6:
        # Swap pairs
        q = a.copy()
        n_swaps = int(rng.integers(5, max(6, n // 50)))
        for _ in range(n_swaps):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = a * np.exp(0.03 * rng.standard_normal(n))
    else:
        # Gaussian bump removal: reduce a local region
        ctr = int(rng.integers(0, n))
        width = int(rng.integers(n // 100, n // 20))
        x = np.arange(n)
        mask = np.exp(-((x - ctr) ** 2) / (2.0 * width ** 2))
        q = a * (1.0 - strength * 0.5 * mask)
        q = np.maximum(q, 0)
    return _normalize(q)


def _generate_inits(n, parent, rng, cap=25):
    inits = []

    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))

    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)
    _add(np.ones(n))
    # Fejer kernel
    k_idx = np.arange(n) - n // 2
    fejer = np.sinc(k_idx / (n / 2.0)) ** 2
    _add(np.maximum(fejer, 0.001))
    # Raised cosine
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        _add(0.5 * (1 + np.cos(np.pi * alpha * (2 * x - 1))))
    # Cosine windows
    for f in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.5, 0.8, 1.2]:
            _add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
    # Bimodal
    for w in [0.08, 0.12, 0.2, 0.3]:
        for ctr in [0.15, 0.25, 0.35]:
            _add(np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) + np.exp(-(((x - (0.5 + ctr)) / w) ** 2)) + 0.01)
    # Exponential decay
    for rate in [2.0, 5.0, 10.0]:
        _add(np.exp(-rate * x) + np.exp(-rate * (1 - x)) + 0.01)
    # Riesz product-like (cosine product)
    for it in range(3):
        freq = int(rng.integers(1, 8))
        riesz = np.ones(n)
        for _ in range(3):
            f2 = int(rng.integers(1, freq + 1))
            riesz *= np.maximum(0.01, 0.5 + 0.5 * np.cos(2.0 * np.pi * f2 * x))
        _add(riesz)
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.0, 2.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        _add(1.0 + float(rng.uniform(0.2, 1.0)) * noise)
    # Pure random
    for _ in range(3):
        _add(0.1 + 0.9 * rng.random(n))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None
    print("Parent: {} (len={})".format("yes" if parent is not None else "no",
                                       parent.size if parent is not None else 0))

    # Determine lengths
    if parent is not None:
        parent_len = min(parent.size, 16384)
        if parent_len < 2048:
            parent_len = 4096
        lengths = [parent_len, 8192, 10240, 6144]
    else:
        lengths = [8192, 10240, 6144, 4096]
    lengths = [L for L in lengths if 2048 <= L <= 16384]
    # Remove duplicates, keep order
    seen = set()
    lengths_uniq = []
    for L in lengths:
        if L not in seen:
            seen.add(L)
            lengths_uniq.append(L)
    lengths = lengths_uniq[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate and evaluate inits (200s)
    p1_end = t_start + 200.0
    all_scores = []
    all_seqs = []
    for n in lengths:
        rem = p1_end - time.monotonic()
        if rem < 5:
            break
        inits = _generate_inits(n, parent, rng, cap=15)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_scores.append(sc)
                all_seqs.append(arr)
    if not all_scores:
        n0 = lengths[0] if lengths else 4096
        all_seqs.append(np.ones(n0) / n0)
        all_scores.append(2.0)
    order = np.argsort(all_scores)
    print("Init best: {:.8f} (len={})".format(all_scores[order[0]], all_seqs[order[0]].size))

    # Select top starts from different lengths (up to 4)
    starts = []
    used_lens = set()
    for idx in order:
        if len(starts) >= 4:
            break
        L = all_seqs[idx].size
        if L not in used_lens:
            starts.append(all_seqs[idx].copy())
            used_lens.add(L)
    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng))

    # Quick-optimize each start
    n_starts = len(starts)
    quick_each = max(20.0, min(50.0, 150.0 / n_starts))
    for i in range(n_starts):
        rem = p1_end - time.monotonic()
        if rem < 10:
            break
        sc, seq = _quick_optimize(starts[i], t_start + min(p1_end, time.monotonic() + quick_each))
        if np.isfinite(sc):
            starts[i] = seq
            print("Quick start{}: {:.8f} (len={})".format(i, sc, seq.size))

    # Global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    print("After phase1: {:.8f}".format(best_score))

    # Phase 2: Main optimization (1200s)
    p2_end = t_start + 1400.0
    # Sort starts by score
    start_scores = [_exact_score(s) for s in starts]
    order_starts = np.argsort(start_scores)
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    elif n_starts == 3:
        fracs = [0.50, 0.30, 0.20]
    else:
        fracs = [0.42, 0.28, 0.18, 0.12]

    main_budget = max(100.0, p2_end - t_start)
    cum = 0.0
    for rank, si in enumerate(order_starts):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(sub_dead, p2_end)
        if time.monotonic() > sub_dead - 20:
            continue
        sq = starts[si].copy()
        # Blend secondary starts with best
        if rank > 0 and len(best_seq) != len(sq):
            best_rs = _resample(best_seq, len(sq))
            t_blend = 0.3 + 0.15 * rank
            sq = _normalize((1 - t_blend) * _normalize(sq) + t_blend * _normalize(best_rs))
        elif rank > 0:
            t_blend = 0.3 + 0.15 * rank
            sq = _normalize((1 - t_blend) * _normalize(sq) + t_blend * _normalize(best_seq))
        sc, seq = _full_optimize(sq, sub_dead, "traj{}".format(rank + 1))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: {:.8f}".format(rank + 1, best_score))

    best_seq = _normalize(best_seq)

    # Phase 3: Perturbation restarts (500s)
    p3_end = t_start + 1900.0
    if p3_end > time.monotonic() + 80:
        n_pert = 6
        chunk = (p3_end - time.monotonic()) / n_pert
        for pi in range(n_pert):
            chunk_dead = time.monotonic() + chunk
            if time.monotonic() > p3_end - 10:
                break
            p = _perturb(best_seq, rng, strength=float(rng.uniform(0.05, 0.25)))
            sc, seq = _quick_optimize(p, min(chunk_dead, p3_end - 10))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    # Phase 4: Final intensive polish (400s)
    final_dead = deadline
    if time.monotonic() < final_dead - 20:
        # High beta LSE
        rem = final_dead - time.monotonic()
        if rem > 30:
            for beta in [300000, 1000000, 3000000]:
                rem = final_dead - time.monotonic()
                if rem < 15:
                    break
                maxfun = int(max(20, min(300, rem * 3)))
                sc, a2 = _lbfgs_lse(best_seq, beta, maxfun)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final LSE b={:.0f}: {:.8f}".format(beta, best_score))
                best_seq = _normalize(a2)
        # Nesterov with various k
        rem = final_dead - time.monotonic()
        if rem > 20:
            for k_act in [2, 3, 1]:
                rem = final_dead - time.monotonic()
                if rem < 10:
                    break
                n_it = int(max(50, rem * 12))
                sc, a2 = _nesterov_soft(best_seq, n_iters=n_it, deadline=final_dead,
                                        k_active=k_act, tol=1e-9, use_soft=False)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("Final Nest k={}: {:.8f}".format(k_act, best_score))
                best_seq = _normalize(a2)
        # Peak-pair descent
        rem = final_dead - time.monotonic()
        if rem > 10:
            n_it = int(max(30, rem * 5))
            sc, a2 = _peak_pair_descent(best_seq, n_iters=n_it, deadline=final_dead)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final PeakPair: {:.8f}".format(best_score))
            best_seq = _normalize(a2)
        # One more mirror descent
        rem = final_dead - time.monotonic()
        if rem > 10:
            n_it = int(max(50, rem * 8))
            sc, a2 = _mirror_descent(best_seq, n_iters=n_it, deadline=final_dead,
                                     k_active=3, tol=1e-9, use_soft=False)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Mirror: {:.8f}".format(best_score))
            best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score
    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
