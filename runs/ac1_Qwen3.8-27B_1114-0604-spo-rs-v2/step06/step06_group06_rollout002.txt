```python
def solve_lp_direction(a, M, active_indices, n_vars):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for k in active_indices, b >= 0"""
    n = len(a)
    K = len(active_indices)
    
    # Build constraint matrix: K x n
    # Row k: row[k, i] = a[k-i] if 0 <= k-i < n
    C_data = []
    C_rows = []
    C_cols = []
    for ki, k in enumerate(active_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            C_data.append(a[k - i])
            C_rows.append(ki)
            C_cols.append(i)
    
    C = sp.csc_matrix((C_data, (C_rows, C_cols)), shape=(K, n))
    
    # LP: max sum(b) s.t. C @ b <= M, b >= 0
    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [C @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, verbose=True)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    
    if b_var.value is None:
        return None
    return np.asarray(b_var.value).ravel()
```

```python
def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def fast_cross_conv(a, b):
    """Compute the middle portion of conv(a, b) (length n)."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n_a = int(a.size)
    n_b = int(b.size)
    if n_a == 0 or n_b == 0:
        return np.empty(0)
    M = n_a + n_b - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return out

    """Compute sum_k w[k] * (a*a)[k] as a function of a (the 'repulsion' vector)."""
    # This gives the gradient direction for the exponentiated gradient
    # corr[a][i] = sum_k w[k] * 2*a[k-i] (the gradient of sum_k w[k]*(a*a)[k] a[i])
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)

    # This is equivalent to the cross-correlation of w and a

    # corr[i] = sum_k w[k] * a[k-i]

    # This is the "cross-correlation" of b and a, which is conv(b, reverse(a)) shifted
    # conv(b, reverse(a))[i + n - 1] = sum_k b[k] * reverse(a)[i + n - 1 - k]
    # = sum_k w[k] * a[n - 1 - (i + n - 1 - k)]
    # = sum_k w[k] * a[k - i]
    # Yes! So corr = conv(w, reverse(a))[n-1 : n-1+n]
    
    N = next_power_of_two(2 * n - 1)

    a = np.asarray(a, dtype=np.float64).ravel()

    a = np.clip(a, 0.0, None)

    if S < 1e-12:

    return a / S

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def exp_grad_update(a, conv, top_k, eps_frac, eta):
    n = len(a)
    dmax = float(np.max(conv))
    if dmax <= 0:
        return a.copy()
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    if active.sum() < max(1, top_k):
        idx = np.argpartition(conv, -max(1, top_k))[-max(1, top_k):]
        active[idx] = True
    max_count = max(top_k, int(0.03 * (2*n-1)))
    if active.sum() > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active2 = np.zeros(conv.size, dtype=bool)
        active2[idx] = True
        active = active2
    w = active.astype(np.float64) / active.sum()
    h = corr_pair(a, w)
    hmax = max(float(np.max(h)), 1e-15)
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)
```
