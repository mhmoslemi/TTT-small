"""
Step function height optimization for minimizing 2n*max(self_conv)/(sum)^2.
Uses adaptive LSE-LBFGS, projected gradient with peak shaving, and multi-restart.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_s = np.exp(scaled - max_scaled)
    p = exp_s / exp_s.sum()
    L = c_max + (max_scaled + float(np.log(exp_s.sum()))) / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * (a_rev * p)[n-1+j]
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    M = 1 << (conv_len - 1).bit_length()
    A_r = np.fft.rfft(a_rev, M)
    P = np.fft.rfft(p, M)
    full_c = np.fft.irfft(A_r * P, M)[:conv_len]
    dL_da = 2.0 * full_c[n-1:2*n-1]
    
    grad = (2.0 * n / (S * S)) * dL_da - (4.0 * n * L) / (S ** 3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=20000):
    """L-BFGS-B optimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_score_grad(x, beta)
    
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

    n = len(a0)
