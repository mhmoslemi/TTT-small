def _subgrad(a, top_p=5, rel_tol=1e-6):

    return val, np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

def _bb_step(a, g, g_prev, a_prev, eta0=0.1):
    """Barzilai-Borwein step size."""
    if g_prev is None:
        gn = np.linalg.norm(g)
        return eta0 / max(gn, 1e-12)
    d = a - a_prev
    y = g - g_prev
    dy = float(d @ y)
    yy = float(y @ y)
    dd = float(d @ d)
    if yy < 1e-20 or dd < 1e-20:
        gn = np.linalg.norm(g)
        return eta0 / max(gn, 1e-12)
    # AB step
    if abs(dy) > 1e-10:
        alpha = dd / max(abs(dy), 1e-10)
        if 0.01 * eta0 < alpha < 10.0 * eta0:
            return alpha
    gn = np.linalg.norm(g)
    return eta0 / max(gn, 1e-12)

    g_prev = None
    a_prev = None
    momentum = 0.0
    y.pos = a.copy()  # Nesterov lookahead

        if time.monotonic() > deadline:
            break
        # Nesterov momentum
        t_iter = float(it + 1)
        beta_m = (t_iter - 1.5 + 0.5 * np.sqrt(1.0 + 6.0 * t_iter * t_iter)) / (t_iter + 0.5 + 0.5 * np.sqrt(1.0 + 6.0 * t_iter * t_iter))
        y_pos = a + beta_m * (a - a_prev) if a_prev is not None else a

        else:

        sc = _score(a)

        if Sy < 1e-12:
            y_pos = a.copy()
        else:
            y_pos = y_pos / Sy

        sc, g = _subgrad(y_pos, top_p, rel_tol, exp_beta)
        if not np.isfinite(sc):
            break

        # BB step size
        d_bb = y_pos - a_prev if g_prev is not None else None
        y_bb = g - g_prev if g_prev is not None else None
        alpha_bb = _bb_alpha(d_bb, y_bb, eta * 0.01, eta * 20.0) if d_bb is not None else None
        if alpha_bb is not None:
            alpha = alpha_bb
        else:
            alpha = eta / (1.0 + 0.5 * np.sqrt(it + 1))

        a_prev = a.copy()
        g_prev = g.copy()

        a_new = y_pos - alpha * g
        a_new = np.clip(a_new, 0.0, None)
        Sa = float(a_new.sum())
        if Sa < 1e-12:
            a_new = y_pos.copy()
        else:
            a_new = a_new / Sa

        a = a_new
        sc_new = _score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()

        # Adaptive eta: reduce if no progress in recent iterations
        if it % 100 == 99 and sc_new >= best_sc + 1e-10:
            eta *= 0.8

    return best_sc, best_a


def _lbfgs_lse(a0, beta, maxfun=400):
    """L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        return _lse_grad(x / S, beta)

    try:
        res = minimize(fun, a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max(30, maxfun // 2),
                                'maxfun': int(maxfun),
                                'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
        out = _normalize(res.x) if res.x is not None else a
    except Exception:
        out = a
    return _score(out), out


def _quick_optimize(a, deadline):
    """Rapid 3-phase optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    rem = deadline - time.monotonic()
    if rem > 15:
        for beta in [1000, 10000, 100000]:
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            mf = int(max(30, min(300, rem * 8)))
            sc, a2 = _lbfgs_lse(a, beta, mf)
            if sc < best_sc:
                best_sc, best_a = sc, a2.copy()
            a = _normalize(a2)
    rem = deadline - time.monotonic()
    if rem > 8:
        nit = int(max(100, rem * 20))
        sc, a2 = _nesterov_subgrad(a, nit, deadline, top_p=3, rel_tol=1e-7)
        if sc < best_sc:
            best_sc, best_a = sc, a2.copy()
        a = _normalize(a2)
    return best_sc, best_a


def _perturb(a, rng, strength=0.3):
    """Generate a diverse perturbation."""
    a = _normalize(a)
    n = a.size
    mode = int(rng.integers(0, 7))
    if mode == 0:
        q = a * np.exp(strength * 0.4 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1.0 - t) * a + t * rng.random(n)
    elif mode == 2:
        # Low-frequency perturbation
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** 2.0)
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        noise /= (np.std(noise) + 1e-10)
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 3:
        # V-shape modulation
        x = np.linspace(0, 1, n)
        amp = float(rng.uniform(0.1, 0.4))
        q = a * (1.0 + amp * np.abs(2.0 * x - 1.0))
    elif mode == 4:
        # Cosine modulation
        f = int(rng.integers(1, 6))
        amp = float(rng.uniform(0.05, 0.3))
        x = np.linspace(0, 1, n)
        q = a * (1.0 + amp * np.cos(2.0 * np.pi * f * x))
    elif mode == 5:
        # Local swap
        q = a.copy()
        for _ in range(int(rng.integers(3, max(5, n // 500)))):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                q[i], q[j] = q[j], q[i]
        q = q * np.exp(0.03 * rng.standard_normal(n))
    else:
        # Blend with reversed
        t = float(rng.uniform(0.1, 0.4))
        q = (1.0 - t) * a + t * a[::-1]
    return _normalize(q)


def _freq_init(n, rng, parent=None):
    """Frequency-domain initialization."""
    inits = []
    inits.append(np.ones(n) / n)
    x = np.linspace(0, 1, n)
    # V-shapes
    for alpha in [0.2, 0.5, 1.0, 1.5]:
        inits.append(_normalize(1.0 + alpha * np.abs(2 * x - 1)))
    # Cosine harmonics
    for f in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.5, 0.8]:
            inits.append(_normalize(1.0 + amp * np.cos(2 * np.pi * f * x)))
            inits.append(_normalize(1.0 + amp * np.cos(np.pi * f * x)))
    # Bimodal
    for ctr in [0.15, 0.25, 0.35]:
        for w in [0.08, 0.15, 0.25]:
            inits.append(_normalize(
                np.exp(-(((x - 0.5 + ctr) / w) ** 2)) +
                np.exp(-(((x - 0.5 - ctr) / w) ** 2)) + 0.01))
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.5, 3.0)))
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        noise = np.fft.irfft(power * np.exp(1j * phase), Nf)[:n]
        noise -= noise.mean()
        inits.append(_normalize(1.0 + float(rng.uniform(0.3, 1.0)) * noise))
    # Parent resampled
    if parent is not None and parent.size > 0:
        inits.append(_resample(parent, n))
        inits.append(_resample(parent[::-1], n))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        pass

    print("Parent: {} (len={})".format(
        "yes" if parent is not None else "no",
        parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        pl = min(parent.size, 20480)
        if pl < 4096:
            pl = 8192
        lengths = [pl, 16384, 12288, 20480]
    else:
        lengths = [16384, 12288, 10240, 8192]
    lengths = sorted(set([L for L in lengths if 4096 <= L <= 20480]))[:4]
    print("Lengths: {}".format(lengths))

    # Phase 1: Find best initialization (10% time)
    phase1_end = t0 + 0.10 * (deadline - t0)
    all_cands = []
    for n in lengths:
        inits = _freq_init(n, rng, parent)
        for arr in inits:
            sc = _score(arr)
            if np.isfinite(sc):
                all_cands.append((sc, arr))
    all_cands.sort(key=lambda x: x[0])
    # Pick top 3 from different lengths
    starts = []
    used = set()
    for sc, arr in all_cands:
        L = arr.size
        if L not in used:
            starts.append(arr)
            used.add(L)
            if len(starts) >= 3:
                break
    while len(starts) < 2:
        starts.append(_perturb(starts[0], rng, 0.3))

    best_score = min(_score(s) for s in starts)
    best_seq = min(starts, key=lambda s: _score(s)).copy()
    print("Phase 1 best: {:.8f} (len={})".format(best_score, best_seq.size))

    # Phase 2: Primary optimization (45% time)
    phase2_end = t0 + 0.55 * (deadline - t0)
    starts.sort(key=lambda s: _score(s))
    fracs = [0.65, 0.35] if len(starts) > 1 else [1.0]
    cum = 0.0
    for rank, sq in enumerate(starts[:2]):
        if rank >= len(fracs):
            break
        cum += fracs[rank]
        sub_dead = phase2_end - 0.45 * (1.0 - cum) * (phase2_end - t0)
        sub_dead = min(sub_dead, phase2_end)
        if time.monotonic() > sub_dead - 15:
            continue

        # Progressive LSE
        betas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
        a = _normalize(sq)
        for beta in betas:
            rem = sub_dead - time.monotonic()
            if rem < 10:
                break
            n = len(a)
            est = 0.02 * np.log2(max(16, n)) / np.log2(8192)
            mf = int(max(40, min(600, rem * 0.85 / max(est, 0.002))))
            sc, a2 = _lbfgs_lse(a, beta, mf)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("  LSE beta={:.0f}: {:.8f}".format(beta, best_score))
            a = _normalize(a2)

        # Subgradient polish
        rem = sub_dead - time.monotonic()
        if rem > 15:
            for tp, rt in [(3, 1e-7), (5, 1e-7), (10, 1e-6)]:
                rem = sub_dead - time.monotonic()
                if rem < 10:
                    break
                nit = int(max(200, rem * 25))
                sc, a2 = _nesterov_subgrad(a, nit, sub_dead, top_p=tp, rel_tol=rt)
                if sc < best_score:
                    best_score = sc
                    best_seq = a2.copy()
                    print("  Subgrad tp={} tol={}: {:.8f}".format(tp, rt, best_score))
                a = _normalize(a2)

        # Exponential-tilt subgradient
        rem = sub_dead - time.monotonic()
        if rem > 10:
            nit = int(max(150, rem * 15))
            sc, a2 = _nesterov_subgrad(a, nit, sub_dead, top_p=8, rel_tol=1e-5,
                                       exp_beta=50.0)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("  ExpTilt: {:.8f}".format(best_score))
            a = _normalize(a2)

    best_seq = _normalize(best_seq)
    print("Phase 2 best: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts (25% time)
    phase3_end = t0 + 0.80 * (deadline - t0)
    n_pert = 6
    p_start = max(time.monotonic(), phase2_end)
    p_end = phase3_end
    if p_end > p_start + 60:
        chunk = (p_end - p_start) / n_pert
        for pi in range(n_pert):
            cd = p_start + (pi + 1) * chunk
            if time.monotonic() > cd - 5:
                break
            # Diverse perturbation
            p = _perturb(best_seq, rng, strength=float(rng.uniform(0.1, 0.5)))
            sc, seq = _quick_optimize(p, cd)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi, best_score))

    best_seq = _normalize(best_seq)
    print("Phase 3 best: {:.8f}".format(best_score))

    # Phase 4: Final intensive polish (15% time)
    final_dead = deadline - 20.0
    if time.monotonic() < final_dead - 30:
        a = best_seq.copy()
        # Very high beta LSE
        for beta in [500000, 2000000, 10000000]:
            rem = final_dead - time.monotonic()
            if rem < 12:
                break
            mf = int(max(30, min(400, rem * 5)))
            sc, a2 = _lbfgs_lse(a, beta, mf)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
            a = _normalize(a2)

        # Exact subgradient - single peak focus
        rem = final_dead - time.monotonic()
        if rem > 15:
            nit = int(max(200, rem * 30))
            sc, a2 = _nesterov_subgrad(a, nit, final_dead, top_p=1, rel_tol=1e-8)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Subgrad k=1: {:.8f}".format(best_score))
            a = _normalize(a2)

        # 2-peak focus
        rem = final_dead - time.monotonic()
        if rem > 10:
            nit = int(max(100, rem * 20))
            sc, a2 = _nesterov_subgrad(a, nit, final_dead, top_p=2, rel_tol=1e-8)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Subgrad k=2: {:.8f}".format(best_score))
            a = _normalize(a2)

        # 3-peak with exponential tilt
        rem = final_dead - time.monotonic()
        if rem > 10:
            nit = int(max(100, rem * 20))
            sc, a2 = _nesterov_subgrad(a, nit, final_dead, top_p=3, rel_tol=1e-8,
                                       exp_beta=100.0)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final ExpTilt k=3: {:.8f}".format(best_score))
            a = _normalize(a2)

        # Very tight LSE
        rem = final_dead - time.monotonic()
        if rem > 10:
            sc, a2 = _lbfgs_lse(a, 50000000, int(max(20, rem * 3)))
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final LSE very high: {:.8f}".format(best_score))

    # Final normalization and output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t0))
    return out
