def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    # Find tau such that sum(max(v - tau, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - 1))[0][-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - tau, 0.0)
    # Fix numerical issues
    S = float(out.sum())
    if S < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return out / S


def _subgrad_compute(a, k_active, tol):
    """
    Compute the subgradient of 2n*max(c)/S^2 at point a (on simplex, S=1).
    Returns (score, gradient, active_indices).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n), np.array([0])

    a_norm = a / S
    c = _fast_self_conv(a_norm)
    cmax = float(c.max())
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax

    # Find active set
    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]

    # Compute subgradient: g[i] = (2n/S^2) * (1/|A|) * sum_{k in A} 2*a_norm[k-i]
    # = (4n/(S^2 * |A|)) * sum_{k in A} a_norm[k-i]
    # Using convolution: sum_{k in A} a_norm[k-i] = conv(a_norm, w_active)[i + n - 1]
    # where w_active[k] = 1/|A| for k in A
    w_active = np.zeros(2 * n - 1, dtype=np.float64)
    w_active[active] = 1.0 / len(active)

    N2 = _next_pow2(3 * n)
    a_rev = a_norm[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    # grad = (4n/S^2) * R - (4n * T)/(S^3) * ones, where T = sum(w_active * c)
    # Actually for the normalized problem (S=1), grad = 4n * R
    # For unnormalized: grad = (2n/S^2) * (sum_{k in A} 2*a[k-i])/|A| - (2n/S^2)*(c_max)*2/S * ones
    # Let me redo: F(a) = 2n * max_c / S^2
    # dF/da[i] = 2n * (1/S^2) * sum_{k in A} (1/|A|) * 2*a[k-i] - 2n * max_c * 2 / S^3
    T = float(w_active.dot(c))
    max_c = T  # since w_active is uniform over active, this is the average active value
    # Actually T = (1/|A|) * sum_{k in A} c[k], which is the average of active c values
    # The max is cmax, not T. Let me use the exact formula.
    
    # F = 2n * cmax / S^2 (approximately, when we use the active average)
    # dF/da[i] = 2n/S^2 * (1/|A|) * sum_{k in A} 2*a_norm[k-i] * S (from d a_norm/da) 
    #           - 2n * c_avg / S^2 * 2/S  (from d(S^2)/da)
    # Hmm, this is getting complicated with the normalization. Let me just work on the simplex directly.
    
    # On the simplex (S=1, a_norm = a): F(a) = 2n * max(c(a))
    # Subgradient: (2n/|A|) * sum_{k in A} 2*a[k-i] for each i
    # But we need to project this onto the tangent space of the simplex (sum of gradient components should be 0 for the projected version)
    # Actually, for the projected gradient method, we just use the raw subgradient and project the step.
    
    g = (4.0 * n / len(active)) * R
    
    # Center the gradient (subtract mean) for better behavior on the simplex
    g = g - g.mean()
    
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), g, active


def _nesterov_subgrad(a0, n_iters, deadline, k_active=3, tol=1e-6, verbose=False):
    """
    Nesterov accelerated subgradient method on the simplex.
    """
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    
    # Initial step size
    score0, g0, _ = _subgrad_compute(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    
    eta0 = 0.1 / g_norm
    eta = eta0
    
    # Backtracking parameters
    ftol = 1e-8
    no_improve_count = 0
    last_best = best_score
    last_best_iter = 0
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        
        y = a + beta_mom * v
        y = _simplex_project(y)
        
        # Compute subgradient at y
        score_y, grad_y, active = _subgrad_compute(y, k_active, tol)
        if not np.isfinite(score_y):
            break
        
        # Adaptive step size with backtracking
        step = eta / (1.0 + 0.05 * it / max(1, n_iters // 10))
        
        # Backtracking: try the step, if score worsens too much, reduce
        x_new = _simplex_project(y - step * grad_y)
        sc_new = _exact_score(x_new)
        
        # Backtracking (max 5 iterations)
        bt = 0
        while sc_new > score_y + step * np.linalg.norm(grad_y) * 0.5 and bt < 5:
            step *= 0.5
            x_new = _simplex_project(y - step * grad_y)
            sc_new = _exact_score(x_new)
            bt += 1
        
        # Update
        v = x_new - a
        a = x_new
        t = t_new
        
        # Track best
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
            no_improve_count = 0
            last_best_iter = it
        else:
            no_improve_count += 1
        
        # Adaptive step size
        if no_improve_count > 100:
            eta *= 0.8
            no_improve_count = 0
        elif no_improve_count > 50:
            eta *= 0.95
        
        if verbose and it % 1000 == 0:
            print("  Nesterov iter {}: score={:.8f} eta={:.2e} |g|={:.2e}".format(
                it, best_score, eta, np.linalg.norm(grad_y)))
    
    return best_score, best_a


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient on the simplex."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    a_n = a / S
    c = _fast_self_conv(a_n)
    d = n * c  # d[k] = n * c[k], and score = 2 * max(d) / S^2 * S^2 = 2*max(d) (since S=1 for a_n)
    # Actually: F = 2n*max(c)/S^2. With a_n = a/S: F = 2n*max(c(a_n))*S^2/S^2 = 2n*max(c(a_n))
    # So on the simplex (S=1): F = 2n * max(c)
    
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L  # = 2n * LSE(c) approximately
    
    # Gradient on the simplex:
    # F(a_n) = 2n * LSE_beta(c(a_n))
    # dF/da_n[i] = 2n * sum_k w[k] * dc[k]/da_n[i] = 2n * sum_k w[k] * 2*a_n[k-i]
    # = 4n * conv(a_n, w)[i+n-1]  (using the reversed convolution trick)
    N2 = _next_pow2(3 * n)
    a_rev = a_n[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    grad = 4.0 * n * R
    grad = grad - grad.mean()  # center for simplex
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, maxfun=200):
    """Run L-BFGS-B on the LSE-smoothed objective with simplex constraint."""
    a = _normalize(a0)
    n = a.size
    
    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad
    
    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-12,
        "maxiter": max(maxfun, 100),
        "maxfun": int(maxfun * 3),
        "maxls": 50,
    }
    
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       bounds=bounds, options=opts)
        a_out = _normalize(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
    return float(_exact_score(a_out)), a_out

    t0 = time.monotonic()
    remaining = max(10.0, float(deadline) - t0 - 5.0)
    
    print("[{}] start: {:.8f} n={}".format(label, best_score, a.size))
    
    # Stage 1: Nesterov with moderate settings (40% of time)
    stage1_end = t0 + 0.40 * remaining
    n_iters = int(max(500, (stage1_end - t0) * 40))
    sc, a2 = _nesterov_subgrad(a, n_iters, stage1_end, k_active=3, tol=1e-5)
    if sc < best_score:
        best_score = sc
        best_a = a2.copy()
        print("[{}] after Nesterov1: {:.8f}".format(label, best_score))
    a = _normalize(a2)
    
    # Stage 2: LSE-LBFGS with increasing beta (30% of time)
    stage2_start = max(stage1_end, time.monotonic())
    stage2_end = t0 + 0.70 * remaining
    if stage2_end > stage2_start + 10.0:
        betas = [500, 2000, 10000, 50000, 200000]
        time_per_beta = (stage2_end - stage2_start) / len(betas)
        for beta in betas:
            rem = stage2_end - time.monotonic()
            if rem < 5.0:
                break
            maxfun = int(max(50, min(500, rem / max(time_per_beta, 1.0) * 0.8)))
            sc, a2 = _lbfgs_lse(a, beta, maxfun=maxfun)
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
            a = _normalize(a2)
    
    # Stage 3: Nesterov with tight tolerance (20% of time)
    stage3_end = t0 + 0.90 * remaining
    if stage3_end > time.monotonic() + 10.0:
        for k_act in [2, 3, 5]:
            rem = stage3_end - time.monotonic()
            if rem < 5.0:
                break
            n_it = int(max(200, rem * 25))
            sc, a2 = _nesterov_subgrad(a, n_it, stage3_end, k_active=k_act, tol=1e-7)
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Nesterov k={} tol=1e-7: {:.8f}".format(label, k_act, best_score))
            a = _normalize(a2)
    
    # Stage 4: Final LSE polish (10% of time)
    stage4_end = float(deadline) - 3.0
    if stage4_end > time.monotonic() + 10.0:
        for beta in [100000, 500000, 2000000]:
            rem = stage4_end - time.monotonic()
            if rem < 5.0:
                break
            maxfun = int(max(30, min(300, rem * 5)))
            sc, a2 = _lbfgs_lse(a, beta, maxfun=maxfun)
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Final LSE beta={:.0f}: {:.8f}".format(label, beta, best_score))
            a = _normalize(a2)
        
        # Final Nesterov
        rem = stage4_end - time.monotonic()
        if rem > 5.0:
            n_it = int(max(100, rem * 30))
            sc, a2 = _nesterov_subgrad(a, n_it, stage4_end, k_active=2, tol=1e-8)
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("[{}] Final Nesterov: {:.8f}".format(label, best_score))
            a = _normalize(a2)
    
    return best_score, best_a


def _perturb(a, rng, strength=None):
    """Generate a perturbation of a."""
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.1, 0.4))
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.15))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        # Subgradient push
        sc, g, _ = _subgrad_compute(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth low-frequency perturbation
        noise = rng.standard_normal(n)
        K = min(21, max(5, n // 500))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    else:
        # Blend with a structured profile
        x = np.linspace(0.0, 1.0, n)
        profiles = [
            1.0 + 0.5 * np.cos(2.0 * np.pi * x),
            1.0 + 0.3 * np.sin(4.0 * np.pi * x) ** 2,
            np.exp(-2.0 * (x - 0.5) ** 2),
        ]
        p = profiles[int(rng.integers(0, len(profiles)))]
        p = p / p.sum()
        t = float(rng.uniform(0.1, 0.3))
        q = (1.0 - t) * a + t * p
    
    return _normalize(np.clip(q, 0.0, None))


def _generate_inits(n, rng, parent, cap=15):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    def _add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() > 1e-12:
            inits.append(_normalize(arr))
    
    if parent is not None:
        pr = _resample(parent, n)
        _add(pr)
        _add(pr[::-1])
    
    x = np.linspace(0.0, 1.0, n)
    
    # Uniform
    _add(np.ones(n))
    # Bump (Gaussian-like)
    for w in [0.15, 0.2, 0.3, 0.4]:
        _add(np.exp(-((x - 0.5) ** 2) / (2 * w * w)))
    # Cosine variants
    for f in [1, 2, 3]:
        for amp in [0.3, 0.6, 1.0]:
            _add(np.maximum(0.1, 1.0 + amp * np.cos(2.0 * np.pi * f * x)))
    # Sine squared
    for f in [1, 2, 3]:
        _add(np.sin(np.pi * f * x) ** 2 + 0.01)
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        _add((x + 0.01) ** p)
        _add(((1 - x) + 0.01) ** p)
    # Random smooth
    Nf = _next_pow2(n)
    for _ in range(3):
        freqs = np.abs(np.fft.rfftfreq(Nf))
        alpha = float(rng.uniform(1.5, 3.0))
        power = 1.0 / (freqs * n + 1.0) ** alpha
        power[0] = 1.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.min()
        _add(noise)
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns list of non-negative floats minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 128 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None
    
    print("Parent loaded: {} (len={})".format("yes" if parent is not None else "no",
                                              parent.size if parent is not None else 0))
    
    # Determine candidate lengths
    if parent is not None:
        parent_len = parent.size
        lengths = [parent_len]
        for L in [8192, 10240, 12288, 16384, 6144]:
            if L != parent_len and 2048 <= L <= 16384:
                lengths.append(L)
        # Limit to top 3 lengths
        lengths = lengths[:4]
    else:
        lengths = [10240, 8192, 16384]
    
    print("Candidate lengths: {}".format(lengths))
    
    # Phase 1: Quick evaluation of initial points (5% of budget)
    phase1_end = t_start + 0.05 * (deadline - t_start)
    all_candidates = []
    
    for n in lengths:
        inits = _generate_inits(n, rng, parent, cap=10)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_candidates.append((sc, arr))
    
    # Sort by score
    all_candidates.sort(key=lambda x: x[0])
    print("Best init: {:.8f} (n={})".format(all_candidates[0][0], all_candidates[0][1].size))
    
    # Select diverse starting points (different lengths)
    starts = []
    used_lens = set()
    for sc, arr in all_candidates:
        L = arr.size
        if L not in used_lens:
            starts.append(arr.copy())
            used_lens.add(L)
        if len(starts) >= 3:
            break
    
    if not starts:
        n0 = lengths[0]
        starts.append(np.ones(n0) / n0)
    
    # Initialize global best
    best_score = float(np.inf)
    best_seq = starts[0].copy()
    for s in starts:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    
    print("Initial best: {:.8f} (n={})".format(best_score, best_seq.size))
    
    # Phase 2: Main optimization on best start (70% of budget)
    main_end = t_start + 0.75 * (deadline - t_start)
    
    # Allocate time to starts (weighted by initial score)
    n_starts = len(starts)
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.70, 0.30]
    else:
        fracs = [0.55, 0.30, 0.15]
    
    cum = 0.0
    for rank in range(n_starts):
        cum += fracs[rank]
        sub_dead = t_start + cum * (main_end - t_start)
        sub_dead = min(sub_dead, main_end)
        
        if time.monotonic() > sub_dead - 15.0:
            continue
        
        sq = starts[rank].copy()
        
        # For secondary starts, blend with current best
        if rank > 0 and best_score < np.inf:
            if len(best_seq) != len(sq):
                best_r = _resample(best_seq, len(sq))
            else:
                best_r = best_seq.copy()
            t_blend = 0.3 + 0.15 * rank
            sq = _normalize((1.0 - t_blend) * _normalize(sq) + t_blend * _normalize(best_r))
        
        sc, seq = _nesterov_lbfgs_pipeline(sq, sub_dead, "main_{}".format(rank + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After main_{}: {:.8f}".format(rank + 1, best_score))
    
    best_seq = _normalize(best_seq)
    print("Post-main best: {:.8f}".format(best_score))
    
    # Phase 3: Perturbation restarts (15% of budget)
    perturb_start = time.monotonic()
    perturb_end = t_start + 0.90 * (deadline - t_start)
    if perturb_end > perturb_start + 30.0:
        n_perturb = 4
        chunk = (perturb_end - perturb_start) / n_perturb
        for pi in range(n_perturb):
            chunk_dead = perturb_start + (pi + 1) * chunk
            if time.monotonic() > chunk_dead - 10.0:
                break
            
            p = _perturb(best_seq, rng, strength=0.15)
            sc, seq = _nesterov_lbfgs_pipeline(p, chunk_dead, "pert_{}".format(pi + 1))
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb {}: {:.8f}".format(pi + 1, best_score))
    
    # Phase 4: Final intensive polish (5% of budget)
    final_dead = deadline - 5.0
    if time.monotonic() < final_dead - 15.0:
        # High-beta LSE
        for beta in [200000, 1000000]:
            rem = final_dead - time.monotonic()
            if rem < 10.0:
                break
            maxfun = int(max(30, min(200, rem * 3)))
            sc, a2 = _lbfgs_lse(best_seq, beta, maxfun=maxfun)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final LSE beta={:.0f}: {:.8f}".format(beta, best_score))
            best_seq = _normalize(a2)
        
        # Final Nesterov with very tight tolerance
        rem = final_dead - time.monotonic()
        if rem > 10.0:
            n_it = int(max(200, rem * 30))
            sc, a2 = _nesterov_subgrad(best_seq, n_it, final_dead, k_active=2, tol=1e-9)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Final Nesterov tight: {:.8f}".format(best_score))
            best_seq = _normalize(a2)
    
    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)
    
    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score
    
    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
