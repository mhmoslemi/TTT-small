"""

LSE-smoothed L-BFGS with correct FFT-based gradients, coordinate descent,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_improved(a, n_iters=500, mode='geometric', gamma=0.7, 
                    sigma=1.0, target='median'):
    """
    Improved Fienup iteration with multiple spectral shaping modes.
    mode: 'geometric', 'power', 'gaussian', 'exponential'
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    a = a / S
    
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        
        # Target level
        if target == 'median':
            C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        elif target == 'mean':
            C = np.mean(amp[1:]) if len(amp) > 1 else amp[0]
        else:
            C = np.median(amp)
        C = max(C, 1e-20)
        
        ratio = amp / C
        
        if mode == 'geometric':
            # Adaptive alpha
            alpha = 0.1 + 0.7 * progress
            amp_new = amp * (C / np.maximum(amp, 1e-20)) ** alpha
        elif mode == 'power':
            # Only compress peaks
            amp_new = amp.copy()
            above = amp > C
            amp_new[above] = C + (amp[above] - C) * (C / amp[above]) ** gamma
        elif mode == 'gaussian':
            # Gaussian compression: amp_new = C + (amp-C) / (1 + sigma*(amp/C - 1)^2)
            diff = (amp - C) / (1.0 + sigma * (ratio - 1.0) ** 2)
            amp_new = C + diff
        elif mode == 'exponential':
            # Exponential smoothing
            lambda_val = 2.0 * progress + 0.5
            amp_new = C * np.exp((np.log(np.maximum(amp, 1e-20)) - np.log(C)) * 1.0 / (1.0 + lambda_val))
        else:
            amp_new = amp
        
        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_score_grad(x, beta):
    """
    LSE-smoothed objective and gradient for L-BFGS.
    L = c_max + (1/beta) * log(sum(exp(beta*(c - c_max))))
    score = 2n * L / S^2
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 0)  # Since c - c_max <= 0
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    p = exp_scaled / exp_sum  # Softmax weights, length M = 2n-1
    
    # G[i] = 2 * sum_k p[k] * x[k-i] (for 0<=k-i<n)
    # This is 2 * cross_correlation(p, x)[i] for i=0,...,n-1
    # Cross-correlation via FFT: ifft(fft(p, N) * conj(fft(x, N)))
    # Need N >= M + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    cross = np.fft.irfft(P * np.conj(X), N)[:n]
    
    G = 2.0 * cross
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (G / S2 - 2.0 * L / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_sweeps=5, step=0.1):
    """
    Coordinate descent: for each element, try increasing/decreasing
    to see if it improves the score.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        for i in range(n):
            # Try changing a[i]
            for delta in [step, -step, step*2, -step*2]:
                a_new = a.copy()
                a_new[i] = max(0.0, a_new[i] + delta)
                S_new = float(a_new.sum())
                if S_new < 1e-12:
                    continue
                # Quick score estimate
                A_new = np.fft.rfft(a_new, M_fft)
                c_new = np.fft.irfft(A_new * A_new, M_fft)[:M]
                sc_new = 2.0 * n * float(c_new.max()) / (S_new * S_new)
                if sc_new < sc - 1e-10:
                    a = a_new.copy()
                    S = S_new
                    sc = sc_new
                    c = c_new
                    M_val = float(c_new.max())
                    break
    
    return best_a


def peak_descent(a, n_iters=5000, n_peaks=10, step_init=0.01, decay=0.9999):
    """Peak-directed gradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top peaks
        k = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        
        # Subgradient
        sg = np.zeros(n)
        for idx in top_idx:
            idx = int(idx)
            js = max(0, idx - n + 1)
            je = min(idx, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_adapt = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step_adapt = min(step_adapt, 0.02)
        
        a -= step_adapt * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a


def build_inits(n, rng, parent=None):

    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None and len(parent) == n:

"""
Multi-phase step function optimization using improved Fienup iterations,
LSE-smoothed L-BFGS with correct FFT cross-correlation gradients,
peak-directed subgradient descent, and perturbation-based restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_improved(a, n_iters=400, mode='geometric', gamma=0.6, sigma=2.0):
    """Fienup iteration with multiple spectral shaping modes."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    a = a / S
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        ratio = amp / C

        if mode == 'geometric':
            alpha = 0.08 + 0.75 * progress
            amp_safe = np.maximum(amp, 1e-20)
            amp_new = amp_safe * (C / amp_safe) ** alpha
        elif mode == 'gaussian':
            beta_eff = sigma * (0.5 + progress)
            amp_new = C + (amp - C) / (1.0 + beta_eff * (ratio - 1.0) ** 2)
        elif mode == 'power':
            amp_new = amp.copy()
            above = amp > C * (1.0 + 1e-10)
            if above.any():
                amp_new[above] = C + (amp[above] - C) * (C / amp[above]) ** gamma
        elif mode == 'exponential':
            lam = 1.0 + 3.0 * progress
            amp_safe = np.maximum(amp, 1e-20)
            log_ratio = np.log(amp_safe) - np.log(C)
            amp_new = C * np.exp(log_ratio / (1.0 + lam))
        else:
            amp_new = amp

        A_new = amp_new * np.exp(1j * phase)
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(x, beta):
    """LSE-smoothed objective and gradient. Returns (score, grad)."""
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)

    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    # p_k = exp(beta*(c_k - c_max)) / Z, length M = 2n-1
    p = exp_scaled / exp_sum

    # Gradient: d(score)/dx_i = 2n * (G_i / S^2 - 2L/S^3)
    # where G_i = 2 * sum_k p[k] * x[k-i] for 0<=k-i<n
    # This is 2 * (cross_correlation of p and x) evaluated at lags 0..n-1
    # cross_corr[i] = sum_k p[k] * x[k-i]
    # Via FFT: cross_corr = ifft(FFT(p, N) * conj(FFT(x, N))) for N >= 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    cross = np.fft.irfft(P * np.conj(X), N)[:n]

    G = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (G / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_descent(a, n_iters=5000, n_peaks=10, step_init=0.01, decay=0.9999):
    """Peak-directed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k = min(n_peaks, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

        sg = np.zeros(n)
        for idx in top_idx:
            idx = int(idx)
            js = max(0, idx - n + 1)
            je = min(idx, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step_adapt = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step_adapt = min(step_adapt, 0.025)

        a = a - step_adapt * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2 * x - 1), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2 * x - 1) ** 2) + 0.05)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x), 0.01))
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    inits.append(np.maximum(0.1 + 0.5 * x ** 0.3 + 0.5 * (1 - x) ** 0.3, 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.sin(3 * np.pi * x) ** 2, 0.01))

    freqs = rng.integers(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    sig = np.ones(n) * 0.4
    for f, ph, am in zip(freqs, phases, amps):
        sig = sig + am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        inits.append(np.clip(parent + 0.05 * rng.standard_normal(n), 0.0, 1000.0))

    return inits
