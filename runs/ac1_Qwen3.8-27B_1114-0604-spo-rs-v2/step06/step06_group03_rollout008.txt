Optimize non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous:

2. Peak-chasing subgradient with backtracking line search
3. Fourier-domain analysis to identify active peak locations
4. Adaptive beta continuation with LSE smooth approximation
5. Multi-start with diverse structured initializations
6. Mirror descent with power iteration (soft cross-correlation update)

import numpy as np
import time
from scipy.optimize import minimize

# Constants
PEN_SUM = 1e-3

def next_power_of_two(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    """Compute self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute cross-correlation: (corr)[j] = sum_i a[i+j]*w[i] aligned properly."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    """Normalize sequence to have sum=target, non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a

def exact_score_a(a):
    """Compute exact score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def resample_sequence(a, n):
    """Resample sequence to length n via linear interpolation."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def d_from_a(a):
    """Compute normalized convolution d = n*conv(a,a)/S^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def lse_score_grad(a, beta):
    """Compute LSE-smoothed objective and its gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z = z - z.max()
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def run_lbfgs(a0, beta, deadline, maxfun):
    """Run L-BFGS-B optimization on LSE objective."""
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a

    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)

    def objective(x):
        return lse_score_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            objective, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def active_weights_topk(d, k):
    """Get uniform weights on top-k positions of d."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)

def peak_chasing_step(a, k_active, eta, warm_start=None):
    """Exact subgradient step with backtracking on the peak."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a

    # Get active set
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a

    # Gradient of score = 2*n*max_b/S^2
    # d(score)/d(a_j) = (4*n*h_j)/S^2 - 4*dot(w,d)/S * (sum constraint handled by normalization)
    grad = (4.0 * float(len(a)) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S

    # Backtracking line search
    f0 = exact_score_a(a)
    step = eta
    best_a = a
    best_f = f0

    for _ in range(20):
        a_trial = normalize_a(np.maximum(a - step * grad / (np.max(np.abs(grad)) + 1e-15), 0.0))
        f_trial = exact_score_a(a_trial)
        if np.isfinite(f_trial) and f_trial < best_f:
            best_f = f_trial
            best_a = a_trial.copy()
        step *= 0.5
        if step < 1e-10:
            break

    return best_a

def mirror_descent_step(a, k_active, t_power):
    """Mirror descent / power iteration step."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a

    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)

    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)

def peak_chasing_phase(a_start, max_iters, deadline, k_list, eta_list):
    """Iterative peak-chasing with adaptive step sizes."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    stall = 0

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            for eta in eta_list:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = peak_chasing_step(a, k, eta)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc - 1e-10:

                    best_a = q.copy()
                    a = q.copy()
                    improved = True
                    stall = 0
                elif np.isfinite(sc) and sc < exact_score_a(a):
                    a = q

        if not improved:
            stall += 1
            if stall > 10:

    return float(best_sc), best_a

def md_phase(a_start, max_iters, deadline, k_list, t_list):
    """Mirror descent phase."""

def inject_asymmetry(a, rng, strength):
    """Inject controlled asymmetry to potentially reduce center convolution."""
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    mode = int(rng.integers(0, 5))

    if mode == 0:
        # Linear tilt
        q = a * (1.0 + strength * (x - 0.5))
    elif mode == 1:
        # Quadratic warp
        q = a * (1.0 + strength * (x - 0.5)**2 * 4.0)
    elif mode == 2:
        # Shift the peak position
        shift = float(rng.uniform(-0.05, 0.05))
        x_shifted = (x - 0.5 + shift) / (1.0 - 2.0*abs(shift))
        x_shifted = np.clip(x_shifted, 0, 1)
        base = np.exp(-((x_shifted - 0.5)**2) / (2 * 0.1**2))
        q = a + strength * base
    elif mode == 3:
        # Frequency-domain phase tilt
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        phases = np.angle(A)
        freqs = np.arange(len(A))
        phases = phases + strength * 0.3 * np.sin(2 * np.pi * 0.5 * freqs / len(A))
        B = np.abs(A) * np.exp(1j * phases)
        q = np.fft.irfft(B, N)[:n]
    else:
        # Random low-frequency perturbation
        N = next_power_of_two(n)
        freqs = np.fft.rfftfreq(N) * N
        power = np.zeros(len(freqs))
        for _ in range(3):
            f0 = float(rng.uniform(0.5, 5.0))
            power += np.exp(-((freqs - f0)**2) / 2.0)
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, N)[:n]
        noise = noise - float(noise.mean())
        q = a + strength * 0.1 * noise

    return normalize_a(np.clip(q, 0.0, None))

def lse_trajectory(a_start, deadline, betas):
    """Run LSE optimization with increasing beta schedule."""
    a = normalize_a(a_start)

def generate_inits(n, rng, parent):

        add(pr * (1.0 - 0.2 * (x - 0.5)))

    # V-shapes with various parameters
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        for p in (1.0, 1.5, 2.0, 3.0):
            add(1.0 + alpha * np.abs(2.0 * x - 1.0) ** p)

    # Inverted V (hill)
    for alpha in (0.3, 0.6, 1.0, 1.5):
        add(1.0 - alpha * (2.0 * x - 1.0)**2)
        add(1.0 + alpha * (2.0 * x - 1.0)**2)

    for f in (2, 3, 4):

        add(1.0 + 0.5 * np.abs(2.0 * x - 1.0) + 0.3 * np.cos(2.0 * np.pi * f * x))

    for width in (0.08, 0.12, 0.15, 0.2, 0.3):

            add(np.exp(-(((x - center) / width) ** 2)) + np.exp(-(((x - center + 1) / width) ** 2)))

        # Asymmetric edge peaks

    # Sine-based
    for f in (1, 2, 3, 4, 6, 8):

        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth
    Nf = next_power_of_two(n)

        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 1.5)) * noise)

    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z = z - z.max()
        e = np.exp(z)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def lse_wrapper(x, beta):
    return lse_score_grad(x, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15, "gtol": 1e-12,
        "maxiter": int(maxiter), "maxfun": int(maxfun), "maxls": 50,
    }
    try:
        res = minimize(lse_wrapper, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def peak_chasing_step(a, k_active, eta):
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a
    grad = (4.0 * float(len(a)) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    gnorm = float(np.max(np.abs(grad)))
    if gnorm < 1e-15:
        return a
    f0 = exact_score_a(a)
    step = eta
    best_a = a.copy()
    best_f = f0
    for _ in range(25):
        a_trial = normalize_a(np.maximum(a - step * grad / gnorm, 0.0))
        f_trial = exact_score_a(a_trial)
        if np.isfinite(f_trial) and f_trial < best_f:
            best_f = f_trial
            best_a = a_trial.copy()
        step *= 0.5
        if step < 1e-12:
            break
    return best_a


def mirror_descent_step(a, k_active, t_power):
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def peak_chasing_phase(a_start, max_iters, deadline, k_list, eta_list):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    stall = 0
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            for eta in eta_list:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = peak_chasing_step(a, k, eta)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
                    stall = 0
                elif np.isfinite(sc) and sc < exact_score_a(a) + 1e-12:
                    a = q
        if not improved:
            stall += 1
            if stall > 15:
                break
    return float(best_sc), best_a


def md_phase(a_start, max_iters, deadline, k_list, t_list):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    stall = 0
    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            if deadline is not None and time.monotonic() > float(deadline):
                break
            for t_p in t_list:
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = mirror_descent_step(a, k, t_p)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
            if stall > 20:
                break
    return float(best_sc), best_a


def inject_asymmetry(a, rng, strength):
    a = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(a)
    x = np.linspace(0.0, 1.0, n)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = a * (1.0 + strength * (x - 0.5) * 2.0)
    elif mode == 1:
        q = a * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 3)
    elif mode == 2:
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        freqs = np.arange(len(A))
        phases = np.angle(A) + strength * 0.5 * np.sin(2 * np.pi * freqs / (2 * len(A)))
        B = np.abs(A) * np.exp(1j * phases)
        q = np.fft.irfft(B, N)[:n]
    elif mode == 3:
        N = next_power_of_two(n)
        freqs = np.fft.rfftfreq(N) * N
        power = np.zeros(len(freqs))
        for _ in range(4):
            f0 = float(rng.uniform(0.5, 8.0))
            w0 = float(rng.uniform(0.5, 2.0))
            power += np.exp(-((freqs - f0) ** 2) / (2 * w0 * w0))
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, N)[:n]
        noise = noise - float(noise.mean())
        q = a + strength * 0.08 * noise
    elif mode == 4:
        shift = float(rng.uniform(-0.08, 0.08))
        x_s = np.clip((x - 0.5 - shift) / (1.0 - 2.0 * abs(shift)), 0, 1)
        bump = np.exp(-((x_s - 0.5) ** 2) / (2 * 0.15 ** 2))
        q = a + strength * 0.3 * bump
    else:
        edges = np.exp(-((x - 0.0) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 1.0) ** 2) / (2 * 0.12 ** 2))
        q = a + strength * 0.2 * edges
    return normalize_a(np.clip(q, 0.0, None))


def lse_trajectory(a_start, deadline, betas, label=""):
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    n_betas = len(betas)
    for bi, beta in enumerate(betas):
        frac_end = min(1.0, (bi + 1) / float(n_betas) * 1.3)
        sub_dead = t0 + frac_end * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            n = len(a)
            est = 0.025 * max(1.0, n / 8192.0) * max(1.0, np.log2(max(16, n)) / 13.0)
            mf = int(min(2500, max(50, (0.65 * rem) / max(est, 1e-4))))
            if mf < 50:
                break
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                if label:
                    print(f"    {label} LSE b={beta:.0f}: {sc:.8f}")
            a = normalize_a(a2)
    return float(best_sc), best_a


def generate_inits(n, rng, parent):
    inits = []
    n = int(n)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.01
        if float(arr.sum()) < 1e-10:
            arr = np.ones(n) * 0.1
        inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        x = np.linspace(0.0, 1.0, n)
        add(pr * (1.0 + 0.15 * (x - 0.5)))
        add(pr * (1.0 - 0.15 * (x - 0.5)))
        add(pr * (1.0 + 0.1 * np.sin(2 * np.pi * x)))

    x = np.linspace(0.0, 1.0, n)

    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0):
        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            add(1.0 + alpha * np.abs(2.0 * x - 1.0) ** p)

    for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
        add(1.0 + alpha * (2.0 * x - 1.0) ** 4)

    for f in (2, 3, 4, 5, 6, 8):
        for amp in (0.3, 0.5, 0.8, 1.0):
            add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
            add(1.0 + amp * np.abs(np.cos(2.0 * np.pi * f * x)))

    for width in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4):
        add(np.exp(-((x / width) ** 2)) + np.exp(-(((x - 1) / width) ** 2)))
        add(np.exp(-((x / (width * 1.3)) ** 2)) + 1.5 * np.exp(-(((x - 1) / width) ** 2)))

    for f in (1, 2, 3, 4, 6):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(np.abs(np.sin(np.pi * f * x)) + 0.05)

    Nf = next_power_of_two(n)
    for _ in range(10):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(0.8, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.2, 2.0)) * noise)

    return inits


def full_optimize(a_start, deadline, verbose=False):
    a = normalize_a(a_start)
    best_sc = exact_score_a(a)
    best_a = a.copy()
    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0))

    d1 = t0 + 0.18 * total
    if time.monotonic() < d1 - 3:
        sc, seq = peak_chasing_phase(a, 300, d1, (1, 2, 3, 5, 8, 12), (0.2, 0.1, 0.05, 0.02, 0.01))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()
            if verbose:
                print(f"    Peak chase 1: {sc:.8f}")

    d2 = t0 + 0.50 * total
    if time.monotonic() < d2 - 3:
        betas = [100.0, 500.0, 2000.0, 8000.0, 30000.0, 100000.0, 300000.0]
        sc, seq = lse_trajectory(best_a, d2, betas, "opt")
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()
            if verbose:
                print(f"    LSE main: {sc:.8f}")

    d3 = t0 + 0.70 * total
    if time.monotonic() < d3 - 3:
        sc, seq = md_phase(best_a, 300, d3, (1, 2, 3, 5, 8), (0.02, 0.05, 0.1, 0.2, 0.35))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()
            if verbose:
                print(f"    MD: {sc:.8f}")

    d4 = t0 + 0.85 * total
    if time.monotonic() < d4 - 3:
        sc, seq = peak_chasing_phase(best_a, 200, d4, (1, 2, 3, 5), (0.1, 0.05, 0.02, 0.01, 0.005))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()
            if verbose:
                print(f"    Peak chase 2: {sc:.8f}")

    if time.monotonic() < deadline - 5:
        sc, seq = lse_trajectory(best_a, deadline, [500000.0, 1500000.0], "final")
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()
            if verbose:
                print(f"    Final LSE: {sc:.8f}")

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print(f"Parent: {parent is not None}, len={parent.size if parent is not None else 0}")

    n_primary = 10240
    if parent is not None:
        n_primary = parent.size
        n_primary = min(n_primary, 16384)
        n_primary = max(n_primary, 2048)

    inits = generate_inits(n_primary, rng, parent)
    print(f"Generated {len(inits)} inits, length={n_primary}")

    scored = []
    for i, ini in enumerate(inits):
        sc = exact_score_a(ini)
        scored.append((sc, i, ini))
    scored.sort(key=lambda x: x[0])

    best_sc = scored[0][0]
    best_a = scored[0][2].copy()
    print(f"Best init score: {best_sc:.8f}")
    for sc, i, _ in scored[:5]:
        print(f"  Init[{i}]: {sc:.8f}")

    # Main optimization: top-3 inits with time split
    top_k = min(3, len(scored))
    fracs = [0.50, 0.30, 0.20][:top_k]
    t_cursor = t_start
    main_end = t_start + 0.72 * (total_budget - 25.0)

    for rank in range(top_k):
        d_end = t_start + (t_cursor - t_start + fracs[rank] * (main_end - t_start)) / 1.0
        d_end = min(main_end, deadline - 20)

        if rank == 0:
            init_opt = scored[0][2].copy()
        elif rank == 1:
            init_opt = normalize_a(0.6 * scored[1][2] + 0.4 * best_a)
            init_opt = inject_asymmetry(init_opt, rng, 0.08)
        else:
            init_opt = scored[2][2].copy()
            init_opt = inject_asymmetry(init_opt, rng, 0.06)

        print(f"Optimizing init rank {rank} (score {scored[rank][0]:.6f}) until t+{d_end - t_start:.0f}s")
        sc, seq = full_optimize(init_opt, d_end, verbose=True)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print(f"  => new best: {best_sc:.8f}")

        t_cursor = max(t_cursor, time.monotonic())
        if np.isfinite(sc) and sc < best_sc + 0.005:
            pass

    # Asymmetry exploration (15% of time)
    asym_start = max(time.monotonic(), t_start + 0.72 * (total_budget - 25.0))
    asym_end = t_start + 0.87 * (total_budget - 25.0)
    asym_end = min(asym_end, deadline - 20)

    if time.monotonic() < asym_end - 20:
        n_attempts = max(1, int((asym_end - time.monotonic()) / 80))
        for attempt in range(min(n_attempts, 6)):
            if time.monotonic() > asym_end - 20:
                break
            a_asym = inject_asymmetry(best_a, rng, float(rng.uniform(0.03, 0.15)))
            # Also try significant perturbation
            if attempt % 2 == 1:
                x = np.linspace(0, 1, len(a_asym))
                a_asym = normalize_a(a_asym * (1.0 + 0.3 * float(rng.uniform(0.1, 0.5)) * np.sin(np.pi * x)))
            chunk_end = min(time.monotonic() + 70, asym_end - 15)
            sc, seq = full_optimize(a_asym, chunk_end, verbose=False)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = seq.copy()
                print(f"  Asym attempt {attempt}: {best_sc:.8f}")

    # Multi-length exploration (8%)
    ml_start = max(time.monotonic(), t_start + 0.87 * (total_budget - 25.0))
    ml_end = t_start + 0.95 * (total_budget - 25.0)
    ml_end = min(ml_end, deadline - 15)

    if time.monotonic() < ml_end - 30:
        other_ns = []
        for candidate_n in [8192, 12288, 14336, 16384]:
            if candidate_n != n_primary and len(other_ns) < 2:
                other_ns.append(candidate_n)

        for n_other in other_ns:
            if time.monotonic() > ml_end - 30:
                break
            a_other = normalize_a(resample_sequence(best_a, n_other))
            sc0 = exact_score_a(a_other)
            # Quick optimization
            d_ml = min(time.monotonic() + 100, ml_end - 15)
            sc, seq = full_optimize(a_other, d_ml, verbose=False)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = seq.copy()
                print(f"  Multi-length n={n_other}: {best_sc:.8f}")

    # Final polish (5%)
    if time.monotonic() < deadline - 5:
        sc, seq = peak_chasing_phase(best_a, 100, deadline - 5,
                                     (1, 2, 3, 5), (0.01, 0.005, 0.002, 0.001))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()

        sc, seq = lse_trajectory(best_a, deadline - 3, [3000000.0, 5000000.0], "polish")
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()

        sc, seq = md_phase(best_a, 50, deadline - 3, (1, 2, 3), (0.01, 0.03, 0.05))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_a = sc, seq.copy()

    # Final output preparation
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print(f"FINAL: internal={best_sc:.8f} verified={verified:.8f} len={len(out)} time={time.monotonic()-t_start:.1f}s")
    return out
