    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def softmax_weights(conv, temperature):
    """Compute softmax weights over convolution positions with given temperature."""
    cmax = float(np.max(conv))
    if cmax <= 0 or temperature <= 0:
        idx = np.argmax(conv)
        w = np.zeros(conv.size)
        w[idx] = 1.0
        return w
    z = (conv - cmax) / temperature
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        idx = np.argmax(conv)
        w = np.zeros(conv.size)
        w[idx] = 1.0
        return w
    return e / s


def excess_weights(conv, fraction=0.05, sharpness=1.0):
    """Compute excess-based weights: focus on positions above a threshold."""
    cmax = float(np.max(conv))
    if cmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = cmax * fraction
    excess = np.maximum(conv - threshold, 0.0)
    excess = excess ** sharpness
    total = float(excess.sum())
    if total < 1e-30:
        idx = np.argmax(conv)
        w = np.zeros(conv.size)
        w[idx] = 1.0
        return w
    return excess / total


def annealed_mw_optimize(a, deadline, verbose=True, tag=""):
    """
    Annealed multiplicative weight optimization with momentum.
    Temperature anneals from low (focused on max) to high (flat focus).
    """
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    # Schedule parameters
    total_time = deadline - t0
    if total_time <= 0:
        return best_sc, best_a

    # Temperature: geometric from T_min to T_max
    conv0 = fast_self_conv(a)
    cmax0 = float(np.max(conv0))
    T_min = max(cmax0 * 0.005, 1e-15)
    T_max = cmax0 * 0.8

    # Learning rate schedule
    eta_start = 0.8
    eta_end = 0.01
    momentum = 0.5
    momentum_end = 0.2

    log_T_min = math.log(T_min)
    log_T_max = math.log(T_max)

    smooth_grad = np.zeros(n)
    it = 0
    stagnant = 0
    max_stagnant = 50

    while time.monotonic() < deadline - 1.0:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)

        # Adaptive temperature (geometric interpolation)
        if progress < 0.1:
            frac = 0.0
        elif progress < 0.7:
            frac = (progress - 0.1) / 0.6
        else:
            frac = 1.0 - 0.3 * (progress - 0.7) / 0.3
        frac = max(0.0, min(1.0, frac))
        log_T = log_T_min + frac * (log_T_max - log_T_min)
        T = math.exp(log_T)

        # Learning rate (decreasing)
        eta = eta_start * (eta_end / eta_start) ** progress

        # Current momentum
        mom = momentum * (momentum_end / momentum) ** progress

        conv = fast_self_conv(a)
        cmax = float(np.max(conv))
        current_sc = 2.0 * n * cmax  # since sum(a)=1

        if cmax < 1e-30:
            break

        # Compute weights: blend softmax and excess
        if frac < 0.5:
            # Early: pure softmax (focused)
            w = softmax_weights(conv, T)
        else:
            # Later: blend softmax and excess weights
            w_soft = softmax_weights(conv, T)
            w_excess = excess_weights(conv, fraction=0.1 * frac, sharpness=1.5)
            blend = (frac - 0.5) * 2.0  # 0 to 1
            w = (1.0 - blend) * w_soft + blend * w_excess
            s = float(w.sum())
            if s > 0:
                w = w / s

        # Compute gradient
        grad = 2.0 * corr_pair(a, w)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        grad_max = float(np.max(grad))
        if grad_max < 1e-30:
            break

        # Normalize gradient
        grad_norm = grad / grad_max

        # Momentum update
        smooth_grad = mom * smooth_grad + (1.0 - mom) * grad_norm

        # Try multiple step sizes
        best_a_iter = a
        best_sc_iter = current_sc

        for eta_mult in [1.0, 0.5, 0.25, 2.0]:
            eta_try = eta * eta_mult
            log_a = np.log(a + 1e-300)
            log_a -= eta_try * smooth_grad
            log_a -= float(np.max(log_a))
            new_a = np.exp(log_a)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_iter:
                best_sc_iter = new_sc
                best_a_iter = new_a

        if best_sc_iter < best_sc - 1e-12:
            best_sc = best_sc_iter
            best_a = best_a_iter
            a = best_a_iter
            stagnant = 0
            if verbose and it % 200 == 0:
                print("  [{} it{}] score={:.9f} T={:.3e} eta={:.4f}".format(
                    tag, it, best_sc, T, eta))
        else:
            stagnant += 1
            if stagnant >= max_stagnant:
                # Reset momentum and reduce step
                smooth_grad *= 0.5
                stagnant = 0
            a = best_a_iter  # still move to best candidate

        it += 1

    # Final: reset to best
    a = best_a
    best_sc = exact_score(a)
    if verbose:
        print("  [{} done] it={} score={:.9f}".format(tag, it, best_sc))
    return best_sc, best_a


def lse_objective_and_grad(a, beta):
    """LSE-smoothed objective and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    scale = n / (S * S)
    d = scale * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs_refine(a, deadline, max_beta=50000, verbose=True, tag=""):
    """Adaptive LSE-LBFGS refinement with increasing beta."""
    from scipy.optimize import minimize
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    betas = [200, 500, 1000, 2500, 5000, 10000, 20000, 50000]
    if max_beta:
        betas = [b for b in betas if b <= max_beta]
        if not betas:
            betas = [200]

    for beta in betas:
        if time.monotonic() > deadline - 10.0:
            break
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        maxfun = max(50, int(rem * 0.6 / 0.12))
        maxfun = min(maxfun, 1500)

        def fun(x):
            return lse_objective_and_grad(x, beta)

        bounds = [(0.0, 10.0)] * n
        opts = {
            "ftol": 1e-15,
            "gtol": 1e-12,
            "maxiter": max(20, maxfun // 3),
            "maxfun": int(maxfun),
            "maxls": 30,
        }
        try:
            res = minimize(fun, a, method="L-BFGS-B", jac=True,
                           args=(float(beta),), bounds=bounds, options=opts)
            a_out = normalize_a(res.x) if res.x is not None else a
        except Exception:
            a_out = a

        sc = exact_score(a_out)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_out.copy()
            a = a_out.copy()
            if verbose:
                print("  [{} lbfgs beta={}] score={:.9f}".format(tag, beta, best_sc))

    return float(best_sc), best_a


def fine_polish(a, deadline, verbose=True, tag=""):
    """Fine-grained subgradient polish with systematic parameter sweep."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    conv = fast_self_conv(a)

    top_fracs = [0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]
    sharps = [0.5, 1.0, 1.5, 2.0]
    etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]

    improved = True
    round_num = 0
    while improved and time.monotonic() < deadline and round_num < 12:
        improved = False
        round_num += 1
        conv = fast_self_conv(best_a)
        cmax = float(np.max(conv))

        for tf in top_fracs:
            if time.monotonic() > deadline:
                break
            for sh in sharps:
                for eta in etas:
                    if time.monotonic() > deadline:
                        break
                    if tf < 0.01:
                        T = cmax * max(tf, 0.005)
                        w = softmax_weights(conv, T)
                    else:
                        w = excess_weights(conv, fraction=tf, sharpness=sh)
                    grad = 2.0 * corr_pair(best_a, w)
                    grad_max = float(np.max(grad))
                    if grad_max < 1e-30:
                        continue
                    log_a = np.log(best_a + 1e-300)
                    log_a -= eta * grad / grad_max
                    log_a -= float(np.max(log_a))
                    new_a = np.exp(log_a)
                    new_a = normalize_a(new_a)
                    sc = exact_score(new_a)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = new_a
                        improved = True
                        conv = fast_self_conv(new_a)
                        cmax = float(np.max(conv))

    if verbose:
        print("  [{} polish r{}] score={:.9f}".format(tag, round_num, best_sc))
    return float(best_sc), best_a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def perturb_excess_directed(a, rng, conv=None, strength=0.15):
    """Perturb in the direction that reduces excess at hot spots."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if conv is None:
        conv = fast_self_conv(p)
    # Direct perturbation: push down at positions contributing to hot spots
    w = excess_weights(conv, fraction=0.05, sharpness=1.0)
    grad = 2.0 * corr_pair(p, w)
    grad_max = max(float(np.max(grad)), 1e-30)
    q = p * np.exp(-strength * grad / grad_max)
    # Add small random noise
    q = q * np.exp(0.02 * rng.standard_normal(n))
    return normalize_a(np.clip(q, 0.0, None))


def perturb_random(a, rng, strength=0.1):
    """Random perturbation for diversification."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    mode = int(rng.integers(0, 5))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        q = p[::-1] * np.exp(0.03 * rng.standard_normal(n))
    elif mode == 2:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 + strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 3:
        k = min(64, max(7, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.3 * noise)
    else:
        q = p * (1.0 - strength * 0.5 * (1.0 - np.abs(2.0 * np.linspace(0, 1, n) - 1.0)) ** 2)
    return normalize_a(np.clip(q, 0.0, None))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        parent = None

    # Choose sequence length
    n = 10240
    if parent is not None:
        n = int(parent.size)
        if n < 2048:
            n = 10240
        elif n > 20000:
            n = 16384
        a0 = resample_sequence(parent, n)
    else:
        a0 = np.ones(n) / n
        x = np.linspace(0.0, 1.0, n)
        candidates = [
            np.ones(n) / n,
            (1.0 - 0.5 * (2.0 * x - 1.0) ** 2) / n,
            (1.0 + 0.5 * (2.0 * x - 1.0) ** 2) / n,
            np.exp(-2.0 * np.abs(x - 0.5)) ,
        ]
        best_init = a0
        best_init_sc = np.inf
        for c in candidates:
            c = normalize_a(c)
            sc = exact_score(c)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = c
        a0 = best_init
        print("No parent. Best init: {:.9f}".format(best_init_sc))

    a = normalize_a(a0)
    best_a = a.copy()
    best_sc = exact_score(a)
    print("Initial: n={}, score={:.9f}".format(n, best_sc))

    # Phase 1: Annealed multiplicative weight (55% of time)
    p1_dead = t_start + 0.55 * total_budget
    sc, a = annealed_mw_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 1: {:.9f}".format(best_sc))

    # Phase 2: LSE-LBFGS refinement (20% of time)
    p2_dead = t_start + 0.75 * total_budget
    sc, a = run_lbfgs_refine(best_a, deadline=p2_dead, max_beta=50000, verbose=True, tag="p2")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 2: {:.9f}".format(best_sc))

    # Phase 3: Fine polish (10% of time)
    p3_dead = t_start + 0.85 * total_budget
    sc, a = fine_polish(best_a, deadline=p3_dead, verbose=True, tag="p3")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 3: {:.9f}".format(best_sc))

    # Phase 4: Another annealed MW pass on best (5% of time)
    p4_dead = t_start + 0.90 * total_budget
    sc, a = annealed_mw_optimize(best_a, deadline=p4_dead, verbose=False, tag="p4")
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After Phase 4: {:.9f}".format(best_sc))

    # Phase 5: Diversification restarts (5% of time)
    p5_dead = deadline - 5.0
    n_restarts = 4
    done_restarts = 0
    while time.monotonic() < p5_dead - 20 and done_restarts < n_restarts:
        rem = p5_dead - 20.0 - time.monotonic()
        if rem < 30:
            break
        done_restarts += 1
        # Alternate between excess-directed and random perturbation
        if done_restarts % 2 == 1:
            p = perturb_excess_directed(best_a, rng, strength=float(rng.uniform(0.05, 0.15)))
        else:
            p = perturb_random(best_a, rng, strength=float(rng.uniform(0.05, 0.2)))
        chunk_dead = time.monotonic() + min(rem * 0.8, 120.0)
        sc, seq = annealed_mw_optimize(p, deadline=chunk_dead, verbose=False, tag="r{}".format(done_restarts))
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
            print("Restart {}: {:.9f}".format(done_restarts, best_sc))

    # Final: quick LBFGS polish if time remains
    if time.monotonic() < deadline - 15:
        sc, a = run_lbfgs_refine(best_a, deadline=deadline - 5, max_beta=10000, verbose=False, tag="fin")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("Final LBFGS: {:.9f}".format(best_sc))

    # Final: quick polish
    if time.monotonic() < deadline - 5:
        sc, a = fine_polish(best_a, deadline=deadline - 2, verbose=False, tag="fin2")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Output
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if len(out) == 0 or sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
