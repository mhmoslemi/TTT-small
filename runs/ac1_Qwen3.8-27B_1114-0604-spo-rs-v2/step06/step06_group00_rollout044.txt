    A = np.fft.rfft(a, M_fft)

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    p = exp_shifted / exp_shifted.sum()
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    score = (2.0 * n * L) / S2
    return score, grad


def adam_lse(a0, beta, max_iters, lr=0.5, decay_lr=False):
    """Adam optimizer for LSE-smoothed objective."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    best_sc = compute_score(a)
    best_a = a.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iters):
        if decay_lr:
            cur_lr = lr * (1.0 - it / max_iters) ** 0.5
        else:
            cur_lr = lr
        
        sc, grad = lse_score_and_grad(a, beta)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            if it % 1000 == 0 or it == max_iters - 1:
                best_a = a.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))
        
        a = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_sc, best_a


def fienup_flat(a, n_iters=300):

        a = a_new
        sc = compute_score(a)
        if sc < best_sc:

    return best_a

def multiplicative_update(a, beta, max_iters, lr=0.1):
    """Multiplicative (exponential) update for LSE objective."""

"""
Adam-based LSE annealing optimizer for minimizing 2n*max(conv(a,a))/S^2.
Combines Fienup spectral shaping for initialization, Adam optimization on
Log-Sum-Exp smoothed objective with aggressive beta annealing, top-K LSE
for targeted peak reduction, and subgradient polish. Uses multiplicative
updates as an alternative to additive Adam for better non-negativity handling.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta, top_k=None):
    """
    Compute LSE-smoothed score and gradient. If top_k is given,
    only include top-k convolution values in the LSE.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k is not None and top_k < len(c):
        # Only use top-k values
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_use = c[idx]
        c_max = float(c_use.max())
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -600, 600)
        exp_shifted = np.exp(shifted)
        log_sum = float(np.log(exp_shifted.sum()))
        L = c_max + log_sum / beta
        p_use = exp_shifted / exp_shifted.sum()
        # Rebuild full p with zeros elsewhere
        p = np.zeros(len(c))
        p[idx] = p_use
    else:
        c_max = float(c.max())
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -600, 600)
        exp_shifted = np.exp(shifted)
        log_sum = float(np.log(exp_shifted.sum()))
        L = c_max + log_sum / beta
        p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da_j = 2 * sum_k p_k * a_{k-j} (convolution-based)
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def adam_lse(a0, beta, max_iters, lr=0.5, top_k=None, lr_decay=True):
    """Adam optimizer for LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    best_sc = compute_score(a)
    best_a = a.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iters):
        if lr_decay and it > max_iters * 0.5:
            cur_lr = lr * (0.5 + 0.5 * (1.0 - (it - max_iters * 0.5) / (max_iters * 0.5)))
        else:
            cur_lr = lr

        sc, grad = lse_score_and_grad(a, beta, top_k=top_k)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))

        a = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_sc, best_a


def mul_lse(a0, beta, max_iters, lr=0.1, top_k=None):
    """Multiplicative (exponential) update for LSE objective. Better for non-negativity."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    best_sc = compute_score(a)
    best_a = a.copy()

    for it in range(max_iters):
        sc, grad = lse_score_and_grad(a, beta, top_k=top_k)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Adaptive learning rate based on gradient magnitude
        g_norm = float(np.sqrt(np.mean(grad * grad)))
        if g_norm < 1e-15:
            break
        cur_lr = lr * (1.0 / (1.0 + g_norm * 0.1))

        log_factor = -cur_lr * grad / g_norm
        log_factor = np.clip(log_factor, -2.0, 2.0)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-30)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_sc, best_a


def fienup_flat(a, n_iters=400):
    """Fienup spectrum flattening for global mixing."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.1 + 0.8 * progress
        P_blend = (1.0 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-30)
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_peak_target(a, n_iters=300):
    """Fienup variant that targets specific peak locations in the convolution."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Compute convolution to find peaks
        c = fast_conv(a)
        c_max = float(c.max())
        # Weight the spectral target to suppress peak-contributing frequencies
        peak_frac = c / (c_max + 1e-30)
        # In frequency domain, peaks at time k correspond to specific structure
        # Use a softer flattening that's more aggressive where spectrum is high
        P_sort = np.sort(P)
        P_90 = P_sort[int(0.9 * len(P_sort))]
        P_50 = P_sort[int(0.5 * len(P_sort))]

        # Adaptive target: flatten more aggressively the top 30%
        thresh = P_90 * (1.0 - 0.3 * progress)
        P_blend = np.minimum(P, thresh)
        P_blend = np.maximum(P_blend, P_50 * 0.3)

        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgrad_shave(a, n_iters, top_k=5, lr_init=0.004):
    """Subgradient descent targeting top-K autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        total_w = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_w += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * total_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step: Polyak-like
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        step_polyak = max(0.01, (sc - lower_bound)) / (gn * gn + 1e-30)
        step = min(lr, step_polyak, 0.05)
        step = max(step, 1e-8)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        lr *= 0.9999

    return best_a


def coordinate_shave(a, n_sweeps=5):
    """Coordinate descent: reduce top contributors to the peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        n_top = min(30, len(contribs))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.4, 0.6, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / 2400s")
    print(f"{'='*60}")
    return out
