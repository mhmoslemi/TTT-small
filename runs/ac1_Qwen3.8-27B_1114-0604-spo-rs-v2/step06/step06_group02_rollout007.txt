```python
def lp_direction(f, M, solver='ECOS'):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    import cvxpy as cp
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    M_conv = 2 * n - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    
    # Convolution constraints in matrix form:
    # (f*b)[k] = sum_j f[j] * b[k-j] (with valid indices)
    # This is a circulant-like matrix times b
    C = np.zeros((M_conv, n))
    for k in range(M_conv):
        for j in range(n):
            if 0 <= k - j < n:
                C[k, j] = f[j]
    
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=getattr(cp, solver))
    
    if b.value is not None:
        b_val = np.maximum(b.value, 0.0)
        S = b_val.sum()
        if S > 1e-12:
            b_val = b_val / S
        return b_val
    return None
```

```python
            if it % 1000 == 0:
                z_max = z.max()
                ez = np.exp(z - z_max)
                a = ez / ez.sum()
                sc = compute_score(a)
                if sc > best_sc * 1.05:
                    lr *= 0.8
                elif sc < best_sc * 0.999:
                    lr = min(lr * 1.1, 5.0)
        
        # End of beta phase: exact check
        z_max = z.max()
        ez = np.exp(z - z_max)
        a = ez / ez.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def lp_direction(f, M_val, n_vars=None, solver='ECOS'):
    """
    LP: max sum(b) s.t. (f*b)[k] <= M_val for all k, b >= 0.
    Uses cvxpy.
    """
    n = len(f)
    if n_vars is not None:
        n = min(n, n_vars)
    
    # Build constraint matrix using FFT would be complex, use direct for moderate n
    # For n > 5000, this might be too slow
    if n > 5000:
        return None
    
    M = 2 * n - 1
    # Build C matrix: C[k, j] = f[j] if 0 <= k-j < n, else 0
    C = np.zeros((M, n))
    f_ext = np.zeros(M)
    f_ext[:n] = f
    for j in range(n):
        C[j:j+n, j] = f_ext[:n]  # This is wrong, let me fix
    
    # Actually: (f*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j]*b[k-j]
    # = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} f[j]*b[k-j]
    # Let i = k-j, so b[i] appears with coefficient f[k-i]
    # C[k, i] = f[k-i] if 0<=k-i<n and 0<=i<n
    C = np.zeros((M, n))
    for k in range(M):
        for i in range(n):
            fj = k - i
            if 0 <= fj < n:
                C[k, i] = f[fj]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [C @ b_var <= M_val]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        # Try another solver
        try:
            prob.solve(solver='SCS')
        except Exception:
            return None
    
    if b_var.value is not None:
        b_val = np.maximum(b_var.value, 0.0)
        S = b_val.sum()
        if S > 1e-12:
            b_val = b_val / S
        return b_val
    return None

def lp_iteration(f, n_total_iters=20, t_start=None, deadline=None):
    """
    LP-based iteration from the state of the art.
    """
    if t_start is None:
        t_start = time.monotonic()
    a = np.maximum(f.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return f
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    M_val = float(c.max())
    
    for it in range(n_total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        g0 = lp_direction(a, M_val)
        dt = time.monotonic() - t0
        
        if g0 is None:
            break
        
        # Rescale g0 to match normalization
        g0 = g0 / g0.sum()
        
        # Blend
        t_blend = 0.3 * (1.0 - it / n_total_iters) + 0.05
        a_new = (1 - t_blend) * a + t_blend * g0
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    LP iter {it}: {sc_new:.8f} (dt={dt:.1f}s)")
        
        a = a_new
        c = fast_conv(a)
        M_val = float(c.max())
    
    return best_a

def initializations(n, rng):
    """Generate various initial sequences of length n."""
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    x = np.arange(n) / n
    rc = np.sin(np.pi * x) ** 2
    rc = np.maximum(rc, 1e-10)
    inits['cosine'] = rc / rc.sum()
    
    # Gaussian
    x = (np.arange(n) - n / 2) / (n / 4)
    g = np.exp(-0.5 * x ** 2)
    g = np.maximum(g, 1e-10)
    inits['gaussian'] = g / g.sum()
    
    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    inits['triangular'] = tri / tri.sum()
    
    # Random (Dirichlet)
    alpha = np.ones(n) * 0.5
    r = rng.dirichlet(alpha)
    inits['random'] = r
    
    # Beta-like (U-shaped)
    x = np.arange(n) / (n - 1)
    u = x ** 2 + (1 - x) ** 2
    u = np.maximum(u, 1e-10)
    inits['u_shape'] = u / u.sum()
    
    # Plateau (flat middle, tapered edges)
    edge = max(1, n // 8)
    plat = np.ones(n)
    for i in range(edge):
        plat[i] = (i + 1) / edge
        plat[n - 1 - i] = (i + 1) / edge
    inits['plateau'] = plat / plat.sum()
    
    return inits

def interpolate_seq(a, target_n):
    """Interpolate a sequence to a different length."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-12:
        a_new = a_new / S
    return a_new
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for non-negative sequences. The previous approach achieved 1.506105, targeting 1.5027.

Key insights for improvement:
1. **Softmax parameterization** (a = softmax(z)) guarantees positivity and normalization, enabling clean unconstrained optimization
2. **Adam with multi-timescale** (large beta for Adam, small beta for L-BFGS) exploits Adam's direction memory at coarser resolution while L-BFGS nails fine details
3. **Peak-hero gradient**: Apply gradient only at the top few convolution peaks for sharper directional signal
4. **Benders LP** using Cauchy-Schwarz on peak-contributing factors for principled direction finding
5. **Multi-scale exploration**: Probe several sequence lengths to find the best basin

```python
"""
Multi-scale Adam + L-BFGS + Benders-LP optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses softmax parameterization, peak-hero gradients, Benders LP direction finding,
and multi-length exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def peak_hero_grad(a, K=3, beta=50):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    vals, idxs = np.partition(c, -K)[-K:], np.argsort(c)[-K:]
    w = vals[**np.argsort(vals)]:0
    w = w / (w.sum() + 1e-30)
    p = np.zeros(2 * n - 1)
    for i in range(K):
        p[int(idxs[np.argsort(vals)[i]])] = w[i]
    p = p / (p.sum() + 1e-30)
    L_val = float(c.max())
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L_val / S3)
    score = 2.0 * n * L_val / S2
    return score, grad


def make_conv_matrix(f):
    n = len(f)
    M = 2 * n - 1
    C = np.zeros((M, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def benders_lp_direction(a, L_too_small_fallback=None):
    """Benders LP: max sum(b) s.t. (a*b)[k] <= L for peaks, b >= 0."""
    n = len(a)
    if n > 4000:
        return None
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return None
    L = float(c.max())
    n_peaks = min(5, 2 * n - 1)
    peak_idx = np.argsort(c)[::-1][:n_peaks]
    C = make_conv_matrix(a)
    C_peaks = C[peak_idx, :]
    b_var = cp.Variable(n, nonneg=True)
    b2_var = cp.Variable(n, nonneg=True)
    a_sqrt = np.sqrt(np.maximum(a, 0.0))
    x_var = b2_var
    x_sqrt_val = x_var * 0 + 1  # placeholder
    # Actually use: factor_k = max_j a[j2] * b[k-j2]; CS gives factor_k <= sqrt(sum_j a[j2]) = 1
    # And factor_k <= L / b[k]... nonlinear. Skip CS, use direct:
    # (a*b)[k] = sum_j a[j] * b[k-j] <= L for peak k
    constraints = [C_peaks @ b_var <= L]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    for solver in ['ECOS', 'SCS', 'GLPK']:
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=getattr(cp, solver))
            if b_var.value is not None:
                b_val = np.maximum(b_var.value, 0.0)
                S_b = b_val.sum()
                if S_b > 1e-12:
                    return b_val / S_b
                break
        except Exception:
            continue
    return None


def benders_iterate(a, n_iters=15, deadline=None):
    """Iterative Benders LP method."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        g0 = benders_lp_direction(a)
        dt = time.monotonic() - t0
        if g0 is None:
            break
        t_blend = 0.3 * (1.0 - it / max(1, n_iters)) + 0.05
        a_new = (1 - t_blend) * a + t_blend * g0
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"      Benders iter {it}: {sc_new:.8f} (lp={dt:.1f}s)")
        a = a_new
    return best_a


def run_adam(a0, deadline, lr_sched='cosine', use_peak_hero=False, K=3,
             beta_start=1000, beta_end=500000, lr_scale=1.0):
    """Adam in softmax parameterization on a规范化 sequence."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    best_a = a.copy()
    best_sc = compute_score(a)
    m1, m2, eps = 0.9, 0.9999, 1e-8
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t = 0
    n_iters = max(2000, int((deadline - time.monotonic()) * 8))
    n_iters = min(n_iters, 200000)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        if lr_sched == 'cosine':
            lr = 0.5 * lr_scale * (1.0 + np.cos(np.pi * frac)) + 0.02 * lr_scale
        elif lr_sched == 'exp':
            lr = 0.1 * lr_scale * (0.3 ** frac) + 0.02 * lr_scale
        else:
            lr = lr_scale
        ez = np.exp(z - z.max())
        a_now = ez / ez.sum()
        if use_peak_hero:
            _, grad_a = peak_hero_grad(a_now, K=K)
        else:
            _, grad_a = lse_score_grad(a_now, beta)
        dot = float(np.dot(a_now, grad_a))
        grad_z = a_now * (grad_a - dot)
        m_s = m1 * m_s + (1 - m1) * grad_z
        v_s = m2 * v_s + (1 - m2) * grad_z ** 2
        m_hat = m_s / (1 - m1 ** t)
        v_hat = v_s / (1 - m2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        if it % 500 == 0 or it == n_iters - 1:
            ez = np.exp(z - z.max())
            a_chk = ez / ez.sum()
            sc = compute_score(a_chk)
            if sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(lse_score_grad, a0, args=(beta,),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-16,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def make_inits(n, rng, parent_hint=None):
    inits = {}
    inits['uniform'] = np.ones(n) / n
    x = np.arange(n) / n
    rc = np.sin(np.pi * x) ** 2 + 1e-12
    inits['cosine'] = rc / rc.sum()
    g = np.exp(-2.0 * ((np.arange(n) - n / 2) / (n / 4)) ** 2) + 1e-12
    inits['gaussian'] = g / g.sum()
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
    inits['triangular'] = tri / tri.sum()
    edge = max(2, n // 10)
    plat = np.ones(n)
    ramp = np.linspace(0.1, 1.0, edge)
    plat[:edge] = ramp
    plat[-edge:] = ramp[::-1]
    inits['plateau'] = plat / plat.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)
    if parent_hint is not None:
        inits['parent'] = parent_hint.copy() / parent_hint.sum()
    for k in inits:
        inits[k] = np.maximum(inits[k], 0.0)
        s = inits[k].sum()
        if s > 1e-12:
            inits[k] /= s
    return inits


def interp_to_n(a, n_target):
    if len(a) == n_target:
        return a.copy()
    x0 = np.linspace(0, 1, len(a))
    x1 = np.linspace(0, 1, n_target)
    out = np.interp(x1, x0, a)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 1e-12:
        out /= s
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: multi-scale Adam + Benders LP + L-BFGS polish."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 30.0
    best_a = None
    best_sc = np.inf

    def upd(sc, a, tag=''):
        nonlocal best_a, best_sc
        if sc < best_sc:
            best_sc = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            el = time.monotonic() - t0
            print(f"  [{tag}] NEW BEST {sc:.8f} n={len(a)} t={el:.0f}s", flush=True)
            return True
        return False

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                sc_p = compute_score(parent)
                print(f"Parent n={len(parent)} sc={sc_p:.8f}", flush=True)
                upd(sc_p, parent, 'parent')
    except Exception:
        pass

    if parent is not None:
        np_ = len(parent)
        n_list = sorted(set([np_, max(512, np_ // 2), min(8192, np_ * 2)]))
    else:
        n_list = [1024, 2048, 4096]
        print("No parent.", flush=True)

    # --- Phase A: Multi-n Adam sweep (40%) ---
    tA_end = t0 + 0.13 * budget
    print(f"\n=== Phase A: Adam sweep (t={tA_end - t0:.0f}s) ===", flush=True)
    sched = ['cosine', 'exp']
    for i, n in enumerate(n_list):
        if time.monotonic() > tA_end:
            break
        print(f"\n--- n={n} ---", flush=True)
        hint = interp_to_n(parent, n) if parent is not None else None
        inits = make_inits(n, rng, hint)
        chunks = 10
        for j, (name, ai) in enumerate(inits.items()):
            if time.monotonic() > tA_end:
                break
            per = max(4, int((tA_end - time.monotonic())) // max(1, (len(inits) - j)))
            de = min(tA_end, time.monotonic() + per)
            sc, ao = run_adam(ai, de, lr_sched=sched[(j) % 2],
                              use_peak_hero=(j % 2 == 1), K=3,
                              beta_start=100, beta_end=10000, lr_scale=1.0)
            upd(sc, ao, f'A-{name}-n{n}')

    # --- Phase B: Benders LP (15%) ---
    tB_end = t0 + 0.21 * budget
    print(f"\n=== Phase B: Benders LP (t={tB_end - t0:.0f}s) ===", flush=True)
    if best_a is not None and len(best_a) <= 3000:
        for n_try in [len(best_a)]:
            if time.monotonic() > tB_end:
                break
            sc_b = benders_iterate(best_a, n_iters=20, deadline=tB_end)
            upd(compute_score(sc_b), sc_b, f'B-anders-n{n_try}')

    # --- Phase C: Extended multi-scale Adam (30%) ---
    tC_end = t0 + 0.56 * budget
    print(f"\n=== Phase C: Extended Adam (t={tC_end - t0:.0f}s) ===", flush=True)
    configs = [
        {'sch': 'cosine', 'ph': False, 'b0': 5000, 'b1': 200000, 'lr': 1.0},
        {'sch': 'exp', 'ph': True, 'b0': 2000, 'b1': 100000, 'lr': 0.8},
        {'sch': 'cosine', 'ph': False, 'b0': 10000, 'b1': 500000, 'lr': 0.5},
        {'sch': 'exp', 'ph': True, 'b0': 5000, 'b1': 300000, 'lr': 0.6},
    ]
    ch = max(1, len(configs))
    for ci, cfg in enumerate(configs):
        if time.monotonic() > tC_end:
            break
        per = max(3, int((tC_end - time.monotonic())) // max(1, (ch - ci)))
        de = min(tC_end, time.monotonic() + per)
        sc, ao = run_adam(best_a, de,
                          lr_sched=cfg['sch'], use_peak_hero=cfg['ph'], K=4,
                          beta_start=cfg['b0'], beta_end=cfg['b1'], lr_scale=cfg['lr'])
        upd(sc, ao, f'C-{ci}')

    # --- Phase D: L-BFGS multi-beta polish (25%) ---
    tD_end = t0 + 0.76 * budget
    print(f"\n=== Phase D: L-BFGS polish (t={tD_end - t0:.0f}s) ===", flush=True)
    for beta in [200, 2000, 20000, 100000, 500000]:
        if time.monotonic() > tD_end:
            break
        rm = tD_end - time.monotonic()
        if rm < 10:
            break
        mi = min(25000, int(rm * 5))
        sc, ao = run_lbfgs(best_a, beta, maxiter=mi, maxfun=mi * 20)
        upd(sc, ao, f'D-lbfgs-{beta}')

    # --- Phase E: Second Benders round (6%) ---
    tE_end = t0 + 0.82 * budget
    print(f"\n=== Phase E: Benders 2 (t={tE_end - t0:.0f}s) ===", flush=True)
    if best_a is not None and len(best_a) <= 2500:
        sc_e = benders_iterate(best_a, n_iters=10, deadline=tE_end)
        upd(compute_score(sc_e), sc_e, 'E-benders')

    # --- Phase F: Final push (10%) ---
    print(f"\n=== Phase F: Final push ===", flush=True)
    if time.monotonic() < deadline - 10:
        mi = min(20000, max(1000, int((deadline - 10 - time.monotonic()) * 4)))
        sc, ao = run_lbfgs(best_a, 1000000, maxiter=mi, maxfun=mi * 15)
        upd(sc, ao, 'F-final-lbfgs')
    if time.monotonic() < deadline - 5:
        sc, ao = run_adam(best_a, deadline - 5,
                          lr_sched='cosine', use_peak_hero=True, K=5,
                          beta_start=20000, beta_end=200000, lr_scale=0.3)
        upd(sc, ao, 'F-final-adam')

    out = [float(max(0.0, min(1000.0, v))) for v in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    ver = float(evaluate_sequence(out))
    el = time.monotonic() - t0
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: internal={best_sc:.8f} verified={ver:.8f}", flush=True)
    print(f"Len={len(out)} time={el:.1f}s/{budget:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    return out
```
