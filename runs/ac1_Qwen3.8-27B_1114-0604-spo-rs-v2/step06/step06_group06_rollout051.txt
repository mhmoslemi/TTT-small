```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using enhanced mirror descent with
adaptive active sets, backtracking line search, spectral regularization,
and local element search.
"""

import numpy as np
import time
import math

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """Compute the correlation: for each lag k weighted by w[k], compute sum_k w[k]*a[k-p]."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    # We want h[p] = sum_k w[k] * a[k-p] for p in [0, n-1]
    # a[k-p] is valid when 0 <= k-p < n, i.e., p <= k < p+n
    # This is a correlation: h = cross_correlation(a[::-1], w) appropriately sliced
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    # full has length n+m-1, and we want the middle n elements
    return full[n - 1:2 * n - 1]

def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent (exponentiated gradient)
with adaptive active sets, backtracking line search, KKT-based convergence checking,
and LSE-LBFGS refinement.
"""

import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """h[p] = sum_k w[k] * a[k-p] for p in 0..n-1 (with a[j]=0 for j out of range)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax)


def full_score(a):
    """Full score with arbitrary sum."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def active_weights(conv, mode="threshold", top_k=10, frac=0.02):
    """Compute active set weights."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    if mode == "threshold":
        thr = dmax * (1.0 - frac)
        active = conv >= thr
        if active.sum() < 1:
            idx = int(np.argmax(conv))
            active[idx] = True
        # Don't let active set be too large
        max_count = max(5, int(0.1 * conv.size))
        if active.sum() > max_count:
            idx = np.argpartition(conv, -max_count)[-max_count:]
            new_active = np.zeros(conv.size, dtype=bool)
            new_active[idx] = True
            active = new_active
        w = active.astype(np.float64) / active.sum()
    elif mode == "topk":
        k = min(top_k, conv.size)
        idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(conv.size)
        w[idx] = 1.0 / k
    elif mode == "softmax":
        # Soft weights proportional to exp(beta * (conv - dmax))
        beta = 50.0
        z = beta * (conv - dmax)
        z = np.clip(z, -100, 0)
        e = np.exp(z)
        s = e.sum()
        if s < 1e-15:
            w = np.zeros(conv.size)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = e / s
    else:
        w = np.ones(conv.size) / conv.size
    return w


def mirror_step(a, q, eta):
    """One mirror descent (exponentiated gradient) step."""
    h = corr_pair(a, q)
    hmax = max(float(np.max(h)), 1e-15)
    # Center to approximately preserve the sum direction
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def kkt_residual(a, q):
    """Compute the KKT residual: how non-uniform is the pressure h on support of a."""
    h = corr_pair(a, q)
    support = a > 1e-15
    if support.sum() < 2:
        return 0.0
    h_sup = h[support]
    return float(np.std(h_sup) / (np.mean(h_sup) + 1e-15))


def run_mirror_descent(a, deadline, verbose=True, tag=""):
    """Main mirror descent optimizer."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()
    
    eta = 0.5
    frac = 0.03
    stagnant = 0
    
    for it in range(20000):
        if time.monotonic() > deadline - 2:
            break
        
        conv = fast_self_conv(a)
        sc = float(2.0 * n * np.max(conv))
        
        # Try multiple step sizes and active set configurations
        best_a_it = a
        best_sc_it = sc
        best_eta = eta
        best_frac = frac
        
        # Compute base active weights
        q = active_weights(conv, "threshold", top_k=10, frac=frac)
        
        # Try several step sizes
        for eta_try in [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0]:
            if eta_try < 0.001:
                continue
            new_a = mirror_step(a, q, eta_try)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_it:
                best_sc_it = new_sc
                best_a_it = new_a
                best_eta = eta_try
        
        # Also try softmax active set
        q_soft = active_weights(conv, "softmax")
        for eta_try in [eta * 0.5, eta, eta * 1.5]:
            new_a = mirror_step(a, q_soft, eta_try)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_it:
                best_sc_it = new_sc
                best_a_it = new_a
                best_eta = eta_try
        
        # Also try topk
        for k_try in [5, 10, 20, 50]:
            q_top = active_weights(conv, "topk", top_k=k_try)
            new_a = mirror_step(a, q_top, best_eta * 0.5)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_it:
                best_sc_it = new_sc
                best_a_it = new_a
        
        if best_sc_it < best_sc - 1e-12:
            best_sc = best_sc_it
            best_a = best_a_it
            a = best_a_it
            eta = best_eta
            stagnant = 0
            if verbose and it % 100 == 0:
                print("  [{} it{}] score={:.9f} eta={:.3f} frac={:.4f}".format(
                    tag, it, best_sc, eta, frac))
        else:
            stagnant += 1
            eta *= 0.8
            if stagnant % 20 == 0:
                frac = min(frac * 1.3, 0.1)
            if stagnant >= 100:
                # Reset with larger step and wider active set
                eta = 0.5
                frac = 0.05
                stagnant = 0
                if verbose:
                    print("  [{} it{}] RESET eta={}, frac={}".format(tag, it, eta, frac))
                
    return best_sc, best_a


def lse_objective(a, beta):
    """LSE-smoothed version of the objective and its gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = c / (S * S)  # normalized conv
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    # Gradient of 2*n*L with respect to a
    # dL/da[i] = sum_k w[k] * d(c[k]/S^2)/da[i]
    # d(c[k])/da[i] = 2*a[k-i] (with zero padding)
    # d(c[k]/S^2)/da[i] = 2*a[k-i]/S^2 - 2*(c[k]/S^3)*S*... 
    # Actually: d(c[k]/S^2)/da[i] = (2*a[k-i]*S^2 - c[k]*2*S*1) / S^4 = 2*a[k-i]/S^2 - 2*c[k]/S^3
    # Hmm this is getting complicated. Let me compute it differently.
    # f = 2n * max(c) / S^2 ≈ 2n * L / ... wait
    # L = dmax + log(sum(exp(beta*(d-dmax))))/beta  where d = c/S^2
    # 2n*L is our smoothed objective
    # d(2n*L)/da[i] = 2n * dL/da[i]
    # dL/da[i] = d(dmax)/da[i] + (1/beta) * d(log(se))/da[i]
    # This is complex. Let me use the chain rule differently.
    # 
    # Actually let me just compute the gradient of 2n * (c/S^2)_{max} using:
    # ∇(2n*c[k]/S^2) = 2n * (2*a_shifted(k) * S^2 - c[k] * 2*S) / S^4
    # = (4n * a_shifted(k)) / S^2 - 4n*c[k]/S^3
    # 
    # For the LSE: replace max with weighted sum
    # ∇(2n * sum_k w[k] * c[k] / S^2)
    # = 2n * sum_k w[k] * ∇(c[k]/S^2)
    # = 2n * sum_k w[k] * (2*a_shifted(k)/S^2 - 2*c[k]/S^3)
    # = (4n/S^2) * sum_k w[k]*a_shifted(k) - (4n/S^3) * sum_k w[k]*c[k]
    # = (4n/S^2) * corr(a, w) - (4n/S^3) * <w, c>
    
    h = corr_pair(a, w)
    wcdotc = float(np.dot(w, c))
    grad = (4.0 * n / (S * S)) * h - (4.0 * n * wcdotc / (S * S * S)) * np.ones(n)
    # Wait, also need to account for the dmax part of LSE
    # dL/da[i] includes d(dmax)/da[i] which is the gradient of the max component
    # Actually L = dmax + (1/beta)*log(se), and se = sum exp(beta*(d-dmax))
    # dL/da[i] = ddmax/da[i] + (1/beta) * (1/se) * sum_k exp(beta*(d[k]-dmax)) * (beta*(d[k]-dmax)' )
    #          = ddmax/da[i] + sum_k w[k] * (dd[k]/da[i])  [since dmax' terms cancel]
    #          = ddmax/da[i] + sum_k w[k] * dd[k]/da[i]
    # 
    # And ddmax/da[i] = gradient of the max d[k], which is d[dmax_idx]/da[i]
    # But with LSE smoothing, this is already handled by the w vector.
    # 
    # Actually the correct gradient of L(d) = dmax + (1/beta)log(sum exp(beta(d-dmax))) is:
    # ∇L = (1/beta) * (sum exp(beta(d-dmax)) * beta * ∇d) / sum exp(beta(d-dmax))
    #     = sum_k w[k] * ∇d[k]
    # where w[k] = exp(beta(d[k]-dmax)) / sum_j exp(beta(d[j]-dmax))
    # 
    # So ∇(2n*L) = 2n * sum_k w[k] * ∇(c[k]/S^2)
    # = 2n * sum_k w[k] * (2*a_shifted(k)/S^2 - 2*c[k]/S^3)
    
    # My formula above already gives this. Let me double-check the sign.
    # We want to MINIMIZE 2n*L, so the minimizer follows -∇(2n*L).
    # But for scipy minimize, we just return the gradient (positive direction = increase).
    
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxiter=200):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5:
        return exact_score(a), a
    
    def fun(x):
        x = np.clip(x, 1e-15, None)
        return lse_objective(x, beta)
    
    bounds = [(1e-15, 10.0)] * n
    opts = {
        "maxiter": maxiter,
        "maxfun": maxiter * 10,
        "ftol": 1e-14,
        "gtol": 1e-10,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = normalize_a(res.x)
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return sc, a_out


def generate_inits(n, rng, parent=None, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            # Resample
            x_old = np.linspace(0, 1, arr.size)
            arr = np.interp(x, x_old, arr)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-12:
            arr = np.ones(n)
        inits.append(arr / arr.sum())
    
    if parent is not None:
        pr = normalize_a(parent)
        if pr.size == n:
            add(pr)
            add(pr[::-1])
    
    add(np.ones(n))
    for alpha in (0.5, 1.0, 2.0, 3.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 4, 5):
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * f * x))
    for rate in (1.0, 3.0, 5.0, 10.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    for p in (0.5, 1.0, 2.0, 4.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Smooth random
    for _ in range(5):
        Nf = next_power_of_two(2 * n - 1)
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** rng.uniform(1.0, 4.0))
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + 0.5 * noise / (np.abs(noise).max() + 1e-15))
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))
    
    return inits
```

Hmm, but I'm worried this is too similar to the previous approach. Let me think about what could be genuinely different.

Key differences I can make:
1. **KKT-based active set**: Instead of using a fixed threshold or top-k, use the KKT conditions to determine the active set. At the optimum, the subgradient should be uniform on the support. I can iteratively add/remove lags from the active set based on whether they're "active" (at the max) and whether the KKT conditions are satisfied.

```python
import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_conv(f, g):
    """Compute cross-convolution f*g (full)."""
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n_f = f.size
    n_g = g.size
    if n_f == 0 or n_g == 0:
        return np.empty(0, dtype=np.float64)
    M = n_f + n_g - 1
    N = next_power_of_two(M)
    F = np.fft.rfft(f, N)
    G = np.fft.rfft(g, N)
    out = np.fft.irfft(F * G, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """h[p] = sum_k w[k] * a[k-p] for p in 0..n-1 (a[j]=0 for j out of [0,n-1])."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax)


def full_score(a):
    """Full score with arbitrary sum."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0 or m == n_new:
        return a.copy() if m == n_new else np.ones(n_new) / n_new
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, a).astype(np.float64)


def downsample(a, factor):
    n = len(a)
    n_new = max(4, n // factor)
    blocks = a[:n_new * factor].reshape(n_new, factor)
    return blocks.mean(axis=1)


def generate_inits(n, rng, parent=None, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            x_old = np.linspace(0, 1, max(1, arr.size))
            arr = np.interp(x, x_old, arr)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.clip(arr, 0.0, None)
        if arr.sum() < 1e-12:
            arr = np.ones(n) * 0.01
        inits.append(arr / arr.sum())

    if parent is not None:
        pr = np.asarray(parent, dtype=np.float64).ravel()
        if pr.size != n:
            pr = resample_sequence(pr, n)
        pr = normalize_a(pr)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0, 3.0):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    for alpha in (0.3, 0.5, 1.0, 1.5):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 4, 5, 7, 10):
        for amp in (0.3, 0.7, 1.2):
            add(1.0 + amp * np.cos(2.0 * np.pi * f * x))
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
        add(np.exp(-rate * np.abs(x - 0.5)) + 0.01)
    for p in (0.3, 0.5, 1.0, 2.0, 4.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.005)
    for width in (0.05, 0.08, 0.12, 0.2):
        for center in (0.15, 0.25, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) +
                np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Spectral inits: flat-ish magnitude spectrum
    for _ in range(6):
        Nf = next_power_of_two(2 * n - 1)
        freqs = np.fft.rfftfreq(Nf) * Nf
        target_power = np.ones(len(freqs))
        target_power[0] = 0.0
        noise = target_power * rng.uniform(0.3, 1.0, size=len(target_power))
        phase = rng.uniform(-np.pi, np.pi, size=len(noise))
        spec = np.sqrt(noise) * np.exp(1j * phase)
        spec[0] = 0.0
        sig = np.fft.irfft(spec, Nf)[:n]
        add(0.5 + np.abs(sig) * 0.5 + 0.01)
    for _ in range(3):
        add(0.05 + 0.95 * rng.random(n))
    # Bimodal smooth
    for wc in [(0.3, 0.7), (0.2, 0.8), (0.25, 0.5, 0.75)]:
        sig = sum(np.exp(-(((x - c) / 0.08) ** 2)) for c in wc) + 0.05
        add(sig)

    return inits


def kkt_residual(a, q):
    """How non-uniform is the pressure on the support of a."""
    h = corr_pair(a, q)
    support = a > 1e-10
    nsup = support.sum()
    if nsup < 2:
        return 0.0
    h_sup = h[support]
    mean_h = float(np.mean(h_sup))
    if mean_h < 1e-15:
        return 0.0
    return float(np.std(h_sup) / mean_h)


def mirror_step(a, h, eta):
    """Mirror descent step given the pressure h."""
    hmax = max(float(np.max(h)), 1e-15)
    log_a = np.log(a + 1e-300)
    log_a -= eta * h / hmax
    mx = float(np.max(log_a))
    log_a -= mx
    return np.exp(log_a)


def solve_lp_direction(f, M, active_lags, timeout=30.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns g or None.
    """
    from scipy.optimize import linprog
    n = len(f)
    n_lags = len(active_lags)
    if n_lags == 0 or n > 5000:
        return None

    c_obj = -np.ones(n)
    A_ub = np.zeros((n_lags, n))
    for idx, k in enumerate(active_lags):
        for p in range(n):
            j = int(k) - p
            if 0 <= j < n:
                A_ub[idx, p] = f[j]
    b_ub = np.full(n_lags, M)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"time_limit": timeout, "presolve": True})
        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.clip(g, 0, None)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    if time.monotonic() - t0 > timeout * 1.2:
        return None
    return None

def _compute_q(conv, mode, frac=0.02, k=None, beta=50.0):
    """Compute active set weights."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size

    if mode == "threshold":
        thr = dmax * (1.0 - frac)
        active = conv >= thr
        if active.sum() < 1:
            active[int(np.argmax(conv))] = True
        max_count = max(3, int(0.08 * conv.size))
        if active.sum() > max_count:
            idx = np.argpartition(conv, -max_count)[-max_count:]
            active_new = np.zeros(conv.size, dtype=bool)
            active_new[idx] = True
            active = active_new
        w = active.astype(np.float64)
        s = w.sum()
        if s < 1e-15:
            return np.ones(conv.size) / conv.size
        return w / s

    elif mode == "softmax":
        z = beta * (conv - dmax)
        z = np.clip(z, -200.0, 0.0)
        e = np.exp(z)
        s = e.sum()
        if s < 1e-15:
            w = np.zeros(conv.size)
            w[int(np.argmax(conv))] = 1.0
            return w
        return e / s

    elif mode == "topk":
        if k is None:
            k = 5
        k = min(k, conv.size)
        idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(conv.size)
        w[idx] = 1.0 / k
        return w

    else:
        return np.ones(conv.size) / conv.size


def lse_objective_and_grad(a, beta):
    """LSE-smoothed objective and gradient for scipy L-BFGS-B."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = c / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 0.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    h = corr_pair(a, w)
    wcdotc = float(np.dot(w, c))
    # grad of 2n*L w.r.t. a:
    # = 2n * sum_k w[k] * d(c[k]/S^2)/da
    # d(c[k]/S^2)/da[i] = 2*a[k-i]/S^2 - 2*c[k]/S^3
    grad = (4.0 * n / (S * S)) * h - (4.0 * n * wcdotc / (S * S * S)) * np.ones(n)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxiter=100):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5:
        return exact_score(a), a

    def fun(x):
        x = np.clip(x, 1e-15, None)
        return lse_objective_and_grad(x, beta)

    bounds = [(1e-15, 10.0)] * n
    opts = {
        "maxiter": maxiter,
        "maxfun": maxiter * 8,
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return sc, a_out


def multi_resolution_optimize(a, deadline, verbose=True):
    """Optimize at multiple resolutions: coarse -> fine."""
    n = len(a)
    a = normalize_a(a)

    # Try downsampling by 2 and 4, optimize, then upsample
    for factor in [4, 2]:
        if time.monotonic() > deadline - 30:
            break
        n_coarse = max(64, n // factor)
        a_coarse = downsample(a, factor)
        a_coarse = normalize_a(a_coarse)
        rem = (deadline - time.monotonic()) * 0.3
        if rem < 20:
            continue
        sc_c, a_c = run_mirror_descent(a_coarse, time.monotonic() + rem,
                                       verbose=False, tag="cr{}".format(factor))
        # Upsample
        a_up = resample_sequence(a_c, n)
        a_up = normalize_a(a_up)
        sc_up = exact_score(a_up)
        if verbose:
            print("  MR factor={}: coarse_sc={:.9f} upsamp_sc={:.9f}".format(factor, sc_c, sc_up))
        # Blend with current best
        for tBlend in [0.3, 0.5, 0.7]:
            blended = (1.0 - tBlend) * a + tBlend * a_up
            blended = normalize_a(blended)
            sc_b = exact_score(blended)
            if sc_b < exact_score(a):
                a = blended
                if verbose:
                    print("    MR blend t={}: {:.9f}".format(tBlend, sc_b))
    return a


def fine_polish(a, deadline, verbose=False):
    """Very fine-tuned mirror descent with small steps and large active set."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Use large, smooth active set
    for it in range(5000):
        if time.monotonic() > deadline - 1:
            break
        conv = fast_self_conv(a)
        dmax = float(np.max(conv))
        sc = 2.0 * n * dmax

        # Multiple active set strategies
        for frac in [0.005, 0.01, 0.02, 0.05]:
            q = _compute_q(conv, "threshold", frac=frac)
            h = corr_pair(a, q)
            for eta in [0.01, 0.03, 0.08, 0.15, 0.3]:
                new_a = mirror_step(a, h, eta)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc - 1e-14:
                    best_sc = new_sc
                    best_a = new_a
                    a = new_a
                    if verbose:
                        print("    fp it{} frac={}: {:.10f}".format(it, frac, best_sc))

        # Softmax with high beta (nearly top-1 but smooth)
        q_s = _compute_q(conv, "softmax", beta=200.0)
        h_s = corr_pair(a, q_s)
        for eta in [0.02, 0.05, 0.1]:
            new_a = mirror_step(a, h_s, eta)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc - 1e-14:
                best_sc = new_sc
                best_a = new_a
                a = new_a

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-resolution mirror descent with KKT-adaptive active sets,
    periodic LP direction finding, LSE-LBFGS polish, and fine-tuning.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p
    except Exception:
        pass

    # Determine n
    n = 10240
    if parent is not None:
        n = int(parent.size)
        if n < 2048:
            parent = resample_sequence(parent, 10240)
            n = 10240
        elif n > 20000:
            parent = resample_sequence(parent, 16384)
            n = 16384

    # Initial sequence
    if parent is not None:
        a = normalize_a(resample_sequence(parent, n))
    else:
        inits = generate_inits(n, rng, None, cap=20)
        a = inits[0]
        best_init_sc = np.inf
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a = init
        print("No parent. Best init: {:.9f}".format(best_init_sc))
        a = normalize_a(a)

    best_a = a.copy()
    best_sc = exact_score(a)
    print("Start: n={}, score={:.10f}".format(n, best_sc))

    # Phase 0: Multi-resolution warm start (5% time)
    p0_dead = t_start + 0.05 * total_budget
    a_mr = multi_resolution_optimize(a, p0_dead, verbose=True)
    sc_mr = exact_score(a_mr)
    if sc_mr < best_sc:
        best_sc = sc_mr
        best_a = a_mr.copy()
        print("After MR warm-start: {:.10f}".format(best_sc))

    # Phase 1: Main mirror descent (45% time)
    p1_dead = t_start + 0.50 * total_budget
    use_lp = (n <= 3000)
    sc, a = run_mirror_descent(best_a, deadline=p1_dead, verbose=True, tag="p1", use_lp=use_lp)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After P1: {:.10f}".format(best_sc))

    # Phase 2: Second mirror descent with different schedule (15% time)
    p2_dead = t_start + 0.65 * total_budget
    sc, a = run_mirror_descent(best_a, deadline=p2_dead, verbose=False, tag="p2", use_lp=use_lp)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After P2: {:.10f}".format(best_sc))

    # Phase 3: LSE-LBFGS refinement (20% time)
    p3_dead = t_start + 0.85 * total_budget
    a_work = normalize_a(best_a)
    betas = [200, 500, 1000, 2500, 5000, 10000, 20000, 50000]
    for beta in betas:
        if time.monotonic() > p3_dead - 15:
            break
        rem = p3_dead - time.monotonic()
        if rem < 20:
            break
        mi = min(80, max(20, int(rem / 0.3)))
        sc, a_work = run_lbfgs(a_work, beta, p3_dead, maxiter=mi)
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()
            print("  P3 beta={}: {:.10f}".format(beta, best_sc))

    # Phase 4: Fine polish (5% time)
    p4_dead = t_start + 0.93 * total_budget
    sc, a = fine_polish(best_a, p4_dead, verbose=False)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
        print("After P4: {:.10f}".format(best_sc))

    # Phase 5: Restart attempts (5% time)
    p5_dead = deadline - 10.0
    max_restarts = 5
    while time.monotonic() < p5_dead - 40 and max_restarts > 0:
        rem = p5_dead - 40.0 - time.monotonic()
        if rem < 50:
            break
        max_restarts -= 1
        # Different perturbation strategies
        mode = rng.integers(0, 4)
        if mode == 0:
            p = best_a * np.exp(0.08 * rng.standard_normal(n))
        elif mode == 1:
            p = best_a * (1.0 + 0.1 * rng.standard_normal(n))
            p = np.clip(p, 1e-6, None)
        elif mode == 2:
            # Gaussian smoothing perturbation
            k = int(rng.integers(7, max(9, n // 500)))
            if k % 2 == 0:
                k += 1
            k = min(k, n if n % 2 == 1 else n - 1)
            if k < 3:
                k = 5
            c_half = k // 2
            sig = max(1.0, c_half / 2.0)
            kern = np.exp(-((np.arange(k) - c_half) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            smoothed = np.convolve(best_a, kern, mode="same")
            p = (1.0 - 0.2) * best_a + 0.2 * smoothed
        else:
            # V-shape perturbation
            x = np.linspace(0.0, 1.0, n)
            for shape in [0, 1]:
                if shape == 0:
                    p = best_a * (1.0 - 0.15 * (2.0 * x - 1.0) ** 2)
                else:
                    p = best_a * (1.0 + 0.15 * (2.0 * x - 1.0) ** 2)
                p = normalize_a(p)
                chunk_dead = time.monotonic() + rem * 0.4
                sc_r, seq = run_mirror_descent(p, deadline=chunk_dead, verbose=False, tag="r")
                if sc_r < best_sc:
                    best_sc = sc_r
                    best_a = seq.copy()
                    print("Restart r{} s{}: {:.10f}".format(max_restarts, shape, best_sc))
        p = normalize_a(p)
        chunk_dead = time.monotonic() + rem * 0.6
        sc_r, seq = run_mirror_descent(p, deadline=chunk_dead, verbose=False, tag="r")
        if sc_r < best_sc:
            best_sc = sc_r
            best_a = seq.copy()
            print("Restart r{}: {:.10f}".format(max_restarts, best_sc))

    # Final micro-polish
    if time.monotonic() < deadline - 5:
        sc, a = fine_polish(best_a, deadline - 3, verbose=False)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final output
    print("Best internal: {:.10f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
```
