from scipy.optimize import linprog

def solve_lp(f, M, n):
    # Maximize sum(g) subject to g >= 0 and f*g <= M (element-wise)
    # Variables: g_0, ..., g_{n-1}
    
    # The convolution (f * g)_k = sum_j f_j * g_{k-j}
    # For k = 0: f_0 * g_0
    # For k = 1: f_0 * g_1 + f_1 * g_0
    # ...
    # For k = n-1: f_0 * g_{n-1} + f_1 * g_{n-2} + ... + f_{n-1} * g_0
    # For k = n: f_1 * g_{n-1} + f_2 * g_{n-2} + ... + f_{n-1} * g_0
    # ...
    
    # Build the constraint matrix A (2n-1 rows, n columns)
    # A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Maximize -c@g where c = -ones(n)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)
    return res.x
```

def weighted_conv(p, a):
    """Compute h[i] = sum_k p[k] * a[k-i] for i=0,...,n-1.
    p has length 2n-1, a has length n."""
    n = len(a)
    a_flipped = a[::-1]
    M = 1 << (3*n - 2 - 1).bit_length()
    P = np.fft.rfft(p, M)
    AF = np.fft.rfft(a_flipped, M)
    result = np.fft.irfft(P * AF, M)
    return result[n-1 : 2*n-1]
```
