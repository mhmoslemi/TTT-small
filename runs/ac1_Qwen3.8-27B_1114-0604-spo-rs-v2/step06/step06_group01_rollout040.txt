Improved multi-phase optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. Exact dual LP solve (greedy interval covering) for fast constraint analysis
2. Multi-peak subgradient descent weighted by dual variables

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import uniform_filter1d

def self_conv(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def solve_dual_lp(a):
    """
    Solve the dual LP using greedy interval covering:
    min sum(y_k) s.t. sum_{k=j}^{j+n-1} y_k >= 1/a[j] for all j, y >= 0
    
    Returns y (length 2n-1) and the optimal value (sum of y).
    """
    n = len(a)
    y = np.zeros(2 * n - 1)
    S = np.zeros(n)  # S[j] = sum(y[j : j+n])
    
    for j in range(n):
        if a[j] > 1e-15:
            target = 1.0 / a[j]
            if S[j] < target:
                delta = target - S[j]
                y[j + n - 1] += delta
                S[j:] += delta
        else:
            # a[j] ~ 0, constraint is 0 >= 0 (always satisfied)
            pass
    
    return y, float(y.sum())


def multi_peak_subgradient_step(a, y_dual, n_steps=200, eta_init=0.01):
    """
    Multi-peak subgradient descent using dual weights.
    g[j] = 2 * sum_k (y[k]/sum(y)) * a[k-j] for valid k-j
    Step: a_new = a - eta * g, projected to {a>=0, sum(a)=1}
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    sy = float(y_dual.sum())
    if sy < 1e-15:
        return best_a
    
    w = y_dual / sy  # normalized weights
    
    N_fft = 1 << (3 * n - 1).bit_length()
    FA = np.fft.rfft(a, N_fft)
    
    for it in range(n_steps):
        FW = np.fft.rfft(w, N_fft)
        corr = np.fft.irfft(np.conj(FA) * FW, N_fft)[:n]
        g = 2.0 * corr
        
        S2 = S * S
        grad = 2.0 * n * (g / S2)  # gradient of 2n * max_approx / S^2 (approx)
        
        # Adaptive step size
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        eta = eta_init / (1.0 + 0.01 * it)
        
        a_try = a - eta * grad
        a_try = np.maximum(a_try, 0.0)
        St = float(a_try.sum())
        if St < 1e-12:
            break
        a_try = a_try / St
        
        sc_try = compute_score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            # Update FFT of a
            FA = np.fft.rfft(a, N_fft)
        else:
            # Halve eta if no improvement
            eta_init = max(eta_init * 0.7, 1e-6)
            # Try smaller step
            a_try2 = a - 0.3 * eta * grad
            a_try2 = np.maximum(a_try2, 0.0)
            St2 = float(a_try2.sum())
            if St2 > 1e-12:
                a_try2 = a_try2 / St2
                sc_try2 = compute_score(a_try2)
                if sc_try2 < best_sc:
                    best_sc = sc_try2
                    best_a = a_try2.copy()
                    a = a_try2
                    FA = np.fft.rfft(a, N_fft)
    
    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    
    # Gradient of L w.r.t. a: (B^T p)[j] = sum_i a[i] * p[i+j] 
    # which equals cross-correlation
    N_fft = 1 << (3 * n - 1).bit_length()
    RA = np.fft.rfft(a, N_fft)
    P = np.fft.rfft(p, N_fft)
    corr = np.fft.irfft(np.conj(RA) * P, N_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_fista_inner(a, M, n_iters=10000, beta=300.0, x_init=None):
    """
    FISTA for: min (1/beta)*log(sum exp(beta*(conv(a,x)-M))) s.t. x >= 0
    Correct gradient via FFT cross-correlation.
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n)
    u = x.copy()
    t_mom = 1.0

    # Lipschitz constant estimate
    sum_a = float(a.sum())
    L_est = beta * n * sum_a * sum_a * 2.0 + 1e-12
    step = 1.0 / L_est

    N_fft = 1 << (3 * n - 1).bit_length()
    RA = np.fft.rfft(a, N_fft)

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = exp_z / exp_sum

        RP = np.fft.rfft(p, N_fft)
        corr = np.fft.irfft(np.conj(RA) * RP, N_fft)[:n]
        grad = corr  # gradient of f w.r.t. x

        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=30, fista_iters=15000, time_limit=None,
                   beta=300.0, use_dual_mixing=True):
    """
    LP-based alternating method with dual-informed mixing.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0
    x_warm = None

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())
        
        # Solve dual for constraint analysis
        y_dual, dual_val = solve_dual_lp(a)
        
        x = lp_fista_inner(a, M, n_iters=fista_iters, beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Use dual value to inform mixing
        if use_dual_mixing and dual_val > 0:
            # LP optimum value is M * dual_val
            # Current feasible value is 1 (sum(a)=1)
            # Improvement ratio
            improvement = M * dual_val
            if improvement > 1.0:
                t_max = min(1.0, 0.5 * (improvement - 1.0) + 0.3)
            else:
                t_max = 0.3
        else:
            t_max = 0.5

        t_candidates = np.linspace(0.05, t_max, 8)
        best_t = 0.2
        best_t_sc = best_sc
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_t_sc:
                best_t_sc = sc_try
                best_t = t
        t = best_t
        a_new = (1.0 - t) * a + t * x

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 8:
                break

        a = a_new

    return best_a, best_sc, x_warm


def error_reduction(a, n_iters=80, alpha=0.7):
    """Newton-like error reduction in frequency domain to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.7 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving_multipeak(a, n_steps=300, top_k=5):
    """
    Multi-peak subgradient: target top-K peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    N_fft = 1 << (3 * n - 1).bit_length()
    FA = np.fft.rfft(a, N_fft)

    eta = 0.005
    no_improve = 0

    for it in range(n_steps):
        c = self_conv(a)
        
        # Find top-K peaks
        c_sorted = np.sort(c)[::-1]
        k_threshold = c_sorted[min(top_k - 1, len(c_sorted) - 1)]
        peak_mask = c >= k_threshold
        n_peaks = int(peak_mask.sum())
        
        if n_peaks == 0:
            break
        
        # Weight peaks by excess above mean
        c_mean = float(c.mean())
        weights = np.maximum(c - c_mean, 0.0)
        weights[~peak_mask] = 0.0
        w_sum = float(weights.sum())
        if w_sum < 1e-15:
            break
        weights = weights / w_sum

        # Compute subgradient: g[j] = 2 * sum_k weights[k] * a[k-j]
        FW = np.fft.rfft(weights, N_fft)
        corr = np.fft.irfft(np.conj(FA) * FW, N_fft)[:n]
        g = 2.0 * corr
        
        # Full gradient of score (with normalization)
        S2 = S * S
        c_max = float(c.max())
        grad = 2.0 * n * (g / S2 - 2.0 * c_max / (S2 * S))
        
        # Adaptive step
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = eta / (gn + 1e-10) * (best_sc - 1.0 + 0.5)
        step = np.clip(step, 1e-6, 0.1)
        
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            a_try = a - step * mult * grad
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                FA = np.fft.rfft(a, N_fft)
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            eta *= 0.8
            if no_improve > 50:
                break
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x + 0.5))
    inits.append(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.7 * (np.cos(2.0 * np.pi * x) ** 4), 0.01, 1000.0))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x + 0.5)
                + 0.1 * np.cos(8.0 * np.pi * x + 1.0))
    inits.append(np.clip(0.3 + 0.3 * x + 0.2 * np.cos(2.0 * np.pi * 3 * x), 0.01, 1000.0))
    # Asymmetric inits (important since optimal may be asymmetric)
    inits.append(np.clip(0.3 + 0.4 * x + 0.3 * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.4 - 0.2 * x + 0.3 * np.sin(2.0 * np.pi * x), 0.01, 1000.0))
    return inits


def propose_candidate(budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: error reduction, multi-peak subgradient, LP alternating,
    L-BFGS polish, and final peak shaving.
    """
    rng = np.random.default_rng(42)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 20.0
    total_budget = float(budget_s) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, "
          f"len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [6962, 8192, 10240, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Quick global search with error reduction (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * total_budget
    print(f"\n--- Phase 1: Global Search (Error Reduction) ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 15:
            break
        print(f"  n={n}, remaining={remaining:.0f}s")
        inits = build_initializations(n, rng, parent)
        max_inits = max(3, int(remaining / 5.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=80, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Multi-peak subgradient + LP alternating (40% time)
    # ==================================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n--- Phase 2: Multi-Peak + LP Alternating ---")

    n_main = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # First: multi-peak subgradient
    if time.monotonic() < phase2_deadline - 60:
        print(f"  Multi-peak subgradient (n={n_main})...")
        t0 = time.monotonic()
        a_mp = peak_shaving_multipeak(a_work, n_steps=500, top_k=5)
        sc = compute_score(a_mp)
        report(sc, a_mp, "multi-peak")
        print(f"    time={time.monotonic()-t0:.1f}s")
        a_work = best_a.copy()
        S = float(a_work.sum)
        if S > 0:
            a_work = a_work / S

    # LP alternating with progressive configs
    configs = [
        (8000, 150.0, 25),
        (12000, 300.0, 25),
        (15000, 500.0, 20),
        (20000, 800.0, 15),
    ]

    x_warm = None
    for fi, (fista_it, beta_val, max_outer) in enumerate(configs):
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"  LP round {fi}: fista={fista_it}, beta={beta_val}, "
              f"outer={max_outer}, remaining={remaining:.0f}s")

        a_out, sc_out, x_warm = lp_alternating(
            a_work,
            max_outer_iters=max_outer,
            fista_iters=fista_it,
            time_limit=phase2_deadline - 20,
            line_search=True,
            beta=beta_val,
            use_dual_mixing=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    # Try another length if time permits
    for n in lengths:
        if n == n_main:
            continue
        if time.monotonic() > phase2_deadline - 50:
            break
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out, _ = lp_alternating(
            a_test, max_outer_iters=12, fista_iters=10000,
            time_limit=phase2_deadline - 10, line_search=True,
            beta=400.0, use_dual_mixing=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Multi-peak refinement + subgradient (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.70 * total_budget
    print(f"\n--- Phase 3: Multi-Peak Refinement ---")

    if time.monotonic() < phase3_deadline - 30:
        # Multiple rounds of multi-peak with different top_k
        for k_val in [3, 5, 8, 10]:
            if time.monotonic() > phase3_deadline - 20:
                break
            a_mp = peak_shaving_multipeak(best_a, n_steps=300, top_k=k_val)
            sc = compute_score(a_mp)
            report(sc, a_mp, f"multi-peak k={k_val}")

        # Error reduction again on the improved solution
        a_er = error_reduction(best_a, n_iters=100, alpha=0.6)
        sc = compute_score(a_er)
        report(sc, a_er, "ER phase3")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: L-BFGS refinement with progressive beta (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.85 * total_budget
    print(f"\n--- Phase 4: L-BFGS Refinement ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S

    beta_schedule = [50, 100, 200, 500, 1000, 2000, 4000, 6000, 8000]
    for beta in beta_schedule:
        if time.monotonic() > phase4_deadline - 20:
            break
        print(f"  L-BFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        maxit = 8000 if beta < 500 else (5000 if beta < 2000 else 3000)
        sc, a_opt, res = run_lbfgs(a_norm, beta, maxiter=maxit, maxfun=300000,
                                    gtol=1e-13)
        elapsed = time.monotonic() - t0
        print(f"score={sc:.8f}, time={elapsed:.1f}s")
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish - LP + multi-peak + verify (15% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Final LP
    if time.monotonic() < deadline - 120:
        print("  Final LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 0:
            a_final_in = a_final_in / S
        a_final, sc_final, _ = lp_alternating(
            a_final_in, max_outer_iters=8, fista_iters=15000,
            time_limit=deadline - 80, line_search=True, beta=600.0,
            use_dual_mixing=True
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # Final multi-peak
    if time.monotonic() < deadline - 60:
        a_mp = peak_shaving_multipeak(best_a, n_steps=300, top_k=5)
        sc = compute_score(a_mp)
        report(sc, a_mp, "final multi-peak")

    # Final L-BFGS with highest beta
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum)
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 10000, maxiter=5000, maxfun=200000,
                                  gtol=1e-14)
        report(sc, a_opt, "final LBFGS b=10000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
