def peak_shave(a, n_iters, n_peaks, deadline):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.01
    patience = 150
    no_improve = 0
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        k_split = min(n_peaks, len(c))
        if k_split >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -k_split)[-k_split:]
        sg = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(ki, n - 1)
            if je >= js:
                seg = a[js: je + 1]
                sg[js: je + 1] += 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        alpha = step
        for _ in range(15):
            a_t = a - alpha * grad
            a_t = np.clip(a_t, 0.0, 1000.0)
            St = float(a_t.sum())
            if St < 1e-12:

            a_t = a_t / St
            At = np.fft.rfft(a_t, M_fft)
            ct = np.fft.irfft(At * At, M_fft)[:2 * n - 1]
            St = float(a_t.sum())
            if St < 1e-12:
                break
            sc_t = 2.0 * n * float(ct.max()) / (St * St)
            if sc_t < sc or alpha < 1e-14:
                a = a_t
                break
            alpha *= 0.5
        step = max(step * 0.9999, 1e-8)
    return best_a

        alpha = a_start + (a_end - a_start) * prog
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real ** 2 + A.imag ** 2)
        ph = np.angle(A)
        if len(amp) > 1:
            C = max(np.median(amp[1:]), 1e-20)
        else:
            C = max(amp[0], 1e-20)
        amp_s = np.maximum(amp, 1e-20)
        amp_n = amp_s * (C / amp_s) ** alpha
        A_n = amp_n * np.exp(1j * ph)
        a = np.fft.irfft(A_n, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def spectral_surgery(a, n_iters, n_bins_target, deadline):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    M = 2 * n - 1
    n_rfft = M_fft // 2 + 1
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        ph = np.angle(A)
        c = np.fft.irfft(A * A, M_fft)[:M]
        S = float(a.sum())
        sc = 2.0 * n * float(c.max()) / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        amp_n = amp.copy()
        if n_rfft > 2:
            non_dc = amp_n[1:].copy()
            nt = min(n_bins_target, len(non_dc))
            if nt > 0:
                top = np.argpartition(non_dc, -nt)[-nt:]
                dec = 0.005 + 0.02 * (1.0 - it / max(1, n_iters))
                amp_n[1 + top] *= (1.0 - dec)
            med = np.median(amp_n[1:])
            amp_n[1:] = np.maximum(amp_n[1:], 0.01 * med)
        A_n = amp_n * np.exp(1j * ph)
        a = np.fft.irfft(A_n, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    return best_a


def coord_descend(a, n_rounds, n_top, delta, deadline):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 2 * n - 1
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Get top elements
        nt = min(n_top, n)
        if nt >= n:
            idx = np.arange(n)
        else:
            idx = np.argpartition(a, -nt)[-nt:]
        idx = idx[np.argsort(a[idx])[::-1]]
        for i in idx:
            if time.monotonic() > deadline:
                break
            i = int(i)
            orig = a[i]
            for sign in [1.0, -1.0]:
                a[i] = max(0.0, orig + sign * delta * orig)
                S_new = float(a.sum())
                if S_new < 1e-12:
                    a[i] = orig
                    continue
                sc_new = compute_score(a)
                if sc_new < sc:
                    sc = sc_new
                else:
                    a[i] = orig
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    return best_a


def resample_seq(a, n_new):
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_inits(n, rng, parent):
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * x), 0.01))
    inits.append(np.exp(-3.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    freqs = rng.integers(1, 10, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.5, size=5)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2.0 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))
    inits.append(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * x)), 0.01))
    inits.append(np.maximum(0.1 + 0.5 * x**0.5 + 0.5 * (1.0 - x)**0.5, 0.01))
    if parent is not None:
        inits.append(resample_seq(parent, n))
    for a in inits:
        S = float(a.sum())
        if S > 1e-12:
            a[:] = a / S
    return inits
