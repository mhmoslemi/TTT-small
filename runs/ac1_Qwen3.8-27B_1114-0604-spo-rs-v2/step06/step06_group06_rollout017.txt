def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1: return np.array([a[0]*a[0]])
    M=2*n-1
    N=1 << (M-1).bit_length()
    A=np.fft.rfft(a, N)
    out=np.fft.irfft(A*A, N)[:M]
    return np.nan_to_num(out, 0,0,0)

"""
Search for non-negative step heights minimizing
    2 * n * max_k conv(a, a)_k / (sum a)^2.

Strategy:
- Work with normalized sums for numerical stability.
- Use FFT-based self-convolution and sparse/dense active-set gradients.
- Multiplicative (exponentiated) subgradient updates with several active
  weightings: top-k uniform, softmax over top-k, excess over top-k, and
  global LSE-smoothed weights.
- Coarse-to-fine initialization: a quick low-resolution survey followed by
  high-resolution refinement.
- Adaptive step-size line search, stagnation restarts, LSE/L-BFGS-B polish,
  and final subgradient refinement.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

from operator import itemgetter


_FFT_N_CACHE = {}
TARGET = 1.5027
DIRECT_LIMIT = 96

ACTIVE_CONFIGS = (
    (0, 16, 0.000, 0.0),
    (1, 24, 0.000, 40.0),
    (0, 32, 0.002, 0.0),
    (2, 48, 0.000, 0.0),
    (1, 64, 0.002, 80.0),
    (0, 80, 0.004, 0.0),
    (2, 96, 0.000, 0.0),

)

LSE_BETAS = (10.0, 25.0, 60.0, 150.0, 400.0, 1000.0)


def _fft_n(length):
    """Next power-of-two FFT size for a linear convolution length."""
    m = int(max(1, length))
    n = _FFT_N_CACHE.get(m)
    if n is None:
        n = 1 << (m - 1).bit_length()
        _FFT_N_CACHE[m] = n
    return n


def fft_ns_for_len(n):
    """FFT sizes for self-convolution and weighted-gradient convolution."""
    n = int(max(1, n))
    return _fft_n(2 * n - 1), _fft_n(3 * n - 2)


def normalize_nonneg(a):
    """Clip to nonnegative finite values and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.zeros(0, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1.0e6, neginf=0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1.0e6)
    s = float(a.sum())
    if s <= 1.0e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def resample_length(a, n):
    """Linearly resample a 1D sequence to length n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(max(1, n))
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, n)
    return np.interp(x1, x0, a).astype(np.float64)


def fast_self_conv(a, N=None):
    """Full linear self-convolution using FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    if N is None:
        N = _fft_n(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    out = np.maximum(out, 0.0)
    return out[:M]


def score_normalized_pre(a, n, N):
    """Score assuming a is already normalized to sum 1."""
    c = fast_self_conv(a, N)
    if c.size == 0:
        return float("inf")
    cm = float(np.max(c))
    if cm < 0.0:
        cm = 0.0
    if not np.isfinite(cm):
        return float("inf")
    return float(2.0 * float(n) * cm)


def internal_score_seq(a):
    """Scale-invariant internal score for arbitrary positive sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.nan_to_num(a, nan=0.0, posinf=1.0e6, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1.0e-8:
        return float("inf")
    n = int(a.size)
    c = fast_self_conv(a / s)
    if c.size == 0:
        return float("inf")
    cm = float(np.max(c))
    if cm < 0.0:
        cm = 0.0
    return float(2.0 * float(n) * cm)


def grad_sparse_direct(a, idx, wvals):
    """
    Sparse active-set gradient h[l] = sum_j a[j] * w[l+j].

    Direct slice addition is faster when the active set is small.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    h = np.zeros(n, dtype=np.float64)
    for k, wk in zip(idx, wvals):
        wk = float(wk)
        if wk == 0.0:
            continue
        k = int(k)
        lw = k - (n - 1)
        if lw < 0:
            lw = 0
        uw = k
        if uw >= n:
            uw = n - 1
        if lw > uw:
            continue
        # h[lw : uw+1] += wk * a[k-lw, k-lw-1, ..., k-uw]
        h[lw:uw + 1] += wk * a[k - lw:k - uw - 1:-1]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(h, 0.0)


def grad_dense(a, w, N=None):
    """Dense active-set gradient h[l] = sum_j a[j] * w[l+j]."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    M = n + m - 1
    if N is None:
        N = _fft_n(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)
    full = np.nan_to_num(full, nan=0.0, posinf=0.0, neginf=0.0)
    out = full[n - 1:2 * n - 1]
    out = np.maximum(out, 0.0)
    return out


def make_active(c, ctype, K, eps, alpha, maxK=None):
    """
    Return (indices, weights) for an active set on convolution c.

    ctype:
      0: uniform
      1: softmax over selected values (alpha > 0)
      2: excess above mean of selected values
    """
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.empty(0, dtype=np.int64), np.empty(0, dtype=np.float64)

    K = int(max(1, min(K, L)))
    if maxK is None:
        maxK = max(K, 64)
    maxK = int(max(1, min(maxK, L)))

    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        idx = np.arange(L, dtype=np.int64)
        return idx, np.ones(L, dtype=np.float64) / float(L)

    if eps > 0.0:
        thr = cmax * (1.0 - float(eps))
        cnt = int(np.count_nonzero(c >= thr))
        if cnt <= maxK:
            idx = np.nonzero(c >= thr)[0]
            if idx.size == 0:
                idx = np.argpartition(c, -K)[-K:]
        else:
            idx = np.argpartition(c, -maxK)[-maxK:]
    else:
        idx = np.argpartition(c, -K)[-K:]

    vals = c[idx]
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    sz = int(vals.size)
    if sz == 0:
        idx = np.argpartition(c, -1)[-1:]
        vals = c[idx]
        sz = 1

    if ctype == 0 or alpha <= 0.0:
        w = np.full(sz, 1.0 / float(sz), dtype=np.float64)
    elif ctype == 1:
        z = float(alpha) * (vals - cmax)
        z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=-700.0)
        z -= float(np.max(z))
        z = np.clip(z, -700.0, 700.0)
        w = np.exp(z)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.full(sz, 1.0 / float(sz), dtype=np.float64)
        else:
            w /= s
    else:
        meanv = float(np.mean(vals))
        w = vals - meanv
        np.clip(w, 0.0, None, out=w)
        s = float(w.sum())
        if not np.isfinite(s) or s <= 1.0e-300:
            w = np.full(sz, 1.0 / float(sz), dtype=np.float64)
        else:
            w /= s

    return idx.astype(np.int64, copy=False), w.astype(np.float64, copy=False)


def soft_weights(c, beta):
    """Dense LSE-style weights over all convolution values."""
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.empty(0, dtype=np.float64)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)

    z = float(beta) * (c / cmax - 1.0)
    z = np.nan_to_num(z, nan=-700.0, posinf=0.0, neginf=-700.0)
    z = np.clip(z, -700.0, 700.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    w /= s

    w[w < 1.0e-10] = 0.0
    s2 = float(w.sum())
    if s2 > 0.0:
        w /= s2
    else:
        w = np.ones(L, dtype=np.float64) / float(L)
    return w


def perturb_smooth(a, rng, strength):
    """Smooth multiplicative log-normal perturbation."""
    a = normalize_nonneg(a)
    n = int(a.size)
    if n == 0:
        return a
    k = int(min(31, max(5, n // 400)))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = 3 if n % 2 == 1 else max(3, n - 1)
    if k < 3:
        k = 3

    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kats = float(kern.sum())
    if kats > 0.0:
        kern /= kats

    noise = np.convolve(rng.standard_normal(n), kern, mode="same")
    noise -= float(np.mean(noise))
    sd = float(np.std(noise))
    if sd > 1.0e-12:
        noise /= sd

    q = a * np.exp(float(strength) * 0.8 * noise)
    return normalize_nonneg(q)


def push_init(inits, cap, arr):
    """Append a normalized nonnegative sequence if cap not reached."""
    if len(inits) >= cap:
        return
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return
    arr = np.nan_to_num(arr, nan=0.0, posinf=1.0e6, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1.0e-12:
        return
    inits.append(arr / s)


def generate_inits(n, rng, parent, cap):
    """Generate diverse smooth/asymmetric initial shapes."""
    n = int(max(16, n))
    inits = []
    x = np.linspace(0.0, 1.0, n)

    push_init(inits, cap, np.ones(n, dtype=np.float64))

    if parent is not None:
        p = resample_length(parent, n)
        push_init(inits, cap, p)
        push_init(inits, cap, p[::-1])

    for r in (0.3, 0.6, 1.0, 1.6, 2.5, 4.0):
        push_init(inits, cap, np.exp(r * x) + 1.0e-4)
        push_init(inits, cap, np.exp(-r * x) + 1.0e-4)
        push_init(inits, cap, np.exp(-r * np.abs(x - 0.5)) + 0.03)

    for th in (0.12, 0.25, 0.50, 0.75, 0.88):
        for wd in (0.07, 0.12, 0.22, 0.35):
            push_init(inits, cap, 0.05 + np.exp(-((x - th) / wd) ** 2))

    for wd in (0.08, 0.15, 0.28):
        for cc in (0.18, 0.28, 0.40, 0.50):
            push_init(
                inits,
                cap,
                0.05
                + np.exp(-((x - cc) / wd) ** 2)
                + 0.5 * np.exp(-((x - (1.0 - cc)) / wd) ** 2),
            )

    for f in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0):
        for amp in (0.2, 0.45, 0.70, 0.95, 1.15):
            for ph in (0.0, 0.7, 1.4, 2.2, 3.1):
                push_init(inits, cap, 1.0 + amp * np.cos(f * np.pi * x + ph))
                if len(inits) < cap:
                    push_init(inits, cap, 1.0 + amp * np.sin(f * np.pi * x + ph))

    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        pwr = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.2, 3.5)))
        pwr[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(pwr))
        spec = pwr * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= float(np.mean(noise))
        amp = float(rng.uniform(0.3, 1.2))
        push_init(inits, cap, 1.0 + amp * noise)

    for _ in range(4):
        push_init(inits, cap, 0.05 + 0.95 * rng.random(n))

    return inits


def clean_output(a):
    """Convert best internal sequence into a valid output list."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return [0.5] * 64
    a = a / s
    return [float(v) for v in a]

def prepare_candidate(arr, n, Ns):
    """Resample and score a candidate at target length n."""
    arr = resample_length(arr, n)
    arr = normalize_nonneg(arr)
    sc = score_normalized_pre(arr, n, Ns)
    return (sc, arr)


def optimize_sequence(
    a,
    deadline,
    rng=None,
    Ns=None,
    Ng=None,
    verbose=False,
    tag="",
    use_restarts=True,
    max_restart=4,
    stagn_limit=240,
):
    """
    Main exponentiated-gradient active-set optimizer.

    Returns (best_score, best_array) with sum normalized to 1.
    """
    a = normalize_nonneg(a)
    n = int(a.size)
    if n == 0:
        return float("inf"), np.zeros(0, dtype=np.float64)

    Ns, Ng = fft_ns_for_len(n)
    if rng is None:
        rng = np.random.default_rng(0)

    two_n = 2.0 * float(n)
    best = a.copy()
    c = fast_self_conv(a, Ns)
    cm = float(np.max(c)) if c.size else 0.0
    best_sc = float(two_n * max(0.0, cm))

    eta = 0.25
    ci = 0
    iter_cnt = 0
    stagn = 0
    restarts = 0
    last_print = best_sc

    if verbose:
        print("  [%s] start n=%d score=%.9f" % (tag, n, best_sc))

    while True:
        now = time.monotonic()
        if now > deadline:
            break

        iter_cnt += 1

        c = fast_self_conv(a, Ns)
        cm = float(np.max(c)) if c.size else 0.0
        sc = float(two_n * max(0.0, cm))
        if not np.isfinite(sc):
            break

        if sc < best_sc - 1.0e-13:
            best_sc = sc
            best = a.copy()
            if verbose and best_sc < last_print - 1.0e-6:
                print("    [%s] it=%d score=%.9f eta=%.4f" % (tag, iter_cnt, best_sc, eta))
                last_print = best_sc

        dense_w = None
        idx = None
        wvals = None

        if iter_cnt % 8 == 0:
            beta = float(LSE_BETAS[(iter_cnt // 8) % len(LSE_BETAS)])
            w_soft = soft_weights(c, beta)
            idx_soft = np.nonzero(w_soft > 1.0e-10)[0]
            if idx_soft.size <= DIRECT_LIMIT and idx_soft.size > 0:
                idx = idx_soft.astype(np.int64, copy=False)
                wvals = w_soft[idx]
                s = float(wvals.sum())
                if s > 0.0:
                    wvals /= s
                else:
                    dense_w = w_soft
            else:
                dense_w = w_soft
        else:
            ctype, K, eps, alpha = ACTIVE_CONFIGS[ci % len(ACTIVE_CONFIGS)]
            maxK = min(int(c.size), max(DIRECT_LIMIT, 2 * int(K)))
            idx, wvals = make_active(c, ctype, K, eps, alpha, maxK)

        try:
            if dense_w is not None:
                h = grad_dense(a, dense_w, Ng)
            else:
                if idx is None or wvals is None or idx.size == 0:
                    continue
                sw = float(wvals.sum())
                if sw <= 0.0 or not np.isfinite(sw):
                    continue
                wvals = wvals / sw
                if idx.size <= DIRECT_LIMIT:
                    h = grad_sparse_direct(a, idx, wvals)
                else:
                    wd = np.zeros(int(c.size), dtype=np.float64)
                    wd[idx] = wvals
                    h = grad_dense(a, wd, Ng)

            h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
            hmax = float(np.max(h))
            if hmax <= 1.0e-14:
                if use_restarts and iter_cnt > 20:
                    a = perturb_smooth(a, rng, 0.12)
                    stagn = 0
                else:
                    break
                continue

            hn = h / hmax
            loga = np.log(a + 1.0e-300)
            loga -= float(np.max(loga))

            trial_sc = sc
            trial_a = a
            trial_eta = eta

            if stagn <= 5:
                factors = (1.0, 1.6, 0.6, 2.4, 0.3)
            else:
                factors = (1.0, 0.6, 0.3, 1.5, 0.25)

            for f in factors:
                if time.monotonic() > deadline - 3.0:
                    break
                e = float(eta) * float(f)
                if e < 0.01 or e > 5.0:
                    continue
                v = loga - e * hn
                v -= float(np.max(v))
                v = np.clip(v, -700.0, 700.0)
                na = np.exp(v)
                na = normalize_nonneg(na)
                cn = fast_self_conv(na, Ns)
                cmax_n = float(np.max(cn)) if cn.size else 0.0
                ns = float(two_n * max(0.0, cmax_n))
                if np.isfinite(ns) and ns < trial_sc:
                    trial_sc = ns
                    trial_a = na
                    trial_eta = e

            if trial_sc < sc - 1.0e-13:
                a = trial_a.copy()
                eta = max(0.02, min(3.0, trial_eta))
                if trial_sc < best_sc - 1.0e-13:
                    best_sc = trial_sc
                    best = a.copy()
                    if verbose and best_sc < last_print - 1.0e-7:
                        print(
                            "    [%s] it=%d score=%.9f eta=%.4f"
                            % (tag, iter_cnt, best_sc, eta)
                        )
                        last_print = best_sc
                stagn = 0
            else:
                stagn += 1
                eta = max(0.02, float(eta) * 0.88)
                if stagn % 5 == 0:
                    ci += 1

                if (
                    use_restarts
                    and restarts < max_restart
                    and stagn in (25, 50, 90, 140, 200)
                ):
                    restarts += 1
                    strength = 0.05 + 0.03 * float(restarts) + float(rng.uniform(0.0, 0.03))
                    a = perturb_smooth(a, rng, strength)
                    eta = 0.30
                    ci = 0
                    stagn = 0

                if stagn > stagn_limit:
                    break

        except Exception:
            break
            break

    if verbose:
        print("  [%s] done n=%d best_score=%.9f" % (tag, n, best_sc))

    return float(best_sc), best


def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and gradient for raw nonnegative x.

    Objective:
        F(x) = 2 n * max_k conv(x,x)_k / (sum x)^2

    Smoothed by LSE over the scaled convolution values.
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(x.size)
    if n == 0:
        return 1.0e100, np.zeros(0, dtype=np.float64)

    a = np.nan_to_num(x, nan=0.0, posinf=100.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1.0e-10:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    Ns, Ng = fft_ns_for_len(n)
    C = fast_self_conv(a, Ns)
    P = C / (S * S)
    pmax = float(np.max(P)) if P.size else 0.0
    if not np.isfinite(pmax) or pmax <= 0.0:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    z = beta * (P - pmax)
    z = np.nan_to_num(z, nan=-700.0, posinf=0.0, neginf=-700.0)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    se = float(e.sum())
    if not np.isfinite(se) or se <= 0.0:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    w = e / se
    L = pmax + float(np.log(se)) / beta
    reg = 1.0e-10
    val = 2.0 * float(n) * L + reg * (S - 1.0) ** 2

    g2 = grad_dense(a, w, Ng)
    Cw = float(np.dot(w, P))
    grad = 4.0 * float(n) * (
        g2 / (S * S) - (Cw / (S * S * S)) * a
    )
    grad += 2.0 * reg * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1.0e8, 1.0e8)

    return float(val), grad


def lse_fun_grad(x, beta):
    """Wrapper for scipy: returns (value, gradient)."""
    return lse_obj_grad(x, beta)


def run_lbfgs(a, deadline, verbose=False):
    """LSE-smoothed L-BFGS-B refinements."""
    if minimize is None:
        return internal_score_seq(a), a

    a = normalize_nonneg(a)
    n = int(a.size)
    Ns, _ = fft_ns_for_len(n)
    best_sc = score_normalized_pre(a, n, Ns)
    best = a.copy()

    start = time.monotonic()
    bounds = [(0.0, 100.0)] * n
    betas = (200.0, 800.0, 3000.0, 12000.0, 50000.0)

    for beta in betas:
        rem = deadline - time.monotonic()
        if rem < 60.0:
            break
        maxfun = int(min(350, max(40, rem / 3.0)))
        try:
            res = minimize(
                lse_fun_grad,
                best,
                method="L-BFGS-B",
                args=(float(beta),),
                bounds=bounds,
                jac=True,
                options={
                    "ftol": 1.0e-15,
                    "gtol": 1.0e-9,
                    "maxiter": max(20, maxfun // 2),
                    "maxfun": maxfun,
                    "maxls": 30,
                },
            )
            a2 = normalize_nonneg(res.x if res is not None and res.x is not None else best)
            sc = score_normalized_pre(a2, n, Ns)
            if np.isfinite(sc) and sc < best_sc - 1.0e-12:
                best_sc = float(sc)
                best = a2.copy()
                if verbose:
                    print("  [lbfgs] beta=%g score=%.9f" % (beta, best_sc))
        except Exception:
            pass

    return float(best_sc), best


def run_survey(n, deadline, parent, rng):
    """Quick low-resolution survey to choose a promising shape."""
    n = int(max(512, n))
    inits = generate_inits(n, rng, parent, 64)
    Ns, Ng = fft_ns_for_len(n)
    two_n = 2.0 * float(n)

    scored = []
    for arr in inits:
        arr = normalize_nonneg(arr)
        c = fast_self_conv(arr, Ns)
        sc = float(two_n * max(0.0, float(np.max(c)))) if c.size else float("inf")
        if np.isfinite(sc):
            scored.append((sc, arr))

    if not scored:
        arr = normalize_nonneg(np.ones(n))
        c = fast_self_conv(arr, Ns)
        sc = float(two_n * max(0.0, float(np.max(c))))
        return sc, arr, []

    scored.sort(key=itemgetter(0))

    best_sc = scored[0][0]
    best_arr = scored[0][1].copy()
    extras = []

    start = time.monotonic()
    top_count = min(8, len(scored))
    if start >= deadline - 10:
        return float(best_sc), best_arr, extras

    per = max(12.0, (deadline - start) / float(top_count))

    for i in range(top_count):
        now = time.monotonic()
        if now >= deadline - 5:
            break
        sc0, arr = scored[i]
        chunk_dead = min(deadline - 2.0, now + 0.75 * per)
        try:
            sc, a = optimize_sequence(
                arr,
                chunk_dead,
                rng,
                Ns,
                Ng,
                verbose=False,
                tag="survey%d" % i,
                use_restarts=False,
                stagn_limit=220,
            )
        except Exception:
            sc, a = sc0, arr

        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_arr = a.copy()

        extras.append(a.copy())
        if len(extras) > 6:
            extras.pop(0)

    return float(best_sc), best_arr, extras


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Entry point. Searches within budget_s seconds and returns a list of
    non-negative heights minimizing the evaluator score.
    """
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 30.0

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    parent = load_parent()

    # Initial best: parent if available, otherwise uniform.
    if parent is not None:
        best_arr = normalize_nonneg(parent)
        best_sc = internal_score_seq(best_arr)
    else:
        best_arr = normalize_nonneg(np.ones(1024, dtype=np.float64))
        best_sc = internal_score_seq(best_arr)

    if np.isfinite(best_sc):
        print("Initial best score=%.9f len=%d" % (best_sc, len(best_arr)))

    # Time plan.
    main_dead = t_start + min(0.62 * total, deadline - 350.0)
    if main_dead < t_start + 60.0:
        main_dead = min(deadline, t_start + 60.0)

    survey_dead = min(main_dead - 60.0, t_start + min(240.0, 0.10 * total))
    if survey_dead < t_start + 20.0:
        survey_dead = max(t_start + 10.0, main_dead - 20.0)

    survey_arr = None
    survey_extras = []

    if total >= 900 and time.monotonic() < survey_dead - 20:
        n_survey = 4096
        try:
            ssc, sarr, survey_extras = run_survey(n_survey, survey_dead, parent, rng)
            if sarr is not None and sarr.size > 0:
                ssc = internal_score_seq(sarr)
                if np.isfinite(ssc) and ssc < best_sc:
                    best_sc = float(ssc)
                    best_arr = sarr.copy()
                    print("Survey best score=%.9f len=%d" % (best_sc, len(best_arr)))
                survey_arr = sarr
        except Exception:
            pass

    # High-resolution main search.
    n_main = 16384
    Ns_main, Ng_main = fft_ns_for_len(n_main)

    cands = []
    if parent is not None:
        cands.append(prepare_candidate(parent, n_main, Ns_main))
        cands.append(prepare_candidate(parent[::-1], n_main, Ns_main))

    if survey_arr is not None and survey_arr.size > 0:
        cands.append(prepare_candidate(survey_arr, n_main, Ns_main))
        cands.append(prepare_candidate(survey_arr[::-1], n_main, Ns_main))

    for ex in survey_extras[:3]:
        if ex is not None and ex.size > 0:
            cands.append(prepare_candidate(ex, n_main, Ns_main))

    if len(cands) < 5:
        try:
            hi_inits = generate_inits(n_main, rng, parent if parent is not None else np.ones(n_main), 24)
            hs = []
            for arr in hi_inits:
                hs.append(prepare_candidate(arr, n_main, Ns_main))
            hs.sort(key=itemgetter(0))
            for item in hs[:5]:
                cands.append(item)
        except Exception:
            pass

    if not cands:
        cands.append(prepare_candidate(np.ones(n_main, dtype=np.float64), n_main, Ns_main))

    cands.sort(key=itemgetter(0))

    main_start = time.monotonic()
    main_dur = max(0.0, main_dead - main_start)
    first_dead = main_start + 0.55 * main_dur
    second_dead = main_start + 0.85 * main_dur
    third_dead = main_dead

    done_cands = 0
    for ci in range(min(3, len(cands))):
        if ci == 0:
            dead = first_dead
        elif ci == 1:
            dead = second_dead
        else:
            dead = third_dead

        now = time.monotonic()
        if now >= dead - 10.0:
            break

        if ci > 0 and best_sc < TARGET - 1.0e-5:
            break

        arr = cands[ci][1]
        try:
            sc, a = optimize_sequence(
                arr,
                dead,
                rng,
                Ns_main,
                Ng_main,
                verbose=True,
                tag="main%d" % ci,
                use_restarts=(ci == 0),
                max_restart=4,
                stagn_limit=300,
            )
        except Exception:
            sc, a = float("inf"), arr

        done_cands = ci + 1

        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_arr = a.copy()
            print("After main%d score=%.9f len=%d" % (ci, best_sc, len(best_arr)))

    # LSE / L-BFGS-B polish.
    lse_dead = min(deadline - 250.0, max(main_dead, time.monotonic()) + min(300.0, 0.12 * total))
    if lse_dead > time.monotonic() + 80.0:
        try:
            sc, a = run_lbfgs(best_arr, lse_dead, verbose=True)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = float(sc)
                best_arr = a.copy()
                print("After LSE score=%.9f len=%d" % (best_sc, len(best_arr)))
        except Exception:
            pass

    # Restart phase: perturbations and untried candidates.
    restart_dead = min(
        deadline - 110.0,
        max(lse_dead, time.monotonic()) + min(300.0, 0.12 * total),
    )

    if restart_dead > time.monotonic() + 60.0:
        attempts = []
        for i in range(3):
            attempts.append(("perturb", i))
        for j in range(done_cands, min(len(cands), done_cands + 3)):
            attempts.append(("cand", j))

        num = len(attempts)
        for idx, (kind, val) in enumerate(attempts):
            now = time.monotonic()
            if now >= restart_dead - 50.0:
                break
            rem = (restart_dead - 50.0 - now) / max(1, num - idx)
            dead = min(restart_dead - 50.0, now + max(25.0, rem))

            if kind == "perturb":
                strength = 0.04 + 0.05 * float(val) + float(rng.uniform(0.0, 0.08))
                p = perturb_smooth(best_arr, rng, strength)
            else:
                p = cands[val][1]

            p = normalize_nonneg(p)
            Ns_r, Ng_r = fft_ns_for_len(len(p))
            try:
                sc, a = optimize_sequence(
                    p,
                    dead,
                    rng,
                    Ns_r,
                    Ng_r,
                    verbose=False,
                    tag="restart%d" % idx,
                    use_restarts=False,
                    stagn_limit=260,
                )
            except Exception:
                sc, a = float("inf"), p

            if np.isfinite(sc) and sc < best_sc:
                best_sc = float(sc)
                best_arr = a.copy()
                print("Restart%d score=%.9f len=%d" % (idx, best_sc, len(best_arr)))

    # Final active-set polish.
    final_dead = deadline - 5.0
    if final_dead > time.monotonic() + 40.0:
        n_final = int(best_arr.size)
        Ns_f, Ng_f = fft_ns_for_len(n_final)
        try:
            sc, a = optimize_sequence(
                best_arr,
                final_dead,
                rng,
                Ns_f,
                Ng_f,
                verbose=True,
                tag="final",
                use_restarts=False,
                max_restart=0,
                stagn_limit=500,
            )
        except Exception:
            sc, a = internal_score_seq(best_arr), best_arr

        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_arr = a.copy()

    # Final cleaning and safe output.
    out = clean_output(best_arr)
    final_score = safe_evaluate(out)

    if not np.isfinite(final_score) and parent is not None:
        pout = clean_output(normalize_nonneg(parent))
        pscore = safe_evaluate(pout)
        if np.isfinite(pscore):
            out = pout
            final_score = pscore

    if not np.isfinite(final_score):
        out = clean_output(normalize_nonneg(np.ones(1024, dtype=np.float64)))
        final_score = safe_evaluate(out)

    print(
        "DONE internal=%.9f verified=%.9f len=%d time=%.1fs"
        % (best_sc, final_score, len(out), time.monotonic() - t_start)
    )
    return out
