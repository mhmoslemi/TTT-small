"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-resolution optimization,
adaptive LSE smoothing, geometric-weighted peak-targeted subgradient,
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)

def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S

def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)

def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

def _subgrad_weighted(a, gamma=10.0):
    """Subgradient with geometric weighting on peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    # Geometric weights: peaks close to max get more weight
    z = gamma * (d - dmax)
    z = np.clip(z, -50, 0)
    w = np.exp(z)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.zeros_like(d)
        w[np.argmax(d)] = 1.0
    else:
        w = w / wsum
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

def _nesterov_opt(a0, n_iters=3000, deadline=None, gamma=10.0):
    """Nesterov with geometrically-weighted subgradient."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    if deadline is not None:
        t0 = time.monotonic()
    score0, g0 = _subgrad_weighted(a, gamma)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.05 / g_norm
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_weighted(y_norm, gamma)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.05 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 100:
            eta *= 0.6
            no_improve = 0
    return best_score, best_a

def _lbfgs_lse(a0, beta, maxfun=300):
    a = _normalize(a0)
    n = a.size
    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad
    bounds = [(0.0, None)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": max(50, maxfun // 2), "maxfun": maxfun, "maxls": 40}
    try:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- Multi-resolution optimization (small -> large)
- Adaptive LSE smoothing with progressive beta
- Geometric-weighted peak-targeted Nesterov subgradient
- Structured perturbation restarts
- Coordinate block polishing

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    beta = float(beta)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_weighted(a, gamma=15.0):
    """Subgradient with geometric weighting emphasizing top peaks."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    z = gamma * (d - dmax)
    z = np.clip(z, -60, 0)
    w = np.exp(z)
    wsum = float(w.sum())
    if wsum < 1e-15:
        w = np.zeros_like(d)
        w[int(np.argmax(d))] = 1.0
    else:
        w = w / wsum
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _nesterov_opt(a0, n_iters=3000, deadline=None, gamma=15.0, eta_scale=0.06):
    """Nesterov accelerated subgradient with geometric peak weighting."""
    a = _normalize(a0).copy()
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t = 1.0
    y = a.copy()
    v = np.zeros(n)
    score0, g0 = _subgrad_weighted(a, gamma)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = eta_scale / g_norm
    no_improve = 0
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 2.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_weighted(y_norm, gamma)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.03 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 150:
            eta *= 0.65
            no_improve = 0
    return best_score, best_a


def _lbfgs_lse(a0, beta, maxfun=300):
    """L-BFGS-B on LSE-smoothed objective."""
    a = _normalize(a0).copy()
    n = a.size
    bounds = [(0.0, None)] * n

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    opts = {"ftol": 1e-15, "gtol": 1e-11, "maxiter": max(50, maxfun // 2),
            "maxfun": int(maxfun), "maxls": 50}
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = _normalize(res.x) if getattr(res, "x", None) is not None else a
    except Exception:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out

def _perturb_structured(a, rng, strength=0.3):
    """Structured perturbation targeting peak structure."""
    a = _normalize(a).copy()
    n = a.size
    c = _fast_self_conv(a)
    d = n * c / (a.sum() ** 2)
    dmax = float(d.max())
    mode = int(rng.integers(0, 5))
    if mode == 0:
        # Multiplicative Gaussian noise
        q = a * np.exp(strength * 0.4 * rng.standard_normal(n))
    elif mode == 1:
        # Frequency-domain perturbation
        Nf = _next_pow2(n)
        A = np.fft.rfft(a, Nf)
        mask = np.ones(len(A))
        mask[0] = 0
        noise = (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))) * strength * 0.1
        A = A * (1 + noise * mask)
        q = np.fft.irfft(A, Nf, s=n)
    elif mode == 2:
        # Peak-targeted: push mass away from peak-contributing positions
        active = np.where(d >= 0.95 * dmax)[0]
        if len(active) > 0:
            # For each active peak at lag k, positions i and k-i contribute
            # Reduce a at positions that contribute to active peaks
            reduction = np.zeros(n)
            for k in active[:10]:
                for i in range(n):
                    j = int(k) - i
                    if 0 <= j < n:
                        reduction[i] += a[i] * a[j]
            reduction /= max(1, reduction.max())
            q = a * (1.0 - strength * 0.5 * reduction)
    elif mode == 3:
        # Blend with a different structured profile
        x = np.linspace(0, 1, n)
        f = int(rng.integers(1, 8))
        phase = rng.uniform(0, 2 * np.pi)
        profile = 1.0 + 0.5 * np.cos(2 * np.pi * f * x + phase)
        profile = np.clip(profile, 0.01, None)
        t = float(rng.uniform(0.1, 0.3))
        q = (1 - t) * a + t * _normalize(profile)
    else:
        # Low-frequency smooth noise
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n * 0.1 + 1.0) ** 2.0)
        power[0] = 0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * strength * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf, s=n)
        q = a * np.exp(0.3 * noise)
    return _normalize(np.clip(q, 0, None))


def _multi_resolution_opt(parent, target_len, deadline):
    """Optimize progressively from small to target length."""
    t0 = time.monotonic()
    total_budget = max(60.0, float(deadline) - t0 - 20.0)

    # Build resolution ladder
    ladder = [512, 1024, 2048, 4096, 8192, 16384]
    ladder = [L for L in ladder if L <= target_len * 2]
    if target_len in ladder:
        ladder.remove(target_len)
    ladder.append(target_len)
    ladder = [L for L in ladder if L >= 512]
    ladder.sort()

    print("Resolution ladder: {}".format(ladder))

    # Allocate time: more time to larger sizes
    weights = []
    for i, L in enumerate(ladder):
        w = float(L) ** 0.5 / len(ladder)
        weights.append(w)
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    best_a = _resample(parent, ladder[0]) if parent is not None else None
    if best_a is None:
        best_a = np.ones(ladder[0]) / ladder[0]

    cum_t = 0.0
    for i, L in enumerate(ladder):
        sub_dead = t0 + (cum_t + weights[i]) * total_budget
        sub_dead = min(sub_dead, float(deadline) - 5.0 - (len(ladder) - 1 - i) * 10.0)
        cum_t += weights[i]

        if time.monotonic() > sub_dead - 10.0:
            continue

        # Upsample current best to this length
        a = _resample(best_a, L)
        print("Level n={} (rem={:.0f}s)".format(L, sub_dead - time.monotonic()))

        sc, a_opt = _optimize_at_level(a, sub_dead, "n={}".format(L))
        if np.isfinite(sc):
            best_a = a_opt
            print("Level n={}: score={:.8f}".format(L, sc))

    return best_a

                if time.monotonic() > fin_dead - 10.0:
                    break
                sc, a2 = _nesterov_opt(best_a, n_iters=n_it, deadline=fin_dead - 5.0,
                                       gamma=gamma, eta_scale=0.02)
                if sc < best_score:
                    best_score = sc
                    best_a = a2
                    print("Final Nest gamma={:.0f}: {:.8f}".format(gamma, best_score))
                best_a = _normalize(a2)

    # Output
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
