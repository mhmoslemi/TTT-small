"""
Coordinate-descent + GS + LSE-LBFGS hybrid for minimizing self-convolution peak.
Key innovation: exact 1D sub-solves in coordinate descent directly minimize the
score with respect to each element, using analytical critical points and
golden-section refinement. Combined with GS initialization and LSE polish.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_flat(a, n_iters=400):
    """GS with constant-modulus frequency target."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    target_mag = np.sqrt(1.0 / M) * np.ones(M)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        alpha = 0.25 * (1.0 - it / max(1, n_iters)) + 0.03
        A = np.fft.fft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        new_mag = A_mag ** (1.0 - alpha) * target_mag ** alpha
        A_new = new_mag * np.exp(1j * A_phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

    best_a = a.copy()
    best_sc = compute_score(a)

    target_mag = np.sqrt(1.0 / M) * np.ones(M)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        # Target: reduce peaks, preserve structure
        b_target = np.zeros(M)
        ratio = c[:2 * n - 1] / (c_max + 1e-30)
        b_target[:2 * n - 1] = 1.0 - 0.5 * ratio + 0.5 * (c_mean / c_max)
        b_target = np.maximum(b_target, 0.01)

        T = np.fft.fft(b_target)
        T_mag = np.maximum(np.abs(T), 1e-20)

        alpha = 0.2 * (1.0 - it / n_iters) + 0.03
        base_mag = np.abs(np.fft.fft(a, M))
        base_phase = np.angle(np.fft.fft(a, M))

        new_mag = base_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.3)
        A_new = new_mag * np.exp(1j * base_phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=3000, maxfun=20000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def coordinate_descent_1d(a, n_cycles=500, n_top=80, recompute_every=500, seed=42):
    """
    Coordinate descent with exact 1D sub-solves.
    For each element i, solve min_{t>=0} 2n*max_k b_k(t)/(S+t)^2.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    S = 1.0

    b = fast_conv(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    rng = np.random.default_rng(seed)
    k_range = np.arange(2 * n - 1, dtype=np.int64)
    total_updates = 0

    for cycle in range(n_cycles):
        order = rng.permutation(n)

        for idx_in_order in range(n):
            i = int(order[idx_in_order])
            t_old = a[i]
            S_fix = S - t_old
            if S_fix < 1e-10:
                continue

            # Partner array: partner[k] = a[k-i] if 0<=k-i<n, else 0
            diff = k_range - i
            valid = (diff >= 0) & (diff < n)
            partner = np.zeros(2 * n - 1)
            partner[valid] = a[diff[valid]]

            # Determine diagonal index
            diag_idx = 2 * i
            has_diag = diag_idx < 2 * n - 1

            # Find top peaks at current state
            top_k = np.argsort(b)[::-1][:n_top]

            # 1D objective: f(t) = 2n * max_k b_k(t) / (S_fix + t)^2
            # b_k(t) = b_k + 2*(t-t_old)*partner[k] for k != diag
            # b_diag(t) = b_diag + t^2 - t_old^2

            def f_at(t):
                # Compute modified top peaks
                indices_eval = top_k
                b_mod = b[indices_eval] + 2.0 * (t - t_old) * partner[indices_eval]
                if has_diag:
                    pos = np.where(indices_eval == diag_idx)[0]
                    if len(pos) > 0:
                        b_mod[pos[0]] = b[diag_idx] + t * t - t_old * t_old
                max_top = float(np.max(b_mod))

                # Safety: check if non-top peaks could exceed
                # Max possible increase for any peak: 2*|t-t_old|*max_partner + |t^2-t_old^2|
                max_incr = 2.0 * abs(t - t_old) * float(partner.max())
                if has_diag:
                    max_incr = max(max_incr, abs(t * t - t_old * t_old))
                b_min_non_top = float(b.min()) - max_incr
                if b_min_non_top > max_top:
                    pass  # impossible
                elif -max_incr > max_top - float(b[top_k[-1]]):
                    # Non-top peak might exceed; do full check
                    b_all = b + 2.0 * (t - t_old) * partner
                    if has_diag:
                        b_all[diag_idx] = b[diag_idx] + t * t - t_old * t_old
                    max_top = float(np.max(b_all))

                S_t = S_fix + t
                if S_t <= 1e-10:
                    return 1e10
                return 2.0 * n * max_top / (S_t * S_t)

            # Find optimal t using critical points + golden section
            # Critical points from top-5 peaks
            t_candidates = set()
            t_candidates.add(t_old)
            t_candidates.add(0.0)
            t_candidates.add(t_old * 0.5)
            t_candidates.add(t_old * 1.5)
            t_candidates.add(S_fix * 0.01)

            for k in top_k[:8]:
                k_int = int(k)
                if has_diag and k_int == diag_idx:
                    if S_fix > 1e-10:
                        t_star = b[diag_idx] / S_fix
                        if 0 < t_star < 1000:
                            t_candidates.add(t_star)
                            t_candidates.add(t_star * 0.5)
                            t_candidates.add(t_star * 1.5)
                else:
                    pk = partner[k_int]
                    if pk > 1e-15:
                        # f_k(t) = (b_k + 2*(t-t_old)*pk) / (S_fix+t)^2
                        # = (b_k - 2*t_old*pk + 2*pk*t) / (S_fix+t)^2
                        # Let A = b_k - 2*t_old*pk, B = 2*pk
                        # f_k(t) = (A + B*t) / (S_fix+t)^2
                        # df/dt = (B*(S_fix+t) - 2*(A+B*t)) / (S_fix+t)^3
                        # = (B*S_fix - B*t - 2*A) / (S_fix+t)^3
                        # Critical: t* = S_fix - 2*A/B = S_fix - (b_k - 2*t_old*pk)/pk
                        A_k = b[k_int] - 2.0 * t_old * pk
                        B_k = 2.0 * pk
                        t_star = S_fix - A_k / pk
                        if 0 < t_star < 1000:
                            t_candidates.add(t_star)
                            t_candidates.add(max(0, t_star * 0.7))
                            t_candidates.add(min(1000, t_star * 1.3))

            t_candidates = [max(0.0, min(1000.0, t)) for t in t_candidates if S_fix + t > 1e-10]

            best_t = t_old
            best_f = f_at(t_old)
            for t_c in t_candidates:
                f_c = f_at(t_c)
                if f_c < best_f - 1e-14:
                    best_f = f_c
                    best_t = t_c

            # Golden section refinement around best_t
            if best_t > 0:
                lo = max(0.0, best_t * 0.8)
                hi = min(1000.0, best_t * 1.2)
                if hi - lo > 1e-6:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    a_g = lo + (1 - gr) * (hi - lo)
                    b_g = lo + gr * (hi - lo)
                    f_a = f_at(a_g)
                    f_b = f_at(b_g)
                    for _ in range(12):
                        if f_a < f_b:
                            hi = b_g
                            b_g = a_g
                            f_b = f_a
                            a_g = lo + (1 - gr) * (hi - lo)
                            f_a = f_at(a_g)
                        else:
                            lo = a_g
                            a_g = b_g
                            f_a = f_b
                            b_g = lo + gr * (hi - lo)
                            f_b = f_at(b_g)
                    t_mid = (lo + hi) / 2.0
                    f_mid = f_at(t_mid)
                    if f_mid < best_f:
                        best_f = f_mid
                        best_t = t_mid

            # Apply update
            if abs(best_t - t_old) > 1e-10:
                delta = best_t - t_old
                b += 2.0 * delta * partner
                if has_diag:
                    # Fix diagonal: we added 2*delta*partner[diag] but need t^2 - t_old^2
                    b[diag_idx] -= 2.0 * delta * partner[diag_idx]
                    b[diag_idx] += best_t * best_t - t_old * t_old
                a[i] = best_t
                S += delta
                total_updates += 1

        # Periodic recompute to avoid drift
        if total_updates % recompute_every < n:
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 1e-10:
                break
            a = a / S
            S = 1.0
            b = fast_conv(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_shave(a, n_rounds=50, n_peaks=20):
    """Directly reduce top peaks by scaling contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner_seg = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner_seg
            # Scale down top contributors
            n_red = min(5, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_red]
            for li in top_local:
                j = j_start + li
                a[j] *= 0.9

        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-10:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def random_perturb(a0, rng, strength=0.03):
    a = a0.copy()
    a += rng.standard_normal(len(a)) * strength
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return a0.copy()
    return a / S


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)

    # Fourier-based inits
    for _ in range(3):
        freqs = np.arange(1, 10)
        coefs = rng.standard_normal(len(freqs))
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
        sig = np.zeros(n)
        for i, f in enumerate(freqs):
            sig += coefs[i] * np.cos(2 * np.pi * f * x + phases[i])
        inits.append(np.maximum(0.5 + 0.25 * sig, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max_conv/(sum)^2.
    Uses GS init, coordinate descent with 1D sub-solves, LSE-LBFGS, peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
    else:
        n = 6962

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"n={n}, initial score={best_score:.8f}")

    # Phase 1: GS warm-up on diverse inits (10% time)
    phase1_end = t_start + 0.08 * (budget_s - 30)
    print(f"\n=== Phase 1: GS warmup ===")

    inits = build_inits(n, rng, parent if len(parent) == n else None)
    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 1e-12:
            a0 = a0 / s
        a1 = gs_flat(a0, n_iters=300)
        sc = compute_score(a1)
        update_best(sc, a1)
        if time.monotonic() < phase1_end:
            a2 = gs_adaptive(a1, n_iters=200)
            sc2 = compute_score(a2)
            update_best(sc2, a2)

    print(f"After Phase 1: {best_score:.8f}")

    # Phase 2: Coordinate descent (main workhorse, 50% time)
    phase2_end = t_start + 0.58 * (budget_s - 30)
    print(f"\n=== Phase 2: Coordinate Descent ===")

    # Run multiple CD runs from different starting points
    cd_starts = [best_a.copy()]
    # Add perturbed versions
    for _ in range(2):
        cd_starts.append(random_perturb(best_a, rng, 0.02))
    # Add GS results if different
    a_gs = gs_flat(best_a, n_iters=200)
    cd_starts.append(a_gs)

    for ci, a_cd in enumerate(cd_starts):
        if time.monotonic() > phase2_end:
            break
        t_cd = time.monotonic()
        n_cyc = 300 + ci * 100
        a_out = coordinate_descent_1d(a_cd, n_cycles=n_cyc, n_top=80, seed=seed + ci)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  CD run {ci}: {sc:.8f} ({time.monotonic()-t_cd:.1f}s)")

    # Continue CD from best
    if time.monotonic() < phase2_end:
        t_cd = time.monotonic()
        a_out = coordinate_descent_1d(best_a, n_cycles=400, n_top=100, seed=seed + 100)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  CD extended: {sc:.8f} ({time.monotonic()-t_cd:.1f}s)")

    print(f"After Phase 2: {best_score:.8f}")

    # Phase 3: LSE-LBFGS refinement (20% time)
    phase3_end = t_start + 0.78 * (budget_s - 30)
    print(f"\n=== Phase 3: LSE-LBFGS ===")

    for beta in [5000, 15000, 50000, 150000]:
        if time.monotonic() > phase3_end:
            break
        t_lb = time.monotonic()
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,), maxiter=4000, maxfun=25000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t_lb:.1f}s)")

    # Peak shaving
    if time.monotonic() < phase3_end:
        a_ps = peak_shave(best_a, n_rounds=30, n_peaks=20)
        sc = compute_score(a_ps)
        update_best(sc, a_ps)

    # CD on LSE-refined
    if time.monotonic() < phase3_end:
        t_cd = time.monotonic()
        a_out = coordinate_descent_1d(best_a, n_cycles=200, n_top=100, seed=seed + 200)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  CD after LBFGS: {sc:.8f} ({time.monotonic()-t_cd:.1f}s)")

    print(f"After Phase 3: {best_score:.8f}")

    # Phase 4: Multi-restart CD + final polish (remaining time)
    print(f"\n=== Phase 4: Multi-restart + polish ===")

    restart_count = 0
    while time.monotonic() < deadline - 60:
        # Perturb and run CD
        a_pert = random_perturb(best_a, rng, strength=0.01 + 0.01 * restart_count)
        t_rs = time.monotonic()

        # Quick GS to diversify
        a_pert = gs_flat(a_pert, n_iters=100)

        # CD
        remaining = deadline - 60 - time.monotonic()
        n_cyc = min(200, max(50, int(remaining * 20)))
        a_out = coordinate_descent_1d(a_pert, n_cycles=n_cyc, n_top=80, seed=seed + 500 + restart_count)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  Restart {restart_count}: {sc:.8f} ({time.monotonic()-t_rs:.1f}s)")

        restart_count += 1

        # Also try LSE from best
        if time.monotonic() < deadline - 60 and restart_count % 2 == 0:
            a_opt = run_lbfgs(best_a, lse_score_grad, (100000,), maxiter=3000, maxfun=15000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

    # Final CD polish
    if time.monotonic() < deadline - 10:
        t_fin = time.monotonic()
        a_out = coordinate_descent_1d(best_a, n_cycles=100, n_top=100, seed=seed + 999)
        sc = compute_score(a_out)
        update_best(sc, a_out)
        print(f"  Final CD: {sc:.8f} ({time.monotonic()-t_fin:.1f}s)")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
