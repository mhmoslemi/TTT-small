def run_lbfgs(a0, beta, maxiter=5000, gtol=1e-13):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_objective_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        args=(beta,),
        options={'ftol': 1e-18, 'gtol': gtol, 'maxiter': maxiter, 'maxfun': maxiter*10, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_descent(a, n_steps=500, rng=None):

        rng = np.random.default_rng(42)
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_steps):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        M_val = float(c[k_star])
        
        # Subgradient of peak: direction of increase of c[k_star]
        direction = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            direction[j_start:j_end + 1] = a[j_end:j_start - 1 : -1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (2.0 * direction / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        base_step = 0.01 / (gn + 1e-10)
