```
def target_shape(c, gamma, cap_frac):
```

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over nonnegative sequences.

Strategy:
- Start from the provided parent sequence if available.
- Use convolution-aware spectral projections (direct A^2 / target-convolution
  shaping) to flatten the linear self-convolution peak.
- Use mirror-descent on the simplex with LSE/top-k activation weights.
- Use small-scale LP iterations from the current peak bound:
  for fixed f, maximize sum(g) subject to f*g <= M, then mix.
- Try larger lengths, random smooth restarts, and a final L-BFGS polish.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog


def _nfft(m):
    return int(np.fft.next_fast_len(int(max(1, m))))


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    M = 2 * n - 1
    N = _nfft(M)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:M]
    return np.maximum(c, 0.0)


def conv_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = fast_conv(a)
    S = float(a.sum())
    if S <= 1e-300:
        return c, np.inf
    return c, float(2.0 * n * float(c.max()) / (S * S))


def compute_score(a):
    _, s = conv_score(a)
    return s


def safe_normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15 or not np.isfinite(s):
        return np.full(a.shape, 1.0 / max(1, a.size))
    return a / s


def make_hermitian(X, N):
    N = int(N)
    X = np.asarray(X, dtype=np.complex128)
    if N == 1:
        return X.copy()
    Y = np.zeros(N, dtype=np.complex128)
    Y[0] = X[0].real
    half = N // 2
    if N % 2 == 0:
        Y[half] = X[half].real
        idx = np.arange(1, half)
    else:
        idx = np.arange(1, half + 1)
    if idx.size > 0:
        y = 0.5 * (X[idx] + np.conj(X[N - idx]))
        Y[idx] = y
        Y[N - idx] = np.conj(y)
    return Y


def conv_project_shape(a, shape, full_spectrum=False):
    a = np.asarray(a, dtype=np.float64).ravel()
    shape = np.asarray(shape, dtype=np.float64).ravel()
    n = a.size
    M = 2 * n - 1
    N = _nfft(M)

    A = np.fft.fft(a.astype(np.complex128), N)
    C = A * A
    St = np.fft.fft(shape.astype(np.complex128), N)

    if full_spectrum:
        denom = np.abs(C)
        safe = np.where(denom > 1e-300, C, (1e-300 + 0j))
        with np.errstate(divide="ignore", invalid="ignore"):
            ratio = np.sqrt(St / safe)
        A_new = A * ratio
    else:
        denom = np.abs(C) + 1e-300
        ratio = np.sqrt(np.abs(St) / denom)
        A_new = A * ratio

    A_new = make_hermitian(A_new, N)
    a_new = np.fft.ifft(A_new)[:n].real
    return safe_normalize(a_new)


def target_shape(c, gamma, cap_frac):
    c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    s = float(c.sum())
    if s <= 1e-30:
        return c.copy()
    maxc = float(c.max())
    if maxc <= 1e-30:
        return c / s

    x = np.clip(c / maxc, 0.0, 1.0)

    if gamma is not None and gamma < 1.0:
        gamma = max(0.05, float(gamma))
        x = np.power(x, gamma)

    t = maxc * x
    if cap_frac is not None and cap_frac < 1.0:
        cap = maxc * max(0.0, float(cap_frac))
        t = np.minimum(t, cap)

    ts = float(t.sum())
    if ts <= 1e-30 or not np.isfinite(ts):
        t = c
        ts = s
    return t / ts


def spectral_flatten(
    a,
    end_time,
    label="SPEC",
    max_iters_param=5000,
    mix=0.65,
    full_from=0.45,
    gamma_start=0.95,
    gamma_end=0.55,
    cap_start=1.0,
    cap_end=0.88,
    use_full_spectrum=True,
):
    a = safe_normalize(a)
    best = a.copy()
    _, best_sc = conv_score(a)

    now = time.monotonic()
    if now >= end_time:
        return best

    limit = max_iters_param if max_iters_param and max_iters_param > 0 else 10**9
    usable = max(0.0, end_time - now)
    est = int(min(max(20, usable * 35), limit))
    est = max(est, 20)

    for it in range(est):
        now = time.monotonic()
        if now >= end_time:
            break

        progress = it / float(est)
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        cap = cap_start + (cap_end - cap_start) * progress

        c, sc = conv_score(a)
        if not np.isfinite(sc):
            a = best.copy()
            continue

        if sc > best_sc * 2.0 + 1e-12:
            a = best.copy()
            c = fast_conv(a)
            continue

        if sc < best_sc:
            best_sc = sc
            best = a.copy()

        shape = target_shape(c, gamma, cap)
        full = use_full_spectrum and (progress >= full_from)
        a_new = conv_project_shape(a, shape, full)

        if mix >= 1.0:
            a = a_new
        else:
            a = safe_normalize((1.0 - mix) * a + mix * a_new)

    return best


def lse_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(c.max())
    beta = float(beta)
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(np.minimum(beta * (c - cmax), 0.0))
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(c.shape, 1.0 / max(1, c.size))
    return e / s


def topk_weights(c, top_k):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    k = int(min(max(1, int(top_k)), L))
    if k >= L:
        return np.full(L, 1.0 / L)
    idx = np.argpartition(c, L - k)[L - k:]
    p = np.zeros(L, dtype=np.float64)
    p[idx] = 1.0 / k
    return p


def weighted_gradient(a, p):
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    Lp = p.size
    N = _nfft(Lp + n - 1)
    PA = np.fft.rfft(p, N)
    RA = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(PA * RA, N)[: Lp + n - 1]
    return h[n - 1 : 2 * n - 1]


def mirror_anneal(
    a,
    end_time,
    label="MIRROR",
    max_iters_param=20000,
    use_lse=True,
    use_topk=True,
    betas=(20, 80, 300, 1000, 3000, 10000, 30000, 100000),
    topk_list=(8, 16, 32, 64, 128),
):
    a = safe_normalize(a)
    best = a.copy()
    _, best_sc = conv_score(a)

    now = time.monotonic()
    if now >= end_time:
        return best

    limit = max_iters_param if max_iters_param and max_iters_param > 0 else 10**9
    usable = max(0.0, end_time - now)
    est = int(min(max(100, usable * 30), limit))
    est = max(est, 50)

    last_improve = 0

    if use_lse:
        lse_budget = int(0.70 * est)
        betas = list(betas)
        it_total = 0
        for bi, beta in enumerate(betas):
            if time.monotonic() >= end_time:
                break
            rem_betas = max(1, len(betas) - bi)
            iters = max(30, lse_budget // rem_betas)
            for it in range(iters):
                if time.monotonic() >= end_time or it_total >= lse_budget:
                    break
                it_total += 1

                c, sc = conv_score(a)
                if not np.isfinite(sc):
                    a = best.copy()
                    break

                if sc > best_sc * 2.0 + 1e-12:
                    a = best.copy()
                    continue

                if sc < best_sc:
                    best_sc = sc
                    best = a.copy()
                    last_improve = it_total

                p = lse_weights(c, beta)
                g = weighted_gradient(a, p)
                gm = float(g.mean())
                denom = float(np.max(np.abs(g - gm))) if g.size else 1.0
                denom = max(denom, 1e-300)

                local_progress = it / float(max(1, iters))
                eta = 0.9 * (1.0 - 0.6 * local_progress) + 0.1
                z = -(g - gm) / denom
                zmin = float(z.min())
                a = np.maximum(a * np.exp(eta * (z - zmin)), 0.0)
                a = safe_normalize(a)

            if it_total - last_improve > 220:
                a = best.copy()
                last_improve = 0

    if use_topk:
        remain_time = max(0.0, end_time - time.monotonic())
        rem_iter_budget = int(min(0.30 * est, max(50, remain_time * 30)))
        topks = list(topk_list)
        for tk in topks:
            if time.monotonic() >= end_time:
                break
            iters = max(20, rem_iter_budget // max(1, len(topks)))
            for _ in range(iters):
                if time.monotonic() >= end_time:
                    break

                c, sc = conv_score(a)
                if not np.isfinite(sc):
                    a = best.copy()
                    break

                if sc > best_sc * 2.0 + 1e-12:
                    a = best.copy()
                    continue

                if sc < best_sc:
                    best_sc = sc
                    best = a.copy()

                p = topk_weights(c, tk)
                g = weighted_gradient(a, p)
                gm = float(g.mean())
                denom = float(np.max(np.abs(g - gm))) if g.size else 1.0
                denom = max(denom, 1e-300)

                eta = 0.55
                z = -(g - gm) / denom
                zmin = float(z.min())
                a = np.maximum(a * np.exp(eta * (z - zmin)), 0.0)
                a = safe_normalize(a)

    return best


def build_conv_matrix(F):
    m = F.size
    C = np.empty((2 * m - 1, m), dtype=np.float64)
    for k in range(2 * m - 1):
        i0 = 0 if k < m else k - m + 1
        i1 = k if k < m else m - 1
        if i1 >= i0:
            C[k, i0 : i1 + 1] = F[k - i1 : k - i0 + 1][::-1]
    return C


def lp_improve_small(f, end_time, max_iters=10, label="LP"):
    f = safe_normalize(f)
    m = f.size
    best = f.copy()
    _, best_sc = conv_score(f)

    stagnant = 0

    for it in range(max_iters):
        now = time.monotonic()
        if now >= end_time:
            break

        allowed = max(1, int(end_time - now - 10))
        if allowed < 1:
            break
        if m > 768:
            break

        scale = np.sqrt(2.0 * m)
        F = f * scale
        cF = fast_conv(F)
        M = float(cF.max())
        M_use = M * (1.0 + 1e-7) + 1e-12

        C = build_conv_matrix(F)
        b = np.full(2 * m - 1, M_use, dtype=np.float64)
        cvec = -np.ones(m, dtype=np.float64)
        bounds = [(0.0, None)] * m

        res = None
        try:
            opts = {"time_limit": float(allowed), "presolve": True}
            res = linprog(cvec, A_ub=C, b_ub=b, bounds=bounds, method="highs", options=opts)
        except Exception:
            try:
                res = linprog(cvec, A_ub=C, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
            except Exception:
                try:
                    res = linprog(cvec, A_ub=C, b_ub=b, bounds=bounds)
                except Exception:
                    res = None

        if res is None or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        if g.size != m or not np.all(np.isfinite(g)):
            break
        g = np.clip(g, 0.0, None)
        Sg = float(g.sum())
        if Sg <= 1e-9 or not np.isfinite(Sg):
            break

        g = g / Sg

        best_local = best
        best_local_sc = best_sc
        for t in (0.15, 0.35, 0.60, 0.85, 1.0):
            mix = safe_normalize((1.0 - t) * f + t * g)
            sc = compute_score(mix)
            if np.isfinite(sc) and sc < best_local_sc:
                best_local_sc = sc
                best_local = mix

        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best = best_local.copy()
            f = best_local.copy()
            stagnant = 0
            print(f"  [{label}] LP iter {it}: improved local {best_sc:.10f}")
        else:
            stagnant += 1
            if Sg > scale * (1.0 + 1e-4):
                f = safe_normalize(0.5 * f + 0.5 * g)
            else:
                break
            if stagnant >= 3:
                break

    return best


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = a.size
    new_n = int(new_n)
    if new_n <= 0:
        return a.copy()[:1]
    if old_n == 0:
        return np.ones(new_n) / new_n
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, a[0])
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, a)
    return safe_normalize(out)


def downsample_sequence(a, m):
    return interpolate_sequence(a, m)


def lowfreq_perturb(a, rng, alpha=0.05):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    x = np.arange(n, dtype=np.float64) / n
    pert = np.zeros(n, dtype=np.float64)
    for m in range(1, 7):
        freq = int(rng.integers(1, 12))
        phase = 2.0 * np.pi * float(rng.uniform())
        pert += (alpha / float(m)) * np.cos(2.0 * np.pi * freq * x + phase)
    out = a * (1.0 + pert)
    return safe_normalize(out)


def lowpass_smooth(a, band=20.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1:
        return a.copy()
    F = np.fft.rfft(a)
    freqs = np.arange(len(F), dtype=np.float64)
    cutoff = max(1.0, n / max(1.0, float(band)))
    F[1:] *= np.exp(-((freqs[1:] / cutoff) ** 2))
    out = np.fft.irfft(F, n)[:n]
    return safe_normalize(out)


def gauss_lowpass_perturb(a, rng, scale=0.02):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1:
        return a.copy()
    a2 = np.maximum(a + scale * rng.standard_normal(n), 0.0)
    F = np.fft.rfft(a2)
    freqs = np.fft.rfftfreq(n) * n
    band = max(1.0, n / 20.0)
    if F.size > 1:
        F[1:] *= np.exp(-((freqs[1:] / band) ** 2))
    a2 = np.fft.irfft(F, n)[:n]
    return safe_normalize(a2)


def candidate_from_family(n, rng):
    x = np.linspace(0.0, 1.0, n)
    typ = int(rng.integers(0, 6))
    if typ == 0:
        a = np.ones(n, dtype=np.float64)
    elif typ == 1:
        a = x + 0.01
    elif typ == 2:
        a = 1.0 - x + 0.01
    elif typ == 3:
        a = np.sqrt(np.maximum(x * (1.0 - x), 1e-6)) + 0.01
    elif typ == 4:
        a = 1.0 / (np.sqrt(x + 1e-3) + np.sqrt(1.0 - x + 1e-3) + 1e-3)
    else:
        a = np.ones(n, dtype=np.float64)
        for m in range(1, 8):
            freq = int(rng.integers(1, 20))
            phase = 2.0 * np.pi * float(rng.uniform())
            a += (0.5 / float(m)) * np.cos(2.0 * np.pi * freq * x + phase)
    return safe_normalize(np.maximum(a, 0.0))


def restart_candidate(base, rng, end_time, label="RESTART"):
    base = safe_normalize(base)
    now = time.monotonic()
    if now >= end_time:
        return base

    n = base.size
    r = int(rng.integers(0, 5))
    if r == 0:
        a = lowfreq_perturb(base, rng, alpha=0.02 + 0.02 * float(rng.uniform()))
    elif r == 1:
        a = gauss_lowpass_perturb(base, rng, scale=0.01 + 0.02 * float(rng.uniform()))
    elif r == 2:
        s = int(rng.integers(1, max(2, n)))
        a = np.roll(base, s)
    elif r == 3:
        a = lowpass_smooth(base, band=8.0 + 12.0 * float(rng.uniform()))
    else:
        a = candidate_from_family(n, rng)

    a = safe_normalize(a)
    sub_budget = max(0.0, end_time - now)
    spec_end = now + 0.45 * sub_budget

    a = spectral_flatten(
        a,
        spec_end,
        label=label,
        max_iters_param=600,
        mix=0.6,
        full_from=0.4,
        gamma_start=0.90,
        gamma_end=0.60,
        cap_start=1.00,
        cap_end=0.92,
    )
    a = mirror_anneal(
        a,
        end_time,
        label=label,
        max_iters_param=700,
        use_lse=True,
        use_topk=True,
        betas=(50, 500, 5000),
        topk_list=(16, 32),
    )
    return a


def lse_score_and_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = fast_conv(a)
    S = float(a.sum())
    if S <= 1e-12:
        return 1e100, np.zeros(n, dtype=np.float64)

    cmax = float(c.max())
    beta = float(beta)
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(np.minimum(beta * (c - cmax), 0.0))
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        p = np.full(c.shape, 1.0 / max(1, c.size))
        L = cmax
    else:
        p = e / s
        L = cmax + np.log(s) / beta

    g = weighted_gradient(a, p)
    grad_L = 2.0 * g
    S2 = S * S
    S3 = S2 * S

    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=1000):
    a0 = safe_normalize(a0)
    n = a0.size
    try:
        bounds = [(0.0, 1000.0)] * n
        maxfun = max(200, int(maxiter * 3))
        res = minimize(
            lse_score_and_grad,
            a0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1e-15,
                "gtol": 1e-10,
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 50,
                "disp": False,
            },
        )
        a = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1000.0)
        a = safe_normalize(a)
        sc = compute_score(a)
        if not np.isfinite(sc):
            return compute_score(a0), a0
        return sc, a
    except Exception:
        return compute_score(a0), a0


def polish_lbfgs(a, end_time, label="LSE"):
    a = safe_normalize(a)
    best = a.copy()
    _, best_sc = conv_score(a)

    betas = (2000, 10000, 40000)
    for beta in betas:
        now = time.monotonic()
        if now >= end_time:
            break
        rem = max(0.0, end_time - now)
        if rem < 40:
            break
        maxiter = int(min(800, max(150, rem * 2.0)))
        sc, a2 = run_lbfgs(a, beta, maxiter)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a2.copy()
            print(f"  [{label}] beta={beta}: {best_sc:.10f}")
        a = safe_normalize(a2)

    return best


def log_update(best_sc, best_a, sc, cand, label):
    if np.isfinite(sc) and sc < best_sc:
        best_sc = float(sc)
        best_a = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0).copy()
        print(f"  [{label}] NEW BEST {best_sc:.10f}  n={len(best_a)}")
        return best_sc, best_a
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    hard_deadline = t0 + max(60.0, float(budget_s) - 30.0)

    parent = None
    try:
        if "height_sequence_1" in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 16 and np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is None or parent.size < 64:
        n0 = 8192
        best_a = np.ones(n0, dtype=np.float64) / float(n0)
        best_sc = compute_score(best_a)
        print(f"No useful parent. Starting with n={n0}, score={best_sc:.10f}")
        for r in range(8):
            cand = candidate_from_family(n0, rng)
            sc = compute_score(cand)
            best_sc, best_a = log_update(best_sc, best_a, sc, cand, "初始 family")
    else:
        if parent.size > 16384:
            parent = interpolate_sequence(parent, 12000)
        best_a = safe_normalize(parent)
        best_sc = compute_score(best_a)
        if not np.isfinite(best_sc):
            best_a = np.ones(8192, dtype=np.float64) / 8192.0
            best_sc = compute_score(best_a)
        print(f"Parent loaded: n={len(best_a)}, score={best_sc:.10f}")

    # Phase 1: direct convolution-aware spectral projection.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    end = now + 0.07 * rem
    cand = spectral_flatten(
        best_a,
        end,
        label="SPEC1",
        max_iters_param=5000,
        mix=0.60,
        full_from=0.30,
        gamma_start=0.95,
        gamma_end=0.60,
        cap_start=1.00,
        cap_end=0.92,
    )
    best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "SPEC1")

    # Phase 2: mirror annealing on full current length.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    end = now + 0.18 * rem
    cand = mirror_anneal(
        best_a,
        end,
        label="MIRROR1",
        max_iters_param=30000,
        betas=(20, 80, 300, 1000, 3000, 10000, 30000, 100000),
        topk_list=(8, 16, 32, 64, 128),
    )
    best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "MIRROR1")

    # Phase 3: small LP fixed-point iterations, then upsample.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    end = now + 0.23 * rem

    current_n = len(best_a)
    small_specs = []
    if current_n > 128:
        small_specs.append(128)
    if current_n > 256:
        small_specs.append(256)
    if current_n > 512:
        small_specs.append(512)

    if small_specs and now < end:
        shares = []
        for m in small_specs:
            if m <= 128:
                shares.append(1.0)
            elif m <= 256:
                shares.append(1.6)
            elif m <= 512:
                shares.append(2.6)
            else:
                shares.append(3.4)
        tot = sum(shares)
        iters_by_size = {128: 18, 192: 14, 256: 11, 384: 8, 512: 6, 768: 4}

        for idx, m in enumerate(small_specs):
            now = time.monotonic()
            if now >= end:
                break
            share_time = (end - now) * (shares[idx] / max(1e-12, sum(shares[idx:])))
            sub_end = min(end, now + max(10.0, share_time))
            if sub_end <= now:
                sub_end = end

            small = downsample_sequence(best_a, m)
            max_iters = iters_by_size.get(m, 5)
            small_best = lp_improve_small(small, sub_end, max_iters=max_iters, label=f"LP-{m}")

            full_small = interpolate_sequence(small_best, current_n)
            sc = compute_score(full_small)
            best_sc, best_a = log_update(best_sc, best_a, sc, full_small, f"LP-{m}->full")

            now = time.monotonic()
            if now < end:
                quick_end = min(end, now + max(4.0, 0.22 * (sub_end - now)))
                if quick_end > now:
                    full_small = spectral_flatten(
                        full_small,
                        quick_end,
                        label=f"SPEC-LP-{m}",
                        max_iters_param=350,
                        mix=0.55,
                        full_from=0.35,
                        gamma_start=0.90,
                        gamma_end=0.65,
                        cap_start=1.00,
                        cap_end=0.92,
                    )
                    sc = compute_score(full_small)
                    best_sc, best_a = log_update(best_sc, best_a, sc, full_small, f"SPEC-LP-{m}")

    # Phase 4: larger-length exploration.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    end = now + 0.12 * rem

    if now < end:
        base_len = len(best_a)
        targets = []
        for fac in (1.20, 1.40, 1.60, 1.80):
            nl = int(base_len * fac)
            if nl > base_len + 50 and nl <= 16384:
                if all(abs(nl - t) >= 150 for t in targets):
                    targets.append(nl)
        targets = targets[:3]

        for idx, nl in enumerate(targets):
            now = time.monotonic()
            if now >= end:
                break
            share_time = (end - now) / max(1, len(targets) - idx)
            phase_end = min(end, now + share_time)
            if phase_end <= now:
                phase_end = end

            cand = interpolate_sequence(best_a, nl)
            mid = now + 0.45 * max(0.0, phase_end - now)
            cand = spectral_flatten(
                cand,
                mid,
                label=f"LARGE-{nl}",
                max_iters_param=1500,
                mix=0.60,
                full_from=0.35,
                gamma_start=0.90,
                gamma_end=0.60,
                cap_start=1.00,
                cap_end=0.90,
            )
            cand = mirror_anneal(
                cand,
                phase_end,
                label=f"LMIR-{nl}",
                max_iters_param=5000,
                betas=(50, 200, 1000, 5000, 20000),
                topk_list=(16, 32, 64),
            )
            best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, f"LARGE-{nl}")

    # Phase 5: random smooth restarts from current best.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    end = now + 0.20 * rem

    restart_idx = 0
    while now < end and restart_idx < 12:
        if end - now < 35.0:
            break
        alloc = min(65.0, max(30.0, 0.22 * (end - now)))
        restart_end = min(end, now + alloc)
        cand = restart_candidate(best_a, rng, restart_end, label=f"R{restart_idx}")
        best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, f"R{restart_idx}")
        restart_idx += 1
        now = time.monotonic()

    # Phase 6: final polish.
    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)

    endf = now + 0.35 * rem
    cand = spectral_flatten(
        best_a,
        endf,
        label="FINAL-SPEC",
        max_iters_param=5000,
        mix=0.50,
        full_from=0.25,
        gamma_start=0.90,
        gamma_end=0.55,
        cap_start=0.98,
        cap_end=0.90,
    )
    best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "FINAL-SPEC")

    now = time.monotonic()
    rem = max(1.0, hard_deadline - now)
    endm = now + 0.25 * rem
    cand = mirror_anneal(
        best_a,
        endm,
        label="FINAL-MIRROR",
        max_iters_param=10000,
        betas=(1000, 5000, 20000, 100000),
        topk_list=(16, 32, 64),
    )
    best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "FINAL-MIRROR")

    now = time.monotonic()
    if hard_deadline - now > 110.0:
        endl = hard_deadline - 10.0
        cand = polish_lbfgs(best_a, endl, label="FINAL-LSE")
        best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "FINAL-LSE")

    now = time.monotonic()
    if hard_deadline - now > 20.0:
        endl = hard_deadline - 3.0
        cand = spectral_flatten(
            best_a,
            endl,
            label="LAST-SPEC",
            max_iters_param=800,
            mix=0.40,
            full_from=0.20,
            gamma_start=0.85,
            gamma_end=0.60,
            cap_start=0.99,
            cap_end=0.92,
        )
        best_sc, best_a = log_update(best_sc, best_a, compute_score(cand), cand, "LAST-SPEC")

    # Sanitize and return.
    out_a = np.clip(np.asarray(best_a, dtype=np.float64).ravel(), 0.0, 1000.0)
    out_a = safe_normalize(out_a)
    if out_a.size > 20000:
        out_a = interpolate_sequence(out_a, 16384)

    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if sum(out) < 0.01:
        out = [0.5]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    elapsed = time.monotonic() - t0
    print("=" * 70)
    print(f"FINAL internal={best_sc:.10f}, verified={verified:.10f}, n={len(out)}, time={elapsed:.1f}s")
    print("=" * 70)

    return out
```
