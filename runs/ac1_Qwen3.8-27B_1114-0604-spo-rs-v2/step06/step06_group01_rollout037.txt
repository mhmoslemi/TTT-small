```python
def self_conv(a):

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def compute_score_fast(a, conv=None):
    """Compute score, optionally using precomputed convolution."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    if conv is None:
        conv = self_conv(a)
    return 2.0 * n * float(conv.max()) / (S * S)
```

```python
"""
Multi-phase optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations: direct coordinate descent on the non-smooth objective (exploiting
incremental convolution updates), Barzilai-Borwein subgradient method with active-set
tracking, and progressive LP alternating via FISTA. Combines spectral initialization,
targeted coordinate search on peak-contributing variables, and adaptive step sizes.
"""

import numpy as np
from scipy.ndimage import uniform_filter1d
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def fista_lp(a, M, n_iters, beta, x_init=None):
    """
    Solve: min (1/beta)*sum exp(beta*((a*x)[k] - M)) s.t. x >= 0
    using FISTA. Returns x (unnormalized).
    """
    n = len(a)
    if x_init is not None:
        x = np.maximum(x_init.astype(np.float64), 0.0).copy()
    else:
        x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    L = beta * 2.0 * n * sum_a2 + 1e-12
    step = 1.0 / L

    n_c = 2 * n - 1
    M_fft = 1 << (n + n_c - 2).bit_length()
    FA = np.fft.rfft(a, M_fft)

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(n_c) / n_c
        else:
            p = exp_z / exp_sum

        FP = np.fft.rfft(p, M_fft)
        corr = np.fft.irfft(np.conj(FA) * FP, M_fft)[:n]
        grad = corr - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer, fista_iters, time_limit, beta, line_search=True):
    """LP-based alternating with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    x_warm = None

    for outer in range(max_outer):
        if time.monotonic() > time_limit:
            break

        c = self_conv(a)
        M = float(c.max())
        x = fista_lp(a, M, n_iters=fista_iters, beta=beta, x_init=x_warm)
        x_warm = x.copy()
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        if line_search:
            t_cands = [0.1, 0.25, 0.4, 0.5, 0.6, 0.8, 1.0]
            best_t = 0.3
            best_t_sc = best_sc
            for t in t_cands:
                a_try = np.maximum((1.0 - t) * a + t * x, 0.0)
                St = float(a_try.sum())
                if St < 1e-12:
                    continue
                a_try = a_try / St
                sc_try = compute_score(a_try)
                if sc_try < best_t_sc:
                    best_t_sc = sc_try
                    best_t = t
            a_new = np.maximum((1.0 - best_t) * a + best_t * x, 0.0)
        else:
            a_new = x

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                break
        a = a_new

    return best_a, best_sc, x_warm


def coord_descent(a, n_passes, n_candidates, time_limit, verbose=True):
    """
    Direct coordinate descent on the non-smooth objective.
    Exploits incremental convolution updates for efficiency.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    conv = self_conv(a)
    best_sc = 2.0 * n * float(conv.max()) / (S * S)

    rng = np.random.default_rng(42)

    for pass_i in range(n_passes):
        if time.monotonic() > time_limit:
            break

        k_star = int(np.argmax(conv))
        improved_count = 0

        # Determine active coordinates (those contributing to peak region)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(k_star, n - 1)
        # Add margin for more thorough search
        margin = max(5, n // (15 + pass_i * 3))
        i_lo = max(0, i_lo - margin)
        i_hi = min(n - 1, i_hi + margin)
        indices = np.arange(i_lo, i_hi + 1)

        # Random subset on later passes for diversity
        if pass_i > 2 and len(indices) > 100:
            sample_size = max(100, len(indices) // 2)
            indices = rng.choice(indices, size=sample_size, replace=False)
            indices.sort()

        start_time = time.monotonic()

        for j in indices:
            if time.monotonic() > time_limit:
                break

            # Compute influence: g[k] = a[k-j] + a[j-k] with zero padding
            g = np.zeros(2 * n - 1)
            lo1 = j
            hi1 = min(j + n, 2 * n - 1)
            if hi1 > lo1:
                g[lo1:hi1] += a[:hi1 - lo1]
            lo2 = max(0, j - n + 1)
            hi2 = j + 1
            if hi2 > lo2:
                g[lo2:hi2] += a[lo2:hi2][::-1]

            current_t = a[j]
            S_base = S - current_t
            if S_base < 1e-15 and current_t == 0:
                continue

            # Generate candidate values
            cands = [0.0]
            if current_t > 1e-15:
                cands.append(current_t)

            gk = g[k_star]
            if gk > 1e-15:
                beta_val = conv[k_star] - current_t * gk
                t_opt = (gk * S_base - 2.0 * beta_val) / gk
                if t_opt > 0 and t_opt < 1000:
                    cands.append(t_opt)
                    cands.append(max(0.0, t_opt * 0.5))
                    cands.append(t_opt * 1.5)
                    cands.append(max(0.0, current_t + 0.1 * (t_opt - current_t)))

            # Scale-based candidates
            if current_t > 1e-15:
                cands.append(current_t * 1.2)
                cands.append(current_t * 0.8)

            cands = [max(0.0, min(1000.0, c)) for c in cands]
            # Deduplicate
            cands = sorted(set(cands))
            if len(cands) > n_candidates + 2:
                # Keep most diverse set
                keep = [cands[0], cands[-1]]
                step_i = max(1, len(cands) // (n_candidates))
                for ci in range(0, len(cands), step_i):
                    if cands[ci] not in keep:
                        keep.append(cands[ci])
                cands = sorted(keep)[:n_candidates + 1]

            best_t = current_t
            best_sc_j = 2.0 * n * float(conv.max()) / (S * S)

            for t in cands:
                if abs(t - current_t) < 1e-14:
                    continue
                new_S = S_base + t
                if new_S < 1e-12:
                    continue
                delta = t - current_t
                new_conv = conv + delta * g
                new_max = float(new_conv.max())
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc_j - 1e-15:
                    best_sc_j = sc
                    best_t = t

            if best_t != current_t:
                delta = best_t - current_t
                a[j] = best_t
                conv += delta * g
                S = S_base + best_t
                if best_sc_j < best_sc - 1e-15:
                    best_sc = best_sc_j
                    improved_count += 1

        elapsed = time.monotonic() - start_time
        if verbose:
            print(f"    CD pass {pass_i}: score={best_sc:.8f}, "
                  f"improved={improved_count} coords, time={elapsed:.1f}s")
        if improved_count == 0 and pass_i > 2:
            break

    return a, conv, best_sc


def subgradient_bb(a, n_iters, time_limit, verbose=True):
    """
    Subgradient descent with Barzilai-Borwein step size.
    Uses average subgradient over active set (near-max lags).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    S = 1.0

    conv = self_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max()) / (S * S)

    x_prev = a.copy()
    g_prev = None
    M_prev = float(conv.max())

    for it in range(n_iters):
        if time.monotonic() > time_limit:
            break

        conv = self_conv(a)
        M_val = float(conv.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find active set: lags where conv[k] > M - epsilon
        eps = 0.001 * M_val
        active = np.where(conv > M - eps)[0]

        # Average subgradient over active set
        # For lag k, subgrad[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # with special case j=k: 2*a[0]
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                val = 0.0
                if 0 <= k - j < n:
                    val += a[k - j]
                if 0 <= j - k < n and j != k:
                    val += a[j - k]
                elif j == k:
                    val = 2.0 * a[0]
                grad[j] += val

        grad /= len(active)

        # Full gradient of score = 2n * (max_conv / S^2 - 2*M*S / S^3) for each direction
        # Actually, the score gradient is:
        # d(score)/da[j] = 2n * (grad[j] / S^2 - 2*M / S^3)
        S2 = S * S
        S3 = S2 * S
        grad_score = 2.0 * n * (grad / S2 - 2.0 * M_val / S3)

        # BB step size
        if g_prev is not None:
            s = a - x_prev
            y = grad_score - g_prev
            ys = float(np.dot(y, s))
            yy = float(np.dot(y, y))
            if yy > 1e-20:
                if ys > 1e-15:
                    alpha_bb = ys / yy
                else:
                    alpha_bb = 1.0 / (yy + 1e-10)
                alpha_bb = min(alpha_bb, 1.0)
                alpha_bb = max(alpha_bb, 1e-8)
            else:
                alpha_bb = 1.0 / (float(np.linalg.norm(grad_score)) + 1e-10)
        else:
            alpha_bb = 1.0 / (float(np.linalg.norm(grad_score)) + 1e-10)

        # Polyak-like step size as backup
        if best_sc > 1.0:
            sc_lower = 1.0
            alpha_polyak = (sc - sc_lower) / (float(np.linalg.norm(grad_score))**2 + 1e-20)
            alpha_polyak = min(alpha_polyak, 2.0)
            alpha = min(alpha_bb, alpha_polyak)
        else:
            alpha = alpha_bb

        # Update
        a_new = a - alpha * grad_score
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        grad_prev = grad_score
        x_prev = a
        a = a_new
        M_prev = M_val

        if verbose and it % 50 == 0:
            print(f"    SubGrad it {it}: score={sc:.8f}, alpha={alpha:.6f}, "
                  f"|active|={len(active)}")

    return best_a, best_sc


def freq_flatten(a, n_iters, alpha, time_limit):
    """Frequency-domain error reduction to flatten autocorrelation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() > time_limit:
            break

        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv.copy()
        conv_mod -= 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-18)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.2 + 0.8 * (it / max(1, n_iters // 2))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.0))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    noise = rng.standard_normal(n)
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 10.0))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x + 0.5))
    inits.append(np.clip(0.3 + 0.3 * x + 0.2 * np.cos(6.0 * np.pi * x), 0.01, 10.0))
    inits.append(0.5 + 0.15 * np.cos(2.0 * np.pi * 2 * x) + 0.1 * np.cos(2.0 * np.pi * 3 * x + 0.3))
    for init in inits:
        s = float(init.sum())
        if s > 1e-12:
            init = init / s
    return inits


def propose_candidate(budget_s=2400):
    """
    Search for the best non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: LP alternating, coordinate descent, BB subgradient, freq flatten.
    """
    rng = np.random.default_rng(777)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.astype(np.float64).copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent score: {ps:.8f}")
        report(ps, parent, "parent")

    # Choose main length
    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 6962

    # Try a couple of lengths
    lengths_to_try = [n_main]
    if n_main != 8192:
        lengths_to_try.append(8192)
    if n_main != 10240:
        lengths_to_try.append(10240)

    print(f"Main length: {n_main}, alternatives: {lengths_to_try}")

    # ============================================================
    # Phase 1: Quick LP alternating (20% time)
    # ============================================================
    p1_deadline = t_start + 0.20 * (float(budget_s) - 30.0)
    print(f"\n=== Phase 1: LP Alternating (t_limit={p1_deadline - t_start:.0f}s) ===")

    for n_try in lengths_to_try:
        if time.monotonic() > p1_deadline - 30:
            break
        inits = build_inits(n_try, rng, parent)
        max_init = max(2, int((p1_deadline - time.monotonic()) / 60.0))
        inits = inits[:max_init]

        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline - 20:
                break
            print(f"  FP1: n={n_try}, init={i}")
            a_out, sc_out, _ = lp_alternating(
                a0, max_outer=15, fista_iters=10000,
                time_limit=p1_deadline - 10, beta=200.0
            )
            sc = compute_score(a_out)
            report(sc, a_out, f"LP n={n_try} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Coordinate Descent (30% time) - KEY NEW METHOD
    # ============================================================
    p2_deadline = t_start + 0.50 * (float(budget_s) - 30.0)
    print(f"\n=== Phase 2: Coordinate Descent (t_limit={p2_deadline - t_start:.0f}s) ===")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Multiple CD passes with different settings
    cd_configs = [
        (20, 7, 1.0),   # (n_passes, n_candidates, focus_fraction)
        (15, 10, 0.8),
        (10, 12, 0.6),
    ]

    for ci, (npass, ncan, frac) in enumerate(cd_configs):
        if time.monotonic() > p2_deadline - 30:
            break
        print(f"  CD config {ci}: passes={npass}, candidates={ncan}")
        a_out, conv_out, sc_out = coord_descent(
            a_work, n_passes=npass, n_candidates=ncan,
            time_limit=p2_deadline - 10, verbose=True
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"CD {ci}")
        a_work = a_out.copy()

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: BB Subgradient + LP refinement (25% time)
    # ============================================================
    p3_deadline = t_start + 0.75 * (float(budget_s) - 30.0)
    print(f"\n=== Phase 3: BB Subgradient + LP (t_limit={p3_deadline - t_start:.0f}s) ===")

    # BB subgradient
    if time.monotonic() < p3_deadline - 60:
        print("  Running BB subgradient...")
        a_sg, sc_sg = subgradient_bb(best_a, n_iters=200,
                                     time_limit=p3_deadline - 120, verbose=True)
        report(sc_sg, a_sg, "subgrad BB")

    # LP refinement
    if time.monotonic() < p3_deadline - 30:
        print("  LP refinement...")
        a_lp_in = best_a.copy()
        S = float(a_lp_in.sum())
        if S > 0:
            a_lp_in = a_lp_in / S
        a_lp, sc_lp, _ = lp_alternating(
            a_lp_in, max_outer=10, fista_iters=15000,
            time_limit=p3_deadline - 20, beta=400.0
        )
        report(sc_lp, a_lp, "LP refine")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (25% time)
    # ============================================================
    p4_deadline = deadline
    print(f"\n=== Phase 4: Final Polish (t_limit={p4_deadline - t_start:.0f}s) ===")

    # Alternate between CD and short LP
    for round_i in range(4):
        if time.monotonic() > p4_deadline - 40:
            break

        # CD pass
        if time.monotonic() < p4_deadline - 80:
            a_out, _, sc_out = coord_descent(
                best_a, n_passes=8, n_candidates=8,
                time_limit=p4_deadline - 60, verbose=False
            )
            sc = compute_score(a_out)
            report(sc, a_out, f"polish CD r{round_i}")

        # Short LP
        if time.monotonic() < p4_deadline - 30:
            a_lp_in = best_a.copy()
            S = float(a_lp_in.sum())
            if S > 0:
                a_lp_in = a_lp_in / S
            a_lp, sc_lp, _ = lp_alternating(
                a_lp_in, max_outer=5, fista_iters=12000,
                time_limit=p4_deadline - 25, beta=600.0
            )
            report(sc_lp, a_lp, f"polish LP r{round_i}")

    # Final CD burst
    if time.monotonic() < p4_deadline - 15:
        a_out, _, sc_out = coord_descent(
            best_a, n_passes=10, n_candidates=10,
            time_limit=p4_deadline - 5, verbose=False
        )
        report(sc_out, a_out, "final CD")

    # ============================================================
    # Verify and return
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
```
