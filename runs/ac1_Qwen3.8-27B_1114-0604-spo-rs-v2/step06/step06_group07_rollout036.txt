"""
Optimized step function construction using adaptive beta LSE-smoothed L-BFGS
with multi-start spectral initialization and post-optimization peak shaving.
"""
import numpy as np
from scipy.optimize import minimize
import time

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    M_fft = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_obj_grad(x, beta):
    """
    LSE-smoothed objective and gradient.
    f(x) = 2n * LSE_beta(conv(x,x)) / sum(x)^2
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M).bit_length()
    
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to prevent overflow
    scaled = np.clip(scaled, -50, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Softmax
    p = exp_scaled / exp_sum
    
    # dLSE/dx
    L_corr = 1 << (M + n - 1).bit_length()
    P = np.fft.rfft(p, L_corr)
    X = np.fft.rfft(x, L_corr)
    corr = np.fft.irfft(np.conj(P) * X, L_corr)[:n]
    
    dLSE = 2.0 * corr / beta
    
    # Diagonal correction
    diag = 2 * np.arange(n)
    valid = diag < M
    dLSE[valid] -= x[valid] * p[diag[valid]] / beta
    
    # Full gradient
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dLSE * S - 2.0 * L) / S3
    
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-12):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
        a0 /= n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return float(compute_score(a_out)), a_out

def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.8):
    """Fienup iteration with geometric amplitude shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n)
        S = 1.0
    a = a / S
    
    M_fft = 1 << (2 * n).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        amp = np.sqrt(A.real**2 + A.imag**2)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def fienup_ellip(a, n_iters=300, beta_shape=3.0):

    n = len(a)

        a = np.ones(n)
        S = 1.0

        progress = it / max(1, n_iters)
        beta = beta_shape * (0.5 + progress)

        else:

        S_new = float(a_new.sum())
        if S_new < 1e-12:

        a = a_new

def peak_shave(a, n_iters=2000, n_top=20, shrink_start=0.02, shrink_end=0.001):
    """
    Direct peak shaving: iteratively reduce elements that contribute most
    to the top convolution peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2 * n).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        progress = it / max(1, n_iters)
        shrink = shrink_start + (shrink_end - shrink_start) * progress
        
        # Find top peaks
        nt = min(n_top, 2 * n - 1)
        top_idx = np.argpartition(c, -nt)[-nt:]
        
        # Compute contribution vector
        contrib = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            ks = k - js
            contrib[js] += a[ks]
            # For diagonal terms (j == k-j), we've already counted it once
            # But dc/dx should count it only once, while 2*a[k-j] counts it twice
            # So we need to subtract the diagonal
            diag_mask = js == ks
            if diag_mask.any():
                contrib[js[diag_mask]] -= a[js[diag_mask]]
        
        # Normalize
        c_sum = contrib.sum()
        if c_sum < 1e-30:
            break
        contrib = contrib / c_sum
        
        # Apply shrink
        a = a - shrink * contrib
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a

def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Edge-enhanced
    inits.append(np.maximum(0.1 + 0.8 * np.abs(2.0 * x - 1.0), 0.01))
    
    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    
    # Gaussian at center
    inits.append(np.exp(-5.0 * (2.0 * x - 1.0)**2) + 0.02)
    
    # Two peaks
    inits.append(np.maximum(0.2 + 0.6 * np.cos(4.0 * np.pi * (x - 0.25)) +
                            0.6 * np.cos(4.0 * np.pi * (x - 0.75)), 0.01))
    
    # Random smooth
    freqs = rng.integers(1, 15, size=6)
    phases = rng.uniform(0, 2 * np.pi, size=6)
    amps = rng.uniform(0.1, 0.5, size=6)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(sig, 0.01))
    
    # Anti-symmetric-ish (high on one half, low on other)
    half = n // 2
    anti = np.zeros(n)
    anti[:half] = 1.0
    anti[half:] = 0.1
    inits.append(anti)
    
    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
    
    return inits
