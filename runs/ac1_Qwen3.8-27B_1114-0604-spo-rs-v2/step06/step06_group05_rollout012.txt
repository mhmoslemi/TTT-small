```python
"""
Multi-resolution peak-suppression optimizer for non-negative self-convolution coefficients.

Method:
- Start from the supplied height_sequence_1 when available, otherwise build several
  smooth/asymmetric densities on [0,1] and optimize at increasing resolutions.
- Optimize the scale-invariant shape objective using:
  * large-beta soft-max / LSE approximations of the maximum convolution coefficient,
  * weighted sums over active peaks (hard top-k and soft top-gamma weights),
  * coordinate-free projected gradients with bounds a_i >= 0,
  * L-BFGS-B refinements, random/asymmetric perturbations, smoothing, symmetrization,
    and final length growth.
- Return the best non-negative sequence found within the time budget.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Globals used by the top-level scipy objective callback.
_T_START = 0.0
_LBFGS_MODE = 0
_LBFGS_BETA = 1.0
_LBFGS_LAM = 1.0e-3
_LBFGS_W = None
_LBFGS_STOP = 0.0


def _next_pow2(length):
    if length <= 1:
        return 1
    return 1 << (length - 1).bit_length()


def _fast_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    return c[:2 * n - 1]


def _cross_corr(p, a):
    """
    Returns y_i = sum_j a_j p_{i+j} for i=0..n-1, where p has length 2n-1.
    This is 0.5 * d(concat peaks)/da for linear self-convolution.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _next_pow2(3 * n - 2)
    ra = a[::-1]
    RA = np.fft.rfft(ra, L)
    PP = np.fft.rfft(p[:2 * n - 1], L)
    y = np.fft.irfft(RA * PP, L)
    return y[n - 1:2 * n - 1]


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a.copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1.0e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S <= 1.0e-12:
        return np.inf
    an = a / S
    c = _fast_conv(an)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = max(0.0, m)
    return float(2.0 * n * m)


def _shape_gradient_lse(a, beta, lam=0.0):
    """
    Gradient/value for F(a) = 2n * LSE_beta(conv(a/sum(a))) + lam*(sum(a)-1)^2.
    This optimizes the shape while keeping the scale mainly at sum=1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 1.0e100, np.empty(0, dtype=np.float64)

    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    S = float(a.sum())
    if S <= 1.0e-12:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    an = a / S
    c = _fast_conv(an)
    if c.size == 0:
        return 1.0e100, np.zeros(n, dtype=np.float64)
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm <= 1.0e-300:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    z = beta * (c - cm)
    zmax = float(np.max(z))
    ex = np.exp(z - zmax)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = int(np.argmax(c))
        ex = np.zeros_like(c, dtype=np.float64)
        ex[idx] = 1.0
        s = 1.0

    L = cm + (float(np.log(s)) / beta)
    p = ex / s

    conv_res = _cross_corr(p, an)
    h = 4.0 * float(n) * conv_res
    hdot = float(np.dot(an, h))
    grad = (h - hdot) / S

    f = 2.0 * float(n) * L
    if lam != 0.0:
        ds = S - 1.0
        grad = grad + 2.0 * lam * ds
        f = f + lam * ds * ds

    if not np.isfinite(f):
        return 1.0e100, np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(f), grad


def _shape_gradient_weighted(a, w, lam=0.0):
    """
    Gradient/value for F(a) = 2n * (w dot conv(a/sum(a))) + lam*(sum(a)-1)^2.
    w is expected to be normalized but is made robust here.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return 1.0e100, np.empty(0, dtype=np.float64)

    S = float(a.sum())
    if S <= 1.0e-12:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    an = a / S
    c = _fast_conv(an)
    if c.size == 0:
        return 1.0e100, np.zeros(n, dtype=np.float64)

    need = 2 * n - 1
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != need:
        tmp = np.zeros(need, dtype=np.float64)
        m = min(w.size, need)
        tmp[:m] = w[:m]
        w = tmp
    ws = float(w.sum())
    if ws <= 1.0e-300:
        w = np.zeros(need, dtype=np.float64)
        w[:] = 1.0 / need
    else:
        w = w / ws

    w_num = float(np.dot(w, c))
    conv_res = _cross_corr(w, an)
    h = 4.0 * float(n) * conv_res
    hdot = float(np.dot(an, h))
    grad = (h - hdot) / S

    f = 2.0 * float(n) * w_num
    if lam != 0.0:
        ds = S - 1.0
        grad = grad + 2.0 * lam * ds
        f = f + lam * ds * ds

    if not np.isfinite(f):
        return 1.0e100, np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(f), grad


def _lbfgs_eval(x):
    if _LBFGS_MODE == 1:
        return _shape_gradient_lse(x, _LBFGS_BETA, _LBFGS_LAM)
    return _shape_gradient_weighted(x, _LBFGS_W, _LBFGS_LAM)


def _lbfgs_func(x):
    try:
        f, g = _lbfgs_eval(x)
        if not np.isfinite(f):
            f = 1.0e100
        g = np.asarray(g, dtype=np.float64).ravel()
        if g.size != x.size:
            g = np.zeros(x.size, dtype=np.float64)
        if not np.all(np.isfinite(g)):
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        return float(f), g
    except Exception:
        return 1.0e100, np.zeros(len(np.asarray(x).ravel()), dtype=np.float64)


def _lbfgs_callback(*args):
    try:
        if time.monotonic() > _LBFGS_STOP:
            return True
    except Exception:
        pass
    return False


def _choose_maxfun(time_limit, eval_cost, cap):
    if eval_cost <= 1.0e-12 or time_limit <= 1.0:
        return cap
    mf = int(float(time_limit) * 0.70 / float(eval_cost))
    if mf < 120:
        return 120
    if mf > cap:
        return cap
    return mf


def _estimate_eval_cost(a):
    a = _normalize(a)
    if a.size == 0:
        return 0.005
    c = _fast_conv(a)
    cm = float(np.max(c)) if c.size else 1e-12
    beta = 50.0 / max(cm, 1e-300)
    t0 = time.monotonic()
    for _ in range(5):
        _shape_gradient_lse(a, beta, 0.0)
    dt = (time.monotonic() - t0) / 5.0
    if not np.isfinite(dt) or dt <= 0.0:
        dt = max(1.0e-4, 1.0e-7 * a.size)
    return float(dt) * 1.6


def _lbfgs_lse(a0, beta, deadline, eval_cost, time_limit, lam=1.0e-3, maxfun_cap=20000):
    a0 = _normalize(a0)
    n = a0.size
    if n == 0:
        return a0
    now = time.monotonic()
    rem = float(deadline) - now
    if rem < 5.0 or time_limit <= 0.0:
        return a0
    time_limit = min(float(time_limit), rem)
    maxfun = _choose_maxfun(time_limit, eval_cost, maxfun_cap)
    if maxfun < 120:
        return a0

    stop = min(float(deadline) - 5.0, now + 0.95 * time_limit)

    global _LBFGS_MODE, _LBFGS_BETA, _LBFGS_LAM, _LBFGS_W, _LBFGS_STOP
    _LBFGS_MODE = 1
    _LBFGS_BETA = float(beta) if np.isfinite(beta) and beta > 0 else 1.0
    _LBFGS_LAM = float(lam)
    _LBFGS_W = None
    _LBFGS_STOP = stop

    x0 = np.clip(a0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": max(300, maxfun),
        "maxfun": maxfun,
        "ftol": 1.0e-12,
        "gtol": 1.0e-8,
        "maxcor": 50,
        "maxls": 50,
        "disp": False,
    }

    try:
        res = minimize(
            _lbfgs_func,
            x0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=opts,
            callback=_lbfgs_callback,
        )
        x = np.asarray(res.x, dtype=np.float64).ravel()
    except Exception:
        return a0

    if x.size != n or not np.all(np.isfinite(x)):
        return a0

    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if s <= 1.0e-12:
        return a0
    return x / s


def _lbfgs_weighted(a0, w, deadline, eval_cost, time_limit, lam=1.0e-3, maxfun_cap=20000):
    a0 = _normalize(a0)
    n = a0.size
    if n == 0:
        return a0
    now = time.monotonic()
    rem = float(deadline) - now
    if rem < 5.0 or time_limit <= 0.0:
        return a0
    time_limit = min(float(time_limit), rem)
    maxfun = _choose_maxfun(time_limit, eval_cost, maxfun_cap)
    if maxfun < 120:
        return a0

    stop = min(float(deadline) - 5.0, now + 0.95 * time_limit)

    w = np.asarray(w, dtype=np.float64).ravel()
    ws = float(w.sum())
    if w.size != 2 * n - 1:
        tmp = np.zeros(2 * n - 1, dtype=np.float64)
        m = min(w.size, 2 * n - 1)
        tmp[:m] = w[:m]
        w = tmp
        ws = float(tmp.sum())
    if ws <= 1.0e-300:
        w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
    else:
        w = w / ws

    global _LBFGS_MODE, _LBFGS_BETA, _LBFGS_LAM, _LBFGS_W, _LBFGS_STOP
    _LBFGS_MODE = 2
    _LBFGS_BETA = 1.0
    _LBFGS_LAM = float(lam)
    _LBFGS_W = np.ascontiguousarray(w, dtype=np.float64)
    _LBFGS_STOP = stop

    x0 = np.clip(a0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": max(300, maxfun),
        "maxfun": maxfun,
        "ftol": 1.0e-12,
        "gtol": 1.0e-8,
        "maxcor": 50,
        "maxls": 50,
        "disp": False,
    }

    try:
        res = minimize(
            _lbfgs_func,
            x0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=opts,
            callback=_lbfgs_callback,
        )
        x = np.asarray(res.x, dtype=np.float64).ravel()
    except Exception:
        return a0

    if x.size != n or not np.all(np.isfinite(x)):
        return a0

    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if s <= 1.0e-12:
        return a0
    return x / s


def _weights_top(c, frac):
    c = np.asarray(c, dtype=np.float64).ravel()
    ln = c.size
    if ln == 0:
        return np.empty(0, dtype=np.float64)
    if frac is None or frac <= 0.0:
        w = np.ones(ln, dtype=np.float64) / ln
        return w
    k = int(round(float(frac) * ln))
    if k <= 0:
        k = 1
    if k >= ln:
        return np.ones(ln, dtype=np.float64) / ln
    idx = np.argpartition(c, ln - k)[ln - k:]
    w = np.zeros(ln, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def _weights_soft(c, gamma):
    c = np.asarray(c, dtype=np.float64).ravel()
    ln = c.size
    if ln == 0:
        return np.empty(0, dtype=np.float64)
    if gamma is None or gamma <= 0.0:
        return np.ones(ln, dtype=np.float64) / ln
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm <= 1.0e-300:
        return np.ones(ln, dtype=np.float64) / ln
    z = gamma * (c / cm - 1.0)
    zmax = float(np.max(z))
    ex = np.exp(z - zmax)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        idx = int(np.argmax(c))
        ex = np.zeros(ln, dtype=np.float64)
        ex[idx] = 1.0
        s = 1.0
    w = ex / s
    # Tiny uniform floor improves stability without materially changing emphasis.
    w = 0.98 * w + 0.02 * (np.ones(ln, dtype=np.float64) / ln)
    return w


def _smooth_seq(a, sigma):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1 or sigma <= 0.0:
        return _normalize(a)
    A = np.fft.rfft(a)
    freq = np.fft.rfftfreq(n, d=1.0)
    mult = np.exp(-2.0 * np.pi * np.pi * freq * freq * sigma * sigma)
    b = np.fft.irfft(A * mult, n)
    b = np.maximum(b, 0.0)
    return _normalize(b)


def _interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(int(max(1, new_n)), dtype=np.float64) / float(max(1, new_n))
    if a.size == int(new_n):
        return a.copy()
    new_n = int(max(1, new_n))
    if a.size == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _make_pattern(n, rng):
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    kind = int(rng.integers(0, 12))

    if kind == 0:
        a = np.ones(n, dtype=np.float64)
    elif kind == 1:
        lam = float(rng.uniform(-1.2, 2.0))
        a = 1.0 + lam * (2.0 * x - 1.0)
        a = np.maximum(a, 0.0)
    elif kind == 2:
        p = float(rng.choice([0.5, 1.0, 2.0, 3.0]))
        a = x ** p
    elif kind == 3:
        p = float(rng.choice([0.5, 1.0, 2.0, 3.0]))
        a = (1.0 - x) ** p
    elif kind == 4:
        f = int(rng.integers(1, 8))
        amp = float(rng.uniform(0.1, 0.9))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + amp * np.cos(2.0 * np.pi * f * x + ph)
        a = np.maximum(a, 0.0)
    elif kind == 5:
        nh = int(rng.integers(3, 12))
        s = np.zeros(n, dtype=np.float64)
        for i in range(nh):
            f = i + 1
            w = float(rng.normal(0.0, 1.0))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            s += w * np.cos(2.0 * np.pi * f * x + ph)
        a = 1.0 + 0.5 * s
        a = np.maximum(a, 0.0)
    elif kind == 6:
        theta = float(rng.uniform(0.15, 0.85))
        ss = float(rng.uniform(2.0, 30.0))
        a = 1.0 / (1.0 + np.exp(-ss * (theta - x)))
    elif kind == 7:
        d = np.abs(x - 0.5)
        p = float(rng.choice([1.0, 2.0, 3.0]))
        a = 0.3 + 0.7 * (d ** p)
    elif kind == 8:
        d = np.abs(x - 0.5)
        p = float(rng.choice([1.0, 2.0, 3.0]))
        a = 1.0 / (0.5 + d ** p)
    elif kind == 9:
        c1 = float(rng.uniform(0.1, 0.9))
        c2 = float(rng.uniform(0.1, 0.9))
        sd1 = float(rng.uniform(0.05, 0.30))
        sd2 = float(rng.uniform(0.05, 0.30))
        w1 = 1.0
        w2 = float(rng.uniform(0.2, 2.0))
        a = w1 * np.exp(-0.5 * ((x - c1) / sd1) ** 2) + w2 * np.exp(-0.5 * ((x - c2) / sd2) ** 2)
    elif kind == 10:
        # Two-sided endpoint U-shape.
        d = np.abs(x - 0.5)
        p = float(rng.choice([0.5, 1.0, 2.0]))
        a = 0.4 + 0.6 * (d ** p)
    else:
        # Endpoint-singular-like but bounded.
        d = np.minimum(x, 1.0 - x)
        a = 0.2 + 1.5 / (1.0 + 20.0 * d)

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1.0e-12)
    s = float(a.sum())
    if s <= 1.0e-12:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _make_inits(n, rng, count=12):
    out = []
    if n <= 0:
        return [np.empty(0, dtype=np.float64)]

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    cand = []

    cand.append(np.ones(n, dtype=np.float64))
    for lam in (-0.75, -0.40, 0.40, 0.80, 1.50):
        cand.append(np.maximum(1.0 + lam * (2.0 * x - 1.0), 0.0))
    for p in (0.5, 1.0, 2.0):
        cand.append(x ** p)
        cand.append((1.0 - x) ** p)
        d = np.abs(x - 0.5)
        cand.append(0.3 + 0.7 * (d ** p))
        cand.append(1.0 / (0.7 + d ** p))
    for f in (1, 2, 3):
        for amp in (0.3, 0.7):
            cand.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * f * x), 0.0))
            cand.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * f * x + 1.5707963), 0.0))
    for theta in (0.25, 0.5, 0.75):
        for ss in (3.0, 10.0):
            cand.append(1.0 / (1.0 + np.exp(-ss * (theta - x))))

    det_count = max(2, count // 2)
    rand_count = max(0, count - det_count)

    for a in cand[:det_count]:
        b = np.asarray(a, dtype=np.float64).ravel()
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        s = float(b.sum())
        if s > 1.0e-12:
            out.append(b / s)

    for _ in range(rand_count):
        out.append(_make_pattern(n, rng))

    i = det_count
    while len(out) < count and i < len(cand):
        a = cand[i]
        b = np.asarray(a, dtype=np.float64).ravel()
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        s = float(b.sum())
        if s > 1.0e-12:
            out.append(b / s)
        i += 1

    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out


def _coarse_to_fine(parent, n_target, deadline, rng):
    base = [64, 128, 256, 512, 1024, 2048]
    if n_target < 4096:
        levels = [L for L in base if L <= max(64, int(n_target * 1.3))]
        if not levels:
            levels = [64]
    else:
        levels = base

    small = None

    for L in levels:
        now = time.monotonic()
        if now > float(deadline):
            break

        sample = np.ones(L, dtype=np.float64) / float(L)
        cost = _estimate_eval_cost(sample)

        inits = []
        if parent is not None and parent.size > 0:
            inits.append(_interpolate_seq(parent, L))
        if small is not None:
            inits.append(_interpolate_seq(small, L))
        inits.extend(_make_inits(L, rng, 8))
        inits = inits[:8]

        best_this = None
        best_this_sc = np.inf

        for a0 in inits:
            now = time.monotonic()
            if now > float(deadline):
                break
            a = _normalize(a0)
            sc = _score(a)
            if np.isfinite(sc) and sc < best_this_sc:
                best_this = a.copy()
                best_this_sc = sc

            c = _fast_conv(a)
            cm = float(np.max(c)) if c.size else 1e-12
            cm = max(cm, 1.0e-30)

            for target in (10.0, 50.0):
                now = time.monotonic()
                if now > float(deadline):
                    break
                beta = target / cm
                op_time = min(18.0, max(2.5, 0.035 * (float(deadline) - now)))
                b = _lbfgs_lse(a, beta, deadline, cost, op_time)
                b = _normalize(b)
                sb = _score(b)
                if np.isfinite(sb) and sb < best_this_sc:
                    best_this = b.copy()
                    best_this_sc = sb
                    a = b

            now = time.monotonic()
            if now > float(deadline):
                break
            cc = _fast_conv(a)
            ww = _weights_top(cc, 0.05)
            op_time = min(10.0, max(2.0, 0.025 * (float(deadline) - now)))
            b = _lbfgs_weighted(a, ww, deadline, cost, op_time)
            b = _normalize(b)
            sb = _score(b)
            if np.isfinite(sb) and sb < best_this_sc:
                best_this = b.copy()
                best_this_sc = sb

        if best_this is not None:
            small = best_this

    if small is not None:
        return _interpolate_seq(small, n_target)
    if parent is not None and parent.size > 0:
        return _interpolate_seq(parent, n_target)
    return _normalize(np.ones(n_target, dtype=np.float64) / float(n_target))


def _local_refine(a, deadline, eval_cost, rng, max_cycles=8, base_op_time=80.0):
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a, np.inf

    best = a.copy()
    best_sc = _score(best)
    current = best.copy()
    no_improve = 0

    ops = (
        ("top", 0.10),
        ("top", 0.03),
        ("soft", 20.0),
        ("top", 0.008),
        ("soft", 80.0),
        ("lse", 40.0),
        ("top", 0.002),
        ("lse", 250.0),
    )

    print(f"  refine n={n} start={best_sc:.8f}")

    for _cycle in range(max_cycles):
        now = time.monotonic()
        if now > float(deadline):
            break

        for op in ops:
            now = time.monotonic()
            rem = float(deadline) - now
            if rem < 25.0:
                break

            op_time = max(10.0, min(float(base_op_time), 0.15 * rem))
            kind = op[0]
            param = op[1]

            c = _fast_conv(current)
            cm = float(np.max(c)) if c.size else 1e-12

            if kind == "top":
                w = _weights_top(c, param)
                b = _lbfgs_weighted(current, w, deadline, eval_cost, op_time)
            elif kind == "soft":
                w = _weights_soft(c, param)
                b = _lbfgs_weighted(current, w, deadline, eval_cost, op_time)
            else:
                if cm <= 1.0e-300:
                    continue
                beta = param / cm
                b = _lbfgs_lse(current, beta, deadline, eval_cost, op_time)

            b = _normalize(b)
            sb = _score(b)
            if np.isfinite(sb) and sb < best_sc:
                best = b
                best_sc = sb
                current = b.copy()
                no_improve = 0
                print(f"    n={n} new best={best_sc:.8f} t={time.monotonic()-_T_START:.0f}s")
            else:
                no_improve += 1
                if np.isfinite(sb) and sb < best_sc * 1.003:
                    current = b
                else:
                    current = best.copy()

        # Cheap deterministic symmetrization / smoothing.
        b = _normalize(0.5 * (best + best[::-1]))
        sb = _score(b)
        if np.isfinite(sb) and sb < best_sc:
            best = b.copy()
            best_sc = sb
            current = best.copy()
            no_improve = 0
            print(f"    sym improved n={n} score={best_sc:.8f}")

        for sig in (1.0, 3.0, 8.0, 16.0):
            b = _smooth_seq(best, sig)
            sb = _score(b)
            if np.isfinite(sb) and sb < best_sc:
                best = b.copy()
                best_sc = sb
                current = best.copy()
                no_improve = 0
                print(f"    smooth({sig:0.1f}) improved n={n} score={best_sc:.8f}")
                break

        # Asymmetric escapes when plateauing.
        if no_improve >= 6:
            now = time.monotonic()
            rem = float(deadline) - now
            if rem > 35.0:
                improved_escape = False
                x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
                ramps = [
                    np.maximum(1.0 - 0.8 * (x - 0.5), 0.0),
                    np.maximum(1.0 + 0.8 * (x - 0.5), 0.0),
                    np.maximum(1.0 - 1.6 * (x - 0.5), 0.0),
                    np.maximum(1.0 + 1.6 * (x - 0.5), 0.0),
                ]

                for t in (0.05, 0.12, 0.25):
                    if improved_escape:
                        break
                    for pat in ramps:
                        now = time.monotonic()
                        rem = float(deadline) - now
                        if rem < 20.0:
                            break
                        mix = _normalize((1.0 - t) * best + t * pat)
                        c = _fast_conv(mix)
                        w = _weights_top(c, 0.05)
                        op_time = max(8.0, min(28.0, 0.05 * rem))
                        b = _lbfgs_weighted(mix, w, deadline, eval_cost, op_time)
                        sb = _score(_normalize(b))
                        if np.isfinite(sb) and sb < best_sc:
                            best = _normalize(b).copy()
                            best_sc = sb
                            current = best.copy()
                            improved_escape = True
                            no_improve = 0
                            print(f"    escape improved n={n} score={best_sc:.8f} t={t:.2f}")
                            break

                if not improved_escape:
                    now = time.monotonic()
                    rem = float(deadline) - now
                    if rem > 25.0:
                        pat = _make_pattern(n, rng)
                        t = 0.15
                        mix = _normalize((1.0 - t) * best + t * pat)
                        c = _fast_conv(mix)
                        w = _weights_top(c, 0.03)
                        op_time = max(8.0, min(30.0, 0.05 * rem))
                        b = _lbfgs_weighted(mix, w, deadline, eval_cost, op_time)
                        sb = _score(_normalize(b))
                        if np.isfinite(sb) and sb < best_sc:
                            best = _normalize(b).copy()
                            best_sc = sb
                            current = best.copy()
                            no_improve = 0
                            print(f"    random escape improved n={n} score={best_sc:.8f}")
                        else:
                            no_improve = max(0, no_improve - 3)

    print(f"  refine n={n} end={best_sc:.8f}")
    return best, best_sc


def propose_candidate(seed=20241, budget_s=2400):
    global _T_START
    _T_START = time.monotonic()
    budget = max(120.0, float(budget_s))
    deadline = _T_START + budget - 45.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            if p.size > 0:
                p = np.where(np.isfinite(p), p, 0.0)
                p = np.clip(p, 0.0, 1000.0)
                if float(p.sum()) > 0.0:
                    parent = p
    except Exception:
        parent = None

    max_n = 12000
    if parent is not None:
        if parent.size > max_n:
            parent = _interpolate_seq(parent, max_n)
        if parent.size < 128:
            parent = _interpolate_seq(parent, 256)
        n_target = int(parent.size)
    else:
        n_target = 6962 if budget > 1500 else 4096

    n_target = int(min(max(64, n_target), max_n))

    if parent is not None:
        best = _normalize(parent)
        best_sc = _score(best)
        if not np.isfinite(best_sc):
            best_sc = np.inf
            best = np.ones(n_target, dtype=np.float64) / float(n_target)
            best_sc = _score(best)
    else:
        best = np.ones(n_target, dtype=np.float64) / float(n_target)
        best_sc = _score(best)

    print(f"Initial n={n_target} score={best_sc:.8f} t={time.monotonic()-_T_START:.0f}s")

    # Coarse-to-fine exploration, especially important if starting point is local.
    if parent is not None:
        coarse_deadline = _T_START + min(150.0, 0.07 * (budget - 60.0))
    else:
        coarse_deadline = _T_START + min(480.0, 0.28 * (budget - 60.0))
    coarse_deadline = min(coarse_deadline, deadline - 300.0)

    if time.monotonic() < coarse_deadline - 5.0:
        try:
            coarse = _coarse_to_fine(parent, n_target, coarse_deadline, rng)
            coarse = _normalize(coarse)
            sc = _score(coarse)
            if np.isfinite(sc) and sc < best_sc:
                best = coarse
                best_sc = sc
                print(f"Coarse-to-fine improved score={best_sc:.8f} t={time.monotonic()-_T_START:.0f}s")
        except Exception:
            pass

    eval_cost = _estimate_eval_cost(best)

    # Main fine-grained refinement at the current length.
    main_deadline = min(deadline - 180.0, _T_START + 0.82 * (budget - 60.0))
    if time.monotonic() < main_deadline - 30.0:
        best, best_sc = _local_refine(best, main_deadline, eval_cost, rng, max_cycles=8, base_op_time=80.0)

    # Try a few length changes. Larger lengths often help, but sometimes a nearby
    # lower length is accidentally better after coarse artifacts.
    factors = (1.10, 1.25, 1.50, 0.80)
    for fac in factors:
        now = time.monotonic()
        if now > deadline - 130.0:
            break
        n_new = int(round(len(best) * float(fac)))
        if n_new < 256 or n_new > max_n:
            continue
        if n_new == len(best):
            continue

        a_new = _interpolate_seq(best, n_new)
        sc_new = _score(a_new)
        if not np.isfinite(sc_new) or sc_new > max(np.isfinite(best_sc) * 1.15, np.inf if not np.isfinite(best_sc) else 2.0):
            continue

        cost_new = _estimate_eval_cost(a_new)
        op_deadline = min(deadline - 60.0, now + 180.0)
        if op_deadline < time.monotonic() + 30.0:
            continue

        b_new, sc_test = _local_refine(a_new, op_deadline, cost_new, rng, max_cycles=1, base_op_time=60.0)
        if np.isfinite(sc_test) and sc_test < best_sc:
            best = b_new
            best_sc = sc_test
            print(f"Length change to {len(best)} improved score={best_sc:.8f} t={time.monotonic()-_T_START:.0f}s")
            if time.monotonic() > deadline - 180.0:
                break

    # Final short polish.
    now = time.monotonic()
    if now < deadline - 35.0:
        eval_cost = _estimate_eval_cost(best)
        final_deadline = min(deadline - 20.0, now + 120.0)
        best, best_sc = _local_refine(best, final_deadline, eval_cost, rng, max_cycles=1, base_op_time=45.0)

    out_arr = np.asarray(best, dtype=np.float64).ravel()
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    s = float(out_arr.sum())
    if s <= 0.1:
        if s > 1.0e-12:
            out_arr = out_arr / s
        else:
            out_arr = np.full(out_arr.size if out_arr.size > 0 else 256, 1.0, dtype=np.float64)
            out_arr = out_arr / out_arr.size

    out_list = [float(max(0.0, min(1000.0, x))) for x in out_arr]
    if not out_list:
        out_list = [0.5] * 256

    verified = np.inf
    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = np.inf

    if not np.isfinite(verified):
        # Defensive fallback: downsample and try again; if still invalid, return a
        # uniformly valid sequence.
        if len(out_list) > 2000:
            small = _interpolate_seq(np.asarray(out_list, dtype=np.float64), 2000)
            out_list = [float(max(0.0, min(1000.0, x))) for x in small]
            try:
                verified = float(evaluate_sequence(out_list))
            except Exception:
                verified = np.inf
        if not np.isfinite(verified):
            out_list = [0.5] * max(64, len(out_arr) if out_arr.size > 0 else 256)
            verified = 2.0

    elapsed = time.monotonic() - _T_START
    print("=" * 70)
    print(f"FINAL n={len(out_list)} internal={best_sc:.8f} evaluated={verified:.8f} time={elapsed:.1f}s")
    print("=" * 70)

    return out_list
```
