def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_objective_grad(a, beta):
    """LogSumExp-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    """Run L-BFGS-B optimization with LogSumExp smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

        lse_objective_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    AR = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AR * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=500000):
    """L-BFGS-B with LogSumExp smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-25, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    return compute_score(a_out), a_out


def spectral_shape(a, n_iters=3000, alpha_init=0.6):
    """
    Fienup-style spectral shaping: flatten the convolution power spectrum
    while maintaining non-negativity via clipping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init * (1.0 - progress) ** 1.5 + 0.02
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = np.fft.irfft(P, M_fft)[:2 * n - 1]
        c_max = float(c.max())
        c_mean = float(c.mean())
        # Target: blend c toward uniform
        c_target = (1.0 - alpha) * c + alpha * c_mean
        P_target = np.fft.rfft(c_target, M_fft)
        P_target = np.maximum(P_target.real, 1e-30)
        # Blend in power domain
        P_new = (1.0 - alpha) * P + alpha * P_target
        P_new = np.maximum(P_new, 1e-20)
        scale = np.sqrt(P_new / (P + 1e-30))
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def targeted_cd(a, n_sweeps=30, deadline=None):
    """
    Peak-targeted coordinate descent:
    At each step, find the worst peak and adjust the single most impactful variable.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        c_max = float(c.max())
        # Find top peaks (within 0.5% of max)
        threshold = c_max * 0.9995
        active_peaks = np.where(c >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([int(np.argmax(c))])

        improved = False
        for k in active_peaks[:8]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end <= j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            # Contribution of a[j] to c[k]: a[j]*a[k-j]
            # If we change a[j] by delta, c[k] changes by delta * a[k-j]
            # To reduce c[k], decrease a[j] if a[k-j] > 0
            # Try scaling a[j] for the top contributors
            contribs = a[js] * a[k - js]
            top_count = min(20, len(contribs))
            top_local = np.argsort(contribs)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global[:10]:
                j = int(j)
                if a[j] < 1e-14:
                    continue
                a_old = a[j]
                best_val = a_old
                best_local_sc = compute_score(a)
                # Try several scale factors
                for f in [0.7, 0.8, 0.85, 0.9, 0.93, 0.96, 1.04, 1.1, 1.25]:
                    a[j] = a_old * f
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old
                        continue
                    a_norm = a / S_tmp
                    sc_tmp = compute_score(a_norm)
                    if sc_tmp < best_local_sc:
                        best_local_sc = sc_tmp
                        best_val = a_old * f
                if best_val != a_old:
                    a[j] = best_val
                    improved = True

        if improved:
            S = a.sum()
            if S > 1e-12:
                a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def multi_beta_lbfgs(a, deadline, betas=None):
    """
    Run L-BFGS with increasing beta until deadline.
    Each beta gets more time as we go higher (finer smoothing).
    """
    if betas is None:
        betas = [200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]

    best_a = a.copy()
    best_sc = compute_score(a)
    total_beta_time = deadline - time.monotonic()
    # Allocate time proportional to index (later betas get more time)
    weights = np.arange(1, len(betas) + 1, dtype=float)
    weights /= weights.sum()
    times = weights * total_beta_time * 0.9  # Leave 10% buffer

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        # Allocate time for this beta
        alloc = min(times[i], remaining - 5)
        if alloc < 5:
            alloc = remaining - 5
        maxiter = int(max(1000, alloc * 8))
        maxfun = maxiter * 30
        print(f"    L-BFGS beta={beta}, alloc={alloc:.0f}s, maxiter={maxiter}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"      -> {sc:.10f} ({dt:.1f}s)")
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()

    return best_a, best_sc


def random_perturb_and_refine(a, rng, n_attempts=8, deadline=None):
    """Try random perturbations followed by quick L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(n_attempts):
        if deadline is not None and time.monotonic() > deadline:
            break
        strategy = r % 4
        if strategy == 0:
            # Small Gaussian
            a_p = a * (1.0 + 0.01 * rng.standard_normal(n))
        elif strategy == 1:
            # Low-freq cosine bump
            f0 = rng.integers(1, 5)
            x = np.arange(n) / n
            a_p = a * (1.0 + 0.02 * np.cos(2 * np.pi * f0 * x))
        elif strategy == 2:
            # Block perturbation
            bs = max(1, n // 32)
            blk = rng.integers(0, n - bs)
            a_p = a.copy()
            a_p[blk:blk + bs] *= (1.0 + 0.05 * rng.standard_normal(bs))
        else:
            # Peak-contributing indices get boosted noise
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j0 = max(0, k_star - n + 1)
            j1 = min(k_star, n - 1)
            a_p = a.copy()
            noise = 0.015 * rng.standard_normal(n)
            noise[j0:j1 + 1] *= 3.0
            a_p = a * (1.0 + noise)

        a_p = np.maximum(a_p, 0.0)
        S = a_p.sum()
        if S < 1e-12:
            continue
        a_p = a_p / S

        # Quick refine
        sc, a_opt = run_lbfgs(a_p, 10000, maxiter=5000, maxfun=100000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"      Perturb {r}: {sc:.10f}")

    return best_a, best_sc


def try_length(parent, target_n, deadline):
    """Interpolate parent to target length and do quick optimization."""
    n_old = len(parent)
    if n_old == target_n:
        a = parent.copy()
    else:
        x_old = np.linspace(0, 1, n_old)
        x_new = np.linspace(0, 1, target_n)
        a = np.interp(x_new, x_old, parent)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        else:
            a = np.ones(target_n) / target_n
    sc = compute_score(a)
    # Quick L-BFGS
    remaining = deadline - time.monotonic()
    if remaining > 20:
        maxiter = int(min(10000, remaining * 5))
        sc, a = run_lbfgs(a, 5000, maxiter=maxiter, maxfun=maxiter * 20)
    return sc, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization focusing on thorough local refinement.
    Phases: spectral init -> multi-beta L-BFGS -> targeted CD -> restarts -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 90.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def record(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        record(sc_p, parent, "parent")
    else:
        print("No parent found, using uniform init")
        sc_u = compute_score(best_a)
        record(sc_u, best_a, "uniform")

    n = len(best_a)
    print(f"Starting: n={n}, score={best_score:.10f}")

    # ================================================================
    # Phase 1: Spectral shaping for a better starting point (10%)
    # ================================================================
    p1_end = t_start + 0.08 * (total_budget - 90.0)
    print(f"\n=== Phase 1: Spectral shaping ===")

    a_spec = spectral_shape(best_a, n_iters=2000, alpha_init=0.5)
    sc = compute_score(a_spec)
    record(sc, a_spec, "spectral")

    # Also try a sinusoidal-modulated initialization
    x = np.arange(n) / n
    for alpha_mod in [0.1, 0.2, 0.3]:
        a_sine = 1.0 + alpha_mod * np.cos(2 * np.pi * x)
        a_sine = np.maximum(a_sine, 0.0)
        a_sine = a_sine / a_sine.sum()
        a_sine = spectral_shape(a_sine, n_iters=500, alpha_init=0.4)
        sc = compute_score(a_sine)
        record(sc, a_sine, f"sine-{alpha_mod}")

    if time.monotonic() > p1_end:
        print("  Phase 1 time exceeded, moving on")

    # ================================================================
    # Phase 2: Multi-beta L-BFGS (45%) - main workhorse
    # ================================================================
    p2_end = t_start + 0.53 * (total_budget - 90.0)
    print(f"\n=== Phase 2: Multi-beta L-BFGS ===")

    a_lb, sc_lb = multi_beta_lbfgs(best_a, p2_end)
    record(sc_lb, a_lb, "multi-lbfgs")

    # ================================================================
    # Phase 3: Targeted coordinate descent (15%)
    # ================================================================
    p3_end = t_start + 0.68 * (total_budget - 90.0)
    print(f"\n=== Phase 3: Targeted CD ===")

    a_cd = targeted_cd(best_a, n_sweeps=40, deadline=p3_end)
    sc = compute_score(a_cd)
    record(sc, a_cd, "cd")

    # L-BFGS after CD
    if time.monotonic() < p3_end - 10:
        remaining = p3_end - 10 - time.monotonic()
        maxiter = int(max(3000, remaining * 6))
        sc2, a2 = run_lbfgs(a_cd, 50000, maxiter=maxiter, maxfun=maxiter * 20)
        record(sc2, a2, "cd+lbfgs")

    # Another CD round
    if time.monotonic() < p3_end:
        a_cd2 = targeted_cd(best_a, n_sweeps=20, deadline=p3_end)
        sc = compute_score(a_cd2)
        record(sc, a_cd2, "cd2")

    # ================================================================
    # Phase 4: Length exploration (10%)
    # ================================================================
    p4_end = t_start + 0.78 * (total_budget - 90.0)
    print(f"\n=== Phase 4: Length exploration ===")

    for target_n in [4096, 8192]:
        if time.monotonic() > p4_end:
            break
        print(f"  Trying n={target_n}")
        sc_t, a_t = try_length(best_a, target_n, p4_end)
        record(sc_t, a_t, f"n={target_n}")

    # Also try a fresh random initialization at 4096
    if time.monotonic() < p4_end:
        n_rand = 4096
        a_rand = rng.random(n_rand) ** 2
        a_rand = a_rand / a_rand.sum()
        a_rand = spectral_shape(a_rand, n_iters=500, alpha_init=0.5)
        remaining = p4_end - time.monotonic()
        if remaining > 15:
            maxiter = int(min(5000, remaining * 5))
            sc_r, a_r = run_lbfgs(a_rand, 5000, maxiter=maxiter, maxfun=maxiter * 15)
            record(sc_r, a_r, "random-4096")

    # ================================================================
    # Phase 5: Random perturbations + refine (12%)
    # ================================================================
    p5_end = t_start + 0.90 * (total_budget - 90.0)
    print(f"\n=== Phase 5: Random perturbations ===")

    a_rp, sc_rp = random_perturb_and_refine(best_a, rng, n_attempts=12, deadline=p5_end)
    record(sc_rp, a_rp, "perturb")

    # ================================================================
    # Phase 6: Final high-beta polish (10%)
    # ================================================================
    print(f"\n=== Phase 6: Final polish ===")

    if time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        maxiter = int(max(5000, remaining * 8))
        sc_f, a_f = run_lbfgs(best_a, 500000, maxiter=maxiter, maxfun=maxiter * 15)
        record(sc_f, a_f, "final-500k")

    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        maxiter = int(max(3000, remaining * 8))
        sc_f2, a_f2 = run_lbfgs(best_a, 1000000, maxiter=maxiter, maxfun=maxiter * 12)
        record(sc_f2, a_f2, "final-1M")

    # Final CD
    if time.monotonic() < deadline - 10:
        a_cd_f = targeted_cd(best_a, n_sweeps=10, deadline=deadline - 10)
        sc = compute_score(a_cd_f)
        record(sc, a_cd_f, "final-cd")

    # Last L-BFGS if time remains
    if time.monotonic() < deadline - 5:
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            maxiter = int(remaining * 5)
            sc_last, a_last = run_lbfgs(best_a, 500000, maxiter=maxiter, maxfun=maxiter * 15)
            record(sc_last, a_last, "absolute-final")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 70}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 70}")
    return out
