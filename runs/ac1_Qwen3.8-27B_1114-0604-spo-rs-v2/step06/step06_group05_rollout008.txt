"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: LP-based active set method (linearized peak reduction),
LSE-LBFGS refinement, spectral GS, and Fourier-parameterized CMA-ES.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    max_scaled = float(np.max(scaled))
    exp_s = np.exp(scaled - max_scaled)
    log_sum = max_scaled + float(np.log(np.sum(exp_s)))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / np.sum(exp_s)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / (S2 * S)
    return score, grad

def solve_lp_active_set(a, n_peaks=200, step_limit=0.1):
    """Solve LP subproblem to reduce top peaks via linearized constraints."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    
    # Top peaks
    peak_idx = np.argsort(c)[::-1][:n_peaks]
    
    # Build LP: min t
    #   s.t. 2*sum_i a[max(0,k...)] δ_i - t <= -c_{p_m}  for each peak
    #        sum δ_i = 0
    #        -a_i <= δ_i <= step_limit
    #        t >= 0
    
    # Variables: [δ_0, ..., δ_{n-1}, t]  (n+1 variables)
    nvars = n + 1
    
    # Objective: minimize t
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0  # t is the last variable
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each peak p: 2 * sum_i a_{valid}(p-i) * δ_i - t <= -c_p
    
    # Build the matrix efficiently
    # For peak p, the coefficient of δ_i is 2*a_{p-i} if 0 <= p-i < n, else 0
    
    A_rows = []
    b_vals = []
    
    for p in peak_idx:
        p = int(p)
        row = np.zeros(nvars)
        # For peak p: coefficients are 2*a_{p-i} for valid i
        i_start = max(0, p - n + 1)
        i_end = min(p, n - 1)
        if i_end >= i_start:
            # a_{p-i} for i in [i_start, i_end]
            vals = a[p - i_end : p - i_start + 1][::-1] * 2.0
            row[i_start:i_end+1] = vals
        row[n] = -1.0  # coefficient of t
        A_rows.append(row)
        b_vals.append(-c[p])
    
    A_ub = np.array(A_rows) if A_rows else np.zeros((0, nvars))
    b_ub = np.array(b_vals) if b_vals else np.zeros(0)
    
    # Equality: sum δ = 0
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -a_i <= δ_i <= step_limit, t >= 0
    lb = np.maximum(a, 0.0)  # -a_i (we want δ_i >= -a_i, so lower bound is -a_i)
    # Wait, linprog bounds are (lower, upper) for each variable
    lb = np.concatenate([-a, [0.0]])
    ub = np.concatenate([np.full(n, step_limit), [np.inf]])
    bounds = list(zip(lb, ub))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return a.copy()
    
    delta = res.x[:n]
    a_new = a + delta
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a.copy()
    a_new = a_new / S_new
    return a_new

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = np.empty(nvars)
    ub = np.empty(nvars)
    lb[:n] = -a  # δ_i >= -a_i
    ub[:n] = step_limit  # δ_i <= step_limit
    lb[n] = 0.0
    ub[n] = np.inf
    bounds = list(zip(lb.tolist(), ub.tolist()))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status != 0:
            return None, c, peak_idx
        delta = res.x[:n]
        return delta, c, peak_idx
    except Exception:
        return None, c, peak_idx

def cosine_init(n, center, width):
    """Generate a cosine-windowed initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (x - center) / width) ** 2
    a[x*0 + 1] = 1  # placeholder, will overwrite
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (x - center) / width)**2
    a[x < 0] = 0
    a[x > n-1] = 0
    a = np.maximum(a, 0)
    S = a.sum()
    if S > 1e-12:
        a = a / S
    else:
        a = np.ones(n) / n
    return a

def diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0, 1000).copy())
        else:
            xo = np.linspace(0, 1, len(parent))
            xn = np.linspace(0, 1, n)
            inits.append(np.interp(xn, xo, parent).copy())
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4*np.cos(2*np.pi*x), 0.01))
    inits.append(0.3 + 0.7*np.cos(2*np.pi*x)**2)
    # Cosine window
    inits.append(np.maximum(np.cos(np.pi*(x - 0.5))**2, 0.001))
    # Symmetric bump
    inits.append(np.maximum(0.2 + 0.8*np.exp(-8*(x-0.5)**2), 0.001))
    # Random
    inits.append(np.clip(0.5 + 0.3*rng.standard_normal(n), 0.01, 10))
    # Multi-frequency
    inits.append(np.maximum(0.4 + 0.3*np.cos(4*np.pi*x) + 0.2*np.cos(2*np.pi*x), 0.01))
    inits.append(np.maximum(0.3 + 0.4*np.cos(6*np.pi*x) + 0.3*np.cos(2*np.pi*x), 0.01))
    return inits
