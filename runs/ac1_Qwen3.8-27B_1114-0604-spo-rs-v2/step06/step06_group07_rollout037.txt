    c = fast_conv(a)
    S = float(a.sum())

        return np.inf

def target_spectrum(n):
    """Compute the target |A[m]| for a flat convolution."""
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    # Target: c[k] = 1/(2n-1) for all k (normalized sum=1)
    # C[m] = FFT(c)[m] = 1/(2n-1) * sum_{k=0}^{2n-2} exp(-2πimk/M_fft)
    # This is a Dirichlet kernel
    c_target = np.ones(M) / M
    C_target = np.fft.rfft(c_target, M_fft)
    # A_target = sqrt(C_target) (complex square root)
    # For the magnitude: |A_target[m]| = sqrt(|C_target[m]|)
    # But C_target can be negative (sinc has negative lobes), so we need to be careful
    # For the phase retrieval, we use: A_target[m] = sqrt(C_target[m]) with appropriate branch
    # Simplification: use |C_target[m]| for the magnitude target
    mag_target = np.sqrt(np.maximum(C_target.real, 0))  # Only use where C is positive (real)
    # Actually C_target is complex in general. Let's use |C_target|
    mag_target = np.sqrt(np.abs(C_target))
    return mag_target, M_fft

def fienup_with_target(a, n_iters=500, alpha_start=0.05, alpha_end=0.8):
    """Fienup iteration with target spectrum magnitude."""
    n = len(a)

    A = np.fft.rfft(a, M_fft)
    amp_current = np.abs(A)
    # Scale target to match the DC component

        scale = amp_current[0] / mag_target[0]
        mag_target = mag_target * scale
    else:
        mag_target = mag_target / mag_target.max() * amp_current.max()

    FA = np.fft.rfft(x[::-1], M2_fft)
    conv_res = np.fft.irfft(P * FA, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S3) * (g_L * S - 2.0 * L)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) * 0.5
        S = n * 0.5
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0.copy()
    return compute_score(a_out), a_out

def build_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Edge-enhanced
    inits.append(np.maximum(0.1 + 0.9 * np.abs(2.0 * x - 1.0), 0.01))
    
    # Cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * x)), 0.01))
    
    # Gaussian center
    inits.append(np.exp(-4.0 * (2.0 * x - 1.0) ** 2) + 0.05)
    
    # Two edges
    inits.append(np.maximum(0.1 + 0.8 * (x ** 3 + (1 - x) ** 3), 0.01))
    
    # Random smooth
    x_fine = np.arange(n) / n
    freqs = rng.integers(1, 15, size=7)
    phases = rng.uniform(0, 2 * np.pi, size=7)
    amps = rng.uniform(0.05, 0.4, size=7)
    sig = np.ones(n) * 0.5
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * x_fine + ph)
    inits.append(np.maximum(sig, 0.01))
    
    # Alternating (high-low pattern)
    alt = np.where(np.arange(n) % 2 == 0, 0.8, 0.2)
    inits.append(alt.astype(np.float64))
    
    # Power law
    inits.append(np.maximum(0.1 + 0.9 * x, 0.01))
    
    return inits
