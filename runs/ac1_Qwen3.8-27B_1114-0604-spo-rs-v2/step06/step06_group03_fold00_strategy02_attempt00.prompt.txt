def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):

def resample_sequence(a, n):

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    # Clip z to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def active_weights_topk(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def exact_subgrad_step(a, k_active, eta):
    """One subgradient step on the exact max objective using top-k lags."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        return a
    grad = (4.0 * float(len(a)) * h) / (S * S)
    LDS = float(np.dot(w, d))
    grad -= (4.0 * LDS) / S
    return a - eta * grad


def mirror_descent_step(a, k_active, t_power):
    """Mirror descent (multiplicative) step using the power iteration insight.
    At optimum: corr_pair(a, w_active) ~ lambda * a.
    We do: a_new = (corr_pair(a, w_active))^t * a^(1-t)
    """
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a
    w = active_weights_topk(d, k_active)
    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)
    # Soft power iteration: blend a with h
    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_power) * log_a + t_power * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def backtracking_line_search(a, grad, obj_fn, max_iter=30):
    """Backtracking line search for step size."""
    n = len(a)
    f0 = obj_fn(a)
    if not np.isfinite(f0):
        return a, f0
    step = 1.0
    scaled_grad = grad / (np.max(np.abs(grad)) + 1e-15)
    for _ in range(max_iter):
        a_trial = normalize_a(np.maximum(a - step * scaled_grad, 0.0))
        f_trial = obj_fn(a_trial)
        if np.isfinite(f_trial) and f_trial < f0 - 1e-12:
            return a_trial, f_trial
        step *= 0.5
        if step < 1e-10:
            break
    return a, f0


def subgrad_polish_exact(a, max_iters=50, deadline=None, k_list=(1, 2, 3, 5, 8, 15)):
    """Polish using exact max subgradient with adaptive step sizes."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        improved = False
        for k in k_list:
            for eta in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                q = exact_subgrad_step(a, k, eta)
                q = normalize_a(np.maximum(q, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < exact_score_a(a):
                    a = q
        if not improved and it > 5:
            break
    return float(best_sc), best_a

def power_iteration_refine(a_start, max_iters=100, deadline=None):
    """Pure power iteration: repeatedly compute h = corr_pair(a, w_active) and replace a with h.
    With soft blend to maintain stability."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        dmax = float(np.max(d))
        if dmax <= 0:
            break
        # Use top-3 active set
        w = active_weights_topk(d, 3)
        h = corr_pair(a, w)
        h = np.maximum(h, 1e-12)
        # Soft update
        for blend in (0.1, 0.3, 0.5, 0.7):
            q = normalize_a(blend * h + (1.0 - blend) * a)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q.copy()
                break

    return float(best_sc), best_a


def generate_inits(n, rng, parent, cap=25):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    add(np.ones(n, dtype=np.float64))

    # V-shapes (key structure)
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Cosine shapes
    for f in (1, 2, 3):
        for alpha in (0.3, 0.6, 1.0, 1.5):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
            add(1.0 + alpha * np.cos(np.pi * x))

    # Bimodal
    for width in (0.1, 0.15, 0.2, 0.3):
        add(np.exp(-(((x - 0.5) / width) ** 2)) + np.exp(-(((x + 0.5) / width) ** 2)))
        add(np.exp(-(((x - 0.25) / width) ** 2)) + np.exp(-(((x - 0.75) / width) ** 2)))

    # Exponential
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x))
        add(np.exp(-rate * (1.0 - x)))
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)))

    # Sine
    for f in (1, 2, 3, 4, 6):
        add(1.0 + 0.5 * np.sin(np.pi * f * x))
        add(1.0 + 0.5 * np.abs(np.sin(np.pi * f * x)))

    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)

    return inits


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 8))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_topk(d, int(rng.integers(1, 6)))
                h = corr_pair(p, w)
                hmax = float(np.max(h))
                if hmax > 1e-12:
                    q = p * np.exp(-strength * h / hmax)
                else:
                    q = p.copy()
            else:
                q = p.copy()
        else:
            q = p.copy()
    elif mode == 4:
        # Smoothed noise in log-space
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.3 * noise
        q = np.exp(logp - float(np.max(logp)))
    elif mode == 5:
        # V-shape perturbation
        x = np.linspace(0.0, 1.0, n)
        vshape = 1.0 + strength * 0.5 * np.abs(2.0 * x - 1.0)
        q = p * vshape
    elif mode == 6:
        # Gaussian smoothing
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        q = np.convolve(p, kern, mode="same")
    else:
        # Targeted: reduce values at positions contributing most to peak
        d, S = d_from_a(p)
        if d.size > 0:
            dmax = float(np.max(d))
            if dmax > 1e-12:
                w = active_weights_topk(d, 2)
                h = corr_pair(p, w)
                hmax = float(np.max(h) + 1e-15)
                q = p * np.exp(-strength * 0.3 * h / hmax)
            else:
                q = p.copy()
        else:
            q = p.copy()

    return normalize_a(np.clip(q, 0.0, None))


def optimize_with_lbfgs_trajectory(a_start, deadline, label):

    a = normalize_a(a_start)

        (200.0, 0.08),
        (500.0, 0.08),
        (1000.0, 0.08),
        (3000.0, 0.10),
        (8000.0, 0.10),
        (20000.0, 0.12),
        (50000.0, 0.12),
        (100000.0, 0.12),
        (300000.0, 0.12),
        (500000.0, 0.10),
    ]

    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        if cum > 1.0:
            cum = 1.0
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())

            mf = estimate_maxfun(rem * 0.85, len(a))

                break
            sc, a2 = run_lbfgs(a, float(beta), deadline, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                if (best_sc < 1.51 or it % 5 == 0):
                    pass  # quiet
            a = normalize_a(a2)

    return float(best_sc), best_a
