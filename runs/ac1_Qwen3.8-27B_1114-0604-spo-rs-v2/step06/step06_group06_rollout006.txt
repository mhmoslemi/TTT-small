def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    result = full[n - 1:2 * n - 1]
    if result.size != n:
        if result.size < n:
            result = np.pad(result, (0, n - result.size))
        else:
            result = result[:n]
    return result


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def compute_active_weights(conv, top_k=10, eps_frac=0.01):
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.10 * conv.size))
    if active.sum() > max_count:
        top_idx = np.argpartition(conv, -max_count)[-max_count:]
        active2 = np.zeros(conv.size, dtype=bool)
        active2[top_idx] = True
        active = active2
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s

def lse_objective_and_grad(a, beta):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    d = (float(n) / (S * S)) * c
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    wdotd = float(np.dot(w, d))
    h = corr_pair(a, w)
    grad = (2.0 * n * n * h) / (S * S) - 4.0 * n * a * wdotd
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * n * L), grad


def run_lbfgs(a, beta, deadline, maxfun=400):
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        return lse_objective_and_grad(x, beta)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-12,
        "maxiter": max(20, maxfun // 2),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n)) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def perturb(a, rng, strength=0.15):
    n = len(a)
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.25))
        q = (1.0 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        # Frequency perturbation
        N = next_power_of_two(n)
        A = np.fft.rfft(a, N)
        mask = np.zeros(A.size, dtype=bool)
        mask[1:-1] = True if A.size < 5 else np.arange(1, A.size - 1) > 0
        smask = np.zeros(A.size, dtype=bool)
        low = max(1, A.size // (2 ** int(rng.integers(2, 5))))
        smask[low:-low if low < A.size - 1 else -1] = True
        noise = (rng.standard_normal(A.size - 1) + 1j * rng.standard_normal(A.size - 1)) * strength * 0.1
        A[smask] += noise[1:len(noise)+1] if len(noise) >= A.size else noise
        q = np.fft.irfft(A, N)[:n]
    elif mode == 3:
        # V-shape
        x = np.linspace(0.0, 1.0, n)
        q = a * (1.0 - strength * (2.0 * x - 1.0) ** 2)
    elif mode == 4:
        # U-shape
        x = np.linspace(0.0, 1.0, n)
        q = a * (1.0 + strength * (2.0 * x - 1.0) ** 2)
    elif mode == 5:
        # Block
        nb = int(rng.integers(4, 16))
        per = max(1, n // nb)
        block = 1.0 + strength * 0.5 * (rng.random(nb) * 2 - 1)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = a * arr[:n]
    elif mode == 6:
        # Smooth Gaussian noise
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    else:

    return normalize_a(np.clip(q, 0.0, None))


def generate_inits(n, rng, parent=None, cap=20):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.01
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 - alpha * (2.0 * x - 1.0) ** 2)
    for alpha in (0.3, 0.5, 0.8, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 4):
        for alpha in (0.3, 0.6, 1.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    for f in (1, 2, 3):
        for alpha in (0.5, 1.0, 1.5):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    for rate in (1.0, 2.0, 3.0, 5.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    for p in (0.5, 1.0, 2.0, 3.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.01)
    # Random smooth
    Nf = 1 << (n - 1).bit_length()
    for _ in range(3):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        add(1.0 + 0.8 * noise)

        parent = None

    if parent is not None:
        print("Parent loaded: n={}, sum={:.4f}".format(len(parent), float(np.sum(parent))))

    global_best_sc = np.inf
    global_best_seq = None

    # Determine lengths to try
    if parent is not None:
        n_parent = int(parent.size)
        n_values = [n_parent]
        # Add nearby and different lengths for diversity
        for n_try in [4096, 6144, 8192, 10240, 12288, 16384, 20480]:
            if n_try not in n_values:
                n_values.append(n_try)
        # Sort: parent first, then by closeness to parent
        n_values.sort(key=lambda x: 0 if x == n_parent else abs(x - n_parent))
    else:
        n_values = [4096, 8192, 10240, 16384]

    # Allocate time: primary (parent) gets 55%, others split remaining 35%, save 10% for final polish
    primary_time = total_budget * 0.55
    secondary_time_each = total_budget * 0.35 / max(1, len(n_values) - 1) if len(n_values) > 1 else 0
    final_time = total_budget * 0.10

    # Primary optimization (parent length or first choice)
    n_primary = n_values[0]
    if parent is not None:
        a0 = normalize_a(resample_sequence(parent, n_primary))
    else:
        inits = generate_inits(n_primary, rng, None, cap=15)
        best_init_sc = np.inf
        a0 = inits[0]
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a0 = init
        print("Best init (n={}): {:.9f}".format(n_primary, best_init_sc))

    t_primary_end = t_start + primary_time
    print("=== Primary: n={} (budget {:.0f}s) ===".format(n_primary, primary_time))
    sc, a_best = optimize_one_length(a0, t_primary_end, verbose=True)
    if sc < global_best_sc:
        global_best_sc = sc
        global_best_seq = a_best.copy()
    print("Primary result: {:.9f}".format(global_best_sc))

    # Secondary lengths
    for i, n_sec in enumerate(n_values[1:], 1):
        rem_for_sec = deadline - final_time - time.monotonic()
        if rem_for_sec < 120:
            break
        # Cap time per secondary
        sec_budget = min(rem_for_sec * 0.4, secondary_time_each)
        if sec_budget < 60:
            break
        t_sec_end = time.monotonic() + sec_budget

        if parent is not None:
            a_sec = normalize_a(resample_sequence(parent, n_sec))
        else:
            a_sec = np.ones(n_sec) / n_sec
            # Add some structure
            x = np.linspace(0, 1, n_sec)
            a_sec = normalize_a(1.0 + 0.3 * np.cos(2 * np.pi * x))

        print("=== Secondary n={} (budget {:.0f}s) ===".format(n_sec, sec_budget))
        sc, a_sec_best = optimize_one_length(a_sec, t_sec_end, verbose=False)
        if sc < global_best_sc:
            global_best_sc = sc
            global_best_seq = a_sec_best.copy()
            print("  New best from n={}: {:.9f}".format(n_sec, global_best_sc))
        else:
            print("  n={} best: {:.9f}".format(n_sec, sc))

    # Final polish on global best
    if global_best_seq is not None and time.monotonic() < deadline - 30:
        n_final = len(global_best_seq)
        a_final = normalize_a(global_best_seq)

        # A few perturb + quick optimize
        t_polish_end = deadline - 10
        n_attempts = min(4, int((t_polish_end - time.monotonic()) / (total_budget * 0.04)))

        for att in range(max(1, n_attempts)):
            if time.monotonic() > t_polish_end - 60:
                break
            p = perturb(a_final, rng, strength=float(rng.uniform(0.03, 0.12)))
            # Quick LSE + AS
            t_quick_end = time.monotonic() + min(120, (t_polish_end - time.monotonic()) * 0.7)
            sc, a_q = lse_continuation(p, t_quick_end, betas=[2000, 5000, 10000, 20000])
            if sc < global_best_sc:
                global_best_sc = sc
                global_best_seq = a_q.copy()
                a_final = a_q.copy()
                print("  Perturb+LSE {:.9f}".format(global_best_sc))
            # Quick active-set
            t_as_end = min(t_polish_end - 10, time.monotonic() + 60)
            if time.monotonic() < t_as_end:
                sc2, a_as = active_set_loop(a_q, t_as_end, verbose=False)
                if sc2 < global_best_sc:
                    global_best_sc = sc2
                    global_best_seq = a_as.copy()
                    a_final = a_as.copy()
                    print("  +AS {:.9f}".format(global_best_sc))

        # Final fine polish
        if time.monotonic() < t_polish_end - 5:
            sc, a_p = fine_polish(a_final, t_polish_end)
            if sc < global_best_sc:
                global_best_sc = sc
                global_best_seq = a_p.copy()

    # Build output
    if global_best_seq is None:
        # Fallback: uniform
        out = [0.5] * 128
    else:
        out_arr = np.asarray(global_best_seq, dtype=np.float64).ravel()
        out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)
        S = float(out_arr.sum())
        if S < 0.01:
            out_arr = np.ones(1024) * 0.5
        else:
            out_arr = out_arr / S
            out_arr = np.clip(out_arr, 0.0, 1000.0)
        out = [float(x) for x in out_arr]

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = global_best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        global_best_sc, verified, len(out), time.monotonic() - t_start))
    return out
