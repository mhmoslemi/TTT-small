def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, b):
    """cross_corr(a,b)[i] = sum_k a[k] * b[k+i] (valid range)."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = int(a.size)
    mb = int(b.size)
    if n == 0 or mb == 0:
        return np.empty(n, dtype=np.float64)
    M = n + mb - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    B = np.fft.rfft(b, N)
    full = np.fft.irfft(A * B, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(max(1, a.size), target / max(1, a.size), dtype=np.float64)
    return a * (target / S)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, a)


def lse_objective_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    c = fast_self_conv(a)
    d = c / (S * S)
    dmax = float(np.max(d))
    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = cross_corr(a, w)
    score = 2.0 * n * L
    grad = (4.0 * n / (S * S)) * h - (4.0 * LDS / S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lse_lbfgs(a0, beta, deadline, maxfun=300):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": int(maxiter),
            "maxfun": int(maxfun), "maxls": 30}

    def fun(x):
        return lse_objective_grad(x, beta)

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, args=(float(beta),),
                       bounds=bounds, options=opts)
    except Exception:
        return exact_score(a), a
    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def smooth_obj_and_grad(a, w):
    """Compute smooth objective sum_k w_k * c_k / S^2 and its gradient for fixed active weights w."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n, dtype=np.float64)
    c = fast_self_conv(a)
    f = float(np.dot(w, c)) / (S * S)
    h = cross_corr(a, w)
    LDS = float(np.dot(w, c))
    grad = (2.0 * n * h) / (S * S) - (2.0 * LDS / S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return f, grad


def active_set_line_search(a, w, max_ls=40):
    """Projective gradient descent with backtracking line search on smooth subproblem."""
    a = normalize_a(a)
    S = float(a.sum())
    f0, g0 = smooth_obj_and_grad(a, w)
    if not np.isfinite(f0):
        return a, f0
    gnorm = float(np.max(np.abs(g0)))
    if gnorm < 1e-14:
        return a, f0
    g0 = g0 / gnorm
    step = 1.0
    best_a = a.copy()
    best_f = f0
    for _ in range(max_ls):
        q = normalize_a(np.maximum(a - step * g0, 0.0))
        fq, _ = smooth_obj_and_grad(q, w)
        if np.isfinite(fq) and fq < best_f - 1e-14:
            best_a = q
            best_f = fq
            if _ < 30:
                continue
            break
        step *= 0.5
        if step < 1e-12:
            break
    return best_a, best_f


def eigenvector_iteration_ls(a_start, k_active, max_iters=150, deadline=None,
                             blend=(0.5,)):
    """Eigenvector iteration with line search on smooth subproblem."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        c = fast_self_conv(a)
        S = float(a.sum())
        d = c / (S * S)
        dmax = float(np.max(d))
        if dmax <= 1e-15:
            break

        idx_sorted = np.argsort(d)[::-1]
        active_idx = idx_sorted[:k_active]
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active_idx] = d[active_idx]
        wsum = float(w.sum())
        if wsum < 1e-15:
            w[active_idx] = 1.0 / k_active
        else:
            w /= wsum

        a_new, _ = active_set_line_search(a, w, max_ls=30)
        if not np.allclose(a_new, a, atol=1e-10):
            sc = exact_score(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
        else:
            # Try blend
            c2 = fast_self_conv(a)
            S2 = float(a.sum())
            d2 = c2 / (S2 * S2)
            idx2 = np.argsort(d2)[::-1]
            act2 = idx2[:k_active]
            w2 = np.zeros(2 * n - 1, dtype=np.float64)
            w2[act2] = 1.0 / k_active
            h = cross_corr(a, w2)
            h = np.maximum(h, 0.0)
            improved = False
            for bl in blend:
                q = normalize_a(bl * h + (1.0 - bl) * a)
                sc = exact_score(q)

                    best_sc = sc
                    best_a = q.copy()

def make_initializations(n, rng, parent, n_init=8):
    inits = []

    def add(arr):
        if len(inits) >= n_init:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        inits.append(normalize_a(arr))

    if parent is not None and parent.size > 0:
        pr = resample(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n, dtype=np.float64))
    for alpha in (0.3, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0):
        add(1.0 + alpha * np.abs(2.0 * x - 1.0))
    for alpha in (0.5, 1.0, 2.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 4):
        add(1.0 + 0.5 * np.cos(2.0 * np.pi * f * x))
    for w in (0.1, 0.15, 0.25):
        add(np.exp(-(((x - 0.5) / w) ** 2)) + np.exp(-(((x - 0.5 + 0.01) / w) ** 2)))
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 5.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= noise.mean()
        add(1.0 + float(rng.uniform(0.3, 1.0)) * noise)
    return inits


def perturb(a_best, rng, n, strength):
    a = normalize_a(np.asarray(a_best, dtype=np.float64).ravel())
    if len(a) != n:
        a = resample(a, n)
        a = normalize_a(a)
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.2))
        q = (1.0 - t) * a + t * np.ones(n) / n
    elif mode == 2:
        q = a[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        c = fast_self_conv(a)
        d = c / (float(a.sum()) ** 2)
        dmax = float(np.max(d))
        if dmax > 1e-12:
            idx = np.argsort(d)[::-1][:int(rng.integers(1, 6))]
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0 / len(idx)
            h = cross_corr(a, w)
            hmax = float(np.max(h) + 1e-15)
            q = a * np.exp(-strength * 0.5 * h / hmax)
        else:
            q = a.copy()
    elif mode == 4:
        logp = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if 3 <= k <= n:
            c2 = k // 2
            sig = max(1.0, c2 / 2.0)
            kern = np.exp(-((np.arange(k) - c2) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode="same")
        noise -= noise.mean()
        q = np.exp(logp + strength * 0.2 * noise)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        q = a * (1.0 + strength * 0.3 * np.abs(2.0 * x - 1.0))
    elif mode == 6:
        k = int(rng.integers(5, max(7, n // 200)))
        if k % 2 == 0:
            k += 1
        k = min(k, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c2 = k // 2
        sig = max(1.0, c2 / 2.0)
        kern = np.exp(-((np.arange(k) - c2) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(a, kern, mode="same")
    else:
        idx = rng.integers(0, n, size=max(10, n // 100))
        q = a.copy()
        q[idx] *= np.exp(strength * 0.3 * rng.standard_normal(len(idx)))
    return normalize_a(np.clip(q, 0, None))

    print("Final polish, best={:.8f}".format(best_sc))
    if time.monotonic() < deadline - 3:
        sc, seq = subgrad_polish(best_seq, max_iters=15, deadline=deadline,
                                  k_list=(1, 2, 3, 5))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
    if time.monotonic() < deadline - 2:
        sc, seq = eigenvector_iteration_ls(best_seq, 3, max_iters=20,
                                           deadline=deadline, blend=(0.1, 0.2))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()

    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
