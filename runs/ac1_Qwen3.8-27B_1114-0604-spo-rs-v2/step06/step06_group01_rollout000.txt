"""
Multi-phase optimizer using improved LP alternating, spectral shaping,
and multi-peak subgradient descent to minimize 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize
import time

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum())
    p = exp_z / exp_sum
    log_sum = float(z.max()) - float(z_shift.max()) + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    # Gradient
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(RA) * P, M2_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000, gtol=1e-12):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': gtol, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res
