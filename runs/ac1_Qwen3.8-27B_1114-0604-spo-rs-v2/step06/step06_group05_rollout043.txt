import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Uses the top-k peaks weighted by softmax."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    shifted = scaled - max_scaled
    exp_s = np.exp(shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 1 << (3 * n - 2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(RA * P, M2)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B optimization with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = max(1.0 / n, 1e-6)
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        if np.any(~np.isfinite(g)):
            g = np.zeros(n)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    a_trimmed = a[:new_n * factor].reshape(new_n, factor).mean(axis=1)
    return a_trimmed


def upsample(a, factor):
    """Upsample by replication and smoothing."""
    new_n = len(a) * factor
    a_up = np.repeat(a, factor)
    # Apply a simple moving average
    kernel = np.ones(factor) / factor
    a_smooth = np.convolve(a_up, kernel, mode='same')[:new_n]
    return np.maximum(a_smooth, 0.0)


def peak_isolation_refine(a0, n_rounds=50, top_k=5, shrink=0.7):
    """
    Identify top peaks in convolution and reduce contributing elements.
    Uses a smarter approach: for each peak, compute a 'pressure' on each element
    proportional to its contribution, then apply a multiplicative update.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / S * S

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For each peak, compute contribution pressure
        pressure = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            # Normalize contribution to sum to 1 over this peak's contributors
            total = contrib.sum()
            if total > 1e-15:
                pressure[j_start:j_end + 1] += contrib / total

        # Apply multiplicative update: reduce elements with high pressure
        # pressure is in [0, top_k] range (sum of normalized contributions)
        pressure = pressure / max(pressure.max(), 1e-15)
        a *= np.exp(-shrink * pressure * 0.3)
        a = np.maximum(a, 0.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new
        S = S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multiplicative_descent(a0, n_iters=5000, lr=0.01, decay=0.99995):
    """
    Multiplicative weight gradient descent on the LSE objective.
    a_i <- a_i * exp(-lr * grad_i), which preserves non-negativity.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    beta = 5000
    step = lr

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)

        # Multiplicative update
        log_update = -step * grad
        # Clip to prevent overflow/underflow
        log_update = np.clip(log_update, -2.0, 2.0)
        a *= np.exp(log_update)
        a = np.maximum(a, 1e-10)
        S_new = a.sum()
        if S_new < 1e-10:
            break
        a = a / S_new

        step *= decay

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def parametric_init(n, rng):
    """
    Generate initializations using parametric families.
    Returns a list of candidate sequences.
    """
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Various smooth positive shapes
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(np.abs(np.sin(np.pi * x)) ** 0.5)
    inits.append(1.0 / (1.0 + np.exp(-5.0 * (2.0 * x - 1.0))))  # sigmoid
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(0.4 + 0.6 * (1.0 - 4.0 * (x - 0.5) ** 2))
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2) * 0.8 + 0.2)

    # Fourier-based shapes
    for trial in range(6):
        n_harm = rng.integers(2, 8)
        weights = rng.standard_normal(n_harm) * 0.3
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        sig = np.zeros(n)
        for i in range(n_harm):
            sig += weights[i] * np.cos(2.0 * np.pi * (i + 1) * x + phases[i])
        init = np.maximum(0.5 + sig, 0.01)
        inits.append(init)

    # Beta-like distributions
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        init = x ** alpha
        inits.append(init)
        init2 = (1.0 - x) ** alpha
        inits.append(init2)
        # Symmetric beta
        init3 = x ** alpha * (1.0 - x) ** alpha
        init3 = np.maximum(init3, 0.01)
        inits.append(init3)

    # W-shaped and M-shaped
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(6.0 * np.pi * x), 0.01))

    return inits


def gs_flat_convolution(a0, n_iters=300, alpha=0.4):
    """
    GS iteration targeting flat convolution.
    The target is c_k = S^2/(2n-1) for all k.
    In FFT domain: |A_m|^2 should equal FFT(target_c)_m.
    Since target_c is a rectangle, its FFT is a sinc.
    """
    n = len(a0)
    M = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0

    # Target: flat convolution at level S^2 / (2n-1)
    target_c = np.zeros(M)
    target_c[:2 * n - 1] = S * S / (2 * n - 1)
    T = np.fft.rfft(target_c)
    T_mag = np.maximum(np.abs(T), 1e-20)
    T_phase = np.angle(T)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        A_mag = np.maximum(np.abs(A), 1e-20)
        A_phase = np.angle(A)

        # Blend magnitude toward target
        new_mag = A_mag ** (1.0 - alpha) * T_mag ** (alpha * 0.5)
        new_phase = (1.0 - alpha) * A_phase + alpha * T_phase * 0.5

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_exact(a0, n_iters=5000, n_peaks=10, step_init=0.02):
    """Subgradient method on the exact max-convolution objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient: average over top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        weights = c[top_idx] - c_max + c_max  # just use equal weights for top peaks
        w_sum = len(top_idx)

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1] / w_sum

        # Adjust for the gradient of 1/S^2 term
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-stage: parametric exploration -> coarse-to-fine LSE-LBFGS ->
    peak isolation -> multiplicative descent -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 60.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    n = 6962
    if parent is not None:
        n = len(parent)

    best_score = np.inf
    best_a = np.ones(n) * 0.5

    def update_best(sc, a):
        nonlocal best_score, best_a
        a = np.clip(a.copy(), 0.0, 1000.0)
        if a.sum() < 0.01:
            return False
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f} (n={n})")

    print(f"\n{'='*60}")
    print(f"Starting optimization: n={n}, initial={best_score:.8f}")
    print(f"Deadline in {total_budget - 60:.0f}s")
    print(f"{'='*60}")

    # ==================================================================
    # Phase 1: Quick parametric exploration (5% of time)
    # ==================================================================
    phase1_end = t_start + 0.05 * (total_budget - 60.0)
    print(f"\n--- Phase 1: Parametric exploration ---")

    inits = parametric_init(n, rng)
    # Quick GS on each to see which shape is promising
    quick_results = []
    for i, a0 in enumerate(inits[:12]):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        a1 = gs_flat_convolution(a0, n_iters=100, alpha=0.3)
        sc = compute_score(a1)
        quick_results.append((sc, a1))

    quick_results.sort(key=lambda x: x[0])
    if quick_results:
        sc_best_param, a_best_param = quick_results[0]
        update_best(sc_best_param, a_best_param)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Coarse-to-fine optimization (20% of time)
    # ==================================================================
    phase2_end = t_start + 0.25 * (total_budget - 60.0)
    print(f"\n--- Phase 2: Coarse-to-fine LSE-LBFGS ---")

    # Work at 1/4 resolution first
    coarse_n = n // 4
    a_coarse = downsample(best_a, 4)
    sc_coarse = compute_score(a_coarse)
    print(f"  Coarse (n={coarse_n}): score={sc_coarse:.8f}")

    # Optimize coarse with LSE
    for beta in [500, 2000, 10000]:
        if time.monotonic() > phase2_end:
            break
        a_coarse = run_lbfgs(a_coarse, lse_score_grad, (beta,),
                             maxiter=3000, maxfun=20000)
        sc = compute_score(a_coarse)
        print(f"  Coarse LBFGS beta={beta}: score={sc:.8f}")

    # Upsample and use as initialization
    a_from_coarse = upsample(a_coarse, 4)
    sc_up = compute_score(a_from_coarse)
    print(f"  Upsampled: score={sc_up:.8f}")
    update_best(sc_up, a_from_coarse)

    # Also try 1/2 resolution
    coarse_n2 = n // 2
    a_coarse2 = downsample(best_a, 2)
    for beta in [2000, 10000]:
        if time.monotonic() > phase2_end:
            break
        a_coarse2 = run_lbfgs(a_coarse2, lse_score_grad, (beta,),
                              maxiter=3000, maxfun=20000)
    a_from_coarse2 = upsample(a_coarse2, 2)
    sc_up2 = compute_score(a_from_coarse2)
    print(f"  Upsampled (2x): score={sc_up2:.8f}")
    update_best(sc_up2, a_from_coarse2)

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Full-resolution LSE-LBFGS with beta annealing (35% of time)
    # ==================================================================
    phase3_end = t_start + 0.60 * (total_budget - 60.0)
    print(f"\n--- Phase 3: Full-res LSE-LBFGS beta annealing ---")

    # Start from current best
    beta_schedule = [500, 2000, 10000, 50000, 200000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,),
                          maxiter=5000, maxfun=30000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Try top-k focused optimization
    if time.monotonic() < phase3_end:
        for k in [5, 10, 20]:
            if time.monotonic() > phase3_end:
                break
            t0 = time.monotonic()
            a_opt = run_lbfgs(best_a, topk_lse_score_grad, (50000, k),
                              maxiter=3000, maxfun=20000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)
            print(f"  TopK-LBFGS k={k}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak isolation + multiplicative descent (20% of time)
    # ==================================================================
    phase4_end = t_start + 0.80 * (total_budget - 60.0)
    print(f"\n--- Phase 4: Peak isolation + multiplicative ---")

    # Peak isolation refinement
    a_pi = peak_isolation_refine(best_a, n_rounds=80, top_k=5, shrink=0.8)
    sc = compute_score(a_pi)
    update_best(sc, a_pi)

    # Then LSE to recover from any damage
    if time.monotonic() < phase4_end:
        a_opt = run_lbfgs(best_a, lse_score_grad, (100000,),
                          maxiter=3000, maxfun=20000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

    # Multiplicative descent
    if time.monotonic() < phase4_end:
        a_md = multiplicative_descent(best_a, n_iters=3000, lr=0.005)
        sc = compute_score(a_md)
        update_best(sc, a_md)

    # Another round of peak isolation with different parameters
    if time.monotonic() < phase4_end:
        a_pi2 = peak_isolation_refine(best_a, n_rounds=50, top_k=10, shrink=0.5)
        sc = compute_score(a_pi2)
        update_best(sc, a_pi2)

    # LSE cleanup
    if time.monotonic() < phase4_end:
        a_opt = run_lbfgs(best_a, lse_score_grad, (50000,),
                          maxiter=3000, maxfun=15000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final intensive polish (15% of time)
    # ==================================================================
    print(f"\n--- Phase 5: Final polish ---")

    polish_rounds = 8
    for pr in range(polish_rounds):
        if time.monotonic() > deadline:
            break

        # Subgradient on exact max
        a_sg = subgradient_exact(best_a, n_iters=2000, n_peaks=5, step_init=0.01)
        sc = compute_score(a_sg)
        update_best(sc, a_sg)

        # Very high beta LSE
        if time.monotonic() > deadline:
            break
        beta_high = 200000 + 200000 * pr
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta_high,),
                          maxiter=3000, maxfun=15000)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)

        # Quick peak isolation
        if time.monotonic() > deadline:
            break
        a_pi = peak_isolation_refine(best_a, n_rounds=30, top_k=3, shrink=1.0)
        sc = compute_score(a_pi)
        update_best(sc, a_pi)

        print(f"  Polish {pr}: best = {best_score:.8f}")

    # ==================================================================
    # Phase 6: Try different sequence lengths (remaining time)
    # ==================================================================
    if time.monotonic() < deadline:
        print(f"\n--- Phase 6: Alternative lengths ---")
        for alt_n in [4096, 8192, 16384]:
            if time.monotonic() > deadline:
                break
            if alt_n == n:
                continue
            # Interpolate current best to new length
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, alt_n)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            a_alt = a_alt / a_alt.sum()

            # Quick LSE optimization
            a_alt = run_lbfgs(a_alt, lse_score_grad, (50000,),
                              maxiter=2000, maxfun=10000)
            sc = compute_score(a_alt)
            print(f"  n={alt_n}: score={sc:.8f}")
            update_best(sc, a_alt)

    # ==================================================================
    # Final: one last LSE with very high beta if time allows
    # ==================================================================
    if time.monotonic() < deadline - 10:
        print(f"\n--- Final LSE polish ---")
        a_final = run_lbfgs(best_a, lse_score_grad, (500000,),
                            maxiter=5000, maxfun=20000)
        sc = compute_score(a_final)
        update_best(sc, a_final)

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
