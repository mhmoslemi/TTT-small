    n = len(a)

    S = float(a.sum())

        return np.ones(n) / n
    a = a / S

    best_a = a.copy()

        omega = omega_start + (omega_end - omega_start) * progress
        
        A = np.fft.rfft(a, M_fft)

        if S_new < 1e-12:

        a_new = a_new / S_new
        
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc

        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-22, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100,
                 'pprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_overrelaxed(a, n_iters=1500, omega_start=1.0, omega_end=1.6, target_flat=True):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    plateaus = []

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Adaptive omega: start gentle, get more aggressive, then settle
        if progress < 0.3:
            omega = omega_start + (1.4 - omega_start) * (progress / 0.3)
        elif progress < 0.8:
            omega = 1.4 + (omega_end - 1.4) * ((progress - 0.3) / 0.5)
        else:
            omega = omega_end - 0.3 * ((progress - 0.8) / 0.2) ** 2

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        if target_flat:
            P_target = P_mean * np.ones_like(P)
        else:
            # Slightly peaked at DC for non-negativity help
            P_target = P_mean * (1.0 + 0.05 * np.exp(-np.arange(len(P)) / (len(P) * 0.5)))
            P_target = P_target / P_target.sum() * P_mean * len(P)

        P_new = (1.0 - omega) * P + omega * P_target
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / (P + 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200 and it > 200:
                plateaus.append(it)
                if len(plateaus) > 5:
                    break
    return best_a


def fienup_peak_targeted(a, n_iters=1000, n_top=5):
    """Fienup that specifically targets the top n_top spectral peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        omega = 1.2 + 0.4 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        c = fast_conv(a)
        c_max = float(c.max())

        # Target: reduce top peaks in the convolution -> shape target spectrum
        # The peaks in c correspond to certain spectral regions.
        # Simpler: use a soft threshold on P
        P_sorted = np.sort(P)[::-1]
        threshold_frac = 0.7 + 0.2 * progress  # Tighten over time
        threshold = P_sorted[max(0, min(n_top, len(P_sorted) - 1))] * threshold_frac

        P_target = P.copy()
        above = P > threshold
        if above.any():
            P_target[above] = threshold + 0.3 * (P[above] - threshold)

        P_new = (1.0 - omega) * P + omega * P_target
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / (P + 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def peak_pair_cd(a, n_sweeps=50, n_top_peaks=8, candidates_per_peak=20, eta=0.01):
    """
    Coordinate descent: for each top peak, find the dominant contributing pairs
    and try to reduce them by shifting mass to quieter positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find active peaks
        c_max = float(c.max())
        active = np.where(c >= 0.95 * c_max)[0]
        # Also include the absolute top
        top_peaks = np.argsort(c)[::-1][:n_top_peaks]
        all_peaks = np.unique(np.concatenate([active, top_peaks]))

        improved_this_sweep = False
        for k in all_peaks:
            k = int(k)
            if k < 0 or k >= 2 * n - 1:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            total = float(contribs.sum())
            if total < 1e-15:
                continue

            # Get top contributing indices
            top_idx = np.argsort(contribs)[::-1][:candidates_per_peak]
            for idx in top_idx:
                j = j_start + int(idx)
                j2 = k - j
                if j2 < 0 or j2 >= n:
                    continue
                if j == j2:
                    continue
                a_j = a[j]
                a_j2 = a[j2]
                if a_j < 1e-10 or a_j2 < 1e-10:
                    continue

                # Try reducing a_j (and/or a_j2) and adding to quiet positions
                # Find quiet positions (low contribution to all top peaks)
                # Simple heuristic: positions with small a value and far from peak-contributing indices
                quiet_candidates = np.argsort(a)[:max(5, n // 100)]
                # Filter out positions that are also contributing to this peak
                quiet_candidates = [q for q in quiet_candidates
                                    if not (max(0, k - n + 1) <= q <= min(k, n - 1) and a[q] > 1e-10)]
                if not quiet_candidates:
                    continue
                quiet_candidates = quiet_candidates[:5]

                # Try different reduction fractions
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    delta = a_j * frac
                    if delta < 1e-12:
                        continue
                    for q in quiet_candidates:
                        a[j] = a_j - delta
                        a[q] = a[q] + delta
                        S_tmp = a.sum()
                        if S_tmp < 1e-12:
                            a[j] = a_j
                            a[q] = a[q] - delta
                            continue
                        # Quick score check
                        c_new = fast_conv(a)
                        sc_new = 2.0 * n * float(c_new.max()) / (S_tmp * S_tmp)
                        if sc_new < best_sc - 1e-12:
                            improved_this_sweep = True
                        a[j] = a_j
                        a[q] = a[q] - delta

        if not improved_this_sweep and sweep > 20:
            break

    return best_a


def spectral_cg(a, n_iters=2000, M_fft=None):
    """
    Spectral Conjugate Gradient: optimize in frequency domain with conjugate
    gradient method, projecting back to time domain for non-negativity.
    """
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    # Work with the phase of A (magnitude is constrained for flat spectrum)
    A = np.fft.rfft(a, M_fft)
    phase = np.angle(A)
    # Target magnitude: sqrt of mean power
    target_mag = np.sqrt(float(np.mean(np.abs(A)**2)))

    d_old = np.zeros(phase.shape)
    g_old = None

    for it in range(n_iters):
        # Current estimate: combine target magnitude with current phase
        A_current = target_mag * np.exp(1j * phase)
        a_current = np.fft.irfft(A_current, M_fft)[:n]
        a_current = np.maximum(a_current, 0.0)
        S_c = a_current.sum()
        if S_c < 1e-12:
            break
        a_current = a_current / S_c
        sc = compute_score(a_current)
        if sc < best_sc:
            best_sc = sc
            best_a = a_current.copy()

        # Compute gradient in phase domain
        c = fast_conv(a_current)
        c_max = c.max()
        beta = 200.0
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -500, 500)
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + np.log(exp_s.sum()) / beta

        # Gradient w.r.t. a
        a_rev = a_current[::-1]
        PA = np.fft.rfft(p, M_fft)
        AA_rev = np.fft.rfft(a_rev, M_fft)
        h = np.fft.irfft(AA_rev * PA, M_fft)[:2*n-1][n-1:2*n-1]
        S2 = S_c * S_c
        S3 = S2 * S_c
        grad_a = 2.0 * n * (h / S2 - 2.0 * L / S3)

        # Map gradient to phase domain: grad_phase_j = Im(conj(A_j) * grad_FFT_j)
        grad_fft = np.fft.rfft(np.concatenate([grad_a, np.zeros(M_fft - n)]), M_fft)
        grad_phase = np.imag(np.conj(A) * grad_fft)

        # Conjugate gradient step
        if g_old is not None:
            alpha = float(np.dot(grad_phase, grad_phase) / (np.dot(g_old, g_old) + 1e-30))
            d_new = -grad_phase + alpha * d_old
        else:
            d_new = -grad_phase
        beta_cg = float(np.dot(grad_phase, grad_phase) / (np.dot(g_old, g_old) + 1e-30)) if g_old is not None else 0.0
        d_old = d_new
        g_old = grad_phase.copy()

        # Line search
        best_step = 0.1
        best_sc_trial = sc
        for step in [0.01, 0.05, 0.1, 0.2, 0.5]:
            phase_trial = phase + step * d_new
            A_trial = target_mag * np.exp(1j * phase_trial)
            a_trial = np.fft.irfft(A_trial, M_fft)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            S_t = a_trial.sum()
            if S_t < 1e-12:
                continue
            a_trial = a_trial / S_t
            sc_trial = compute_score(a_trial)
            if sc_trial < best_sc_trial:
                best_sc_trial = sc_trial
                best_step = step
        phase = phase + best_step * d_new

    return best_a


def adaptive_lbfgs(a, deadline, beta_schedule=None):
    """Long adaptive L-BFGS with increasing beta."""
    if beta_schedule is None:
        beta_schedule = [500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000]

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        # Allocate iterations proportional to remaining time
        maxiter = int(min(15000, max(1000, remaining * 12)))
        maxfun = maxiter * 30
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    L-BFGS beta={beta}: {sc:.10f} ({dt:.1f}s)")
        else:
            print(f"    L-BFGS beta={beta}: {sc:.10f} (no improve, {dt:.1f}s)")

    return best_a, best_sc


def fine_coord_descent(a, n_iters=5000, n_top=5):
    """
    Very fine-grained coordinate descent: for each of the top n_top peaks,
    do a ternary search on the single most important coordinate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        c = fast_conv(a)
        c_max = float(c.max())
        # Top peaks
        top_k = np.argsort(c)[::-1][:n_top]

        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            contribs = a[js] * a[k - js]
            # Top 3 contributing indices
            top_idx = np.argsort(contribs)[::-1][:3]

            for idx in top_idx:
                j = int(j_start + idx)
                a_j = a[j]
                if a_j < 1e-10:
                    continue
                # Ternary search on multiplier in [0.85, 1.15]
                lo, hi = 0.85, 1.15
                for _ in range(5):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    a[j] = a_j * m1
                    c1 = fast_conv(a)
                    s1 = 2.0 * n * float(c1.max()) / (a.sum() ** 2)
                    a[j] = a_j * m2
                    c2 = fast_conv(a)
                    s2 = 2.0 * n * float(c2.max()) / (a.sum() ** 2)
                    if s1 < s2:
                        hi = m2
                    else:
                        lo = m1
                best_mult = (lo + hi) / 2
                a[j] = a_j * best_mult
                S_new = a.sum()
                if S_new > 1e-12:
                    a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def stochastic_refine(a, rng, n_restarts=15, deadline=None):
    """Small perturbations + quick L-BFGS to escape local minima."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(n_restarts):
        if deadline and time.monotonic() > deadline:
            break

        strategy = r % 5
        if strategy == 0:
            # Gaussian noise
            noise_scale = 0.01 + 0.02 * (r / n_restarts)
            a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        elif strategy == 1:
            # Targeted noise at peak-contributing indices
            c = fast_conv(a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            a_pert = a.copy()
            a_pert[j_start:j_end+1] *= (1.0 + 0.03 * rng.standard_normal(j_end - j_start + 1))
        elif strategy == 2:
            # Low-frequency modulation
            x = np.arange(n) / n
            freq = rng.integers(1, max(2, n // 200))
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a * (1.0 + 0.03 * np.cos(2 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Block swap: swap two small blocks
            block_size = max(2, n // 100)
            b1 = rng.integers(0, n - 2 * block_size)
            b2 = b1 + block_size + rng.integers(0, n - 2 * block_size - block_size)
            a_pert = a.copy()
            temp = a_pert[b1:b1+block_size].copy()
            a_pert[b1:b1+block_size] = a_pert[b2:b2+block_size]
            a_pert[b2:b2+block_size] = temp
        else:
            # Smooth perturbation
            kernel_size = max(3, n // 80)
            kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2) ** 2 / (kernel_size**2 / 6.0))
            kernel /= kernel.sum()
            smooth_noise = np.convolve(rng.standard_normal(n), kernel, mode='same')
            a_pert = a * (1.0 + 0.02 * smooth_noise)

        a_pert = np.maximum(a_pert, 0.0)
        S_p = a_pert.sum()
        if S_p < 1e-12:
            continue
        a_pert = a_pert / S_p

        # Quick L-BFGS
        sc, a_opt = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=60000)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Stochastic restart {r}: {sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization combining over-relaxed Fienup,
    adaptive LSE-LBFGS, peak-targeted spectral CG, fine coordinate descent,
    and stochastic refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 16:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None and len(parent) >= 16:
        n = len(parent)
        # Ensure n is reasonable (power of 2 for FFT efficiency)
        if n > 8192:
            # Downsample to 4096
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, 4096)
            parent = np.interp(x_new, x_old, parent)
            parent = np.maximum(parent, 0.0)
            parent = parent / parent.sum()
            n = 4096
        a_init = parent / parent.sum()
        sc_init = compute_score(a_init)
        print(f"Parent: n={n}, score={sc_init:.10f}")
        update(sc_init, a_init, "parent")
    else:
        n = 2048
        a_init = np.ones(n) / n
        sc_init = compute_score(a_init)
        print(f"No parent: uniform n={n}, score={sc_init:.10f}")
        update(sc_init, a_init, "uniform-init")

    print(f"Initial best: {best_score:.10f}")

    # ==================================================================
    # Phase 1: Fienup variants (12% of time)
    # ==================================================================
    p1_end = t_start + 0.12 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (until t={p1_end - t_start:.0f}s) ===")

    if time.monotonic() < p1_end:
        a_f1 = fienup_overrelaxed(best_a, n_iters=1500, omega_start=1.0, omega_end=1.6)
        sc = compute_score(a_f1)
        update(sc, a_f1, "FIENUP-OR")

    if time.monotonic() < p1_end:
        a_f2 = fienup_peak_targeted(best_a, n_iters=800, n_top=5)
        sc = compute_score(a_f2)
        update(sc, a_f2, "FIENUP-PT")

    if time.monotonic() < p1_end - 30:
        a_f3 = fienup_overrelaxed(best_a, n_iters=1000, omega_start=1.1, omega_end=1.4, target_flat=False)
        sc = compute_score(a_f3)
        update(sc, a_f3, "FIENUP-shape")

    # ==================================================================
    # Phase 2: Spectral Conjugate Gradient (8% of time)
    # ==================================================================
    p2_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Spectral CG (until t={p2_end - t_start:.0f}s) ===")

    if time.monotonic() < p2_end:
        a_cgr = spectral_cg(best_a, n_iters=1500)
        sc = compute_score(a_cgr)
        update(sc, a_cgr, "SPEC-CG")

    # ==================================================================
    # Phase 3: Adaptive L-BFGS (40% of time) - main workhorse
    # ==================================================================
    p3_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Adaptive L-BFGS (until t={p3_end - t_start:.0f}s) ===")

    # Custom beta schedule: start moderate, go very high
    beta_sched = [500, 1000, 2000, 5000, 10000, 20000, 50000, 100000, 200000, 500000, 1000000]
    a_lb, sc_lb = adaptive_lbfgs(best_a, p3_end, beta_schedule=beta_sched)
    update(sc_lb, a_lb, "ADAPTIVE-LBFGS")

    # ==================================================================
    # Phase 4: Fine coordinate descent (15% of time)
    # ==================================================================
    p4_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Fine Coord Descent (until t={p4_end - t_start:.0f}s) ===")

    if time.monotonic() < p4_end:
        remaining = p4_end - time.monotonic()
        n_iters_cd = int(min(8000, max(2000, remaining * 3)))
        a_cd = fine_coord_descent(best_a, n_iters=n_iters_cd, n_top=5)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINE-CD")

    # L-BFGS polish after CD
    if time.monotonic() < p4_end - 10:
        sc2, a2 = run_lbfgs(best_a, 100000, maxiter=8000, maxfun=160000)
        update(sc2, a2, "CD+POLISH")

    # ==================================================================
    # Phase 5: Stochastic refinement (10% of time)
    # ==================================================================
    p5_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 5: Stochastic (until t={p5_end - t_start:.0f}s) ===")

    a_sr, sc_sr = stochastic_refine(best_a, rng, n_restarts=15, deadline=p5_end)
    update(sc_sr, a_sr, "STOCHASTIC")

    # ==================================================================
    # Phase 6: Final high-precision polish (7% of time)
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        maxiter = int(min(15000, max(2000, remaining * 10)))
        sc, a_fin = run_lbfgs(best_a, 500000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_fin, "FINAL-LBFGS-500k")

    # Even higher beta
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        maxiter = int(min(10000, max(1000, remaining * 8)))
        sc, a_fin = run_lbfgs(best_a, 2000000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_fin, "FINAL-LBFGS-2M")

    # Final fine CD
    if time.monotonic() < deadline - 15:
        a_cd2 = fine_coord_descent(best_a, n_iters=3000, n_top=3)
        sc = compute_score(a_cd2)
        update(sc, a_cd2, "FINAL-CD")

    # Final L-BFGS
    if time.monotonic() < deadline - 5:
        sc, a_fin = run_lbfgs(best_a, 5000000, maxiter=3000, maxfun=60000)
        update(sc, a_fin, "FINAL-LBFGS-5M")

    # Ultra-fine CD
    if time.monotonic() < deadline - 2:
        a_cd3 = fine_coord_descent(best_a, n_iters=2000, n_top=3)
        sc = compute_score(a_cd3)
        update(sc, a_cd3, "ULTRA-CD")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
