"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE annealing with L-BFGS, global flattening, multi-scale optimization,
and adaptive diversification. Key improvements: aggressive beta schedule,
proper line search, multi-length exploration, and "peak deflation" steps.

import numpy as np
import time
import math


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def cross_corr(a, w):
    """Cross-correlation: gradient of (a*w) w.r.t. a, restricted to valid positions."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, 1.0 / max(1, a.size))
    return a / S


def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed objective and gradient (assuming sum(a)=1)."""
    n = len(a)
    c = fast_self_conv(a)
    d = float(n) * c  # Since sum(a)=1, score = 2n*max(c), use d = n*c for LSE
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    z = np.clip(z, -500.0, 500.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    # Gradient: d(LSE)/da = sum_k w_k * d(c_k)/da * n
    # d(c_k)/da = cross_corr(a, e_k) which summed with weights gives cross_corr(a, w)
    h = cross_corr(a, w)
    grad = n * h
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(2.0 * L), grad


def run_lbfgs_lse(a, beta, deadline, maxfun=300):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    from scipy.optimize import minimize
    a = normalize_a(a)
    n = len(a)
    if time.monotonic() > deadline - 5.0:
        return exact_score(a), a

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = lse_objective_and_grad(xn, beta)
        # Chain rule: d/dx [f(x/S)] = (1/S) * f'(x/S) - f'(x/S) * x / S^2
        # Actually: f is homogeneous of degree 0, so gradient w.r.t. x = grad/S - x*sum(grad)/S^2
        # For constrained optimization with normalization, just use grad directly
        return val, grad

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(20, maxfun // 3),
        "maxfun": int(maxfun),
        "maxls": 30,
    }
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return float(sc), a_out


def peak_deflation(a, n_peaks=20, strength=0.1, deadline=None):
    """Apply a deflation step: reduce mass at positions contributing to peaks."""
    n = len(a)
    conv = fast_self_conv(a)
    dmax = float(np.max(conv))
    if dmax < 1e-15:
        return a.copy()
    # Soft active set: weight proportional to excess over mean
    cmean = float(np.mean(conv))
    excess = np.maximum(conv - cmean, 0.0)
    total_excess = float(np.sum(excess))
    if total_excess < 1e-15:
        return a.copy()
    w = excess / total_excess
    h = cross_corr(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    # Multiple step sizes
    best_a = a.copy()
    best_sc = exact_score(a)
    for eta in [strength * 0.25, strength * 0.5, strength, strength * 2, strength * 4]:
        log_a = np.log(a + 1e-300) - eta * h / hmax
        log_a -= float(np.max(log_a))
        new_a = normalize_a(np.exp(log_a))
        sc = exact_score(new_a)
        if sc < best_sc:
            best_sc = sc
            best_a = new_a
    return best_a


def exp_grad_step(a, conv, w, eta):
    """Single exponential gradient step with given weights."""
    h = cross_corr(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy()
    log_a = np.log(a + 1e-300) - eta * h / hmax
    log_a -= float(np.max(log_a))
    return np.exp(log_a)


def softmax_active_set(conv, temperature):
    """Compute soft active set weights using softmax."""
    dmax = float(np.max(conv))
    if dmax < 1e-15:
        return np.ones(conv.size) / conv.size
    z = (conv - dmax) / max(temperature, 1e-10)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return e / s


def resample_sequence(a, n_new):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n_new)) / float(max(1, n_new))
    if m == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, np.maximum(a, 0.0)).astype(np.float64)


def top_k_weights(conv, top_k=10, eps_frac=0.02):
    """Hard active set: uniform over top-k or positions within eps of max."""
    dmax = float(np.max(conv))
    if dmax <= 0:
        return np.ones(conv.size) / conv.size
    threshold = dmax * (1.0 - eps_frac)
    active = conv >= threshold
    min_count = max(1, top_k)
    if active.sum() < min_count:
        idx = np.argpartition(conv, -min_count)[-min_count:]
        active[idx] = True
    max_count = max(min_count, int(0.03 * conv.size))
    if active.sum() > max_count:
        idx = np.argpartition(conv, -max_count)[-max_count:]
        active = np.zeros(conv.size, dtype=bool)
        active[idx] = True
    w = active.astype(np.float64)
    s = w.sum()
    if s < 1e-15:
        return np.ones(conv.size) / conv.size
    return w / s


def active_set_optimize(a, deadline, verbose=True, tag=""):
    """Active-set exponential gradient with adaptive parameters and line search."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    top_k = 5
    eps_frac = 0.03
    eta = 0.3
    stagnant = 0
    max_stagnant = 40

    for phase in range(15):
        if time.monotonic() > deadline:
            break
        phase_dead = min(deadline, t0 + (deadline - t0) * (phase + 1) / 15.0 + 5.0)

        for it in range(300):
            if time.monotonic() > phase_dead - 2.0:
                break

            conv = fast_self_conv(a)
            sc = float(2.0 * n * np.max(conv))

            best_a_iter = a
            best_sc_iter = sc
            best_eta = eta

            # Hard active set
            w_hard = top_k_weights(conv, top_k=top_k, eps_frac=eps_frac)
            etas = [eta * 0.25, eta * 0.5, eta, eta * 1.5, eta * 2.0, eta * 3.0, eta * 5.0]
            for eta_try in etas:
                if eta_try < 0.001:
                    continue
                new_a = exp_grad_step(a, conv, w_hard, eta_try)
                new_a = normalize_a(new_a)
                new_sc = exact_score(new_a)
                if new_sc < best_sc_iter - 1e-12:
                    best_sc_iter = new_sc
                    best_a_iter = new_a
                    best_eta = eta_try

            # Soft active set (LSE-like)
            for temp in [0.01, 0.05, 0.1]:
                w_soft = softmax_active_set(conv, temp)
                for eta_try in [best_eta * 0.5, best_eta, best_eta * 2]:
                    new_a = exp_grad_step(a, conv, w_soft, eta_try)
                    new_a = normalize_a(new_a)
                    new_sc = exact_score(new_a)
                    if new_sc < best_sc_iter - 1e-12:
                        best_sc_iter = new_sc
                        best_a_iter = new_a

            # Peak deflation
            new_a = peak_deflation(a, n_peaks=20, strength=best_eta * 0.5)
            new_a = normalize_a(new_a)
            new_sc = exact_score(new_a)
            if new_sc < best_sc_iter - 1e-12:
                best_sc_iter = new_sc
                best_a_iter = new_a

                best_sc = best_sc_iter
                best_a = best_a_iter.copy()
                a = best_a_iter
                eta = best_eta
                stagnant = 0

        if stagnant >= max_stagnant:

            eps_frac = min(eps_frac * 1.4, 0.12)
            eta = max(0.05, eta * 0.7)
            stagnant = 0
        else:
            top_k = max(3, int(top_k * 0.97))

def lse_anneal(a, deadline, verbose=True, tag=""):
    """LSE optimization with beta annealing from low to high."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Beta schedule: start smooth, get sharper
    betas = [30, 50, 100, 200, 500, 1000, 3000, 8000, 20000, 50000, 100000]
    # For each beta, run L-BFGS with allotted time
    total_lse_time = deadline - time.monotonic()
    if total_lse_time < 30:
        return best_sc, best_a

    # Allocate more time to mid-range betas
    time_weights = [1, 1.5, 2, 3, 3, 2.5, 2, 1.5, 1, 0.8, 0.5]
    total_w = sum(time_weights)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 10:
            break
        rem = deadline - time.monotonic() - 5
        if rem < 15:
            break
        alloc = rem * (time_weights[i] / total_w)
        alloc = max(15, min(alloc, rem * 0.4))
        maxfun = max(50, int(alloc / 0.5))
        maxfun = min(maxfun, 3000)

        sc, a = run_lbfgs_lse(a, float(beta), time.monotonic() + alloc, maxfun)
        a = normalize_a(a)
        sc = exact_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose:
                print("  [{} lse beta={}] score={:.9f}".format(tag, beta, best_sc))
        total_w -= time_weights[i]

    return float(best_sc), best_a


def fine_polish(a, deadline, verbose=False, tag=""):
    """Systematic fine polish trying many parameter combinations with multi-step."""
    n = len(a)
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score(a)

    # Try different (top_k, eta, steps) combinations
    configs = []
    for tk in [1, 2, 3, 5, 8, 12, 20, 40, 80]:
        for eta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.8]:
            for steps in [1, 3, 10, 50]:
                configs.append((tk, eta, steps))

    # Sort by expected promise (smaller eta, moderate steps first)
    configs.sort(key=lambda x: x[1] * x[2])

    for tk, eta, steps in configs:
        if time.monotonic() > deadline - 5:
            break
        if steps * exact_score_cost(n) > (deadline - time.monotonic()) * 0.8:
            continue

        test_a = a.copy()
        improved = False
        for s in range(steps):
            if time.monotonic() > deadline - 5:
                break
            conv = fast_self_conv(test_a)
            w = top_k_weights(conv, top_k=tk, eps_frac=0.01)
            test_a = exp_grad_step(test_a, conv, w, eta)
            test_a = normalize_a(test_a)
            sc_test = exact_score(test_a)
            if sc_test >= best_sc + 0.01:
                break  # Getting worse, abort this config
        else:
            sc_test = exact_score(test_a)
            if sc_test < best_sc - 1e-12:
                best_sc = sc_test
                best_a = test_a
                a = test_a
                improved = True
                if verbose:
                    print("  [{} fp tk={} eta={:.3f} s={}] -> {:.9f}".format(
                        tag, tk, eta, steps, best_sc))

    return float(best_sc), best_a


def exact_score_cost(n):
    """Estimate cost of one exact score computation."""
    return 0.01 + n / 50000.0  # Rough estimate in seconds


def generate_diverse_inits(n, rng, parent=None, cap=15):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    add(np.ones(n))

    # V-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        add(1.0 + alpha * (2.0 * x - 1.0) ** 2)
    # U-shapes
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        add(1.0 - alpha * (1.0 - (2.0 * x - 1.0) ** 2))
    # Cosine
    for f in (1, 2, 3, 5, 8):
        for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
            add(1.0 + alpha * np.cos(2.0 * np.pi * f * x))
    # Sine
    for f in (1, 2, 3, 5):
        for alpha in (0.5, 1.0, 1.5):
            add(0.5 + alpha * np.abs(np.sin(np.pi * f * x)))
    # Exponential
    for rate in (1.0, 2.0, 3.0, 5.0, 8.0, 15.0):
        add(np.exp(-rate * x) + np.exp(-rate * (1.0 - x)) + 0.01)
    # Bimodal
    for width in (0.03, 0.05, 0.1, 0.15):
        for center in (0.15, 0.2, 0.3, 0.35):
            add(np.exp(-(((x - center) / width) ** 2)) +
                np.exp(-(((x - (1.0 - center)) / width) ** 2)) + 0.01)
    # Power shapes
    for p in (0.3, 0.5, 1.0, 2.0, 4.0):
        add((1.0 - np.abs(2.0 * x - 1.0)) ** p + 0.001)
    # Random smooth
    Nf = next_power_of_two(2 * n)
    for _ in range(5):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.0, 4.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise -= float(noise.mean())
        add(1.0 + float(rng.uniform(0.3, 1.5)) * noise)
    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def perturb(a, rng, strength=None):
    """Generate a perturbed version of a."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if strength is None:
        strength = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 8))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        conv = fast_self_conv(p)
        w = top_k_weights(conv, top_k=15, eps_frac=0.02)
        h = cross_corr(p, w)
        hmax = max(float(np.max(h)), 1e-15)
        q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        k = min(31, max(5, n // 500))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(rng.standard_normal(n), kern, mode="same")
        noise -= noise.mean()
        q = p * np.exp(strength * 0.5 * noise)
    elif mode == 5:
        x = np.linspace(0.0, 1.0, n)
        q = p * (1.0 - strength * 0.5 * (2.0 * x - 1.0) ** 2)
    elif mode == 6:
        nb = int(rng.integers(4, 20))
        block = 1.0 + strength * 0.3 * (rng.random(nb) * 2 - 1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.full(n - arr.size, block[-1])])
        q = p * arr[:n]
    else:
        q = p * (0.85 + 0.3 * rng.random(n))
    return normalize_a(np.clip(q, 0.0, None))


def multi_length_search(parent, deadline, rng, verbose=True):
    """Try multiple sequence lengths and return the best."""
    candidates = []
    # Determine which lengths to try
    n_values = [8192, 10240, 12288, 16384]
    # If parent has a specific length, include it
    if parent is not None:
        pn = len(parent)
        # Round to nearest power-of-two-ish
        for nv in [8192, 10240, 12288, 16384, 20480]:
            if abs(nv - pn) / pn < 0.15 and nv not in n_values:
                n_values = list(set(n_values + [nv]))
                break
    n_values = sorted(set(n_values))

    # Allocate time proportionally
    total_time = deadline - time.monotonic()
    if total_time < 60:
        n_values = [n_values[1] if len(n_values) > 1 else n_values[0]]
        total_time = deadline - time.monotonic()

    # Try at most 3 lengths
    if len(n_values) > 3:
        # Prioritize lengths close to parent
        if parent is not None:
            n_values = sorted(n_values, key=lambda x: abs(x - len(parent)))[:3]
        else:
            n_values = n_values[:3]

    for i, n in enumerate(n_values):
        if time.monotonic() > deadline - 30:
            break
        rem = deadline - time.monotonic() - 20
        if rem < 60:
            break
        alloc = rem / max(1, len(n_values) - i)
        chunk_dead = time.monotonic() + alloc * 0.9

        if verbose:
            print("Trying n={} (alloc {:.0f}s)".format(n, alloc))

        # Get initial sequence
        if parent is not None:
            a0 = resample_sequence(parent, n)
        else:
            inits = generate_diverse_inits(n, rng, None, cap=10)
            a0 = inits[0]
            best_sc = np.inf
            for init in inits:
                sc = exact_score(init)
                if sc < best_sc:
                    best_sc = sc
                    a0 = init

        a = normalize_a(a0)
        sc = exact_score(a)
        if verbose:
            print("  n={} initial: {:.9f}".format(n, sc))

        # Phase A: Active set optimization (50% of allocation)
        dead_a = time.monotonic() + alloc * 0.45
        sc_a, a = active_set_optimize(a, dead_a, verbose=False, tag="n{}a".format(n))
        sc = min(sc, sc_a)

        # Phase B: LSE annealing (35% of allocation)
        dead_b = time.monotonic() + alloc * 0.32
        sc_b, a = lse_anneal(a, dead_b, verbose=False, tag="n{}b".format(n))
        sc = min(sc, sc_b)

        # Phase C: Fine polish (15% of allocation)
        dead_c = chunk_dead
        sc_c, a = fine_polish(a, dead_c, verbose=False, tag="n{}c".format(n))
        sc = min(sc, sc_c)

        if verbose:
            print("  n={} best: {:.9f}".format(n, sc))
        candidates.append((sc, a, n))

    # Return best
    candidates.sort(key=lambda x: x[0])
    return candidates[0][1], candidates[0][0]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main entry point: search for the best sequence."""
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and float(np.sum(p)) > 0.01:
                parent = p / float(np.sum(p))
                print("Loaded parent: length={}, score={}".format(
                    len(parent), exact_score(parent)))
    except Exception:
        parent = None

    # Determine primary n
    if parent is not None:
        n_primary = len(parent)
        if n_primary < 4096:
            n_primary = 8192
        elif n_primary > 24000:
            n_primary = 16384
    else:
        n_primary = 10240

    print("Starting search: n_primary={}, budget={}s".format(n_primary, total_budget))

    best_a = None
    best_sc = np.inf

    # Strategy: Multi-phase with time allocation
    # Phase 1 (40%): Active set optimization on primary length
    # Phase 2 (25%): LSE annealing
    # Phase 3 (15%): Fine polish
    # Phase 4 (10%): Multi-length exploration
    # Phase 5 (10%): Diversification restarts

    # Phase 1: Active set optimization
    if parent is not None:
        a = resample_sequence(parent, n_primary)
    else:
        inits = generate_diverse_inits(n_primary, rng, None, cap=15)
        a = inits[0]
        best_init_sc = np.inf
        for init in inits:
            sc = exact_score(init)
            if sc < best_init_sc:
                best_init_sc = sc
                a = init
        print("Best init: {:.9f}".format(best_init_sc))

    a = normalize_a(a)
    sc0 = exact_score(a)
    print("Phase 0 - Initial: n={}, score={:.9f}".format(n_primary, sc0))
    best_a = a.copy()
    best_sc = sc0

    p1_dead = t_start + 0.35 * total_budget
    sc, a = active_set_optimize(a, deadline=p1_dead, verbose=True, tag="p1")
    a = normalize_a(a)
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("After Phase 1: {:.9f}".format(best_sc))

    # Phase 2: LSE annealing
    p2_dead = t_start + 0.60 * total_budget
    sc, a = lse_anneal(a, deadline=p2_dead, verbose=True, tag="p2")
    a = normalize_a(a)
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("After Phase 2: {:.9f}".format(best_sc))

    # Phase 3: Fine polish
    p3_dead = t_start + 0.75 * total_budget
    sc, a = fine_polish(a, deadline=p3_dead, verbose=True, tag="p3")
    a = normalize_a(a)
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    print("After Phase 3: {:.9f}".format(best_sc))

    # Phase 4: Multi-length exploration
    if time.monotonic() < deadline - 120:
        p4_dead = t_start + 0.88 * total_budget
        a_multi, sc_multi = multi_length_search(best_a, p4_dead, rng, verbose=True)
        if sc_multi < best_sc:
            best_sc = sc_multi
            best_a = a_multi.copy()
        print("After Phase 4: {:.9f}".format(best_sc))

    # Phase 5: Diversification restarts
    p5_dead = deadline - 30.0
    n_restarts = 5
    restart_count = 0
    while time.monotonic() < p5_dead - 40 and restart_count < n_restarts:
        rem = p5_dead - 40.0 - time.monotonic()
        if rem < 50:
            break
        restart_count += 1
        p = perturb(best_a, rng, strength=float(rng.uniform(0.03, 0.2)))
        # Quick optimization: active set + short LSE
        chunk_dead = time.monotonic() + rem * 0.7
        sc_r, a_r = active_set_optimize(p, deadline=chunk_dead - rem * 0.3,
                                        verbose=False, tag="r{}".format(restart_count))
        a_r = normalize_a(a_r)
        # Short LSE polish
        dead_lse = min(chunk_dead, time.monotonic() + rem * 0.3)
        sc_r2, a_r = lse_anneal(a_r, deadline=dead_lse, verbose=False, tag="rl{}".format(restart_count))
        sc_r = min(sc_r, exact_score(a_r))
        if sc_r < best_sc:
            best_sc = sc_r
            best_a = a_r.copy()
            print("Restart {}: {:.9f}".format(restart_count, best_sc))

    # Phase 6: Final polish
    if time.monotonic() < deadline - 15:
        sc, a = fine_polish(best_a, deadline=deadline - 10, verbose=False, tag="final")
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Final output
    print("Best final: {:.9f}".format(best_sc))
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("DONE: internal={:.9f} verified={:.9f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
