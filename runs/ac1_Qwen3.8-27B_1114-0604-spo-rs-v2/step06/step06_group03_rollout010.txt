"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: LSE continuation, active-set fixed-point (KKT-based), subgradient polish,
and multi-length exploration. The key innovation is a principled active-set
fixed-point that directly enforces the KKT condition: at optimum, the
correlation of a with the active set must be uniform on the support.

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Compute full linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_with_active(a, active_indices):
    """Compute h[i] = sum_{k in active} a[k-i] / |active| for valid k-i.
    This is the subgradient direction."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    active_indices = np.asarray(active_indices)
    if active_indices.size == 0:
        return np.zeros(n, dtype=np.float64)
    # h[i] = (1/|A|) * sum_{k in A} a[k-i] where 0 <= k-i < n
    # This is a cross-correlation type operation
    # Use FFT: h = ifft(fft(a_rev) * fft(w)) where w is active indicator
    M = 2 * n - 1
    N = next_power_of_two(M)
    # a[k-i] summed over k: this is convolution of a with active_indicator (reversed)
    w = np.zeros(2 * n - 1, dtype=np.float64)
    valid = (active_indices >= 0) & (active_indices < 2 * n - 1)
    w[active_indices[valid]] = 1.0
    w /= float(active_indices[valid].size)
    # h[i] = sum_k w[k] * a[k-i] = (w * a)[i] in a specific sense
    # Actually: h[i] = sum_k w[k] * a[k-i]
    # = cross-correlation of w and a at lag related to i
    # Let's use direct computation via FFT:
    # h = ifft(fft(a, N) * conj(fft(w, N))) doesn't give what we want directly
    # Let me just do: for each k in active, add a[k-i] shifted
    # Efficient: use the fact that h[i] = (w_rev * a)[something]
    # Actually the simplest: h[i] = sum_k w[k] * a[k-i]
    # = sum_j a[j] * w[i+j]  (substituting j = k-i, k = i+j)
    # = (a * w)[i]  (linear convolution of a and w, evaluated at i)
    Aw = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # (a * w)[i] = ifft(A * W)[i]
    full = np.fft.irfft(Aw * W, N)
    return full[:n]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    """Log-sum-exp smooth approximation with gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta < 1e-3:
        beta = 100.0

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        e = np.exp(z - z.max())
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    # Gradient: 2*n/S^2 * sum_k w[k] * a[k-i] - 2*n*LDS/S
    # Use corr_with_active for the first term
    h = corr_with_active(a, np.arange(2 * n - 1))
    # But we need weighted: sum_k w[k]*a[k-i], not uniform active
    # Let's compute directly via FFT
    M = 2 * n - 1
    N = next_power_of_two(M)
    Aw = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(Aw * W, N)[:n]

    grad = (4.0 * full) / S
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(2.0 * L), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun):
    """Run L-BFGS-B on the LSE objective."""
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    maxfun = max(64, int(maxfun))
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }
    try:
        res = minimize(fun_lse, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def estimate_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 8192.0)
    return base * scale


def active_set_threshold(d, frac=0.99, k_min=1, k_max=None):
    """Get active set as entries above a fraction of max."""
    d = np.asarray(d)
    dmax = float(np.max(d))
    if dmax <= 0:
        return np.array([0], dtype=int)
    threshold = dmax * frac
    idx = np.where(d >= threshold)[0]
    # Ensure at least k_min entries
    if idx.size < k_min:
        idx = np.argsort(d)[-k_min:]
    if k_max is not None and idx.size > k_max:
        idx = np.argsort(d)[-k_max:]
    return np.sort(idx)


def active_set_fixedpoint(a, frac=0.99, max_iters=200, deadline=None, verbose=False):
    """Fixed-point iteration based on KKT conditions.
    At optimum: corr(a, active) is constant on support(a)."""
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        c = fast_self_conv(a)
        d = c / max(float(np.max(c)), 1e-30)
        M = float(np.max(c))
        active = active_set_threshold(d, frac=frac, k_min=1, k_max=min(200, 2 * n - 1))

        # Compute h[i] = (1/|A|) * sum_{k in A} a[k-i]
        h = corr_with_active(a, active)
        h = np.maximum(h, 0.0)
        h_sum = float(h.sum())
        if h_sum < 1e-15:
            break

        # Check convergence: h should be ~constant on support
        support = np.where(a > 1e-10)[0]
        if support.size > 0:
            h_support = h[support]
            h_range = float(h_support.max() - h_support.min())
            h_mean = float(h_support.mean())
            rel_range = h_range / (h_mean + 1e-30)
            if verbose and it % 20 == 0:
                sc = exact_score_a(a)
                print("  FP iter {}: score={:.8f}, rel_range={:.6f}, |A|={}".format(
                    it, sc, rel_range, len(active)))

        # Update: blend a with normalized h
        h_norm = h / h_sum
        # Multiple blend factors
        best_candidate = None
        best_candidate_sc = best_sc
        for alpha in (1.0, 0.8, 0.7, 0.5, 0.3, 0.2, 0.1):
            q = normalize_a(alpha * h_norm + (1.0 - alpha) * a)
            sc = exact_score_a(q)
            if np.isfinite(sc) and sc < best_candidate_sc:
                best_candidate_sc = sc
                best_candidate = q
        if best_candidate is not None:
            a = best_candidate.copy()
            if best_sc > best_candidate_sc:
                best_sc = best_candidate_sc
                best_a = a.copy()
        else:
            # No improvement from any blend; try perturbations
            improved = False
            for eta in (0.01, 0.03, 0.1, 0.3):
                log_a = np.log(np.maximum(a, 1e-15))
                log_h = np.log(np.maximum(h_norm, 1e-15))
                log_new = (1.0 - eta) * log_a + eta * log_h
                q = normalize_a(np.exp(log_new - np.max(log_new)))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved = True
            if not improved and it > 3:
                break

    return float(best_sc), best_a


def subgrad_polish(a, max_iters=100, deadline=None):
    """Polish using exact subgradient with various active set sizes."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        c = fast_self_conv(a)
        S = float(np.sum(a))
        dmax = float(np.max(c))
        if dmax < 1e-15:
            break

        improved_this_iter = False
        for k in (1, 2, 3, 5, 8, 13, 21, 50):
            if k >= 2 * n - 1:
                break
            idx = np.argpartition(c, -k)[-k:]
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0 / k
            h = corr_with_active(a, idx)
            hmax = float(np.max(np.abs(h)))
            if hmax < 1e-15:
                continue
            grad = (2.0 * h) / S - (2.0 * float(np.dot(w, (n * c) / (S * S)))) / S

            for eta in (0.001, 0.003, 0.01, 0.03, 0.1):
                q = normalize_a(np.maximum(a - eta * grad, 0.0))
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    improved_this_iter = True
        if not improved_this_iter and it > 10:
            break
    return float(best_sc), best_a


def perturb_sequence(a, rng, strength=0.1):
    """Create a perturbed version of the sequence."""
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy() * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Frequency-domain perturbation
        N = next_power_of_two(n)
        spec = np.fft.rfft(p, N)
        spec *= np.exp(strength * 0.5 * rng.standard_normal(spec.size))
        q = np.fft.irfft(spec, N)[:n]
    elif mode == 4:
        # Local perturbation
        center = int(rng.integers(n // 4, 3 * n // 4))
        width = int(rng.integers(n // 20, n // 5))
        mask = np.exp(-((np.arange(n) - center) ** 2) / (2.0 * width * width))
        q = p * (1.0 + strength * 0.5 * mask * rng.standard_normal(n))
    else:
        # Random phase in frequency domain
        N = next_power_of_two(n)
        spec = np.fft.rfft(p, N)
        phase = rng.uniform(-0.3 * strength, 0.3 * strength, spec.size)
        spec[0] = abs(spec[0])  # keep DC
        spec = spec * np.exp(1j * phase)
        q = np.fft.irfft(spec, N)[:n]
    return normalize_a(np.clip(q, 0.0, None))


def lse_trajectory(a_start, deadline, label=""):
    """Run LSE optimization with increasing beta schedule."""
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    stages = [
        (200.0, 0.06), (500.0, 0.06), (1000.0, 0.06), (3000.0, 0.07),
        (8000.0, 0.07), (20000.0, 0.07), (50000.0, 0.07), (100000.0, 0.07),
        (300000.0, 0.08), (1000000.0, 0.08), (3000000.0, 0.08),
        (10000000.0, 0.08), (50000000.0, 0.09),
    ]

    t0 = time.monotonic()
    total = max(10.0, float(deadline - t0 - 5.0))
    cum = 0.0
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + min(cum, 1.0) * total
        sub_dead = min(float(sub_dead), float(deadline - 3.0))
        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            est = estimate_eval_time(len(a))
            mf = int(min(2000, max(64, (0.7 * rem) / est)))
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                if verbose_lbfgs(sc, label):
                    print("  {} beta={:.0f}: {:.8f}".format(label, beta, best_sc))
            a = normalize_a(a2)
    return float(best_sc), best_a

    # === Phase 1: LSE optimization (35% of time) ===
    t1_end = t_start + 0.35 * total_budget
    print("Phase 1: LSE trajectory (t_end={:.0f}s)".format(t1_end - t_start))
    sc, seq = lse_trajectory(a_start, t1_end, "LSE")
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
    print("After LSE: {:.8f}".format(best_sc))

    # === Phase 2: Active-set fixed-point (25% of time) ===
    t2_end = t_start + 0.60 * total_budget
    print("Phase 2: Active-set FP (t_end={:.0f}s)".format(t2_end - t_start))
    for frac in (0.99, 0.98, 0.95, 0.90, 0.80):
        if time.monotonic() > t2_end - 30.0:
            break
        sub_dead = min(t2_end - 10.0, time.monotonic() + 180.0)
        sc, seq = active_set_fixedpoint(best_seq, frac=frac, max_iters=300, deadline=sub_dead)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("  FP frac={}: {:.8f}".format(frac, best_sc))
        # Follow with quick LSE
        if time.monotonic() < t2_end - 60.0:
            sc, seq = lse_trajectory(best_seq, min(t2_end - 30.0, time.monotonic() + 120.0), "FP+LSE")
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("  FP+LSE frac={}: {:.8f}".format(frac, best_sc))
    print("After FP: {:.8f}".format(best_sc))

    # === Phase 3: Subgradient polish (15% of time) ===
    t3_end = t_start + 0.75 * total_budget
    print("Phase 3: Subgrad polish (t_end={:.0f}s)".format(t3_end - t_start))
    sc, seq = subgrad_polish(best_seq, max_iters=200, deadline=t3_end)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
    print("After subgrad: {:.8f}".format(best_sc))

    # === Phase 4: High-beta LSE + FP cycle (15% of time) ===
    t4_end = t_start + 0.90 * total_budget
    print("Phase 4: High-beta + FP cycle (t_end={:.0f}s)".format(t4_end - t_start))
    a = normalize_a(best_seq)
    for beta in (500000.0, 2000000.0, 10000000.0, 50000000.0, 200000000.0):
        if time.monotonic() > t4_end - 60.0:
            break
        rem = float(t4_end - time.monotonic())
        if rem < 30.0:
            break
        est = estimate_eval_time(len(a))
        mf = int(min(1500, max(64, (0.6 * rem) / est)))
        sc, a2 = run_lbfgs(a, beta, t4_end - 30.0, mf)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = a2.copy()
            print("  HB beta={:.0e}: {:.8f}".format(beta, best_sc))
        a = normalize_a(a2)
        # Quick FP
        if time.monotonic() < t4_end - 30.0:
            sc, a2 = active_set_fixedpoint(a, frac=0.99, max_iters=50,
                                           deadline=t4_end - 20.0)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  HB+FP beta={:.0e}: {:.8f}".format(beta, best_sc))
            a = normalize_a(a2)
    print("After Phase 4: {:.8f}".format(best_sc))

    # === Phase 5: Multi-length + perturbation restarts (10% of time) ===
    t5_end = deadline - 10.0
    print("Phase 5: Multi-length + perturb (t_end={:.0f}s)".format(t5_end - t_start))

    # Try other lengths
    other_ns = [8192, 12288, 16384]
    for n_other in other_ns:
        if time.monotonic() > t5_end - 60.0:
            break
        if n_other == n_primary:
            continue
        a_other = normalize_a(resample_sequence(best_seq, n_other))
        sc_other = exact_score_a(a_other)
        # Quick LSE
        if time.monotonic() < t5_end - 40.0:
            sc, a2 = lse_trajectory(a_other, min(t5_end - 30.0, time.monotonic() + 90.0), "n={}".format(n_other))
            if np.isfinite(sc) and sc < sc_other:
                sc_other = sc
                a_other = a2
        # Quick FP
        if time.monotonic() < t5_end - 20.0:
            sc, a2 = active_set_fixedpoint(a_other, frac=0.99, max_iters=50, deadline=t5_end - 15.0)
            if np.isfinite(sc) and sc < sc_other:
                sc_other = sc
                a_other = a2
        if np.isfinite(sc_other) and sc_other < best_sc:
            best_sc = sc_other
            best_seq = a_other.copy()
            print("  n={}: {:.8f}".format(n_other, best_sc))

    # Perturbation restarts
    perturb_count = 0
    while time.monotonic() < t5_end - 40.0 and perturb_count < 3:
        left = float(t5_end - time.monotonic() - 40.0)
        if left < 60.0:
            break
        chunk_end = time.monotonic() + left / float(max(1, 3 - perturb_count)) - 10.0
        p = perturb_sequence(best_seq, rng, strength=float(rng.uniform(0.03, 0.15)))
        for beta in (100000.0, 500000.0):
            if time.monotonic() > chunk_end:
                break
            rem = float(chunk_end - time.monotonic())
            if rem < 20.0:
                break
            est = estimate_eval_time(len(p))
            mf = int(min(500, max(64, (0.6 * rem) / est)))
            sc, a2 = run_lbfgs(p, beta, chunk_end, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Perturb {} b={:.0f}: {:.8f}".format(perturb_count, beta, best_sc))
            p = normalize_a(a2)
        if time.monotonic() < chunk_end - 10.0:
            sc, a2 = active_set_fixedpoint(p, frac=0.99, max_iters=50, deadline=chunk_end)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Perturb {} FP: {:.8f}".format(perturb_count, best_sc))
        perturb_count += 1

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 15.0:
        sc, seq = active_set_fixedpoint(best_seq, frac=0.99, max_iters=100, deadline=deadline - 10.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
        sc, seq = subgrad_polish(best_seq, max_iters=30, deadline=deadline - 5.0)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()

    # Final verification
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
