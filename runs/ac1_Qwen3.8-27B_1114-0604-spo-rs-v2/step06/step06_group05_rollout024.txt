import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(x):
    """Smallest power of 2 >= x."""
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = next_pow2(M2)
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def weighted_score_grad(a, w):
    """Weighted convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    w_num = float(np.dot(w, c))
    score = 2.0 * n * w_num / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = next_pow2(M2)
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * w_num) / S3
    return score, grad


def run_lbfgs(a0, obj_func, obj_params, maxiter=5000, maxfun=50000, ftol=1e-18, gtol=1e-15):
    """Run L-BFGS-B optimization with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = obj_func(x, *obj_params)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def gs_spectral_target(a, c_target, n_iters=500, alpha=0.4):
    """GS iteration: match non-negativity in time domain and spectral target."""
    n = len(a)
    M = next_pow2(2 * n - 1)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        # Compute current spectrum
        A = np.fft.rfft(a, M)
        A_mag = np.abs(A)
        A_phase = np.angle(A)

        # Target spectrum: sqrt of target convolution's spectrum
        C_target = np.fft.rfft(c_target, M)
        # sqrt magnitude, preserve phase relationship
        C_mag = np.abs(C_target)
        C_phase = np.angle(C_target)

        # Blend: new_A = alpha * target_A + (1-alpha) * current_A
        # Use magnitude blending for stability
        T_mag = np.sqrt(np.maximum(C_mag, 1e-30))
        T_phase = C_phase / 2.0  # Phase of sqrt

        # Use weighted geometric mean of magnitudes and phase interpolation
        new_mag = A_mag ** (1.0 - alpha) * np.maximum(T_mag, 1e-30) ** alpha
        # Phase: interpolate (handle wraparound)
        d_phase = T_phase - A_phase / 2.0
        d_phase = np.arctan(np.tan(d_phase))
        new_phase = A_phase / 2.0 + alpha * d_phase

        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_adaptive(a, n_iters=300):
    """GS with adaptive target based on current convolution."""
    n = len(a)
    M = next_pow2(2 * n - 1)

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())

    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        # Target: blend current c with flat
        t = 0.2 + 0.3 * (it / n_iters)
        flat = np.ones(2 * n - 1) * float(c.mean())
        c_target = (1 - t) * c + t * flat
        c_target = np.maximum(c_target, 1e-10)

        alpha = 0.3 * (1.0 - 0.5 * it / n_iters) + 0.1

        a = gs_spectral_target(a, c_target, n_iters=1, alpha=alpha)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def nesterov_lse(a0, beta, n_iters=15000, lr_init=0.01, lr_decay=0.99995, momentum=0.9):
    """Nesterov momentum on LSE objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)

        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            a_look = a
        else:
            a_look = a_look / S_look

        if it % 20 == 0:
            _, grad_look = lse_score_grad(a_look, beta)
            v = momentum * v + lr * grad_look
        else:
            v = momentum * v + lr * grad

        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        lr *= lr_decay

        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_a


def peak_reduction(a0, n_rounds=50, top_k=30, reduction=0.8):
    """Directly reduce elements contributing to top peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:

    for rnd in range(n_rounds):

        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            partner = a[k - j_end:k - j_start + 1][::-1]
            contrib = seg * partner
            n_reduce = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_reduce]
            for li in top_local:
                j = j_start + li
                a[j] *= reduction

        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def crossover(a1, a2, rng):
    """Create a new candidate by combining two solutions."""
    n = min(len(a1), len(a2))
    strategies = [
        'min', 'max', 'mean', 'geomean', 'random_mix'
    ]
    strategy = strategies[rng.integers(len(strategies))]

    a1 = a1[:n]
    a2 = a2[:n]

    if strategy == 'min':
        result = np.minimum(a1, a2)
    elif strategy == 'max':
        result = np.maximum(a1, a2)
    elif strategy == 'mean':
        result = 0.5 * (a1 + a2)
    elif strategy == 'geomean':
        result = np.sqrt(np.maximum(a1 * a2, 1e-20))
    else:  # random_mix
        t = rng.random(n)
        result = t * a1 + (1 - t) * a2

    result = np.maximum(result, 0.0)
    S = result.sum()
    if S < 1e-12:
        return 0.5 * (a1 + a2)
    return result / S


def perturb(a0, rng, strength=0.03):

    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:

    return a / S

            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        if len(parent) == n:
            best_a = parent.copy()
        else:
            best_a = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(parent)), parent).copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f}")

    # ============================================================
    # Phase 1: Quick spectral GS warmup (5% time)
    # ============================================================
    p1_end = t_start + 0.05 * (budget_s - 30.0)
    print(f"\n=== Phase 1: GS warmup ({p1_end - t_start:.0f}s) ===")

    inits = build_inits(n, rng, parent)
    for a0 in inits[:4]:
        if time.monotonic() > p1_end:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / max(a0.sum(), 1e-12)
        a1 = gs_adaptive(a0, n_iters=200)
        update_best(compute_score(a1), a1)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS continuation (35% time)
    # ============================================================
    p2_end = t_start + 0.40 * (budget_s - 30.0)
    print(f"\n=== Phase 2: LSE-LBFGS continuation ({p2_end - t_start:.0f}s) ===")

    # Continue beta from low to high
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for beta in beta_schedule:
        if time.monotonic() > p2_end:
            break
        t0 = time.monotonic()
        remaining = p2_end - t0
        # Scale maxiter based on remaining time
        maxiter = min(10000, max(1000, int(remaining / 0.5)))
        maxfun = min(50000, max(5000, maxiter * 5))

        a_opt = run_lbfgs(best_a, lse_score_grad, (beta,),
                          maxiter=maxiter, maxfun=maxfun)
        sc = compute_score(a_opt)
        update_best(sc, a_opt)
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Additional LSE runs from perturbed versions
    if time.monotonic() < p2_end:
        for strength in [0.02, 0.05]:
            if time.monotonic() > p2_end:
                break
            a_pert = perturb(best_a, rng, strength)
            a_opt = run_lbfgs(a_pert, lse_score_grad, (5000,),
                              maxiter=5000, maxfun=20000)
            sc = compute_score(a_opt)
            if sc < best_score:
                # Found better, update
                update_best(sc, a_opt)

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Iterative reweighting (25% time)
    # ============================================================
    p3_end = t_start + 0.65 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Iterative reweighting ({p3_end - t_start:.0f}s) ===")

    # Main IR run
    remaining_iters = max(10, int((p3_end - time.monotonic()) / 5.0))
    a_ir = iterative_reweighting(best_a, n_outer=min(30, remaining_iters),
                                  n_inner=2000, maxfun_per_inner=3000)
    update_best(compute_score(a_ir), a_ir)

    # IR from perturbed
    if time.monotonic() < p3_end:
        a_pert = perturb(best_a, rng, 0.03)
        a_ir2 = iterative_reweighting(a_pert, n_outer=15,
                                       n_inner=1500, maxfun_per_inner=2000)
        update_best(compute_score(a_ir2), a_ir2)

    # LSE polish after IR
    if time.monotonic() < p3_end:
        a_opt = run_lbfgs(best_a, lse_score_grad, (20000,),
                          maxiter=5000, maxfun=20000)
        update_best(compute_score(a_opt), a_opt)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Diversification - multi-restart + crossover (20% time)
    # ============================================================
    p4_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 4: Diversification ({p4_end - t_start:.0f}s) ===")

    # Crossover with perturbed versions
    pool = [best_a.copy()]
    for _ in range(3):
        if time.monotonic() > p4_end:
            break
        a_pert = perturb(best_a, rng, 0.04)
        pool.append(a_pert)

    for i in range(len(pool)):
        for j in range(i + 1, len(pool)):
            if time.monotonic() > p4_end:
                break
            a_cross = crossover(pool[i], pool[j], rng)
            a_opt = run_lbfgs(a_cross, lse_score_grad, (5000,),
                              maxiter=3000, maxfun=10000)
            sc = compute_score(a_opt)
            update_best(sc, a_opt)

    # Nesterov from best
    if time.monotonic() < p4_end:
        a_nm = nesterov_lse(best_a, beta=10000, n_iters=8000,
                            lr_init=0.005, lr_decay=0.9999, momentum=0.9)
        update_best(compute_score(a_nm), a_nm)

    # Peak reduction
    if time.monotonic() < p4_end:
        a_pk = peak_reduction(best_a, n_rounds=40, top_k=50, reduction=0.85)
        sc = compute_score(a_pk)
        if update_best(sc, a_pk):
            # If peak reduction helped, polish with LSE
            a_opt = run_lbfgs(best_a, lse_score_grad, (10000,),
                              maxiter=3000, maxfun=10000)
            update_best(compute_score(a_opt), a_opt)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (10% time)
    # ============================================================
    print(f"\n=== Phase 5: Final polish ===")

    polish_end = deadline
    for pr in range(10):
        if time.monotonic() > polish_end:
            break

        # GS refinement
        a_gs = gs_adaptive(best_a, n_iters=100)
        update_best(compute_score(a_gs), a_gs)

        # LSE with very high beta
        if time.monotonic() > polish_end:
            break
        beta_high = 50000 + 20000 * pr
        a_opt = run_lbfgs(best_a, lse_score_grad, (beta_high,),
                          maxiter=3000, maxfun=10000,
                          ftol=1e-20, gtol=1e-16)
        update_best(compute_score(a_opt), a_opt)

        # Quick IR
        if time.monotonic() > polish_end:
            break
        a_ir = iterative_reweighting(best_a, n_outer=3, n_inner=1000,
                                     maxfun_per_inner=2000)
        update_best(compute_score(a_ir), a_ir)

        # Nesterov fine-tune
        if time.monotonic() > polish_end:
            break
        a_nm = nesterov_lse(best_a, beta=30000, n_iters=3000,
                            lr_init=0.002, lr_decay=0.9999, momentum=0.9)
        update_best(compute_score(a_nm), a_nm)

        print(f"  Polish {pr}: best = {best_score:.8f}")

    # ============================================================
    # Try different n values if time permits
    # ============================================================
    if time.monotonic() < deadline - 60:
        print(f"\n=== Trying alternate n ===")
        for n_alt in [len(best_a) + 500, len(best_a) - 500]:
            if time.monotonic() > deadline - 30:
                break
            if n_alt < 100:
                continue
            # Interpolate best_a to new length
            x_old = np.linspace(0, 1, len(best_a))
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            a_alt = a_alt / a_alt.sum()

            # Quick LSE
            a_opt = run_lbfgs(a_alt, lse_score_grad, (5000,),
                              maxiter=3000, maxfun=10000)
            sc = compute_score(a_opt)
            print(f"  n={n_alt}: score={sc:.8f}")
            update_best(sc, a_opt)

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
