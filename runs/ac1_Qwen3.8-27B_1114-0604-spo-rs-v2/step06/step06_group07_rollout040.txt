"""
Optimized step function using LP-based direction finding with 
constraint generation, adaptive LSE gradient, and spectral shaping.

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

def fast_conv(a, b=None):
    """Fast linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def fast_conv_square(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    c = fast_conv_square(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_lp_matrix(f):
    """Build the sparse constraint matrix for the LP.
    A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    Shape: (2n-1, n)
    """
    n = len(f)
    # Each column i has nonzeros in rows i to min(i+n-1, 2n-2)
    # Use a more efficient construction
    n_rows = 2 * n - 1
    n_cols = n
    
    # For each column i, the nonzero entries are f[0], f[1], ..., f[n-1]
    # starting at row i
    rows = np.concatenate([np.arange(i, min(i+n, n_rows)) for i in range(n_cols)])
    cols = np.concatenate([np.full(min(n, n_rows-i), i) for i in range(n_cols)])
    vals = np.tile(f, n_cols)  # Each column has f[0], f[1], ..., f[n-1]
    
    # Wait, this isn't right. For column i, row k, the value is f[k-i].
    # So for column i, the values are f[0], f[1], ..., f[min(n-1, n_rows-1-i)]
    # Let me redo this.
    
    # Actually for column i:
    # Row k (for k = i, i+1, ..., min(i+n-1, n_rows-1)): A[k,i] = f[k-i]
    # So the values in column i are f[0], f[1], ..., f[min(n-1, n_rows-1-i)]
    # Since n_rows-1 = 2n-2, and i <= n-1, we have n_rows-1-i >= n-1
    # So each column has exactly n entries: f[0], f[1], ..., f[n-1]
    
    total = n * n_cols
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    
    for i in range(n_cols):
        idx_start = i * n_cols
        idx_end = (i + 1) * n_cols
        rows[idx_start:idx_end] = np.arange(i, i + n)
        cols[idx_start:idx_end] = i
        vals[idx_start:idx_end] = f
    
    A_sparse = sp.csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))
    return A_sparse

def lp_direction(f, M, time_limit=30):
    """
    Solve LP: max sum(g) s.t. conv(f, g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if infeasible/too slow.
    """
    n = len(f)
    A_sparse = build_lp_matrix(f)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return res.x
        elif res.status == 6:  # timeout
            return None
        else:
            return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None

def lp_iteration(f, n_lp_iter=5, time_limit=30):
    """
    Perform LP-based iterations.
    f is normalized to sum=1.
    Returns improved f.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S
    
    best_f = f.copy()
    best_sc = compute_score(f)
    t = 0.1
    
    for it in range(n_lp_iter):
        c = fast_conv_square(f)
        M = float(c.max())
        S = float(f.sum())
        # M in terms of the normalized problems
        # We want conv(f, g)[k] <= M for all k
        # where M = max(conv(f, f))
        
        g = lp_direction(f, M, time_limit=time_limit)
        if g is None:
            break
        
        g = np.maximum(g, 0)
        Sg = g.sum()
        if Sg < 1e-12:
            break
        g = g / Sg  # Normalize g to sum=1
        
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            break
        f_new = f_new / S_new
        
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
            f = f_new
            # Increase t slightly
            t = min(t * 1.5, 0.5)
        else:
            # Decrease t
            t = max(t * 0.5, 0.01)
            f = f_new  # Still move, but smaller step
            S_new = f.sum()
            f = f / S_new
    
    return best_f

def fienup_flat_spectral(a, n_iters=200, alpha=0.5):
    """
    Fienup iteration targeting flat 2n-point spectrum of zero-padded a.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1
    a = a / S
    
    # Zero-pad to 2n
    M = 2 * n
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Compute 2n-point DFT of zero-padded a
        A = np.fft.rfft(a, M)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat amplitude
        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)
        
        amp_new = amp * (C / np.maximum(amp, 1e-20)) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0)
        
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def lse_smoothed_grad(x, beta):
    """
    LSE-smoothed objective and gradient.
    Objective: 2n * LSE_beta(conv(x,x)) / (sum(x))^2
    """
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient of LSE
    p = exp_scaled / exp_sum  # Softmax weights
    
    # dL/dx_i = 2 * sum_k p[k] * dc_k/dx_i
    # dc_k/dx_i = 2*x[k-i] - x[i] (if 2i==k, then just x[i])
    # More precisely: c_k = sum_j x[j]*x[k-j]
    # dc_k/dx_i = x[k-i] + x[k-i] = 2*x[k-i] (if k-i != i, i.e., 2i != k)
    #            = x[i] (if 2i == k)
    # For the LSE, we use: dL/dx_i = 2 * sum_k p[k] * (x[k-i] + [k-i==i] * x[i] ... 
    # Actually, let's use the convolution approach:
    # dL/dx_i = 2 * (p * x_reversed)[i + n - 1] ... let me verify
    # 
    # c_k = sum_j x[j] * x[k-j]
    # dc_k/dx_i: j=i gives x[k-i]; k-j=i gives j=k-i, term is x[k-i]
    # If i != k-i: dc_k/dx_i = 2*x[k-i]
    # If i == k-i (k=2i): dc_k/dx_i = x[i]
    #
    # sum_k p[k] * dc_k/dx_i = sum_k p[k] * 2*x[k-i] - sum_k p[k]*x[k-i]*[k=2i]
    # = 2 * sum_k p[k]*x[k-i] - p[2i]*x[i] (if 2i is in range)
    #
    # The first term is a cross-correlation: (p * x_rev)[i + n - 1] where x_rev = x[::-1]
    # Let me verify: conv(p, x_rev)[m] = sum_k p[k] * x_rev[m-k] = sum_k p[k] * x[n-1-m+k]
    # For m = i + n - 1: = sum_k p[k] * x[n-1-(i+n-1)+k] = sum_k p[k] * x[k-i]
    # Yes! So conv(p, x_rev)[i+n-1] = sum_k p[k]*x[k-i]
    
    # Now handle the diagonal correction:
    # sum_k p[k]*dc_k/dx_i = 2*conv(p, x_rev)[i+n-1] - p[2i]*x[i] (if 0 <= 2i < 2n-1)
    
    x_rev = x[::-1]
    # conv(p, x_rev) has length (2n-1) + n - 1 = 3n-2
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    PP = np.fft.rfft(p, M2_fft)
    RX = np.fft.rfft(x_rev, M2_fft)
    conv_res = np.fft.irfft(PP * RX, M2_fft)[:3*n-2]
    
    # conv_res[i + n - 1] = sum_k p[k] * x[k-i]
    cross = conv_res[n-1:2*n-1]  # cross[i] = sum_k p[k]*x[k-i] for i=0,...,n-1
    
    # Diagonal correction
    diag_corr = np.zeros(n)
    for i in range(n):
        k = 2 * i
        if 0 <= k < 2*n-1:
            diag_corr[i] = p[k] * x[i]
    
    dL_dx = 2.0 * cross - diag_corr
    
    S2 = S * S
    grad = (2.0 * n) * (dL_dx / S2 - 2.0 * L / (S2 * S))
    
    return score, grad

def run_gradient_descent(a0, beta, n_iters=5000, lr_init=0.01, decay=0.999):
    """
    Projected gradient descent on the LSE-smoothed objective.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        score, grad = lse_smoothed_grad(a, beta)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        lr *= decay
        
        if it % 500 == 0:
            print(f"    GD iter {it}: score={sc:.8f}")
    
    return best_a

def fienup_geometric(a, n_iters=300, alpha_start=0.1, alpha_end=0.8):
    """Geometric Fienup on the 2n-point spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    M = 2 * n
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M)
        amp = np.abs(A)
        phase = np.angle(A)
        
        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def peak_descent(a, n_iters=2000, n_peaks=5, step_init=0.01, decay=0.9999):
    """Peak-directed coordinate descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    M_fft = 1 << (2*n-1-1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        top_idx = np.argpartition(c, -min(n_peaks, len(c)))[-min(n_peaks, len(c)):]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[js:je+1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, max((sc - sc_lb) / (gn*gn + 1e-30), 1e-10))
        step = min(step, 0.02)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

"""
Optimized step function construction using LP-based direction finding with
constraint generation, adaptive LSE gradient descent, and spectral shaping.
Key novelty: The LP-based direction finder solves a linear program to find the
optimal "flattening" direction at each step, which is complementary to gradient
methods and provides larger global moves.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_conv_square(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Fast linear convolution a*b via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv_square(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lp_direction_cg(f, M, max_rounds=15, n_init_cons=200, tol=1e-6):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses constraint generation for efficiency.
    """
    n = len(f)
    c_f = fast_conv_square(f)
    
    # Initial active constraints: positions where f*f is largest
    n_active_init = min(n_init_cons, 2 * n - 1)
    top_k = np.argpartition(c_f, -n_active_init)[-n_active_init:]
    active_k = list(top_k.astype(int))
    
    g = None
    for rnd in range(max_rounds):
        n_cons = len(active_k)
        # Build dense constraint matrix (n_cons x n)
        A = np.zeros((n_cons, n), dtype=np.float64)
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                A[i, js:je + 1] = f[js:je + 1][::-1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(n_cons, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.status != 0:
                return None
            g = res.x
        except Exception:
            return None
        
        # Check for violated constraints
        fg = fast_cross_conv(f, np.maximum(g, 0))
        violated = np.where(fg > M * (1.0 + tol))[0]
        
        if len(violated) == 0:
            return np.maximum(g, 0)
        
        # Add violated constraints (limit total)
        max_active = min(800, 2 * n - 1)
        active_k = list(set(active_k + [int(v) for v in violated]))
        if len(active_k) > max_active:
            # Keep most violated
            fg_active = np.array([fg[k] for k in active_k])
            order = np.argsort(fg_active)[::-1][:max_active]
            active_k = [active_k[i] for i in order]
    
    return np.maximum(g, 0) if g is not None else None


def lp_iteration(f, n_iters=5, time_per_lp=15.0):
    """
    Perform LP-based iterations on f (normalized to sum=1).
    Returns the best f found.
    """
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S
    
    best_f = f.copy()
    best_sc = compute_score(f)
    t = 0.3
    
    for it in range(n_iters):
        t_lp_start = time.monotonic()
        c = fast_conv_square(f)
        M = float(c.max())
        
        g = lp_direction_cg(f, M)
        t_lp_end = time.monotonic()
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        Sg = float(g.sum())
        if Sg < 1e-12:
            break
        g = g / Sg
        
        # Line search on t
        best_t = 0.0
        best_sc_try = best_sc
        for t_try in [0.5, 0.3, 0.15, 0.1, 0.05]:
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0.0)
            St = float(f_try.sum())
            if St < 1e-12:
                continue
            f_try = f_try / St
            sc_try = compute_score(f_try)
            if sc_try < best_sc_try:
                best_sc_try = sc_try
                best_t = t_try
        
        if best_t > 0:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            Sf = float(f_new.sum())
            f_new = f_new / Sf
            sc = compute_score(f_new)
            if sc < best_sc:
                best_sc = sc
                best_f = f_new.copy()
            f = f_new
        else:
            # Try direct jump
            sc_g = compute_score(g)
            if sc_g < best_sc:
                best_sc = sc_g
                best_f = g.copy()
                f = g.copy()
            else:
                break
    
    return best_f


def fienup_geometric(a, n_iters=300, alpha_start=0.05, alpha_end=0.7):
    """Geometric Fienup targeting flat 2n-point spectrum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    a = a / S
    
    M = 2 * n
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M)
        amp = np.abs(A)
        phase = np.angle(A)
        
        C = np.median(amp[1:]) if len(amp) > 1 else amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
