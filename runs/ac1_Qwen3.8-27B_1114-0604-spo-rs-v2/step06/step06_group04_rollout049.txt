"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. LSE-smoothed L-BFGS-B with progressive beta increase

3. Multi-restart perturbation search

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lse_value_grad(a, beta):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L
    T = float(w.dot(c))
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _subgrad_exact(a, k_active=3, tol=1e-6):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())
    score = 2.0 * dmax
    threshold = dmax - tol * dmax
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    T = float(w_active.dot(c))
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_optimize(a0, beta, deadline, maxfun=500):
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, 1.0)] * n
    opts = {"ftol": 1e-14, "gtol": 1e-10, "maxiter": max(maxfun // 2, 50),
            "maxfun": int(maxfun), "maxls": 40}
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds,
                       options=opts, callback=lambda x: None)
    except Exception:
        return _exact_score(a), a
    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _nesterov_optimize(a0, n_iters=2000, deadline=None, k_active=3, tol=1e-7):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    t_param = 1.0
    y = a.copy()
    v = np.zeros(n)
    score0, g0 = _subgrad_exact(a, k_active, tol)
    g_norm = np.linalg.norm(g0)
    if g_norm < 1e-12:
        return best_score, best_a
    eta = 0.05 / g_norm
    stale_count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > float(deadline) - 1.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta_mom = (t_param - 1.0) / t_new
        y = a + beta_mom * v
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
            Sy = 1.0
        y_norm = y / Sy
        score_y, grad_y = _subgrad_exact(y_norm, k_active, tol)
        if not np.isfinite(score_y):
            break
        step = eta / (1.0 + 0.05 * it)
        x_new = y_norm - step * grad_y
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        v = x_new - a
        a = x_new
        t_param = t_new
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            stale_count = 0
        else:
            stale_count += 1
        if it > 0 and it % 100 == 0:
            if stale_count > 80:
                eta *= 0.5
                stale_count = 0
                if eta < 1e-10:
                    break
    return best_score, best_a

def _perturb(a, rng, strength=None):
    a = _normalize(np.asarray(a, dtype=np.float64).ravel())
    n = a.size
    if n == 0:
        return a
    if strength is None:
        strength = float(rng.uniform(0.05, 0.4))
    mode = int(rng.integers(0, 8))
    x = np.linspace(0.0, 1.0, n)

    if mode == 0:
        q = a * np.exp(strength * 0.5 * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.2))
        q = (1.0 - t) * a + t / n
    elif mode == 2:
        sc, g = _subgrad_exact(a, k_active=3, tol=1e-6)
        if np.isfinite(sc):
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                q = a - strength * 0.2 * g / gn
            else:
                q = a.copy()
        else:
            q = a.copy()
    elif mode == 3:
        # Smooth low-freq noise
        K = min(31, max(5, n // 200))
        if K % 2 == 0:
            K -= 1
        if K >= 3:
            c = K // 2
            sig = max(1.0, c / 2.5)
            kern = np.exp(-((np.arange(K) - c) ** 2) / (2.0 * sig * sig))
            kern /= kern.sum()
            noise = np.convolve(rng.standard_normal(n), kern, mode="same")
            noise -= noise.mean()
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
        q = a * np.exp(strength * 0.3 * noise)
    elif mode == 4:
        # Cosine modulation
        f = int(rng.integers(1, 8))
        amp = float(rng.uniform(0.1, 0.5))
        q = a * (1.0 + amp * np.cos(2.0 * np.pi * f * x + rng.uniform(0, 2 * np.pi)))
    elif mode == 5:
        # Bimodal blend
        ctr = float(rng.uniform(0.15, 0.45))
        w = float(rng.uniform(0.1, 0.3))
        bimo = np.exp(-(((x - (0.5 - ctr)) / w) ** 2)) + np.exp(-(((x - (0.5 + ctr)) / w) ** 2))
        bimo = bimo / bimo.max()
        t = float(rng.uniform(0.2, 0.5))
        q = (1.0 - t) * a + t * bimo
    elif mode == 6:
        # Power law shaping
        alpha = float(rng.uniform(0.5, 2.0))
        shape = (1.0 - x + 1e-10) ** alpha + (x + 1e-10) ** alpha
        shape = shape / shape.mean()
        t = float(rng.uniform(0.3, 0.6))
        q = (1.0 - t) * a + t * shape
    else:
        # Random smooth via FFT
        Nf = _next_pow2(n)
        freqs = np.fft.rfftfreq(Nf)
        power = 1.0 / ((np.abs(freqs) * n + 1.0) ** float(rng.uniform(1.5, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - noise.mean()
        q = a * (1.0 + strength * 0.3 * noise)

    return _normalize(np.clip(q, 0.0, None))

            best_seq = _normalize(a2)

    # Final output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
