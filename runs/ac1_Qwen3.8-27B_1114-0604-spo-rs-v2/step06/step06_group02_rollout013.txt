```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative step heights.

Main changes versus the supplied predecessor:
  * Correct analytic gradient for weighted sums of convolution coefficients
      c_k(a) = sum_i a_i a_{k-i}
    is 2 * a_{k-j} when 0 <= k-j < n, summed over active/smoothed constraints.
  * Mirror-descent / multiplicative subgradient method using that correct gradient.
  * L-BFGS optimization of a log-sum-exp smooth approximation with the correct gradient.
  * Spectral square-root projection toward flatter polynomial-square spectra.
  * Multi-length, multi-start exploration with a hard real-time budget.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize
import warnings

warnings.filterwarnings("ignore")

# -----------------------------------------------------------------------------
# Global state for time limited L-BFGS objective calls.
# -----------------------------------------------------------------------------
_GLOBAL_RNG = np.random.default_rng(20250405)

_LBFGS_TIMEOUT = 0.0
_LBFGS_BETA = 100.0
_LBFGS_M2 = 0
_LBFGS_M3 = 0
_LBFGS_BEST_X = None
_LBFGS_BEST_F = float("inf")
_LBFGS_BEST_EXACT_P = None
_LBFGS_BEST_EXACT_F = float("inf")


# -----------------------------------------------------------------------------
# Basic helpers.
# -----------------------------------------------------------------------------
def _fft_len(m):
    if m <= 1:
        return 1
    m = int(m)
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    a = a / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s2


def _compute_b(a, mfft=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = 2 * n - 1
    if mfft is None or int(mfft) < N:
        mfft = _fft_len(N)
    mfft = int(mfft)
    A = np.fft.rfft(a, mfft)
    b = np.fft.irfft(A * A, mfft)[:N]
    b = np.maximum(b, 0.0)
    return b


def _score_norm(p, b=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return float("inf")
    if b is None:
        b = _compute_b(p)
    if b.size == 0:
        return float("inf")
    return float(2.0 * n * np.max(b))


def _score_general(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    b = _compute_b(a)
    return float(2.0 * n * np.max(b) / (s * s))


def _resample(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64) / max(1, new_n)
    if a.size == int(new_n):
        return _normalize(a)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    y = np.interp(x_new, x_old, a)
    return _normalize(y)


def _perturb_sequence(a, scale=0.05, rng=None):
    r = _GLOBAL_RNG if rng is None else rng
    a = _normalize(a)
    n = a.size
    if n <= 2:
        return a
    kind = int(r.integers(0, 4))
    if kind == 0:
        q = a * (1.0 + scale * r.standard_normal(n))
    elif kind == 1:
        x = np.arange(n, dtype=np.float64) / max(1.0, float(n - 1))
        f = int(r.integers(1, 8))
        ph = float(r.uniform(0.0, 2.0 * math.pi))
        q = a * (1.0 + 0.08 * np.cos(2.0 * math.pi * f * x + ph))
    elif kind == 2:
        bw = max(1, n // 40)
        bi = int(r.integers(0, n - bw))
        q = a.copy()
        q[bi:bi + bw] *= (1.0 + 0.25 * r.standard_normal(bw))
    else:
        q = a.copy()
        bw = max(1, n // 100)
        bi = int(r.integers(0, n - bw))
        q[bi:bi + bw] *= 0.2
        add_scale = max(float(np.max(a)), 1e-12) * 0.1
        q += r.uniform(0.0, add_scale, size=n)
    return _normalize(np.maximum(q, 0.0))


def _make_random_start(n, mode=0, rng=None):
    r = _GLOBAL_RNG if rng is None else rng
    n = int(max(1, n))
    if mode == 0:
        a = np.ones(n, dtype=np.float64)
    elif mode == 1:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - 0.7 * (2.0 * x - 1.0) ** 2
    elif mode == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(math.pi * x) ** 1.5 + 0.05
    elif mode == 3:
        white = r.standard_normal(n)
        F = np.fft.rfft(white)
        freqs = np.fft.rfftfreq(n)
        mask = 1.0 / (1.0 + (20.0 * np.abs(freqs)) ** 2)
        a = np.abs(np.fft.irfft(F * mask, n)) + 0.1
    else:
        x = np.linspace(0.0, 1.0, n)
        f = int(r.integers(1, 6))
        ph = float(r.uniform(0.0, 2.0 * math.pi))
        a = 1.0 + 0.25 * np.cos(2.0 * math.pi * f * x + ph)
    return _normalize(a)


# -----------------------------------------------------------------------------
# Constraint weights and exact weighted gradient.
#
# For c_k(a)=sum_i a_i a_{k-i},
# d/d a_j (sum_k w_k c_k) = 2 * sum_i a_i w_{i+j},
# where w_i is zero outside [0, 2n-2].
# -----------------------------------------------------------------------------
def _make_weights(b, beta, hard=False):
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0:
        return np.array([], dtype=np.float64)
    bmax = float(np.max(b))
    if not math.isfinite(bmax) or bmax <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    if hard or beta > 20000.0:
        tol = 1e-3 * bmax + 1e-18
        idx = np.flatnonzero(b >= bmax - tol)
        if idx.size == 0:
            idx = np.array([int(np.argmax(b))], dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx] = 1.0 / float(idx.size)
        return w

    if beta <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    x = beta * (b - bmax)
    x = np.clip(x, -50.0, 50.0)
    w = np.exp(x)
    wsum = float(w.sum())
    if wsum <= 0.0 or not math.isfinite(wsum):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / wsum


def _weighted_grad_direct(p, active_idx, active_w):
    n = p.size
    s = np.zeros(n, dtype=np.float64)
    for ii in range(active_idx.size):
        k = int(active_idx[ii])
        wk = float(active_w[ii])
        if wk == 0.0:
            continue
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        if end >= start:
            js = np.arange(start, end + 1, dtype=np.int64)
            s[js] += 2.0 * wk * p[k - js]
    return s


def _weighted_grad_fft(p, w, m3=None):
    n = p.size
    N = 2 * n - 1
    if m3 is None or int(m3) < n + N - 1:
        m3 = _fft_len(n + N - 1)
    m3 = int(m3)

    Fp = np.fft.rfft(p, m3)
    Fw = np.fft.rfft(w[::-1], m3)
    c = np.fft.irfft(Fp * Fw, m3)
    # A_j = sum_i p_i w_{i+j}; result index N-1-j for j=0..n-1.
    part = c[N - n:N][::-1]
    part = np.maximum(part, 0.0)
    return 2.0 * part


def _weighted_grad(p, w, m3=None):
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != 2 * n - 1:
        w = _make_weights(w, 1.0)
    wmax = float(w.max()) if w.size else 0.0
    threshold = max(1e-15, wmax * 1e-10)
    active = np.flatnonzero(w > threshold)
    if active.size == 0:
        return np.zeros(n, dtype=np.float64)

    # For very concentrated weights, direct summation is faster and exact enough.
    if active.size <= 80:
        return _weighted_grad_direct(p, active, w[active])
    return _weighted_grad_fft(p, w, m3)


# -----------------------------------------------------------------------------
# Mirror descent / multiplicative subgradient method on the exact max objective.
# -----------------------------------------------------------------------------
def _mirror_step(p, ratio, alpha):
    ratio = np.asarray(ratio, dtype=np.float64)
    ratio = np.clip(ratio, 1e-12, 1e12)
    logf = -float(alpha) * np.log(ratio)
    logf = np.clip(logf, -20.0, 20.0)
    q = p * np.exp(logf)
    return _normalize(q)


def _mirror_descent(a, budget_s, beta_start=50.0, beta_end=100000.0,
                    alpha_start=0.35, alpha_end=0.05):
    p = _normalize(a)
    n = p.size
    if n == 0:
        return p, float("inf")
    if budget_s < 3.0:
        return p, _score_norm(p)

    N = 2 * n - 1
    m2 = _fft_len(N)
    m3 = _fft_len(3 * n - 2)

    best_p = p.copy()
    b = _compute_b(best_p, m2)
    best_sc = float(2.0 * n * np.max(b))

    t0 = time.monotonic()
    t_end = t0 + max(1.0, float(budget_s))
    alpha = float(alpha_start)
    beta = float(beta_start)
    stuck = 0

    while time.monotonic() < t_end - 0.5:
        p = best_p.copy()
        b = _compute_b(p, m2)
        sc = float(2.0 * n * np.max(b))
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_p = p.copy()

        # Occasionally try small stochastic kicks to escape the current basin.
        if stuck > 30:
            improved = False
            for _ in range(5):
                cand = _perturb_sequence(best_p, scale=0.015 + 0.02 * np.random.rand())
                bc = _compute_b(cand, m2)
                scc = float(2.0 * n * np.max(bc))
                if scc < best_sc - 1e-15:
                    best_sc = scc
                    best_p = cand.copy()
                    improved = True
                    break
            stuck = 0
            alpha = float(alpha_start)
            if improved:
                beta = min(beta * 1.25, 500000.0)
                continue

        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / max(1e-6, float(budget_s)))

        if beta_start > 0.0:
            beta_target = beta_start * (beta_end / beta_start) ** prog
        else:
            beta_target = beta_end
        if beta < beta_target:
            beta = beta_target
        if stuck > 10:
            beta = min(beta * 1.25, 500000.0)

        w = _make_weights(b, beta, hard=(beta > 30000.0))
        s = _weighted_grad(p, w, m3)
        pos = s[s > 1e-300]
        if pos.size == 0:
            break
        scale = float(np.median(pos))
        if not math.isfinite(scale) or scale <= 0.0:
            scale = 1e-12
        ratio = s / scale
        ratio = np.clip(ratio, 1e-6, 1e6)

        alpha_target = float(alpha_start + (alpha_end - alpha_start) * prog)
        if alpha < 0.005:
            alpha = float(alpha_start)
        elif alpha < 0.3 * alpha_target:
            alpha = alpha_target

        accepted = False
        for al in (alpha, alpha * 0.5, alpha * 0.25):
            if not math.isfinite(al) or al <= 0.0:
                continue
            cand = _mirror_step(p, ratio, al)
            if float(np.sum(np.abs(cand - best_p))) <= 1e-14:
                continue
            bc = _compute_b(cand, m2)
            scc = float(2.0 * n * np.max(bc))
            if scc < best_sc - 1e-15:
                best_sc = scc
                best_p = cand.copy()
                alpha = min(al * 1.05, 1.0)
                accepted = True
                break

        if not accepted:
            stuck += 1
            alpha *= 0.75

    return best_p, best_sc


# -----------------------------------------------------------------------------
# Log-sum-exp L-BFGS.  The state is a nonnegative simplex vector x in [0,1];
# p = x / sum(x) is the normalized sequence actually scored.
# -----------------------------------------------------------------------------
def _lse_core_p(p, beta, m2=None, m3=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    b = _compute_b(p, m2)
    cmax = float(np.max(b)) if b.size else 0.0
    exact = float(2.0 * n * cmax)

    if beta <= 0.0:
        w = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        L = cmax
    else:
        scaled = beta * (b - cmax)
        scaled = np.clip(scaled, -50.0, 50.0)
        u = np.exp(scaled)
        usum = float(u.sum())
        if usum <= 0.0 or not math.isfinite(usum):
            u = np.ones_like(b, dtype=np.float64)
            usum = float(b.size)
        L = cmax + math.log(usum) / beta
        w = u / usum

    s = _weighted_grad(p, w, m3)
    lse_score = float(2.0 * n * L)
    grad = 2.0 * n * s
    return lse_score, grad, exact


def _lbfgs_objective(x):
    global _LBFGS_BEST_X, _LBFGS_BEST_F
    global _LBFGS_BEST_EXACT_P, _LBFGS_BEST_EXACT_F

    if time.monotonic() > _LBFGS_TIMEOUT:
        raise TimeoutError("L-BFGS timeout")

    x = np.asarray(x, dtype=np.float64).ravel()
    p = np.maximum(x, 0.0)
    S = float(p.sum())
    if S <= 1e-14:
        return 1e10, np.zeros_like(x)

    p = p / S
    beta = float(_LBFGS_BETA)
    lse_sc, gp, exact_sc = _lse_core_p(p, beta, _LBFGS_M2, _LBFGS_M3)

    if not math.isfinite(lse_sc):
        lse_sc = 1e10
        gp = np.zeros_like(p)
    if not math.isfinite(exact_sc):
        exact_sc = 1e10

    if lse_sc < _LBFGS_BEST_F:
        _LBFGS_BEST_F = lse_sc
        _LBFGS_BEST_X = p.copy()
    if exact_sc < _LBFGS_BEST_EXACT_F:
        _LBFGS_BEST_EXACT_F = exact_sc
        _LBFGS_BEST_EXACT_P = p.copy()

    dot = float(np.dot(p, gp))
    grad = (gp - p * dot) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return lse_sc, grad


def _lbfgs_run(a0, beta, deadline):
    global _LBFGS_TIMEOUT, _LBFGS_BETA, _LBFGS_M2, _LBFGS_M3
    global _LBFGS_BEST_X, _LBFGS_BEST_F
    global _LBFGS_BEST_EXACT_P, _LBFGS_BEST_EXACT_F

    n = int(np.asarray(a0).size)
    if n <= 0:
        return a0, float("inf")

    p0 = _normalize(a0)
    exact0 = _score_norm(p0)

    _LBFGS_TIMEOUT = float(deadline)
    _LBFGS_BETA = float(beta)
    _LBFGS_M2 = _fft_len(2 * n - 1)
    _LBFGS_M3 = _fft_len(3 * n - 2)
    _LBFGS_BEST_X = p0.copy()
    _LBFGS_BEST_F = float("inf")
    _LBFGS_BEST_EXACT_P = p0.copy()
    _LBFGS_BEST_EXACT_F = exact0

    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": 20000,
        "maxfun": 200000,
        "maxls": 12,
        "ftol": 1e-20,
        "gtol": 1e-8,
        "maxcor": 12,
    }

    x_out = None
    try:
        res = minimize(
            _lbfgs_objective,
            p0,
            jac=True,
            bounds=bounds,
            method="L-BFGS-B",
            options=opts,
        )
        x_out = res.x
    except Exception:
        x_out = None

    cands = []
    if _LBFGS_BEST_EXACT_P is not None:
        cands.append(_LBFGS_BEST_EXACT_P)
    if _LBFGS_BEST_X is not None:
        cands.append(_LBFGS_BEST_X)
    if x_out is not None:
        cands.append(x_out)

    best_p = p0.copy()
    best_sc = exact0
    for cand in cands:
        pc = _normalize(cand)
        sc = _score_norm(pc)
        if sc < best_sc:
            best_sc = sc
            best_p = pc
    return best_p, best_sc


def _lbfgs_schedule(a, deadline):
    p = _normalize(a)
    best_p = p.copy()
    best_sc = _score_norm(p)
    n = p.size

    if n <= 4096:
        betas = [20.0, 80.0, 300.0, 1500.0, 8000.0, 50000.0]
    elif n <= 12000:
        betas = [50.0, 300.0, 2000.0, 15000.0]
    else:
        betas = [100.0, 800.0, 6000.0, 40000.0]

    for beta in betas:
        now = time.monotonic()
        if now > deadline - 25.0:
            break
        remaining = deadline - now
        if remaining < 30.0:
            break
        alloc = min(170.0, remaining * 0.32)
        if alloc < 20.0:
            alloc = max(20.0, 0.5 * remaining)
        if alloc > remaining - 5.0:
            alloc = max(5.0, remaining - 5.0)

        p2, sc2 = _lbfgs_run(p, beta, now + alloc)
        if sc2 < best_sc:
            best_sc = sc2
            best_p = p2

    return best_p, best_sc

# Spectral square-root projection: push A(z)^2 toward flatter coefficients,
# then take a square-root in Fourier domain and project back to nonnegative a.
# -----------------------------------------------------------------------------
def _spectral_square_refine(a, budget_s, alpha_start=0.30):
    p = _normalize(a)
    n = p.size
    if n == 0 or budget_s < 10.0:
        return p, _score_norm(p)

    N = 2 * n - 1
    M = _fft_len(N)
    best_p = p.copy()
    b = _compute_b(best_p, M)
    best_sc = float(2.0 * n * np.max(b))

    t0 = time.monotonic()
    t_end = t0 + max(1.0, float(budget_s))
    alpha = float(alpha_start)
    stuck = 0

    while time.monotonic() < t_end - 0.5:
        p = best_p.copy()
        b = _compute_b(p, M)
        sc = float(2.0 * n * np.max(b))
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_p = p.copy()

        if stuck > 25:
            cand = _perturb_sequence(best_p, scale=0.05)
            cb = _compute_b(cand, M)
            cc = float(2.0 * n * np.max(cb))
            if cc < best_sc - 1e-15:
                best_sc = cc
                best_p = cand.copy()
            alpha = float(alpha_start)
            stuck = 0
            continue

        q_t = 0.70 * b + 0.30 * (1.0 / float(N))
        q_t = np.maximum(q_t, 0.0)
        q_sum = float(q_t.sum())
        if q_sum <= 0.0:
            q_t = np.full(N, 1.0 / N, dtype=np.float64)
        else:
            q_t = q_t / q_sum

        Q = np.fft.rfft(q_t, M)
        A = np.fft.rfft(p, M)
        P = np.sqrt(Q.astype(complex))
        A_new = (1.0 - alpha) * A + alpha * P
        p_new = np.fft.irfft(A_new, M)[:n]
        p_new = _normalize(np.maximum(p_new, 0.0))

        b_new = _compute_b(p_new, M)
        sc_new = float(2.0 * n * np.max(b_new))
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_p = p_new.copy()
            alpha = min(alpha * 1.05, 0.9)
        else:
            stuck += 1
            alpha *= 0.75
            if alpha < 0.03:
                alpha = float(alpha_start)

    return best_p, best_sc


# -----------------------------------------------------------------------------
# One full local optimization pipeline for a starting sequence.
# -----------------------------------------------------------------------------
def _optimize_start(a, end_time):
    p = _normalize(a)
    sc = _score_norm(p)
    now = time.monotonic()
    if now >= end_time - 5.0:
        return p, sc

    # Phase 1: multiplicative subgradient warm-start.
    remain = end_time - now
    budget = min(remain, remain * 0.25, 160.0)
    if budget > 15.0:
        p2, sc2 = _mirror_descent(p, budget, beta_start=50.0, beta_end=60000.0,
                                  alpha_start=0.35, alpha_end=0.05)
        if sc2 < sc:
            p, sc = p2, sc2

    # Phase 2: L-BFGS smooth approximation.
    now = time.monotonic()
    remain = end_time - now
    if remain > 60.0:
        lb_budget = min(remain * 0.72, 700.0)
        lb_end = now + lb_budget
        p2, sc2 = _lbfgs_schedule(p, lb_end)
        if sc2 < sc:
            p, sc = p2, sc2

    # Phase 3: final exact mirror polish.
    now = time.monotonic()
    remain = max(0.0, end_time - now)
    budget = min(remain * 0.30, 100.0)
    if budget > 10.0:
        p2, sc2 = _mirror_descent(p, budget, beta_start=800.0, beta_end=300000.0,
                                  alpha_start=0.12, alpha_end=0.02)
        if sc2 < sc:
            p, sc = p2, sc2

    # Phase 4: high-beta exact L-BFGS if time remains.
    now = time.monotonic()
    remain = max(0.0, end_time - now)
    if remain > 70.0:
        p2, sc2 = _lbfgs_run(p, 150000.0, end_time - 10.0)
        if sc2 < sc:
            p, sc = p2, sc2

    # Phase 5: spectral square-root escape/polish.
    now = time.monotonic()
    remain = max(0.0, end_time - now)
    budget = min(remain * 0.50, 75.0)
    if budget > 15.0:
        p2, sc2 = _spectral_square_refine(p, budget, alpha_start=0.25)
        if sc2 < sc:
            p, sc = p2, sc2

    # One last tiny exact mirror pass.
    now = time.monotonic()
    remain = max(0.0, end_time - now)
    if remain > 12.0:
        p2, sc2 = _mirror_descent(p, remain - 2.0, beta_start=5000.0, beta_end=500000.0,
                                  alpha_start=0.08, alpha_end=0.01)
        if sc2 < sc:
            p, sc = p2, sc2

    return p, sc

# Top-level search.
# -----------------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative height sequence with low objective value."""
    if budget_s is None:
        budget_s = 2400.0
    budget_s = float(budget_s)
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 2400.0

    rng = np.random.default_rng(int(seed) if seed is not None else 0)

    # Optional parent from the competition state.
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 16 and p.sum() > 1e-12:
                parent = _normalize(p)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(4096, dtype=np.float64)
        parent = _normalize(parent)

    # Keep final length conservative so evaluate_sequence's np.convolve is fast.
    MAX_FINAL_N = 18000
    if parent.size > MAX_FINAL_N:
        parent = _resample(parent, 15000)

    MAX_OPT_BUDGET = max(60.0, budget_s - 75.0)
    t0 = time.monotonic()
    opt_deadline = t0 + MAX_OPT_BUDGET

    best_a = parent.copy()
    best_sc = _score_norm(best_a)
    print(f"Initial parent: n={best_a.size}, internal_score={best_sc:.8f}")

    def update_best(a, label):
        nonlocal best_a, best_sc
        if a is None:
            return
        ap = _normalize(a)
        ap = np.clip(ap, 0.0, 1000.0)
        s = _score_norm(ap)
        if s < best_sc:
            best_sc = s
            best_a = ap.copy()
            elapsed = time.monotonic() - t0
            print(f"  [{label}] NEW BEST: {s:.8f} (n={ap.size}, t={elapsed:.0f}s)")

    # Choose start lengths.
    parent_len = int(best_a.size)
    n2 = min(MAX_FINAL_N, max(4096, 2 * parent_len))
    if n2 == parent_len:
        n2 = min(MAX_FINAL_N, parent_len + 2048)
    n3 = 8192
    if n3 == parent_len or n3 == n2:
        n3 = 12000
    if n3 > MAX_FINAL_N:
        n3 = MAX_FINAL_N

    starts = []
    starts.append(("parent", best_a))
    starts.append(("parent-resample", _resample(best_a, n2)))
    starts.append(("explore-random", _make_random_start(n3, 3, rng)))

    # Time allocation.  If a parent exists, spend the most time polishing it.
    if parent_len > 64:
        fracs = [0.55, 0.28, 0.17]
    else:
        fracs = [0.40, 0.35, 0.25]

    acc = 0.0
    for i, (label, a0) in enumerate(starts):
        now = time.monotonic()
        end1 = t0 + (acc + fracs[i]) * MAX_OPT_BUDGET
        end2 = t0 + (acc + fracs[i + 1]) * MAX_OPT_BUDGET if i + 1 < len(starts) else opt_deadline
        end_time = max(now + 20.0, min(end1, end2))
        if now >= end_time - 20.0:
            break

        print(f"\n=== Start {i + 1}: {label}, n={a0.size}, budget end={end_time - now:.0f}s ===")
        p_out, sc_out = _optimize_start(a0, end_time)
        update_best(p_out, f"{label}-optimized")
        acc += fracs[i]

    # Final safety: one more global best check and a final tiny high-beta polish.
    now = time.monotonic()
    if now < t0 + MAX_OPT_BUDGET - 25.0:
        p_out, sc_out = _mirror_descent(best_a, max(10.0, t0 + MAX_OPT_BUDGET - now - 10.0),
                                        beta_start=10000.0, beta_end=700000.0,
                                        alpha_start=0.06, alpha_end=0.005)
        update_best(p_out, "final-tiny-mirror")

    out_a = _normalize(best_a)
    out_a = np.clip(out_a, 0.0, 1000.0)
    out_list = [float(x) for x in out_a]

    if sum(out_list) < 0.01:
        out_list = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = float("inf")

    print("\n" + "=" * 70)
    print(f"FINAL internal : {best_sc:.10f}")
    print(f"FINAL verified : {verified:.10f}")
    print(f"Length          : {len(out_list)}")
    print(f"Elapsed         : {time.monotonic() - t0:.1f}s / {budget_s:.1f}s")
    print("=" * 70)

    if not math.isfinite(verified) or verified > 1e6:
        return [0.5] * 64
    return out_list
```
