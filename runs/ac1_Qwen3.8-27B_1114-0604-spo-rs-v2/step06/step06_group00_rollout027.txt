def fast_conv(a):

    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta
    
    p = exp_shifted / exp_shifted.sum()
    
    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    
    score = (2.0 * n * L) / S2
    return score, grad

def run_lbfgs(a0, beta, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0

def subgradient_peak_shave(a, n_iters=3000, top_k=10, step_init=0.005, decay=0.9998):
    """Subgradient descent on top-K peaks with Polyak step."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if sc - lower_bound < 1e-8:
            break
        
        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_polyak = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.02)
        step = max(step, 1e-12)
        
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a

def fienup_flat(a, n_iters=200):
    """Fienup spectrum flattening."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.2 + 0.6 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/S^2.
Combines: approximate LP-direction (damage-based), LSE-annealing with L-BFGS-B,
subgradient peak shaving, Fienup spectral flattening, diverse restarts,
and n-exploration. Focuses on beating 1.5027 target.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a
    return a / S


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -600, 600)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta

    p = exp_shifted / exp_shifted.sum()

    a_rev = a[::-1]
    M3 = (2 * n - 1) + n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    PA = np.fft.rfft(p, M3_fft)
    RA = np.fft.rfft(a_rev, M3_fft)
    conv_res = np.fft.irfft(PA * RA, M3_fft)[:M3]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    score = (2.0 * n * L) / S2
    return score, grad


def run_lbfgs(a0, beta, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def lp_direction_damage(a, M_target, n_iters=200, lr=0.05):
    """
    Approximate LP direction using damage-based multiplicative updates.
    Finds b >= 0 that maximizes sum(b) subject to conv(a,b) <= M_target.
    """
    n = len(a)
    log_b = np.zeros(n)  # b starts at 1

    for it in range(n_iters):
        b = np.exp(log_b)
        ab = conv_ab(a, b)
        excess = np.maximum(ab - M_target, 0)
        if excess.max() < 1e-12 * M_target:
            break

        # Compute grad: d(sum_k excess[k])/db[j] = sum_k I[ab[k]>M] * a[k-j]
        # This is conv(I[ab>M], a_reversed)
        e_ind = (ab > M_target).astype(np.float64)
        a_rev = a[::-1]
        M3 = len(e_ind) + n - 1
        M3_fft = 1 << (M3 - 1).bit_length()
        E = np.fft.rfft(e_ind, M3_fft)
        AR = np.fft.rfft(a_rev, M3_fft)
        grad_b = np.fft.irfft(E * AR, M3_fft)[:M3]
        grad_b = grad_b[n - 1:2 * n - 1]

        # Multiplicative update in log space
        b_safe = np.maximum(b, 1e-15)
        log_b -= lr * grad_b / b_safe
        log_b = np.minimum(log_b, 0.0)

    b = np.exp(log_b)
    b = np.maximum(b, 1e-15)
    return b / b.sum()


def lp_optimize_step(a, n_lp_iters=300, blend=0.3):
    """
    One LP-based optimization step:
    Find b with conv(a,b) <= max(conv(a,a)), then blend a toward b.
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())

    b = lp_direction_damage(a, M, n_iters=n_lp_iters)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return a

    # Scale b to match a's sum
    S = a.sum()
    b = b * (S / b_sum)

    a_new = (1.0 - blend) * a + blend * b
    a_new = np.maximum(a_new, 0.0)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a
    a_new = a_new / S_new
    return a_new


def subgradient_peak_shave(a, n_iters=3000, top_k=8, step_init=0.006, decay=0.9998):
    """Subgradient descent targeting top-K autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if sc - lower_bound < 1e-8:
            break

        top_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        total_weight = 0.0
        for k in top_idx:
            k = int(k)
            w = c[k] / (M_val + 1e-30)
            total_weight += w
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        avg_w = total_weight / len(top_idx)
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * avg_w / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step_polyak = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, step_polyak, 0.025)
        step = max(step, 1e-12)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def fienup_flat(a, n_iters=300):
    """Fienup spectrum flattening."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.15 + 0.6 * progress
        P_target = np.full_like(P, P_mean)
        P_blend = (1.0 - alpha) * P + alpha * P_target
        P_blend = np.maximum(P_blend, 1e-30)
        ratio = np.sqrt(P_blend / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=300):
    """Fienup with quantile-based spectral clipping."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        q = 0.70 + 0.25 * progress
        thresh = np.quantile(P, q) * (1.0 + 0.3 * (1.0 - progress))
        P_clipped = P.copy()
        above = P_clipped > thresh
        P_clipped[above] = thresh
        ratio = np.sqrt(P_clipped / np.maximum(P, 1e-30))
        A_new = A * ratio
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        a = a_new
        sc = compute_score(a)

            best_sc = sc
            best_a = a.copy()

def coordinate_descent(a, n_sweeps=8, n_top=40):
    """Coordinate descent: reduce top contributors to peak lag."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(n_top, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_val = a_old
            best_local = best_sc
            for f in [0.3, 0.5, 0.7, 0.8, 0.9]:
                a[j] = a_old * f
                a_n = normalize(a)
                sc = compute_score(a_n)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    if time.monotonic() < phase2_end - 10:
        remaining = phase2_end - 10 - time.monotonic()
        maxiter = max(500, int(remaining * 8))
        sc, a_opt = run_lbfgs(a_work, 2000, maxiter=maxiter, maxfun=maxiter * 20)
        update(sc, a_opt, "LP+LSE")

    # ==================================================================
    # Phase 3: LSE Annealing (25% time)
    # ==================================================================
    phase3_end = t_start + 0.56 * total_budget
    print(f"\n=== Phase 3: LSE Annealing (budget {phase3_end - t_start:.0f}s) ===")

    a_lse = best_a.copy()
    betas = [200, 500, 1000, 2000, 4000, 8000, 15000, 25000, 50000]
    for i, beta in enumerate(betas):
        if time.monotonic() > phase3_end - 15:
            break
        remaining = phase3_end - 15 - time.monotonic()
        if remaining < 12:
            break
        # Spend more time at mid-range betas
        weight = 1.0 / (1.0 + 0.4 * abs(i - 4))
        maxiter = max(400, int(remaining * 10 * weight))
        maxfun = maxiter * 25
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_lse, beta, maxiter=maxiter, maxfun=maxfun)
        dt = time.monotonic() - t0
        print(f"  LSE beta={beta:>6}: score={sc:.8f} ({dt:.1f}s)")
        update(sc, a_opt, f"LSE b={beta}")
        a_lse = a_opt.copy()

    # ==================================================================
    # Phase 4: Subgradient peak shaving (8% time)
    # ==================================================================
    phase4_end = t_start + 0.64 * total_budget
    print(f"\n=== Phase 4: Peak shaving (budget {phase4_end - t_start:.0f}s) ===")

    for top_k in [5, 10, 15]:
        if time.monotonic() > phase4_end - 10:
            break
        remaining_iters = max(1000, int((phase4_end - 10 - time.monotonic()) * 2500))
        remaining_iters = min(remaining_iters, 4000)
        a_ps = subgradient_peak_shave(best_a, n_iters=remaining_iters, top_k=top_k,
                                      step_init=0.005)
        sc = compute_score(a_ps)
        update(sc, a_ps, f"PSAVE k={top_k}")

        if sc < best_score and time.monotonic() < phase4_end - 8:
            sc2, a2 = run_lbfgs(a_ps, 5000, maxiter=1500, maxfun=30000)
            update(sc2, a2, f"PSAVE+LSE k={top_k}")

    # ==================================================================
    # Phase 5: Coordinate descent (5% time)
    # ==================================================================
    phase5_end = t_start + 0.69 * total_budget
    print(f"\n=== Phase 5: Coordinate Descent ===")

    a_cd = coordinate_descent(best_a, n_sweeps=8, n_top=30)
    sc = compute_score(a_cd)
    update(sc, a_cd, "COORD")

    if time.monotonic() < phase5_end - 8:
        sc2, a2 = run_lbfgs(a_cd, 8000, maxiter=2000, maxfun=40000)
        update(sc2, a2, "COORD+LSE")

    # ==================================================================
    # Phase 6: Diverse restarts (18% time)
    # ==================================================================
    phase6_end = t_start + 0.87 * total_budget
    print(f"\n=== Phase 6: Diverse restarts (budget {phase6_end - t_start:.0f}s) ===")

    n_restarts = 25
    for r in range(n_restarts):
        if time.monotonic() > phase6_end - 15:
            break
        remaining = phase6_end - 15 - time.monotonic()
        if remaining < 20:
            break

        strategy = r % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.008 + 0.004 * (r % 6)
            a_pert = best_a + scale * rng.standard_normal(len(best_a))
            a_pert = normalize(a_pert)
        elif strategy == 1:
            # Multiplicative low-freq structure
            n = len(best_a)
            x = np.arange(n, dtype=np.float64) / n
            pert = np.zeros(n)
            for m in range(1, 10):
                freq = rng.integers(1, 40)
                phase_val = 2.0 * np.pi * rng.uniform(0, 1)
                pert += (0.015 / m) * np.cos(2.0 * np.pi * freq * x + phase_val)
            a_pert = best_a * (1.0 + pert)
            a_pert = normalize(a_pert)
        elif strategy == 2:
            # LP-based perturbation
            c = fast_conv(best_a)
            M = float(c.max())
            b = lp_direction_damage(best_a, M * 1.05, n_iters=150)
            a_pert = 0.75 * best_a + 0.25 * b
            a_pert = normalize(a_pert)
        elif strategy == 3:
            # Window perturbation (Gaussian bump)
            n = len(best_a)
            center = rng.uniform(0.1, 0.9) * n
            width = rng.uniform(0.1, 0.5) * n
            x = (np.arange(n) - center) / width
            w = 1.0 + 0.2 * np.exp(-x ** 2)
            a_pert = best_a * w
            a_pert = normalize(a_pert)
        else:
            # Focused: add to non-peak areas
            n = len(best_a)
            c = fast_conv(best_a)
            k_peak = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_peak - n + 1)
            j_end = min(k_peak, n - 1)
            peak_mask = np.ones(n, dtype=bool)
            peak_mask[j_start:j_end + 1] = False
            non_peak_idx = np.where(peak_mask)[0]
            n_add = max(50, n // 15)
            if len(non_peak_idx) > n_add:
                add_idx = rng.choice(non_peak_idx, n_add, replace=False)
            else:
                add_idx = non_peak_idx
            delta = 0.003 / n
            a_pert[add_idx] += delta
            a_pert = normalize(a_pert)

        # Optimize with LSE
        remaining_opt = remaining - 8
        if remaining_opt < 10:
            continue
        beta = 1000 + r * 400
        maxiter = max(300, int(remaining_opt * 5))
        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxiter * 18)
        update(sc, a_opt, f"RE {r} s{strategy}")

        # Follow-up with LP if close to best
        if sc < best_score + 0.002 and time.monotonic() < phase6_end - 12:
            a_lp = lp_optimize_step(a_opt, n_lp_iters=200, blend=0.2)
            sc_lp = compute_score(a_lp)
            update(sc_lp, a_lp, f"RE {r}+LP")

    # ==================================================================
    # Phase 7: n-exploration (7% time)
    # ==================================================================
    phase7_end = t_start + 0.94 * total_budget
    print(f"\n=== Phase 7: n-exploration (budget {phase7_end - t_start:.0f}s) ===")

    current_n = len(best_a)
    target_ns = []
    if current_n < 12000:
        target_ns.append(12000)
    if current_n > 8000:
        target_ns.append(8000)
    if current_n < 15000:
        target_ns.append(15000)

    for new_n in target_ns:
        if time.monotonic() > phase7_end - 25:
            break
        remaining = phase7_end - 25 - time.monotonic()
        if remaining < 25:
            break

        a_large = interpolate_sequence(best_a, new_n)
        a_large = normalize(a_large)
        sc = compute_score(a_large)
        print(f"  n={new_n}: score={sc:.8f}")
        update(sc, a_large, f"n={new_n}")

        if time.monotonic() < phase7_end - 12:
            maxiter = max(400, int((phase7_end - 12 - time.monotonic()) * 5))
            sc2, a2 = run_lbfgs(a_large, 3000, maxiter=maxiter, maxfun=maxiter * 18)
            update(sc2, a2, f"n={new_n}+LSE")

            if sc2 < best_score and time.monotonic() < phase7_end - 10:
                a_f = fienup_flat(a2, n_iters=100)
                sc3 = compute_score(a_f)
                update(sc3, a_f, f"n={new_n}+Fienup")

    # ==================================================================
    # Phase 8: Final polish (6% time)
    # ==================================================================
    print(f"\n=== Phase 8: Final polish ===")

    # High-beta LSE
    if time.monotonic() < deadline - 25:
        remaining = deadline - 25 - time.monotonic()
        maxiter = max(500, int(remaining * 7))
        sc, a_final = run_lbfgs(best_a, 30000, maxiter=maxiter, maxfun=maxiter * 15)
        update(sc, a_final, "FINAL-LSE-30k")

    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        maxiter = max(400, int(remaining * 6))
        sc, a_final = run_lbfgs(best_a, 80000, maxiter=maxiter, maxfun=maxiter * 12)
        update(sc, a_final, "FINAL-LSE-80k")

    # Final peak shave
    if time.monotonic() < deadline - 8:
        remaining_iters = max(500, int((deadline - 8 - time.monotonic()) * 2500))
        a_ps = subgradient_peak_shave(best_a, n_iters=min(remaining_iters, 3000),
                                       top_k=5, step_init=0.003)
        sc = compute_score(a_ps)
        update(sc, a_ps, "FINAL-PSAVE")

    # Final coordinate descent
    if time.monotonic() < deadline - 4:
        a_cd = coordinate_descent(best_a, n_sweeps=5, n_top=25)
        sc = compute_score(a_cd)
        update(sc, a_cd, "FINAL-CD")

    # Final LP step
    if time.monotonic() < deadline - 2:
        a_lp = lp_optimize_step(best_a, n_lp_iters=200, blend=0.15)
        sc = compute_score(a_lp)
        update(sc, a_lp, "FINAL-LP")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
