    m = int(max(1, m))

    """Compute full linear self-convolution via FFT."""

    else:
        lam = amask.astype(np.float64) / amask.sum()

    h = _xkkt(a, lam)
    h = np.maximum(h, 1e-15)
    return h, lam

            w = np.zeros(c.size, dtype=np.float64)
            w[idx] = 1.0 / kk
            h = _xkkt(a, w)
            for eta in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
                if time.monotonic() > dl - 3.0:
                    break
                q = a - eta * h / (S * S)
                q = _norm(np.maximum(q, 0.0))
                sc = _score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best = q.copy()
                    improved = True
                if np.isfinite(sc) and sc < _score(a) - 1e-10:
                    a = q
        if not improved and it > 10:
            break
    return float(best_sc), best


def _gen_inits(nn, rng, parent):
    """Generate diverse initial sequences of length nn."""
    inits = []
    x = np.linspace(0.0, 1.0, nn)

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != nn:
            arr = _rsmp(arr, nn)
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e4, neginf=0.0)
        mn = float(arr.min())
        if mn < 0:
            arr = arr - mn + 0.01
        inits.append(_norm(arr))

    if parent is not None:
        pr = _rsmp(parent, nn)
        add(pr)
        add(pr[::-1])

    add(np.ones(nn))
    for al in (0.3, 0.5, 0.8, 1.2, 2.0):
        add(1.0 + al * np.abs(2.0 * x - 1.0))
        add(1.0 + al * (2.0 * x - 1.0) ** 2)
    for f in (1, 2, 3, 5):
        for al in (0.3, 0.7, 1.0, 1.5):
            add(1.0 + al * np.cos(2.0 * np.pi * f * x))
    for wd in (0.08, 0.12, 0.2, 0.3):
        add(np.exp(-((x - 0.5) / wd) ** 2) + 0.1)
        add(np.exp(-((x - 0.25) / wd) ** 2) + np.exp(-((x - 0.75) / wd) ** 2) + 0.1)
    for rt in (1.0, 3.0, 7.0):
        add(np.exp(-rt * x) + np.exp(-rt * (1.0 - x)) + 0.1)
    for f in (1, 2, 3, 4, 7):
        add(1.0 + 0.4 * np.sin(np.pi * f * x))
        add(1.0 + 0.4 * np.abs(np.sin(np.pi * f * x)))
    # Smooth random
    Nf = _np2(nn)
    for _ in range(4):
        fr = np.fft.rfftfreq(Nf) * Nf
        pw = 1.0 / ((np.abs(fr) + 1.0) ** float(rng.uniform(1.5, 4.0)))
        pw[0] = 0.0
        ph = rng.uniform(-np.pi, np.pi, size=len(pw))
        nz = np.fft.irfft(pw * np.exp(1j * ph), Nf)
        nn2 = nz[:nn]
        nn2 = nn2 - nn2.mean()
        add(1.0 + float(rng.uniform(0.2, 0.8)) * nn2)
    return inits


def _perturb(a, rng, s=None):
    p = _norm(a)
    n = p.size
    if n == 0:
        return p
    if s is None:
        s = float(rng.uniform(0.05, 0.3))
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(s * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.2))
        q = (1 - t) * p + t * np.ones(n) / n
    elif mode == 2:
        c = _sconv(p)
        cmax = c.max()
        if cmax > 1e-15:
            idx = np.argpartition(-c, 4)[:5]
            w = np.zeros(c.size)
            w[idx] = 1.0 / 5
            h = _xkkt(p, w)
            h = np.maximum(h, 1e-15)
            q = p * np.exp(-s * 0.5 * h / h.max())
        else:
            q = p.copy()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        q = p * (1.0 + s * 0.3 * np.abs(2 * x - 1))
    elif mode == 4:
        k = min(max(3, n // 300) | 1, n if n % 2 == 1 else n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sg = max(1.0, c / 2.0)
        kn = np.exp(-((np.arange(k) - c) ** 2) / (2 * sg * sg))
        kn /= kn.sum()
        q = np.convolve(p, kn, mode='same')
    else:
        q = p * np.exp(s * 0.5 * rng.standard_normal(n))
        q = np.maximum(q, 0.0)
    return _norm(q)


def propose_candidate(seed=42, budget_s=2400):
    """Main search function. Returns best sequence found within budget."""
    t0 = time.monotonic()
    total = float(budget_s)
    dl = t0 + total - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e4, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    n_main = 10240
    if parent is not None:
        n_main = min(parent.size, 16384)
        if n_main < 2048:
            n_main = 2048

    print("Parent: {} len={}".format("yes" if parent is not None else "no",
                                      parent.size if parent is not None else 0))
    print("Main length: {}".format(n_main))

    # === PHASE 1: Coarse optimization (10% time) ===
    t1 = t0 + 0.10 * total
    n_coarse = max(256, n_main // 8)
    a_c = _norm(_rsmp(parent, n_coarse)) if parent is not None else np._ones(n_coarse) / n_coarse
    x = np.linspace(0, 1, n_coarse)
    a_c = _norm(1.0 + 0.8 * np.abs(2.0 * x - 1.0))
    if parent is not None:
        a_c = _norm(_rsmp(parent, n_coarse))

    sc_c, a_c = _lbfgs(a_c, 500.0, t1, maxfev=200)
    sc_c, a_c = _run_md(a_c, t1, 50, 0.1)
    sc_c, a_c = _lbfgs(a_c, 2000.0, min(t1, time.monotonic() + 30), maxfev=150)
    print("Coarse (n={}): {:.8f}".format(n_coarse, sc_c))

    # Upsample
    a_med = _norm(_rsmp(a_c, n_main // 2))
    if parent is not None:
        a_med = 0.5 * a_med + 0.5 * _norm(_rsmp(parent, n_main // 2))
        a_med = _norm(a_med)

    # === PHASE 2: Medium resolution (15% time) ===
    t2 = t0 + 0.25 * total
    sc_m, a_m = _lbfgs(a_med, 2000.0, t2, maxfev=300)
    sc_m, a_m = _run_md(a_m, t2, 80, 0.1)
    sc_m, a_m = _lbfgs(a_m, 10000.0, t2, maxfev=200)
    print("Medium (n={}): {:.8f}".format(n_main // 2, sc_m))

    # Upsample to full
    a_full = _norm(_rsmp(a_m, n_main))
    if parent is not None:
        a_par = _norm(_rsmp(parent, n_main))
        a_full = _norm(0.5 * a_full + 0.5 * a_par)

    best_sc = _score(a_full)
    best_a = a_full.copy()
    print("Full init (n={}): {:.8f}".format(n_main, best_sc))

    # === PHASE 3: Full resolution LSE optimization (20% time) ===
    t3 = t0 + 0.45 * total
    a_work = a_full.copy()
    for beta in (500.0, 2000.0, 5000.0, 10000.0, 30000.0, 80000.0):
        if time.monotonic() > t3 - 5.0:
            break
        rem = t3 - time.monotonic()
        if rem < 10.0:
            break
        sc, a2 = _lbfgs(a_work, beta, t3)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  LSE beta={:.0f}: {:.8f}".format(beta, best_sc))
        a_work = _norm(a2)

    # === PHASE 4: FFT mirror descent (35% time) ===
    t4 = t0 + 0.80 * total
    a_work = best_a.copy()
    for afrac in (0.05, 0.1, 0.15, 0.2, 0.3):
        if time.monotonic() > t4 - 10.0:
            break
        sc, a2 = _run_md(a_work, t4, 100, afrac)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  MD afrac={:.2f}: {:.8f}".format(afrac, best_sc))
        a_work = _norm(a2)

    # Interleave with LSE
    for beta in (50000.0, 100000.0, 200000.0):
        if time.monotonic() > t4 - 10.0:
            break
        rem = t4 - time.monotonic()
        if rem < 8.0:
            break
        sc, a2 = _lbfgs(best_a, beta, t4)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  LSE beta={:.0f} (MD phase): {:.8f}".format(beta, best_sc))
        a_work = _norm(a2)
        if time.monotonic() < t4 - 5.0:
            sc, a2 = _run_md(a_work, min(t4 - 5.0, time.monotonic() + 60), 40, 0.1)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                a_work = _norm(a2)

    # === PHASE 5: Subgradient polish (10% time) ===
    t5 = t0 + 0.90 * total
    sc, a2 = _subgrad_polish(best_a, t5, 60)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a2.copy()
        print("  Subgrad: {:.8f}".format(best_sc))

    # === PHASE 6: Perturbation restarts (5% time) ===
    t6 = dl - 10.0
    n_pert = 0
    while time.monotonic() < t6 and n_pert < 3:
        left = t6 - time.monotonic()
        if left < 20.0:
            break
        chunk = time.monotonic() + left / (3 - n_pert) - 3.0
        p = _perturb(best_a, rng)
        for beta in (20000.0, 100000.0):
            if time.monotonic() > chunk:
                break
            sc, a2 = _lbfgs(p, beta, chunk, maxfev=150)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  Pert {0} b={1:.0f}: {2:.8f}".format(n_pert, beta, best_sc))
            p = _norm(a2)
        if time.monotonic() < chunk - 2.0:
            sc, a2 = _run_md(p, chunk, 30, 0.1)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
        n_pert += 1

    # === PHASE 7: Final cleanup (5%) ===
    if time.monotonic() < dl - 5.0:
        sc, a2 = _run_md(best_a, dl - 5.0, 30, 0.05)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        sc, a2 = _subgrad_polish(best_a, dl - 3.0, 20)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    # Final output
    out = np.asarray(best_a, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out /= S
    result = [float(v) for v in out]

    try:
        ver = float(evaluate_sequence(result))
    except Exception:
        ver = best_sc
    print("FINAL: internal={:.8f} verified={:.8f} n={} t={:.1f}s".format(
        best_sc, ver, len(result), time.monotonic() - t0))
    return result
