"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses multiplicative subgradient, adaptive LSE L-BFGS, and block coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA_rev = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA_rev * PA, M2_fft)[:M2]
    h = h_full[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=400000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

    Multiplicative updates using exact subgradient of max(c).
    Natural handling of non-negativity.

    a = np.maximum(a.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init
    M_fft = 1 << (2 * n - 1).bit_length()

        S = float(a.sum())
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        
        if sc < best_sc:

            best_a = a.copy()

def fienup_annealed(a, n_iters=500):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        alpha = 0.7 * (1.0 - progress) ** 2 + 0.03
        P_mean = float(P.mean())
        P_blend = (1 - alpha) * P + alpha * P_mean
        P_blend = np.maximum(P_blend, 1e-20)

        A_new = A * (np.sqrt(P_blend / (P + 1e-30)) ** alpha)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def block_coordinate_descent(a, n_sweeps=30, block_size=32, deadline=None):
    """Block CD with FFT-accelerated updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n)/n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    n_blocks = n // block_size + (1 if n % block_size else 0)
    
    for sweep in range(n_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        for b in range(n_blocks):
            if deadline and time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            if start >= n:
                break
            
            a_block = a[start:end].copy()
            
            # For this block, the score depends on a through the convolution
            # We want to minimize max(c) by adjusting a_block
            # This is a small subproblem
            
            # Simple approach: try multiplying the block by various factors
            best_mult = 1.0
            best_block_sc = sc
            
            for mult in [0.5, 0.7, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.3, 1.5]:
                a_test = a.copy()
                a_test[start:end] = a_block * mult
                S_t = a_test.sum()
