"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based block coordinate descent with constraint generation
2. LSE (Log-Sum-Exp) smoothing with L-BFGS-B for smooth refinement
3. Nesterov accelerated exact subgradient for non-smooth polishing
4. Multi-phase strategy with adaptive time allocation
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _fast_cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    M = n + m - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _lp_solve(f, M, max_constraints=200, deadline=None):
    """
    Solve the LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Uses constraint generation (cutting plane) approach.
    
    Returns: g (optimal solution), or None if failed.
    """
    n = len(f)
    
    # Ensure f has some positive entries
    f = np.maximum(f, 0.0)
    S_f = f.sum()
    if S_f < 1e-15:
        return None
    
    # Upper bound for g components
    f_max = f.max()
    if f_max < 1e-15:
        return None
    g_bound = M / f_max * 1.01  # Generous upper bound
    
    # Start with the peak constraint
    c_f = _fast_self_conv(f)
    k_peak = int(np.argmax(c_f))
    active = [k_peak]
    
    g = None
    for outer_iter in range(50):
        if deadline is not None and time.monotonic() > deadline - 5.0:
            break
        
        # Build constraint matrix for active set
        A_ub_list = []
        for k in active:
            row = np.zeros(n)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                row[i] = f[k - i]
            A_ub_list.append(row)
        
        if not A_ub_list:
            # No constraints: g is unbounded, set to uniform
            g = np.ones(n) / n
            break
        
        A_ub = np.array(A_ub_list)
        b_ub = M * np.ones(len(active))
        c_obj = -np.ones(n)
        
        bounds = [(0.0, g_bound)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            # Try with fewer constraints
            if len(active) > 10:
                # Keep only the most violated ones
                continue
            break
        
        g = res.x
        
        # Check all constraints
        h = _fast_cross_conv(f, g)
        violations = h > M * (1 + 1e-8)
        
        if not np.any(violations):
            break  # Feasible solution found
        
        # Add most violated constraints
        viol_idx = np.where(violations)[0]
        viol_vals = h[viol_idx]
        # Add top violations (up to 10 at a time)
        top_n = min(10, len(viol_idx))
        top_idx = viol_idx[np.argsort(viol_vals)[-top_n:]]
        
        for idx in top_idx:
            if int(idx) not in active:
                active.append(int(idx))
        
        # Cap on active constraints
        if len(active) > max_constraints:
            # Keep only the most violated
            active = [int(k) for k in np.argsort(h)[-max_constraints:]]
    
    if g is not None:
        g = np.clip(g, 0.0, None)
        S_g = g.sum()
        if S_g > 1e-15:
            g = g / S_g  # Normalize to sum 1
    return g


def _lp_block_step(f, deadline=None, max_constraints=200):
    """
    One block coordinate descent step using LP.
    Returns (new_f, score, g) or (f, score, None) if LP failed.
    """
    f = _normalize(np.asarray(f, dtype=np.float64).ravel())
    n = f.size
    score = _exact_score(f)
    
    c_f = _fast_self_conv(f)
    M = float(c_f.max())
    
    g = _lp_solve(f, M, max_constraints=max_constraints, deadline=deadline)
    
    if g is None:
        return f, score, None
    
    g = _normalize(g)
    
    # Line search: try different t values
    best_score = score
    best_f = f.copy()
    best_t = 0.0
    
    for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
        f_new = _normalize((1 - t) * f + t * g)
        s = _exact_score(f_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_f = f_new
            best_t = t
    
    return best_f, best_score, g


def _lse_value_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _fast_self_conv(a)
    d = n * c / (S * S)
    dmax = float(d.max())

    z = beta * (d - dmax)
    z = np.clip(z, -400, 400)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)

    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    score = 2.0 * L

    T = float(w.dot(c))

    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]

    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def _lbfgs_lse(a0, beta, deadline, maxfun=500):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = _normalize(a0)
    n = a.size

    def fun(x):
        x = np.clip(x, 0.0, None)
        S = float(x.sum())
        if S < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / S
        val, grad = _lse_value_grad(xn, beta)
        return val, grad

    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxiter": max(maxfun // 2, 50),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True,
                       args=(), bounds=bounds, options=opts)
    except Exception:
        return _exact_score(a), a

    if getattr(res, "x", None) is not None:
        a_out = _normalize(res.x)
    else:
        a_out = a

    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out
