```python
"""
Hybrid optimization for minimum peak-to-average self-convolution sequences.
Combines LSE-LBFGS, LP-based peak shaving, adaptive GS, and multi-resolution refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_shifted = scaled - scaled.max()
    exp_s = np.exp(scaled_shifted)
    log_sum = scaled.max() + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g
    
    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    return np.clip(res.x, 0.0, 1000.0)

def lp_peak_shave(a, n_peaks=40, target_reduction=0.02, max_iters=20):
    """
    LP-based peak shaving: iteratively solve LP to reduce top peaks.
    Returns improved sequence.
    """
    n = len(a)
    S = float(a.sum())
    a = np.maximum(a, 0.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(max_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        S = float(a.sum())
        
        # Top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Target: reduce the maximum by target_reduction
        T_target = c_max * (1.0 - target_reduction)
        
        # LP: minimize T s.t. linearized peak constraints
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for k_i in top_idx:
            k_i = int(k_i)
            row = np.zeros(n)
            for m in range(n):
                km = k_i - m
                if 0 <= km < n:
                    row[m] = a[km]
            # Constraint: sum_m a_{k_i-m} * a'_m - T <= c(k_i)/2
            A_ub_row = np.zeros(n_vars)
            A_ub_row[:n] = row
            A_ub_row[-1] = -1.0
            A_ub_list.append(A_ub_row)
            b_ub_list.append(float(c[k_i]) / 2.0)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        # Equality: sum(a') = S
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        a_new = res.x[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new * S  # Renormalize to same sum
        
        # Check actual improvement
        sc_new = compute_score(a_new)
        
        # Trust region check: if the solution moved too much, scale back
        diff = np.linalg.norm(a_new - a)
        avg_val = S / n
        if diff > 2.0 * avg_val:
            # Scale back
            for alpha in [0.5, 0.3, 0.1]:
                a_try = a + alpha * (a_new - a)
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try * S
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                    break
            else:
                break
        else:
            if sc_new < best_sc:
                a = a_new
                best_a = a_new.copy()
                best_sc = sc_new
            else:
                # No improvement, try smaller reduction
                target_reduction *= 0.5
                if target_reduction < 0.001:
                    break
    
    return best_a

    n = len(a)

"""
```

```python
Combines high-beta LSE-LBFGS, adaptive GS with peak-aware spectral shaping,
LP-based peak shaving with trust region, and diverse multi-start exploration.
Novel: LP peak shaving, adaptive GS with dynamic peak-aware targets, and

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-6:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    scaled_shifted = scaled - max_scaled
    exp_s = np.exp(scaled_shifted)
    log_sum = max_scaled + float(np.log(exp_s.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g * S - 2.0 * L) / S3
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=50000):
    """Run L-BFGS-B optimization on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def wrapped(x):
        sc, g = lse_score_grad(x, beta)
        return sc, g

    res = minimize(
        wrapped, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 80}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out


def gs_peak_aware(a, n_iters=500, n_peaks=30, alpha_base=0.15, alpha_end=0.03):

    to be flat except at current peak positions, where it is reduced.
    This directly attacks the peaks while maintaining smoothness elsewhere.
    """

    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())

        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_min = float(np.min(c[c > 0])) if np.any(c > 0) else c.min()

        # Build adaptive target

        target = np.ones(2 * n - 1) / (2 * n - 1)

        # At peak positions, reduce the target (push these down)

                reduction = min(0.8, 0.3 * (ratio / 2.0))

        target[below_mean] *= 1.1

        target = np.maximum(target, 1e-10)

        T_mag = np.maximum(np.abs(T), 1e-30)
        T_phase = np.angle(T)

        alpha = alpha_base + (alpha_end - alpha_base) * progress

        A = np.fft.fft(a, M)

        A_phase = np.angle(A)

        a_new = np.fft.ifft(A_new, M).real[:n]

def lp_peak_shave(a, n_peaks=30, reduction=0.01, max_iters=15, verbose=False):
    """
    LP-based peak shaving with trust region.
    At each iteration, linearize the convolution at top peaks and solve an LP
    to find a new sequence that reduces the peak. Use trust region to ensure
    the linearization is valid.
    """
    n = len(a)
    S = float(a.sum())
    a = np.maximum(a, 0.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        # Identify top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # LP formulation:
        # min T
        # s.t. for each peak k_i: sum_m a_{k_i-m} * x_m - T <= c(k_i)/2
        #      sum(x) = S
        #      x >= 0, T >= 0
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0

        A_ub_rows = []
        b_ub_vals = []

        for k_i in top_idx:
            k_i = int(k_i)
            row = np.zeros(n)
            # row[m] = a[k_i - m] if 0 <= k_i - m < n
            m_vals = np.arange(n)
            km = k_i - m_vals
            valid = (km >= 0) & (km < n)
            row[valid] = a[km[valid]]
            A_ub_row = np.zeros(n_vars)
            A_ub_row[:n] = row
            A_ub_row[-1] = -1.0
            A_ub_rows.append(A_ub_row)
            b_ub_vals.append(float(c[k_i]) / 2.0)

        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception:
            break

        if not res.success:
            break

        a_new = np.maximum(res.x[:n], 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new * S

        # Trust region: check if the step is reasonable
        step_norm = np.linalg.norm(a_new - a) / (S / n + 1e-30)

        if step_norm > 3.0:
            # Scale back
            alpha = 3.0 / step_norm
            a_new = a + alpha * (a_new - a)
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                break
            a_new = a_new / S_new * S

        sc_new = compute_score(a_new)

        if sc_new < best_sc:
            a = a_new
            best_a = a.copy()
            best_sc = sc_new
            if verbose:
                print(f"    LP iter {it}: score={best_sc:.8f}, T_lp={res.x[-1]:.6f}")
        else:
            # Try a smaller step
            for alpha in [0.5, 0.3, 0.1]:
                a_try = a + alpha * (a_new - a)
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try < 1e-12:
                    continue
                a_try = a_try / S_try * S
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                    break
            else:
                # No improvement, reduce aggressiveness
                reduction *= 0.5
                if reduction < 0.001:
                    break

    return best_a


def spectral_shaping(a, n_iters=300, n_attenuate=50):
    """
    Direct spectral shaping: identify frequency bins that contribute most to
    the peak convolution and attenuate them.
    """
    n = len(a)
    M = 1 << (2 * n - 2).bit_length()

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    F = np.fft.fft(a, M)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())

        # Compute |F|^2
        power = np.abs(F) ** 2
        total_power = power.sum()

        # The contribution of frequency bin k to c(m) is power[k] * exp(2j*pi*k*m/M)
        # Peaks in c are caused by constructive interference of certain frequency bins
        # Simply attenuate the largest frequency bins (top n_attenuate)
        top_freqs = np.argsort(power)[::-1][:n_attenuate]

        # Attenuate
        attenuation = 1.0 - 0.05 * (1.0 - it / n_iters) - 0.02
        for k in top_freqs:
            F[k] *= np.sqrt(attenuation)
            F[M - k] *= np.sqrt(attenuation)  # Conjugate symmetry

        a_new = np.fft.ifft(F, M).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        a = a_new
        F = np.fft.fft(a, M)

        if it % 50 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.4 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2.0 * np.pi * 3 * x) + 0.3 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.2 + 0.6 * (1.0 - 2.0 * x) ** 2)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 5.0))

    # Random Fourier
    for _ in range(3):
        n_harm = rng.integers(3, 12)
        weights = rng.standard_normal(n_harm)
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        sig = np.zeros(n)
        for i in range(n_harm):
            sig += weights[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
        inits.append(np.maximum(0.5 + 0.3 * sig, 0.01))

    for ini in inits:
        ini = np.maximum(ini, 0.0)
        S = ini.sum()
        if S > 1e-12:
            ini = ini / S
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(a*a)/(sum(a))^2.
    Uses adaptive GS, LSE-LBFGS, LP peak shaving, and spectral shaping.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_time = float(budget_s)
    deadline = t_start + total_time - 40.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_a = parent.astype(np.float64).copy()
    else:
        n = 7000
        best_a = np.ones(n) * 0.5

    best_score = compute_score(best_a)
    print(f"Starting: n={n}, score={best_score:.8f}")

    def update_best(a_candidate):
        nonlocal best_a, best_score
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        if a_candidate.sum() < 0.01:
            return
        sc = compute_score(a_candidate)
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.8f} (t={elapsed:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: GS warmup on best init (8% of time)
    # ============================================================
    p1_end = t_start + 0.08 * (total_time - 40.0)
    print(f"\n=== Phase 1: GS warmup ===")

    a_gs = gs_peak_aware(best_a, n_iters=300, n_peaks=30, alpha_base=0.2, alpha_end=0.03)
    update_best(a_gs)

    if time.monotonic() < p1_end:
        a_gs2 = gs_flat_momentum(best_a, n_iters=300)
        update_best(a_gs2)

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-LBFGS with increasing beta (35% of time)
    # ============================================================
    p2_end = t_start + 0.43 * (total_time - 40.0)
    print(f"\n=== Phase 2: LSE-LBFGS ===")

    # Start with moderate beta, increase for finer optimization
    beta_schedule = [500, 2000, 8000, 30000, 80000, 200000]
    for beta in beta_schedule:
        if time.monotonic() > p2_end:
            break
        t0 = time.monotonic()
        # Use more iterations for lower beta (coarser), fewer for high beta
        max_it = max(3000, 15000 - beta // 20)
        max_fun = max(10000, 50000 - beta // 5)
        a_opt = run_lbfgs(best_a, beta, maxiter=max_it, maxfun=max_fun)
        sc = compute_score(a_opt)
        elapsed_beta = time.monotonic() - t0
        print(f"  LBFGS beta={beta}: score={sc:.8f} ({elapsed_beta:.1f}s)")
        update_best(a_opt)

    # Try from a slightly perturbed version
    if time.monotonic() < p2_end:
        a_pert = best_a + 0.01 * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum() * best_a.sum()
        a_opt = run_lbfgs(a_pert, 50000, maxiter=5000, maxfun=20000)
        sc = compute_score(a_opt)
        print(f"  LBFGS perturbed: score={sc:.8f}")
        update_best(a_opt)

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: LP peak shaving (20% of time)
    # ============================================================
    p3_end = t_start + 0.63 * (total_time - 40.0)
    print(f"\n=== Phase 3: LP Peak Shaving ===")

    lp_configs = [
        (20, 0.015, 20),
        (30, 0.010, 20),
        (50, 0.008, 15),
        (40, 0.005, 20),
        (60, 0.004, 15),
    ]
    for n_peaks, reduction, max_it in lp_configs:
        if time.monotonic() > p3_end:
            break
        t0 = time.monotonic()
        a_lp = lp_peak_shave(best_a, n_peaks=n_peaks, reduction=reduction,
                            max_iters=max_it, verbose=True)
        sc = compute_score(a_lp)
        elapsed_lp = time.monotonic() - t0
        print(f"  LP (peaks={n_peaks}, red={reduction:.3f}): score={sc:.8f} ({elapsed_lp:.1f}s)")
        update_best(a_lp)

        # Quick LSE polish after each LP
        if time.monotonic() < p3_end:
            a_pol = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=10000)
            update_best(a_pol)

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Spectral shaping + GS (15% of time)
    # ============================================================
    p4_end = t_start + 0.78 * (total_time - 40.0)
    print(f"\n=== Phase 4: Spectral shaping + GS ===")

    a_spec = spectral_shaping(best_a, n_iters=300, n_attenuate=40)
    sc = compute_score(a_spec)
    print(f"  Spectral shaping: score={sc:.8f}")
    update_best(a_spec)

    if time.monotonic() < p4_end:
        a_gs3 = gs_peak_aware(best_a, n_iters=400, n_peaks=40, alpha_base=0.12, alpha_end=0.02)
        update_best(a_gs3)

    if time.monotonic() < p4_end:
        a_gs4 = gs_flat_momentum(best_a, n_iters=300, momentum=0.6)
        update_best(a_gs4)

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish - interleaved (12% of time)
    # ============================================================
    p5_end = deadline
    print(f"\n=== Phase 5: Final polish ===")

    polish_iters = 15
    for it in range(polish_iters):
        if time.monotonic() > p5_end:
            break

        # Short GS
        a_gs = gs_peak_aware(best_a, n_iters=100, n_peaks=25, alpha_base=0.08, alpha_end=0.015)
        update_best(a_gs)

        if time.monotonic() > p5_end:
            break

        # Short LP
        a_lp = lp_peak_shave(best_a, n_peaks=25, reduction=0.004, max_iters=8)
        update_best(a_lp)

        if time.monotonic() > p5_end:
            break

        # Short LSE
        beta = 100000 + 20000 * it
        a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=10000)
        update_best(a_opt)

        if it % 3 == 0:
            print(f"  Polish iter {it}: best={best_score:.8f}")

    # ============================================================
    # Phase 6: Very high beta LSE final push
    # ============================================================
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 6: Final high-beta push ===")
        for beta in [300000, 500000]:
            if time.monotonic() > deadline - 10:
                break
            t0 = time.monotonic()
            a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=15000)
            sc = compute_score(a_opt)
            print(f"  Final LBFGS beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")
            update_best(a_opt)

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * n

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
