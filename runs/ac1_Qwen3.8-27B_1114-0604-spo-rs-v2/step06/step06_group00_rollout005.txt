def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_score_and_grad, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        if a_out.sum() < 1e-6:
            a_out = a0
        return compute_score(a_out), a_out
    except Exception:
        return compute_score(a0), a0


def fista_optimize(a0, beta, n_iters=5000, gamma_init=0.05):
    """Nesterov-accelerated proximal gradient on LSE-smoothed objective."""
    n = len(a0)
    x = normalize(a0.copy())
    y = x.copy()
    t_k = 1.0
    gamma = gamma_init
    best_sc = compute_score(x)
    best_a = x.copy()
    
    for k in range(n_iters):
        # Extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + (t_k - 1.0) / t_new * (x - y)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy < 1e-15:
            break
        y = y / Sy
        
        # Gradient at y
        sc_y, grad_y = lse_score_and_grad(y, beta)
        
        # Backtracking line search
        gnorm2 = float(grad_y @ grad_y)
        if gnorm2 < 1e-30:
            break
        gamma = min(gamma * 1.5, 0.5 / np.sqrt(gnorm2))  # L-smoothness adaptive
        
        for bt in range(30):
            a_trial = y - gamma * grad_y
            a_trial = np.maximum(a_trial, 0.0)
            Sa = a_trial.sum()
            if Sa < 1e-15:
                gamma *= 0.5
                continue
            a_trial = a_trial / Sa
            sc_trial, _ = lse_score_and_grad(a_trial, beta)
            if sc_trial <= sc_y - 0.5 * gamma * gnorm2:
                break
            gamma *= 0.5
        
        x_new = y - gamma * grad_y
        x_new = np.maximum(x_new, 0.0)
        Sx = x_new.sum()
        if Sx < 1e-15:
            break
        x_new = x_new / Sx
        
        x = x_new
        t_k = t_new
        
        # Evaluate objective (expensive, do less often)
        if k % 50 == 0 or k == n_iters - 1:
            sc_x = compute_score(x)
            if sc_x < best_sc:
                best_sc = sc_x
                best_a = x.copy()
    
    sc_final = compute_score(x)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = x.copy()
    
    return best_sc, best_a


def quadratic_peak_deflate(a0, n_iters=2000, top_k=3):
    """
    Quadratic approximation of the peak reduction problem.
    For the top-K peaks, find the 2nd-order optimal step.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find active peaks (within 1% of max)
        threshold = M_val * 0.995
        active = np.where(c >= threshold)[0]
        active = active[:top_k * 3]  # cap
        if len(active) == 0:
            break
        
        # Build subgradient from active peaks
        sg = np.zeros(n)
        for k_idx in active:
            k_idx = int(k_idx)
            w = c[k_idx] / (M_val + 1e-30)
            j_start = max(0, k_idx - n + 1)
            j_end = min(k_idx, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        avg_w = len(active)  # for the subgradient of max
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        # Line search with quadratic model
        gnorm2 = float(grad @ grad)
        if gnorm2 < 1e-30:
            break
        
        # Try a few step sizes
        best_step = 0.0
        best_trial_sc = sc
        base_steps = 0.02 / np.sqrt(gnorm2 + 1e-30)
        for f in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            step = base_steps * f
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / max(a_trial.sum(), 1e-15)
            sc_trial = compute_score(a_trial)
            if sc_trial < best_trial_sc:
                best_trial_sc = sc_trial
                best_step = step
        
        if best_step == 0:
            break
        
        a = a - best_step * grad
        a = np.maximum(a, 0.0)
        a = a / max(a.sum(), 1e-15)
    
    return best_a


def fienup_quick(a, n_iters=200):
    """Quick Fienup spectrum flattening."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        alpha = 0.2 + 0.6 * (it / n_iters)
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 1e-30)
        ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
        A_new = A * ratio
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def coordinate_polish(a, n_sweeps=5):
    """Coordinate descent: reduce top contributors to peak."""
    n = len(a)
    a = normalize(a.copy())
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        js = np.arange(j_start, j_end + 1)
        contribs = a[js] * a[k_star - js]
        top_count = min(20, len(contribs))
        top_local = np.argsort(contribs)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            if a_old < 1e-14:
                continue
            best_local = best_sc
            best_val = a_old
            for f in [0.3, 0.5, 0.7, 0.85, 0.95]:
                a[j] = a_old * f
                a_tmp = normalize(a)
                sc = compute_score(a_tmp)
                a[j] = a_old
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * f
            a[j] = best_val
            a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def perturb_spectral(a, rng, strength=0.02):
    """Perturb in frequency domain for structured diversity."""
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    # Add random phase perturbation
    phase = rng.uniform(-strength, strength, size=len(A))
    A_pert = A * np.exp(1j * phase)
    a_new = np.fft.irfft(A_pert, M_fft)[:n]
    return normalize(a_new)


def perturb_additive(a, rng, strength=0.01):
    """Additive Gaussian perturbation."""
    a_new = a + strength * rng.standard_normal(len(a))
    return normalize(a_new)
