def _fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def _score(a):
    n = len(a)
    c = _fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def _lse_objective_grad(a, beta):
    n = len(a)
    c = _fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -500, 500)
    exp_s = np.exp(scaled)
    L = c_max + float(np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient w.r.t. a
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a_rev, M2_fft)
    h_full = np.fft.irfft(AA * PA, M2_fft)[:M2]
    h = h_full[n-1:2*n-1]
    S2 = S * S
    grad = 2.0 * n * (h / S2 - 2.0 * L / (S2 * S))
    return score, grad
