def lse_smooth_score_grad(a, beta):
    a = a.astype(float)
    S = a.sum()
    if S<1e-12: return inf, zeros(a.shape)
    p = a/S
    c = conv(p)
    cmax = c.max()
    shifted = beta*(c-cmax)
    shifted = np.clip(shifted, -700, 700)
    ex = np.exp(shifted)
    exsum = ex.sum()
    L = cmax + np.log(exsum)/beta
    W = ex/exsum
    Ncorr = 1 << (len(c)+len(p)-2).bit_length()
    Wpad = np.zeros(Ncorr); Wpad[:len(c)] = W
    ppad = np.zeros(Ncorr); ppad[:len(p)] = p
    corr = np.fft.irfft(np.fft.rfft(Wpad,Ncorr) * np.conj(np.fft.rfft(ppad,Ncorr)), Ncorr)
    r = corr[:len(p)].real
    gradF = 4.0 * len(p) * r  # for normalized F
    # Optionally include gradient of normalization? If we use p fixed sum1, direction tangent.
    # If caller updates p and normalizes, fine.
    return 2*len(p)*L, gradF, c, W

"""
Asymmetric anti-convolution search for step-function heights.

The routine searches for non-negative coefficients minimizing
    2*n*max_k ((a*a)[k]) / (sum a)^2
using Fienup-free active-equilibrium updates in the time domain, LSE-smoothed
subgradients, and a short softmax L-BFGS polish.  It starts from a broad
library of asymmetric ramp/power/exponential shapes, which are much better
than the symmetric local minima of many naive spectral methods.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_len(m):
    """Next power of two >= m (at least 1)."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def linear_conv_full(a):
    """Full linear convolution a*a via FFT, with tiny negative roundoff removed."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    N = fft_len(m)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:m]
    neg = c < 0.0
    if neg.any():
        c[neg] = 0.0
    return c


def compute_score(a):
    """Internal evaluation of 2*n*max(a*a)/(sum a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S <= 1e-300:
        return np.inf
    c = linear_conv_full(a)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    if maxc < 0.0:
        maxc = 0.0
    return float(2.0 * n * maxc / (S * S))


def normalize_nonnegative(a):
    """Clip to non-negative finite reals and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a[a < 0.0] = 0.0
    S = float(np.sum(a))
    n = a.size
    if S <= 1e-300:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / S


def correlation_weights_p(w, p):
    """
    Return R_i = sum_j p_j w_{i+j}.

    w has length 2n-1 and p has length n.  This is the valid cross-correlation
    w(p) needed for gradients of convolution coefficients.
    """
    n = int(len(p))
    m = int(len(w))
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    N = fft_len(m + n - 1)
    W = np.zeros(N, dtype=np.float64)
    P = np.zeros(N, dtype=np.float64)
    W[:m] = w
    P[:n] = p
    Wr = np.fft.rfft(W, N)
    Pr = np.fft.rfft(P, N)
    corr = np.fft.irfft(Wr * np.conjugate(Pr), N)
    return corr[:n].copy()


def make_weights(c, mode, beta, topk):
    """
    Build nonnegative weights over convolution coefficients.

    Modes:
      'top'   : uniform weights on top `topk` coefficients.
      'lse'   : log-sum-exp weights, weight_k ∝ exp(beta*c_k).
      'mixed' : half top-uniform + half LSE.
    """
    m = int(len(c))
    if m == 0:
        return np.zeros(0, dtype=np.float64)

    W = np.zeros(m, dtype=np.float64)
    use_top = (mode == "top" or mode == "mixed")
    if use_top:
        k = int(topk)
        if k <= 0:
            k = 1
        if k > m:
            k = m
        if k == 1:
            idx = np.array([int(np.argmax(c))], dtype=np.int64)
        elif k < m:
            idx = np.argpartition(c, -k)[-k:]
        else:
            idx = np.arange(m)
        W[idx] = 1.0 / float(len(idx))

    use_lse = (mode == "lse" or mode == "mixed")
    if use_lse:
        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            W.fill(1.0 / m)
            return W
        beta = float(beta)
        if beta <= 0.0:
            ex = np.ones(m, dtype=np.float64)
        else:
            shifted = beta * (c - cmax)
            shifted = np.clip(shifted, -745.0, 700.0)
            ex = np.exp(shifted)
        s = float(np.sum(ex))
        if s <= 0.0 or not np.isfinite(s):
            ex = np.ones(m, dtype=np.float64)
            s = float(m)
        else:
            ex = ex / s
        if mode == "lse":
            W = ex
        elif mode == "mixed":
            W = 0.5 * W + 0.5 * ex

    s = float(np.sum(W))
    if s <= 0.0 or not np.isfinite(s):
        W = np.ones(m, dtype=np.float64) / float(m)
    else:
        W = W / s
    return W


def active_equilibrium_step(a, w, alpha, log_clip=2.0):
    """
    Multiplicative equilibrium step.

    For active/lse weights w, compute R_i = sum_j a_j w_{i+j}.  For a convex
    active objective, the preferred stationary condition is R_i approximately
    constant on the support.  This step moves log(a) toward that condition.
    """
    R = correlation_weights_p(w, a)
    R = np.maximum(R, 0.0)
    n = a.size

    mask = a > 1e-14
    if not np.any(mask):
        return a.copy()

    Rpos = R[mask]
    if Rpos.size == 0:
        return a.copy()

    maxR = float(np.max(Rpos))
    minR = float(np.min(Rpos))
    floor = max(maxR * 1e-12, 1e-300)
    if minR < floor:
        floor = max(minR, 1e-300) if minR > 0.0 else 1e-300

    Rsafe = np.maximum(R, floor)
    logR = np.log(Rsafe)

    amask = np.maximum(a, 0.0)
    smask = float(np.sum(amask[mask]))
    if smask <= 0.0:
        loglam = float(np.mean(logR))
    else:
        loglam = float(np.average(logR[mask], weights=amask[mask]))

    loga = np.full(n, -300.0, dtype=np.float64)
    loga[mask] = np.log(np.maximum(a[mask], 1e-300))

    logf = float(alpha) * (loglam - logR)
    logf = np.clip(logf, -float(log_clip), float(log_clip))
    lognew = loga + logf
    lognew = np.clip(lognew, -300.0, 50.0)
    anew = np.exp(lognew)
    anew[~np.isfinite(anew)] = 0.0

    S = float(np.sum(anew))
    if S <= 1e-300:
        return a.copy()
    anew = anew / S

    small = anew < 1e-16
    if small.any():
        anew[small] = 1e-16
        anew = anew / np.sum(anew)

    return anew


def active_equilibrium_optimize(a0, max_time=1200.0, seed=1, verbose=True):
    """
    Run a schedule of active-equilibrium multiplicative updates.

    Keeps a running exact best and tries several top/LSE weight mixes.
    """
    if max_time is None or max_time <= 5.0:
        a = normalize_nonnegative(a0)
        return a, compute_score(a)

    a = normalize_nonnegative(a0)
    best_a = a.copy()
    best_sc = compute_score(a)
    if verbose:
        print(f"  active start score: {best_sc:.10f}, n={a.size}")

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)
    n = a.size

    # (mode, beta_factor, topk, alpha, log_clip)
    params = [
        ("top",   0.0,    3, 0.25, 2.30),
        ("top",   0.0,    5, 0.18, 2.20),
        ("top",   0.0,    8, 0.14, 2.10),
        ("top",   0.0,   12, 0.11, 2.00),
        ("top",   0.0,   20, 0.09, 1.90),
        ("mixed", 10.0,   5, 0.20, 2.30),
        ("mixed", 40.0,  10, 0.16, 2.10),
        ("mixed", 150.0, 20, 0.12, 2.00),
        ("mixed", 600.0, 50, 0.09, 1.90),
        ("lse",   10.0,   0, 0.20, 2.30),
        ("lse",   50.0,   0, 0.15, 2.10),
        ("lse",  200.0,   0, 0.11, 2.00),
        ("lse",  800.0,   0, 0.07, 1.80),
    ]

    for mode, beta_factor, topk, alpha, log_clip in params:
        if time.monotonic() > deadline:
            break

        cur_a = best_a.copy()
        cur_sc = best_sc
        beta = float(beta_factor) * float(n) if beta_factor > 0.0 else 0.0
        stall = 0
        restarts = 0

        for _ in range(100000):
            if time.monotonic() > deadline:
                break

            c = linear_conv_full(cur_a)
            W = make_weights(c, mode, beta, topk)
            cand = active_equilibrium_step(cur_a, W, alpha, log_clip)
            cand_sc = compute_score(cand)

            if cand_sc < cur_sc - 1e-13:
                cur_a = cand
                cur_sc = cand_sc
                stall = 0
                if cur_sc < best_sc:
                    best_a = cur_a.copy()
                    best_sc = cur_sc
                    if verbose and (best_sc < 1e3) and (cur_sc < -1.0 or best_sc < 1.5 or cur_sc < 1.5):
                        rel = time.monotonic() - t0
                        print(f"    active new best: {best_sc:.10f} mode={mode} beta={beta_factor} "
                              f"topk={topk} alpha={alpha} t={rel:.1f}s")
            else:
                stall += 1
                if stall > 40:
                    restarts += 1
                    if restarts > 6:
                        break
                    noise = 0.015 + 0.008 * float(restarts)
                    pert = best_a * (1.0 + noise * rng.standard_normal(n))
                    cur_a = normalize_nonnegative(pert)
                    cur_sc = compute_score(cur_a)
                    stall = 0

    if verbose:
        print(f"  active end best: {best_sc:.10f}")
    return best_a, float(best_sc)


def initial_candidate_list(n, seed=2024):
    """
    Generate a diverse set of asymmetric, non-negative initial shapes.

    The asymmetry is important: symmetric sequences are poor for this objective
    because the central convolution coefficient becomes sum(a_i^2).
    """
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    inv = 1.0 - t
    raw = []

    # Overshoot-free linear ramps.  offset=0 is near the optimal continuous ramp.
    for off in (0.0, 0.05, 0.10, 0.25, 0.5, 1.0, 2.0):
        raw.append(t + off)

    # Power laws.
    for g in (0.3, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0):
        raw.append(t ** g)
        raw.append(inv ** g)

    # Asymmetric beta-like shapes.
    for aa, bb in ((2, 1), (1, 2), (3, 1), (1, 3),
                   (4, 1), (1, 4), (4, 2), (2, 4),
                   (3, 2), (2, 3), (5, 2), (2, 5),
                   (5, 3), (3, 5)):
        raw.append(t ** aa * inv ** bb)
        raw.append(inv ** aa * t ** bb)

    # Exponentials.
    for g in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        raw.append(np.exp(g * t))
        raw.append(np.exp(-g * t))

    # Low-degree asymmetric polynomials.
    for c in (0.5, 1.0, 2.0):
        raw.append(t + c * t * t)
        raw.append(t + c * inv * inv)

    # A few smooth random asymmetric curves.
    rng = np.random.default_rng(seed)
    for _ in range(8):
        f1 = int(rng.integers(1, 6))
        f2 = int(rng.integers(1, 12))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a1 = float(rng.uniform(0.2, 0.8))
        a2 = float(rng.uniform(0.1, 0.5))
        tr = float(rng.uniform(-0.5, 0.5))
        q = (1.0
             + a1 * np.sin(2.0 * np.pi * f1 * t + p1)
             + a2 * np.sin(2.0 * np.pi * f2 * t + p2)
             + tr * t)
        raw.append(np.maximum(q, 1e-3))

    out = []
    q = raw
    for arr in q:
        arr = np.asarray(arr, dtype=np.float64)
        arr = np.maximum(arr, 1e-9)
        arr = normalize_nonnegative(arr)
        out.append(arr)
    return out


def add_candidate(candidates, arr, label, max_len=20000):
    """Evaluate arr and append (score, normalized_array, label) if valid."""
    if arr is None:
        return
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0 or a.size > int(max_len):
        return
    a = normalize_nonnegative(a)
    sc = compute_score(a)
    if np.isfinite(sc):
        candidates.append((float(sc), a.copy(), str(label)))


def candidate_key(item):
    return float(item[0])


def softmax_stable(x):
    """Numerically stable softmax for finite x."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    y = x - float(np.max(x))
    y = np.clip(y, -745.0, 700.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def lse_softmax_grad(x, beta_factor):
    """
    Value and gradient of LSE-smoothed objective with respect to logits x.

    x parameterizes p = softmax(x).
    """
    n = x.size
    p = softmax_stable(x)
    c = linear_conv_full(p)
    if c.size == 0:
        return 1e100, np.zeros(n, dtype=np.float64)

    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return 1e100, np.zeros(n, dtype=np.float64)

    beta = float(beta_factor) * float(n)
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -745.0, 700.0)
    ex = np.exp(shifted)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        W = np.ones(c.size, dtype=np.float64) / float(c.size)
        L = cmax
    else:
        W = ex / s
        L = cmax + float(np.log(s)) / beta

    R = correlation_weights_p(W, p)
    gp = 4.0 * float(n) * R
    G = float(np.dot(p, gp))
    gx = p * (gp - G)

    val = 2.0 * float(n) * L
    if not np.isfinite(val):
        val = 1e100
    gx[~np.isfinite(gx)] = 0.0
    return float(val), gx


def run_lbfgs_softmax(a0, beta_factor, maxfun=120):
    """Populate a sequence with short softmax-L-BFGS polish on LSE objective."""
    p0 = normalize_nonnegative(a0)
    x0 = np.log(np.maximum(p0, 1e-300))
    finite = np.isfinite(x0)
    if finite.any():
        x0[finite] = x0[finite] - float(np.median(x0[finite]))

    try:
        res = minimize(
            lse_softmax_grad,
            x0,
            args=(float(beta_factor),),
            method="L-BFGS",
            jac=True,
            options={
                "maxiter": 120,
                "maxfun": int(maxfun),
                "ftol": 1e-18,
                "gtol": 1e-14,
                "maxls": 50,
            },
        )
        x1 = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x1 = x0

    p1 = softmax_stable(x1)
    p1 = normalize_nonnegative(p1)
    sc = compute_score(p1)
    return float(sc), p1


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search entry point.

    Returns a Python list of non-negative floats.
    """
    if seed is None:
        seed = 42
    if budget_s is None:
        budget_s = 2400

    total = min(float(budget_s), 2400.0)
    t_start = time.monotonic()
    # Leave a generous finalization margin for the official evaluator.
    deadline = t_start + max(60.0, total - 180.0)

    candidates = []

    # Parent sequence, if present and reasonably sized.
    try:
        parent = height_sequence_1
    except Exception:
        parent = None
    if parent is not None:
        add_candidate(candidates, parent, "parent", 20000)

    # Main asymmetric shape library.  8192 is fast for the official direct
    # np.convolve check; 16384 gives a slightly denser limit and is still cheap.
    for n_len in (8192, 16384):
        inits = initial_candidate_list(n_len, seed)
        for a in inits:
            add_candidate(candidates, a, f"init n={n_len}", 20000)

    # Guaranteed fallback.
    if not candidates:
        n = 8192
        a = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        add_candidate(candidates, a, "ramp fallback", 20000)

    candidates.sort(key=candidate_key)
    best_a = np.asarray(candidates[0][1], dtype=np.float64).copy()
    best_score = float(candidates[0][0])
    print(f"Initial best: {best_score:.10f}, n={best_a.size}, "
          f"t={time.monotonic() - t_start:.1f}s")

    # Free candidate list memory.
    candidates = None

    # Main active-equilibrium refinement.
    if best_a.size <= 20000:
        remaining = deadline - time.monotonic()
        if remaining > 300.0:
            max_time = min(1200.0, remaining - 300.0)
            best_a, best_score = active_equilibrium_optimize(
                best_a, max_time=max_time, seed=int(seed) + 1, verbose=True
            )

    # Second continuation from the current best with different RNG.
    if best_a.size <= 20000:
        remaining = deadline - time.monotonic()
        if remaining > 300.0:
            max_time = min(700.0, remaining - 300.0)
            best_a, best_score = active_equilibrium_optimize(
                best_a, max_time=max_time, seed=int(seed) + 101, verbose=True
            )

    # Smooth LSE / softmax L-BFGS polish.
    if best_a.size <= 20000:
        for beta_factor in (80.0, 300.0):
            if time.monotonic() > deadline - 120.0:
                break
            remaining = deadline - time.monotonic()
            if remaining < 120.0:
                break
            sc_l, a_l = run_lbfgs_softmax(best_a, beta_factor, maxfun=120)
            if sc_l < best_score:
                best_a = a_l.copy()
                best_score = float(sc_l)
                print(f"  L-BFGS new best: {best_score:.10f}, beta_factor={beta_factor}")

    # Final small active polish.
    if best_a.size <= 20000:
        remaining = deadline - time.monotonic()
        if remaining > 120.0:
            max_time = min(300.0, remaining - 60.0)
            best_a, best_score = active_equilibrium_optimize(
                best_a, max_time=max_time, seed=int(seed) + 202, verbose=True
            )

    best_a = normalize_nonnegative(best_a)
    out_arr = np.clip(best_a, 0.0, 1000.0)
    if float(np.sum(out_arr)) < 0.01:
        out_arr = np.ones(max(1, out_arr.size), dtype=np.float64) / max(1, out_arr.size)
        out_arr *= 1.0

    out = [float(max(0.0, min(1000.0, x))) for x in out_arr]

            verified = evaluate_sequence(out)

                  f"n={len(out)}, t={time.monotonic() - t_start:.1f}s")
    except Exception as e:
        print(f"Verification skipped: {e}")

    return out
