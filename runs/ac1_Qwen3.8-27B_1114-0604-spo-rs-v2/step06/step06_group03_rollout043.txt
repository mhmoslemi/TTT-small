"""

Uses a multi-phase approach combining:

2. LSE optimization with aggressive beta schedule

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """Compute cross-correlation: result[j] = sum_k w[k] * a[k-j] (valid range)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    z = np.clip(z, -100.0, 100.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        sum_e = float(e.sum())
        if sum_e <= 0.0:
            return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse(a, beta):
    return lse_score_grad(a, float(beta))


def run_lbfgs(a0, beta, deadline, maxfun=200):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 3.0:
            return exact_score_a(a), a

    maxfun = max(20, int(maxfun))
    maxiter = max(20, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 20,
    }

    try:
        res = minimize(
            fun_lse, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def exact_subgrad(a, active_lags):
    """Compute exact subgradient at given active lags. Returns (d, grad, w)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return d_from_a(a)[0], np.zeros(n)

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)

    # Find weights for active lags (proportional to how close to max)
    cmax = float(np.max(c))
    if cmax < 1e-15:
        return d, np.zeros(n)

    # Use uniform weights on active lags
    m = len(active_lags)
    w = np.zeros(2 * n - 1)
    for k in active_lags:
        if 0 <= k < len(c):
            w[k] = 1.0
    w /= m

    h = corr_pair(a, w)
    LDS = float(np.dot(w, d))

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    return d, grad


def subgrad_step_exact(a, eta, active_width=3):
    """One subgradient step using exact peak and its neighbors."""
    n = len(a)
    c = fast_self_conv(a)
    peak_idx = int(np.argmax(c))

    # Active set: peak and its neighbors within active_width
    lo = max(0, peak_idx - active_width)
    hi = min(2 * n - 2, peak_idx + active_width)
    active_lags = list(range(lo, hi + 1))

    d, grad = exact_subgrad(a, active_lags)
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize_a(a_new)


def backtracking_subgrad(a, obj_fn, active_width=3, max_tries=20):
    """Backtracking line search for subgradient step."""
    f0 = obj_fn(a)
    eta = 0.1
    n = len(a)
    # Scale eta by problem size
    c = fast_self_conv(a)
    S = float(a.sum())
    d = (float(n) * c) / (S * S)
    dmax = float(np.max(d))
    if dmax > 1e-10:
        eta = 0.5 / dmax

    for _ in range(max_tries):
        a_new = subgrad_step_exact(a, eta, active_width)
        f_new = obj_fn(a_new)
        if np.isfinite(f_new) and f_new < f0:
            return a_new, f_new
        eta *= 0.5
        if eta < 1e-12:
            break
    return a, f0


def mirror_descent_step(a, k_active, t_blend):
    """Mirror descent step using active set."""
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return a
    dmax = float(np.max(d))
    if dmax <= 0:
        return a

    # Top-k active set
    m = len(d)
    if k_active >= m:
        w = np.ones(m) / m
    else:
        idx = np.argpartition(d, m - k_active)[m - k_active:]
        w = np.zeros(m)
        w[idx] = 1.0 / k_active

    h = corr_pair(a, w)
    h = np.maximum(h, 1e-15)

    log_a = np.log(np.maximum(a, 1e-15))
    log_h = np.log(h)
    log_new = (1.0 - t_blend) * log_a + t_blend * log_h
    a_new = np.exp(log_new - np.max(log_new))
    return normalize_a(a_new)


def polynomial_log_space_score(coeffs, n):
    """Score of a sequence parameterized as a[i] = exp(poly(i/n))."""
    x = np.linspace(0.0, 1.0, n)
    # Evaluate polynomial
    log_a = np.zeros(n)
    for i, c in enumerate(coeffs):
        log_a += c * x ** i
    # Shift to avoid overflow
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    a = normalize_a(a)
    return exact_score_a(a)


def optimize_polynomial(n, degree, deadline, x0=None):
    """Optimize polynomial coefficients in log-space."""
    if x0 is None:
        x0 = np.zeros(degree + 1)
        x0[1] = 1.0  # Start with linear tilt

    def obj(c):
        return polynomial_log_space_score(c, n)

    try:

        if res.fun < obj(x0):
            x0 = res.x
    except Exception:
        pass

    # Also try L-BFGS-B with finite differences

    except Exception:

def coordinate_descent_polish(a, deadline, max_rounds=5):
    """Polish by trying small perturbations on the most important coordinates."""
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Find the peak lag
        c = fast_self_conv(best_a)
        peak_idx = int(np.argmax(c))

        # Find which coordinates contribute most to the peak
        S = float(best_a.sum())
        # contribution[i] = best_a[peak_idx - i] for valid i
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = best_a[j]
            j2 = i - peak_idx
            if 0 <= j2 < n and j2 != j:
                contrib[i] += best_a[j2]

        # Try perturbing top contributors
        top_k = 30
        idx = np.argsort(contrib)[-top_k:]

        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            for frac in [0.01, 0.05, 0.1, -0.01, -0.05, -0.1]:
                a_trial = best_a.copy()
                a_trial[i] = max(0.0, a_trial[i] * (1.0 + frac))
                a_trial = normalize_a(a_trial)
                sc = exact_score_a(a_trial)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = a_trial
                    improved = True
                    break
        if not improved:
            break

    return best_sc, best_a

def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # Parent resampled
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        inits.append(normalize_a(pr))
        inits.append(normalize_a(pr[::-1]))

    # Power-law ramps (key structure for this problem)
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        a = (x + 0.01) ** p
        inits.append(normalize_a(a))
        a = (1.0 - x + 0.01) ** p
        inits.append(normalize_a(a))

    # Exponential tilts
    for rate in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = np.exp(rate * x)
        inits.append(normalize_a(a))
        a = np.exp(-rate * x)
        inits.append(normalize_a(a))

    # Mixed: ramp + noise
    for _ in range(5):
        p = rng.uniform(0.5, 2.0)
        base = (x + 0.01) ** p
        noise = 0.1 * rng.standard_normal(n)
        # Smooth noise
        k = 11
        kern = np.ones(k) / k
        noise = np.convolve(noise, kern, mode='same')
        a = base * np.exp(noise)
        inits.append(normalize_a(np.maximum(a, 0.0)))

    # Quadratic/cubic in log space
    for c1 in [1.0, 2.0, 4.0]:
        for c2 in [-1.0, 0.0, 1.0, 2.0]:
            log_a = c1 * x + c2 * x ** 2
            a = np.exp(log_a - np.max(log_a))
            inits.append(normalize_a(a))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: length={}".format(parent.size if parent is not None else 0))

    # Choose primary length
    n_primary = 10240
    if parent is not None:
        n_primary = parent.size
        if n_primary > 16384:
            n_primary = 16384
        if n_primary < 4096:
            n_primary = max(4096, n_primary)

    # Starting point
    if parent is not None:
        a_start = normalize_a(resample_sequence(parent, n_primary))
    else:
        x = np.linspace(0.0, 1.0, n_primary)
        a_start = normalize_a((x + 0.01) ** 1.0)

    best_sc = exact_score_a(a_start)
    best_seq = a_start.copy()
    print("Initial score: {:.10f} (n={})".format(best_sc, n_primary))

    # === Phase 1: Polynomial parameterization (10% time) ===
    t1_end = t_start + 0.10 * total_budget
    print("Phase 1: Polynomial search...")
    for degree in [2, 3, 4]:
        if time.monotonic() > t1_end - 10:
            break
        # Try several starting points for the polynomial
        for x0 in [np.array([0, 1, 0, 0][:degree+1]),
                   np.array([0, 2, -1, 0][:degree+1]),
                   np.array([0, 1, 1, 0][:degree+1])]:
            if time.monotonic() > t1_end - 5:
                break
            sc, a = optimize_polynomial(n_primary, degree, t1_end, x0)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a.copy()
                print("  Poly deg={} improved: {:.10f}".format(degree, best_sc))

    print("After Phase 1: {:.10f}".format(best_sc))

    # === Phase 2: LSE optimization with aggressive beta (30% time) ===
    t2_end = t_start + 0.40 * total_budget
    print("Phase 2: LSE optimization...")
    a = normalize_a(best_seq)

    beta_schedule = [
        (500.0, 0.08),
        (2000.0, 0.08),
        (10000.0, 0.08),
        (50000.0, 0.10),
        (200000.0, 0.10),
        (1000000.0, 0.12),
        (5000000.0, 0.12),
        (20000000.0, 0.12),
        (50000000.0, 0.10),
        (100000000.0, 0.10),
    ]

    cum = 0.0
    for beta, frac in beta_schedule:
        cum += frac
        sub_dead = t_start + cum * (t2_end - t_start) + t_start
        sub_dead = min(sub_dead, t2_end - 3.0)

        while time.monotonic() < sub_dead - 2.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            # Estimate maxFUN based on remaining time
            mf = max(30, min(300, int(rem * 2.0)))
            sc, a2 = run_lbfgs(a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  LSE beta={:.0e}: {:.10f}".format(beta, best_sc))
            a = normalize_a(a2)

    print("After Phase 2: {:.10f}".format(best_sc))

    # === Phase 3: Active-set subgradient polish (25% time) ===
    t3_end = t_start + 0.65 * total_budget
    print("Phase 3: Active-set subgradient...")
    sc, seq = active_set_polish(best_seq, t3_end, max_iters=500)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("  Active-set improved: {:.10f}".format(best_sc))

    # Additional subgradient with varying parameters
    a = normalize_a(best_seq)
    for aw in [1, 2, 3, 5]:
        if time.monotonic() > t3_end - 5:
            break
        for _ in range(50):
            if time.monotonic() > t3_end - 3:
                break
            a_new, sc = backtracking_subgrad(a, exact_score_a, active_width=aw, max_tries=15)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_seq = a_new.copy()
                a = a_new
            else:
                a = a_new  # Still move, just don't update best

    print("After Phase 3: {:.10f}".format(best_sc))

    # === Phase 4: Multi-start exploration (20% time) ===
    t4_end = t_start + 0.85 * total_budget
    print("Phase 4: Multi-start exploration...")

    inits = generate_diverse_inits(n_primary, rng, best_seq)
    n_inits = min(len(inits), 12)
    time_per_init = (t4_end - time.monotonic() - 10.0) / max(1, n_inits)

    for idx in range(n_inits):
        if time.monotonic() > t4_end - 10.0:
            break
        init_a = normalize_a(inits[idx])
        init_sc = exact_score_a(init_a)

        # Skip if already worse by a lot
        if init_sc > best_sc * 1.5:
            continue

        sub_dead = time.monotonic() + max(5.0, time_per_init)

        # Quick LSE
        for beta in [10000.0, 100000.0, 1000000.0]:
            if time.monotonic() > sub_dead - 3:
                break
            rem = float(sub_dead - time.monotonic())
            if rem < 3.0:
                break
            mf = max(20, min(100, int(rem * 3.0)))
            sc, a2 = run_lbfgs(init_a, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Multi-start [{}] beta={:.0e}: {:.10f}".format(idx, beta, best_sc))
            init_a = normalize_a(a2)

        # Quick active-set
        if time.monotonic() < sub_dead - 5:
            sc, a2 = active_set_polish(init_a, sub_dead, max_iters=50)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Multi-start [{}] active: {:.10f}".format(idx, best_sc))

    # Try a different length
    if time.monotonic() < t4_end - 30.0:
        n_alt = 8192 if n_primary != 8192 else 12288
        a_alt = normalize_a(resample_sequence(best_seq, n_alt))
        sub_dead = t4_end - 10.0
        for beta in [50000.0, 500000.0]:
            if time.monotonic() > sub_dead - 5:
                break
            rem = float(sub_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = max(30, min(100, int(rem * 2.0)))
            sc, a2 = run_lbfgs(a_alt, float(beta), sub_dead, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Alt length n={}: {:.10f}".format(n_alt, best_sc))
            a_alt = normalize_a(a2)
        if time.monotonic() < sub_dead - 10:
            sc, a2 = active_set_polish(a_alt, sub_dead, max_iters=30)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Alt length n={} active: {:.10f}".format(n_alt, best_sc))

    print("After Phase 4: {:.10f}".format(best_sc))

    # === Phase 5: Coordinate descent + final polish (15% time) ===
    t5_end = deadline
    print("Phase 5: Final polish...")

    # Coordinate descent
    sc, seq = coordinate_descent_polish(best_seq, min(t5_end - 30, time.monotonic() + 120), max_rounds=10)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_seq = seq.copy()
        print("  Coord descent: {:.10f}".format(best_sc))

    # More active-set polish
    if time.monotonic() < t5_end - 20:
        sc, seq = active_set_polish(best_seq, t5_end - 10, max_iters=200)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("  Final active-set: {:.10f}".format(best_sc))

    # One more LSE at very high beta
    if time.monotonic() < t5_end - 15:
        a = normalize_a(best_seq)
        rem = float(t5_end - time.monotonic() - 5)
        if rem > 10:
            mf = max(30, min(100, int(rem)))
            sc, a2 = run_lbfgs(a, 100000000.0, t5_end - 5, mf)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = a2.copy()
                print("  Final LSE: {:.10f}".format(best_sc))

    # Final coordinate descent
    if time.monotonic() < t5_end - 8:
        sc, seq = coordinate_descent_polish(best_seq, t5_end - 3, max_rounds=5)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("  Final coord: {:.10f}".format(best_sc))

    # === Final verification and cleanup ===
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
