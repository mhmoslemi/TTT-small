"""
Optimized step function construction using momentum subgradient descent,
adaptive spectral shaping, LP-inspired peak reduction, and L-BFGS polish.
Key improvements over previous: (1) Nesterov momentum in subgradient phase,
(2) adaptive step size with backtracking, (3) LP-inspired peak reduction,
(4) better time management and multi-start strategy.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Fast cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to sum 1."""
    S = float(a.sum())
    if S < 1e-12:
        return a
    return a / S


def era_geometric(a, n_iters, alpha_start, alpha_end, M_fft):
    """ERA with geometric spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)
        
        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)
        
        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_momentum(a, n_iters, M_fft, lr_init, momentum, sc_lb):
    """Subgradient descent with Nesterov momentum and adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # Momentum velocity
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(np.max(c))
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Adaptive learning rate
        gap = max(sc - sc_lb, 1e-6)
        lr = lr_init * gap / (n + 1)
        lr = min(lr, 0.01)
        
        # Compute subgradient of max(c) w.r.t. a
        # Use top peaks for a "soft" subgradient
        n_peaks = min(20, len(c))
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                w = c[k]  # Weight by the peak value
                sg[js:je + 1] += 2.0 * a[js:je + 1][::-1] * w
        
        # Normalize subgradient
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm > 1e-15:
            sg = sg / sg_norm
        
        # Nesterov momentum update
        v = momentum * v + lr * sg

def lse_objective(x, beta, M_fft):

    if exp_sum < 1e-30:

    L = c_max + float(np.log(exp_sum)) / beta
    return 2.0 * n * L / (S * S)

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)

"""
Optimized step function construction using Nesterov momentum subgradient descent,
adaptive spectral shaping (ERA), LP-inspired peak reduction, and L-BFGS polish.
Key improvements: (1) Nesterov momentum with adaptive step, (2) weighted subgradient
using top-K peaks, (3) multi-start with diverse inits, (4) perturbation restarts
to escape local minima, (5) coordinate descent fine-tuning.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective value: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_seq(a):
    """Normalize sequence to sum 1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    return a / S


def era_geometric(a, n_iters, alpha_start, alpha_end, M_fft):
    """ERA with adaptive geometric spectral shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize_seq(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_safe = np.maximum(amp, 1e-20)
        amp_new = amp_safe * (C / amp_safe) ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def era_power_law(a, n_iters, gamma, M_fft):
    """ERA with power-law shaping (only reduce above-median amplitudes)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize_seq(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        amp = np.abs(A)
        phase = np.angle(A)

        if len(amp) > 1:
            C = np.median(amp[1:])
        else:
            C = amp[0]
        C = max(C, 1e-20)

        amp_new = amp.copy()
        above = amp > C * (1.0 + 1e-10)
        if above.any():
            amp_new[above] = amp[above] * (C / amp[above]) ** gamma

        A_new = amp_new * np.exp(1j * phase)

def subgradient_momentum(a, n_iters, M_fft, lr_init, momentum_decay, sc_lb, n_peaks):
    """
    Subgradient descent with Nesterov momentum and adaptive step size.
    Uses weighted subgradient from top-K peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize_seq(a)

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    lr = lr_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(np.max(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        gap = max(sc - sc_lb, 1e-8)

        # Adaptive learning rate (Polyak-inspired)
        lr = lr_init * gap / (1.0 + 0.01 * it)
        lr = min(lr, 0.005)
        lr = max(lr, 1e-6)

        # Compute weighted subgradient from top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
            weights = c.copy()
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
            weights = c[top_idx]

        # Normalize weights
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            weights = np.ones(len(top_idx))
        else:
            weights = weights / w_sum

        sg = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = int(idx)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je + 1] += 2.0 * w * a[js:je + 1][::-1]

        # Also subtract the "uniform" component to center the gradient
        sg -= np.sum(sg) / n

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm

        # Nesterov momentum
        v = momentum_decay * v + lr * sg
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        # Accept if not too much worse
        sc_new = compute_score(a_new)
        if sc_new <= sc + 0.05 * gap:
            a = a_new
        else:
            # Halve momentum on rejection
            v *= 0.5

    return best_a


def peak_coordinate_descent(a, n_iters, M_fft, n_peaks, step_size):
    """
    Coordinate descent that directly targets top convolution peaks
    by adjusting individual elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize_seq(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        sc = 2.0 * n * float(np.max(c)) / (float(a.sum()) ** 2)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top peaks
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # For each peak, find the element that contributes most and reduce it
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je < js:
                continue

            # Contributions: a[j] * a[k-j] for j in [js, je]
            contrib = a[js:je + 1] * a[k - je:k - js + 1]
            JM = int(np.argmax(contrib))
            j_star = js + JM

            # Reduce a[j_star] slightly
            delta = step_size * a[j_star]
            if delta < 1e-12:
                continue
            a_try = a.copy()
            a_try[j_star] = max(0.0, a_try[j_star] - delta)

            # Also try increasing the "complementary" element
            comp = k - j_star
            if 0 <= comp < n:
                a_try[comp] += delta * 0.5  # Partially compensate

            a_try = normalize_seq(a_try)
            sc_try = compute_score(a_try)
            if sc_try < sc:
                a = a_try
                sc = sc_try

    return best_a

    S3 = S2 * S

        if time.monotonic() < phase2_end - 5:
            a_pol = era_geometric(a_sg, n_iters=100, alpha_start=0.02, alpha_end=0.4, M_fft=M_fft)
            sc = compute_score(a_pol)
            update_best(sc, a_pol, f"SG-{ci}-ERA")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS with LSE (25% time)
    # ============================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (t<={phase3_end - t_start:.0f}s) ===")

    betas = [50, 200, 500, 2000, 5000, 15000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        print(f"  beta={beta}, remaining={remaining:.0f}s")
        maxiter = min(5000, int(remaining * 20))
        maxfun = maxiter * 20
        try:
            sc, a_opt = run_lbfgs(best_a, beta, M_fft, maxiter=maxiter, maxfun=maxfun)
            update_best(sc, a_opt, f"LBFGS b={beta}")
        except Exception as e:
            print(f"    L-BFGS failed: {e}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Coordinate descent + perturbation restarts (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Polish (t<={deadline - t_start:.0f}s) ===")

    # Coordinate descent
    if time.monotonic() < deadline - 60:
        print("  Coordinate descent...")
        a_cd = peak_coordinate_descent(best_a, n_iters=200, M_fft=M_fft,
                                       n_peaks=10, step_size=0.01)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD")

    # More subgradient with different params
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60 - time.monotonic()
        a_sg2 = subgradient_momentum(
            best_a, int(remaining * 1000), M_fft,
            lr_init=0.005, momentum_decay=0.99,
            sc_lb=sc_lb, n_peaks=30
        )
        sc = compute_score(a_sg2)
        update_best(sc, a_sg2, "SG-final")

    # Final L-BFGS
    if time.monotonic() < deadline - 30:
        try:
            sc, a_opt = run_lbfgs(best_a, 3000, M_fft, maxiter=2000, maxfun=30000)
            update_best(sc, a_opt, "LBFGS-final")
        except Exception:
            pass

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(5):
            if time.monotonic() > deadline - 15:
                break
            noise_level = 0.01 * (1 + 0.5 * trial)
            a_p = best_a * (1.0 + noise_level * rng.standard_normal(len(best_a)))
            a_p = np.clip(a_p, 0.0, 1000.0)
            a_p = normalize_seq(a_p)

            # Quick optimization from perturbed start
            a_r = era_geometric(a_p, n_iters=50, alpha_start=0.03, alpha_end=0.5, M_fft=M_fft)
            a_r = subgradient_momentum(a_r, 5000, M_fft, lr_init=0.008,
                                       momentum_decay=0.95, sc_lb=sc_lb, n_peaks=10)
            sc = compute_score(a_r)
            update_best(sc, a_r, f"RESTART-{trial}")

    # ============================================================
    # Final verification and output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
