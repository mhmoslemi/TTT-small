"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov accelerated subgradient
with adaptive step size and LSE-smoothed polishing.
Focus on fast convergence from a good starting point (parent sequence).
"""

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (int(m) - 1).bit_length()


def _fast_self_conv(a):
    """Compute full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = _fast_self_conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / S


def _resample(a, n):
    """Resample sequence to length n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def _subgrad_score(a, k_active=5, tol=1e-4):
    """
    Compute score and subgradient of 2*n*max(c)/S^2.
    a is assumed to be normalized (sum=1, non-negative).
    Returns (score, gradient) where gradient is the subgradient of the function.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c = _fast_self_conv(a)
    d = n * c / (S * S)  # normalized: score = 2*max(d)
    dmax = float(d.max())
    score = 2.0 * dmax
    
    # Find active set
    threshold = dmax * (1.0 - tol)
    active = np.where(d >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(d)])
    if len(active) > k_active:
        # Keep only top k_active
        top_idx = np.argsort(d[active])[-k_active:]
        active = active[top_idx]
    
    # Subgradient: grad[i] = 2n/S^2 * sum_{k in active} w_k * a[k-i] - 2n*max_c/S^3
    # where w_k = 1/|active|
    # For normalized a (S=1): grad[i] = 2n * mean(a[active[i]-i]) - 2n * max_c
    # But let's keep it general.
    
    # R[i] = mean of a[active[j] - i] for j in range(len(active)), where 0 <= active[j]-i < n
    # Use FFT: R = convolve(a_rev, w_active) where w_active has 1/|active| at active positions
    w_active = np.zeros(2 * n - 1)
    w_active[active] = 1.0 / len(active)
    
    N2 = _next_pow2(3 * n)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, N2)
    WR = np.fft.rfft(w_active, N2)
    conv_full = np.fft.irfft(AR * WR, N2)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    # c[active] mean = mean of d[active] * S^2 / n = (mean(d[active]) * S^2) / n
    # Actually: T = mean(c[active[j]])
    T = float(c[active].mean())
    
    # grad[i] = (2n/S^2) * R[i] - (2n * T) / S^3
    grad = (2.0 * n / (S * S)) * R - (2.0 * n * T) / (S * S * S)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad

Now, for the Nesterov method, I need to be careful about:
1. The function is defined on the simplex {a >= 0, sum(a) = 1}.
2. The subgradient is computed in R^n, but the effective gradient should be projected onto the tangent space of the simplex.
3. However, if we use the "projected" subgradient method (project back onto the simplex after each step), we don't need to explicitly project the gradient.

The projected subgradient method is:
a_{k+1} = proj_simplex(a_k - α_k * g_k)

where g_k is the subgradient at a_k.

For Nesterov acceleration:
y_k = a_k + ((t_k - 1) / t_{k+1}) * (a_k - a_{k-1})
a_{k+1} = proj_simplex(y_k - α_k * g(y_k))
t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

The step size α_k for Nesterov is typically α_k = β / (k + 1) or α_k = β / sqrt(k+1), where β is a constant related to the Lipschitz constant.

For the O(1/T^2) rate with Nesterov, we need α_k = c / (k+1) where c depends on the Lipschitz constant L and the initial distance R.

In practice, we can use:
α_k = α_0 / (1 + λ*k) for some α_0 and λ.

Or we can use an adaptive scheme:
- Track the best score seen so far.
- If no improvement in K iterations, reduce α by a factor.

def _nesterov_subgrad(a0, deadline, k_active=3, tol=1e-4):
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_score = _exact_score(a)
    
    t = 1.0
    a_prev = a.copy()
    alpha = 0.01  # initial step size
    alpha_decay = 0.9999  # decay per iteration
    last_improve = 0
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new
        y = a + beta_mom * (a - a_prev)
        a_prev = a.copy()
        t = t_new
        
        # Project y to simplex
        y = np.clip(y, 0.0, None)
        Sy = float(y.sum())
        if Sy < 1e-12:
            y = a.copy()
        else:
            y = y / Sy
        
        # Compute subgradient at y
        score_y, grad = _subgrad_score(y, k_active=k_active, tol=tol)
        if not np.isfinite(score_y):
            break
        
        # Gradient step
        x_new = y - alpha * grad
        
        # Project to simplex
        x_new = np.clip(x_new, 0.0, None)
        Sx = float(x_new.sum())
        if Sx < 1e-12:
            x_new = a.copy()
        else:
            x_new = x_new / Sx
        
        a = x_new
        alpha *= alpha_decay
        
        # Track best
        sc = _exact_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve = it
        
        # Adaptive step size
        if it > 100 and it - last_improve > 500:
            alpha *= 0.5
            last_improve = it
    
    return best_score, best_a
