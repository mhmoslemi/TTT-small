    """

                    cols.append(i)
                    vals.append(f[j])

        m = len(active_set)

"""
Step function optimization using LP-based direction finding, Fienup spectral
shaping, and L-BFGS LSE polishing. Key novelty: the LP direction-finding
step finds the maximally-sum non-negative sequence b such that the
cross-convolution (f*b) stays below the current peak, providing a principled
improvement direction that exploits the linear structure of the constraint.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    """Cross-correlation/convolution of a (len m) and b (len n), full mode."""
    m, n = len(a), len(b)
    L = m + n - 1
    M_fft = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:L]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

"""
Step function optimization using LP-based direction finding (cutting-plane),
Fienup spectral shaping, and L-BFGS LSE polishing. Key novelty: the LP step
finds the maximum-sum non-negative direction b such that the cross-convolution
(f*b) stays below the current peak, providing a principled improvement direction.
This is combined with adaptive LSE smoothing and multi-scale peak suppression.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    m, n = len(a), len(b)
    L = m + n - 1
    M_fft = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:L]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def fienup_init(a, n_iters=100, alpha=0.5):

        amp_new = amp * (C / np.maximum(amp, 1e-20)) ** alpha

def build_con_mat(f, active_indices):
    n = len(f)
    m = len(active_indices)
    if m == 0:
        return csr_matrix((0, n))
    total_nnz = 0
    for k in active_indices:
        k = int(k)
        total_nnz += min(k + 1, n, 2 * n - 1 - k)
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    pos = 0
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_s = max(0, k - n + 1)
        j_e = min(k, n - 1)
        nnz = j_e - j_s + 1
        if nnz <= 0:
            continue
        jj = np.arange(j_s, j_e + 1)
        rows[pos:pos+nnz] = idx
        cols[pos:pos+nnz] = jj
        vals[pos:pos+nnz] = f[k - jj]
        pos += nnz
    if pos < total_nnz:
        rows = rows[:pos]
        cols = cols[:pos]
        vals = vals[:pos]
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def lp_direction(f, M_target, K_init=100, max_rounds=50, time_limit_lp=20.0):

def lse_objective_and_grad(x, beta):
    n = len(x)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(x, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # grad_L[i] = 2 * sum_k p[k] * a[k-i] where 0<=k-i<n
    # Use cross-correlation via convolution
    a_rev = x[::-1]
    L2 = M + n - 1
    M_fft2 = 1 << (L2 - 1).bit_length()
    P = np.fft.rfft(p, M_fft2)
    AR = np.fft.rfft(a_rev, M_fft2)
    conv_res = np.fft.irfft(P * AR, M_fft2)[:L2]
    G = conv_res[n - 1:2 * n - 1]
    g_L = 2.0 * G
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )

    if a_out.sum() < 1e-6:

def peak_directed_descent(a, n_iters=2000, n_peaks=5, step_init=0.015, decay=0.99995):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if n_peaks >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                seg = a[js:je + 1]
                sg[js:je + 1] += 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(step, max((sc - sc_lb) / (gn * gn + 1e-30), 1e-10))
        step = min(step, 0.025)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    return best_a

                S_b = float(b_alt.sum())
                if S_b > 1e-12:
                    b_sc = b_alt * (S_curr / S_b)
                    a2 = (1.0 - 0.15) * a_cur + 0.15 * b_sc
                    a2 = np.maximum(a2, 0.0)
                    a2 = a2 / a2.sum()
                    sc2 = compute_score(a2)
                    if sc2 < score_cur - 1e-10:
                        a_cur = a2
                        update_best(sc2, a_cur, f"LP-alt {n_lp_iters}")
                    else:
                        # No improvement, break
                        break
        n_lp_iters += 1
        if n_lp_iters % 20 == 0:
            print(f"  LP iter {n_lp_iters}: score={compute_score(a_cur):.8f}, t={time.monotonic()-t_start:.0f}s")

    print(f"Phase 2 done: {n_lp_iters} LP iters, best={best_score:.8f}")

    # Phase 3: L-BFGS with LSE
    print(f"\n=== Phase 3: L-BFGS LSE ===")
    phase3_end = t_start + 0.65 * (total_budget - 30.0)
    a_cur = best_a.copy()
    betas = [50, 200, 1000, 3000, 10000, 30000]
    for beta in betas:
        if time.monotonic() > phase3_end - 60:
            break
        remaining = phase3_end - time.monotonic()
        maxiter = min(8000, int(remaining * 20))
        maxfun = maxiter * 20
        print(f"  beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            update_best(sc, a_opt, f"LBFGS b={beta}")
            a_cur = a_opt.copy()
        else:
            # Try from best_a directly
            sc2, a_opt2 = run_lbfgs(best_a, beta, maxiter=maxiter // 2, maxfun=maxfun // 2)
            if sc2 < best_score:
                update_best(sc2, a_opt2, f"LBFGS-best b={beta}")
                a_cur = a_opt2.copy()

    print(f"Phase 3 done: best={best_score:.8f}")

    # Phase 4: Peak-directed descent + final polish
    print(f"\n=== Phase 4: Peak Descent + Polish ===")
    for n_peaks in [3, 5, 10, 20]:
        if time.monotonic() > deadline - 60:
            break
        a_pd = peak_directed_descent(best_a, n_iters=3000, n_peaks=n_peaks,
                                     step_init=0.01, decay=0.99998)
        sc = compute_score(a_pd)
        update_best(sc, a_pd, f"PEAK m={n_peaks}")

    # More LP iterations from improved starting point
    if time.monotonic() < deadline - 120:
        a_lp2 = best_a.copy()
        for _ in range(30):
            if time.monotonic() > deadline - 60:
                break
            a_new, sc_new = lp_blend_iteration(a_lp2, blend=0.15, M_reduction=0.002)
            if sc_new < compute_score(a_lp2) - 1e-10:
                a_lp2 = a_new
                update_best(sc_new, a_lp2, "LP-final")
            else:
                break

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=2000, maxfun=40000)
        update_best(sc, a_opt, "LBFGS-final")

    # Perturbation restarts
    if time.monotonic() < deadline - 20:
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            noise = 0.01 * (1 + trial)
            perturb = best_a * (1.0 + noise * rng.standard_normal(len(best_a)))
            perturb = np.clip(perturb, 0.0, 1000.0)
            S_p = perturb.sum()
            if S_p < 1e-12:
                break
            perturb = perturb / S_p
            a_res = fienup_init(perturb, n_iters=30, alpha=0.3)
            a_res = peak_directed_descent(a_res, n_iters=500, n_peaks=5, step_init=0.005)
            sc = compute_score(a_res)
            update_best(sc, a_res, f"RESTART {trial}")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
