def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for threshold
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.maximum(v - t, 0)

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.where(cond)[0]) if cond.any() else 0
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
