    return a

"""
Peak-directed LSE optimization with aggressive high-beta schedule,
multi-start perturbation, and coordinate-descent finishing.
Starts from parent, uses smooth beta continuation up to very high values,
and applies targeted peak-reduction local search for final improvement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """Compute LSE-smoothed objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    # Clip for numerical safety
    scaled = np.clip(scaled, -700.0, 0.0)
    exp_scaled = np.exp(scaled)
    exp_sum = np.sum(exp_scaled)
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient via FFT convolution
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200,
                 'iprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Iterative spectral flattening for initialization."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    total = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if k == 0 or (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (total / S)
    return a


def peak_directed_update(a, alpha=0.01, n_top=50):
    """Reduce entries contributing most to the peak, redistribute to low contributors."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        for j in range(j_start, j_end + 1):
            other = k_star - j
            if 0 <= other < n:
                contrib[j] = a[other]
    n_top = min(n_top, n // 2)
    top_idx = np.argsort(contrib)[::-1][:n_top]
    bottom_idx = np.argsort(contrib)[:n_top]
    a_new = a.copy()
    total_reduce = 0.0
    for j in top_idx:
        amt = a_new[j] * alpha
        a_new[j] = max(a_new[j] - amt, 0.0)
        total_reduce += amt
    if total_reduce > 0 and len(bottom_idx) > 0:
        per = total_reduce / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per
    return np.clip(a_new, 0.0, 1000.0)


def multi_peak_update(a, alpha=0.005, n_peaks=3):
    """Reduce contributions to top few peak lags simultaneously."""
    n = len(a)
    c = fast_conv(a)
    # Get top peak lags
    top_lags = np.argsort(c)[::-1][:n_peaks]
    contrib = np.zeros(n)
    for k in top_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            other = k - j
            if 0 <= other < n:
                contrib[j] += a[other]
    n_top = max(20, n // 200)
    top_idx = np.argsort(contrib)[::-1][:n_top]
    bottom_idx = np.argsort(contrib)[:n_top]
    a_new = a.copy()
    total_reduce = 0.0
    for j in top_idx:
        amt = a_new[j] * alpha
        a_new[j] = max(a_new[j] - amt, 0.0)
        total_reduce += amt
    if total_reduce > 0 and len(bottom_idx) > 0:
        per = total_reduce / len(bottom_idx)
        for j in bottom_idx:
            a_new[j] += per
    return np.clip(a_new, 0.0, 1000.0)


def subgrad_polish(a, n_iters=2000):
    """Exact subgradient method for the non-smooth objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        # Backtracking line search
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a_new = np.clip(a - step * grad, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new <= sc:
            a = a_new
        else:
            a = np.clip(a - 0.01 * step * grad, 0.0, 1000.0)
    return best_sc, best_a
