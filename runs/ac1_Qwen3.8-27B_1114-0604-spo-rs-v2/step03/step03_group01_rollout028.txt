"""
Hybrid optimization for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses multiplicative weight updates, LP-based iterations with constraint selection,
L-BFGS polishing, and targeted coordinate descent. Different from previous approach
by using MWU as primary method and LP-based direction finding.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def mwu_iteration(a, eta, alpha, top_k_lags=50, n_iters=1, t_start=None, time_limit=None):
    """
    Multiplicative Weight Update iteration.
    Reduces positions that contribute to peak lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a_norm = a / S

    for it in range(n_iters):
        if t_start is not None and time_limit is not None:
            if time.monotonic() - t_start > time_limit:
                break

        c = fast_conv(a_norm)
        M = c.max()
        if M < 1e-15:
            break

        # Select top-K lags for the update
        if top_k_lags < len(c):
            sorted_idx = np.argsort(c)[::-1][:top_k_lags]
            # Only include lags that are within a factor of the peak
            threshold = 0.5 * M
            active_idx = [k for k in sorted_idx if c[k] >= threshold]
        else:
            active_idx = list(range(len(c)))

        if len(active_idx) == 0:
            break

        active_idx = np.array(active_idx)
        w = c[active_idx] / M  # normalized weights

        # Compute load on each position
        # load_j = sum_{k in active} w_k * a_norm[k-j] (with bounds)
        load = np.zeros(n)
        for k in active_idx:
            k = int(k)
            wk = float(w[list(active_idx).index(k)]) if False else float(c[k] / M)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                load[js:je + 1] += wk * a_norm[idx]

        # Multiplicative update: reduce loaded positions
        factor = 1.0 / (1.0 + eta * load) ** alpha
        a_norm = a_norm * factor

        # Re-normalize
        S_new = a_norm.sum()
        if S_new < 1e-12:
            a_norm = np.ones(n) / n
        else:
            a_norm = a_norm / S_new

    return a_norm * S


def lp_direction(a, M_frac=0.7, max_constraints=300, use_highs=True):
    """
    LP-based direction finding.
    Solves: max sum(g) s.t. g >= 0, (a*g)_k <= M for top-K lags.
    Returns the LP solution rescaled to match sum(a).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()

    c = fast_conv(a)
    M = float(c.max())

    # Select constraints: lags where c_k >= M_frac * M
    threshold = M_frac * M
    active_mask = c >= threshold
    active_lags = np.where(active_mask)[0]

    if len(active_lags) == 0:
        active_lags = np.array([int(np.argmax(c))])
    elif len(active_lags) > max_constraints:
        # Take the top max_constraints
        sorted_active = active_lags[np.argsort(c[active_lags])[::-1]][:max_constraints]
        active_lags = sorted_active

    # Build constraint matrix (sparse -> dense for small problems)
    n_cons = len(active_lags)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For memory efficiency, build row by row
    A = np.zeros((n_cons, n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            A[i, js:je + 1] = a[idx]

    # Solve: max 1^T g s.t. A g <= M, g >= 0
    # linprog minimizes, so: min -1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(
        c_obj,
        A_ub=A,
        b_ub=np.full(n_cons, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )

    if res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        g_sum = float(g.sum())
        if g_sum > 1e-12:
            # Rescale to match sum(a)
            g = g * S / g_sum
            return g
    return a.copy()


def lp_iteration(a, n_iters=5, blend=0.3, M_frac=0.7, max_constraints=300,
                 t_start=None, time_limit=None):
    """
    Multiple LP-based iterations with blending.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if t_start is not None and time_limit is not None:
            if time.monotonic() - t_start > time_limit:
                break

        t0 = time.monotonic()
        g = lp_direction(a, M_frac=M_frac, max_constraints=max_constraints)
        elapsed_lp = time.monotonic() - t0

        if g is not None:
            # Check if LP gives improvement
            # Compute cross-conv a*g to verify
            cross = fast_cross_conv(a, g)
            cross_max = float(cross.max())
            M_old = float(fast_conv(a).max())

            if cross_max < M_old * (1 - 1e-6):
                # Blending
                a_new = (1.0 - blend) * a + blend * g
                a_new = np.clip(a_new, 0, None)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller blend
                for b in [0.1, 0.05, 0.02]:
                    a_new = (1.0 - b) * a + b * g
                    a_new = np.clip(a_new, 0, None)
                    sc = compute_score(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        a = a_new
                        break

    return best_score, best_a


def lse_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def targeted_coord_descent(a, n_sweeps=20, top_k=50, t_start=None, time_limit=None):
    """
    Coordinate descent targeting positions that contribute to peak lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit if (t_start is not None and time_limit is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        current_score = 2.0 * n * M / (S * S)

        if current_score < best_score:
            best_score = current_score

        # Find top contributing positions to the peak lags
        top_lags = np.argsort(c)[::-1][:10]
        hot_positions = set()
        for k in top_lags:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                hot_positions.add(j)

        hot_positions = list(hot_positions)[:top_k]
        # Also add some random positions
        extra = np.random.randint(0, n, size=min(20, n))
        hot_positions = list(set(hot_positions + list(extra)))

        for j in hot_positions:
            if time.monotonic() > deadline:
                break

            a_old = a[j]
            if a_old < 1e-15:
                candidates = [0.0, 0.001, 0.01, 0.1]
            else:
                ratios = [0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.2, 1.5, 2.0]
                candidates = [a_old * r for r in ratios]
                candidates.append(0.0)

            best_x = a_old
            best_sc = current_score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, a


def structured_init(n, strategy='sine', rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(0)

    if strategy == 'sine':
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif strategy == 'sine2':
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.3 * np.cos(2 * np.pi * 7 * x + 1.0)
    elif strategy == 'triangle':
        x = np.arange(n) / n
        a = 1.0 - np.abs(2 * x - 1)
    elif strategy == 'bell':
        x = np.arange(n) / n
        a = np.exp(-5.0 * (x - 0.5) ** 2)
    elif strategy == 'random_smooth':
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    elif strategy == 'two_sine':
        x = np.arange(n) / n
        f1 = 1 + int(rng.integers(0, 5))
        f2 = 3 + int(rng.integers(0, 8))
        p1 = 2 * np.pi * rng.random()
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x + p1) + 0.2 * np.cos(2 * np.pi * f2 * x)
    else:
        a = np.ones(n)

    a = np.clip(a, 0, None)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def adam_direct(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                time_limit_s=None, t_start=None, normalize=True):
    """Adam optimizer in a-space with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad_a(a, beta_lse)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s

        if t % 10000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def mwu_optimize(a_init, eta_init=0.5, alpha=1.0, n_iters=500,
                 top_k_lags=50, t_start=None, time_limit=None):
    """
    Full MWU optimization loop with adaptive eta.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    a_norm = a / S if S > 1e-12 else np.ones(n) / n

    best_a = a.copy()
    best_score = compute_score(a)
    eta = eta_init
    patience = 0
    best_eta = eta

    for it in range(n_iters):
        if t_start is not None and time_limit is not None:
            if time.monotonic() - t_start > time_limit:
                break

        # Single MWU step
        c = fast_conv(a_norm)
        M = float(c.max())
        if M < 1e-15:
            break

        # Select active lags
        threshold = 0.6 * M
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) > top_k_lags:
            active_lags = active_lags[np.argsort(c[active_lags])[::-1]][:top_k_lags]

        # Compute load
        load = np.zeros(n)
        for k in active_lags:
            k = int(k)
            wk = float(c[k] / M)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                load[js:je + 1] += wk * a_norm[idx]

        # Update
        factor = 1.0 / (1.0 + eta * load) ** alpha
        a_norm_new = a_norm * factor
        s = a_norm_new.sum()
        if s < 1e-12:
            a_norm_new = np.ones(n) / n
        else:
            a_norm_new = a_norm_new / s

        # Evaluate
        a_candidate = a_norm_new * S
        sc = compute_score(a_candidate)

        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            a_norm = a_norm_new
            a = a_candidate
            patience = 0
            best_eta = eta
        else:
            patience += 1
            if patience > 50:
                # Reduce eta and reset
                eta *= 0.7
                patience = 0
                if eta < 1e-4:
                    break

        # Slowly decay eta
        eta *= 0.999

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_score, best_a

        sc, a_opt = targeted_coord_descent(best_a, n_sweeps=10, top_k=60,
                                           t_start=t_start, time_limit=min(15.0, final_deadline - time.monotonic() - 3))
        try_update(sc, a_opt, "final coord")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
