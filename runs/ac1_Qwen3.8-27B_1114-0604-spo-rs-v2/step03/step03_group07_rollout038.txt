        sg = np.zeros(n)
        j_range = np.arange(j_start, j_end + 1)
        sg[j_range] = 2.0 * a[k_star - j_range]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad.dot(grad))
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def multi_peak_subgrad(a0, max_iters=5000):
    """Subgradient using weighted average of top peaks for smoother direction."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    n_top = min(5, n)

    for it in range(max_iters):
        b = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(b.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top n_top peaks
        top_idx = np.argsort(b)[::-1][:n_top]
        # Weight by proximity to max
        weights = np.exp(-0.5 * (b[top_idx] - M) / (M * 0.05 + 1e-10))
        weights /= weights.sum()

        sg = np.zeros(n)
        for w, k in zip(weights, top_idx):
            j_start = max(0, int(k) - n + 1)
            j_end = min(int(k), n - 1)
            j_range = np.arange(j_start, j_end + 1)
            sg[j_range] += w * 2.0 * a[int(k) - j_range]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad.dot(grad))
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def block_coordinate_descent(a0, block_size=64, n_sweeps=20, iters_per_block=200):
    """Cyclic block coordinate descent with L-BFGS on each block."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        for bi in range(n_blocks):
            start = bi * block_size
            end = min(start + block_size, n)
            if end - start < 2:
                continue

            def block_obj(x, s=start, e=end):
                a_tmp = a.copy()
                a_tmp[s:e] = x
                sc = compute_score(a_tmp)
                if np.isinf(sc):
                    return 1e10, np.zeros(len(x))
                b = fast_conv(a_tmp)
                S = float(a_tmp.sum())
                bmax = float(b.max())
                k_star = int(b.argmax())
                j_start = max(0, k_star - n + 1)
                j_end = min(k_star, n - 1)
                grad = np.zeros(len(x))
                for local_j in range(len(x)):
                    gj = s + local_j
                    gval = 0.0
                    if j_start <= gj <= j_end:
                        partner = k_star - gj
                        if 0 <= partner < n:
                            gval = 2.0 * a_tmp[partner]
                    S2 = S * S
                    S3 = S2 * S
                    grad[local_j] = 2.0 * n * (gval / S2 - 2.0 * bmax / S3)
                return sc, grad

            x0 = a[start:end].copy()
            bounds = [(0.0, 1000.0)] * (end - start)
            try:
                res = minimize(block_obj, x0, method='L-BFGS-B', jac=True,
                               bounds=bounds,
                               options={'maxiter': iters_per_block, 'maxfun': iters_per_block * 5})
                a[start:end] = np.clip(res.x, 0.0, 1000.0)
            except Exception:
                pass

            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_sc, best_a


def sharpen(a, factor=1.5):
    """Increase contrast: amplify above-mean, reduce below-mean."""
    mean_val = a.mean()
    a_sharp = np.clip(mean_val + factor * (a - mean_val), 0.0, 1000.0)
    if a_sharp.sum() < 1e-6:
        return a.copy()
    return a_sharp


def build_inits(n, rng, parent):
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        inits.append(sharpen(parent, 1.3))
        inits.append(sharpen(parent, 2.0))

    # Power-law ramps (anti-palindromic)
    for p in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.clip(0.01 + 0.99 * xn ** p, 0.01, 1000.0))
        inits.append(np.clip(0.01 + 0.99 * (1 - xn) ** p, 0.01, 1000.0))

    # Sinusoidal anti-symmetric (about midpoint)
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(2 * np.pi * freq * 2 * xn - np.pi / 2), 0.01, 1000.0))

    # Combined ramp + texture
    inits.append(np.clip(0.2 + 0.6 * xn + 0.1 * np.sin(2 * np.pi * 13 * xn), 0.01, 1000.0))
    inits.append(np.clip(0.2 + 0.6 * (1 - xn) + 0.1 * np.cos(2 * np.pi * 11 * xn), 0.01, 1000.0))

    # Gaussian bump at edge
    center = 0.0
    width = 0.3
    inits.append(np.clip(0.1 + 0.9 * np.exp(-((xn - center) / width) ** 2), 0.01, 1000.0))
    center = 1.0
    inits.append(np.clip(0.1 + 0.9 * np.exp(-((xn - center) / width) ** 2), 0.01, 1000.0))

    # Random smooth
    ksize = max(3, n // 100)
    a_rand = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(a_rand, kernel, mode='same'), 0.01, 1000.0))

    # Resample parent to this length
    if parent is not None and len(parent) != n:
        from numpy import interp
        p_x = np.linspace(0, 1, len(parent))
        n_x = np.linspace(0, 1, n)
        resampled = interp(n_x, p_x, parent)
        inits.append(np.clip(resampled, 0.01, 1000.0))
        inits.append(np.clip(resampled[::-1], 0.01, 1000.0))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [4096, 6962, 8192, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths: {lengths}")
    print(f"Parent: {'yes (n=' + str(parent.size) + ')' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: Quick LSE sweep to find good starting points
    # ============================================================
    p1_end = t_start + 0.20 * total_budget
    print(f"\n=== Phase 1: Quick LSE Sweep (until t={p1_end-t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        time_per_init = max(10, int((p1_end - time.monotonic()) / len(inits)))
        for i, init in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            t0 = time.monotonic()
            a_cur = init.copy()
            for beta in [100, 500]:
                if time.monotonic() > p1_end:
                    break
                maxit = max(500, min(3000, int((time_per_init - (time.monotonic()-t0)) / 2)))
                if maxit < 200:
                    break
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 10)
                a_cur = a_opt
                report(sc, a_opt, f"P1 n={n} i={i} b={beta}")

    print(f"  After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: High-beta LSE continuation on best solution
    # ============================================================
    p2_end = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: High-Beta Refinement (until t={p2_end-t_start:.0f}s) ===")

    n_best = len(best_a)
    beta_schedule = [1000, 3000, 10000, 30000, 100000, 300000, 500000]

    for beta in beta_schedule:
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        # More iterations for higher beta (needs more care)
        maxit = max(2000, min(20000, int(remaining * 0.3 / max(1, n_best / 8192))))
        maxfun = maxit * 15
        print(f"  beta={beta}, maxiter={maxit}, remaining={remaining:.0f}s")
        sc, a_opt, success = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxfun)
        report(sc, a_opt, f"P2 beta={beta}")

        # Also try from sharpened version
        a_sharp = sharpen(best_a, 1.5)
        sc2, a_opt2, _ = run_lbfgs(a_sharp, beta, maxiter=maxit // 2, maxfun=maxit)
        report(sc2, a_opt2, f"P2s beta={beta}")

    print(f"  After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Try other lengths from best solution
    # ============================================================
    p3_end = t_start + 0.78 * total_budget
    print(f"\n=== Phase 3: Multi-length Exploration ===")

    for n in lengths:
        if time.monotonic() > p3_end:
            break
        if n == len(best_a):
            continue
        # Resample best solution to new length
        from numpy import interp
        old_x = np.linspace(0, 1, len(best_a))
        new_x = np.linspace(0, 1, n)
        a_resampled = interp(new_x, old_x, best_a)
        a_resampled = np.clip(a_resampled, 0.01, 1000.0)

        for beta in [5000, 50000, 200000]:
            if time.monotonic() > p3_end:
                break
            remaining = p3_end - time.monotonic()
            maxit = max(1000, min(10000, int(remaining * 0.3)))
            sc, a_opt, _ = run_lbfgs(a_resampled, beta, maxiter=maxit, maxfun=maxit * 10)
            report(sc, a_opt, f"P3 n={n} b={beta}")

    print(f"  After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Subgradient polish (exact objective)
    # ============================================================
    p4_end = t_start + 0.90 * total_budget
    print(f"\n=== Phase 4: Subgradient Polish ===")

    if time.monotonic() < p4_end:
        n_iters = max(2000, int((p4_end - time.monotonic()) * 50))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "subgrad")

    if time.monotonic() < p4_end:
        n_iters = max(1000, int((p4_end - time.monotonic()) * 30))
        sc, a_opt = multi_peak_subgrad(best_a, max_iters=n_iters)
        report(sc, a_opt, "multi-peak subgrad")

    print(f"  After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Block coordinate descent
    # ============================================================
    p5_end = deadline - 10
    print(f"\n=== Phase 5: Block Coordinate Descent ===")

    if time.monotonic() < p5_end:
        n_best = len(best_a)
        remaining = p5_end - time.monotonic()
        block_size = max(32, min(128, n_best // 64))
        n_sweeps = max(5, int(remaining / (n_best / block_size * 0.5)))
        n_sweeps = min(n_sweeps, 30)
        print(f"  block_size={block_size}, n_sweeps={n_sweeps}")
        sc, a_opt = block_coordinate_descent(best_a, block_size=block_size,
                                             n_sweeps=n_sweeps, iters_per_block=100)
        report(sc, a_opt, "block CD")

    # ============================================================
    # Phase 6: Final local search with perturbations
    # ============================================================
    print(f"\n=== Phase 6: Final Local Search ===")
    n_best = len(best_a)
    trial = 0
    while time.monotonic() < deadline - 5 and trial < 15:
        trial += 1
        strat = trial % 5
        if strat == 0:
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            a_pert = sharpen(best_a, 1.2 + 0.5 * rng.random())
        elif strat == 2:
            # Reduce top contributors at peak
            b = fast_conv(best_a)
            k_star = int(b.argmax())
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            contrib = np.zeros(n_best)
            j_range = np.arange(j_start, j_end + 1)
            contrib[j_range] = best_a[j_range] * best_a[k_star - j_range]
            top_count = max(5, n_best // 50)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for j in top_idx:
                amt = a_pert[j] * 0.05
                a_pert[j] = max(a_pert[j] - amt, 0.0)
                total_red += amt
            bottom_idx = np.argsort(contrib)[:-top_count]
            if len(bottom_idx) > 0:
                a_pert[bottom_idx] += total_red / len(bottom_idx)
        elif strat == 3:
            # Frequency domain smoothing
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert if 'a_pert' in dir() else best_a, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.05 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Anti-palindromic nudge
            a_pert = best_a.copy()
            center = n_best // 2
            for j in range(n_best):
                if j < center:
                    a_pert[j] *= (1.0 - 0.03)
                else:
                    a_pert[j] *= (1.0 + 0.03)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        sc, a_opt, _ = run_lbfgs(a_pert, 100000, maxiter=3000, maxfun=30000)
        report(sc, a_opt, f"P6 #{trial}")

        # Also do quick subgrad on the result
        sc2, a_opt2 = subgrad_polish(a_opt, max_iters=2000)
        report(sc2, a_opt2, f"P6s #{trial}")

    # ============================================================
    # Final subgradient polish
    # ============================================================
    if time.monotonic() < deadline - 3:
        sc, a_opt = subgrad_polish(best_a, max_iters=5000)
        report(sc, a_opt, "FINAL subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
