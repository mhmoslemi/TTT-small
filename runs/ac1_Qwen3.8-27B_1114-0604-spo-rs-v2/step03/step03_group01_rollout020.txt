"""
Hybrid optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements over previous approach:
1. Palindrome constraint to halve effective dimensionality
2. LP-based dual updates (multiplicative weight) for escaping local minima
3. Aggressive LSE beta continuation with geometric schedule
4. Structured diverse restarts (Gaussian, raised-cosine, Fourier-param)
5. Multi-scale optimization (small-n then upsample)
6. Adaptive coordinate descent with peak-targeting
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_score_grad_pal(half_a, beta):
    """LSE-smoothed score and gradient w.r.t. half of palindromic sequence."""
    n_half = len(half_a)
    n = 2 * n_half
    # Reconstruct full palindrome: a = [half, half[::-1]] for even n
    # Actually for palindrome of length n: a[i] = a[n-1-i]
    # If n is even: a = [h[0], h[1], ..., h[n/2-1], h[n/2-1], ..., h[1], h[0]]
    # That's length n = 2*(n/2) = 2*n_half... wait that gives n = 2*n_half
    # a = concatenate(half, half[::-1]) gives length 2*n_half
    a = np.concatenate([half_a, half_a[::-1]])
    sc, grad_full = lse_score_grad_a(a, beta)
    # Jacobian: da/dhalf[i] = e_i + e_{n-1-i} (for i=0..n_half-1)
    # grad_half[i] = grad_full[i] + grad_full[n-1-i]
    grad_half = grad_full[:n_half] + grad_full[::-1][:n_half]
    return sc, grad_half


def subgrad_score_a(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse_direct(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start,
                    normalize=True, verbose_interval=5000):
    """Adam with LSE smoothing and scheduled beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a

            lr_t = lr / (1.0 + 1e-5 * t)
            score, grad = lse_score_grad_a(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if (bi * steps_per_beta + t) % verbose_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

def lp_dual_update(a, M, max_iter=200, lr=0.1):
    """
    LP-based dual update: find g maximizing sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Uses projected subgradient on the dual for scalability.
    Returns the optimal g (normalized to same sum as a).
    """
    n = len(a)
    # The primal: max 1^T g s.t. A g <= M*1, g >= 0
    # where Ag[k] = sum_j a[k-j]*g[j] (valid range)
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # Where A^T y [j] = sum_k a[k-j]*y[k]
    #
    # We solve the dual using projected subgradient, then recover primal from y.
    # Actually, let's use a simpler multiplicative weight approach.
    
    # Multiplicative weight: start with g = uniform, update based on constraints.
    g = np.ones(n) / n
    y = np.zeros(2 * n - 1)  # dual variables for each constraint
    
    for it in range(max_iter):
        # Compute constraint violations: (a*g)[k] - M for each k
        c = _cross_conv(a, g)
        violations = np.maximum(0, c - M)
        
        # Update dual variables
        y += lr * violations
        
        # Recover primal from dual: g[j] proportional to exp(-sum_k a[k-j]*y[k])
        # A^T y [j] = sum_k a[max(0, k-j)] * y[k] ... 
        # Let's compute: for each j, sum_k a[k-j]*y[k] where 0<=k-j<n
        # = sum_i a[i]*y[i+j] where i = k-j, so j = k-i
        # = (a * y)[j + n - 1] ... approximately
        # Let me just compute it directly via FFT
        M_fft = 1 << (3 * n - 2).bit_length()
        A_fft = np.fft.rfft(a, M_fft)
        Y_fft = np.fft.rfft(y, M_fft)
        Ay = np.fft.irfft(A_fft * Y_fft, M_fft)[:3 * n - 2]
        # A^T y [j] = sum_k y[k] * a[k-j] = (y * a_rev)[j + n - 1]
        a_rev = a[::-1]
        M_fft2 = 1 << (3 * n - 2).bit_length()
        Y_fft2 = np.fft.rfft(y, M_fft2)
        AR_fft2 = np.fft.rfft(a_rev, M_fft2)
        Aty = np.fft.irfft(Y_fft2 * AR_fft2, M_fft2)[:3 * n - 2]
        Aty = Aty[n - 1:2 * n - 1]
        
        # Multiplicative weight update on g
        g = g * np.exp(-lr * Aty)
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    
    return g


def _cross_conv(a, b):
    """Linear cross-convolution of a and b."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def multiplicative_weight_update(a, n_iters=100, lr=0.5):
    """
    Direct multiplicative weight update targeting the max convolution lag.
    At each step, find the peak lag and disproportionate mass away from
    elements contributing to that peak.
    """
    n = len(a)
    g = a.astype(np.float64).copy()
    g = g / g.sum()
    
    for it in range(n_iters):
        c = fast_conv(g)
        k_star = int(c.argmax())
        M = c[k_star]
        
        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            contrib[js:je + 1] = g[idx]
        
        # Reduce elements with high contribution
        g = g * np.exp(-lr * contrib / max(M, 1e-10))
        g = np.maximum(g, 1e-15)
        g = g / g.sum()
    
    return g


def coord_descent_peak_target(a, n_sweeps=50, time_limit_s=None, t_start=None):
    """
    Coordinate descent that targets peak-reducing moves.
    At each step, identify elements contributing to the peak and try to reduce them,
    redistributing mass to non-contributing elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Reduce peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing
            for frac in [0.8, 0.6, 0.4, 0.2]:
                a[j] = a_old * frac
                sc = compute_score(a)
                # Redistribute to non-contributing elements
                removed = a_old - a[j]
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    a[non_contrib] += removed / non_contrib.sum()
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            a[j] = a[j]  # keep best

        # Also try increasing non-contributing elements
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) > 0:
            # Pick a random non-contributing element and increase it
            j = nc_idx[np.random.randint(len(nc_idx))]
            a_old = a[j]
            S = a.sum()
            for mult in [1.2, 1.5, 2.0]:
                a[j] = a_old * mult
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            # Revert if no improvement
            if sc >= score:
                a[j] = a_old

    return best_score, a


def generate_initializations(rng, n):
    """Generate diverse initializations for a sequence of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (centered)
    x = np.arange(n) / n
    sigma = 0.15 + 0.2 * rng.random()
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Sinusoidal modulation
    freq = 2 + 8 * rng.random()
    phase = 2 * np.pi * rng.random()
    s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Multi-frequency
    f1 = 1 + 3 * rng.random()
    f2 = 4 + 8 * rng.random()
    p1 = 2 * np.pi * rng.random()
    p2 = 2 * np.pi * rng.random()
    m = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * x + p1) + 0.1 * np.cos(2 * np.pi * f2 * x + p2)
    m = np.maximum(m, 0.01)
    inits.append(m / m.sum())

    # Two-bump
    c1 = 0.25 + 0.1 * rng.random()
    c2 = 0.65 + 0.1 * rng.random()
    s1 = 0.05 + 0.1 * rng.random()
    s2 = 0.05 + 0.1 * rng.random()
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    # Smooth with a running average
    k = max(3, n // 50)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.5
    h_smooth = np.convolve(h, np.ones(max(3, half_n // 50)) / max(3, half_n // 50), mode='same')
    pal = np.concatenate([h_smooth, h_smooth[::-1]])
    inits.append(pal / pal.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def resample_sequence(a, new_n):
    """Resample to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Diverse initializations + quick Adam screening
    Phase 2: LSE continuation Adam (palindrome) on best candidates
    Phase 3: L-BFGS polishing (palindrome)
    Phase 4: LP/multiplicative weight updates
    Phase 5: Peak-targeting coordinate descent
    Phase 6: Structured restarts
    Phase 7: Final intensive polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==========================================
    # Phase 1: Diverse initializations + screening (15% time)
    # ==========================================
    p1_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Diverse init + screening (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    # Choose working lengths
    if parent is not None:
        n_parent = len(parent)
        n_work = n_parent
        print(f"Parent: n={n_parent}, score={compute_score(parent):.8f}")
        # Try to exploit palindrome structure
        n_half = n_parent // 2
        half_parent = parent[:n_half].copy()
        # Symmetrize
        half_parent = (parent[:n_half] + parent[::-1][:n_half]) / 2.0
        pal_parent = np.concatenate([half_parent, half_parent[::-1]])
        pal_score = compute_score(pal_parent)
        print(f"  Palindromized parent: {pal_score:.8f}")
        if pal_score < best_score:
            best_score = pal_score
            best_a = pal_parent.copy()
    else:
        n_work = 4096
        print("No parent available, starting fresh.")

    # Generate and screen initializations
    n_screen = n_work
    inits = generate_initializations(rng, n_screen)
    # Add parent if available
    if parent is not None and len(parent) == n_screen:
        inits.append(parent.copy())
    if parent is not None and len(parent) != n_screen:
        inits.append(resample_sequence(parent, n_screen))

    candidates = []
    for i, init in enumerate(inits):
        if time.monotonic() - t_start > t_start + p1_time - 5:
            break
        # Quick Adam on each
        beta_sched = [20, 50, 100]
        sc, a_opt = adam_lse_direct(init, beta_sched, lr=0.01, n_steps=3000,
                                     time_limit_s=3.0, t_start=time.monotonic() - 0.5)
        candidates.append((sc, a_opt))
        print(f"  Init {i}: score={sc:.8f}")

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    top_candidates = [c[1] for c in candidates[:5]]
    for sc, a_opt in candidates[:5]:
        try_update(sc, a_opt, f"screening init")

    print(f"  Best after screening: {best_score:.8f}")

    # ==========================================
    # Phase 2: LSE continuation Adam - Palindrome (30% time)
    # ==========================================
    p2_time = effective_budget * 0.30
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LSE continuation Adam - Palindrome (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    # Work in palindrome space
    n_best = len(best_a)
    n_half = n_best // 2
    # Ensure even length for palindrome
    if n_best % 2 != 0:
        n_best = n_best + 1
        best_a = resample_sequence(best_a, n_best)
        n_half = n_best // 2

    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    # Geometric beta schedule
    betas = [10, 30, 100, 300, 1000, 3000, 10000]
    lr_sched = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]

    t_alloc = p2_time / len(betas)
    for i, (beta, lr) in enumerate(zip(betas, lr_sched)):
        remaining = t_start + p2_time - time.monotonic()
        if remaining < 2:
            break
        t0 = time.monotonic()
        sc, h_opt = adam_lse_pal(half_best, [beta], lr=lr, n_steps=100000,
                                  time_limit_s=min(t_alloc * 1.2, remaining),
                                  t_start=t_start)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"Adam-pal beta={beta}")
        half_best = h_opt

    # Also try non-palindrome Adam on the best
    remaining = t_start + p2_time - time.monotonic()
    if remaining > 10:
        betas2 = [100, 500]
        for beta, lr in zip(betas2, [0.005, 0.002]):
            remaining = t_start + p2_time - time.monotonic()
            if remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = adam_lse_direct(best_a, [beta], lr=lr, n_steps=100000,
                                         time_limit_s=min(remaining, p2_time * 0.1),
                                         t_start=t_start)
            try_update(sc, a_opt, f"Adam-direct beta={beta}")

    # ==========================================
    # Phase 3: L-BFGS Polishing - Palindrome (20% time)
    # ==========================================
    p3_time = effective_budget * 0.20
    print(f"\n{'=' * 60}")
    print(f"Phase 3: L-BFGS Polish - Palindrome (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    n_half = len(best_a) // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    for beta in [300, 1000, 3000, 8000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = run_lbfgs_pal(half_best, beta, maxiter=5000, maxfun=80000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"LBFGS-pal beta={beta}")
        half_best = h_opt
        print(f"  LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # L-BFGS without palindrome constraint
    for beta in [500, 2000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"LBFGS-direct beta={beta}")
        print(f"  LBFGS-direct beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 4: LP / Multiplicative Weight Updates (10% time)
    # ==========================================
    p4_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 4: LP / MW Updates (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining = t_start + p4_time - time.monotonic()
    if remaining > 5:
        # Multiplicative weight update
        t0 = time.monotonic()
        g = multiplicative_weight_update(best_a, n_iters=50, lr=0.3)
        # Blend with current best
        for t_blend in [0.3, 0.5, 0.7]:
            a_blend = (1 - t_blend) * best_a + t_blend * g
            a_blend = np.clip(a_blend, 0, 1000)
            sc = compute_score(a_blend)
            try_update(sc, a_blend, f"MW blend={t_blend}")
        print(f"  MW: best={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 5: Peak-targeting coordinate descent (10% time)
    # ==========================================
    p5_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Peak-targeting Coord Descent (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    t0 = time.monotonic()
    sc, a_opt = coord_descent_peak_target(best_a, n_sweeps=100,
                                           time_limit_s=p5_time, t_start=t_start)
    try_update(sc, a_opt, "Peak coord descent")
    print(f"  Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 6: Structured restarts (10% time)
    # ==========================================
    p6_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Structured Restarts (budget={p6_time:.0f}s)")
    print(f"{'=' * 60}")

    n_restarts = 8
    restart_budget = p6_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > t_start + effective_budget - 30:
            break

        strategy = r % 4
        n_cur = len(best_a)

        if strategy == 0:
            # Gaussian perturbation in log-space
            log_a = np.log(np.clip(best_a, 1e-10, 1e10))
            noise_scale = 0.05 + 0.05 * rng.random()
            log_pert = log_a + noise_scale * rng.standard_normal(n_cur)
            a_pert = np.exp(log_pert - np.max(log_pert))
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Peak-avoidance: reduce elements at peak lag, boost others
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_cur + 1)
            je = min(k_star, n_cur - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= (0.5 + 0.2 * rng.random())
            non_peak = np.ones(n_cur, dtype=bool)
            non_peak[js:je + 1] = False
            if non_peak.sum() > 0:
                a_pert[non_peak] *= (1.0 + 0.1 * rng.random())
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Fourier perturbation
            x = np.arange(n_cur) / n_cur
            freqs = rng.uniform(1, 10, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amps = rng.uniform(0.02, 0.08, size=3)
            mod = 1.0
            for f, p, a_ in zip(freqs, phases, amps):
                mod += a_ * np.cos(2 * np.pi * f * x + p)
            a_pert = best_a * np.maximum(mod, 0.1)
            a_pert = a_pert / a_pert.sum()
        else:
            # Block modification
            n_blocks = 4 + int(4 * rng.random())
            a_pert = best_a.copy()
            block_size = n_cur // n_blocks
            for b in range(n_blocks):
                s = b * block_size
                e = min(s + block_size, n_cur)
                a_pert[s:e] *= rng.uniform(0.4, 1.6)
            a_pert = a_pert / a_pert.sum()

        # Quick refinement
        time_left = t_start + effective_budget - 30 - time.monotonic()
        if time_left < 3:
            break
        adam_t = min(restart_budget * 0.7, time_left)
        sc, a_r = adam_lse_direct(a_pert, [50, 200], lr=0.008, n_steps=30000,
                                   time_limit_s=adam_t, t_start=t_start)
        # Quick L-BFGS
        if time.monotonic() < t_start + effective_budget - 25:
            sc2, a_r2 = run_lbfgs_a(a_r, 500, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")

    # ==========================================
    # Phase 7: Final intensive polish (5% time)
    # ==========================================
    p7_time = effective_budget * 0.05
    print(f"\n{'=' * 60}")
    print(f"Phase 7: Final Polish (budget={p7_time:.0f}s)")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget - 5.0

    # High-beta L-BFGS in palindrome space
    n_half = len(best_a) // 2
    if n_half * 2 != len(best_a):
        n_half = len(best_a) // 2
        best_a = resample_sequence(best_a, 2 * n_half)
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    for beta in [5000, 15000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, h_opt = run_lbfgs_pal(half_best, beta, maxiter=8000, maxfun=150000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"final LBFGS-pal beta={beta}")
        half_best = h_opt
        print(f"  Final LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final Adam with very high beta
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic()
        sc, a_opt = adam_lse_direct(best_a, [5000], lr=0.0005, n_steps=50000,
                                     time_limit_s=min(remaining, p7_time),
                                     t_start=t_start)
        try_update(sc, a_opt, "final Adam")
        print(f"  Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # One more coordinate descent pass
    if time.monotonic() < final_deadline - 2:
        sc, a_opt = coord_descent_peak_target(best_a, n_sweeps=20,
                                               time_limit_s=min(5.0, final_deadline - time.monotonic() - 1),
                                               t_start=t_start)
        try_update(sc, a_opt, "final coord")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
