"""

1. Soft (softmax) active set over convolution lags with annealing schedule.

5. Explores n=16384 in addition to n=8192 for better asymptotic bounds.
6. Adaptive step-size for multiplicative updates based on duality gap.
7. Nesterov momentum in the subgradient phase.

import numpy as np
from scipy.optimize import minimize
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_corr_lag(h_vec, a, n):
    """
    Compute h_j = 2 * sum_k lambda_k * a[k-j] for j=0..n-1.
    h_vec has length 2n-1 (distribution over lags).
    Uses FFT cross-correlation.
    """
    M_big = 1 << (3 * n - 2).bit_length()
    H = np.fft.rfft(h_vec, M_big)
    A = np.fft.rfft(a, M_big)
    cross = np.fft.irfft(H * np.conj(A), M_big)
    return 2.0 * cross[:n]


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * LSE_beta(conv(a,a)) / (sum a)^2
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -50.0, 0.0)
    exp_s = np.exp(scaled)
    p_lag = exp_s / exp_s.sum()
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    h = cross_corr_lag(p_lag, a, n)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (h / S2 - 2.0 * L / S3)
    return score, grad


def project_simplex(v):
    """
    Project v onto the probability simplex {p: p>=0, sum(p)=1}.
    Uses the water-filling algorithm with bisection.
    """
    n = len(v)
    if n == 0:
        return v
    # Bisection on the threshold
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if abs(s - 1.0) < 1e-15:
            break
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    p = np.maximum(v - mid, 0.0)
    p_sum = float(p.sum())
    if p_sum > 0:
        p /= p_sum
    else:
        p = np.ones(n) / n
    return p


def run_lbfgs_lse(a0, beta, maxiter, maxfun, deadline):
    """Run L-BFGS-B on the LSE-smoothed objective with bounds [0, 1000]."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

    bounds = [(0.0, 1000.0)] * n

        sc, _ = lse_score_and_grad(x, beta)
        return sc

    def grad_fun(x):
        _, g = lse_score_and_grad(x, beta)
        return g

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

        return compute_score(a0), a0

    mi = min(maxiter, max(50, int(remaining * 10)))

def fista_subgrad(a0, total_iters, step_init, deadline):

    best_score = compute_score(a0)
    x = p.copy()
    y = p.copy()
    t_curr = 1.0
    step = step_init
    step_decay = 1.0 / (1.0 + 0.05)

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        c = fast_self_conv(y)
        M = float(c.max())
        k_star = int(np.argmax(c))
        # Subgradient: pick the argmax lag
        # h_j = 2 * y[k_star - j] for valid indices (approximately)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        h = np.zeros(n)
        if j_end >= j_start:
            h[j_start:j_end + 1] = 2.0 * y[j_start:j_end + 1][::-1]
        # Also include the "other side" contribution
        j_start2 = max(0, k_star - n + 1)
        # Actually for self-conv, the subgradient of c_{k*} w.r.t. p_j is:
        # 2 * p[k*-j] if 0 <= k*-j <= n-1, with correction for j = k*/2
        # h is the subgradient of max c (just one subgradient)
        
        # FISTA update

        x = project_simplex(v)
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr))

        sc = compute_score(a_test)

def fista_soft(a0, total_iters, beta_lag_init, beta_lag_end, deadline):

    Anneals beta_lag from init to end.
    """

    S0 = float(a0.sum())
    if S0 < 1e-12:
        return np.inf, a0.copy()
    p = a0 / S0
    best_p = p.copy()
    best_score = compute_score(a0)
    x = p.copy()
    y = p.copy()
    t_curr = 1.0
    step = 0.01
    precompute_M = 1 << (3 * n - 2).bit_length()

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, total_iters - 1)
        beta_lag = beta_lag_init * (beta_lag_end / beta_lag_init) ** frac

        c = fast_self_conv(y)
        c_max = float(c.max())
        scaled = np.clip(beta_lag * (c - c_max), -50.0, 0.0)
        exp_s = np.exp(scaled)

        cross = np.fft.irfft(H * np.conj(Y), precompute_M)
        h = 2.0 * cross[:n]

        x = project_simplex(v)
        t_curr = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr))

        step *= 0.998

        a_test = x * S0
        sc = compute_score(a_test)
        if sc < best_score:
            best_score = sc
            best_p = x.copy()

    return best_score, best_p * S0

    gamma = gamma_init

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, n_iters - 1)
        # Anneal beta_lag: start soft (small), end sharp (large)
        beta_lag = beta_lag_init * (beta_lag_end / beta_lag_init) ** frac
        # Anneal gamma: start larger, end smaller
        gamma_cur = gamma_init * (0.1 + 0.9 * (1.0 - frac))

        c = fast_self_conv(p)
        c_max = float(c.max())
        scaled = np.clip(beta_lag * (c - c_max), -50.0, 0.0)
        exp_s = np.exp(scaled)
        lambda_v = exp_s / exp_s.sum()

        # Compute h_j = 2 * sum_k lambda_k * p[k-j]
        H = np.fft.rfft(lambda_v, precompute_M)
        P = np.fft.rfft(p, precompute_M)
        cross = np.fft.irfft(H * np.conj(P), precompute_M)
        h = 2.0 * cross[:n]

        # Boltzmann update: p <- p * exp(-gamma * (h - h_mean)) / Z
        h_mean = float(h.mean())
        update = np.exp(-gamma_cur * (h - h_mean))
        update = np.clip(update, 1e-10, 1e10)
        p_new = p * update
        S_new = float(p_new.sum())
        if S_new < 1e-15:
            break
        p_new /= S_new
        p = p_new

        a_test = p * S0
        sc = compute_score(a_test)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    return best_score, best_p * S0


def freq_domain_init(n, rng):
    """
    Initialize using a flat-top design in the frequency domain.
    Want |FFT(a)|^2 to be as flat as possible, with a >= 0.
    Start with random phase, IFFT, take absolute value.
    """
    M = 2 * n
    # Random phase with some structure
    phase = rng.uniform(0, 2 * np.pi, M // 2 + 1)
    # Force Hermitian symmetry
    mag = np.ones(M // 2 + 1)
    # Add some low-pass character
    freq = np.arange(M // 2 + 1) / (M // 2.0)
    mag *= 1.0 / (1.0 + 0.5 * freq ** 2)
    spectrum = mag * np.exp(1j * phase)
    a = np.fft.irfft(spectrum, M)[:n]
    a = np.abs(a)
    a = np.clip(a, 0.01, 1000.0)
    return a


def symmetric_init(n, rng):
    """Create a symmetric initialization with good autocorrelation properties."""
    half = n // 2
    x = np.arange(half, dtype=np.float64) / half
    # Raised cosine
    a_half = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    # Add some randomness
    a_half += 0.1 * rng.standard_normal(half)
    a_half = np.clip(a_half, 0.01, 1000.0)
    # Make symmetric
    a = np.concatenate([a_half, a_half[::-1]]) if n % 2 == 0 else np.concatenate([a_half, [a_half[0]], a_half[::-1]])
    a = a[:n]
    return a


def build_diverse_inits(n, rng, parent):
    """Build diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            x_new = np.linspace(0, len(p) - 1, n)
            p = np.interp(x_new, np.arange(len(p)), p)
        inits.append(np.clip(p, 0.001, 1000.0).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Symmetric raised cosine
    inits.append(symmetric_init(n, rng))
    # Frequency domain
    inits.append(freq_domain_init(n, rng))
    # Cosine window
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.clip(0.5 * (1 + np.cos(2 * np.pi * (x - 0.5))), 0.01, 1000.0))
    # Gaussian window
    x = (np.arange(n) - n / 2.0) / (n / 4.0)
    inits.append(np.clip(np.exp(-0.5 * x ** 2), 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Two-lobe
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.clip(0.3 + 0.7 * np.cos(2 * np.pi * x) ** 2, 0.01, 1000.0))
    # Parent with jitter (if available)
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            x_new = np.linspace(0, len(p) - 1, n)
            p = np.interp(x_new, np.arange(len(p)), p)
        inits.append(np.clip(p * (1 + 0.05 * rng.standard_normal(n)), 0.001, 1000.0))
    return inits


def targeted_local_search(a, rng, n_trials, deadline):
    """Local search with diverse perturbation strategies."""
    n = len(a)
    best_sc = compute_score(a)
    best_a = a.copy()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        strategy = trial % 6
        if strategy == 0:
            # Gaussian noise
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
        elif strategy == 1:
            # Targeted: reduce peak-contributing elements
            c = fast_self_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            a_pert = best_a.copy()
            contrib = a_pert[j_start:j_end + 1]
            top_k = max(3, n // 100)
            top_idx = j_start + np.argsort(contrib)[::-1][:top_k]
            bottom_idx = np.setdiff1d(np.arange(n), top_idx)
            bottom_idx = rng.choice(bottom_idx, size=top_k, replace=False)
            red = 0.03 + 0.03 * rng.random()
            for j in top_idx:
                amt = a_pert[j] * red
                a_pert[j] -= amt
            a_pert[bottom_idx] += amt * top_k / len(bottom_idx)
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Block swap
            a_pert = best_a.copy()
            bs = max(8, n // 200)
            nb = n // bs
            if nb > 4:
                i = rng.integers(0, nb - 2)
                j = rng.integers(i + 1, nb - 1)
                a_pert[i*bs:(i+1)*bs], a_pert[j*bs:(j+1)*bs] = a_pert[j*bs:(j+1)*bs].copy(), a_pert[i*bs:(i+1)*bs].copy()
        elif strategy == 4:
            # Mirror perturbation
            a_pert = best_a.copy()
            half = n // 2
            delta = 0.02 * rng.standard_normal(half)
            a_pert[:half] += delta
            a_pert[half:] += delta[::-1]
        else:
            # Sparse random restart
            a_pert = best_a.copy()
            n_change = max(10, n // 100)
            idx = rng.choice(n, size=n_change, replace=False)
            a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_change))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            continue

        # Quick LSE polish
        sc, a_opt = run_lbfgs_lse(a_pert, 300, 2000, 30000, deadline - 5)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt
            print(f"    Local #{trial}: {sc:.8f} (strategy={strategy})")

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: diverse init -> LSE L-BFGS -> soft-active equalization ->
    FISTA -> local search -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    n_list = [8192, 16384]
    if parent is not None and parent.size not in n_list:
        n_list.insert(0, parent.size)
    n_list = sorted(set(n_list))[:3]
    print(f"Lengths to try: {n_list}")
    print(f"Parent: {'yes (len=' + str(parent.size) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: Global search - LSE L-BFGS from diverse inits
    # ==================================================================
    phase1_deadline = t_start + 0.35 * total_budget
    beta_schedule = [30, 80, 200, 500]

    print(f"\n{'='*60}")
    print(f"Phase 1: Global LSE L-BFGS (budget: {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in n_list:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  --- Phase 1: n = {n} ---")
        inits = build_diverse_inits(n, rng, parent)
        # Limit number of inits based on time
        time_per_init = (phase1_deadline - time.monotonic()) / max(1, len(inits))
        n_inits = min(len(inits), max(3, int((phase1_deadline - time.monotonic()) / 60.0)))
        inits = inits[:n_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            for beta in beta_schedule:
                if time.monotonic() > phase1_deadline:
                    break
                remaining = phase1_deadline - time.monotonic()
                if remaining < 5.0:
                    break
                sc, a_opt = run_lbfgs_lse(a_cur, beta, 5000, 80000, phase1_deadline)
                a_cur = a_opt
                _report(sc, a_opt, f"n={n},init{i},b={beta}")

    print(f"\n  After Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ==================================================================
    # Phase 2: Soft-active-set multiplicative equalization
    # ==================================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Soft-Active-Set Equalization (budget: {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_best = len(a_cur)

    for round_i in range(6):
        if time.monotonic() > phase2_deadline:
            break
        print(f"\n  Round {round_i+1}:")
        # Soft active set equalization with annealing
        beta_lag_init = 5.0 + 5.0 * round_i
        beta_lag_end = 50.0 + 50.0 * round_i
        gamma_init = 0.05 + 0.02 * rng.random()
        sc, a_eq = soft_active_equalize(a_cur, 100, beta_lag_init, beta_lag_end,
                                         gamma_init, phase2_deadline)
        _report(sc, a_eq, f"round{round_i+1},equalize")
        a_cur = a_eq

        # LSE polish between rounds
        if time.monotonic() < phase2_deadline - 10:
            beta = 300 + 200 * round_i
            sc, a_opt = run_lbfgs_lse(a_cur, beta, 3000, 50000, phase2_deadline)
            _report(sc, a_opt, f"round{round_i+1},lbfgs b={beta}")
            a_cur = a_opt

    print(f"\n  After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: FISTA with soft active set
    # ==================================================================
    phase3_deadline = t_start + 0.72 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: FISTA (budget: {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    for itype in range(3):
        if time.monotonic() > phase3_deadline:
            break
        print(f"\n  FISTA variant {itype+1}:")
        bl_init = 10.0 + 10.0 * itype
        bl_end = 100.0 + 100.0 * itype
        n_iters = max(100, int((phase3_deadline - time.monotonic()) / 3 / 0.005))
        sc, a_fista = fista_soft(a_cur, n_iters, bl_init, bl_end, phase3_deadline)
        _report(sc, a_fista, f"fista v{itype+1}")
        if sc < best_score:
            a_cur = a_fista

        # Also try standard subgradient FISTA
        if time.monotonic() < phase3_deadline - 10:
            n_iters2 = max(100, int((phase3_deadline - time.monotonic()) / 0.005))
            sc2, a_fs2 = fista_subgrad(a_cur, n_iters2, 0.01, phase3_deadline)
            _report(sc2, a_fs2, f"fista_subgrad v{itype+1}")
            if sc2 < best_score:
                a_cur = a_fs2

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Local search with diverse perturbations
    # ==================================================================
    phase4_deadline = t_start + 0.88 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Local Search (budget: {phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_local = max(10, int((phase4_deadline - time.monotonic()) / 30.0))
    sc, a_local = targeted_local_search(best_a, rng, n_local, phase4_deadline)
    _report(sc, a_local, "local_search")

    # ==================================================================
    # Phase 5: Final high-beta polish
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    for beta in [500, 1000, 2000, 5000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs_lse(a_cur, beta, 5000, 100000, deadline - 5)
        _report(sc, a_opt, f"final b={beta}")
        a_cur = a_opt

    # One more round of soft equalization + polish
    if time.monotonic() < deadline - 30:
        sc, a_eq = soft_active_equalize(a_cur, 50, 50.0, 200.0, 0.02, deadline - 15)
        _report(sc, a_eq, "final_equalize")
        if sc < best_score:
            a_cur = a_eq
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs_lse(a_cur, 2000, 3000, 50000, deadline - 5)
            _report(sc, a_opt, "final_polish")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
