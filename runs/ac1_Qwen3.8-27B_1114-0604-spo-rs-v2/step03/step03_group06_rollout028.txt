"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines spectral perturbation reinitialization, multi-peak subgradient descent,
L-BFGS-B with LSE continuation, and LP-based direction finding.
Novel elements: Fourier-domain perturbations, weighted multi-peak subgradients,
adaptive length exploration, and aggressive final polishing.

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def multi_peak_subgrad_descent(a, n_iters=500, K=5, temp_decay=0.995):
    """
    Multi-peak subgradient descent on the exact max-conv objective.
    Uses weighted average of top-K peak subgradients for smoother descent.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-K peaks
        c_sorted_idx = np.argsort(c)[::-1]
        top_idx = c_sorted_idx[:K]
        top_vals = c[top_idx]
        # Weight: exponential of proximity to max
        weights = np.exp(-top_vals * 0.0 + (M - top_vals) / (M * 0.01 + 1e-10))
        weights = np.exp(-(M - top_vals) / (0.01 * M + 1e-10))
        weights /= weights.sum()

        # Combined subgradient
        sg = np.zeros(n)
        for ki, k in enumerate(top_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * weights[ki] * seg

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polak-ribiere-like step size
        step = (sc - sc_lower) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        step = max(step, 1e-8)

        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Line search: try a few step sizes
        best_trial_sc = compute_score(a_new)
        best_trial_a = a_new.copy()
        for mult in [0.5, 0.25, 2.0]:
            a_trial = a - step * mult * grad
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            if a_trial.sum() < 0.01:
                continue
            sc_trial = compute_score(a_trial)
            if sc_trial < best_trial_sc:
                best_trial_sc = sc_trial
                best_trial_a = a_trial.copy()

        if best_trial_sc < sc:
            a = best_trial_a
        else:
            # Keep current but add tiny random perturbation to escape
            a = a + 1e-6 * np.random.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_perturb(a, rng, mode='phase', sigma=0.3):
    """
    Perturb the sequence in the Fourier domain to generate diverse nearby solutions.
    mode='phase': random phase perturbation
    mode='magnitude': random magnitude perturbation
    mode='both': both
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = np.fft.fft(a_padded)

    if mode == 'phase':
        phase_pert = sigma * 2 * np.pi * rng.standard_normal(N)
        A_new = A * np.exp(1j * phase_pert)
    elif mode == 'magnitude':
        mag_pert = 1.0 + sigma * rng.standard_normal(N)
        A_new = A * mag_pert
    else:  # both
        phase_pert = sigma * np.pi * rng.standard_normal(N)
        mag_pert = 1.0 + sigma * 0.5 * rng.standard_normal(N)
        A_new = A * mag_pert * np.exp(1j * phase_pert)

    # Enforce conjugate symmetry for real output
    A_new[0] = np.real(A_new[0])
    if N % 2 == 0:
        A_new[N // 2] = np.real(A_new[N // 2])
    for f in range(1, N // 2 + 1):
        A_new[N - f] = np.conj(A_new[f])

    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 0.01:
        a_new = a.copy()
    return a_new


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def solve_lp_direction(a, max_wall_s=25.0):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 5000:
        return None
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()

    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None

    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    # Asymmetric (important for reducing center peak)
    inits.append(0.3 + 0.4 * x)
    inits.append(0.3 + 0.4 * (1.0 - x))
    inits.append(0.2 + 0.6 * np.sin(np.pi * x) ** 0.5)
    # Bimodal
    inits.append(0.3 + 0.5 * (np.cos(4.0 * np.pi * x - np.pi / 2) ** 2))
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, 1000.0))
    # Block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_block = np.repeat(vals, per)
        if len(a_block) < n:
            a_block = np.concatenate([a_block, vals[:n - len(a_block)]])
        inits.append(a_block[:n].copy())
    return inits


def gs_flat_iter(a, N, n_iters=200, relax=1.02):
    """
    Gerchberg-Saxton style iteration: flatten the non-DC spectrum.
    Returns the final sequence.
    """
    n = len(a)
    a_cur = a.astype(np.float64).copy()

    for it in range(n_iters):
        a_padded = np.zeros(N)
        a_padded[:n] = a_cur
        A = np.fft.fft(a_padded)

        # Flatten non-DC magnitudes
        mags = np.abs(A)
        dc_val = A[0]
        non_dc_mag_mean = np.mean(mags[1:])
        if non_dc_mag_mean < 1e-15:
            break

        phases = np.angle(A)
        A_new = np.zeros(N, dtype=complex)
        A_new[0] = dc_val
        A_new[1:] = non_dc_mag_mean * np.exp(1j * phases[1:])

        # Relaxation
        A_blend = (1.0 - relax) * A + relax * A_new

        # Enforce conjugate symmetry
        A_blend[0] = np.real(A_blend[0])
        if N % 2 == 0:
            A_blend[N // 2] = np.real(A_blend[N // 2])
        for f in range(1, N // 2 + 1):
            A_blend[N - f] = np.conj(A_blend[f])

        a_new_full = np.fft.ifft(A_blend).real
        a_cur = np.maximum(a_new_full[:n], 0.0)

    return a_cur

            a_pert = np.fft.ifft(A).real[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = a_pert / max(a_pert.sum(), 1e-10) * best_a.sum()
        else:
            # Mix with structured sequence
            x = np.arange(n_best) / n_best
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.05 + 0.1 * rng.random()
            a_struct = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_struct, 0.0, 1000.0)

        # Polish
        if a_pert.sum() < 0.01:
            trial += 1
            continue
        sc, a_opt = run_lbfgs(a_pert, 800, maxiter=2000, maxfun=5000)
        report(sc, a_opt, f"local #{trial} strat={strategy}")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final aggressive polishing (10% time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polishing")
    print(f"{'='*60}")

    # Very high beta LSE
    for beta in [5000, 10000, 20000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final LBFGS beta={beta}")
        if sc < best_score:
            best_a = a_opt

    # Multi-peak subgradient final
    if time.monotonic() < deadline - 10:
        sc, a_pol = multi_peak_subgrad_descent(best_a, n_iters=2000, K=8)
        report(sc, a_pol, "final subgrad")
        if sc < best_score:
            best_a = a_pol

    # One more L-BFGS from polished
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final final LBFGS")
        if sc < best_score:
            best_a = a_opt

    # =========================================================================
    # Final verification
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
