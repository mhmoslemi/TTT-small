"""
Optimized peak-reduction solver using Adam on LSE-smoothed objective,
peak-targeted coordinate descent, and spectral shaping.
Targets improvement from 1.5075 to below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta):
    """Compute LSE-smoothed score and gradient (with sum normalization built in)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: 2 * (p * a_reversed) for the convolution
    # conv(a,a)[k] = sum_j a[j] * a[k-j]
    # dL/da[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    grad_L = 2.0 * np.fft.irfft(RA * P, M2_fft)[:n]
    # d(score)/da = 2n * (grad_L / S^2 - 2L * grad_S / S^3)
    # where grad_S = 1 for all components (S = sum(a))
    S2 = S * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / (S2 * S)) * np.ones(n)
    # Wait, dS/da[j] = 1, so d(score)/da[j] = 2n * (dL/da[j] / S^2 - L * 2*S' / S^3)
    # = 2n * (grad_L[j] / S^2 - 2*L / S^3)
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / (S2 * S))
    return score, grad

def adam_optimize(a0, beta, n_iters=5000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_iters + 1):
        score, grad = lse_grad(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
        # Renormalize
        S = a.sum()
        if S < 1e-12:
            a = a0.copy()
            break
        a = a / S
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_score, best_a

def peak_shave(a, n_iters=200, strength=0.1):
    """Directly reduce the convolution peak by modifying contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        # Find contributing indices
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        
        # Contributions to the peak
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        
        # Reduce top contributors
        top_n = max(5, n // 50)
        top_idx = np.argsort(contrib)[::-1][:top_n]
        
        # Reduce them and add to bottom contributors
        total_red = 0.0
        for idx in top_idx:
            red = strength * a[idx] / len(top_idx)
            a[idx] -= red
            a[idx] = max(a[idx], 0.0)
            total_red += red
        
        # Distribute to non-peak-contributing elements
        bottom_idx = np.argsort(contrib)[:top_n]
        for idx in bottom_idx:
            a[idx] += total_red / len(bottom_idx)
        
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

def coordinate_descent_exact(a, n_sweeps=5, n_top=100):
    """Exact 1D coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Sort by contribution
        contrib = np.array([a[j] * a[k_star - j] for j in range(j_start, j_end + 1)])
        top_n = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_n]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old = a[j]
            # Try multiple values
            best_val = a_old
            best_sc = best_score
            for factor in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 0.0]:
                cv = a_old * factor
                a[j] = cv
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
            a[j] = best_val
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

def spectral_shaping(a, n_iters=100, target='flat'):
    """
    Shape the spectrum to reduce convolution peak.
    In frequency domain, make |A(f)|^2 flatter.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.fft(a, M)
        # Power spectrum
        P = np.abs(A)**2
        P_mean = P.mean()
        
        # Compress the power spectrum
        alpha = 0.5 + 0.5 * (it / n_iters)  # Start gentle, get more aggressive
        P_new = P_mean + (P - P_mean) * (1 - alpha * 0.3)
        P_new = np.maximum(P_new, 1e-10)
        
        # New amplitude (keep phase)
        A_new = A * np.sqrt(P_new / P)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-12:
            break
        a_new = a_new / S
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_score, best_a

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a0, beta, n_iters, lr, momentum_decay=0.9, max_step=0.5):
    """Adam-variant optimizer for LSE-smoothed objective with peak-aware scaling."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    lr0 = lr

    for t in range(1, n_iters + 1):
        score, grad = lse_score_grad(a, beta)
        # Adaptive learning rate (cosine-like decay)
        progress = (t - 1) / max(1, n_iters - 1)
        lr_t = lr0 * (0.3 + 0.7 * (1.0 - 0.5 * (1.0 + np.cos(np.pi * progress))))

        m = momentum_decay * m + (1.0 - momentum_decay) * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - momentum_decay ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        step_vec = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        # Clip step size per-coordinate
        step_max = max_step / (S + 1e-10)
        step_vec = np.clip(step_vec, -step_max, step_max)

        a = a - step_vec
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_score, best_a


def peak_aware_adam(a0, beta, n_iters, lr):

def spectral_flatten(a, n_iters=200, strength=0.3):
    """Flatten the power spectrum to reduce convolution peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()

        return a

    for it in range(n_iters):

        # Compress power spectrum toward flat

        P_new = P.mean() + (P - P.mean()) * (1.0 - alpha * 0.5)
        P_new = np.maximum(P_new, 1e-20)
        A_new = A * np.sqrt(P_new / np.maximum(P, 1e-30))
        a_new = np.fft.ifft(A_new).real[:n]

            best_a = a_new.copy()
        # Interpolate for stability
        a = (1.0 - 0.5) * a + 0.5 * a_new
        S = a.sum()
        a = a / S

    return best_a


def peak_shave_adaptive(a, n_iters=300):
    """Adaptive peak shaving: reduce peak contributors, redistribute to valleys."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        peak_val = float(c[k_star])
        mean_val = float(c.mean())
        if peak_val - mean_val < 1e-12:
            break

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        # Compute contributions
        indices = np.arange(j_start, j_end + 1)
        contrib = a[indices] * a[k_star - indices]
        total_contrib = contrib.sum()

        # Adaptive strength: stronger when peak is much above mean
        excess_ratio = (peak_val - mean_val) / (peak_val + 1e-30)
        strength = 0.02 + 0.15 * excess_ratio

        # Reduce top contributors
        n_top = max(10, min(n // 20, len(contrib)))
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local

        # Find where to add (elements not contributing to peak)
        non_peak = np.zeros(n)
        non_peak[indices] = 1.0
        # Add to elements with low values
        candidates = np.where(non_peak == 0)[0]
        if len(candidates) == 0:
            candidates = indices[np.argsort(contrib)[:n_top]]

        total_red = 0.0
        for idx in top_global:
            red = strength * a[idx] / len(top_global)
            a[idx] -= red
            a[idx] = max(a[idx], 0.0)
            total_red += red

        # Distribute to low-value non-peak elements
        n_add = min(len(candidates), n_top)
        add_targets = candidates[np.argsort(a[candidates])[:n_add]]
        per_add = total_red / max(1, n_add)
        for idx in add_targets:
            a[idx] += per_add

        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def coord_descent_top(a, n_sweeps=5, n_top=60):
    """Coordinate descent on top peak contributors with golden-section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    gr = (1.0 + np.sqrt(5.0)) / 2.0

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        indices = np.arange(j_start, j_end + 1)
        contrib = a[indices] * a[k_star - indices]
        top_n = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_n]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-15:
                continue
            # Golden section search in [0, 2*a_old]
            lo = 0.0
            hi = 2.0 * a_old
            x1 = hi - (hi - lo) / gr
            x2 = lo + (hi - lo) / gr
            a[j] = x1
            f1 = compute_score(a)
            a[j] = x2
            f2 = compute_score(a)

            for _ in range(8):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - (hi - lo) / gr
                    a[j] = x1
                    f1 = compute_score(a)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + (hi - lo) / gr
                    a[j] = x2
                    f2 = compute_score(a)

            a[j] = x1 if f1 < f2 else x2

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B on LSE objective."""

def error_reduction_v2(a, n_iters=200, alpha=0.7):

        sc = compute_score(a_new)

        print(f"  ER alpha={alpha}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_new = error_reduction_v2_score(best_a, 200, alpha)
        print(f"took {time.monotonic()-t0:.1f}s -> {sc:.8f}")
        report(sc, a_new, f"ER a={alpha}")

    if time.monotonic() < p2_end:
        print("  Spectral flatten...", end=" ", flush=True)
        t0 = time.monotonic()
        a_sf = spectral_flatten(best_a, 150, 0.4)
        sc = compute_score(a_sf)
        print(f"took {time.monotonic()-t0:.1f}s -> {sc:.8f}")
        report(sc, a_sf, "Spectral")

    print(f"  Phase 2 best: {best_score:.8f}")

    # Phase 3: L-BFGS refinement (20% time)
    p3_end = t_start + 0.80 * T
    print(f"\n=== Phase 3: L-BFGS ===")

    for beta in [5000, 10000, 20000]:
        if time.monotonic() > p3_end:
            break
        print(f"  LBFGS beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_new = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        print(f"took {time.monotonic()-t0:.1f}s -> {sc:.8f}")
        report(sc, a_new, f"LBFGS beta={beta}")

    # Try from spectral result too
    if time.monotonic() < p3_end:
        a_sf2 = spectral_flatten(best_a, 100, 0.5)
        sc, a_new = run_lbfgs(a_sf2, 5000, maxiter=2000, maxfun=30000)
        report(sc, a_new, "LBFGS from spectral")

    print(f"  Phase 3 best: {best_score:.8f}")

    # Phase 4: Peak shaving + coordinate descent (15% time)
    p4_end = t_start + 0.95 * T
    print(f"\n=== Phase 4: Peak Reduction ===")

    sc, a_new = peak_shave_adaptive(best_a, 300)
    report(sc, a_new, "PeakShave")

    sc, a_new = coord_descent_top(best_a, n_sweeps=3, n_top=40)
    report(sc, a_new, "CoordDesc")

    # More peak shaving after CD
    sc, a_new = peak_shave_adaptive(best_a, 200)
    report(sc, a_new, "PeakShave2")

    # Final Adam from best
    if time.monotonic() < p4_end:
        sc, a_new = peak_aware_adam(best_a, 10000, 2000, lr=0.003)
        report(sc, a_new, "Final Adam high-beta")

    # Final short ER
    sc, a_new = error_reduction_v2_score(best_a, 100, 0.85)
    report(sc, a_new, "Final ER")

    # One more L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_new = run_lbfgs(best_a, 50000, maxiter=1000, maxfun=20000)
        report(sc, a_new, "Final LBFGS beta=50000")

    # Final peak shave
    sc, a_new = peak_shave_adaptive(best_a, 100)
    report(sc, a_new, "Very Final PeakShave")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {T}s")
    print(f"{'=' * 60}")
    return out
