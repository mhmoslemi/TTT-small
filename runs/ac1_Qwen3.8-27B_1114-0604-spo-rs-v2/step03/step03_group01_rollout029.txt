"""
Peak-directed optimization with spectral decomposition, simulated annealing,
and L-BFGS polishing for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad_a(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def softmax_fn(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def run_lbfgs_a(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def peak_directed_descent(a, n_steps=50000, lr_init=0.01, momentum=0.9,
                          time_limit_s=None, t_start=None, normalize=True):
    """
    Subgradient descent with momentum, directed at reducing the peak.
    Uses Lagrange multiplier to maintain sum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Subgradient of max(conv) w.r.t. a
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        # Full gradient of score = 2n * M / S^2
        # d(score)/d(a_j) = 2n * (dM/da_j * S^2 - M * 2*S * 1) / S^4
        #                   = 2n * (sg_j / S^2 - 2M / S^3)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Adaptive learning rate
        lr_t = lr_init / (1.0 + 0.01 * t * (S / S0)**2)
        
        # Momentum update
        v = momentum * v - lr_t * grad
        a = a + v
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            a = a * (S0 / a.sum())
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    
    return best_score, best_a

    """Find the LP direction: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""

    # The constraint matrix A has A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    
    # This is a Toeplitz matrix. We can build it efficiently.
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1 and 0 <= j <= n-1
    #          = 0 otherwise
    
    # For k = 0, ..., 2n-2 and j = 0, ..., n-1:
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0

    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_and_grad_info(a):
    """Return convolution, max value, argmax, and sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M, S


def subgrad_at_peak(a, k_star=None):
    """Compute subgradient of the peak of self-convolution w.r.t. a."""
    n = len(a)
    c, k_star, M, S = conv_and_grad_info(a)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
    return c, k_star, M, S, sg


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def run_lbfgs_a(a0, beta, maxiter=2000, maxfun=50000, time_budget=None, t_start=None):
    """L-BFGS-B in a-space with bounds and LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def objective_and_grad(x):
        return lse_score_grad_a(x, beta)

    if time_budget is not None and t_start is not None:
        deadline = t_start + time_budget
        maxfun_adj = maxfun
        def timed_obj(x):
            if time.monotonic() > deadline:
                raise TimeoutError
            return lse_score_grad_a(x, beta)
        try:
            res = minimize(
                timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-14, 'gtol': 1e-11,
                         'maxiter': maxiter, 'maxfun': maxfun_adj, 'maxls': 50}
            )
        except (TimeoutError, Exception):
            a_out = a0
            exact_sc = compute_score(a_out)
            return exact_sc, a_out
    else:
        res = minimize(
            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-11,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def golden_section_1d(a, j, n_evals=25):
    """
    Golden-section search for optimal value of a[j].
    Returns the best value found.
    """
    a_old = a[j]
    if a_old < 1e-15:
        # Try small positive values and zero
        candidates = [0.0, 0.01 * a.max(), 0.05 * a.max(), 0.1 * a.max()]
    else:
        # Search in [0, a_old * 5] centered around a_old
        lo = 0.0
        hi = a_old * 4.0
        if hi < 1e-10:
            hi = a.max() * 2.0

    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def coord_descent_golden(a, n_sweeps=3, top_k=50, time_budget=60.0, t_start=None):
    """
    Coordinate descent with golden-section 1D search on top-k coordinates.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c, k_star, M, S, sg = subgrad_at_peak(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Gradient of the score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Select top-k coordinates by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=15)
            if sc < best_score:
                best_score = sc

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, a


def adam_direct(a_init, lr=0.005, n_steps=100000, beta_lse=200,
                time_budget=None, t_start=None, normalize=True,
                lr_decay=1e-5):
    """Adam optimizer in a-space with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        lr_t = lr / (1.0 + lr_decay * t)
        score, grad = lse_score_grad_a(a, beta_lse)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a * (S0 / s)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def spectral_perturbation(a, rng, strength=0.1):
    """
    Perturb the sequence in the DFT domain: modify amplitudes and/or phases.
    This creates diverse structures that differ in their frequency content.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    # Modify amplitudes with random multiplicative factors
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)

    # Modify phases (careful with DC and Nyquist)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))

    A_new = A * amp_factors * np.exp(1j * phase_shifts)

    # Enforce conjugate symmetry (rfft already handles this implicitly)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)

    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())

    return a_new


def block_swap_perturbation(a, rng, n_blocks=2):
    """Swap random blocks of the sequence."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        block_len = int(rng.integers(n // 20, n // 5))
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if i != j:
            a_new[i:i+block_len], a_new[j:j+block_len] = a_new[j:j+block_len].copy(), a_new[i:i+block_len].copy()
    return a_new


def gaussian_bump_perturbation(a, rng, n_bumps=3):
    """Add/subtract Gaussian bumps at random locations."""
    n = len(a)
    a_new = a.copy()
    S0 = a.sum()
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(n // 50, n // 10))
        amplitude = float(rng.uniform(-0.15, 0.15)) * S0 / n
        x = np.arange(n, dtype=np.float64)
        bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new += bump
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def antipalindromic_perturbation(a, rng, strength=0.1):
    """
    Add an anti-palindromic component: a_j increases where a_{n-1-j} decreases.
    This reduces the middle convolution peak.
    """
    n = len(a)
    a_new = a.copy()
    # Create a random "tilt" vector that is anti-palindromic
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / center  # Linear tilt: -1 to 1
    # Add some frequency variation
    freq = float(rng.uniform(1, 5))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    # Make it anti-palindromic: keep only the odd part
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new += strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def chirp_perturbation(a, rng):
    """Add a chirp (frequency-modulated sinusoid) component."""
    n = len(a)
    a_new = a.copy()
    x = np.arange(n, dtype=np.float64) / n
    f0 = float(rng.uniform(1, 3))
    f1 = float(rng.uniform(5, 15))
    phase = float(rng.uniform(0, 2 * np.pi))
    chirp = np.sin(2.0 * np.pi * (f0 * x + (f1 - f0) * x * x / 2.0 + phase))
    amplitude = float(rng.uniform(0.02, 0.08)) * a.max()
    a_new += amplitude * chirp
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def continuation_lbfgs(a0, betas, time_budget, t_start, maxiter_per_stage=2000):
    """
    Continuation method: run L-BFGS with increasing beta.
    Each stage warms up from the previous result.
    """
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        time_alloc = min(remaining * 0.4, 60.0)
        sc, a_out, _ = run_lbfgs_a(a, beta, maxiter=maxiter_per_stage,
                                    maxfun=10000, time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Continuation beta={beta}: score={sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a = a_out

    return best_score, best_a


def refine_candidate(a_init, time_budget, t_start):
    """
    Multi-stage refinement: Adam -> L-BFGS continuation -> Coord descent.
    """
    a = a_init.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False

    # Stage 1: Adam (40% of budget)
    if time.monotonic() < deadline - 5:
        adam_time = min(remaining := deadline - time.monotonic(), time_budget * 0.4)
        sc, a_opt = adam_direct(a, lr=0.003, n_steps=100000, beta_lse=200,
                                time_budget=adam_time, t_start=t_start)
        try_update(sc, a_opt, "Adam")
        a = best_a

    # Stage 2: L-BFGS continuation (40% of budget)
    if time.monotonic() < deadline - 5:
        lbfgs_time = min(deadline - time.monotonic(), time_budget * 0.4)
        sc, a_opt = continuation_lbfgs(a, [100, 400, 1500, 5000], lbfgs_time, t_start,
                                        maxiter_per_stage=1500)
        try_update(sc, a_opt, "L-BFGS cont")
        a = best_a

    # Stage 3: Coordinate descent (20% of budget)
    if time.monotonic() < deadline - 3:
        coord_time = min(deadline - time.monotonic() - 3, time_budget * 0.2)
        sc, a_opt = coord_descent_golden(a, n_sweeps=5, top_k=30,
                                         time_budget=coord_time, t_start=t_start)
        try_update(sc, a_opt, "CoordDesc")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function: Continuation L-BFGS + Spectral restarts + Coord descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: Intensive continuation L-BFGS from parent (35% of budget)
    # ==================================================================
    p1_budget = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 1: Continuation L-BFGS from parent (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    # First, a quick Adam to get out of any bad local structure
    quick_adam_time = min(20.0, p1_budget * 0.15)
    sc, a_quick = adam_direct(parent, lr=0.005, n_steps=50000, beta_lse=100,
                              time_budget=quick_adam_time, t_start=t_start)
    try_update(sc, a_quick, "Quick Adam")

    # Continuation L-BFGS
    remaining = p1_budget - (time.monotonic() - t_start)
    if remaining > 10:
        sc, a_cont = continuation_lbfgs(best_a, [50, 200, 800, 3000, 10000],
                                         remaining, t_start, maxiter_per_stage=2000)
        try_update(sc, a_cont, "Continuation LBFGS")

    # ==================================================================
    # Phase 2: Diverse restarts with structural perturbations (40% of budget)
    # ==================================================================
    p2_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget
    n_restarts = 10
    per_restart_time = p2_budget / (n_restarts + 1)

    # Define perturbation strategies
    strategies = [
        ("spectral", lambda a, r: spectral_perturbation(a, r, strength=0.15)),
        ("spectral_strong", lambda a, r: spectral_perturbation(a, r, strength=0.3)),
        ("block_swap", lambda a, r: block_swap_perturbation(a, r, n_blocks=3)),
        ("gaussian", lambda a, r: gaussian_bump_perturbation(a, r, n_bumps=5)),
        ("anti-palin", lambda a, r: antipalindromic_perturbation(a, r, strength=0.15)),
        ("chirp", lambda a, r: chirp_perturbation(a, r)),
        ("spectral+anti", None),  # Combined, handled below
        ("gaussian+swap", None),  # Combined, handled below
        ("spectral_weak", lambda a, r: spectral_perturbation(a, r, strength=0.08)),
        ("block_swap_1", lambda a, r: block_swap_perturbation(a, r, n_blocks=1)),
    ]

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 10:
            break

        t_remain = p2_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.9, per_restart_time * 1.5)
        if time_alloc < 15:
            break

        # Generate perturbation
        strategy_name, strategy_fn = strategies[ri % len(strategies)]
        if strategy_fn is not None:
            a_pert = strategy_fn(best_a, rng)
        else:
            # Combined strategies
            if strategy_name == "spectral+anti":
                a_pert = antipalindromic_perturbation(
                    spectral_perturbation(best_a, rng, 0.1), rng, 0.1)
            else:
                a_pert = block_swap_perturbation(
                    gaussian_bump_perturbation(best_a, rng, 3), rng, 2)

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick refine
        sc, a_opt = refine_candidate(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"restart#{ri}: {strategy_name}")

    # ==================================================================
    # Phase 3: Intensive optimization of best so far (15% of budget)
    # ==================================================================
    p3_budget = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < t_start + effective_budget - 30:
        remaining = p3_budget
        # High-precision L-BFGS
        for beta in [5000, 15000, 50000]:
            if time.monotonic() > t_start + effective_budget - 30:
                break
            t0 = time.monotonic()
            sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=50000)
            try_update(sc, a_opt, f"Intensive LBFGS beta={beta}")
            print(f"    beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

        # Coordinate descent with more sweeps
        remaining = t_start + effective_budget - 30 - time.monotonic()
        if remaining > 10:
            sc, a_opt = coord_descent_golden(best_a, n_sweeps=10, top_k=40,
                                             time_budget=min(remaining, 60.0),
                                             t_start=t_start)
            try_update(sc, a_opt, "Intensive CoordDesc")

    # ==================================================================
    # Phase 4: Try different lengths via resampling (10% of budget)
    # ==================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Different lengths (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    resample_targets = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
        tl = int(n_best * factor)
        if 512 <= tl <= 30000:
            resample_targets.append(tl)

    for tl in resample_targets:
        if time.monotonic() > t_start + effective_budget - 15:
            break
        time_left = t_start + effective_budget - 15 - time.monotonic()
        if time_left < 20:
            break

        # Resample
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)

        time_alloc = min(time_left * 0.8, 40.0)
        sc, a_opt = refine_candidate(a_res, time_alloc, t_start)
        try_update(sc, a_opt, f"n={tl}")

    # ==================================================================
    # Phase 5: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # Final high-beta L-BFGS
    if time.monotonic() < final_deadline - 10:
        sc, a_opt, _ = run_lbfgs_a(best_a, 20000, maxiter=10000, maxfun=100000)
        try_update(sc, a_opt, "Final LBFGS")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = coord_descent_golden(best_a, n_sweeps=8, top_k=50,
                                         time_budget=min(remaining - 2, 30.0),
                                         t_start=t_start)
        try_update(sc, a_opt, "Final CoordDesc")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
