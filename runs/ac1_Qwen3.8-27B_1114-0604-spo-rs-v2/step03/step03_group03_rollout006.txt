Optimized step function search: minimizes 2n * max(self-conv) / (sum)^2 over non-negative sequences.
Uses a combination of:
  1. Fast coordinate descent with warm-started 1D Newton solves
  2. Active-set gradient steps for handling near-tied convolution peaks
  3. LSE-smoothed L-BFGS for global moves

import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    Score = 2n * LSE_max(conv) / (sum(a))^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    LSE_val = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum  # softmax weights
    
    # Gradient of LSE_val w.r.t. a:
    # grad_LSE[i] = 2 * sum_j p[i+j] * a[j] (cross-correlation)
    grad_LSE = 2.0 * np.correlate(p, a, mode='valid')
    
    score = 2.0 * n * LSE_val / (S * S)

    S3 = S * S * S
    grad = (2.0 * n) * (grad_LSE * S - 2.0 * LSE_val) / S3
    
    return score, grad
