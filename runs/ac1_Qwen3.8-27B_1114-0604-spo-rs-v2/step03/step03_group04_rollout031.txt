def fast_conv(a):

    A = np.fft.rfft(a, M_fft)

**Approach: Semidefinite relaxation (SDR)**

We can relax this to: minimize t subject to a^T Q_k a <= t for all k, sum(a) = 1, a >= 0.

The relaxation: introduce X = aa^T (rank-1 PSD matrix). Then a^T Q_k a = tr(Q_k X). The constraints become tr(Q_k X) <= t, tr(diag(1) X) = 1, X >= 0, X >= 0 (element-wise, for a >= 0).

But this is an SDP with a rank constraint, which is non-convex. Dropping the rank constraint gives a valid relaxation (the optimal value of the relaxation is a lower bound).

For n=6962, the SDP would be 6962x6962, which is way too large for standard SDP solvers.

However, we can use the dual: the dual would have 2n-1 + 1 = 13925 variables (one for each convolution coefficient plus one for the sum constraint), and the constraint is that a certain matrix is PSD. But the matrix is still 6962x6962.

So SDR is not feasible for this size. Let's skip it.

**Approach: First-order methods with variance reduction**

Since the objective is f(a) = max_k g_k(a) where g_k(a) = 2*n*a^T Q_k a / S^2, we can use:
- Subgradient method with Nesterov acceleration
- Variance-reduced stochastic subgradient (randomly sample k)

The issue with the subgradient method is that it converges slowly (O(1/sqrt(T)) for non-smooth). But with LSE smoothing (beta-approximation), we get a smooth function with Lipschitz gradient, and we can achieve O(1/T) convergence with Nesterov's method.

"""
Optimized peak-autocorrelation minimizer using multi-phase LSE-LBFGS + subgradient + peak shaving.
Improves on previous approach with higher beta values, adaptive step sizes, and targeted local search.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_val_grad(a, beta):
    """Compute LSE-smoothed log-sum-exp of convolution and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    L = c_max + np.log(sum_exp + 1e-30) / beta
    p = exp_scaled / (sum_exp + 1e-30)
    
    # Gradient of LSE w.r.t. a
    # dLSE/da[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j in [0,n-1])
    # Using FFT:
    # Create u = p reversed (length 2n-1), w = a padded to length 2n-1
    # conv(u, w) at position (2n-2-j) gives sum_l p[l+j]*w[l] = sum_l p[l+j]*a[l]
    M2 = 1 << (4 * n - 3).bit_length()
    u = np.zeros(M2)
    u[:2*n-1] = p[::-1]
    w = np.zeros(M2)
    w[:n] = a
    conv = np.fft.irfft(np.fft.rfft(u) * np.fft.rfft(w), M2)
    grad_LSE = 2.0 * conv[n-1:2*n-1][::-1]
    
    return L, grad_LSE

def score_and_grad(a, beta):
    """Compute smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    L = c_max + np.log(sum_exp + 1e-30) / beta
    p = exp_scaled / (sum_exp + 1e-30)
    
    M2 = 1 << (4 * n - 3).bit_length()
    u = np.zeros(M2)
    u[:2*n-1] = p[::-1]
    w = np.zeros(M2)
    w[:n] = a
    conv = np.fft.irfft(np.fft.rfft(u) * np.fft.rfft(w), M2)
    grad_LSE = 2.0 * conv[n-1:2*n-1][::-1]
    
    score = 2.0 * n * L / (S * S)
    grad = 2.0 * n * (grad_LSE / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

def lbfgs_optimize(a0, beta, maxiter=3000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    res = minimize(
        score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=[(0.0, 1000.0)] * n,
        options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

def subgradient_optimize(a0, n_iters=5000, lr_init=0.01):
    """Exact subgradient method with adaptive step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = lr_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        # Subgradient of max: gradient of c[k_star] w.r.t. a
        # c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = 2*a[k-j] (for valid k-j)
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[k_star - j_end:k_star - j_start + 1][::-1]
        
        # Full gradient of score = 2*n*max(c)/S^2
        # d(score)/da[j] = 2*n*(dc[k*]/da[j]/S^2 - 2*max(c)/S^3)
        grad = 2.0 * n * (sg / (S * S) - 2.0 * M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size (Barzilai-Borwein style)
        # Simple: step = (sc - 1.0) / gn^2 (theoretical lower bound is 2 for non-negative)
        # Actually the lower bound is 2, so:
        step = max(0.001, (sc - 2.0) / (gn * gn + 1e-30))
        step = min(step, 0.1)
        
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Check if this step improved
        sc_new = compute_score(a_new)
        if sc_new > sc * 1.01:
            step *= 0.5
            a_new = np.clip(a - step * grad, 0.0, 1000.0)
        
        a = a_new
    
    return best_sc, best_a

def peak_shave(a, n_iters=200):
    """Targeted peak shaving: reduce the worst convolution coefficient."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the worst k
        k_star = int(np.argmax(c))
        # Variables that affect c[k_star]: a[j] for j in [max(0,k_star-n+1), min(k_star, n-1)]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        active = slice(j_start, j_end + 1)
        active_len = j_end - j_start + 1
        
        if active_len < 2:
            break
        
        # Try to reduce c[k_star] by adjusting the active variables
        # c[k_star] = sum_{i=j_start}^{j_end} a[i]*a[k_star-i]
        # To reduce this, we want to make the product a[i]*a[k_star-i] smaller
        # Strategy: equalize the products (by AM-GM, fixed sum -> min product is at extremes)
        # Actually, to minimize the sum of products with fixed sum, we want to concentrate mass.
        # But we also need to keep other convolution coefficients low.
        
        # Simple heuristic: reduce the variable that contributes most to c[k_star]
        # and distribute to a variable that contributes least
        contrib = np.zeros(active_len)
        for idx in range(active_len):
            j = j_start + idx
            contrib[idx] = a[j] * a[k_star - j]
        
        max_idx = int(np.argmax(contrib))
        min_idx = int(np.argmin(contrib))
        
        if max_idx == min_idx:
            break
        
        j_max = j_start + max_idx
        j_min = j_start + min_idx
        
        # Move a small amount from j_max to j_min
        amt = a[j_max] * 0.01
        a[j_max] -= amt
        a[j_min] += amt
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a
