```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Step function optimization using FFT-accelerated subgradient/Adam methods.
Key insight: the subgradient of max(conv(a,a)) can be computed in O(n log n)
using FFT, enabling hundreds of thousands of optimization iterations.
Strategy: LSE warmup -> Adam subgradient -> LSE polish -> subgradient finish.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def subgradient_fft(a, w):
    """
    Compute the weighted subgradient g = sum_k w[k] * dc_k/da
    using FFT. O(n log n).
    
    g[j] = sum_m a[m] * (w[j+m] + w[j-m]) for valid indices.
    
    X1[j] = conv(a_rev, w)[j + n - 1]
    X2[j] = conv(w, a)[j]
    """
    n = len(a)
    # Length of w is 2n-1
    # conv(a_rev, w) has length n + 2n - 1 - 1 = 3n - 2
    # conv(w, a) has length 2n - 1 + n - 1 = 3n - 2
    L = 3 * n - 2
    L_fft = 1 << (L - 1).bit_length()
    
    a_rev = a[::-1]
    W = np.fft.rfft(w, L_fft)
    A = np.fft.rfft(a, L_fft)
    AR = np.fft.rfft(a_rev, L_fft)
    
    # X1 = conv(a_rev, w)[:3n-2], take [n-1 : 2n-1]
    conv1 = np.fft.irfft(AR * W, L_fft)[:L]
    X1 = conv1[n-1:2*n-1]
    
    # X2 = conv(w, a)[:3n-2], take [0 : n]
    conv2 = np.fft.irfft(W * A, L_fft)[:L]
    X2 = conv2[:n]
    
    return X1 + X2


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (for scipy L-BFGS-B)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient using FFT
    g = subgradient_fft(a, p)
    g *= 2.0  # because dc_k/da = 2 * (the "half-gradient")... 
    
    # Wait, let me recheck. c_k = sum_i a_i * a_{k-i}
    # dc_k/da_j = a_{k-j} + a_{j-k}
    # In my subgradient_fft, I compute sum_k w_k * (a_{k-j} + a_{j-k})
    # which IS dc/da (the full gradient of the weighted sum).
    # So g = sum_k w_k * dc_k/da = sum_k w_k * (a_{k-j} + a_{j-k})
    # This is already the full gradient, no factor of 2 needed.
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def adam_subgrad_optimize(a, iters, beta=5.0, lr=0.005, verbose=True, deadline=None):
    """
    Adam optimizer with FFT-accelerated subgradient of smooth max (LSE with given beta).
    Works on normalized sequence (sum = 1).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S  # normalize to simplex
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for t in range(1, iters + 1):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max  # since sum(a) = 1
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute softmax weights
        scaled = beta * (c - c_max)
        scaled = np.clip(scaled, -700.0, 700.0)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # Compute subgradient via FFT
        g = subgradient_fft(a, w)
        
        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        # Update: we want to minimize, so subtract
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

def mirror_descent_optimize(a, iters, beta=5.0, lr=0.01, verbose=True, deadline=None):

    a = a / S

    best_sc = compute_score(a)
```

```python
"""
Step function optimization using FFT-accelerated Adam/AdamW subgradient methods.
Key innovation: O(n log n) subgradient computation enables millions of iterations,
far exceeding the L-BFGS approach. Uses adaptive beta LSE continuation within Adam,
plus mirror descent and exact subgradient polishing.
Target: beat 1.5027 (previous best: 1.509299).
"""

import numpy as np
from scipy.optimize import minimize
import time

t_start_global = time.monotonic()


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def subgradient_fft(a, w):
    """
    Compute weighted subgradient g = sum_k w[k] * dc_k/da using FFT. O(n log n).
    g[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices.
    Decomposed as X1 + X2 where:
      X1[j] = conv(a_rev, w)[j+n-1]
      X2[j] = conv(w, a)[j]
    """
    n = len(a)
    L = 3 * n - 2
    L_fft = 1 << (L - 1).bit_length()

    a_rev = a[::-1]
    W = np.fft.rfft(w, L_fft)
    A = np.fft.rfft(a, L_fft)
    AR = np.fft.rfft(a_rev, L_fft)

    conv1 = np.fft.irfft(AR * W, L_fft)[:L]
    X1 = conv1[n - 1:2 * n - 1]

    conv2 = np.fft.irfft(W * A, L_fft)[:L]
    X2 = conv2[:n]

    return X1 + X2


def softmax_weights(c, beta):
    """Compute softmax weights over convolution values."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return exp_scaled / total


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for scipy L-BFGS-B."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L_val = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L_val / (S * S)
    g = subgradient_fft(a, p)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L_val / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def adam_optimize(a, iters, beta_start=3.0, beta_end=50.0, lr=0.01,
                  verbose=True, deadline=None):
    """
    Adam optimizer with adaptive-beta LSE subgradient.
    Works on normalized sequence (sum=1) which we maintain internally.
    Returns (best_score, best_normalized_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8

    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    for t in range(1, iters + 1):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Adaptive beta: increase from beta_start to beta_end
        frac = (t - 1) / max(1, iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac

        c = fast_conv(a)
        c_max = float(c.max())
        sc = 2.0 * n * c_max

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        w = softmax_weights(c, beta)
        g = subgradient_fft(a, w)

        # Adam update
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        if verbose and t % 20000 == 0:
            el = time.monotonic() - t_start_global
            print(f"      adam t={t}, beta={beta:.1f}, score={sc:.8f}, "
                  f"best={best_sc:.8f}, elapsed={el:.0f}s")

    return best_sc, best_a

def exact_subgrad_optimize(a, iters, lr_start=0.005, lr_decay=0.9999,

        # Simpler approach:

    # ===================================================================
    phase4_end = t_start_global + 0.90 * (total_budget - 15.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Exact subgradient polish (budget to {phase4_end - t_start_global:.0f}s)")
    print(f"{'='*60}")

    budget4 = phase4_end - time.monotonic()
    if budget4 > 5:
        # Multi-peak subgradient
        iters4a = int(budget4 * 0.5 * 80)
        iters4a = min(iters4a, 2000000)
        sc, a_mp = multi_peak_subgrad(a_cur, iters=iters4a, top_k=5,
                                      lr=0.003, deadline=phase4_end, verbose=True)
        report(sc, a_mp, "multi-peak subgrad")
        if sc < compute_score(a_cur):
            a_cur = a_mp

        # Single peak exact subgradient
        if time.monotonic() < phase4_end:
            iters4b = int((phase4_end - time.monotonic()) * 80)
            iters4b = min(iters4b, 2000000)
            sc, a_sp = exact_subgrad_fft(a_cur, iters=iters4b,
                                         lr_start=0.005, lr_decay=0.99999,
                                         deadline=phase4_end, verbose=True)
            report(sc, a_sp, "exact subgrad fft")
            if sc < compute_score(a_cur):
                a_cur = a_sp

    # ===================================================================
    # Phase 5: Restarts and final polish (remaining time)
    # ===================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Restarts & final polish")
    print(f"{'='*60}")

    n_best = len(a_cur)
    restart_count = 0
    while time.monotonic() < deadline - 10 and restart_count < 15:
        t0 = time.monotonic()
        strategy = restart_count % 5

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_cur * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Targeted: reduce top contributors to peak
            c = fast_conv(a_cur)
            k_star = int(np.argmax(c))
            a_pert = a_cur.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            contrib = np.zeros(n_best)
            contrib[j_start:j_end + 1] = a_cur[j_start:j_end + 1][::-1]
            top_idx = np.argsort(contrib)[::-1][:max(10, n_best // 30)]
            total = a_pert[top_idx].sum() * 0.03
            a_pert[top_idx] -= total / len(top_idx)
            a_pert = np.maximum(a_pert, 0.0)
            bottom_idx = np.argsort(contrib)[:n_best - len(top_idx)]
            a_pert[bottom_idx] += total / max(1, len(bottom_idx))
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Frequency domain perturbation
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.fft(a_cur, L)
            A_mag = np.abs(A) * (1.0 + 0.01 * rng.standard_normal(L))
            A_mag = np.clip(A_mag, 0, None)
            A_new = A_mag * np.exp(1j * np.angle(A))
            a_pert = np.fft.ifft(A_new).real[:n_best]
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() < 1e-12:
                a_pert = a_cur.copy()
            else:
                a_pert = a_pert / a_pert.sum()
        elif strategy == 3:
            # Block swap
            block_size = max(10, n_best // 50)
            num_blocks = n_best // block_size
            i = rng.integers(0, num_blocks)
            j = rng.integers(0, num_blocks)
            a_pert = a_cur.copy()
            block_i = a_pert[i * block_size:(i + 1) * block_size].copy()
            block_j = a_pert[j * block_size:(j + 1) * block_size].copy()
            a_pert[i * block_size:(i + 1) * block_size] = block_j
            a_pert[j * block_size:(j + 1) * block_size] = block_i
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
        else:
            # Sinusoidal modulation
            freq = 1.0 + 3.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_cur * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

        # Quick Adam optimization from perturbed start
        time_left = deadline - 10 - time.monotonic()
        if time_left > 5:
            iters_r = int(min(time_left * 50, 200000))
            iters_r = max(iters_r, 5000)
            sc, a_r = adam_optimize(a_pert, iters=iters_r,
                                    beta_start=10.0, beta_end=50.0,
                                    lr=0.008, deadline=deadline - 10,
                                    verbose=False)
            report(sc, a_r, f"restart #{restart_count} (strat={strategy})")
            if sc < compute_score(a_cur):
                a_cur = a_r

        restart_count += 1
        print(f"  restart {restart_count}: best={best_score:.8f}, "
              f"cur={compute_score(a_cur):.8f}, elapsed={time.monotonic()-t_start_global:.0f}s")

    # Final: one more Adam pass on best
    if time.monotonic() < deadline - 5:
        time_left = deadline - 5 - time.monotonic()
        iters_final = int(min(time_left * 60, 500000))
        iters_final = max(iters_final, 5000)
        sc, a_final = adam_optimize(best_a, iters=iters_final,
                                    beta_start=20.0, beta_end=100.0,
                                    lr=0.005, deadline=deadline - 3,
                                    verbose=False)
        report(sc, a_final, "final Adam")

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start_global
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
