"""
LP-based optimization with active set reduction for minimizing
2n * max(self-conv)/(sum)^2 over non-negative sequences.
Combined with Adam (LSE-smoothed), coordinate descent, and L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L with respect to a

Wait, I need to double-check my FFT formulas. Let me use a different approach that I'm more confident about.

For real signals `a` (length n) and `p` (length 2n-1), with zero-padding to length N:

The linear convolution `(a * p)[m] = sum_i a[i] * p[m-i]` can be computed as:
`ifft(fft(a, N) * fft(p, N))[:n + 2n-1 - 1]`

The cross-correlation `R[j] = sum_i a[i] * p[i+j]` (for j >= 0) is:
This equals `(a * p_rev)[j + (2n-2)]` where `p_rev = p[::-1]`... 

Actually, let me just use a direct approach for the gradient since it's only used in the Adam phase (not the main LP phase).

For the gradient computation, let me use a simpler but correct approach:

```python
def compute_grad_L(a, p, beta):
    """Compute dL/da where L is the LSE-smoothed max of conv."""
    n = len(a)
    # dL/da[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (terms are 0 if index out of [0, n-1])
    # So dL/da[j] = sum_k p[k] * a[k-j] (for 0<=k-j<n) + sum_k p[k] * a[j-k] (for 0<=j-k<n)
    
    # For efficiency, note that p is mostly concentrated at a few indices (where conv is near max)
    # So we can just iterate over the non-negligible p values
    
    grad = np.zeros(n)
    # Find indices where p is significant
    threshold = 1e-10
    significant_p = p > threshold
    idx = np.where(significant_p)[0]
    
    for k in idx:
        k = int(k)
        pk = p[k]
        # First term: a[k-j] for 0<=k-j<n => j in [k-n+1, k] ∩ [0, n-1]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end+1] += pk * a[k - np.arange(j_start, j_end+1)]
        # Second term: a[j-k] for 0<=j-k<n => j in [k, k+n-1] ∩ [0, n-1]
        j_start2 = max(k, 0)
        j_end2 = min(k + n - 1, n - 1)
        if j_end2 >= j_start2:
            grad[j_start2:j_end2+1] += pk * a[np.arange(j_start2, j_end2+1) - k]
    
    return grad
```

This is O(n_active * n) where `n_active` is the number of significant `p` values. With LSE smoothing and moderate beta, `n_active` might be a few hundred, so this is manageable.

Actually wait, with LSE smoothing, `p` is spread over more indices than just the argmax. With `beta = 100`, `p` will be significant over maybe 100-500 indices. So `n_active ≈ 500` and the loop is `500 * 7000 = 3.5M` operations per gradient evaluation. That's fine.

But actually, I realize there's a simpler way to think about this. Since `p` sums to 1 and is concentrated, the gradient is essentially a "weighted average" of the vectors `[a[k-j] for valid j]` weighted by `p[k]`. Each such vector is just a shifted version of `a`. So the gradient is a weighted sum of shifted copies of `a`.

OK let me just implement the direct approach and move on.

```python
"""
LP-based active set optimization for minimizing 2n*max(self-conv)/(sum)^2.
Primary method: LP with active constraint reduction.
Secondary: LSE-smoothed Adam, coordinate descent, L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smoothed(a, beta):
    """LSE-smoothed max of self-convolution."""
    c = fast_conv(a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_max = shifted.max()
    exp_s = np.exp(shifted - shifted_max)
    return c_max + (shifted_max + np.log(exp_s.sum())) / beta


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_max = shifted.max()
    exp_s = np.exp(shifted - shifted_max)
    log_sum = shifted_max + np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    # Compute gradient: dL/da[j] = sum_k p[k] * (a[k-j] + a[j-k])
    grad_L = np.zeros(n)
    # Only iterate over significant p values
    sig_thresh = 1e-8
    significant = p > sig_thresh
    idx = np.where(significant)[0]
    
    for k in idx:
        k = int(k)
        pk = float(p[k])
        # Term 1: a[k-j] for j where 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            grad_L[js:je+1] += pk * a[k - np.arange(js, je + 1)]
        # Term 2: a[j-k] for j where 0 <= j-k < n
        js2 = max(k, 0)
        je2 = min(k + n - 1, n - 1)
        if je2 >= js2:
            grad_L[js2:je2+1] += pk * a[np.arange(js2, je2 + 1) - k]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def build_lp_matrix(a, active_indices, n):
    """Build sparse matrix for LP constraints (only active rows)."""
    rows = []
    cols = []
    data = []
    for k in active_indices:
        k = int(k)
        # (a * g)[k] = sum_{j} a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        # Coefficient of g[l] (where l = k-j) is a[j] = a[k-l]
        # l ranges from max(0, k-n+1) to min(k, n-1)
        l_start = max(0, k - n + 1)
        l_end = min(k, n - 1)
        if l_end >= l_start:
            l_range = np.arange(l_start, l_end + 1)
            j_range = k - l_range
            rows.extend([k] * len(l_range))
            cols.extend(l_range.tolist())
            data.extend(a[j_range].tolist())
    mat = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return mat.tocsr()


def lp_single_step(a, M, n, threshold=0.98):
    """Solve LP: max sum(g) s.t. a*g <= M (elementwise), g >= 0.
    Returns g or None if infeasible/failed."""
    conv = fast_conv(a)
    active = np.where(conv >= threshold * M)[0]
    
    A = build_lp_matrix(a, active, n)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, method='highs',
                  bounds=[(0, None)] * n)
    
    if res.status != 0:
        return None
    return res.x


def lp_optimize(a_init, n_iters=500, threshold=0.98, t_blend=0.15,
                time_limit_s=None, t_start=None):
    """LP-based optimization loop."""
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / len(a)
        S = 1.0
    a = a / S
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break
        
        conv = fast_conv(a)
        M = float(conv.max())
        
        g = lp_single_step(a, M, n, threshold=threshold)
        if g is None:
            break
        
        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-8:
            break
        
        g_resc = g / g_sum
        t = min(t_blend, max(0.01, 0.5 * (g_sum - 1.0)))
        a_new = (1.0 - t) * a + t * g_resc
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            if it % 20 == 0:
                print(f"    LP iter {it}: score={sc:.8f}, sum_g={g_sum:.6f}")
        
        a = a_new
    
    return best_score, best_a


def adam_lse(a_init, lr=0.01, n_steps=50000, beta=200,
             time_limit_s=None, t_start=None, normalize=True):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        
        if normalize:
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def coordinate_descent(a_init, top_k=30, n_sweeps=20,
                       time_limit_s=None, t_start=None):
    """Selective coordinate descent."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        c_max = float(conv.max())
        S = float(a.sum())
        score = 2.0 * n * c_max / (S * S)
        if score < best_score:
            best_score = score
        
        # Find peak location
        k_star = int(conv.argmax())
        
        # Select coordinates with highest contribution to peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        # Compute contribution of each coordinate to the peak
        contrib = np.zeros(n)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            contrib[js:je+1] = a[idx]
        
        # Also consider coordinates near other high-conv points
        top_lags = np.argsort(conv)[::-1][:5]
        for k in top_lags:
            k = int(k)
            js2 = max(0, k - n + 1)
            je2 = min(k, n - 1)
            if je2 >= js2:
                idx2 = k - np.arange(js2, je2 + 1)
                contrib[js2:je2+1] += a[idx2] * (conv[k] / c_max)
        
        # Select top-k coordinates
        order = np.argsort(contrib)[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            
            # Try reducing this coordinate
            candidates = a_old * np.array([0.0, 0.2, 0.4, 0.6, 0.75, 0.9])
            best_x = a_old
            best_sc = score
            
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def lbfgs_polish(a_init, beta=1000, maxiter=5000, maxfun=50000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact = compute_score(a_out)
    return exact, a_out
