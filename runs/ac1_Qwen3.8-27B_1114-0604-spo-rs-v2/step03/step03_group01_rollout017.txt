    """Build the sparse constraint matrix A for the LP.
    A[k, i] = f[k-i] if 0 <= k-i < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    # Build efficiently
    cols_all = []
    vals_all = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        count = i_end - i_start + 1

    cols = np.concatenate([x[1] for x in cols_all])

    """One iteration of the LP-based multiplicative update."""
    g, t_lp = solve_lp(f, M)
    if g is None:

    if g_sum > 1e-12:

    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        count = i_end - i_start + 1
        if count > 0:
            rows_list.append(np.full(count, k, dtype=np.int32))
            cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
            vals_list.append(f[k - np.arange(i_start, i_end + 1)])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_prebuilt=None):
    n = len(f)
    if A_prebuilt is None:
        A_prebuilt = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    res = linprog(c, A_ub=A_prebuilt, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    elapsed = time.monotonic() - t0
    if res.status != 0:
        return None, elapsed
    return res.x, elapsed


def lp_phase(f, target_sum, time_budget, t_start):
    """Run LP-based multiplicative updates until time budget exhausted."""
    n = len(f)
    f = f.copy()
    f = f * target_sum / f.sum()
    M = float(fast_conv(f).max())
    score = compute_score(f)
    best_f = f.copy()
    best_M = M
    best_score = score
    step = 0.5
    deadline = t_start + time_budget
    A_prebuilt = build_lp_matrix(f)

    print(f"  LP phase start: M={M:.10f}, score={score:.10f}")

    for it in range(200):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 30:
            break

        g, t_lp = solve_lp_direction(f, M, A_prebuilt)
        if g is None:
            print(f"    LP iter {it}: failed, stopping")
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g * target_sum / g_sum

        # Line search on step size
        for _ in range(6):
            new_f = (1.0 - step) * f + step * g
            new_f = np.clip(new_f, 0.0, 1e10)
            new_sum = new_f.sum()
            if new_sum < 1e-12:
                break
            new_f = new_f * target_sum / new_sum
            new_M = float(fast_conv(new_f).max())
            if new_M <= M * 1.0001:
                break
            step *= 0.5
        else:
            step = max(0.01, step * 0.5)
            new_f = (1.0 - step) * f + step * g
            new_f = np.clip(new_f, 0.0, 1e10)
            new_sum = new_f.sum()
            if new_sum > 1e-12:
                new_f = new_f * target_sum / new_sum
            new_M = float(fast_conv(new_f).max())

        f = new_f
        M = new_M

        # Increase step if M decreased a lot
        if M < best_M * 0.995:
            step = min(1.0, step * 1.2)
            best_M = M
            best_f = f.copy()
            score = compute_score(f)
            best_score = score
            print(f"    LP iter {it}: M={M:.10f} (score={score:.10f}) [step={step:.3f}, lp_time={t_lp:.1f}s]")

        # Rebuild LP matrix if f changed significantly
        if it % 5 == 4:
            A_prebuilt = build_lp_matrix(f)

    return best_score, best_f


def adam_lse(a_init, lr=0.01, n_steps=50000, beta=100,
             time_limit_s=None, t_start=None):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad_a(a, beta)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        if t % 10000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def run_lbfgs(a0, beta=500, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def coord_descent(a, top_k=30, time_limit_s=None, t_start=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(50):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        k_star = int(c.argmax())
        # Gradient from peak lag
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if grad[j] > 1e-10:
                ratios = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05])
            elif grad[j] < -1e-10:
                ratios = np.array([0.5, 0.8, 1.0, 1.2, 1.5, 2.0])
            else:
                continue
            candidates = a_old * ratios
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.10f}")

    best_score = parent_score
    best_a = parent.copy()
    n = len(best_a)

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Normalize for LP phase
    target_sum = np.sqrt(2.0 * n)
    f_norm = best_a * target_sum / best_a.sum()

    # ============================================================
    # Phase 1: LP-based multiplicative updates (40% time)
    # ============================================================
    p1 = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 1: LP-based updates (budget={p1:.0f}s, n={n})")
    print(f"{'='*60}")

    t1_start = time.monotonic()
    sc_lp, a_lp = lp_phase(f_norm, target_sum, p1, t1_start)
    # Convert back to score scale
    a_lp_unscaled = a_lp.copy()
    sc_lp_full = compute_score(a_lp_unscaled)
    try_update(sc_lp_full, a_lp_unscaled, "LP phase")

    # ============================================================
    # Phase 2: LSE-Adam continuation (25% time)
    # ============================================================
    p2 = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE-Adam (budget={p2:.0f}s)")
    print(f"{'='*60}")

    a_work = best_a.copy()
    for beta, lr in [(50, 0.02), (150, 0.01), (400, 0.005), (1000, 0.002)]:
        remaining = p2 - (time.monotonic() - t_start)
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.35, 120.0)
        sc, a_opt = adam_lse(a_work, lr=lr, n_steps=200000, beta=beta,
                             time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_work = a_opt

    # ============================================================
    # Phase 3: L-BFGS polish (20% time)
    # ============================================================
    p3 = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS Polish (budget={p3:.0f}s)")
    print(f"{'='*60}")

    for beta in [300, 800, 2000, 5000]:
        remaining = p3 - (time.monotonic() - t_start)
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # Phase 4: Coordinate descent (5% time)
    # ============================================================
    p4 = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 4: Coord Descent (budget={p4:.0f}s)")
    print(f"{'='*60}")

    t0 = time.monotonic()
    sc, a_opt = coord_descent(best_a, top_k=40, time_limit_s=p4, t_start=t_start)
    try_update(sc, a_opt, "Coord descent")
    print(f"    Coord descent: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # Phase 5: Restart + final polish (10% time)
    # ============================================================
    p5 = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Restarts + Final (budget={p5:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    restart_deadline = t_start + effective_budget - 10.0

    for r in range(8):
        if time.monotonic() > restart_deadline:
            break
        strategy = r % 4
        if strategy == 0:
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= 0.7
            removed = float(best_a[js:je + 1].sum() - a_pert[js:je + 1].sum())
            mask = np.ones(n_best, dtype=bool)
            mask[js:je + 1] = False
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + 0.02 * np.cos(2 * np.pi * freq * x + phase))
        else:
            center = int(rng.integers(n_best // 4, 3 * n_best // 4))
            width = max(20, n_best // 40)
            a_pert = best_a.copy()
            a_pert[center - width:center + width] *= rng.uniform(0.4, 1.2)

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        time_left = restart_deadline - time.monotonic()
        if time_left < 5:
            break
        adam_time = min(15.0, time_left * 0.4)
        sc, a_r = adam_lse(a_pert, lr=0.005, n_steps=30000, beta=300,
                           time_limit_s=adam_time, t_start=t_start)
        try_update(sc, a_r, f"restart#{r}s{strategy}")

    # Final L-BFGS
    if time.monotonic() < restart_deadline:
        sc, a_opt = run_lbfgs(best_a, 3000, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, "final LBFGS")

    # Final coord descent
    if time.monotonic() < restart_deadline:
        sc, a_opt = coord_descent(best_a, top_k=50,
                                  time_limit_s=min(10.0, restart_deadline - time.monotonic()),
                                  t_start=t_start)
        try_update(sc, a_opt, "final coord")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
