Step function optimization via SLP (Sequential Linear Programming) + LSE-LBFGS.

Key ideas:
  1. SLP with constraint generation: iteratively solve an LP that linearizes
     the peak-convolution constraint. Only include "active" lags (near the peak).
  2. Damped updates to ensure stability.
  3. LSE-smoothed L-BFGS for local refinement.
  4. Multi-phase approach with time management.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def get_active_lags(c, M, ratio=0.1):
    """Get indices of lags where c[k] >= M * (1-ratio)."""
    threshold = M * (1.0 - ratio)
    return np.where(c >= threshold)[0]


def build_slp_lp(a, active_indices, c_0):
    """
    Build the LP for one SLP iteration.
    
    LP: minimize t
        subject to: 2 * sum_j a_0[k-j] * a[j] - t <= c_0[k]  for k in active
                    sum(a) = 1
                    a >= 0
                    t >= 0

    """

    # Objective: minimize t => c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # For each active lag k: [2*a_0[k], 2*a_0[k-1], ..., 2*a_0[k-n+1], -1] @ [a, t] <= c_0[k]
    # (with a_0[m] = 0 for m < 0 or m >= n)

Actually, I realize I should think more carefully about the FISTA approach. The LSE objective is:

`L_beta(a) = c_max + (1/beta) * log(exp(beta * (c[a,i] - c_max)))`

where the sum is over i. This is a smooth approximation to `max_i c[a,i]`.

The gradient is: `grad_L[j] = sum_i p[i] * d(c[a,i])/d(a[j])` where `p[i] = exp(beta*(c[a,i]-c_max)) / sum_l exp(beta*(c[a,l]-c_max))`.

And `d(c[a,i])/d(a[j]) = 2*a[i-j]` (with zeros for out-of-range).

So `grad_L[j] = 2 * sum_i p[i] * a[i-j]` (with zeros for out-of-range).

This is a cross-correlation of `p` and `a`, which can be computed via FFT in O(n log n).

The full objective (including the normalization) is:
`F(a) = 2n * L_beta(a) / (sum(a))^2`

The gradient of F is:
`grad_F[j] = 2n * (grad_L[j] / (sum(a))^2 - 2 * L_beta(a) * grad_sum[j] / (sum(a))^3)`

where `grad_sum[j] = 1` for all j.

  2. SLP with constraint generation: solve small LPs on active lags for large coordinated steps.

  4. Final L-BFGS polish with very high beta for local refinement.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / ar
    idx = np.where(cond > 0)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    k = idx[-1] + 1
    theta = (cssv[k - 1] - 1.0) / k
    return np.maximum(v - theta, 0.0)


def lse_value_grad(a, beta):
    """
    Compute LSE-smoothed max-convolution value and gradient.
    a is on the simplex (sum=1, a>=0).
    Returns (L, grad) where grad has same shape as a.

    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical safety
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()

    P = np.fft.rfft(p, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_pa = np.fft.irfft(P * AR, M_fft)[:M_conv]
    grad = 2.0 * conv_pa[n - 1:2 * n - 1]

    FISTA on LSE-smoothed objective, projected onto simplex.
    a0 should be on the simplex.
    Returns best a found.

    x = a0.copy()
    y = a0.copy()
    t = 1.0
    best_x = x.copy()
    best_val = lse_value_grad(x, beta)[0]
    step = step_scale / (2.0 * beta + 1e-10)

    for k in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:

        # Gradient step
        x_new = y - step * grad
        # Project to simplex

    active_set = set(top_lags.tolist())

    for gen_iter in range(8):
        active_list = sorted(active_set)
        K = len(active_list)
        if K > max_constraints:
            # Keep only top max_constraints
            active_list = sorted(top_lags[:max_constraints].tolist())
            K = max_constraints

        # Build LP: min t s.t. 2*sum_j a_0[k-j]*a_new[j] - t <= c[k] for active k
        #               sum(a_new) = 1, a_new >= 0, t >= 0
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)

        for i, k in enumerate(active_list):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                A_ub[i, j_start:j_end + 1] = 2.0 * a[k - js]
            A_ub[i, -1] = -1.0
            b_ub[i] = c[k]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0.0, None)] * (n + 1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return a

        if not res.success:
            return a

        a_new = np.clip(res.x[:n], 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            return a
        a_new = a_new / S_new

        # Check for violated constraints
        c_new = fast_conv(a_new)
        M_new = float(c_new.max())
        target = float(res.x[-1]) + 1e-10
        violated = np.where(c_new > target)[0]
        violated = violated[~np.isin(violated, sorted(active_set))]

        if len(violated) == 0 or M_new >= M - 1e-12:
            return a_new

        # Add top violated constraints
        violated_top = violated[np.argsort(c_new[violated])[::-1][:200]]
        for v in violated_top:
            active_set.add(int(v))

    return a_new


def slp_damped(a, alpha=0.5):
    """Damped SLP step."""
    a_slp = slp_step(a)
    a_new = (1.0 - alpha) * a + alpha * a_slp
    a_new = np.clip(a_new, 0.0, None)
    S = a_new.sum()
    if S < 1e-12:
        return a
    return a_new / S


def lse_full_score_grad(a, beta):
    """
    LSE-smoothed full objective (with normalization) and gradient for L-BFGS.
    F(a) = 2n * L_beta(conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()

    # grad_L[j] = 2 * sum_i p[i] * a[i-j]
    a_rev = a[::-1]
    M_conv = 2 * n - 1
    M_fft = 1 << (M_conv - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AR = np.fft.rfft(a_rev, M_fft)
    conv_pa = np.fft.irfft(P * AR, M_fft)[:M_conv]
    grad_L = 2.0 * conv_pa[n - 1:2 * n - 1]

    score = 2.0 * n * L / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_func(x):
        sc, g = lse_full_score_grad(x, beta)
        return sc, g

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_full_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 40)))

    try:
        res = minimize(obj_func, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n, all on the simplex."""
    inits = []
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        p = np.clip(p, 0.01, 1000.0)
        S = p.sum()
        if S > 0:
            inits.append(p / S)

    x = np.arange(n, dtype=np.float64) / n
    # Uniform
    inits.append(np.ones(n) / n)
    # Cosine modulations
    for freq, amp in [(1, 0.3), (2, 0.4), (3, 0.2), (5, 0.15), (8, 0.1)]:
        v = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
        v = np.clip(v, 0.01, None)
        inits.append(v / v.sum())
    # Quadratic
    v = 0.1 + 0.9 * (2.0 * x - 1.0) ** 2
    inits.append(v / v.sum())
    # Random
    v = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None)
    inits.append(v / v.sum())
    v = np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None)
    inits.append(v / v.sum())
    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        a = a[:n]
        inits.append(a / a.sum())
    # Triangular
    v = np.abs(1.0 - 2.0 * x) + 0.1
    inits.append(v / v.sum())
    # Bimodal
    v = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2) + 0.01
    inits.append(v / v.sum())
    return inits


def exact_subgrad_polish(a, total_iters=500, deadline=None):
    """Projected subgradient descent on the exact max-conv objective (on simplex)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M  # (sum=1, so score = 2n*M)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max at k_star
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] = 2.0 * a[k_star - js]

        # Step size (Polyak-like)
        step = 1.0 / (np.linalg.norm(sg) * (it + 1) ** 0.5 + 1e-10)
        step = min(step, 0.1)
        a = a - step * sg
        a = project_simplex(a)

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024, dtype=np.float64) / 1024.0

    def _check(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = [4096, 8192, 16384]
    if parent is not None:
        pa = parent.size
        if pa not in lengths:
            lengths.insert(0, pa)
    lengths = sorted(set(lengths))[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}" + (f" (len={parent.size})" if parent is not None else ""))

    # ============================================================
    # Phase 1: FISTA + LSE global search (50% time)
    # ============================================================
    phase1_deadline = t_start + 0.50 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: FISTA + LSE Global Search")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- Phase 1: n = {n} ---")

        inits = build_inits(n, rng, parent)
        n_init_budget = max(2, int((phase1_deadline - time.monotonic()) / (total_budget * 0.08)))
        inits = inits[:min(n_init_budget, len(inits))]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            a_cur = a0.copy()

            # Beta continuation with FISTA
            for beta in [10, 50, 200, 500]:
                if time.monotonic() > phase1_deadline:
                    break
                remaining = phase1_deadline - time.monotonic()
                if remaining < 10.0:
                    break
                n_iters = int(min(3000, remaining * 100))
                a_cur = fista_lse(a_cur, beta, n_iters, phase1_deadline)

                # Check score
                c = fast_conv(a_cur)
                sc = 2.0 * n * float(c.max())  # sum=1
                _check(sc, a_cur, f"FISTA n={n} init={i} beta={beta}")

                # Periodic SLP boost
                if time.monotonic() < phase1_deadline - 15:
                    a_slp = slp_damped(a_cur, alpha=0.5)
                    c_s = fast_conv(a_slp)
                    sc_s = 2.0 * n * float(c_s.max())
                    _check(sc_s, a_slp, f"SLP n={n} init={i} beta={beta}")
                    if sc_s < sc:
                        a_cur = a_slp

            # Final FISTA at high beta
            if time.monotonic() < phase1_deadline - 5:
                a_cur = fista_lse(a_cur, 1000, 2000, phase1_deadline)
                c = fast_conv(a_cur)
                sc = 2.0 * n * float(c.max())
                _check(sc, a_cur, f"FISTA final n={n} init={i}")

            elapsed = time.monotonic() - t0
            print(f"  init {i}: {elapsed:.1f}s, best so far: {best_score:.6f}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ============================================================
    # Phase 2: SLP refinement (20% time)
    # ============================================================
    phase2_deadline = t_start + 0.70 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: SLP Refinement")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    S = a_cur.sum()
    if S > 0:
        a_cur = a_cur / S

    for slp_round in range(30):
        if time.monotonic() > phase2_deadline:
            break
        a_slp = slp_step(a_cur, n_active=1000)
        S_s = a_slp.sum()
        if S_s > 0:
            a_slp = a_slp / S_s
        c_s = fast_conv(a_slp)
        sc_s = 2.0 * len(a_slp) * float(c_s.max())
        improved = _check(sc_s, a_slp, f"SLP round={slp_round}")

        if improved:
            a_cur = a_slp
        else:
            # Try damped
            a_damp = slp_damped(a_cur, alpha=0.3)
            c_d = fast_conv(a_damp)
            sc_d = 2.0 * len(a_damp) * float(c_d.max())
            if _check(sc_d, a_damp, f"SLP damp={slp_round}"):
                a_cur = a_damp
            else:
                # Try with more active constraints
                a_slp2 = slp_step(a_cur, n_active=3000)
                S_s2 = a_slp2.sum()
                if S_s2 > 0:
                    a_slp2 = a_slp2 / S_s2
                c_s2 = fast_conv(a_slp2)
                sc_s2 = 2.0 * len(a_slp2) * float(c_s2.max())
                if _check(sc_s2, a_slp2, f"SLP big={slp_round}"):
                    a_cur = a_slp2
                else:
                    # Stall - perturb
                    pert = 0.02 * rng.standard_normal(len(a_cur))
                    a_pert = a_cur + pert
                    a_pert = project_simplex(a_pert)
                    a_cur = a_pert

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS refinement (20% time)
    # ============================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS Refinement")
    print(f"{'='*60}")

    for beta in [500, 1000, 2000, 5000]:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000, deadline=phase3_deadline)
        _check(sc, a_opt, f"L-BFGS beta={beta}")

    # Try from SLP result too
    if time.monotonic() < phase3_deadline - 10:
        for beta in [1000, 3000]:
            if time.monotonic() > phase3_deadline:
                break
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=60000, deadline=phase3_deadline)
            _check(sc, a_opt, f"L-BFGS from SLP beta={beta}")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Local search with perturbations (10% time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n{'='*60}")
    print(f"Phase 4: Local Search")
    print(f"{'='*60}")

    n_best = len(best_a)
    a_ref = best_a.copy()

    for trial in range(15):
        if time.monotonic() > phase4_deadline:
            break
        strategy = trial % 4
        if strategy == 0:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = a_ref * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, None)
            S_p = a_pert.sum()
            if S_p > 0:
                a_pert = a_pert / S_p
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(a_ref)
            k_star = int(np.argmax(c))
            a_pert = a_ref.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            # Reduce top contributors
            contrib_range = a_ref[j_start:j_end + 1]
            top_n = max(3, len(contrib_range) // 20)
            if top_n > 0:
                idx_in_range = np.argsort(contrib_range)[::-1][:top_n]
                reduce_total = 0.0
                for idx in idx_in_range:
                    j = j_start + idx
                    amt = a_pert[j] * 0.1
                    a_pert[j] -= amt
                    a_pert[j] = max(a_pert[j], 0.0)
                    reduce_total += amt
                # Redistribute to bottom
                bottom_mask = np.ones(n_best, dtype=bool)
                bottom_mask[j_start:j_end + 1] = False
                bottom_idx = np.where(bottom_mask)[0]
                if len(bottom_idx) > 0 and reduce_total > 0:
                    a_pert[bottom_idx] += reduce_total / len(bottom_idx)
            S_p = a_pert.sum()
            if S_p > 0:
                a_pert = a_pert / S_p
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 0.5 + 10.0 * rng.random()
            phase_val = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_ref * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_val))
            a_pert = np.clip(a_pert, 0.0, None)
            S_p = a_pert.sum()
            if S_p > 0:
                a_pert = a_pert / S_p
        else:
            # FISTA micro-step
            a_pert = fista_lse(a_ref, 2000, 200, phase4_deadline)

        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=2000, maxfun=30000, deadline=phase4_deadline)
        _check(sc, a_opt, f"Local #{trial} strat={strategy}")
        if sc < best_score:
            a_ref = a_opt

    # ============================================================
    # Final polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Final Polish")
    print(f"{'='*60}")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000, deadline=deadline - 10)
        _check(sc, a_opt, "Final L-BFGS")

    # Subgradient polish
    if time.monotonic() < deadline - 5:
        a_sub = exact_subgrad_polish(best_a, total_iters=200, deadline=deadline - 3)
        sc = compute_score(a_sub)
        _check(sc, a_sub, "Subgrad polish")

    # One more high-beta L-BFGS
    if time.monotonic() < deadline - 3:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=2000, maxfun=30000, deadline=deadline - 1)
        _check(sc, a_opt, "Very final L-BFGS")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
