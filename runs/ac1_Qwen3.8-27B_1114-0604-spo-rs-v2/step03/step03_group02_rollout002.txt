import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: 2 * cross-correlation of a_rev and p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 40)))

    try:
        res = minimize(
            obj_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def deflation_perturb(a, rng, strength=0.05):
    """
    Target the peak convolution lag and reduce contributing elements,
    while adding to elements that contribute to valleys.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    c_max = float(c.max())
    c_min = float(c.min())
    c_range = c_max - c_min
    if c_range < 1e-15:
        return a.copy()

    # Contribution of each element j to the peak lag
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

    # Contribution to the minimum lag
    k_min = int(np.argmin(c))
    contrib_min = np.zeros(n)
    j_start2 = max(0, k_min - n + 1)
    j_end2 = min(k_min, n - 1)
    if j_end2 >= j_start2:
        contrib_min[j_start2:j_end2 + 1] = a[j_start2:j_end2 + 1][::-1]

    a_pert = a.copy()
    # Reduce top contributors to peak
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    total_reduce = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - amt, 0.0)
        total_reduce += amt

    # Add to elements that help the valley
    bottom_count = max(5, n // 30)
    bottom_idx = np.argsort(contrib_min)[:bottom_count]
    if len(bottom_idx) > 0 and total_reduce > 0:
        per = total_reduce / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per

    return np.clip(a_pert, 0.0, 1000.0)

            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 'deflation':
            a_pert = deflation_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strategy == 'sinusoidal':
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.05 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 'block':
            block_size = max(16, n // 80)
            n_blocks = n // block_size
            if n_blocks > 2:
                perm = rng.permutation(n_blocks)
                for b in range(n_blocks):
                    src = perm[b] * block_size
                    dst = b * block_size
                    a_pert[dst:dst + block_size] = best_a[src:src + block_size]
                a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:  # crossover
            if len(candidates) > 1:
                other = candidates[rng.integers(1, len(candidates))]
                t_mix = 0.3 + 0.4 * rng.random()
                a_pert = t_mix * best_a + (1.0 - t_mix) * other
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            else:
                a_pert = deflation_perturb(best_a, rng, strength=0.05)

        sc, a_opt = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000, deadline=deadline)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    Population trial {trials} ({strategy}): improved to {sc:.8f}")

        # Add to candidate pool
        if compute_score(a_opt) < np.inf and len(candidates) < 8:
            candidates.append(a_opt.copy())
        trials += 1

    return best_sc, best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.sin(3.0 * np.pi * x) ** 2)
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    # Gapped
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    a[1::2] = 0.1
    inits.append(a.copy())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0  # 20s buffer

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        pa = parent.size
        if pa <= 16384:
            lengths.append(pa)
    for n in [8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {parent is not None}" + (f" (len={parent.size})" if parent is not None else ""))

    # ==================================================================
    # Phase 1: Annealed multiplicative equalization from parent + inits
    # ==================================================================
    phase1_deadline = t_start + 0.35 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Annealed Multiplicative Equalization")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = build_diverse_inits(n, rng, parent)
        n_tries = min(len(inits), max(4, int((phase1_deadline - time.monotonic()) / (total_budget * 0.03))))
        inits = inits[:n_tries]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            sc, a_eq = annealed_multiplicative(a0, n_iters=300, deadline=phase1_deadline)
            improved = _report(sc, a_eq, f"n={n}, init={i}, annealed")
            elapsed = time.monotonic() - t0
            print(f"    init={i}: sc={sc:.8f}, elapsed={elapsed:.1f}s")
            if improved:
                # Follow up with L-BFGS polish
                if time.monotonic() < phase1_deadline - 10:
                    sc2, a2 = run_lbfgs(a_eq, 100, maxiter=5000, maxfun=80000,
                                         deadline=phase1_deadline)
                    _report(sc2, a2, f"n={n}, init={i}, lbfgs after anneal")

    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ==================================================================
    # Phase 2: Iterative refinement loop (multiplicative + L-BFGS)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Iterative Refinement")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_best = len(a_cur)

    for round_i in range(20):
        if time.monotonic() > phase2_deadline:
            break

        # Multiplicative equalization with moderate iterations
        sc, a_eq = annealed_multiplicative(a_cur, n_iters=100, deadline=phase2_deadline)
        _report(sc, a_eq, f"refine r={round_i}, mult")
        a_cur = a_eq

        # L-BFGS with increasing beta
        beta = 100 + 50 * round_i
        remaining = phase2_deadline - time.monotonic()
        if remaining > 15:
            sc2, a2 = run_lbfgs(a_cur, beta, maxiter=4000, maxfun=60000,
                                 deadline=phase2_deadline)
            _report(sc2, a2, f"refine r={round_i}, lbfgs b={beta}")
            if sc2 < best_score:
                a_cur = a2

        if round_i % 5 == 4 and time.monotonic() < phase2_deadline - 30:
            # Periodic deflation perturbation
            a_pert = deflation_perturb(a_cur, rng, strength=0.03)
            sc3, a3 = run_lbfgs(a_pert, 200, maxiter=3000, maxfun=40000,
                                 deadline=phase2_deadline)
            _report(sc3, a3, f"refine r={round_i}, deflation")
            if sc3 < best_score:
                a_cur = a3

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Population-based local search
    # ==================================================================
    phase3_deadline = t_start + 0.80 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Population-Based Search")
    print(f"{'='*60}")

    # Build initial population
    candidates = [best_a.copy()]
    n_pop = len(best_a)
    # Add some diverse candidates
    a_def = deflation_perturb(best_a, rng, strength=0.05)
    candidates.append(a_def)
    a_gauss = best_a * (1.0 + 0.03 * rng.standard_normal(n_pop))
    candidates.append(np.clip(a_gauss, 0.0, 1000.0))
    a_sin = best_a * (1.0 + 0.04 * np.cos(2.0 * np.pi * 3.0 * np.arange(n_pop) / n_pop))
    candidates.append(np.clip(a_sin, 0.0, 1000.0))

    sc, a_best = population_search(candidates, n_pop, rng, phase3_deadline)
    _report(sc, a_best, "population search")

    # Second round of population with updated best
    candidates = [a_best.copy()]
    for _ in range(3):
        a_p = a_best * (1.0 + 0.02 * rng.standard_normal(n_pop))
        candidates.append(np.clip(a_p, 0.0, 1000.0))
    candidates.append(deflation_perturb(a_best, rng, strength=0.04))

    sc, a_best2 = population_search(candidates, n_pop, rng, phase3_deadline)
    _report(sc, a_best2, "population search round 2")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final high-precision polishing
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    a_final = best_a.copy()

    # High-beta L-BFGS
    for beta in [300, 500, 1000, 2000]:
        if time.monotonic() > deadline - 60:
            break
        sc, a_opt = run_lbfgs(a_final, beta, maxiter=8000, maxfun=150000,
                               deadline=deadline - 30)
        _report(sc, a_opt, f"final lbfgs beta={beta}")
        if sc < best_score:
            a_final = a_opt

    # Nesterov subgradient polish
    if time.monotonic() < deadline - 30:
        sc, a_ns = nesterov_subgrad(a_final, total_iters=500, deadline=deadline - 15)
        _report(sc, a_ns, "nesterov subgrad")
        if sc < best_score:
            a_final = a_ns

    # One final multiplicative pass
    if time.monotonic() < deadline - 20:
        sc, a_mp = annealed_multiplicative(a_final, n_iters=100, deadline=deadline - 10)
        _report(sc, a_mp, "final multiplicative")
        if sc < best_score:
            a_final = a_mp

    # Final L-BFGS
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(a_final, 500, maxiter=5000, maxfun=80000,
                               deadline=deadline - 5)
        _report(sc, a_opt, "absolute final lbfgs")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
