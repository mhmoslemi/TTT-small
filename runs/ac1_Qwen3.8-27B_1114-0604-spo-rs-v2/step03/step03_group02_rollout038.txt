import numpy as np

def _fft_len(m): return 1 << (m-1).bit_length()

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2*n - 1
    M = _fft_len(m)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A*A, M)[:m]

def cross_weight(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = 3*n - 2
    M = _fft_len(m)
    W = np.fft.rfft(w, M)
    R = np.fft.rfft(a[::-1], M)
    out = np.fft.irfft(R * W, M)
    return 2.0 * out[n-1:2*n-1]

    S = float(a.sum())
    n = a.size
    if S <= 1e-300: return np.inf, a.copy()
    an = a / S
    c = fast_conv(an)
    sc = 2.0 * n * float(c.max())
    return sc, an

def lse_norm_fgrad(x, beta):
    x = np.asarray(x, dtype=np.float64)
    # Do not clip? For safety: x = np.where(x > 0, x, 0.0)
    S = float(x.sum())
    n = x.size
    if S <= 1e-300 or not np.isfinite(S):
       return (1e10, np.zeros(n))
    a = x / S

def multiplicative_phase(a0, beta, n_iters, alpha, deadline, verbose=False, min_improvement=1e-15):
    sc, a, c = score_and_norm(a0)
    best_sc = sc; best_a = a.copy()
    cur_sc = sc
    alpha_cur = max(1e-6, float(alpha))
    K = c.size
    n = a.size
    stall = 0
    it = 0
    while it < int(n_iters):
        it += 1
        if deadline is not None and time.monotonic() > deadline:
            break
        M = float(np.max(c))
        if beta > 1e-12:
            z = beta * (c - M)
            z = np.minimum(z, 0.0)
            z = np.maximum(z, -745.0)
            ez = np.exp(z)
            s = float(np.sum(ez))
            if s <= 1e-300:
                w = np.zeros(K); w[int(np.argmax(c))] = 1.0
            else:
                w = ez / s
        else:
            tol = max(1e-10 * M, 1e-14)
            active = (c >= M - tol).astype(np.float64)
            s = float(active.sum())
            if s <= 0:
                w = np.zeros(K); w[int(np.argmax(c))] = 1.0
            else:
                w = active / s
        h = cross_weight(a, w)
        mu = float(np.dot(a, h))
        if not np.isfinite(mu) or mu <= 1e-300:
            break
        h_safe = np.maximum(h, mu * 1e-8)
        ratio = mu / h_safe
        ratio = np.clip(ratio, 1e-6, 1e6)
        factor = np.power(ratio, alpha_cur)
        factor = np.clip(factor, 0.02, 50.0)
        a_new = a * factor
        a_new = np.clip(a_new, 0.0, 1000.0)
        Sn = float(np.sum(a_new))
        if not np.isfinite(Sn) or Sn <= 1e-300:
            break
        a_new /= Sn
        c_new = fast_conv(a_new)
        sc_new = 2.0 * n * float(np.max(c_new))
        if not np.isfinite(sc_new):
            break
        if sc_new <= cur_sc + 1e-12 * max(1.0, cur_sc):
            if sc_new < cur_sc:
                stall = 0
            else:
                stall += 1
            a = a_new; c = c_new; cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new; best_a = a_new.copy()
                if verbose:
                    print(f"[mult beta={beta:.0f} it={it}] improved to {best_sc:.8f}")
            if stall >= 1000:
                alpha_cur *= 0.5
                stall = 0
            if alpha_cur < 5e-4:
                break
        else:
            alpha_cur *= 0.6
            stall += 1
            if alpha_cur < 5e-4:
                break
            if stall >= 5000:
                break
    # evaluate final
    if cur_sc < best_sc:
        best_sc=cur_sc; best_a=a.copy()
    return best_sc, best_a

def resample_to_n(p, n):
   p=np.asarray(p,dtype=np.float64).ravel()
   if p.size == n: return p
   if p.size ==0: return np.ones(n)*0.5
   old=np.linspace(0.0,p.size-1.0,p.size)
   new=np.linspace(0.0,n-1.0,n)
   return np.interp(new, old, p)

"""
Optimize a length-8192 non-negative step function to minimize
    score = 2*n*max(convolve(a,a)) / (sum(a))^2.

The implementation uses:
  - scale-invariant LSE-smoothed objectives with an explicit normalized gradient,
  - multiplicative equalization updates driven by softmax/active-set Lagrange weights,
  - L-BFGS-B polishing with adaptive LSE temperature,
  - several diverse initializations, including central-binomial "square-root" starts,
  - occasional perturbations and exact active-lag polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize

_N_TARGET = 8192

_BOUNDS_CACHE = {}
_CENTRAL_CACHE = {}

_LBFGS_DEADLINE = None
_LBFGS_ITER = 0
_LBFGS_BEST_SCORE = np.inf
_LBFGS_BEST_A = None


def _fft_len(m):
    """Smallest power of two at least m."""
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    """Full linear convolution of a sequence with itself via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    M = _fft_len(m)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:m]


def cross_weight(a, w):
    """
    Gradient-related operator for weighted convolution.

    For F(a) = sum_l w_l (a*a)_l, this returns dF/da (without normalization).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = 3 * n - 2
    M = _fft_len(m)

    R = np.fft.rfft(a[::-1], M)
    W = np.fft.rfft(w, M)
    out = np.fft.irfft(R * W, M)
    return 2.0 * out[n - 1:2 * n - 1]


def normalize(a):
    """Clip to [0, 1000] and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a

    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 1e-300:
        a.fill(1.0 / max(1, a.size))
        return a

    return a / S


def score_and_norm(a):
    """Return (exact scale-invariant score, normalized array, convolution)."""
    a = normalize(a)
    c = fast_conv(a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf, a, c
    sc = 2.0 * a.size * mx
    return sc, a, c


def exact_score(a):
    """Exact scale-invariant score."""
    sc, _, _ = score_and_norm(a)
    return sc


def central_binom_init(n, alpha=1.0):
    """
    Coefficients proportional to central binomial coefficients raised to alpha.

    For alpha=1, the generating function is (1-z)^(-1/2), whose self-convolution
    is flat on its infinite support. This is a strong spectral initialization.
    """
    n = int(n)
    key = (n, float(alpha))
    cached = _CENTRAL_CACHE.get(key)
    if cached is not None:
        return cached.copy()

    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    if alpha != 1.0:
        c = np.power(np.maximum(c, 0.0), float(alpha))

    _CENTRAL_CACHE[key] = c.copy()
    return c.copy()


def resample_to_n(p, n):
    """Resample a one-dimensional sequence to length n by linear interpolation."""
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.ones(n, dtype=np.float64) * 0.5
    if p.size == n:
        return p.copy()

    old = np.linspace(0.0, p.size - 1.0, p.size)
    new = np.linspace(0.0, n - 1.0, n)
    return np.interp(new, old, p)


def build_inits(n, rng, parent=None):
    """Build a diverse pool of initial candidates."""
    inits = []

    if parent is not None:
        p = resample_to_n(parent, n)
        p = np.clip(p, 0.0, 1000.0)
        inits.append(p)
        inits.append(p[::-1])
        inits.append(0.5 * (p + p[::-1]))

        for sig in (0.03, 0.07, 0.15, 0.30):
            q = p * np.exp(sig * rng.standard_normal(n))
            inits.append(np.clip(q, 0.0, 1000.0))

    for alpha in (1.0, 0.95, 0.85, 0.75, 1.10):
        inits.append(central_binom_init(n, alpha))

    x = np.arange(n, dtype=np.float64) / float(n)
    for f, amp in ((1.0, 0.15), (2.0, 0.25), (3.0, 0.18), (5.0, 0.15)):
        q = 0.5 + amp * np.cos(2.0 * np.pi * f * x)
        inits.append(np.clip(q, 1e-8, 1000.0))

    inits.append(np.clip(0.5 + 0.35 * rng.standard_normal(n), 1e-8, 1000.0))

    return inits


def select_top(scored, k):
    """Select top k (score, array) pairs without using lambdas."""
    pool = list(scored)
    top = []
    if not pool:
        return top

    while pool and len(top) < k:
        best_idx = 0
        best_val = pool[0][0]
        for i in range(1, len(pool)):
            if pool[i][0] < best_val:
                best_val = pool[i][0]
                best_idx = i
        top.append(pool.pop(best_idx))
    return top


def update_best(best_sc, best_a, sc, a):
    """Return improved pair if sc is finite and better."""
    if np.isfinite(sc) and sc < best_sc:
        if best_a is None:
            return float(sc), normalize(a).copy()
        return float(sc), normalize(a).copy()
    if best_a is None:
        return float(best_sc if np.isfinite(best_sc) else np.inf), normalize(a).copy()
    return float(best_sc), best_a


def get_bounds(n):
    """Cached bounds for L-BFGS-B."""
    b = _BOUNDS_CACHE.get(n)
    if b is None:
        b = [(0.0, 1000.0)] * n
        _BOUNDS_CACHE[n] = b
    return b


def lse_norm_fgrad(x, beta):
    """
    Scale-invariant LSE-smoothed objective and gradient.

    The objective is 2n * log-average of exp(beta*c_l), where c_l are convolution
    coefficients of the normalized input. The beta -> infinity limit is the exact
    max-convolution score.
    """
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return (1e10, np.zeros(0, dtype=np.float64))

    xw = np.maximum(x, 0.0)
    S = float(np.sum(xw))
    if not np.isfinite(S) or S <= 1e-300:
        return (1e10, np.zeros(n, dtype=np.float64))

    a = xw / S
    c = fast_conv(a)
    K = c.size
    M = float(np.max(c))
    if not np.isfinite(M):
        return (1e10, np.zeros(n, dtype=np.float64))

    z = beta * (c - M)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)

    ez = np.exp(z)
    ez_sum = float(np.sum(ez))

    if ez_sum <= 1e-300:
        w = np.zeros(K, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        L = M
    else:
        w = ez / ez_sum
        # log-mean LSE: subtract log(K) to keep the small-beta limit bounded.
        L = M + (float(np.log(ez_sum)) - float(np.log(K))) / beta

    h = cross_weight(a, w)
    mu = float(np.dot(a, h))

    if not np.isfinite(L):
        return (1e10, np.zeros(n, dtype=np.float64))

    f = 2.0 * n * L
    g = 2.0 * n * (h - mu) / S

    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    if not np.isfinite(f):
        f = 1e10

    return (float(f), g)


def lbfgs_callback(xk):
    """Stop L-BFGS when its deadline is hit and track a periodic exact best."""
    global _LBFGS_ITER, _LBFGS_BEST_SCORE, _LBFGS_BEST_A

    if _LBFGS_DEADLINE is not None and time.monotonic() >= _LBFGS_DEADLINE:
        return True

    _LBFGS_ITER += 1
    if _LBFGS_ITER % 30 == 0:
        sc, an, _ = score_and_norm(xk)
        if np.isfinite(sc) and sc < _LBFGS_BEST_SCORE:
            _LBFGS_BEST_SCORE = sc
            _LBFGS_BEST_A = an.copy()

    return False


def run_lbfgs_norm(a0, beta, deadline, max_seconds=None, maxiter=3000, maxfun=20000):
    """
    Run L-BFGS-B on the normalized LSE objective.

    Returns (best_array, best_score) seen during this call.
    """
    global _LBFGS_DEADLINE, _LBFGS_ITER, _LBFGS_BEST_SCORE, _LBFGS_BEST_A

    if max_seconds is not None:
        deadline = min(deadline, time.monotonic() + float(max_seconds))

    if time.monotonic() >= deadline:
        a0 = normalize(a0)
        return a0, exact_score(a0)

    a0 = normalize(a0)

    _LBFGS_ITER = 0
    _LBFGS_BEST_SCORE = np.inf
    _LBFGS_BEST_A = None

    sc_init, a_init, _ = score_and_norm(a0)
    _LBFGS_BEST_SCORE = sc_init
    _LBFGS_BEST_A = a_init.copy()

    options = {
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "ftol": 1e-15,
        "gtol": 1e-10,
        "maxls": 50,
        "maxcor": 30,
    }

    x_final = a_init
    try:
        _LBFGS_DEADLINE = deadline
        res = minimize(
            lse_norm_fgrad,
            a_init,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=get_bounds(a_init.size),
            options=options,
            callback=lbfgs_callback,
        )
        if res is not None and hasattr(res, "x") and np.asarray(res.x).size == a_init.size:
            x_final = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x_final = a_init
    finally:
        _LBFGS_DEADLINE = None

    x_final = np.clip(x_final, 0.0, 1000.0)
    sc_final, a_final, _ = score_and_norm(x_final)

    best_a = _LBFGS_BEST_A if _LBFGS_BEST_A is not None else a_final
    best_sc = _LBFGS_BEST_SCORE

    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final

    return best_a, best_sc


def multiplicative_phase(a0, beta, n_iters, alpha, deadline, verbose=False):
    """
    Multiplicative equalization.

    beta > 0 uses softmax weights exp(beta*c).
    beta == 0 uses uniform weights over lags within a small tolerance of max(c).

    Each update is accepted only if the exact score does not increase, with
    adaptive damping if stagnating.
    """
    sc, a, c = score_and_norm(a0)
    best_sc = sc
    best_a = a.copy()
    cur_sc = sc

    alpha_cur = max(5e-4, float(alpha))
    n = a.size
    K = c.size
    stall = 0
    it = 0
    n_iters = int(n_iters)

    while it < n_iters:
        it += 1
        if deadline is not None and time.monotonic() > deadline:
            break

        M = float(np.max(c))
        if not np.isfinite(M):
            break

        if beta > 1e-12:
            z = beta * (c - M)
            z = np.minimum(z, 0.0)
            z = np.maximum(z, -745.0)
            ez = np.exp(z)
            s = float(np.sum(ez))
            if s <= 1e-300:
                w = np.zeros(K, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ez / s
        else:
            tol = max(1e-3 * M, 1e-14)
            active = (c >= M - tol).astype(np.float64)
            s = float(np.sum(active))
            if s <= 1e-300:
                w = np.zeros(K, dtype=np.float64)
                w[int(np.argmax(c))] = 1.0
            else:
                w = active / s

        h = cross_weight(a, w)
        mu = float(np.dot(a, h))
        if not np.isfinite(mu) or mu <= 1e-300:
            break

        h_safe = np.maximum(h, mu * 1e-8)
        ratio = mu / h_safe
        ratio = np.clip(ratio, 1e-6, 1e6)
        factor = np.power(ratio, alpha_cur)
        factor = np.clip(factor, 0.02, 50.0)

        a_new = a * factor
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = float(np.sum(a_new))
        if not np.isfinite(S_new) or S_new <= 1e-300:
            break

        a_new /= S_new
        c_new = fast_conv(a_new)
        mx_new = float(np.max(c_new))
        if not np.isfinite(mx_new):
            break

        sc_new = 2.0 * n * mx_new

        if sc_new <= cur_sc + 1e-12 * max(1.0, cur_sc):
            if sc_new < cur_sc:
                stall = 0
            else:
                stall += 1

            a = a_new
            c = c_new
            cur_sc = sc_new

            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                if verbose and sc_new < best_sc - 1e-9:
                    print(f"[multiplicative beta={float(beta):.0f} it={it}] best={best_sc:.9f}")

            if stall >= 1000:
                alpha_cur *= 0.5
                stall = 0
                if alpha_cur < 5e-4:
                    break
        else:
            alpha_cur *= 0.6
            stall += 1
            if alpha_cur < 5e-4:
                break
            if stall >= 5000:
                break

    if np.isfinite(cur_sc) and cur_sc < best_sc:
        best_sc = cur_sc
        best_a = a.copy()

    return best_sc, best_a


def perturb_candidate(a, rng, strategy):
    """Create a randomized perturbation of a candidate."""
    a = normalize(a)
    n = a.size
    s = int(strategy) % 6

    if s == 0:
        sig = 0.03 + 0.07 * rng.random()
        q = a * np.exp(sig * rng.standard_normal(n))
    elif s == 1:
        amp = 0.02 + 0.06 * rng.random()
        q = a * (1.0 + amp * rng.standard_normal(n))
    elif s == 2:
        freq = 1.0 + 6.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = 0.02 + 0.05 * rng.random()
        x = np.arange(n, dtype=np.float64)
        q = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase))
    elif s == 3:
        b = max(8, n // 80)
        nb = n // b
        q = a.copy()
        if nb > 1:
            perm = rng.permutation(nb)
            q2 = np.empty(n, dtype=np.float64)
            for i in range(nb):
                src = int(perm[i]) * b
                dst = i * b
                q2[dst:dst + b] = a[src:src + b]
            rem = n % b
            if rem:
                q2[n - rem:] = a[n - rem:]
            q = q2
    elif s == 4:
        q = 0.5 * a + 0.5 * a[::-1]
    else:
        x = np.arange(n, dtype=np.float64)
        ramp = 0.5 + 0.5 * x / max(1, n - 1)
        q = a * (0.5 + 0.5 * ramp)

    return normalize(np.clip(q, 0.0, 1000.0))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.

    Returns a Python list of floats. All values are in [0, 1000].
    """
    n = _N_TARGET
    budget = float(budget_s)
    if budget <= 20.0:
        a = normalize(central_binom_init(n, 1.0))
        return [float(v) for v in a]

    t0 = time.monotonic()
    deadline = t0 + budget - 15.0
    if deadline <= t0 + 5.0:
        deadline = t0 + max(5.0, budget * 0.95)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        if "height_sequence_1" in globals():
            parent = globals()["height_sequence_1"]
    except Exception:
        parent = None

    inits = build_inits(n, rng, parent)
    scored = []
    for a in inits:
        sc, an, _ = score_and_norm(a)
        scored.append((sc, an))

    best_sc = np.inf
    best_a = None
    for sc, an in scored:
        best_sc, best_a = update_best(best_sc, best_a, sc, an)

    if best_a is None:
        best_a = normalize(central_binom_init(n, 1.0))
        best_sc = exact_score(best_a)

    print(f"[init] best={best_sc:.9f} length={best_a.size} parent_available={parent is not None}")

    top_pool = select_top(scored, 10)

    # Fixed phase boundaries.  Using absolute budgets prevents later phases from
    # being skipped solely because earlier phases converged early.
    phase1_end = min(deadline, t0 + 0.08 * budget)
    phase2_end = min(deadline, t0 + 0.30 * budget)
    phase3_end = min(deadline, t0 + 0.55 * budget)
    phase4_end = min(deadline, t0 + 0.82 * budget)

    # -----------------------------------------------------------------
    # Phase 1: fast multiplicative warm-up on the best initial candidate.
    # -----------------------------------------------------------------
    p1a = t0 + (phase1_end - t0) * 0.40
    p1b = t0 + (phase1_end - t0) * 0.75

    if phase1_end > time.monotonic() + 5:
        sc, a = multiplicative_phase(best_a, 500.0, 10**9, 0.50, p1a, True)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

        sc, a = multiplicative_phase(best_a, 2000.0, 10**9, 0.35, p1b, True)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

        sc, a = multiplicative_phase(best_a, 8000.0, 10**9, 0.25, phase1_end, True)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

    print(f"[phase1] best={best_sc:.9f} t={time.monotonic()-t0:.1f}s")

    # -----------------------------------------------------------------
    # Phase 2: L-BFGS continuation with increasing LSE temperature.
    # -----------------------------------------------------------------
    lbfgs_betas = (500.0, 2000.0, 8000.0)
    for idx, beta in enumerate(lbfgs_betas):
        rem = phase2_end - time.monotonic()
        if rem <= 15.0:
            break
        sec = min(120.0, rem / float(len(lbfgs_betas) - idx))
        if sec <= 5.0:
            break
        end = min(phase2_end, time.monotonic() + sec)
        a, sc = run_lbfgs_norm(best_a, beta, end, max_seconds=sec, maxiter=2500, maxfun=25000)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

    if phase2_end - time.monotonic() > 20.0:
        sc, a = multiplicative_phase(best_a, 0.0, 10**9, 0.18, phase2_end, True)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

    print(f"[phase2] best={best_sc:.9f} t={time.monotonic()-t0:.1f}s")

    # -----------------------------------------------------------------
    # Phase 3: explore alternative qualified starts.
    # -----------------------------------------------------------------
    explored = 0
    for sc0, a0 in top_pool:
        if time.monotonic() >= phase3_end:
            break
        if explored >= 7:
            break
        if not np.isfinite(sc0):
            continue
        if sc0 > best_sc + 0.06:
            continue

        explored += 1
        rem0 = phase3_end - time.monotonic()
        if rem0 <= 70.0:
            break

        sec1 = min(28.0, rem0 * 0.25)
        end1 = min(phase3_end, time.monotonic() + sec1)
        a, sc = multiplicative_phase(a0, 1000.0, 10**9, 0.40, end1, False)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

        rem2 = phase3_end - time.monotonic()
        if rem2 <= 30.0:
            continue

        sec2 = min(45.0, rem2 * 0.50)
        end2 = min(phase3_end, time.monotonic() + sec2)
        a, sc = run_lbfgs_norm(a, 8000.0, end2, max_seconds=sec2, maxiter=1800, maxfun=18000)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

        rem3 = phase3_end - time.monotonic()
        if rem3 > 30.0:
            sec3 = min(25.0, rem3 * 0.50)
            end3 = min(phase3_end, time.monotonic() + sec3)
            a, sc = multiplicative_phase(best_a, 0.0, 10**9, 0.15, end3, False)
            best_sc, best_a = update_best(best_sc, best_a, sc, a)

    print(f"[phase3] best={best_sc:.9f} t={time.monotonic()-t0:.1f}s")

    # -----------------------------------------------------------------
    # Phase 4: repeated multiplicative / L-BFGS refinement on the global best.
    # -----------------------------------------------------------------
    phase4_schedule = (
        (3000.0, 10000.0, 0.20),
        (10000.0, 20000.0, 0.15),
        (0.0, 30000.0, 0.12),
    )
    phase4_idx = 0

    while time.monotonic() < phase4_end - 25.0:
        rem = phase4_end - time.monotonic()
        if rem <= 40.0:
            break

        beta_m, beta_l, alpha_m = phase4_schedule[phase4_idx % len(phase4_schedule)]

        # Multiplicative / active-lag phase.
        sec_m = min(35.0, rem * 0.45)
        if sec_m > 8.0:
            end_m = min(phase4_end, time.monotonic() + sec_m)
            sc, a = multiplicative_phase(best_a, beta_m, 10**9, alpha_m, end_m, True)
            best_sc, best_a = update_best(best_sc, best_a, sc, a)

        # L-BFGS polish.
        rem = phase4_end - time.monotonic()
        if rem > 35.0:
            sec_l = min(35.0, rem * 0.50)
            end_l = min(phase4_end, time.monotonic() + sec_l)
            a, sc = run_lbfgs_norm(best_a, beta_l, end_l, max_seconds=sec_l, maxiter=1800, maxfun=18000)
            best_sc, best_a = update_best(best_sc, best_a, sc, a)

        # Occasionally branch from the current best to escape shallow local minima.
        phase4_idx += 1
        rem = phase4_end - time.monotonic()
        if phase4_idx % 3 == 0 and rem > 80.0:
            strat = phase4_idx % 5
            p = perturb_candidate(best_a, rng, strat)
            sec_p = min(20.0, rem * 0.20)
            end_p = min(phase4_end, time.monotonic() + sec_p)
            p, sp = multiplicative_phase(p, 3000.0, 10**9, 0.25, end_p, False)

            if np.isfinite(sp) and sp < best_sc + 0.002:
                best_sc, best_a = update_best(best_sc, best_a, sp, p)
                rem = phase4_end - time.monotonic()
                if rem > 25.0:
                    sec_q = min(25.0, rem * 0.35)
                    end_q = min(phase4_end, time.monotonic() + sec_q)
                    q, sq = run_lbfgs_norm(best_a, 20000.0, end_q, max_seconds=sec_q, maxiter=1200, maxfun=12000)
                    best_sc, best_a = update_best(best_sc, best_a, sq, q)

    print(f"[phase4] best={best_sc:.9f} t={time.monotonic()-t0:.1f}s")

    # -----------------------------------------------------------------
    # Phase 5: final active-lag and high-tone L-BFGS polish.
    # -----------------------------------------------------------------
    if deadline - time.monotonic() > 25.0:
        end = min(deadline, time.monotonic() + 25.0)
        sc, a = multiplicative_phase(best_a, 0.0, 10**9, 0.10, end, True)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

    if deadline - time.monotonic() > 12.0:
        end = min(deadline, time.monotonic() + 12.0)
        a, sc = run_lbfgs_norm(best_a, 30000.0, end, max_seconds=12.0, maxiter=1000, maxfun=10000)
        best_sc, best_a = update_best(best_sc, best_a, sc, a)

    # Final sanity conversion and one authoritative check.
    out = []
    for v in best_a:
        if not np.isfinite(v):
            out.append(0.0)
        else:
            out.append(float(max(0.0, min(1000.0, v))))

    if sum(out) < 0.01:
        out = [1.0] + [0.0] * (n - 1)

    try:
        verified = float(evaluate_sequence(out))
        if np.isfinite(verified):
            print(f"[final] internal={best_sc:.9f} verified={verified:.9f} t={time.monotonic()-t0:.1f}s")
        else:
            print(f"[final] internal={best_sc:.9f} verified=inf")
    except Exception as e:
        print(f"[final] internal={best_sc:.9f} verify_error={e}")

    return out
