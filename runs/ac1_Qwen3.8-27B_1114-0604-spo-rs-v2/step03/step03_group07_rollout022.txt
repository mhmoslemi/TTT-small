    n = len(a)

    best_a = a.copy()

    # Lower bound: use the target (slightly optimistic)

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:

        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if verbose and it % 1000 == 0:
                print(f"    subgrad it={it}: {sc:.8f}")
        
        # Find all (near-)maximizing lags
        threshold = M * 0.999
        active_k = np.where(c >= threshold)[0]
        
        # Compute subgradient (average of gradients at active lags)
        sg = np.zeros(n)
        for k in active_k:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= len(active_k)
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        # Polyak step
        step = (sc - sc_lb) / gn2
        # Cap the step
        step = min(step, 1.0 / np.sqrt(1.0 + it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a
