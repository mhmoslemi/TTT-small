"""
Iterative Reweighted QP + L-BFGS-B + Subgradient optimizer for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Key innovation: exploits the reversed-Hankel structure of Q_w for O(n log n) matvecs,
solves the inner QP via FISTA with simplex projection, and uses temperature-annealed
reweighting to concentrate on peak convolution lags.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def q_matvec(a, w, w_fft_cache=None):
    """
    Compute Q_w @ a where (Q_w)_{i,j} = w[i+j].
    Uses FFT: (Qa)_i = [a_rev * w]_{n-1+i}
    Returns (result, w_fft) for caching.
    """
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    A_fft = np.fft.rfft(a_rev, L)
    if w_fft_cache is not None:
        W_fft = w_fft_cache
    else:
        W_fft = np.fft.rfft(w, L)
    conv = np.fft.irfft(A_fft * W_fft, L)
    result = conv[n - 1:2 * n - 1]
    return result, W_fft


def estimate_lipschitz(w, a_init, num_powers=20):
    """Estimate 2*lambda_max(Q_w) using power iteration."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    W_fft = np.fft.rfft(w, L)
    v = np.random.default_rng(0).standard_normal(n)
    v /= np.linalg.norm(v)
    lam = 0.0
    for _ in range(num_powers):
        Qv, _ = q_matvec(v, w, W_fft)
        lam_new = float(np.dot(v, Qv))
        v_new = Qv / (np.linalg.norm(Qv) + 1e-30)
        v = v_new
        lam = lam_new
    return 2.0 * lam


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1))[0][-1]
    tau = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)


def solve_qp_fista(w, a_init, num_iters, lipschitz=None):
    """
    Solve: min a^T Q_w a s.t. sum(a)=1, a>=0 using FISTA.
    Returns optimal a (sum=1) and objective value.
    """
    n = len(w // 2)
    a = a_init.copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    if lipschitz is None:
        lipschitz = estimate_lipschitz(w, a)
    alpha = 1.0 / max(lipschitz, 1e-10)

    L_fft = 1 << (3 * len(a) - 2).bit_length()
    W_fft = np.fft.rfft(w, L_fft)

    a_prev = a.copy()
    t = 1.0
    y = a.copy()

    for k in range(num_iters):
        Qy, _ = q_matvec(y, w, W_fft)
        grad = 2.0 * Qy
        a_new = project_simplex(y - alpha * grad)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        a_prev = a
        a = a_new
        t = t_new

    obj = float(np.dot(a, q_matvec(a, w)[0]))
    return a, obj


def iterative_reweighted_qp(a_init, budget_s, ragged_seed=42):
    """
    Main iterative reweighted QP loop.
    a_init: initial sequence (will be normalized to sum=1)
    Returns best sequence (sum=1) and best exact score.
    """
    rng = np.random.default_rng(ragged_seed)
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    deadline = t_start + budget_s

    # Tau schedule: geometric from 1.0 down to 0.02
    tau_values = np.geomspace(1.0, 0.02, num=25)

    for tau in tau_values:
        if time.monotonic() > deadline - 30.0:
            break

        # Update weights
        c = fast_conv(a)
        c_max = c.max()
        if c_max < 1e-15:
            break
        scaled = (c - c_max) / tau
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # Solve QP
        remaining = deadline - time.monotonic()
        # Estimate how many FISTA iterations we can afford
        # Each iteration ~2 FFTs of size ~32768 for n~7000, ~0.3ms each
        # Plus projection ~0.1ms
        # Budget ~remaining/2 per tau step
        est_time_per_iter = 0.001  # 1ms per FISTA iteration (conservative)
        max_iters = int(min(1000, max(100, remaining / 3.0 / est_time_per_iter / len(tau_values))))
        max_iters = max(100, min(max_iters, 800))

        a_new, qp_obj = solve_qp_fista(w, a, max_iters)
        a = a_new

        # Check exact score
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  tau={tau:.4f}: score={sc:.8f} (qp_obj={qp_obj:.8f})")

    return best_a, best_score


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: d/da [2n*L/S^2] = 2n*(grad_L/S^2 - 2L*grad_S/S^3)
    # grad_L = conv with p: (grad_L)[i] = 2*sum_j p[i+j]*a[j] (for the quadratic form)
    # Actually: L = p^T c where c = conv(a,a), so dL/da = 2 * (Q_p @ a) where (Q_p)_{i,j}=p[i+j]
    # And grad_S = ones(n)
    Mp = len(p)
    L_fft = 1 << (Mp + n - 1 - 1).bit_length()
    # Need conv_res[i] = sum_j p[i+j] * a[j] for i=0..n-1
    # Same as q_matvec but with p instead of w
    a_rev = a[::-1]
    A_fft = np.fft.rfft(a_rev, L_fft)
    P_fft = np.fft.rfft(p, L_fft)
    conv_full = np.fft.irfft(A_fft * P_fft, L_fft)
    conv_res = conv_full[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def lbfgs_obj(x):
        return lse_score_grad(x, beta)

    res = minimize(
        lbfgs_obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def exact_subgrad_polish(a, total_iters=2000):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        # Use uniform over top-k max values for better subgradient
        top_k = min(5, len(c))
        thresh = np.sort(c)[::-1][top_k - 1]
        mask = c >= thresh
        k_star_indices = np.where(mask)[0]
        t = it / max(total_iters - 1, 1)
        step_base = (best_sc - sc_lb) / (1.0 + it * 0.01)
        step = min(step_base, 0.1 / (1 + 0.1 * t))

        for k_star in k_star_indices:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            sg_M = np.zeros(n)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
            S2 = S * S
            S3 = S2 * S
            grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-15:
                continue
            a -= (step / len(k_star_indices)) * grad / (gn / (np.sqrt(n) + 1e-10))
            a = np.clip(a, 0.0, 1000.0)

        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a = best_a.copy()
            break
        # Don't renormalize - score is scale invariant

        sc = 2.0 * n * float(c.max()) / (S * S)  # use old S for score of old a
        c_new = fast_conv(a)
        S_actual = float(a.sum())
        sc_new = 2.0 * n * float(c_new.max()) / (S_actual * S_actual)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()

    return best_score_check(best_a), best_a


def best_score_check(a):
    return compute_score(a)


def coordinate_perturb(a, rng, n_trials=100):
    """Random coordinate perturbations with hill climbing."""
    n = len(a)
    a = a.copy()
    best_sc = compute_score(a)
    for _ in range(n_trials):
        n_pert = max(3, n // 200)
        idx = rng.integers(0, n, size=n_pert)
        a_new = a.copy()
        a_new[idx] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            a = a_new
    return best_sc, a

            other_idx = np.setdiff1d(np.arange(n_best), np.arange(j_start, j_end + 1))
            if len(other_idx) > 0:
                add_amt = a_pert[j_start:j_end + 1].sum() * 0.05 / len(other_idx)
                a_pert[other_idx] += add_amt
        elif strategy == 2:
            # Frequency perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block shuffle
            block_sz = max(10, n_best // 30)
            b1 = int(rng.integers(0, n_best - 2 * block_sz))
            b2 = int(rng.integers(0, n_best - 2 * block_sz))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        else:
            # Sparse perturbation
            n_sparse = max(10, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Polish with L-BFGS
        remaining = max(60, phase3_deadline - time.monotonic())
        sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=2000, maxfun=4000)
        report(sc, a_opt, f"local #{trial} (strat={strategy})")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final refinement (5% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Refinement")
    print(f"{'='*60}")

    # High-beta LSE
    if time.monotonic() < deadline - 10:
        sc, a_opt, _ = run_lbfgs(best_a, 10000, maxiter=5000, maxfun=10000)
        report(sc, a_opt, "final LBFGS beta=10000")

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=500)
        report(sc, a_pol, "subgrad polish")

    # One final L-BFGS
    if time.monotonic() < deadline - 5:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=6000)
        report(sc, a_opt, "final final LBFGS")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
