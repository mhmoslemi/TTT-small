```python
def mirror_descent_step(a, eta, n_peaks=1):
    """One step of mirror descent on non-negative simplex."""
    n = len(a)
    conv = fast_conv(a)
    # Find top n_peaks lags
    top_lags = np.argsort(conv)[::-1][:n_peaks]
    # Subgradient: weighted average of gradients at top peaks
    g = np.zeros(n)
    for k in top_lags:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            g[i] += a[k - i]
    g /= n_peaks  # average
    # Mirror descent update
    a_new = a * np.exp(-eta * g)
    a_new /= a_new.sum()
    return a_new
```

```python
    n = len(a)
```
```python
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    n = len(a)

    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)
```

```python
    """Compute subgradient of max_k conv[k]."""
    n = len(a)
    conv = fast_conv(a)
    top_lags = np.argsort(conv, kind='stable')[::-1][:top_k]

    for k in top_lags:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= float(top_k)
    return g, conv

    log_ratio -= log_ratio.max()  # numerical stability

    s = a_new.sum()

        a_new = np.ones_like(a) / len(a)
    else:
        a_new /= s
    return a_new
```

```python
"""
Hybrid mirror descent + coordinate descent + LSE optimizer for minimum peak autocorrelation.
Key innovation: Entropy-regularized mirror descent (subgradient method on simplex)
provides efficient global moves by directly targeting the convolution peak,
combined with exact coordinate descent for local refinement.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_subgradient(a, top_k=3):
    """Compute subgradient of max_k conv[k] at point a. Returns (gradient, conv)."""
    n = len(a)
    conv = fast_conv(a)
    top_lags = np.argsort(conv, kind='stable')[::-1][:top_k]
    g = np.zeros(n)
    for k in top_lags:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= float(top_k)
    return g, conv


def mirror_descent_phase(a, duration, rng, eta0=None):
    """
    Mirror descent with entropy regularizer on the simplex.
    Uses adaptive step size and periodic coordinate descent.
    Returns (best_score, best_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S_orig = float(a.sum())
    if S_orig < 1e-12:
        a[:] = 1.0 / n
        S_orig = 1.0
    a_norm = a / S_orig  # normalize to sum=1

    if eta0 is None:
        eta0 = 0.3 * n

    best_local = compute_score(a_norm * S_orig)
    best_local_a = a.copy()

    t_start = time.monotonic()
    iteration = 0
    a_avg = a_norm.copy()
    momentum = np.zeros(n)
    beta_mom = 0.9

    while time.monotonic() - t_start < duration:
        iteration += 1
        g, conv = compute_subgradient(a_norm, top_k=min(3 + iteration // 5000, 10))

        # Momentum
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            momentum = np.zeros(n)
            eta = 0.0
        else:
            momentum = beta_mom * momentum + (1.0 - beta_mom) * g
            eta = min(eta0 / (1.0 + 0.01 * iteration), 0.8 / max(g_norm, 1e-15))

        if eta > 0:
            # Mirror descent step on normalized sequence
            log_ratio = -eta * momentum
            log_ratio -= float(log_ratio.max())
            log_ratio = np.clip(log_ratio, -80, 80)
            weights = np.exp(log_ratio)
            a_new = a_norm * weights
            a_new = np.maximum(a_new, 1e-30)
            s = float(a_new.sum())
            if s > 1e-30:
                a_new /= s
            else:
                a_new = a_norm.copy()
            a_norm = a_new

        # Update running average
        a_avg = a_avg + (a_norm - a_avg) / float(iteration)

        # Periodic evaluation every 500 steps

            sc = compute_score(a_norm * S_orig)
            if sc < best_local:
                best_local = sc
                best_local_a = (a_norm * S_orig).copy()
                if iteration % 5000 == 0:
                    print(f"    MD iter {iteration}: {sc:.8f}")

            sc_avg = compute_score(a_avg * S_orig)

                best_local = sc_avg
                best_local_a = (a_avg * S_orig).copy()
                print(f"    MD iter {iteration} (avg): {sc_avg:.8f} (NEW BEST)")

                best_local_a = a_cd.copy()
                print(f"    MD iter {iteration} (CD): {sc_cd:.8f} (NEW BEST)")
                a_norm = a_cd / a_cd.sum()
                a_avg = a_norm.copy()

    return best_local, best_local_a


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent (top-level, no closure)."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, n_sweeps=1):
    """
    One or more full sweeps of coordinate descent with incremental conv updates.
    Returns (score, updated_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            i = int(i)
            a_i = a[i]

            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()

            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-8}
            )
            t_opt = float(res.x)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Phase 5: Final polish (10%)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    a_cur = best_a.copy()
    polish_sweeps = 0
    while time.monotonic() < deadline - 10.0:
        polish_sweeps += 1
        if polish_sweeps % 4 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if polish_sweeps % 5 == 0:
                print(f"  P5 sweep {polish_sweeps}: {sc:.8f} (NEW BEST)")

    # One final LSE if time permits
    if time.monotonic() < deadline - 20.0:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=4000, maxfun=80000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
            # Final CD
            a_cur = best_a.copy()
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
