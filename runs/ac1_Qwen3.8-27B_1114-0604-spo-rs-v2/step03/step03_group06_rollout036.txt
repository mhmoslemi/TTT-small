"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Primary method: iterative LP-based direction finding with L-BFGS-B polishing.
Secondary: LSE-continuation and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact evaluation score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    # Gradient: 2n * (2 * (rev_a * p) / S^2 - 2L / S^3)
    # where (rev_a * p)[j] = sum_k p_k * a_{k-j} (reversed)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (2.0 * conv_res / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M_val=None, max_time=20.0):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns optimal b rescaled to have same sum as a, or None.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    if n > 12000:
        return None
    t0 = time.monotonic()
    if M_val is None:
        M_val = float(np.max(fast_conv(a)))
    if M_val < 1e-15:
        return a.copy()
    
    A = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M_val]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        remaining = max_time - (time.monotonic() - t0)
        if remaining <= 0:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_time - (time.monotonic() - t0)
            if remaining <= 0:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    
    # Rescale to match sum of a
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step(a, max_time=25.0, aggressive=False):

    cur_sc = compute_score(a)
    n = len(a)
    
    # Solve LP with slight relaxation
    M_cur = float(np.max(fast_conv(a)))

def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_obj_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].copy())
    
    # Various structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 4))
    
    # Sinc-like
    z = x - 0.5
    sinc_val = np.sinc(3.0 * z)
    inits.append(np.clip(0.5 + 0.3 * sinc_val, 0.01, 100.0))
    
    # Gaussian
    inits.append(np.clip(np.exp(-4.0 * (2 * x - 1) ** 2), 0.01, 100.0))

    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(np.clip(0.3 + 0.5 * rng.random(n), 0.01, 100.0))
    
    # Chirp
    inits.append(0.5 + 0.3 * np.cos(np.pi * x * x * 20))
    
    return inits


def subgrad_polish(a, n_iters=200):
    """Exact subgradient descent on the max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of M w.r.t. a
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a = np.clip(a - step * grad, 0.0, 1000.0)
    
    return best_sc, best_a

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Primary method: iterative LP-based direction finding (maximize sum(b) s.t. a*b <= M)
with L-BFGS-B polishing using LSE-smoothed objective, multi-length exploration,
and subgradient refinement.
"""

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact evaluation: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient via FFT-based cross-correlation."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    p = exp_s / exp_s.sum()
    # dL/da_j = 2 * cross_corr(j) where cross_corr(j) = (rev_a * p)[j+n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    cross_corr = conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (2.0 * cross_corr / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2
    return score, grad


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] for linear convolution."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M_val=None, max_time=20.0):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns optimal b rescaled to same sum as a, or None on failure.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    if n > 12000:
        return None
    t0 = time.monotonic()
    if M_val is None:
        M_val = float(np.max(fast_conv(a)))
    if M_val < 1e-15:
        return a.copy()

    A = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M_val]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        remaining = max_time - (time.monotonic() - t0)
        if remaining <= 1.0:
            return None
        prob.solve(solver=cp.PDLP, max_time=remaining)
    except Exception:
        try:
            remaining = max_time - (time.monotonic() - t0)
            if remaining <= 1.0:
                return None
            prob.solve(solver=cp.ECOS, max_time=remaining)
        except Exception:
            return None

    if time.monotonic() - t0 > max_time * 1.5:
        return None
    if b_var.value is None:
        return None

    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None

    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step(a, max_time=20.0, M_factor=1.0):
    """One LP-based direction step with multiple trial step sizes."""
    cur_sc = compute_score(a)
    M_cur = float(np.max(fast_conv(a)))
    M_try = M_cur * M_factor

    b_opt = solve_lp_direction(a, M_val=M_try, max_time=max_time)
    if b_opt is None:
        return cur_sc, a

    best_sc = cur_sc
    best_a = a.copy()

    step_sizes = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in step_sizes:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B minimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_obj_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    sc = compute_score(a_out)
    return sc, a_out


def make_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 4))
    z = x - 0.5
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * z), 0.01, 100.0))
    inits.append(np.clip(np.exp(-4.0 * (2 * x - 1) ** 2), 0.01, 100.0))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.5 + 0.3 * np.cos(np.pi * x * x * 20.0))
    inits.append(0.4 + 0.2 * np.cos(6 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x))
    return inits


def subgrad_polish(a, n_iters=300):
    """Exact subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a = np.clip(a - step * grad, 0.0, 1000.0)

    return best_sc, best_a


def upsample_sequence(a, new_n):
    """Upsample sequence to new_n via frequency-domain zero-padding."""
    n = len(a)
    if new_n <= n:
        idx = np.linspace(0, n - 1, new_n).astype(int)
        return a[idx].copy()
    # FFT-based interpolation
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    # Zero-pad in frequency domain for interpolation
    new_M = 1 << (2 * new_n - 1).bit_length()
    A_new = np.zeros(new_M // 2 + 1, dtype=np.complex128)
    copy_len = min(M_fft // 2 + 1, A_new.shape[0])
    A_new[:copy_len] = A[:copy_len]
    a_new = np.fft.irfft(A_new, new_M * 2 - 2)[:2 * new_n - 1]
    # Actually, let's just do linear interpolation for simplicity
    idx = np.linspace(0, n - 1, new_n)
    a_new = np.interp(idx, np.arange(n), a)
    return np.clip(a_new, 0.0, 1000.0)

    print(f"\n{'='*60}\nPhase 3: Local search (to t={phase3_end - t_start:.0f}s)\n{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase3_end and trial < 20:
        strategy = trial % 5
        if strategy == 0:
            scale = 0.03 + 0.07 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Target peak-contributing elements
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_s = max(0, k_star - n_best + 1)
            j_e = min(k_star, n_best - 1)
            contrib = np.zeros(n_best)
            contrib[j_s:j_e + 1] = best_a[j_s:j_e + 1][::-1]
            top_k = np.argsort(contrib)[::-1][:max(10, n_best // 30)]
            a_pert = best_a.copy()
            a_pert[top_k] *= 0.8
            a_pert = a_pert * (best_a.sum() / a_pert.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency modulation
            freq = 1.0 + 6.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.03 + 0.05 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block shuffle
            bs = max(10, n_best // 15)
            a_pert = best_a.copy()
            b1 = int(rng.integers(0, n_best - 2 * bs))
            b2 = int(rng.integers(0, n_best - 2 * bs))
            if abs(b1 - b2) > bs:
                a_pert[b1:b1 + bs], a_pert[b2:b2 + bs] = \
                    a_pert[b2:b2 + bs].copy(), a_pert[b1:b1 + bs].copy()
        else:
            # Mix with structured
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2 * np.pi * rng.random() * 4 * x)
            a_pert = 0.8 * best_a + 0.2 * a_struct * (best_a.sum() / a_struct.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=2000, maxfun=20000)
        report(sc, a_opt, f"LS#{trial} s={strategy}")
        trial += 1

    print(f"  After P3: {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final high-precision refinement (7% time)
    # =========================================================================
    print(f"\n{'='*60}\nPhase 4: Final refinement\n{'='*60}")

    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 5:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=100000)
        report(sc, a_opt, f"fin b={beta}")

    if time.monotonic() < deadline - 10:
        sc, a_pol = subgrad_polish(best_a, n_iters=500)
        report(sc, a_pol, "subgrad")

    if HAVE_CVXPY and time.monotonic() < deadline - 30 and len(best_a) <= 10000:
        sc, a_new = lp_step(best_a, max_time=20.0)
        report(sc, a_new, "final LP")
        if sc < best_score:
            sc, a_opt = run_lbfgs(a_new, 5000, maxiter=3000, maxfun=30000)
            report(sc, a_opt, "final LP+pol")

    # =========================================================================
    # Return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
