```python
"""
LP cutting-plane method for minimizing peak autocorrelation of non-negative sequences.
Uses linearized convolution constraints solved as LPs, with adaptive active set management.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
```

"""
LP cutting-plane + LSE-LBFGS hybrid for peak autocorrelation minimization.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    M_len = 2 * n - 1
    M_fft = 1 << (M_len - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M_len]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad

def solve_step(a, c, active_idx):
    """
    Solve LP to find descent direction.
    Returns (direction d, predicted_new_peak t).
    """
    n = len(a)
    m = len(active_idx)
    n_vars = n + 1  # d_0...d_{n-1}, t
    
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n_vars))
    b_ub = np.zeros(m)
    
    for i, k in enumerate(active_idx):
        A_ub[i, -1] = -1.0
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            # grad component: 2 * a[k-j]
            # j ranges from j_start to j_end
            # a[k-j] means we need a at positions k-j_end, ..., k-j_start
            # which is a[k-j_end : k-j_start+1], reversed
            A_ub[i, j_start:j_end+1] = 2.0 * a[k-j_end : k-j_start+1][::-1]
        b_ub[i] = -c[k]
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(0, None)]
    for j in range(n):
        bounds[j] = (-a[j], None)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if not res.success:
        return None, None
    
    return res.x[:n], res.x[-1]

def lp_optimize(a_init, n_active=50, max_iter=1000, deadline=None, verbose=False):
    """
    Main LP cutting-plane optimization.
    a_init is normalized to sum=1.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    best_sc = 2.0 * n * best_M  # since sum=1
    
    c = fast_conv(a)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        M = float(c.max())
        
        # Active set: top n_active lags
        if n_active < 2*n-1:
            active = np.argsort(c)[::-1][:n_active]
        else:
            active = np.arange(2*n-1)
        
        d, t = solve_step(a, c, active)
        if d is None:

        # If predicted improvement is negligible, try with more active constraints
        if t > M * (1 - 1e-8):
            # Stuck, try more constraints
            n_active = min(2*n-1, n_active * 2)
            continue
        
        # Damped step
        alpha = 1.0
        step_taken = False
        for _ in range(30):
            a_try = a + alpha * d
            if np.min(a_try) >= -1e-10:
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try > 1e-12:
                    a_try = a_try / S_try
                    c_try = fast_conv(a_try)
                    M_try = float(c_try.max())
                    if M_try < M:
                        a = a_try
                        c = c_try
                        sc = 2.0 * n * M_try
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        step_taken = True
                        break
            alpha *= 0.5
        
        if not step_taken:
            # Tiny step in subgradient direction as fallback
            k_star = int(np.argmax(c))
            sg = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
            sg = sg - sg.mean()  # enforce sum=0
            a = np.maximum(a - 0.01 * sg, 0.0)
            S = a.sum()
            a = a / S
            c = fast_conv(a)
        
        # Convergence
        if M - best_M < 1e-12:
            break
    
    return best_sc, best_a

def run_lbfgs_polish(a0, beta, maxiter=5000, deadline=None):
    """L-BFGS-B polish with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_score_grad(x, beta)
    
    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': 100000}

        if remaining < 5:
            return compute_score(a0), a0
        options['maxfun'] = min(100000, max(100, int(remaining * 100)))
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out

"""
LP cutting-plane method with adaptive active set for minimizing peak autocorrelation
of non-negative sequences, combined with LSE-smoothed L-BFGS polishing.

The key idea is to iteratively linearize the peak autocorrelation constraint and
solve a LP to find the best descent direction, using an adaptive active set of
constraints that are "close to the peak."
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_step(a, c, active_idx):
    """
    Solve LP to find best descent direction.
    Minimize t subject to linearized conv constraints at active lags.
    Returns (direction, predicted_peak).
    """
    n = len(a)
    m = len(active_idx)
    n_vars = n + 1
    
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((m, n_vars))
    b_ub = np.zeros(m)
    
    for i, k in enumerate(active_idx):
        A_ub[i, -1] = -1.0
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A_ub[i, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
        b_ub[i] = -c[k]
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(0.0, None)]
    for j in range(n):
        bounds[j] = (-a[j], None)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if not res.success:
        return None, None
    
    return res.x[:n], float(res.x[-1])


def lp_optimize(a_init, n_active_init=30, n_active_max=200, max_iter=5000, deadline=None, verbose=False):
    """
    LP cutting-plane optimization.
    a_init should be non-negative with sum > 0.
    Returns (best_score, best_array).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S
    
    best_a = a.copy()
    c = fast_conv(a)
    best_M = float(c.max())
    best_sc = 2.0 * n * best_M
    
    n_active = n_active_init
    stall_count = 0
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        M = float(c.max())
        
        # Adaptive active set
        if n_active < 2 * n - 1:
            active = np.argsort(c)[::-1][:n_active]
        else:
            active = np.arange(2 * n - 1)
        
        d, t = solve_lp_step(a, c, active)
        if d is None:
            break
        
        # Check if improvement is meaningful
        if t > M * (1.0 - 1e-9):
            # Stuck: increase active set or perturb
            stall_count += 1
            if stall_count < 10:
                n_active = min(n_active_max, n_active + 10)
            elif stall_count < 30:
                # Perturb
                rng = np.random.default_rng(it)
                a = a + 0.001 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                S = a.sum()
                a = a / S
                c = fast_conv(a)
                n_active = n_active_init
            else:
                break
            continue
        
        stall_count = 0
        
        # Damped step with backtracking line search
        alpha = 1.0
        step_taken = False
        for _ in range(20):
            a_try = a + alpha * d
            if np.min(a_try) >= -1e-10:
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try > 1e-12:
                    a_try = a_try / S_try
                    c_try = fast_conv(a_try)
                    M_try = float(c_try.max())
                    if M_try < M * (1.0 - 1e-12):
                        a = a_try
                        c = c_try
                        sc = 2.0 * n * M_try
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        step_taken = True
                        break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not step_taken:
            # Fallback: tiny subgradient step
            k_star = int(np.argmax(c))
            sg = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
            sg = sg - sg.mean()
            step_size = min(0.01, 0.01 * M / (np.max(np.abs(sg)) + 1e-15))
            a = np.maximum(a - step_size * sg, 0.0)
            S = a.sum()
            a = a / S
            c = fast_conv(a)
        
        # Convergence check
        if M - best_M < 1e-13:
            break
    
    return best_sc, best_a


def run_lbfgs_polish(a0, beta, maxiter=5000, deadline=None):
    """L-BFGS-B polish with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        return lse_score_grad(x, beta)
    
    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': 100000}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            return compute_score(a0), a0
        options['maxfun'] = min(100000, max(200, int(remaining * 80)))
        options['maxiter'] = min(maxiter, max(200, int(remaining * 20)))
    
    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def multiplicative_equalize(a, n_iters=50, alpha=0.3, deadline=None):
    """LP-dual-inspired multiplicative update."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        threshold = M * 0.995
        lambda_v = (c >= threshold).astype(np.float64)
        if lambda_v.sum() == 0:
            lambda_v[int(np.argmax(c))] = 1.0
        
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

            best_a = a.copy()

            return True
        return False
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    # Determine working lengths
    lengths = [8192]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)
    lengths = lengths[:2]  # Limit to 2 lengths for time
    
    print(f"Lengths: {lengths}, Parent: {parent is not None}")
    
    # ========== Phase 1: LP Cutting-Plane (main) ==========
    phase1_deadline = t_start + 0.55 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Cutting-Plane (deadline at t+{phase1_deadline-t_start:.0f}s)")
    print(f"{'='*60}")
    
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        
        # Get starting point
        if parent is not None and len(parent) == n:
            a0 = parent.copy()
        elif parent is not None:
            a0 = np.interp(np.linspace(0, len(parent)-1, n), np.arange(len(parent)), parent)
        else:
            a0 = np.ones(n) / n
        
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0[:] = 1.0 / n
        
        print(f"\n--- LP optimize: n={n} ---")
        sc, a_opt = lp_optimize(a0, n_active_init=30, n_active_max=300,
                                 max_iter=10000, deadline=phase1_deadline)

        if sc < best_score:
            # Continue with more refinement
            pass
    
    # ========== Phase 2: LSE-LBFGS Polish ==========
    phase2_deadline = t_start + 0.80 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE-LBFGS Polish (deadline at t+{phase2_deadline-t_start:.0f}s)")
    print(f"{'='*60}")
    
    if best_a is not None:
        for beta in [100, 300, 1000, 3000]:
            if time.monotonic() > phase2_deadline - 30:
                break
            sc, a_opt = run_lbfgs_polish(best_a, beta, maxiter=3000,
                                          deadline=phase2_deadline - 30)
            _report(sc, a_opt, f"LBFGS beta={beta}")
            
            # Interleave with multiplicative equalization
            if time.monotonic() < phase2_deadline - 60:
                sc2, a_eq = multiplicative_equalize(best_a, n_iters=30, alpha=0.2,
                                                     deadline=phase2_deadline - 30)
                _report(sc2, a_eq, "multiplicative")
    
    # ========== Phase 3: More LP from polished solution ==========
    phase3_deadline = deadline - 30
    print(f"\n{'='*60}")
    print(f"Phase 3: Final LP + Polish (deadline at t+{phase3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")
    
    if best_a is not None:
        # One more round of LP
        sc, a_opt = lp_optimize(best_a, n_active_init=50, n_active_max=500,
                                 max_iter=5000, deadline=phase3_deadline - 60)
        _report(sc, a_opt, "LP final")
        
        # Final LSE-LBFGS
        for beta in [5000, 10000]:
            if time.monotonic() > phase3_deadline - 20:
                break
            sc, a_opt = run_lbfgs_polish(best_a, beta, maxiter=5000,
                                          deadline=phase3_deadline - 10)
            _report(sc, a_opt, f"final LBFGS beta={beta}")
    
    # ========== Final verification ==========
    if best_a is None:
        out = [0.5] * 1024
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 64

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out

"""
LP cutting-plane method with adaptive active set + LSE-smoothed L-BFGS for
minimizing peak autocorrelation of non-negative sequences.

Key idea: Iteratively linearize the peak convolution constraint and solve a
linear program to find the best descent direction. The active set of constraints
is adapted to include lags closest to the peak. This is interleaved with
LSE-smoothed gradient-based optimization for final polishing.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def solve_lp_step(a, c, active_idx):
    """
    Solve LP to find best descent direction.
    Minimize t subject to linearized conv constraints at active lags.
    Returns (direction_array, predicted_new_peak) or (None, None).
    """
    n = len(a)
    m = len(active_idx)
    n_vars = n + 1

    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0

    A_ub = np.zeros((m, n_vars))
    b_ub = np.zeros(m)

    for i, k in enumerate(active_idx):
        A_ub[i, -1] = -1.0
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            A_ub[i, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
        b_ub[i] = -c[k]

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] * n + [(0.0, None)]
    for j in range(n):
        bounds[j] = (-a[j], None)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None

    if not res.success:
        return None, None

    return res.x[:n], float(res.x[-1])


def lp_optimize(a_init, n_active_init=30, n_active_max=300, max_iter=10000,
                deadline=None, verbose=False):
    """
    LP cutting-plane optimization.
    a_init should be non-negative with sum > 0.
    Returns (best_score, best_array).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    c = fast_conv(a)
    best_M = float(c.max())
    best_sc = 2.0 * n * best_M

    n_active = n_active_init
    stall_count = 0
    iter_count = 0

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        iter_count += 1

        M = float(c.max())

        # Adaptive active set: top lags by convolution value
        if n_active < 2 * n - 1:
            active = np.argsort(c)[::-1][:n_active]
        else:
            active = np.arange(2 * n - 1)

        d, t = solve_lp_step(a, c, active)
        if d is None:
            break

        # Check if improvement is meaningful
        if t > M * (1.0 - 1e-9):
            stall_count += 1
            if stall_count <= 5:
                n_active = min(n_active_max, n_active + 15)
            elif stall_count <= 20:
                # Perturb to escape
                rng = np.random.default_rng(12345 + it)
                a = a + 0.002 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                S = a.sum()
                if S < 1e-12:
                    a[:] = 1.0 / n
                    S = 1.0
                a = a / S
                c = fast_conv(a)
                n_active = n_active_init
            else:
                break
            continue

        stall_count = 0

        # Damped step with backtracking line search
        alpha = 1.0
        step_taken = False
        for _ in range(25):
            a_try = a + alpha * d
            if np.min(a_try) >= -1e-10:
                a_try = np.maximum(a_try, 0.0)
                S_try = a_try.sum()
                if S_try > 1e-12:
                    a_try = a_try / S_try
                    c_try = fast_conv(a_try)
                    M_try = float(c_try.max())
                    if M_try < M * (1.0 - 1e-12):
                        a = a_try
                        c = c_try
                        sc = 2.0 * n * M_try
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a.copy()
                        step_taken = True
                        break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if not step_taken:
            # Fallback: subgradient step
            k_star = int(np.argmax(c))
            sg = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
            sg = sg - sg.mean()
            sg_norm = np.max(np.abs(sg))
            if sg_norm > 1e-15:
                step_size = min(0.005, 0.005 * M / sg_norm)
                a = np.maximum(a - step_size * sg, 0.0)
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                    c = fast_conv(a)

        # Convergence
        if M - best_M < 1e-14:
            break

        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best_sc={best_sc:.8f}, "
                  f"n_active={n_active}, t={time.monotonic()-(deadline-0.55*2400) if deadline else 0:.1f}s")

    return best_sc, best_a


def run_lbfgs_polish(a0, beta, maxiter=5000, deadline=None):
    """L-BFGS-B polish with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_score_grad(x, beta)

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': 100000}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            return compute_score(a0), a0
        options['maxfun'] = min(100000, max(200, int(remaining * 80)))
        options['maxiter'] = min(maxiter, max(200, int(remaining * 20)))

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def multiplicative_equalize(a, n_iters=50, alpha=0.3, deadline=None):
    """LP-dual-inspired multiplicative update."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        threshold = M * 0.995
        lambda_v = (c >= threshold).astype(np.float64)
        if lambda_v.sum() == 0:
            lambda_v[int(np.argmax(c))] = 1.0

        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a
