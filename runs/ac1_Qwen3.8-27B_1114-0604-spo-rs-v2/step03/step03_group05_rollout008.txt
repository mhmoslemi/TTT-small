import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def build_conv_matrix(n, f):
    """Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if the index k-j is valid (0 <= k-j <= n-1), else 0.
    Shape: (2n-1, n)
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp_waters(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g.
    """
    n = len(f)
    # For efficiency, build the matrix once
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def peak_shave(a, iters=3000):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

                c = fast_conv(a)
                S = float(a.sum())
                M = float(c.max())
                k2 = int(np.argmax(c))
                j_start = max(0, k2 - n + 1)
                j_end = min(k2, n - 1)
                sg = np.zeros(n)
                if j_end >= j_start:
                    sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
                S2 = S * S
                S3 = S2 * S
                grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
                gn = float(np.linalg.norm(grad))
                if gn < 1e-15:
                    break
                step = 0.1 / (gn * S + 1e-20)
                step = min(step, 0.5 / S)
                a -= step * grad
                a = np.clip(a, 0.0, 1000.0)
                sc = 2.0 * n * float(c.max()) / (S * S)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to the peak, redistribute to others."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(5, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def lp_waters_filling(a, rows, cols, deadline, t=0.4, verbose=True):
    """LP-based waters filling optimization. Runs until deadline."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    if verbose:
        print(f"  LP waters filling start: score={best_sc:.8f}")

    stagnant = 0
    for it in range(10000):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M
        g = solve_lp_waters(a, M, rows, cols, n, time_limit=90)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g / g_sum
        diff = float(np.linalg.norm(g - a))
        if verbose and (it % 20 == 0 or it < 5):
            print(f"    LP iter {it}: score={sc:.8f}, sum_g_raw={g_sum:.6f}, ||g-a||={diff:.6f}")
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        a = a_new
        if sc_new < best_sc - 1e-10:
            best_sc = sc_new
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                t = max(0.05, t * 0.7)
                stagnant = 0
        if diff < 1e-7:
            if verbose:
                print(f"    Converged at iter {it}, score={best_sc:.8f}")
            break
    if verbose:
        print(f"  LP waters filling done: best score={best_sc:.8f}")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LSE continuation, LP-based waters filling, peak shaving, and structured restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: score={sc:.8f}, length={len(parent)}")
        n_use = len(parent)
    else:
        n_use = 2048
        best_a = np.ones(n_use) * 0.5
        sc = compute_score(best_a)
        report(sc, best_a, "init")
        print(f"Initialized: n={n_use}, score={sc:.8f}")

    # ===================================================================
    # Phase 1: LSE optimization - 20% time
    # ===================================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 1: LSE Optimization (deadline t={phase1_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    beta_schedule = [30, 60, 150, 300, 600, 1200, 2500]
    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj_func, (beta,), maxiter=800, maxfun=20000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE beta={beta}")
        print(f"  beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 2: LP Waters Filling - 50% time
    # ===================================================================
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LP Waters Filling (deadline t={phase2_end - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    n_cur = len(a_cur)
    print(f"  Building convolution pattern for n={n_cur}...")
    t0 = time.monotonic()
    conv_rows, conv_cols = build_conv_pattern(n_cur)
    print(f"  Pattern built in {time.monotonic()-t0:.1f}s, nnz={len(conv_rows)}")

    lp_deadline = min(phase2_end, time.monotonic() + 0.45 * (total_budget - 30.0))
    sc, a_lp = lp_waters_filling(a_cur, conv_rows, conv_cols, lp_deadline, t=0.4)
    report(sc, a_lp, "LP waters")

    # Second round of LP with smaller step for refinement
    if time.monotonic() < phase2_end:
        lp_deadline2 = phase2_end
        sc, a_lp2 = lp_waters_filling(a_lp, conv_rows, conv_cols, lp_deadline2, t=0.15)
        report(sc, a_lp2, "LP waters refine")
        a_cur = a_lp2 if sc < compute_score(a_lp) else a_lp

    # ===================================================================
    # Phase 3: Peak Shaving - 15% time
    # ===================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak Shaving (deadline t={phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        iters = max(1000, int(remaining * 50))
        t0 = time.monotonic()
        sc, a_ps = peak_shave(best_a, iters=iters)
        report(sc, a_ps, "peak shave")
        print(f"  Peak shave took {time.monotonic()-t0:.1f}s")

        if time.monotonic() < phase3_end:
            sc, a_mp = multi_peak_shave(best_a, iters_per_peak=500, num_peaks=6)
            report(sc, a_mp, "multi-peak")

    # ===================================================================
    # Phase 4: Restarts - 15% time
    # ===================================================================
    print(f"\n=== Phase 4: Restarts ===")

    n_best = len(best_a)
    restart_count = 0
    while time.monotonic() < deadline and restart_count < 30:
        t0 = time.monotonic()
        strategy = restart_count % 6

        if strategy == 0:
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 6.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            a_pert = best_a.copy()
            thresh = np.percentile(a_pert, 50)
            high_mask = a_pert > thresh
            low_mask = ~high_mask
            if high_mask.sum() > 0 and low_mask.sum() > 0:
                transfer = 0.04 * a_pert[high_mask].sum() / high_mask.sum()
                a_pert[high_mask] -= transfer
                a_pert[low_mask] += transfer * high_mask.sum() / low_mask.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Block shuffle perturbation
            a_pert = best_a.copy()
            block = max(10, n_best // 50)
            for _ in range(5):
                start = rng.integers(0, n_best - 2 * block)
                swap_amt = 0.02 * best_a[start:start + block].sum()
                a_pert[start:start + block] -= swap_amt / block
                target = rng.integers(0, n_best - block)
                a_pert[target:target + block] += swap_amt / block
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Frequency-domain perturbation
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.fft(best_a, L)
            A_mag = np.abs(A) * (1.0 + 0.015 * rng.standard_normal(L))
            A_mag = np.clip(A_mag, 0, None)
            A_phase = np.angle(A)
            a_pert = np.fft.ifft(A_mag * np.exp(1j * A_phase)).real[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Try quick LSE then LP
        a_opt = run_lbfgs(a_pert, lse_obj_func, (1500,), maxiter=300, maxfun=6000)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"restart {restart_count} (strat={strategy})")

        # If improved, try LP refinement
        if sc < best_score + 0.01 and time.monotonic() < deadline - 60:
            n_opt = len(a_opt)
            if n_opt == n_best:
                t0_inner = time.monotonic()
                inner_deadline = t0_inner + 60
                if inner_deadline > deadline:
                    inner_deadline = deadline
                sc_lp, a_lp_r = lp_waters_filling(a_opt, conv_rows, conv_cols,
                                                   inner_deadline, t=0.3, verbose=False)
                report(sc_lp, a_lp_r, f"restart {restart_count} LP")

        restart_count += 1

    # ===================================================================
    # Final polish
    # ===================================================================
    print(f"\n=== Final Polish ===")
    if time.monotonic() < deadline:
        a_final = run_lbfgs(best_a, lse_obj_func, (10000,), maxiter=500, maxfun=10000)
        sc = compute_score(a_final)
        report(sc, a_final, "final LSE")

        sc, a_pol = peak_shave(best_a, iters=2000)
        report(sc, a_pol, "final subgrad")

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
