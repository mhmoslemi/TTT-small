"""
Step function optimization using spectral flattening (Gerchberg-Saxton),
log-space Adam with LSE continuation, and subgradient polishing.
Uses larger sequence lengths for better theoretical bounds.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def gs_iterate(a, n_iter=100, relax=0.5, over_sample=1):
    """Gerchberg-Saxton type iteration for spectral flattening."""
    n = len(a)
    L = (1 << (2*n - 1).bit_length()) * over_sample
    a_cur = a.copy()
    for i in range(n_iter):
        # FFT of zero-padded a
        A = np.fft.fft(a_cur, L)
        # Flatten magnitude
        mag_mean = np.mean(np.abs(A))
        phase = A / (np.abs(A) + 1e-30)
        A_new = mag_mean * phase
        # IFFT and project to non-negative
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        # Relaxed update
        a_cur = (1 - relax) * a_cur + relax * a_new
        # Normalize
        if a_cur.sum() > 0:
            a_cur = a_cur / a_cur.sum() * np.sqrt(2*n)
    return a_cur


def adam_optimize(a0, beta_schedule, max_steps_per_beta=5000, lr=0.01, eps=1e-8):
    """Adam optimizer in log-space with LSE objective."""
    n = len(a0)
    # Initialize in log-space
    a = a0.copy()
    a = np.maximum(a, 1e-10)  # Ensure positive for log
    x = np.log(a)
    v_m = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_sc = compute_score(a)
    best_a = a.copy()
    
    for beta in beta_schedule:
        for step in range(max_steps_per_beta):
            t += 1
            # Compute score and grad in a-space
            sc, grad_a = lse_score_grad(a, beta)
            # Chain rule: grad_x = grad_a * a (since a = exp(x))
            grad_x = grad_a * a
            # Adam update
            beta1, beta2 = 0.9, 0.999
            v_m = beta1 * v_m + (1 - beta1) * grad_x
            v_v = beta2 * v_v + (1 - beta2) * grad_x ** 2
            v_m_hat = v_m / (1 - beta1 ** t)
            v_v_hat = v_v / (1 - beta2 ** t)
            x -= lr * v_m_hat / (np.sqrt(v_v_hat) + eps)
            a = np.exp(x)
            # Clip to avoid overflow
            a = np.clip(a, 0, 1000)
            # Update best
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            # Normalize to keep values reasonable
            S = a.sum()
            if S > 0:
                a = a / S * np.sqrt(2 * n)
                x = np.log(np.maximum(a, 1e-10))
    return best_sc, best_a


def peak_shave(a, iters=5000):

    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def gs_initialize(n, rng, n_iter=2000, relax=0.7):
    """
    Gerchberg-Saxton type initialization for flat-spectrum sequences.
    Alternatively start from a smooth profile and refine.
    """
    # Start from a smooth (non-uniform) profile to break symmetry
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * x) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum() * np.sqrt(2 * n)

    L = 1 << (2 * n - 1).bit_length()

    for i in range(n_iter):
        # Frequency domain
        A = np.fft.fft(a, L)
        mag = np.abs(A)
        # Replace magnitude with its mean (flatten spectrum)
        mag_mean = np.mean(mag)
        phase = A / (mag + 1e-30)
        A_new = mag_mean * phase
        # Back to time domain
        a_new = np.fft.ifft(A_new).real[:n]
        # Project to non-negative (soft: shift up rather than hard clip)
        a_new_shifted = a_new - a_new.min()
        # Relaxed combination
        if i < n_iter // 3:
            # Early: more aggressive flattening
            a = relax * a_new_shifted + (1.0 - relax) * a
        else:
            # Later: more conservative
            a = 0.8 * a + 0.2 * a_new_shifted
        # Ensure non-negativity and normalize
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            a = 0.5 + 0.1 * rng.random(n)
        a = a / a.sum() * np.sqrt(2 * n)

    return a


def gs_initialize_v2(n, rng, n_iter=3000):
    """
    Alternative GS initialization: start from random, use iterative
    magnitude/phase separation with non-negativity projection.
    """

    L = 1 << (2 * n - 1).bit_length()

    target_mag = np.ones(L) * np.sqrt(2.0 * n * n / L)

    for i in range(n_iter):
        # Step 1: FFT
        A = np.fft.fft(a, L)
        # Step 2: Replace magnitude with target (keeping phase)
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase
        # Step 3: IFFT
        a_new = np.fft.ifft(A_new).real
        # Step 4: Keep first n elements, project to non-negative
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        # Relaxation
        t = 0.5 if i < n_iter // 2 else 0.3
        a = (1.0 - t) * a + t * a_new
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            a = 0.5 + 0.1 * rng.random(n)
        a = a / a.sum() * np.sqrt(2 * n)

    return a


def adam_log_optimize(a0, beta_schedule, max_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, time_limit=None, t_start=None):
    """
    Adam optimizer in log-space with LSE-smoothed objective.
    Handles non-negativity naturally via log parameterization.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-10).astype(np.float64)
    x = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_sc = compute_score(a)
    best_a = a.copy()
    no_improve_count = 0

    for beta in beta_schedule:
        for step in range(max_steps):
            t += 1
            if time_limit is not None and t_start is not None:
                if time.monotonic() > t_start + time_limit:
                    return best_sc, best_a

            # Compute objective and gradient
            sc, grad_a = lse_score_grad(a, beta)
            if not np.isfinite(sc):
                break

            # Update best (using exact score every few steps for efficiency)
            if step % 50 == 0 or sc < best_sc:
                exact_sc = compute_score(a)
                if exact_sc < best_sc:
                    best_sc = exact_sc
                    best_a = a.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 50

            # Chain rule for log-space: dx/dx = a * da_dx
            grad_x = grad_a * a

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * grad_x * grad_x
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # Update a from x
            a = np.exp(np.clip(x, -30, 30))
            a = np.clip(a, 0.0, 1000.0)

            # Periodic normalization to keep values in reasonable range
            S = a.sum()
            if S > 0 and S > 1e6:
                a = a / S * np.sqrt(2.0 * n)
                x = np.log(np.maximum(a, 1e-30))

            # Adaptive learning rate: reduce if oscillating
            if no_improve_count > 200:
                lr *= 0.95
                no_improve_count = 0

    return best_sc, best_a


def subgrad_polish(a, iters, time_limit=None, t_start=None):
    """
    Subgradient descent on exact max-conv objective with Polyak step.
    Focuses on shaving the dominant peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        if time_limit is not None and t_start is not None:
            if time.monotonic() > t_start + time_limit:
                break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient contribution
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size with safeguards
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 2.0 / (S + 1e-10))
        step = max(step, 1e-8 / (gn * S + 1e-30))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a

def variance_obj(a):
    """Variance-based flatness objective (for L-BFGS)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    # Gradient of sum(c^2) w.r.t. a
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    E = np.abs(A) ** 2 * np.conj(A)
    E_ifft = np.fft.ifft(E).real
    idx = np.concatenate([[0], np.arange(L - 1, L - n, -1)])
    dc2_da = 4.0 * E_ifft[idx]
    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (dc2_da / S4 - 4.0 * c2 / S5)
    return score, grad


def run_lbfgs(a0, func, n, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0 = a0 + 0.1
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def targeted_perturb(a, rng, strength=0.05):

        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

    top_idx = np.argsort(contrib)[::-1][:top_count]

    a_pert = a.copy()

        amt = a_pert[j] * strength
        a_pert[j] -= amt

        total_red += amt
    if len(bottom_idx) > 0 and total_red > 0:
        per = total_red / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)

            continue

        print(f"\n{'='*60}")
        print(f"Length n={n}, time available: {seg_time_left:.0f}s")
        print(f"{'='*60}")

        # --- Initialization ---
        if parent is not None and len(parent) == n:
            a_cur = parent.copy()
            print(f"  Starting from parent, score={compute_score(a_cur):.8f}")
        else:
            # GS initialization
            t0 = time.monotonic()
            gs_time = min(20.0, seg_time_left * 0.1)
            gs_iters = int(gs_time * 30)  # ~30 iters/sec for moderate n
            gs_iters = max(500, min(gs_iters, 5000))
            a_cur = gs_initialize(n, rng, n_iter=gs_iters, relax=0.6)
            sc_gs = compute_score(a_cur)
            print(f"  GS init: score={sc_gs:.8f}, time={time.monotonic()-t0:.1f}s, iters={gs_iters}")
            if sc_gs < best_score:
                best_score = sc_gs
                best_a = a_cur.copy()
                print(f"  *** NEW BEST from GS: {sc_gs:.8f} ***")

            # Also try a simple random init + quick optimization
            a_rand = 0.5 + 0.5 * rng.random(n)
            sc_rand = compute_score(a_rand)
            print(f"  Random init score: {sc_rand:.8f}")

        # --- Phase 1: Variance flattening (quick) ---
        if time.monotonic() < seg_deadline - seg_time_left * 0.3:
            t0 = time.monotonic()
            a_v = run_lbfgs(a_cur, variance_obj, n, maxiter=300, maxfun=6000)
            sc_v = compute_score(a_v)
            print(f"  Variance flatten: score={sc_v:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_v < best_score:
                best_score = sc_v
                best_a = a_v.copy()
                print(f"  *** NEW BEST from variance: {sc_v:.8f} ***")
            if sc_v < compute_score(a_cur):
                a_cur = a_v

        # --- Phase 2: Adam in log-space (main optimization) ---
        if time.monotonic() < seg_deadline - seg_time_left * 0.2:
            remaining = seg_deadline - time.monotonic() - seg_time_left * 0.15
            # Beta schedule: start moderate, increase
            if n <= 8192:
                betas = [30, 60, 100, 200, 400, 800, 1600, 3000, 6000, 12000]
            else:
                betas = [30, 60, 100, 200, 400, 800, 1600, 3000, 6000]

            # Estimate steps per beta
            est_time_per_step = 0.02 if n <= 8192 else 0.05
            est_total_steps = int(remaining / est_time_per_step)
            steps_per_beta = max(100, est_total_steps // len(betas))
            steps_per_beta = min(steps_per_beta, 5000)

            t0 = time.monotonic()
            sc_adam, a_adam = adam_log_optimize(
                a_cur, betas, steps_per_beta,
                lr=0.01, time_limit=remaining, t_start=time.monotonic()
            )
            print(f"  Adam: score={sc_adam:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_adam < best_score:
                best_score = sc_adam
                best_a = a_adam.copy()
                print(f"  *** NEW BEST from Adam: {sc_adam:.8f} ***")
            a_cur = a_adam.copy()

        # --- Phase 3: L-BFGS at high beta ---
        if time.monotonic() < seg_deadline - 10:
            for beta_hi in [5000, 10000]:
                if time.monotonic() > seg_deadline - 5:
                    break
                t0 = time.monotonic()
                a_lb = run_lbfgs(a_cur, lse_obj_wrapper, n, maxiter=1500, maxfun=30000)
                # Fix the fact that lse_obj_wrapper needs beta arg
                # Actually let's just do it manually
                sc_lb = compute_score(a_lb)
                print(f"  L-BFGS (beta={beta_hi}): score={sc_lb:.8f}, time={time.monotonic()-t0:.1f}s")
                if sc_lb < best_score:
                    best_score = sc_lb
                    best_a = a_lb.copy()
                    print(f"  *** NEW BEST from L-BFGS: {sc_lb:.8f} ***")
                if sc_lb < compute_score(a_cur):
                    a_cur = a_lb

        # --- Phase 4: Subgradient polishing ---
        if time.monotonic() < seg_deadline - 3:
            remaining = seg_deadline - time.monotonic() - 3
            iters = int(remaining * 150)
            iters = max(1000, min(iters, 50000))
            t0 = time.monotonic()
            sc_sg, a_sg = subgrad_polish(a_cur, iters, time_limit=remaining, t_start=time.monotonic())
            print(f"  Subgrad: score={sc_sg:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_sg < best_score:
                best_score = sc_sg
                best_a = a_sg.copy()
                print(f"  *** NEW BEST from subgrad: {sc_sg:.8f} ***")
            a_cur = a_sg.copy()

        # --- Phase 5: Multi-peak shaving ---
        if time.monotonic() < seg_deadline - 2:
            remaining = seg_deadline - time.monotonic() - 2
            t0 = time.monotonic()
            sc_mp, a_mp = multi_subgrad(a_cur, iters_per_peak=100, n_peaks=5,
                                         time_limit=remaining, t_start=time.monotonic())
            print(f"  Multi-peak: score={sc_mp:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_mp < best_score:
                best_score = sc_mp
                best_a = a_mp.copy()
                print(f"  *** NEW BEST from multi-peak: {sc_mp:.8f} ***")

    # ===================================================================
    # Final phase: intensive polishing of the best found
    # ===================================================================
    if time.monotonic() < deadline and best_a is not None:
        print(f"\n{'='*60}")
        print(f"Final polishing phase (best score={best_score:.8f}, n={len(best_a)})")
        print(f"{'='*60}")

        final_deadline = deadline
        a_best = best_a.copy()

        # More GS iterations on the best
        t0 = time.monotonic()
        gs_time = min(15.0, (final_deadline - time.monotonic()) * 0.15)
        gs_iters = int(gs_time * 30)
        gs_iters = max(200, min(gs_iters, 3000))
        a_gs2 = gs_initialize(len(a_best), rng, n_iter=gs_iters, relax=0.3)
        # Blend with current best
        a_blend = 0.5 * a_best + 0.5 * a_gs2
        a_blend = np.maximum(a_blend, 0)
        sc_blend = compute_score(a_blend)
        print(f"  GS blend: score={sc_blend:.8f}")
        if sc_blend < best_score:
            best_score = sc_blend
            best_a = a_blend.copy()
            a_best = a_blend.copy()

        # Adam refinement at high beta
        if time.monotonic() < final_deadline - 10:
            remaining = final_deadline - time.monotonic() - 10
            t0 = time.monotonic()
            sc_f, a_f = adam_log_optimize(a_best, [8000, 15000, 30000], 2000,
                                           lr=0.005, time_limit=remaining, t_start=time.monotonic())
            print(f"  Final Adam: score={sc_f:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_f < best_score:
                best_score = sc_f
                best_a = a_f.copy()
                a_best = a_f.copy()

        # Final subgradient
        if time.monotonic() < final_deadline - 3:
            remaining = final_deadline - time.monotonic() - 3
            iters = int(remaining * 120)
            iters = max(2000, min(iters, 100000))
            t0 = time.monotonic()
            sc_final, a_final = subgrad_polish(a_best, iters, time_limit=remaining, t_start=time.monotonic())
            print(f"  Final subgrad: score={sc_final:.8f}, time={time.monotonic()-t0:.1f}s")
            if sc_final < best_score:
                best_score = sc_final
                best_a = a_final.copy()

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 100

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: score={best_score:.8f}")
    print(f"Sequence length: {len(out)}")
    print(f"Total time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
