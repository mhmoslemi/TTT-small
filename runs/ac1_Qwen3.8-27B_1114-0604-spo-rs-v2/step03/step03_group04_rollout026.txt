Uses softmax reparameterization (a[i]=softmax(x)[i]) to handle non-negativity smoothly,

def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""

    exact_sc = compute_score(a_opt)
    return exact_sc, a_opt, res


def subgradient_polish(a, n_iters=500):
    """Exact subgradient descent on the max-convolution objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.2)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Small additional step for progress
        a = a - (0.0005 / (1.0 + 0.005 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def spectral_flatten_flat(a0, n_iters=300, lam=0.6):
    """Iterative spectral flattening targeting uniform magnitude spectrum."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: flat magnitude (excluding DC and Nyquist)
        non_dc = mag[1:N // 2]
        target = np.median(non_dc) if len(non_dc) > 0 else mag[1]
        new_mag = mag.copy()
        for k in range(1, N // 2 + 1):
            if N % 2 == 0 and k == N // 2:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def peak_shave(a, rng, n_shaves=50, strength=0.03):
    """Iterative peak shaving: reduce contributions to max convolution peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for s in range(n_shaves):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        # Contributions to c[k_star]
        contrib = np.zeros(n)
        seg = a[j_start:j_end + 1]
        contrib[j_start:j_end + 1] = seg[::-1]
        # Reduce top contributors, add to bottom
        n_top = max(5, n // 100)
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:n_top]
        total_red = 0.0
        for j in top_idx:
            amt = a[j] * strength
            a[j] -= amt
            a[j] = max(a[j], 0.0)
            total_red += amt
        if total_red > 0 and len(bot_idx) > 0:
            per = total_red / len(bot_idx)
            for j in bot_idx:
                a[j] += per
        a = np.clip(a, 0.0, 1000.0)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_sc, best_a


def chirp_inits(n, rng, count=5):
    """Generate chirp-based non-negative initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    for _ in range(count):
        freq = 1.0 + 6.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        base = np.cos(np.pi * freq * x * x / (2.0 * n) + phase)
        a = base ** 2 + 0.05
        inits.append(np.clip(a, 0.01, 100.0))
    # Linear chirp
    freq = 1.0 + 4.0 * rng.random()
    a = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
    inits.append(np.clip(a, 0.01, 100.0))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses softmax-parametrized L-BFGS with LSE continuation, spectral initialization,
    and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = parent.size
    else:
        n_work = 4096
    print(f"Working length: n={n_work}")
    print(f"Parent available: {parent is not None}")

    # ================================================================
    # Phase 1: Generate diverse initializations
    # ================================================================
    print("\n--- Phase 1: Initialization ---")
    t_phase1_end = t_start + 0.15 * (deadline - t_start)

    candidates = []
    if parent is not None:
        candidates.append(('parent', parent.copy()))
        # Perturbed parent
        for scale in [0.02, 0.05, 0.1]:
            pert = parent * (1.0 + scale * rng.standard_normal(n_work))
            pert = np.clip(pert, 0.01, 1000.0)
            candidates.append((f'parent_pert_{scale}', pert))
        # Spectrally flattened parent
        sf_parent = spectral_flatten_flat(parent, n_iters=100, lam=0.5)
        candidates.append(('sf_parent', sf_parent))

    # Chirp-based
    chirps = chirp_inits(n_work, rng, count=4)
    for i, ch in enumerate(chirps):
        candidates.append((f'chirp_{i}', ch))
    # Spectrally flattened chirp
    sf_chirp = spectral_flatten_flat(chirps[0], n_iters=150, lam=0.5)
    candidates.append(('sf_chirp', sf_chirp))

    # Spectrally flattened random
    rand_init = np.clip(0.5 + 0.3 * rng.standard_normal(n_work), 0.01, 10.0)
    sf_rand = spectral_flatten_flat(rand_init, n_iters=150, lam=0.5)
    candidates.append(('sf_rand', sf_rand))

    # Evaluate and sort
    scored_cands = []
    for name, a in candidates:
        if time.monotonic() > t_phase1_end:
            break
        sc = compute_score(a)
        scored_cands.append((sc, name, a))
        print(f"    {name}: score={sc:.6f}")
    scored_cands.sort(key=lambda x: x[0])
    print(f"  Top candidate: {scored_cands[0][1]} with score={scored_cands[0][0]:.6f}")
    report(scored_cands[0][0], scored_cands[0][2], scored_cands[0][1])

    # ================================================================
    # Phase 2: L-BFGS optimization with softmax parametrization
    # ================================================================
    print("\n--- Phase 2: L-BFGS (softmax) Continuation ---")
    t_phase2_end = t_start + 0.65 * (deadline - t_start)

    beta_schedule = [30, 80, 200, 500, 1000, 2000, 5000]

    # Take top 4 candidates
    top_cands = scored_cands[:4]
    for ci, (sc_init, name, a_init) in enumerate(top_cands):
        if time.monotonic() > t_phase2_end:
            break
        print(f"\n  >> Candidate {ci}: {name} (init score={sc_init:.6f})")
        t0 = time.monotonic()

        # Convert to log-params: x = log(a) (softmax will re-normalize)
        a_norm = a_init / a_init.sum()
        a_norm = np.maximum(a_norm, 1e-10)
        x = np.log(a_norm)

        a_cur = a_init.copy()
        for bi, beta in enumerate(beta_schedule):
            if time.monotonic() > t_phase2_end:
                break
            remaining = t_phase2_end - time.monotonic()
            if remaining < 10:
                break
            # Scale iterations
            est_iter_time = 0.003 * (n_work / 4096)
            maxiter = min(5000, max(200, int(remaining * 0.7 / max(est_iter_time * 15, 0.05))))
            maxfun = min(80000, maxiter * 25)

            sc, a_opt, res = run_lbfgs_softmax(x, beta, maxiter=maxiter, maxfun=maxfun)
            if sc < best_score:
                a_cur = a_opt
                x = np.log(np.maximum(a_opt / a_opt.sum(), 1e-10))
                report(sc, a_opt, f"sm n={n_work} c={ci} b={beta}")
            else:
                # Try bounded L-BFGS as alternative
                sc_b, a_b, _ = run_lbfgs_bounded(a_cur, beta, maxiter=max(200, maxiter//2), maxfun=max(1000, maxfun//2))
                if sc_b < best_score:
                    a_cur = a_b
                    x = np.log(np.maximum(a_b / a_b.sum(), 1e-10))
                    report(sc_b, a_b, f"bd n={n_work} c={ci} b={beta}")

        # Final score from this trajectory
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"traject_{ci}_{name}")
        elapsed = time.monotonic() - t0
        print(f"    Trajectory {ci} done: score={sc_final:.6f} ({elapsed:.1f}s)")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: High-beta refinement on best solution
    # ================================================================
    print("\n--- Phase 3: High-Beta Refinement ---")
    t_phase3_end = t_start + 0.80 * (deadline - t_start)

    a_norm = best_a / best_a.sum()
    a_norm = np.maximum(a_norm, 1e-10)
    x_best = np.log(a_norm)

    for beta in [3000, 8000, 15000, 30000]:
        if time.monotonic() > t_phase3_end:
            break
        remaining = t_phase3_end - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(8000, max(500, int(remaining * 0.5 / 0.01)))
        sc, a_opt, _ = run_lbfgs_softmax(x_best, beta, maxiter=maxiter, maxfun=max(2000, maxiter * 20))
        if sc < best_score:
            report(sc, a_opt, f"refine_s b={beta}")
            x_best = np.log(np.maximum(a_opt / a_opt.sum(), 1e-10))

    # Also try bounded
    for beta in [5000, 15000]:
        if time.monotonic() > t_phase3_end:
            break
        remaining = t_phase3_end - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(5000, max(300, int(remaining * 0.4 / 0.01)))
        sc, a_opt, _ = run_lbfgs_bounded(best_a, beta, maxiter=maxiter, maxfun=max(1000, maxiter * 20))
        if sc < best_score:
            report(sc, a_opt, f"refine_b b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Subgradient polish + peak shaving
    # ================================================================
    print("\n--- Phase 4: Subgradient + Peak Shaving ---")
    t_phase4_end = t_start + 0.92 * (deadline - t_start)

    # Subgradient
    if time.monotonic() < t_phase4_end:
        sc, a_sg = subgradient_polish(best_a, n_iters=1000)
        report(sc, a_sg, "subgrad")

    # Peak shaving
    if time.monotonic() < t_phase4_end:
        sc, a_ps = peak_shave(best_a, rng, n_shaves=200, strength=0.02)
        report(sc, a_ps, "peak_shave")

    # L-BFGS from subgradient result
    if time.monotonic() < t_phase4_end:
        a_sg_norm = a_sg / a_sg.sum()
        a_sg_norm = np.maximum(a_sg_norm, 1e-10)
        x_sg = np.log(a_sg_norm)
        remaining = t_phase4_end - time.monotonic()
        if remaining > 20:
            sc, a_opt, _ = run_lbfgs_softmax(x_sg, 5000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, "post_sg_lbfgs")

    # ================================================================
    # Phase 5: Diverse perturbation + local optimization
    # ================================================================
    print("\n--- Phase 5: Diverse Local Search ---")

    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > deadline - 30:
            break
        strategy = trial % 5
        n_best = len(best_a)

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        elif strategy == 1:
            # Targeted perturbation at peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            n_red = max(10, n_best // 200)
            if j_end >= j_start:
                contrib = np.zeros(n_best)
                contrib[j_start:j_end+1] = best_a[j_start:j_end+1][::-1]
                top = np.argsort(contrib)[::-1][:n_red]
                bot = np.argsort(contrib)[:n_red]
                tot = 0.0
                for j in top:
                    amt = a_pert[j] * 0.05
                    a_pert[j] -= amt
                    tot += amt
                for j in bot:
                    a_pert[j] += tot / len(bot)
        elif strategy == 2:
            # Frequency domain perturbation
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise_real = 0.01 * rng.standard_normal(len(A))
            noise_imag = 0.01 * rng.standard_normal(len(A))
            A_noisy = A + noise_real + 1j * noise_imag
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        elif strategy == 3:
            # Chirp mixing
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 8.0 * rng.random()
            pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 1e-6, 1000.0)
        else:
            # Spectral re-flattening
            a_pert = spectral_flatten_flat(best_a, n_iters=50, lam=0.3)

        # Quick L-BFGS polish
        remaining = deadline - 30 - time.monotonic()
        if remaining < 10:
            break
        a_pert_norm = a_pert / a_pert.sum()
        a_pert_norm = np.maximum(a_pert_norm, 1e-10)
        x_pert = np.log(a_pert_norm)
        maxiter = min(2000, max(100, int(remaining * 0.3 / 0.01)))
        sc, a_opt, _ = run_lbfgs_softmax(x_pert, 2000, maxiter=maxiter, maxfun=max(500, maxiter * 15))
        report(sc, a_opt, f"local_{trial}_s{strategy}")

    # ================================================================
    # Phase 6: Final polish
    # ================================================================
    print("\n--- Phase 6: Final Polish ---")

    # Subgradient
    sc, a_final = subgradient_polish(best_a, n_iters=2000)
    report(sc, a_final, "final_sg")

    # One more L-BFGS
    if time.monotonic() < deadline - 10:
        a_f_norm = a_final / a_final.sum()
        a_f_norm = np.maximum(a_f_norm, 1e-10)
        x_f = np.log(a_f_norm)
        sc, a_opt, _ = run_lbfgs_softmax(x_f, 10000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final_lbfgs")

    # Final subgradient
    sc, a_final2 = subgradient_polish(best_a, n_iters=1000)
    report(sc, a_final2, "final_sg2")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
