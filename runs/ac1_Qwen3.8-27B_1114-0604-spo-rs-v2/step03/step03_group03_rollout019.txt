from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def cd_optimal_1d(base, alpha, beta, outside_max, S_rest, n, t_upper):

    base, alpha, beta: arrays of length L (affected lags)

    f(t) = 2n * max(max_k conv_new[k], outside_max) / (S_rest + t)^2
    """
    L = len(base)
    
    # Candidate t values
    candidates = [0.0, t_upper]

                            t2 = (-b2 - sq) / (2.0 * a2)
                            if 0.0 < t1 < t_upper:
                                candidates.append(t1)
                            if 0.0 < t2 < t_upper:
                                candidates.append(t2)

            # Evaluate candidates using top-k only
            best_t = 0.0
            best_val = np.inf
            top_base = base[top_idx]
            top_alpha = alpha[top_idx]
            top_beta = beta[top_idx]

            for t in candidates:
                t = max(0.0, min(t_upper, t))
                vals_t = top_base + t * top_alpha + (t * t) * top_beta
                inner_max = float(np.max(vals_t))
                tot_max = inner_max if inner_max > outside_max else outside_max
                S_new = S_rest + t
                if S_new < 1e-12:
                    continue
                val = 2.0 * n * tot_max / (S_new * S_new)
                if val < best_val:
                    best_val = val
                    best_t = t

            t_opt = best_t

        # Update
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base + t_opt * alpha + (t_opt * t_opt) * beta


def cd_verify_sweep(a, conv, order, top_k=20):
    """CD sweep with full verification (slower but exact)."""
    n = len(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for i in order:
        i = int(i)
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        L = k_hi - k_lo + 1

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        seg = conv[k_lo:k_hi + 1]
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_idxs = ks - i
        a_vals = a[a_idxs]
        is_2i = (ks == 2 * i)

        alpha = 2.0 * a_vals
        alpha[is_2i] = 0.0
        beta = np.zeros(L)
        beta[is_2i] = 1.0

        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base = seg - old_contrib

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        # Full evaluation of candidates
        current_max = max(float(np.max(base)), outside_max)
        if current_max < 1e-15:
            t_opt = t_upper
        else:
            threshold = current_max * 0.7
            top_mask = base >= threshold
            top_idx = np.where(top_mask)[0]
            if len(top_idx) > top_k:
                vals = base[top_idx]
                ord2 = np.argsort(vals)[::-1][:top_k]
                top_idx = top_idx[ord2]

            candidates = [0.0, t_upper]
            n_top = len(top_idx)

            for k in top_idx:
                A = base[k]
                B = alpha[k]
                C = beta[k]
                D = S_rest
                denom = 2.0 * C - B
                if abs(denom) > 1e-14:
                    t_c = (2.0 * A - D * B) / denom
                    if 0.0 < t_c < t_upper:
                        candidates.append(t_c)

            for ii in range(n_top):
                ki = top_idx[ii]
                for jj in range(ii + 1, n_top):
                    kj = top_idx[jj]
                    a2 = beta[ki] - beta[kj]
                    b2 = alpha[ki] - alpha[kj]
                    c2 = base[ki] - base[kj]
                    if abs(a2) < 1e-14:
                        if abs(b2) > 1e-14:
                            t_int = -c2 / b2
                            if 0.0 < t_int < t_upper:
                                candidates.append(t_int)
                    else:
                        disc = b2 * b2 - 4.0 * a2 * c2
                        if disc >= 0:
                            sq = np.sqrt(disc)
                            t1 = (-b2 + sq) / (2.0 * a2)
                            t2 = (-b2 - sq) / (2.0 * a2)
                            if 0.0 < t1 < t_upper:
                                candidates.append(t1)
                            if 0.0 < t2 < t_upper:
                                candidates.append(t2)

            if outside_max > 1e-15:
                for k in top_idx:
                    a2 = beta[k]
                    b2 = alpha[k]
                    c2 = base[k] - outside_max
                    if abs(a2) < 1e-14:
                        if abs(b2) > 1e-14:
                            t_int = -c2 / b2
                            if 0.0 < t_int < t_upper:
                                candidates.append(t_int)
                    else:
                        disc = b2 * b2 - 4.0 * a2 * c2
                        if disc >= 0:
                            sq = np.sqrt(disc)
                            t1 = (-b2 + sq) / (2.0 * a2)
                            t2 = (-b2 - sq) / (2.0 * a2)
                            if 0.0 < t1 < t_upper:
                                candidates.append(t1)
                            if 0.0 < t2 < t_upper:
                                candidates.append(t2)

            # Full evaluation (exact max over all affected lags)
            best_t = 0.0
            best_val = np.inf

            for t in candidates:
                t = max(0.0, min(t_upper, t))
                conv_t = base + t * alpha + (t * t) * beta
                inner_max = float(np.max(conv_t))
                tot_max = inner_max if inner_max > outside_max else outside_max
                S_new = S_rest + t
                if S_new < 1e-12:
                    continue
                val = 2.0 * n * tot_max / (S_new * S_new)
                if val < best_val:
                    best_val = val
                    best_t = t

            t_opt = best_t

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base + t_opt * alpha + (t_opt * t_opt) * beta


def targeted_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to peak, redistribute to others."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_mask = np.ones(n, dtype=bool)
    bottom_mask[top_idx] = False
    bottom_idx = np.where(bottom_mask)[0]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def gaussian_perturb(a, rng, scale):
    """Gaussian multiplicative perturbation."""
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_pert, 0.0, 1000.0)


def block_perturb(a, rng):
    """Zero a random block and redistribute."""
    n = len(a)
    block_size = max(5, n // 80)
    start = int(rng.integers(0, n - block_size))
    a_pert = a.copy()
    saved = a_pert[start:start + block_size].sum()
    a_pert[start:start + block_size] = 0.0
    rest_mask = np.ones(n, dtype=bool)
    rest_mask[start:start + block_size] = False
    a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
    return np.clip(a_pert, 0.0, 1000.0)


def sinusoidal_perturb(a, rng):
    """Apply sinusoidal modulation."""
    n = len(a)
    freq = 1.0 + 8.0 * rng.random()
    amp = 0.01 + 0.04 * rng.random()
    x = np.arange(n, dtype=np.float64) / n
    phase = 2.0 * np.pi * rng.random()
    a_pert = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    return np.clip(a_pert, 0.0, 1000.0)


def create_structured_init(n, rng):
    """Create diverse structured initializations."""
    x = np.arange(n, dtype=np.float64) / n
    strategy = int(rng.integers(0, 6))
    if strategy == 0:
        a = np.ones(n) * 0.5
    elif strategy == 1:
        a = np.sin(np.pi * x) ** 2
    elif strategy == 2:
        sigma = 0.1 + 0.3 * rng.random()
        center = 0.2 + 0.6 * rng.random()
        a = np.exp(-((x - center) ** 2) / (2 * sigma * sigma))
    elif strategy == 3:
        freq = 1 + int(rng.integers(0, 5))
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + 2 * np.pi * rng.random())
    elif strategy == 4:
        # Two peaks
        c1 = 0.2 + 0.3 * rng.random()
        c2 = 0.5 + 0.3 * rng.random()
        s = 0.05 + 0.1 * rng.random()
        a = np.exp(-((x - c1) ** 2) / (2 * s * s)) + np.exp(-((x - c2) ** 2) / (2 * s * s))
    else:
        # Random smooth
        a = np.abs(rng.standard_normal(n)).cumsum()
        a = a / a.max()
    a += 0.01 * np.abs(rng.standard_normal(n))
    return np.clip(a, 0.0, 1000.0)


def shorten_sequence(a, target_len):
    """Shorten by averaging adjacent pairs."""
    n = len(a)
    if target_len >= n:
        return a.copy()
    # Pad to even length if needed
    if n % 2 != 0:
        a = np.concatenate([a, [a[-1]]])
        n += 1
    # Repeatedly halve until we reach target length
    while n > target_len * 2:
        a = (a[0::2] + a[1::2]) / 2.0
        n = len(a)
    # Final resize
    if len(a) > target_len:
        idx = np.linspace(0, len(a) - 1, target_len).astype(int)
        a = a[idx]
    elif len(a) < target_len:
        a = np.interp(np.linspace(0, 1, target_len), np.linspace(0, 1, len(a)), a)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence using fast analytical CD + multi-restart + LSE.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent, using structured init length 2048.")
        best_a = create_structured_init(2048, rng)
        best_score = compute_score(best_a)
        print(f"Initial score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: Fast CD on parent (50% of time)
    # ============================================================
    phase1_deadline = t_start + 0.50 * total_budget
    print(f"\n=== Phase 1: Fast CD ({phase1_deadline - t_start:.0f}s budget) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    prev_score = compute_score(a_cur)
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < phase1_deadline:
        sweep_count += 1

        # Choose order: alternate between random and hot
        if sweep_count % 5 == 0:
            # Hot ordering: prioritize coordinates near peak
            k_star = int(np.argmax(conv_cur))
            contrib = np.zeros(n_work)
            j_start = max(0, k_star - n_work + 1)
            j_end = min(k_star, n_work - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a_cur[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        else:
            order = rng.permutation(n_work)

        # Use fast sweep
        cd_fast_sweep(a_cur, conv_cur, order, top_k=12)

        # Every 30 sweeps, do a verify sweep for accuracy
        if sweep_count % 30 == 0:
            cd_verify_sweep(a_cur, conv_cur, order, top_k=20)

        sc = compute_score(a_cur)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 100 == 0:
                print(f"  Sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 100 == 0:
            print(f"  Sweep {sweep_count}: {sc:.8f} (no_improve={no_improve})")

        # Perturb when stuck
        if no_improve >= 80:
            strat = int(rng.integers(0, 4))
            if strat == 0:
                a_cur = targeted_perturb(a_cur, rng, strength=0.02 + 0.03 * rng.random())
            elif strat == 1:
                a_cur = gaussian_perturb(a_cur, rng, 0.02 + 0.03 * rng.random())
            elif strat == 2:
                a_cur = block_perturb(a_cur, rng)
            else:
                a_cur = sinusoidal_perturb(a_cur, rng)
            conv_cur = fast_conv(a_cur)
            no_improve = 0
            print(f"  Perturbed (strategy {strat}) at sweep {sweep_count}")

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: Multi-restart (20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.72 * total_budget
    print(f"\n=== Phase 2: Multi-Restart ({phase2_deadline - t_start:.0f}s budget) ===")

    for trial in range(100):
        if time.monotonic() > phase2_deadline - 5.0:
            break

        t_trial = time.monotonic()
        n_cur = len(best_a)

        # Create perturbation
        strat = trial % 7
        if strat == 0:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 1:
            a_pert = gaussian_perturb(best_a, rng, 0.03 + 0.05 * rng.random())
        elif strat == 2:
            a_pert = block_perturb(best_a, rng)
        elif strat == 3:
            a_pert = sinusoidal_perturb(best_a, rng)
        elif strat == 4:
            # Blend with structured init
            a_struct = create_structured_init(n_cur, rng)
            mix = 0.3 + 0.4 * rng.random()
            a_pert = (1 - mix) * best_a + mix * a_struct * (best_a.sum() / a_struct.sum())
        elif strat == 5:
            # Segment swap
            seg_len = max(10, n_cur // 20)
            s1 = int(rng.integers(0, max(1, n_cur - seg_len)))
            s2 = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg_len].copy()
            a_pert[s1:s1 + seg_len] = a_pert[s2:s2 + seg_len]
            a_pert[s2:s2 + seg_len] = tmp
        else:
            # Random block zeroing + fill
            n_blocks = 3 + int(rng.integers(0, 5))
            a_pert = best_a.copy()
            for _ in range(n_blocks):
                bs = max(5, n_cur // 50)
                start = int(rng.integers(0, n_cur - bs))
                saved = a_pert[start:start + bs].sum()
                a_pert[start:start + bs] = 0.0
                rest = a_pert.copy()
                rest_mask = np.ones(n_cur, dtype=bool)
                rest_mask[start:start + bs] = False
                if rest_mask.sum() > 0:
                    a_pert[rest_mask] += saved / rest_mask.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        conv_pert = fast_conv(a_pert)

        # Run CD from perturbation
        n_sweeps = 0
        while time.monotonic() < phase2_deadline - 5.0:
            n_sweeps += 1
            order = rng.permutation(n_cur)
            cd_fast_sweep(a_pert, conv_pert, order, top_k=12)
            if n_sweeps % 20 == 0:
                cd_verify_sweep(a_pert, conv_pert, order, top_k=20)
            if n_sweeps >= 30:
                break

        sc_pert = compute_score(a_pert)
        elapsed = time.monotonic() - t_trial

        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  Trial {trial} (s={strat}, {n_sweeps} sweeps, {elapsed:.1f}s): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  Trial {trial} (s={strat}, {n_sweeps} sweeps, {elapsed:.1f}s): {sc_pert:.8f}")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE refinement (18% of time)
    # ============================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n=== Phase 3: LSE Refinement ({phase3_deadline - t_start:.0f}s budget) ===")

    for beta in [100, 300, 800, 2000, 5000]:
        if time.monotonic() > phase3_deadline - 10.0:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=4000, maxfun=80000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # CD after LSE
    if time.monotonic() < phase3_deadline - 5.0:
        a_cur = best_a.copy()
        conv_cur = fast_conv(a_cur)
        for sw in range(50):
            if time.monotonic() > phase3_deadline - 5.0:
                break
            order = rng.permutation(len(a_cur))
            cd_fast_sweep(a_cur, conv_cur, order, top_k=12)
            if sw % 10 == 0:
                cd_verify_sweep(a_cur, conv_cur, order, top_k=20)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Post-LSE CD: {sc:.8f} (NEW BEST)")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Try different lengths + final polish (10% of time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Length variation + final polish ===")

    # Try shortened version
    if time.monotonic() < phase4_deadline - 30.0:
        for target_len in [4096, 2048, 8192]:
            if time.monotonic() > phase4_deadline - 15.0:
                break
            if target_len >= len(best_a):
                continue
            a_short = shorten_sequence(best_a, target_len)
            a_short = np.clip(a_short, 0.0, 1000.0)
            conv_short = fast_conv(a_short)
            # Run CD
            n_short = len(a_short)
            for sw in range(30):
                if time.monotonic() > phase4_deadline - 15.0:
                    break
                order = rng.permutation(n_short)
                cd_fast_sweep(a_short, conv_short, order, top_k=12)
            sc_short = compute_score(a_short)
            if sc_short < best_score:
                best_score = sc_short
                best_a = a_short.copy()
                print(f"  Shortened to {target_len}: {sc_short:.8f} (NEW BEST)")
            else:
                print(f"  Shortened to {target_len}: {sc_short:.8f}")

    # Final CD polish on best
    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    for sw in range(50):
        if time.monotonic() > deadline - 3.0:
            break
        if sw % 5 == 0:
            # Hot ordering
            k_star = int(np.argmax(conv_cur))
            contrib = np.zeros(len(a_cur))
            j_start = max(0, k_star - len(a_cur) + 1)
            j_end = min(k_star, len(a_cur) - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a_cur[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        else:
            order = rng.permutation(len(a_cur))
        cd_fast_sweep(a_cur, conv_cur, order, top_k=12)
        if sw % 10 == 0:
            cd_verify_sweep(a_cur, conv_cur, order, top_k=20)

    sc_final = compute_score(a_cur)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_cur.copy()
        print(f"  Final CD: {sc_final:.8f} (NEW BEST)")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:

    elapsed_total = time.monotonic() - t_start

    print(f"{'=' * 60}")
    return out
