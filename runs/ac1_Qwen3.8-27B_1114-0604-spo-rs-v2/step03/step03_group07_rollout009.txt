"""
Optimized step function sequence minimizer for 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS with adaptive beta, Nesterov-accelerated subgradient,
and diverse restart strategies. Focuses computational budget on local refinement
of the parent sequence rather than broad global exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: shift
    max_scaled = scaled.max()
    exp_s = np.exp(scaled - max_scaled)
    L = c_max + (max_scaled + np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient: d/d a[k] of c(k') = sum of a[j] where k' = j + k (for each lag k')
    # More precisely: dc(k')/da[k] = a[k'] + a[k'-k] (with bounds checking)
    # Using FFT: conv(a, p)[n-1 : 2n-1] gives the cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Scipy wrapper."""
    val, grad = lse_grad(a, beta)
    return val, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """L-BFGS-B with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def nesterov_subgrad(a, max_iters, sc_lb):
    """Nesterov-accelerated subgradient method for the max-conv objective."""
    n = len(a)

        new_y = np.clip(new_y, 0.0, 1000.0)
        a = (1 - t_k) * a + t_k * new_y

def targeted_reduce(a, n_top=10, strength=0.02):
    """Reduce the top contributors to peak convolution lags."""
    n = len(a)
    c = fast_conv(a)
    # Find top lags
    top_lags = np.argsort(c)[::-1][:n_top]
    a_new = a.copy()
    S = float(a.sum())
    
    for k in top_lags:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je < js:
            continue
        # Contributors at this lag: a[j] * a[k-j] for j in [js, je]
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            jk = k - j
            if 0 <= j < n and 0 <= jk < n:
                contrib[j] += a[j] * a[jk]
        
        # Reduce top contributors, add to bottom
        total = contrib.sum()
        if total < 1e-15:
            continue
        frac = contrib / total
        # Reduce elements with high contribution
        top_idx = np.argsort(contrib)[::-1][:max(3, n // 50)]
        bot_idx = np.argsort(contrib)[:max(3, n // 50)]
        red_amt = strength * total / n_top
        for j in top_idx:
            a_new[j] = max(0, a_new[j] - red_amt * frac[j] * n)
        for j in bot_idx:
            a_new[j] += red_amt * frac[j] * n
    
    return np.clip(a_new, 0.0, 1000.0)

"""
Optimized step function sequence minimizer for 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS with adaptive beta, Nesterov-accelerated subgradient,
and diverse restart strategies. Focuses computational budget on local refinement
of the parent sequence rather than broad global exploration.
Key improvements: exponential/power-law anti-palindromic inits, multi-lag subgradient,
frequency-domain perturbation, and careful time budgeting.
"""
import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = float(scaled.max())
    exp_s = np.exp(scaled - max_scaled)
    L = c_max + (max_scaled + np.log(exp_s.sum())) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via FFT cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    val, grad = lse_grad(a, beta)
    return val, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with non-negative bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_obj, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def nesterov_subgrad(a, max_iters, sc_lb):
    """Nesterov-accelerated subgradient with multi-lag smoothing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Normalize to sum=1 for cleaner subgradient geometry
    S_init = float(a.sum())
    if S_init < 1e-12:
        return best_sc, best_a
    a = a / S_init
    
    y = a.copy()
    t_val = 1.0
    
    for it in range(max_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        if Sy < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (Sy * Sy)
        
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        # Multi-lag subgradient (top 3 lags)
        c_sorted = np.sort(c)[::-1]
        n_lags = min(3, len(c_sorted))
        threshold = c_sorted[n_lags - 1]
        near_max = np.where(c >= threshold * 0.95)[0]
        
        sg = np.zeros(n)
        weights = np.zeros(len(near_max))
        for i, k in enumerate(near_max):
            weights[i] = c[k]
        if weights.sum() < 1e-30:
            break
        weights = weights / weights.sum()
        
        for i, k in enumerate(near_max):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                seg = y[js:je + 1]
                sg[js:je + 1] += weights[i] * 2.0 * seg[::-1]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (np.sqrt(it + 1) * 0.5 + 1.0))
        step = max(step, 1e-10)
        
        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = y - step * grad
        y = np.clip(y, 0.0, 1000.0)
        # Renormalize
        Sy2 = float(y.sum())
        if Sy2 > 1e-12:
            y = y / Sy2
        a = (1.0 - t_val / t_new) * a + (t_val / t_new) * y
        a = np.clip(a, 0.0, 1000.0)
        Sa = float(a.sum())
        if Sa > 1e-12:
            a = a / Sa
        t_val = t_new

def targeted_peak_reduction(a, rng, strength=0.02, n_peak_lags=5):
    """Reduce elements contributing most to peak convolution lags."""
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:n_peak_lags]
    a_new = a.copy()
    
    for k in top_lags:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je < js:
            continue
        # Vectorized contribution computation
        j_arr = np.arange(js, je + 1)
        jk_arr = k - j_arr
        valid = (jk_arr >= 0) & (jk_arr < n)
        j_valid = j_arr[valid]
        contrib_vals = a_new[j_valid] * a_new[jk_arr[valid]]
        total = contrib_vals.sum()
        if total < 1e-15:
            continue
        # Find top and bottom contributors
        top_count = max(2, n // 200)
        order = np.argsort(contrib_vals)
        bottom_idx = j_valid[order[:top_count]]
        top_idx = j_valid[order[::-1][:top_count]]
        red_total = strength * total / n_peak_lags
        per_red = red_total / max(1, len(top_idx))
        for j in top_idx:
            a_new[j] = max(0.0, a_new[j] - per_red)
        per_add = red_total / max(1, len(bottom_idx))
        for j in bottom_idx:
            a_new[j] += per_add
    
    return np.clip(a_new, 0.0, 1000.0)

        inits.append((a, f'exp{rate}'))
    
    # Power-law ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (xn + 0.01) ** p
        a = 0.5 * a / a.max()
        inits.append((a, f'pow{p}'))
    
    # Anti-palindromic sinusoidal

    # Two-block

    # Random with smoothing

    a_smooth = np.convolve(a_rand, kernel, mode='same')

    return inits

    print(f"Base length: {base_n}")
    if parent is not None:
        parent_sc = compute_score(parent)
        print(f"Parent score: {parent_sc:.8f}")
        report(parent_sc, parent, "parent")
    
    # ============================================================
    # Phase 1: LSE Continuation from diverse inits (35%)
    # ============================================================
    phase1_end = t_start + 0.35 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE Continuation (until t={phase1_end-t_start:.0f}s)")
    print(f"{'='*60}")
    
    inits = build_inits(base_n, rng, parent)
    
    # Score and sort initializations
    scored_inits = []
    for a, name in inits:
        sc = compute_score(a)
        scored_inits.append((sc, a, name))
    scored_inits.sort(key=lambda x: x[0])
    
    # Try top initializations with LSE continuation
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    n_inits_to_try = min(10, len(scored_inits))
    
    for trial in range(n_inits_to_try):
        if time.monotonic() > phase1_end:
            break
        sc0, a0, name = scored_inits[trial]
        t0 = time.monotonic()
        print(f"  Init '{name}' (score={sc0:.6f}):")
        
        a_cur = a0.copy()
        for beta in beta_schedule:
            if time.monotonic() > phase1_end - 10:
                break
            remaining = phase1_end - 10 - time.monotonic()
            if remaining < 10:
                break
            est_call_time = 5.0 + 0.003 * base_n
            maxit = max(200, min(3000, int(remaining / est_call_time * 3)))
            sc, a_opt, ok = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
            a_cur = a_opt
            report(sc, a_opt, f"P1 {name} b={beta}")
        
        final_sc = compute_score(a_cur)
        print(f"    Done in {time.monotonic()-t0:.1f}s, final score: {final_sc:.8f}")
    
    print(f"\n  After Phase 1: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 2: Nesterov Subgradient Polish (10%)
    # ============================================================
    phase2_end = t_start + 0.45 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Nesterov Subgradient (until t={phase2_end-t_start:.0f}s)")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    sc_lb = 2.0 * n_best / (2.0 * n_best - 1.0)
    max_iters = int((phase2_end - time.monotonic()) * 80)
    max_iters = max(2000, min(50000, max_iters))
    sc, a_opt = nesterov_subgrad(best_a, max_iters=max_iters, sc_lb=sc_lb)
    report(sc, a_opt, "P2 nesterov")
    
    # Also try Nesterov from the best Phase-1 solution specifically
    if parent is not None and len(parent) == base_n:
        sc_p, a_p = nesterov_subgrad(parent, max_iters=max(1000, max_iters//3), sc_lb=sc_lb)
        report(sc_p, a_p, "P2 nesterov parent")
    
    print(f"\n  After Phase 2: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 3: High-beta LSE Refinement (20%)
    # ============================================================
    phase3_end = t_start + 0.65 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: High-Beta Refinement (until t={phase3_end-t_start:.0f}s)")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    for beta in [3000, 8000, 20000, 50000]:
        if time.monotonic() > phase3_end - 15:
            break
        remaining = phase3_end - 15 - time.monotonic()
        if remaining < 15:
            break
        est_time = 5.0 + 0.003 * n_best
        maxit = max(500, int(remaining / est_time * 4))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, f"P3 beta={beta}")
        # Quick Nesterov between LSE runs
        if sc < best_score and time.monotonic() < phase3_end - 20:
            sc2, a2 = nesterov_subgrad(a_opt, max_iters=2000, sc_lb=sc_lb)
            report(sc2, a2, f"P3 beta={beta} nesterov")
    
    print(f"\n  After Phase 3: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 4: Local Search (25%)
    # ============================================================
    phase4_end = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Local Search (until t={phase4_end-t_start:.0f}s)")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    trial = 0
    sc_lb = 2.0 * n_best / (2.0 * n_best - 1.0)
    
    while time.monotonic() < phase4_end - 20 and trial < 50:
        trial += 1
        strat = trial % 6
        
        if strat == 0:
            # Small Gaussian perturbation
            scale = 0.015 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted peak reduction
            a_pert = targeted_peak_reduction(
                best_a, rng,
                strength=0.01 + 0.02 * rng.random(),
                n_peak_lags=3 + (trial % 4)
            )
        elif strat == 2:
            # Anti-palindromic modulation
            a_pert = best_a.copy()
            center = n_best / 2.0
            width = n_best / (6 + trial % 10)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.04 * sign * np.exp(-d * d))
        elif strat == 3:
            # Frequency-domain high-pass boost (sharpen)
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            # Boost mid-frequencies
            boost = 1.0 + 0.1 * np.exp(-((freqs - len(A)//3) / (len(A)//6)) ** 2)
            A *= boost
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        elif strat == 4:
            # Block-wise rescaling
            a_pert = best_a.copy()
            n_blocks = 8 + (trial % 8)
            block_size = n_best // n_blocks
            for b in range(n_blocks):
                start = b * block_size
                end = min(start + block_size, n_best)
                if end > start:
                    block_mean = a_pert[start:end].mean()
                    if block_mean > 1e-10:
                        target = best_a.mean() * (1.0 + 0.1 * rng.standard_normal())
                        a_pert[start:end] *= target / block_mean
        else:
            # Larger perturbation + quick recovery
            scale = 0.08 + 0.08 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            # Quick LSE to get back on track
            sc_q, a_q, _ = run_lbfgs(a_pert, 100, maxiter=500, maxfun=8000)
            a_pert = a_q
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        remaining = phase4_end - 20 - time.monotonic()
        if remaining < 10:
            break
        est_time = 5.0 + 0.003 * n_best
        maxit = max(500, min(5000, int(remaining / est_time * 3)))
        sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=maxit, maxfun=maxit * 15)
        improved = report(sc, a_opt, f"P4 #{trial} s={strat}")
        
        # Quick subgradient polish if improved
        if improved and time.monotonic() < phase4_end - 30:
            sc2, a2 = nesterov_subgrad(a_opt, max_iters=1000, sc_lb=sc_lb)
            report(sc2, a2, f"P4 #{trial} subgrad")
    
    print(f"\n  After Phase 4: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 5: Final Polish (remaining)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    sc_lb = 2.0 * n_best / (2.0 * n_best - 1.0)
    
    # Very high beta LSE for final precision
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        if remaining > 30:
            maxit = max(1000, int(remaining / 4))
            sc, a_opt, _ = run_lbfgs(best_a, 100000, maxiter=maxit, maxfun=maxit * 20)
            report(sc, a_opt, "P5 high-beta")
    
    # Final Nesterov
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        max_iters = max(2000, int(remaining * 50))
        max_iters = min(max_iters, 100000)
        sc, a_opt = nesterov_subgrad(best_a, max_iters=max_iters, sc_lb=sc_lb)
        report(sc, a_opt, "P5 final nesterov")
    
    # One more targeted reduction + LSE
    if time.monotonic() < deadline - 10:
        a_t = targeted_peak_reduction(best_a, rng, strength=0.005, n_peak_lags=3)
        remaining = deadline - 10 - time.monotonic()
        if remaining > 20:
            maxit = max(500, int(remaining / 5))
            sc, a_opt, _ = run_lbfgs(a_t, 50000, maxiter=maxit, maxfun=maxit * 20)
            report(sc, a_opt, "P5 targeted+lbfgs")
    
    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if len(out) < 1 or sum(out) < 0.01:
        out = [0.5] * 64
        print("WARNING: fallback to default!")
    
    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
