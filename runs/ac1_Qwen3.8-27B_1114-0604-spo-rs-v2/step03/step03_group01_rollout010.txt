This is exactly what I had before. Good.

So the "damage" to c[k] from reducing a[j] is proportional to a[k-j]. And the "benefit" is reducing c[k] by a[j] * a[k-j] for each k in the hot set.

        c_mean = c.mean()
        # w[k] = max(0, c[k] - c_mean) / (c_max - c_mean)
        if c_max > c_mean:
            w = np.maximum(0, c - c_mean) / (c_max - c_mean)
        else:
            w = np.ones(2*n-1) / (2*n-1)

        W_rev = w[::-1]
        M4 = 2 * n

        WR4 = np.fft.rfft(W_rev, M4_fft)
        conv_result = np.fft.irfft(A4 * WR4, M4_fft)
        load = conv_result[n-1:2*n-1][::-1]
        
        load_max = load.max()
        if load_max > 0:
            load = load / load_max
        
        a = a * np.exp(-eta * load)
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 1.0 / n
        else:
            a = a / S_new
        a = np.clip(a, 0, None)
        a = a / a.sum()
        
        eta *= decay
        
        if (t + 1) % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    MWT-soft t={t+1}: score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad_a(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/d(a[j]) = sum_k p[k] * a[k-j] (for valid k-j)
    # Same as the "load" computation
    W_rev = p[::-1]
    M2 = 2 * n
    M2_fft = 1 << (M2 - 1).bit_length()
    A2 = np.fft.rfft(a, M2_fft)
    WR2 = np.fft.rfft(W_rev, M2_fft)
    conv_res = np.fft.irfft(A2 * WR2, M2_fft)
    dL_da = conv_res[n-1:2*n-1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def mwt_optimize(a, n_iter=50000, eta_init=0.05, eta_decay=0.9995,
                 time_limit_s=None, t_start=None, verbose_interval=5000):
    """Multiplicative Weight Update optimization."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    eta = eta_init
    
    for t in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = c.max()
        c_mean = c.mean()
        
        if c_max > c_mean + 1e-15:
            w = np.maximum(0.0, c - c_mean)
            w = w / w.sum()
        else:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        
        # Compute load
        W_rev = w[::-1]
        M2 = 2 * n
        M2_fft = 1 << (M2 - 1).bit_length()
        A2 = np.fft.rfft(a, M2_fft)
        WR2 = np.fft.rfft(W_rev, M2_fft)
        conv_res = np.fft.irfft(A2 * WR2, M2_fft)
        load = conv_res[n-1:2*n-1][::-1]
        
        load_max = load.max()
        if load_max > 1e-15:
            load = load / load_max
        
        a = a * np.exp(-np.minimum(eta * load, 10.0))  # Cap to prevent underflow
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 1.0 / n
        else:
            a = a / S_new
        a = np.clip(a, 0.0, None)
        if a.sum() > 0:
            a = a / a.sum()
        
        eta *= eta_decay
        
        if (t + 1) % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    MWT t={t+1}: score={sc:.8f}, eta={eta:.6f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def adam_lse(a_init, lr=0.003, n_steps=50000, beta_lse=200,
             time_limit_s=None, t_start=None, verbose_interval=10000):
    """Adam with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad_a(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    Adam t={t}: score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
