def spectral_flatten_v2(a, N_fft, n_iters=300, lam_schedule=None):
    """Improved spectral flattening with adaptive lambda."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        # Adaptive lambda: start aggressive, end gentle
        lam = 0.7 * (1.0 - progress) + 0.1
        
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: use mean of non-DC magnitudes (instead of median for better properties)
        non_dc = mag[1:]
        target = float(non_dc.mean())
        
        new_mag = mag.copy()
        # Keep DC and Nyquist fixed
        half = N_fft // 2
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N_fft - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    
    return a

def peak_shave_subgrad(a, n_iters=200, lr_init=0.1):
    """Subgradient method for peak shaving with adaptive learning rate."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        
        # Find all near-peak locations (to avoid waterbed effect)
        threshold = M * 0.95
        peak_locs = np.where(c >= threshold)[0]
        
        # Compute subgradient: average of gradients at near-peak locations
        g = np.zeros(n)
        for k in peak_locs:
            # dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
            j_max = min(k, n - 1)
            j_min = max(0, k - n + 1)
            if j_max >= j_min:
                indices = np.arange(j_min, j_max + 1)
                g[indices] += 2.0 * a[k - indices]
        
        g /= len(peak_locs)
        
        # Also need the gradient of the full score (including the S dependence)
        # score = 2n*M/S^2
        # d(score)/da[j] = 2n * (dM/da[j] / S^2 - 2*M*S_j / S^3)
        # where S_j = dS/da[j] = 1
        S2 = S * S
        grad = (2.0 * n) * (g / S2 - 2.0 * M / (S2 * S))
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        lr = lr_init / (1.0 + 0.01 * it)
        a = a - lr * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S / S_new)
    
    return best_sc, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_and_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def chirp_sequence(n, rng):
    """Generate CAZAC-inspired non-negative sequences."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Quadratic chirp (squared cosine)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    
    # Multiple frequency chirps
    for freq in [1.0, 2.0, 3.0, 5.0]:
        inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    
    # Linear + quadratic mix
    f1 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 
                 0.2 * np.cos(2.0 * np.pi * rng.random() * x / n))
    
    # B-window (raised cosine)
    inits.append(np.sin(np.pi * x / n) ** 2)
    
    # Gaussian-like
    sigma = n * (0.2 + 0.3 * rng.random())
    inits.append(np.exp(-0.5 * ((x - n/2) / sigma) ** 2) + 0.01)
    
    return [np.clip(s, 0.01, 1000.0) for s in inits]

def difference_set_like(n, rng):
    """Generate sequences inspired by difference sets (have flat autocorrelation)."""
    # A perfect difference set of order k has parameters (v, k, lambda)
    # where v = k^2 - k + 1. The indicator sequence has flat autocorrelation.
    # We'll approximate this with non-negative reals.
    
    # For v = k^2 - k + 1, we need k such that v is close to n
    k_best = int(np.sqrt(n + 0.25) + 0.5)
    v = k_best * k_best - k_best + 1
    
    if v <= n:
        # Generate a Singer difference set (simplified)
        # Use a primitive root modulo v (when v is prime power)
        # For simplicity, use a random subset of size k
        a = np.zeros(n)
        positions = rng.choice(n, size=min(k_best, n), replace=False)
        a[positions] = 1.0
        # Smooth it
        kernel_size = max(1, n // 50)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1) / (kernel_size//3 + 1))**2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.clip(a, 0.0, 1000.0)
        return a
    
    return np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0)

"""
Multi-strategy optimizer for minimum peak convolution of non-negative sequences.
Uses alternating spectral flattening, LSE-smoothed L-BFGS with continuation,
multi-peak subgradient shaving, and diverse initializations (chirps, difference sets,
low-pass filtered noise) to minimize 2n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
- Multi-peak subgradient (averages gradients over near-peak locations to avoid waterbed effect)
- Adaptive spectral flattening with mean-based target
- More aggressive beta continuation schedule
- Frequency-domain and block-swap perturbations in local search
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_and_grad(a, beta):
    """Compute LSE-smoothed objective value and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient via cross-correlation of p and a
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(x, beta):
    """Wrapper for scipy minimize: returns (objective, gradient)."""
    return lse_and_grad(x, beta)


def spectral_flatten(a, N_fft, n_iters=300, lam=0.5):
    """Iterative spectral flattening: alternate between flat-spectrum and non-negativity."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S0 = float(a.sum())
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        cur_lam = 0.7 * (1.0 - progress) + 0.15
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        target = float(non_dc.mean())
        new_mag = mag.copy()
        half = N_fft // 2
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - cur_lam) * mag[k] + cur_lam * target
            new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def spectral_flatten_fast(a, N_fft, n_iters=100, lam=0.5):
    """Vectorized spectral flattening (faster for short runs)."""
    a = a.astype(np.float64).copy()
    n = len(a)
    S0 = float(a.sum())
    half = N_fft // 2
    keep_mask = np.zeros(N_fft, dtype=bool)
    keep_mask[0] = True
    if N_fft % 2 == 0:
        keep_mask[half] = True
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        cur_lam = 0.7 * (1.0 - progress) + 0.15
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = A / np.where(mag > 0, mag, 1.0)
        target = float(mag[1:].mean())
        new_mag = np.where(keep_mask, mag, (1.0 - cur_lam) * mag + cur_lam * target)
        new_mag[N_fft - np.arange(1, half)] = new_mag[np.arange(1, half)]
        new_A = new_mag * phase
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def peak_shave_subgrad(a, n_iters=300, lr_init=0.1):
    """Multi-peak subgradient method for peak shaving."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        # Find near-peak locations
        threshold = M * 0.9
        peak_locs = np.where(c >= threshold)[0]
        # Compute averaged subgradient over near-peaks
        g = np.zeros(n)
        for k in peak_locs:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                g[indices] += 2.0 * a[k - indices]
        g /= float(len(peak_locs))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        lr = lr_init / (1.0 + 0.005 * it)
        a = a - lr * grad
        a = np.clip(a, 0.0, 1000.0)
        S_new = float(a.sum())
        if S_new > 1e-12:
            a = a * (S / S_new)
    return best_sc, best_a


def multiplicative_peak_reduction(a, n_iters=200):
    """Multiplicative update that naturally preserves non-negativity.
    Each step: a[j] *= exp(-eta * grad[j]) which is equivalent to
    gradient descent on log(a) with the constraint a > 0."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        threshold = M * 0.9
        peak_locs = np.where(c >= threshold)[0]
        g = np.zeros(n)
        for k in peak_locs:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                g[indices] += 2.0 * a[k - indices]
        g /= float(len(peak_locs))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * M / S3)
        eta = 0.1 / (1.0 + 0.01 * it)
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, 1000.0)
        S_new = float(a.sum())
        if S_new > 1e-12:
            a = a * (S / S_new)
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def generate_chirps(n, rng):
    """Generate CAZAC-inspired non-negative sequences."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    inits.append(0.5 + 0.4 * np.cos(np.pi * 2.0 * x * x / (2.0 * n)))
    inits.append(0.5 + 0.3 * np.cos(np.pi * 3.0 * x * x / (2.0 * n)))
    f1 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) +
                 0.2 * np.cos(2.0 * np.pi * rng.random() * x / n))
    inits.append(np.sin(np.pi * x / n) ** 2 + 0.01)
    sigma = n * (0.2 + 0.3 * rng.random())
    inits.append(np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2) + 0.01)
    # Sine blend
    f2 = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * f2 * x / n) *
                 np.cos(2.0 * np.pi * (f2 + 1) * x / n))
    return [np.clip(s, 0.01, 1000.0) for s in inits]


def generate_diffset(n, rng):
    """Generate difference-set-inspired sequences with approximately flat autocorrelation."""
    k_best = int(np.sqrt(n + 0.25) + 0.5)
    a = np.zeros(n)
    positions = rng.choice(n, size=min(k_best, n), replace=False)
    a[positions] = 1.0
    kernel_size = max(3, n // 40)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel_x = np.arange(kernel_size) - kernel_size // 2
    sigma_k = max(1.0, kernel_size / 6.0)
    kernel = np.exp(-0.5 * (kernel_x / sigma_k) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    return np.clip(a, 0.0, 1000.0)


def lowpass_random(n, rng, N_fft, cutoff_frac=0.25):
    """Generate a random sequence passed through a low-pass filter."""
    a = rng.random(n) + 0.1
    A = np.fft.rfft(a, N_fft)
    cutoff = int(N_fft * cutoff_frac)
    A[cutoff:] = 0.0
    a_lp = np.fft.irfft(A, N_fft)[:n]
    return np.clip(a_lp, 0.01, 10.0)


def update_best(best_score, best_a, sc, a, label, t_start):

        print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.1f}s)")
        return sc, a.copy(), True
    return best_score, best_a, False

        else:
            # Quick spectral flattening
            a_pert = spectral_flatten_fast(best_a, N_fft_best, n_iters=15, lam=0.3)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            continue

        # L-BFGS polish
        remaining = phase4_deadline - time.monotonic()
        if remaining < 8:
            break
        maxiter = min(2000, int(remaining / 0.06))
        sc, a_opt, _ = run_lbfgs(a_pert, 2000, maxiter=max(100, maxiter), maxfun=max(500, maxiter * 15))
        best_score, best_a, _ = update_best(best_score, best_a, sc, a_opt, f"local#{trial}s{strategy}", t_start)

    # ================================================================
    # Phase 5: Final polish
    # ================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Subgradient
    sc, a_pol = peak_shave_subgrad(best_a, n_iters=500, lr_init=0.02)
    best_score, best_a, _ = update_best(best_score, best_a, sc, a_pol, "final_subgrad", t_start)

    # Multiplicative
    if time.monotonic() < deadline - 20:
        sc, a_mp = multiplicative_peak_reduction(best_a, n_iters=300)
        best_score, best_a, _ = update_best(best_score, best_a, sc, a_mp, "final_multir", t_start)

    # Final L-BFGS at very high beta
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        maxiter = min(5000, int(remaining * 0.8 / 0.02))
        sc, a_opt, _ = run_lbfgs(best_a, 50000, maxiter=max(500, maxiter), maxfun=max(2000, maxiter * 20))
        best_score, best_a, _ = update_best(best_score, best_a, sc, a_opt, "final_lbfgs", t_start)

    # Final subgradient
    sc, a_pol2 = peak_shave_subgrad(best_a, n_iters=400, lr_init=0.01)
    best_score, best_a, _ = update_best(best_score, best_a, sc, a_pol2, "last_subgrad", t_start)

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
