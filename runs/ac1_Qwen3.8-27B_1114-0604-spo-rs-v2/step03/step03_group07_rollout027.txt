import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def generalized_lse_score_grad(a, beta, top_k=1):
    """
    Generalized LSE-smoothed objective targeting top-k peaks.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta, top_k):
    return generalized_lse_score_grad(a, beta, top_k)

    # Actually, let's just use the full LSE gradient but with modified p

    S2 = S * S
    S3 = S2 * S

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, use_log=False, k_peaks=1):
    """Run L-BFGS-B on smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()

            a = np.clip(a, 0.0, 1000.0)
            sc, ga = generalized_lse_score_grad(a, beta, k_peaks)
            gb = ga * a  # chain rule: d/ db = d/da * da/db = d/da * a
            return sc, gb
    else:
        b0 = a0
        bounds = [(0.0, 1000.0)] * n
        
        def log_obj(b):
            a = np.clip(b, 0.0, 1000.0)
            if k_peaks > 1:

            return sc, ga
    
    t0 = time.monotonic()
    try:
        res = minimize(
            log_obj, b0,
            method='L-BFGS-B', jac=True, bounds=bounds,

    if use_log:
        a_out = np.exp(np.clip(res.x, -30, 20))

        a_out = np.clip(res.x, 0.0, 1000.0)

def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else np.ones(n) * 0.5
    
    if parent is not None and len(parent) >= n:
        # Resample parent to length n
        from scipy.interpolate import interp1d
        x_interp = np.linspace(0, len(parent)-1, n)
        parent_interp = interp1d(np.arange(len(parent)), parent, kind='cubic', fill_value='extrapolate')(x_interp)
        inits.append(np.clip(parent_interp, 0.01, 1000.0))
    elif parent is not None:
        # Pad or truncate
        if len(parent) > n:
            inits.append(parent[:n].copy())
        else:
            inits.append(np.resize(parent, n).copy())
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Anti-palindromic linear ramp
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        inits.append(np.clip(0.5 + alpha * (xn - 0.5), 0.01, 1000.0))
        inits.append(np.clip(0.5 - alpha * (xn - 0.5), 0.01, 1000.0))
    
    # Anti-palindromic quadratic
    for alpha in [0.3, 0.5, 0.7]:
        t = xn - 0.5
        inits.append(np.clip(0.5 + alpha * t * np.abs(t), 0.01, 1000.0))
    
    # Cosine anti-palindromic (odd harmonics)
    for freq in [1, 3, 5]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * xn), 0.01, 1000.0))
    
    # Power-law decay (strong anti-palindromic)
    for p in [0.5, 1.0, 1.5, 2.0]:
        inits.append(np.clip((xn + 0.1) ** p, 0.01, 1000.0))
        inits.append(np.clip((1 - xn + 0.1) ** p, 0.01, 1000.0))
    
    # Exponential decay
    for rate in [1.0, 2.0, 3.0, 5.0]:
        inits.append(np.clip(np.exp(-rate * xn), 0.01, 1000.0))
        inits.append(np.clip(np.exp(-rate * (1 - xn)), 0.01, 1000.0))
    
    # Two-level anti-palindromic
    inits.append(np.concatenate([np.ones(n//2) * 0.3, np.ones(n - n//2) * 0.7]))
    inits.append(np.concatenate([np.ones(n//2) * 0.7, np.ones(n - n//2) * 0.3]))
    
    # Three-level
    third = n // 3
    inits.append(np.concatenate([np.ones(third) * 0.2, np.ones(n - 2*third) * 0.8, np.ones(third) * 0.2]))
    inits.append(np.concatenate([np.ones(third) * 0.8, np.ones(n - 2*third) * 0.3, np.ones(third) * 0.8]))
    
    # Random with anti-palindromic bias
    r = rng.standard_normal(n)
    r_anti = r - r[::-1]  # anti-palindromic perturbation
    inits.append(np.clip(0.5 + 0.2 * r_anti, 0.01, 1000.0))
    
    # Smoothed random
    r2 = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    ksize = max(1, n // 50)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(r2, kernel, mode='same'), 0.01, 1000.0))
    
    # "Optimal-looking" profile: superposition of anti-palindromic modes
    a = 0.5 * np.ones(n)
    a += 0.25 * (xn - 0.5)
    a += 0.1 * np.sin(3 * np.pi * xn)
    a += 0.05 * np.sin(7 * np.pi * xn)
    inits.append(np.clip(a, 0.01, 1000.0))
    
    return inits


def peak_balancing_search(a, rng, n_iters=10000, time_limit=None):
    """
    Local search that targets the top peaks in the convolution.
    Uses coordinate-style moves: pick two positions, transfer mass.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        if it % 500 == 0 and it > 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            
            # Occasionally try larger perturbations
            if it % 5000 == 0:
                a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
                a = np.clip(a, 0.0, 1000.0)
                continue
        
        # Compute convolution and find peaks
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Find contributors to this peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_end >= j_start:
            # contrib[j] = a[j] * a[k_star - j]
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k_star - js]
            
            # Try to reduce the largest contributor
            # Pick the index with largest contribution
            max_idx = js[np.argmax(contrib)]
            
            # Find a "cold" index: one that doesn't contribute much to this peak
            # and is at a position that's low at this lag
            noise = contrib + rng.standard_normal(j_end - j_start + 1) * 0.01

            # Transfer mass from hot to cold
            transfer = a[max_idx] * (0.01 + 0.05 * rng.random())
            a[max_idx] -= transfer
            a[cold_idx] += transfer
            
        a = np.clip(a, 0.0, 1000.0)
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_sc, best_a


def subgrad_polish(a, max_iters=5000, time_limit=None):
    """Exact subgradient descent with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t0 = time.monotonic()
    step_scale = 0.1
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Polyak-like step size
        lb = 2.0 * n / (2.0 * n - 1.0)
        step = (sc - lb) / (float(grad @ grad) + 1e-30)
        step = min(step, step_scale / (1.0 + 0.01 * it))
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Decay step scale
        if it % 1000 == 0:
            step_scale *= 0.5
    
    return best_sc, best_a


def coarse_to_fine(a_small, n_target, rng):
    """
    Take an optimized small sequence and create a good initialization
    for a larger sequence by interpolating the profile.
    """
    n_small = len(a_small)
    if n_small == n_target:
        return a_small.copy()
    
    x_small = np.linspace(0, 1, n_small)
    x_target = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_target, x_small, a_small)
    
    # Add small anti-palindromic perturbation
    pert = rng.standard_normal(n_target)
    pert = pert - pert[::-1]
    a_interp += 0.02 * pert
    
    return np.clip(a_interp, 0.01, 1000.0)

"""
Optimized step-function sequence minimizer.
Strategy:
1. Log-parametrization (a=exp(b)) for automatic non-negativity in L-BFGS
2. LSE-continuation with aggressive beta schedule on multiple diverse inits
3. Coarse-to-fine: optimize small sequences, interpolate profile to target length
4. Multi-peak weighted LSE for balanced peak reduction
5. Peak-targeted coordinate-descent local search
6. Subgradient polish with Polyak step sizes
Target: 2n*max(conv(a,a))/(sum(a))^2 < 1.5027

import numpy as np

from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def multi_lse_score_grad(a, beta, k_peaks):
    """Multi-peak LSE: targets top-k peaks with equal weight via LSE."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    return compute_score(a_out), a_out, res.success


def run_lbfgs_log(a0, beta, maxiter=5000, k_peaks=1):
    """L-BFGS-B in log-space (a=exp(b)), automatically non-negative."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    b0 = np.log(a0).astype(np.float64)
    bounds = [(-25.0, 13.0)] * n
    try:
        if k_peaks > 1:
            res = minimize(objective_log_multi, b0, args=(beta, k_peaks),
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'ftol': 1e-18, 'gtol': 1e-15,
                                    'maxiter': maxiter, 'maxfun': maxiter * 25, 'maxls': 50})
        else:
            res = minimize(objective_log, b0, args=(beta,),
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'ftol': 1e-18, 'gtol': 1e-15,
                                    'maxiter': maxiter, 'maxfun': maxiter * 25, 'maxls': 50})
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.exp(np.clip(res.x, -30.0, 13.0))
    a_out = np.clip(a_out, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def build_inits(n, rng, parent=None):
    """Generate diverse initializations emphasizing anti-palindromic structure."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else np.ones(n) * 0.5

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        elif len(parent) > n:
            x_interp = np.linspace(0, len(parent) - 1, n)
            from numpy import interp
            inits.append(np.clip(interp(x_interp, np.arange(len(parent)), parent), 0.01, 1000.0))
        else:
            x_interp = np.linspace(0, n - 1, len(parent))
            interp_vals = interp(x_interp, np.arange(n), parent)
            inits.append(np.clip(interp_vals, 0.01, 1000.0))

    inits.append(np.ones(n) * 0.5)

    # Anti-palindromic ramps
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        inits.append(np.clip(0.5 + alpha * (xn - 0.5), 0.01, 1000.0))
        inits.append(np.clip(0.5 - alpha * (xn - 0.5), 0.01, 1000.0))

    # Quadratic anti-palindromic
    t = xn - 0.5
    for alpha in [0.3, 0.5, 0.8]:
        inits.append(np.clip(0.5 + alpha * t * np.abs(t), 0.01, 1000.0))

    # Sinusoidal anti-palindromic
    for freq in [1, 3, 5, 7]:
        inits.append(np.clip(0.5 + 0.35 * np.sin(np.pi * freq * xn), 0.01, 1000.0))

    # Power-law
    for p in [0.5, 1.0, 1.5, 2.0]:
        inits.append(np.clip((xn + 0.05) ** p, 0.01, 1000.0))
        inits.append(np.clip((1.0 - xn + 0.05) ** p, 0.01, 1000.0))

    # Exponential
    for rate in [1.0, 2.0, 4.0]:
        inits.append(np.clip(np.exp(-rate * xn) + 0.1, 0.01, 1000.0))
        inits.append(np.clip(np.exp(-rate * (1.0 - xn)) + 0.1, 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.2, np.ones(n - half) * 0.8]))
    inits.append(np.concatenate([np.ones(half) * 0.8, np.ones(n - half) * 0.2]))

    # Smoothed random with anti-palindromic bias
    r = rng.standard_normal(n)
    r_anti = r - r[::-1]
    inits.append(np.clip(0.5 + 0.25 * r_anti, 0.01, 1000.0))

    # Combined profile
    a = 0.5 + 0.3 * (xn - 0.5) + 0.1 * np.sin(3 * np.pi * xn) + 0.05 * np.sin(7 * np.pi * xn)
    inits.append(np.clip(a, 0.01, 1000.0))

    return inits


def peak_balance_search(a, rng, n_iters, time_limit=None):
    """Coordinate-descent style local search targeting convolution peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue

        js = np.arange(j_start, j_end + 1)
        contrib = a[js] * a[k_star - js]

        # Try multiple transfer candidates
        n_tries = 8
        for _ in range(n_tries):
            a_try = a.copy()
            # Pick hot index (high contribution)
            hot_weights = contrib + 1e-10
            hot_weights /= hot_weights.sum()
            hot_idx_local = rng.choice(len(js), p=hot_weights)
            hot_idx = js[hot_idx_local]

            # Pick cold index (low contribution)
            cold_weights = 1.0 / (contrib + 0.1)
            cold_weights /= cold_weights.sum()
            cold_idx_local = rng.choice(len(js), p=cold_weights)
            cold_idx = js[cold_idx_local]

            if hot_idx == cold_idx:
                continue

            # Transfer mass
            frac = 0.01 + 0.08 * rng.random()
            transfer = a_try[hot_idx] * frac
            a_try[hot_idx] -= transfer
            a_try[cold_idx] += transfer
            a_try = np.clip(a_try, 0.0, 1000.0)

            # Accept if better (stochastic acceptance occasionally)
            sc_try = compute_score(a_try)
            if sc_try < sc or (rng.random() < 0.05 and sc_try < sc * 1.001):
                a = a_try
                sc = sc_try
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

    final_sc = compute_score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    return best_sc, best_a


def subgrad_polish(a, max_iters=5000, time_limit=None):
    """Subgradient descent with Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - lb) / (gn2 + 1e-30)
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def coarse_to_fine(a_small, n_target, rng):
    """Interpolate a small optimized sequence to target length."""
    if len(a_small) == n_target:
        return a_small.copy()
    x_small = np.linspace(0, 1, len(a_small))
    x_target = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_target, x_small, a_small)
    pert = rng.standard_normal(n_target)
    pert = pert - pert[::-1]
    a_interp += 0.02 * pert
    return np.clip(a_interp, 0.01, 1000.0)


def run_continuation(a_start, beta_schedule, time_limit, use_log_from_beta=2000):
    """Run LSE continuation from a starting point. Returns (best_score, best_a)."""
    best_sc = compute_score(a_start)
    best_a = a_start.copy()
    t0 = time.monotonic()
    a_cur = a_start.copy()

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        n = len(a_cur)
        maxit = max(500, min(10000, int(remaining * 4)))

        if beta >= use_log_from_beta:
            sc, a_opt, success = run_lbfgs_log(a_cur, beta, maxiter=maxit)
        else:
            sc, a_opt, success = run_lbfgs_direct(a_cur, beta, maxiter=maxit)

        a_cur = a_opt
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Multi-phase:
    Phase 1: Coarse-to-fine (small n optimization)
    Phase 2: Full LSE continuation on target lengths
    Phase 3: Log-param refinement with high beta
    Phase 4: Peak-balancing local search
    Phase 5: Subgradient + final local search
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0

    best_score = np.inf
    best_a = None

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.01, 1000.0)
                print(f"Parent loaded: n={p.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    # Phase time allocation
    p1_end = t_start + 0.08 * total_budget
    p2_end = t_start + 0.55 * total_budget
    p3_end = t_start + 0.75 * total_budget
    p4_end = t_start + 0.88 * total_budget

    # ============================================================
    # Phase 1: Coarse-to-fine
    # ============================================================
    print(f"\n=== Phase 1: Coarse-to-Fine ===")
    scales = [256, 512, 1024, 2048, 4096]
    best_small = None

    for n_scale in scales:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n_scale} (t={time.monotonic()-t_start:.0f}s)")
        inits = build_inits(n_scale, rng, best_small)
        # Score all inits and pick top 3
        scored = sorted([(compute_score(a), a) for a in inits], key=lambda x: x[0])
        best_here_sc = np.inf
        best_here_a = None

        for sc_init, a_init in scored[:3]:
            if time.monotonic() > p1_end:
                break
            sc, a_opt = run_continuation(a_init, [200, 1000, 5000],
                                         time_limit=(p1_end - time.monotonic()) * 0.3,
                                         use_log_from_beta=3000)
            if sc < best_here_sc:
                best_here_sc = sc
                best_here_a = a_opt

        if best_here_sc < np.inf:
            best_small = best_here_a
            if best_here_sc < best_score:
                best_score = best_here_sc
                best_a = best_here_a.copy()
                print(f"    Best at n={n_scale}: {best_score:.8f}")

    # Phase 1 result
    if parent is not None:
        sc_p = compute_score(parent)
        if sc_p < best_score:
            best_score = sc_p
            best_a = parent.copy()
            print(f"  Parent is best: {best_score:.8f}")

    print(f"  Phase 1 done: best={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 2: Full LSE continuation on target lengths
    # ============================================================
    print(f"\n=== Phase 2: Full LSE Continuation ===")

    target_lengths = []
    if parent is not None:
        target_lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 12288]:
        if n_try not in target_lengths:
            target_lengths.append(n_try)
    target_lengths = sorted(set(target_lengths))[:3]
    print(f"  Target lengths: {target_lengths}")

    beta_schedule = [50, 200, 800, 3000, 10000, 30000, 80000]

    for n_target in target_lengths:
        if time.monotonic() > p2_end:
            break
        print(f"\n  --- n={n_target} ---")

        # Build initialization pool
        init_pool = []
        if parent is not None:
            init_pool.append(coarse_to_fine(parent, n_target, rng))
        if best_small is not None:
            init_pool.append(coarse_to_fine(best_small, n_target, rng))

        all_inits = build_inits(n_target, rng, parent)
        init_pool.extend(all_inits[:8])

        # Score and sort
        scored_inits = []
        for a_init in init_pool:
            if a_init is None:
                continue
            sc = compute_score(a_init)
            if sc < np.inf:
                scored_inits.append((sc, a_init))
        scored_inits.sort(key=lambda x: x[0])

        # Run continuation on top initializations
        n_to_try = min(len(scored_inits), 4)
        time_per = (p2_end - time.monotonic()) / max(1, n_to_try)

        for i in range(n_to_try):
            if time.monotonic() > p2_end:
                break
            a_init = scored_inits[i][1]
            print(f"    Init {i}: sc={scored_inits[i][0]:.6f} (t={time.monotonic()-t_start:.0f}s)")
            sc, a_opt = run_continuation(a_init, beta_schedule,
                                         time_limit=time_per,
                                         use_log_from_beta=2000)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"    NEW BEST: {sc:.8f} (init={i}, t={time.monotonic()-t_start:.0f}s)")
            else:
                print(f"    No improvement: {sc:.8f} vs best {best_score:.8f}")

    print(f"  Phase 2 done: best={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 3: Log-param refinement with very high beta
    # ============================================================
    print(f"\n=== Phase 3: High-Beta Log-Param Refinement ===")
    n_best = len(best_a)

    for beta in [50000, 100000, 300000, 1000000]:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 30:
            break
        maxit = max(1000, int(remaining * 5))
        sc, a_opt, success = run_lbfgs_log(best_a, beta, maxiter=maxit)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  beta={beta}: NEW BEST {sc:.8f}")
        else:
            print(f"  beta={beta}: {sc:.8f} (no improvement)")

    # Also try multi-peak with moderate beta
    for k_peaks in [3, 5]:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 30:
            break
        maxit = max(1000, int(remaining * 3))
        sc, a_opt, success = run_lbfgs_log(best_a, 20000, maxiter=maxit, k_peaks=k_peaks)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  multi k={k_peaks}: NEW BEST {sc:.8f}")

    print(f"  Phase 3 done: best={best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Phase 4: Peak-balancing local search
    # ============================================================
    print(f"\n=== Phase 4: Peak-Balance Local Search ===")
    if time.monotonic() < p4_end:
        t_left = p4_end - time.monotonic()
        n_iters = int(t_left * 300)
        sc, a_opt = peak_balance_search(best_a, rng, n_iters=n_iters, time_limit=t_left)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Peak balance: NEW BEST {sc:.8f}")
        else:
            print(f"  Peak balance: {sc:.8f} (best stays {best_score:.8f}")
    else:
        print(f"  Skipped (no time)")

    # ============================================================
    # Phase 5: Subgradient + final targeted search
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 15:
        t_left = deadline - 15 - time.monotonic()
        sc, a_opt = subgrad_polish(best_a, max_iters=int(t_left * 100), time_limit=t_left * 0.5)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Subgrad: NEW BEST {sc:.8f}")

    # Targeted local search attempts
    trial = 0
    while time.monotonic() < deadline - 5 and trial < 8:
        trial += 1
        if trial % 3 == 1:
            noise = rng.standard_normal(n_best)
            noise = noise - noise[::-1]
            a_pert = best_a * (1.0 + 0.02 * noise)
        elif trial % 3 == 2:
            # Frequency smoothing
            M_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.rfft(best_a, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.1 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Peak-targeted
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                contrib = a_pert[js] * a_pert[k_star - js]
                p90 = np.percentile(contrib, 90)
                p20 = np.percentile(contrib, 20)
                a_pert[js[contrib > p90]] *= 0.92
                a_pert[js[contrib < p20]] *= 1.08
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            continue

        sc, a_opt, _ = run_lbfgs_log(a_pert, 50000, maxiter=3000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  P5 trial {trial}: NEW BEST {sc:.8f}")

    # Final subgradient
    if time.monotonic() < deadline - 2:
        t_left = max(5, deadline - 2 - time.monotonic())
        sc, a_opt = subgrad_polish(best_a, max_iters=5000, time_limit=t_left)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final subgrad: NEW BEST {sc:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"First 10: {[round(x, 4) for x in out[:10]]}")
    print(f"Last 10:  {[round(x, 4) for x in out[-10:]]}")
    print(f"{'=' * 60}")
    return out
