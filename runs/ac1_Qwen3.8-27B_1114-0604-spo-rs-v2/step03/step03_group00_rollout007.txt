Strategy: Arcsine-inspired init + FFT-based projected gradient + LSE L-BFGS + polish.

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def variance_grad_descent(a, n_iters=1000, step_init=0.1):
    """Projected gradient descent on variance of convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_var = np.inf
    best_sc = compute_score(a)
    
    M = 2*n - 1
    M_fft = 1 << (M - 1).bit_length()
    target = 1.0 / M  # target flat value
    
    # Precompute FFT of a
    A = np.fft.rfft(a, M_fft)
    
    for it in range(n_iters):
        # Compute convolution
        conv = np.fft.irfft(A * A, M_fft)[:M]
        
        # Compute variance (proxy for max)
        c_mean = conv.mean()
        c_var = float(np.mean((conv - c_mean)**2))
        if c_var < best_var:
            best_var = c_var
            best_sc = compute_score(a)
            best_a = a.copy()
        
        # Gradient of ||conv - target||^2 w.r.t. a
        # = 2 * cross_correlation(conv - target, a)
        err = conv - target
        # cross_correlation: sum_k err_k * a_{k-j} for j = 0,...,n-1
        # = ifft(FFT(err) * conj(FFT(a)))[:n]
        E = np.fft.rfft(err, M_fft)  # err has length M = 2n-1, pad to M_fft
        cross = np.fft.irfft(E * np.conj(A), M_fft)[:n]
        grad = 2.0 * cross
        
        # Also need to account for the normalization constraint (sum a = 1)
        # The gradient on the simplex: project grad to be orthogonal to the all-ones vector
        grad_mean = grad.mean()
        grad_central = grad - grad_mean  # this handles the normalization
        
        # Adaptive step size
        step = step_init / (1.0 + it * 0.1)
        # Backtracking
        for _ in range(20):
            a_new = a - step * grad_central
            a_new = np.maximum(a_new, 0.0)
            S_new = a_new.sum()
            if S_new > 1e-12:
                a_new = a_new / S_new
                # Check if variance decreased
                A_new = np.fft.rfft(a_new, M_fft)
                conv_new = np.fft.irfft(A_new * A_new, M_fft)[:M]
                var_new = float(np.mean((conv_new - conv_new.mean())**2))
                if var_new < c_var:
                    break
            step *= 0.5
        else:
            break
        
        a = a_new
        A = np.fft.rfft(a, M_fft)
    
    return best_a

def arcsine_init(n):
    """Generate arcsine-distributed initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((j + 1) * (n - j))
    a /= a.sum()
    return a

def u_shape_init(n, power=2.0):
    """Generate U-shaped initialization."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0  # in [-1, 1]
    a = 1.0 + power * np.abs(x) ** 2
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def v_shape_init(n, power=1.0):
    """Generate V-shaped initialization."""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / (n - 1) - 1.0
    a = 0.1 + power * np.abs(x) ** power
    a /= a.sum()
