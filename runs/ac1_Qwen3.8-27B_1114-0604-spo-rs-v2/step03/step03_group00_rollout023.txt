import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv2(a, b):
    """Linear convolution of a and b via real FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient via FFT: 2 * (rev_a * p) aligned properly
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_rp = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_rp[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(f, M, max_active=150, max_rounds=8):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for active lags k, g >= 0.
    Uses constraint generation for efficiency.
    """
    n = len(f)
    conv_ff = fast_conv(f)

    # Initialize active set with top peaks
    top_k = min(40, len(conv_ff))
    active_set = set(np.argsort(conv_ff)[::-1][:top_k].tolist())

    g = None
    for rnd in range(max_rounds):
        active_list = sorted(active_set)
        m = len(active_list)
        if m == 0:
            break

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_list):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(f[k - j])

        if len(vals) == 0:
            break

        A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        bounds_lp = [(0, None)] * n

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp,
                      method='highsoptions' if False else 'highs')

        if not res.success:
            break

        g = res.x
        S_g = g.sum()
        if S_g < 1e-10:
            break

        # Check for violated constraints
        conv_fg = fast_conv2(f, g)
        tol = M * 1e-8 + 1e-15
        violated = np.where(conv_fg > M + tol)[0]

        if len(violated) == 0:
            break

        # Add most violated constraints (up to 30 new)
        worst = violated[np.argsort(conv_fg[violated])[::-1][:30]]
        for lag in worst:
            active_set.add(int(lag))
        # Cap active set size
        if len(active_set) > max_active:
            # Keep the most violated ones
            conv_vals = {k: conv_fg[k] for k in active_set}
            sorted_lags = sorted(conv_vals.keys(), key=lambda x: conv_vals[x], reverse=True)
            active_set = set(sorted_lags[:max_active])

    return g


def lp_iteration(f, max_active=150, max_rounds=8, ts=None):
    """One full LP iteration: find direction, line search, return updated f."""
    n = len(f)
    S = float(f.sum())
    if S < 1e-12:
        return f, compute_score(f)
    f_norm = f / S
    conv_ff = fast_conv(f_norm)
    M = float(conv_ff.max())

    g = lp_direction(f_norm, M, max_active, max_rounds)
    if g is None or g.sum() < 1e-10:
        return f, compute_score(f)

    # Normalize g to same scale as f
    g = g / g.sum() * S

    if ts is None:
        ts = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]

    best_f = f.copy()
    best_sc = compute_score(f)

    for t in ts:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        S_new = f_new.sum()
        if S_new < 1e-12:
            continue
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()

    return best_f, best_sc


def multiplicative_updates(f, n_iters=500, eta0=0.1, decay=0.999):
    """
    Mirror descent with KL divergence.
    a_new[j] = a[j] * exp(-eta * subgrad[j]) / normalization
    """
    n = len(f)
    a = f.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return f
    a = a / S  # normalize to sum=1

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M

        if sc < best_sc * 0.999:  # relative improvement threshold for tracking
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max conv (at the peak lag)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        subgrad = np.zeros(n)
        seg = a[j_start:j_end + 1]
        subgrad[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Mirror descent update
        # a_new = a * exp(-eta * (subgrad/S - 2*M/S^2)) but S=1 after normalization
        # Simplified: a_new = a * exp(-eta * (subgrad - 2*M))
        # Scale eta adaptively
        sg_range = subgrad.max() - subgrad.min()
        if sg_range > 1e-12:
            log_updates = -eta * (subgrad - subgrad.mean()) / sg_range
        else:
            break

        a_new = a * np.exp(log_updates)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        eta *= decay

    return best_a * (f.sum() / best_a.sum()) if best_a.sum() > 1e-12 else f


def soft_multiplicative_updates(f, n_iters=300, beta=50.0, eta0=0.5, decay=0.998):
    """
    Soft version: use softmax-weighted subgradient (averages over near-peak lags).
    """
    n = len(f)
    a = f.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return f
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        # Softmax weights over conv values
        scaled = beta * (c - c_max)
        p = np.exp(scaled) / np.exp(scaled).sum()

        # Weighted subgradient
        subgrad = np.zeros(n)
        for k in range(len(c)):
            if p[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                subgrad[j_start:j_end + 1] += p[k] * 2.0 * seg[::-1]

        L = c_max + np.log(np.exp(scaled).sum()) / beta

        # Update
        sg_centered = subgrad - subgrad.mean()
        sg_range = sg_centered.max() - sg_centered.min()
        if sg_range < 1e-12:
            break

        log_updates = -eta * sg_centered / sg_range
        a_new = a * np.exp(log_updates)
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        eta *= decay

    return best_a * (f.sum() / best_a.sum()) if best_a.sum() > 1e-12 else f


def coordinate_descent(f, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak-contributing elements with quadratic refinement."""
    n = len(f)
    a = f.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return f
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old = a[j]
            if a_old < 1e-15:
                continue
            # Try geometric sequence of values
            candidates = [a_old * r for r in [0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.2, 1.5]]
            candidates.append(0.0)
            best_val = a_old
            best_local = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old) < 1e-12:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
            a[j] = best_val

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    # U-shaped
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    # Inverted U
    inits.append(0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 2))
    # Cosine
    inits.append(0.3 + 0.5 * np.cos(2.0 * np.pi * x) ** 2)
    # Double hump
    inits.append(0.1 + 0.8 * np.cos(4.0 * np.pi * x) ** 2 * np.cos(2.0 * np.pi * x) ** 2 + 0.3 * (2*x-1)**2)
    # Block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.1 + 0.9 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].clip(0.01, 10.0))
    # Random smooth
    freqs = rng.choice([1, 2, 3, 4, 6, 8], size=3, replace=False)
    phases = 2 * np.pi * rng.random(3)
    amps = rng.random(3) * 0.5
    sig = 0.4
    for fr, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * fr * x + ph)
    inits.append(np.clip(sig, 0.01, 10.0))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 20.0

    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    n_target = len(parent) if parent is not None else 6962
    # Round to power of 2 for FFT efficiency
    n_pow2 = 1 << (n_target - 1).bit_length()
    if n_pow2 > 16384:
        n_pow2 = 16384
    if n_pow2 < 1024:
        n_pow2 = 1024

    lengths = [n_target if parent is not None else n_pow2]
    # Add a couple alternatives
    if n_target != n_pow2:
        lengths.append(n_pow2)
    lengths = sorted(set(lengths))[:3]

    print(f"Target length: {n_target}, Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # ==================================================================
    # Phase 1: Quick LSE + L-BFGS to get good starting points (25% time)
    # ==================================================================
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE + L-BFGS (coarse to fine)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  n={n}")
        inits = build_initializations(n, rng, parent)
        # Limit based on time
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 10.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            # LSE with increasing beta
            for beta in [20, 50, 100, 300]:
                if time.monotonic() > phase1_deadline:
                    break
                sc, a_opt = run_lbfgs(a0, beta, maxiter=3000, maxfun=50000)
                report(sc, a_opt, f"P1 n={n} i={i} b={beta}")
                a0 = a_opt

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Multiplicative updates (15% time)
    # ==================================================================
    phase2_deadline = t_start + 0.40 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Multiplicative Updates (mirror descent)")
    print(f"{'='*60}")

    n_best = len(best_a)
    # Hard version
    if time.monotonic() < phase2_deadline:
        a_mu = multiplicative_updates(best_a, n_iters=500, eta0=0.1, decay=0.999)
        sc = compute_score(a_mu)
        report(sc, a_mu, "MU hard")

    # Soft version
    if time.monotonic() < phase2_deadline:
        for beta_mu in [20, 50, 100]:
            if time.monotonic() > phase2_deadline:
                break
            a_sm = soft_multiplicative_updates(best_a, n_iters=300, beta=beta_mu, eta0=0.5, decay=0.998)
            sc = compute_score(a_sm)
            report(sc, a_sm, f"MU soft b={beta_mu}")

    # Combine: take best, then L-BFGS refine
    if time.monotonic() < phase2_deadline:
        for beta in [200, 500]:
            if time.monotonic() > phase2_deadline:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"P2 LBFGS b={beta}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: LP-based direction finding (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.70 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: LP Direction Finding")
    print(f"{'='*60}")

    n_best = len(best_a)
    a_cur = best_a.copy() / best_a.sum()
    sc_cur = compute_score(a_cur)

    # Adaptive: start with small active set, grow it
    n_lp_iters = 0
    for it in range(200):
        if time.monotonic() > phase3_deadline:
            break
        # Adaptive active set size
        if it < 10:
            max_active = 50
        elif it < 30:
            max_active = 100
        else:
            max_active = 200
        max_rounds = 5

        t0 = time.monotonic()
        a_new, sc_new = lp_iteration(a_cur, max_active=max_active, max_rounds=max_rounds)
        dt = time.monotonic() - t0

        if sc_new < sc_cur:
            sc_cur = sc_new
            a_cur = a_new
            report(sc_cur, a_cur, f"LP it={it}")
        else:
            # No improvement, try smaller step or break
            if it > 10 and dt > 30:
                break

        n_lp_iters += 1
        if n_lp_iters % 10 == 0:
            print(f"    LP iter {n_lp_iters}: score={sc_cur:.8f}, dt={dt:.1f}s")

    print(f"  After Phase 3 ({n_lp_iters} LP iters): best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Fine LSE + L-BFGS (20% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Fine LSE + L-BFGS")
    print(f"{'='*60}")

    # Use the best from LP phase as starting point
    a_start = best_a.copy()
    for beta in [200, 500, 1000, 2000, 5000]:
        if time.monotonic() > phase4_deadline:
            break
        sc, a_opt = run_lbfgs(a_start, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"P4 LBFGS b={beta}")
        a_start = a_opt

    # Also try from LP result directly
    if time.monotonic() < phase4_deadline and sc_cur < best_score:
        for beta in [500, 2000]:
            if time.monotonic() > phase4_deadline:
                break
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"P4 LBFGS-LP b={beta}")

    print(f"  After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Coordinate descent + final polish (5% time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate Descent + Polish")
    print(f"{'='*60}")

    # Coordinate descent
    a_cd = coordinate_descent(best_a, n_sweeps=3, n_top=80)
    sc = compute_score(a_cd)
    report(sc, a_cd, "CD")

    # Final L-BFGS from CD
    if time.monotonic() < deadline - 30:
        sc, a_opt = run_lbfgs(a_cd, 2000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "CD+LBFGS")

    # One more round of multiplicative updates from best
    if time.monotonic() < deadline - 30:
        a_mu2 = multiplicative_updates(best_a, n_iters=200, eta0=0.05, decay=0.999)
        sc = compute_score(a_mu2)
        report(sc, a_mu2, "MU final")

    # Try LP one more time with larger active set
    if time.monotonic() < deadline - 30:
        a_lp_final, sc_lp = lp_iteration(best_a, max_active=300, max_rounds=10)
        report(sc_lp, a_lp_final, "LP final")
        if sc_lp < best_score:
            sc, a_opt = run_lbfgs(a_lp_final, 3000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, "LP+LBFGS final")

    # ==================================================================
    # Final: try a few random perturbations + local opt
    # ==================================================================
    if time.monotonic() < deadline - 30:
        n_best = len(best_a)
        for trial in range(5):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=2000, maxfun=30000)
            report(sc, a_opt, f"perturb {trial}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
