def kkt_multiplicative_iteration(a, p=50, alpha=0.5, n_iters=1000, 
                                  time_limit_s=None, t_start=None,
                                  verbose_interval=100):
    """
    KKT-based multiplicative iteration to flatten the self-convolution.
    a is normalized to sum=1.
    """
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > t_start + time_limit_s:
                break
        
        # Compute convolution
        c = fast_conv(a)
        c_max = c.max()
        
        # Compute soft Lagrange multipliers
        lam = (c / c_max) ** p
        
        # Compute g_j = 2 * sum_i lam[j+i] * a_i
        g = 2.0 * cross_corr_lam_a(lam, a)
        
        # KKT update: a_j *= (nu / g_j)^alpha
        # nu = mean of g (approximation of the KKT multiplier)
        g_pos = g[g > 1e-15]
        if len(g_pos) == 0:
            break
        nu = np.mean(g_pos)
        
        # Clip ratio to avoid extreme updates
        ratio = nu / (g + 1e-15)
        ratio = np.clip(ratio, 0.1, 10.0)
        
        a = a * ratio ** alpha
        a = np.maximum(a, 0.0)
        
        # Renormalize
        S = a.sum()
        if S < 1e-12:
            a = np.ones(n) / n
            break
        a = a / S
        
        # Track best
        if it % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    KKT iter {it}: score={sc:.8f}, c_max={c_max:.6e}")
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def kkt_direct_solve(a_init, active_tol=0.01, n_solves=50,

    a = a / a.sum()
    
    best_score = compute_score(a)
    best_a = a.copy()

    c = fast_conv(a)
    c_max = c.max()
    active_lags = np.where(c >= c_max * (1 - active_tol))[0]

    free_set = set(range(n))
    
    for solve_iter in range(n_solves):
        if time_limit_s is not None and t_start is not None:

                break
