def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    # Shift for numerical stability
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient computation
    rev_a = a[::-1]
    M2 = 2 * n  # need conv of rev_a (length n) with p (length 2n-1)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

"""

import numpy as np

import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def nesterov_subgrad_v2(a, total_iters=1000, K=8, lr_init=0.005):
    """Nesterov accelerated subgradient with top-K active set."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    a = a / S0
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    mu = 0.85

    for it in range(total_iters):
        a_hat = a + mu * v
        a_hat = np.maximum(a_hat, 0)
        S_hat = a_hat.sum()
        if S_hat < 1e-12:
            break
        a_hat = a_hat / S_hat
        c = fast_conv(a_hat)
        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_k_idx = np.argsort(c)[::-1][:K]
        sg = np.zeros(n)
        for k in top_k_idx:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            js = np.arange(j_lo, j_hi + 1)
            partners = k - js
            valid1 = (partners >= 0) & (partners < n)
            sg[js[valid1]] += a_hat[partners[valid1]]
            valid2 = (js >= 0) & (js < n)
            sg[partners[valid2]] += a_hat[js[valid2]]
        # Account for double-counting when j == k-j (i.e., 2j == k)
        for k in top_k_idx:
            if k % 2 == 0:
                j_mid = k // 2
                j_lo2 = max(0, k - n + 1)
                j_hi2 = min(k, n - 1)
                if j_lo2 <= j_mid <= j_hi2:
                    sg[j_mid] = (sg[j_mid] + a_hat[j_mid])  # was added twice, need a[k-j]+a[j] = 2a[j]
                    # Actually it was added once from js and once from partners, giving 2*a_hat[j_mid]
                    # which is correct for k=2j_mid
        grad = sg - np.mean(sg)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        t = (it + 1)
        lr = lr_init / (1.0 + 0.01 * t)
        v_new = mu * v - lr * grad / gn
        a = a + v_new
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        else:
            a = best_a.copy()
    final_sc = compute_score(a * S0)
    if final_sc < best_sc / (S0 * S0) * 2 * n:
        pass
    result_sc = compute_score(best_a * S0)
    return result_sc, best_a * S0


def subgrad_polish(a, total_iters=500, K=5):
    """Polished subgradient with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S0 = a.sum()

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_k = np.argsort(c)[::-1][:K]
        sg = np.zeros(n)
        for k in top_k:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            js = np.arange(j_lo, j_hi + 1)
            partners = k - js
            sg[js] += a[partners]
            sg[partners] += a[js]
        for k in top_k:
            if k % 2 == 0:
                j_mid = k // 2
                j_lo2 = max(0, k - n + 1)
                j_hi2 = min(k, n - 1)
                if j_lo2 <= j_mid <= j_hi2:
                    sg[j_mid] += a[j_mid]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / (K * S2) - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1 / (1.0 + 0.005 * it))
        a = a - step * grad
        a = np.maximum(a, 0.0)
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S0 / S_new)
        else:
            a = best_a.copy()
    return best_sc, best_a


def block_perturb(a, rng, n_blocks=32, strength=0.1):
    """Perturb blocks of the sequence to escape local minima."""
    n = len(a)
    a_pert = a.copy()
    per = max(1, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - per)))
        end = min(n, start + per)
        scale = 1.0 + strength * (rng.random() - 0.5) * 2
        a_pert[start:end] *= scale
    a_pert = np.maximum(a_pert, 0.0)
    S = a_pert.sum()
    if S > 1e-12:
        a_pert = a_pert * (a.sum() / S)
    return a_pert


def chirp_sequence(n, rng):
    """Generate CAZAC-inspired chirp initializations (non-negative)."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    f = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * f * x * x / (2.0 * n)))
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    a1 = np.cos(np.pi * x * x / (2.0 * n)) ** 2
    shift = int(n * rng.uniform(0.1, 0.4))
    a2 = np.roll(a1, shift)
    inits.append(0.5 * (a1 + a2) + 0.01)
    return [np.clip(s, 0.01, 1000.0) for s in inits]

            np.ones(n) * 0.5,
            np.cos(np.pi * np.arange(n) ** 2 / (2.0 * n)) ** 2 + 0.01,
            np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
            0.5 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n) / n),
        ]
        for bi, binit in enumerate(base_inits):
            if time.monotonic() > sf_deadline:
                break
            t0 = time.monotonic()
            sf_a = spectral_flatten_wf(binit, N_fft, n_iters=250)
            sc = compute_score(sf_a)
            key = f"sf_n{n}_b{bi}"
            sf_results[key] = (sc, sf_a)
            report(sc, sf_a, key)
            elapsed = time.monotonic() - t0
            print(f"    {key}: score={sc:.6f} ({elapsed:.1f}s)")

    # ================================================================
    # Phase 2: L-BFGS continuation from best initializations
    # ================================================================
    lbfgs_deadline = t_start + 0.60 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Continuation (deadline t={lbfgs_deadline - t_start:.0f}s) ---")

    beta_schedule = [50, 150, 400, 1000, 3000]

    for n in lengths:
        if time.monotonic() > lbfgs_deadline:
            break
        print(f"\n  >> n = {n}")
        # Collect candidates for this length
        cand_n = []
        for key, (sc, a) in sf_results.items():
            if len(a) == n:
                cand_n.append(a)
        if parent is not None and len(parent) == n:
            cand_n.append(parent.copy())
        if not cand_n:
            N_fft = 1 << (2 * n - 1).bit_length()
            cand_n.append(spectral_flatten_wf(np.ones(n) * 0.5, N_fft, 100))
            cand_n.append(np.ones(n) * 0.5)

        # Sort by score, take top
        scored_cands = [(compute_score(c), c) for c in cand_n]
        scored_cands.sort(key=lambda x: x[0])
        scored_cands = scored_cands[:min(4, len(scored_cands))]

        for ci, (sc_init, a_init) in enumerate(scored_cands):
            if time.monotonic() > lbfgs_deadline:
                break
            t0 = time.monotonic()
            a_cur = a_init.copy()
            for beta in beta_schedule:
                if time.monotonic() > lbfgs_deadline:
                    break
                remaining = lbfgs_deadline - time.monotonic()
                if remaining < 10:
                    break
                # Scale iterations
                scale_factor = max(0.3, min(1.0, remaining / 60.0))
                maxiter = int(3000 * scale_factor)
                maxfun = int(60000 * scale_factor)
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=max(100, maxiter), maxfun=max(500, maxfun))
                a_cur = a_opt
                report(sc, a_opt, f"lbfgs n={n} ci={ci} beta={beta}")
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, f"lbfgs_final n={n} ci={ci}")
            elapsed = time.monotonic() - t0
            print(f"    n={n} ci={ci}: score={sc_final:.6f} ({elapsed:.1f}s)")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Nesterov subgradient refinement
    # ================================================================
    nesterov_deadline = t_start + 0.75 * effective_budget
    print(f"\n--- Phase 3: Nesterov Subgradient ---")

    for K_val in [5, 10, 20]:
        if time.monotonic() > nesterov_deadline:
            break
        remaining = nesterov_deadline - time.monotonic()
        if remaining < 10:
            break
        n_iters = int(min(2000, remaining * 2))
        sc, a_opt = nesterov_subgrad_v2(best_a, total_iters=n_iters, K=K_val, lr_init=0.003)
        report(sc, a_opt, f"nesterov K={K_val}")

    # ================================================================
    # Phase 4: Block perturbation local search
    # ================================================================
    local_deadline = t_start + 0.92 * effective_budget
    print(f"\n--- Phase 4: Block Perturbation Local Search ---")
    n_best = len(best_a)
    n_trials = 60
    for trial in range(n_trials):
        if time.monotonic() > local_deadline:
            break
        strategy = trial % 4
        if strategy == 0:
            # Block perturbation
            n_blocks = 16 + 32 * int(rng.random())
            strength = 0.05 + 0.15 * rng.random()
            a_pert = block_perturb(best_a, rng, n_blocks=n_blocks, strength=strength)
        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.02 + 0.06 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0.0)
        elif strategy == 2:
            # Spectral perturbation
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise = 0.03 * rng.standard_normal(len(A)) + 0.03j * rng.standard_normal(len(A))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.maximum(a_pert, 0.0)
        else:
            # Targeted: reduce contributions to peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                seg_len = j_end - j_start + 1
                reduce_idx = rng.choice(seg_len, size=max(3, seg_len // 10), replace=False)
                for ri in reduce_idx:
                    j = j_start + ri
                    a_pert[j] *= (1.0 - 0.1 * rng.random())
                a_pert = np.maximum(a_pert, 0.0)
                S = a_pert.sum()
                if S > 1e-12:
                    a_pert *= (best_a.sum() / S)

        # Quick L-BFGS polish
        remaining = local_deadline - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(2000, int(remaining / 0.03))
        sc, a_opt, _ = run_lbfgs(a_pert, 800, maxiter=max(100, maxiter), maxfun=max(500, maxiter * 20))
        report(sc, a_opt, f"local #{trial} s={strategy}")

    print(f"\nAfter Phase 4: best = {best_score:.8f}")

    # ================================================================
    # Phase 5: Final high-beta polish + subgradient
    # ================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    for beta in [3000, 8000, 20000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final lbfgs beta={beta}")

    sc, a_opt = subgrad_polish(best_a, total_iters=1000, K=8)
    report(sc, a_opt, "final subgrad")

    # One more L-BFGS from the polished result
    if time.monotonic() < deadline - 10:
        sc, a_opt2, _ = run_lbfgs(a_opt, 5000, maxiter=3000, maxfun=60000)
        report(sc, a_opt2, "post-subgrad lbfgs")

    # Final subgradient
    sc, a_opt3 = subgrad_polish(best_a, total_iters=500, K=10)
    report(sc, a_opt3, "final-final subgrad")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
