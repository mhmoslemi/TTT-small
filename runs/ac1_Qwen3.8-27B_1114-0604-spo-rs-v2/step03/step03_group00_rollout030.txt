```python
"""
Hybrid optimizer combining LP-based algorithm iteration, L-BFGS, and multiplicative updates.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. LP-based algorithm iteration (maximize sum(g) s.t. conv(f,g) <= M, g >= 0)

import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_iteration(a, n_iters=10, blend=0.5):
    """
    LP-based algorithm iteration.
    Given f = a (normalized), find g maximizing sum(g) s.t. g >= 0, ||f*g||_inf <= M.
    Blend: a_new = (1-blend)*a + blend*g_normalized.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S  # normalize to sum=1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        # Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
        # (a*g)[k] = sum_j a[j] * g[k-j] (with bounds 0 <= j,k-j < n)
        # This is: sum_j a[j] * g[j'] <= M where j' = k-j, so 0 <= k-j < n => k-n+1 <= j <= k
        # In matrix form: A_lp @ g <= M * ones(2n-1)
        # where A_lp[k, j] = a[j] if k-n+1 <= j <= min(k, n-1), else 0
        # Actually A_lp[k, j] = a[k-j] if 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k,n-1)
        
        # For large n, building the full matrix is expensive. Let's use cvxpy.
        g = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(g))
        
        # Build constraints: for each k, sum_j a[max(0,k-n+1):min(k+1,n)] @ g[...reversed index] <= M
        # (a*g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * g[k-j]
        # Let i = k-j, then i ranges from max(0,k-n+1) to min(k,n-1)
        # (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * g[i]
        
        # This is a sparse matrix. Let's build it.
        # A_lp[k, i] = a[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
        
        # For efficiency, use scipy sparse
        from scipy.sparse import lil_matrix
        A_lp = lil_matrix((2*n-1, n))
        for k in range(2*n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                A_lp[k, k - j] = a[j]
        
        A_lp = A_lp.tocsr()
        constraints = [A_lp @ g <= M]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-6)
        except Exception:
            break
        
        if g.value is None:
            break
        
        g_val = np.maximum(g.value, 0.0)
        g_sum = float(g_val.sum())
        if g_sum < 1e-12:
            break
        
        g_norm = g_val / g_sum  # normalize to sum=1
        
        # Blend
        a_new = (1 - blend) * a + blend * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * d
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.2 + 0.6 * np.abs(2.0 * x - 1.0))
    
    # Ahmed-Zaidi inspired: (1 - 2x)^2 shape
    inits.append((1.0 - 2.0 * x) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - 2.0 * x) ** 4)
    
    # Random block structure
    for nb in (8, 16, 32, 64):
        per = max(1, n // nb)
        vals = 0.1 + 0.9 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits


def coordinate_descent_peak(a, n_sweeps=3, n_top=50):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3]]
            candidates.append(0.0)
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
```

```python
def simplified_lp_iteration(a, n_iters=20, blend=0.5, n_top_peaks=50):
    """
    Simplified LP algorithm iteration using only top K peak constraints.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        
        # Get top K peaks
        k_indices = np.argsort(c)[::-1][:n_top_peaks]
        
        # Build constraint matrix: for each peak k, (a*g)[k] <= M
        # (a*g)[k] = sum_j a[j] * g[k-j] where 0<=j<n, 0<=k-j<n
        # = sum_i a[k-i] * g[i] where i = k-j, max(0,k-n+1) <= i <= min(k, n-1)
        
        # Use scipy sparse for the constraint matrix
        from scipy.sparse import lil_matrix
        A_con = lil_matrix((n_top_peaks, n))
        for idx, k in enumerate(k_indices):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                A_con[idx, k - j] = a[j]
        
        A_con = A_con.tocsr()
        
        # Solve: max sum(g) s.t. A_con @ g <= M, g >= 0
```

```python
"""
Hybrid optimizer: Multiplicative Updates + Error Reduction + Simplified LP + L-BFGS + Subgradient.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Innovations over previous approach:
1. Multiplicative update scheme (naturally maintains non-negativity, O(n log n) per step)
2. Multi-peak targeting with adaptive weighting
3. Simplified LP algorithm iteration (top-K peak constraints only)
4. Adaptive error reduction with progressive sharpening
5. Enhanced local search with norm-based perturbations
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import cvxpy as cp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = np.log(exp_scaled.sum() + 1e-30)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / (exp_scaled.sum() + 1e-30)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def multiplicative_update(a, n_iters=300, alpha=0.3, peak_power=5, thresh_frac=0.9):
    """
    Multiplicative update scheme for flattening the convolution.
    Uses multi-peak weighted pressure to guide updates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_pad = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        if c_max < 1e-15:
            break

        # Multi-peak weighting: emphasize peaks above threshold
        thresh = thresh_frac * c_max
        w = np.zeros(len(c))
        mask = c >= thresh
        w[mask] = (c[mask] / c_max) ** peak_power

        if w.sum() < 1e-15:
            # Fallback: single peak
            k_star = int(np.argmax(c))
            w[k_star] = 1.0

        # Compute pressure: press[j] = sum_k w[k] * a[k-j] (with valid bounds)
        # This is cross-correlation of w and a
        W_fft = np.fft.rfft(w, M_pad)
        A_fft = np.fft.rfft(a, M_pad)
        press = np.fft.irfft(np.conj(W_fft) * A_fft, M_pad)[:n]
        press = np.maximum(press, 1e-15)

        target = float(press.mean())
        if target < 1e-15:
            break

        # Adaptive alpha: decrease over iterations for stability
        alpha_eff = alpha * (1.0 - 0.3 * it / max(1, n_iters))
        alpha_eff = max(alpha_eff, 0.05)

        ratio = target / press
        a = a * (ratio ** alpha_eff)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def error_reduction(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        # Progressive sharpening: start broad, get narrow
        alpha_eff = alpha + (1.0 - alpha) * min(1.0, it / max(1, n_iters * 0.6))
        thresh = conv_mean + (1.0 - alpha_eff) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.3 + 0.6 * min(1.0, it / max(1, n_iters // 3))
        A_new = A + damp * d

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

def simplified_lp_iteration(a, n_iters=10, blend=0.4, n_top_peaks=30):
    """
    Simplified LP algorithm iteration using only top-K peak constraints.
    Maximizes sum(g) s.t. g >= 0 and (a*g)[k] <= M for top peaks.
    """
    n = len(a)
    if n > 10000:
        return a  # Too large for LP

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())

        # Get top K peaks
        k_indices = np.argsort(c)[::-1][:n_top_peaks]

        # Build constraint matrix (sparse)
        A_con = lil_matrix((n_top_peaks, n))
        for idx, k in enumerate(k_indices):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                A_con[idx, k - j] = a[j]
        A_con = A_con.tocsr()

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(g_var))
        cons = [A_con @ g_var <= M_val]
        prob = cp.Problem(obj, cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-7)
        except Exception:
            break

        if g_var.value is None:
            break

        g_val = np.maximum(g_var.value, 0.0)
        g_sum = float(g_val.sum())
        if g_sum < 1e-12:
            break

        g_norm = g_val / g_sum
        a_new = (1 - blend) * a + blend * g_norm
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def multi_peak_subgradient(a, total_iters=3000, n_peaks=10):
    """Subgradient descent using top-K peaks instead of just the max."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top K peaks with softmax weighting
        top_k = np.argsort(c)[::-1][:n_peaks]
        c_top = c[top_k]
        # Weight peaks by their height (normalized)
        w_peaks = c_top / c_top.sum()

        # Combined subgradient
        sg = np.zeros(n)
        for idx, k in enumerate(top_k):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                indices = np.arange(j_start, j_end + 1)
                sg[indices] += w_peaks[idx] * 2.0 * a[k - indices]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def coordinate_descent_peak(a, n_sweeps=3, n_top=80):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3]]
            candidates.append(0.0)
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

                threshes=[0.85, 0.9, 0.93, 0.97]
            )
            sc = compute_score(a_mu)
            report(sc, a_mu, "MU n=%d i=%d" % (n, i))

            # Error reduction from MU result
            a_er = error_reduction(a_mu, n_iters=150, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, "ER n=%d i=%d" % (n, i))

            # Multi-alpha error reduction
            a_er2 = error_reduction_multi(a_mu, n_total_iters=200)
            sc = compute_score(a_er2)
            report(sc, a_er2, "ERm n=%d i=%d" % (n, i))

    print("After Phase 1: best = %.8f" % best_score)

    # ==================================================================
    # Phase 2: Simplified LP iteration (15% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.45 * total_budget
    print("\n--- Phase 2: Simplified LP Iteration ---")

    n_lp = len(best_a)
    if n_lp > 10000:
        n_lp = 8192
    a_lp = best_a if len(best_a) == n_lp else np.interp(
        np.linspace(0, 1, n_lp), np.linspace(0, 1, len(best_a)), best_a
    )

    if time.monotonic() < phase2_deadline and n_lp <= 10000:
        a_lp_res = simplified_lp_iteration(a_lp, n_iters=12, blend=0.4, n_top_peaks=30)
        sc = compute_score(a_lp_res)
        report(sc, a_lp_res, "LP n=%d" % n_lp)

        # L-BFGS from LP result
        if time.monotonic() < phase2_deadline:
            sc, a_opt, _ = run_lbfgs(a_lp_res, 300, maxiter=5000, maxfun=80000)
            report(sc, a_opt, "LP+LBFGS")

        # Error reduction from LP result
        if time.monotonic() < phase2_deadline:
            a_er_lp = error_reduction(a_lp_res, n_iters=100, alpha=0.8)
            sc = compute_score(a_er_lp)
            report(sc, a_er_lp, "LP+ER")

    print("After Phase 2: best = %.8f" % best_score)

    # ==================================================================
    # Phase 3: L-BFGS refinement (30% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * total_budget
    print("\n--- Phase 3: L-BFGS Refinement ---")

    beta_schedule = [50, 100, 200, 500, 1000, 1500]

    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        report(sc, a_opt, "LBFGS beta=%d" % beta)

    # Try from interpolated different lengths
    for n_alt in lengths[:3]:
        if time.monotonic() > phase3_deadline:
            break
        if n_alt == len(best_a):
            continue
        a_alt = np.interp(
            np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_a)), best_a
        )
        for beta in [200, 500]:
            if time.monotonic() > phase3_deadline:
                break
            sc, a_opt, _ = run_lbfgs(a_alt, beta, maxiter=5000, maxfun=80000)
            report(sc, a_opt, "LBFGS n=%d b=%d" % (n_alt, beta))

    print("After Phase 3: best = %.8f" % best_score)

    # ==================================================================
    # Phase 4: Local search + polish (25% of time)
    # ==================================================================
    print("\n--- Phase 4: Local Search & Polish ---")

    n_best = len(best_a)
    n_trials = 12
    for trial in range(n_trials):
        if time.monotonic() > deadline - 40:
            break
        strategy = trial % 6
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak reduction with redistribution
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(10, n_best // 60)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.05
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Multi-peak subgradient from perturbed start
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = a_pert / a_pert.sum()
            a_pert = multi_peak_subgradient(a_pert, total_iters=2000, n_peaks=5)
        elif strategy == 4:
            # Error reduction restart
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = error_reduction(a_pert, n_iters=80, alpha=0.8)
        else:
            # Multiplicative update from perturbed start
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = multiplicative_update(a_pert, n_iters=100, alpha=0.3,
                                           peak_power=8, thresh_frac=0.95)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_p = a_pert.sum()
        if S_p > 1e-12:
            a_pert = a_pert / S_p

        sc, a_opt, _ = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "local#%d s=%d" % (trial, strategy))

    # Coordinate descent
    if time.monotonic() < deadline - 30:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=3, n_top=80)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_descent")
        sc, a_opt, _ = run_lbfgs(a_cd, 500, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "CD+LBFGS")

    # Multi-peak subgradient polish
    if time.monotonic() < deadline - 30:
        a_mp = multi_peak_subgradient(best_a, total_iters=3000, n_peaks=8)
        sc = compute_score(a_mp)
        report(sc, a_mp, "multi_peak_sg")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 25:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final_LBFGS_b2000")

    # Final error reduction
    if time.monotonic() < deadline - 15:
        a_er_final = error_reduction(best_a, n_iters=150, alpha=0.85)
        sc = compute_score(a_er_final)
        report(sc, a_er_final, "final_ER")

    # Final multiplicative update
    if time.monotonic() < deadline - 10:
        a_mu_final = multiplicative_update(best_a, n_iters=150, alpha=0.2,
                                          peak_power=10, thresh_frac=0.97)
        sc = compute_score(a_mu_final)
        report(sc, a_mu_final, "final_MU")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print("\n" + "=" * 60)
    print("FINAL: internal=%.8f, verified=%.8f" % (best_score, verified))
    print("Length: %d, Time: %.1fs / %ds" % (len(out), elapsed_total, budget_s))
    print("=" * 60)
    return out
```
