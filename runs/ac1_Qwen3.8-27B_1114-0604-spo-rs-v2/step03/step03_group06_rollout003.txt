"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based LSE-continuation with peak-targeted local search, diverse restarts,
and multi-length exploration. Key improvements: refined continuation schedule,
active-set-aware peak shaving, and tournament-based candidate management.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def active_set_lse_grad(a, k_active):
    """
    Compute gradient of the average of active-set convolution values.
    k_active: list of convolution lag indices to average.
    This gives a smoother objective than LSE when the active set is known.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    # Use the active set lags
    eps = 1e-10
    # Soft active set: weight = exp(c[k] / temp) for k in active set
    c_active = np.array([c[k] for k in k_active])
    temp = max((c_active.max() - c_active.min()), eps) * 0.1
    w = np.exp((c_active - c_active.max()) / temp)
    w = w / w.sum()
    L = float(np.dot(w, c_active))
    score = 2.0 * n * L / (S * S)
    # Build full p vector (sparse)
    p = np.zeros(2 * n - 1)
    for i, k in enumerate(k_active):
        p[k] = w[i]
    # Gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def peak_shave_local(a, n_iters=200, rng=None):
    """
    Local search that targets the peak convolution lags directly.
    Reduces elements contributing most to peaks, redistributes to non-peak contributors.
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find peak lags (top few)
        k_star = int(np.argmax(c))
        # Subgradient direction: reduce a[j] where j contributes to c[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        contrib = np.zeros(n)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        # Find non-peak-contributing positions
        all_contrib = np.zeros(2 * n - 1)
        # Compute total contribution to each element from ALL lags
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                all_contrib[js:je + 1] += a[js:je + 1][::-1] * (c[k] / M)

        # Reduce from high-contributing, add to low-contributing
        top_k = max(3, n // 30)
        bottom_k = max(3, n // 15)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_idx = np.argsort(all_contrib)[:bottom_k]

        reduce_frac = 0.02 + 0.03 * rng.random()
        total_reduced = 0.0
        for j in top_idx:
            amt = a[j] * reduce_frac
            a[j] -= amt
            a[j] = max(a[j], 0.0)
            total_reduced += amt

        if total_reduced > 0:
            per = total_reduced / len(bottom_idx)
            for j in bottom_idx:
                a[j] += per

        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            a = best_a.copy()

    return best_sc, best_a


def diverse_init(n, rng, parent=None):
    """Generate a diverse set of initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
        # Scaled/noisy parent
        inits.append(np.clip(parent_resized(parent, n) * (1.0 + 0.1 * rng.standard_normal(n)), 0.01, 1000.0))

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine family
    for freq in [1, 2, 3, 5]:
        inits.append(np.clip(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x), 0.01, 1000.0))
    # Sine-modulated
    inits.append(np.clip(0.5 + 0.2 * np.sin(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    # Bump
    inits.append(np.clip(0.2 + 0.8 * np.exp(-4.0 * (x - 0.5) ** 2), 0.01, 1000.0))
    # Two bumps
    inits.append(np.clip(0.3 + 0.5 * np.exp(-6.0 * (x - 0.25) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.75) ** 2), 0.01, 1000.0))
    # Power law decay
    inits.append(np.clip(0.5 * (1 - x) ** 0.5 + 0.01, 0.01, 1000.0))
    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    amps = 0.1 + 0.2 * rng.random(5)
    phases = 2 * np.pi * rng.random(5)
    sig = np.zeros(n)
    for f, a, ph in zip(freqs, amps, phases):
        sig += a * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.clip(0.5 + sig, 0.01, 1000.0))
    # Random noise
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in [4, 8, 16]:
        per = max(1, n // nb)
        vals = 0.1 + 0.9 * rng.random(nb)
        blk = np.repeat(vals, per)
        if len(blk) < n:
            blk = np.concatenate([blk, vals[:n - len(blk)]])
        inits.append(blk[:n].copy())

    return inits


def parent_resized(parent, n):
    """Resize parent to length n via interpolation."""
    m = len(parent)
    if m == n:
        return parent.astype(np.float64).copy()
    idx = np.linspace(0, m - 1, n)
    lo = idx.astype(int)
    hi = np.minimum(lo + 1, m - 1)
    frac = idx - lo
    return parent[lo] * (1 - frac) + parent[hi] * frac


def subgradient_polish(a, n_iters=500, rng=None):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(c)
        # Get all argmax lags for a more robust subgradient
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(5, len(c_sorted) - 1)]
        # Average over near-max lags
        near_max_mask = c >= threshold * 0.999
        k_active = np.where(near_max_mask)[0]

        sg = np.zeros(n)
        for k in k_active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg /= len(k_active)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lower) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    Uses LSE-continuation, peak shaving, subgradient polish, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [-512, 512, -1024, 1024, 2048]:
            cand = pa + dl
            if 1024 <= cand <= 20000:
                lengths.append(cand)
    for n in [4096, 6144, 8192, 10240, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit to 5 lengths max
    lengths = lengths[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes' if parent is not None else 'no'}, len={parent.size if parent is not None else 0}")

    # Phase 1: LSE continuation for each length (60% of budget)
    phase1_deadline = t_start + 0.60 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE Continuation (until t={phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = diverse_init(n, rng, parent)

        # Score initializations and sort (keep best ones)
        init_scores = []
        for a0 in inits:
            sc = compute_score(a0)
            init_scores.append((sc, a0))
        init_scores.sort(key=lambda x: x[0])

        # Try top initializations with continuation
        n_inits_to_try = min(len(init_scores), max(3, int((phase1_deadline - time.monotonic()) / 60.0)))
        print(f"  Trying {n_inits_to_try} of {len(init_scores)} initializations")

        for i, (init_sc, a0) in enumerate(init_scores[:n_inits_to_try]):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy()
            # Continuation schedule
            betas = [10, 30, 100, 300, 1000, 3000, 10000]
            for bi, beta in enumerate(betas):
                if time.monotonic() > phase1_deadline:
                    break
                remaining = phase1_deadline - time.monotonic()
                maxiter = min(int(remaining / 5.0), 3000)
                if maxiter < 50:
                    break
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=max(maxiter * 2, 1000))
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # Phase 2: Peak shading + subgradient (20% of budget)
    phase2_deadline = t_start + 0.80 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak Shaving & Subgradient (until t={phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_work = best_a.copy()
    remaining_iters = int((phase2_deadline - time.monotonic()) / 3.0)
    sc, a_new = peak_shave_local(a_work, n_iters=max(100, remaining_iters // 2), rng=rng)
    report(sc, a_new, "peak_shave")
    a_work = a_new

    sc, a_new = subgradient_polish(a_work, n_iters=max(200, remaining_iters // 3))
    report(sc, a_new, "subgrad")
    a_work = a_new

    # Alternate peak shave and subgradient
    n_rounds = max(3, int((phase2_deadline - time.monotonic()) / 10.0))
    for r in range(n_rounds):
        if time.monotonic() > phase2_deadline:
            break
        sc, a_new = peak_shave_local(a_work, n_iters=100, rng=rng)
        report(sc, a_new, f"alt_ps_{r}")
        a_work = a_new
        sc, a_new = subgradient_polish(a_work, n_iters=200)
        report(sc, a_new, f"alt_sg_{r}")
        a_work = a_new
        # Small LSE polish
        if time.monotonic() < phase2_deadline - 5:
            sc, a_new = run_lbfgs(a_work, 5000, maxiter=2000, maxfun=4000)
            report(sc, a_new, f"alt_lbfgs_{r}")
            a_work = a_new

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # Phase 3: Diversified restarts (12% of budget)
    phase3_deadline = t_start + 0.92 * (budget_s - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Diversified Local Search (until t={phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase3_deadline and trial < 30:
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.06 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Frequency modulation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.05 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Sparse perturbation
            n_sparse = max(10, n_best // 40)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.5 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Mix with structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 8 * x + rng.random() * 6.28)
            t_mix = 0.2 + 0.3 * rng.random()
            a_pert = (1 - t_mix) * best_a + t_mix * a_struct * (best_a.sum() / max(a_struct.sum(), 1e-10))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            # Reduce top contributors at peak
            contrib = np.zeros(n_best)
            contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            top_n = max(5, n_best // 50)
            top_idx = np.argsort(contrib)[::-1][:top_n]
            reduce_amt = best_a[top_idx].sum() * 0.05
            for j in top_idx:
                a_pert[j] -= reduce_amt / top_n
            # Add to random non-peak positions
            bottom_idx = np.argsort(contrib)[:top_n * 2]
            add_amt = reduce_amt / max(len(bottom_idx), 1)
            for j in bottom_idx:
                a_pert[j] += add_amt
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Polish
        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter=2000, maxfun=4000)
        report(sc, a_opt, f"local_{trial}_s{strategy}")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # Phase 4: Final intensive polish (8% of budget)
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    # High beta LSE
    for beta in [5000, 10000, 20000, 50000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final_b{beta}")

    # Subgradient
    if time.monotonic() < deadline - 5:
        sc, a_pol = subgradient_polish(best_a, n_iters=1000)
        report(sc, a_pol, "final_subgrad")

    # One more peak shave
    if time.monotonic() < deadline - 3:
        sc, a_pol = peak_shave_local(best_a, n_iters=500, rng=rng)
        report(sc, a_pol, "final_peaks")

    # Final LSE from best
    if time.monotonic() < deadline - 2:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "very_final_lbfgs")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
