            new_score = evaluate_sequence(res.x)
            if new_score < best_score:
                best_score = new_score
                best_candidate = res.x.copy()
                print(f"Improve score: {best_score:.8f} at beta={beta_val}")

        # Random perturbation if not improved for this damping level
        if damping_level == "low":
            # Introduce white noise for exploration
            perturb_amount_high = 3.0
            perturb_scale = 0.5 * np.ones(len(sequence_to_optimize))
            perturb_amount = len(sequence_to_optimize) * 0.1
            res = minimize(objective_func, best_candidate + perturb_amount_high * np.random.rand(len(perturb_scale)), method='L-BFGS-B', bounds=boundaries)
            new_score = evaluate_sequence(res.x)
            if new_score < best_score:
                best_score = new_score
                best_candidate = res.x.copy()

    # Final defensivere-evaluation and return
    final_scores, final_candidate = [], []
    for seq in [best_candidate, fallback_sequence]:
        score_here = evaluate_sequence(seq)
        if score_here is not np.inf:
            final_scores.append(score_here)
            final_candidate.append(seq)
    return np.array(final_candidate).flatten().tolist() if final_candidate else fallback_sequence.tolist()
