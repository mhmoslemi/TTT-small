    grad_L = 2.0 * conv_bp[n-1:2*n-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * log_sum / S3)
    return score, grad

"""
Optimized self-convolution peak minimizer using multi-scale L-BFGS with
verified FFT-based gradients, diverse chirp initializations, and targeted
local search. Exploits the homogeneity of the objective and uses careful
beta continuation for the log-sum-exp smoothed peak function.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of L w.r.t. a[j] = 2 * sum_i a[i]*p[j+i]
    # = 2 * conv(a_reversed, p)[j + n - 1]
    b = a[::-1]
    M_conv = 3 * n - 2
    M_fft = 1 << (M_conv - 1).bit_length()
    B = np.fft.rfft(b, M_fft)
    P = np.fft.rfft(p, M_fft)
    conv_bp = np.fft.irfft(B * P, M_fft)
    grad_L = 2.0 * conv_bp[n - 1:2 * n - 1]
    # Full gradient of 2*n*L/(S^2)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def make_lbfgs_obj(a, beta):
    """Wrapper for scipy minimize (returns (f, g) tuple)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            make_lbfgs_obj, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def chirp_init(n, rng, variant=0):
    """Generate chirp-based initializations."""
    x = np.arange(n, dtype=np.float64)
    if variant == 0:
        # Quadratic chirp
        return np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01
    elif variant == 1:
        # Linear frequency modulation
        f = 1.0 + 3.0 * rng.random()
        return np.clip(0.5 + 0.4 * np.cos(np.pi * f * x * x / (2.0 * n)), 0.01, None)
    elif variant == 2:
        # Cubic chirp
        return np.clip(0.5 + 0.3 * np.cos(np.pi * x * x * x / (2.0 * n * n)), 0.01, None)
    elif variant == 3:
        # Mixed frequencies
        f1 = 1.0 + 4.0 * rng.random()
        f2 = 1.0 + 4.0 * rng.random()
        return np.clip(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) +
                        0.2 * np.cos(2.0 * np.pi * f2 * x / n), 0.01, None)
    elif variant == 4:
        # Golay-like: sum of shifted chirps
        a1 = np.cos(np.pi * x * x / (2.0 * n)) ** 2
        shift = int(n * rng.uniform(0.1, 0.4))
        a2 = np.roll(a1, shift)
        return np.clip(0.5 * (a1 + a2) + 0.01, 0.01, None)
    elif variant == 5:
        # pseudorandom with structure
        return np.clip(0.3 + 0.7 * np.abs(np.sin(2 * np.pi * rng.random() * x / n + rng.random() * 2 * np.pi)), 0.01, None)
    else:
        # Higher-order chirp
        p = 2 + int(rng.integers(0, 3))
        return np.clip(0.5 + 0.3 * np.cos(np.pi * x ** p / (n ** (p - 1))), 0.01, None)


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Iterative spectral flattening for the self-convolution spectrum."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # We want A^2 (element-wise) to be flat, i.e., mag^2 to be flat
        # So we want mag to be flat (constant)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + (1 if N % 2 else 0)):
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            if k < N - k:
                new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def parametric_chirp(n, params, rng):
    """Generate a parametric chirp sequence from parameters."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n) * params['base']
    for i in range(params.get('n_terms', 3)):
        freq = params['freqs'][i]
        phase = params['phases'][i]
        amp = params['amps'][i]
        order = params.get('orders', [1, 2, 3])
        if order == 1:
            a = a + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
        elif order == 2:
            a = a + amp * np.cos(np.pi * freq * x * x / n + phase)
        else:
            a = a + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.clip(a, 0.0, 1000.0)


def subgradient_descent(a, n_iters=500, verbose=False):
    """Subgradient method with line search for the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 1.0
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max at argmax positions
        k_stars = np.where(np.isclose(c, M, atol=1e-10 * M))[0]
        sg = np.zeros(n)
        for k in k_stars:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                sg[j] += 2.0 * a[k - j] / len(k_stars)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step size
        step = step_scale / (gn + 1e-10)
        step = min(step, 0.1)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        # Line search: try a few step sizes
        best_step_sc = sc
        for fraction in [0.5, 0.25, 0.1]:
            a_try = np.clip(a - step * fraction * grad, 0.0, 1000.0)
            sc_try = compute_score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                a_new = a_try
        a = a_new
        if best_step_sc >= sc:
            step_scale *= 0.7
        else:
            step_scale = min(step_scale * 1.2, 10.0)
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.03):
    """Perturb elements that contribute most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Find which pairs (i, k-i) contribute to c[k_star]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    for j in range(j_start, j_end + 1):
        contrib[j] = a[j] * a[k_star - j]
    # Reduce top contributors, increase bottom ones
    top_count = max(2, n // 100)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count * 2]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += amt
    if total_reduced > 1e-15 and len(bottom_idx) > 0:
        for j in bottom_idx:
            a_pert[j] += total_reduced / len(bottom_idx)
    return np.clip(a_pert, 0.0, 1000.0)


def coordinate_descent(a, n_rounds=5, n_top=100):
    """Coordinate descent on the top-contributing elements."""
    n = len(a)
    best_sc = compute_score(a)
    best_a = a.copy()
    for rnd in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        # Identify top contributing indices
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_idx = np.argsort(contrib)[::-1][:n_top]
        improved = False
        for j in top_idx:
            if a[j] < 1e-10:
                continue
            # Try small decreases
            for frac in [0.01, 0.005, 0.002]:
                a_try = a.copy()
                a_try[j] *= (1.0 - frac)
                sc_try = compute_score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            break
    return best_sc, best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate a diverse set of initializations of length n."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Chirps

        inits.append(chirp_init(n, rng, variant=v))
    # Random with various sparsity
    for scale in [0.2, 0.5, 0.8, 1.0]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, None))
    # Structured: sum of chirps
    for _ in range(3):
        n_terms = rng.integers(2, 6)
        params = {
            'base': 0.3 + 0.3 * rng.random(),
            'n_terms': int(n_terms),
            'freqs': rng.random(int(n_terms)) * 8 + 1,
            'phases': rng.random(int(n_terms)) * 2 * np.pi,
            'amps': (0.2 + 0.3 * rng.random(int(n_terms))),
            'orders': list(rng.integers(1, 4, size=int(n_terms)))
        }
        inits.append(parametric_chirp(n, params, rng))
    # Block structure
    for nb in [4, 8, 16]:
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, rng.random(n - len(a)) * 0.5 + 0.2])
        inits.append(a[:n].copy())
    return inits


def pre_screen(init, beta=20, maxiter=300):
    """Quick L-BFGS to pre-optimize an initialization."""
    try:
        sc, a_opt = run_lbfgs(init, beta, maxiter=maxiter, maxfun=maxiter * 15)
        return sc, a_opt
    except Exception:
        return compute_score(init), init

                if time.monotonic() > phase2_deadline:
                    break
                remaining = phase2_deadline - time.monotonic()
                if remaining < 5:
                    break
                # Scale maxiter based on n and remaining time
                if n <= 4096:
                    maxiter = min(5000, int(remaining * 0.6 / 0.003))
                elif n <= 8192:
                    maxiter = min(3000, int(remaining * 0.6 / 0.006))
                else:
                    maxiter = min(2000, int(remaining * 0.6 / 0.012))
                maxiter = max(100, maxiter)
                maxfun = max(500, maxiter * 20)
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
                a_cur = a_opt
                report(sc, a_opt, f"LH n={n} c={ci} b={beta}")
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, f"LH_final n={n} c={ci}")
            elapsed = time.monotonic() - t0
            print(f"    c={ci}: score={sc_final:.6f} ({elapsed:.1f}s)")
        # Update top_candidates with refined results
        all_results = []
        for sc_c, a_c in cands:
            all_results.append((sc_final, a_cur))
        # Add the refined best
        all_results.append((compute_score(a_cur), a_cur))
        all_results.sort(key=lambda x: x[0])
        top_candidates[n] = all_results[:4]

    print(f"\nAfter Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ================================================================
    # Phase 3: Subgradient + targeted local search (15% time)
    # ================================================================
    phase3_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 3: Local Search (t={phase3_deadline - t_start:.0f}s) ---")

    n_best = len(best_a)
    # First: subgradient from best
    sc_sg, a_sg = subgradient_descent(best_a, n_iters=300)
    report(sc_sg, a_sg, "subgrad")

    # Targeted perturbations + short L-BFGS
    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 6
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
        elif strategy == 2:
            # Frequency-domain perturbation
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise_amp = 0.01 + 0.03 * rng.random()
            noise = noise_amp * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
            A_noisy = A + noise
            A_noisy[0] = A[0]  # preserve DC
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Chirp modulation
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 6.0 * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(np.pi * f * x * x / (2.0 * n_best)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Swap-based: move mass from peak contributors to non-contributors
            a_pert = targeted_perturb(best_a, rng, strength=0.05)
        else:
            # Random restart blend
            t_blend = 0.05 + 0.1 * rng.random()
            a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n_best), 0.0, None)
            a_pert = (1 - t_blend) * best_a + t_blend * a_rand
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick L-BFGS polish
        remaining = phase3_deadline - time.monotonic()
        if remaining < 3:
            break
        maxiter = min(2000, int(remaining * 0.4 / 0.005))
        maxiter = max(100, maxiter)
        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=maxiter, maxfun=max(500, maxiter * 15))
        report(sc, a_opt, f"local#{trial} s={strategy}")

    # ================================================================
    # Phase 4: Final polish (remaining time)
    # ================================================================
    print(f"\n--- Phase 4: Final Polish ---")
    print(f"  Current best: {best_score:.8f} (n={len(best_a)})")

    # Very high beta L-BFGS
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=60000)
        report(sc, a_opt, f"polish beta={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 5:
        sc_sg2, a_sg2 = subgradient_descent(best_a, n_iters=400)
        report(sc_sg2, a_sg2, "final_subgrad")

    # Coordinate descent
    if time.monotonic() < deadline - 5:
        sc_cd, a_cd = coordinate_descent(best_a, n_rounds=3, n_top=50)
        report(sc_cd, a_cd, "coord_desc")

    # One more high-beta pass
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=2000, maxfun=40000)
        report(sc, a_opt, "final_lbfgs")

    # Final subgradient
    if time.monotonic() < deadline - 3:
        sc_sg3, a_sg3 = subgradient_descent(best_a, n_iters=200)
        report(sc_sg3, a_sg3, "very_final_sg")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 1024

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
