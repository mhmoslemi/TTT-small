"""
Step function optimization using normalized LSE + multi-peak shaving + spectral restarts.
Minimizes max(conv(a,a)) subject to sum(a)=sqrt(2n), a>=0.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score_normalized(a):
    """Score = max(c) when sum(a) = sqrt(2n)."""
    n = len(a)
    c = fast_conv(a)
    return float(np.max(c))


def compute_score(a):
    """Original score: 2n * max(c) / S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_a(a, target_sum=None):
    """Normalize a so sum(a) = target_sum (default sqrt(2n))."""
    n = len(a)
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    S = float(a.sum())
    if S < 1e-12:
        return a.copy()
    return a * (target_sum / S)


def lse_obj_norm(a, beta):
    """LSE objective for normalized problem: LSE_beta(c) where c=conv(a,a)."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        L = c_max
    else:
        L = c_max + np.log(exp_sum) / beta
    # Add constraint penalty for sum != sqrt(2n)
    target = np.sqrt(2.0 * n)
    S = float(a.sum())
    penalty = 1e6 * (S - target) ** 2
    return L + penalty


def lse_obj_norm_grad(a, beta):
    """LSE objective and gradient for normalized problem."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
        L = c_max
    else:
        p = exp_scaled / exp_sum
        L = c_max + np.log(exp_sum) / beta
    
    # Gradient of LSE w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    
    # Gradient of penalty
    target = np.sqrt(2.0 * n)
    S = float(a.sum())
    grad_pen = 2e6 * (S - target)
    
    grad = grad_L + grad_pen
    return L + 1e6 * (S - target) ** 2, grad

def run_lbfgs(a0, func, args=(), maxiter=2000, maxfun=50000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-20, 'gtol': 1e-15,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def subgradient_step(a, k_list, weights, step_size):
    """
    Take a subgradient step for the max(c) objective.
    k_list: indices of peaks to shave
    weights: weights for each peak
    """
    n = len(a)
    c = fast_conv(a)
    grad = np.zeros(n)
    for k, w in zip(k_list, weights):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]
    # Normalize gradient
    grad *= 2.0  # derivative of c_k = sum a_j a_{k-j} w.r.t. a is 2*a (by symmetry)
    return grad


def multi_peak_shave(a, n_peaks=10, iters=5000):
    """
    Multi-peak shaving: simultaneously reduce the top n_peaks peaks.
    Uses adaptive step sizes based on the gap between peaks.
    """
    n = len(a)
    a = normalize_a(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = float(np.max(fast_conv(a)))
    
    for it in range(iters):
        c = fast_conv(a)
        M = float(c.max())
        if M < best_sc:
            best_sc = M
            best_a = a.copy()
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Weights: emphasize peaks that are above the "target"
        # Target is the average
        avg_c = float(c.sum()) / len(c)
        weights = np.maximum(0, c[sorted_idx] - avg_c)
        if weights.max() < 1e-10:
            break
        weights = weights / weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for k, w in zip(sorted_idx, weights):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += w * a[j_start:j_end + 1][::-1]
        grad *= 2.0
        
        # Polyak step: step = (M - target) / ||grad||^2
        # target = avg_c (we can't go below average since sum is fixed)
        target = avg_c
        gap = M - target
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = gap / (gn * gn + 1e-20)
        step = min(step, 1.0 / (float(a.sum()) + 1e-15))
        step = max(step, 1e-8 / gn)
        
        a -= step * grad
        a = np.clip(a, 0.0, None)
        # Re-normalize
        a = normalize_a(a)
    
    return best_sc, best_a


def spectral_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain, then project back to non-negative."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.fft(a, L)
    # Perturb phase and magnitude
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    A_mag *= (1.0 + strength * rng.standard_normal(L))
    A_mag = np.clip(A_mag, 0, None)
    A_phase += strength * 0.1 * rng.standard_normal(L)
    A_new = A_mag * np.exp(1j * A_phase)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.clip(a_new, 0.0, None)
    a_new = normalize_a(a_new)
    return a_new

        gap = sc - 2.0 * n * avg_c / (S * S)
        step0 = max(gap / (gn * gn + 1e-20), 1e-6 / (gn * S + 1e-20))
        step0 = min(step0, 2.0 / S)
        step = step0
        accepted = False
        for ls_iter in range(20):
            a_trial = a - step * grad
            a_trial = np.clip(a_trial, 0.0, None)
            # Renormalize to maintain sum
            S_trial = float(a_trial.sum())
            if S_trial < 1e-12:
                break
            a_trial = a_trial * (S / S_trial)
            sc_trial = compute_score(a_trial)
            if sc_trial < sc - 1e-12:
                a = a_trial
                accepted = True
                break
            step *= 0.5
        if not accepted:
            # Take a very small step anyway to escape stagnation
            step = 1e-7 / (gn * S + 1e-20)
            a -= step * grad
            a = np.clip(a, 0.0, None)
            S2 = float(a.sum())
            if S2 > 1e-12:
                a = a * (S / S2)

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.03):
    """Reduce elements contributing most to the peak, redistribute to others."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = max(5, n // 30)
    bottom_idx = np.argsort(contrib)[:bottom_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = float(a_pert.sum())
    if S > 1e-12:
        a_pert = a_pert * (float(a.sum()) / S)
    return a_pert


def spectral_perturb(a, rng, strength=0.05):
    """Perturb in frequency domain, project to non-negative."""
    n = len(a)
    L = 1 << (n - 1).bit_length()
    A = np.fft.fft(a, L)
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    A_mag_new = A_mag * (1.0 + strength * rng.standard_normal(L))
    A_mag_new = np.clip(A_mag_new, 0, None)
    A_phase_new = A_phase + strength * 0.05 * rng.standard_normal(L)
    A_new = A_mag_new * np.exp(1j * A_phase_new)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.clip(a_new, 0.0, None)
    S = float(a_new.sum())
    if S > 1e-12:
        a_new = a_new * (float(a.sum()) / S)
    else:
        a_new = a.copy()
    return a_new


def structured_perturb(a, rng, mode=0):
    """Various structured perturbations that preserve the sum."""
    n = len(a)
    S_orig = float(a.sum())
    a_new = a.copy()

    if mode == 0:
        # Gaussian
        a_new = a * (1.0 + 0.04 * rng.standard_normal(n))
    elif mode == 1:
        # Sinusoidal
        freq = 1.0 + 6.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        x = np.arange(n, dtype=np.float64) / n
        a_new = a * (1.0 + 0.04 * np.cos(2.0 * np.pi * freq * x + phase))
    elif mode == 2:
        # Tail boost (make ends larger, middle smaller)
        center = n // 2
        boost = 0.03 + 0.04 * rng.random()
        mid_start = center - n // 5
        mid_end = center + n // 5
        a_new[mid_start:mid_end] *= (1.0 - boost)
        mask = np.ones(n, dtype=bool)
        mask[mid_start:mid_end] = False
        a_new[mask] *= (1.0 + boost * 0.7)
    elif mode == 3:
        # Random block modulation
        num_blocks = 3 + int(4 * rng.random())
        block_size = max(10, n // (2 * num_blocks))
        for _ in range(num_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            sign = 1.0 if rng.random() > 0.5 else -1.0
            a_new[start:start + block_size] *= (1.0 + sign * (0.03 + 0.04 * rng.random()))
    elif mode == 4:
        # Exponential tilt
        rate = (rng.random() - 0.5) * 3.0
        x = np.arange(n, dtype=np.float64) / max(1, n - 1)
        tilt = np.exp(rate * (x - 0.5))
        a_new = a * tilt
    else:
        # Periodic modulation
        period = max(2, int(n / (3 + 5 * rng.random())))
        amp = 0.02 + 0.05 * rng.random()
        x = np.arange(n, dtype=np.float64)
        a_new = a * (1.0 + amp * np.sin(2.0 * np.pi * x / period))

    a_new = np.clip(a_new, 0.0, None)
    S = float(a_new.sum())
    if S > 1e-12:
        a_new = a_new * (S_orig / S)
    return a_new


def resize_sequence(a, new_n):
    """Resize sequence via linear interpolation, preserving non-negativity and sum."""
    n_old = len(a)
    S_old = float(a.sum())
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    S_new = float(a_new.sum())
    if S_new > 1e-12:
        a_new = a_new * (S_old / S_new)
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Strategy:
    - Phase 1: LSE continuation with high iteration budget (40%)
    - Phase 2: Subgradient peak shaving with line search (20%)
    - Phase 3: LSE refinement at high beta (10%)
    - Phase 4: Diverse restarts with multiple perturbation strategies (25%)
    - Phase 5: Final polish (5%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # State tracking (using list to avoid closure issues)
    state = {'best_score': np.inf, 'best_a': None}

    def report(sc, a, label):
        if sc < state['best_score']:
            state['best_score'] = sc
            state['best_a'] = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent score: {sc:.8f}, length={len(parent)}")
    else:
        n_init = 4096
        a_init = np.ones(n_init) * 0.5
        sc = compute_score(a_init)
        report(sc, a_init, "uniform init")

    if state['best_a'] is None:
        # Fallback - should not happen
        a_fb = np.ones(1024) * 0.5
        report(compute_score(a_fb), a_fb, "fallback")

    n_main = len(state['best_a'])
    print(f"Main sequence length: {n_main}")

    # ===================================================================
    # Phase 1: LSE Continuation (40% time)
    # ===================================================================
    phase1_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 1: LSE Continuation ===")

    a_cur = state['best_a'].copy()

    beta_schedule = [10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        remaining = phase1_end - time.monotonic()
        maxit = int(min(3000, remaining * 40))
        if maxit < 100:
            maxit = 100
        maxfun = maxit * 30
        a_cur = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxfun)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE b={beta}")
        print(f"  beta={beta}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s, iters_used~{maxit}")

    # ===================================================================
    # Phase 2: Subgradient Peak Shaving with Line Search (20% time)
    # ===================================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Subgradient Peak Shaving ===")

    if time.monotonic() < phase2_end:
        t0 = time.monotonic()
        iters = int(min(15000, (phase2_end - time.monotonic()) * 80))
        sc, a_shaved = subgradient_shave_line_search(state['best_a'], iters=max(2000, iters), n_peaks=5)
        report(sc, a_shaved, "subgrad k=5")
        print(f"  Subgrad k=5: t={time.monotonic()-t0:.1f}s")

        if time.monotonic() < phase2_end:
            t0 = time.monotonic()
            iters2 = int(min(10000, (phase2_end - time.monotonic()) * 60))
            sc, a_shaved = subgradient_shave_line_search(state['best_a'], iters=max(1000, iters2), n_peaks=10)
            report(sc, a_shaved, "subgrad k=10")
            print(f"  Subgrad k=10: t={time.monotonic()-t0:.1f}s")

        if time.monotonic() < phase2_end:
            t0 = time.monotonic()
            iters3 = int(min(5000, (phase2_end - time.monotonic()) * 40))
            sc, a_shaved = subgradient_shave_line_search(state['best_a'], iters=max(500, iters3), n_peaks=20)
            report(sc, a_shaved, "subgrad k=20")
            print(f"  Subgrad k=20: t={time.monotonic()-t0:.1f}s")

    # ===================================================================
    # Phase 3: LSE Refinement after shaving (10% time)
    # ===================================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: LSE Refinement ===")

    if time.monotonic() < phase3_end:
        a_cur = state['best_a'].copy()
        for beta in [5000, 10000, 20000]:
            if time.monotonic() > phase3_end:
                break
            t0 = time.monotonic()
            maxit = int(min(2000, (phase3_end - time.monotonic()) * 30))
            if maxit < 100:
                maxit = 100
            a_cur = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 25)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"refine b={beta}")
            print(f"  beta={beta}: score={sc:.8f}, t={time.monotonic()-t0:.1f}s")

        # Quick subgradient after LSE
        if time.monotonic() < phase3_end:
            sc, a_pol = subgradient_shave_line_search(state['best_a'], iters=3000, n_peaks=5)
            report(sc, a_pol, "post-LSE subgrad")

    # ===================================================================
    # Phase 4: Diverse Restarts (25% time)
    # ===================================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Diverse Restarts ===")

    restart_count = 0
    # Try different sequence lengths in restarts
    alt_lengths = [n_main, n_main, 8192, 4096, n_main]

    while time.monotonic() < phase4_end and restart_count < 60:
        t0 = time.monotonic()
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break

        strategy = restart_count % 7

        # Choose starting point
        if strategy == 6 and restart_count > 5:
            # Different length: resize best_a
            alt_n = alt_lengths[restart_count % len(alt_lengths)]
            if alt_n != n_main:
                a_start = resize_sequence(state['best_a'], alt_n)
                # Quick LSE at this length
                a_pert = run_lbfgs(a_start, 1000, maxiter=200, maxfun=4000)
            else:
                a_pert = state['best_a'].copy()
        else:
            a_pert = state['best_a'].copy()
            # Apply perturbation
            if strategy == 0:
                a_pert = structured_perturb(a_pert, rng, mode=0)
            elif strategy == 1:
                a_pert = structured_perturb(a_pert, rng, mode=2)
            elif strategy == 2:
                a_pert = spectral_perturb(a_pert, rng, strength=0.03 + 0.04 * rng.random())
            elif strategy == 3:
                a_pert = targeted_perturb(a_pert, rng, strength=0.02 + 0.04 * rng.random())
            elif strategy == 4:
                a_pert = structured_perturb(a_pert, rng, mode=1)
            elif strategy == 5:
                a_pert = spectral_perturb(a_pert, rng, strength=0.08 + 0.08 * rng.random())

        # Optimize from perturbed start
        if remaining > 60:
            a_opt = run_lbfgs(a_pert, 3000, maxiter=400, maxfun=8000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"r{restart_count} s{strategy}")

            # If improved or close, do subgradient polish
            if sc < state['best_score'] * 1.005 and remaining > 30:
                sc2, a_pol = subgradient_shave_line_search(a_opt, iters=2000, n_peaks=5)
                report(sc2, a_pol, f"r{restart_count} s{strategy}+sg")

                # And one more LSE pass
                if remaining > 20 and sc2 < state['best_score']:
                    a_opt2 = run_lbfgs(a_pol, 10000, maxiter=300, maxfun=6000)
                    sc3 = compute_score(a_opt2)
                    report(sc3, a_opt2, f"r{restart_count} s{strategy}+sg+LSE")
        elif remaining > 20:
            sc, a_pol = subgradient_shave_line_search(a_pert, iters=1000, n_peaks=5)
            report(sc, a_pol, f"r{restart_count} s{strategy} sg-only")

        restart_count += 1
        if restart_count % 10 == 0:
            print(f"  ... restart #{restart_count}, best={state['best_score']:.8f}, "
                  f"remaining={remaining:.0f}s")

    # ===================================================================
    # Phase 5: Final Polish
    # ===================================================================
    print(f"\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline:
        #Final LSE at very high beta
        t_rem = deadline - time.monotonic()
        maxit_final = int(min(1000, t_rem * 30))
        if maxit_final >= 50:
            a_final = run_lbfgs(state['best_a'], 15000, maxiter=maxit_final, maxfun=maxit_final * 20)
            sc = compute_score(a_final)
            report(sc, a_final, "final LSE b=15000")

        # Final subgradient
        if time.monotonic() < deadline:
            t_rem = deadline - time.monotonic()
            iters_final = int(min(5000, t_rem * 50))
            if iters_final >= 100:
                sc, a_pol = subgradient_shave_line_search(state['best_a'], iters=iters_final, n_peaks=5)
                report(sc, a_pol, "final subgrad")

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in state['best_a']]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={state['best_score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
