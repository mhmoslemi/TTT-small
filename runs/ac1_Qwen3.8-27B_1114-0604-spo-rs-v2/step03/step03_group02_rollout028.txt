1. Exact coordinate descent on the max-convolution objective (no smoothing needed for the main driver).

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_score_from_conv(c, S, n):
    """Compute score given precomputed convolution and sum."""
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Prevent overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000, deadline=None):
    """Run L-BFGS-B on LSE-smoothed score. Returns (exact_score, result_array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(50, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 40)))

    try:
        res = minimize(obj_wrapper, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coordinate_descent(a, n_passes=10, deadline=None, seed=42, verbose=False):
    """
    Coordinate descent on the exact max-convolution objective.
    For each coordinate j, try several candidate values and pick the best.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 0.5
        S = a.sum()
    c = fast_conv(a)
    best_sc = compute_score_from_conv(c, S, n)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    two_n_minus_1 = 2 * n - 1

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Randomize coordinate order
        order = rng.permutation(n)
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            t_old = a[j]
            if t_old < 1e-15:
                # Try small positive values
                candidates = [0.01, 0.05, 0.1, 0.5]
            else:
                # Generate candidates around current value
                candidates = [0.0, t_old * 0.3, t_old * 0.6, t_old * 0.8,
                              t_old * 1.2, t_old * 1.5, t_old * 2.0]
                # Also add subgradient-informed candidate
                # Find current max lag
                k_star = int(np.argmax(c))
                kj = k_star - j
                if 0 <= kj < n and k_star != 2 * j:
                    # Gradient suggests: if a[kj] > 0, reducing a[j] helps
                    # (since it reduces c[k_star])
                    pass
                # Add a candidate based on balancing
                candidates.append(t_old * 0.5)
                candidates = list(set(candidates))  # deduplicate
                candidates = [max(0.0, min(1000.0, x)) for x in candidates]

            # Determine affected range
            k_start = j
            k_end = min(j + n, two_n_minus_1)
            if k_end <= k_start:
                continue
            profile = a[0:k_end - k_start]  # a[k-j] for k in [k_start, k_end)
            has_sq = (2 * j < two_n_minus_1)
            sq_idx = 2 * j

            best_f = best_sc
            best_t = t_old

            for t in candidates:
                delta = t - t_old
                if abs(delta) < 1e-12 and t == t_old:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                # Apply update
                c[k_start:k_end] += 2.0 * delta * profile
                if has_sq:
                    c[sq_idx] += t * t - t_old * t_old
                f = compute_score_from_conv(c, S_new, n)
                # Rollback
                c[k_start:k_end] -= 2.0 * delta * profile
                if has_sq:
                    c[sq_idx] -= t * t - t_old * t_old
                if f < best_f:
                    best_f = f
                    best_t = t

            # Commit the best
            if best_t != t_old:
                delta = best_t - t_old
                c[k_start:k_end] += 2.0 * delta * profile
                if has_sq:
                    c[sq_idx] += best_t * best_t - t_old * t_old
                S += delta
                a[j] = best_t
                best_sc = best_f
                best_a = a.copy()

        if verbose and (p % 5 == 0):
            print(f"    CD pass {p+1}/{n_passes}: score={best_sc:.8f}")

    return best_sc, best_a


def kkt_soft_update(a, n_iters=200, alpha=0.5, beta_soft=50.0, deadline=None, verbose=False):
    """
    KKT-satisfying multiplicative update with soft active set.
    At optimum, h[j] = sum_k w_k * dc_k/da_j should be constant on support of a.
    Uses exp(soft) weighting for active lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 0.5
        S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    two_n_minus_1 = 2 * n - 1

    # Precompute FFT size for cross-correlation
    M_fft = 1 << (two_n_minus_1 - 1).bit_length()

    # Anderson acceleration storage
    a_0 = a.copy()
    h_0 = None
    m_anderson = 5
    a_history = [a.copy()]
    h_history = [None]

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break

        # Soft active set weights
        # w_k = exp(beta_soft * (c[k] - M)) / sum(...)
        # This gives weight 1 to the max and decays for others
        w = np.exp(beta_soft * (c - M))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(two_n_minus_1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Compute h[j] = sum_k w[k] * dc[k]/da[j]
        # dc[k]/da[j] = 2*a[k-j] (if valid, k!=2j) + a[j]*(if k==2j)
        # h[j] = sum_{k: 0<=k-j<n} w[k]*2*a[k-j] + a[j]*sum_{k: k=2j} w[k]
        #      = 2 * sum_{m: 0<=m<n, j+m < 2n-1} w[j+m]*a[m] + a[j]*w[2j] (if 2j valid)

        # First term: cross-correlation of w with a, evaluated at position j
        # h_cross[j] = sum_m w[j+m] * a[m] for valid m and j+m
        # This is the cross-correlation (w * a)[j] where we use the appropriate definition

        # Actually: sum_m w[j+m]*a[m] for m=0..n-1 and 0<=j+m<=2n-2
        # Let's compute via FFT: this is (w_convolve_RevA) at specific positions
        # Or more directly: it's the "cross-correlation" of w and a at offset j

        # h_cross[j] = sum_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * a[m]
        # This is the cross-correlation of w and a: (w ⊛ a)[j]

        # Using FFT: if W = rfft(w), A = rfft(a), then
        # circular cross-correlation at lag j is ifft(W * conj(A))[j]
        # But we need linear, so use zero-padding (already done via M_fft >= 2n-1)

        RA = np.fft.rfft(a, M_fft)
        W = np.fft.rfft(w, M_fft)
        # cross[j] = sum_m w[j+m]*a[m] = sum_m a[m]*w[j+m]
        # This is the cross-correlation at lag j
        # In FFT: ifft(A * conj(W)) gives circular cross-correlation
        # Let's verify: circular cross-correlation at lag j = sum_m a[m+j]*a'[m]
        # We want sum_m w[j+m]*a[m] = sum_m a[m]*w[m+j]
        # = circular correlation of a with w at lag j (if we define corr_a_w[j] = sum a[m]*w[m+j])
        # = ifft(conj(A) * W)[j] ... let me just compute it directly for clarity
        # Actually for n=8192, direct computation is O(n^2) which is too slow.
        # Let's use the FFT approach carefully.

        # sum_m a[m]*w[m+j] for m = 0 to n-1-j (where w[m+j] is defined)
        # This is the "cross-correlation" (a ⋆ w)[j] = sum_m a[m] * w[m+j]
        # In frequency: IFFT(FFT(a) * conj(FFT(w)))... no.
        #
        # Let's use the standard formula:
        # (a ⋆ w)[j] = sum_m a[m] * conj(w[m-j]) (for real: a[m]*w[m-j])
        # That's not quite what we want.
        #
        # What we want: h[j] = sum_{m=0}^{n-1} a[m] * w[m+j] (where w is defined on [0, 2n-2])
        # = sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * w[m+j]
        #
        # Let u[m] = w[m+j] for m in valid range. This is a shifted version of w.
        # So h[j] = dot(a[:min(n, 2n-1-j)], w[j:j+min(n,2n-1-j)])
        #
        # This is the cross-correlation of a and w at "negative lag" j:
        # If we define (a * w_rev)[k] where w_rev is reversed w...
        #
        # Simplest FFT approach:
        # h[j] = sum_m a[m] * w[m+j] (valid range)
        # Let b[m] = w[m] (length 2n-1), then h[j] = sum_m a[m] * b[m+j]
        # This is the cross-correlation (a ⋆ b)[j] = sum_m a[m]*b[m+j]
        # In FFT: let A = FFT(a padded to M_fft), B = FFT(b padded to M_fft)
        # Then ifft(A * conj(B)) gives circular cross-correlation (a ⋆ b)[j] = sum a[m+j]*b[m]... 
        # I'm getting confused with conventions. Let me just use the direct formula with FFT:
        #
        # Define: cross[j] = sum_{m} a[m] * w[m+j]
        # This equals (circular convolution of a with reversed-w) at a certain index.
        # Or: it's the "uncorrelated" version.
        #
        # Let me just compute it as: cross = ifft(fft(a, M) * fft(w_shifted, M)) ... no.
        #
        # The cleanest way: 
        # cross[j] = sum_m a[m] * w[m+j]
        # Let v[m] = w[m] (length 2n-1, zero-padded to M_fft)
        # Then cross[j] = sum_m a[m] * v[m+j] = (a convolved with v_flipped) at index (2n-2-j) ... 
        #
        # OK let me just use a different formulation:
        # cross[j] = sum_m a[m] * w[m+j] = sum_k w[k] * a[k-j] (substituting k=m+j)
        # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (where a is defined on [0,n-1])
        # 
        # This is the "cross-correlation" that appears in signal processing:
        # (w ⋆ a)[j] = sum_k w[k] * a[k-j] (for valid range)
        # = ifft(rFFT(w) * conj(rFFT(a)))... 
        #
        # Standard: R_xy[k] = sum_n x[n+k]*y[n] = ifft(X * conj(Y))[k] (circular, with padding for linear)
        # Here we have: sum_k w[k]*a[k-j] = sum_n a[n]*w[n+j] (letting n=k-j)
        # = R_aw[j] = ifft(FFT(a) * conj(FFT(w)))[j] (linear, with zero-padding)

        A_fft = np.fft.rfft(a, M_fft)
        W_fft = np.fft.rfft(w, M_fft)
        cross = np.fft.irfft(A_fft * np.conj(W_fft), M_fft)
        # cross[j] = sum_m a[m] * w[m+j] (approximately, for j in valid range with padding ensuring linearity)

        # Second term: a[j] * w[2j] (if 2j is valid)
        sq_term = np.zeros(n)
        valid_sq = 2 * np.arange(n) < two_n_minus_1
        sq_term[valid_sq] = a[valid_sq] * w[2 * np.arange(n)[valid_sq]]

        h = 2.0 * cross[:n] + sq_term
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Damped multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a_new = a * factor
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new * (S / S_new)  # preserve sum

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        if verbose and (it % 50 == 0):
            print(f"    KKT iter {it}/{n_iters}: score={sc:.8f}, h_var={h.std():.6f}")

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Find which j's contribute to c[k_star]
    # c[k_star] = sum_j a[j]*a[k_star-j]
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[k_star - j_end:k_star - j_start + 1]  # a[k_star - j]
    top_count = max(5, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def exact_subgrad_polish(a, total_iters=300, deadline=None):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # theoretical lower bound for our purposes

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max at k_star
        # dc[k_star]/da[j] = 2*a[k_star-j] (if valid, k_star != 2j) + a[j] (if k_star == 2j)
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[k_star - j_end:k_star - j_start + 1]
            if j_start <= 2 * j_start - 2 * j_start + k_star <= j_end:  # check if k_star = 2j for some j
                pass
        # Handle the k=2j case: if k_star is even, j = k_star/2
        if k_star % 2 == 0 and 0 <= k_star // 2 < n:
            j_sq = k_star // 2
            sg[j_sq] += a[j_sq]  # extra a[j]^2 term gives derivative 2*a[j], we already have 2*a[j] from the cross term
            # Wait: dc[k]/da[j] when k=2j: the a[j]^2 term gives 2*a[j], and the cross term also gives 2*a[j]
            # So total is 4*a[j]? No...
            # c[k] = sum_i a[i]*a[k-i]. When k=2j: the term i=j gives a[j]*a[j]=a[j]^2.
            # dc[2j]/da[j]: a[j] appears once as a[i] (i=j) and once as a[k-i] (i=k-j=j), so it's a[j]^2.
            # d(a[j]^2)/da[j] = 2*a[j].
            # But for i != j: a[i]*a[k-i] doesn't involve a[j].
            # So dc[2j]/da[j] = 2*a[j] (from the a[j]^2 term).
            # For i where k-i = j (i.e., i = k-j = j): same term.
            # So the total derivative is 2*a[j] (not 4*a[j]).
            # But in my formula above, I have sg[j] = 2*a[k-j] = 2*a[j] for the cross term.
            # And I'd be adding another a[j] which is wrong.
            # Let me fix: for k=2j, the correct derivative is 2*a[j] (from a[j]^2).
            # My formula gives 2*a[k-j] = 2*a[j], which is correct. No extra term needed.
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64).ravel()
        if len(p) != n:
            x_old = np.linspace(0, 1, len(p))
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.sin(3.0 * np.pi * x) ** 2)
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    # Triangular
    inits.append(np.abs(1.0 - 2.0 * x))
    # Bimodal
    inits.append(0.3 + 0.7 * np.exp(-10.0 * (x - 0.3)**2) + 0.7 * np.exp(-10.0 * (x - 0.7)**2))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0  # 20s buffer

    # Load parent solution if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    def report(sc, arr, label):
        nonlocal best_score, best_a
        if sc < best_score and sc < np.inf:
            best_score = sc
            best_a = arr.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(arr)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [8192]
    if parent is not None:
        pa = parent.size
        if 64 <= pa <= 16384 and pa != 8192:
            lengths.append(pa)
    lengths = sorted(set(lengths))
    lengths = lengths[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}" + (f" (len={parent.size})" if parent is not None else ""))

    # ============================================================
    # Phase 1: LSE-smoothed L-BFGS with continuation
    # ============================================================
    phase1_deadline = t_start + 0.30 * total_budget
    beta_schedule = [30, 80, 200, 500]
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE L-BFGS continuation (deadline at t={phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- Phase 1: n = {n} ---")
        inits = build_inits(n, rng, parent)
        # Limit initializations
        n_init_budget = max(2, int((phase1_deadline - time.monotonic()) / (total_budget * 0.04)))
        inits = inits[:min(n_init_budget, len(inits))]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            a_cur = a0.copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_deadline:
                    break
                remaining = phase1_deadline - time.monotonic()
                if remaining < 5.0:
                    break
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=50000,
                                       deadline=phase1_deadline)
                a_cur = a_opt
                improved = report(sc, a_opt, f"n={n},init={i},b={beta}")
                if not improved and bi > 1:
                    break

    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ============================================================
    # Phase 2: Coordinate descent (main workhorse)
    # ============================================================
    phase2_deadline = t_start + 0.55 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Coordinate Descent (deadline at t={phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_cd = len(a_cur)
