Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Multi-peak subgradient with Nesterov momentum and adaptive step size
2. Spectral density shaping to reduce long-range correlation
3. Adaptive n selection based on progress
4. Block-coordinate refinement for top-contributing positions

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def power_mean_score_grad(a, p):
    """Power-mean smoothed score and gradient. M_p = (sum c^p)^(1/p)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    # Clip c to avoid issues
    c_pos = np.clip(c, 0, None)
    M_p = (np.sum(c_pos ** p)) ** (1.0 / p)
    score = 2.0 * n * M_p / (S * S)
    # Gradient: dM_p/da[m] = M_p^(1-p) * sum_k c[k]^(p-1) * 2*a[k-m]*1_{valid}
    w = c_pos ** (p - 1)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    W = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * W, M2_fft)[n - 1:2 * n - 1]
    g_Mp = 2.0 * conv_res * (M_p ** (1 - p))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_Mp / S2 - 2.0 * M_p / S3)
    return score, grad

def subgrad_multi_peak(a, max_iters=5000, top_k=8, momentum_coeff=0.85):
    """Multi-peak subgradient descent with momentum and Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        a[:] = 1.0 / n
        return compute_score(a), a
    # Work with normalized sequence (sum=1) for better conditioning
    a = a / S0
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    v = np.zeros(n)

    for it in range(max_iters):
        c = fast_conv(a)
        S = 1.0
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top-k peaks
        sorted_idx = np.argsort(c)[::-1]
        peak_idx = sorted_idx[:top_k]
        peak_vals = c[peak_idx]
        # Weights: relative to max
        weights = peak_vals / M

        # Compute averaged subgradient
        sg = np.zeros(n)
        for ki in range(len(peak_idx)):
            k = peak_idx[ki]
            w = weights[ki]
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            # sg[j] += w * 2 * a[k-j]
            sg[j_start:j_end + 1] += w * 2.0 * a[k - j_end:k - j_start][::-1]

        # Average over peaks
        sg /= (weights.sum() + 1e-15)

        # Center for sum preservation
        sg_centered = sg - sg.mean()

        # Step size (Polyak-like)
        gap = sc - sc_lb
        gn2 = float(sg_centered @ sg_centered)
        if gn2 < 1e-30:
            break
        step = gap / gn2
        step = min(step, 0.3 / (1.0 + 0.002 * it))

        # Momentum
        v = momentum_coeff * v + sg_centered
        a = a - step * v
        a = np.clip(a, 0.0, None)
        S_new = a.sum()
        if S_new > 1e-12:
            a = a / S_new
        else:
            a = best_a.copy()
            break

    # Scale back
    best_a = best_a * S0
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.04):
    """Reduce elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    a_pert = a.copy()
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_count = max(5, n // 80)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_idx = np.argsort(contrib)[:-top_count]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)

            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [8192, 6962, 10240, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes n=' + str(parent.size) if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: LSE-continuation L-BFGS from parent & structured inits
    # ============================================================
    phase1_end = t_start + 0.40 * total_budget
    print(f"\n=== Phase 1: LSE-Continuation (until t={phase1_end - t_start:.0f}s) ===")

    beta_schedule = [50, 150, 400, 1000, 3000, 8000, 15000]

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  n={n}")

        # Build inits
        inits = []
        if parent is not None and len(parent) == n:
            inits.append(parent.copy())
        inits.extend(build_structured_inits(n, rng))

        # Try each init through beta schedule
        n_inits = min(len(inits), 8)
        time_per_init = (phase1_end - time.monotonic()) / max(1, n_inits)

        for i in range(n_inits):
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            a_cur = inits[i].copy()
            for beta in beta_schedule:
                if time.monotonic() > phase1_end:
                    break
                remaining = phase1_end - time.monotonic()
                maxit = max(300, min(8000, int(remaining / (0.005 + 0.0005 * n))))
                sc, a_opt = run_lbfgs_lse(a_cur, beta, maxiter=maxit)
                a_cur = a_opt
                report(sc, a_opt, f"P1 n={n} i={i} b={beta}")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Power-mean smoothing (alternative landscape)
    # ============================================================
    phase2_end = t_start + 0.60 * total_budget
    print(f"\n=== Phase 2: Power-Mean Smoothing (until t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    for p_val in [4, 8, 16, 32]:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        maxit = max(500, min(5000, int(remaining / (0.01 + 0.001 * n_best))))
        sc, a_opt = run_lbfgs_pm(best_a, p_val, maxiter=maxit)
        report(sc, a_opt, f"P2 pm p={p_val}")
        # After power-mean, refine with LSE
        if sc < best_score:
            sc2, a_opt2 = run_lbfgs_lse(a_opt, 5000, maxiter=3000)
            report(sc2, a_opt2, f"P2 pm+LSE p={p_val}")

    print(f"\n  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Multi-peak subgradient with restarts
    # ============================================================
    phase3_end = t_start + 0.78 * total_budget
    print(f"\n=== Phase 3: Multi-Peak Subgradient (until t={phase3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    for trial in range(5):
        if time.monotonic() > phase3_end:
            break
        # Slightly perturb the best solution
        scale = 0.02 + 0.02 * trial
        a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a.copy()

        n_iters = max(1000, int((phase3_end - time.monotonic()) / 5 / (0.001 + 0.0002 * n_best)))
        sc, a_opt = subgrad_multi_peak(a_pert, max_iters=n_iters, top_k=8, momentum_coeff=0.85)
        report(sc, a_opt, f"P3 subgrad #{trial}")

        # Refine with LSE
        if sc < best_score:
            sc2, a_opt2 = run_lbfgs_lse(a_opt, 10000, maxiter=3000)
            report(sc2, a_opt2, f"P3 subgrad+LSE #{trial}")

    print(f"\n  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Targeted local search
    # ============================================================
    phase4_end = t_start + 0.92 * total_budget
    print(f"\n=== Phase 4: Targeted Local Search (until t={phase4_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase4_end and trial < 20:
        trial += 1
        strat = trial % 5
        if strat == 0:
            # Gaussian perturbation
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.03 * rng.random())
        elif strat == 2:
            # Anti-palindromic bump
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 12)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.1 * sign * np.exp(-d * d))
        elif strat == 3:
            # Frequency smoothing
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.15 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Block swap: swap two random blocks
            a_pert = best_a.copy()
            blk = max(10, n_best // 200)
            i1 = rng.integers(0, n_best - 2 * blk)
            i2 = rng.integers(0, n_best - 2 * blk)
            if abs(i1 - i2) > blk:
                tmp = a_pert[i1:i1 + blk].copy()
                a_pert[i1:i1 + blk] = a_pert[i2:i2 + blk]
                a_pert[i2:i2 + blk] = tmp

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        remaining = phase4_end - time.monotonic()
        maxit = max(500, min(5000, int(remaining / 10 / (0.003 + 0.0003 * n_best))))
        sc, a_opt = run_lbfgs_lse(a_pert, 5000, maxiter=maxit)
        report(sc, a_opt, f"P4 #{trial} s={strat}")

    print(f"\n  After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final high-beta refinement + subgradient
    # ============================================================
    print(f"\n=== Phase 5: Final Refinement ===")

    n_best = len(best_a)
    for beta in [20000, 50000]:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - 20 - time.monotonic()
        maxit = max(2000, min(10000, int(remaining / (0.005 + 0.0005 * n_best))))
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=maxit)
        report(sc, a_opt, f"P5 LSE b={beta}")

    # Final subgradient polish
    if time.monotonic() < deadline - 10:
        n_iters = max(2000, int((deadline - 10 - time.monotonic()) * 50))
        sc, a_opt = subgrad_multi_peak(best_a, max_iters=n_iters, top_k=10, momentum_coeff=0.9)
        report(sc, a_opt, "P5 final subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
