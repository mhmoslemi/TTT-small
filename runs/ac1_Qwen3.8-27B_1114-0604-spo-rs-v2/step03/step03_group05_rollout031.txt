def run_lbfgs(a0, func, args, maxiter=1000, maxfun=30000):
    """Run L-BFGS-B optimization."""

def qp_peak_reduction(a, c_peak, S, mu=1.0):
    """
    Solve QP: min sum((x_j - a_j)^2) + mu * sum c_j x_j
    s.t. sum x_j = S, x_j >= 0
    
    Closed form: x_j = max(0, a_j - mu*c_j/2 - lambda/2)
    Find lambda so sum(x) = S via bisection.
    """
    n = len(a)
    
    def sum_x(lam):
        x = np.maximum(0.0, a - mu * c_peak / 2.0 - lam / 2.0)
        return x.sum()
    
    # Find lambda range
    lo = a.min() - mu * c_peak.max() / 2.0 - a.max()
    hi = a.max() + mu * c_peak.min() / 2.0 + a.max()
    
    # Ensure sum_x(lo) > S and sum_x(hi) < S
    for _ in range(20):
        if sum_x(lo) < S:
            lo -= 1.0
        if sum_x(hi) > S:
            hi += 1.0
        if sum_x(lo) >= S and sum_x(hi) <= S:
            break
    
    try:
        lam = brentq(lambda l: sum_x(l) - S, lo, hi, xtol=1e-14, maxiter=100)
    except Exception:
        lam = (lo + hi) / 2.0
    
    x = np.maximum(0.0, a - mu * c_peak / 2.0 - lam / 2.0)
    # Renormalize to exactly sum S (handle numerical issues)
    if x.sum() > 1e-15:
        x = x * S / x.sum()
    return x

"""
Step function optimization: QP-based peak reduction with adaptive LSE continuation.
Key innovations over previous approach:
1. Closed-form QP (water-filling) for exact peak reduction - much faster convergence
2. Adaptive top-k peak targeting
3. Better LSE beta schedule with convergence detection
4. Diversified restarts with QP finishing
5. Hybrid QP+LSE polishing

import numpy as np
from scipy.optimize import minimize, brentq
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    L_fft = 1 << (3 * n - 2).bit_length()
    RA = np.fft.rfft(rev_a, L_fft)
    P = np.fft.rfft(p, L_fft)
    conv_res = np.fft.irfft(RA * P, L_fft)[:3 * n - 2]
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy minimize with args=(beta,)."""
    return lse_score_and_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=1000, maxfun=30000):
    """Run L-BFGS-B optimization. Returns optimized array."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def run_trf(a0, func, args, maxiter=500, maxfun=5000):
    """Run trust-region reflective optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='trf', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-10,
                                'maxiter': maxiter, 'maxfun': maxfun})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def qp_solve_peak(a, c_vec, S, mu):
    """
    Solve: min sum((x_j - a_j)^2) + mu * sum(c_j * x_j)
    s.t. sum(x_j) = S, x_j >= 0
    
    Closed form: x_j = max(0, a_j - mu*c_j/2 - lam/2)
    Find lam via bisection.
    """
    n = len(a)
    offset = a - mu * c_vec / 2.0

    def sum_x(lam):
        return np.maximum(0.0, offset - lam / 2.0).sum()

    # Bracket finding
    lo = -2.0 * (float(offset.max()) + 1.0)
    hi = 2.0 * (float(-offset.min()) + 1.0) + 2.0

    # Expand bracket if needed
    for _ in range(30):
        if sum_x(lo) <= S:
            lo -= (hi - lo) + 1.0
        if sum_x(hi) >= S:
            hi += (hi - lo) + 1.0
        if sum_x(lo) > S and sum_x(hi) < S:
            break

    try:
        lam = brentq(lambda l: sum_x(l) - S, lo, hi, xtol=1e-14, maxiter=80)
    except Exception:
        lam = (lo + hi) / 2.0

    x = np.maximum(0.0, offset - lam / 2.0)
    sx = x.sum()
    if sx > 1e-15:
        x = x * (S / sx)
    return x


def multi_qp_reduce(a, top_k, iters, mu_init, mu_decay, step_size, score_threshold):
    """
    Iteratively reduce top peaks using QP. Returns (best_score, best_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    mu = mu_init
    no_improve = 0

    for it in range(iters):
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200 and it > 100:
            break
        if best_sc < score_threshold:
            break

        # Find top peaks
        sorted_idx = np.argsort(c)[::-1][:top_k]

        for k_star in sorted_idx:
            k_star = int(k_star)
            c_vec = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                c_vec[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

            x = qp_solve_peak(a, c_vec, S, mu)
            a = (1.0 - step_size) * a + step_size * x
            a = np.clip(a, 0.0, 1000.0)
            S = float(a.sum())
            if S < 1e-6:
                a[:] = 0.5
                S = float(a.sum())

        mu *= mu_decay
        mu = max(mu, 0.0005)

    return best_sc, best_a

        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0001) / (gn * gn + 1e-20)
        step = min(step, 0.3 / S)
        step = max(step, 1e-8 / (gn * S + 1e-20))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def targeted_perturb(a, rng, strength):
    """Reduce peak contributors, redistribute to low areas."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(5, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def multi_qp_reduce_v2(a, iters, step_size):
    """
    Enhanced QP reduction with adaptive top-k and step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for it in range(iters):
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300 and it > 200:
            break

        # Adaptive top-k: start with fewer, increase as peaks flatten
        c_range = float(c.max()) - float(c.min())
        if it < 100:
            top_k = 3
        elif it < 500:
            top_k = 5
        else:
            top_k = 8

        sorted_idx = np.argsort(c)[::-1][:top_k]

        # Adaptive step size
        if no_improve > 50:
            cur_step = step_size * 0.5
        elif no_improve > 20:
            cur_step = step_size * 0.7
        else:
            cur_step = step_size

        for k_star in sorted_idx:
            k_star = int(k_star)
            c_vec = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                c_vec[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

            # Adaptive mu: larger when peaks are concentrated
            peak_ratio = float(c[k_star]) / (float(c.sum()) / (2 * n - 1) + 1e-30)
            mu = 0.05 * min(peak_ratio, 10.0) + 0.01

            x = qp_solve_peak(a, c_vec, S, mu)
            a = (1.0 - cur_step) * a + cur_step * x
            a = np.clip(a, 0.0, 1000.0)
            S = float(a.sum())
            if S < 1e-6:
                a[:] = 0.5
                S = float(a.sum())

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for optimal non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses adaptive LSE, closed-form QP peak reduction, and diversified restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # State tracking (avoiding closure)
    state = {'best_score': np.inf, 'best_a': None}

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc = compute_score(parent)
        state['best_score'] = sc
        state['best_a'] = parent.copy()
        n = len(parent)
        print(f"Parent: score={sc:.8f}, n={n}")
    else:
        n = 4096
        state['best_a'] = np.ones(n) * 0.5
        state['best_score'] = compute_score(state['best_a'])
        print(f"Uniform init: score={state['best_score']:.8f}, n={n}")

    def try_report(sc, a, label):
        if sc < state['best_score']:
            state['best_score'] = sc
            state['best_a'] = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    a_cur = state['best_a'].copy()
    n = len(a_cur)
    effective_budget = total_budget - 30.0

    # ===================================================================
    # Phase 1: LSE Continuation - 20% time
    # ===================================================================
    phase1_end = t_start + 0.20 * effective_budget
    print(f"\n=== Phase 1: LSE Continuation (until t={phase1_end - t_start:.0f}s) ===")

    beta_schedule = [15, 30, 60, 120, 250, 500, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_objective, (beta,), maxiter=1000, maxfun=25000)
        sc = compute_score(a_cur)
        try_report(sc, a_cur, f"LSE b={beta}")
        print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 2: QP Peak Reduction (main) - 30% time
    # ===================================================================
    phase2_end = t_start + 0.50 * effective_budget
    print(f"\n=== Phase 2: QP Peak Reduction (until t={phase2_end - t_start:.0f}s) ===")

    if time.monotonic() < phase2_end:
        t0 = time.monotonic()
        available = phase2_end - time.monotonic()
        # Estimate: each iteration ~5ms for n~7000 with 5 peaks
        est_iters = int(min(available * 150, 15000))
        est_iters = max(500, est_iters)
        sc, a_qp = multi_qp_reduce_v2(state['best_a'], iters=est_iters, step_size=0.35)
        try_report(sc, a_qp, "QP main")
        print(f"  QP main: {sc:.8f} ({time.monotonic()-t0:.1f}s, {est_iters} iters)")

        # Second round of QP from updated best
        if time.monotonic() < phase2_end:
            t0 = time.monotonic()
            available2 = phase2_end - time.monotonic()
            est_iters2 = int(min(available2 * 150, 10000))
            est_iters2 = max(300, est_iters2)
            sc2, a_qp2 = multi_qp_reduce(state['best_a'], top_k=5, iters=est_iters2,
                                          mu_init=0.1, mu_decay=0.9995, step_size=0.25,
                                          score_threshold=state['best_score'] - 0.001)
            try_report(sc2, a_qp2, "QP round2")
            print(f"  QP round2: {sc2:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 3: High-beta LSE refinement - 15% time
    # ===================================================================
    phase3_end = t_start + 0.65 * effective_budget
    print(f"\n=== Phase 3: High-Beta LSE (until t={phase3_end - t_start:.0f}s) ===")

    a_cur = state['best_a'].copy()
    for beta in [2000, 5000, 10000, 20000]:
        if time.monotonic() > phase3_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_objective, (beta,), maxiter=2000, maxfun=50000)
        sc = compute_score(a_cur)
        try_report(sc, a_cur, f"hiLSE b={beta}")
        print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 4: QP + Subgradient combined polish - 15% time
    # ===================================================================
    phase4_end = t_start + 0.80 * effective_budget
    print(f"\n=== Phase 4: Combined Polish (until t={phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end:
        # QP polish with smaller steps
        t0 = time.monotonic()
        available = phase4_end - time.monotonic()
        est_iters = int(min(available * 100, 8000))
        est_iters = max(300, est_iters)
        sc, a_polish = multi_qp_reduce_v2(state['best_a'], iters=est_iters, step_size=0.2)
        try_report(sc, a_polish, "QP polish")
        print(f"  QP polish: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

        # Subgradient polish
        if time.monotonic() < phase4_end:
            t0 = time.monotonic()
            iters_sg = int((phase4_end - time.monotonic()) * 80)
            iters_sg = max(500, min(iters_sg, 5000))
            sc_sg, a_sg = subgradient_shave(state['best_a'], iters=iters_sg)
            try_report(sc_sg, a_sg, "subgrad")
            print(f"  Subgrad: {sc_sg:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 5: Diversified restarts - 15% time
    # ===================================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Restarts (until t={phase5_end - t_start:.0f}s) ===")

    n_best = len(state['best_a'])
    restart_count = 0
    while time.monotonic() < phase5_end and restart_count < 40:
        t0 = time.monotonic()
        strategy = restart_count % 7

        if strategy == 0:
            # Small Gaussian
            scale = 0.008 + 0.015 * rng.random()
            a_pert = state['best_a'] * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted
            a_pert = targeted_perturb(state['best_a'], rng, strength=0.015 + 0.03 * rng.random())
        elif strategy == 2:
            # Sinusoidal
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.015 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = state['best_a'] * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Frequency domain
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.fft(state['best_a'], L)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            A_mag_new = A_mag * (1.0 + 0.015 * rng.standard_normal(L))
            A_mag_new = np.clip(A_mag_new, 0, None)
            A_new = A_mag_new * np.exp(1j * A_phase)
            a_pert = np.fft.ifft(A_new).real[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Aggressive peak reduction then QP
            c = fast_conv(state['best_a'])
            top_peaks = np.argsort(c)[::-1][:5]
            a_pert = state['best_a'].copy()
            for k_star in top_peaks:
                k_star = int(k_star)
                j_s = max(0, k_star - n_best + 1)
                j_e = min(k_star, n_best - 1)
                if j_e >= j_s:
                    seg = a_pert[j_s:j_e + 1].copy()
                    top_in_seg = np.argsort(seg)[::-1][:max(3, len(seg) // 15)]
                    for idx in top_in_seg:
                        a_pert[j_s + idx] *= 0.80
            S_t = float(state['best_a'].sum())
            S_p = float(a_pert.sum())
            if S_p > 0 and S_p < S_t:
                deficit = S_t - S_p
                peak_mask = np.zeros(n_best, dtype=bool)
                for k_star in top_peaks:
                    k_star = int(k_star)
                    j_s = max(0, k_star - n_best + 1)
                    j_e = min(k_star, n_best - 1)
                    peak_mask[j_s:j_e + 1] = True
                non_peak = ~peak_mask
                if non_peak.sum() > 0:
                    a_pert[non_peak] += deficit / non_peak.sum()
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Antisymmetric perturbation (break symmetry)
            half = n_best // 2
            a_pert = state['best_a'].copy()
            noise = rng.standard_normal(half) * 0.02
            a_pert[:half] += noise
            a_pert[-half:] -= noise[::-1]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block exchange
            bs = max(10, n_best // 8)
            i1 = rng.integers(0, n_best - 2 * bs)
            i2 = rng.integers(0, n_best - 2 * bs)
            if abs(i1 - i2) > bs:
                a_pert = state['best_a'].copy()
                tmp = a_pert[i1:i1 + bs].copy()
                a_pert[i1:i1 + bs] = a_pert[i2:i2 + bs]
                a_pert[i2:i2 + bs] = tmp
            else:
                a_pert = state['best_a'].copy()
                a_pert[:bs] += 0.01 * rng.standard_normal(bs)
                a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LSE refinement
        a_opt = run_lbfgs(a_pert, lse_objective, (3000,), maxiter=400, maxfun=10000)
        sc = compute_score(a_opt)
        try_report(sc, a_opt, f"R{restart_count} LSE")

        # QP finishing from the refined point
        if time.monotonic() < phase5_end:
            remaining = phase5_end - time.monotonic()
            if remaining > 2.0:
                qp_iters = int(min(remaining * 80, 1500))
                qp_iters = max(100, qp_iters)
                sc2, a_qp_f = multi_qp_reduce_v2(a_opt, iters=qp_iters, step_size=0.25)
                try_report(sc2, a_qp_f, f"R{restart_count} +QP")

        restart_count += 1
        if restart_count % 5 == 0:
            print(f"  After {restart_count} restarts: best={state['best_score']:.8f}")

    # ===================================================================
    # Final: Verify and output
    # ===================================================================
    best_a_final = state['best_a']
    out = [float(max(0.0, min(1000.0, x))) for x in best_a_final]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={state['best_score']:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
