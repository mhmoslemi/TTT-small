```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)
```

```python
"""
LP-direction + coordinate descent optimizer for minimum peak linear autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Uses the LP-based direction-finding approach: at each step, solve an LP to find
the best direction that doesn't increase the peak convolution, then take a step
in that direction. Combined with coordinate descent polishing and perturbation.
"""

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def solve_lp_direction(f, M, active_lags):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    n = len(f)
    g_var = cp.Variable(n)
    constraints = [g_var >= 0]
    
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_vals = np.arange(j_start, j_end + 1)
        coeffs = f[k - j_vals]
        constraints.append(coeffs @ g_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        try:
            prob.solve(verbose=False)
        except:
            return None
    
    if g_var.value is not None:
        return np.clip(np.asarray(g_var.value, dtype=np.float64), 0, None)
    return None


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, order=None):
    """One full sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if order is None:
        order = np.arange(n)
    
    for i in order:
        i = int(i)
        a_i = a[i]
        
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)
        
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a

def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

            lp_fail_count = 0

            g_scaled = (S_cur / g_opt.sum()) * g_opt

            best_t = 0.1

                a_trial = np.clip(a_trial, 0.0, 1000.0)
                sc = compute_score(a_trial)
                if sc < best_step_score:

                    best_t = t

            a_cur = np.clip(a_cur, 0.0, 1000.0)

    phase3_deadline = t_start + 0.90 * total_budget

        elif strategy == 1:

            a_pert[start:start + block_size] = 0.0

            rest_mask[start:start + block_size] = False
            a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
        elif strategy == 3:

            a_pert = best_a.copy()
            edge = max(5, n_cur // 50)

            a_pert[:edge] *= boost

        else:

            a_pert = best_a.copy()
            kernel_size = max(3, n_cur // 200)

            a_pert += smooth_noise * best_a.mean()

        # Quick LP step
        conv_pert = fast_conv(a_pert)
        M_pert = float(np.max(conv_pert))
        S_pert = float(a_pert.sum())
        K = min(300, 2 * n_cur - 1)
        active_lags = np.argsort(conv_pert)[::-1][:K]
        g_opt = solve_lp_direction(a_pert, M_pert, active_lags)
        
        if g_opt is not None and g_opt.sum() > S_pert * 0.9:
            g_scaled = (S_pert / g_opt.sum()) * g_opt
            a_pert = 0.7 * a_pert + 0.3 * g_scaled
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # CD polishing
        for sw in range(5):
            if time.monotonic() > phase3_deadline - 10.0:
                break
            if sw % 2 == 0:
                sc_pert, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)
            else:
                sc_pert, a_pert = coordinate_descent_sweep(a_pert)
        
        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P3 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 5 == 0:
            print(f"  P3 trial {trial} (s={strategy}): {sc_pert:.8f}")
    
    # ============================================================
    # Phase 4: Final polish (10% of time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")
    
    a_cur = best_a.copy()
    for _ in range(30):
        if time.monotonic() > deadline:
            break
        sc, a_cur = coordinate_descent_sweep(a_cur, order=rng.permutation(n_work))
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
    
    # One more LP + CD if time permits
    if time.monotonic() < deadline - 30.0:
        conv_cur = fast_conv(best_a)
        M_cur = float(np.max(conv_cur))
        K = min(300, 2 * len(best_a) - 1)
        active_lags = np.argsort(conv_cur)[::-1][:K]
        g_opt = solve_lp_direction(best_a, M_cur, active_lags)
        if g_opt is not None and g_opt.sum() > best_a.sum() * 0.9:
            S_cur = float(best_a.sum())
            g_scaled = (S_cur / g_opt.sum()) * g_opt
            for t in [0.05, 0.1, 0.15]:
                a_trial = (1 - t) * best_a + t * g_scaled
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                sc = compute_score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    print(f"  Final LP (t={t}): {sc:.8f} (NEW BEST)")
                    break
            # CD after
            a_cur = best_a.copy()
            sc, a_cur = coordinate_descent_sweep(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} (NEW BEST)")
    
    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"LP solves: {lp_solve_count}, LP fails: {lp_fail_count}")
    print(f"{'=' * 60}")
    return out
```
