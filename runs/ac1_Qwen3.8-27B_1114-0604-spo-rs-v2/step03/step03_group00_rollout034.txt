def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    S = float(a.sum())
    if S < 1e-12:
        return np.ones_like(a) / len(a)
    return a / S

def freq_flatten_newton(a, n_iters=200, gamma=0.8, alpha=0.7):
    """Newton-like frequency domain flattening."""
    n = len(a)
    M_fft = 1 << (2*n - 1 - 1).bit_length()
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, M_fft)
        conv = np.fft.ifft(A * A).real[:2*n-1]
        
        conv_mean = conv.sum() / (2*n-1)
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        
        conv_mod = conv - gamma * excess
        conv_mod = np.maximum(conv_mod, 0.0)
        
        # Newton step in frequency domain
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2*n-1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A)
        d[np.abs(A) < 1e-10] = 0.0
        
        damp = 0.5 + 0.5 * min(1.0, it / (0.25 * n_iters))
        A_new = A + damp * d
        a = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a, 0.0)
        a = normalize(a)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

"""
Hybrid optimizer for step function sequence minimization.
Uses frequency-domain flattening (Newton + magnitude spectrum methods),
LSE-smoothed L-BFGS, multi-peak active-set subgradient, and coordinate descent.
Key innovations: multi-peak handling, magnitude-spectrum GS, adaptive blending,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize_a(a):
    """Normalize to sum 1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones_like(a) / len(a)
    return a / S


def freq_flatten_newton(a, n_iters=300, gamma=0.8, alpha=0.7, top_k=1):
    """Newton-like frequency domain flattening with multi-peak handling."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    a = normalize_a(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.fft(a, M_fft)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = conv.sum() / (2 * n - 1)
        conv_sorted = np.sort(conv)[::-1]
        # Adaptive: handle top-k peaks
        k_eff = min(top_k, 2 * n - 1)
        peak_level = conv_sorted[k_eff - 1] if k_eff > 1 else conv_sorted[0]
        conv_max = float(conv_sorted[0])
        if conv_max - conv_mean < 1e-14:
            break

        # Adaptive threshold: between mean and peak
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - gamma * excess
        conv_mod = np.maximum(conv_mod, 0.0)

        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A)
        d[np.abs(A) < 1e-10] = 0.0

        # Adaptive damping: start aggressive, become conservative
        progress = it / max(1, n_iters)
        damp = 0.6 + 0.4 * (1.0 - progress)
        A_new = A + damp * d
        a = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a, 0.0)
        a = normalize_a(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def freq_flatten_magnitude(a, n_iters=300, gamma=0.5, alpha=0.7):
    """Direct magnitude spectrum adjustment (GS-style)."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    a = normalize_a(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.fft(a, M_fft)
        power = np.abs(A) ** 2
        conv = np.fft.ifft(power).real[:2 * n - 1]

        conv_mean = conv.sum() / (2 * n - 1)
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        # Soft compression: use power-law for smoother transition
        conv_mod = conv - gamma * (excess ** 1.2)
        conv_mod = np.maximum(conv_mod, 0.0)

        # Go to frequency domain via FFT of modified convolution
        conv_mod_pad = np.zeros(M_fft, dtype=np.float64)
        conv_mod_pad[:2 * n - 1] = conv_mod
        power_new = np.fft.fft(conv_mod_pad).real
        power_new = np.maximum(power_new, 0.0)

        # Blend with original power for stability
        blend = 0.5 + 0.5 * (it / max(1, n_iters))
        power_final = (1.0 - blend) * power + blend * power_new

        phase = np.angle(A)
        mag_new = np.sqrt(power_final)
        A_new = mag_new * np.exp(1j * phase)

        a = np.fft.ifft(A_new).real[:n]
        a = np.maximum(a, 0.0)
        a = normalize_a(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def freq_flatten_adaptive(a, n_total=500):
    """Run both Newton and magnitude methods, return best."""
    results = []
    for alpha in [0.6, 0.7, 0.8, 0.85]:
        a1, sc1 = freq_flatten_newton(a, n_iters=n_total // 4, gamma=0.8, alpha=alpha)
        results.append((sc1, a1))
        a2, sc2 = freq_flatten_magnitude(a, n_iters=n_total // 4, gamma=0.5, alpha=alpha)
        results.append((sc2, a2))
    results.sort(key=lambda x: x[0])
    return results[0][1], results[0][0]


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (vectorized)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # = 2 * IFFT(conj(FFT(a)) * FFT(p))[j]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    corr = np.fft.irfft(np.conj(A_fft) * P_fft, M2_fft)[:n]

    dL_da = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    bounds = [(0.0, 1e6)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,

    )
    a_out = np.clip(res.x, 0.0, 1e6)
    sc = compute_score(a_out)
    return sc, a_out


def multi_peak_subgradient(a, total_iters=3000, top_k=5):
    """Subgradient method targeting multiple peak lags simultaneously."""
    n = len(a)
    a = normalize_a(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 1.0  # theoretical lower bound

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k peak lags
        top_indices = np.argsort(c)[::-1][:top_k]
        c_mean = c.sum() / (2 * n - 1)

        # Weighted subgradient: focus on peaks above mean
        weights = np.zeros(2 * n - 1)
        for k in top_indices:
            excess = c[k] - c_mean
            if excess > 0:
                weights[k] = excess

        # Compute gradient as weighted sum of subgradients
        M2 = 2 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        A_fft = np.fft.rfft(a, M2_fft)
        W_fft = np.fft.rfft(weights, M2_fft)
        corr = np.fft.irfft(np.conj(A_fft) * W_fft, M2_fft)[:n]
        sg = 2.0 * corr

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = (sc - sc_lower) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        # Decreasing step
        step *= max(0.1, 1.0 - 0.3 * it / total_iters)

        a -= step * grad
        a = np.clip(a, 0.0, 1e6)
        a = normalize_a(a)

    return best_a, best_sc


def coordinate_descent_multi(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting top peak-contributing elements."""
    n = len(a)
    a = normalize_a(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Get top few peak lags
        top_lags = np.argsort(c)[::-1][:3]

        for k_star in top_lags:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old = a[j]
                # Try reducing this element
                for factor in [0.7, 0.8, 0.85, 0.5, 0.0]:
                    a[j] = a_old * factor
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    else:
                        a[j] = a_old
                        break
                a[j] = best_a[j] if sc < best_sc else a_old
                a = normalize_a(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def blend_sequences(a1, a2, t):
    """Interpolate between two sequences, adjusting for different lengths."""
    if len(a1) == len(a2):
        return normalize_a(t * a1 + (1 - t) * a2)
    # Resample to same length
    n = max(len(a1), len(a2))
    x1 = np.linspace(0, 1, len(a1))
    x2 = np.linspace(0, 1, len(a2))
    x_new = np.linspace(0, 1, n)
    a1_r = np.interp(x_new, x1, a1)
    a2_r = np.interp(x_new, x2, a2)
    return normalize_a(t * a1_r + (1 - t) * a2_r)

    # Envelope: high at edges to counteract triangular convolution
    candidates.append(0.3 + 0.7 * np.sin(np.pi * x) ** 2)
    # Multi-frequency
    candidates.append(0.5 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x))
    return candidates


def upsample_sequence(a, factor):
    """Upsample a sequence by smooth interpolation."""
    n_old = len(a)
    n_new = n_old * factor
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def downsample_sequence(a, factor):
    """Downsample a sequence by block averaging."""
    n_old = len(a)
    n_new = n_old // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses hybrid frequency-domain + LSE-LBFGS + subgradient approach.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    # Try to get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1e6)
                print(f"Parent loaded: length {len(parent)}, score {compute_score(parent):.8f}")
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def check_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1e6)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")

    # ================================================================
    # Phase 1: Multi-scale frequency domain flattening (45% time)
    # ================================================================
    phase1_end = t_start + 0.45 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 1: Frequency Domain Flattening (until t={phase1_end-t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  Processing n={n}...")

        # Build initializations
        if parent is not None and len(parent) == n:
            inits = [parent.copy()]
        elif parent is not None:
            inits = [np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)]
        else:
            inits = build_good_init(n, rng)

        # Add random inits for diversity
        for _ in range(2):
            inits.append(np.clip(0.3 + 0.5 * rng.random(n), 0.01, 10.0))

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a0 = normalize_a(a0.astype(np.float64).copy())
            sc0 = compute_score(a0)
            print(f"    Init {i}: score={sc0:.6f}")

            # Newton method with different parameters
            for alpha in [0.65, 0.75, 0.85]:
                if time.monotonic() > phase1_end:
                    break
                a_er, sc_er = freq_flatten_newton(a0, n_iters=200, gamma=0.8, alpha=alpha)
                check_update(sc_er, a_er, f"Newton n={n} a={alpha} i={i}")

            # Magnitude method
            if time.monotonic() > phase1_end:
                break
            a_mag, sc_mag = freq_flatten_magnitude(a0, n_iters=200, gamma=0.5, alpha=0.75)
            check_update(sc_mag, a_mag, f"MagSpec n={n} i={i}")

            # Continue refining from best of the above
            a_cur = best_a if len(best_a) != n else best_a.copy()
            if len(a_cur) != n:
                a_cur = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a_cur)), a_cur)
            a_er2, sc_er2 = freq_flatten_newton(a_cur, n_iters=300, gamma=0.7, alpha=0.8, top_k=3)
            check_update(sc_er2, a_er2, f"Newton2 n={n} i={i}")

    print(f"\n  After Phase 1: best = {best_score:.8f} (n={len(best_a)})")

    # ================================================================
    # Phase 2: LSE-LBFGS refinement (35% time)
    # ================================================================
    phase2_end = t_start + 0.80 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 2: LSE-LBFGS Refinement ===")

    n_best = len(best_a)
    # If best is too large for L-BFGS, use a smaller version first
    n_lbfgs = min(n_best, 8192)
    if n_lbfgs < n_best:
        a_start = downsample_sequence(best_a, max(1, n_best // n_lbfgs))
    else:
        a_start = best_a.copy()

    for beta in [50, 100, 200, 500, 1000, 2000]:
        if time.monotonic() > phase2_end:
            break
        print(f"  L-BFGS beta={beta}, n={n_lbfgs}")
        sc, a_opt = run_lbfgs(a_start, beta, maxiter=8000, maxfun=200000)
        updated = check_update(sc, a_opt, f"LBFGS b={beta}")
        if updated:
            a_start = best_a.copy() if len(best_a) == n_lbfgs else \
                np.interp(np.linspace(0, 1, n_lbfgs), np.linspace(0, 1, len(best_a)), best_a)
        else:
            a_start = a_opt  # Continue from this point

    # Try L-BFGS at the best length
    if len(best_a) != n_lbfgs and len(best_a) <= 16384:
        if time.monotonic() < phase2_end - 60:
            for beta in [200, 500]:
                if time.monotonic() > phase2_end:
                    break
                print(f"  L-BFGS at native n={len(best_a)}, beta={beta}")
                sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
                check_update(sc, a_opt, f"LBFGS native b={beta}")

    # Local search with perturbations
    if time.monotonic() < phase2_end:
        print(f"  Local search perturbations...")
        n_ps = len(best_a)
        for trial in range(8):
            if time.monotonic() > phase2_end:
                break
            strategy = trial % 4
            if strategy == 0:
                a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_ps))
            elif strategy == 1:
                # Reduce peak contributors
                c = fast_conv(best_a)
                k_star = int(np.argmax(c))
                j_start = max(0, k_star - n_ps + 1)
                j_end = min(k_star, n_ps - 1)
                a_pert = best_a.copy()
                seg = a_pert[j_start:j_end + 1]
                contrib = seg[::-1].copy()
                top_count = max(5, n_ps // 200)
                top_idx = np.argsort(contrib)[::-1][:top_count]
                for li in top_idx:
                    a_pert[j_start + li] *= 0.85
            elif strategy == 2:
                # Sinusoidal perturbation
                freq = 1.0 + 10.0 * rng.random()
                phase = 2.0 * np.pi * rng.random()
                x = np.arange(n_ps) / n_ps
                a_pert = best_a * (1.0 + 0.05 * np.cos(2 * np.pi * freq * x + phase))
            else:
                # Frequency domain restart
                a_pert = best_a * (1.0 + 0.1 * rng.standard_normal(n_ps))

            a_pert = normalize_a(a_pert)
            sc_p, a_opt_p = freq_flatten_newton(a_pert, n_iters=100, gamma=0.8, alpha=0.8)
            check_update(sc_p, a_opt_p, f"perturb #{trial} s={strategy}")
            if time.monotonic() < phase2_end:
                sc_l, a_opt_l = run_lbfgs(a_opt_p, 300, maxiter=3000, maxfun=50000)
                check_update(sc_l, a_opt_l, f"perturb #{trial} LBFGS")

    print(f"\n  After Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ================================================================
    # Phase 3: Subgradient + coordinate descent polish (15% time)
    # ================================================================
    phase3_end = deadline
    print(f"\n=== Phase 3: Polish ===")

    # Multi-peak subgradient
    a_sg, sc_sg = multi_peak_subgradient(best_a, total_iters=3000, top_k=5)
    check_update(sc_sg, a_sg, "subgrad")

    # Coordinate descent
    if time.monotonic() < phase3_end - 30:
        a_cd, sc_cd = coordinate_descent_multi(best_a, n_sweeps=4, n_top=80)
        check_update(sc_cd, a_cd, "coord_desc")

    # Final LSE optimization with high beta
    if time.monotonic() < phase3_end - 30:
        sc_f, a_f = run_lbfgs(best_a, 3000, maxiter=3000, maxfun=60000)
        check_update(sc_f, a_f, "final LBFGS b=3000")

    # One more round of frequency domain from best
    if time.monotonic() < phase3_end - 20:
        a_er_f, sc_er_f = freq_flatten_newton(best_a, n_iters=200, gamma=0.6, alpha=0.9, top_k=3)
        check_update(sc_er_f, a_er_f, "final ER")
        a_mag_f, sc_mag_f = freq_flatten_magnitude(best_a, n_iters=200, gamma=0.4, alpha=0.9)
        check_update(sc_mag_f, a_mag_f, "final MagSpec")

    # Try blending with parent (if available) for diversity
    if parent is not None and time.monotonic() < phase3_end - 15:
        for t in [0.3, 0.5, 0.7]:
            a_blend = blend_sequences(best_a, parent, t)
            sc_b = compute_score(a_blend)
            check_update(sc_b, a_blend, f"blend t={t}")
            if sc_b < best_score + 0.01:
                a_er_b, sc_er_b = freq_flatten_newton(a_blend, n_iters=100, gamma=0.8, alpha=0.8)
                check_update(sc_er_b, a_er_b, f"blend+ER t={t}")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1e6, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 1024

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
