    subject to sum(a)=1, a>=0.
    
    Uses soft-max subgradient with continuation on sharpness.
    Step size: eta = damping * (M - lb) / (||g||_inf^2).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c0 = fast_conv(a)
    best_sc = 2.0 * n * float(c0.max())

    lb = 1.0 / (2.0 * n - 1)
    M_fft = 1 << (3 * n - 2).bit_length()

    n_continuation = 6
    iters_per_phase = max(100, n_iters // n_continuation)

    for phase in range(n_continuation):
        frac = phase / max(1, n_continuation - 1)
        beta = initial_beta * (final_beta / initial_beta) ** frac

        if phase < n_continuation - 1:
            phase_iters = iters_per_phase
        else:
            phase_iters = n_iters - phase * iters_per_phase

        for it in range(phase_iters):
            if deadline is not None and time.monotonic() > deadline:
                return best_sc, best_a

            c = fast_conv(a)
            M = float(c.max())
            sc = 2.0 * n * M

            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            if M <= lb * (1.0 + 1e-10):
                return best_sc, best_a

            # Soft-max weights
            scaled = beta * (c - M)
            weights = np.exp(scaled)
            weights /= weights.sum()

            # grad[m] = 2 * sum_k weights[k] * a[k-m]
            W = np.fft.rfft(weights, M_fft)
            AR = np.fft.rfft(a[::-1], M_fft)
            conv_res = np.fft.irfft(W * AR, M_fft)
            grad = 2.0 * conv_res[n-1:2*n-1]

            g_max = float(np.max(np.abs(grad)))
            if g_max < 1e-15:
                break

            # Step size with damping
            eta = damping * (M - lb) / (g_max * g_max)
            eta = min(eta, 3.0)

            # Mirror descent update
            log_update = -eta * grad
            log_update -= log_update.max()
            update = np.exp(log_update)
            a = a * update
            S_new = a.sum()
            if S_new < 1e-30:
                break
            a = a / S_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        if verbose:
            print(f"    MD phase {phase+1}/{n_continuation} (beta={beta:.1f}): best_score={best_sc:.8f} [{label}]")

    return best_sc, best_a


def targeted_mirror_descent(a, n_iters=5000, deadline=None, beta=200.0, top_k=10, verbose=False):
    """
    Targeted mirror descent that focuses on the top-k peak lags.
    More aggressive reduction of peak lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    lb = 1.0 / (2.0 * n - 1)
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if M <= lb * (1.0 + 1e-9):
            break

        # Focus on top-k lags with exponential weighting
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:top_k]
        top_vals = c[top_idx]
        # Exponential weighting: peaks get more weight
        w_scaled = beta * (top_vals - top_vals[0])
        w_top = np.exp(w_scaled)
        w_top /= w_top.sum()

        weights = np.zeros(len(c))
        weights[top_idx] = w_top

        W = np.fft.rfft(weights, M_fft)
        AR = np.fft.rfft(a[::-1], M_fft)
        conv_res = np.fft.irfft(W * AR, M_fft)
        grad = 2.0 * conv_res[n-1:2*n-1]

        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break

        eta = 0.7 * (M - lb) / (g_max * g_max)
        eta = min(eta, 2.0)

        log_update = -eta * grad
        log_update -= log_update.max()
        update = np.exp(log_update)
        a = a * update
        S_new = a.sum()
        if S_new < 1e-30:
            break
        a = a / S_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    return best_sc, best_a


def palindromic_mirror_descent(a, n_iters=10000, deadline=None, initial_beta=10.0, final_beta=500.0):
    """Mirror descent enforcing palindromic symmetry a[j] = a[n-1-j]."""
    n = len(a)
    # Enforce symmetry
    half = n // 2
    a = a.astype(np.float64).copy()
    a_sym = np.zeros(n)
    if n % 2 == 0:
        a_sym[:half] = (a[:half] + a[half::-1]) / 2.0
        a_sym[half:] = a_sym[:half][::-1]
    else:
        a_sym[:half] = (a[:half] + a[half::-1]) / 2.0
        a_sym[half] = a[half]
        a_sym[half+1:] = a_sym[:half][::-1]
    a = np.maximum(a_sym, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    lb = 1.0 / (2.0 * n - 1)
    half = (n + 1) // 2
    M_fft = 1 << (3 * n - 2).bit_length()

    n_continuation = 4
    iters_per_phase = max(100, n_iters // n_continuation)

    for phase in range(n_continuation):
        frac = phase / max(1, n_continuation - 1)
        beta = initial_beta * (final_beta / initial_beta) ** frac
        phase_iters = iters_per_phase if phase < n_continuation - 1 else (n_iters - phase * iters_per_phase)

        for it in range(phase_iters):
            if deadline is not None and time.monotonic() > deadline:
                return best_sc, best_a

            c = fast_conv(a)
            M = float(c.max())
            sc = 2.0 * n * M
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

            if M <= lb * (1.0 + 1e-10):
                break

            scaled = beta * (c - M)
            weights = np.exp(scaled)
            weights /= weights.sum()

            W = np.fft.rfft(weights, M_fft)
            AR = np.fft.rfft(a[::-1], M_fft)
            conv_res = np.fft.irfft(W * AR, M_fft)
            grad = 2.0 * conv_res[n-1:2*n-1]

            g_max = float(np.max(np.abs(grad)))
            if g_max < 1e-15:
                break

            eta = 0.5 * (M - lb) / (g_max * g_max)
            eta = min(eta, 3.0)

            log_update = -eta * grad
            log_update -= log_update.max()
            update = np.exp(log_update)
            a = a * update
            a = np.maximum(a, 1e-15)

            # Enforce palindromic symmetry
            if n % 2 == 0:
                a[:half] = (a[:half] + a[half-1::-1]) / 2.0
                a[half:] = a[:half][::-1]
            else:
                a[:half] = (a[:half] + a[half::-1]) / 2.0
                a[half+1:] = a[:half][::-1]

            a = a / a.sum()

        if phase % 2 == 0:
            print(f"      PalinMD phase {phase+1}: score={best_sc:.8f}")

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p)-1, n), np.arange(len(p)), p)
        p = np.clip(p, 0.01, 1000.0)
        inits.append(p / p.sum() * np.sqrt(2*n))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5 / n * np.sqrt(2*n))
    inits.append((0.5 + 0.4 * np.cos(2.0 * np.pi * x)) / n * np.sqrt(2*n))
    inits.append((0.3 + 0.5 * np.cos(4.0 * np.pi * x)) / n * np.sqrt(2*n))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None) / n * np.sqrt(2*n))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None) / n * np.sqrt(2*n))
    # Triangular
    inits.append(np.abs(1.0 - 2.0*x) / n * np.sqrt(2*n))
    # Bell-shaped
    inits.append(np.exp(-4.0*(x-0.5)**2) / n * np.sqrt(2*n))
    # Edge-concentrated
    edge = np.exp(-20.0 * np.minimum(x, 1-x)**2) + 0.1
    inits.append(edge / edge.sum() * np.sqrt(2*n))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses mirror descent as primary optimizer with LSE-LBFGS polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Parent loaded: length={parent.size}, score={compute_score(parent):.8f}")
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def _report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n in [8192, 4096, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to try: {lengths}")

    # ============================================================
    # Phase 1: Mirror descent global search
    # ============================================================
    phase1_end = t_start + 0.50 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Mirror Descent Global Search")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n--- n = {n} ---")
        inits = build_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            a_norm = a0 / a0.sum()

            # Main mirror descent
            sc, a_md = mirror_descent(a_norm, n_iters=15000, deadline=phase1_end,
                                       initial_beta=5.0, final_beta=800.0,
                                       damping=0.5, verbose=(i == 0), label=f"n={n},i={i}")
            _report(sc, a_md, f"MD n={n},init={i}")

            # If time remains, do targeted MD
            if time.monotonic() < phase1_end - 30:
                sc2, a_tgt = targeted_mirror_descent(a_md, n_iters=5000, deadline=phase1_end,
                                                       beta=200.0, top_k=8)
                _report(sc2, a_tgt, f"TM n={n},init={i}")
                a_md = a_tgt if sc2 < sc else a_md

        # Palindromic attempt for this length
        if time.monotonic() < phase1_end - 60:
            a_pal_in = np.ones(n) * 0.5
            if parent is not None:
                p = parent.astype(np.float64)
                if len(p) != n:
                    p = np.interp(np.linspace(0, len(p)-1, n), np.arange(len(p)), p)
                a_pal_in = p
            sc_pal, a_pal = palindromic_mirror_descent(a_pal_in, n_iters=8000, deadline=phase1_end - 60)
            _report(sc_pal, a_pal, f"PalinMD n={n}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}, n = {len(best_a)}")

    # ============================================================
    # Phase 2: LSE-LBFGS polishing on best solution
    # ============================================================
    phase2_end = t_start + 0.70 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE-LBFGS Polish")
    print(f"{'='*60}")

    n_best = len(best_a)
    a_cur = best_a.copy()

    beta_schedule = [50, 100, 200, 400, 800, 1500]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=50000, deadline=phase2_end)
        _report(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_cur = a_opt

    # ============================================================
    # Phase 3: Alternating MD + LBFGS refinement
    # ============================================================
    phase3_end = t_start + 0.88 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Alternating Refinement")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    n_rounds = 12

    for rnd in range(n_rounds):
        if time.monotonic() > phase3_end:
            break
        print(f"  Round {rnd+1}/{n_rounds}:")

        # Mirror descent step
        a_norm = a_cur / a_cur.sum()
        beta_md = 100.0 + 100.0 * rnd
        sc_md, a_md = mirror_descent(a_norm, n_iters=3000, deadline=phase3_end,
                                       initial_beta=beta_md, final_beta=beta_md*5,
                                       damping=0.4, verbose=False)
        _report(sc_md, a_md, f"alt MD r={rnd}")
        if sc_md < best_score:
            a_cur = a_md

        # Targeted MD
        if time.monotonic() < phase3_end - 20:
            sc_tgt, a_t = targeted_mirror_descent(a_cur / a_cur.sum(), n_iters=2000,
                                                    deadline=phase3_end, beta=300.0, top_k=12)
            _report(sc_tgt, a_t, f"alt TM r={rnd}")
            if sc_tgt < best_score:
                a_cur = a_t

        # Short LBFGS
        if time.monotonic() < phase3_end - 15:
            sc_l, a_l = run_lbfgs(a_cur, 500 + 200*rnd, maxiter=1500, maxfun=20000,
                                    deadline=phase3_end)
            _report(sc_l, a_l, f"alt LBFGS r={rnd}")
            if sc_l < best_score:
                a_cur = a_l

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Diversification - try different perturbations
    # ============================================================
    phase4_end = t_start + 0.95 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Diversification")
    print(f"{'='*60}")

    n_best = len(best_a)
    n_diversify = 10

    for d in range(n_diversify):
        if time.monotonic() > phase4_end:
            break
        strategy = d % 4
        a_base = best_a / best_a.sum()

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.04 * rng.random()
            a_pert = a_base * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Frequency-domain perturbation
            A = np.fft.rfft(a_base)
            freq = rng.integers(1, len(A))
            A[freq] *= (1.0 + 0.1 * rng.standard_normal())
            if freq + 1 < len(A):
                A[freq+1] *= (1.0 + 0.05 * rng.standard_normal())
            a_pert = np.fft.irfft(A, 2*n_best-1)[:n_best]
        elif strategy == 2:
            # Edge boost
            a_pert = a_base.copy()
            edge_width = max(4, n_best // 100)
            boost = 1.0 + 0.2 * rng.random()
            a_pert[:edge_width] *= boost
            a_pert[-edge_width:] *= boost
        else:
            # Block swap
            a_pert = a_base.copy()
            bs = max(16, n_best // 50)
            b1 = rng.integers(0, n_best - 2*bs)
            b2 = rng.integers(0, n_best - 2*bs)
            if b1 != b2:
                a_pert[b1:b1+bs], a_pert[b2:b2+bs] = a_pert[b2:b2+bs].copy(), a_pert[b1:b1+bs].copy()

        a_pert = np.clip(a_pert, 1e-10, None)
        a_pert = a_pert / a_pert.sum()

        sc_p, a_opt = mirror_descent(a_pert, n_iters=3000, deadline=phase4_end,
                                       initial_beta=50.0, final_beta=500.0, damping=0.5)
        _report(sc_p, a_opt, f"div #{d} (strat={strategy})")
        if sc_p < best_score:
            # Quick LBFGS polish
            if time.monotonic() < phase4_end - 10:
                sc_l, a_l = run_lbfgs(a_opt, 500, maxiter=2000, maxfun=30000, deadline=phase4_end)
                _report(sc_l, a_l, f"div #{d} polish")

    # ============================================================
    # Phase 5: Final intensive polishing
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    a_final = best_a.copy()
    for beta in [500, 1000, 2000, 4000]:
        if time.monotonic() > deadline - 30:
            break
        sc, a_opt = run_lbfgs(a_final, beta, maxiter=5000, maxfun=80000, deadline=deadline - 30)
        _report(sc, a_opt, f"final b={beta}")
        if sc < best_score:
            a_final = a_opt

    # One more MD round on the best
    if time.monotonic() < deadline - 20:
        sc_md, a_md = mirror_descent(a_final / a_final.sum(), n_iters=5000,
                                       deadline=deadline - 20, initial_beta=200.0,
                                       final_beta=2000.0, damping=0.3)
        _report(sc_md, a_md, "final MD")

    if time.monotonic() < deadline - 15:
        sc, a_opt = run_lbfgs(a_final, 3000, maxiter=3000, maxfun=50000, deadline=deadline - 10)
        _report(sc, a_opt, "final LBFGS 3000")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
