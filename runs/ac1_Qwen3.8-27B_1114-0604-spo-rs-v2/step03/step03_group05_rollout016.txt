from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        log_sum = np.log(exp_sum)
        p = exp_scaled / exp_sum
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

Wait, if I normalize $\sum a = 1$ as a constraint, then the subgradient of $\max_k c_k$ subject to $\sum a = 1$ is more complex. Let me just use the unconstrained subgradient and let the normalization handle the rest.

Actually, for the multiplicative weights approach, I should be optimizing $\max_k c_k / S^2$ (the actual score divided by 2n). The gradient of this with respect to $a_j$ is:
$$\frac{\partial}{\partial a_j} \left( \frac{M}{S^2} \right) = \frac{g_j \cdot S^2 - M \cdot 2S \cdot 1}{S^4} = \frac{g_j}{S^2} - \frac{2M}{S^3}$$

where $g_j = 2a_{k^*-j}$ (the subgradient of $M = c_{k^*}$).

For the multiplicative update:
$a_j \leftarrow a_j \cdot \exp(-\tau \cdot (\frac{g_j}{S^2} - \frac{2M}{S^3}))$

The second term $-\frac{2M}{S^3}$ is the same for all $j$, so it just scales the whole vector and gets removed by normalization. So effectively:
$a_j \leftarrow a_j \cdot \exp(-\tau \cdot \frac{g_j}{S^2})$

$a_j \leftarrow \frac{a_j \cdot \exp(-\tau \cdot 2a_{k^*-j}/S^2)}{\sum_l a_l \cdot \exp(-\tau \cdot 2a_{k^*-l}/S^2)}$

This is a soft-attention-like update that redistributes mass away from elements that contribute to the peak.

"""
Step function optimization: LSE continuation + mirror descent (multiplicative weights)
+ Nesterov-accelerated subgradient + adaptive restarts.
Improves on prior 1.5093 by using multiplicative updates that naturally preserve
non-negativity and a more aggressive LSE schedule with momentum.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize with args=(beta,)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization. Returns optimized array."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def nesterov_subgrad(a, iters=5000, verbose=False):
    """Nesterov-accelerated subgradient descent on max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = float(a.sum())
    a_norm = a / S0  # work with normalized (sum=1)

    best_score = compute_score(a_norm)
    best_a = a_norm.copy()
    prev_a = a_norm.copy()
    y = a_norm.copy()
    sc_lb = 1.0  # lower bound on score

    for t in range(iters):
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = y.copy()

        # Subgradient of max c_k / S^2
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            for j in range(j_start, j_end + 1):
                other = k_star - j
                if 0 <= other < n:
                    sg[j] = 2.0 * y[other]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Step size: Polyak-like with decay
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 0.5 / S)
        step = max(step, 1e-7 / (gn * S + 1e-30))

        # Nesterov momentum
        momentum = (t - 1) / (t + 2) if t > 0 else 0.0
        x_new = y - step * grad
        x_new = np.clip(x_new, 0.0, 1000.0)
        y_new = x_new + momentum * (x_new - y)
        y_new = np.clip(y_new, 0.0, 1000.0)
        prev_a = y.copy()
        y = y_new

    return best_score, best_a

        # Compute weighted subgradient
        sg = np.zeros(n)
        for k in range(len(c)):

                j_end = min(k, n - 1)

def freq_domain_init(n, rng):

    L = 1 << (2 * n - 2).bit_length()

    # Ensure real sequence: Z symmetry
    A = np.zeros(L, dtype=np.complex128)
    for i in range(L):
        A[i] = np.exp(1j * phases[i])
    # Enforce Hermitian symmetry for real output
    for i in range(1, L // 2 + 1):

    A[0] = 1.0  # DC component positive
    a = np.fft.ifft(A).real[:n]
    a = np.abs(a)
    if a.sum() < 1e-10:

    a = a / a.sum()
    return a

    print("\n=== Phase 6: Final Restarts ===")

    n_best = len(best_a)
    restart_count = 0
    while time.monotonic() < phase6_end and restart_count < 15:
        t0 = time.monotonic()
        strategy = restart_count % 5

        if strategy == 0:
            # Perturb and LSE
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_opt = run_lbfgs(a_pert, lse_obj, (5000,), maxiter=800, maxfun=15000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"restart {restart_count} (perturb+LSE)")

        elif strategy == 1:
            # Top-peak targeted reduction
            a_pert = best_a.copy()
            c = fast_conv(a_pert)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            contrib = np.zeros(n_best)
            for j in range(j_start, j_end + 1):
                other = k_star - j
                if 0 <= other < n_best:
                    contrib[j] = a_pert[other]
            top_idx = np.argsort(contrib)[::-1][:max(10, n_best // 50)]
            for j in top_idx:
                a_pert[j] *= 0.8
            a_pert = a_pert / a_pert.sum() * best_a.sum()
            # LSE polish
            a_opt = run_lbfgs(a_pert, lse_obj, (5000,), maxiter=500, maxfun=10000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"restart {restart_count} (peak cut)")

        elif strategy == 2:
            # Multiplicative weights from perturbed start
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            iters_mw = 1000
            sc, a_mw_r = multiplicative_weights(a_pert, iters=iters_mw, tau_init=0.5)
            report(sc, a_mw_r, f"restart {restart_count} (MW perturb)")

        elif strategy == 3:
            # Frequency domain perturbation
            L = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.rfft(best_a, L)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            A_mag_new = A_mag * (1.0 + 0.03 * rng.standard_normal(len(A_mag)))
            A_mag_new = np.clip(A_mag_new, 0, None)
            A_new = A_mag_new * np.exp(1j * A_phase)
            a_pert = np.fft.irfft(A_new, L)[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            if a_pert.sum() > 1e-6:
                a_opt = run_lbfgs(a_pert, lse_obj, (3000,), maxiter=500, maxfun=10000)
                sc = compute_score(a_opt)
                report(sc, a_opt, f"restart {restart_count} (freq pert)")

        else:
            # Try slightly different length
            n_trial = n_best + rng.integers(-512, 513)
            n_trial = max(1024, min(12000, n_trial))
            if n_trial != n_best:
                # Resample
                x_old = np.linspace(0, 1, n_best)
                x_new = np.linspace(0, 1, n_trial)
                a_trial = np.interp(x_new, x_old, best_a)
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                a_opt = run_lbfgs(a_trial, lse_obj, (2000,), maxiter=500, maxfun=10000)
                sc = compute_score(a_opt)
                report(sc, a_opt, f"restart {restart_count} (n={n_trial})")

        restart_count += 1
        print(f"  Restart {restart_count}: {time.monotonic()-t0:.1f}s, best={best_score:.8f}")

    # ===================================================================
    # Final: one more LSE + Nesterov on the best
    # ===================================================================
    print("\n=== Final Polish ===")
    if time.monotonic() < deadline:
        a_final = run_lbfgs(best_a, lse_obj, (20000,), maxiter=1000, maxfun=20000)
        sc = compute_score(a_final)
        report(sc, a_final, "final LSE")

        sc, a_pol = nesterov_subgrad(best_a, iters=2000)
        report(sc, a_pol, "final Nesterov")

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
