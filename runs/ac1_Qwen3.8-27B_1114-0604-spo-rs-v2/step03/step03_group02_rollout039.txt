def _cross_h(p, w, n, M):
    rev = p[::-1]

```python
"""
Search for a non-negative step function minimizing
    score = 2 * n * max(convolve(a, a)) / (sum(a))**2
by optimizing probability mass p = a / sum(a) on the simplex.

The main components are:
  - fast FFT convolution/cross-correlation,
  - a multiplicative/log-simplex line search driven by the LP-dual quantity
    h_j = sum_k lambda_k p_{k-j}, where lambda_k are soft active lags,
  - periodic L-BFGS polishing on a log-sum-exp smoothed objective,
  - controlled exploration by small log-noise restarts,
  - an optional attempt at length 16384 seeded by interpolation of the best
    length-8192 solution.

All returned sequences are normalized to sum 1. This is scale invariant for
the objective.
"""

import numpy as np
import time
from scipy.optimize import minimize


_BEST_SCORE = float('inf')
_BEST_P = None
_PRINTED_SCORE = float('inf')
_LAST_PRINT_TIME = 0.0


def _register_candidate(score, p):
    global _BEST_SCORE, _BEST_P, _PRINTED_SCORE, _LAST_PRINT_TIME
    if p is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not np.isfinite(score):
        return
    try:
        a = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return
    if a.size == 0:
        return
    if score < _BEST_SCORE - 1e-15:
        _BEST_SCORE = score
        _BEST_P = a.copy()
        now = time.monotonic()
        if score < _PRINTED_SCORE - 1e-7 or (now - _LAST_PRINT_TIME) > 30.0:
            print(f"new best score={score:.10f} len={a.size} elapsed={now:.1f}s")
            _PRINTED_SCORE = score
            _LAST_PRINT_TIME = now


def _fft_size(n):
    need = 3 * int(n) - 2
    return 1 << (need - 1).bit_length()


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / max(1, z.size), dtype=np.float64)
    return e / s


def _resample_to_length(p, new_n):
    if p is None:
        return np.full(int(new_n), 1.0 / float(new_n), dtype=np.float64)
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.full(int(new_n), 1.0 / float(new_n), dtype=np.float64)
    old_n = p.size
    if old_n == new_n:
        q = p.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        q = np.interp(x_new, x_old, p)
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-15:
        q = np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    else:
        q /= s
    return q


def _z_from_p(p):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    if s <= 1e-15:
        p = np.full(p.size, 1.0 / max(1, p.size), dtype=np.float64)
    else:
        p /= s
    z = np.log(p)
    z -= float(np.mean(z))
    return z


def _conv_self(p, n, M):
    A = np.fft.rfft(p, M)
    c = np.fft.irfft(A * A, M)
    c = c[:2 * n - 1]
    c[c < 0.0] = 0.0
    return c


def _cross_h(p, w, n, M):
    # h[j] = sum_k w[k] * p[k-j], p outside [0,n-1] is 0.
    rev = p[::-1]
    R = np.fft.rfft(rev, M)
    W = np.fft.rfft(w, M)
    out = np.fft.irfft(R * W, M)
    h = out[n - 1:2 * n - 1]
    h[h < 0.0] = 0.0
    return h


def _exact_score_p(p, n, M):
    c = _conv_self(p, n, M)
    return 2.0 * int(n) * float(c.max())


def _lse_grad_p(p, beta, n, M):
    # p is normalized, sum(p)=1.
    # Use d = n * conv(p,p), so max_d = n * max_c and exact score = 2*max_d.
    c = _conv_self(p, n, M)
    d = float(n) * c
    dmax = float(d.max())
    exact = 2.0 * dmax

    scaled = float(beta) * (d - dmax)
    smax = float(scaled.max())
    exps = np.exp(scaled - smax)
    sumexp = float(exps.sum())
    if sumexp <= 0.0 or not np.isfinite(sumexp):
        w = np.ones(2 * n - 1, dtype=np.float64) / float(2 * n - 1)
    else:
        w = exps / sumexp

    h = _cross_h(p, w, n, M)
    # Objective F = 2*L, where L = (1/beta) logsumexp(beta*d).
    # dL/dp = 2*n*h, hence dF/dp = 4*n*h.
    gp = 4.0 * float(n) * h

    logsumexp_scaled = smax + float(np.log(max(sumexp, 1e-300)))
    L = dmax + logsumexp_scaled / float(beta)
    F = 2.0 * L
    return float(F), gp, w, h, float(exact)


def _log_lse_objective(z, beta, n, M, state):
    p = _softmax(z)
    F, gp, w, h, exact = _lse_grad_p(p, beta, n, M)

    if np.isfinite(exact):
        _register_candidate(exact, p)
        if state is not None and exact < float(state[0]):
            state[0] = float(exact)
            state[1] = p.copy()

    if not np.isfinite(F):
        return 1e100, np.zeros_like(z, dtype=np.float64)

    T = float(np.dot(p, gp))
    gz = p * (gp - T)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(F), gz


def _lbfgs_refine(z0, beta, n, M, deadline, state):
    if state is None:
        state = [float('inf'), None]

    remaining = float(deadline - time.monotonic())
    if remaining < 30.0:
        return

    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    if z.size != n:
        return
    z = z - float(np.max(z))

    maxfun = int(min(25000, max(800, remaining * 30)))
    maxiter = int(min(6000, max(800, remaining * 8)))
    options = {
        'maxiter': maxiter,
        'maxfun': maxfun,
        'ftol': 1e-15,
        'gtol': 1e-14,
        'maxls': 50,
        'm': 12,
    }

    try:
        minimize(
            _log_lse_objective,
            z,
            args=(beta, int(n), int(M), state),
            method='L-BFGS-B',
            jac=True,
            options=options,
        )
    except Exception:
        pass


def _line_search_phase(z, p, cur_sc, beta, n, M, deadline,
                       max_no_improve=8, tau=0.2, max_iters=1500):
    """
    Local search on z = log(p). Direction is d_j = mean(log h_j) - log h_j,
    which drives the dual variable h toward constancy.
    """
    z = np.asarray(z, dtype=np.float64).ravel().copy()
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    cur_sc = float(cur_sc)
    tau = float(tau)

    it = 0
    no_improve = 0

    while it < int(max_iters) and time.monotonic() < float(deadline):
        p = _softmax(z)
        F, gp, w, h, exact = _lse_grad_p(p, float(beta), int(n), int(M))

        if exact < cur_sc:
            cur_sc = exact
            _register_candidate(cur_sc, p)

        h = np.maximum(h, 1e-15)
        log_h = np.log(h)
        dirv = float(np.mean(log_h)) - log_h
        dirv = np.where(np.isfinite(dirv), dirv, 0.0)

        if float(np.max(np.abs(dirv))) < 1e-13:
            break

        dirv = np.clip(dirv, -3.0, 3.0)

        old_tau = tau
        candidates = (
            min(1.0, max(0.002, old_tau)),
            min(0.8, max(0.002, old_tau * 0.5)),
            min(1.0, max(0.004, old_tau * 2.0)),
            min(0.6, max(0.001, old_tau * 0.25)),
        )

        accepted = False
        for idx, t in enumerate(candidates):
            if time.monotonic() > float(deadline):
                break
            zc = z + float(t) * dirv
            zc = zc - float(np.mean(zc))
            pc = _softmax(zc)
            sc = _exact_score_p(pc, int(n), int(M))
            if sc < cur_sc - 1e-14:
                z = zc
                p = pc
                cur_sc = float(sc)
                accepted = True

                # Adaptive step policy.
                if idx == 2:
                    tau = min(0.8, max(0.02, old_tau * 1.15))
                elif t >= old_tau:
                    tau = min(0.8, max(0.02, old_tau * 1.1))
                else:
                    tau = max(0.01, float(t))
                _register_candidate(cur_sc, p)
                no_improve = 0
                break

        if not accepted:
            no_improve += 1
            tau = max(0.005, old_tau * 0.5)
            if no_improve >= int(max_no_improve):
                break

        it += 1

    return z, p, cur_sc, tau


def _refine_sequence(p0, n, M, deadline, max_iters_per_beta=1500):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    if s <= 1e-15:
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        p /= s

    z = _z_from_p(p)
    cur_sc = _exact_score_p(p, int(n), int(M))
    _register_candidate(cur_sc, p)

    beta_schedule = [
        20, 50, 100, 200, 500,
        30, 1000, 20, 2000, 10,
        800, 5000, 100, 15000, 3000
    ]

    tau = 0.2
    for beta in beta_schedule:
        if time.monotonic() > float(deadline):
            break
        remaining = float(deadline - time.monotonic())
        if remaining < 30.0:
            break
        # For very long runs, let each beta have a modest budget.
        this_max = int(max(80, min(int(max_iters_per_beta), int(remaining * 0.10))))
        z, p, cur_sc, tau = _line_search_phase(
            z, p, cur_sc, float(beta), int(n), int(M), dead line if False else deadline,
            max_no_improve=8,
            tau=tau,
            max_iters=this_max
        )

    state = [cur_sc, p.copy()]
    for beta in [100, 400, 1600, 6400]:
        if time.monotonic() > float(deadline) - 30.0:
            break
        _lbfgs_refine(z, float(beta), int(n), int(M), float(deadline) - 15.0, state)
        if state is not None and state[0] < cur_sc:
            cur_sc = float(state[0])
            p = np.asarray(state[1], dtype=np.float64).ravel().copy()
            z = _z_from_p(p)

    # Final exact-flavored line searches with high beta.
    for beta in [20000, 60000, 150000]:
        if time.monotonic() > float(deadline) - 10.0:
            break
        z, p, cur_sc, tau = _line_search_phase(
            z, p, cur_sc, float(beta), int(n), int(M),
            float(deadline) - 5.0,
            max_no_improve=3,
            tau=0.15,
            max_iters=600
        )

    _register_candidate(cur_sc, p)
    return z, p, cur_sc


def _perturb_restarts(p_best, n, M, deadline, rng, max_restarts=3, max_budget_each=60):
    p_best = np.asarray(p_best, dtype=np.float64).ravel().copy()
    sc_best = _exact_score_p(p_best, int(n), int(M))
    _register_candidate(sc_best, p_best)

    for _ in range(int(max_restarts)):
        if time.monotonic() > float(deadline):
            break

        scale = 0.01 + 0.03 * float(rng.random())
        z = _z_from_p(p_best)
        z = z + scale * rng.standard_normal(n)
        z = z - float(np.mean(z))
        p_try = _softmax(z)
        sc_try = _exact_score_p(p_try, int(n), int(M))

        # Only spend time if the perturbation is not catastrophically worse.
        if sc_try < sc_best * 1.08 + 1e-5:
            local_deadline = min(float(deadline), time.monotonic() + float(max_budget_each))
            for beta in [200, 800, 30]:
                if time.monotonic() > local_deadline:
                    break
                z, p_try, sc_try, _ = _line_search_phase(
                    z, p_try, sc_try, float(beta), int(n), int(M),
                    local_deadline,
                    max_no_improve=4,
                    tau=0.15,
                    max_iters=400
                )
            if sc_try < sc_best:
                sc_best = float(sc_try)
                p_best = np.asarray(p_try, dtype=np.float64).ravel().copy()
                _register_candidate(sc_best, p_best)

    return p_best, sc_best


def _load_parent():
    try:
        if 'height_sequence_1' in globals():
            arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        else:
            return None
    except Exception:
        return None

    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-12:
        return None
    return arr / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _BEST_SCORE, _BEST_P, _PRINTED_SCORE, _LAST_PRINT_TIME

    _BEST_SCORE = float('inf')
    _BEST_P = None
    _PRINTED_SCORE = float('inf')
    _LAST_PRINT_TIME = time.monotonic()

    t0 = time.monotonic()
    total = float(max(60.0, budget_s))
    reserve = 180.0
    if total < reserve + 180.0:
        reserve = 0.2 * total
    deadline = t0 + max(30.0, total - reserve)
    span = max(1.0, deadline - t0)

    rng = np.random.default_rng(seed)
    parent = _load_parent()

    n8 = 8192
    M8 = _fft_size(n8)

    if parent is not None:
        p8 = _resample_to_length(parent, n8)
    else:
        # A small set of cheap initializations if the parent is unavailable.
        best_init = None
        best_init_sc = float('inf')

        cands = [np.full(n8, 1.0 / float(n8), dtype=np.float64)]
        x = np.arange(n8, dtype=np.float64) / float(n8)
        for freq, amp in [(1, 0.35), (2, 0.40), (3, 0.30), (5, 0.50), (1, 0.70)]:
            q = 0.5 + amp * np.cos(2.0 * np.pi * freq * x)
            q = np.maximum(q, 1e-12)
            q /= q.sum()
            cands.append(q)

        for _ in range(12):
            z = rng.normal(0.0, 0.6, size=n8)
            cands.append(_softmax(z))

        for q in cands:
            sc = _exact_score_p(q, n8, M8)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = q.copy()
        p8 = best_init
        _register_candidate(best_init_sc, p8)

    print(f"Starting n=8192 phase, initial/random best score={_BEST_SCORE:.10f}")

    dl8 = t0 + 0.42 * span
    z8, p8, sc8 = _refine_sequence(p8, n8, M8, dl8, max_iters_per_beta=2200)
    print(f"After n=8192 main refinement best={_BEST_SCORE:.10f}")

    # Small number of exploratory restarts from the best 8192 solution.
    base8 = _BEST_P if _BEST_P is not None and _BEST_P.size == n8 else p8
    dl_restart = t0 + 0.55 * span
    if base8 is not None and time.monotonic() < dl_restart:
        base8, _ = _perturb_restarts(base8, n8, M8, dl_restart, rng,
                                     max_restarts=4, max_budget_each=70)
        print(f"After n=8192 restarts best={_BEST_SCORE:.10f}")

    # Optional longer solution, seeded by interpolating the best 8192 mass.
    n16 = 16384
    M16 = _fft_size(n16)
    dl16 = deadline - max(240.0, 0.13 * span)

    if _BEST_P is not None and time.monotonic() < dl16 - 120.0:
        if _BEST_P.size == n8:
            base16 = _BEST_P
        elif _BEST_P.size == n16:
            base16 = _BEST_P
        else:
            base16 = base8 if base8 is not None else p8

        p16 = _resample_to_length(base16, n16)
        sc16 = _exact_score_p(p16, n16, M16)
        print(f"n=16384 seeded score={sc16:.10f}, global best={_BEST_SCORE:.10f}")
        _register_candidate(sc16, p16)

        if time.monotonic() < dl16:
            z16, p16, sc16 = _refine_sequence(p16, n16, M16, dl16, max_iters_per_beta=900)
            print(f"After n=16384 refinement best={_BEST_SCORE:.10f}")

    # Final short polish on whichever global solution survived.
    if _BEST_P is not None:
        final_dl = deadline - 10.0
        pfin = np.asarray(_BEST_P, dtype=np.float64).ravel().copy()
        nfin = int(pfin.size)

        if nfin < 1024 or nfin > 32768:
            pfin = np.full(8192, 1.0 / 8192.0, dtype=np.float64)
            nfin = 8192

        Mfin = _fft_size(nfin)
        zfin = _z_from_p(pfin)
        scfin = _exact_score_p(pfin, nfin, Mfin)
        _register_candidate(scfin, pfin)

        for beta in [100, 500, 2000, 8000]:
            if time.monotonic() > final_dl - 5.0:
                break
            zfin, pfin, scfin, _ = _line_search_phase(
                zfin, pfin, scfin, float(beta), nfin, Mfin,
                final_dl - 5.0,
                max_no_improve=3,
                tau=0.18,
                max_iters=350
            )

        state = [scfin, pfin.copy()]
        if time.monotonic() < final_dl - 25.0:
            _lbfgs_refine(zfin, 2500.0, nfin, Mfin, final_dl - 15.0, state)

        if state is not None and state[0] < _BEST_SCORE:
            _register_candidate(state[0], state[1])

    # Construct and verify final sequence.
    if _BEST_P is None:
        out_arr = np.full(8192, 1.0 / 8192.0, dtype=np.float64)
    else:
        out_arr = np.asarray(_BEST_P, dtype=np.float64).ravel().copy()
        out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    s = float(out_arr.sum())
    if out_arr.size == 0 or s < 0.01:
        out_arr = np.full(8192, 1.0 / 8192.0, dtype=np.float64)
    else:
        out_arr = out_arr / s

    out = [float(x) for x in out_arr.tolist()]

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float('inf')

    print(f"Final internal best={_BEST_SCORE:.10f} verified={verified:.10f} len={len(out)} elapsed={time.monotonic()-t0:.1f}s")

    if not np.isfinite(verified):
        out = [1.0 / 8192.0] * 8192

    return out
```
