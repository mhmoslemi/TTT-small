    M = 2*n - 1

    A = np.fft.rfft(a, M_fft)

    c = fast_conv(a)
    S = a.sum()

def proj_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0}."""
    n = len(v)
    if n == 0: return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Project v onto simplex {x: sum(x)=target, x>=0}."""
    n = len(v)

"""
Step function optimization: Projected LSE with BB step, peak variance equalization,
multi-peak subgradient polishing, and strategic restarts.
Targets score < 1.5027 by exploiting active-set structure and variance flattening.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def proj_simplex(v, target=1.0):
    """Project v onto {x: sum(x)=target, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if target <= 0:
        return np.zeros(n)
    w = v / target
    u = np.sort(w)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) > 0:
        rho = idx[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(w - theta, 0.0) * target


def lse_score_grad_unnorm(a, beta):
    """
    LSE-smoothed score and gradient for unnormalized a.
    score = 2n * LSE_beta(conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    e_sum = float(e.sum())
    if e_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / e_sum

    L = c_max + np.log(e_sum) / beta
    score = 2.0 * n * L / (S * S)

    # Compute g[i] = sum_k p[k] * dc[k]/da[i]
    # g[i] = cross_corr(p,a)[i] + conv(p,a)[i]
    Lp = 2 * n - 1  # length of p
    L_fft = 1 << (3 * n - 3).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)

    # conv(p, a)
    pa = np.fft.irfft(P * A, L_fft)

    # cross_corr(p,a)[i] = conv(p, a_rev)[n-1+i]
    a_rev = a[::-1].copy()
    AR = np.fft.rfft(a_rev, L_fft)
    pa_rev = np.fft.irfft(P * AR, L_fft)

    cross = pa_rev[n - 1:2 * n - 1]
    conv_part = pa[:n]
    g = cross + conv_part

    # Gradient of 2n*L/S^2: = 2n*(g*S - 2*L)/S^3
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g * S - 2.0 * L) / S3

    return score, grad


def lse_obj_wrapper(a, beta):
    """Objective + gradient wrapper for scipy minimize."""
    return lse_score_grad_unnorm(a, beta)


def variance_obj(a):
    """Variance of convolution as objective (smooth, for initial shaping)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    # Gradient: d/dc2 of c[k] = 2*a convolved with a (shifted)
    # d(comove k)/a[i] = a[k-i] + a[i-k], same as before
    # d(c^2)/a[i] = 2 * sum_k c[k] * dc[k]/a[i]
    Lp = 2 * n - 1
    L_fft = 1 << (3 * n - 3).bit_length()
    C = np.fft.rfft(c, L_fft)
    A = np.fft.rfft(a, L_fft)
    ca = np.fft.irfft(C * A, L_fft)
    a_rev = a[::-1].copy()
    AR = np.fft.rfft(a_rev, L_fft)
    ca_rev = np.fft.irfft(C * AR, L_fft)
    cross = ca_rev[n - 1:2 * n - 1]
    conv_part = ca[:n]
    gc = 2.0 * (cross + conv_part)

    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (gc / S4 - 4.0 * c2 / S5)
    return score, grad


def peak_variance_grad(a, top_p=20):
    """
    Gradient of variance of top-P convolution peaks.
    Returns the direction that should be descended to equalize peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n), 0.0
    sorted_idx = np.argsort(c)[::-1][:top_p]
    vals = c[sorted_idx]
    c_bar = vals.mean()
    weights = 2.0 * (vals - c_bar) / top_p

    # dV/da = sum_j weights[j] * dc[k_j]/da
    # dc[k]/da[i] = a[k-i] + a[i-k]
    Lp = 2 * n - 1
    L_fft = 1 << (3 * n - 3).bit_length()
    W = np.zeros(Lp)
    for j in range(len(sorted_idx)):
        W[sorted_idx[j]] = weights[j]
    PW = np.fft.rfft(W, L_fft)
    A = np.fft.rfft(a, L_fft)
    pa = np.fft.irfft(PW * A, L_fft)
    a_rev = a[::-1].copy()
    AR = np.fft.rfft(a_rev, L_fft)
    pa_rev = np.fft.irfft(PW * AR, L_fft)
    cross = pa_rev[n - 1:2 * n - 1]
    conv_part = pa[:n]
    grad = cross + conv_part

    # Scale by 2n/S^2 for consistency with score gradient
    score = 2.0 * n * float(np.var(vals)) / (S * S)
    return grad * 2.0 * n / (S * S), score


def subgradient_polish(a, iters=5000, top_p=5):
    """
    Subgradient descent on the exact max with multi-peak variance reduction.
    Uses Polyak step size.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find top P peaks
        sorted_idx = np.argsort(c)[::-1][:top_p]
        top_vals = c[sorted_idx]
        c_bar = top_vals.mean()

        # Compute combined gradient: max + variance
        # Max subgradient
        k_star = int(sorted_idx[0])
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_max = np.zeros(n)
        if j_end >= j_start:
            sg_max[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Variance gradient
        Lp = 2 * n - 1
        L_fft = 1 << (3 * n - 3).bit_length()
        W = np.zeros(Lp)
        weights = 2.0 * (top_vals - c_bar) / top_p
        for j in range(len(sorted_idx)):
            W[sorted_idx[j]] = weights[j]
        PW = np.fft.rfft(W, L_fft)
        A = np.fft.rfft(a, L_fft)
        pa = np.fft.irfft(PW * A, L_fft)
        a_rev = a[::-1].copy()
        AR = np.fft.rfft(a_rev, L_fft)
        pa_rev = np.fft.irfft(PW * AR, L_fft)
        sg_var = pa_rev[n - 1:2 * n - 1] + pa[:n]

        # Combined
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * ((sg_max + sg_var) / S2 - 2.0 * M / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step
        sc_lb = 1.0
        step = (sc - sc_lb) / (gn * gn + 1e-20)
        step = min(step, 0.5 / (S + 1e-20))
        step = max(step, 1e-8 / (gn * S + 1e-20))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        # Renormalize to keep sum stable
        S_new = a.sum()
        if S_new > 1e-12:
            a *= S / S_new

    return best_sc, best_a


def lbfgs_optimize(a0, func, args, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B optimization with bounds [0, 1000]."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")
        return a0


def try_truncate(a, max_remove_frac=0.3):
    """Try removing small elements from the ends to reduce n."""
    n = len(a)
    best = compute_score(a)
    best_a = a.copy()
    max_remove = int(n * max_remove_frac)
    left = 0
    right = 0
    # Try removing from left
    for i in range(1, max_remove + 1):
        if a[i - 1] < 0.01 * a.max():
            cand = a[i:]
            sc = compute_score(cand)
            if sc < best:
                best = sc
                best_a = cand.copy()
            left = i
        else:
            break
    # Try removing from right
    for i in range(1, max_remove + 1):
        if a[n - i] < 0.01 * a.max():
            cand = a[:n - i]
            sc = compute_score(cand)
            if sc < best:
                best = sc
                best_a = cand.copy()
            right = i
        else:
            break
    # Try removing from both
    if left > 0 and right > 0:
        cand = a[left:n - right]
        sc = compute_score(cand)
        if sc < best:
            best = sc
            best_a = cand.copy()
    return best, best_a

            if maxit < 50:
                break
            a_cur = lbfgs_optimize(a_cur, lse_obj_wrapper, (beta,),
                                   maxiter=maxit, maxfun=maxit * 30)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"LSE b={beta} r{rnd}")
        sc = compute_score(a_cur)
        print(f"  beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===== Phase 3: Peak Variance Equalization (15% time) =====
    print("\n=== Phase 3: Peak Variance Equalization ===")
    phase3_end = t_start + 0.70 * total_budget
    a_cur = best_a.copy()
    n = len(a_cur)

    pv_iters = int(min(3000, (phase3_end - time.monotonic()) * 80))
    if pv_iters > 100 and time.monotonic() < phase3_end:
        t0 = time.monotonic()
        top_p = max(5, n // 500)
        a_pved = a_cur.copy()
        S_orig = a_pved.sum()
        prev_sc = compute_score(a_pved)

        for it in range(pv_iters):
            if time.monotonic() > phase3_end:
                break
            grad, _ = peak_variance_grad(a_pved, top_p=top_p)
            S = a_pved.sum()
            S2 = S * S
            S3 = S2 * S
            # Add the sum-constraint correction
            c = fast_conv(a_pved)
            M = c.max()
            sg_max = np.zeros(n)
            k_star = int(np.argmax(c))
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            if je >= js:
                sg_max[js:je + 1] = 2.0 * a_pved[js:je + 1][::-1]
            combined = grad + 0.3 * (2.0 * n * (sg_max / S2 - 2.0 * M / S3))

            gn = np.linalg.norm(combined)
            if gn < 1e-15:
                break
            step = 0.1 / (gn * S + 1e-20)
            step = min(step, 0.3 / S)

            a_pved -= step * combined
            a_pved = np.clip(a_pved, 0.0, 1000.0)
            S_new = a_pved.sum()
            if S_new > 1e-12:
                a_pved *= S / S_new

            if it % 500 == 0:
                sc = compute_score(a_pved)
                report(sc, a_pved, f"PV eq it{it}")

        sc = compute_score(a_pved)
        report(sc, a_pved, "PV equalized")
        print(f"  PV phase: {time.monotonic()-t0:.1f}s, score={sc:.8f}")
        if sc < best_score:
            a_cur = a_pved

    # ===== Phase 4: High-beta LSE refinement (10% time) =====
    print("\n=== Phase 4: High-Beta Refinement ===")
    phase4_end = t_start + 0.80 * total_budget
    a_cur = best_a.copy()
    for beta in [4000, 8000, 15000]:
        if time.monotonic() > phase4_end:
            break
        t0 = time.monotonic()
        a_cur = lbfgs_optimize(a_cur, lse_obj_wrapper, (beta,),
                               maxiter=1000, maxfun=25000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"refine b={beta}")
        print(f"  beta={beta}: score={sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # ===== Phase 5: Subgradient Polishing (10% time) =====
    print("\n=== Phase 5: Subgradient Polishing ===")
    phase5_end = t_start + 0.90 * total_budget
    if time.monotonic() < phase5_end:
        t0 = time.monotonic()
        remaining = phase5_end - time.monotonic()
        iters = int(min(8000, remaining * 150))
        sc, a_pol = subgradient_polish(best_a, iters=max(500, iters), top_p=8)
        report(sc, a_pol, "subgrad polish")
        print(f"  Subgrad: {time.monotonic()-t0:.1f}s, score={sc:.8f}")

        # Second round with different top_p
        if time.monotonic() < phase5_end:
            sc2, a_pol2 = subgradient_polish(best_a, iters=2000, top_p=3)
            report(sc2, a_pol2, "subgrad polish P=3")

    # ===== Phase 6: Targeted Restarts (10% time) =====
    print("\n=== Phase 6: Targeted Restarts ===")
    restart_count = 0
    while time.monotonic() < deadline and restart_count < 15:
        t0 = time.monotonic()
        strategy = restart_count % 5

        if strategy == 0:
            # Small perturbation + LSE
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Reduce top contributors to the max peak, boost others
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            js = max(0, k_star - len(best_a) + 1)
            je = min(k_star, len(best_a) - 1)
            contrib = np.zeros(len(best_a))
            if je >= js:
                contrib[js:je + 1] = best_a[js:je + 1][::-1]
            top_idx = np.argsort(contrib)[::-1][:max(5, len(best_a) // 50)]
            bot_idx = np.argsort(contrib)[:max(5, len(best_a) - len(top_idx))]
            a_pert = best_a.copy()
            total_red = 0.0
            for j in top_idx:
                red = a_pert[j] * 0.05
                a_pert[j] -= red
                a_pert[j] = max(a_pert[j], 0.0)
                total_red += red
            if len(bot_idx) > 0 and total_red > 0:
                per = total_red / len(bot_idx)
                a_pert[bot_idx] += per
        elif strategy == 2:
            # Frequency domain perturbation
            nn = len(best_a)
            L = 1 << (2 * nn - 2).bit_length()
            A = np.fft.rfft(best_a, L)
            A_mag = np.abs(A)
            A_phase = np.angle(A)
            A_mag_new = A_mag * (1.0 + 0.03 * rng.standard_normal(len(A)))
            A_mag_new = np.clip(A_mag_new, 0, None)
            A_new = A_mag_new * np.exp(1j * A_phase)
            a_pert = np.fft.irfft(A_new, L)[:nn]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Try truncation
            sc_tr, a_tr = try_truncate(best_a, max_remove_frac=0.1)
            a_pert = a_tr
        else:
            # Sinusoidal modulation
            nn = len(best_a)
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.02 * rng.random()
            x = np.arange(nn, dtype=np.float64) / nn
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick optimization from perturbed start
        a_opt = lbfgs_optimize(a_pert, lse_obj_wrapper, (3000,),
                               maxiter=500, maxfun=12000)
        sc = compute_score(a_opt)
        report(sc, a_opt, f"restart {restart_count} (s{strategy})")

        # Quick subgradient polish
        sc2, a_pol = subgradient_polish(a_opt, iters=500, top_p=5)
        report(sc2, a_pol, f"restart {restart_count} polish")

        restart_count += 1

    # ===== Phase 7: Final Polish =====
    print("\n=== Phase 7: Final Polish ===")
    if time.monotonic() < deadline:
        # One more subgradient pass
        sc, a_fin = subgradient_polish(best_a, iters=3000, top_p=10)
        report(sc, a_fin, "final subgrad")

        # Final LSE
        a_fin2 = lbfgs_optimize(best_a, lse_obj_wrapper, (5000,),
                                maxiter=500, maxfun=10000)
        sc2 = compute_score(a_fin2)
        report(sc2, a_fin2, "final LSE")

    # ===== Output =====
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
