"""
Improved optimizer for minimum peak autocorrelation of non-negative sequences.
Uses aggressive spectral flattening, L-BFGS beta-continuation (up to beta=20000),
exact subgradient polishing with Polyak steps, and iterative peak reduction.
Explores parent length plus alternative lengths for diversity.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=300, lam=0.7):
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def spectral_flatten_anneal(a0, N, n_iters=400):
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = 0.9 * (1.0 - progress) + 0.3
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def subgrad_polish(a, total_iters=2000, lower_bound=1.0):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        gap = sc - lower_bound
        if gap < 1e-12:
            gap = 1e-6
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step2 = 0.005 / (1.0 + 0.05 * it)
        a -= step2 * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def peak_reduction(a, n_steps=200, strength=0.03):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for s in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        sc = 2.0 * n * float(c[k_star]) / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        contrib = np.zeros(n)
        seg = a[j_start:j_end + 1]
        contrib[j_start:j_end + 1] = seg[::-1]
        top_k = max(5, n // 50)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bot_idx = np.argsort(contrib)[:top_k]
        reduce_total = 0.0
        for j in top_idx:
            amt = a[j] * strength
            a[j] -= amt
            a[j] = max(a[j], 0.0)
            reduce_total += amt
        if reduce_total > 0 and len(bot_idx) > 0:
            per = reduce_total / len(bot_idx)
            for j in bot_idx:
                a[j] += per
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

            sf_results[key] = (sc, sf_a)
            report(sc, sf_a, key)
            elapsed = time.monotonic() - t0
            print(f"    {key}: score={sc:.6f} ({elapsed:.1f}s)")

    # Phase 2: L-BFGS beta continuation
    lbfgs_deadline = t_start + 0.65 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Beta Continuation (<= {lbfgs_deadline - t_start:.0f}s) ---")

    candidates = list(sf_results.values())
    if parent is not None:
        candidates.append((compute_score(parent), parent.copy()))

    beta_schedule = [15, 40, 100, 300, 1000, 3000, 8000]

    for n in lengths:
        if time.monotonic() > lbfgs_deadline:
            break
        print(f"\n  >> Optimizing n = {n}")
        cand_n = [(sc, a) for sc, a in candidates if len(a) == n]
        if not cand_n:
            N_fft = 1 << (2 * n - 1).bit_length()
            cand_n.append((compute_score(np.ones(n)*0.5), np.ones(n)*0.5))
            cand_n.append((compute_score(np.clip(0.5+0.3*rng.standard_normal(n),0.01,10)), np.clip(0.5+0.3*rng.standard_normal(n),0.01,10)))
        cand_n.sort(key=lambda x: x[0])
        top_cands = cand_n[:min(5, len(cand_n))]

        for ci, (sc_init, a_init) in enumerate(top_cands):
            if time.monotonic() > lbfgs_deadline:
                break
            t0 = time.monotonic()
            a_cur = a_init.copy()
            for beta in beta_schedule:
                if time.monotonic() > lbfgs_deadline:
                    break
                remaining = lbfgs_deadline - time.monotonic()
                if remaining < 3:
                    break
                est_per_iter = 0.001 * (n / 4096)
                maxiter = min(5000, int(remaining * 0.7 / max(est_per_iter * 15, 0.05)))
                maxiter = max(200, maxiter)
                maxfun = max(1000, maxiter * 25)
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
                a_cur = a_opt
                if report(sc, a_opt, f"lbfgs n={n} i={ci} b={beta}"):
                    pass
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, f"lbfgs_done n={n} i={ci}")
            elapsed = time.monotonic() - t0
            print(f"    n={n} i={ci}: score={sc_final:.8f} ({elapsed:.1f}s)")

    print(f"\nAfter Phase 2: best = {best_score:.10f} (n={len(best_a)})")

    # Phase 3: Subgradient + peak reduction polishing
    sg_deadline = t_start + 0.82 * effective_budget
    print(f"\n--- Phase 3: Subgradient + Peak Reduction (<= {sg_deadline - t_start:.0f}s) ---")

    n_best = len(best_a)
    for round_idx in range(3):
        if time.monotonic() > sg_deadline:
            break
        sc, a_sg = subgrad_polish(best_a, total_iters=2000, lower_bound=1.0)
        report(sc, a_sg, f"subgrad_r{round_idx}")
        sc2, a_pr = peak_reduction(a_sg, n_steps=300, strength=0.02)
        report(sc2, a_pr, f"peakred_r{round_idx}")
        sc3, a_sg2 = subgrad_polish(a_pr, total_iters=1000, lower_bound=1.0)
        report(sc3, a_sg2, f"subgrad2_r{round_idx}")

    # Phase 4: Local search with diverse perturbations
    local_deadline = t_start + 0.94 * effective_budget
    print(f"\n--- Phase 4: Local Search (<= {local_deadline - t_start:.0f}s) ---")

    n_l = len(best_a)
    strategies = 50
    for trial in range(strategies):
        if time.monotonic() > local_deadline:
            break
        strategy = trial % 6
        if strategy == 0:
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_l))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = best_a.copy()
            idx = rng.choice(n_l, size=n_l // 20, replace=False)
            a_pert[idx] *= (0.5 + rng.random())
            renorm = a_pert.sum() / best_a.sum()
            a_pert *= renorm
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.05 * rng.random()
            x = np.arange(n_l, dtype=np.float64) / n_l
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            N_fft = 1 << (2 * n_l - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise = 0.03 * rng.standard_normal(len(A)) + 0.03j * rng.standard_normal(len(A))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_l]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_l + 1)
            j_end = min(k_star, n_l - 1)
            contrib = np.zeros(n_l)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            top_k = max(5, n_l // 40)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            a_pert = best_a.copy()
            red = 0.0
            for j in top_idx:
                amt = a_pert[j] * 0.05
                a_pert[j] -= amt
                a_pert[j] = max(a_pert[j], 0.0)
                red += amt
            bot_idx = np.argsort(contrib)[:top_k]
            for j in bot_idx:
                a_pert[j] += red / top_k
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            x = np.arange(n_l, dtype=np.float64)
            f = 1.0 + 6.0 * rng.random()
            pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n_l))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        remaining = local_deadline - time.monotonic()
        if remaining < 2:
            break
        maxiter = min(3000, int(remaining * 0.3 / max(0.001 * (n_l / 4096), 0.02)))
        maxiter = max(200, maxiter)
        sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=maxiter, maxfun=max(1000, maxiter * 20))
        sc2, a_pg = subgrad_polish(a_opt, total_iters=500, lower_bound=1.0)
        if sc2 < sc:
            sc, a_opt = sc2, a_pg
        report(sc, a_opt, f"local#{trial} s={strategy}")

    # Phase 5: Final intensive polish
    print(f"\n--- Phase 5: Final Polish ---")
    sc, a_pol = subgrad_polish(best_a, total_iters=3000, lower_bound=1.0)
    report(sc, a_pol, "final_subgrad")

    if time.monotonic() < deadline - 10:
        for beta in [5000, 15000, 20000]:
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            maxiter = min(8000, int(remaining / max(0.002 * (len(best_a) / 4096), 0.05)))
            maxiter = max(500, maxiter)
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=max(2000, maxiter * 20))
            report(sc, a_opt, f"final_lbfgs b={beta}")

    sc, a_pol2 = peak_reduction(best_a, n_steps=500, strength=0.01)
    report(sc, a_pol2, "final_peakred")

    sc, a_pol3 = subgrad_polish(best_a, total_iters=2000, lower_bound=1.0)
    report(sc, a_pol3, "final_sg2")

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
