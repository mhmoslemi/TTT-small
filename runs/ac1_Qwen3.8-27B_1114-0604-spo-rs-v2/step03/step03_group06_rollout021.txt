Optimized step function minimizer using coordinate descent with analytical
per-coordinate optimization, LSE-smoothed L-BFGS-B, and LP-based refinement.
Key innovation: fast incremental coordinate descent that exploits the structure
of the self-convolution to achieve O(n) per-coordinate updates, enabling

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient via FFT."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient computation via cross-correlation
    N_fft = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    cross = np.fft.irfft(np.conj(A_fft) * P_fft, N_fft)[:n]
    g = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)

    def objective(x):
        sc, gr = lse_score_grad(x, beta)
        return sc, gr

    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,

                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:

    exact_sc = compute_score(a_out)

                # Also try neighbors of current value
                candidates.append(max(0.0, x_old * 0.5))
                candidates.append(x_old * 1.5)

                # Evaluate candidates using incremental update
                best_x = x_old
                best_local_sc = float('inf')
                k_all = np.arange(2 * n - 1)
                abs_kj = np.abs(k_all - j)
                mask_valid = abs_kj < n

                for x_new in candidates:
                    dx = x_new - x_old
                    if abs(dx) < 1e-12:
                        sc = compute_score_local(conv, S, n)
                    else:
                        # Incremental update
                        delta = np.zeros(2 * n - 1)
                        delta[mask_valid] = 2.0 * dx * a[abs_kj[mask_valid]]
                        if 2 * j < 2 * n - 1:
                            delta[2 * j] += dx * (x_new + x_old)
                        conv_new = conv + delta
                        S_new = S + dx
                        sc = compute_score_local(conv_new, S_new, n)

                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_x = x_new

                # Apply best update
                dx = best_x - x_old
                if abs(dx) > 1e-12:
                    delta = np.zeros(2 * n - 1)
                    delta[mask_valid] = 2.0 * dx * a[abs_kj[mask_valid]]
                    if 2 * j < 2 * n - 1:
                        delta[2 * j] += dx * (best_x + x_old)
                    conv = conv + delta
                    S += dx
                    a[j] = best_x

                    sc_now = compute_score_local(conv, S, n)
                    if sc_now < best_sc:
                        best_sc = sc_now
                        best_a = a.copy()

        # Periodically re-verify with exact computation
        if pass_num % 20 == 19:
            conv = fast_conv(a)
            S = float(a.sum())
            sc_exact = compute_score(a)
            if sc_exact < best_sc:
                best_sc = sc_exact
                best_a = a.copy()
            top_lags = np.argsort(conv)[::-1][:top_k_active]

    return best_sc, best_a


def compute_score_local(conv, S, n):
    """Compute score from precomputed convolution and sum."""
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)


def solve_lp_direction(a, max_wall_s=30.0):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()
    if n > 5000:
        # For larger n, use a banded/sparse formulation
        pass

    # Build constraint matrix (banded)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step(a, max_wall_s=20.0):
    """Perform LP-based direction step with adaptive step size."""
    old_sc = compute_score(a)
    b_opt = solve_lp_direction(a, max_wall_s=max_wall_s)
    if b_opt is None:
        return old_sc, a
    best_sc = old_sc
    best_a = a
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
    return best_sc, best_a


def exact_subgrad_polish(a, total_iters=200, deadline=None):
    """Projected subgradient descent on exact max-conv objective with Nesterov momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    y = a.copy()  # Nesterov position
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    beta_mom = 0.9

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        # Subgradient of max(c)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = y[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)

        # Nesterov update
        v = beta_mom * v - step * grad
        y_new = y + v
        y = np.clip(y_new, 0.0, 1000.0)
        a = np.clip(0.5 * (a + y), 0.0, 1000.0)

        sc_y = compute_score(y)
        if sc_y < best_sc:
            best_sc = sc_y
            best_a = y.copy()
        sc_a = compute_score(a)
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a.copy()

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    z = 2.0 * np.pi * x
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))

    # Block structure
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    # Alternating
    a_alt = np.ones(n)
    a_alt[0::2] = 1.0
    a_alt[1::2] = 0.3
    inits.append(a_alt * 0.5)

    # Thue-Morse inspired
    tm = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = 1.0 if cnt % 2 == 0 else 0.1
    inits.append(tm * 0.5)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses coordinate descent, L-BFGS-B with LSE, LP refinement, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [0, 200, -200, 400, -400, 800, -800]:
            cand = pa + dl
            if cand >= 2048:
                lengths.append(cand)
    for n in [4096, 6962, 7200, 8192]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")
    print(f"CVXPY available: {HAVE_CVXPY}")

    # =========================================================================
    # Phase 1: Coordinate Descent (40% of time)
    # =========================================================================
    cd_deadline = t_start + 0.40 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: Coordinate Descent (deadline at {cd_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > cd_deadline:
            break
        print(f"\n--- Coordinate Descent: n = {n} ---")
        inits = build_inits(n, rng, parent)
        remaining = cd_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 60.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > cd_deadline:
                break
            a_init = np.clip(a0, 0.0, 1000.0)
            if a_init.sum() < 0.01:
                a_init[:] = 0.5
            sc = compute_score(a_init)
            report(sc, a_init, f"CD init n={n}, i={i}")

            # Run coordinate descent
            max_passes = max(50, int((cd_deadline - time.monotonic()) / 5.0))
            sc, a_opt = coordinate_descent(a_init, max_passes, cd_deadline,
                                          top_k_active=12, seed=seed + i * 100 + n)
            report(sc, a_opt, f"CD n={n}, init={i}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # =========================================================================
    # Phase 2: L-BFGS-B with LSE continuation (25% of time)
    # =========================================================================
    lbfgs_deadline = t_start + 0.65 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS-B LSE (deadline at {lbfgs_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    beta_schedule = [50, 200, 500, 1000, 2000, 4000, 8000]

    for beta in beta_schedule:
        if time.monotonic() > lbfgs_deadline:
            break
        remaining = lbfgs_deadline - time.monotonic()
        maxiter = min(5000, int(remaining / 10.0))
        if maxiter < 100:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=max(maxiter * 3, 1000))
        report(sc, a_opt, f"LBFGS beta={beta}")

    # Try also from slightly different starting points
    if time.monotonic() < lbfgs_deadline - 30:
        for trial in range(3):
            if time.monotonic() > lbfgs_deadline:
                break
            scale = 0.05 + 0.1 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=3000, maxfun=6000)
            report(sc, a_opt, f"LBFGS perturb {trial}")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================================
    # Phase 3: More Coordinate Descent on best solution (15% of time)
    # =========================================================================
    cd2_deadline = t_start + 0.80 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate Descent Round 2 (deadline at {cd2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < cd2_deadline:
        max_passes = max(100, int((cd2_deadline - time.monotonic()) / 3.0))
        sc, a_opt = coordinate_descent(best_a, max_passes, cd2_deadline,
                                      top_k_active=15, seed=seed + 999)
        report(sc, a_opt, "CD round 2")

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: LP-based refinement (10% of time)
    # =========================================================================
    lp_deadline = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: LP Refinement (deadline at {lp_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    if HAVE_CVXPY and len(best_a) <= 5000 and time.monotonic() < lp_deadline:
        a_lp = best_a.copy()
        for it_lp in range(20):
            if time.monotonic() > lp_deadline:
                break
            sc, a_new = lp_step(a_lp, max_wall_s=15.0)
            if sc < best_score:
                report(sc, a_new, f"LP step {it_lp}")
            a_lp = a_new
            # L-BFGS polish after LP
            if time.monotonic() < lp_deadline - 10:
                sc, a_pol, _ = run_lbfgs(a_lp, 1000, maxiter=2000, maxfun=4000)
                report(sc, a_pol, f"LP+LBFGS {it_lp}")
                a_lp = a_pol
    else:
        print("  Skipping LP (no CVXPY or too long)")
        # Alternative: subgradient + CD
        if time.monotonic() < lp_deadline:
            sc, a_opt = exact_subgrad_polish(best_a, total_iters=200, deadline=lp_deadline)
            report(sc, a_opt, "subgrad phase 4")

    print(f"\nAfter Phase 4: best = {best_score:.8f}")

    # =========================================================================
    # Phase 5: Final refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Refinement")
    print(f"{'='*60}")

    # High-beta LSE
    for beta in [3000, 6000, 10000]:
        if time.monotonic() > deadline - 15:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final LBFGS beta={beta}")

    # Coordinate descent final
    if time.monotonic() < deadline - 20:
        remaining_s = deadline - time.monotonic() - 20
        max_passes = max(50, int(remaining_s / 3.0))
        sc, a_opt = coordinate_descent(best_a, max_passes, deadline - 20,
                                      top_k_active=15, seed=seed + 7777)
        report(sc, a_opt, "final CD")

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=500, deadline=deadline - 5)
        report(sc, a_pol, "final subgrad")

    # Final L-BFGS
    if time.monotonic() < deadline - 5:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final final LBFGS")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
