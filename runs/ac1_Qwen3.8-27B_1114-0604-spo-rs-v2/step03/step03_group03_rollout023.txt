"""
Multi-strategy optimizer for minimum peak autocorrelation of non-negative sequences.
Uses LSE gradient descent (fast, many iterations) combined with coordinate descent
(for precise local optimization) and L-BFGS refinement. Key innovation: centered
LSE gradient steps that maintain the sum while reducing the peak efficiently.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient: dL/da[j] = sum_k p[k]*(a[k-j] + a[j-k])
    # = (p*a)[j] + (p*rev_a)[j+n-1]
    rev_a = a[::-1]
    M_total = 3 * n - 2
    M_fft = 1 << (M_total - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    RA = np.fft.rfft(rev_a, M_fft)
    p_conv_a = np.fft.irfft(P * A, M_fft)[:n]
    p_conv_reva = np.fft.irfft(P * RA, M_fft)[n - 1:n - 1 + n]
    grad_c = p_conv_a + p_conv_reva
    S2 = S * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * L / (S2 * S))
    return score, grad


def lse_obj_jac(x, beta):
    """Wrapper for scipy minimize."""
    return lse_grad(x, beta)


def lse_gradient_descent(a, betas, max_iter=10000, t_start=0.0, time_limit=2000.0):
    """
    LSE gradient descent with centered steps, non-negativity projection,
    and adaptive step size. This is the main exploration engine.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    
    for beta in betas:
        if time.monotonic() - t_start > time_limit:
            break
        score_cur = compute_score(a)
        lr = 0.1 / (np.sqrt(n) * beta * 0.01 + 1.0)
        no_improve = 0
        
        for it in range(max_iter):
            if time.monotonic() - t_start > time_limit:
                break
            if it % 2000 == 0 and it > 0:
                sc = compute_score(a)
                print(f"    LSE-GD beta={beta} iter={it}: {sc:.8f}")
                no_improve = 0
                
            _, grad = lse_grad(a, beta)
            # Center to maintain sum
            grad_c = grad - np.mean(grad)
            # Step
            a_new = a - lr * grad_c
            a_new = np.maximum(a_new, 0.0)
            
            new_score = compute_score(a_new)
            
            if new_score < score_cur:
                a = a_new
                score_cur = new_score
                no_improve = 0
                if new_score < score_cur * 1.001:
                    lr *= 1.02
            else:
                no_improve += 1
                lr *= 0.8
                if no_improve > 50:
                    break
        
        final_sc = compute_score(a)
        print(f"  LSE-GD beta={beta} done: {final_sc:.8f}")
    
    return a


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent (top-level)."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def cd_sweep(a, order_type='random', rng=None):
    """
    One coordinate descent sweep with exact 1D minimization.
    order_type: 'sequential', 'random', 'hot', 'coldest'
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if order_type == 'random' and rng is not None:
        order = rng.permutation(n)
    elif order_type == 'hot':
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    elif order_type == 'coldest':
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)
    else:
        order = np.arange(n)

    for i in order:
        i = int(i)
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def multi_peak_deflation(a, n_peaks=3, n_steps=200, t_start=0.0, time_limit=2000.0):
    """
    Multi-peak deflation: simultaneously reduce the top peaks using a combined
    gradient. This is faster than coordinate descent for global moves.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    beta = 200.0  # Focus on top peaks
    lr = 0.5 / (n ** 0.5)
    
    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        
        # LSE gradient (focuses on top peaks due to high beta)
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        
        # Compute gradient of sum_k p_k * c_k w.r.t. a
        rev_a = a[::-1]
        M_total = 3 * n - 2
        M_fft = 1 << (M_total - 1).bit_length()
        P = np.fft.rfft(p, M_fft)
        A = np.fft.rfft(a, M_fft)
        RA = np.fft.rfft(rev_a, M_fft)
        p_conv_a = np.fft.irfft(P * A, M_fft)[:n]
        p_conv_reva = np.fft.irfft(P * RA, M_fft)[n - 1:n - 1 + n]
        grad_c = p_conv_a + p_conv_reva
        
        # Center and step
        grad_c = grad_c - np.mean(grad_c)
        a_new = a - lr * grad_c
        a_new = np.maximum(a_new, 0.0)
        
        new_score = compute_score(a_new)
        cur_score = 2.0 * n * float(c.max()) / (S * S)
        
        if new_score < cur_score:
            a = a_new
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break
    
    return a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_jac, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def perturb_targeted(a, rng, strength=0.03):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def perturb_gaussian(a, rng, scale=0.02):
    """Gaussian perturbation."""
    n = len(a)
    a_pert = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_pert, 0.0, 1000.0)


def perturb_block(a, rng):
    """Block zeroing + redistribution."""
    n = len(a)
    a_pert = a.copy()
    block_size = max(20, n // 80)
    start = int(rng.integers(0, max(1, n - block_size)))
    saved = a_pert[start:start + block_size].sum()
    a_pert[start:start + block_size] = 0.0
    rest_mask = np.ones(n, dtype=bool)
    rest_mask[start:start + block_size] = False
    a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
    return np.clip(a_pert, 0.0, 1000.0)
