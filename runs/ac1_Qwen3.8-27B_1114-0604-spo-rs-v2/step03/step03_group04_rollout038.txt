Optimized active-set + L-BFGS + coordinate descent for minimum peak autocorrelation.
Targets improvement from 1.509 toward 1.5027 by using multi-lag subgradient,
adaptive continuation, and coordinate refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, c, 0
    score = 2.0 * n * float(np.max(c)) / (S * S)
    return score, c, int(np.argmax(c))


def lse_obj_grad(a, beta):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: grad_L[i] = 2 * sum_{m=0}^{n-1} a[m] * p[m+i]
    # Use FFT: conv(rev_a, p)[n-1+i] for i=0..n-1
    # conv length = n + M - 1 = 3n - 2
    M2 = 1 << (3 * n - 3).bit_length()
    rev_a = a[::-1]
    FA = np.fft.rfft(rev_a, M2)
    FP = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(FA * FP, M2)
    r = conv_res[n - 1:2 * n - 1]
    grad_L = 2.0 * r
    S2 = S * S
    grad = (2.0 * n / (S2 * S)) * (S * grad_L - 2.0 * L)
    return score, grad

    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc, _, _ = compute_score(a_out)
    return sc, a_out

    """Take a subgradient step using the top n_lags autocorrelation peaks."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    M = len(c)
    # Get top lags
    top_idx = np.argsort(c)[::-1][:n_lags]
    # Compute weighted gradient
    # For each active lag k, grad_k[i] = 2*a[k-i] (if 0 <= k-i < n)

    FP = np.fft.rfft(p_ext, M2)
    conv_res = np.fft.irfft(FA * FP, M2)
    r = conv_res[n - 1:2 * n - 1]
    grad = 2.0 * r
    # Normalize and take step
    gn = float(np.linalg.norm(grad))
    if gn < 1e-15:
        return a
    # Step size: use (score - lower_bound) / ||grad||^2
    sc = 2.0 * n * float(c.max()) / (S * S)
    step = step_scale * (sc - 1.0) / (gn * gn + 1e-30)
    step = min(step, 0.1)
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new

"""
Dense L-BFGS continuation + multi-lag subgradient + coordinate descent
for minimizing peak autocorrelation ratio of non-negative sequences.
Uses adaptive beta scheduling, line-search subgradient, and targeted
coordinate refinement to push below 1.503.
"""

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, c, 0
    score = 2.0 * n * float(np.max(c)) / (S * S)
    return score, c, int(np.argmax(c))


def lse_obj_grad(a, beta):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:M]
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    M2 = 1 << (3 * n - 3).bit_length()
    rev_a = a[::-1]
    FA = np.fft.rfft(rev_a, M2)
    FP = np.fft.rfft(p, M2)
    conv_res = np.fft.irfft(FA * FP, M2)
    r = conv_res[n - 1:2 * n - 1]
    grad_L = 2.0 * r
    S2 = S * S
    grad = (2.0 * n / (S2 * S)) * (S * grad_L - 2.0 * L)
    return score, grad

        options={'ftol': 1e-18, 'gtol': 1e-14,

    )

def multi_lag_grad(a, n_lags=10):
    n = len(a)
    c = fast_conv(a)
    M = len(c)
    S = float(a.sum())
    S2 = S * S
    sc = 2.0 * n * float(c.max()) / S2
    top_idx = np.argsort(c)[::-1][:n_lags]
    weights = c[top_idx].copy()
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return sc, np.zeros(n), c
    p_active = np.zeros(M)
    p_active[top_idx] = weights / w_sum
    M2 = 1 << (3 * n - 3).bit_length()
    rev_a = a[::-1]
    FA = np.fft.rfft(rev_a, M2)

    grad_c = 2.0 * r
    grad_score = (2.0 * n / (S2 * S)) * (S * grad_c - 2.0 * float(c.max()))

    for it in range(n_iters):
        A_full = np.fft.fft(a, M_fft)
        mag = np.abs(A_full)
        phase = np.angle(A_full)
        half = M_fft // 2
        non_dc = mag[1:half]
        if len(non_dc) == 0:
            break
        target = np.median(non_dc)
        new_mag = mag.copy()
        for k in range(1, half):
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[M_fft - k] = new_mag[k]
        if M_fft % 2 == 0:
            new_mag[half] = (1.0 - lam) * mag[half] + lam * target
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def resample_seq(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    Uses L-BFGS continuation + multi-lag subgradient + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0
    deadline = t_start + effective_budget

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is not None:
        a = parent.copy()
        n = len(a)
    else:
        n = 4096
        a = np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01
        a = np.clip(a, 0.0, 1000.0)

    sc, c, k_star = compute_score(a)
    report(sc, a, "init")
    print(f"Start: n={n}, score={sc:.8f}")

    # Phase 1: L-BFGS continuation (55% time)
    p1_deadline = t_start + 0.55 * effective_budget
    print(f"\n--- Phase 1: L-BFGS (t={p1_deadline - t_start:.0f}s) ---")

    beta_schedule = [20, 50, 100, 200, 350, 600, 1000, 2000, 4000, 8000]
    prev_sc = sc
    stuck_count = 0
    for beta in beta_schedule:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 15:
            break
        maxiter = min(5000, max(300, int(remaining * 0.65 / max(0.025, 0.004 * n / 1024))))
        maxfun = min(120000, maxiter * 25)
        t0 = time.monotonic()
        sc_new, a_new = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxfun)
        a = a_new
        report(sc_new, a, f"LBFGS b={beta}")
        elapsed = time.monotonic() - t0
        print(f"    b={beta}: {sc_new:.8f} ({elapsed:.1f}s)")
        if sc_new > prev_sc - 1e-10:
            stuck_count += 1
        else:
            stuck_count = 0
        prev_sc = sc_new
        if stuck_count >= 3:
            # Perturb and restart
            a = np.clip(a + 0.01 * rng.standard_normal(n), 0.0, 1000.0)
            stuck_count = 0
            sc, _, _ = compute_score(a)
            print(f"    Perturbed, score={sc:.8f}")

    # Phase 2: Subgradient + coordinate descent (25% time)
    p2_deadline = t_start + 0.80 * effective_budget
    print(f"\n--- Phase 2: Subgrad+CD (t={p2_deadline - t_start:.0f}s) ---")

    it_count = 0
    while time.monotonic() < p2_deadline:
        it_count += 1
        n_lags = min(25, 5 + it_count // 5)
        a_new, sc_new = subgrad_step(a, n_lags=n_lags, max_step=0.03)
        if sc_new < compute_score(a)[0] - 1e-12:
            a = a_new
            report(sc_new, a, f"sg {it_count}")
        if it_count % 15 == 0:
            sc_cd, a_cd = coord_descent(a, n_trials=25, rng=rng)
            sc_cur, _, _ = compute_score(a)
            if sc_cd < sc_cur:
                a = a_cd
                report(sc_cd, a, f"cd {it_count}")
        if it_count % 50 == 0:
            sc_cur, _, _ = compute_score(a)
            print(f"    iter {it_count}: score={sc_cur:.8f}")

    # Phase 3: Try fresh init at different params (10% time)
    p3_deadline = t_start + 0.90 * effective_budget
    print(f"\n--- Phase 3: Alt inits (t={p3_deadline - t_start:.0f}s) ---")

    # Try spectral flatten from parent
    if time.monotonic() < p3_deadline:
        a_sf = spectral_flatten(a, n_iters=100, lam=0.4)
        sc_sf, _, _ = compute_score(a_sf)
        report(sc_sf, a_sf, "spectral")
        if sc_sf < compute_score(a)[0]:
            a = a_sf
            for beta in [200, 1000, 5000]:
                if time.monotonic() > p3_deadline:
                    break
                remaining = p3_deadline - time.monotonic()
                if remaining < 10:
                    break
                maxiter = min(3000, max(200, int(remaining * 0.7 / 0.03)))
                sc_new, a_new = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxiter * 20)
                a = a_new
                report(sc_new, a, f"sf+lbfgs b={beta}")

    # Phase 4: Final aggressive polish (5% time)
    print(f"\n--- Phase 4: Final Polish ---")
    if best_a is not None:
        a = best_a.copy()

    for beta in [3000, 10000, 30000]:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(3000, max(200, int(remaining * 0.8 / 0.04)))
        sc_new, a_new = run_lbfgs(a, beta, maxiter=maxiter, maxfun=maxiter * 20)
        a = a_new
        report(sc_new, a, f"final b={beta}")

    # Final subgradient burst
    for it in range(200):
        if time.monotonic() > deadline - 3:
            break
        n_lags = min(30, 10 + it // 3)
        a_new, sc_new = subgrad_step(a, n_lags=n_lags, max_step=0.02)
        sc_cur, _, _ = compute_score(a)
        if sc_new < sc_cur:
            a = a_new
            report(sc_new, a, f"fin_sg {it}")

    # Final coordinate descent
    sc_cd, a_cd = coord_descent(a, n_trials=50, rng=rng)
    sc_cur, _, _ = compute_score(a)
    if sc_cd < sc_cur:
        a = a_cd
    report(sc_cd if sc_cd < sc_cur else sc_cur, a_cd if sc_cd < sc_cur else a, "final")

    # Output
    out_a = best_a if best_a is not None else a
    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
