"""
Multi-strategy optimizer for step function peak minimization.
Key innovations over previous approach:
1. Nesterov-accelerated subgradient with adaptive step size
2. Top-k peak LSE smoothing (smoother than full LSE near the optimum)

5. Multi-length exploration with parent interpolation

import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad_topk(a, beta, top_k=None):
    """LSE-smoothed score with optional top-k restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Restrict to top_k peaks for smoother optimization
        threshold_idx = np.argsort(c)[::-1][top_k - 1]
        c_clipped = np.minimum(c, threshold_idx)
    
    c_max = float(c_clipped.max())
    scaled = beta * (c_clipped - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))  # numerical stability
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_score_grad_plain(a, beta):
    """Standard LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective with tight criteria."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    if top_k is not None:
        func = lambda x: lse_score_grad_topk(x, beta, top_k)
    else:
        func = lambda x: lse_score_grad_plain(x, beta)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

            if time.monotonic() > deadline - 20:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=50000, maxfun=200000)
            report(sc, a_opt, f"final beta={beta}")
    
    # One more Nesterov pass
    if time.monotonic() < deadline - 20:
        a_ns2 = nesterov_subgrad(best_a, n_iters=3000, f_lb=1.0)
        sc = compute_score(a_ns2)
        report(sc, a_ns2, "final nesterov")
    
    # Try top-k final
    if time.monotonic() < deadline - 15:
        for top_k in [2, 3]:
            sc_k, a_k, _ = run_lbfgs(best_a, 100000, maxiter=30000,
                                      maxfun=100000, top_k=top_k)
            report(sc_k, a_k, f"final topk={top_k}")
    
    # ==================================================================
    # Try a different length if time permits
    # ==================================================================
    if time.monotonic() < deadline - 60 and parent is not None:
        print(f"\n--- Phase 5: Length Exploration ---")
        for n_try in [len(parent) // 2, int(len(parent) * 1.5)]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 256 or n_try > 50000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt, _ = run_lbfgs(a_interp, 500, maxiter=20000, maxfun=80000)
            report(sc, a_opt, f"len={n_try}")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
