def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad_a(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

"""
Multi-strategy optimizer combining LP-based refinement, LSE-smoothed Adam,
structural exploration, and coordinate descent for minimizing
2n*max(self-conv)/(sum)^2 over non-negative sequences.

Key innovations over previous approach:
1. LP-based direction finding (solves exact LP for optimal step direction)
2. Multi-M LP exploration (varies constraint tightness)
3. Peak-reduction heuristic (structurally reduces peak contributors)
4. Better multi-phase time management with adaptive allocation
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def build_conv_constraint_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    M_total = 2 * n - 1
    # Column j has nonzeros at rows j..j+n-1 with values a[0]..a[n-1]
    # Total nnz = sum over j of min(n, M_total - j)
    nnz_total = sum(min(n, M_total - j) for j in range(n))
    rows = np.empty(nnz_total, dtype=np.int32)
    cols = np.empty(nnz_total, dtype=np.int32)
    vals = np.empty(nnz_total, dtype=np.float64)
    idx = 0
    for j in range(n):
        length = min(n, M_total - j)
        rows[idx:idx + length] = np.arange(j, j + length)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[:length]
        idx += length
    A_mat = csr_matrix((vals, (rows, cols)), shape=(M_total, n))
    return A_mat


def lp_solve_direction(a, M_mult=1.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(a)
    c = fast_conv(a)
    M = M_mult * float(c.max())
    M_total = 2 * n - 1

    # Estimate matrix size
    nnz_estimate = n * min(n, M_total)
    if nnz_estimate > 25_000_000:
        return None

    A_mat = build_conv_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(M_total, M)

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0)
            S_g = g.sum()
            if S_g > 1e-12:
                return g / S_g
    except Exception:
        pass
    return None

def resample_sequence(a, new_n):

def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer: LP refinement + Adam + L-BFGS + peak reduction + coord descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) * 0.5
        parent_score = compute_score(parent)
        print(f"No parent, uniform n=4096")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ============================================================
    # PHASE 1: LP Refinement on parent (if size allows)
    # ============================================================
    p1_budget = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 1: LP Refinement (budget={p1_budget:.0f}s)")
    print(f"{'='*50}")

    p1_deadline = t_start + p1_budget
    if len(parent) <= 3500:
        elapsed_ok = time.monotonic() < p1_deadline
        if elapsed_ok:
            print(f"  Parent n={len(parent)}, doing LP refinement...")
            sc, a_lp = lp_refine_iteration(parent, n_iter=12, t=0.25,
                                           M_mults=(1.0, 1.05, 1.1), verbose=True)
            try_update(sc, a_lp, "LP refine parent")
    else:
        print(f"  Parent n={len(parent)} too large for LP, trying sub-sample")
        # Subsample parent to size 3000, do LP, then upscale
        n_sub = min(3000, len(parent))
        a_sub = resample_sequence(parent, n_sub)
        a_sub = a_sub / a_sub.sum()
        sc_sub, a_sub_opt = lp_refine_iteration(a_sub, n_iter=10, t=0.3,
                                                M_mults=(1.0, 1.1), verbose=True)
        # Upscale back
        a_up = resample_sequence(a_sub_opt, len(parent))
        a_up = a_up / a_up.sum()
        # Quick Adam to recover
        sc_up, a_up_opt = adam_direct(a_up, lr=0.003, n_steps=30000, beta_lse=200,
                                      time_limit_s=min(30, p1_deadline - time.monotonic()),
                                      t_start=t_start)
        try_update(sc_up, a_up_opt, "LP sub-sample + upscale")

    # ============================================================
    # PHASE 2: Adam with continuation (30% time)
    # ============================================================
    p2_budget = effective_budget * 0.30
    print(f"\n{'='*50}")
    print(f"Phase 2: Adam Continuation (budget={p2_budget:.0f}s)")
    print(f"{'='*50}")

    p2_deadline = t_start + p2_budget
    a_work = best_a / best_a.sum()

    betas_lrs = [(30, 0.01), (80, 0.005), (200, 0.003), (500, 0.002)]
    for beta, lr in betas_lrs:
        remaining = p2_deadline - time.monotonic()
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.35, 60.0)
        sc, a_opt = adam_direct(a_work, lr=lr, n_steps=100000, beta_lse=beta,
                                time_limit_s=time_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam beta={beta}")
        a_work = a_opt / a_opt.sum()

    # ============================================================
    # PHASE 3: L-BFGS Polish (15% time)
    # ============================================================
    p3_budget = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS Polish (budget={p3_budget:.0f}s)")
    print(f"{'='*50}")

    p3_deadline = t_start + p3_budget
    for beta in [200, 500, 1000, 2000]:
        if time.monotonic() > p3_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_a(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # PHASE 4: Peak Reduction + LP (20% time)
    # ============================================================
    p4_budget = effective_budget * 0.20
    print(f"\n{'='*50}")
    print(f"Phase 4: Peak Reduction + LP (budget={p4_budget:.0f}s)")
    print(f"{'='*50}")

    p4_deadline = t_start + p4_budget

    # Peak reduction
    if time.monotonic() < p4_deadline - 20:
        t0 = time.monotonic()
        sc, a_pr = peak_reduction(best_a, n_rounds=50, verbose=True)
        try_update(sc, a_pr, "Peak reduction")
        print(f"    Peak reduction: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # LP on best (if size allows)
    if time.monotonic() < p4_deadline - 10 and len(best_a) <= 3500:
        t0 = time.monotonic()
        sc, a_lp2 = lp_refine_iteration(best_a, n_iter=10, t=0.2,
                                        M_mults=(1.0, 1.03, 1.08), verbose=True)
        try_update(sc, a_lp2, "LP refine best")
        print(f"    LP refine best: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # PHASE 5: Exploratory search with new initializations (10% time)
    # ============================================================
    p5_budget = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 5: Exploratory (budget={p5_budget:.0f}s)")
    print(f"{'='*50}")

    p5_deadline = t_start + p5_budget
    n_best = len(best_a)

    strategies = [
        ("reverse", lambda: best_a[::-1].copy()),
        ("gauss_pert", lambda: np.maximum(best_a * (1 + 0.05 * rng.standard_normal(n_best)), 0)),
        ("block_mod", lambda: _block_modify(best_a, rng)),
        ("peak_avoid", lambda: _peak_avoid_init(best_a, rng)),
    ]

    for name, gen_fn in strategies:
        if time.monotonic() > p5_deadline - 10:
            break
        try:
            a_pert = gen_fn()
            a_pert = np.clip(a_pert, 0, 1000)
            S = a_pert.sum()
            if S < 1e-6:
                continue
            a_pert = a_pert / S
            time_left = p5_deadline - time.monotonic()
            if time_left < 5:
                break
            sc, a_opt = adam_direct(a_pert, lr=0.003, n_steps=30000, beta_lse=300,
                                    time_limit_s=min(20, time_left * 0.6), t_start=t_start)
            if time.monotonic() < p5_deadline - 3:
                sc2, a_opt2, _ = run_lbfgs_a(a_opt, 500, maxiter=2000, maxfun=30000)
                if sc2 < sc:
                    sc, a_opt = sc2, a_opt2
            try_update(sc, a_opt, f"explore:{name}")
        except Exception as e:
            print(f"    {name} failed: {e}")

    # ============================================================
    # PHASE 6: Try different lengths (5% time)
    # ============================================================
    print(f"\n{'='*50}")
    print("Phase 6: Different Lengths")
    print(f"{'='*50}")

    time_left_total = t_start + effective_budget - time.monotonic()
    target_fracs = [0.7, 1.3, 1.5]
    for frac in target_fracs:
        if time_left_total < 20:
            break
        tl = max(512, min(20000, int(n_best * frac)))
        if tl == n_best:
            continue
        a_res = resample_sequence(best_a, tl)
        a_res = a_res / a_res.sum()
        sc, a_opt = adam_direct(a_res, lr=0.003, n_steps=30000, beta_lse=200,
                                time_limit_s=min(15, time_left_total * 0.3), t_start=t_start)
        if time.monotonic() < t_start + effective_budget - 5:
            sc2, a_opt2, _ = run_lbfgs_a(a_opt, 500, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"n={tl}")
        time_left_total = t_start + effective_budget - time.monotonic()

    # ============================================================
    # PHASE 7: Final intensive polish
    # ============================================================
    print(f"\n{'='*50}")
    print("Phase 7: Final Polish")
    print(f"{'='*50}")

    final_deadline = t_start + effective_budget - 5.0

    # Coordinate descent
    if time.monotonic() < final_deadline - 5:
        sc, a_cd = selective_coord_descent(best_a, top_k=50, n_sweeps=10,
                                           time_limit_s=min(30, final_deadline - time.monotonic() - 5),
                                           t_start=t_start)
        try_update(sc, a_cd, "final coord")

    # High-beta L-BFGS
    if time.monotonic() < final_deadline - 3:
        sc, a_opt, _ = run_lbfgs_a(best_a, 3000, maxiter=5000, maxfun=100000)
        try_update(sc, a_opt, "final LBFGS hi-beta")

    # One more LP if possible
    if time.monotonic() < final_deadline - 2 and len(best_a) <= 3500:
        sc, a_lp3 = lp_refine_iteration(best_a, n_iter=5, t=0.15,
                                        M_mults=(1.0, 1.02), verbose=False)
        try_update(sc, a_lp3, "final LP")

    # Final coord descent
    if time.monotonic() < final_deadline - 1:
        sc, a_cd2 = selective_coord_descent(best_a, top_k=60, n_sweeps=3,
                                            time_limit_s=min(10, final_deadline - time.monotonic() - 1),
                                            t_start=t_start)
        try_update(sc, a_cd2, "very final coord")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out


def _block_modify(a, rng):
    """Modify a random block of the sequence."""
    n = len(a)
    a_new = a.copy()
    center = int(rng.integers(n // 4, 3 * n // 4))
    width = max(10, n // 40)
    a_new[center - width:center + width] *= rng.uniform(0.3, 1.5)
    return a_new


def _peak_avoid_init(a, rng):
    """Create initialization that avoids the current peak structure."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    a_new = a.copy()
    # Reduce peak contributors significantly
    for j in range(js, je + 1):
        a_new[j] *= 0.4
    # Boost non-contributors
    mask = np.ones(n, dtype=bool)
    mask[js:je + 1] = False
    if mask.sum() > 0:
        boost = (a[js:je + 1].sum() - a_new[js:je + 1].sum()) / mask.sum()
        a_new[mask] += boost * (0.5 + 0.5 * rng.random(mask.sum()))
    return np.maximum(a_new, 0)
