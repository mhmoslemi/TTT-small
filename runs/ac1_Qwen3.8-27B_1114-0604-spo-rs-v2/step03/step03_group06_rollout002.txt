The issue is that in step 4, I'm throwing away information (the values at positions `n` to `2n-2`). This is necessary because our sequence only has `n` free parameters. But it means the algorithm might not converge nicely.

A potential improvement: instead of hard truncation, I could use a soft transition:
`a_new[j] = max(0, a_full[j] * smoothstep(n - j))` for `j` near `n`.

But this complicates things. Let me stick with hard truncation for now.

Another thought: **what if I use a different constraint projection?** Instead of projecting to `[0, ∞)^n`, I could project to the simplex `{x >= 0, sum(x) = 1}`. This would maintain the normalization automatically. But for L-BFGS-B, it's easier to work with the box constraint `[0, 1000]^n`.

OK, I think I have enough ideas. Let me write the code.

5. **Combined optimization loop** (modified)

"""
Spectral IFTA + L-BFGS-B hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: Iterative Fourier Transform Algorithm (IFTA) with spectral magnitude
compression, combined with gradient-based L-BFGS-B refinement. The IFTA step directly
targets spectral flatness (equivalently, autocorrelation flatness) and provides a
globally-informed search direction that complements local gradient methods.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def ifta_compress(a, alpha):
    """One IFTA iteration: compress spectrum, project back to time domain."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    
    A = np.fft.rfft(a, N_fft)
    mag = np.abs(A)
    phase = np.angle(A)
    
    mean_mag = np.mean(mag)
    if mean_mag < 1e-15:
        return a.copy()
    
    ratio = mag / mean_mag
    # Avoid log(0) issues
    ratio = np.maximum(ratio, 1e-10)
    mag_new = mean_mag * (ratio ** alpha)
    
    A_new = mag_new * np.exp(1j * phase)
    a_full = np.fft.irfft(A_new, N_fft)
    a_new = np.maximum(a_full[:n], 0.0)
    
    return a_new

def ifta_continuation(a, alphas, iters_per_alpha=50, normalize_sum=None):
    """
    Run IFTA with alpha continuation.
    alphas: list of alpha values (decreasing from 1 to small)
    iters_per_alpha: number of IFTA iterations at each alpha
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    if normalize_sum is None:
        normalize_sum = float(a.sum())
    
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            a = ifta_compress(a, alpha)
            if a.sum() < 1e-12:
                break
            # Normalize to maintain scale
            a = a * (normalize_sum / a.sum())
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_sc, best_a

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def build_inits(n, rng, parent):
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
    
    # Structured
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Multi-frequency
    for freq in [3, 5, 7, 11]:
        inits.append(0.5 + 0.2 * np.cos(2.0 * np.pi * freq * x) + 0.1 * np.cos(4.0 * np.pi * x))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # Block
    for nb in (8, 16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits

def exact_subgrad_polish(a, total_iters=500):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        a -= 0.001 * grad / (gn + 1e-15)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

def freq_domain_perturb(a, rng, n_swaps=5):
    """Perturb in frequency domain: swap or combine frequency bins."""
    n = len(a)
    N_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, N_fft)
    for _ in range(n_swaps):
        i, j = rng.integers(0, len(A), size=2)
        if i == j:
            continue
        # Swap magnitudes between two bins (keep phases)
        mi, mj = np.abs(A[i]), np.abs(A[j])
        pi, pj = np.angle(A[i]), np.angle(A[j])
        A[i] = mj * np.exp(1j * pi)
        A[j] = mi * np.exp(1j * pj)
    a_full = np.fft.irfft(A, N_fft)
    a_new = np.maximum(a_full[:n], 0.0)
    if a_new.sum() > 1e-12:
        a_new *= a.sum() / a_new.sum()
    return a_new
