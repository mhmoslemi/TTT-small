The idea: if c[k*] is the peak, and the contributing pairs are (i, k*-i) for i in S_k*, then:

def peak_anti_correlate(a, k_star, strength=0.01):
    """Modify a to reduce c[k_star] by making a anti-correlated with its k*-reflection."""
    n = len(a)
    a = a.copy()
    # rev_a[j] = a[k_star - j] for valid j
    rev_a = np.zeros(n)
    for j in range(n):
        k_minus_j = k_star - j
        if 0 <= k_minus_j < n:
            rev_a[j] = a[k_minus_j]
    
    # Project rev_a onto the subspace sum=0
    rev_centered = rev_a - rev_a.mean()
    
    # delta = -strength * rev_centered / ||rev_centered||
    norm = np.linalg.norm(rev_centered)
    if norm < 1e-15:
        return a
    delta = -strength * rev_centered / norm
    
    a_new = a + delta
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-12:
        return a
    a_new = a_new / S
    return a_new
