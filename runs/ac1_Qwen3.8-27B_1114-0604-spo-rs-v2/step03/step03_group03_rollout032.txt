The strategy mentions solving a linear program to find the optimal direction. This is the "sieve method" from analytic number theory. The idea is:

Given the current sequence f with peak M = max(f*f), find a non-negative direction g such that:

"""
LP-guided coordinate descent + LSE optimization for minimum peak autocorrelation.
Innovation: Uses cutting-plane LP to find guaranteed first-order improvement directions,

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    return lse_score_grad(a, beta)


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])

        k_hi = min(M_conv - 1, i + n - 1)

            outside_max = 0.0

        inside_len = k_hi - k_lo + 1

def solve_lp_direction(f, M, max_constraints=400, time_limit=8):
    """
    Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0.
    Uses cutting plane method. Returns (g, sum_g).
    """
    n = len(f)
    g_var = cp.Variable(n, nonneg=True)

    conv_f = fast_conv(f)
    top_k = np.argsort(conv_f)[::-1][:80]

    constraints = []
    active_ks = set()
    t_start = time.monotonic()

    def add_constraint(k):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        coeffs = f[k - i_hi:k - i_lo + 1][::-1]
        return coeffs @ g_var[i_lo:i_hi + 1] <= M

    for k in top_k:
        constraints.append(add_constraint(k))
        active_ks.add(int(k))

    best_g = np.zeros(n)
    best_sum = 0.0

    for iteration in range(40):
        if time.monotonic() - t_start > time_limit:
            break

        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        solved = False
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, abs_tol=1e-7, rel_tol=1e-7)
            solved = True
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-7)
                solved = True
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.CLARABEL)
                solved = True
            except Exception:
                pass

        if not solved or g_var.value is None:
            break

        g_val = np.asarray(g_var.value).flatten()
        g_val = np.clip(g_val, 0, None)
        if g_val.sum() > best_sum:
            best_g = g_val.copy()
            best_sum = float(g_val.sum())

        conv_fg = cross_conv(f, g_val)
        violations = conv_fg - M
        max_viol = float(np.max(violations))

        if max_viol < 1e-5 * max(M, 1e-10):
            break

        if len(constraints) >= max_constraints:
            break

        worst_k = int(np.argmax(violations))
        if worst_k not in active_ks:
            constraints.append(add_constraint(worst_k))
            active_ks.add(worst_k)

    return best_g, best_sum


def lp_guided_step(a, M, threshold=0.001):
    """
    Try to take an LP-guided step. Returns (new_a, improved, score).
    """
    n = len(a)
    S = float(a.sum())

    g, sum_g = solve_lp_direction(a, M)

    if sum_g <= S * (1.0 + threshold):
        return a, False, np.inf

    # Find optimal step size
    def step_obj(t):
        a_new = a + t * g
        c = fast_conv(a_new)
        S_new = S + t * sum_g
        if S_new < 1e-12:
            return 1e10
        return 2.0 * n * float(np.max(c)) / (S_new * S_new)

    t_upper = min(0.5, 500.0 / max(1e-10, float(np.max(g))))
    res = minimize_scalar(step_obj, bounds=(1e-8, t_upper), method='bounded',
                          options={'xatol': 1e-8})
    t_opt = float(res.x)
    a_new = np.clip(a + t_opt * g, 0.0, 1000.0)
    sc = compute_score(a_new)
    return a_new, sc < np.inf, sc

    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
        )
    except Exception:
        return compute_score(a0), a0, None
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def block_lbfgs_step(a, block_size=64, beta=500, maxiter=200):
    """Optimize one block of variables using L-BFGS on LSE objective."""
    n = len(a)
    # Random block
    start = int(np.random.randint(0, max(1, n - block_size)))
    end = min(start + block_size, n)
    b = end - start

    def block_obj(x):
        a_temp = a.copy()
        a_temp[start:end] = x
        return lse_score_grad(a_temp, beta)[0]

    def block_grad(x):
        a_temp = a.copy()
        a_temp[start:end] = x
        _, grad_full = lse_score_grad(a_temp, beta)
        return grad_full[start:end]

    x0 = a[start:end].copy()
    bounds = [(0.0, 1000.0)] * b
    try:
        res = minimize(block_obj, x0, jac=block_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'maxfun': maxiter * 10,
                                               'ftol': 1e-12, 'gtol': 1e-10})
        a_new = a.copy()
        a_new[start:end] = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_new = a.copy()
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP-guided directions + coordinate descent + LSE refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    if parent is not None and parent.sum() > 0.01:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent available, using random init.")
        best_a = 0.5 + 0.3 * rng.random(2048)
        best_score = compute_score(best_a)
        print(f"Random init score: {best_score:.8f}")

    print(f"Working length: {n_work}")

    # Phase 1: Initial coordinate descent (15% time)
    # ============================================================
    phase1_deadline = t_start + 0.15 * total_budget

    a_cur = best_a.copy()

    # ============================================================
    phase2_deadline = t_start + 0.60 * total_budget

        lp_time = time.monotonic() - t0

        if improved and sc_lp < score_cur:
            a_cur = a_new
            lp_improved += 1

            if lp_improved <= 10 or lp_improved % 20 == 0:
                print(f"  LP {lp_count} (improved {lp_improved}): {sc_lp:.8f} ({lp_time:.1f}s)")
        else:
            stall_count += 1
            if stall_count % 20 == 0:
                print(f"  LP {lp_count}: no improvement (stall={stall_count}, best={score_cur:.8f})")
