def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S

def subgrad_score_a(a):

    S3 = S2 * S

def multiple_subgrad_score_a(a, top_k=5):
    """Score and averaged subgradient over top-k convolution lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    # Use top-k lags for a "smoothed" subgradient
    sorted_idx = np.argsort(c)[::-1][:top_k]
    weights = c[sorted_idx] - c.max() + 1.0  # Ensure positive
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for k in sorted_idx:

        je = min(int(k), n - 1)
        if je >= js:

    S3 = S2 * S

    return score, grad


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with bounds and LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_score_grad_x(x, beta):
    """LSE-smoothed score and gradient w.r.t. x (softmax parametrization)."""
    a = softmax_fn(x)
    score, grad_a = lse_score_grad_a(a, beta)
    d = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - d)
    return score, grad_x


def run_lbfgs_x(x0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in x-space (softmax parametrization)."""
    res = minimize(
        lse_score_grad_x, x0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = softmax_fn(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.x


def adam_direct(a_init, lr=0.005, n_steps=50000, beta_lse=100,
                time_limit_s=None, t_start=None, normalize=True,
                verbose_interval=20000):
    """
    Adam optimizer in a-space with LSE-smoothed gradient.
    Optionally normalizes to sum=1 after each step.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        
        score, grad = lse_score_grad_a(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        if t % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    Adam t={t}: score={sc:.8f}")
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def selective_coord_descent(a, top_k=30, n_sweeps=10, time_limit_s=None, t_start=None):
    """
    Coordinate descent optimizing top-K coordinates per sweep.
    Uses exact score evaluation for candidate selection.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        score, grad = subgrad_score_a(a)
        if score < best_score:
            best_score = score
        
        # Select top-k coordinates by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            
            # Generate candidates based on gradient direction
            if grad[j] > 1e-10:
                # Positive gradient: reducing a[j] helps
                ratios = np.array([0.2, 0.4, 0.6, 0.75, 0.85, 0.95, 1.0, 1.05])
                candidates = a_old * ratios
            elif grad[j] < -1e-10:
                # Negative gradient: increasing a[j] helps
                ratios = np.array([0.5, 0.8, 1.0, 1.1, 1.3, 1.5, 2.0, 3.0])
                candidates = a_old * ratios
                candidates = np.concatenate([[0.0], candidates])
            else:
                candidates = np.array([a_old])
            
            candidates = np.clip(candidates, 0.0, 1000.0)
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    
    return best_score, a


def nesterov_subgrad(a_init, lr_init=0.01, n_steps=20000, time_limit_s=None, t_start=None):
    """Nesterov-accelerated subgradient descent (exact, not smoothed)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        momentum = 2.0 / (t + 2.0)
        lr_t = lr_init * momentum
        
        score, grad = subgrad_score_a(y)
        
        a_new = y - lr_t * grad
        a_new = np.clip(a_new, 0.0, 1e10)
        
        y = a_new + (momentum - 1.0) * (a_new - a)
        y = np.clip(y, 0.0, 1e10)
        a = a_new
        
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
