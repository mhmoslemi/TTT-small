import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """FFT-based linear self-convolution. O(n log n)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """FFT-based linear cross-convolution of a and b."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. O(n log n)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

def topk_score_grad(a, k):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

        return 1e10, np.zeros(n)

    gap = max(threshold - next_val, 1e-10)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, use_topk=False, topk_k=5):
    """Run L-BFGS-B minimization."""

    else:

    res = minimize(
        fun, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,

    )

def subgradient_polish(a, n_iters=2000, step_scale=1.0):

        sc = 2.0 * n * M / (S * S)

            best_a = a.copy()
        # Subgradient of max(c) w.r.t. a

        sg = np.zeros(n)

        step = min(step, 0.5)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def coordinate_polish(a, n_rounds=50, rng=None, max_points=200):
    """Simple coordinate descent: try each element independently."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    S = float(a.sum())
    for rnd in range(n_rounds):
        improved = False
        idxs = np.arange(n)
        if rng is not None:
            rng.shuffle(idxs)
        for i in idxs[:max_points]:
            if S < 1e-10:
                break
            old_val = a[i]
            # Try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 1.5,
                          old_val + 0.01, old_val - 0.01,
                          max(0, old_val - 0.1), old_val + 0.1]
            best_v = old_val
            best_local_sc = compute_score(a)
            for v in candidates:
                v = max(0.0, min(1000.0, v))

                    best_v = v
            a[i] = best_v

                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    c = fast_conv(a)

    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.sin(4.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.3 + 0.4 * x)
    inits.append(0.3 + 0.4 * (1 - x))
    # Multi-frequency
    inits.append(0.5 + 0.15 * np.cos(2*np.pi*x) + 0.15 * np.cos(4*np.pi*x) + 0.1 * np.cos(6*np.pi*x))
    # Block structure
    nb = 16
    per = max(1, n // nb)
    vals = 0.2 + 0.8 * rng.random(nb)
    inits.append(np.repeat(vals, per)[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing the evaluation function.
    Uses continuation L-BFGS-B with LSE smoothing, subgradient polish,
    and coordinate descent for final refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 10.0

    # Load parent solution
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine length
    if parent is not None:
        n_primary = parent.size
        print(f"Using parent length: {n_primary}")
    else:
        n_primary = 8192
        print(f"No parent, using length: {n_primary}")

    # Try a few nearby lengths for FFT efficiency
    # Round to next power of 2 minus 1 for fast convolution
    n_fft = 1 << (n_primary - 1).bit_length()
    print(f"n_primary={n_primary}, n_fft_pad={2*n_primary-1}, n_fft_use={1 << ((2*n_primary-2).bit_length())}")

    lengths_to_try = [n_primary]
    if parent is not None:
        for dl in [-256, 256, -512, 512, -1024, 1024]:
            cand = n_primary + dl
            if cand >= 1024 and cand not in lengths_to_try:
                lengths_to_try.append(cand)
    lengths_to_try = lengths_to_try[:3]  # Limit to 3 lengths
    print(f"Lengths: {lengths_to_try}")

    # =========================================================================
    # Phase 1: Primary optimization on parent length (60% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.60 * (total_budget - 10.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: Primary L-BFGS-B optimization")
    print(f"{'='*60}")

    n = lengths_to_try[0]  # Primary length
    inits = diverse_init(n, rng, parent)

    # For the parent, do a long continuation
    a_cur = inits[0].copy()
    sc0 = compute_score(a_cur)
    print(f"  Initial score (parent): {sc0:.10f}")
    report(sc0, a_cur, "parent")

    beta_schedule = [10, 30, 100, 300, 1000, 3000, 5000, 10000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 10:
            break
        maxiter = min(8000, int(remaining / 0.05))
        maxfun = min(100000, maxiter * 3)
        print(f"  Beta={beta}, maxiter={maxiter}, remaining={remaining:.0f}s")
        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        a_cur = a_opt
        report(sc, a_opt, f"beta={beta}")

    # Try other initializations for the primary length
    for i, a0 in enumerate(inits[1:], start=1):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        # Quick optimization with moderate beta
        sc, a_opt, _ = run_lbfgs(a0, 200, maxiter=min(3000, int(remaining/0.03)),
                                  maxfun=min(20000, int(remaining/0.01)))
        report(sc, a_opt, f"init{i}")
        # If it's close to best, do more optimization
        if compute_score(a_opt) < best_score + 0.01:
            for beta in [1000, 5000]:
                if time.monotonic() > phase1_deadline:
                    break
                remaining = phase1_deadline - time.monotonic()
                if remaining < 10:
                    break
                sc, a_opt2, _ = run_lbfgs(a_opt, beta,
                                           maxiter=min(5000, int(remaining/0.03)),
                                           maxfun=min(30000, int(remaining/0.01)))
                a_opt = a_opt2
                report(sc, a_opt, f"init{i}_beta{beta}")

    print(f"\nAfter Phase 1: best = {best_score:.10f}")

    # =========================================================================
    # Phase 2: Try other lengths (15% of time)
    # =========================================================================
    phase2_deadline = t_start + 0.75 * (total_budget - 10.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Other lengths")
    print(f"{'='*60}")

    for n_alt in lengths_to_try[1:]:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\n  Trying n={n_alt}, remaining={remaining:.0f}s")
        # Interpolate parent to new length
        if parent is not None and len(parent) != n_alt:
            idx = np.linspace(0, len(parent) - 1, n_alt).astype(int)
            a0_alt = parent[idx].copy()
        elif len(best_a) == n_alt:
            a0_alt = best_a.copy()
        else:
            a0_alt = np.ones(n_alt) * (best_a.sum() / len(best_a)) if len(best_a) > 0 else np.ones(n_alt) * 0.5

        sc = compute_score(a0_alt)
        report(sc, a0_alt, f"n={n_alt}_init")

        for beta in [100, 500, 2000]:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            sc, a_opt, _ = run_lbfgs(a0_alt, beta,
                                      maxiter=min(5000, int(remaining/0.03)),
                                      maxfun=min(20000, int(remaining/0.01)))
            a0_alt = a_opt
            report(sc, a_opt, f"n={n_alt}_beta{beta}")

    print(f"\nAfter Phase 2: best = {best_score:.10f}")

    # =========================================================================
    # Phase 3: Top-k and subgradient refinement (10% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 10.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Top-k and subgradient refinement")
    print(f"{'='*60}")

    a_refine = best_a.copy()
    # Top-k optimization
    for k in [3, 5, 10]:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 10:
            break
        sc, a_opt, _ = run_lbfgs(a_refine, 100, use_topk=True, topk_k=k,
                                  maxiter=min(3000, int(remaining/0.03)),
                                  maxfun=min(10000, int(remaining/0.01)))
        a_refine = a_opt
        report(sc, a_opt, f"topk_k={k}")

    # Subgradient polish
    if time.monotonic() < phase3_deadline - 5:
        remaining = phase3_deadline - time.monotonic()
        n_iters = min(5000, int(remaining / 0.005))
        sc, a_opt = subgradient_polish(a_refine, n_iters=n_iters)
        report(sc, a_opt, "subgrad")
        a_refine = a_opt

    print(f"\nAfter Phase 3: best = {best_score:.10f}")

    # =========================================================================
    # Phase 4: Final high-precision refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement")
    print(f"{'='*60}")

    a_final = best_a.copy()
    for beta in [5000, 10000, 20000, 50000]:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - 5 - time.monotonic()
        if remaining < 5:
            break
        sc, a_opt, _ = run_lbfgs(a_final, beta,
                                  maxiter=min(10000, int(remaining/0.01)),
                                  maxfun=min(50000, int(remaining/0.005)))
        a_final = a_opt
        report(sc, a_opt, f"final_beta={beta}")

    # One last subgradient pass
    if time.monotonic() < deadline - 3:
        remaining = deadline - 3 - time.monotonic()
        n_iters = min(5000, int(remaining / 0.005))
        if n_iters > 100:
            sc, a_opt = subgradient_polish(a_final, n_iters=n_iters)
            report(sc, a_opt, "final_subgrad")

    # Coordinate polish
    if time.monotonic() < deadline - 2:
        remaining = deadline - 2 - time.monotonic()
        n_rounds = min(20, int(remaining / 1.0))
        if n_rounds > 1:
            sc, a_opt = coordinate_polish(a_final, n_rounds=n_rounds, rng=rng,
                                           max_points=min(500, len(a_final)))
            report(sc, a_opt, "coord_polish")

    # =========================================================================
    # Final output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
