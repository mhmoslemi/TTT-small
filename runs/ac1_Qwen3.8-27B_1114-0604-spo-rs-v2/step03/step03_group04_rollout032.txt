"""
Multi-phase optimizer for minimum peak linear autocorrelation of non-negative sequences.
Uses LSE-smoothed objective with adaptive beta, L-BFGS-B optimization,
spectral flattening initialization, and subgradient polishing.
Key improvements: asymmetric initialization, targeted peak reduction, Adam fallback.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute the exact score: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fft_convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    # Shift for numerical stability
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    L = c_max + log_sum / beta
    p = exp_scaled / np.sum(exp_scaled)
    rev_a = a[::-1].copy()
    # Convolution of p (length 2n-1) and rev_a (length n) -> length 3n-2
    M2 = (2 * n - 1) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[:M2]
    v = conv_res[n - 1:2 * n - 1]
    score = 2.0 * n * L / (S * S)
    grad = (4.0 * n / (beta * S * S)) * v - (4.0 * n * L) / (S * S * S)
    return score, grad


def lse_objective(a, beta):
    """Objective-and-gradient for scipy minimize."""
    return lse_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def spectral_flatten(a0, N_fft, n_iters=200, lam=0.5, rng=None):
    """Iterative spectral flattening: alternate flat spectrum with non-negativity."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:N_fft//2] if N_fft % 2 == 0 else mag[1:(N_fft+1)//2]
        if len(non_dc) > 0:
            target = np.median(non_dc)
        else:
            target = mag[1] if N_fft > 1 else mag[0]
        new_mag = mag.copy()
        half = N_fft // 2
        for k in range(1, half + 1):
            if N_fft % 2 == 0 and k == half:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            else:
                new_mag[k] = (1.0 - lam) * mag[k] + lam * target
                new_mag[N_fft - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a

    freq = 1.0 + 5.0 * rng.random()
    inits.append(np.exp(-rate * x / n) * (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)) + 0.01)
    for s in inits:
        inits[-1] = np.clip(s, 0.001, 1000.0)
    return inits


def subgradient_polish(a, n_iters=500, lr_scale=1.0):
    """Subgradient method on the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fft_convolve(a, a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max(c) w.r.t. a
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        # Gradient of full score
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        # Polyak step size
        sc_lb = 1.4  # crude lower bound
        step = lr_scale * (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        # Add small exploration
        a -= step * grad
        a -= (0.005 / (1.0 + 0.05 * it)) * grad / (gn + 1e-30) * lr_scale
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def adam_optimize(a0, beta, n_steps=2000, lr=0.01):
    """Adam optimizer on the LSE-smoothed objective with projection."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    for t in range(1, n_steps + 1):
        sc, grad = lse_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        grad = np.maximum(grad, 0) - np.minimum(grad, 0)  # no-op, just for clarity
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a

"""
Multi-phase optimizer for minimum peak linear autocorrelation of non-negative sequences.
Combines spectral flattening initialization, LSE-smoothed L-BFGS-B optimization with
adaptive beta continuation, subgradient polishing, and targeted local search.
Key improvements: asymmetric initialization, higher beta values, better time allocation,
and a more aggressive subgradient method.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute the exact score: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = fft_convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """Compute LSE-smoothed score and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = np.sum(exp_scaled)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    p = exp_scaled / exp_sum
    rev_a = a[::-1].copy()
    M2 = (2 * n - 1) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[:M2]
    v = conv_res[n - 1:2 * n - 1]
    score = 2.0 * n * L / (S * S)
    grad = (4.0 * n / (beta * S * S)) * v - (4.0 * n * L) / (S * S * S)
    return score, grad


def lse_objective(a, beta):
    """Objective-and-gradient for scipy minimize."""
    return lse_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def spectral_flatten(a0, N_fft, n_iters=200, lam=0.5):
    """Iterative spectral flattening: alternate flat spectrum with non-negativity."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        half = N_fft // 2
        non_dc = mag[1:half]
        if len(non_dc) > 0:
            target = np.median(non_dc)
        else:
            target = mag[1] if N_fft > 1 else mag[0]
        new_mag = mag.copy()
        for k in range(1, half + 1):
            val = (1.0 - lam) * mag[k] + lam * target
            new_mag[k] = val
            if N_fft % 2 == 0 and k == half:
                pass
            else:
                new_mag[N_fft - k] = val
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def spectral_flatten_anneal(a0, N_fft, n_iters=300):
    """Spectral flattening with annealing: start strong, end weak."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = 0.8 * (1.0 - progress) + 0.15
        A = np.fft.fft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        half = N_fft // 2
        non_dc = mag[1:half]
        if len(non_dc) > 0:
            target = np.median(non_dc)
        else:
            target = mag[1] if N_fft > 1 else mag[0]
        new_mag = mag.copy()
        for k in range(1, half + 1):
            val = (1.0 - lam) * mag[k] + lam * target
            new_mag[k] = val
            if N_fft % 2 == 0 and k == half:
                pass
            else:
                new_mag[N_fft - k] = val
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    return a


def chirp_init_list(n, rng):
    """Generate chirp-based initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    f = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * f * x * x / (2.0 * n) + 0.5))
    ramp = x / n
    inits.append(0.2 + 0.3 * ramp + 0.3 * np.cos(2.0 * np.pi * 3.0 * x / n) ** 2)
    center = n * (0.3 + 0.4 * rng.random())
    width = n * (0.3 + 0.4 * rng.random())
    t = (x - center) / width
    inits.append(0.5 * (1.0 + np.cos(np.pi * np.clip(t, -1, 1))) ** 2 + 0.01)
    rate = 1.0 + 2.0 * rng.random()
    freq = 1.0 + 5.0 * rng.random()
    inits.append(np.exp(-rate * x / n) * (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)) + 0.01)
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    result = []
    for s in inits:
        result.append(np.clip(s, 0.001, 1000.0))
    return result


def subgradient_polish(a, n_iters=500, lr_scale=1.0):
    """Subgradient method on the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(n_iters):
        c = fft_convolve(a, a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        sc_lb = 1.4
        step = lr_scale * max(sc - sc_lb, 0.01) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a = a - step * grad
        a = a - (0.003 / (1.0 + 0.03 * it)) * lr_scale * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def adam_optimize(a0, beta, n_steps=2000, lr=0.01):
    """Adam optimizer on LSE-smoothed objective with projection."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, n_steps + 1):
        sc, grad = lse_grad(a, beta)
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def update_best(best_score, best_a, sc, a, label, t_start):
    """Update best if improved. Returns (new_best_score, new_best_a)."""
    if sc < best_score:
        elapsed = time.monotonic() - t_start
        print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return sc, a.copy()
    return best_score, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb by reducing peak-contributing elements and boosting others."""
    n = len(a)
    c = fft_convolve(a, a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    bottom_idx = np.argsort(contrib)[:top_count]
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: spectral init, LSE-LBFGS continuation, subgradient, local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 12.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(512) * 0.5

    # Determine sequence length
    if parent is not None:
        n = parent.size
    else:
        n = 4096
    print(f"Using n = {n}, parent: {'yes' if parent is not None else 'no'}")
    N_fft = 1 << (2 * n - 1).bit_length()

    # ================================================================
    # Phase 1: Generate diverse initializations
    # ================================================================
    p1_end = t_start + 0.10 * effective_budget
    print(f"\n--- Phase 1: Init (until t={p1_end - t_start:.0f}s) ---")

    inits = []
    if parent is not None:
        inits.append(parent.copy())

    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x / n))
    inits.append(0.3 + 0.4 * (x / n) ** 2)
    inits.append(0.5 * np.exp(-2.0 * x / n) + 0.1)
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * 2.0 * x / n + 1.0))
    inits.extend(chirp_init_list(n, rng))
    for _ in range(3):
        inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))

    # Spectral flattening
    sf_bases = [
        np.ones(n) * 0.5,
        np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
        np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01,
        0.3 + 0.4 * (x / n) ** 2,
    ]
    for bi, b in enumerate(sf_bases):
        if time.monotonic() > p1_end:
            break
        t0 = time.monotonic()
        sf_a = spectral_flatten(b, N_fft, n_iters=200, lam=0.5)
        inits.append(sf_a)
        sc_sf = compute_score(sf_a)
        print(f"  SF base {bi}: score={sc_sf:.6f} ({time.monotonic()-t0:.1f}s)")

    # One annealed version
    if time.monotonic() < p1_end:
        sf_a2 = spectral_flatten_anneal(np.ones(n) * 0.5, N_fft, n_iters=250)
        inits.append(sf_a2)
        print(f"  SF anneal: score={compute_score(sf_a2):.6f}")

    # Score and sort
    scored = []
    for a_init in inits:
        sc = compute_score(a_init)
        scored.append((sc, a_init))
    scored.sort(key=lambda tt: tt[0])
    print(f"  Best init: {scored[0][0]:.6f}")
    best_score, best_a = update_best(best_score, best_a, scored[0][0], scored[0][1], "init", t_start)

    top_inits = [a for _, a in scored[:min(6, len(scored))]]

    # ================================================================
    # Phase 2: LSE-LBFGS continuation (main optimization)
    # ================================================================
    p2_end = t_start + 0.55 * effective_budget
    print(f"\n--- Phase 2: LSE-LBFGS (until t={p2_end - t_start:.0f}s) ---")

    beta_schedule = [20, 50, 100, 300, 800, 2000]
    scale_factor = max(1.0, n / 2048.0)

    for ci, a_init in enumerate(top_inits):
        if time.monotonic() > p2_end:
            break
        t0 = time.monotonic()
        a_cur = a_init.copy()
        for beta in beta_schedule:
            if time.monotonic() > p2_end:
                break
            remaining = p2_end - time.monotonic()
            if remaining < 15:
                break
            est_per_iter = 0.004 * scale_factor
            maxiter = min(4000, max(300, int(remaining * 0.6 / (est_per_iter * 40))))
            maxfun = min(80000, maxiter * 25)
            sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            best_score, best_a = update_best(best_score, best_a, sc, a_opt, f"lse ci={ci} b={beta}", t_start)
        # Adam refinement if time allows
        if time.monotonic() < p2_end - 40 and n <= 8192:
            remaining = p2_end - time.monotonic()
            n_steps = min(3000, int(remaining / (0.006 * scale_factor)))
            if n_steps > 200:
                sc_adam, a_adam = adam_optimize(a_cur, 800, n_steps=n_steps, lr=0.005 / scale_factor)
                best_score, best_a = update_best(best_score, best_a, sc_adam, a_adam, f"adam ci={ci}", t_start)
                if sc_adam < compute_score(a_cur):
                    a_cur = a_adam
        sc_final = compute_score(a_cur)
        best_score, best_a = update_best(best_score, best_a, sc_final, a_cur, f"p2end ci={ci}", t_start)
        elapsed = time.monotonic() - t0
        print(f"  ci={ci}: score={sc_final:.6f} ({elapsed:.1f}s)")
        # Feed best trajectory forward
        if ci + 1 < len(top_inits) and sc_final < compute_score(top_inits[ci + 1]):
            top_inits[ci + 1] = a_cur.copy()

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: High-beta refinement
    # ================================================================
    p3_end = t_start + 0.78 * effective_budget
    print(f"\n--- Phase 3: High-beta refinement (until t={p3_end - t_start:.0f}s) ---")

    a_refine = best_a.copy()
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 15:
            break
        maxiter = min(5000, max(500, int(remaining * 0.7 / (0.005 * scale_factor * 40))))
        maxfun = min(100000, maxiter * 25)
        sc, a_opt = run_lbfgs(a_refine, beta, maxiter=maxiter, maxfun=maxfun)
        a_refine = a_opt
        best_score, best_a = update_best(best_score, best_a, sc, a_opt, f"refine b={beta}", t_start)

    # Subgradient
    if time.monotonic() < p3_end - 30:
        n_sg = min(1500, int((p3_end - time.monotonic()) / (0.015 * scale_factor)))
        if n_sg > 100:
            sc_sg, a_sg = subgradient_polish(a_refine, n_iters=n_sg, lr_scale=1.0)
            best_score, best_a = update_best(best_score, best_a, sc_sg, a_sg, "subgrad1", t_start)

    # ================================================================
    # Phase 4: Targeted local search
    # ================================================================
    p4_end = t_start + 0.92 * effective_budget
    print(f"\n--- Phase 4: Local Search (until t={p4_end - t_start:.0f}s) ---")

    n_best = len(best_a)
    for trial in range(80):
        if time.monotonic() > p4_end:
            break
        strategy = trial % 6
        if strategy == 0:
            scale_p = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale_p * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strategy == 2:
            A = np.fft.fft(best_a, N_fft)
            half_idx = N_fft // 2 + 1
            noise = 0.015 * (rng.standard_normal(half_idx) + 1j * rng.standard_normal(half_idx))
            noise[0] = 0
            A_mod = A.copy()
            A_mod[:half_idx] *= (1.0 + noise[:half_idx])
            if N_fft % 2 == 0:
                A_mod[half_idx:] = np.conj(A_mod[1:half_idx][::-1])
            else:
                A_mod[half_idx:] = np.conj(A_mod[1:half_idx][::-1])
            a_pert = np.real(np.fft.ifft(A_mod))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            xc = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            ph = 2.0 * np.pi * rng.random()
            pert = 0.03 * np.cos(np.pi * f * xc * xc / (2.0 * n_best) + ph)
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            nb = int(4 + 12 * rng.random())
            per = max(1, n_best // nb)
            block_noise = 0.04 * rng.standard_normal(nb)
            a_pert = best_a.copy()
            for bb in range(nb):
                st = bb * per
                en = min(st + per, n_best)
                a_pert[st:en] *= (1.0 + block_noise[bb])
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            sf_a = spectral_flatten(best_a, N_fft, n_iters=40, lam=0.3)
            a_pert = 0.7 * best_a + 0.3 * sf_a
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        remaining = p4_end - time.monotonic()
        if remaining < 8:
            break
        maxiter = min(2000, max(150, int(remaining * 0.25 / (0.005 * scale_factor * 35))))
        sc, a_opt = run_lbfgs(a_pert, 800, maxiter=maxiter, maxfun=maxiter * 20)
        best_score, best_a = update_best(best_score, best_a, sc, a_opt, f"loc {trial} s{strategy}", t_start)

    # ================================================================
    # Phase 5: Final polish
    # ================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    sc_sg, a_sg = subgradient_polish(best_a, n_iters=min(800, int(60 / (0.015 * scale_factor))), lr_scale=1.5)
    best_score, best_a = update_best(best_score, best_a, sc_sg, a_sg, "final_sg", t_start)

    if time.monotonic() < deadline - 15:
        for beta in [5000, 15000]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=60000)
            best_score, best_a = update_best(best_score, best_a, sc, a_opt, f"final b={beta}", t_start)

    if time.monotonic() < deadline - 5:
        sc_sg2, a_sg2 = subgradient_polish(best_a, n_iters=400, lr_scale=2.0)
        best_score, best_a = update_best(best_score, best_a, sc_sg2, a_sg2, "final_sg2", t_start)

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
