```python
def compute_score(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

    scaled = beta * (c - c_max)

    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    g = 2.0 * conv_res
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
```

```python
def block_optimize(a, B, n_steps, beta, deadline):
    """Optimize random blocks of size B."""
    n = len(a)
    rng = np.random.default_rng()
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        start = rng.integers(0, n - B + 1)
        end = start + B
        
        # Compute gradient for the block
        score, grad = lse_score_grad(a, beta)
        
        # Take a step on the block
        block_grad = grad[start:end]
        block = a[start:end]
        
        # Simple gradient descent with adaptive step size
        lr = 0.1 / (np.max(np.abs(block_grad)) + 1e-10)
        new_block = block - lr * block_grad
        new_block = np.clip(new_block, 0, 1000)
        
        a[start:end] = new_block
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a
```

```python
def block_lbfgs(a, B, n_blocks, beta, deadline):
    """Optimize random blocks of size B using L-BFGS."""
    n = len(a)
    rng = np.random.default_rng()
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for _ in range(n_blocks):
        if time.monotonic() > deadline:
            break
        start = rng.integers(0, n - B + 1)
        end = start + B
        
        # Define objective for the block
        def block_obj(x_block):
            a_trial = a.copy()
            a_trial[start:end] = np.clip(x_block, 0, 1000)
            return lse_score_grad(a_trial, beta)
        
        x0 = a[start:end].copy()
        bounds = [(0, 1000)] * B
        res = minimize(block_obj, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                      options={'maxiter': 100, 'maxfun': 200})
        a[start:end] = np.clip(res.x, 0, 1000)
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a
```
