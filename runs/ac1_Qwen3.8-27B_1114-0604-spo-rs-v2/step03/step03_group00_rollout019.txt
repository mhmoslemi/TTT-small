    # where a_flipped = a[::-1]
    
    a_flipped = a[::-1]
    M2 = (2 * n - 1) + n - 1  # length of full conv of p (len 2n-1) and a (len n)
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AF = np.fft.rfft(a_flipped, M2_fft)
    corr_full = np.fft.irfft(P * AF, M2_fft)
    # corr_full has length M2_fft, the useful part is corr_full[n-1:n-1+n]
    # Actually, conv(p, a_flipped)[k] = sum_i p[i] * a_flipped[k-i] = sum_i p[i] * a[n-1-(k-i)]
    # We want corr[j] = sum_i p[i+j] * a[i]
    # conv(p, a_flipped)[n-1+j] = sum_i p[i] * a_flipped[n-1+j-i] = sum_i p[i] * a[n-1-(n-1+j-i)] = sum_i p[i] * a[i-j]
    # Hmm, that's not quite what we want.
    # Let me use: corr[j] = sum_i p[i+j]*a[i] = sum_i p[j+i]*a[i]
    # This is the cross-correlation (p ⋆ a)[j] = sum_i p[i+j]*a[i]
    # In terms of convolution: (p ⋆ a)[j] = (p * a_flipped)[n-1+j]
    # where * denotes linear convolution and a_flipped[i] = a[n-1-i]
    # (p * a_flipped)[k] = sum_i p[i] * a_flipped[k-i] = sum_i p[i] * a[n-1-k+i]
    # For k = n-1-j: sum_i p[i] * a[n-1-(n-1-j)+i] = sum_i p[i] * a[j+i]
    # That's not right either. Let me just use:
    # corr[j] = sum_i p[i+j]*a[i]
    # = sum_i' p[i']*a[i'-j] where i' = i+j
    # = (p * a_flipped)[j + (n-1)]  ... I'm getting confused.
    
    # Let me just use the direct FFT approach:
    # We want g[j] = sum_{i=0}^{n-1} p[i+j] * a[i] for j = 0, ..., n-1
    # This is the "cross-correlation" of p and a at lag j.
    # Using FFT: g = ifft( fft(p, N) * conj(fft(a, N)) )[:n] where N >= 2n-1+n-1 = 3n-2
    # Let me verify: ifft( P * conj(A) )[j] = sum_k p[k] * conj(a[k-j]) 
    # For real a: = sum_k p[k] * a[k-j]
    # We want sum_i p[i+j]*a[i]. Let k = i+j: sum_k p[k]*a[k-j]. Yes!
    # So g[j] = ifft( fft(p, N) * conj(fft(a, N)) )[j]
    
    N = 1 << ((3 * n - 2) - 1).bit_length()
    P_fft = np.fft.rfft(p, N)
    A_fft = np.fft.rfft(a, N)
    g = np.fft.irfft(P_fft * np.conj(A_fft), N)[:n]
    
    # dL/da[j] = 2 * g[j]
    dL = 2.0 * g
    grad = (2.0 * n) * (dL / (S * S) - 2.0 * L / (S ** 3))
    return score, grad


def annealed_lbfgs(a0, betas, maxiter_per_beta=3000):
    """
    Run L-BFGS with annealing beta schedule.
    Start with smooth beta, increase to sharp beta.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 1.0 / n
    # Normalize
    a = a / a.sum()
    
    best_a = a.copy()
    best_exact = compute_score(a)
    bounds = [(0.0, 1000.0)] * n
    
    for beta in betas:
        res = minimize(
            lse_score_grad, a, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-16, 'maxiter': maxiter_per_beta,
                     'maxfun': maxiter_per_beta * 3, 'maxls': 100}
        )
        a = np.clip(res.x, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            a[:] = 1.0 / n
            S = a.sum()
        a = a / S
        
        sc = compute_score(a)
        if sc < best_exact:
            best_exact = sc
            best_a = a.copy()
    
    return best_exact, best_a


def peak_targeted_cd(a, n_sweeps=5, n_top=None):
    """
    Coordinate descent targeting peak-contributing elements.
    Uses 1D optimization for each variable.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    if n_top is None:
        n_top = max(20, n // 20)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue
        
        # Contribution of a[j] to c[k_star]: 2*a[k_star-j] for j in [j_start, j_end]
        seg_idx = np.arange(j_start, j_end + 1)
        contrib = 2.0 * a[k_star - seg_idx]
        
        # Sort by contribution (descending)
        order = np.argsort(contrib)[::-1]
        top_count = min(n_top, len(order))
        top_idx = seg_idx[order[:top_count]]
        
        improved = False
        for j in top_idx:
            a_old_j = a[j]
            if a_old_j < 1e-15:
                # Try small positive values
                candidates = [1e-3 / n, 1e-2 / n, 0.01 / n]
            else:
                # Try relative changes
                candidates = [
                    a_old_j * 0.3, a_old_j * 0.5, a_old_j * 0.7,
                    a_old_j * 0.85, a_old_j * 1.0, a_old_j * 1.15,
                    a_old_j * 1.3, a_old_j * 1.5, a_old_j * 2.0,
                    0.0,
                ]
            
            best_val = a_old_j
            best_local = best_sc
            
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-14:
                    continue
                a[j] = cv
                S = a.sum()
                if S < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
            
            a[j] = best_val
            if best_val != a_old_j:
                improved = True
        
        # Renormalize
        S = a.sum()
        if S > 1e-12:
            a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def spectral_flatten(a, n_iters=50):
    """
    Try to flatten the convolution by adjusting in the frequency domain.
    Reduces power at frequencies that contribute to the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_mean = float(c.mean())
        c_max = float(c.max())
        if c_max - c_mean < 1e-14:
            break
        
        # Soft-threshold the convolution
        alpha = 0.7 + 0.2 * (it / n_iters)  # decreasing aggressiveness
        thresh = c_mean + (1.0 - alpha) * (c_max - c_mean)
        target = np.where(c > thresh, thresh + alpha * (c - thresh), c)
        
        # Compute error
        err = target - c
        err_pad = np.zeros(M_fft)
        err_pad[:2*n-1] = err
        E = np.fft.rfft(err_pad)
        
        A = np.fft.rfft(a, M_fft)
        
        # Newton step: A_new = A + E / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = E / (2.0 * A + 1e-15j)
        dA[np.abs(A) < 1e-10] = 0
        
        # Damping
        damp = 0.5 * (1.0 - it / (2.0 * n_iters))
        A_new = A + damp * dA
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_inits(n, rng, parent=None):
    """Build diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    inits.append(np.ones(n) * (1.0 / n))
    inits.append(0.5 * (1.0 + 0.5 * np.cos(2.0 * np.pi * x)) / n * n * (1.0/n))
    # U-shaped (to counteract triangular convolution)
    inits.append((0.3 + 0.7 * (2.0 * x - 1.0) ** 2).copy())
    inits.append((0.2 + 0.8 * np.abs(2.0 * x - 1.0)).copy())
    # Multi-freq
    inits.append((0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x) * np.cos(2.0 * np.pi * x)).copy())
    inits.append((0.5 + 0.2 * np.sin(2.0 * np.pi * 2.0 * x) + 0.15 * np.cos(2.0 * np.pi * 5.0 * x)).copy())
    # Random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None))
    # Block structure
    for nb in (4, 8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: adaptive L-BFGS with LSE smoothing, spectral flattening, peak-targeted coordinate
descent with 1D golden-section search, and multi-length exploration with random restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, gradient)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    shifted_max = float(shifted.max())
    exp_shifted = np.exp(shifted - shifted_max)
    log_sum = shifted_max + np.log(exp_shifted.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_shifted / exp_shifted.sum()

    # Gradient: dL/da[j] = 2 * sum_i p[i+j] * a[i] (cross-correlation of p and a)
    N = 1 << ((3 * n - 2) - 1).bit_length()
    P_fft = np.fft.rfft(p, N)
    A_fft = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_fft * np.conj(A_fft), N)[:n]
    dL = 2.0 * corr

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                 'maxfun': maxiter * 3, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def annealed_lbfgs(a0, betas, maxiter_per_beta=3000):
    """L-BFGS with progressive beta (annealing)."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 1.0 / n
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        sc, a_opt = run_lbfgs(a, beta, maxiter=maxiter_per_beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a = a_opt

    return best_sc, best_a


def spectral_flatten(a, n_iters=50):
    """Newton-like spectral flattening of the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        alpha = 0.6 + 0.3 * (it / max(1, n_iters))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M_fft)
        err_pad[:2 * n - 1] = err
        E = np.fft.rfft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            dA = E / (2.0 * A + 1e-15j)
        dA[np.abs(A) < 1e-10] = 0.0

        damp = 0.6 * (1.0 - 0.5 * it / max(1, n_iters))
        A_new = A + damp * dA
        a = np.fft.irfft(A_new, M_fft)[:n]
        a = np.maximum(a, 0.0)
        S_new = a.sum()
        if S_new < 1e-12:
            break
        a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_cd(a, n_sweeps=5, n_top=None):
    """Peak-targeted coordinate descent with 1D golden-section refinement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    if n_top is None:
        n_top = max(15, n // 40)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg_idx = np.arange(j_start, j_end + 1)
        contrib = 2.0 * a[k_star - seg_idx]
        order = np.argsort(contrib)[::-1]
        top_count = min(n_top, len(order))
        top_idx = seg_idx[order[:top_count]]

        improved = False
        for j in top_idx:
            a_old_j = a[j]
            if a_old_j < 1e-14:
                grid = [1e-4 / n, 1e-3 / n, 1e-2 / n, 0.01 / n, 0.0]
            else:
                grid = [a_old_j * f for f in [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.0]]

            best_val = a_old_j
            best_local = best_sc

            for cv in grid:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-14:
                    continue
                a[j] = cv
                S_cur = a.sum()
                if S_cur < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_val = cv

            # Golden-section refinement around best_val
            if best_val > 1e-14:
                lo = best_val * 0.5
                hi = best_val * 2.0
                if lo < 0:
                    lo = 0.0
                for _ in range(15):
                    m1 = hi - (hi - lo) * 0.618
                    m2 = lo + (hi - lo) * 0.618
                    a[j] = m1
                    S1 = a.sum()
                    sc1 = compute_score(a) if S1 > 1e-12 else best_local
                    a[j] = m2
                    S2 = a.sum()
                    sc2 = compute_score(a) if S2 > 1e-12 else best_local
                    if sc1 < sc2:
                        hi = m2
                    else:
                        lo = m1
                    if abs(hi - lo) < 1e-12:
                        break
                    a[j] = 0.5 * (lo + hi)
                    S_mid = a.sum()
                    sc_mid = compute_score(a) if S_mid > 1e-12 else best_local
                    if sc_mid < best_local:
                        best_local = sc_mid
                        best_val = 0.5 * (lo + hi)
                a[j] = best_val

            if best_val != a_old_j:
                improved = True

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if not improved:
            break

    return best_a


def build_inits(n, rng, parent):
    """Build diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    inits.append(np.ones(n) * (1.0 / n))
    inits.append(0.5 + 0.4 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * np.abs(2.0 * x - 1.0))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.25 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(0.4 + 0.3 * np.cos(2.0 * np.pi * 3 * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, None))
    # Block structures
    for nb in (8, 16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_b = np.repeat(vals, per)
        if len(a_b) < n:
            a_b = np.concatenate([a_b, vals[:n - len(a_b)]])
        inits.append(a_b[:n].copy())
    return inits
