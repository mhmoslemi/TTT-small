"""
Hybrid optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2.
Uses LSE-smoothed L-BFGS with progressive beta, spectral initialization,
and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time

# --- Core functions ---

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Handle overflow
    exp_scaled = np.exp(scaled - scaled.max())  # already shifted by c_max
    # Actually scaled is already shifted: scaled = beta*(c - c_max), so max(scaled) = 0
    exp_scaled = np.exp(scaled)  # max is exp(0) = 1
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # where q[j] = sum_k p[k] * a[k-j]
    # q = cross_correlation(p, a) (first n values)
    N3 = 1 << (3*n - 1).bit_length()
    P_full = np.zeros(N3)
    P_full[:2*n-1] = p
    A_full = np.zeros(N3)
    A_full[:n] = a
    q = np.fft.irfft(np.fft.rfft(P_full) * np.conj(np.fft.rfft(A_full)))[:n]
    
    S2 = S * S
    grad = 4.0 * n * (q / S2 - L / (S2 * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) * 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def spectral_flatten(a, n_iters=30, gamma=0.3):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2*n - 1).bit_length()
    for _ in range(n_iters):
        A = np.fft.rfft(a, M)
        power = A.real**2 + A.imag**2
        target = np.sqrt(power.sum() / (2*n-1))
        scale = (1-gamma) + gamma * target / (np.sqrt(power) + 1e-15)
        A *= scale
        a = np.fft.irfft(A, M)[:n]
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    return a

def seed_initializer(a, size=5):
    """Normalize and seed for next run."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(len(a)) * 0.5
    return a / S
