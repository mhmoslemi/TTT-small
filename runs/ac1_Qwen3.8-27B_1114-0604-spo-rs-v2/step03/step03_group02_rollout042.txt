def lse_score_grad(x, beta):
    a=np.asarray(x, dtype=np.float64)
    n=a.size
    S=float(a.sum())
    if S <= 1e-12:
       return 1e100, np.zeros(n)
    c=fft_conv_self(a)
    c_max=float(np.max(c))

    return float(F), grad

def make_lag_weights(c, mode, gamma=5.0, gap=1e-3, min_count=16, max_count=1024):

    cmax=float(np.max(c))
    if cmax <= 1e-300:
       w=np.ones(L)/L; return w
    r=c/cmax

"""
Search for low-peak non-negative step-function sequences.

The objective is homogeneous, so internal optimization mostly normalizes the
sum to one.  The method combines:
  1. Resampled parent / structured / Fourier / circular-square-root starts.
  2. Multiplicative KKT-style peak equalization for the self-convolution.
  3. LP-inspired generation: with a fixed f, optimize g by equalizing
     grad_g convolution(f, g) while controlling convolution(f, g).
  4. LSE-smoothed exact-peak polishing with L-BFGS-B.
  5. Deep active-lag polishing and late perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time

_BOUNDS_CACHE = {}
_EVAL_DEADLINE = 0.0


def make_bounds(n):
    b = _BOUNDS_CACHE.get(n)
    if b is None:
        b = [(0.0, 1000.0)] * n
        _BOUNDS_CACHE[n] = b
    return b


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_array(a, min_sum=1e-12):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < min_sum:
        a[:] = 1.0 / float(a.size)
    else:
        a = a / s
    return np.clip(a, 0.0, 1.0)


def fft_conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    N = next_pow2(L)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:L]


def fft_cross(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    N = next_pow2(L)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:L]


def weighted_grad(a, w):
    """
    For w of length 2n-1, return h[j] = sum_k w[k] * a[k-j],
    where only valid indices are used.  This is the derivative structure
    needed for convolution gradients.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    b = a[::-1].copy()
    m = 3 * n - 2
    N = next_pow2(m)
    B = np.fft.rfft(b, N)
    W = np.fft.rfft(w, N)
    out = np.fft.irfft(B * W, N)[n - 1:2 * n - 1]
    out = np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(out, 0.0)


def exact_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if n == 0 or s < 1e-12:
        return np.inf
    c = fft_conv_self(a)
    m = float(np.max(c))
    if m < 0.0 and m > -1e-14:
        m = 0.0
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def top_mask(c, count):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    count = int(max(0, min(count, L)))
    if count == 0:
        return np.zeros(L, dtype=bool)
    if count == L:
        return np.ones(L, dtype=bool)
    idx = np.argpartition(c, L - count)[L - count:]
    m = np.zeros(L, dtype=bool)
    m[idx] = True
    return m


def make_lag_weights(c, cmax, mode="soft", gamma=5.0, gap=1e-3):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if cmax <= 1e-300 or L == 0:
        return np.ones(L, dtype=np.float64) / max(1, L)

    r = c / cmax
    r = np.nan_to_num(r, nan=0.0, posinf=1.0, neginf=0.0)

    min_count = max(8, L // 500)
    max_count = min(4096, max(32, L // 4))

    if mode == "top":
        thresh = 1.0 - float(gap)
        active = r >= thresh
        cnt = int(np.count_nonzero(active))
        if cnt < min_count:
            active = top_mask(c, min_count)
        elif cnt > max_count:
            active = top_mask(c, max_count)
        w = np.zeros(L, dtype=np.float64)
        cnt = int(np.count_nonzero(active))
        if cnt == 0:
            w[0] = 1.0
        else:
            w[active] = 1.0 / cnt
        return w

    # soft weights
    if gamma <= 1e-8:
        return make_lag_weights(c, cmax, "top", gap=gap)

    w = np.exp(gamma * (r - 1.0))
    w[r < 0.8] *= 1e-8
    w = np.nan_to_num(w, nan=0.0, posinf=1.0, neginf=0.0)
    w += 1e-300
    z = float(w.sum())
    if z <= 1e-300:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return w / z


def lse_score_grad(x, beta):
    global _EVAL_DEADLINE
    if _EVAL_DEADLINE > 1.0 and time.monotonic() > _EVAL_DEADLINE:
        try:
            n = int(len(x))
        except Exception:
            n = 0
        return 1e100, np.zeros(max(0, n), dtype=np.float64)

    a = np.asarray(x, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)

    s = float(a.sum())
    if s <= 1e-12:
        return 1e100, np.zeros(n, dtype=np.float64)

    c = fft_conv_self(a)
    cmax = float(np.max(c))
    if cmax < 0.0 and cmax > -1e-14:
        cmax = 0.0
    if not np.isfinite(cmax) or cmax <= 1e-300:
        return 1e100, np.zeros(n, dtype=np.float64)

    scale = beta * (c - cmax)
    scale = np.minimum(scale, 0.0)
    scale = np.nan_to_num(scale, nan=-1e6, posinf=0.0, neginf=-1e6)
    e = np.exp(scale)
    z = float(e.sum())
    if z <= 1e-300 or not np.isfinite(z):
        L = cmax
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        L = cmax + np.log(z) / float(beta)
        p = e / z

    f = float(2.0 * n * L / (s * s))
    if not np.isfinite(f):
        f = 1e100

    g = 2.0 * weighted_grad(a, p)
    s2 = s * s
    s3 = s2 * s
    grad = 2.0 * n * (g / s2 - 2.0 * L / s3)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(f), grad


def lbfgs_fun(x, beta):
    return lse_score_grad(x, beta)


def estimate_eval_rate(a, beta, trials=4):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return 100.0
    t0 = time.monotonic()
    for _ in range(max(1, int(trials))):
        lse_score_grad(a, float(beta))
    dt = time.monotonic() - t0
    if dt <= 1e-9:
        return 1000.0
    return max(1e-3, float(trials) / dt)


def scale_raw_to_peak(a, target_peak=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(a.sum())
    if a.size == 0:
        return a
    if s < 1e-12:
        a[:] = 1.0 / float(a.size)
        return np.clip(a, 0.0, 1000.0)
    c = fft_conv_self(a)
    m = float(np.max(c))
    if m <= 1e-300:
        return np.clip(a, 0.0, 1000.0)
    out = a * np.sqrt(float(target_peak) / m)
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    return np.clip(out, 0.0, 1000.0)


def run_lbfgs(a0, beta, deadline, rate, maxfun_cap=30000):
    global _EVAL_DEADLINE

    a0 = np.asarray(a0, dtype=np.float64).copy()
    if a0.size == 0:
        return np.inf, np.ones(1, dtype=np.float64)
    a0 = np.nan_to_num(a0, nan=0.0, posinf=1000.0, neginf=0.0)
    a0 = np.clip(a0, 0.0, 1000.0)

    deadline = float(deadline)
    remaining = deadline - time.monotonic()
    if remaining < 4.0:
        return np.inf, normalize_array(a0)

    if not np.isfinite(rate) or rate <= 0.0:
        rate = 100.0

    a0 = scale_raw_to_peak(a0, 0.1)
    s = float(a0.sum())
    if s < 1e-12:
        a0 = scale_raw_to_peak(np.ones(a0.size) / float(a0.size), 0.1)

    sc0 = exact_score(a0)
    if not np.isfinite(sc0):
        a0 = scale_raw_to_peak(np.ones(a0.size) / float(a0.size), 0.1)
        sc0 = exact_score(a0)

    remaining = deadline - time.monotonic()
    if remaining < 4.0:
        return float(sc0), normalize_array(a0)

    maxfun = int(min(maxfun_cap, max(50, remaining * 0.35 * rate)))
    maxiter = int(min(20000, max(50, maxfun // 10)))

    _EVAL_DEADLINE = deadline
    try:
        res = minimize(
            lbfgs_fun,
            a0,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            bounds=make_bounds(a0.size),
            options={
                "disp": False,
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1e-12,
                "gtol": 1e-6,
                "maxls": 30,
            },
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    finally:
        _EVAL_DEADLINE = 0.0

    if time.monotonic() > deadline:
        return float(sc0), normalize_array(a0)

    sc1 = exact_score(a_out)
    if not np.isfinite(sc1) or sc1 >= sc0:
        return float(sc0), normalize_array(a0)
    return float(sc1), normalize_array(a_out)


def multiplicative_equalize(a, deadline, mode="soft", gamma=5.0, alpha=0.3, gap=1e-3, max_iters=1000000):
    a = normalize_array(a)
    n = a.size
    if n == 0:
        return np.inf, a

    best_a = a.copy()
    c = fft_conv_self(a)
    m = float(np.max(c))
    if m < 0.0 and m > -1e-14:
        m = 0.0
    best_sc = float(2.0 * n * m)

    deadline = float(deadline)
    it = 0
    while it < max_iters:
        if time.monotonic() > deadline:
            break
        if m <= 1e-300:
            break

        w = make_lag_weights(c, m, mode=mode, gamma=gamma, gap=gap)
        h = 2.0 * weighted_grad(a, w)
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
        h = np.maximum(h, 0.0)
        hmax = float(np.max(h))
        if hmax <= 1e-300:
            break

        hmean = float(np.mean(h))
        delta = float(alpha) * (h - hmean) / hmax
        delta = np.clip(delta, -10.0, 10.0)

        a_pos = np.maximum(a, 1e-30)
        anew = a_pos * np.exp(-delta)
        s = float(anew.sum())
        if not np.isfinite(s) or s <= 1e-300:
            break
        anew = anew / s
        anew = np.clip(anew, 0.0, 1.0)

        c = fft_conv_self(anew)
        m = float(np.max(c))
        if m < 0.0 and m > -1e-14:
            m = 0.0
        sc = float(2.0 * n * m)

        if sc < best_sc:
            best_sc = sc
            best_a = anew.copy()

        if np.max(np.abs(h - hmean)) <= 1e-15 * hmax:
            break

        a = anew
        it += 1

    return float(best_sc), best_a


def lp_generate(f, budget_s, mode="soft", gamma=5.0, alpha=0.5, gap=1e-3, init=None, max_iters=1000000):
    f = normalize_array(f)
    n = f.size
    if n == 0:
        return np.inf, f

    if init is None:
        g = f.copy()
    else:
        g = normalize_array(init)

    g = np.clip(g, 0.0, 1.0)
    s = float(g.sum())
    if s < 1e-12:
        g = f.copy()

    best_g = g.copy()
    best_sc = exact_score(g)

    deadline = time.monotonic() + float(budget_s)
    it = 0
    while it < max_iters:
        if time.monotonic() > deadline:
            break

        c = fft_cross(f, g)
        m = float(np.max(c))
        if m <= 1e-300:
            break

        w = make_lag_weights(c, m, mode=mode, gamma=gamma, gap=gap)
        h = weighted_grad(f, w)
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
        h = np.maximum(h, 0.0)
        hmax = float(np.max(h))
        if hmax <= 1e-300:
            break

        hmean = float(np.mean(h))
        delta = float(alpha) * (h - hmean) / hmax
        delta = np.clip(delta, -10.0, 10.0)

        g_pos = np.maximum(g, 1e-30)
        gnew = g_pos * np.exp(-delta)
        snew = float(gnew.sum())
        if not np.isfinite(snew) or snew <= 1e-300:
            break
        gnew = gnew / snew
        g = np.clip(gnew, 0.0, 1.0)

        if (it % 4) == 0:
            sc = exact_score(g)
            if sc < best_sc:
                best_sc = sc
                best_g = g.copy()

        if np.max(np.abs(h - hmean)) <= 1e-15 * hmax:
            break

        it += 1

    sc = exact_score(g)
    if sc < best_sc:
        best_sc = sc
        best_g = g.copy()

    return float(best_sc), best_g


def update_best(state, sc, a, label=""):
    if a is None:
        return False
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return False
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-8:
        return False
    a = a / s
    if not np.isfinite(sc):
        return False
    if sc < state["score"]:
        state["score"] = float(sc)
        state["a"] = a.copy()
        print(f"NEW BEST {sc:.8f} n={a.size} {label}")
        return True
    return False


def random_binary_init(n, rng, p=0.5):
    a = (rng.random(n) < float(p)).astype(np.float64)
    if a.sum() == 0:
        a[0] = 1.0
    return normalize_array(a)


def smooth_noise_init(n, rng, bands=8, amp=0.35):
    a = np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n)
    upper = min(int(bands), n // 2)
    for m in range(1, upper + 1):
        am = float(amp) / (float(m) ** 0.8)
        phc = 2.0 * np.pi * float(rng.random())
        phs = 2.0 * np.pi * float(rng.random())
        a = a + am * (np.cos(2.0 * np.pi * m * x + phc) + np.sin(2.0 * np.pi * m * x + phs))
    return normalize_array(a)


def block_pattern_init(n, rng, blocks=8):
    n = int(n)
    blocks = max(1, min(int(blocks), n))
    per = max(1, n // blocks)
    blocks = max(1, n // per)
    vals = np.clip(1.0 + 0.8 * rng.standard_normal(blocks), 0.05, 3.0)
    a = np.repeat(vals, per)
    if a.size < n:
        a = np.concatenate([a, vals[:n - a.size]])
    a = a[:n]
    return normalize_array(a)


def circular_sqrt_init(n, rng, decay=0.5):
    n = int(n)
    N = next_pow2(n)
    k = np.minimum(np.arange(N), N - np.arange(N))
    C = np.exp(-float(decay) * (k / float(max(1, N // 2))) ** 2)
    C = np.nan_to_num(C, nan=0.0, posinf=1.0, neginf=0.0)
    C[0] = 1.0

    A = np.zeros(N, dtype=np.complex128)
    if N > 1:
        half = N // 2
        if half > 1:
            signs = 2.0 * (rng.random(half - 1) > 0.5).astype(np.float64) - 1.0
            vals = signs * np.sqrt(C[1:half])
            A[1:half] = vals
            A[N - 1:N - half:-1] = vals
        if N % 2 == 0:
            A[N // 2] = (1.0 if rng.random() > 0.5 else -1.0) * np.sqrt(C[N // 2])
        A[0] = np.sqrt(C[0])

    a_full = np.fft.ifft(A).real
    a = a_full[:n]
    a = np.clip(a, 0.0, 1.0)
    s = float(a.sum())
    if s < 1e-6:
        a = np.clip(1.0 + 0.3 * rng.standard_normal(n), 0.0, 1.0)
        s = float(a.sum())
    if s < 1e-12:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s
    return a


def load_parent():
    try:
        p = globals().get("height_sequence_1", None)
        if p is None:
            return None
        arr = np.asarray(p, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.sum() < 1e-12:
            return None
        return arr
    except Exception:
        return None


def resample_to_length(p, n):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if p.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if p.size == n:
        return normalize_array(p)
    xs = np.linspace(0.0, float(p.size - 1), n)
    q = np.interp(xs, np.arange(p.size, dtype=np.float64), p)
    q = np.nan_to_num(q, nan=0.0, posinf=1000.0, neginf=0.0)
    q = np.clip(q, 0.0, 1000.0)
    return normalize_array(q)


def medium_refine(a, budget_s, state, label="cand"):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s))

    local_a = normalize_array(a)
    local_sc = exact_score(local_a)
    update_best(state, local_sc, local_a, f"{label} init")

    passes = [
        ("soft", 1.0, 1.0, 0.16, 1e-3),
        ("soft", 3.0, 0.6, 0.16, 1e-3),
        ("top", 0.0, 0.3, 0.14, 1e-4),
        ("soft", 10.0, 0.3, 0.14, 1e-3),
        ("top", 0.0, 0.15, 0.12, 1e-5),
        ("soft", 30.0, 0.15, 0.10, 1e-4),
    ]

    for mode, gamma, alpha, frac, gap in passes:
        now = time.monotonic()
        if now >= deadline:
            break
        ptime = float(budget_s) * float(frac)
        ptime = max(5.0, min(ptime, deadline - now))
        sc, an = multiplicative_equalize(
            local_a, now + ptime, mode=mode, gamma=gamma, alpha=alpha, gap=gap, max_iters=1000000
        )
        local_a = an
        update_best(state, sc, an, f"{label} mpe {mode} g{gamma} a{alpha}")

    rem = deadline - time.monotonic()
    if rem > 25.0:
        a_scaled = scale_raw_to_peak(local_a, 0.1)
        rate = estimate_eval_rate(a_scaled, 100.0, 4)
        for beta in (100.0, 300.0, 1000.0):
            now = time.monotonic()
            rem = deadline - now
            if rem < 12.0:
                break
            sub = min(rem * 0.4, 60.0)
            if sub < 8.0:
                break
            sc, an = run_lbfgs(local_a, beta, now + sub, rate)
            local_a = an
            update_best(state, sc, an, f"{label} lbfgs b{int(beta)}")

    return state["a"]


def lp_phase(state, deadline, rng):
    f = state["a"].copy()
    n = f.size

    inits = [("f", f.copy())]
    inits.append(("smooth", smooth_noise_init(n, rng, 8, 0.4)))
    inits.append(("binary", random_binary_init(n, rng, 0.5)))
    inits.append(("block", block_pattern_init(n, rng, 10)))
    if n in (4096, 8192, 16384):
        inits.append(("circ", circular_sqrt_init(n, rng, 0.3)))

    print(f"LP phase: n={n}, inits={len(inits)}, remaining={max(0.0, deadline-time.monotonic()):.1f}s")

    sub_budget = max(1.0, (deadline - time.monotonic()) / float(max(1, len(inits))))

    for label, init in inits:
        now = time.monotonic()
        if now >= deadline:
            break
        b = min(sub_budget, deadline - now)
        if b < 12.0:
            break

        sc, g = lp_generate(
            state["a"],
            b * 0.6,
            mode="soft",
            gamma=5.0,
            alpha=0.5,
            gap=1e-3,
            init=init,
        )
        update_best(state, sc, g, f"lp gen {label}")

        now = time.monotonic()
        if now + 10.0 < deadline:
            sc2, gm = multiplicative_equalize(
                g,
                min(deadline - 5.0, now + b * 0.25),
                mode="top",
                gap=1e-4,
                alpha=0.25,
                max_iters=1000000,
            )
            update_best(state, sc2, gm, f"lp mpe {label}")

            blend = normalize_array(0.5 * state["a"] + 0.5 * gm)
            scb = exact_score(blend)
            update_best(state, scb, blend, f"lp blend {label}")

            now = time.monotonic()
            if now + 10.0 < deadline:
                sc3, bm = multiplicative_equalize(
                    blend,
                    min(deadline - 5.0, now + b * 0.15),
                    mode="soft",
                    gamma=10.0,
                    alpha=0.25,
                    gap=1e-3,
                    max_iters=1000000,
                )
                update_best(state, sc3, bm, f"lp blend mpe {label}")


def deep_polish(state, deadline):
    stages = [
        {"type": "mpe", "mode": "soft", "gamma": 1.0, "alpha": 1.0, "gap": 1e-3, "sec": 25.0},
        {"type": "mpe", "mode": "soft", "gamma": 3.0, "alpha": 0.6, "gap": 1e-3, "sec": 25.0},
        {"type": "lbfgs", "beta": 1000.0, "sec": 40.0},
        {"type": "mpe", "mode": "top", "gamma": 0.0, "alpha": 0.3, "gap": 1e-4, "sec": 25.0},
        {"type": "mpe", "mode": "soft", "gamma": 10.0, "alpha": 0.3, "gap": 1e-3, "sec": 25.0},
        {"type": "lbfgs", "beta": 3000.0, "sec": 40.0},
        {"type": "mpe", "mode": "top", "gamma": 0.0, "alpha": 0.2, "gap": 1e-5, "sec": 25.0},
        {"type": "mpe", "mode": "soft", "gamma": 30.0, "alpha": 0.15, "gap": 1e-4, "sec": 25.0},
        {"type": "lbfgs", "beta": 10000.0, "sec": 40.0},
        {"type": "mpe", "mode": "top", "gamma": 0.0, "alpha": 0.1, "gap": 1e-6, "sec": 25.0},
        {"type": "mpe", "mode": "soft", "gamma": 100.0, "alpha": 0.1, "gap": 1e-4, "sec": 20.0},
        {"type": "lbfgs", "beta": 30000.0, "sec": 30.0},
    ]

    rate = None
    last_rate = 0.0

    for st in stages:
        now = time.monotonic()
        sec = min(float(st["sec"]), deadline - now - 10.0)
        if sec < 4.0:
            break

        if st["type"] == "mpe":
            sc, an = multiplicative_equalize(
                state["a"],
                now + sec,
                mode=st["mode"],
                gamma=st["gamma"],
                alpha=st["alpha"],
                gap=st["gap"],
                max_iters=1000000,
            )
            update_best(state, sc, an, f"deep mpe {st['mode']} g{st['gamma']} a{st['alpha']}")
        else:
            beta = float(st["beta"])
            if rate is None or (time.monotonic() - last_rate) > 90.0:
                a_scaled = scale_raw_to_peak(state["a"], 0.1)
                rate = estimate_eval_rate(a_scaled, beta, 3)
                last_rate = time.monotonic()
            sc, an = run_lbfgs(state["a"], beta, now + sec, rate)
            update_best(state, sc, an, f"deep lbfgs b{int(beta)}")

    # Use any final sliver for exact active-lag polishing.
    while time.monotonic() < deadline - 5.0:
        now = time.monotonic()
        sec = min(12.0, deadline - now - 5.0)
        if sec < 3.0:
            break
        sc, an = multiplicative_equalize(
            state["a"],
            now + sec,
            mode="top",
            gamma=0.0,
            alpha=0.08,
            gap=1e-6,
            max_iters=1000000,
        )
        update_best(state, sc, an, "deep final top")


def targeted_perturb(a, strength=0.05):
    a = normalize_array(a)
    n = a.size
    if n < 8:
        return a

    c = fft_conv_self(a)
    k = int(np.argmax(c))
    g = np.zeros(n, dtype=np.float64)

    jstart = max(0, k - (n - 1))
    jend = min(k, n - 1)
    if jend >= jstart:
        idx = np.arange(jstart, jend + 1)
        u = k - idx
        g[idx] = 2.0 * a[u]

    top_count = max(8, min(n // 50, 512))
    if top_count >= n:
        top_count = max(1, n // 2)
    if top_count <= 0:
        return a

    top_idx = np.argpartition(g, -top_count)[-top_count:]
    gtop = g[top_idx]
    gmax = float(np.max(gtop))
    if gmax <= 1e-30:
        return a

    a2 = a.copy()
    red = a2[top_idx] * (float(strength) * (gtop / gmax))
    a2[top_idx] -= red
    total = float(red.sum())

    if total > 0.0:
        bottom_count = max(8, top_count // 2)
        bottom_idx = np.argpartition(g, bottom_count)[:bottom_count]
        a2[bottom_idx] += total / float(max(1, bottom_idx.size))

    return normalize_array(a2)


def refine_perturbed(a, budget_s, state, label, rng=None):

            gamma=0.0,
            alpha=0.2,
            gap=1e-4,
            max_iters=1000000,
        )
        local_a = an
        update_best(state, sc, an, f"{label} mpe")

    now = time.monotonic()
    if now + 15.0 < deadline:
        a_scaled = scale_raw_to_peak(local_a, 0.1)
        rate = estimate_eval_rate(a_scaled, 1000.0, 3)
        sc, an = run_lbfgs(local_a, 1000.0, min(deadline - 2.0, now + max(5.0, budget_s * 0.4)), rate)
        update_best(state, sc, an, f"{label} lbfgs")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = t_start + total - 45.0
    if deadline <= t_start + 60.0:
        deadline = t_start + max(60.0, total - 10.0)

    rng = np.random.default_rng(int(seed))

    parent = load_parent()
    n_primary = 8192
    if parent is not None and parent.size > 0:
        if parent.size in (4096, 8192, 16384):
            n_primary = int(parent.size)

    state = {"score": np.inf, "a": np.ones(n_primary, dtype=np.float64) / float(n_primary)}
    update_best(state, exact_score(state["a"]), state["a"], "uniform")

    parent_primary = None
    if parent is not None and parent.size > 0:
        parent_primary = resample_to_length(parent, n_primary)
        sc = exact_score(parent_primary)
        update_best(state, sc, parent_primary, "parent resampled")

    print(f"Primary n={n_primary}, parent available={parent is not None}, current best={state['score']:.8f}")

    # Phase 1: parent + diverse initializations.
    phase1_deadline = min(deadline, t_start + 0.20 * total)
    if parent_primary is not None:
        b = min(0.10 * total, phase1_deadline - time.monotonic())
        if b > 30.0:
            medium_refine(parent_primary, b, state, "parent")

    alt_candidates = [
        ("smooth", smooth_noise_init(n_primary, rng, 8, 0.4)),
        ("circ1", circular_sqrt_init(n_primary, rng, 0.35) if n_primary in (4096, 8192, 16384) else smooth_noise_init(n_primary, rng, 10, 0.5)),
        ("circ2", circular_sqrt_init(n_primary, rng, 0.15) if n_primary in (4096, 8192, 16384) else block_pattern_init(n_primary, rng, 12)),
        ("binary", random_binary_init(n_primary, rng, 0.5)),
        ("block", block_pattern_init(n_primary, rng, 10)),
    ]

    if parent_primary is not None:
        noised = parent_primary * (1.0 + 0.02 * rng.standard_normal(n_primary))
        alt_candidates.insert(0, ("parent_noise", normalize_array(noised)))

    alt_budget = max(30.0, min(90.0, 0.035 * total))
    for label, cand in alt_candidates:
        if time.monotonic() >= phase1_deadline:
            break
        sc0 = exact_score(cand)
        update_best(state, sc0, cand, f"{label} init")
        b = min(alt_budget, phase1_deadline - time.monotonic())
        if b < 20.0:
            break
        medium_refine(cand, b, state, label)

    # Phase 2: LP-inspired complementary generation.
    lp_deadline = min(deadline, phase1_deadline + 0.18 * total)
    if time.monotonic() < lp_deadline - 20.0:
        lp_phase(state, lp_deadline, rng)

    # Phase 3: optional larger length.
    n_secondary = 16384 if n_primary < 16384 else 4096
    sec_deadline = min(deadline, lp_deadline + 0.10 * total)
    if n_secondary in (4096, 8192, 16384) and time.monotonic() < sec_deadline - 70.0:
        if parent is not None:
            a2 = resample_to_length(parent, n_secondary)
        else:
            a2 = circular_sqrt_init(n_secondary, rng, 0.3)
        b2 = min(0.06 * total, sec_deadline - time.monotonic())
        if b2 > 50.0:
            medium_refine(a2, b2, state, "sec_parent")

        if time.monotonic() < sec_deadline - 40.0:
            a2u = resample_to_length(state["a"], n_secondary)
            b2u = min(0.04 * total, sec_deadline - time.monotonic())
            if b2u > 40.0:
                medium_refine(a2u, b2u, state, "sec_upsample")

    # Phase 4: deep polish.
    polish_deadline = deadline - 30.0
    if time.monotonic() < polish_deadline - 20.0:
        deep_polish(state, polish_deadline)

    # Phase 5: late perturbations.
    if time.monotonic() < deadline - 25.0:
        late_budget = min(90.0, max(0.0, deadline - time.monotonic() - 20.0))
        if late_budget > 30.0:
            for i in range(3):
                if time.monotonic() >= deadline - 20.0:
                    break
                if i == 0:
                    ap = normalize_array(state["a"] * (1.0 + 0.02 * rng.standard_normal(state["a"].size)))
                elif i == 1:
                    ap = targeted_perturb(state["a"], 0.05)
                else:
                    ap = normalize_array(state["a"] * (1.0 + 0.05 * rng.standard_normal(state["a"].size)))
                refine_perturbed(ap, max(12.0, late_budget / 3.0), state, f"late{i}", rng)

    out_a = np.asarray(state["a"], dtype=np.float64)
    out_a = np.nan_to_num(out_a, nan=0.0, posinf=1000.0, neginf=0.0)
    out_a = np.clip(out_a, 0.0, 1000.0)
    s = float(out_a.sum())
    if out_a.size == 0 or s < 0.01:
        out_a = np.ones(8192, dtype=np.float64) / 8192.0
    else:
        out_a = out_a / s

    out = [float(x) for x in out_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = np.inf

    if (not np.isfinite(verified)) or verified > 2.05:
        fb = [1.0 / 8192.0] * 8192
        vf = np.inf
        try:
            vf = float(evaluate_sequence(fb))
        except Exception:
            vf = np.inf
        if np.isfinite(vf) or not np.isfinite(verified):
            out = fb
            verified = vf

    print(f"FINAL internal={state['score']:.8f} verified={verified:.8f} len={len(out)} time={time.monotonic()-t_start:.1f}")
    return out
