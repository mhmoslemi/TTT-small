def compute_original_score(a):
    """Compute the original T score: 2n*max(conv)/sum²"""
    n_len = 2 * len(a) - 1
    conv_linear = np.convolve(a, a, mode='full')
    total_sum = np.sum(a)
    if total_sum < 0.01:
        return np.inf
    return float(2.0 * len(a) * np.max(conv_linear) / (total_sum ** 2))
