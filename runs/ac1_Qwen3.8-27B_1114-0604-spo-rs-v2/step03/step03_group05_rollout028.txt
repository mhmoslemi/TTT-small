"""
LP-based iterative refinement for step function optimization.
Uses the linear programming approach from the state-of-the-art:
at each step, find g that maximizes sum(g) subject to g>=0 and (f*g)_inf <= M,
then blend f with g. Combined with LSE smoothing and peak shaving.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where A @ b = convolve(f, b)."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        length = j_hi - j_lo + 1
        if length > 0:
            rows.append(np.full(length, k, dtype=np.int32))
            cols.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
            data.append(f[j_lo:j_hi+1])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(f, M):
    """Solve the LP: maximize sum(b) s.t. b >= 0, (f*b)(k) <= M for all k."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0

def peak_shave(a, iters=3000):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

"""
LP-based iterative refinement for step function optimization.
Uses the linear programming dual structure to find optimal blending directions,
combined with LSE smoothing, reciprocal heuristics, and peak shaving.
Key innovation: exploits the interval-matrix structure of the convolution LP
for fast dual solutions, enabling many more iterations within the time budget.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_index(n):
    """Pre-compute row and column indices for the convolution matrix."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        length = j_hi - j_lo + 1
        if length > 0:
            rows_parts.append(np.full(length, k, dtype=np.int32))
            cols_parts.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
    return np.concatenate(rows_parts), np.concatenate(cols_parts)


def solve_lp_direction(f, M, row_idx, col_idx):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)(k)<=M. Uses pre-computed indices."""
    n = len(f)
    data = f[col_idx]
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def solve_dual_greedy(f, M):
    """
    Solve the dual LP using a greedy sliding-window algorithm.
    Dual: min M*sum(y) s.t. for each j, sum(y[j:j+n]) >= 1/f[j], y >= 0.
    Returns the dual solution y (length 2n-1) and the window sums W.
    """
    n = len(f)
    # y has length 2n-1, indices 0 to 2n-2
    y = np.zeros(2 * n - 1)
    W = np.zeros(n)  # W[j] = sum(y[j:j+n])
    
    # Running window sum
    window_sum = 0.0
    for j in range(n):
        # Window j covers y[j] to y[j+n-1]
        # Update window_sum: add y[j+n-1], remove y[j-1] (if j>0)
        if j == 0:
            window_sum = 0.0
        else:
            window_sum += y[j + n - 1] - y[j - 1]
        
        req = 1.0 / max(f[j], 1e-30)
        if window_sum < req:
            deficit = req - window_sum
            pos = j + n - 1
            if pos < 2 * n - 1:
                y[pos] += deficit
                window_sum += deficit
        W[j] = window_sum
    
    return y, W


def reciprocal_direction(a):
    """Compute the reciprocal direction: b = 1/(a+eps), normalized to match sum(a)."""
    eps = 1e-10
    b = 1.0 / (a + eps)
    S = float(b.sum())
    if S < 1e-30:
        return a.copy()
    S_a = float(a.sum())
    b = b * (S_a / S)
    return b


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(func, a0, args=args, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-18, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def peak_shave(a, iters=3000):
    """Exact subgradient descent on max-conv objective with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-20)
        step = min(step, 1.0 / S)
        step = max(step, 1e-6 / (gn * S))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

            direction = solve_lp_direction(a_lp, M, row_idx, col_idx)
            if direction is not None and float(direction.sum()) > 1e-6:
                direction = direction * (S / float(direction.sum()))
                dir_type = "LP"
            else:
                direction = reciprocal_direction(a_lp)
                dir_type = "RECIP"
        else:
            # Fast greedy dual + reciprocal blend
            y_dual, W_dual = solve_dual_greedy(a_lp, M)
            # Use reciprocal as base direction, modulated by dual information
            direction = reciprocal_direction(a_lp)
            # Scale positions where dual constraint is tight (W[j] = 1/f[j])
            tight_mask = np.abs(W_dual * a_lp - 1.0) < 0.1
            if tight_mask.sum() > n_lp // 4:
                # Most constraints tight - reciprocal is good
                pass
            else:
                # Blend with uniform
                direction = 0.7 * direction + 0.3 * np.full(n_lp, S / n_lp)
            dir_type = "GRD"

        direction = np.clip(direction, 0.0, 1000.0)

        # Line search over step sizes
        best_t = 0.0
        best_sc_t = sc_cur
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1.0 - t) * a_lp + t * direction
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc_try = compute_score(a_try)
            if sc_try < best_sc_t:
                best_sc_t = sc_try
                best_t = t

        if best_t > 0:
            a_lp = (1.0 - best_t) * a_lp + best_t * direction
            a_lp = np.clip(a_lp, 0.0, 1000.0)

        sc_new = compute_score(a_lp)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_lp.copy()
            print(f"  LP iter {lp_iter} [{dir_type}]: NEW BEST {sc_new:.8f} "
                  f"(t={best_t:.2f}, {time.monotonic()-t_start:.0f}s)")
        elif lp_iter % 30 == 0:
            print(f"  LP iter {lp_iter} [{dir_type}]: {sc_new:.8f} "
                  f"(t={best_t:.2f}, dt={time.monotonic()-t0:.2f}s)")

        if sc_new < prev_sc - 1e-10:
            no_improve = 0
        else:
            no_improve += 1
        prev_sc = sc_new
        lp_iter += 1

        # Periodic LSE smoothing
        if lp_iter % 25 == 0 and time.monotonic() < phase2_end:
            a_lp = run_lbfgs(a_lp, lse_score_grad, (1000,), maxiter=150, maxfun=3000)
            sc_smooth = compute_score(a_lp)
            if sc_smooth < best_score:
                best_score = sc_smooth
                best_a = a_lp.copy()
                print(f"  After LSE smoothing: NEW BEST {sc_smooth:.8f}")
            no_improve = 0

        # Perturb if stuck
        if no_improve > 40:
            strategy = rng.integers(0, 3)
            if strategy == 0:
                a_lp = a_lp * (1.0 + 0.08 * rng.standard_normal(n_lp))
            elif strategy == 1:
                a_lp = targeted_perturb(a_lp, rng, strength=0.05)
            else:
                # Frequency domain perturbation
                L = 1 << (2 * n_lp - 2).bit_length()
                A = np.fft.fft(a_lp, L)
                A_mag = np.abs(A) * (1.0 + 0.03 * rng.standard_normal(L))
                A_mag = np.clip(A_mag, 0, None)
                A_new = A_mag * np.exp(1j * np.angle(A))
                a_lp = np.fft.ifft(A_new).real[:n_lp]
            a_lp = np.clip(a_lp, 0.0, 1000.0)
            no_improve = 0
            print(f"  Perturbed at iter {lp_iter} (strategy={strategy})")

    print(f"  Phase 2 complete: {lp_iter} LP iterations, best={best_score:.8f}")

    # ===================================================================
    # Phase 3: Full LP refinement from best (15% time)
    # ===================================================================
    phase3_end = t_start + 0.73 * (total_budget - 25.0)
    print("\n=== Phase 3: Full LP Refinement ===")

    a_lp2 = best_a.copy()
    n_lp2 = len(a_lp2)
    row_idx2, col_idx2 = build_conv_index(n_lp2)

    lp2_iter = 0
    while time.monotonic() < phase3_end:
        t0 = time.monotonic()
        sc_cur = compute_score(a_lp2)
        c = fast_conv(a_lp2)
        S = float(a_lp2.sum())
        M = float(c.max())

        direction = solve_lp_direction(a_lp2, M, row_idx2, col_idx2)
        if direction is not None and float(direction.sum()) > 1e-6:
            direction = direction * (S / float(direction.sum()))
        else:
            direction = reciprocal_direction(a_lp2)

        direction = np.clip(direction, 0.0, 1000.0)

        best_t = 0.0
        best_sc_t = sc_cur
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = (1.0 - t) * a_lp2 + t * direction
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc_try = compute_score(a_try)
            if sc_try < best_sc_t:
                best_sc_t = sc_try
                best_t = t

        if best_t > 0:
            a_lp2 = (1.0 - best_t) * a_lp2 + best_t * direction
            a_lp2 = np.clip(a_lp2, 0.0, 1000.0)

        sc_new = compute_score(a_lp2)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_lp2.copy()
            print(f"  LP2 iter {lp2_iter}: NEW BEST {sc_new:.8f} ({time.monotonic()-t_start:.0f}s)")
        elif lp2_iter % 10 == 0:
            print(f"  LP2 iter {lp2_iter}: {sc_new:.8f} (dt={time.monotonic()-t0:.1f}s)")

        lp2_iter += 1

        # Occasional LSE
        if lp2_iter % 15 == 0:
            a_lp2 = run_lbfgs(a_lp2, lse_score_grad, (2000,), maxiter=100, maxfun=2000)
            sc_s = compute_score(a_lp2)
            if sc_s < best_score:
                best_score = sc_s
                best_a = a_lp2.copy()
                print(f"  LP2+LSE: NEW BEST {sc_s:.8f}")

    print(f"  Phase 3 complete: {lp2_iter} iterations, best={best_score:.8f}")

    # ===================================================================
    # Phase 4: Peak shaving (10% time)
    # ===================================================================
    phase4_end = t_start + 0.83 * (total_budget - 25.0)
    print("\n=== Phase 4: Peak Shaving ===")
    if time.monotonic() < phase4_end:
        t0 = time.monotonic()
        iters = max(500, int((phase4_end - time.monotonic()) * 80))
        sc, a_polished = peak_shave(best_a, iters=iters)
        if sc < best_score:
            best_score = sc
            best_a = a_polished.copy()
            print(f"  Peak shave: NEW BEST {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        else:
            print(f"  Peak shave: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 5: Multi-peak shaving (5% time)
    # ===================================================================
    phase5_end = t_start + 0.88 * (total_budget - 25.0)
    print("\n=== Phase 5: Multi-Peak Shaving ===")
    if time.monotonic() < phase5_end:
        t0 = time.monotonic()
        sc, a_mp = multi_peak_shave(best_a, iters_per_peak=200, num_peaks=6)
        if sc < best_score:
            best_score = sc
            best_a = a_mp.copy()
            print(f"  Multi-peak: NEW BEST {sc:.8f}")
        else:
            print(f"  Multi-peak: {sc:.8f}")

    # ===================================================================
    # Phase 6: Structured restarts (5% time)
    # ===================================================================
    phase6_end = deadline
    print("\n=== Phase 6: Restarts ===")
    n_best = len(best_a)
    restart_count = 0
    while time.monotonic() < phase6_end and restart_count < 15:
        t0 = time.monotonic()
        strategy = restart_count % 4
        if strategy == 0:
            a_pert = best_a * (1.0 + 0.04 * rng.standard_normal(n_best))
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03)
        elif strategy == 2:
            freq = 1.0 + 3.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        else:
            a_pert = best_a.copy()
            thresh = np.median(a_pert)
            high_mask = a_pert > thresh
            low_mask = ~high_mask
            if high_mask.sum() > 0 and low_mask.sum() > 0:
                transfer = 0.04 * a_pert[high_mask].sum() / high_mask.sum()
                a_pert[high_mask] -= transfer
                a_pert[low_mask] += transfer * high_mask.sum() / low_mask.sum()

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LSE + a couple LP steps
        a_opt = run_lbfgs(a_pert, lse_score_grad, (2000,), maxiter=300, maxfun=6000)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Restart #{restart_count}: NEW BEST {sc:.8f}")

        # Try one LP step from the perturbed point
        c = fast_conv(a_opt)
        S = float(a_opt.sum())
        M = float(c.max())
        if len(row_idx) == n_best:
            d = solve_lp_direction(a_opt, M, row_idx, col_idx)
        else:
            d = solve_lp_direction(a_opt, M, *build_conv_index(n_best))
        if d is not None and float(d.sum()) > 1e-6:
            d = d * (S / float(d.sum()))
            d = np.clip(d, 0.0, 1000.0)
            for t in [0.1, 0.3, 0.5]:
                a_try = np.clip((1 - t) * a_opt + t * d, 0.0, 1000.0)
                sc_try = compute_score(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    print(f"  Restart #{restart_count}+LP: NEW BEST {sc_try:.8f}")
                    break

        restart_count += 1

    # ===================================================================
    # Phase 7: Final polish (2% time)
    # ===================================================================
    print("\n=== Phase 7: Final Polish ===")
    if time.monotonic() < deadline:
        sc, a_polished = peak_shave(best_a, iters=1500)
        if sc < best_score:
            best_score = sc
            best_a = a_polished.copy()
            print(f"  Final polish: NEW BEST {sc:.8f}")

    # ===================================================================
    # Final output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
