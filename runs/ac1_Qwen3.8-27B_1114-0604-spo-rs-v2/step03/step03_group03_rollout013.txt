import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def _solve_coord_1d(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper):
    """
    Solve the 1D subproblem for one coordinate.
    Returns optimal t.
    """
    # Quick check: if increasing t only increases the numerator (via affected)
    # and the denominator, we can sometimes determine the answer analytically.
    
    # Check if outside_max dominates throughout [0, t_upper]
    affected_max_at_upper = float(np.max(base_inside + t_upper * alpha + (t_upper * t_upper) * beta_coef))
    if affected_max_at_upper <= outside_max:
        # f(t) = outside_max / (S_rest + t)^2, minimized at t_upper
        return t_upper
    
    # Check t=0
    base_max = float(np.max(base_inside))
    f0 = 2.0 * n * max(base_max, outside_max) / (S_rest * S_rest) if S_rest > 0 else 1e10
    
    # Check t_upper
    f_up = 2.0 * n * max(affected_max_at_upper, outside_max) / ((S_rest + t_upper) ** 2)
    
    # If f at upper is much better, likely optimum is near upper
    # Use minimize_scalar for the general case
    res = minimize_scalar(
        _coord_1d_obj,
        args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
        bounds=(0.0, t_upper),
        method='bounded',
        options={'xatol': 1e-8}
    )
    return float(res.x)


def coordinate_descent_sweep(a, conv, order=None):
    """
    One full sweep of coordinate descent with incremental conv updates.
    Mutates a and conv in place. Returns nothing.
    """
    n = len(a)
    M_conv = 2 * n - 1

    if order is None:
        order = np.arange(n)

    for i in order:
        i = int(i)
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = float(a.sum()) - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        t_opt = _solve_coord_1d(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper)

        a[i] = t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef


def hot_spot_order(a, conv, n_focus=None):

def spectral_projection_step(a, n_iter=5, alpha=0.1):
    """
    Alternating projections: flatten spectrum + enforce non-negativity.
    This helps reduce peak autocorrelation by shaping the power spectrum.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    a_work = a.astype(np.float64).copy()
    S_orig = float(a_work.sum())
    
    for it in range(n_iter):
        # Compute spectrum
        A = np.fft.rfft(a_work, M_fft)
        C = A.real**2 + A.imag**2  # |A(f)|^2
        
        # Target: flatten C (reduce peaks in power spectrum)
        C_mean = C.mean()
        if C_mean < 1e-12:
            break
        
        # Damp the peaks: scale A down where |A|^2 is above mean
        # Use soft thresholding
        ratio = C / C_mean
        # Where ratio > 1, reduce; where ratio < 1, leave alone (or slightly boost)
        scale = np.ones_like(ratio)
        above = ratio > 1.0
        scale[above] = 1.0 - alpha * (ratio[above] - 1.0) / (ratio[above] - 1.0 + 1.0)
        # Simplified: scale = 1 - alpha * max(0, ratio - 1) / ratio
        # scale[above] = 1.0 / np.sqrt(ratio[above]) ** alpha  # softer
        
        # Use: scale = (1/(ratio))^alpha for ratio > 1, else 1
        # This pushes toward flat spectrum
        scale[above] = (1.0 / ratio[above]) ** alpha
        scale = np.clip(scale, 0.01, 1.0)
        
        # Apply scaling to A (preserving phase)
        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Enforce non-negativity (projection onto non-negative cone)
        a_new = np.maximum(a_new, 0.0)
        
        # Rescale to maintain sum
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a_new *= S_orig / S_new
        else:
            a_new = a_work.copy()
            break
        
        a_work = a_new
    
    return a_work


def targeted_spectral_perturb(a, conv, rng, strength=0.03):

    # Also consider second peak

    if len(c_sorted) > 1 and c_sorted[1] > 0.5 * c_sorted[0]:
        # Find second peak location
        conv_no0 = conv.copy()
        conv_no0[k_star] = -1
        k_star2 = int(np.argmax(conv_no0))
        contrib2 = np.zeros(n)
        j2_start = max(0, k_star2 - n + 1)

        contrib += 0.5 * contrib2
    
    # Top contributors get reduced, others get increased

    top_idx = np.argsort(contrib)[::-1][:top_count]

        per = total_reduced / len(bottom_idx)

    return np.clip(a_pert, 0.0, 1000.0)

    # We also need the self term: p[2i] * 2a[i] for the diagonal
    # Actually the FFT approach above gives sum_k p[k] * a[|k-i|] approximately
    # Let me be more careful: dc[k]/da[i] for fixed i is a[k-i] if 0<=k-i<n
    # So sum_k p[k]*dc[k]/da[i] = sum_k p[k]*a[k-i] (for valid k-i)
    # = (p reversed) convolved with a, evaluated at position i
    # conv_res[i] should give this
    
    # Add the diagonal contribution (when k=2i, dc[k]/da[i] = 2a[i] instead of a[i])
    # Actually the formula dc[k]/da[i] = a[k-i] already handles this correctly
    # since when k=2i, a[k-i] = a[i], and the total derivative is a[i] + a[i] = 2a[i]
    # But in the convolution sum, it appears only once... let me verify.
    # c[k] = sum_j a[j]*a[k-j]. dc[k]/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i, j=k-i)
    # If i != k-i: dc[k]/da[i] = a[k-i] + a[i]... wait no.
    # c[k] = sum_j a[j]*a[k-j]. The term involving a[i] is: a[i]*a[k-i] (when j=i) + a[k-i]*a[i] (when k-j=i, j=k-i)
    # If i != k-i: two terms, dc[k]/da[i] = a[k-i] + a[i]
    # If i == k-i (k=2i): one term a[i]^2, dc[k]/da[i] = 2*a[i]
    # So in general: dc[k]/da[i] = a[k-i] + (a[i] if i != k-i else a[i]) = a[k-i] + a[i] for i!=k-i
    # Wait that's not right either. Let me be very careful.
    # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # d(c[k])/d(a[i]) = sum over terms where a[i] appears:
    #   - a[i] appears as a[j] when j=i: contributes a[k-i] (if k-i is valid)
    #   - a[i] appears as a[k-j] when k-j=i, i.e., j=k-i: contributes a[k-i] (if k-i is valid)
    # If i != k-i: both terms exist, total = a[k-i] + a[k-i] = 2*a[k-i]
    # If i == k-i (k=2i): same term counted once, total = 2*a[i]
    # Actually when k=2i: j=i and k-j=i are the same, so it's a[i]*a[i], derivative = 2*a[i]
    # When k!=2i: j=i gives a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
    #            j=k-i gives a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
    #            total = 2*a[k-i]
    # So: dc[k]/da[i] = 2*a[k-i] (if k-i is in [0,n-1] and k != 2i)
    #                  = 2*a[i] (if k = 2i)
    #                  = a[k-i] (edge cases at boundaries)
    
    # The FFT computation above gives: sum_k p[k] * a[|k-i|] approximately
    # which is sum_k p[k] * a[k-i] for the valid range
    # This should be correct as an approximation
    
    g = 2.0 * conv_res  # factor of 2 from the derivative
    # But we might be missing the diagonal... let's add it
    # The FFT gives sum_k p[k]*a[k-i] which for k=2i gives p[2i]*a[i]
    # But we need 2*p[2i]*a[i] there. So we need to add p[2i]*a[i] for each i.
    # This is a sparse correction.
    # For now, the FFT approximation should be close enough.
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        # Chirp-based initialization
        n_init = 2048
        x = np.arange(n_init)
        chirp = np.cos(np.pi * x * (x + 1) / n_init)
        best_a = np.abs(chirp) + 0.1
        best_score = compute_score(best_a)
        print(f"Init score: {best_score:.8f}, length: {n_init}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # Precompute FFT size
    M_conv = 2 * n_work - 1
    M_fft = 1 << (M_conv - 1).bit_length()

    # ============================================================
    # Phase 1: Coordinate descent with hot-spot ordering (45% of time)
    # ============================================================
    phase1_deadline = t_start + 0.45 * total_budget
    print(f"\n=== Phase 1: CD with hot-spot priority ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    sweep_count = 0
    no_improve = 0

    while time.monotonic() < phase1_deadline:
        sweep_count += 1

        # Alternate orderings
        if sweep_count % 5 == 0:
            order, k_star = hot_spot_order(a_cur, conv_cur)
        elif sweep_count % 3 == 0:
            order = rng.permutation(n_work)
        else:
            order = np.arange(n_work)

        t0 = time.monotonic()
        coordinate_descent_sweep(a_cur, conv_cur, order=order)
        sweep_time = time.monotonic() - t0

        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 30 == 0:
                print(f"  Sweep {sweep_count}: {sc:.8f} (NEW BEST, {sweep_time:.2f}s/sweep)")
        else:
            no_improve += 1

        if sweep_count % 10 == 0:
            print(f"  Sweep {sweep_count}: {sc:.8f} (no_improve={no_improve}, {sweep_time:.2f}s)")

        # Perturb if stuck
        if no_improve >= 30:
            print(f"  Stuck at sweep {sweep_count}, perturbing...")
            a_cur = targeted_spectral_perturb(a_cur, conv_cur, rng, strength=0.02 + 0.04 * rng.random())
            conv_cur = fast_conv(a_cur)
            no_improve = 0

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: Spectral projection + CD (20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: Spectral Projection + CD ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    sp_count = 0

    while time.monotonic() < phase2_deadline:
        sp_count += 1

        # Spectral projection step
        a_sp = spectral_projection_step(a_cur, n_iter=3, alpha=0.08 + 0.04 * rng.random())
        a_sp = np.clip(a_sp, 0.0, 1000.0)

        # Follow up with a few CD sweeps
        conv_sp = fast_conv(a_sp)
        for sw in range(3):
            if time.monotonic() > phase2_deadline:
                break
            order, _ = hot_spot_order(a_sp, conv_sp)
            coordinate_descent_sweep(a_sp, conv_sp, order=order)

        sc_sp = compute_score(a_sp)
        if sc_sp < best_score:
            best_score = sc_sp
            best_a = a_sp.copy()
            a_cur = a_sp.copy()
            conv_cur = fast_conv(a_cur)
            print(f"  SP iter {sp_count}: {sc_sp:.8f} (NEW BEST)")
        elif sp_count % 5 == 0:
            print(f"  SP iter {sp_count}: {sc_sp:.8f}")

    print(f"Phase 2 done: {sp_count} iterations, best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE + L-BFGS (15% of time)
    # ============================================================
    phase3_deadline = t_start + 0.80 * total_budget
    print(f"\n=== Phase 3: LSE + L-BFGS ({phase3_deadline - t_start:.0f}s) ===")

    n_lbfgs = n_work
    bounds = [(0.0, 1000.0)] * n_lbfgs

    for beta in [200, 800, 3000, 8000]:
        if time.monotonic() > phase3_deadline - 30:
            break
        t0 = time.monotonic()
        try:
            res = minimize(
                lse_objective, best_a, args=(beta,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-14, 'gtol': 1e-12,
                         'maxiter': 3000, 'maxfun': 60000,
                         'maxls': 50}
            )
            a_opt = np.clip(res.x, 0.0, 1000.0)
            sc_opt = compute_score(a_opt)
            elapsed = time.monotonic() - t0
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"  LSE beta={beta}: {sc_opt:.8f} (NEW BEST, {elapsed:.1f}s)")
            else:
                print(f"  LSE beta={beta}: {sc_opt:.8f} ({elapsed:.1f}s)")
        except Exception as e:
            print(f"  LSE beta={beta} failed: {e}")

    # CD polish after LSE
    if time.monotonic() < phase3_deadline - 10:
        a_cur = best_a.copy()
        conv_cur = fast_conv(a_cur)
        for _ in range(5):
            if time.monotonic() > phase3_deadline - 5:
                break
            order, _ = hot_spot_order(a_cur, conv_cur)
            coordinate_descent_sweep(a_cur, conv_cur, order=order)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Post-LSE CD: {sc:.8f} (NEW BEST)")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Exploration with diverse perturbations (10% of time)
    # ============================================================
    phase4_deadline = t_start + 0.90 * total_budget
    print(f"\n=== Phase 4: Exploration ===")

    trial = 0
    while time.monotonic() < phase4_deadline - 10:
        trial += 1
        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            # Sinusoidal perturbation at peak-related frequency
            conv_tmp = fast_conv(best_a)
            k_star = int(np.argmax(conv_tmp))
            freq = k_star / (2.0 * n_cur)
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_cur, dtype=np.float64)
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 2:
            # Targeted spectral perturbation
            conv_tmp = fast_conv(best_a)
            a_pert = targeted_spectral_perturb(best_a, conv_tmp, rng, strength=0.02 + 0.05 * rng.random())
        elif strategy == 3:
            # Block shuffle
            seg_len = max(20, n_cur // 20)
            num_segs = n_cur // seg_len
            new_order = rng.permutation(num_segs)
            a_pert = np.empty(n_cur)
            for new_i in range(num_segs):
                old_i = new_order[new_i]
                a_pert[new_i * seg_len:(new_i + 1) * seg_len] = best_a[old_i * seg_len:(old_i + 1) * seg_len]
            a_pert[n_cur - seg_len:] = best_a[n_cur - seg_len:]
        elif strategy == 4:
            # Spectral projection with strong alpha
            a_pert = spectral_projection_step(best_a, n_iter=10, alpha=0.2)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Random restart of a random block
            block_size = max(50, n_cur // 10)
            start = int(rng.integers(0, n_cur - block_size))
            a_pert = best_a.copy()
            # Replace block with smoothed version of neighbors
            left = max(0, start - block_size // 2)
            right = min(n_cur, start + block_size * 3 // 2)
            if right > left + 1:
                a_pert[start:start + block_size] = np.interp(
                    np.linspace(0, right - left - 1, block_size),
                    np.arange(right - left),
                    a_pert[left:right]
                )

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD from perturbed point
        conv_pert = fast_conv(a_pert)
        n_cd = 4
        for sw in range(n_cd):
            if time.monotonic() > phase4_deadline - 10:
                break
            if sw % 2 == 0:
                order, _ = hot_spot_order(a_pert, conv_pert)
            else:
                order = rng.permutation(n_cur)
            coordinate_descent_sweep(a_pert, conv_pert, order=order)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 3 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ============================================================
    # Final polish (10% of time)
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()
    conv_cur = fast_conv(a_cur)
    polish_count = 0

    while time.monotonic() < deadline - 5:
        polish_count += 1
        if polish_count % 3 == 0:
            order, _ = hot_spot_order(a_cur, conv_cur)
        else:
            order = np.arange(n_work)
        coordinate_descent_sweep(a_cur, conv_cur, order=order)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if polish_count % 10 == 0:
                print(f"  Polish {polish_count}: {sc:.8f} (NEW BEST)")

    # One final LSE attempt if time permits
    if time.monotonic() < deadline - 30:
        try:
            res = minimize(
                lse_objective, best_a, args=(5000,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-15, 'gtol': 1e-13,
                         'maxiter': 2000, 'maxfun': 40000,
                         'maxls': 30}
            )
            a_opt = np.clip(res.x, 0.0, 1000.0)
            sc_opt = compute_score(a_opt)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"  Final LSE: {sc_opt:.8f} (NEW BEST)")
            # CD after
            a_cur = best_a.copy()
            conv_cur = fast_conv(a_cur)
            coordinate_descent_sweep(a_cur, conv_cur, order=np.arange(n_work))
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} (NEW BEST)")
        except Exception as e:
            print(f"  Final LSE failed: {e}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
