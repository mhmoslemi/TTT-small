    log_sum = np.log(exp_scaled.sum()) + scaled.max()

    p = exp_scaled / exp_scaled.sum()

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x, b):
        return lse_score_and_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_and_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(50, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(50, int(remaining * 40)))

    try:
        res = minimize(obj, a0, args=(beta,), method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def softmax_weights(c, beta):
    """Compute softmax weights on the autocorrelation values."""
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = scaled - scaled.max()  # for numerical stability
    w = np.exp(scaled)
    w /= w.sum()
    return w


def multiplicative_equalize(a, n_iters=100, alpha_start=0.5, alpha_end=0.1, beta_schedule=None, deadline=None):
    """
    LP-dual-inspired multiplicative update with cooling schedule.
    Uses softmax weights instead of hard thresholding.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    if beta_schedule is None:
        # Default: increasing beta (sharper weights)
        beta_schedule = np.linspace(10, 200, n_iters)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        beta = beta_schedule[min(it, len(beta_schedule)-1)]
        
        # Smooth weights (softmax)
        w = softmax_weights(c, beta)
        
        # Compute h[j] = 2 * sum_k w[k] * a[k-j]
        h = weighted_conv_gradient(a, w)
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        # Cooling: decrease alpha over iterations
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(n_iters - 1, 1))
        
        # Multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

        # Compute "error" at each lag: e[k] = c[k] - c_mean
        # We want to reduce the positive errors (above mean)
        # Gradient of sum_k max(0, c[k] - c_mean)^2 w.r.t. a:
        # = 2 * sum_{k: c[k]>c_mean} (c[k]-c_mean) * d(c[k])/da
        # = 2 * weighted_conv_gradient with weights = max(0, c - c_mean)
        
        excess = np.maximum(c - c_mean, 0)
        if excess.sum() < 1e-15:
            break
        excess = excess / excess.sum()  # normalize to sum to 1
        
        h = weighted_conv_gradient(a, excess)
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        alpha = 0.3
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a


def block_coordinate_descent(a, block_size=128, n_rounds=5, deadline=None):
    """
    Optimize blocks of variables while keeping the rest fixed.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n_blocks = (n + block_size - 1) // block_size
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for b in range(n_blocks):
            if deadline is not None and time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            n_block = len(block)
            
            def block_obj(x, s, e, full_a):
                full_a[s:e] = x
                sc = compute_score(full_a)
                # Compute gradient w.r.t. block only
                c = fast_conv(full_a)
                S = full_a.sum()
                k_star = int(np.argmax(c))
                M = c[k_star]
                # d(score)/da_j for j in [s,e)
                g = np.zeros(n)
                j_start = max(0, k_star - n + 1)
                j_end = min(k_star, n - 1)
                if j_end >= j_start:
                    g[j_start:j_end+1] = 2.0 * full_a[j_start:j_end+1][::-1]
                grad = (2.0 * n) * (g / (S*S) - 2.0 * M / (S**3))
                return sc, grad[start:end]
            
            # Use L-BFGS on the block
            try:
                res = minimize(
                    block_obj, block, args=(start, end, a),
                    method='L-BFGS-B', jac=True,
                    bounds=[(0.0, 1000.0)] * n_block,
                    options={'maxiter': 200, 'maxfun': 2000, 'gtol': 1e-8}
                )
                new_block = np.clip(res.x, 0.0, 1000.0)
                a[start:end] = new_block
            except Exception:
                pass
            
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_sc, best_a


def subgradient_polish(a, n_iters=200, deadline=None):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max at k_star
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg_M[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size (lower bound is 1.0)
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 1.0)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Small fixed step
        a -= 0.001 * grad / (gn + 1e-15)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a
