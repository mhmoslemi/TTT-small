"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE-continuation, top-k smoothing, multi-peak subgradient, and random restarts.
Improvements: adaptive continuation schedule, multi-peak subgradient, diverse restarts,
length exploration, and iterative refinement with exact objective polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and its gradient w.r.t. x (assumed non-negative)."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. x_j = 2 * (x_rev * p)[n-1+j]
    x_rev = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(x_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L * x / S3)
    return score, grad


def topk_obj_grad(x, k, sharpness):
    """Soft top-k convolution score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[min(k, len(c_sorted) - 1)]
        gap = max(c_sorted[min(k + 2, len(c_sorted) - 1)] - threshold, 1e-8)
        width = gap * 0.3 + 1e-8
    else:
        threshold = c_sorted[-1]
        width = 1.0
    w = 1.0 / (1.0 + np.exp(-sharpness * (c - threshold) / width))
    w_sum = float(w.sum())
    if w_sum < 1e-15:
        w = np.ones(len(c)) / len(c)
        w_sum = len(c)
    w = w / w_sum
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    x_rev = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(x_rev, M2_fft)
    P = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g_L / S2 - 2.0 * L * x / S3)
    return score, grad


def make_lse_func(beta):
    """Create a function handle for LSE objective (no closure needed, uses global)."""
    return beta  # We'll just pass beta directly


def lbfgs_lse(x0, beta, maxiter, maxfun):
    """Run L-BFGS-B with LSE objective."""
    n = len(x0)
    bounds = [(0.0, 1000.0)] * n

    def obj_and_jac(x):
        return lse_obj_grad(np.maximum(x, 0.0), beta)

    res = minimize(
        obj_and_jac, x0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = np.maximum(x0, 0.0).copy()
    sc = compute_score(a_out)
    return sc, a_out


def lbfgs_topk(x0, k, sharpness, maxiter, maxfun):
    """Run L-BFGS-B with top-k objective."""
    n = len(x0)
    bounds = [(0.0, 1000.0)] * n

    def obj_and_jac(x):
        return topk_obj_grad(np.maximum(x, 0.0), k, sharpness)

    res = minimize(
        obj_and_jac, x0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = np.maximum(x0, 0.0).copy()
    sc = compute_score(a_out)
    return sc, a_out


def multi_peak_subgrad(a, k, step_frac, n_iters):
    """Subgradient descent using top-k peaks of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lower_bound = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        S2 = S * S
        M = float(c.max())
        sc = 2.0 * n * M / S2

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k peaks
        top_k_idx = np.argsort(c)[::-1][:k]
        # Average subgradient over top-k peaks
        sg = np.zeros(n)
        for k_star in top_k_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1][::-1]
                sg[j_start:j_end + 1] += 2.0 * seg
        sg /= k
        # Full gradient of score
        grad = 2.0 * n * (sg / S2 - 2.0 * M * a / (S2 * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, step_frac, 0.5)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 1e-6:
            break
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Halve the step
            step /= 2.0
            a_new = a - step * grad
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() >= 1e-6:
                sc_new2 = compute_score(a_new)
                if sc_new2 < sc:
                    a = a_new
            else:
                a = a - (step / 2.0) * grad
                a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_subgrad_step(a, k, n_sub_iters):
    """Targeted subgradient: reduce elements contributing to top peaks, increase others."""
    n = len(a)
    a_work = a.copy()
    best_a = a_work.copy()
    best_sc = compute_score(a_work)

    for it in range(n_sub_iters):
        c = fast_conv(a_work)
        S = float(a_work.sum())
        S2 = S * S
        M = float(c.max())
        sc = 2.0 * n * M / S2
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()

        top_k_idx = np.argsort(c)[::-1][:k]
        # Find elements contributing most to these peaks
        contrib = np.zeros(n)
        for k_star in top_k_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] += a_work[j_start:j_end + 1][::-1]

        # Reduce top contributors, increase bottom
        top_count = max(5, n // 30)
        bottom_count = max(10, n // 10)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_idx = np.argsort(contrib)[:bottom_count]

        reduce_frac = 0.01 + 0.02 * (1.0 - it / max(n_sub_iters, 1))
        total_red = 0.0
        for j in top_idx:
            amt = a_work[j] * reduce_frac
            a_work[j] -= amt
            a_work[j] = max(0.0, a_work[j])
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            for j in bottom_idx:
                a_work[j] += per
        a_work = np.clip(a_work, 0.0, 1000.0)

    return best_sc, best_a


def build_inits(n, rng, parent_arr):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent_arr is not None:
        if len(parent_arr) == n:
            inits.append(parent_arr.copy())
        else:
            idx = np.linspace(0, len(parent_arr) - 1, n).astype(int)
            inits.append(parent_arr[idx].copy())
        # Perturbed parent
        p = parent_arr.copy()
        if len(p) != n:
            idx = np.linspace(0, len(p) - 1, n).astype(int)
            p = p[idx].copy()
        inits.append(np.clip(p * (1.0 + 0.05 * rng.standard_normal(n)), 0.01, 1000.0))
        inits.append(np.clip(p * (1.0 + 0.1 * rng.standard_normal(n)), 0.01, 1000.0))

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x + 0.5))
    inits.append(0.3 + 0.5 * np.cos(np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    # Multi-frequency
    inits.append(0.5 + 0.15 * np.cos(2*np.pi*1*x) + 0.1*np.cos(2*np.pi*3*x+1) + 0.05*np.cos(2*np.pi*7*x+2))

            # Targeted: reduce top contributors to peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if j_end >= j_start:
                seg_start = j_start
                seg_end = j_end
                contrib = a_pert[seg_start:seg_end + 1][::-1]
                top_idx = seg_start + np.argsort(contrib)[::-1][:max(5, n_best // 50)]
                for j in top_idx:
                    a_pert[j] *= 0.95
                a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase_ph = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            xx = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * xx + phase_ph))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Sparse modification
            n_sparse = max(20, n_best // 30)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Block swap
            block_sz = max(10, n_best // 25)
            b1 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            b2 = int(rng.integers(0, max(1, n_best - 2 * block_sz)))
            a_pert = best_a.copy()
            tmp = a_pert[b1:b1 + block_sz].copy()
            a_pert[b1:b1 + block_sz] = a_pert[b2:b2 + block_sz]
            a_pert[b2:b2 + block_sz] = tmp
        elif strategy == 5:
            # Mix with structured
            xx = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * (1 + 4 * rng.random()) * xx + rng.random() * 6.28)
            a_struct = np.clip(a_struct, 0.01, 1000.0)
            a_struct = a_struct * (best_a.sum() / a_struct.sum())
            a_pert = 0.8 * best_a + 0.2 * a_struct
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Amplify low regions
            a_pert = best_a.copy()
            median_val = float(np.median(a_pert))
            low_mask = a_pert < median_val * 0.7
            a_pert[low_mask] *= 1.1
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick LSE polish
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - 15 - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(4000, max(200, int(remaining / 5.0) * 30))
        sc, a_opt = lbfgs_lse(a_pert, 1000, maxiter, maxiter * 2)
        if sc < best_score:
            report(sc, a_opt, f"P4 restart #{trial} strat={strategy}")
        trial += 1

    print(f"After Phase 4: best = {best_score:.8f}")

    # =========================================================================
    # Phase 5: Final high-precision polish
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")

    for beta in [5000, 10000, 20000]:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - 5 - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(10000, max(1000, int(remaining / 3.0) * 50))
        sc, a_opt = lbfgs_lse(best_a, beta, maxiter, maxiter * 2)
        if sc < best_score:
            report(sc, a_opt, f"P5 LBFGS beta={beta}")
        best_a = a_opt

    # Final subgradient
    if time.monotonic() < deadline - 3:
        sc, a_opt = multi_peak_subgrad(best_a, 3, 0.2, 500)
        if sc < best_score:
            report(sc, a_opt, "P5 final subgrad")
        best_a = a_opt

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 256

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
