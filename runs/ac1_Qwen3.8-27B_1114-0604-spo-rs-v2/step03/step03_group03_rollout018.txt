```python
def greedy_lp(f, M, n):
    """Solve max sum(g) s.t. (f*g)(k) <= M, g >= 0 using greedy coordinate ascent."""
    g = np.zeros(n)
    conv_fg = np.zeros(2*n-1)  # (f*g)(k)
    
    # Process coordinates in order of smallest "bottleneck"
    # For coordinate j, max g[j] = min over k of (M - conv_fg[k]) / f[k-j]
    # where k-j is in [0, n-1]
    
    for j in range(n):
        # For each k where f[k-j] > 0, the constraint is:
        # conv_fg[k] + g[j] * f[k-j] <= M
        # So g[j] <= (M - conv_fg[k]) / f[k-j]
        
        k_lo = j
        k_hi = min(2*n-2, j+n-1)
        ks = np.arange(k_lo, k_hi+1)
        f_vals = f[ks - j]  # f[k-j] for k in [k_lo, k_hi]
        valid = f_vals > 1e-12
        if not np.any(valid):
            continue
        max_gj = (M - conv_fg[ks[valid]]) / f_vals[valid]
        max_gj = np.clip(max_gj, 0, None)
        g[j] = np.min(max_gj)
        conv_fg[k_lo:k_hi+1] += g[j] * f[ks-j]
    
    return g
```

```python
def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def greedy_lp(f, M, n, order=None):
    """
    Greedy LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns the greedy solution g.
    """
    g = np.zeros(n)
    conv_fg = np.zeros(2 * n - 1)
    
    if order is None:
        order = np.arange(n)
    
    for j in order:
        j = int(j)
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        f_vals = f[np.arange(k_lo, k_hi + 1) - j]
        valid = f_vals > 1e-12
        if not np.any(valid):
            continue
        available = M - conv_fg[np.arange(k_lo, k_hi + 1)][valid]
        max_gj = np.min(available / f_vals[valid])
        max_gj = max(0.0, max_gj)
        if max_gj > 0:
            g[j] = max_gj
            conv_fg[k_lo:k_hi + 1] += max_gj * f_vals
    
    return g, conv_fg


def frank_wolfe_step(f, rng, n_max_greedy_passes=3):
    """
    One Frank-Wolfe step: find direction g via greedy LP, then line search.
    Returns (new_f, score_improved).
    """
    n = len(f)
    conv_f = fast_conv(f)
    M = float(np.max(conv_f))
    S_f = float(f.sum())
    
    # Solve greedy LP with multiple orderings
    best_g = None
    best_sum = -1
    
    orders = [np.arange(n), np.arange(n-1, -1, -1)]
    for _ in range(n_max_greedy_passes - 2):
        orders.append(rng.permutation(n))
    
    for order in orders:
        g, _ = greedy_lp(f, M, n, order)
        s = float(g.sum())
        if s > best_sum:
            best_sum = s
            best_g = g
    
    if best_sum <= 0:
        return f.copy(), False
    
    # Normalize g to have same sum as f
    g = best_g * (S_f / best_sum)
    
    # Line search
    best_t = 0.0
    best_score = compute_score(f)
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1 - t) * f + t * g
        sc = compute_score(f_cand)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    if best_t == 0:
        return f.copy(), False
    return (1 - best_t) * f + best_t * g, True


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]
        
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)
        
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a
```

```python
"""
Frank-Wolfe + Coordinate Descent hybrid optimizer for minimum peak autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation: Uses a greedy LP solver to compute Frank-Wolfe descent directions
that simultaneously consider all coordinates, combined with exact coordinate descent
for local refinement. The LP direction maximizes sum(g) subject to (f*g)(k) <= M,
which directly attacks the structure of the objective function.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def greedy_lp(f, M, n, order):
    """
    Greedy LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Processes coordinates in given order, setting each to max feasible value.
    Returns (g, sum_g).
    """
    g = np.zeros(n)
    conv_fg = np.zeros(2 * n - 1)

    for j in order:
        j = int(j)
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        idx = np.arange(k_lo, k_hi + 1) - j
        f_vals = f[idx]
        valid = f_vals > 1e-12
        if not np.any(valid):
            continue
        idx_valid = idx[valid]
        f_valid = f_vals[valid]
        available = M - conv_fg[idx_valid]
        max_gj = float(np.min(available / f_valid))
        max_gj = max(0.0, max_gj)
        if max_gj > 1e-15:
            g[j] = max_gj
            conv_fg[idx] += max_gj * f_vals

    return g, float(g.sum())


def frank_wolfe_step(f, rng, n_passes=3):
    """
    One Frank-Wolfe step: find direction via greedy LP (multiple orderings),
    then line search on convex combination.
    Returns (new_f, score, improved).
    """
    n = len(f)
    conv_f = fast_conv(f)
    M = float(np.max(conv_f))
    S_f = float(f.sum())

    if S_f < 1e-12 or M <= 0:
        return f.copy(), compute_score(f), False

    # Try multiple orderings
    best_g = None
    best_sum = -1.0

    orders = [np.arange(n), np.arange(n - 1, -1, -1)]
    for _ in range(max(0, n_passes - 2)):
        orders.append(rng.permutation(n))

    # Also try order by f values (ascending = small f first, leaves room for large f later)
    orders.append(np.argsort(f))

    for order in orders:
        g, s = greedy_lp(f, M, n, order)
        if s > best_sum:
            best_sum = s
            best_g = g

    if best_sum <= 1e-12:
        return f.copy(), compute_score(f), False

    # Normalize g to have same sum as f
    g = best_g * (S_f / best_sum)

    # Line search: try several t values
    current_score = compute_score(f)
    best_t = 0.0
    best_sc = current_score
    best_f = f.copy()

    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t in t_candidates:
        f_cand = (1.0 - t) * f + t * g
        if f_cand.sum() < 1e-12:
            continue
        sc = compute_score(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
            best_f = f_cand

    if best_t == 0.0:
        return f.copy(), current_score, False
    return best_f, best_sc, True


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None):
    """One full sweep of exact coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(len(ks))
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def hot_coordinate_sweep(a, conv=None):
    """Coordinate descent prioritizing coordinates that contribute to the peak."""
    n = len(a)
    if conv is None:
        conv = fast_conv(a)
    a = a.astype(np.float64).copy()
    conv = conv.copy()
    S = float(a.sum())
    M_conv = 2 * n - 1

    k_star = int(np.argmax(conv))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    order = np.argsort(contrib)[::-1]

    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(len(ks))
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def fw_cd_block(a, rng, n_fw=5, n_cd=3, deadline=None):
    """Alternating FW and CD block."""
    for i in range(n_fw):
        if deadline is not None and time.monotonic() > deadline:
            break
        a, sc, improved = frank_wolfe_step(a, rng, n_passes=3)
    for i in range(n_cd):
        if deadline is not None and time.monotonic() > deadline:
            break
        sc, a = coordinate_descent_sweep(a, rng=rng)
    return a
```
