"""
Spectral water-filling optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses iterative peak clipping in time domain with phase-preserving spectral updates,
LP-based refinement, and L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_waterfill(a, n_iters=200, start_lambda=0.1, end_lambda=0.9, 
                       use_phase=True, verbose=False):
    """
    Iterative spectral water-filling:
    1. Compute convolution c
    2. Clip peak: c_target = min(c, threshold)
    3. Compute P_target = DFT(c_target_padded)
    4. Update: A_new = sqrt(max(P_target,0)) * phase(A)
    5. a_new = IFFT(A_new), project to non-negative, renormalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S  # normalize to unit sum
    
    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Interpolate lambda
        frac = it / max(1, n_iters - 1)
        lam = start_lambda + (end_lambda - start_lambda) * frac
        
        # Compute FFT
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Compute convolution
        P = np.abs(A) ** 2
        c_pad = np.fft.ifft(P).real
        c_lin = c_pad[:2 * n - 1]
        
        c_max = float(c_lin.max())
        c_mean = float(c_lin.sum()) / (2 * n - 1)
        
        if c_max - c_mean < 1e-12:
            break
        
        # Water-fill: clip at threshold
        threshold = c_mean + lam * (c_max - c_mean)
        c_target = np.minimum(c_lin, threshold)
        
        # Compute target power spectrum
        ct_pad = np.zeros(M_fft, dtype=np.complex128)
        ct_pad[:2 * n - 1] = c_target
        P_target = np.fft.fft(ct_pad).real  # should be real if c is symmetric
        
        # Handle small imaginary part
        # P_target = np.fft.fft(ct_pad)  # keep complex for safety
        P_target_real = np.fft.fft(ct_pad).real
        
        # Project to non-negative
        P_target_pos = np.maximum(P_target_real, 0.0)
        
        # Check: if too much was zeroed out, reduce clipping
        frac_zero = np.sum(P_target_pos < 1e-10) / M_fft
        if frac_zero > 0.5:
            # Too aggressive, use less clipping
            threshold = c_max - 0.01 * (c_max - c_mean)
            c_target = np.minimum(c_lin, threshold)
            ct_pad = np.zeros(M_fft, dtype=np.complex128)
            ct_pad[:2 * n - 1] = c_target
            P_target_real = np.fft.fft(ct_pad).real
            P_target_pos = np.maximum(P_target_real, 0.0)
            frac_zero = np.sum(P_target_pos < 1e-10) / M_fft
            if frac_zero > 0.5:
                # Give up on this iteration
                continue
        
        # Phase-preserving update
        A_phase = A / (np.abs(A) + 1e-30)  # unit phase
        A_mag_new = np.sqrt(P_target_pos)
        
        if use_phase:
            A_new = A_mag_new * A_phase
        else:
            A_new = A_mag_new + 0j  # zero phase
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Blending for stability
        if it < n_iters // 4:
            a = 0.5 * a + 0.5 * a_new
        else:
            a = a_new
        
        if verbose and it % 20 == 0:
            print(f"    iter {it}: score={best_sc:.8f}, c_max={c_max:.4f}, c_mean={c_mean:.4f}")
    
    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_refine(a, max_time=60.0):
    """
    LP-based refinement from the SoTA description.
    Given f with peak M, find g maximizing sum(g) subject to conv(f,g) <= M, g >= 0.
    Then blend.
    """
    n = len(a)
    if n > 4096:
        # Too large for LP, skip
        return a
    
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    c = fast_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Build LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0
    # conv(a,g)[k] = sum_i a[i] * g[k-i] (with appropriate bounds)
    # This is a sparse LP with 2n-1 constraints and n variables
    
    from scipy.sparse import csr_matrix
    
    # Build constraint matrix A_lp where A_lp @ g <= M * ones
    # A_lp[k, j] = a[j] if 0 <= k-j <= n-1 and 0 <= j <= n-1, else 0
    # i.e., A_lp[k, j] = a[j] if max(0, k-(n-1)) <= j <= min(k, n-1)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A_lp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max c^T g s.t. A_lp @ g <= M * 1, 0 <= g <= 1000
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    t_start = time.monotonic()
    try:
        res = linprog(
            c_obj,
            A_ub=A_lp,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, 1000)] * n,
            method='highs',
            options={'time_limit': max_time}
        )
    except Exception as e:
        print(f"    LP failed: {e}")
        return best_a
    
    if time.monotonic() - t_start > max_time:
        return best_a
    
    if res.status == 0:
        g = res.x
        sum_g = float(g.sum())
        if sum_g > 1e-12:
            g = g / sum_g  # normalize
            # Blend
            for t_mix in [0.3, 0.5, 0.7, 1.0]:
                a_blend = (1 - t_mix) * a + t_mix * g
                a_blend = np.maximum(a_blend, 0.0)
                S_b = float(a_blend.sum())
                if S_b < 1e-12:
                    continue
                a_blend = a_blend / S_b
                sc = compute_score(a_blend)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_blend.copy()
    
    return best_a


def multiplicative_update(a, n_iters=100, verbose=False):
    """
    Multiplicative update inspired by NMF / EM.
    For each coordinate, update based on ratio of "expected" to "observed" contribution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2 * n - 2).bit_length()
    
    for it in range(n_iters):
        A = np.fft.fft(a, M_fft)
        P = np.abs(A) ** 2
        c_pad = np.fft.ifft(P).real
        c_lin = c_pad[:2 * n - 1]
        
        c_max = float(c_lin.max())
        c_mean = float(c_lin.sum()) / (2 * n - 1)
        
        # Target: flat convolution at level c_mean
        # Update rule: a[i] *= (target / observed_contribution)^alpha
        # observed contribution of a[i] to max(c) is proportional to the peak-row correlation
        
        # Find peak
        k_star = int(np.argmax(c_lin))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # "Responsibility" of each a[j] for the peak
        resp = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            resp[j_start:j_end + 1] = seg[::-1]
        
        # Multiplicative update: reduce elements that contribute to peak
        # a[i] *= (c_mean / c_max)^alpha * (1 + beta * (1 - resp[i]/max_resp))
        ratio = c_mean / max(c_max, 1e-12)
        alpha = 0.1 + 0.1 * (it / n_iters)  # increasing aggressiveness
        max_resp = max(float(resp.max()), 1e-12)
        resp_norm = resp / max_resp
        
        a_new = a * (ratio ** alpha) * (1.0 + 0.5 * (1.0 - resp_norm))
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # U-shaped (counteracts triangular conv)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.2 + 0.6 * np.abs(2.0 * x - 1.0))
    # Periodic
    for freq in [2, 3, 5]:
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x))
    # Random blocks
    for nb in [8, 16, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_blk = np.repeat(vals, per)
        if len(a_blk) < n:
            a_blk = np.concatenate([a_blk, vals[:n - len(a_blk)]])
        inits.append(a_blk[:n].copy())
    return inits

"""
Spectral water-filling optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: iterative peak clipping in time domain with phase-preserving
power spectrum updates, combined with adaptive soft-clipping, LP refinement,
and multi-scale local search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def spectral_waterfill(a, n_iters=300, start_lam=0.05, end_lam=0.95,
                       use_phase=True, soft_power=1.0, blend_schedule='ramp',
                       verbose=False):
    """
    Iterative spectral water-filling with soft clipping.
    1. Compute convolution c
    2. Soft-clip: c_target = c - alpha*max(0, c-thresh)^p
    3. P_target = DFT(c_target_padded), project to non-negative
    4. A_new = sqrt(P_target) * phase(A)
    5. a_new = IFFT(A_new), project non-negative, renormalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        lam = start_lam + (end_lam - start_lam) * frac

        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        P = np.abs(A) ** 2
        c_pad = np.fft.ifft(P).real
        c_lin = c_pad[:2 * n - 1]

        c_max = float(c_lin.max())
        c_mean = float(c_lin.sum()) / (2 * n - 1)

        if c_max - c_mean < 1e-12:
            break

        threshold = c_mean + lam * (c_max - c_mean)

        if soft_power == 1.0:
            c_target = np.minimum(c_lin, threshold)
        else:
            excess = np.maximum(c_lin - threshold, 0.0)
            c_target = c_lin - np.power(excess, soft_power)
            c_target = np.maximum(c_target, 0.0)

        ct_pad = np.zeros(M_fft, dtype=np.complex128)
        ct_pad[:2 * n - 1] = c_target
        P_target = np.fft.fft(ct_pad)
        P_target_real = P_target.real

        P_target_pos = np.maximum(P_target_real, 0.0)

        # Check quality: fraction of spectrum zeroed
        total_power = float(P_target_pos.sum())
        if total_power < 0.1 * float(P.sum()):
            # Too destructive, use milder clipping
            threshold = c_max - 0.02 * (c_max - c_mean)
            c_target = np.minimum(c_lin, threshold)
            ct_pad = np.zeros(M_fft, dtype=np.complex128)
            ct_pad[:2 * n - 1] = c_target
            P_target_real = np.fft.fft(ct_pad).real
            P_target_pos = np.maximum(P_target_real, 0.0)
            total_power = float(P_target_pos.sum())
            if total_power < 0.05 * float(P.sum()):
                continue

        A_mag_new = np.sqrt(P_target_pos)
        if use_phase:
            A_abs = np.abs(A)
            A_phase = np.where(A_abs > 1e-30, A / (A_abs + 1e-30), 1.0 + 0j)
            A_new = A_mag_new * A_phase
        else:
            A_new = A_mag_new + 0j

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Blending
        if blend_schedule == 'ramp':
            if it < n_iters // 3:
                alpha_blend = 0.5
            elif it < 2 * n_iters // 3:
                alpha_blend = 0.7
            else:
                alpha_blend = 1.0
        elif blend_schedule == 'aggressive':
            alpha_blend = 1.0
        else:
            alpha_blend = 0.8

        a = (1.0 - alpha_blend) * a + alpha_blend * a_new
        a = np.maximum(a, 0.0)
        S_a = float(a.sum())
        if S_a < 1e-12:
            break
        a = a / S_a

        if verbose and it % 50 == 0:
            print(f"    WF iter {it}: best={best_sc:.8f}, c_ratio={c_max/c_mean:.4f}")

    return best_a


def frequency_shaping(a, n_iters=200, verbose=False):
    """
    Direct frequency-domain shaping: reduce power at frequencies
    that constructively interfere at the peak location.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        P = np.abs(A) ** 2
        c_pad = np.fft.ifft(P).real
        c_lin = c_pad[:2 * n - 1]

        k_star = int(np.argmax(c_lin))
        c_max = c_lin[k_star]
        c_mean = float(c_lin.sum()) / (2 * n - 1)

        if c_max - c_mean < 1e-12:
            break

        # Influence of each frequency on c[k_star]
        k_freq = np.arange(M_fft)
        phase = np.exp(2j * np.pi * k_freq * k_star / M_fft)
        # Re(phase) tells us how much each freq contributes to c[k_star]
        influence = np.real(phase)

        # Normalize influence to [-1, 1]
        inf_max = float(np.max(np.abs(influence)))
        if inf_max < 1e-10:
            break
        influence = influence / inf_max

        # Reduce power where influence is positive, increase where negative
        alpha_step = 0.05 + 0.15 * (it / max(1, n_iters - 1))
        P_new = P * (1.0 - alpha_step * influence)
        P_new = np.maximum(P_new, 0.0)

        # Renormalize power to maintain sum
        P_scale = float(P.sum()) / max(float(P_new.sum()), 1e-30)
        P_new = P_new * P_scale

        A_mag = np.sqrt(P_new)
        A_abs = np.abs(A)
        A_phase = np.where(A_abs > 1e-30, A / (A_abs + 1e-30), 1.0 + 0j)
        A_new = A_mag * A_phase

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Blend
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0.0)
        S_a = float(a.sum())
        if S_a < 1e-12:
            break
        a = a / S_a

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def lp_refine(a, max_time=60.0):
    """
    LP-based refinement: find g maximizing sum(g) s.t. conv(f,g)<=M, g>=0.
    Then blend with f.
    """
    n = len(a)
    if n > 4096:
        return a

    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    c = fast_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_sc = compute_score(a)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])

    A_lp = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)

    t_start = time.monotonic()
    try:
        res = linprog(
            c_obj, A_ub=A_lp, b_ub=np.full(2 * n - 1, M),
            bounds=[(0, 1000)] * n, method='highs',
            options={'time_limit': max_time}
        )
    except Exception as e:
        print(f"    LP failed: {e}")
        return best_a

    if time.monotonic() - t_start > max_time or res.status != 0:
        return best_a

    g = res.x
    sum_g = float(g.sum())
    if sum_g < 1e-12:
        return best_a
    g = g / sum_g

    for t_mix in [0.3, 0.5, 0.7, 1.0]:
        a_blend = (1 - t_mix) * a + t_mix * g
        a_blend = np.maximum(a_blend, 0.0)
        S_b = float(a_blend.sum())
        if S_b < 1e-12:
            continue
        a_blend = a_blend / S_b
        sc = compute_score(a_blend)
        if sc < best_sc:
            best_sc = sc
            best_a = a_blend.copy()

    return best_a


def peak_chasing(a, n_iters=500, verbose=False):
    """
    Iterative peak chasing: at each step, find the peak and
    take an optimal step to reduce it while maintaining non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Gradient of c[k_star] w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue

        # c[k_star] = sum_{j=j_start}^{j_end} a[j] * a[k_star - j]
        # d(c[k_star])/da[i] involves a[k_star - i] for i in [j_start, j_end]
        # and a[i] for k_star - i in [j_start, j_end]
        dck = np.zeros(n)
        for j in range(j_start, j_end + 1):
            partner = k_star - j
            dck[j] += a[partner]
            if partner != j:
                dck[partner] += a[j]

        # We want to reduce c[k_star]: step in -dck direction
        # But also account for sum constraint (score = 2n*c_max/S^2)
        # d(score)/da[i] = 2n*(dck[i]*S^2 - c_max*2*S) / S^4
        #                = 2n*(dck[i]*S - 2*c_max) / S^3
        grad_score = 2.0 * n * (dck * S - 2.0 * c_max) / (S * S * S)

        # Line search: find best step alpha
        best_alpha = 0.0
        best_local_sc = sc
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_test = a - alpha * grad_score
            a_test = np.maximum(a_test, 0.0)
            S_test = float(a_test.sum())
            if S_test < 1e-6:
                continue
            a_test = a_test / S_test
            sc_test = compute_score(a_test)
            if sc_test < best_local_sc:
                best_local_sc = sc_test
                best_alpha = alpha

        if best_alpha == 0.0:
            # Try just reducing peak contributors
            for j in range(j_start, j_end + 1):
                a_old = a[j]
                for factor in [0.8, 0.9, 0.95]:
                    a[j] = a_old * factor
                    S_t = float(a.sum())
                    if S_t < 1e-6:
                        a[j] = a_old
                        continue
                    a_norm = a / S_t
                    sc_t = compute_score(a_norm)
                    if sc_t < best_local_sc:
                        best_local_sc = sc_t
                        a = a_norm
                        best_a = a.copy()
                        best_sc = sc_t
                        break
                a[j] = a[j]  # restore if not improved (already normalized)

            a = a / float(a.sum())
            continue

        a = a - best_alpha * grad_score
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    avg_step = 0.01

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step with running average
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, avg_step)
        avg_step = 0.9 * avg_step + 0.1 * step

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.2 + 0.6 * np.abs(2.0 * x - 1.0))
    for freq in [2, 3, 5]:
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x))
    # Chirp
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * (x + 2.0 * x * x)))
    # Random blocks
    for nb in [8, 16, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a_blk = np.repeat(vals, per)
        if len(a_blk) < n:
            a_blk = np.concatenate([a_blk, vals[:n - len(a_blk)]])
        inits.append(a_blk[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: spectral water-filling, frequency shaping, peak chasing,
    LP refinement, L-BFGS polish, subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 8192, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # ================================================================
    # Phase 1: Spectral Water-Filling (45% of time)
    # ================================================================
    phase1_deadline = t_start + 0.45 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Spectral Water-Filling ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(3, int(remaining / 20.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break

            # Hard clip water-filling (main)
            a_wf = spectral_waterfill(a0, n_iters=400, start_lam=0.03, end_lam=0.95,
                                      use_phase=True, soft_power=1.0)
            sc = compute_score(a_wf)
            report(sc, a_wf, f"WF n={n} i={i}")

            # Soft clip (p=2)
            a_wf2 = spectral_waterfill(a0, n_iters=200, start_lam=0.1, end_lam=0.8,
                                       use_phase=True, soft_power=2.0)
            sc2 = compute_score(a_wf2)
            report(sc2, a_wf2, f"WFor n={n} i={i}")

            # Frequency shaping
            a_fs = frequency_shaping(best_a if len(best_a) == n else a_wf, n_iters=150)
            sc3 = compute_score(a_fs)
            report(sc3, a_fs, f"FS n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Peak Chasing + LP (20% of time)
    # ================================================================
    phase2_deadline = t_start + 0.65 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: Peak Chasing + LP ---")

    # Peak chasing on best
    n_best = len(best_a)
    a_pc = peak_chasing(best_a, n_iters=300)
    sc = compute_score(a_pc)
    report(sc, a_pc, "peak_chase")

    # LP refinement for smaller lengths
    lp_lengths = [n for n in lengths if n <= 4096]
    if n_best not in lp_lengths and n_best <= 4096:
        lp_lengths.append(n_best)

    for n_lp in lp_lengths[:2]:
        if time.monotonic() > phase2_deadline:
            break
        a_in = best_a if len(best_a) == n_lp else np.interp(
            np.linspace(0, 1, n_lp), np.linspace(0, 1, len(best_a)), best_a
        )
        time_left = max(10.0, phase2_deadline - time.monotonic() - 5)
        a_lp = lp_refine(a_in, max_time=time_left)
        sc = compute_score(a_lp)
        report(sc, a_lp, f"LP n={n_lp}")

        # Water-fill from LP result
        if time.monotonic() < phase2_deadline - 10:
            a_wf_lp = spectral_waterfill(a_lp, n_iters=200, start_lam=0.2, end_lam=0.9)
            sc2 = compute_score(a_wf_lp)
            report(sc2, a_wf_lp, f"WF+LP n={n_lp}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: L-BFGS Refinement (20% of time)
    # ================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    beta_schedule = [50, 100, 300, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        remaining_time = phase3_deadline - time.monotonic()
        if remaining_time < 5:
            break
        max_iter = min(6000, int(remaining_time * 3))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=max_iter, maxfun=max_iter * 20)
        report(sc, a_opt, f"LBFGS b={beta}")

    # From alternative starts
    for n_alt in lengths[:2]:
        if time.monotonic() > phase3_deadline:
            break
        a_alt = best_a if len(best_a) == n_alt else np.interp(
            np.linspace(0, 1, n_alt), np.linspace(0, 1, len(best_a)), best_a
        )
        for beta in [100, 500]:
            if time.monotonic() > phase3_deadline:
                break
            sc, a_opt, _ = run_lbfgs(a_alt, beta, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"LBFGS alt n={n_alt} b={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final Polish (10% of time)
    # ================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    # Subgradient
    a_sp = subgradient_polish(best_a, total_iters=3000)
    sc = compute_score(a_sp)
    report(sc, a_sp, "subgrad")

    # Final water-filling
    if time.monotonic() < deadline - 30:
        a_wf_final = spectral_waterfill(best_a, n_iters=300, start_lam=0.2, end_lam=0.95)
        sc = compute_score(a_wf_final)
        report(sc, a_wf_final, "final_WF")

    # Peak chasing from water-filled
    if time.monotonic() < deadline - 25:
        a_pc_final = peak_chasing(best_a, n_iters=200)
        sc = compute_score(a_pc_final)
        report(sc, a_pc_final, "final_PC")

    # High-beta L-BFGS
    if time.monotonic() < deadline - 15:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final_LBFGS")

    # Another round: WF -> peak_chase -> WF
    if time.monotonic() < deadline - 20:
        a_combo = spectral_waterfill(best_a, n_iters=100, start_lam=0.3, end_lam=0.9)
        a_combo = peak_chasing(a_combo, n_iters=100)
        a_combo = spectral_waterfill(a_combo, n_iters=100, start_lam=0.4, end_lam=0.95)
        sc = compute_score(a_combo)
        report(sc, a_combo, "combo")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
