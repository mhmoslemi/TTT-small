import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):

def adam_lse(a_init, lr=0.005, n_steps=50000, beta=200,
             time_limit_s=None, t_start=None):
    """Adam with LSE-smoothed gradient, cosine LR schedule."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        # Cosine annealing
        lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * t / n_steps))
        lr_t = max(lr_t, lr * 0.01)

        score, grad = lse_score_grad(a, beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def simulated_annealing(a_init, n_iters=50000, t_init=0.1, t_final=0.001,
                        time_limit_s=None, t_start=None):
    """Simulated annealing with diverse move types."""
    n = len(a_init)
    a = a_init.copy()
    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, n_iters + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        temp = t_init * (t_final / t_init) ** (it / n_iters)

        move_type = it % 4
        a_new = a.copy()

        if move_type == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.05 * temp
            a_new = a * (1.0 + scale * rng.standard_normal(n))
        elif move_type == 1:
            # Sinusoidal perturbation
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            amp = 0.03 + 0.05 * temp
            a_new = a * (1.0 + amp * np.cos(2 * np.pi * freq * x + phase))
        elif move_type == 2:
            # Block modification
            center = int(rng.integers(n // 4, 3 * n // 4))
            width = max(5, n // 80)
            factor = 1.0 + 0.3 * temp * (rng.random() * 2 - 1)
            a_new[center - width:center + width] *= factor
        else:
            # Pair swap (transfer mass)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                transfer = a[i] * 0.1 * (1 + rng.random())
                a_new[i] -= transfer
                a_new[j] += transfer

        a_new = np.clip(a_new, 0.0, 1e10)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new = a_new / s

        sc_new = compute_score(a_new)
        sc_old = compute_score(a) if it % 10 == 0 else best_score

        if sc_new < best_score or rng.random() < np.exp(-(sc_new - best_score) / (temp * 0.01)):
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()

    return best_score, best_a


def run_lbfgs(a0, beta=500, maxiter=3000, maxfun=50000):
    """L-BFGS-B polish with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer: LSE gradient descent + peak shaving +
    simulated annealing + L-BFGS polish + multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) * 0.5
        parent_score = compute_score(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==================================================================
    # Phase 1: LSE Adam with beta continuation (50% of time)
    # ==================================================================
    p1 = effective_budget * 0.50
    print(f"\n{'='*50}")
    print(f"Phase 1: LSE Adam (budget={p1:.0f}s)")
    print(f"{'='*50}")

    a_norm = best_a / best_a.sum()

    beta_schedule = [30, 80, 200, 500, 1000, 2000, 4000]
    lr_schedule = [0.01, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0005]

    n_phases = len(beta_schedule)
    for i in range(n_phases):
        remaining = p1 - (time.monotonic() - t_start)
        if remaining < 5:
            break
        time_alloc = remaining / (n_phases - i)
        beta = beta_schedule[i]
        lr = lr_schedule[i]
        print(f"  beta={beta}, lr={lr}, time_alloc={time_alloc:.0f}s")
        sc, a_opt = adam_lse(a_norm, lr=lr, n_steps=200000, beta=beta,
                             time_limit_s=time_alloc, t_start=t_start)
        if sc < best_score:
            try_update(sc, a_opt, f"Adam beta={beta}")
        a_norm = a_opt

    # ==================================================================
    # Phase 2: Peak shaving (15% of time)
    # ==================================================================
    p2 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 2: Peak Shaving (budget={p2:.0f}s)")
    print(f"{'='*50}")

    a_norm = best_a / best_a.sum()
    remaining = p2 - (time.monotonic() - t_start)
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = peak_shave(a_norm, n_iters=500, step_scale=0.05)
        try_update(sc, a_opt, "Peak shave")
        print(f"    Peak shave: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

        # More aggressive peak shaving with smaller steps
        remaining = p2 - (time.monotonic() - t_start)
        if remaining > 5:
            t0 = time.monotonic()
            sc, a_opt = peak_shave(best_a / best_a.sum(), n_iters=1000, step_scale=0.02)
            try_update(sc, a_opt, "Peak shave fine")
            print(f"    Peak shave fine: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Phase 3: Simulated annealing (15% of time)
    # ==================================================================
    p3 = effective_budget * 0.15
    print(f"\n{'='*50}")
    print(f"Phase 3: Simulated Annealing (budget={p3:.0f}s)")
    print(f"{'='*50}")

    remaining = p3 - (time.monotonic() - t_start)
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_opt = simulated_annealing(best_a, n_iters=100000,
                                        time_limit_s=remaining * 0.8, t_start=t_start)
        try_update(sc, a_opt, "Simulated annealing")
        print(f"    SA: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Phase 4: Restart with perturbations + quick optimize (10% of time)
    # ==================================================================
    p4 = effective_budget * 0.10
    print(f"\n{'='*50}")
    print(f"Phase 4: Restarts (budget={p4:.0f}s)")
    print(f"{'='*50}")

    n_best = len(best_a)
    restart_deadline = t_start + effective_budget * 0.90

    n_restarts = 8
    for r in range(n_restarts):
        if time.monotonic() > restart_deadline:
            break

        strategy = r % 4
        if strategy == 0:
            # Gaussian
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= 0.7
            removed = float(best_a[js:je + 1].sum() - a_pert[js:je + 1].sum())
            mask = np.ones(n_best, dtype=bool)
            mask[js:je + 1] = False
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            # Multi-sinusoidal
            x = np.arange(n_best, dtype=np.float64) / n_best
            f1 = 1.0 + 4.0 * rng.random()
            f2 = 3.0 + 7.0 * rng.random()
            p1p = 2 * np.pi * rng.random()
            p2p = 2 * np.pi * rng.random()
            a_pert = best_a * (1.0 + 0.03 * np.cos(2 * np.pi * f1 * x + p1p)
                               + 0.02 * np.cos(2 * np.pi * f2 * x + p2p))
        else:
            # Block
            center = int(rng.integers(n_best // 4, 3 * n_best // 4))
            width = max(10, n_best // 60)
            a_pert = best_a.copy()
            a_pert[center - width:center + width] *= rng.uniform(0.4, 1.4)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        s = a_pert.sum()
        if s > 1e-12:
            a_pert = a_pert / s

        time_left = restart_deadline - time.monotonic()
        if time_left < 5:
            break
        adam_time = min(15.0, time_left * 0.6)
        sc, a_r = adam_lse(a_pert, lr=0.003, n_steps=50000, beta=300,
                           time_limit_s=adam_time, t_start=t_start)
        if time.monotonic() < restart_deadline - 3:
            sc2, a_r2 = run_lbfgs(a_r, beta=1000, maxiter=1000, maxfun=20000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart#{r} s{strategy}")

    # ==================================================================
    # Phase 5: L-BFGS polish (7% of time)
    # ==================================================================
    p5 = effective_budget * 0.07
    print(f"\n{'='*50}")
    print(f"Phase 5: L-BFGS Polish (budget={p5:.0f}s)")
    print(f"{'='*50}")

    a_norm = best_a / best_a.sum() * best_a.sum()  # keep original scale
    for beta in [1000, 3000]:
        if time.monotonic() > t_start + effective_budget - 30:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(a_norm, beta=beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        a_norm = a_opt

    # ==================================================================
    # Phase 6: Different lengths (3% of time)
    # ==================================================================
    print(f"\n{'='*50}")
    print("Phase 6: Different Lengths")
    print(f"{'='*50}")

    n_parent = len(best_a)
    for factor in [0.9, 1.1]:
        tl = int(n_parent * factor)
        if tl < 512 or tl > 30000:
            continue
        if time.monotonic() > t_start + effective_budget - 20:
            break
        a_res = resample_sequence(best_a, tl)
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_res, lr=0.003, n_steps=30000, beta=200,
                             time_limit_s=min(20.0, effective_budget - (time.monotonic() - t_start) - 10),
                             t_start=t_start)
        if time.monotonic() < t_start + effective_budget - 5:
            sc2, a_opt2 = run_lbfgs(a_opt, beta=500, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt, f"n={tl}")
        print(f"    n={tl}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==================================================================
    # Final: one more LSE + peak shave cycle
    # ==================================================================
    final_deadline = t_start + effective_budget - 3.0
    if time.monotonic() < final_deadline:
        a_norm = best_a / best_a.sum()
        sc, a_opt = adam_lse(a_norm, lr=0.001, n_steps=50000, beta=2000,
                             time_limit_s=min(15.0, final_deadline - time.monotonic()),
                             t_start=t_start)
        try_update(sc, a_opt, "final Adam")

        if time.monotonic() < final_deadline - 3:
            sc, a_opt = peak_shave(best_a / best_a.sum(), n_iters=200, step_scale=0.03)
            try_update(sc, a_opt, "final peak shave")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
