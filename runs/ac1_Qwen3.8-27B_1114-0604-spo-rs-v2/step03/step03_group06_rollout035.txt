"""
Multiplicative Descent + L-BFGS-B hybrid optimizer for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovation: Exponentiated gradient (mirror descent with entropy)
with top-k peak averaging, combined with LSE-smoothed L-BFGS-B polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS-B."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: sum_k p[k] * ∂b[k]/∂a
    # ∂b[k]/∂a[m] = 2*a[k-m] (valid)
    # Use FFT: gradient = 2 * conv(rev_p, rev_a) shifted appropriately
    # Actually, let's compute it directly via correlation
    # grad_a[m] = 2 * sum_k p[k] * a[k-m] (valid)
    # This is 2 * (p * rev_a) at the right shift
    # p has length 2n-1, rev_a has length n
    # conv(p, rev_a) at index m gives sum_j p[j] * rev_a[m-j]
    # = sum_j p[j] * a[n-1-(m-j)] = sum_j p[j] * a[n-1-m+j]
    # We want sum_k p[k] * a[k-m] = sum_k p[k] * rev_a[n-1-k+m]
    # Let's use: grad[m] = 2 * sum_k p[k] * a[k-m]
    # = 2 * (cross_correlation of p and a) at lag m
    # = 2 * (p * rev_a)[n-1+m] (in conv terms)
    # Actually: let q = rev_a (q[j] = a[n-1-j])
    # (p * q)[k] = sum_j p[j] * q[k-j] = sum_j p[j] * a[n-1-k+j]
    # We want: sum_k p[k] * a[k-m]
    # Let k' = k-m: sum_{k'} p[k'+m] * a[k']
    # = sum_{k'} a[k'] * p[m+k']
    # This is (a * rev_p)[n-1-m] where rev_p[j] = p[2n-2-j]
    # Hmm, this is getting complicated. Let me just use a direct approach.
    
    # Direct approach: for each m, grad[m] = 2 * sum_k p[k]*a[k-m] (valid k)
    # This is a cross-correlation. Use FFT.
    #Corr between p (length 2n-1) and a (length n):
    # r[m] = sum_k p[k+m] * a[k] (for appropriate m)
    # Using: r = ifft(conj(fft(p)) * fft(a)) ... no, that's circular.
    
    # Let's use the fact that:
    # sum_k p[k] * a[k-m] = (p * rev_a)[(2n-2) - (n-1) + m] (conv trick)
    # Actually, let me just use np.correlate or do it with FFT.
    
    # Simplest: use the FFT approach from the original code
    rev_a = a[::-1]
    M2 = (2*n-1) + n - 1  # = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[:3*n-2]
    # conv_res[k] = sum_j p[j] * rev_a[k-j] = sum_j p[j] * a[n-1-k+j]
    # We want grad[m] = 2 * sum_k p[k] * a[k-m]
    # Let k = j+m: sum_j p[j+m] * a[j] = sum_j a[j] * p[j+m]
    # = conv_res at position: n-1-j + j+m = n-1+m ... 
    # Hmm let me think differently.
    # conv_res[k] = sum_j p[j] * a[n-1-k+j] (valid j)
    # Let's substitute: we want sum_k p[k]*a[k-m]
    # = sum_k p[k] * a[k-m]
    # In conv_res: conv_res[n-1+(k-m)-... ]
    # Let me just verify with the original code's approach.
    
    # The original code does:
    # conv_res = irfft(RA * P, M2_fft)[n-1:2*n-1]
    # This gives n elements. Let's see what they are.
    # conv_res_full[k] = sum_j rev_a[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]
    # At k = n-1+m: sum_j a[n-1-j] * p[n-1+m-j]
    # Let i = n-1-j: sum_i a[i] * p[m+i]
    # = sum_i a[i] * p[i+m]
    # This is exactly what we want! grad[m] = 2 * sum_i a[i] * p[i+m]

    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.res

"""
Hybrid optimizer: Multiplicative (mirror) descent with top-k peak averaging,
combined with LSE-smoothed L-BFGS-B polishing, for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovation over prior work: exponentiated gradient (mirror descent with
entropy) as the primary workhorse, which naturally preserves non-negativity
and handles the combinatorial max via top-k averaging with momentum.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for use with L-BFGS-B."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad

    return lse_score_grad(x, beta)

def compute_topk_gradient(a, b, top_k):
    """Compute averaged gradient over top-k peaks of convolution."""
    n = len(a)
    Lb = len(b)
    if top_k >= Lb:
        top_idx = np.arange(Lb)

    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()

            best_sc = sc
            best_a = a.copy()

    # =========================================================================
    # Phase 3: L-BFGS-B high-beta continuation (20% of time)
    # =========================================================================
    p3_deadline = t_start + 0.85 * (budget_s - 20)
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS-B high-beta polish (until t={p3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    beta_schedule = [500, 1000, 2000, 4000, 8000, 15000]
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > p3_deadline:
            break
        remaining = p3_deadline - time.monotonic()
        maxiter = min(10000, max(1000, int(remaining / 10.0)))
        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxiter * 2)
        _report(sc, a_opt, f"P3 beta={beta}")
        a_cur = a_opt

    print(f"After Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final refinement (10% of time)
    # =========================================================================
    p4_deadline = deadline
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement (until t={p4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # Subgradient polish
    if time.monotonic() < p4_deadline - 30:
        sc, a_sg = subgradient_polish(best_a, n_iters=5000, deadline=p4_deadline - 30)
        _report(sc, a_sg, "subgrad")

    # Final high-beta L-BFGS
    if time.monotonic() < p4_deadline - 10:
        sc, a_final = run_lbfgs(best_a, 20000, maxiter=10000, maxfun=200000)
        _report(sc, a_final, "final LBFGS")

    # One more subgradient pass
    if time.monotonic() < p4_deadline - 3:
        sc, a_sg2 = subgradient_polish(best_a, n_iters=3000, deadline=p4_deadline - 3)
        _report(sc, a_sg2, "final subgrad")

    # =========================================================================
    # Also try a shorter length (sometimes helps)
    # =========================================================================
    if time.monotonic() < deadline - 120 and n > 3000:
        n_short = max(2048, n // 2)
        print(f"\nTrying shorter length n={n_short}...")
        a_short = np.ones(n_short) / n_short
        if parent is not None:
            idx = np.linspace(0, len(parent) - 1, n_short).astype(int)
            a_short = np.clip(parent[idx], 1e-10, None)
            a_short /= a_short.sum()
        for beta in [200, 1000, 5000]:
            if time.monotonic() > deadline - 30:
                break
            sc, a_opt = run_lbfgs(a_short, beta, maxiter=3000, maxfun=6000)
            _report(sc, a_opt, f"short n={n_short}, beta={beta}")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
