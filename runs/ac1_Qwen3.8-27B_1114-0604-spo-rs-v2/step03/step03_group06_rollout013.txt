def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)

    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        score = 2.0 * n * c_max / (S * S)
        return score, np.zeros(n)
    log_sum = np.log(exp_sum) + max_scaled / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient computation: dL/da_j = 2 * sum_k p[k] * a[k-j] (valid k-j)
    # = 2 * (rev_a * p)[n-1+j] for j=0..n-1

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_main = parent.size
    else:
        n_main = 6962
    print(f"Parent length: {n_main if parent is not None else 'N/A'}")

    # Normalize parent for optimization
    if parent is not None:
        a_work = normalize_sequence(parent.copy())
        parent_score = compute_score(a_work)
        print(f"Parent score (normalized): {parent_score:.10f}")
        report(parent_score, a_work, "parent init")
    else:
        a_work = np.ones(n_main) * 0.5
        report(compute_score(a_work), a_work, "uniform init")

    # =========================================================================
    # Phase 1: LSE L-BFGS with aggressive beta schedule (45% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.45 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LSE L-BFGS optimization (until t={phase1_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = a_work.copy()
    beta_schedule = [30, 100, 300, 1000, 3000, 8000, 20000]
    
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_deadline:
            break
        remaining_time = phase1_deadline - time.monotonic()
        # Scale maxiter based on remaining time and beta
        # Higher beta needs fewer iterations (already close)
        if beta <= 100:
            maxiter = int(min(5000, max(500, remaining_time / 3.0)))
            maxfun = maxiter * 3
        elif beta <= 1000:
            maxiter = int(min(3000, max(300, remaining_time / 5.0)))
            maxfun = maxiter * 3
        else:
            maxiter = int(min(2000, max(200, remaining_time / 8.0)))
            maxfun = maxiter * 3
        
        print(f"  LSE beta={beta}, maxiter={maxiter}")
        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        print(f"    Score: {sc:.10f}, Converged: {res.message}")
        report(sc, a_opt, f"LSE beta={beta}")
        a_cur = a_opt

    # Try top-k objective as alternative
    if time.monotonic() < phase1_deadline:
        for k in [3, 5, 10]:
            if time.monotonic() > phase1_deadline:
                break
            remaining_time = phase1_deadline - time.monotonic()
            maxiter = int(min(2000, max(200, remaining_time / 10.0)))
            sc, a_opt, _ = run_lbfgs(a_cur, 0, maxiter=maxiter, maxfun=maxiter*3,
                                     use_topk=True, topk_k=k)
            report(sc, a_opt, f"topk={k}")
            a_cur = a_opt

    print(f"\nAfter Phase 1: best = {best_score:.10f}")

    # =========================================================================
    # Phase 2: Alternating exact subgradient + LSE polish (25% of time)
    # =========================================================================
    phase2_deadline = t_start + 0.70 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient + LSE alternating (until t={phase2_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_alt = best_a.copy()
    iteration = 0
    while time.monotonic() < phase2_deadline:
        iteration += 1
        # Exact subgradient steps
        sc_sg, a_sg = multi_lag_subgrad_steps(a_alt, n_steps=200, top_k_lags=5,
                                              lower_bound=best_score - 0.5, max_step=0.3)
        report(sc_sg, a_sg, f"subgrad iter {iteration}")
        a_alt = a_sg

        # LSE polish
        if time.monotonic() < phase2_deadline - 5.0:
            remaining = phase2_deadline - time.monotonic()
            maxiter = int(min(1000, max(100, remaining / 15.0)))
            sc_lb, a_lb, _ = run_lbfgs(a_alt, 5000, maxiter=maxiter, maxfun=maxiter * 2)
            report(sc_lb, a_lb, f"LSE polish iter {iteration}")
            a_alt = a_lb

        # Also try single-lag subgradient
        if time.monotonic() < phase2_deadline - 3.0:
            sc_1, a_1 = exact_subgrad_steps(a_alt, n_steps=100,
                                            lower_bound=best_score - 0.3, max_step=0.2)
            report(sc_1, a_1, f"exact sg iter {iteration}")
            a_alt = a_1

    print(f"\nAfter Phase 2: best = {best_score:.10f}")

    # =========================================================================
    # Phase 3: Coordinate descent + fine LSE (20% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.90 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate descent + fine polish (until t={phase3_deadline-t_start:.0f}s)")
    print(f"{'='*60}")

    a_cd = best_a.copy()
    # Random coordinate descent
    n_cd_total = 3000  # total coordinates to try
    sc_cd, a_cd = random_coordinate_descent(a_cd, n_cd_total, rng, phase3_deadline)
    report(sc_cd, a_cd, "coord descent")

    # LSE polish after coordinate descent
    if time.monotonic() < phase3_deadline:
        remaining = phase3_deadline - time.monotonic()
        for beta in [10000, 30000]:
            if time.monotonic() > phase3_deadline - 5.0:
                break
            maxiter = int(min(2000, max(200, remaining / 10.0)))
            sc, a_opt, _ = run_lbfgs(a_cd, beta, maxiter=maxiter, maxfun=maxiter * 2)
            report(sc, a_opt, f"fine LSE beta={beta}")
            a_cd = a_opt

    print(f"\nAfter Phase 3: best = {best_score:.10f}")

    # =========================================================================
    # Phase 4: Final refinement (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement")
    print(f"{'='*60}")

    # Very high beta LSE
    a_final = best_a.copy()
    for beta in [50000, 100000]:
        if time.monotonic() > deadline - 10.0:
            break
        maxiter = int(min(3000, max(500, (deadline - time.monotonic() - 5.0) / 5.0)))
        sc, a_opt, _ = run_lbfgs(a_final, beta, maxiter=maxiter, maxfun=maxiter * 2)
        report(sc, a_opt, f"final LSE beta={beta}")
        a_final = a_opt

    # Final subgradient polish
    if time.monotonic() < deadline - 5.0:
        sc_sg, a_sg = exact_subgrad_steps(a_final, n_steps=500,
                                          lower_bound=best_score - 0.1, max_step=0.1)
        report(sc_sg, a_sg, "final subgrad")

    # Targeted perturbation + quick polish
    if time.monotonic() < deadline - 10.0:
        for strength in [0.01, 0.02, 0.03]:
            if time.monotonic() > deadline - 5.0:
                break
            a_pert = targeted_perturb(best_a, rng, strength)
            sc, a_opt, _ = run_lbfgs(a_pert, 20000, maxiter=1000, maxfun=2000)
            report(sc, a_opt, f"perturb s={strength}")

    # =========================================================================
    # Final verification
    # =========================================================================
    out_a = best_a
    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
