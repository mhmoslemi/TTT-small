"""
Advanced optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2.
Combines high-beta LSE-L-BFGS, LP-based fixed-point iteration, and peak equalization.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags as sparse_diags
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a:
    # dL/da = 2 * (p * a_reversed)[n-1 : 2n-1] (the cross-correlation)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1 : 2 * n - 1]
    score = 2.0 * n * L / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * conv_res / S2 - 2.0 * L / S3)
    return score, grad

def lbfgs_wrapper(a, beta):
    return lse_grad(a, beta)

    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lbfgs_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': maxls}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def build_conv_matrix_sparse(f):
    """Build the convolution matrix A where A[k,i] = f[k-i] for valid k-i."""
    n = len(f)
    # n diagonals with offsets 0, -1, ..., -(n-1)
    # Each diagonal has n entries with value f[d] for diagonal d
    diags_vals = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -(n), -1))
    A = sparse_diags(diags_vals, offsets, shape=(2*n-1, n), format='csc')
    return A

def lp_step(f, M, time_limit=60):

    return None
