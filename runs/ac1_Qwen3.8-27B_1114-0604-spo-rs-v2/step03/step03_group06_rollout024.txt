def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lambda x: lse_score_grad(x, beta), a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def mw_direction(a, M, n_iters=1000, eta=None):
    """
    Find a direction b >= 0, sum(b)=1, that approximately maximizes sum(b)
    subject to (a*b)[k] <= M for all k, using multiplicative weights.
    
    Returns b (normalized to sum 1).
    """
    n = len(a)
    b = np.ones(n) / n
    
    # Precompute: for each k, the row A[k,:] is a shifted version of a.
    # (a*b)[k] = sum_j a[j] * b[k-j] (with appropriate bounds)
    # We can compute all (a*b)[k] via cross-correlation (FFT).
    
    if eta is None:
        eta = 0.1  # Will tune this
    
    for t in range(n_iters):
        # Compute (a*b)
        # (a*b)[k] = sum_j a[j] * b[k-j]
        # This is the cross-correlation, which we can compute via FFT.
        # Actually, this is just the convolution of a and b!
        ab = fast_conv_cross(a, b)  # (a*b)[k] for k=0,...,2n-2
        
        # Find the most violated constraint
        ratios = ab / M
        k_star = int(np.argmax(ratios))
        
        # Update: b[j] *= exp(-eta * a[k_star - j]) for valid j
        # A[k_star, j] = a[k_star - j] if 0 <= k_star - j < n, else 0
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            # A[k_star, j] = a[k_star - j] for j in [j_start, j_end]
            # k_star - j ranges from k_star - j_end to k_star - j_start
            coeff = a[k_star - j_end : k_star - j_start + 1][::-1]
            # Or more simply: coeff[j] = a[k_star - j] for j in [j_start, j_end]
            log_update = -eta * a[k_star - np.arange(j_start, j_end + 1)]
            b[j_start:j_end+1] *= np.exp(log_update)
        
        # Normalize
        b_sum = b.sum()
        if b_sum < 1e-30:
            b = np.ones(n) / n
        else:
            b /= b_sum
    
    return b

"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights (MW) for fast LP-like direction finding, 
LSE-smoothed L-BFGS-B for polishing, and multi-start exploration.
Key improvement over previous: MW-based directions are ~100x faster than LP,
allowing many more optimization steps in the same time budget.
"""
import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except ImportError:
    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _lse_obj_wrapper(x, beta):
    """Objective wrapper for L-BFGS-B (no lambda)."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B minimization with LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    # Create a class-based objective to avoid lambda
    class ObjFn:
        def __init__(self, beta):
            self.beta = beta
        def __call__(self, x):
            return lse_score_grad(x, self.beta)
    
    obj_fn = ObjFn(beta)
    res = minimize(
        obj_fn, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-11,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def mw_direction(a, M, n_iters=500, eta_init=1.0, eta_decay=0.999):
    """
    Find a descent direction using multiplicative weights.
    Maximizes sum(b) subject to b>=0, (a*b)[k] <= M for all k.
    Returns b normalized to have same sum as a.
    """
    n = len(a)
    b = np.ones(n) / n
    eta = eta_init
    
    for t in range(n_iters):
        # Compute cross-correlation (a*b)
        ab = fast_cross_conv(a, b)
        # Find most violated constraint
        max_ab = float(np.max(ab))
        if max_ab <= M * (1 + 1e-10):
            break  # All constraints satisfied
        
        k_star = int(np.argmax(ab))
        
        # Update: b[j] *= exp(-eta * A[k_star, j])
        # A[k_star, j] = a[k_star - j] if 0 <= k_star - j < n
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            coeff = a[k_star - js]  # a[k_star - j] for j in [j_start, j_end]
            b[js] *= np.exp(-eta * coeff)
        
        # Normalize to keep sum = 1
        b_sum = b.sum()
        if b_sum < 1e-30:
            b = np.ones(n) / n
        else:
            b /= b_sum
        
        # Decay eta
        eta *= eta_decay
    
    # Rescale to match sum of a
    target_sum = float(a.sum())
    if b.sum() > 1e-10:
        b = b * (target_sum / b.sum())
    return b


def mw_line_search(a, direction, M, max_steps=20):
    """
    Line search: find best t in [0, 1] such that (1-t)*a + t*direction
    minimizes the score.
    """
    n = len(a)
    best_t = 0.0
    best_sc = compute_score(a)
    
    # Try logarithmically spaced t values
    ts = np.logspace(-3, 0, max_steps)  # 0.001 to 1
    for t in ts:
        a_new = (1.0 - t) * a + t * direction
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t > 0:
        a_new = (1.0 - best_t) * a + best_t * direction
        a_new = np.clip(a_new, 0.0, 1000.0)
    else:
        a_new = a.copy()
    
    return best_sc, a_new


def nesterov_subgrad(a, n_iters=2000, lr_init=0.1):
    """
    Nesterov-accelerated subgradient descent on the exact max-conv objective.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # Lower bound on the score
    
    for it in range(n_iters):
        # Nesterov lookahead
        y = a + (it / (it + 2)) * v  # Hmm, this isn't quite right
        # Actually, let's use the standard Nesterov scheme
        # y = a + momentum * v
        # grad = subgrad at y
        # v_new = momentum * v - lr * grad
        # a_new = y + v_new - v = a + v_new
        
        # Simpler: just use Polyak step size subgradient
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(c) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, lr_init / (1.0 + it * 0.01))
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())
    
    # Structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # U-shape
    inits.append(np.clip(0.3 + 0.7 * (2.0 * x - 1.0) ** 2, 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits


def solve_lp_direction(a, max_wall_s=30.0):
    """Solve the LP for the optimal direction (fallback if MW is not enough)."""
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if n > 8000:
        return None
    
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()
    
    # Build constraint matrix
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None
    
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pa = parent.size
        lengths.append(pa)
        for dl in [-500, -1000, 500, 1000, -2000, 2000]:
            cand = pa + dl
            if 2048 <= cand <= 20000:
                lengths.append(cand)

        lengths = [4096, 6144, 8192, 10240]
    
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")
    print(f"CVXPY available: {HAVE_CVXPY}")

    # =========================================================================
    # Phase 1: MW-based optimization (60% of time)
    # =========================================================================
    phase1_deadline = t_start + 0.60 * (total_budget - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: MW-based optimization")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = build_inits(n, rng, parent)
        
        # Limit initializations based on remaining time

        max_inits = max(2, min(len(inits), int(remaining / 60.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break

            a_cur = np.clip(a_cur, 0.0, 1000.0)
            if a_cur.sum() < 0.01:
                a_cur[:] = 0.5
            
            sc0 = compute_score(a_cur)
            report(sc0, a_cur, f"n={n}, init={i}")
            
            # MW-based optimization loop
            n_mw_rounds = 50
            for mw_round in range(n_mw_rounds):
                if time.monotonic() > phase1_deadline:
                    break

                S = float(a_cur.sum())
                M = float(np.max(c))
                sc = 2.0 * n * M / (S * S)
                
                # Find MW direction
                b_dir = mw_direction(a_cur, M, n_iters=300, eta_init=0.5)
