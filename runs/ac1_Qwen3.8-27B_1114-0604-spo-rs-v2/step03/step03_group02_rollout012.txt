"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_score_and_grad(a, beta):
    """LSE-smoothed version of max(conv(a,a)) and its gradient."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())  # numerical stability
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a
    # dL/da[j] = sum_m p[m] * d(c[m])/da[j] = sum_m p[m] * a[m-j] (if valid)
    # This is the cross-correlation of p and a, evaluated at lags 0..n-1
    M_fft = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    # cross[m] = sum_j a[j] * p[m+j] ... hmm, let me be more careful
    # c[m] = sum_i a[i] * a[m-i]
    # dc[m]/da[j] = a[m-j] (if 0 <= m-j <= n-1)
    # dL/da[j] = sum_m p[m] * a[m-j] (if 0 <= m-j <= n-1)
    # = sum_{m-j=0}^{n-1} p[m] * a[m-j]
    # Let k = m-j, then m = k+j:
    # = sum_{k=0}^{n-1} p[k+j] * a[k]
    # This is the cross-correlation of p and a at lag j
    # cross_corr[j] = sum_k p[k+j] * a[k]
    
    # Using FFT: cross_corr = ifft(conj(FFT(a)) * FFT(p_padded))
    # Actually, let's use a different approach:
    # cross_corr[j] = sum_k a[k] * p[j+k] for k=0..n-1
    # This is (a * p_rev)[j + n - 1] ... let me just compute it directly with FFT
    
    # a * p (linear convolution) has length 2n-1
    # (a*p)[m] = sum_k a[k] * p[m-k]
    # We want sum_k a[k] * p[j+k] = (a * p_rev)[j + (n-1)] where p_rev is reverse of p
    # Hmm, this is getting confusing. Let me use the direct formula:
    # dL/da[j] = sum_{m=max(j,0)}^{min(n-1+j, 2n-2)} p[m] * a[m-j]
    # Since a is defined for indices 0..n-1, m-j must be in 0..n-1, so m in j..j+n-1.
    # And m must be in 0..2n-2.
    # So dL/da[j] = sum_{m=j}^{min(j+n-1, 2n-2)} p[m] * a[m-j]
    
    # I'll compute this using FFT cross-correlation
    # Let f[j] = sum_m p[m] * a[m-j] for m = j..j+n-1
    # This is the correlation of p and a at lag j
    # corr[j] = sum_m p[m] * a[m-j] = (p * a_rev)[n-1+j] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    M2 = 3 * n - 2  # length of p * a_rev
    M2_fft = 1 << (M2 - 1).bit_length()
    P2 = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P2 * AR, M2_fft)
    # (p * a_rev)[n-1+j] for j = 0..n-1
    grad_L = conv_pa[n-1:2*n-1]
    
    return L, grad_L

def lse_objective_and_grad(a, beta):
    """Full LSE score (with normalization) and gradient."""
    n = len(a)
    L, grad_L = lse_score_and_grad(a, beta)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    score = 2.0 * n * L / (S * S)
    grad = (2.0 * n) * (grad_L / (S * S) - 2.0 * L / (S**3) * a)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    n = len(a0)
    a0 = np.clip(a0.copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_func(x):
        sc, g = lse_objective_and_grad(x, beta)
        return sc, g
    
    options = {'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 300)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 50)))
    
    try:
        res = minimize(obj_func, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out

def geometric_mean_equalize(a, n_iters=200, alpha=0.5, n_active=20, deadline=None):
    """
    Geometric mean multiplicative update for equalizing the dual condition.
    At optimum: sum_k lambda_k * a[k-i] = const for all i.
    We iteratively update a to make this more uniform.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        
        # Find active lags (near the maximum)
        threshold = M * (1.0 - 1.0 / n_active)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute h[i] = mean of a[k-i] over active lags k
        # h[i] = (1/|A|) * sum_{k in A} a[k-i] (with a[j]=0 for j out of range)
        h = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                h[j_start:j_end+1] += a[k - np.arange(j_start, j_end+1)]
        h /= len(active)
        
        # Geometric mean update: a[i] <- a[i] * (h_mean / h[i])^alpha
        h = np.maximum(h, 1e-15)
        h_mean = float(h.mean())
        factor = (h_mean / h) ** alpha
        a = a * factor
        a = np.clip(a, 1e-10, 1000.0)
        
        # Renormalize
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def fista_lse(a0, beta, n_iters=500, deadline=None):
    """FISTA (proximal gradient) for min LSE_score(a) subject to a>=0."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0/n
    S = a.sum()
    a = a / S  # normalize to sum=1
    z = a.copy()
    t = 1.0
    L = beta / (n * 0.1)  # Lipschitz constant estimate (conservative)
    # Actually, L = beta * max_k ||grad (a*a)_k||^2 = beta * max_k sum_j a[k-j]^2 <= beta * sum_j a[j]^2
    # On the simplex with sum=1, sum a[j]^2 <= 1, so L <= beta
    L = beta * float(np.sum(a**2)) + 1e-10
    eta = 1.0 / L
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        sc, grad = lse_objective_and_grad(z, beta)
        # Gradient step
        y = z - eta * grad
        # Project onto {a >= 0, sum = 1} (simplex)
        # But we also need a >= 0... the projection onto {a>=0, sum(a)=1} is project_simplex
        a_new = project_simplex(y)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = a_new + ((t - 1.0) / t_new) * (a_new - a)
        z = np.maximum(z, 0.0)
        if z.sum() > 0:
            z = z / z.sum()
        else:
            z = a_new
        a = a_new
        t = t_new
        
        exact_sc = compute_score(a)
        if exact_sc < best_sc:
            best_sc = exact_sc
            best_a = a.copy()
    
    return best_sc, best_a

def targeted_local_search(a, rng, n_trials=50, deadline=None):
    """Aggressive local search with targeted perturbations."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        strategy = trial % 6
        if strategy == 0:
            # Small Gaussian
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
        elif strategy == 1:
            # Targeted: reduce peak-contributing elements
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            contrib = best_a[j_start:j_end+1].copy()
            total = contrib.sum()
            if total > 0:
                # Reduce top contributors
                top_k = max(3, n // 100)
                top_idx = np.argsort(contrib)[::-1][:top_k]
                reduce_total = 0
                for j in top_idx:
                    amt = contrib[j] * 0.1
                    a_pert[j_start + j] -= amt
                    reduce_total += amt
                # Add to random other elements
                other = np.setdiff1d(np.arange(n), np.arange(j_start, j_end+1)[:top_k])
                if len(other) > 0:
                    a_pert[other] += reduce_total / len(other)
        elif strategy == 2:
            # Sinusoidal modulation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.06 * rng.random()
            x = np.arange(n, dtype=np.float64) / n
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 3:
            # Block flip
            a_pert = best_a.copy()
            block = max(8, n // 64)
            b1 = int(rng.integers(0, n // block - 1)) * block
            b2 = int(rng.integers(0, n // block - 1)) * block
            if b1 != b2:
                a_pert[b1:b1+block], a_pert[b2:b2+block] = a_pert[b2:b2+block].copy(), a_pert[b1:b1+block].copy()
        elif strategy == 4:
            # Geometric mean equalize for a few steps
            _, a_eq = geometric_mean_equalize(best_a, n_iters=10, alpha=0.7, n_active=15, deadline=deadline)
            a_pert = a_eq
        else:
            # Random restart with blending
            blend = 0.1 + 0.3 * rng.random()
            a_random = np.clip(0.5 * rng.random(n) + 0.01, 0.0, 1000.0)
            a_pert = (1.0 - blend) * best_a + blend * a_random * (best_a.sum() / a_random.sum())
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-10:
            continue
        
        # Short L-BFGS refinement
        sc, a_opt = run_lbfgs(a_pert, 200, maxiter=1000, maxfun=20000, deadline=deadline)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
    
    return best_sc, best_a
