```python
"""
Convex optimization approach to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LSE-smoothed L-BFGS with aggressive beta continuation,
(2) exact subgradient polish on the true max, (3) peak-targeted local search,
(4) active-set-inspired cutting to focus on binding constraints.
Leverages convexity: any local minimum is global.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact score 2n*maxb/S^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    Compute LSE-smoothed score and gradient.
    LSE(c, beta) = (1/beta)*log(sum exp(beta*c_k))
    score = 2n * LSE / S^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    # Numerically stable LSE
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da_j = 2 * sum_m a[m] * p[j+m]
    # Computed as: conv(rev_a, p)[n-1:2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_step(a, step_size):
    """One projected subgradient step on the exact max-based score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a
    M = float(c.max())
    k_star = int(np.argmax(c))
    # Subgradient of M w.r.t. a_j
    sg = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * M / S2
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new, score


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, ftol=1e-18, gtol=1e-14):
    """Run L-BFGS-B on the LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res


def subgrad_polish(a, n_iters=1000, start_lr=0.01):
    """Polish using exact subgradient with decreasing step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound ~1
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        # Polyak step size
        lr = (sc - sc_lower) / (gn * gn + 1e-30)
        lr = min(lr, start_lr / (1.0 + 0.05 * it))
        lr = max(lr, 1e-10)
        a = a - lr * grad
        a = np.clip(a, 0.0, 1000.0)
        # Extra small step for fine tuning
        a = a - (1e-5 / (1.0 + 0.1 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def multi_peak_subgrad(a, n_iters=500, start_lr=0.005):
    """Subgradient using top-k peaks for smoother gradient."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lower = 2.0 * n / (2.0 * n - 1.0)
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        # Use soft-max over top peaks
        top_k = min(20, len(c))
        c_sorted = np.sort(c)[::-1]
        # Soft weights: exponential decay
        weights = np.exp(-(c_sorted - M) * 50.0)
        weights /= weights.sum()
        top_indices = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for wi, ki in zip(weights, top_indices):
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * wi * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        lr = min((sc - sc_lower) / (gn * gn + 1e-30), start_lr / (1.0 + 0.05 * it))
        lr = max(lr, 1e-10)
        a = a - lr * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def spectral_flatten(a0, N, n_iters=200, lam=0.5):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a


def peak_targeted_perturb(a, rng, strength=0.03):
    """Perturb specifically to reduce the peak lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    a_pert = a.copy()
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    # Reduce elements that contribute to the peak
    for j in range(j_start, j_end + 1):
        reduce_amt = a_pert[j] * strength * rng.random()
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
    # Redistribute to low-contribution elements
    total_reduced = a.sum() - a_pert.sum()
    if total_reduced > 0:
        # Find elements NOT in the peak-contributing range
        mask = np.ones(n, dtype=bool)
        mask[j_start:j_end + 1] = False
        candidates = np.where(mask)[0]
        if len(candidates) > 0:
            n_add = min(len(candidates), max(10, n // 10))
            add_idx = rng.choice(candidates, size=n_add, replace=False)
            per = total_reduced / n_add
            a_pert[add_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Strategy: LSE-LBFGS continuation with very high betas, interleaved with
    exact subgradient steps, peak-targeted local search, and active-set refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working lengths
    if parent is not None:
        n_parent = parent.size
        print(f"Parent length: {n_parent}, parent score: {compute_score(parent):.8f}")
        # Try parent length and a few nearby
        lengths = [n_parent]
        for dn in [50, 100, 200, -100, -200, 300]:
            nl = n_parent + dn
            if nl >= 512 and nl not in lengths:
                lengths.append(nl)
        lengths.sort()
    else:
        lengths = [4096, 6962, 8192]

    print(f"Lengths to try: {lengths}")
    report(compute_score(parent) if parent is not None else np.inf,
           parent if parent is not None else np.ones(100)*0.5, "initial")

    # ================================================================
    # Phase 1: Quick spectral flattening for diverse starts (~20% time)
    # ================================================================
    sf_deadline = t_start + 0.15 * effective_budget
    print(f"\n--- Phase 1: Spectral Flattening (deadline +{0.15*effective_budget:.0f}s) ---")

    N_fft = 1 << (2 * max(lengths) - 1).bit_length()

    sf_candidates = {}  # (n, init_id) -> (score, a)
    for n in lengths:
        if time.monotonic() > sf_deadline:
            break
        N_n = 1 << (2 * n - 1).bit_length()
        inits = [
            np.ones(n) * 0.5,
            np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
            np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01,
            0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * np.arange(n) / n),
        ]
        if parent is not None and len(parent) == n:
            inits.insert(0, parent.copy())

        for ii, init in enumerate(inits):
            if time.monotonic() > sf_deadline:
                break
            sf_a = spectral_flatten(init, N_n, n_iters=150, lam=0.4)
            sc = compute_score(sf_a)
            sf_candidates[(n, ii)] = (sc, sf_a)
            report(sc, sf_a, f"sf n={n} i={ii}")

    # ================================================================
    # Phase 2: L-BFGS continuation - the main workhorse (~55% time)
    # ================================================================
    lbfgs_deadline = t_start + 0.70 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Continuation (deadline +{0.55*effective_budget:.0f}s) ---")

    # Collect all candidates sorted by score
    all_cands = [(sc, a) for (n, ii), (sc, a) in sf_candidates.items()]
    if parent is not None:
        all_cands.append((compute_score(parent), parent.copy()))
    all_cands.sort(key=lambda x: x[0])
    # Take top candidates
    top_cands = all_cands[:min(8, len(all_cands))]

    # Beta schedule - go much higher than before
    beta_schedule = [50, 100, 300, 800, 2000, 5000, 10000, 20000]

    for ci, (sc_init, a_init) in enumerate(top_cands):
        if time.monotonic() > lbfgs_deadline:
            break
        n = len(a_init)
        print(f"\n  >> Candidate {ci}: n={n}, init_score={sc_init:.8f}")
        a_cur = a_init.copy()
        for beta in beta_schedule:
            if time.monotonic() > lbfgs_deadline:
                break
            remaining = lbfgs_deadline - time.monotonic()
            if remaining < 5:
                break
            # Estimate iterations based on time
            # Each iteration ~ n/1024 * 0.01s for n up to ~8000
            time_per_iter = 0.008 * (n / 1024.0)
            maxiter = min(20000, int(remaining * 0.7 / max(time_per_iter, 0.001)))
            maxiter = max(maxiter, 200)
            maxfun = maxiter * 25

            t0 = time.monotonic()
            sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            report(sc, a_opt, f"lbfgs c={ci} beta={beta}")
            el = time.monotonic() - t0
            print(f"    beta={beta}: score={sc:.8f}, iters={res.niter}, time={el:.1f}s")

            # Interleave subgradient polish every 2 betas
            if beta >= 800 and beta < 20000:
                remaining = lbfgs_deadline - time.monotonic()
                if remaining > 30:
                    sc_sg, a_sg = subgrad_polish(a_cur, n_iters=min(200, int(remaining * 5)))
                    if sc_sg < compute_score(a_cur):
                        a_cur = a_sg
                        report(sc_sg, a_cur, f"subgrad interleave beta={beta}")

        # Final score for this trajectory
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"traject c={ci} final")

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Very high beta refinement (~15% time)
    # ================================================================
    refine_deadline = t_start + 0.85 * effective_budget
    print(f"\n--- Phase 3: High-Beta Refinement ---")

    n_best = len(best_a)
    for beta in [20000, 50000, 100000]:
        if time.monotonic() > refine_deadline:
            break
        remaining = refine_deadline - time.monotonic()
        if remaining < 3:
            break
        time_per_iter = 0.008 * (n_best / 1024.0)
        maxiter = min(30000, int(remaining * 0.8 / max(time_per_iter, 0.001)))
        maxiter = max(maxiter, 500)
        sc, a_opt, res = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxiter*25,
                                    ftol=1e-20, gtol=1e-15)
        report(sc, a_opt, f"refine beta={beta}")
        # Subgradient after each high-beta run
        remaining = refine_deadline - time.monotonic()
        if remaining > 20:
            sc_sg, a_sg = subgrad_polish(a_opt, n_iters=min(300, int(remaining * 5)))
            report(sc_sg, a_sg, f"subgrad after beta={beta}")

    # ================================================================
    # Phase 4: Peak-targeted local search (~10% time)
    # ================================================================
    local_deadline = deadline
    print(f"\n--- Phase 4: Peak-Targeted Local Search ---")

    n_best = len(best_a)
    n_trials = 60
    for trial in range(n_trials):
        if time.monotonic() > local_deadline:
            break
        strategy = trial % 6

        if strategy == 0:
            # Peak-targeted perturbation
            a_pert = peak_targeted_perturb(best_a, rng, strength=0.01 + 0.03 * rng.random())
        elif strategy == 1:
            # Small Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Frequency-domain perturbation
            N_fft_b = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft_b)
            noise_scale = 0.01 + 0.02 * rng.random()
            noise = noise_scale * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]  # Preserve DC
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Chirp mixing
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 6.0 * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            pert = amp * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Multi-peak reduction: reduce top contributors at ALL high peaks
            c = fast_conv(best_a)
            a_pert = best_a.copy()
            top_k_peaks = np.argsort(c)[::-1][:10]
            for ki in top_k_peaks:
                j_start = max(0, ki - n_best + 1)
                j_end = min(ki, n_best - 1)
                for j in range(j_start, j_end + 1):
                    a_pert[j] *= (1.0 - 0.005 * rng.random())
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            # Rescale to maintain similar sum
            S_orig = best_a.sum()
            S_new = a_pert.sum()
            if S_new > 1e-6:
                a_pert *= S_orig / S_new
        else:
            # Spectral re-equalization: flatten the autocorrelation spectrum
            c = fast_conv(best_a)
            target_level = np.median(c)
            # Reduce elements that create high autocorrelation peaks
            k_peak = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_peak - n_best + 1)
            j_end = min(k_peak, n_best - 1)
            # Reduce peak contributors, increase others
            for j in range(j_start, j_end + 1):
                a_pert[j] *= (1.0 - 0.02 * rng.random())
            mask = np.ones(n_best, dtype=bool)
            mask[j_start:j_end+1] = False
            idx = np.where(mask)[0]
            a_pert[idx] *= (1.0 + 0.01 * rng.random())
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick L-BFGS from perturbed
        remaining = local_deadline - time.monotonic()
        if remaining < 3:
            break
        time_per_iter = 0.008 * (n_best / 1024.0)
        maxiter = min(5000, int(remaining * 0.15 / max(time_per_iter, 0.001)))
        maxiter = max(maxiter, 300)
        sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=maxiter*25,
                                  ftol=1e-18, gtol=1e-14)
        report(sc, a_opt, f"local#{trial} s={strategy}")

        # If improved, do a subgradient polish
        if sc < best_score + 1e-8 and time.monotonic() < local_deadline - 5:
            sc_sg, a_sg = subgrad_polish(a_opt, n_iters=100)
            report(sc_sg, a_sg, f"local#{trial} sg")

    # ================================================================
    # Phase 5: Final polish
    # ================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Multi-peak subgradient
    sc, a_mp = multi_peak_subgrad(best_a, n_iters=300)
    report(sc, a_mp, "multi-peak sg")

    # Standard subgradient
    sc, a_sg = subgrad_polish(best_a, n_iters=500, start_lr=0.005)
    report(sc, a_sg, "final subgrad")

    # One more L-BFGS at extreme beta from the best
    if time.monotonic() < deadline - 10:
        for beta in [100000, 300000]:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000,
                                      ftol=1e-20, gtol=1e-15)
            report(sc, a_opt, f"extreme beta={beta}")
            sc_sg, a_sg = subgrad_polish(a_opt, n_iters=200)
            report(sc_sg, a_sg, f"extreme beta={beta} + sg")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
