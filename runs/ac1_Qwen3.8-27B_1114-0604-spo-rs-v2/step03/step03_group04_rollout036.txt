        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                 'factr': 1e12}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res


def peak_reducing_gradient(a, top_k=10, beta_smooth=50):
    """
    Compute a gradient that targets the top-K peaks.
    This is a smoothed subgradient that spreads the push across the top peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    S2 = S * S
    
    # Sort peak values
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(top_k, len(c_sorted))-1]
    
    # Weight each lag by its "excess" over threshold, smoothed
    excess = np.maximum(c - threshold, 0)
    # Normalize to get a probability distribution
    w_sum = float(excess.sum())
    if w_sum < 1e-15:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w = excess / w_sum
    
    # grad[j] = (1/S2) * sum_k w[k] * 2*a[k-j] - (2*max_c / S3)
    # Use FFT for efficiency
    M_xf = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, M_xf)
    A = np.fft.rfft(a, M_xf)
    cross = np.fft.irfft(W * np.conj(A), M_xf)
    grad_M = (2.0 / S2) * cross[:n]
    
    M = float(np.max(c))
    S3 = S2 * S
    grad = (2.0 * n) * (grad_M - 2.0 * M * np.ones(n) / S3)
    return grad, M


def projected_step(a, grad, step, S_target):
    """Take a gradient step and project onto {a >= 0, sum a = S_target}."""
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s > 1e-15:
        a_new = a_new * (S_target / s)
    return a_new


def adaptive_subgradient(a, max_iters=5000, top_k=20):
    """
    Adaptive subgradient descent targeting top-K peaks.
    Uses backtracking line search.
    """
    n = len(a)
    S = float(a.sum())
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Adam-like momentum
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        grad, M = peak_reducing_gradient(a, top_k=top_k)
        
        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive step size
        sc = 2.0 * n * M / (S * S)
        sc_lb = 2.0 * n / ((2 * n - 1) * S * S)  # theoretical lower bound
        gap = max(sc - sc_lb, 1e-10)
        lr = 0.1 * gap / (float(np.linalg.norm(m_hat)) + 1e-10)
        lr = min(lr, 0.5 / n)
        
        a_new = a - lr * m_hat
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)
        
        # Check improvement
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_sc, best_a


def spectral_flatten_v2(a0, N, n_iters=300, lam_start=0.8, lam_end=0.1):
    """
    Improved spectral flattening:
    - Uses mean (not median) as target
    - Annealing from strong to weak
    - Preserves DC component exactly
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_target = float(a.sum())
    
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start * (1 - progress) + lam_end * progress
        
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean of non-DC magnitudes
        non_dc_mag = mag[1:]
        target = float(np.exp(np.mean(np.log(non_dc_mag + 1e-30))))
        
        new_mag = mag.copy()
        half = N // 2
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        
        s = float(a.sum())
        if s > 1e-12:
            a = a * (S_target / s)
    
    return a


def swap_local_search(a, n_swaps=50000, rng_np=None):
    """
    Local search via random mass transfers.
    Transfer small amounts from high-contributing positions to low-contributing ones.
    """
    if rng_np is None:
        rng_np = np.random.default_rng(42)
    
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    current_max = float(np.max(c))
    
    for it in range(n_swaps):
        # Pick a random position to reduce and a random position to increase
        i = rng_np.integers(0, n)
        j = rng_np.integers(0, n)
        if i == j or a[i] < 1e-10:
            continue
        
        # Try transferring a small amount from i to j
        delta = min(a[i] * 0.1, 0.01 * S / n)
        
        a[i] -= delta
        a[j] += delta
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            c = fast_conv(a)
            current_max = float(np.max(c))
        else:
            # Revert
            a[i] += delta
            a[j] -= delta
    
    return best_sc, best_a


def targeted_perturb(a, rng_np, n_peaks=5, strength=0.03):
    """Perturb positions that contribute most to the top peaks."""
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:n_peaks]
    
    a_pert = a.copy()
    for k in top_lags:
        # Positions contributing to lag k
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            contrib = a[j_start:j_end+1][::-1]
            # Reduce top contributors
            top_idx_in_range = np.argsort(contrib)[::-1][:max(1, len(contrib)//5)]
            for idx in top_idx_in_range:
                actual_idx = j_end - idx  # because of the reverse
                a_pert[actual_idx] *= (1 - strength)
    
    # Redistribute
    S = float(a.sum())
    S_pert = float(a_pert.sum())
    if S_pert > 1e-15:
        # Add back to low positions
        deficit = S - S_pert
        low_positions = np.argsort(a_pert)[:n//10]
        per = deficit / len(low_positions)
        a_pert[low_positions] += per
    
    return np.clip(a_pert, 0.0, 1000.0)

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Uses: (1) diverse spectral initialization, (2) L-BFGS with LSE smoothing,
(3) IPF-style multiplicative peak flattening, (4) adaptive subgradient with momentum,
(5) swap-based local search, (6) high-beta L-BFGS polish.
Key innovation: IPF multiplicative update that directly targets convolution peaks,
combined with multi-start perturbation strategy.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Fast full linear convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed max convolution objective and its gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 0)
    exp_scaled = np.exp(scaled)
    p_sum = float(exp_scaled.sum())
    L = c_max + np.log(p_sum) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient computation
    p = exp_scaled / p_sum
    # cross[ell] = sum_m p[ell+m]*a[m] for ell=0..n-1
    M_xf = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, M_xf)
    A = np.fft.rfft(a, M_xf)
    cross = np.fft.irfft(P * np.conj(A), M_xf)
    # dL/da[ell] = (2/beta) * cross[ell]
    # (factor of 2 because d(conv[k])/da[j] = 2*a[k-j] for valid k-j)
    grad_L = (2.0 / beta) * cross[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * np.ones(n) / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def ipf_flatten(a0, n_iters=2000, top_fraction=0.1, damping=0.5):
    """
    Iterative Proportional Fitting for flattening the autocorrelation peak.
    Multiplicative update: reduce a[j] proportional to its contribution to top peaks.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S_target = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)

    top_k = max(5, int(n * top_fraction * 2))  # Number of top lags to target

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(np.max(c))
        sc = 2.0 * n * M / (S_target * S_target)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find the threshold for "top" peaks
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(top_k - 1, len(c_sorted) - 1)]

        # Weight: how much each lag exceeds the threshold
        excess = np.maximum(c - threshold, 0.0)
        w_sum = float(excess.sum())
        if w_sum < 1e-20:
            break
        w = excess / w_sum

        # Responsibility of each a[j] for the excess peaks:
        # resp[j] = sum_k w[k] * 2*a[k-j] (for valid k-j)
        # This is 2 * cross-correlation of w and a
        M_xf = 1 << (3 * n - 1).bit_length()
        W = np.fft.rfft(w, M_xf)
        A = np.fft.rfft(a, M_xf)
        cross = np.fft.irfft(W * np.conj(A), M_xf)
        resp = 2.0 * cross[:n]

        # Multiplicative update: a[j] *= (threshold_target / resp[j])^damping
        # Where threshold_target is the average excess per unit responsibility
        avg_resp = float(resp.sum()) / n
        ratio = np.where(resp > 1e-20, avg_resp / resp, 1.0)
        # Clip the ratio to avoid extreme changes
        ratio = np.clip(ratio, 0.5, 2.0)
        update = ratio ** damping
        a = a * update

        # Project onto {a >= 0, sum a = S_target}
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s > 1e-15:
            a = a * (S_target / s)
        else:
            break

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def adaptive_momentum_optimize(a0, max_iters=10000, top_k=20, lr_base=0.05):
    """
    Adaptive momentum (Adam-like) optimization targeting top-K convolution peaks.
    Uses the smoothed subgradient with top-K peak weighting.
    """
    n = len(a0)
    S = float(a0.sum())
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    for it in range(max_iters):
        c = fast_conv(a)
        M = float(np.max(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Build weighted subgradient targeting top-K peaks
        c_sorted = np.sort(c)[::-1]
        k_actual = min(top_k, len(c_sorted))
        threshold = c_sorted[k_actual - 1]
        excess = np.maximum(c - threshold, 0.0)
        w_sum = float(excess.sum())
        if w_sum < 1e-20:
            w = np.full(len(c), 1.0 / len(c))
        else:
            w = excess / w_sum

        # Gradient of weighted max: grad[j] = 2 * sum_k w[k] * a[k-j]
        M_xf = 1 << (3 * n - 1).bit_length()
        W = np.fft.rfft(w, M_xf)
        A = np.fft.rfft(a, M_xf)
        cross = np.fft.irfft(W * np.conj(A), M_xf)
        grad_M = 2.0 * cross[:n] / (S * S)

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_M - 2.0 * M * np.ones(n) / S3)

        # Adam update
        t = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        # Adaptive step size based on gap to lower bound
        sc_lb = 2.0 * n / ((2 * n - 1) * S * S)
        gap = max(sc - sc_lb, 1e-10)
        norm_grad = float(np.linalg.norm(m_hat))
        lr = lr_base * gap / (norm_grad + 1e-10)
        lr = min(lr, 1.0 / (n * 0.1))

        a = a - lr * m_hat
        a = np.maximum(a, 0.0)
        s_new = float(a.sum())
        if s_new > 1e-15:
            a = a * (S / s_new)

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def spectral_flatten_v2(a0, N, n_iters=300, lam_start=0.9, lam_end=0.15):
    """
    Spectral flattening with geometric mean target and annealing.
    """
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_target = float(a.sum())

    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start * (1 - progress) + lam_end * progress

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Geometric mean of non-DC magnitudes as target
        non_dc_mag = mag[1:]
        target = float(np.exp(np.mean(np.log(non_dc_mag + 1e-30))))

        new_mag = mag.copy()
        half = N // 2
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]

        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)

        s = float(a.sum())
        if s > 1e-12:
            a = a * (S_target / s)

    return a


def swap_local_search(a, n_swaps, rng_np):
    """
    Local search via random mass transfers between positions.
    Accepts a move only if it reduces the score.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    accepted = 0

    for _ in range(n_swaps):
        i = int(rng_np.integers(0, n))
        j = int(rng_np.integers(0, n))
        if i == j or a[i] < 1e-10:
            continue

        # Random delta (positive or negative transfer)
        max_delta = min(a[i], a[j]) * 0.2
        delta = float(rng_np.uniform(-max_delta, max_delta))
        if abs(delta) < 1e-12:
            continue

        # Ensure non-negativity after swap
        if delta > 0 and a[i] < delta:
            delta = a[i] * 0.5
        if delta < 0 and a[j] < -delta:
            delta = -a[j] * 0.5

        a[i] -= delta
        a[j] += delta

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            accepted += 1
        else:
            a[i] += delta
            a[j] -= delta

    return best_sc, best_a


def targeted_perturb(a, rng_np, n_peaks=5, strength=0.04):
    """Perturb positions contributing most to top peaks, redistribute to low contributors."""
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:n_peaks]
    S = float(a.sum())

    a_pert = a.copy()
    total_reduced = 0.0

    for k in top_lags:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            seg = a_pert[j_start:j_end + 1][::-1]
            top_count = max(2, len(seg) // 10)
            top_indices = np.argsort(seg)[::-1][:top_count]
            for idx in top_indices:
                actual_idx = j_end - idx
                reduce_amt = a_pert[actual_idx] * strength
                a_pert[actual_idx] -= reduce_amt
                a_pert[actual_idx] = max(a_pert[actual_idx], 0.0)
                total_reduced += reduce_amt

    # Redistribute to low positions
    if total_reduced > 1e-15:
        low_count = max(10, n // 20)
        low_positions = np.argsort(a_pert)[:low_count]
        per = total_reduced / len(low_positions)
        for pos in low_positions:
            a_pert[pos] += per

    return np.clip(a_pert, 0.0, 1000.0)
