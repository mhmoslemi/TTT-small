    # And sum_k p[k] * a[i-k]:
    # = sum_k p[k] * a[i-k] (valid when 0≤i-k<n)
    # Let j = i-k, so k = i-j:
    # = sum_j p[i-j] * a[j] (valid when 0≤j<n and 0≤i-j≤2n-2)
    # = conv(p, a)[i]
    
    # So G[i] = conv(flip(p), a)[2n-2-i] + conv(p, a)[i]
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    
    # conv(p, a)
    pa = np.fft.irfft(P * A, M_fft)[:M]
    
    # conv(flip(p), a)
    p_flip = p[::-1]
    PF = np.fft.rfft(p_flip, M_fft)
    pflip_a = np.fft.irfft(PF * A, M_fft)[:M]
    
    # G[i] = pflip_a[M-1-i] + pa[i]
    G = pflip_a[M-1::-1] + pa  # pflip_a[M-1-i] for i=0..M-1, reversed
    
    # Wait, G[i] = pflip_a[M-1-i] + pa[i] for i = 0, ..., n-1
    # But pa has length M = 2n-1, and we only need i = 0..n-1
    # And pflip_a has length M, we need indices M-1-i for i=0..n-1, which is M-1, M-2, ..., M-n
    
    G = pflip_a[M-1:M-n:-1] + pa[0:n]
    # Hmm, M-1:M-n:-1 gives indices M-1, M-2, ..., M-n (that's n elements)
    # Wait: M-1, M-2, ..., M-n. That's n elements. Good.
    # But we need pflip_a[M-1-i] for i=0 to n-1:
    # i=0: pflip_a[M-1]
    # i=1: pflip_a[M-2]
    # ...
    # i=n-1: pflip_a[M-n]
    # So it's pflip_a[M-n:M] reversed = pflip_a[M-1:M-n:-1]
    # In numpy: pflip_a[M-1:M-n-1:-1] or pflip_a[M-1::-1][:n]
    
    G = pflip_a[M-1::-1][:n] + pa[:n]
    
    # Now the gradient of the score:
    # score = 2n * L / S^2
    # d(score)/d(a[i]) = (2n/S^2) * G[i] - 2*(2n*L/S^3) * 1
    # = (2n/S^2) * (G[i] - 2L/S)
    
    grad = (2.0 * n / (S * S)) * (G - 2.0 * L / S)
    
    return score, grad


def spectral_flatten_soft(a0, N, n_iters=300, lam=0.5, soft_eps=0.01):
    """Spectral flattening with soft non-negativity enforcement."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    target_sum = a.sum()
    
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        
        new_mag = mag.copy()
        half = N // 2
        decay = (1.0 - i / max(1, n_iters)) * lam + 0.1
        for k in range(1, N):
            if k == half and N % 2 == 0:
                continue
            nk = N - k if k > half else k
            if nk == 0:
                continue
            new_mag[k] = (1.0 - decay) * mag[k] + decay * target
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        # Soft non-negativity: use exp reparameterization implicitly
        a = np.maximum(a_new, 0.0)
        a = a + soft_eps  # small floor to avoid exact zeros
        S = a.sum()
        if S > 1e-12:
            a = a * (target_sum / S)
    
    return a


def subgradient_step(a, top_k=5, step_scale=1.0):
    """
    One subgradient step: reduce contributions to the top-k worst lags.
    Returns updated a and the score.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return a, np.inf
    
    # Find top-k lags
    top_lags = np.argsort(c)[::-1][:top_k]
    
    # Subgradient of max(c) at the argmax lag k*:
    # dc[k*]/da[i] = a[k*-i] + a[i-k*] (with bounds)
    # For the LSE-softened version with top-k:
    M_max = c[top_lags[0]]
    weights = np.zeros(2*n-1)
    for idx in top_lags:
        weights[idx] = c[idx] / M_max  # weight proportional to value
    
    # Compute subgradient
    M = 2*n - 1
    M_fft = 1 << (M - 1).bit_length()
    P = np.fft.rfft(weights, M_fft)
    A = np.fft.rfft(a, M_fft)
    pa = np.fft.irfft(P * A, M_fft)[:M]
    p_flip = weights[::-1]
    PF = np.fft.rfft(p_flip, M_fft)
    pflip_a = np.fft.irfft(PF * A, M_fft)[:M]
    G = pflip_a[M-1::-1][:n] + pa[:n]
    
    # Gradient of the objective
    L = M_max  # Using the max (exact subgradient)
    grad = (2.0 * n / (S * S)) * (G - 2.0 * L / S)
    
    # Normalized step
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return a, compute_score(a)
    
    # Step size: use (score - lower_bound) / ||grad||^2 as in昏暗 projection
    sc = 2.0 * n * M_max / (S * S)
    lower_bound = 1.0  # Theoretical lower bound
    step = step_scale * (sc - lower_bound) / (gn * gn + 1e-30)
    step = min(step, 0.1)  # Cap step size
    
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    # Renormalize to keep sum fixed (optional, since objective is scale-invariant)
    # Actually, since the objective is scale-invariant, we don't need to.
    # But keeping the sum roughly constant helps numerical stability.
    
    return a_new, compute_score(a_new)


def lag_aware_perturb(a, rng, n_lags=3):
    """Perturb a by reducing elements that contribute to the worst lags and increasing others."""
    n = len(a)
    c = fast_conv(a)
    # Find worst lags
    worst_lags = np.argsort(c)[::-1][:n_lags]
    
    # For each worst lag k, the contributing pairs are (i, k-i) for valid i
    # Compute contribution of each element to the worst lags
    contrib = np.zeros(n)
    for k in worst_lags:
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[i] = a[k-i] + a[i-k] (with bounds)
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                contrib[i] += a[ki]
            ik = i - k
            if 0 <= ik < n:
                contrib[i] += a[ik]
    
    # Reduce high-contributing elements, increase low-contributing ones
    a_pert = a.copy()
    threshold_high = np.percentile(contrib, 80)
    threshold_low = np.percentile(contrib, 20)
    
    high_idx = contrib > threshold_high
    low_idx = contrib < threshold_low
    
    total_reduced = 0.0
    strength = 0.03 + 0.05 * rng.random()
    for i in np.where(high_idx)[0]:
        reduce_amt = a_pert[i] * strength
        a_pert[i] -= reduce_amt
        a_pert[i] = max(a_pert[i], 0.0)
        total_reduced += reduce_amt
    
    low_positions = np.where(low_idx)[0]
    if len(low_positions) > 0 and total_reduced > 0:
        per = total_reduced / len(low_positions)
        for i in low_positions:
            a_pert[i] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def objective(a):
        return lse_score_and_grad(a, beta)
    
    res = minimize(
        objective, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def chirp_init(n, rng):
    """Generate chirp-based initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    freq = 1.0 + 5.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    inits.append(0.5 + 0.3 * np.cos(np.pi * x * x * x / (2.0 * n * n)))
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits
