"""
Improved optimizer for minimum peak autocorrelation of non-negative sequences.
Uses Adam with LSE smoothing, projected gradient descent, multi-scale perturbation,
and coordinate descent. Different from previous approach by using Adam instead of
L-BFGS, adding projected gradient steps, and employing frequency-domain perturbations.

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def adam_optimize(a0, beta, lr_start, lr_end, n_iter, t_check=200):
    """Adam optimizer for LSE objective with LR scheduling."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    best_local = compute_score(a)
    for t in range(n_iter):
        if (t + 1) % t_check == 0:
            sc = compute_score(a)
            if sc < best_local:
                best_local = sc
        lr = lr_start * (lr_end / lr_start) ** (t / max(1, n_iter - 1))
        score, grad = lse_obj_grad(a, beta)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - update
        a = np.clip(a, 0.0, 1000.0)
    return a


def projected_gradient_steps(a0, beta, n_steps, lr_init=0.005, t_check=100):
    """Projected gradient descent with backtracking line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    sc0, _ = lse_obj_grad(a, beta)
    best_local = compute_score(a)
    lr = lr_init
    for t in range(n_steps):
        if (t + 1) % t_check == 0:
            sc = compute_score(a)
            if sc < best_local:
                best_local = sc
            lr = max(lr * 0.95, lr_init * 0.01)
        score, grad = lse_obj_grad(a, beta)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new, _ = lse_obj_grad(a_new, beta)
        if sc_new > score * 1.01 + 1e-10:
            lr *= 0.5
        else:
            a = a_new
    return a


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, n_sweeps=1):
    """Coordinate descent sweep with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for idx in range(n):
            i = int(order[idx])
            a_i = a[i]
            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-7}
            )
            t_opt = float(res.x)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return compute_score(a), a

def freq_domain_perturb(a, rng, num_bands=8, strength=0.1):
    """Perturb in frequency domain for structured perturbations."""
    n = len(a)
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Perturb magnitude in bands
    bin_size = max(1, len(A) // num_bands)
    new_mag = mag.copy()
    for b in range(num_bands):
        lo = b * bin_size
        hi = min(len(A), (b + 1) * bin_size)
        scale = 1.0 + strength * (2.0 * rng.random() - 1.0)
        new_mag[lo:hi] *= scale
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def multi_scale_perturb(a, rng):
    """Perturb at multiple frequency scales simultaneously."""
    n = len(a)
    a_pert = a.copy()
    for scale, strength in [(0.5, 0.03), (2.0, 0.05), (8.0, 0.07), (32.0, 0.04)]:
        freq = scale
        amp = strength
        x = np.arange(n, dtype=np.float64) / n
        phase = 2.0 * np.pi * rng.random()
        mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
        a_pert = a_pert * mod
    return np.clip(a_pert, 0.0, 1000.0)


def momentum_steps(a0, beta, n_steps, mu=0.9, lr=0.003, t_check=200):
    """Heavy-ball momentum method for LSE objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    v = np.zeros(n)
    best_local = compute_score(a)
    for t in range(n_steps):
        if (t + 1) % t_check == 0:
            sc = compute_score(a)
            if sc < best_local:
                best_local = sc
            lr = max(lr * 0.97, 1e-5)
        score, grad = lse_obj_grad(a, beta)
        v = mu * v - lr * grad
        a = a + v
        a = np.clip(a, 0.0, 1000.0)
    return a


def block_cd_sweep(a, rng, block_size=10):
    """Block coordinate descent: optimize blocks of coordinates together."""
    n = len(a)
    a = a.astype(np.float64).copy()
    n_blocks = (n + block_size - 1) // block_size
    order = rng.permutation(n_blocks)

    for b_idx in order:
        start = int(b_idx) * block_size
        end = min(n, start + block_size)
        B = end - start
        if B < 2:
            continue

        # Get current values of the block
        a_block = a[start:end].copy()
        if a_block.sum() < 1e-12:
            a_block = np.ones(B) * 0.01

        # Define sub-objective: LSE score with only block variables free
        def block_obj(x):
            a_trial = a.copy()
            a_trial[start:end] = np.clip(x, 0.0, 1000.0)
            sc, _ = lse_obj_grad(a_trial, 200.0)
            return sc, _

        # Optimize the block using L-BFGS-B
        bounds = [(0.0, 1000.0)] * B
        try:
            res = minimize(
                block_obj, a_block,
                method='L-BFGS-B', jac=True,
                bounds=bounds,
                options={'maxiter': 30, 'maxfun': 200, 'ftol': 1e-12}
            )
            a[start:end] = np.clip(res.x, 0.0, 1000.0)
        except Exception:
            pass

    return compute_score(a), a

        # CD from perturbed point (4-6 sweeps)
        n_cd = 4 + int(2 * rng.random())
        for sw in range(n_cd):
            if time.monotonic() > phase4_end - 5.0:
                break
            if sw % 2 == 0:
                sc_p, a_pert = hot_spot_sweep(a_pert, n_sweeps=1)
            else:
                sc_p, a_pert = coordinate_descent_sweep(a_pert, rng=rng)

        sc_final = compute_score(a_pert)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_final:.8f} (NEW BEST)")
        elif trial % 8 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_final:.8f}")

    # ============================================================
    # Phase 5: Block CD (10% of time)
    # ============================================================
    phase5_end = t_start + 0.80 * total_budget
    print(f"\n=== Phase 5: Block CD ===")
    a_blk = best_a.copy()
    for bs in [8, 15]:
        if time.monotonic() > phase5_end:
            break
        t0 = time.monotonic()
        sc_blk, a_blk = block_cd_sweep(a_blk, rng, block_size=bs)
        elapsed = time.monotonic() - t0
        if sc_blk < best_score:
            best_score = sc_blk
            best_a = a_blk.copy()
            print(f"  BlockCD bs={bs}: {sc_blk:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  BlockCD bs={bs}: {sc_blk:.8f} ({elapsed:.1f}s)")

    # ============================================================
    # Phase 6: Final CD + Adam polish (10% of time)
    # ============================================================
    phase6_end = deadline
    print(f"\n=== Phase 6: Final Polish ===")
    a_fin = best_a.copy()

    # More CD sweeps
    for sweep in range(40):
        if time.monotonic() > phase6_end - 30.0:
            break
        if sweep % 4 == 0:
            sc, a_fin = hot_spot_sweep(a_fin, n_sweeps=1)
        else:
            sc, a_fin = coordinate_descent_sweep(a_fin, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_fin.copy()
            if sweep % 8 == 0:
                print(f"  P6 sweep {sweep}: {sc:.8f} (NEW BEST)")

    # Final Adam at low beta
    if time.monotonic() < phase6_end - 30.0:
        a_adam2 = best_a.copy()
        a_adam2 = adam_optimize(a_adam2, 50, lr_start=0.003, lr_end=0.0005, n_iter=200, t_check=100)
        sc_a2 = compute_score(a_adam2)
        if sc_a2 < best_score:
            best_score = sc_a2
            best_a = a_adam2.copy()
            print(f"  Final Adam: {sc_a2:.8f} (NEW BEST)")
        # CD after
        sc_cd2, a_adam2 = coordinate_descent_sweep(a_adam2, rng=rng, n_sweeps=3)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_a = a_adam2.copy()
            print(f"  Final CD: {sc_cd2:.8f} (NEW BEST)")

    # Very final CD
    a_fin = best_a.copy()
    for _ in range(5):
        if time.monotonic() > deadline - 5.0:
            break
        sc, a_fin = coordinate_descent_sweep(a_fin, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_fin.copy()

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
