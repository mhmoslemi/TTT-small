    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500.0, 0.0))
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        L = c_max
        p = np.full(len(c), 1.0 / len(c))
    else:
        L = c_max + np.log(exp_sum) / beta
        p = exp_scaled / exp_sum
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def multiplicative_weights(a, total_iters=20000, eta0=0.05, eta_final=1e-5, deadline=None):
    """Multiplicative weights (mirror descent) on the peak objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = float(2.0 * n * float(np.max(fast_conv(a))))
    
    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Learning rate schedule (geometric)
        progress = it / max(1, total_iters - 1)
        eta = eta0 * (eta_final / eta0) ** progress
        
        # Multiplicative update
        update = -eta * 2.0 * n * sg
        update = np.clip(update, -10.0, 10.0)  # prevent overflow
        a = a * np.exp(update)
        
        # Renormalize
        S = float(a.sum())
        if S < 1e-30:
            a = np.ones(n) / n
        else:
            a = a / S
    
    return best_sc, best_a

def spectral_perturb(a, rng, delta=0.1):
    """Perturb the phase of the FFT while keeping magnitude approximately constant."""
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, N)
    # Add random phase perturbation
    phase_noise = delta * rng.standard_normal(len(A))
    # Apply: multiply by exp(i * phase_noise)
    A_pert = A * np.exp(1j * phase_noise)
    # Keep DC component
    A_pert[0] = A[0]
    a_pert = np.real(np.fft.ifft(A_pert))[:n]
    return np.clip(a_pert, 0.0, None)

def lbfgs_lse(a, beta, maxiter=5000, deadline=None):
    """L-BFGS on LSE-smoothed objective."""
    n = len(a)
    a0 = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        if deadline is not None and time.monotonic() > deadline:
            return 1e10, np.zeros(n)
        return lse_score_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def subgrad_polish(a, total_iters=2000, deadline=None):
    """Subgradient descent with Nesterov momentum on the peak objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 0.5
        S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = float(2.0 * n * float(np.max(fast_conv(a))))
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    
    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Gradient of F = 2n*M (with sum=1)
        # We need to project onto the tangent space (sum of gradient = 0)
        g = 2.0 * n * sg
        g = g - np.mean(g)  # project to sum(g) = 0
        
        # Nesterov momentum
        v = mu * v + (1 - mu) * g
        step = 0.01 / (1.0 + 0.01 * it)
        a = a - step * v
        a = np.clip(a, 1e-10, None)
        a = a / a.sum()
    
    return best_sc, best_a
