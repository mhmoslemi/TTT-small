def fast_conv(a):

    N = 1 << (M - 1).bit_length()

        p_full = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j]
    # Computed via FFT cross-correlation
    N3 = 1 << (3 * n - 2).bit_length()
    P = np.fft.fft(p_full, N3)
    A = np.fft.fft(a, N3)
    R = np.fft.ifft(P * np.conj(A)).real
    g = 2.0 * R[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, topK, maxiter, maxfun):
    """Run L-BFGS-B with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        sc, gr = lse_score_grad(x, beta, topK)
        return sc, gr
    
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def spectral_flatten_conv(a, n_iters=30, gamma=0.3):
    """
    Spectral-domain iterative flattening of the self-convolution.
    Uses multiplicative update: A_new = A * (C_target/C)^(gamma/2)
    """
    n = len(a)
    S = float(a.sum())
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    
    target_val = S * S / M
    c_target_seq = np.zeros(N)
    c_target_seq[:M] = target_val
    C_target = np.fft.fft(c_target_seq)
    
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.fft(a, N)
        C = A * A
        # Avoid division by zero
        C_safe = C + 1e-30 * (abs(C).max() + 1e-30)
        ratio = C_target / C_safe
        A_new = A * np.power(ratio, gamma / 2.0)
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new > 1e-12:
            a_new = a_new * (S / S_new)
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc


def spectral_flatten_spectrum(a, n_iters=200, lam=0.5):
    """
    Spectral flattening of the input signal's magnitude spectrum.
    Makes |A[k]| more uniform (excludes DC and Nyquist).
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = mag[1:]
        target = float(np.median(non_dc))
        new_mag = mag.copy()
        half = N // 2
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S > 1e-12:
            a = a * (S / (S))  # maintain sum
    return a


def subgradient_polish(a, total_iters=500):
    """Subgradient method on the exact max objective with Polyak step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lower_bound = 2.0 * n / (2 * n - 1)
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max at k_star: sg[j] = 2*a[k_star - j] (if in bounds)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Small additional step for progress
        a = a - (0.0005 / (1.0 + 0.05 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


def multi_lag_subgrad(a, total_iters=300, topK=5):
    """Subgradient using average of top-K active lags."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lower_bound = 2.0 * n / (2 * n - 1)
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        # Get top-K lags
        k_idx = np.argpartition(c, -topK)[-topK:]
        # Only use lags within 1% of max
        threshold = M_val * 0.99
        k_active = k_idx[c[k_idx] >= threshold]
        if len(k_active) == 0:
            k_active = k_idx[:1]
        
        # Average subgradient over active lags
        sg_M = np.zeros(n)
        for k in k_active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg_M[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg_M /= len(k_active)
        
        # L = average of active c values
        L = float(c[k_active].mean())
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg_M / S2 - 2.0 * L / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        sc = 2.0 * n * M_val / (S * S)
        step = (sc - lower_bound) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        a = a - (0.0003 / (1.0 + 0.05 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
        
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
    
    return best_sc, best_a


def generate_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        inits.append(parent.astype(np.float64).copy())
    
    x = np.arange(n, dtype=np.float64)
    xn = x / n
    pi = np.pi
    
    # Constant
    inits.append(np.ones(n) * 0.5)
    
    # Chirp-based
    inits.append(np.cos(pi * x * x / (2.0 * n)) ** 2 + 0.01)
    inits.append(0.5 + 0.4 * np.cos(pi * 2.0 * x * x / (2.0 * n)))
    inits.append(0.5 + 0.3 * np.cos(pi * x * x / n) * np.cos(2 * pi * 3 * xn))
    
    # Cosine based
    for freq in [2, 5, 8, 13]:
        inits.append(0.5 + 0.3 * np.cos(2 * pi * freq * xn))
        inits.append(0.5 + 0.3 * np.sin(2 * pi * freq * xn))
    
    # Parabolic
    inits.append(0.1 + 0.9 * (2.0 * xn - 1.0) ** 2)
    inits.append(0.5 + 0.4 * (1.0 - (2.0 * xn - 1.0) ** 2))
    
    # Random
    for scale in [0.1, 0.3, 0.5]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, 10.0))
    
    # Spectral flattening from a few starts
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    inits.append(spectral_flatten_spectrum(np.ones(n) * 0.5, n_iters=100, lam=0.5))
    inits.append(spectral_flatten_spectrum(
        np.cos(pi * x * x / (2.0 * n)) ** 2 + 0.01, n_iters=100, lam=0.4))
    
    # Spectral convolution flattening
    sf_a, _ = spectral_flatten_conv(np.ones(n) * 0.5, n_iters=20, gamma=0.3)
    inits.append(sf_a)
    sf_a2, _ = spectral_flatten_conv(
        np.cos(pi * x * x / (2.0 * n)) ** 2 + 0.01, n_iters=20, gamma=0.3)
    inits.append(sf_a2)
    
    for i in range(len(inits)):
        inits[i] = np.clip(inits[i], 0.0, 1000.0)
        if inits[i].sum() < 0.01:
            inits[i] = np.ones(n) * 0.5
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(convolve(a,a))/(sum(a))^2.
    Uses spectral-domain flattening, adaptive top-K LSE, L-BFGS continuation,
    multi-lag subgradient polish, and diverse local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent[parent.sum() > 0.01]
    except Exception:
        pass
    
    if parent is not None:
        n = len(parent)
    else:
        n = 8192
    
    print(f"Using n = {n}")
    if parent is not None:
        parent_sc = compute_score(parent)
        print(f"Parent score: {parent_sc:.8f}")
    
    best_score = compute_score(parent) if parent is not None else np.inf
    best_a = parent.copy() if parent is not None else np.ones(n) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (t={elapsed:.0f}s)")
            return True
        return False
    
    # ================================================================
    # Phase 1: Generate and evaluate diverse initializations (10%)
    # ================================================================
    p1_deadline = t_start + 0.10 * effective_budget
    print(f"\n=== Phase 1: Initialization (deadline +{0.10*effective_budget:.0f}s) ===")
    
    inits = generate_inits(n, rng, parent)
    print(f"Generated {len(inits)} initializations")
    
    scored_inits = []
    for i, a_init in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        sc = compute_score(a_init)
        scored_inits.append((sc, a_init))
        report(sc, a_init, f"init_{i}")
    
    scored_inits.sort(key=lambda x: x[0])
    print(f"Top init: {scored_inits[0][0]:.8f}")
    
    # ================================================================
    # Phase 2: Coarse L-BFGS continuation (30%)
    # ================================================================
    p2_deadline = t_start + 0.35 * effective_budget
    print(f"\n=== Phase 2: Coarse L-BFGS (deadline +{0.35*effective_budget:.0f}s) ===")
    
    # Take top 4 initializations
    top_inits = scored_inits[:min(4, len(scored_inits))]
    
    beta_schedule_coarse = [(50, 0), (100, 0), (200, 10), (500, 20)]
    
    for ii, (sc_init, a_init) in enumerate(top_inits):
        if time.monotonic() > p2_deadline:
            break
        t0 = time.monotonic()
        a_cur = a_init.copy()
        for beta, topK in beta_schedule_coarse:
            if time.monotonic() > p2_deadline:
                break
            remaining = p2_deadline - time.monotonic()
            if remaining < 10:
                break
            maxiter = min(3000, int(remaining * 0.6 / 0.05))
            maxfun = max(1000, maxiter * 20)
            sc, a_opt = run_lbfgs(a_cur, beta, topK, maxiter, maxfun)
            a_cur = a_opt
            report(sc, a_opt, f"p2 ii={ii} beta={beta} topK={topK}")
        
        # Spectral convolution flattening between L-BFGS stages
        if time.monotonic() < p2_deadline - 10:
            a_sf, sc_sf = spectral_flatten_conv(a_cur, n_iters=30, gamma=0.2)
            report(sc_sf, a_sf, f"p2_sf ii={ii}")
            if sc_sf < compute_score(a_cur):
                a_cur = a_sf
            # Quick L-BFGS from spectral result
            if time.monotonic() < p2_deadline - 5:
                sc, a_opt = run_lbfgs(a_cur, 500, 20, min(2000, int((p2_deadline - time.monotonic()) * 10)), 50000)
                report(sc, a_opt, f"p2_sf_lse ii={ii}")
        
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"p2_final ii={ii}")
        elapsed = time.monotonic() - t0
        print(f"  ii={ii}: {sc_final:.8f} ({elapsed:.1f}s)")
    
    # ================================================================
    # Phase 3: Fine L-BFGS (25%)
    # ================================================================
    p3_deadline = t_start + 0.65 * effective_budget
    print(f"\n=== Phase 3: Fine L-BFGS (deadline +{0.65*effective_budget:.0f}s) ===")
    print(f"  Starting from: {best_score:.10f}")
    
    beta_schedule_fine = [(1000, 30), (2000, 30), (5000, 50), (10000, 100), (20000, 200)]
    
    for beta, topK in beta_schedule_fine:
        if time.monotonic() > p3_deadline:
            break
        remaining = p3_deadline - time.monotonic()
        if remaining < 15:
            break
        maxiter = min(5000, int(remaining * 0.7 / 0.08))
        maxfun = max(2000, maxiter * 25)
        sc, a_opt = run_lbfgs(best_a, beta, topK, maxiter, maxfun)
        report(sc, a_opt, f"p3 beta={beta} topK={topK}")
        
        # Interleave subgradient polish
        if time.monotonic() < p3_deadline - 10:
            sc_sg, a_sg = multi_lag_subgrad(a_opt, total_iters=100, topK=3)
            report(sc_sg, a_sg, f"p3_sg beta={beta}")
            if sc_sg < sc:
                best_a = a_sg
                best_score = sc_sg
                a_opt = a_sg
    
    # ================================================================
    # Phase 4: Multi-start local search (25%)
    # ================================================================
    p4_deadline = t_start + 0.90 * effective_budget
    print(f"\n=== Phase 4: Local Search (deadline +{0.90*effective_budget:.0f}s) ===")
    print(f"  Current best: {best_score:.10f}")
    
    n_best = len(best_a)
    n_trials = 50
    
    for trial in range(n_trials):
        if time.monotonic() > p4_deadline:
            break
        remaining = p4_deadline - time.monotonic()
        if remaining < 15:
            break
        
        strategy = trial % 6
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Spectral phase perturbation
            M_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, M_fft)
            noise_phase = 0.02 * rng.standard_normal(len(A))
            A_noisy = A * np.exp(1j * noise_phase)
            A_noisy[0] = A[0]  # preserve DC
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Targeted: reduce peak-contributing elements
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            contrib = np.zeros(n_best)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            top_count = max(5, n_best // 100)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for j in top_idx:
                amt = a_pert[j] * 0.03
                a_pert[j] -= amt
                total_red += amt
            bottom_idx = np.argsort(contrib)[:top_count]
            for j in bottom_idx:
                a_pert[j] += total_red / top_count
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Spectral convolution flattening perturbation
            a_sf, _ = spectral_flatten_conv(best_a, n_iters=5, gamma=0.1)
            a_pert = a_sf
        elif strategy == 4:
            # Multi-lag targeted: reduce multiple peaks
            c = fast_conv(best_a)
            k_idx = np.argsort(c)[::-1][:5]
            a_pert = best_a.copy()
            for k in k_idx:
                j_start = max(0, k - n_best + 1)
                j_end = min(k, n_best - 1)
                if j_end >= j_start:
                    seg = a_pert[j_start:j_end + 1]
                    seg_rev = seg[::-1]
                    # Reduce the highest contributors
                    top_j = j_start + np.argmax(seg_rev)
                    amt = a_pert[top_j] * 0.02
                    a_pert[top_j] -= amt
                    # Add to a low-contributing position
                    bot_j = j_start + np.argmin(seg_rev)
                    a_pert[bot_j] += amt
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Chirp-modulated perturbation
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            pert = 0.02 * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick L-BFGS polish from perturbed point
        maxiter = min(2000, int(remaining * 0.3 / 0.06))
        maxfun = max(500, maxiter * 15)
        sc, a_opt = run_lbfgs(a_pert, 2000, 50, max(100, maxiter), maxfun)
        report(sc, a_opt, f"p4 #{trial} s={strategy}")
        
        # If improvement, do a bit more polish
        if sc < best_score + 1e-6 and time.monotonic() < p4_deadline - 10:
            sc2, a_opt2 = multi_lag_subgrad(a_opt, total_iters=50, topK=3)
            report(sc2, a_opt2, f"p4 #{trial} s={strategy} sg")
    
    # ================================================================
    # Phase 5: Final polish (10%)
    # ================================================================
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Before: {best_score:.10f}")
    
    # High-beta L-BFGS
    if time.monotonic() < deadline - 30:
        for beta, topK in [(5000, 50), (10000, 100), (20000, 200), (50000, 500)]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - 30 - time.monotonic()
            if remaining < 10:
                break
            maxiter = min(8000, int(remaining * 0.8 / 0.1))
            sc, a_opt = run_lbfgs(best_a, beta, topK, max(200, maxiter), maxiter * 20)
            report(sc, a_opt, f"p5 beta={beta} topK={topK}")
    
    # Multi-lag subgradient
    if time.monotonic() < deadline - 20:
        sc_sg, a_sg = multi_lag_subgrad(best_a, total_iters=500, topK=3)
        report(sc_sg, a_sg, "p5 multi_sg")
        sc_sg2, a_sg2 = multi_lag_subgrad(best_a, total_iters=300, topK=5)
        report(sc_sg2, a_sg2, "p5 multi_sg5")
    
    # Final single-lag subgradient
    if time.monotonic() < deadline - 10:
        sc_fin, a_fin = subgradient_polish(best_a, total_iters=300)
        report(sc_fin, a_fin, "p5 final_sg")
    
    # Spectral flattening as last resort
    if time.monotonic() < deadline - 5:
        a_sf, sc_sf = spectral_flatten_conv(best_a, n_iters=20, gamma=0.1)
        report(sc_sf, a_sf, "p5 spectral")
    
    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
