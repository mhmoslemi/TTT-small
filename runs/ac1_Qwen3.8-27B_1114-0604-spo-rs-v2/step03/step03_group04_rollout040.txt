```
def spectral_flatten_sequence(a, n_iters=100, lam=0.5):
    n=a.size; N=next_pow2(2*n-1)
    a=np.asarray(a,float); S0=a.sum()
    H=N//2
    for it in range(n_iters):
       progress=it/(n_iters-1) if n_iters>1 else 1
       l = lam*(1-progress)+0.2*progress
       A=np.fft.fft(a,N)
       mag=np.abs(A); phase=np.angle(A)
       non_dc=np.concatenate? mag[1:]

       idx=np.arange(1,H)
       vals=(1-l)*mag[idx]+l*target
       new_mag[idx]=vals
```

```
def resample_sequence(p, new_n):
   if new_n <=0: return uniform
   if p.size==new_n: return p.copy()
   cdf=np.cumsum(p); total=cdf[-1]
   if total<=0: return np.full(new_n,1/new_n)
   cdf=cdf/total
   xp=(np.arange(p.size)+0.5)/p.size
   xq=(np.arange(new_n)+0.5)/new_n
   ncdf=np.interp(xq, xp, cdf)
   out=np.diff(np.concatenate((np.array([0.0]), ncdf, np.array([1.0]))))
   out=np.maximum(out,0.0); s=out.sum(); if s<=0: out=np.full(new_n,1/new_n) else out/=s
   return out
```

```python
"""
Symmetric/log-domain optimizer for non-negative sequences minimizing
2*n*max(conv(a,a))/(sum(a))^2.

The code combines:
  - initialization from the supplied parent when available,
  - symmetrized and full log-softmax L-BFGS polishing with log-sum-exp
    approximations of the peak,
  - active-lag multiplicative balancing passes,
  - a few low-cost perturbation/polishing attempts.

All helpers are top-level; no lambdas or nested closures are used.
"""

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_vector(a):
    a = np.asarray(a, dtype=float).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        if a.size == 0:
            return a
        return np.full(a.shape, 1.0 / a.size, dtype=float)
    return a / s


def softmax_plain(x):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    mx = float(np.max(x))
    if not np.isfinite(mx):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(x - mx)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    return e / s


def conv_full(a):
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.array([], dtype=float)
    m = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)[:2 * n - 1]
    return np.maximum(c, 0.0)


def cross_corr_full(p, q):
    p = np.asarray(p, dtype=float)
    q = np.asarray(q, dtype=float)
    n = p.size
    if n == 0:
        return p
    m = next_pow2(3 * n - 2)
    P = np.fft.rfft(p, m)
    Q = np.fft.rfft(q[::-1], m)
    c = np.fft.irfft(P * Q, m)
    h = c[n - 1:2 * n - 1][::-1]
    return np.maximum(h, 0.0)


def cross_corr_active(p, idx, weights):
    p = np.asarray(p, dtype=float)
    n = p.size
    h = np.zeros(n, dtype=float)
    for t in range(len(idx)):
        w = float(weights[t])
        if w <= 0.0:
            continue
        l = int(idx[t])
        i0 = max(0, l - n + 1)
        i1 = min(l, n - 1)
        if i1 < i0:
            continue
        j0 = l - i1
        j1 = l - i0
        h[i0:i1 + 1] += w * p[j0:j1 + 1][::-1]
    return h


def softmax_weights_c(c, K):
    c = np.asarray(c, dtype=float)
    nn = c.size
    if nn == 0:
        return c
    if K <= 0:
        return np.full(nn, 1.0 / nn, dtype=float)
    beta = max(1.0, float(K) * float(nn))
    cmax = float(np.max(c))
    shifted = beta * (c - cmax)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 0.0:
        return np.full(nn, 1.0 / nn, dtype=float)
    return e / s


def exact_score(a):
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.inf
    c = conv_full(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (s * s))


def measure_conv_time(a):
    t0 = time.monotonic()
    conv_full(a)
    return max(1e-5, time.monotonic() - t0)


def symmetrize_sequence(p):
    p = np.asarray(p, dtype=float).copy()
    out = 0.5 * (p + p[::-1])
    s = float(np.sum(out))
    if s <= 1e-300:
        return np.full(p.size, 1.0 / max(1, p.size), dtype=float)
    return normalize_vector(out)


def log_full(p):
    p = np.asarray(p, dtype=float)
    p = np.maximum(p, 1e-300)
    z = np.log(p)
    z = z - float(np.mean(z))
    return z


def log_sym(p, n):
    p = np.asarray(p, dtype=float)
    n = int(n)
    m = (n + 1) // 2
    if n % 2 == 0:
        y = 0.5 * (p[:m] + p[::-1][:m])
    else:
        y = np.empty(m, dtype=float)
        if m == 1:
            y[0] = p[0]
        else:
            y[:m - 1] = 0.5 * (p[:m - 1] + p[::-1][:m - 1])
            y[m - 1] = p[m - 1]
    y = np.maximum(y, 1e-300)
    z = np.log(y)
    z = z - float(np.mean(z))
    return z


def sequence_from_z_sym(z, n):
    z = np.asarray(z, dtype=float)
    n = int(n)
    m = z.size
    if m == 0 or n == 0:
        return np.concatenate([z, np.array([], dtype=float)])
    if 2 * m != n and 2 * m - 1 != n:
        return np.full(n, 1.0 / n, dtype=float)

    mx = float(np.max(z))
    if not np.isfinite(mx):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(z - mx)
    se = float(np.sum(e))
    if (not np.isfinite(se)) or se <= 0.0:
        e = np.exp(np.zeros(m, dtype=float))
        se = float(m)

    if 2 * m == n:
        T = 2.0 * se
        first = e / T
        p = np.empty(n, dtype=float)
        p[:m] = first
        p[m:] = first[::-1]
    else:
        if m == 1:
            p = np.array([1.0], dtype=float)
        else:
            left = e[:m - 1]
            center = e[m - 1]
            T = 2.0 * float(np.sum(left)) + center
            first = left / T
            p = np.empty(n, dtype=float)
            p[:m - 1] = first
            p[m - 1] = center / T
            p[m:] = first[::-1]

    s = float(np.sum(p))
    if (not np.isfinite(s)) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=float)
    return p / s


def lse_value_grad_full(x, K, ridge):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return 1e100, np.array([], dtype=float)

    p = softmax_plain(x)
    c = conv_full(p)
    nn = 2 * n - 1
    cmax = float(np.max(c))
    beta = max(1.0, float(K) * float(nn))

    shifted = beta * (c - cmax)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 0.0:
        q = np.full(nn, 1.0 / nn, dtype=float)
        L = cmax
    else:
        q = e / s
        L = cmax + float(np.log(s)) / beta

    h = cross_corr_full(p, q)
    raw = 4.0 * float(n) * h
    dot = float(np.dot(p, raw))
    base_grad = p * (raw - dot)

    xx = float(np.dot(x, x))
    if not np.isfinite(xx):
        return 1e100, np.zeros(n, dtype=float)

    scale = float(n)
    val = scale * (2.0 * float(n) * L + ridge * xx)
    grad = scale * base_grad + 2.0 * scale * ridge * x
    if not np.isfinite(val):
        val = 1e100
        grad = np.zeros(n, dtype=float)
    return float(val), grad


def lse_value_grad_sym(x, K, ridge, n):
    x = np.asarray(x, dtype=float)
    n = int(n)
    m = x.size
    if m == 0 or n == 0:
        return np.inf, np.array([], dtype=float)
    if 2 * m != n and 2 * m - 1 != n:
        return 1e100, np.zeros(m, dtype=float)

    p = sequence_from_z_sym(x, n)
    c = conv_full(p)
    nn = 2 * n - 1
    cmax = float(np.max(c))
    beta = max(1.0, float(K) * float(nn))

    shifted = beta * (c - cmax)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if (not np.isfinite(s)) or s <= 0.0:
        q = np.full(nn, 1.0 / nn, dtype=float)
        L = cmax
    else:
        q = e / s
        L = cmax + float(np.log(s)) / beta

    h = cross_corr_full(p, q)
    raw = 4.0 * float(n) * h
    dot = float(np.dot(p, raw))

    base = np.zeros(m, dtype=float)
    if 2 * m == n:
        pl = p[:m]
        base = pl * (raw[:m] + raw[::-1][:m]) - 2.0 * pl * dot
    else:
        if m == 1:
            pc = float(p[0])
            base[0] = pc * raw[0] - pc * dot
        else:
            pl = p[:m - 1]
            base[:m - 1] = pl * (raw[:m - 1] + raw[::-1][:m - 1]) - 2.0 * pl * dot
            pc = float(p[m - 1])
            base[m - 1] = pc * raw[m - 1] - pc * dot

    xx = float(np.dot(x, x))
    if not np.isfinite(xx):
        return 1e100, np.zeros(m, dtype=float)

    scale = float(n)
    val = scale * (2.0 * float(n) * L + ridge * xx)
    grad = scale * base + 2.0 * scale * ridge * x
    if not np.isfinite(val):
        val = 1e100
        grad = np.zeros(m, dtype=float)
    return float(val), grad


def run_lbfgs_full(z0, K, ridge, ridge_maxiter, ridge_maxfun):
    z0 = np.asarray(z0, dtype=float)
    if z0.size == 0:
        return np.inf, np.array([], dtype=float), None
    options = {
        'maxiter': int(max(50, ridge_maxiter)),
        'maxfun': int(max(50, ridge_maxfun)),
        'ftol': 1e-11,
        'gtol': 1e-8,
        'maxls': 20,
        'maxcor': 20,
        'disp': 0,
    }
    try:
        res = minimize(lse_value_grad_full, z0, args=(K, ridge),
                       method='L-BFGS-B', jac=True, options=options)
    except Exception:
        return np.inf, z0, None
    if not np.all(np.isfinite(res.x)):
        return np.inf, z0, res
    p = normalize_vector(softmax_plain(res.x))
    sc = exact_score(p)
    return sc, p, res

            if st <= 1e-300:
                break
            trial = trial / st
            trial_s = exact_score(trial)

            if trial_s < current_s - 1e-13:
                current = trial
                current_s = trial_s
                accepted = True
                no_improve = max(0, no_improve - 2)
                alpha = max(0.08, alpha * 0.97)
                break

        if not accepted:
            alpha = max(0.08, alpha * 0.85)

        if it == 0 or (it > 0 and it % print_every == 0):
            print(f"      balance K={K} topk={topk} sym={prefer_sym} it={it} best={best_s:.8f}")

    return best, best_s


def suggested_maxfun(n):
    n = int(n)
    if n <= 1024:
        return 12000
    if n <= 2500:
        return 8000
    if n <= 6000:
        return 4000
    if n <= 12000:
        return 2500
    if n <= 20000:
        return 1500
    return 800


def spectral_flatten_sequence(a, n_iters=30, lam=0.5):
    a = normalize_vector(a)
    n = a.size
    if n == 0:
        return a
    N = next_pow2(2 * n - 1)
    H = N // 2
    S0 = float(np.sum(a))

    for it in range(max(1, int(n_iters))):
        progress = float(it) / max(1, int(n_iters) - 1)
        l = lam * (1.0 - progress) + 0.2 * progress

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        target = float(np.median(mag[1:])) if N > 1 else 0.0

        new_mag = mag.copy()
        if H > 1:
            idx = np.arange(1, H, dtype=int)
            vals = (1.0 - l) * mag[idx] + l * target
            new_mag[idx] = vals
            new_mag[N - idx] = vals
        if N % 2 == 0:
            new_mag[H] = mag[H]

        new_A = new_mag * np.exp(1j * phase)
        a = np.maximum(np.fft.ifft(new_A).real[:n], 0.0)
        S = float(np.sum(a))
        if S > 1e-300:
            a = a * (S0 / S)
        else:
            a = np.full(n, 1.0 / n, dtype=float)
    return normalize_vector(a)


def build_default_sequence(n=2048):
    n = max(32, int(n))
    x = np.linspace(0.0, 1.0, n)
    bases = [
        np.ones(n, dtype=float),
        0.25 + 0.75 * np.cos(np.pi * x) ** 2,
        0.5 + 0.4 * np.cos(2.0 * np.pi * x),
        0.5 + 0.4 * np.sin(2.0 * np.pi * 2.0 * x),
        0.2 + 0.8 * np.maximum(0.0, 1.0 - np.abs(2.0 * x - 1.0) * 1.5),
    ]

    best = np.full(n, 1.0 / n, dtype=float)
    best_s = np.inf
    for a in bases:
        a = normalize_vector(a)
        sc = exact_score(a)
        if sc < best_s:
            best = a.copy()
            best_s = sc

        af = spectral_flatten_sequence(a, 25, 0.5)
        scf = exact_score(af)
        if scf < best_s:
            best = af.copy()
            best_s = scf

    return best


def perturb_sequence(p, rng, mode=0, strength=0.05):
    p = np.asarray(p, dtype=float)
    n = p.size
    a = p.copy()

    mode = int(mode) % 5
    if mode == 0:
        a = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        shift = int((0.02 + 0.35 * rng.random()) * n)
        a = np.roll(p, shift)
    elif mode == 2:
        N = next_pow2(2 * n - 1)
        A = np.fft.fft(p, N)
        rn = 0.25 * strength * rng.standard_normal(A.size)
        im = 0.25 * strength * rng.standard_normal(A.size)
        A = A * (1.0 + rn + 1j * im)
        if A.size > 0:
            A[0] = float(np.real(A[0]))
        a = np.fft.ifft(A).real[:n]
    elif mode == 3:
        f = 1.0 + 8.0 * rng.random()
        ph = 2.0 * np.pi * rng.random()
        t = np.arange(n, dtype=float) / n
        a = a * (1.0 + strength * np.cos(2.0 * np.pi * f * t + ph))
    else:
        f1 = 1.0 + 10.0 * rng.random()
        f2 = 1.0 + 10.0 * rng.random()
        t = np.arange(n, dtype=float) / n
        a = a * (1.0 + 0.5 * strength * np.cos(2.0 * np.pi * f1 * t)
                 + 0.5 * strength * np.sin(2.0 * np.pi * f2 * t))

    return normalize_vector(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative step height sequence minimizing the target peak
    convolution score. Returns the best list of floats found before the budget
    expires.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + max(60.0, total_budget - 25.0)
    early_threshold = 1.5025
    target_reference = 1.5027
    ridge = 1e-12

    # Load parent if present.
    p = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None:
            h = np.asarray(h, dtype=float).ravel()
            h = np.where(np.isfinite(h), h, 0.0)
            h = np.clip(h, 0.0, 1000.0)
            if h.size >= 8 and float(np.sum(h)) > 0.0:
                p = normalize_vector(h)
    except Exception:
        p = None

    if p is None:
        p = build_default_sequence(2048)
        print("No parent sequence available; using constructed default.")

    n = int(p.size)
    best_p = p.copy()
    best_score = exact_score(best_p)
    print(f"Initial length={n}, score={best_score:.10f}, budget={total_budget:.0f}s")
    print(f"CP_reference target={target_reference:.4f}")

    # Try a symmetrized version; it is often very helpful.
    p_sym = symmetrize_sequence(best_p)
    sym_score = exact_score(p_sym)
    if sym_score < best_score - 1e-14:
        best_p = p_sym.copy()
        best_score = sym_score
        p_sym = best_p.copy()
        print(f"Symmetrization improved score to {best_score:.10f}")

    conv_time = measure_conv_time(best_p)
    print(f"Approx convolution time: {conv_time*1000:.3f} ms")

    base_maxfun = suggested_maxfun(n)
    cost_est = 0.02

    # ------------------------------------------------------------------
    # Symmetric log-softmax L-BFGS polishing.
    # ------------------------------------------------------------------
    if sym_score <= best_score * 1.02 + 0.005:
        z = log_sym(p_sym, n)
        sym_Ks = [20, 100, 500, 2000, 10000, 30000]
        for K in sym_Ks:
            if best_score < early_threshold:
                break
            remaining = deadline - time.monotonic() - 150.0
            if remaining < 30.0:
                break
            allowed = min(150.0, max(20.0, 0.30 * remaining))
            maxfun = int(min(base_maxfun, max(300, allowed / max(cost_est, 1e-3))))
            maxiter = max(100, maxfun)
            t0 = time.monotonic()
            sc, p_run, res = run_lbfgs_sym(z, n, K, ridge, maxiter, maxfun)
            dt = time.monotonic() - t0
            if res is not None:
                nfev = int(getattr(res, 'nfev', 0))
                if nfev > 0 and np.isfinite(dt):
                    cost_est = min(1.0, max(1e-4, dt / float(nfev)))

            if np.isfinite(sc):
                p_run = normalize_vector(np.asarray(p_run, dtype=float))
                if sc < best_score - 1e-15:
                    best_p = p_run
                    best_score = sc
                    z = log_sym(best_p, n)
                    print(f"  sym L-BFGS K={K}: NEW BEST {sc:.10f}")
                elif sc <= best_score * 1.02:
                    z = log_sym(p_run, n)
                    print(f"  sym L-BFGS K={K}: continue from score {sc:.10f}")

    # ------------------------------------------------------------------
    # Full log-softmax L-BFGS polishing.
    # ------------------------------------------------------------------
    if best_score >= early_threshold and (deadline - time.monotonic()) > 300.0:
        z = log_full(best_p)
        full_Ks = [100, 1000, 10000]
        for K in full_Ks:
            if best_score < early_threshold:
                break
            remaining = deadline - time.monotonic() - 240.0
            if remaining < 40.0:
                break
            allowed = min(120.0, max(20.0, 0.30 * remaining))
            maxfun = int(min(max(300, base_maxfun // 2), max(300, allowed / max(cost_est, 1e-3))))
            maxiter = max(100, maxfun)
            t0 = time.monotonic()
            sc, p_run, res = run_lbfgs_full(z, K, ridge, maxiter, maxfun)
            dt = time.monotonic() - t0
            if res is not None:
                nfev = int(getattr(res, 'nfev', 0))
                if nfev > 0 and np.isfinite(dt):
                    cost_est = min(1.0, max(1e-4, dt / float(nfev)))

            if np.isfinite(sc):
                p_run = normalize_vector(np.asarray(p_run, dtype=float))
                if sc < best_score - 1e-15:
                    best_p = p_run
                    best_score = sc
                    z = log_full(best_p)
                    print(f"  full L-BFGS K={K}: NEW BEST {sc:.10f}")
                elif sc <= best_score * 1.02:
                    z = log_full(p_run)
                    print(f"  full L-BFGS K={K}: continue from score {sc:.10f}")

    # ------------------------------------------------------------------
    # Active-lag multiplicative balancing.
    # ------------------------------------------------------------------
    if best_score >= early_threshold:
        balance_configs = [
            (20, 1, 0.55),
            (80, 5, 0.45),
            (300, 20, 0.38),
            (1200, 80, 0.32),
            (5000, 300, 0.25),
        ]

        for K, topk, alpha in balance_configs:
            if best_score < early_threshold:
                break
            for prefer_sym in (False, True):
                if best_score < early_threshold:
                    break
                rem = deadline - time.monotonic() - 60.0
                if rem < 20.0:
                    break
                allowed = min(45.0, max(10.0, 0.10 * rem))
                iter_cost = max(1e-3, 4.0 * conv_time + 0.005)
                maxit = int(min(5000, max(100, allowed / iter_cost)))
                p_start = best_p.copy()
                if prefer_sym:
                    p_start = symmetrize_sequence(p_start)
                stop = min(deadline - 30.0, time.monotonic() + allowed)
                p_out, s_out = balancing_pass(p_start, K, topk, alpha, maxit, stop, prefer_sym)
                if np.isfinite(s_out) and s_out < best_score - 1e-14:
                    best_p = normalize_vector(np.asarray(p_out, dtype=float))
                    best_score = s_out
                    conv_time = max(1e-5, conv_time * 0.75)
                    print(f"  balance K={K} topk={topk} sym={prefer_sym}: NEW BEST {best_score:.10f}")

    # ------------------------------------------------------------------
    # Final high-smoothness polish.
    # ------------------------------------------------------------------
    if best_score >= early_threshold and (deadline - time.monotonic()) > 150.0:
        p_sym = symmetrize_sequence(best_p)
        sym_score = exact_score(p_sym)
        if sym_score <= best_score * 1.01 + 0.001:
            z = log_sym(p_sym, n)
            for K in (10000, 30000):
                if best_score < early_threshold:
                    break
                rem = deadline - time.monotonic() - 90.0
                if rem < 25.0:
                    break
                allowed = min(70.0, max(15.0, 0.4 * rem))
                maxfun = int(min(base_maxfun, max(300, allowed / max(cost_est, 1e-3))))
                sc, p_run, _ = run_lbfgs_sym(z, n, K, ridge, max(100, maxfun), maxfun)
                if np.isfinite(sc):
                    p_run = normalize_vector(np.asarray(p_run, dtype=float))
                    if sc < best_score - 1e-15:
                        best_p = p_run
                        best_score = sc
                        print(f"  final sym L-BFGS K={K}: NEW BEST {sc:.10f}")
                    z = log_sym(p_run, n)

        z = log_full(best_p)
        for K in (5000, 20000):
            if best_score < early_threshold:
                break
            rem = deadline - time.monotonic() - 70.0
            if rem < 25.0:
                break
            allowed = min(70.0, max(15.0, 0.4 * rem))
            maxfun = int(min(max(300, base_maxfun // 2), max(300, allowed / max(cost_est, 1e-3))))
            sc, p_run, _ = run_lbfgs_full(z, K, ridge, max(100, maxfun), maxfun)
            if np.isfinite(sc):
                p_run = normalize_vector(np.asarray(p_run, dtype=float))
                if sc < best_score - 1e-15:
                    best_p = p_run
                    best_score = sc
                    print(f"  final full L-BFGS K={K}: NEW BEST {sc:.10f}")
                z = log_full(p_run)

    # ------------------------------------------------------------------
    # A few short perturbation escapes and rebalancing attempts.
    # ------------------------------------------------------------------
    if best_score >= early_threshold and (deadline - time.monotonic()) > 160.0:
        rng = np.random.default_rng(98765 + int(seed))
        for mode in range(3):
            rem = deadline - time.monotonic() - 30.0
            if rem < 25.0:
                break
            p_try = perturb_sequence(best_p, rng, mode, 0.04)
            allowed = min(18.0, max(6.0, 0.1 * rem))
            iter_cost = max(1e-3, 4.0 * conv_time + 0.005)
            maxit = int(min(2500, max(80, allowed / iter_cost)))
            stop = min(deadline - 25.0, time.monotonic() + allowed)
            p_out, s_out = balancing_pass(p_try, 2000, 100, 0.30, maxit, stop, False)
            if np.isfinite(s_out) and s_out < best_score - 1e-14:
                best_p = normalize_vector(np.asarray(p_out, dtype=float))
                best_score = s_out
                print(f"  perturb mode={mode}: NEW BEST {best_score:.10f}")

    # ------------------------------------------------------------------
    # Very final short symmetric balancing polish.
    # ------------------------------------------------------------------
    if best_score >= early_threshold and (deadline - time.monotonic()) > 30.0:
        p_start = symmetrize_sequence(best_p)
        allowed = min(25.0, max(6.0, deadline - time.monotonic() - 15.0))
        iter_cost = max(1e-3, 4.0 * conv_time + 0.005)
        maxit = int(min(2500, max(80, allowed / iter_cost)))
        stop = min(deadline - 10.0, time.monotonic() + allowed)
        p_out, s_out = balancing_pass(p_start, 2000, 100, 0.25, maxit, stop, True)
        if np.isfinite(s_out) and s_out < best_score - 1e-14:
            best_p = normalize_vector(np.asarray(p_out, dtype=float))
            best_score = s_out
            print(f"  final sym balance: NEW BEST {best_score:.10f}")

    # ------------------------------------------------------------------
    # Final validation and packaging.
    # ------------------------------------------------------------------
    out_arr = np.asarray(best_p, dtype=float)
    out_arr = np.where(np.isfinite(out_arr), out_arr, 0.0)
    out_arr = np.maximum(out_arr, 0.0)
    s = float(np.sum(out_arr))
    if s > 0.0:
        out_arr = out_arr / s
    else:
        out_arr = np.full(64, 1.0 / 64.0, dtype=float)

    out = [float(x) for x in out_arr]
    internal_score = exact_score(out_arr)

    try:
        verified = float(evaluate_sequence(out))
        print(f"Final internal={internal_score:.10f}, verified={verified:.10f}, length={len(out)}")
    except Exception as ex:
        print(f"Final internal={internal_score:.10f}, verification error: {ex}")

    return out
```
