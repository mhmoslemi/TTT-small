= Σ_k p_k * a[k-i]

where p_k = exp(β*conv[k]) / Σ_j exp(β*conv[j]) is a softmax over the convolution values.

For large β, p_k concentrates on the peak lag k*, and the gradient becomes approximately a[k*-i], which is what we expect.

So the LSE approach is good. The previous code used this with L-BFGS.

"""

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: subtract max before exp
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

def frank_wolfe(a, beta, n_iters=1000, tol=1e-12):
    """
    Frank-Wolfe (conditional gradient) method for minimizing LSE_β(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S  # Normalize to simplex
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        score, grad = lse_score_grad(a, beta)
        
        # Linear minimization oracle: min_{b in simplex} grad · b
        # = b is a unit vector at argmin(grad)
        i_min = np.argmin(grad)
        
        # Step size (line search or standard FW step size)
        gamma = 2.0 / (it + 2)  # Standard FW step size
        # Or use backtracking line search
        a_new = (1 - gamma) * a + gamma * np.zeros(n)
        a_new[i_min] = gamma
        
        # Backtracking line search
        alpha = 1.0
        for _ in range(30):
            a_try = (1 - alpha) * a + alpha * a_new
            score_try, _ = lse_score_grad(a_try, beta)
            if score_try <= score - 1e-12 * alpha * np.dot(grad, a_new - a):
                break
            alpha *= 0.5
        
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Convergence check
        gap = np.dot(grad, a - a_new)  # Duality gap (up to normalization)
        if gap < tol:
            break
    
    return best_score, best_a


def spectral_flatten_v2(a0, N, n_iters=300, lam=0.5, rng=None):
    """Improved spectral flattening with adaptive lambda."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        # Adaptive lambda: start strong, end weak
        lam_i = lam * (1.0 - 0.5 * progress)
        
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean of non-DC magnitudes (more aggressive than median)
        non_dc_mag = mag[1:N//2 + 1]
        if len(non_dc_mag) > 0:
            target = np.exp(np.log(np.maximum(non_dc_mag, 1e-30)).mean())
        else:
            target = 1.0
        
        new_mag = mag.copy()
        new_mag[0] = mag[0]  # Preserve DC
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        
        for k in range(1, half + 1):
            if k == 0 or (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam_i) * mag[k] + lam_i * target
            new_mag[N - k] = new_mag[k]
        
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (S0 / S)
    return a

def peak_shave(a, n_shaves=50, beta=1000):
    """
    Targeted peak shaving: identify the peak lags and make small
    modifications to reduce them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_shaves):
        c = fast_conv(a)
        S = a.sum()
        M = c.max()
        k_star = np.argmax(c)
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find the "active set": lags where conv is within 5% of max
        threshold = M * 0.95
        active_lags = np.where(c >= threshold)[0]
        
        # For each active lag, compute the gradient contribution
        # and try to reduce it
        # The subgradient of max at k_star is: 2 * a shifted
        # For multiple active lags, average the subgradients
        
        # Compute subgradient at the peak
        sg = np.zeros(n)
        for k in active_lags[:10]:  # Limit to top 10 active lags
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += seg[::-1]
        sg /= max(1, len(active_lags[:10]))
        sg *= 2.0  # Factor of 2 from autocorrelation
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Step: reduce in direction of subgradient
        # Use a line search
        step = 1e-4 * S / sg_norm
        for trial_step in [step, step*0.5, step*0.25, step*0.1]:

            a_try = np.clip(a_try, 0.0, 1000.0)
            if a_try.sum() < 1e-12:
                continue
            sc_try = compute_score(a_try)
            if sc_try < sc:
                a = a_try
                break
        
        if it % 10 == 0:
            print(f"    peak_shave it={it}: score={sc:.8f}, k_star={k_star}, n_active={len(active_lags)}")
    
    return best_sc, best_a


def adam_optimize(a0, beta, lr=1e-3, n_steps=5000, weight_decay=1e-5):
    """Adam optimizer for the LSE objective with non-negativity."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_a = a.copy()
    best_sc = compute_score(a)
    b1, b2 = 0.9, 0.999
    
    for t in range(1, n_steps + 1):
        sc, grad = lse_score_grad(a, beta)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Weight decay
        grad = grad + weight_decay * a
        grad = np.maximum(grad, 0)  # Project gradient to respect non-negativity? No, just clip a after update
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        lr_t = lr * np.sqrt(1 - b2**t) / (1 - b1**t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.clip(a, 0.0, 1000.0)
    
    final_sc = compute_score(best_a)
    return final_sc, best_a
