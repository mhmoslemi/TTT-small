```python
import numpy as np
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def adam_update(a, grad, state, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. State is (m, v, t)."""
    m, v, t = state
    t += 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, (m, v, t)

def adam_optimize(a0, beta, steps=50000, lr=0.01, patience=5000):
    """Optimize using Adam with LSE objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a[:] = 1.0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_score = compute_score(a)
    best_a = a.copy()
    best_t = 0
    
    for step in range(steps):
        score, grad = lse_score_grad(a, beta)
        a, (m, v, t) = adam_update(a, grad, (m, v, t), lr=lr)
        a = np.clip(a, 0.0, 1000.0)
        
        if step % 100 == 0:
            exact = compute_score(a)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                best_t = step
            if step - best_t > patience:
                # Restore best and reduce learning rate
                a = best_a.copy()
                lr *= 0.5
                if lr < 1e-8:
                    break
                patience = max(1000, patience // 2)
    
    return best_score, best_a
```

```python
"""
Step function optimization: Multi-scale LSE + Adam hybrid with bowl-shaped initialization,

Key innovations over previous approach:
1. Multiple sequence lengths (n) explored simultaneously
2. Bowl-shaped and structured initializations to escape local minima
3. Adam optimizer (parallel-friendly, good for large n) combined with L-BFGS-B
4. Adaptive peak shaving with oscillating step sizes
5. Cosine-annealed LSE beta schedule

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1: 2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=50000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def adam_optimize(a0, beta, steps=20000, lr=0.05, lr_end=0.001, verbose=False):
    """Adam optimization with cosine-annealed learning rate on LSE objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a[:] = 1.0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_score = compute_score(a)
    best_a = a.copy()

    for step in range(steps):
        # Cosine annealing
        progress = step / max(1, steps - 1)
        lr_cur = lr_end + 0.5 * (lr - lr_end) * (1 + np.cos(np.pi * progress))

        score, grad = lse_score_grad(a, beta)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if step % 500 == 0:
            exact = compute_score(a)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
            if verbose and step % 2000 == 0:
                print(f"    Adam step {step}/{steps}: score={exact:.8f}, beta={beta}, lr={lr_cur:.6f}")

    exact = compute_score(a)
    if exact < best_score:
        best_score = exact
        best_a = a.copy()

    return best_score, best_a


def peak_shave_adaptive(a, iters=5000, verbose=False):
    """Adaptive subgradient descent with oscillating step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    stall_count = 0
    base_step = 0.1

    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step: Polyak-like with oscillation
        sc_lb = 1.0
        polyak_step = (sc - sc_lb) / (gn * gn + 1e-20)
        # Oscillate between small and medium steps
        osc = 0.5 + 0.5 * np.sin(0.1 * it)
        step = base_step * osc * min(polyak_step, 1.0 / S)
        step = max(step, 1e-8 / (gn * S + 1e-30))
        step = min(step, 0.5 / S)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

        # If stalled, try a larger perturbation
        if stall_count > 500 and stall_count % 500 == 0:
            a = best_a.copy()
            base_step = max(0.01, base_step * 0.5)

    return best_sc, best_a

def make_bowl_init(n, alpha=1.0):
    """Bowl-shaped initialization: high at edges, low in middle."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + alpha * (2.0 * x - 1.0) ** 2
    return np.clip(a, 0.1, 100.0)


def make_sine_init(n, freq=1.0, amp=0.3, phase=0.0):
    """Sinusoidal modulation initialization."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    return np.clip(a, 0.05, 100.0)


def resize_sequence(a, new_n):
    """Resize a sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
```
