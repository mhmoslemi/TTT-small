"""
Step function optimization using LP-based direction finding (inspired by convex
relaxation of the peak-shaving problem), LSE continuation, and momentum
subgradient. Combines multiple strategies across different sequence lengths
to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import minimize
from scipy import sparse
import cvxpy as cp
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    """Objective wrapper for scipy.minimize with args=(beta,)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization. Returns optimized array."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def build_cross_corr_constraint(f, k):
    """
    Build the coefficient vector for constraint (f*b)[k] <= M.
    Returns a sparse row (1 x n) where entry j is f[k-j] if 0<=k-j<n.
    """
    n = len(f)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_end < j_start:
        return sparse.csr_matrix((1, n))
    indices = np.arange(j_start, j_end + 1)
    vals = f[k - indices]  # f[k-j] for j in range
    row = sparse.csr_matrix((vals, (np.zeros(len(indices), dtype=int), indices)), shape=(1, n))
    return row


def solve_lp_direction(f, M_bound, top_lags, slack=0.0):
    """
    Solve: maximize sum(b) s.t. b >= 0, (f*b)[k] <= M_bound*(1+slack) for k in top_lags.
    Returns optimal b or None on failure.
    """
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    rhs = M_bound * (1.0 + slack)

    # Build sparse constraint matrix
    rows_data = []
    cols_data = []
    vals_data = []
    row_indices = []
    for ri, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end < j_start:
            continue
        indices = np.arange(j_start, j_end + 1)
        vals = f[k - indices]
        for idx, v in zip(indices, vals):
            if v > 1e-12:
                rows_data.append(ri)
                cols_data.append(int(idx))
                vals_data.append(float(v))
        row_indices.append(ri)

    if len(row_indices) == 0:
        return None

    row_indices = np.array(row_indices)
    cols_data = np.array(cols_data)
    vals_data = np.array(vals_data)

    A_spp = sparse.csr_matrix((vals_data, (row_indices, cols_data)),
                               shape=(len(top_lags), n))
    # Remove rows that are all zero
    nonzero_rows = np.any(A_spp != 0, axis=1)
    A_spp = A_spp[nonzero_rows, :]
    n_cons = A_spp.shape[0]
    if n_cons == 0:
        return None

    constraints.append(A_spp @ b <= rhs)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iter=200000, rel_tol=1e-8, abs_tol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None

    b_val = b.value
    if b_val is None:
        return None
    b_val = np.clip(np.asarray(b_val, dtype=np.float64), 0.0, 1000.0)
    if b_val.sum() < 1e-12:
        return None
    return b_val


def lp_based_iteration(f, top_P=30, slack=0.05, max_tries=8):
    """
    One LP-based peak shaving iteration.
    Returns (success, new_f, alpha_used).
    """
    n = len(f)
    S_f = float(f.sum())
    c = fast_conv(f)
    M = float(c.max())
    k_star = int(np.argmax(c))
    sc_cur = 2.0 * n * M / (S_f * S_f)

    # Find top lags (where c is close to M)
    threshold = M * 0.95
    peak_lags = np.where(c >= threshold)[0]
    if len(peak_lags) < top_P:
        sorted_lags = np.argsort(c)[::-1]
        peak_lags = sorted_lags[:top_P]
    else:
        # Use the top_P lags by value
        sorted_lags = np.argsort(c)[::-1]
        peak_lags = sorted_lags[:top_P]

    # Solve LP
    b = solve_lp_direction(f, M, peak_lags, slack=slack)
    if b is None:
        return False, f.copy(), 0.0

    # Normalize b to match sum of f
    S_b = float(b.sum())
    if S_b < 1e-12:
        return False, f.copy(), 0.0
    b_norm = b * (S_f / S_b)

    # Line search on alpha
    best_alpha = 0.0
    best_sc = sc_cur
    for i in range(max_tries):
        alpha = 0.3 * (0.7 ** i)
        f_new = (1.0 - alpha) * f + alpha * b_norm
        f_new = np.clip(f_new, 0.0, 1000.0)
        sc_new = compute_score(f_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_alpha = alpha
        if best_alpha > 0.0 and i >= 3:
            break

    if best_alpha < 1e-6:
        return False, f.copy(), 0.0

    f_new = (1.0 - best_alpha) * f + best_alpha * b_norm
    f_new = np.clip(f_new, 0.0, 1000.0)
    return True, f_new, best_alpha


def subgradient_with_momentum(a, max_iters, initial_step=None, verbose=False):
    """Subgradient descent on the sharp max-convolution objective with momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, a
    if initial_step is None:
        initial_step = 1.0 / (n * a.max() + 1e-12)

    step = initial_step
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    best_a = a.copy()
    best_sc = compute_score(a)
    best_sc_lb = 1.0  # theoretical lower bound

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of the score
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[k_star - np.arange(j_start, j_end + 1)]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M * a / S3)

        # Momentum update
        v = mu * v - step * grad
        a -= v
        a = np.clip(a, 0.0, 1000.0)

        # Adaptive step: Polyak-like
        if sc > best_sc * 1.05:
            step *= 0.7
        elif it % 200 == 0 and step < 1.0 / (S + 1e-12):
            step *= 1.05

        # Periodically reduce momentum
        if it > max_iters // 2:
            mu = max(0.5, mu - 0.001)

    return best_sc, best_a


def variance_score_grad(a):
    """Score and gradient for flatness objective: 2n*sum(c^2)/S^4."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c2 = float(np.sum(c * c))
    score = 2.0 * n * c2 / (S ** 4)
    L = 1 << (2 * n - 2).bit_length()
    A = np.fft.fft(a, L)
    E = np.abs(A) ** 2 * np.conj(A)
    E_ifft = np.fft.ifft(E).real
    idx = np.concatenate([[0], np.arange(L - 1, L - n, -1)])
    dc2_da = 4.0 * E_ifft[idx]
    S4 = S ** 4
    S5 = S4 * S
    grad = 2.0 * n * (dc2_da / S4 - 4.0 * c2 / S5)
    return score, grad


def variance_obj(a):
    """Objective wrapper for variance minimization."""
    return variance_score_grad(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-based peak shaving, LSE continuation, subgradient with momentum,
    and structured restarts across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent: n={len(parent)}, score={sc:.8f}")

    # Initial candidate if no parent or parent is bad
    if best_a is None:
        n_init = 2048
        a_init = np.ones(n_init) * 0.5
        best_a = a_init.copy()
        best_score = compute_score(a_init)
        print(f"Initialized with uniform n={n_init}, score={best_score:.8f}")

    # ===================================================================
    # Phase 1: LSE continuation from parent (20% time)
    # ===================================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print("\n=== Phase 1: LSE Continuation (warmup) ===")

    a_cur = best_a.copy()
    beta_schedule = [20, 50, 100, 200, 500, 1000]
    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=500, maxfun=10000)
        sc = compute_score(a_cur)
        report(sc, a_cur, f"LSE beta={beta}")
        print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 2: LP-based peak shaving (50% time) - MAIN PHASE
    # ===================================================================
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print("\n=== Phase 2: LP-Based Peak Shaving ===")

    a_lp = best_a.copy()
    n_lp = len(a_lp)
    lp_iter_count = 0
    no_improve_count = 0

    # Adaptive parameters
    top_P = max(10, n_lp // 100)
    slack = 0.05

    while time.monotonic() < phase2_end:
        t0 = time.monotonic()
        success, a_lp, alpha = lp_based_iteration(a_lp, top_P=top_P, slack=slack, max_tries=6)
        lp_iter_count += 1

        if success:
            sc = compute_score(a_lp)
            if report(sc, a_lp, f"LP iter {lp_iter_count}"):
                no_improve_count = 0
                # Occasionally reduce slack for tighter constraints
                if lp_iter_count % 20 == 0:
                    slack = max(0.01, slack * 0.8)
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        # If stuck, try larger slack or different top_P
        if no_improve_count >= 10:
            slack = min(0.2, slack * 1.5)
            top_P = max(5, top_P // 2)
            no_improve_count = 0
            # Perturb to escape
            a_lp = a_lp * (1.0 + 0.01 * rng.standard_normal(n_lp))
            a_lp = np.clip(a_lp, 0.0, 1000.0)

        if lp_iter_count % 50 == 0:
            print(f"  LP iter {lp_iter_count}: best={best_score:.8f}, "
                  f"slack={slack:.3f}, top_P={top_P}, alpha={alpha:.4f}")

    print(f"  LP phase done: {lp_iter_count} iterations")

    # ===================================================================
    # Phase 3: Subgradient with momentum (15% time)
    # ===================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print("\n=== Phase 3: Subgradient with Momentum ===")

    if time.monotonic() < phase3_end:
        t0 = time.monotonic()
        n_sg = len(best_a)
        iters_sg = int(min(5000, (phase3_end - time.monotonic()) * 300))
        step_init = 1.0 / (n_sg * float(best_a.max()) + 1e-12)
        sc_sg, a_sg = subgradient_with_momentum(best_a, max_iters=max(500, iters_sg),
                                                initial_step=step_init)
        report(sc_sg, a_sg, "subgradient momentum")
        print(f"  Subgradient: {sc_sg:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 4: Shorter sequence attempts (10% time)
    # ===================================================================
    phase4_end = deadline
    print("\n=== Phase 4: Shorter Sequences ===")

    for n_try in [1024, 2048, 4096]:
        if time.monotonic() > phase4_end:
            break
        t0 = time.monotonic()

        # Initialize with a good pattern
        a_try = np.ones(n_try) * 0.5
        # Add structure: raised cosine window
        x = np.arange(n_try) / n_try
        a_try = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a_try = np.clip(a_try, 0.01, None)

        # Quick LSE
        for beta in [50, 200, 1000]:
            if time.monotonic() > phase4_end:
                break
            a_try = run_lbfgs(a_try, lse_obj, (beta,), maxiter=300, maxfun=6000)

        sc = compute_score(a_try)
        report(sc, a_try, f"short n={n_try} LSE")

        # Quick LP
        n_lp_short = max(10, n_try // 100)
        for _ in range(30):
            if time.monotonic() > phase4_end:
                break
            success, a_try, _ = lp_based_iteration(a_try, top_P=n_lp_short, slack=0.05)
            if success:
                sc = compute_score(a_try)
                report(sc, a_try, f"short n={n_try} LP")

        print(f"  n={n_try}: best_so_far={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 5: Final polish on best sequence (5% time)
    # ===================================================================
    print("\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline and best_a is not None:
        a_final = best_a.copy()

        # High-beta LSE polish
        a_final = run_lbfgs(a_final, lse_obj, (5000,), maxiter=500, maxfun=10000)
        sc = compute_score(a_final)
        report(sc, a_final, "final LSE high-beta")

        # A few more LP iterations
        n_lp_final = max(10, len(a_final) // 100)
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            success, a_final, _ = lp_based_iteration(a_final, top_P=n_lp_final, slack=0.02)
            if success:
                sc = compute_score(a_final)
                report(sc, a_final, "final LP")

        # Short subgradient
        sc_sg2, a_sg2 = subgradient_with_momentum(best_a, max_iters=500,
                                                   initial_step=1.0/(len(best_a)*float(best_a.max())+1e-12))
        report(sc_sg2, a_sg2, "final subgrad")

    # ===================================================================
    # Final output
    # ===================================================================
    if best_a is None:
        out = [0.5] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
