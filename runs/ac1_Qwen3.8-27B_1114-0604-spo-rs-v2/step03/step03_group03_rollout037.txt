```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k."""
    n = len(f)
    
    # Build constraint matrix A (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for j in range(i_min, i_max + 1):
            # A[k, j] = f[k - j]
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_constraint = M * np.ones(2*n - 1)
    
    # Bounds: 0 <= b_j <= some upper bound
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_constraint, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
Hybrid optimizer for minimum peak autocorrelation of non-negative sequences.
Combines Adam on LSE-smoothed objective, LP-based direction finding,
and exact coordinate descent to minimize 2n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
- Adam first-order method for fast local optimization (many more iterations)
- LP-based global direction finding for escaping local minima
- Adaptive beta (LSE temperature) schedule
- More aggressive diversification strategies
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/d a[i] = sum_k p[k] * d(conv(a,a)_k)/da[i]
    # d(conv(a,a)_k)/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0 (with double count at k=2i handled correctly)
    # sum_k p[k] * 2*a[k-i] = 2 * (p_cross_a)[i]
    # where (p_cross_a)[i] = sum_k p[k] * a[k-i] (cross-correlation)
    
    # Use FFT for cross-correlation
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, M2_fft)
    A = np.fft.rfft(a, M2_fft)
    cross = np.fft.irfft(P_rev * A, M2_fft)[n - 1:2 * n - 1]
    
    grad_L = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n * grad_L / S2) - (2.0 * score / S)
    
    return score, grad


def adam_lse_optimize(a0, beta, lr, max_iter, deadline):
    """
    Adam optimizer on LSE-smoothed objective.
    Returns (final_score, final_a).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    prev_score = compute_score(a)
    no_improve = 0
    
    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break
        
        score_val, grad = lse_grad(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, 1000.0)

        if t % 50 == 0 or t == max_iter:
            exact_sc = compute_score(a_new)
            if exact_sc < prev_score - 1e-10:
                prev_score = exact_sc
                no_improve = 0
            else:
                no_improve += 50
            
            if no_improve > 200:
                break
        
        a = a_new
    
    final_score = compute_score(a)
    return final_score, a


def build_conv_matrix_sparse(f):
    """
    Build sparse matrix A such that A @ b = convolve(f, b).
    A has shape (2n-1, n). Each column j has n non-zeros.
    """
    n = len(f)
    m = 2 * n - 1
    
    indptr = np.arange(n + 1, dtype=np.int32) * n
    # For column j, rows are j, j+1, ..., j+n-1
    # indices = tile(arange(n), n) + repeat(arange(n), n)
    base_idx = np.tile(np.arange(n, dtype=np.int32), n)
    offsets = np.repeat(np.arange(n, dtype=np.int32), n)
    indices = base_idx + offsets
    data = np.tile(f, n)
    
    A = csr_matrix((data, indices, indptr), shape=(m, n))
    return A


def solve_lp_direction(f, M, time_limit=25):
    """
    Solve LP: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns solution b or None.
    """
    n = len(f)
    
    try:
        A = build_conv_matrix_sparse(f)
    except MemoryError:
        return None
    
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    
    if res.success or (res.x is not None and np.all(res.x >= -1e-8)):
        return np.clip(res.x, 0.0, None)
    else:
        return None


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None):
    """One full sweep of exact coordinate descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for i in order:
        i = int(i)
        a_i = a[i]
        
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)
        
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a


def hot_cd_sweep(a, n_sweeps=1):
    """CD sweep prioritizing coordinates that contribute to the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
        
        for i in order:
            i = int(i)
            a_i = a[i]
            
            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)
            
            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0
            
            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()
            
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib
            
            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0
            
            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)
            
            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-7}
            )
            t_opt = float(res.x)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a


def targeted_perturb(a, rng, strength=0.03):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    
    top_count = max(3, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0
    
    # Load parent
    best_a = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                best_a = np.clip(p, 0.0, 1000.0).copy()
    except Exception:
        pass
    
    if best_a is None:
        n = 2048
        best_a = np.ones(n) * 0.5
    
    n = len(best_a)
    best_score = compute_score(best_a)
    print(f"Initial: n={n}, score={best_score:.8f}")
    
    # ============================================================
    # Phase 1: CD polish (10% time)
    # ============================================================
    phase1_end = t_start + 0.10 * total_budget
    print(f"\n=== Phase 1: CD Polish ===")
    a_cur = best_a.copy()
    for i in range(50):
        if time.monotonic() > phase1_end:
            break
        if i % 3 == 0:
            sc, a_cur = hot_cd_sweep(a_cur)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if i < 5 or i % 10 == 0:
                print(f"  CD {i}: {sc:.8f}")
    print(f"Phase 1 done: {best_score:.8f}")
    
    # ============================================================
    # Phase 2: Adam LSE optimization (35% time)
    # Adaptive beta schedule: start smooth, get finer
    # ============================================================
    phase2_end = t_start + 0.45 * total_budget
    print(f"\n=== Phase 2: Adam LSE ===")
    
    a_cur = best_a.copy()
    betas = [20, 50, 100, 300, 800, 2000, 5000]
    
    for beta in betas:
        if time.monotonic() > phase2_end:
            break
        # Learning rate: start moderate, decrease
        lr = 0.05 / np.sqrt(n)
        max_iter = 2000
        t0 = time.monotonic()
        sc, a_cur = adam_lse_optimize(a_cur, beta, lr, max_iter, phase2_end)
        elapsed = time.monotonic() - t0
        
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Adam beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  Adam beta={beta}: {sc:.8f} ({elapsed:.1f}s)")
        
        # Quick CD after each Adam phase
        for _ in range(2):
            if time.monotonic() > phase2_end:
                break
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    
    print(f"Phase 2 done: {best_score:.8f}")
    
    # ============================================================
    # Phase 3: LP-based direction finding (30% time)
    # ============================================================
    phase3_end = t_start + 0.75 * total_budget
    print(f"\n=== Phase 3: LP Direction Finding ===")
    
    a_cur = best_a.copy()
    n_lp = len(a_cur)
    lp_success = 0
    lp_fail = 0
    
    for lp_iter in range(100):
        if time.monotonic() > phase3_end:
            break
        if time.monotonic() > deadline - 30:
            break
        
        # Normalize to sum = sqrt(2n)
        S_cur = a_cur.sum()
        if S_cur < 1e-12:
            break
        a_norm = a_cur / S_cur * np.sqrt(2.0 * n_lp)
        
        # Compute M
        conv = fast_conv(a_norm)
        M = float(conv.max())
        
        # Solve LP
        t0 = time.monotonic()
        b_sol = solve_lp_direction(a_norm, M, time_limit=20)
        lp_time = time.monotonic() - t0
        
        if b_sol is None:
            lp_fail += 1
            if lp_fail > 3:
                print(f"  LP failed 4 times, stopping LP phase")
                break
            continue
        
        lp_success += 1
        b_sum = float(b_sol.sum())
        if b_sum < 1e-10:
            continue
        
        # Rescale g
        g = b_sol / b_sum * np.sqrt(2.0 * n_lp)
        
        # Line search over blend parameter
        best_t = 0.0
        best_sc_phase3 = compute_score(a_cur)
        for t_blend in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = (1.0 - t_blend) * a_norm + t_blend * g
            sc_try = compute_score(a_try)
            if sc_try < best_sc_phase3:
                best_sc_phase3 = sc_try
                best_t = t_blend
        
        if best_t == 0.0:
            print(f"  LP {lp_iter}: no improvement (lp_time={lp_time:.1f}s)")
            # Add small perturbation to escape
            a_cur = targeted_perturb(a_cur, rng, strength=0.01)
            continue
        
        a_cur = (1.0 - best_t) * a_norm + best_t * g
        
        # CD polish after LP step
        for _ in range(3):
            if time.monotonic() > phase3_end:
                break
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        
        sc_final = compute_score(a_cur)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_cur.copy()
            print(f"  LP {lp_iter}: {sc_final:.8f} (NEW BEST, t={best_t:.2f}, lp_time={lp_time:.1f}s)")
        else:
            print(f"  LP {lp_iter}: {sc_final:.8f} (t={best_t:.2f}, lp_time={lp_time:.1f}s)")
    
    print(f"Phase 3 done: {lp_success} LP success, {lp_fail} fail. Best: {best_score:.8f}")
    
    # ============================================================
    # Phase 4: Diversification (20% time)
    # ============================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Diversification ===")
    
    for trial in range(50):
        if time.monotonic() > phase4_end - 5:
            break
        
        strategy = trial % 6
        n_cur = len(best_a)
        
        if strategy == 0:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
        elif strategy == 1:
            # Gaussian perturbation
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 2:
            # Block redistribution
            block_size = max(20, n_cur // 80)
            start = int(rng.integers(0, max(1, n_cur - block_size)))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            mask = np.ones(n_cur, dtype=bool)
            mask[start:start + block_size] = False
            a_pert[mask] += saved / max(1, int(mask.sum()))
        elif strategy == 3:
            # Frequency-domain perturbation
            A = np.fft.rfft(a_pert if 'a_pert' in dir() else best_a)
            A_real = A.real * (1.0 + 0.05 * rng.standard_normal(len(A)).real)
            a_pert = np.fft.irfft(A_real, 2 * n_cur - 1)[:n_cur]
        elif strategy == 4:
            # Sinusoidal modulation
            freq = 2.0 + 8.0 * rng.random()
            amp = 0.02 + 0.05 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        else:
            # Segment swap
            seg_len = max(20, n_cur // 25)
            s1 = int(rng.integers(0, max(1, n_cur - seg_len)))
            s2 = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg_len].copy()
            a_pert[s1:s1 + seg_len] = a_pert[s2:s2 + seg_len]
            a_pert[s2:s2 + seg_len] = tmp
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick optimization from perturbed point
        # Short Adam
        lr_short = 0.03 / np.sqrt(n_cur)
        sc_pert, a_pert = adam_lse_optimize(a_pert, 200, lr_short, 300, phase4_end - 5)
        
        # Short CD
        for _ in range(3):
            if time.monotonic() > phase4_end - 5:
                break
            sc_pert, a_pert = coordinate_descent_sweep(a_pert, rng=rng)
        
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  D4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
    
    # ============================================================
    # Phase 5: Final polish (CD + Adam)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    a_cur = best_a.copy()
    
    for _ in range(50):
        if time.monotonic() > deadline - 3:
            break
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
    
    # One more Adam pass at high beta
    if time.monotonic() < deadline - 20:
        lr_final = 0.02 / np.sqrt(len(best_a))
        sc, a_cur = adam_lse_optimize(best_a.copy(), 5000, lr_final, 500, deadline - 5)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final Adam: {sc:.8f} (NEW BEST)")
        
        # Final CD
        sc, a_cur = coordinate_descent_sweep(best_a.copy(), rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} (NEW BEST)")
    
    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
        best_score = np.inf
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
