```python
"""
Hybrid optimizer: Fast exact coordinate descent + LP-based direction finding + LSE refinement.
"""
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)
```

```python
def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. T_a @ g <= M*ones, g >= 0 where T_a is the Toeplitz matrix."""
    n = len(a)
    # Build the constraint matrix efficiently
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # This is a Toeplitz matrix with first row [a[0], a[1], ..., a[n-1], 0, ..., 0]
    # and first column [a[0], 0, 0, ..., 0]
    # Actually it's more like a "convolution matrix"
    
    # For efficiency, let's build it as a scipy sparse matrix
    from scipy.sparse import csr_matrix
    # Row k, col j: non-zero if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve LP
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or GLPK, CBC
    if g.value is not None:
        return np.clip(g.value, 0, 1000)
    return None
```

```python
def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    # T has shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n
    rows = np.zeros(0, dtype=int)
    cols = np.zeros(0, dtype=int)
    vals = np.zeros(0, dtype=float)
    
    # For efficiency, build row by row
    row_parts = []
    col_parts = []
    val_parts = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        row_parts.append(np.full(length, k, dtype=int))
        col_parts.append(np.arange(j_start, j_end + 1, dtype=int))
        val_parts.append(a[k - j_end : k - j_start : -1])  # a[k-j] for j from j_end down to j_start
    
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
    # For linear terms: t = (c*SR - 2*b) / c where c = alpha[k]

            idx = idx[order]
        t_candidates = (alpha[idx] * S_rest - 2.0 * base_inside[idx]) / alpha[idx]
        t_candidates = t_candidates[(t_candidates > 0) & (t_candidates < t_upper)]
        candidates.extend(t_candidates.tolist())
    
    # Evaluate at all candidates
```

```python
"""
Hybrid optimizer: Fast exact 1D coordinate descent + LSE refinement + LP-based direction.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Key improvements: exact 1D solver (no minimize_scalar), LP direction finding, multi-phase strategy.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient using FFT-based convolution."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j] (zero-extended)
    # = 2 * valid part of conv(a_rev, p)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fast_1d_solve(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper):
    """
    Fast exact 1D solver for coordinate descent subproblem.
    Minimize f(t) = 2n * max(OM, max_k(bk + t*ak + t^2*dk)) / (SR+t)^2
    Returns (optimal_t, optimal_value).
    """
    S_rest = float(S_rest)
    if S_rest < 1e-12:
        S_rest = 1e-12

    candidates = [0.0, t_upper]

    # Quadratic critical point: t = b2 / SR
    quad_idx = np.where(beta_coef > 0.5)[0]
    if len(quad_idx) > 0:
        b2 = float(base_inside[quad_idx[0]])
        if b2 > 0:
            t_q = b2 / S_rest
            if 0 < t_q < t_upper:
                candidates.append(float(t_q))

    # Linear critical points: t = (c*SR - 2*b) / c for c > 0
    sig = alpha[alpha > 1e-8]
    base_sig = base_inside[np.where(alpha > 1e-8)[0]]
    if len(sig) > 0:
        top_k = min(40, len(sig))
        if len(sig) > top_k:
            order = np.argsort(sig)[::-1][:top_k]
            sig_k = sig[order]
            base_k = base_sig[order]
        else:
            sig_k = sig
            base_k = base_sig
        t_cands = (sig_k * S_rest - 2.0 * base_k) / sig_k
        t_cands = t_cands[(t_cands > 0) & (t_cands < t_upper)]
        candidates.extend(t_cands.astype(float).tolist())

    # Also add some heuristic points
    if S_rest > 0:
        t_mean = max(0.0, min(t_upper, S_rest / n * 2.0))
        candidates.append(float(t_mean))
        candidates.append(float(min(t_upper, t_mean * 1.5)))

    # Evaluate
    best_t = 0.0
    S_new0 = S_rest
    max0 = max(float(outside_max), float(np.max(base_inside)))
    best_f = 2.0 * n * max0 / (S_new0 * S_new0)

    for t in candidates:
        t = float(max(0.0, min(t_upper, t)))
        S_new = S_rest + t
        if S_new < 1e-12:
            continue
        conv_in = base_inside + t * alpha + (t * t) * beta_coef
        im = float(np.max(conv_in))
        tot_max = im if im > outside_max else float(outside_max)
        f_t = 2.0 * n * tot_max / (S_new * S_new)
        if f_t < best_f:
            best_f = f_t
            best_t = t

    return best_t, best_f


def coord_descent_sweep(a, rng=None, n_sweeps=1, hot=False):
    """
    Coordinate descent sweep with fast 1D solver and incremental conv updates.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _sweep in range(n_sweeps):
        if hot:
            k_star = int(np.argmax(conv))
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
            order = np.argsort(contrib)[::-1]
        elif rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            i = int(i)
            a_i = a[i]

            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_slice = conv[k_lo:k_hi + 1].copy()
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i].astype(np.float64)
            is_2i = (ks == 2 * i)

            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside = inside_slice - old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros_like(alpha)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 4.0)
            t_upper = min(t_upper, 2000.0)

            t_opt, _ = fast_1d_solve(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper)

            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def targeted_perturb(a, rng, strength=0.03):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[n - top_count:]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if total_reduced > 0 and len(bottom_idx) > 0:
        a_pert[bottom_idx] += total_reduced / len(bottom_idx)
    return np.clip(a_pert, 0.0, 1000.0)


def lp_direction(a, M, timeout=30):
    """
    Solve LP: max sum(g) s.t. T_a @ g <= M*ones(2n-1), g >= 0.
    T_a is the Toeplitz convolution matrix of a.
    Returns the optimal g or None if timeout/failure.
    """
    import cvxpy as cp
    t0 = time.monotonic()

    n = len(a)
    # Build sparse convolution matrix
    # Row k, col j: a[k-j] if 0 <= k-j < n
    row_parts = []
    col_parts = []
    val_parts = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        if length > 0:
            row_parts.append(np.full(length, k, dtype=np.int32))
            col_parts.append(np.arange(j_start, j_end + 1, dtype=np.int32))
            # a[k-j] for j = j_start..j_end => indices k-j_end .. k-j_start
            val_parts.append(a[k - j_end:k - j_start + 1][::-1].astype(np.float64))

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    if time.monotonic() - t0 > timeout * 0.5:
        return None

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GLOP, time_limit=timeout)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, time_limit=timeout)
        except Exception:
            return None

    if time.monotonic() - t0 > timeout * 1.5:
        return None

    if g.value is not None and np.isfinite(g.value).all():
        return np.clip(g.value, 0.0, 1000.0)
    return None


def lp_update(a, timeout=20):
    """
    LP-based update: compute M, solve LP, normalize, line search.
    Returns (score, updated_a) or (inf, a) on failure.
    """
    n = len(a)
    conv = fast_conv(a)
    M = float(np.max(conv))
    S = float(a.sum())

    g = lp_direction(a, M, timeout=timeout)
    if g is None:
        return compute_score(a), a

    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return compute_score(a), a

    # Normalize g to have same sum as a
    g = g * (S / g_sum)

    # Line search on (1-t)*a + t*g
    best_sc = compute_score(a)
    best_t = 0.0
    for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
        a_try = (1.0 - t) * a + t * g
        a_try = np.clip(a_try, 0.0, 1000.0)
        sc = compute_score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.clip(a_new, 0.0, 1000.0)
    return best_sc, a_new

            print(f"  LP trial {trial}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LP trial {trial}: {sc:.8f} ({elapsed:.1f}s)")

        # CD polish after LP
        a_cur = a_new.copy()
        for _ in range(5):
            if time.monotonic() > phase3_deadline - 5:
                break
            sc2, a_cur = coord_descent_sweep(a_cur, rng=rng, hot=True)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_cur.copy()

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Perturbation + CD cycles (10%)
    # ============================================================
    phase4_deadline = t_start + 0.90 * total_budget
    print(f"\n=== Phase 4: Perturbation Cycles ({phase4_deadline - t_start:.0f}s) ===")

    for trial in range(40):
        if time.monotonic() > phase4_deadline - 5:
            break
        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            scale = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.04 * rng.random())
        elif strategy == 2:
            # Block zeroing + redistribution
            block_size = max(20, n_cur // 50)
            start = int(rng.integers(0, n_cur - block_size))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            rest = np.ones(n_cur, dtype=bool)
            rest[start:start + block_size] = False
            a_pert[rest] += saved / max(1, int(rest.sum()))
        elif strategy == 3:
            # Sinusoidal modulation
            freq = 1.0 + 15.0 * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        elif strategy == 4:
            # Segment swap
            seg_len = max(20, n_cur // 20)
            s1 = int(rng.integers(0, max(1, n_cur - seg_len)))
            s2 = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg_len].copy()
            a_pert[s1:s1 + seg_len] = a_pert[s2:s2 + seg_len]
            a_pert[s2:s2 + seg_len] = tmp
        else:
            # Mirror perturbation
            half = n_cur // 2
            a_pert = best_a.copy()
            mix = 0.3 + 0.4 * rng.random()
            a_pert[:half] = (1 - mix) * a_pert[:half] + mix * a_pert[n_cur - 1 - np.arange(half)][::-1]

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick CD from perturbed
        for sw in range(6):
            if time.monotonic() > phase4_deadline - 5:
                break
            sc_pert, a_pert = coord_descent_sweep(a_pert, rng=rng, hot=(sw % 2 == 0))

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 8 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Phase 5: Final Polish (5%)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")
    a_cur = best_a.copy()
    for _ in range(30):
        if time.monotonic() > deadline - 3:
            break
        sc, a_cur = coord_descent_sweep(a_cur, rng=rng, hot=True)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One more LSE if time permits
    if time.monotonic() < deadline - 20:
        try:
            sc, a_opt = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=40000)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  Final LSE: {sc:.8f} (NEW BEST)")
            # Quick CD after
            sc2, a_cur = coord_descent_sweep(best_a, rng=rng, hot=True)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_cur.copy()
                print(f"  Final CD: {sc2:.8f} (NEW BEST)")
        except Exception:
            pass

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
