Uses improved spectral flattening for initialization, high-beta L-BFGS for continuous

import numpy as np

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)

    if np.sum(a0) < 1e-6:

def spectral_flatten(a0, n_iters=500, lam_start=0.9, lam_end=0.1):
    """Iterative spectral flattening with annealing."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_orig = float(np.sum(a))
    N = 1 << (2 * n - 1).bit_length()
    half = N // 2
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start + (lam_end - lam_start) * progress
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target: median of non-DC magnitudes
        if N > 2:
            non_dc = mag[1:half+1] if N % 2 == 0 else mag[1:half+1]
            target = np.median(non_dc)
        else:
            target = mag[0]
        new_mag = mag.copy()
        for k in range(1, half + 1):
            new_val = (1.0 - lam) * mag[k] + lam * target
            new_mag[k] = new_val
            if k < N - k:
                # rfft only stores up to N//2, so we don't need to set N-k
                pass
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.irfft(new_A, N))[:n]
        a = np.maximum(a_new, 0.0)
        S = float(np.sum(a))
        if S > 1e-12:
            a = a * (S_orig / S)
    return a


def spectral_flatten_full(a0, n_iters=500, lam_start=0.9, lam_end=0.1):
    """Spectral flattening using full FFT (handles both sides)."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_orig = float(np.sum(a))
    N = 1 << (2 * n - 1).bit_length()
    half = N // 2
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start + (lam_end - lam_start) * progress
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = np.concatenate([mag[1:half], mag[N-half+1:]]) if N > 2 else mag[1:]
        if len(non_dc) > 0:
            target = np.median(non_dc)
        else:
            target = 0
        new_mag = mag.copy()
        for k in range(1, half + 1):
            new_val = (1.0 - lam) * mag[k] + lam * target
            new_mag[k] = new_val
            new_mag[N - k] = new_val
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = float(np.sum(a))
        if S > 1e-12:

    return a


def subgrad_polish(a, total_iters=1000):
    """Subgradient method for the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Polyak step
        step = max(0.0, (sc - 1.0)) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        # Check if it improved
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
        else:
            # Halve the step
            a = a - 0.5 * step * grad
            a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def multi_peak_subgrad(a, total_iters=500, n_peaks=5):
    """Subgradient method tracking multiple peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # Weighted subgradient
        weights = np.exp(-0.5 * (c[top_idx] - M))  # Important peaks get more weight
        weights = weights / np.sum(weights)
        
        sg_M = np.zeros(n)
        for w, k in zip(weights, top_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg_M[j_start:j_end + 1] += 2.0 * w * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = max(0.0, (2.0 * n * M / (S * S) - 1.0)) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        if sc_new <= 2.0 * n * M / (S * S):
            a = a_new
        else:
            a = a - 0.5 * step * grad
            a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_reduction(a, strength=0.03):
    """Reduce values that contribute most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    a_pert = a.copy()
    # Reduce top contributors, add to bottom
    top_count = max(3, n // 100)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = max(3, n // 50)
    bottom_idx = np.argsort(contrib)[:bottom_count]
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)

"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Uses annealed spectral flattening for initialization, aggressive L-BFGS with high-beta
LSE smoothing, multi-peak subgradient methods, and guided local search.
Key improvements over prior approach: annealed flattening, much higher betas,
peak-aware search, and iterative refinement loops.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    n = len(x)
    c = fast_conv(x)
    S = float(np.sum(x))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    shifted = beta * (c - c_max)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / np.sum(exp_s)
    a_rev = x[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:2 * n - 1]
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):

    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_flatten_annealed(a0, n_iters, lam_start, lam_end):
    a = a0.astype(np.float64).copy()
    n = len(a)
    S_orig = float(np.sum(a))
    if S_orig < 1e-12:
        a[:] = 0.5
        S_orig = n * 0.5
    N = 1 << (2 * n - 1).bit_length()
    half = N // 2
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = lam_start + (lam_end - lam_start) * progress
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        mask = np.ones(N, dtype=bool)
        mask[0] = False
        if N % 2 == 0:
            mask[half] = False
        non_dc = mag[mask]
        target = np.median(non_dc) if len(non_dc) > 0 else mag[0]
        new_mag = mag.copy()
        for k in range(1, N):
            if mask[k]:
                new_val = (1.0 - lam) * mag[k] + lam * target
                new_mag[k] = new_val
                new_mag[N - k if N - k < N else 0] = new_val
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = float(np.sum(a))
        if S > 1e-12:
            a = a * (S_orig / S)
    return a


def spectral_flatten_fast(a0, n_iters, lam):

    half = N // 2

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc = np.concatenate([mag[1:half], mag[N - half + 1:]]) if N > 2 else np.array([1.0])
        target = np.median(non_dc)
        new_mag = mag.copy()

def subgrad_polish(a, total_iters):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

        c = fast_conv(a)

        if S < 1e-12:
            break
        M = float(np.max(c))
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:

        step = max(0.0, sc - 1.5) / (gn * gn + 1e-30)

        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new <= sc:
            a = a_new
        else:
            a = a - 0.5 * step * grad
            a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def multi_peak_subgrad(a, total_iters, n_peaks):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[top_idx]
        weights = np.exp(-1.0 * (peak_vals - M))
        weights = weights / np.sum(weights)
        sg_M = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg_M[j_start:j_end + 1] += 2.0 * wi * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        sc = 2.0 * n * M / S2
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = max(0.0, sc - 1.5) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        if sc_new <= sc:
            a = a_new
        else:
            a = a - 0.5 * step * grad
            a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_reduction(a, strength):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    a_pert = a.copy()
    top_count = max(3, n // 80)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_count = max(3, n // 40)
    bottom_idx = np.argsort(contrib)[:bottom_count]
    total_reduced = 0.0
    for j in top_idx:
        amt = a_pert[j] * strength
        a_pert[j] -= amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def generate_base_inits(n, rng):
    inits = []
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    x = np.arange(n, dtype=np.float64)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x / n))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * 7.0 * x / n))
    t = x / n
    inits.append(0.1 + 0.8 * (2.0 * t - 1.0) ** 2)
    inits.append(0.5 + 0.4 * np.sin(np.pi * t))
    inits.append(0.3 + 0.4 * np.cos(np.pi * x * x / n))
    for sc in [0.1, 0.2, 0.5]:
        inits.append(np.clip(0.5 + sc * rng.standard_normal(n), 0.01, 10.0))
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer: annealed spectral flattening -> aggressive L-BFGS ->
    multi-peak subgradient -> guided local search -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    margin = 15.0
    deadline = t_start + budget_s - margin

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print("  [{}] NEW BEST: {:.8f} (n={}, t={:.0f}s)".format(label, sc, len(a), elapsed))
            return True
        return False

    if parent is not None:
        p_score = compute_score(parent)
        print("Parent: n={}, score={:.8f}".format(len(parent), p_score))
        report(p_score, parent, "parent")
    else:
        print("No parent available")

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 8192]:
        if n not in lengths:
            lengths.append(n)
    if budget_s > 1800:
        lengths.append(16384)
    lengths = sorted(set(lengths))[:4]
    print("Lengths: {}".format(lengths))

    # ============ Phase 1: Spectral Flattening Initialization ============
    sf_deadline = t_start + 0.12 * (budget_s - margin)
    print("\n--- Phase 1: Spectral Flattening (until {:.0f}s) ---".format(sf_deadline - t_start))

    sf_results = []
    for n in lengths:
        if time.monotonic() > sf_deadline:
            break
        base_inits = generate_base_inits(n, rng)
        for bi, binit in enumerate(base_inits):
            if time.monotonic() > sf_deadline:
                break
            t0 = time.monotonic()
            # Annealed flattening
            sf_a = spectral_flatten_annealed(binit, 300, 0.9, 0.1)
            sc = compute_score(sf_a)
            sf_results.append((sc, sf_a, "sf_n{}_b{}".format(n, bi)))
            report(sc, sf_a, "sf_n{}_b{}".format(n, bi))
            el = time.monotonic() - t0
            print("    n={} b={}: score={:.6f} ({:.1f}s)".format(n, bi, sc, el))

    # ============ Phase 2: L-BFGS with aggressive beta continuation ============
    lbfgs_deadline = t_start + 0.55 * (budget_s - margin)
    print("\n--- Phase 2: L-BFGS (until {:.0f}s) ---".format(lbfgs_deadline - t_start))

    # Build candidate pool
    pool = []
    if parent is not None:
        pool.append((compute_score(parent), parent.copy(), "parent"))
    for sc, a, label in sf_results:
        pool.append((sc, a.copy(), label))
    pool.sort(key=lambda x: x[0])

    for n in lengths:
        if time.monotonic() > lbfgs_deadline:
            break
        print("\n  >> n = {}".format(n))
        cand_n = [(sc, a, lab) for sc, a, lab in pool if len(a) == n]
        if not cand_n:
            a_fresh = spectral_flatten_fast(np.ones(n) * 0.5, 150, 0.7)
            cand_n.append((compute_score(a_fresh), a_fresh, "fresh"))
        cand_n = cand_n[:min(5, len(cand_n))]

        for ci, (sc_init, a_init, label) in enumerate(cand_n):
            if time.monotonic() > lbfgs_deadline:
                break
            print("    [{}/{}] {} init={:.6f}".format(ci, len(cand_n), label, sc_init))
            a_cur = a_init.copy()
            beta_schedule = [50, 200, 1000, 5000, 20000]
            for beta in beta_schedule:
                if time.monotonic() > lbfgs_deadline:
                    break
                remaining = lbfgs_deadline - time.monotonic()
                if remaining < 10:
                    break
                est_iter_time = 0.004 * (n / 1024.0) + 0.003
                maxiter = min(8000, max(300, int(remaining * 0.6 / max(est_iter_time, 0.01))))
                maxfun = max(2000, maxiter * 40)
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter, maxfun)
                a_cur = a_opt
                report(sc, a_opt, "lbfgs n={} c={} b={}".format(n, ci, beta))
            sc_final = compute_score(a_cur)
            report(sc_final, a_cur, "lbfgs_done n={} c={}".format(n, ci))
            print("    n={} c={}: final={:.6f}".format(n, ci, sc_final))

    print("\nAfter Phase 2: best={:.8f}".format(best_score))

    # ============ Phase 3: High-beta refinement loop ============
    refine_deadline = t_start + 0.72 * (budget_s - margin)
    print("\n--- Phase 3: High-Beta Refinement (until {:.0f}s) ---".format(refine_deadline - t_start))

    if best_a is not None:
        for beta in [10000, 50000, 100000]:
            if time.monotonic() > refine_deadline:
                break
            remaining = refine_deadline - time.monotonic()
            if remaining < 10:
                break
            maxiter = min(8000, max(500, int(remaining * 0.5 / 0.02)))
            maxfun = max(3000, maxiter * 30)
            sc, a_opt = run_lbfgs(best_a, beta, maxiter, maxfun)
            report(sc, a_opt, "refine beta={}".format(beta))

    # ============ Phase 4: Guided local search ============
    local_deadline = t_start + 0.88 * (budget_s - margin)
    print("\n--- Phase 4: Local Search (until {:.0f}s) ---".format(local_deadline - t_start))

    trial = 0
    while time.monotonic() < local_deadline and best_a is not None:
        trial += 1
        n_best = len(best_a)
        strategy = trial % 7

        if strategy == 0:
            scale = 0.015 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_reduction(best_a, 0.02 + 0.03 * rng.random())
        elif strategy == 2:
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            noise = 0.015 * rng.standard_normal(len(A)) + 0.015j * rng.standard_normal(len(A))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            pert = 0.025 * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Peak-aware: reduce at positions contributing to peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            contrib = np.zeros(n_best)
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            if j_end >= j_start:
                contrib[j_start:j_end + 1] = best_a[j_start:j_end + 1][::-1]
            # Reduce top contributors, spread to bottom
            a_pert = best_a.copy()
            top_k = max(5, n_best // 50)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            bot_idx = np.argsort(contrib)[:top_k]
            total_red = 0.0
            for j in top_idx:
                amt = a_pert[j] * 0.05
                a_pert[j] -= amt
                total_red += amt
            per = total_red / max(1, len(bot_idx))
            for j in bot_idx:
                a_pert[j] += per
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 5:
            # Mix with fresh spectral-flattened
            fresh = spectral_flatten_fast(np.clip(0.5 + 0.3 * rng.standard_normal(n_best), 0.01, 10.0),
                                          80, 0.6)
            t_mix = 0.1 + 0.3 * rng.random()
            a_pert = (1.0 - t_mix) * best_a + t_mix * fresh
            S_orig = float(np.sum(best_a))
            a_pert = a_pert * (S_orig / max(1e-12, float(np.sum(a_pert))))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block perturbation
            block_size = max(1, n_best // 128)
            n_blocks = n_best // block_size + 1
            block_pert = (rng.random(n_blocks) - 0.5) * 0.06
            a_pert = best_a * (1.0 + np.repeat(block_pert, block_size)[:n_best])
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        remaining = local_deadline - time.monotonic()
        if remaining < 5:
            break
        maxiter = min(4000, max(300, int(remaining * 0.3 / 0.015)))
        maxfun = max(1500, maxiter * 30)
        sc, a_opt = run_lbfgs(a_pert, 3000, maxiter, maxfun)
        report(sc, a_opt, "local #{} s={}".format(trial, strategy))

    # ============ Phase 5: Final polish ============
    print("\n--- Phase 5: Final Polish ---")

    if best_a is not None:
        # Multi-peak subgradient
        sc, a_mp = multi_peak_subgrad(best_a, 600, 5)
        report(sc, a_mp, "multi_peak")

        # Standard subgradient
        sc, a_sg = subgrad_polish(best_a, 1000)
        report(sc, a_sg, "subgrad")

        # L-BFGS from best
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs(best_a, 50000, 5000, 150000)
            report(sc, a_opt, "final_lbfgs_50k")

        # Another subgradient
        sc, a_sg2 = subgrad_polish(best_a, 500)
        report(sc, a_sg2, "final_subgrad")

        # One more high-beta L-BFGS
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs(best_a, 200000, 3000, 100000)
            report(sc, a_opt, "final_lbfgs_200k")

    # ============ Final output ============
    if best_a is None:
        out = [0.5] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]

    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print("\n" + "=" * 60)
    print("FINAL: internal={:.8f}, verified={:.8f}".format(best_score, verified))
    print("Length: {}, Time: {:.1f}s / {}s".format(len(out), elapsed_total, budget_s))
    print("=" * 60)
    return out
