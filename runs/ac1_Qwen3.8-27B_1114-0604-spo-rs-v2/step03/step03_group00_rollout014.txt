def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for scipy minimize."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_max = float(scaled.max())
    exp_scaled = np.exp(scaled - scaled_max)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    p = exp_scaled / exp_sum
    L = c_max + (scaled_max + np.log(exp_sum)) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    cross = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def error_reduction(a, n_iters=200, alpha=0.7, threshold_frac=0.0):
    """

    best_sc = compute_score(a)

        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15 * max(1.0, conv_max):

        err_pad = np.zeros(M, dtype=np.complex128)

        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):

        A_new = A + d
        a_new = np.fft.ifft(A_new).real[:n]

        if sc < best_sc:

def spectral_shaping(a, n_iters=100, alpha=0.3):

    """

    S = float(a.sum())

    for it in range(n_iters):

        a_pad[:n] = a

        mag = np.abs(A)
        phase = np.angle(A)
        power = mag * mag

        # Compute current convolution peak

            reduction[mask] = 1.0 - alpha * excess

        # Also apply mild global compression
        power_new = power * reduction
        # Normalize to maintain energy

def coordinate_descent_peak(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak-contributing elements with golden section."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for ji in top_global:
            a_old_j = a[ji]
            if a_old_j < 1e-10:
                continue
            # Golden section search in [0, 2*a_old_j]
            lo, hi = 0.0, 2.0 * a_old_j
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for _ in range(8):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                a[ji] = m1
                sc1 = compute_score(a)
                a[ji] = m2
                sc2 = compute_score(a)
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
                a[ji] = best_val
                if abs(hi - lo) < 1e-10:
                    break
            # Try boundary values too
            for cv in [0.0, a_old_j * 0.3, a_old_j * 1.5]:
                a[ji] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[ji] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    step_sizes = []
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent, clip=True).copy())
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Tapered (reduces edge peaks)
    taper = (1.0 - (2.0 * x - 1.0) ** 4)
    inits.append(0.1 + 0.9 * taper)
    # Multi-lobe
    inits.append(0.3 + 0.5 * np.abs(np.cos(2.0 * np.pi * x)))
    #.block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 6962, 8192, 12288]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: len={0 if parent is None else len(parent)}")

    # ==================================================================
    # Phase 1: Global search via error reduction + spectral shaping (35%)
    # ==================================================================
    phase1_deadline = t_start + 0.35 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 1: Global Search (deadline t={phase1_deadline - t_start:.0f}s) ===")

    candidates = []
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 8.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0)
            if a0.sum() < 1e-6:
                continue
            a0 = a0 / a0.sum()

            # Error reduction with different params
            a_er = error_reduction(a0, n_iters=120, alpha=0.7)
            sc_er = compute_score(a_er)
            report(sc_er, a_er, f"ER n={n} i={i}")
            candidates.append((sc_er, a_er))

            # Spectral shaping
            a_ss = spectral_shaping(a_er, n_iters=80, alpha=0.2)
            sc_ss = compute_score(a_ss)
            report(sc_ss, a_ss, f"SS n={n} i={i}")
            candidates.append((sc_ss, a_ss))

    # Sort candidates and keep top few
    candidates.sort(key=lambda x: x[0])
    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS refinement with adaptive beta (35%)
    # ==================================================================
    phase2_deadline = t_start + 0.70 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 2: L-BFGS Refinement (deadline t={phase2_deadline - t_start:.0f}s) ===")

    # Take the best candidate and refine with increasing beta
    start_a = best_a.copy()
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 4000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt, _ = run_lbfgs(start_a, beta, maxiter=8000, maxfun=150000)
        improved = report(sc, a_opt, f"LBFGS b={beta}")
        if improved:
            start_a = a_opt.copy()

    # Also try from second-best candidate
    if len(candidates) > 1 and time.monotonic() < phase2_deadline:
        sc2, a2 = candidates[1]
        if abs(len(a2) - len(start_a)) <= 0.1 * len(start_a):
            for beta in [200, 500]:
                if time.monotonic() > phase2_deadline:
                    break
                sc, a_opt, _ = run_lbfgs(a2, beta, maxiter=5000, maxfun=80000)
                report(sc, a_opt, f"LBFGS alt b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Active-set Newton + local search (20%)
    # ==================================================================
    phase3_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 3: Active-Set + Local Search (deadline t={phase3_deadline - t_start:.0f}s) ===")

    # Active-set Newton
    a_asn = active_set_newton(best_a, n_iters=200, top_k=5)
    sc_asn = compute_score(a_asn)
    report(sc_asn, a_asn, "ActiveSet")

    # L-BFGS from active-set result
    if time.monotonic() < phase3_deadline:
        sc, a_opt, _ = run_lbfgs(a_asn, 1000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "AS+LBFGS")

    # Diverse perturbations
    n_best = len(best_a)
    perturbations = []
    strategies = [
        "noise_small", "noise_med", "taper", "block",
        "noise_large", "spectral", "noise_small2", "noise_med2"
    ]
    for si, strat in enumerate(strategies):
        if time.monotonic() > phase3_deadline:
            break
        a_pert = best_a.copy()
        if strat == "noise_small":
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
        elif strat == "noise_med":
            a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
        elif strat == "noise_large":
            a_pert = best_a * (1.0 + 0.1 * rng.standard_normal(n_best))
        elif strat == "taper":
            x = np.arange(n_best) / n_best
            taper = 1.0 - 0.3 * (2 * x - 1) ** 2
            a_pert = best_a * taper
        elif strat == "block":
            nb = 10 + int(rng.integers(0, 10))
            bs = max(1, n_best // nb)
            for b in range(nb):
                s = b * bs
                e = min(s + bs, n_best)
                a_pert[s:e] *= 0.7 + 0.6 * rng.random()
        elif strat == "spectral":
            a_pert = spectral_shaping(best_a, n_iters=50, alpha=0.15)
        elif strat == "noise_small2":
            x = np.arange(n_best, dtype=np.float64) / n_best
            freq = 2.0 + 6.0 * rng.random()
            a_pert = best_a * (1.0 + 0.03 * np.cos(2 * np.pi * freq * x))
        else:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(10, n_best // 50)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.08
                a_pert[gi] -= red
                a_pert[gi] = max(a_pert[gi], 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
                a_pert = a_pert / a_pert.sum()

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = a_pert.sum()
        if S < 1e-6:
            continue
        a_pert = a_pert / S

        # Short L-BFGS
        sc, a_opt, _ = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"pert {strat}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (remaining time)
    # ==================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Coordinate descent
    a_cd = coordinate_descent_peak(best_a, n_sweeps=4, n_top=80)
    sc_cd = compute_score(a_cd)
    report(sc_cd, a_cd, "coord_descent")

    # Subgradient polish
    if time.monotonic() < deadline - 30:
        a_sp = subgradient_polish(best_a, total_iters=3000)
        sc_sp = compute_score(a_sp)
        report(sc_sp, a_sp, "subgrad")

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS b=5000")

    # Error reduction from best
    if time.monotonic() < deadline - 20:
        a_er = error_reduction(best_a, n_iters=100, alpha=0.8, threshold_frac=0.5)
        sc_er = compute_score(a_er)
        report(sc_er, a_er, "final ER")

    # One more round: L-BFGS from the absolute best
    if time.monotonic() < deadline - 15:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS2")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
