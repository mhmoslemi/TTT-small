        # This is (p * a_flipped)[n-1-j] where a_flipped[i] = a[n-1-i]
        # Or: it's the (n-1-j)-th element of convolve(p, a_flipped)
        # Actually simplest: it's the convolution of p with a, evaluated at index (n-1-j) + offset
        # Let me just use np.convolve:
        # conv(p, a)[n-1-j] = sum_i p[i]*a[n-1-j-i] ... not quite what we want.
        # 
        # OK let me just do: for each j, sum_k p[k]*a[k-j] where 0<=k<n and 0<=k-j<n
        # = sum_{k=max(j,0)}^{min(n-1, n-1+j)} p[k]*a[k-j] ... but j is in [0,n-1]
        # For j in [0,n-1]: k ranges from j to n-1, and a[k-j] ranges from a[0] to a[n-1-j]
        # So sum_{k=j}^{n-1} p[k]*a[k-j] = sum_{m=0}^{n-1-j} p[m+j]*a[m]
        # This is dot(p[j:], a[:n-j])
        # 
        # For efficiency with large n, use the fact that this is a cross-correlation:
        # crosscorr[j] = dot(p[j:], a[:n-j]) for j=0,...,n-1
        # This can be computed as: ifft(FFT(p, M) * conj(FFT(a, M)))[n-1 : 2n-1] ... 
        # Actually the standard formula: (f ⋆ g)[j] = sum_k f[k] g[k+j] = ifft(FFT(f) * conj(FFT(g)))[j + len(g) - 1]
        # Hmm let me just be explicit:
        # We want h[j] = sum_{m=0}^{n-1-j} p[m+j]*a[m] for j=0,...,n-1
        # This is the (n-1-j) element of the correlation... 
        # Let's use: h = np.correlate(p, a, mode='full')[n-1:2*n-1][::-1]
        # Or more simply, compute via FFT:
        M2_fft = 1 << (2 * n).bit_length()
        P2 = np.fft.rfft(p, M2_fft)
        A2 = np.fft.rfft(a, M2_fft)
        # ifft(P2 * conj(A2)) gives cross-correlation: (p ⋆ a)[k] = sum_i p[i] a[i+k]
        cc2 = np.fft.irfft(P2 * np.conj(A2), M2_fft)
        # h[j] = sum_m p[m+j]*a[m] = (a ⋆ p)[j] = sum_m a[m] p[m+j]
        # (a ⋆ p)[j] = sum_m a[m]*p[m+j] = ifft(conj(FFT(p)) * FFT(a))[j] (with appropriate indexing)
        # Standard: cross-correlation (f ⋆ g)[k] = sum_n f[n+k] g[n] = ifft(conj(FFT(f)) FFT(g))[k + len(g) - 1]
        # So (a ⋆ p)[j] = sum_n a[n+j] p[n] = ifft(conj(FFT(a)) * FFT(p))[j + len(p) - 1]
        # = ifft(conj(FA) * FP)[j + n - 1]
        cross = np.fft.irfft(np.conj(A2) * P2, M2_fft)
        h = cross[n-1:2*n-1]
        # h[j] = sum_m a[m+j]*p[m] = sum_m p[m]*a[m+j] ... this is what we want!
        # Wait: we want sum_k p[k]*a[k-j] = sum_m p[m+j]*a[m] (substituting k=m+j)
        # = h[j] where h[j] = sum_m p[m+j]*a[m]
        # And from cross-correlation: (p ⋆ a)[j] = sum_m p[m+j]*a[m]
        # (p ⋆ a)[j] = ifft(conj(FFT(p)) * FFT(a))[j + n - 1]
        cross2 = np.fft.irfft(np.conj(P2) * A2, M2_fft)
        h2 = cross2[n-1:2*n-1]
        # h2[j] should be sum_m p[m+j]*a[m] = sum_k p[k]*a[k-j]
        grad = h2 + a  # a[j] * sum_k p[k] = a[j] * 1 = a[j]
    
    # For normalized a, the gradient of the score w.r.t. a (before normalization) is:
    # score = 2n * L (with a already normalized to sum=1)
    # But we need to account for the normalization constraint.
    # If we optimize in the "unconstrained" space with the softmax-free approach:
    # grad_score[j] = 2n * grad[j] - 2n * L * (2*a[j]/S) ... this gets complex.
    # 
    # Simpler: since we normalize after each step, just return the gradient of 2n*L w.r.t. a
    # (treating a as having sum=1 already), and the optimizer will handle the rest.
    grad_score = 2.0 * n * grad
    return score, grad_score


def adam_lse_adaptive(a_init, n_steps=50000, beta_start=50, beta_end=5000,
                      lr=0.01, time_limit_s=None, t_start=None,
                      normalize=True):
    """Adam with adaptive LSE beta (annealing from smooth to sharp)."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, 1e10).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Adaptive beta: exponential annealing
        frac = t / n_steps
        beta_t = beta_start * (beta_end / beta_start) ** frac
        
        lr_t = lr * (1.0 / (1.0 + 0.01 * t))
        
        score, grad = lse_score_grad_a(a, beta_t)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        if normalize:
            a = normalize_a(a)
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_smooth(a_init, beta=500, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE smoothing on normalized a."""
    n = len(a_init)
    a0 = np.clip(a_init, 1e-10, 1e10).copy()
    a0 = a0 / a0.sum()
    
    def obj_and_grad(x):
        x = np.clip(x, 1e-10, None)
        x = x / x.sum()
        score, grad = lse_score_grad_a(x, beta)
        return float(score), grad.astype(np.float64)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1e10)
    s = a_out.sum()
    if s < 1e-12:
        a_out = np.ones(n) / n
    else:
        a_out = a_out / s
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def block_coord_descent(a, block_size=20, n_sweeps=50, time_limit_s=None, t_start=None):
    """Block coordinate descent: optimize groups of coordinates jointly via simple line search."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1e10).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf
    
    n_blocks = max(1, n // block_size)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Shuffle block order
        perm = np.arange(n_blocks)
        np.random.shuffle(perm)
        
        for bi in perm:
            if time.monotonic() > deadline:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            
            a_block = a[start:end].copy()
            best_block = a_block.copy()
            best_local = compute_score(a)
            
            # Try several perturbations of the block
            for trial in range(5):
                if trial == 0:
                    candidate = a_block.copy()
                elif trial == 1:
                    candidate = a_block * np.random.uniform(0.5, 1.5, size=end - start)
                elif trial == 2:
                    candidate = a_block * np.exp(np.random.normal(0, 0.3, size=end - start))
                elif trial == 3:
                    # Redistribute: reduce peak, increase trough
                    mx = candidate.max()
                    mn = candidate.min()
                    if mx > 0:
                        candidate = candidate * 0.8 + 0.2 * mx
                    candidate = candidate / candidate.sum() * a_block.sum()
                else:
                    # Mirror: reverse the block
                    candidate = a_block[::-1].copy()
                
                candidate = np.clip(candidate, 0.0, 1e10)
                a[start:end] = candidate
                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_block = candidate.copy()
            
            a[start:end] = best_block
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
    
    return best_score, a


def simulated_annealing(a_init, n_iters=50000, t_start_temp=0.1, t_end_temp=0.001,
                        perturbation_scale=0.05, time_limit_s=None, t_start=None):
    """Simulated annealing with structured perturbations."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, 1e10).copy()
    a = normalize_a(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    for t in range(1, n_iters + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        progress = t / n_iters
        temperature = t_start_temp * (t_end_temp / t_start_temp) ** progress
        
        # Generate perturbation
        strategy = np.random.randint(0, 5)
        a_new = a.copy()
        
        if strategy == 0:
            # Gaussian perturbation
            a_new = a + perturbation_scale * (1 - progress) * np.random.normal(0, 1, n)
        elif strategy == 1:
            # Block swap
            i = np.random.randint(0, n - 10)
            w = np.random.randint(5, max(6, n // 10))
            j = np.random.randint(0, n - w)
            a_new = a.copy()
            a_new[i:i+w], a_new[j:j+w] = a_new[j:j+w].copy(), a_new[i:i+w].copy()
        elif strategy == 2:
            # Frequency-domain perturbation
            A = np.fft.rfft(a, 2 * n)
            A += perturbation_scale * (1 - progress) * np.random.normal(0, 1, len(A))
            a_new = np.fft.irfft(A, 2 * n)[:n]
        elif strategy == 3:
            # Targeted: reduce high-variance regions
            c = fast_conv(a)
            k_max = int(c.argmax())
            js = max(0, k_max - n + 1)
            je = min(k_max, n - 1)
            a_new[js:je+1] *= np.random.uniform(0.5, 0.9, je - js + 1)
        else:
            # Random coordinate amplification
            idx = np.random.choice(n, size=max(1, n // 10), replace=False)
            a_new[idx] *= np.random.uniform(1.1, 2.0, size=len(idx))
        
        a_new = np.clip(a_new, 0.0, 1e10)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new = a_new / s
        
        new_score = compute_score(a_new)
        
        # Accept/reject
        if new_score < current_score:
            accept = True
        else:
            delta = new_score - current_score
            accept = np.random.random() < np.exp(-delta / max(temperature, 1e-10))
        
        if accept:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    
    return best_score, best_a


def gen_linear_decrease(n):
    """Linearly decreasing sequence."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def gen_exponential_decrease(n, rate=0.5):
    """Exponentially decreasing sequence."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    return a / a.sum()


def gen_cosine_window(n):
    """Cosine-squared window."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** 2
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def gen_bimodal(n):
    """Bimodal: large at ends, small in middle."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.cos(2 * np.pi * x) ** 2
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def gen_ramp_up(n):
    """Linearly increasing."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def gen_two_level(n):
    """Two-level sequence (Lempel-Garden type)."""
    a = np.zeros(n)
    # Random pattern with two levels
    pattern = np.random.randint(0, 2, n)
    a[pattern == 0] = 1.0
    a[pattern == 1] = 0.3
    return a / a.sum()


def gen_frequency_flat(n):
    """Sequence designed to have flat spectrum (via random phases)."""
    # Try to make |A(omega)|^2 as flat as possible
    # A[m] = |A[m]| * exp(i*phase_m)
    # We want a = IDFT(A) >= 0
    # Start with |A| = 1 (flat), random phases
    for _ in range(100):
        phases = np.random.uniform(0, 2 * np.pi, n // 2 + 1)
        A = np.exp(1j * phases)
        a = np.fft.irfft(A * np.sqrt(n), 2 * n)[:n]
        if np.all(a >= -1e-10):
            a = np.clip(a, 0, None)
            if a.sum() > 1e-10:
                return a / a.sum()
    # Fallback
    return gen_linear_decrease(n)


def gen_golay_inspired(n):
    """Generate a sequence inspired by Golay/Barker constructions."""
    # Use a Barker-like pattern and expand
    base = np.array([1, 1, 1, -1, 1, -1, 1], dtype=np.float64)
    # Take absolute value and tile
    a = np.tile(np.abs(base), n // len(base) + 1)[:n].astype(np.float64)
    # Add some structure: multiply by a window
    window = np.linspace(1, 0.3, n)
    a = a * window
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def gen_reversal_mixed(n, seed=42):
    """Mix of forward and reversed triangular shapes."""
    rng = np.random.RandomState(seed)
    a1 = np.linspace(1, 0.1, n // 2)
    a2 = np.linspace(0.1, 0.5, n - n // 2)
    a = np.concatenate([a1, a2])
    # Add noise
    a += rng.normal(0, 0.05, n)
    a = np.clip(a, 1e-8, None)
    return a / a.sum()


def gen_quadratic_shape(n, c1=2.0, c2=1.0):
    """Quadratic shape: a[i] = c1 + c2*(i/(n-1) - 0.5)^2."""
    x = np.arange(n, dtype=np.float64) / (n - 1) - 0.5
    a = c1 + c2 * x * x
    a = np.clip(a, 1e-8, None)
    return a / a.sum()
