def peak_targeted_refine(a, time_budget, t_start):
    """Refine by targeting the peak of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)
        if score < best_score:
            best_score = score
        
        k_star = int(np.argmax(c))
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        improved = False
        # Try reducing peak-contributing elements one by one
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                return best_score, a
            old_val = a[j]
            if old_val < 1e-15:
                continue
            best_local = old_val
            best_local_score = best_score
            
            # Binary search for optimal value of a[j]
            lo, hi = 0.0, old_val * 1.5
            for _ in range(8):
                mid = (lo + hi) / 2
                a[j] = mid
                sc = compute_score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = mid
                    lo = mid
                else:
                    hi = mid
            
            a[j] = best_local
            if best_local_score < best_score:
                best_score = best_local_score
                improved = True
            else:
                a[j] = old_val
        
        if not improved:
            # Try moving mass from peak to non-peak
            non_peak = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
            if len(non_peak) == 0:
                break
            
            # Random moves
            for _ in range(50):
                if time.monotonic() > deadline:
                    return best_score, a
                j_src = int(rng.integers(js, je + 1))
                j_dst = int(non_peak[int(rng.integers(len(non_peak)))])

                a[j_dst] += move

                    best_score = sc
                    improved = True
                else:
                    a[j_src] += move
                    a[j_dst] -= move

                break

def make_arcsine_init(n, taper=0.0):
    """Create an arcsine-like initialization."""
    x = np.arange(n, dtype=np.float64) / n
    # Avoid singularities at 0 and 1
    eps = 1e-6
    x = np.clip(x, eps, 1 - eps)
    a = 1.0 / (np.pi * np.sqrt(x * (1 - x)))
    a = np.clip(a, 0, None)
    # Apply edge taper
    if taper > 0:
        for i in range(int(taper * n)):
            t = (i + 1) / (taper * n)
            a[i] *= t * t
            a[n - 1 - i] *= t * t
    a /= a.sum()
    return a

def make_hann_init(n):
    """Hann window initialization."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

def make_sin_init(n):
    """Sine initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

def make_flat_init(n, edge_width_frac=0.1):
    """Flat middle with tapered edges."""
    a = np.ones(n)
    w = max(1, int(n * edge_width_frac))
    for i in range(w):
        t = (i + 1) / w
        a[i] = 0.5 * (1 - np.cos(np.pi * t))
        a[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * t))
    a /= a.sum()
    return a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

"""
Minimize 2n*max(self-conv)/sum^2 for non-negative sequences.
Strategy: Multi-length exploration with Adam (LSE-smoothed gradients),
peak-targeted coordinate descent, and multi-beta continuation.
Key improvements: longer sequences, tangent-space gradient projection,
and focused time allocation on the most promising candidates.

import numpy as np
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_a(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    b = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    B = np.fft.rfft(b, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_bp = np.fft.irfft(B * P, M2_fft)
    dL_da = 2.0 * conv_bp[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    # Project onto tangent space of simplex (remove scaling component)
    grad = grad - np.mean(grad)
    return score, grad


def subgrad_score_a(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())

    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    grad = grad - np.mean(grad)
    return score, grad
