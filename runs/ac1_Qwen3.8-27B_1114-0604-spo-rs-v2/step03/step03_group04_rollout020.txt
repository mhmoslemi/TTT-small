Optimized peak-convolution minimizer using LSE-smoothed L-BFGS with aggressive

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # Numerically stable LSE
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta  # This is the smoothed max
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a
    # dL/da[j] = sum_k p(k) * dc(k)/da[j]
    # dc(k)/da[j] = a[k-j] if 0 <= k-j < n
    # So dL/da[j] = sum_{k=j}^{j+n-1} p(k) * a[k-j]
    # This is cross-correlation of p and a, evaluated at... let me think.
    # dL/da[j] = sum_{l=0}^{n-1} p(j+l) * a[l] (where p is zero outside [0,2n-2])
    
    p = exp_scaled / exp_scaled.sum()
    
    # Compute dL/da using FFT
    # dL/da[j] = sum_{l=0}^{n-1} p[j+l] * a[l]
    # This is the cross-correlation (p, a) at lag j
    # = ifft(fft(p, L) * conj(fft(a, L))) but we need the right part
    # Actually: cross_corr(p, a)[j] = sum_l p[j+l] * a[l]
    # In terms of convolution: cross_corr(p,a) = conv(p, flip(a)) shifted
    # Let's use: conv(p, a_rev)[j + n - 1] where a_rev = a[::-1]
    # Hmm, let me just use the direct FFT approach.
    
    # dL/da[j] = sum_{l=0}^{n-1} p[j+l] * a[l]
    # Let q = p (length 2n-1), and we want sum_l q[j+l] * a[l] for l=0..n-1
    # This is the "valid" cross-correlation of q and a, but q is longer than a.
    # Using FFT: rfft approach
    # Actually, let's define: result[j] = sum_{l=0}^{n-1} q[j+l] * a[l]
    # This is equivalent to: result = convolve(q, a, mode='valid') but with q as the first arg
    # convolve(q, a, 'valid')[j] = sum_{l=0}^{n-1} q[j+l] * a[n-1-l]
    # That's not quite right either.
    
    # Let me use a different approach:
    # result[j] = sum_{l=0}^{n-1} q[j+l] * a[l]
    # = <q[j:j+n], a> (dot product)
    # This is the cross-correlation of q and a at lag j (with q being the "long" signal)
    
    # Using FFT: 
    # cross_corr(q, a)[j] = sum_l q[j+l] * conj(a[l])
    # For real a: = sum_l q[j+l] * a[l]  ✓
    
    # cross_corr via FFT: ifft(FFT(q) * conj(FFT(a)))
    # But we need zero-padding to avoid circular effects.
    # Length of q is 2n-1, length of a is n.
    # For linear cross-correlation, we need FFT size >= (2n-1) + n - 1 = 3n-2
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    Q = np.fft.rfft(q, M2_fft) if False else np.fft.rfft(p, M2_fft)
    A_fft = np.fft.rfft(a, M2_fft)
    conv_res = np.fft.irfft(Q * np.conj(A_fft), M2_fft)
    # The cross-correlation is in conv_res[n-1 : n-1 + (2n-1)]
    # Wait, I need to be more careful.
    
    # ifft(FFT(p, L) * conj(FFT(a, L))) gives the circular cross-correlation
    # For linear cross-correlation with zero-padding to L >= len(p)+len(a)-1:
    # result = ifft(FFT(p,L) * conj(FFT(a,L)))
    # The linear cross-correlation is in result[0:len(p)-1] (or a specific segment)
    
    # Actually, the standard result is:
    # ifft(FFT(x, L) * conj(FFT(y, L))) = circular cross-correlation of x and y
    # With L >= len(x) + len(y) - 1, this equals the linear cross-correlation
    # The linear cross-correlation has length len(x) + len(y) - 1
    # cross_corr[m] = sum_n x[n+m] * conj(y[n]) for valid m
    # For x = p (length 2n-1), y = a (length n):
    # cross_corr[m] = sum_n p[n+m] * a[n] for n where both are valid
    # We want result[j] = sum_{l=0}^{n-1} p[j+l] * a[l] = cross_corr[j]
    
    # With zero-padding to M2_fft:
    # The output has M2_fft elements, representing cross-correlation at lags 0, 1, ..., M2_fft-1
    # The "useful" lags are 0 to (2n-1)+n-1-(n) = 2n-2... 
    # Actually for cross-correlation of signals of lengths Nx and Ny:
    # The output has length Nx + Ny - 1 (for linear)
    # With FFT of size L >= Nx+Ny-1, the linear cross-correlation appears at:
    # positions 0 to (Nx-1) in the first part, and positions L-(Ny-1) to L-1 in the second part
    # Wait, I'm overcomplicating this.
    
    # Let me just use a simpler approach. The gradient dL/da can be computed as:
    # dL/da = convolve(p_reversed_segment, a) or similar
    
    # Actually, the cleanest way:
    # dL/da[j] = sum_{k} p[k] * (dc[k]/da[j]) = sum_k p[k] * a[k-j] (where 0<=k-j<n)
    # = sum_{k=j}^{j+n-1} p[k] * a[k-j]
    # Let m = k-j: = sum_{m=0}^{n-1} p[j+m] * a[m]
    
    # This is: (p cross-correlated with a) at lag j, where p is the "template"
    
    # Using the property: cross_corr(p, a) = conv(p, reverse(a))
    # conv(p, reverse(a))[j + n - 1] = sum_{m} p[m] * reverse(a)[j+n-1-m]
    # = sum_m p[m] * a[n-1-(j+n-1-m)] = sum_m p[m] * a[m-j]
    # Hmm, that gives sum_m p[m] * a[m-j], which is cross_corr at lag -j of a and p...
    
    # I think the safest approach is to just use numpy's correlate or do a direct FFT.
    
    # Let me use: dL/da[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # Define r[m] = a[m] for m=0..n-1 (padded with zeros to length 2n-1 or M2_fft)
    # Define s[k] = p[k] for k=0..2n-2
    # Then dL/da[j] = sum_m s[j+m] * r[m] = (s cross-correlated with r)[j]
    # = ifft(FFT(s, L) * conj(FFT(r, L)))[j]  for the right range of j
    
    # With L = M2_fft:
    Q = np.fft.rfft(p, M2_fft)
    A_pad = np.fft.rfft(a, M2_fft)
    xc = np.fft.irfft(Q * np.conj(A_pad), M2_fft)
    # xc[j] = sum_m p[j+m] * a[m] (for the valid range)
    # The valid range for j is 0 to (2n-1)-1 = 2n-2, but we only need 0 to n-1
    # Wait, actually with zero-padding, xc[j] for j=0..M2_fft-1 represents:
    # xc[j] = sum_m p[(j+m) mod M2_fft] * a[m]  (circular, but linear since padded)
    # For j = 0 to 2n-2 (the length of p minus 1), this equals sum_m p[j+m] * a[m]
    # (since j+m < 2n-1 <= M2_fft for the range we care about)
    
    dL_da = xc[:n]  # We only need j = 0 to n-1
    
    # Now the full objective: score = 2n * L / S^2
    # d(score)/da[j] = 2n * (dL/da[j] / S^2 - 2*L/(S^3) * 1)
    # (since d(S)/da[j] = 1)
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_objective_and_grad(x, beta)
    
    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def subgradient_polish(a, total_iters=500, cooling=0.999):
    """Subgradient descent on the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = 0.1
    step_scale = alpha
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv(a,a)):
        # dM/da[j] = a[k_star - j] if 0 <= k_star - j < n
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        
        # Full gradient of the objective
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        a -= step_scale * grad / gn
        a = np.clip(a, 0.0, 1000.0)
        step_scale *= cooling
    
    return best_sc, best_a


def spectral_flatten(a0, N, n_iters=300, lam=0.6):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def chirp_init(n, rng):
    """Generate chirp-based initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    freq = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits


def build_diverse_inits(n, rng, parent=None):
    """Build a diverse set of initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    
    inits.append(np.ones(n) * 0.5)
    x = np.arange(n, dtype=np.float64) / n
    
    for freq in [1, 2, 3, 5, 8]:
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x))
    
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.4 * (1.0 - (2.0 * x - 1.0) ** 2))
    
    inits.extend(chirp_init(n, rng))
    
    for scale in [0.1, 0.3, 0.5, 0.8]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, 1000.0))
    
    # Spectral flattening from a few bases
    N_fft = 1 << (2 * n - 1).bit_length()
    sf1 = spectral_flatten(np.ones(n) * 0.5, N_fft, 150, 0.6)
    inits.append(sf1)
    sf2 = spectral_flatten(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
                          N_fft, 150, 0.6)
    inits.append(sf2)
    
    chirp = np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01
    sf3 = spectral_flatten(chirp, N_fft, 200, 0.5)
    inits.append(sf3)
    
    return inits


def targeted_perturb(a, rng, strength=0.05):
    """Perturb by reducing top contributors to the peak and redistributing."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count * 2]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    Uses LSE-smoothed L-BFGS with aggressive beta continuation,
    subgradient polishing, and targeted local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 20.0
    deadline = t_start + effective_budget
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n in [8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(parent.size) + ', score=' + str(compute_score(parent)) + ')' if parent is not None else 'no'}")
    
    # ================================================================
    # Phase 1: Generate initializations and quick evaluation (20% time)
    # ================================================================
    phase1_deadline = t_start + 0.20 * effective_budget
    print(f"\n--- Phase 1: Initialization (deadline t={phase1_deadline - t_start:.0f}s) ---")
    
    candidates = {}  # key: (score, a)
    
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        n_parent = parent if (parent is not None and parent.size == n) else None
        inits = build_diverse_inits(n, rng, n_parent)
        
        for i, a_init in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_init = np.clip(a_init, 0.0, 1000.0)
            if a_init.sum() < 1e-6:
                continue
            sc = compute_score(a_init)
            candidates[f"n{n}_i{i}"] = (sc, a_init)
            report(sc, a_init, f"init n={n} i={i}")
    
    # Sort candidates by score, keep top ones
    sorted_cands = sorted(candidates.items(), key=lambda x: x[1][0])
    print(f"\nTop 5 initializations:")
    for key, (sc, a) in sorted_cands[:5]:
        print(f"  {key}: {sc:.6f} (n={len(a)})")
    
    # ================================================================
    # Phase 2: L-BFGS with increasing beta (55% time)
    # ================================================================
    phase2_deadline = t_start + 0.75 * effective_budget
    print(f"\n--- Phase 2: L-BFGS Optimization (deadline t={phase2_deadline - t_start:.0f}s) ---")
    
    # Take top candidates and do deep optimization
    top_cands = sorted_cands[:min(8, len(sorted_cands))]
    
    beta_schedule = [50, 100, 300, 800, 2000, 5000, 15000, 50000]
    
    for ci, (key, (sc_init, a_init)) in enumerate(top_cands):
        if time.monotonic() > phase2_deadline:
            break
        n = len(a_init)
        t0 = time.monotonic()
        a_cur = a_init.copy()
        
        for bi, beta in enumerate(beta_schedule):
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            
            # Adapt maxiter based on remaining time and dimension
            # Estimate time per function eval: roughly O(n log n) for FFT
            est_tpe = 0.003 * (n / 1024.0)  # seconds per function eval (rough)
            maxfun = max(100, int(remaining * 0.7 / max(est_tpe, 0.001)))
            maxfun = min(maxfun, 50000)
            maxiter = max(50, maxfun // 3)
            
            t_beta_start = time.monotonic()
            sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
            a_cur = a_opt
            elapsed_beta = time.monotonic() - t_beta_start
            report(sc, a_opt, f"lbfgs {key} beta={beta}")
            
            # Subgradient polish between L-BFGS calls
            if bi < len(beta_schedule) - 1 and elapsed_beta < remaining * 0.5:
                sc_sg, a_sg = subgradient_polish(a_cur, total_iters=100)
                report(sc_sg, a_sg, f"subgrad {key} beta={beta}")
                if sc_sg < sc:
                    a_cur = a_sg
        
        sc_final = compute_score(a_cur)
        report(sc_final, a_cur, f"final {key}")
        elapsed = time.monotonic() - t0
        print(f"    {key}: final score={sc_final:.6f} ({elapsed:.1f}s)")
    
    print(f"\nAfter Phase 2: best = {best_score:.8f}")
    
    # ================================================================
    # Phase 3: Very high beta refinement on the best (10% time)
    # ================================================================
    phase3_deadline = t_start + 0.88 * effective_budget
    print(f"\n--- Phase 3: High-Beta Refinement ---")
    
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 10:
            break
        n = len(best_a)
        est_tpe = 0.003 * (n / 1024.0)
        maxfun = max(200, int(remaining * 0.8 / max(est_tpe, 0.001)))
        maxfun = min(maxfun, 30000)
        maxiter = max(100, maxfun // 3)
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"refine beta={beta}")
    
    # Subgradient polish
    sc, a_polished = subgradient_polish(best_a, total_iters=1000, cooling=0.9995)
    report(sc, a_polished, "subgrad phase3")
    
    # ================================================================
    # Phase 4: Targeted local search (10% time)
    # ================================================================
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Targeted Local Search ---")
    
    n_best = len(best_a)
    n_trials = 30
    
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 10:
            break
        
        strategy = trial % 6
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.05 * rng.random())
        elif strategy == 2:
            # Frequency-domain perturbation
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            sigma = 0.005 + 0.02 * rng.random()
            noise = sigma * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]  # Preserve DC
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Chirp mixing
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            pert = amp * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            # Swap-based: swap a few elements with neighbors
            a_pert = best_a.copy()
            for _ in range(5):
                i = rng.integers(1, n_best - 1)
                j = i + 1
                a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
        else:
            # Partial re-optimization: zero out a random segment and re-fill
            a_pert = best_a.copy()
            seg_len = max(5, n_best // 50)
            start = rng.integers(0, n_best - seg_len)
            saved_sum = a_pert[start:start+seg_len].sum()
            a_pert[start:start+seg_len] = 0.0
            # Re-fill from spectral flattening
            N_fft = 1 << (2 * n_best - 1).bit_length()
            base = np.zeros(n_best)

"""
Peak-convolution minimizer using LSE-smoothed L-BFGS with aggressive beta
continuation, subgradient polishing, and targeted local search.
Exploits spectral flattening for diverse initializations and deep local
optimization from the parent seed.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute the exact objective value."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """
    LSE-smoothed objective and gradient.
    score = 2n/(S^2) * LSE_max(beta, c) where c = conv(a,a), S = sum(a).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta

    score = 2.0 * n * L / (S * S)

    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is cross-correlation of p (len 2n-1) and a (len n) at lag j
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    Q = np.fft.rfft(p, M2_fft)
    A_fft = np.fft.rfft(a, M2_fft)
    xc = np.fft.irfft(Q * np.conj(A_fft), M2_fft)
    dL_da = xc[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_objective_and_grad(x, beta)

    try:
        res = minimize(
            obj_and_grad, a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
    except Exception:
        return compute_score(a0), a0, None
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def subgradient_polish(a, total_iters=500, cooling=0.999):
    """Subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.1

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        a -= step_scale * grad / gn
        a = np.clip(a, 0.0, 1000.0)
        step_scale *= cooling

    return best_sc, best_a


def spectral_flatten(a0, N, n_iters=300, lam=0.6):
    """Iterative spectral flattening."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if N % 2 == 0 and k == half:
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def chirp_init(n, rng):
    """Generate chirp-based initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    freq = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits


def build_diverse_inits(n, rng, parent=None):
    """Build a diverse set of initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    inits.append(np.ones(n) * 0.5)
    x = np.arange(n, dtype=np.float64) / n

    for freq in [1, 2, 3, 5, 8]:

    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.4 * (1.0 - (2.0 * x - 1.0) ** 2))

    inits.extend(chirp_init(n, rng))

    for scale in [0.1, 0.3, 0.5, 0.8]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, 1000.0))

    inits.append(sf2)

    chirp = np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01

def targeted_perturb(a, rng, strength=0.05):
    """Perturb by reducing top contributors to the peak and redistributing."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count * 2]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)

            break
        n = len(best_a)
        est_tpe = 0.003 * (n / 1024.0)
        maxfun = max(200, int(remaining * 0.7 / max(est_tpe, 0.001)))
        maxfun = min(maxfun, 30000)
        maxiter = max(100, maxfun // 3)
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"refine beta={beta}")

    sc, a_polished = subgradient_polish(best_a, total_iters=800, cooling=0.9995)
    report(sc, a_polished, "subgrad phase3")

    # Phase 4: Targeted local search
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Targeted Local Search ---")

    n_best = len(best_a)
    n_trials = 30

    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining = phase4_deadline - time.monotonic()
        if remaining < 10:
            break

        strategy = trial % 6
        if strategy == 0:
            scale = 0.01 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.05 * rng.random())
        elif strategy == 2:
            N_fft = 1 << (2 * n_best - 1).bit_length()
            A = np.fft.fft(best_a, N_fft)
            sigma = 0.005 + 0.02 * rng.random()
            noise = sigma * (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A)))
            A_noisy = A * (1.0 + noise)
            A_noisy[0] = A[0]
            a_pert = np.real(np.fft.ifft(A_noisy))[:n_best]
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            x = np.arange(n_best, dtype=np.float64)
            f = 1.0 + 5.0 * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            pert = amp * np.cos(np.pi * f * x * x / (2.0 * n_best))
            a_pert = best_a * (1.0 + pert)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 4:
            a_pert = best_a.copy()
            for _ in range(5):
                i = rng.integers(1, n_best - 1)
                a_pert[i], a_pert[i + 1] = a_pert[i + 1], a_pert[i]
        else:
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        n = n_best
        est_tpe = 0.003 * (n / 1024.0)
        maxfun = max(100, int(remaining * 0.25 / max(est_tpe, 0.001)))
        maxfun = min(maxfun, 10000)
        maxiter = max(50, maxfun // 3)

        sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"local #{trial} s={strategy}")

        sc_sg, a_sg = subgradient_polish(a_pert, total_iters=50)
        report(sc_sg, a_sg, f"local_sg #{trial} s={strategy}")

    # Phase 5: Final polish
    print(f"\n--- Phase 5: Final Polish ---")

    sc, a_polished = subgradient_polish(best_a, total_iters=2000, cooling=0.9998)
    report(sc, a_polished, "final_subgrad")

    if time.monotonic() < deadline - 30:
        for beta in [50000, 100000]:
            if time.monotonic() > deadline - 30:
                break
            sc, a_opt, _ = run_lbfgs(a_polished, beta, maxiter=5000, maxfun=30000)
            report(sc, a_opt, f"final_lbfgs beta={beta}")

    sc, a_final = subgradient_polish(best_a, total_iters=1000, cooling=0.9999)
    report(sc, a_final, "abs_final_subgrad")

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
