import numpy as np
from scipy.optimize import minimize, minimize_scalar

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2

    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize (returns (f, grad))."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def fast_1d_minimize(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper, a_i):
    """
    Fast 1D minimization using analytical critical points + local search.
    Much faster than scipy minimize_scalar for this problem structure.
    """
    best_t = a_i
    best_f = coord_1d_obj(a_i, base_inside, alpha, beta_coef, outside_max, S_rest, n)

    # Evaluate at boundaries
    f0 = coord_1d_obj(0.0, base_inside, alpha, beta_coef, outside_max, S_rest, n)
    if f0 < best_f:
        best_f = f0
        best_t = 0.0

    ft = coord_1d_obj(t_upper, base_inside, alpha, beta_coef, outside_max, S_rest, n)
    if ft < best_f:
        best_f = ft
        best_t = t_upper

    # Find critical points: d/dt [2n * q(t) / (S+t)^2] = 0
    # where q(t) = base[k] + t*alpha[k] + t^2*beta[k] for the active k
    # For a fixed active k: t* = (2*base[k] - alpha[k]*S_rest) / (2*beta[k]*S_rest - alpha[k])
    candidates = np.array([0.0, t_upper])

    # Active quadratics: those with beta > 0 or alpha != 0
    active = np.where((beta_coef != 0) | (alpha != 0))[0]
    if len(active) > 0:
        bk = base_inside[active]
        ak = alpha[active]
        ck = beta_coef[active]
        num = 2.0 * bk - ak * S_rest
        den = 2.0 * ck * S_rest - ak
        valid = np.abs(den) > 1e-14
        if valid.any():
            t_crit = num[valid] / den[valid]
            t_crit = t_crit[(t_crit > 1e-10) & (t_crit < t_upper * 0.999)]
            # Take the top 20 closest to current best
            if len(t_crit) > 30:
                idx = np.argsort(np.abs(t_crit - best_t))[:30]
                t_crit = t_crit[idx]
            if len(t_crit) > 0:
                candidates = np.concatenate([candidates, t_crit])

    # Add a few grid points around the current best
    span = max(0.05 * t_upper, 0.01 * (S_rest / max(1, n)))
    grid = best_t + np.linspace(-span, span, 8)
    grid = np.clip(grid, 0, t_upper)
    candidates = np.concatenate([candidates, grid])

    # Remove duplicates (within tolerance)
    candidates = np.unique(np.round(candidates, 10))
    candidates = np.clip(candidates, 0, t_upper)

    # Evaluate all candidates
    for t in candidates:
        t = float(t)
        f = coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        if f < best_f:
            best_f = f
            best_t = t

    # Local refinement with golden section around best
    lo = max(0.0, best_t - span * 0.5)
    hi = min(t_upper, best_t + span * 0.5)
    if hi - lo > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = coord_1d_obj(c1, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        f2 = coord_1d_obj(c2, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        for _ in range(20):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = coord_1d_obj(c1, base_inside, alpha, beta_coef, outside_max, S_rest, n)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = coord_1d_obj(c2, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        t_mid = (lo + hi) / 2.0
        f_mid = coord_1d_obj(t_mid, base_inside, alpha, beta_coef, outside_max, S_rest, n)
        if f_mid < best_f:
            best_f = f_mid
            best_t = t_mid

    return best_t


def coordinate_descent_sweep(a, rng=None, n_coords=None):
    """One full sweep of coordinate descent with incremental conv updates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        t_opt = fast_1d_minimize(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper, a_i)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def hot_coordinate_sweep(a, n_sweeps=1):
    """Coordinate descent prioritizing coordinates contributing to the current peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]

        for i in order:
            i = int(i)
            a_i = a[i]
            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            t_opt = fast_1d_minimize(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper, a_i)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def pressure_step(a, eta, quantile=0.85):
    """
    One step of pressure-based gradient update.
    Uses quantile-based targeting: only the top (1-quantile) fraction of
    convolution bins contribute to the gradient. This focuses optimization
    on the most problematic peaks.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    # Quantile-based weighting
    T = np.quantile(c, quantile)
    h = np.maximum(0.0, c - T)
    h_norm = float(h.sum())
    if h_norm < 1e-12:
        return a.copy()

    # Compute gradient of sum_k h[k] * c[k] w.r.t. a
    # Using FFT trick: grad[i] = 2 * ifft(FFT(rev_a) * FFT(h))[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    H = np.fft.rfft(h, M2_fft)
    conv_res = np.fft.irfft(RA * H, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res

    # Scale gradient
    g_max = float(np.max(np.abs(g)))
    if g_max < 1e-12:
        return a.copy()

    a_new = a - eta * g / g_max
    a_new = np.maximum(a_new, 0.0)
    s_new = float(a_new.sum())
    if s_new < 1e-12:
        return a.copy()
    a_new = a_new * S / s_new
    return a_new


def lse_gradient_step(a, beta, eta):
    """One step of gradient descent on the LSE-smoothed objective."""
    n = len(a)
    score, grad = lse_score_grad(a, beta)

    g_norm = float(np.linalg.norm(grad))
    if g_norm < 1e-14:
        return a.copy(), score

    a_new = a - eta * grad / g_norm
    a_new = np.maximum(a_new, 0.0, 1000.0)
    return a_new, score


def run_pressure_iterations(a, t_end, initial_eta=0.3, decay=0.995, quantile=0.85, check_interval=50):
    """
    Run pressure-based iterations until time limit.
    Adaptively adjusts step size based on improvement.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(a.sum())
    best_score = compute_score(a)
    best_a = a.copy()
    eta = initial_eta
    no_improve = 0

    while time.monotonic() < t_end:
        a_new = pressure_step(a, eta, quantile)
        # Check score every few steps
        if no_improve % check_interval == 0:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += check_interval
                eta *= decay
            a = a_new
        else:
            no_improve += 1
            a = a_new

        # Periodically try a slightly different quantile
        if no_improve > 200:
            quantile = 0.7 + 0.4 * np.random.rand()
            no_improve = 0
            eta = initial_eta * 0.5

    final_sc = compute_score(best_a)
    return final_sc, best_a


def spectral_perturb(a, rng):
    """
    Spectral-domain perturbation: modify the magnitude spectrum to reduce
    peakiness, then project back to non-negative.
    """
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    mag = np.abs(A)
    phase = np.angle(A)

    # Compute current convolution (circular approx)
    c_circ = np.fft.irfft(A * A, M_fft)[:M]
    c_max = float(np.max(c_circ))
    c_mean = float(np.mean(c_circ))
    if c_max < 1e-12:
        return a.copy()

    # Flatten the spectrum: reduce peaks in |A|^2 that contribute to time-domain peaks
    # Use the inverse transform of the "excess" to identify which frequencies to damp
    excess = np.maximum(0.0, c_circ - c_mean)
    E = np.fft.rfft(excess, M_fft)
    damp = np.exp(-0.3 * (np.abs(E) / (float(np.max(np.abs(E))) + 1e-12)))

    A_new = mag * damp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-12:
        return a.copy()
    a_new = a_new * float(a.sum()) / s
    return a_new


def block_perturb(a, rng, block_frac=0.05):
    """Perturb a random block of the sequence."""
    n = len(a)
    block_size = max(5, int(n * block_frac))
    start = int(rng.integers(0, n - block_size))
    a_new = a.copy()
    S_block = float(a_new[start:start + block_size].sum())
    a_new[start:start + block_size] = 0.0
    # Redistribute to other positions
    mask = np.ones(n, dtype=bool)
    mask[start:start + block_size] = False
    mask_idx = np.where(mask)[0]
    # Concentrate in a nearby region
    near_start = max(0, start - block_size)
    near_end = min(n, start + 2 * block_size)
    near_idx = [j for j in mask_idx if near_start <= j < near_end]
    if len(near_idx) > 0:
        a_new[near_idx] += S_block / len(near_idx)
    else:
        a_new[mask_idx] += S_block / len(mask_idx)
    return np.clip(a_new, 0.0, 1000.0)


def gaussian_perturb(a, rng, scale=0.03):
    """Gaussian perturbation preserving non-negativity."""
    n = len(a)
    a_new = a * (1.0 + scale * rng.standard_normal(n))
    return np.clip(a_new, 0.0, 1000.0)


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def sinusoidal_perturb(a, rng):
    """Apply a sinusoidal modulation to the sequence."""
    n = len(a)
    freq = 2.0 + 20.0 * rng.random()
    amp = 0.01 + 0.05 * rng.random()
    x = np.arange(n, dtype=np.float64) / n
    a_new = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses a hybrid approach: pressure updates, coordinate descent, LSE-optimization,
    and diverse perturbation strategies.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        best_a = parent.copy()
        best_score = compute_score(best_a)
        print(f"Parent score: {best_score:.8f}, length: {len(best_a)}")
    else:
        n_init = 2048
        best_a = np.ones(n_init) * 0.5
        best_score = compute_score(best_a)
        print(f"No parent. Random init score: {best_score:.8f}, length: {n_init}")

    n = len(best_a)
    print(f"Working length: {n}")

    # ============================================================
    # Phase 1: Quick coordinate descent (15% time)
    # Establish a good local minimum quickly
    # ============================================================
    phase1_deadline = t_start + 0.15 * total_budget
    print(f"\n=== Phase 1: Quick CD ({phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    sweep_count = 0
    while time.monotonic() < phase1_deadline:
        sweep_count += 1
        if sweep_count % 3 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            if sweep_count <= 5 or sweep_count % 10 == 0:
                print(f"  P1 sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        if sweep_count % 10 == 0:
            print(f"  P1 sweep {sweep_count}: {sc:.8f}")

    print(f"Phase 1 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 2: Pressure-based iterations (30% time)
    # Fast gradient steps with adaptive step size
    # ============================================================
    phase2_deadline = t_start + 0.45 * total_budget
    print(f"\n=== Phase 2: Pressure Updates ({phase2_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    sub_deadline = t_start + (phase2_deadline - t_start) * 0.4
    sc, a_cur = run_pressure_iterations(a_cur, sub_deadline, initial_eta=0.5, decay=0.99, quantile=0.85)
    if sc < best_score:
        best_score = sc
        best_a = a_cur.copy()
        print(f"  P2 pressure (q=0.85): {sc:.8f} (NEW BEST)")
    else:
        print(f"  P2 pressure (q=0.85): {sc:.8f}")

    # Try different quantiles
    for q in [0.75, 0.9, 0.95]:
        if time.monotonic() > phase2_deadline - 30:
            break
        a_cur = best_a.copy()
        sc, a_cur = run_pressure_iterations(a_cur, min(time.monotonic() + 120, phase2_deadline - 10),
                                            initial_eta=0.3, decay=0.995, quantile=q)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  P2 pressure (q={q}): {sc:.8f} (NEW BEST)")
        else:
            print(f"  P2 pressure (q={q}): {sc:.8f}")

    # CD polish after pressure
    a_cur = best_a.copy()
    for _ in range(10):
        if time.monotonic() > phase2_deadline - 5:
            break
        sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: LSE-based L-BFGS (25% time)
    # Varying beta for multi-scale optimization
    # ============================================================
    phase3_deadline = t_start + 0.70 * total_budget
    print(f"\n=== Phase 3: LSE L-BFGS ({phase3_deadline - t_start:.0f}s) ===")

    for beta in [50, 150, 400, 1000, 3000, 8000]:
        if time.monotonic() > phase3_deadline - 30:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

        # Quick CD after each LSE
        if time.monotonic() < phase3_deadline - 20:
            a_cur = best_a.copy()
            sc2, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc2 < best_score:

                best_a = a_cur.copy()
                print(f"    CD after LSE beta={beta}: {sc2:.8f} (NEW BEST)")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Diverse perturbations + rapid CD (25% time)
    # Try many different perturbations and quickly optimize each
    # ============================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Perturbation + CD Cycles ===")

    trial = 0
    strategies = [
        ("gauss", lambda: gaussian_perturb(best_a, rng, 0.02 + 0.04 * rng.random())),
        ("targeted", lambda: targeted_perturb(best_a, rng, 0.02 + 0.04 * rng.random())),
        ("spectral", lambda: spectral_perturb(best_a, rng)),
        ("block", lambda: block_perturb(best_a, rng, 0.03 + 0.05 * rng.random())),
        ("sine", lambda: sinusoidal_perturb(best_a, rng)),
    ]

    while time.monotonic() < phase4_deadline - 20:
        strategy_name, strategy_fn = strategies[trial % len(strategies)]
        a_pert = strategy_fn()
        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Quick optimization: 3-5 CD sweeps + 1 pressure phase
        for sw in range(4):
            if time.monotonic() > phase4_deadline - 15:
                break
            if sw % 2 == 0:
                sc_pert, a_pert = hot_coordinate_sweep(a_pert, n_sweeps=1)
            else:
                sc_pert, a_pert = coordinate_descent_sweep(a_pert, rng=rng)

        # Brief pressure phase
        sub_dl = min(time.monotonic() + 60, phase4_deadline - 10)
        if time.monotonic() < sub_dl:
            sc_pert, a_pert = run_pressure_iterations(a_pert, sub_dl,
                                                      initial_eta=0.3, decay=0.99, quantile=0.85)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} ({strategy_name}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 5 == 0:
            print(f"  P4 trial {trial} ({strategy_name}): {sc_pert:.8f}")
        trial += 1

    # ============================================================
    # Phase 5: Final polish (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    a_cur = best_a.copy()
    for _ in range(30):
        if time.monotonic() > deadline - 5:
            break
        if _ % 3 == 0:
            sc, a_cur = hot_coordinate_sweep(a_cur, n_sweeps=1)
        else:
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # Final LSE if time permits
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 5000, maxiter=3000, maxfun=50000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
        # Final CD
        if time.monotonic() < deadline - 5:
            a_cur = best_a.copy()
            sc, a_cur = coordinate_descent_sweep(a_cur, rng=rng)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"  Final CD: {sc:.8f} (NEW BEST)")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
