def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_smoothed_max(c, beta):
    """LSE approximation of max."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    log_sum = np.log(np.exp(scaled).sum())
    return c_max + log_sum / beta

def nesterov_lse_descent(a0, beta_schedule, total_iters, budget_fn):
    """
    Nesterov momentum gradient descent on LSE-smoothed objective.
    Adaptive step size with backtracking.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_sc = compute_score(a)
    
    gamma = 0.9  # momentum
    step = 0.01  # initial step size
    
    for it in range(total_iters):
        if budget_fn():
            break
        
        score, grad = lse_score_and_grad(a, beta_schedule)
        if score < best_sc:
            best_sc = score
            best_a = a.copy()
        
        # Nesterov lookahead
        a_look = a - gamma * step * grad
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look > 1e-12:
            a_look = a_look / S_look
        
        score_look, grad_look = lse_score_and_grad(a_look, beta_schedule)
        
        # Update momentum
        v = gamma * v + grad_look
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        # Adaptive step
        step = 0.5 / (v_norm * np.sqrt(it + 1))
        step = min(step, 0.1)
        
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        a = a_new
        
        # Track best (using exact score periodically)
        if it % 50 == 0:
            exact_sc = compute_score(a)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = a.copy()
    
    # Final check
    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc
        best_a = a.copy()
    
    return best_a, best_sc

def peak_targeted_cd(a, n_sweeps=5, n_top_peaks=10, n_top_coords=100, budget_fn=None):

    and optimize them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        if budget_fn and budget_fn():
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Find top peaks
        peak_threshold = c_max * 0.95
        peak_indices = np.where(c >= peak_threshold)[0]
        
        # Also include the global max
        peak_indices = list(set(list(peak_indices) + [int(np.argmax(c))]))
        peak_indices.sort()
        
        for k_star in peak_indices:
            # Coordinates contributing to c[k_star]: j where 0<=j<n and 0<=k_star-j<n
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()  # contribution of a[j] to c[k_star] is a[k_star-j]
            top_count = min(n_top_coords, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            
            for li in top_local:
                j = j_start + li
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                
                # Try reducing this coordinate
                best_val = old_val
                best_local_sc = best_sc  # use current best as reference
                
                # Actually, let's compute current score
                cur_sc = compute_score(a)
                best_local_sc = cur_sc
                best_val = old_val
                
                for factor in [0.99, 0.98, 0.95, 0.9, 0.8, 0.7]:
                    a[j] = old_val * factor
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = old_val * factor
                    if factor < 0.85:
                        break  # diminishing returns
                
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

        # We want amp^2 to be flatter, i.e., amp to be flatter
        # Use: amp_new = amp^alpha (which flattens the amplitude spectrum)
        amp_new = amp ** (alpha ** 0.5)  # since amp^2 ~ conv, we want amp^2 to be flatter
        # Actually: conv ~ |A|^2 (approximately for large M)
        # We want |A_new|^2 = (|A|^2)^alpha = |A|^(2*alpha)
        # So |A_new| = |A|^alpha
        amp_new = amp ** alpha
        
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Mix for stability
        a = 0.5 * a + 0.5 * a_new
        S = a.sum()
        if S > 1e-12:
            a = a / S
    
    return best_a

def subgradient_polish_v2(a, total_iters=3000, budget_fn=None):
    """
    Improved subgradient descent with averaging.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    avg_a = a.copy()
    sc_lb = 1.0
    
    for it in range(total_iters):
        if budget_fn and budget_fn():
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max wrt a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Also include near-max positions (within 5% of max)
        threshold = M * 0.95
        near_max = np.where(c >= threshold)[0]
        for k in near_max:
            if k == k_star:
                continue
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                seg = a[js:je + 1]
                weight = (c[k] / M) ** 2  # weight by how close to max
                sg_M[js:je + 1] += weight * 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Running average
        avg_a = avg_a + (a - avg_a) / (it + 1)
        
        # Check averaged solution periodically
        if it % 100 == 0 and it > 100:
            avg_sc = compute_score(avg_a)
            if avg_sc < best_sc:
                best_sc = avg_sc
                best_a = avg_a.copy()
    
    # Final check on average
    avg_sc = compute_score(avg_a)
    if avg_sc < best_sc:
        best_sc = avg_sc
        best_a = avg_a.copy()
    
    return best_a

"""
Multi-phase optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2.
Strategy: Adaptive LSE L-BFGS with momentum augmentation, multi-peak subgradient
with running average, spectral flattening, and peak-targeted coordinate descent.
Key innovations: (1) adaptive beta escalation based on convergence, (2) near-peak
aware subgradient with Nesterov-type averaging, (3) spectral amplitude reshaping.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for the objective 2n*max(conv)/sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta

    # dL/da_j = 2 * sum_l a_l * p_{j+l} (cross-correlation)
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(A_rev * P, M2_fft)[:M2]
    dL_da = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def normalize_a(a):

    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    avg_a = a.copy()
    sc_lb = 1.0 + 1e-6

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Multi-peak subgradient
        threshold = M * 0.98
        peak_indices = np.where(c >= threshold)[0]
        sg = np.zeros(n)
        for k in peak_indices:
            weight = (c[k] / M) ** 4  # emphasize peaks more
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += weight * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step with cap
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        if use_average:
            avg_a = avg_a + (a - avg_a) / (it + 1)
            if it % 200 == 0 and it > 100:
                avg_sc = compute_score(avg_a)
                if avg_sc < best_sc:
                    best_sc = avg_sc
                    best_a = avg_a.copy()

    if use_average:
        avg_sc = compute_score(avg_a)
        if avg_sc < best_sc:
            best_sc = avg_sc
            best_a = avg_a.copy()
    return best_a, best_sc


def spectral_flattening(a, n_iters=100, alpha=0.85, deadline=None):
    """
    Reshape the amplitude spectrum to flatten the convolution.
    amp_new = amp^alpha reduces spectral peaks.
    """
    n = len(a)
    a = normalize_a(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        amp = np.abs(A)
        phase = np.angle(A)
        # Flatten amplitude
        amp_new = amp ** alpha
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        # Blend
        a = 0.4 * a + 0.6 * a_new
        a = normalize_a(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def peak_coordinate_descent(a, n_sweeps=5, n_top_coords=150, deadline=None):
    """Coordinate descent on peak-contributing coordinates."""
    n = len(a)
    a = normalize_a(a.astype(np.float64))
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        c_max = float(c.max())
        # Top peaks
        thresh = c_max * 0.97
        peak_inds = np.where(c >= thresh)[0]

        for k_star in peak_inds[:8]:  # limit peaks
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = min(n_top_coords, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]

            for li in top_local:
                j = j_start + li
                old_val = a[j]
                if old_val < 1e-10:
                    continue
                best_val = old_val
                best_local = compute_score(a)
                for factor in [0.97, 0.94, 0.90, 0.85]:
                    a[j] = old_val * factor
                    sc = compute_score(a)
                    if sc < best_local:
                        best_local = sc
                        best_val = old_val * factor
                    else:
                        break
                a[j] = best_val
                a = normalize_a(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def build_inits(n, rng, parent):
    """Build diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent))
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.5 * (2.0 * x - 1.0) ** 2)
    inits.append(0.4 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.15 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    # Block structure
    for nb in (8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return [np.clip(a, 0.001, 1000.0) for a in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2n * max(conv(a,a)) / (sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def do_report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        pl = len(parent)
        lengths.append(pl)
        for delta in [-256, -128, 128, 256, 512]:
            alt = pl + delta
            if alt >= 1024 and alt not in lengths:
                lengths.append(alt)
    for n in [4096, 6144, 8192, 10240]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths: {lengths}")
    print(f"Parent: {parent is not None} (len={0 if parent is None else len(parent)})")

    # ================================================================
    # Phase 1: LSE L-BFGS with adaptive beta (45% time)
    # ================================================================
    p1_end = t_start + 0.45 * float(budget_s)
    print(f"\n=== Phase 1: LSE L-BFGS (deadline {p1_end - t_start:.0f}s) ===")

    beta_list = [30, 60, 120, 250, 500, 1000]
    n_focus = lengths[:3]

    for n in n_focus:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        # Limit inits for large n
        remaining = p1_end - time.monotonic()
        max_inits = max(2, int(remaining / 20.0))
        inits = inits[:max_inits]

        for ii, a0 in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            a_cur = normalize_a(a0)
            for beta in beta_list:
                if time.monotonic() > p1_end - 5:
                    break
                sc, a_opt = lbfgs_lse(a_cur, beta, maxiter=3000, maxfun=60000, deadline=p1_end)
                do_report(sc, a_opt, f"P1 n={n} i={ii} b={beta}")
                a_cur = best_a if best_a is not None and len(best_a) == n else a_opt
            # Nesterov from current best
            if time.monotonic() < p1_end - 30:
                a_nest, sc_nest = nesterov_lse_descent(a_cur, beta_list[-1], 800, p1_end)
                do_report(sc_nest, a_nest, f"P1 n={n} i={ii} nesterov")

    print(f"  After P1: {best_score:.8f}")

    # ================================================================
    # Phase 2: Spectral flattening + high-beta L-BFGS (20% time)
    # ================================================================
    p2_end = t_start + 0.65 * float(budget_s)
    print(f"\n=== Phase 2: Spectral + High-beta L-BFGS ===")

    if best_a is not None:
        for alpha in [0.8, 0.9, 0.95]:
            if time.monotonic() > p2_end:
                break
            a_flat, sc_flat = spectral_flattening(best_a, n_iters=80, alpha=alpha, deadline=p2_end)
            do_report(sc_flat, a_flat, f"P2 alpha={alpha}")

        # High beta L-BFGS
        for beta in [1000, 2000, 3000]:
            if time.monotonic() > p2_end - 10:
                break
            sc, a_opt = lbfgs_lse(best_a, beta, maxiter=4000, maxfun=80000, deadline=p2_end)
            do_report(sc, a_opt, f"P2 LBFGS b={beta}")

        # Try length variations
        n_cur = len(best_a)
        for n_try in [n_cur + 256, n_cur - 256]:
            if n_try < 512 or n_try > 25000:
                continue
            if time.monotonic() > p2_end - 10:
                break
            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_try)
            a_try = np.interp(x_new, x_old, best_a)
            sc, a_opt = lbfgs_lse(a_try, 200, maxiter=3000, maxfun=50000, deadline=p2_end)
            do_report(sc, a_opt, f"P2 n={n_try}")

    print(f"  After P2: {best_score:.8f}")

    # ================================================================
    # Phase 3: Multi-peak subgradient with averaging (20% time)
    # ================================================================
    p3_end = t_start + 0.85 * float(budget_s)
    print(f"\n=== Phase 3: Multi-peak Subgradient ===")

    if best_a is not None:
        a_sub, sc_sub = multi_peak_subgradient(best_a, total_iters=5000, deadline=p3_end, use_average=True)
        do_report(sc_sub, a_sub, "P3 subgrad")

        # Nesterov from subgradient result
        if time.monotonic() < p3_end - 30:
            a_ne, sc_ne = nesterov_lse_descent(a_sub, 1000, 500, p3_end)
            do_report(sc_ne, a_ne, "P3 nesterov")

        # L-BFGS from subgradient
        if time.monotonic() < p3_end - 20:
            sc, a_opt = lbfgs_lse(best_a, 1500, maxiter=3000, maxfun=60000, deadline=p3_end)
            do_report(sc, a_opt, "P3 LBFGS b=1500")

    print(f"  After P3: {best_score:.8f}")

    # ================================================================
    # Phase 4: Peak CD + final polish (15% time)
    # ================================================================
    print(f"\n=== Phase 4: Peak CD + Final Polish ===")

    if best_a is not None:
        a_cd, sc_cd = peak_coordinate_descent(best_a, n_sweeps=5, n_top_coords=200, deadline=deadline)
        do_report(sc_cd, a_cd, "P4 CD")

        # L-BFGS from CD
        if time.monotonic() < deadline - 30:
            sc, a_opt = lbfgs_lse(a_cd, 2000, maxiter=4000, maxfun=60000, deadline=deadline)
            do_report(sc, a_opt, "P4 LBFGS from CD")

        # Final subgradient
        if time.monotonic() < deadline - 20:
            a_sub2, sc_sub2 = multi_peak_subgradient(best_a, total_iters=3000, deadline=deadline, use_average=True)
            do_report(sc_sub2, a_sub2, "P4 subgrad2")

        # Final CD
        if time.monotonic() < deadline - 10:
            a_cd2, sc_cd2 = peak_coordinate_descent(best_a, n_sweeps=3, n_top_coords=300, deadline=deadline)
            do_report(sc_cd2, a_cd2, "P4 CD2")

    # ================================================================
    # Final output
    # ================================================================
    if best_a is None:
        best_a = np.ones(1024) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
