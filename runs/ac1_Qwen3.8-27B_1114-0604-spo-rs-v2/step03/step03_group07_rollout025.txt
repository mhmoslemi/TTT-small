```python
"""
Optimized step function search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS, subgradient polish, and novel spectral shaping.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        a_out = a0
        return compute_score(a_out), a_out, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def subgrad_polish(a, max_iters=5000, lb_override=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    if lb_override is None:
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
    else:
        sc_lb = lb_override

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_shape(a, alpha=0.3):
    """Flatten the power spectrum to reduce autocorrelation peak."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    power = np.abs(A) ** 2
    # Flatten: A_new = A * (1/sqrt(power))^alpha = A * power^(-alpha/2)
    # But be careful with zeros
    eps = 1e-10
    power_safe = np.maximum(power, eps)
    A_new = A * np.power(power_safe, -alpha / 2.0)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def build_initializations(n, rng, parent=None):
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
        # Scaled copies
        sum_p = parent.sum()
        if sum_p > 0:

    # U-shaped (high at edges, low in middle)
    inits.append(0.1 + 0.9 * (2 * xn - 1) ** 2)
    # V-shaped
    inits.append(0.1 + 0.9 * np.abs(2 * xn - 1))
    # Inverted V (low at edges, high in middle) - opposite
    inits.append(0.1 + 0.9 * (1 - (2 * xn - 1) ** 2))

    # Anti-palindromic ramp
    inits.append(0.05 + 0.95 * xn)
    inits.append(0.05 + 0.95 * (1 - xn))

    # Quadratic
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1 - xn) ** 2)

    # Sinusoidal (odd harmonics are anti-palindromic)
    inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * xn), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * np.sin(3 * np.pi * xn), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.5 * np.sin(np.pi * xn) + 0.1 * np.sin(5 * np.pi * xn), 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 1.0, np.ones(n - half) * 0.1]))
    inits.append(np.concatenate([np.ones(half) * 0.1, np.ones(n - half) * 1.0]))

    # Random smooth
    a_rand = 0.5 + 0.4 * rng.standard_normal(n)
    kernel_size = max(1, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))

    # Pure random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    return inits


def targeted_perturb(a, rng, strength=0.05):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_count = max(5, n // 200)
        sorted_idx = np.argsort(contrib)
        top_idx = sorted_idx[-top_count:]
        bottom_idx = sorted_idx[:-top_count]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)
```

```python
"""
Optimized step function search minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LSE-continuation L-BFGS, multi-peak subgradient, spectral shaping,
and greedy peak-reduction. Builds on parent solution (n=6962, score~1.5095).
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_grad(a, beta)

    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-16,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        a_out = a0
        return compute_score(a_out), a_out, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def subgrad_polish(a, max_iters=5000, lb_val=None):
    """Exact subgradient descent with Polyak step size and safe cap."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    if lb_val is None:
        lb_val = 2.0 * n / (2.0 * n - 1.0)
    lb_val = min(lb_val, best_sc)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - lb_val) / gn2
        step = min(step, 2.0 / (1.0 + 0.02 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def subgrad_multi_peak(a, max_iters=3000, top_k=3, lb_val=None):
    """Subgradient using top-k peaks for more stable descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    if lb_val is None:
        lb_val = 2.0 * n / (2.0 * n - 1.0)
    lb_val = min(lb_val, best_sc)

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top-k peak lags
        top_lags = np.argsort(c)[::-1][:top_k]
        # Weight peaks by their relative height
        peak_vals = c[top_lags]
        weights = peak_vals / peak_vals.sum()

        sg = np.zeros(n)
        for w, k_star in zip(weights, top_lags):
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - lb_val) / gn2
        step = min(step, 1.5 / (1.0 + 0.01 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_shape(a, alpha=0.2):
    """Flatten the power spectrum to reduce autocorrelation concentration."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    power = np.abs(A) ** 2
    eps = 1e-12
    power_safe = np.maximum(power, eps)
    A_new = A * np.power(power_safe, -alpha / 2.0)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def spectral_sharpen(a, alpha=0.1):
    """Sharpen the spectrum (opposite of flatten) - sometimes helps."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    power = np.abs(A) ** 2
    eps = 1e-12
    power_safe = np.maximum(power, eps)
    A_new = A * np.power(power_safe, alpha / 2.0)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def greedy_peak_reduce(a, top_k=3, frac=0.2):
    """Reduce elements contributing most to top-k autocorrelation peaks."""
    n = len(a)
    c = fast_conv(a)
    top_lags = np.argsort(c)[::-1][:top_k]
    a_new = a.copy()

    for k_star in top_lags:
        k_star = int(k_star)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a_new[j] * a_new[k_star - j]
        # Reduce top contributors
        threshold = np.sort(contrib)[::-1][max(0, n // 50)]
        reduce_mask = contrib > threshold
        total_red = 0.0
        for j in np.where(reduce_mask)[0]:
            amt = a_new[j] * frac
            a_new[j] -= amt
            a_new[j] = max(a_new[j], 0.0)
            total_red += amt
        # Redistribute to low contributors
        add_mask = contrib < threshold * 0.3
        add_indices = np.where(add_mask)[0]
        if len(add_indices) > 0 and total_red > 0:
            per = total_red / len(add_indices)
            a_new[add_indices] += per

    return np.clip(a_new, 0.0, 1000.0)


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / max(n - 1, 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    # U-shaped (high at edges)
    inits.append(0.1 + 0.9 * (2.0 * xn - 1.0) ** 2)
    inits.append(0.05 + 0.95 * np.abs(2.0 * xn - 1.0))

    # Anti-palindromic ramp
    inits.append(0.05 + 0.95 * xn)
    inits.append(0.05 + 0.95 * (1.0 - xn))

    # Quadratic
    inits.append(0.05 + 0.95 * xn ** 2)
    inits.append(0.05 + 0.95 * (1.0 - xn) ** 2)

    # Sinusoidal anti-palindromic
    inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * xn), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.3 * np.sin(3.0 * np.pi * xn), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * np.sin(np.pi * xn) + 0.2 * np.sin(5.0 * np.pi * xn), 0.01, 1000.0))

    # Two-block
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 1.0, np.ones(n - half) * 0.1]))
    inits.append(np.concatenate([np.ones(half) * 0.1, np.ones(n - half) * 1.0]))

    # Random smooth
    a_rand = 0.5 + 0.4 * rng.standard_normal(n)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    a_smooth = np.convolve(a_rand, kernel, mode='same')
    inits.append(np.clip(a_smooth, 0.01, 1000.0))

    # Pure random
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    return inits


def targeted_perturb(a, rng, strength=0.04):
    """Reduce elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a_pert[j] * a_pert[k_star - j]
        top_count = max(5, n // 300)
        sorted_idx = np.argsort(contrib)
        top_idx = sorted_idx[-top_count:]
        bottom_idx = sorted_idx[:-top_count]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer for minimizing the autocorrelation-based score.
    Phases: initial exploration -> LSE continuation -> subgradient -> spectral
    shaping -> local search -> final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    # Determine primary length
    if parent is not None:
        n_primary = parent.size
    else:
        n_primary = 8192
        parent = np.ones(n_primary) * 0.5
        best_a = parent.copy()
        best_score = compute_score(parent)

    print(f"Parent length: {n_primary}, initial score: {best_score:.8f}")

    # ============================================================
    # Phase 1: Quick LSE from parent with elevated beta
    # ============================================================
    phase1_end = t_start + 0.08 * budget_s
    print(f"\n=== Phase 1: Quick LSE refinement (budget {phase1_end - t_start:.0f}s) ===")

    a_cur = parent.copy()
    for beta in [500, 2000]:
        if time.monotonic() > phase1_end:
            break
        remaining = max(1.0, phase1_end - time.monotonic())
        maxit = max(500, int(remaining * 2))
        sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [P1 beta={beta}] {sc:.8f}")
        a_cur = a_opt

    # ============================================================
    # Phase 2: LSE Continuation (main optimization)
    # ============================================================
    phase2_end = t_start + 0.45 * budget_s
    print(f"\n=== Phase 2: LSE Continuation (budget {phase2_end - t_start:.0f}s) ===")

    n_cur = len(best_a)
    a_cur = best_a.copy()
    beta_schedule = [5000, 10000, 20000, 50000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = max(1.0, phase2_end - time.monotonic())
        # More iterations at higher beta
        maxit = max(2000, min(15000, int(remaining * 1.5)))
        print(f"  LSE beta={beta}, maxiter={maxit}, elapsed={time.monotonic()-t_start:.0f}s")
        t0 = time.monotonic()
        sc, a_opt, success = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [P2 beta={beta}] NEW BEST: {sc:.8f} (took {elapsed:.1f}s)")
        else:
            print(f"  [P2 beta={beta}] no improve: {sc:.8f} (took {elapsed:.1f}s)")
        a_cur = a_opt

    # ============================================================
    # Phase 3: Subgradient polish (exact, multi-peak)
    # ============================================================
    phase3_end = t_start + 0.60 * budget_s
    print(f"\n=== Phase 3: Subgradient Polish (budget {phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        # Single-peak subgradient
        n_iters = max(500, int(remaining * 30))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters, lb_val=best_score * 0.999)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [P3 subgrad] NEW BEST: {sc:.8f}")

        # Multi-peak subgradient
        if time.monotonic() < phase3_end:
            remaining = phase3_end - time.monotonic()
            n_iters = max(500, int(remaining * 20))
            sc, a_opt = subgrad_multi_peak(best_a, max_iters=n_iters, top_k=3,
                                           lb_val=best_score * 0.999)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  [P3 multi-peak] NEW BEST: {sc:.8f}")

    # ============================================================
    # Phase 4: Spectral shaping + re-optimization
    # ============================================================
    phase4_end = t_start + 0.78 * budget_s
    print(f"\n=== Phase 4: Spectral Shaping (budget {phase4_end - t_start:.0f}s) ===")

    a_spec_base = best_a.copy()
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.08, 0.12]

    for alpha in alphas:
        if time.monotonic() > phase4_end:
            break
        a_shaped = spectral_shape(a_spec_base, alpha=alpha)
        if a_shaped.sum() < 1e-6:
            continue
        sc_init = compute_score(a_shaped)
        if sc_init < best_score:
            best_score = sc_init
            best_a = a_shaped.copy()
            a_spec_base = a_shaped.copy()
            print(f"  [P4 alpha={alpha}] direct improve: {sc:.8f}")
        # Re-optimize from shaped start
        sc, a_opt, _ = run_lbfgs(a_shaped, beta=20000, maxiter=4000, maxfun=80000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a_spec_base = a_opt.copy()
            print(f"  [P4 alpha={alpha} + opt] NEW BEST: {sc:.8f}")

    # ============================================================
    # Phase 5: Local search with diverse perturbations
    # ============================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Local Search (budget {phase5_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase5_end - 20 and trial < 25:
        trial += 1
        strat = trial % 6

        if strat == 0:
            # Small Gaussian perturbation
            scale = 0.015 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted peak reduction
            a_pert = targeted_perturb(best_a, rng, strength=0.02 + 0.03 * rng.random())
        elif strat == 2:
            # Greedy multi-peak reduction
            a_pert = greedy_peak_reduce(best_a, top_k=3, frac=0.1 + 0.1 * rng.random())
        elif strat == 3:
            # Anti-palindromic center bump
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 25)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.04 * sign * np.exp(-d * d))
        elif strat == 4:
            # Spectral smoothing (attenuate high freq)
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.03 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Edge boost + center reduce
            a_pert = best_a.copy()
            edge_amt = 0.03 * best_a.mean()
            a_pert[0] += edge_amt
            a_pert[-1] += edge_amt
            mid_start = n_best // 5
            mid_end = 4 * n_best // 5
            a_pert[mid_start:mid_end] -= 2.0 * edge_amt / (mid_end - mid_start)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.95

        # Quick LSE optimization from perturbation
        sc, a_opt, _ = run_lbfgs(a_pert, beta=20000, maxiter=3000, maxfun=60000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [P5 #{trial} s={strat}] NEW BEST: {sc:.8f}")

    # ============================================================
    # Final: High-beta LSE + subgradient
    # ============================================================
    print(f"\n=== Final Polish ===")
    if time.monotonic() < deadline - 15:
        remaining = deadline - 15 - time.monotonic()
        maxit = max(3000, int(remaining * 1.5))
        sc, a_opt, _ = run_lbfgs(best_a, beta=100000, maxiter=maxit, maxfun=maxit * 20)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final LSE 100k] NEW BEST: {sc:.8f}")

    if time.monotonic() < deadline - 5:
        sc, a_opt = subgrad_polish(best_a, max_iters=3000, lb_val=best_score * 0.999)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final subgrad] NEW BEST: {sc:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
