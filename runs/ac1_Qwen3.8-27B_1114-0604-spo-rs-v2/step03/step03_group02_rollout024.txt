"""
Step function optimization via coordinate descent + multiplicative equalization + L-BFGS.

Key innovations over previous approach:
1. Coordinate descent with periodic approximation for the 1D subproblem.

   of the multiplicative update.
2. Soft-weighted multiplicative equalization with annealing.
3. Combined approach: alternate between multiplicative updates and coordinate sweeps.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_and_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_and_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 40)))

    try:

                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:

    return compute_score(a_out), a_out


def multiplicative_equalize_soft(a, n_iters=200, beta_start=50, beta_end=2000, 
                                  alpha=0.3, deadline=None):
    """
    Soft-weighted multiplicative equalization with beta annealing.
    
    Instead of hard thresholding for active lags, use softmax weights:
    w[k] = exp(beta * (c[k] - max(c))) / sum(exp(beta * (c - max(c))))
    
    Then: a[j] <- a[j] * (1 - alpha + alpha * h_mean / h[j])
    where h[j] = 2 * sum_k w[k] * a[|k-j|] ... 
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal beta
        t = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Soft weights
        scaled = beta * (c - c_max)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Compute h[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        M_fft = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(w, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(W * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        S_new = a.sum()
        if S_new < 1e-12:

def coordinate_descent_sweep(a, max_sweeps=10, top_k=20, deadline=None):
    """
    Coordinate descent for the step function optimization.
    
    For each j, find the optimal a[j] given all other elements fixed.
    Uses a periodic approximation: only consider top-k lags in the 1D subproblem.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Precompute convolution
    conv = fast_conv(a).copy()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Find top lags (for the 1D subproblem approximation)
        # We'll update this periodically
        top_lags = np.argsort(conv)[::-1][:top_k]
        
        # Random order for this sweep
        order = np.random.permutation(n)
        
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            a_j = a[j]
            
            # d[k] = a[k-j] for valid k (i.e., 0 <= k-j < n)
            # Valid k range: j <= k <= j + n - 1
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)
            
            if k_max < k_min:
                continue
            
            # d[k] for valid k
            # a[k-j] where k-j ranges from 0 to n-1
            d_vals = a[:min(n, k_max - k_min + 1)]  # a[0], a[1], ..., a[k_max-k_min]
            valid_ks = np.arange(k_min, k_max + 1)
            
            # For the 1D subproblem, we only need to check top_lags
            # that are in the valid range
            mask = (top_lags >= k_min) & (top_lags <= k_max)
            relevant_lags = top_lags[mask]
            
            if len(relevant_lags) == 0:
                # Use all valid ks but capped
                relevant_lags = valid_ks[:top_k]
            
            # Get d values for relevant lags: d[k] = a[k-j]
            d_relevant = a[relevant_lags - j]
            c_relevant = conv[relevant_lags] - a_j * d_relevant
            
            # Special case: if 2*j is in relevant_lags, the term is x^2 not x*a[j]
            # Actually, when k = 2j, d[k] = a[k-j] = a[j], and the term is a[j]^2
            # So for k=2j: f_k(x) = c[k] + x^2 (not c[k] + x*a[j])
            # For other k: f_k(x) = c[k] + x*d[k]
            
            # Objective: g(x) = 2n * max_k f_k(x) / (S - a_j + x)^2
            # We want to minimize this for x >= 0
            
            # Simple approach: try a few candidate x values
            # x candidates: 0, a_j, a_j * 1.5, a_j * 0.5, a_j * 2, a_j * 0.25
            # Plus the value that makes the top lag equal to the second top lag
            
            candidates = [0.0, a_j]
            if a_j > 0:
                candidates.extend([a_j * 0.5, a_j * 1.5, a_j * 0.25, a_j * 2.0,
                                   a_j * 0.75, a_j * 1.25])
            
            # Also try the value that equalizes the top two lags
            if len(relevant_lags) >= 2:
                # Sort by current value
                order_relev = np.argsort(c_relevant + a_j * d_relevant)[::-1]
                k1 = relevant_lags[order_relev[0]]
                k2 = relevant_lags[order_relev[1]]
                d1 = a[k1 - j]
                d2 = a[k2 - j]
                c1 = c_relevant[order_relev[0]]
                c2 = c_relevant[order_relev[1]]
                
                # c1 + x*d1 = c2 + x*d2 => x = (c2 - c1) / (d1 - d2)
                if abs(d1 - d2) > 1e-15:
                    x_eq = (c2 - c1) / (d1 - d2)
                    if x_eq > 0:
                        candidates.append(x_eq)
            
            # Evaluate g(x) at each candidate
            S_base = S - a_j
            best_x = a_j
            best_g = np.inf
            
            for x in candidates:
                x = max(0.0, min(x, 1000.0))
                new_S = S_base + x
                if new_S < 1e-12:
                    continue
                
                # Compute max of f_k(x) for relevant lags
                f_vals = c_relevant + x * d_relevant
                # Check if 2j is in relevant lags (quadratic term)
                if 2 * j in relevant_lags:
                    idx = np.where(relevant_lags == 2 * j)[0][0]
                    f_vals[idx] = c_relevant[idx] + x * x  # quadratic term
                
                g_val = 2.0 * n * float(np.max(f_vals)) / (new_S * new_S)
                
                if g_val < best_g:
                    best_g = g_val
                    best_x = x
            
            # Update a[j]
            delta = best_x - a_j
            if abs(delta) < 1e-15:
                continue
            
            a[j] = best_x
            S += delta
            
            # Update convolution: conv[k] += delta * a[k-j] for valid k
            # Valid k: j <= k <= j + n - 1
            conv[k_min:k_max + 1] += delta * a[:k_max - k_min + 1]
            
            # Track best
            # (Skip per-iteration score computation for speed; check at end of sweep)
        
        # End of sweep: compute exact score
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # Reconvolve for accuracy
            conv = fast_conv(a).copy()
    
    return best_sc, best_a


def soft_multiplicative_with_backtrack(a, n_iters=100, deadline=None):
    """
    Multiplicative equalization with backtracking line search on alpha.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (3 * n - 2).bit_length()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = float(c.max())
        
        # Use all lags with exponential weighting (soft)
        beta = 50 + 50 * it  # Increasing sharpness
        scaled = beta * (c - c_max)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        W = np.fft.rfft(w, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(W * np.conj(A), M_fft)
        h = 2.0 * cross[:n]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()
        
        # Try different alphas via backtracking
        alpha = 0.5
        for _ in range(10):
            factor = 1.0 - alpha + alpha * (h_mean / h)
            a_trial = a * factor
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            S_new = a_trial.sum()
            if S_new < 1e-12:
                alpha *= 0.5
                continue
            a_trial = a_trial * (S / S_new)
            sc_trial = compute_score(a_trial)
            if sc_trial < best_sc * 1.0001:  # Accept if not much worse
                a = a_trial
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a.copy()
                break
            alpha *= 0.5
        
    return best_sc, best_a


def spectral_init(n, rng):
    """
    Initialize with a flat power spectrum, then project to non-negative.
    """
    # Flat power spectrum means |A(ω)|² = constant
    # This corresponds to a "random" phase sequence
    # We'll create a sequence whose FFT has roughly constant magnitude
    
    # Start with random phases
    M_fft = 1 << (2 * n - 1).bit_length()
    freqs = np.fft.rfftfreq(M_fft)
    
    # Create flat spectrum with random phases
    A = np.ones(M_fft // 2 + 1, dtype=np.complex128)
    A *= np.exp(1j * rng.uniform(0, 2 * np.pi, size=A.shape))
    # Enforce Hermitian symmetry (already done by rfft construction)
    
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.abs(a)  # Non-negative
    if a.sum() < 1e-12:
        a = np.ones(n) * 0.5
    return a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p)-1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
    
    inits.append(np.ones(n) * 0.5)
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(spectral_init(n, rng))
    
    # Block structure
    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    
    return inits

"""
Step function optimization via coordinate descent + soft multiplicative equalization + L-BFGS.

Key innovations over previous approach:
1. Coordinate descent with periodic approximation for the 1D subproblem - provides
   a different optimization path that can escape stagnation of multiplicative updates.
2. Soft-weighted multiplicative equalization with geometric annealing of beta.
3. Interleaved optimization: alternate between multiplicative updates and coordinate
   sweeps for complementary progress.
4. Aggressive high-beta L-BFGS polishing at the end.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """Run L-BFGS-B to minimize the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_and_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_and_grad(a0, beta)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(100, int(remaining * 150)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 40)))

    try:
        res = minimize(obj_wrapper, a0, args=(beta,),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out


def multiplicative_equalize_soft(a, n_iters=200, beta_start=50, beta_end=2000,
                                  alpha=0.3, deadline=None):
    """
    Soft-weighted multiplicative equalization with beta annealing.
    Uses softmax weights over convolution lags instead of hard thresholding.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        t = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t

        c = fast_conv(a)
        c_max = float(c.max())

        scaled = beta * (c - c_max)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        W = np.fft.rfft(w, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(W * np.conj(A), M_fft)
        h = 2.0 * cross[:n]

        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def coordinate_descent_sweep(a, max_sweeps=20, top_k=30, deadline=None, rng=None):
    """
    Coordinate descent for the step function optimization.
    For each j, find the optimal a[j] given all other elements fixed.
    Uses periodic approximation: only considers top-k lags in the 1D subproblem.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)

    if rng is None:
        rng = np.random.default_rng(0)

    conv = fast_conv(a).copy()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        top_lags = np.argsort(conv)[::-1][:top_k]
        order = rng.permutation(n)

        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break

            a_j = a[j]

            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)

            if k_max < k_min:
                continue

            d_vals = a[:k_max - k_min + 1]
            valid_ks = np.arange(k_min, k_max + 1)

            mask = (top_lags >= k_min) & (top_lags <= k_max)
            relevant_lags = top_lags[mask]

            if len(relevant_lags) == 0:
                relevant_lags = valid_ks[:min(top_k, len(valid_ks))]

            d_relevant = a[relevant_lags - j]
            c_relevant = conv[relevant_lags] - a_j * d_relevant

            # Check if 2j is in relevant lags (quadratic term)
            has_quad = (2 * j >= k_min) and (2 * j <= k_max)
            if has_quad:
                quad_idx = np.where(relevant_lags == 2 * j)[0]
                if len(quad_idx) > 0:
                    quad_idx = quad_idx[0]
                    # For k=2j: f(x) = c + x^2, but we stored c = conv[2j] - a_j*a[j]
                    # So the correct form is c_relevant[quad_idx] + x^2
                    # We need to handle this in the evaluation
                else:
                    has_quad = False
            else:
                quad_idx = -1

            S_base = S - a_j

            # Build candidate x values
            candidates = [0.0, a_j]
            if a_j > 0:
                candidates.extend([a_j * 0.5, a_j * 1.5, a_j * 0.25,
                                   a_j * 2.0, a_j * 0.75, a_j * 1.25])

            if len(relevant_lags) >= 2:
                cur_vals = c_relevant + a_j * d_relevant
                if has_quad and quad_idx >= 0:
                    cur_vals = cur_vals.copy()
                    cur_vals[quad_idx] = c_relevant[quad_idx] + a_j * a_j
                sorted_idx = np.argsort(cur_vals)[::-1]
                k1 = relevant_lags[sorted_idx[0]]
                k2 = relevant_lags[sorted_idx[1]]
                d1 = a[k1 - j]
                d2 = a[k2 - j]
                c1 = c_relevant[sorted_idx[0]]
                c2 = c_relevant[sorted_idx[1]]
                if k1 == 2 * j:
                    pass  # quadratic, skip equalization
                elif k2 == 2 * j:
                    pass
                elif abs(d1 - d2) > 1e-15:
                    x_eq = (c2 - c1) / (d1 - d2)
                    if x_eq > 0 and x_eq < 1000:
                        candidates.append(x_eq)

            # Remove duplicates and sort
            candidates = sorted(set(max(0.0, min(1000.0, x)) for x in candidates))

            best_x = a_j
            best_g = np.inf

            for x in candidates:
                new_S = S_base + x
                if new_S < 1e-12:
                    continue

                f_vals = c_relevant + x * d_relevant
                if has_quad and quad_idx >= 0:
                    f_vals = f_vals.copy()
                    f_vals[quad_idx] = c_relevant[quad_idx] + x * x

                g_val = 2.0 * n * float(np.max(f_vals)) / (new_S * new_S)

                if g_val < best_g:
                    best_g = g_val
                    best_x = x

            delta = best_x - a_j
            if abs(delta) < 1e-15:
                continue

            a[j] = best_x
            S += delta

            # Update convolution incrementally
            conv[k_min:k_max + 1] += delta * a[:k_max - k_min + 1]
            # Fix the quadratic term at k=2j
            if has_quad and 2 * j <= 2 * n - 2:
                # The linear update gave delta * a[j]=delta*a_j (old value)
                # Correct is delta * (a_j + best_x) = delta * (2*a_j + delta)
                # Extra: delta^2
                conv[2 * j] += delta * delta

        # End of sweep: recompute conv exactly
        conv = fast_conv(a).copy()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def subgradient_polish(a, total_iters=200, deadline=None):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step_decay = 1.0 / (1.0 + 0.05 * it)
        a -= step_decay * 0.0005 * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_init(n, rng):
    """Initialize with near-flat power spectrum, projected to non-negative."""
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.ones(M_fft // 2 + 1, dtype=np.complex128)
    A *= np.exp(1j * rng.uniform(0, 2 * np.pi, size=A.shape))
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.abs(a)
    if a.sum() < 1e-12:
        a = np.ones(n) * 0.5
    return a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())

    inits.append(np.ones(n) * 0.5)

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.clip(0.5 + 0.3 * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * np.cos(4.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * np.sin(3.0 * np.pi * x) ** 2, 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(spectral_init(n, rng))

    for nb in (4, 8, 16):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    return inits
