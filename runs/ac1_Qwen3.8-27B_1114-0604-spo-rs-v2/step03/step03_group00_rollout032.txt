from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_and_grad(a, beta):
    """Compute LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of L w.r.t. a
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    # Compute via cross-correlation using FFT
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    gL = 2.0 * conv_res  # gradient of L
    
    # Full gradient of F = 2n * L / S^2
    S2 = S * S
    grad = (2.0 * n / S2) * (gL - 2.0 * L / S)
    fval = 2.0 * n * L / S2
    return fval, grad

    """Adam optimization with annealed LSE temperature."""
    n = len(a0)

    if S < 1e-12:
        a = np.ones(n) / n
        return a
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

        lr = lr_init

            t += 1
            fval, grad = lse_and_grad(a, beta)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S  # Normalize

"""
Optimizes non-negative sequences to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses Adam-based LSE optimization with adaptive temperature, mirror descent,
peak shaving, coordinate descent, and L-BFGS polish.
Key innovations: adaptive beta from peak-gap analysis, multiplicative (mirror) updates,
direct linear-system-based pressure equalization, and multi-scale length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_and_grad(a, beta):
    """Compute LSE-smoothed objective value and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[i] = 2 * sum_j p[i+j] * a[j]
    # Computed as cross-correlation via FFT
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1].astype(np.float64), M2_fft)
    P = np.fft.rfft(p.astype(np.float64), M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    gL = 2.0 * conv_res
    S2 = S * S
    grad = (2.0 * n / S2) * (gL - 2.0 * L / S)
    fval = 2.0 * n * L / S2
    return fval, grad


def adam_optimize(a0, beta_schedule, steps_per_beta, lr_init=0.05, lr_decay=0.9995):
    """Adam optimization with annealed LSE temperature."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10

    for beta in beta_schedule:
        lr = lr_init
        for step in range(steps_per_beta):
            t += 1
            fval, grad = lse_and_grad(a, beta)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
            lr *= lr_decay
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    Adam beta={beta}: score={sc:.8f}")
    return best_a


def mirror_descent(a0, beta_schedule, steps_per_beta, lr_init=0.01, lr_decay=0.9998):
    """Multiplicative (mirror) descent on LSE-smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 1e-10, 1000.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    for beta in beta_schedule:
        lr = lr_init
        for step in range(steps_per_beta):
            fval, grad = lse_and_grad(a, beta)
            # Multiplicative update: a[i] *= exp(-lr * grad[i])
            # Use log for numerical stability
            log_updates = -lr * grad
            log_updates = np.clip(log_updates, -5.0, 5.0)
            a = a * np.exp(log_updates)
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S < 1e-12:
                break
            a = a / S
            lr *= lr_decay
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    Mirror beta={beta}: score={sc:.8f}")
    return best_a


def adaptive_beta_from_conv(c, target_gap=2.0):
    """Choose beta so that beta*(cmax - c2nd) ≈ target_gap."""
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    if len(c_sorted) > 1:
        c_2nd = c_sorted[1]
    else:
        c_2nd = c_max
    gap = c_max - c_2nd
    if gap < 1e-15:
        return 1000.0
    return min(10000.0, max(5.0, target_gap / gap))


def pressure_equalize(a, lam, n_iters=50):
    """
    Solve for a such that T_lam * a is approximately constant on support.
    T_lam[i,j] = 2*lam[j-i] (with bounds). Uses iterative water-filling.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    LAM = np.fft.rfft(lam.astype(np.float64), M2_fft)

    for it in range(n_iters):
        # Compute g = T_lam * a (the "pressure" at each position)
        RA = np.fft.rfft(a[::-1].astype(np.float64), M2_fft)
        conv_res = np.fft.irfft(RA * LAM, M2_fft)[n - 1:2 * n - 1]
        g = 2.0 * conv_res

        # Water-filling: a[i] proportional to max(nu - g[i], 0)
        g_mean = np.mean(g)
        g_std = np.std(g) + 1e-15
        # Soft version: a[i] ∝ exp(-g[i] / temperature)
        temp = g_std * 0.5
        a_new = np.exp(-g / temp)
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        # Blend
        a = 0.7 * a + 0.3 * a_new
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S > 1e-12:
            a = a / S
    return a


def peak_shave(a, n_iters=200, soft_alpha=0.1):
    """Soft peak shaving: reduce high-contributing elements, boost low ones."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = np.array([a[k_star - j] for j in range(j_start, j_end + 1)])
        # Soft reduction: reduce proportional to contribution
        contrib_max = contrib.max()
        if contrib_max < 1e-15:
            break
        reduction = soft_alpha * contrib / contrib_max
        seg_new = seg - reduction
        seg_new = np.clip(seg_new, 0.0, 1000.0)
        # Redistribute the removed mass to low-contributing elements
        removed = seg.sum() - seg_new.sum()
        if removed > 1e-15:
            low_mask = contrib < 0.3 * contrib_max
            low_count = low_mask.sum()
            if low_count > 0:
                per = removed / low_count
                seg_new[low_mask] += per
        a[j_start:j_end + 1] = seg_new
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def coord_descent_peak(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            continue
        seg = a[j_start:j_end + 1]
        contrib = np.array([a[k_star - j] for j in range(j_start, j_end + 1)])
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            best_val = a_old_j
            best_local_sc = best_sc
            for scale in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 0.0]:
                cv = a_old_j * scale
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def lse_objective_wrapper(a, beta):
    """Wrapper for scipy minimize: returns (fval, grad)."""
    return lse_and_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_and_grad(x, beta)

    res = minimize(
        obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) / n)
    inits.append((0.3 + 0.7 * (2.0 * x - 1.0) ** 2))
    inits.append(0.5 + 0.35 * np.cos(2.0 * np.pi * x))
    inits.append(0.4 + 0.6 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.25 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 100.0))
    # IFFT-based
    M_fft = 1 << (2 * n).bit_length()
    phases = rng.uniform(-np.pi, np.pi, M_fft)
    a_ifft = np.fft.ifft(np.exp(1j * phases)).real[:n]
    a_ifft = np.abs(a_ifft)
    if a_ifft.sum() > 1e-12:
        inits.append(a_ifft / a_ifft.sum())
    # Bimodal
    center = n // 2
    inits.append(0.3 + 0.7 * np.exp(-((x - 0.15) ** 2) / 0.01) + 0.7 * np.exp(-((x - 0.85) ** 2) / 0.01))
    # Block structure
    nb = 16
    per = max(1, n // nb)
    vals = 0.2 + 0.8 * rng.random(nb)
    a_block = np.repeat(vals, per)
    if len(a_block) < n:
        a_block = np.concatenate([a_block, vals[:n - len(a_block)]])
    a_block = a_block[:n]
    if a_block.sum() > 1e-12:
        inits.append(a_block / a_block.sum())
    return inits

    # Determine lengths to try
    lengths = [5000, 7000, 9000, 12000]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes' if parent is not None else 'no'}")

    # Phase 1: Multi-start Adam/Mirror (45% time)
    phase1_end = t_start + 0.45 * (budget_s - 20)
    print(f"\n{'='*50}")
    print("Phase 1: Multi-start optimization (Adam + Mirror)")
    print(f"{'='*50}")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"\n  --- n={n} ---")
        inits = build_inits(n, rng, parent)

        remaining = phase1_end - time.monotonic()
        n_inits = len(inits)
        # Time budget per init-method pair
        time_per_pair = max(10.0, remaining / (n_inits * 2))
        # Estimate steps: ~0.3ms per step for n=7000
        est_time_per_step = n * 3e-8
        steps_per_beta = max(100, int(time_per_pair / (5 * est_time_per_step * 3)))
        steps_per_beta = min(steps_per_beta, 5000)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            if len(a0) != n:
                continue
            a0 = a0.astype(np.float64).copy()
            a0 = np.clip(a0, 0.0, 1000.0)
            S = a0.sum()
            if S < 1e-12:
                continue
            a0 = a0 / S

            # Adam
            beta_sched = [20, 50, 100, 200, 500, 1000]
            a_opt = adam_optimize(a0, beta_sched, steps_per_beta, lr_init=0.04)
            sc = compute_score(a_opt)
            update_best(sc, a_opt, f"Adam n={n} init={i}")

            # Mirror descent from Adam result
            if time.monotonic() < phase1_end - 30:
                a_mirror = mirror_descent(a_opt, [100, 300, 500, 1000],
                                          max(50, steps_per_beta // 2), lr_init=0.005)
                sc = compute_score(a_mirror)
                update_best(sc, a_mirror, f"Mirror n={n} init={i}")

    print(f"\nAfter Phase 1: best={best_score:.8f}, n={len(best_a)}")

    # Phase 2: L-BFGS with adaptive beta (30% time)
    phase2_end = t_start + 0.75 * (budget_s - 20)
    print(f"\n{'='*50}")
    print("Phase 2: L-BFGS refinement with adaptive beta")
    print(f"{'='*50}")

    n_best = len(best_a)
    # Get adaptive beta from current solution
    c_best = fast_conv(best_a)
    beta_adapt = adaptive_beta_from_conv(c_best, target_gap=2.0)
    print(f"  Adaptive beta from peak gap: {beta_adapt:.0f}")

    betas_to_try = [50, 100, 200, 500, int(beta_adapt), 2000, 5000]
    betas_to_try = sorted(set(betas_to_try))

    for beta in betas_to_try:
        if time.monotonic() > phase2_end:
            break
        print(f"  L-BFGS beta={beta}...")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, f"LBFGS beta={beta}")

    # Try from a few perturbed starts
    for trial in range(5):
        if time.monotonic() > phase2_end:
            break
        scale = 0.01 + 0.02 * trial
        a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = a_pert.sum()
        if S > 1e-12:
            a_pert = a_pert / S
        sc, a_opt = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, f"LBFGS pert{trial}")

    print(f"\nAfter Phase 2: best={best_score:.8f}")

    # Phase 3: Peak shaving + pressure equalization (15% time)
    phase3_end = t_start + 0.90 * (budget_s - 20)
    print(f"\n{'='*50}")
    print("Phase 3: Peak shaving + pressure equalization")
    print(f"{'='*50}")

    # Peak shaving
    a_shaved = peak_shave(best_a, n_iters=500, soft_alpha=0.05)
    sc = compute_score(a_shaved)
    update_best(sc, a_shaved, "peak_shave_soft")

    a_shaved2 = peak_shave(best_a, n_iters=300, soft_alpha=0.1)
    sc = compute_score(a_shaved2)
    update_best(sc, a_shaved2, "peak_shave_strong")

    # Pressure equalization with various lambda
    n_pe = len(best_a)
    c_cur = fast_conv(best_a)
    for gamma in [50, 100, 200, 500]:
        if time.monotonic() > phase3_end:
            break
        lam = np.exp(gamma * (c_cur - c_cur.max()))
        lam = lam / lam.sum()
        a_pe = pressure_equalize(best_a, lam, n_iters=100)
        sc = compute_score(a_pe)
        update_best(sc, a_pe, f"pressure_eq gamma={gamma}")

    # Coordinate descent
    a_cd = coord_descent_peak(best_a, n_sweeps=5, n_top=100)
    sc = compute_score(a_cd)
    update_best(sc, a_cd, "coord_descent")

    print(f"\nAfter Phase 3: best={best_score:.8f}")

    # Phase 4: Final polish (10% time)
    print(f"\n{'='*50}")
    print("Phase 4: Final polish")
    print(f"{'='*50}")

    # High-beta L-BFGS
    if time.monotonic() < deadline - 60:
        c_final = fast_conv(best_a)
        beta_final = adaptive_beta_from_conv(c_final, target_gap=1.0)
        beta_final = max(500, min(10000, beta_final))
        print(f"  Final L-BFGS with beta={beta_final:.0f}")
        sc, a_opt = run_lbfgs(best_a, beta_final, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "final_LBFGS")

    # Final Adam with very low lr
    if time.monotonic() < deadline - 60:
        a_fin = adam_optimize(best_a, [1000, 2000, 5000], 500, lr_init=0.01, lr_decay=0.999)
        sc = compute_score(a_fin)
        update_best(sc, a_fin, "final_Adam")

    # One more peak shave
    if time.monotonic() < deadline - 30:
        a_ps = peak_shave(best_a, n_iters=200, soft_alpha=0.03)
        sc = compute_score(a_ps)
        update_best(sc, a_ps, "final_shave")

    # Interpolation between lengths
    if parent is not None and len(parent) != len(best_a) and time.monotonic() < deadline - 60:
        n_interp = (len(parent) + len(best_a)) // 2
        x_old1 = np.linspace(0, 1, len(parent))
        x_old2 = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_interp)
        a_interp = 0.5 * np.interp(x_new, x_old1, parent) + 0.5 * np.interp(x_new, x_old2, best_a)
        a_interp = np.clip(a_interp, 0.0, 1000.0)
        S = a_interp.sum()
        if S > 1e-12:
            a_interp = a_interp / S
        a_opt = adam_optimize(a_interp, [100, 300, 500], 300, lr_init=0.03)
        sc = compute_score(a_opt)
        update_best(sc, a_opt, "interp")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
