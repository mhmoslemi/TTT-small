```
    n=a.size

    ra=np.fft.rfft(a[::-1], M_fft)

    h=np.fft.irfft(ra*P, M_fft)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a)^2) for non-negative a.

Algorithm:
1. Use a strong analytic family a_i = 1 + A*sin(2*pi*i/n + phi).
   This family is motivated by continuous density autocorrelation flattening
   and attains scores very near 1.5 for large n.
2. Grid + local refinement over (A, phi) using fast precomputed convolution
   arrays for that one sinusoid basis.
3. Add a few small harmonic perturbations to break finite-n symmetry errors.
4. Polish with L-BFGS-B on an LSE-smoothed peak-convolution objective.
5. Finish with an active-set projected subgradient polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Globals used by top-level objective callbacks (no closures/lambdas).
OPT_BETA = 100.0
REG_EPS = 1e-10

FAM_N = 0
FAM_NARR = None
FAM_LS = None
FAM_LC = None
FAM_QSS = None
FAM_QCC = None
FAM_QSC = None
FAM_SUM_S = 0.0
FAM_SUM_C = 0.0


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    M = 2 * n - 1
    Mfft = _next_pow2(M)
    A = np.fft.rfft(a, Mfft)
    c = np.fft.irfft(A * A, Mfft)[:M]
    return c


def cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na = int(a.size)
    nb = int(b.size)
    M = na + nb - 1
    Mfft = _next_pow2(M)
    A = np.fft.rfft(a, Mfft)
    B = np.fft.rfft(b, Mfft)
    c = np.fft.irfft(A * B, Mfft)[:M]
    return c


def fast_conv_cross_rev(a, p):
    """
    For gradient of sum_k p_k * (a*a)[k]:
    returns g_j = sum_k p_k * a_{k-j} (where valid), length len(a).
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = int(a.size)
    m = int(p.size)
    M = n + m - 1
    Mfft = _next_pow2(M)
    ra = np.fft.rfft(a[::-1], Mfft)
    P = np.fft.rfft(p, Mfft)
    h = np.fft.irfft(ra * P, Mfft)
    return h[n - 1:2 * n - 1]


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    S = float(np.sum(a))
    if S < 1e-12:
        return float("inf")
    c = fast_conv(a)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return float("inf")
    return float(2.0 * a.size * maxc / (S * S))


def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return 1e100, np.zeros(0)
    if not np.all(np.isfinite(a)):
        return 1e100, np.zeros(n)

    S = float(np.sum(a))
    if S < 1e-12:
        return 1e100, np.zeros(n)

    c = fast_conv(a)
    cmax = float(np.max(c))
    if cmax <= 1e-300:
        return 0.0, np.zeros(n)

    if beta <= 0.0:
        beta = 1.0

    scaled = beta * (c - cmax)
    expv = np.exp(scaled)
    se = float(np.sum(expv))
    if se < 1e-300:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        L = cmax
    else:
        p = expv / se
        L = cmax + float(np.log(se)) / beta

    g = 2.0 * fast_conv_cross_rev(a, p)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    score = 2.0 * n * L / S2

    # Tiny scale anchor: keeps mean near 1, negligible impact on exact score.
    if REG_EPS > 0.0:
        dmean = S / float(n) - 1.0
        if abs(dmean) > 1e-14:
            score += REG_EPS * dmean * dmean
            grad += (2.0 * REG_EPS * dmean / float(n)) * np.ones(n, dtype=np.float64)

    if not np.isfinite(score):
        score = 1e100
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return float(score), grad


def lse_fun(x):
    return lse_score_grad(x, OPT_BETA)


def run_lbfgs(a0, beta, maxiter, maxfun):
    global OPT_BETA
    n = int(len(a0))
    x0 = np.asarray(a0, dtype=np.float64).copy()
    x0 = np.clip(x0, 0.0, 10.0)
    if np.sum(x0) < 1e-8:
        x0 = np.ones(n, dtype=np.float64) * 0.5

    bounds = [(0.0, 10.0)] * n
    OPT_BETA = float(beta)

    options = {
        "ftol": 1e-14,
        "gtol": 1e-11,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            lse_fun,
            x0,
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
    except Exception as e:
        print("L-BFGS failed:", e)
        return compute_score(x0), x0, None

    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        x = x0
    x = np.clip(x, 0.0, 10.0)
    if np.sum(x) < 1e-8:
        x = x0
    return compute_score(x), x, res


def active_subgrad_polish(a, max_iters, deadline):
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    best_a = a.copy()
    best_sc = compute_score(a)

    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    for _ in range(int(max_iters)):
        if time.monotonic() >= deadline:
            break

        c = fast_conv(a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        tol = max(1.0, 0.005 * M)
        active = c >= (M - tol)
        if not np.any(active):
            active[int(np.argmax(c))] = True

        p = active.astype(np.float64) / float(np.sum(active))
        L = float(np.dot(p, c))

        g = 2.0 * fast_conv_cross_rev(a, p)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

        dmean = S / float(n) - 1.0
        if abs(dmean) > 1e-14:
            grad = grad + (2.0 * REG_EPS * dmean / float(n)) * np.ones(n, dtype=np.float64)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        step = (sc - lower_bound) / (gn * gn + 1e-12)
        step = min(step, 0.5)

        cur_a = a
        cur_sc = sc
        factors = (1.0, 0.5, 0.25, 0.1, 0.05)
        for fac in factors:
            cand = a - step * fac * grad
            cand = np.clip(cand, 0.0, 10.0)
            cm = float(np.mean(cand))
            if cm > 0.0:
                cand = cand / cm
            if np.sum(cand) < 1e-8:
                continue
            cs = compute_score(cand)
            if cs < cur_sc:
                cur_sc = cs
                cur_a = cand

        a = cur_a

    final_sc = compute_score(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return best_sc, best_a


def family_score_current(A, phi):
    if FAM_N <= 0 or FAM_NARR is None:
        return float("inf")

    cp = float(np.cos(phi))
    sp = float(np.sin(phi))
    A = float(A)

    S = float(FAM_N) + A * (cp * FAM_SUM_S + sp * FAM_SUM_C)
    if S <= 1e-12:
        return float("inf")

    b2 = A * A
    cvec = (
        FAM_NARR
        + 2.0 * A * (cp * FAM_LS + sp * FAM_LC)
        + b2 * (cp * cp * FAM_QSS + 2.0 * sp * cp * FAM_QSC + sp * sp * FAM_QCC)
    )
    maxc = float(np.max(cvec))
    if not np.isfinite(maxc):
        return float("inf")
    return float(2.0 * FAM_N * maxc / (S * S))


def optimize_family(n, deadline):
    global FAM_N, FAM_NARR, FAM_LS, FAM_LC, FAM_QSS, FAM_QCC, FAM_QSC, FAM_SUM_S, FAM_SUM_C

    n = int(n)
    i = np.arange(n, dtype=np.float64)
    ang = 2.0 * np.pi * i / float(n)
    s = np.sin(ang)
    c = np.cos(ang)

    sum_s = float(np.sum(s))
    sum_c = float(np.sum(c))

    N = np.empty(2 * n - 1, dtype=np.float64)
    N[:n] = np.arange(1, n + 1, dtype=np.float64)
    N[n:] = np.arange(n - 1, 0, -1, dtype=np.float64)

    ones = np.ones(n, dtype=np.float64)
    LS = cross_conv(ones, s)
    LC = cross_conv(ones, c)
    QSS = fast_conv(s)
    QCC = fast_conv(c)
    QSC = cross_conv(s, c)

    FAM_N = n
    FAM_NARR = N
    FAM_LS = LS
    FAM_LC = LC
    FAM_QSS = QSS
    FAM_QCC = QCC
    FAM_QSC = QSC
    FAM_SUM_S = sum_s
    FAM_SUM_C = sum_c

    best_sc = float("inf")
    best_A = 0.88
    best_phi = np.pi

    # Deterministic good starts
    initial_points = (
        (0.88, np.pi),
        (0.86, np.pi),
        (1.0 / np.sqrt(2.0), np.pi),
        (0.80, np.pi),
        (0.70, np.pi),
    )
    for A, phi in initial_points:
        sc = family_score_current(A, phi)
        if sc < best_sc:
            best_sc = sc
            best_A = A
            best_phi = phi

    # Coarse 2D grid.
    if time.monotonic() < deadline:
        A_vals = np.linspace(0.0, 1.0, 51)
        phi_vals = np.linspace(0.0, 2.0 * np.pi, 64, endpoint=False)
        for A in A_vals:
            if time.monotonic() >= deadline:
                break
            for phi in phi_vals:
                sc = family_score_current(float(A), float(phi))
                if sc < best_sc:
                    best_sc = sc
                    best_A = float(A)
                    best_phi = float(phi)

    # Refine around best.
    if time.monotonic() < deadline:
        da = 0.05
        dp = 0.20
        A_grid = np.linspace(max(0.0, best_A - da), min(1.0, best_A + da), 41)
        phi_grid = np.linspace(best_phi - dp, best_phi + dp, 41)
        for A in A_grid:
            if time.monotonic() >= deadline:
                break
            for phi in phi_grid:
                sc = family_score_current(float(A), float(phi))
                if sc < best_sc:
                    best_sc = sc
                    best_A = float(A)
                    best_phi = float(phi)

    if time.monotonic() < deadline:
        da = 0.012
        dp = 0.045
        A_grid = np.linspace(max(0.0, best_A - da), min(1.0, best_A + da), 21)
        phi_grid = np.linspace(best_phi - dp, best_phi + dp, 21)
        for A in A_grid:
            if time.monotonic() >= deadline:
                break
            for phi in phi_grid:
                sc = family_score_current(float(A), float(phi))
                if sc < best_sc:
                    best_sc = sc
                    best_A = float(A)
                    best_phi = float(phi)

    cp = float(np.cos(best_phi))
    sp = float(np.sin(best_phi))
    a = 1.0 + float(best_A) * (cp * s + sp * c)
    a = np.clip(a, 0.0, 1000.0)
    mn = float(np.mean(a))
    if mn > 0.0:
        a = a / mn

    exact_sc = compute_score(a)
    if exact_sc < best_sc:
        best_sc = exact_sc

    return best_sc, a, best_A, best_phi


def harmonic_refine(a, deadline):
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    best_a = a.copy()
    best_sc = compute_score(a)
    idx = np.arange(n, dtype=np.float64)

    for freq in (2, 3, 4):
        if time.monotonic() >= deadline:
            break

        amps = np.linspace(-0.05, 0.05, 11)
        phases = np.linspace(0.0, 2.0 * np.pi, 24, endpoint=False)

        for amp in amps:
            amp = float(amp)
            if abs(amp) < 1e-12:
                continue
            for phase in phases:
                if time.monotonic() >= deadline:
                    break
                cand = best_a + amp * np.sin(2.0 * np.pi * freq * idx / float(n) + float(phase))
                cand = np.clip(cand, 0.0, 10.0)
                cm = float(np.mean(cand))
                if cm <= 0.0:
                    continue
                cand = cand / cm
                sc = compute_score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_a = cand.copy()

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + max(60.0, budget - 25.0)

    print(f"Starting candidate search with budget {budget:.1f}s")

    # Length 8192 gives a large safety margin and still evaluates quickly.
    n = 8192

    best_a = None
    best_sc = float("inf")

    # Phase A: strong sine family.
    fam_deadline = min(deadline, t0 + 180.0)
    try:
        sc, a, A, phi = optimize_family(n, fam_deadline)
        if np.isfinite(sc):
            best_sc = sc
            best_a = a.astype(np.float64).copy()
            print(f"Sine family: score={sc:.10f}, A={A:.6f}, phi={phi:.6f}, t={time.monotonic()-t0:.1f}s")
    except Exception as e:
        print("Sine family failed:", e)

    # Ensure a universally safe fallback, even if the grid had a bad phase.
    try:
        t_safe = time.monotonic()
        i = np.arange(n, dtype=np.float64)
        a_safe = 1.0 - (1.0 / np.sqrt(2.0)) * np.sin(2.0 * np.pi * i / float(n))
        a_safe = np.clip(a_safe, 0.0, 1000.0)
        mn = float(np.mean(a_safe))
        if mn > 0.0:
            a_safe = a_safe / mn
        sc_safe = compute_score(a_safe)
        if sc_safe < best_sc:
            best_sc = sc_safe
            best_a = a_safe.astype(np.float64).copy()
            print(f"Safe fallback: score={sc_safe:.10f}, t={time.monotonic()-t_safe:.1f}s")
    except Exception as e:
        print("Safe fallback failed:", e)

    if best_a is None or not np.isfinite(best_sc):
        # Very cheap deterministic final fallback.
        i = np.arange(n, dtype=np.float64)
        best_a = 1.0 - (1.0 / np.sqrt(2.0)) * np.sin(2.0 * np.pi * i / float(n))
        best_sc = compute_score(best_a)
        print(f"Emergency fallback score={best_sc:.10f}")

    # Phase B: small harmonic perturbations.
    if time.monotonic() < deadline - 300.0:
        h_deadline = min(deadline, time.monotonic() + 180.0)
        try:
            sc2, a2 = harmonic_refine(best_a, h_deadline)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a2.astype(np.float64).copy()
                print(f"Harmonic refine: score={sc2:.10f}, t={time.monotonic()-t0:.1f}s")
            else:
                print(f"Harmonic refine: no improvement, best remains {best_sc:.10f}")
        except Exception as e:
            print("Harmonic refine failed:", e)

    # Phase C: L-BFGS polishing with continuation in beta.
    if time.monotonic() < deadline - 120.0:
        betas = (80.0, 200.0, 500.0, 1000.0, 2000.0)
        for beta in betas:
            if time.monotonic() >= deadline - 120.0:
                break
            rem = deadline - time.monotonic()
            if rem > 1200.0:
                maxfun = 20000
                maxiter = 10000
            elif rem > 600.0:
                maxfun = 10000
                maxiter = 5000
            elif rem > 240.0:
                maxfun = 5000
                maxiter = 2500
            else:
                maxfun = 2000
                maxiter = 1000

            a = best_a.astype(np.float64).copy()
            mn = float(np.mean(a))
            if mn > 0.0:
                a = a / mn

            try:
                sc2, a2, _ = run_lbfgs(a, beta, maxiter, maxfun)
                if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
                    best_sc = sc2
                    best_a = a2.astype(np.float64).copy()
                    print(f"L-BFGS beta={beta:.0f}: score={sc2:.10f}, t={time.monotonic()-t0:.1f}s")
                else:
                    print(f"L-BFGS beta={beta:.0f}: no improvement, best remains {best_sc:.10f}")
            except Exception as e:
                print(f"L-BFGS beta={beta:.0f} failed:", e)

    # Phase D: active-set subgradient polish.
    if time.monotonic() < deadline - 60.0:
        try:
            sc2, a2 = active_subgrad_polish(best_a, 400, deadline - 60.0)
            if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_a = a2.astype(np.float64).copy()
                print(f"Subgradient polish: score={sc2:.10f}, t={time.monotonic()-t0:.1f}s")
            else:
                print(f"Subgradient polish: no improvement, best remains {best_sc:.10f}")
        except Exception as e:
            print("Subgradient polish failed:", e)

    # Final cleanup.
    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)

    if best_a.size == 0 or np.sum(best_a) < 0.01:
        i = np.arange(n, dtype=np.float64)
        best_a = 1.0 - (1.0 / np.sqrt(2.0)) * np.sin(2.0 * np.pi * i / float(n))

    mn = float(np.mean(best_a))
    if mn > 0.0:
        best_a = best_a / mn

    out = [float(x) for x in best_a]

    # Final verification if there is time.
    try:
        if time.monotonic() < deadline - 5.0:
            verified = float(evaluate_sequence(out))
            print(f"FINAL: internal={best_sc:.10f}, verified={verified:.10f}")
        else:
            print(f"FINAL: internal={best_sc:.10f}, skipped verification for timeout safety")
    except Exception as e:
        print("Verification failed:", e)

    print(f"Returned length {len(out)} after {time.monotonic()-t0:.1f}s")
    return out
```
