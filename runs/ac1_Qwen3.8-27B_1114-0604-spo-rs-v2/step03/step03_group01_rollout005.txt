"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines FISTA, multiplicative updates, structured perturbations, and L-BFGS.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / sum^2. Returns (score, max_c, argmax)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.inf, 0
    max_c = float(c.max())
    score = 2.0 * n * max_c / (S * S)
    return score, max_c, int(c.argmax())

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    N_fft = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    # f[j] = sum_m a[m] * p[m+j]
    f = np.fft.irfft(A_fft * np.conj(P_fft), N_fft)[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (f / S2 - 2.0 * L / S3)
    
    return score, grad

def exact_subgrad(a):
    """Exact score and subgradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    # Subgradient: for each j in argmax(c), the contribution to da[j] is a[k-j]
    # where k ranges over argmax positions
    k_star = int(c.argmax())
    sg = np.zeros(n)
    # c[k_star] = sum_j a[j] * a[k_star - j] for valid j
    # dc[k_star]/da[j] = a[k_star - j] if 0 <= k_star - j < n
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad

def topk_subgrad(a, top_k=10):
    """Score and averaged subgradient over top-k convolution lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    # Weight by proximity to max
    weights = c[sorted_idx] - c_max + 1.0
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for i, k in enumerate(sorted_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += weights[i] * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
    return score, grad

def fista_lse(a_init, beta, n_steps=20000, time_limit=60.0, lr=None):
    """FISTA (proximal gradient with Nesterov momentum) in a-space with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    # Normalize to sum=1
    a = a / a.sum()
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)[0]
    t_start = time.monotonic()
    
    if lr is None:
        # Estimate Lipschitz constant
        # For LSE with parameter beta, the Hessian is bounded by beta * ||something||
        # Use a simple heuristic
        lr = 1.0 / (beta * n * 0.1)
        # Actually, let's use backtracking
        lr = 1.0 / (beta * 10.0)  # Start conservative
    
    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        
        score, grad = lse_score_grad(y, beta)
        
        # Gradient step
        x = y - lr * grad
        x = np.clip(x, 0.0, None)
        if x.sum() < 1e-12:
            x = a.copy()
        else:
            x = x / x.sum()
        
        # Update y with momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - a)
        y = np.clip(y, 0.0, None)
        if y.sum() > 1e-12:
            y = y / y.sum()
        t = t_new
        a = x
        
        if step % 5000 == 0 or step == n_steps - 1:
            sc = compute_score(a)[0]
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)[0]
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def multiplicative_updates(a_init, n_steps=5000, eta=0.1, time_limit=60.0):
    """Multiplicative/exponentiated gradient updates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)[0]
    t_start = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        
        score, grad = exact_subgrad(a)
        
        # Multiplicative update: a[j] *= exp(-eta * grad[j])
        # Normalize grad to have zero mean (for sum-preserving updates)
        grad_centered = grad - grad.mean()
        a = a * np.exp(-eta * grad_centered)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if step % 1000 == 0:
            sc = compute_score(a)[0]
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)[0]
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def flattening_heuristic(a, n_iters=100, time_limit=30.0):
    """Direct flattening: iteratively reduce peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)[0]
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        score, max_c, k_star = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find peak region
        c = fast_conv(a)
        threshold = c.max() - 0.1 * c.max()
        peak_positions = np.where(c >= threshold)[0]
        
        # For each peak position, find the a[j] that contribute most
        # and reduce them slightly
        reduce_mask = np.zeros(n, dtype=bool)
        for k in peak_positions:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                reduce_mask[j] = True
        
        # Reduce the top contributors
        if reduce_mask.sum() > 0:
            # Find the a[j] values at peak positions and reduce the largest ones
            peak_vals = a[reduce_mask]
            sort_idx = np.argsort(peak_vals)[::-1][:max(1, n // 10)]
            # Get actual indices
            mask_indices = np.where(reduce_mask)[0]
            reduce_indices = mask_indices[sort_idx]
            # Reduce by a small factor
            a[reduce_indices] *= 0.95
            # Renormalize
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a = a / s
    
    sc_final = compute_score(a)[0]
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def run_lbfgs(a_init, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init, 1e-10, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    
    def fun_and_grad(x):
        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-10:
            return 1e10, np.zeros(n)
        a_norm = a / s
        sc, g = lse_score_grad(a_norm, beta)
        # Chain rule through normalization: da_norm/da[j] = (n*delta[j] - 1)/n * ... 
        # Actually, for the score which is scale-invariant, we can just use the gradient
        # of the score w.r.t. a_norm, projected onto the tangent space of sum=1
        # Since score is homogeneous of degree 0 (scale invariant), the gradient
        # in the original space is just the gradient in the normalized space
        # projected to the zero-sum subspace
        g_centered = g - g.mean()
        return sc, g_centered
    
    bounds = [(1e-15, 1000.0)] * n
    res = minimize(fun_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    s = a_out.sum()
    if s > 1e-10:
        a_out = a_out / s
    exact_sc = compute_score(a_out)[0]
    return exact_sc, a_out

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

        time_left = effective_budget - (time.monotonic() - t_start)
        if time_left < 15:
            break
        sc, a_opt = multiplicative_updates(best_a, n_steps=10000, eta=eta,
                                          time_limit=min(20.0, time_left * 0.3))
        try_update(sc, a_opt, f"MultUpd eta={eta}")
        print(f"    MultUpd eta={eta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 4: Flattening heuristic
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Flattening heuristic")
    print(f"{'='*60}")
    
    t0 = time.monotonic()
    sc, a_opt = flattening_heuristic(best_a, n_iters=200,
                                    time_limit=min(30.0, effective_budget * 0.05))
    try_update(sc, a_opt, "Flattening")
    print(f"    Flattening: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 5: L-BFGS polish (high beta)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: L-BFGS polish")
    print(f"{'='*60}")
    
    for beta in [500, 1000, 2000, 5000]:
        t0 = time.monotonic()
        time_left = effective_budget - (time.monotonic() - t_start)
        if time_left < 30:
            break
        sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=5000, maxfun=100000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Phase 6: Structured restarts
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Structured restarts")
    print(f"{'='*60}")
    
    n_best = len(best_a)
    restart_deadline = t_start + effective_budget - 60.0
    
    strategies = [
        ('low_freq_sin', 0),
        ('peak_reduce', 1),
        ('block_mod', 2),
        ('gaussian', 3),
        ('low_freq_sin', 4),
        ('peak_reduce', 5),
        ('gaussian', 6),
        ('block_mod', 7),
    ]
    
    for name, strat in strategies:
        if time.monotonic() > restart_deadline:
            break
        
        if strat == 0:
            # Low-frequency sinusoidal perturbation
            freq = 1.0 + 3.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            amp = 0.05 + 0.1 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(2*np.pi*freq*x + phase))
        elif strat == 1:
            # Peak reduction: reduce elements contributing to max convolution
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            # Reduce peak contributors
            for j in range(js, je + 1):
                a_pert[j] *= 0.7
            # Redistribute
            removed = float(best_a[js:je+1].sum() - a_pert[js:je+1].sum())
            mask = np.zeros(n_best, dtype=bool)
            mask[:js] = True
            mask[je+1:] = True
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strat == 2:
            # Block modification
            n_blocks = 4
            block_size = n_best // n_blocks
            block = int(rng.integers(0, n_blocks))
            a_pert = best_a.copy()
            a_pert[block*block_size:(block+1)*block_size] *= rng.uniform(0.5, 1.5)
        else:
            # Gaussian perturbation
            scale = 0.05 + 0.1 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        s = a_pert.sum()
        if s > 1e-10:
            a_pert = a_pert / s
        
        # Quick optimization
        time_left = restart_deadline - time.monotonic()
        if time_left < 10:
            break
        sc, a_r = fista_lse(a_pert, beta=200, n_steps=10000,
                           time_limit=min(20.0, time_left * 0.5))
        time_left = restart_deadline - time.monotonic()
        if time_left > 15:
            sc2, a_r2 = run_lbfgs(a_r, beta=1000, maxiter=3000, maxfun=50000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r, f"restart {name}")
    
    # ==================================================================
    # Phase 7: Final intensive polish
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")
    
    final_deadline = t_start + effective_budget - 5.0
    
    # High-beta L-BFGS
    for beta in [3000, 5000, 10000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, f"final LBFGS beta={beta}")
        print(f"    Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final FISTA
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        sc, a_opt = fista_lse(best_a, beta=5000, n_steps=50000,
                             time_limit=min(30.0, final_deadline - time.monotonic()))
        try_update(sc, a_opt, "final FISTA")
        print(f"    Final FISTA: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # One more flattening pass
    if time.monotonic() < final_deadline - 5:
        t0 = time.monotonic()
        sc, a_opt = flattening_heuristic(best_a, n_iters=100,
                                        time_limit=min(15.0, final_deadline - time.monotonic()))
        try_update(sc, a_opt, "final flatten")
        print(f"    Final flatten: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
