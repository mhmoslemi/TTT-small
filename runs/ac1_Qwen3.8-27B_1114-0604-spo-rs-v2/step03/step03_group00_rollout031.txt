import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)

    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Handles edge cases."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_scaled.sum()) + scaled.max()
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of LSE w.r.t. a: convolve(a, p) shifted
    # dL/d a[j] = sum_k p[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = 2*a[k-j] (if valid)
    # So dL/d(a[j]) = 2 * sum_k p[k] * a[k-j] = 2 * (a * p)[j] (cross correlation)
    # (a * p)[j] = sum_k a[k] * p[j-k] ... need to be careful
    # Actually: conv[k] = sum_i a[i]*a[k-i], d(conv[k])/d(a[j]) = a[k-j] + a[k-j] = 2*a[k-j]
    # dL/d(a[j]) = sum_k p[k] * 2*a[k-j]
    # = 2 * sum_k p[k] * a[k-j]
    # Let m = k-j: = 2 * sum_m p[m+j] * a[m] = 2 * (palindromic cross-correlation)
    # More simply: define q = p reversed/shifted, then dL/d(a) = 2 * ifft(fft(a)*fft(q))
    # Actually the cleanest: dL/d(a[j]) = 2 * (p * a_rev)[n-1-j] where * is linear conv
    # Or: dL/d(a) = 2 * np.convolve(a, p, 'full')[n-1:2*n-1]  ... let me verify
    # conv_full[k] = sum_i a[i]*p[k-i]
    # dL/d(a[j]) = 2*sum_k p[k]*a[k-j], let m=k-j: 2*sum_m p[m+j]*a[m]
    # = 2*(p_shifted * a) where p_shifted[j] = p[j]... 
    # Let's use: dL/d(a[j]) = 2 * sum_{k} p[k] * a[k-j] (k-j in [0,n-1])
    # This is 2 * (cross_correlation of p and a at lag j)
    # cross_corr[j] = sum_k p[k] * a[k-j]
    # In FFT: cross_corr = ifft(fft(p) * conj(fft(a))) ... for circular
    # For linear, we need to be more careful. Let's just use direct convolution.
    # dL/d(a[j]) = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j]  (when k and k-j are both in [0,n-1])
    # = 2 * sum_{m=0}^{n-1} p[m+j] * a[m] (where p[m+j] = 0 if m+j >= len(p))
    # This is 2 * (p * a)[j] where (p*a) is the cross-correlation
    # cross_corr[j] = <p[j:], a[:len(p)-j]> ... 
    # Simplest: use FFT
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a, M2_fft)
    # We want: g[j] = sum_k p[k] * a[k-j] for j=0..n-1
    # This is the cross-correlation corr[j] = <p, shift_j(a)>
    # In FFT: corr = ifft(conj(fft(p)) * fft(a)) ... for circular correlation
    # For linear, we need zero-padding.
    # Actually: g[j] = sum_{k} p[k]*a[k-j] = (p * a)[2n-2-j] ... no
    # Let me just compute it directly using the convolution identity:
    # sum_k p[k]*a[k-j] = (p * a_rev)[n-1-j] where a_rev = a[::-1] and * is linear conv
    # Define b = a[::-1], c = np.convolve(p, b, 'full')
    # c[m] = sum_i p[i]*b[m-i] = sum_i p[i]*a[n-1-(m-i)]
    # c[n-1-j] = sum_i p[i]*a[n-1-n+1+j-i] = sum_i p[i]*a[j-i]
    # Hmm, that's sum_i p[i]*a[j-i] which is (p*a)[j] (the linear convolution)
    # That's not what we want. We want sum_k p[k]*a[k-j].
    # Let's use: sum_k p[k]*a[k-j] = sum_m p[m+j]*a[m] (sub m=k-j)
    # = (p_shifted * a)[0] where p_shifted[m] = p[m+j]
    # This is just the dot product <p[j:], a[:n-j]>
    # For efficiency, use the fact that this is the cross-correlation:
    # g[j] = sum_m a[m]*p[m+j] = cross_correlation(p, a, lag=j)
    # In FFT: g = ifft(conj(FFT(p, M2_fft)) * FFT(a, M2_fft))[:n]
    # Wait, let me verify: for circular correlation of length M:
    # corr[j] = sum_m x[m] * y[(m-j) mod M]
    # We want: g[j] = sum_m a[m] * p[m+j] (no wrap)
    # If M is large enough (>= 2n-1), then for j in [0,n-1] and m in [0,n-1]:
    #   m+j is in [j, n-1+j] which is < 2n-1 <= M
    #   So no wrap occurs, and the circular correlation equals the linear one.
    # corr[j] = sum_m a[m] * p[(m-j) mod M]
    # We want sum_m a[m]*p[m+j] = corr[(-j) mod M] = corr[M-j]
    # So g[j] = corr[M-j] = ifft(conj(FA) * FP)[M-j]
    # Alternatively, g[j] = ifft(FA * conj(FP))[j] ... let me just verify numerically
    # Actually the standard formula: cross_corr(x, y)[j] = sum_m x[m+y] * conj(y[m]) -- varies by convention
    # Let me just compute it the safe way:
    # g[j] = sum_{m=0}^{n-1} a[m] * p[m+j] (p[m+j]=0 if m+j >= 2n-1)
    # This equals (a * p_rev)[n-1-j] where p_rev = p[::-1] and * is full linear conv
    # (a * p_rev)[k] = sum_i a[i]*p_rev[k-i] = sum_i a[i]*p[2n-2-k+i]
    # (a * p_rev)[n-1-j] = sum_i a[i]*p[2n-2-n+1+j+i] = sum_i a[i]*p[n-1+j+i]
    # That doesn't look right either. Let me just do it with a direct approach using FFT:
    # g = np.zeros(n)
    # For efficiency with FFT:
    # g[j] = sum_m a[m] * p[m+j]
    # Consider the sequence a_padded of length M2_fft with a in first n positions
    # and p_padded of length M2_fft with p in first (2n-1) positions
    # The circular cross-correlation (ifft(FA * conj(FP))) gives:
    #   c[j] = sum_m a[m] * p[(m - j) mod M2_fft]
    # For m in [0,n-1] and j in [0,n-1]: m-j in [-(n-1), n-1]
    # If m-j >= 0: c[j] includes a[m]*p[m-j]
    # If m-j < 0: c[j] includes a[m]*p[m-j+M2_fft] which is 0 (since p is only defined for [0,2n-2])
    # So c[j] = sum_{m=j}^{n-1} a[m]*p[m-j] = sum_{k=0}^{n-1-j} a[k+j]*p[k]
    # That's not quite g[j] = sum_m a[m]*p[m+j]
    
    # Let me just use the straightforward approach: g[j] = sum_m a[m]*p[m+j]
    # = (a * p)[2n-2-j] ... no
    # Let's define h = p reversed: h[i] = p[2n-2-i]
    # (a * h)[k] = sum_i a[i]*h[k-i] = sum_i a[i]*p[2n-2-k+i]
    # (a*h)[2n-2-j] = sum_i a[i]*p[2n-2-2n+2+j+i] = sum_i a[i]*p[j+i]
    # That's sum_i a[i]*p[i+j] which IS g[j]! (with i = m)
    # So g = (a * p[::-1])[2n-2 : 2n-2+n-1 : -1] ... 
    # (a * h) has length 2n-1 + n-1 = 3n-2
    # g[j] = (a*h)[2n-2+j] ... wait let me recheck
    # (a*h)[k] = sum_i a[i]*h[k-i] where h[t] = p[2n-2-t]
    # (a*h)[k] = sum_i a[i]*p[2n-2-(k-i)] = sum_i a[i]*p[2n-2-k+i]
    # We want: g[j] = sum_i a[i]*p[i+j]
    # Set 2n-2-k = -j: k = 2n-2+j
    # (a*h)[2n-2+j] = sum_i a[i]*p[i+j] = g[j] ✓
    # So g[j] = (a * p[::-1])[2n-2+j] for j = 0, ..., n-1
    # (a * p[::-1]) has length (n-1) + (2n-2-1) + 1 = 3n-2
    # We need indices 2n-2 through 2n-2+n-1 = 3n-3, which is valid (max index is 3n-3)
    
    # For efficiency, use FFT:
    a_rev_for_grad = a  # We need a * p[::-1]
    p_rev = p[::-1]
    M3 = n + len(p_rev) - 1  # = n + 2n-2 - 1 = 3n-3
    M3_fft = 1 << (M3 - 1).bit_length()
    FA = np.fft.rfft(a, M3_fft)
    FP = np.fft.rfft(p_rev, M3_fft)
    conv_ap = np.fft.irfft(FA * FP, M3_fft)
    g = conv_ap[2 * n - 2:3 * n - 2]  # n elements: g[0]..g[n-1]
    
    grad_L = 2.0 * g
    # Full gradient of score = 2n*L/(S^2):
    # d(score)/d(a[j]) = 2n * (grad_L[j]/S^2 - 2*L*a[j]/S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L * a / S3)
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_fun(x):
        s, g = lse_score_grad(x, beta)
        return s, g
    
    res = minimize(
        obj_fun, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_exact(a, total_iters=5000, sc_lower=1.0):
    """Projected subgradient on exact max-conv objective with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(conv) at a:
        # d(max conv)/d(a[j]) = 2*a[k_star - j] if 0 <= k_star - j < n, else a[k_star-j] (one side)
        sg_M = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                sg_M[j] = 2.0 * a[k2]
            else:
                sg_M[j] = 0.0
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Adaptive step: (current - lower_bound) / ||grad||^2
        step = (sc - sc_lower) / (gn * gn + 1e-30)
        step = min(step, 2.0 / (it + 1))
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a


def peak_coordinate_descent(a, n_sweeps=5, n_top=100):
    """Coordinate descent targeting peak contributors with exact 1D optimization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        
        # Find top contributors to the peak
        contrib = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib[j] = 2.0 * a[k2]
        
        top_count = min(n_top, n)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        
        for j in top_idx:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            # Try several scales
            best_val = a_old_j
            best_local = best_sc
            for factor in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3]:
                try_val = a_old_j * factor
                if abs(try_val - a_old_j) < 1e-12:
                    continue
                a[j] = try_val
                S = a.sum()
                if S < 1e-12:
                    a[j] = a_old_j
                    continue
                a_norm = a / S
                sc = compute_score(a_norm)
                if sc < best_local:
                    best_local = sc
                    best_val = try_val
                a[j] = a_old_j
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def error_reduction_fft(a, n_iters=200, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    # U-shaped (counteracts triangular convolution envelope)
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(0.05 + 0.95 * (2.0 * x - 1.0) ** 4)
    inits.append(0.1 + 0.9 * np.exp(5.0 * (2.0 * x - 1.0) ** 2))
    # Cosine-based
    inits.append(np.abs(np.cos(np.pi * x)) + 0.01)
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * x))
    # Bimodal
    inits.append(np.exp(-20.0 * (x - 0.15) ** 2) + np.exp(-20.0 * (x - 0.85) ** 2) + 0.01)
    inits.append(np.exp(-10.0 * (x - 0.2) ** 2) + np.exp(-10.0 * (x - 0.8) ** 2) + 0.05)
    # Random structured
    nb = 16
    per = max(1, n // nb)
    vals = 0.1 + 0.9 * rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    inits.append(a[:n].copy())
    # Pure random
    inits.append(np.clip(0.3 + 0.7 * rng.random(n), 0.01, 1000.0))
    
    for a_init in inits:
        a_init = np.clip(a_init, 0.0, 1000.0)
        s = a_init.sum()
        if s > 1e-6:
            a_init = a_init / s
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses multi-phase: error reduction init, LSE-annealed L-BFGS, subgradient, coord descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) / 1024

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine target length
    if parent is not None and len(parent) >= 1024:
        n_target = len(parent)
    else:
        n_target = 8192
    
    print(f"Target length: {n_target}, parent available: {parent is not None}")
    print(f"Time budget: {budget_s}s, deadline at t={deadline - t_start:.0f}s")

    # ============================================================
    # Phase 1: Multi-init error reduction (25% time)
    # ============================================================
    phase1_end = t_start + 0.25 * (float(budget_s) - 20.0)
    print(f"\n{'='*50}")
    print(f"Phase 1: Error Reduction ({n_target} samples)")
    print(f"{'='*50}")
    
    inits = build_initializations(n_target, rng, parent)
    n_inits = min(len(inits), max(4, int((phase1_end - t_start) / 8.0)))
    
    for i in range(n_inits):
        if time.monotonic() > phase1_end:
            break
        a0 = inits[i].copy()
        a_er = error_reduction_fft(a0, n_iters=150, alpha=0.7)
        sc = compute_score(a_er)
        report(sc, a_er, f"ER init={i}")
        
        # Second pass with different alpha
        a_er2 = error_reduction_fft(a_er, n_iters=100, alpha=0.85)
        sc2 = compute_score(a_er2)
        report(sc2, a_er2, f"ER2 init={i}")

    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ============================================================
    # Phase 2: LSE-annealed L-BFGS (50% time)
    # ============================================================
    phase2_end = t_start + 0.75 * (float(budget_s) - 20.0)
    print(f"\n{'='*50}")
    print(f"Phase 2: LSE Annealed L-BFGS")
    print(f"{'='*50}")
    
    # Adaptive beta schedule: geometric progression
    betas = []
    b = 20.0
    while b <= 50000.0:
        betas.append(b)
        b *= 1.4
    # Ensure we have a good schedule
    if len(betas) < 10:
        betas = [20, 50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600, 50000]
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        # Allocate more iterations to lower betas (easier), fewer to higher
        maxiter = max(500, int(3000 * (1.0 / (1.0 + idx * 0.3))))
        maxfun = maxiter * 5
        
        t_beta = time.monotonic()
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        elapsed_beta = time.monotonic() - t_beta
        report(sc, a_opt, f"LBFGS beta={beta:.0f} ({elapsed_beta:.1f}s)")
        
        # If we made significant progress, do an extra pass at same beta
        if sc < best_score * 0.999 and time.monotonic() < phase2_end:
            sc2, a_opt2 = run_lbfgs_lse(a_opt, beta, maxiter=maxiter//2, maxfun=maxfun//2)
            report(sc2, a_opt2, f"LBFGS beta={beta:.0f} extra")
    
    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient + Coordinate Descent (15% time)
    # ============================================================
    phase3_end = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n{'='*50}")
    print(f"Phase 3: Subgradient + Coord Descent")
    print(f"{'='*50}")
    
    # Subgradient with multiple restarts
    for restart in range(3):
        if time.monotonic() > phase3_end:
            break
        if restart == 0:
            a_start = best_a.copy()
        else:
            # Perturb
            scale = 0.03 * (restart ** 0.5)
            a_pert = best_a * (1.0 + scale * rng.standard_normal(len(best_a)))
            a_start = np.clip(a_pert, 0.0, 1000.0)
            s = a_start.sum()
            if s > 1e-6:
                a_start = a_start / s
        
        a_sg = subgradient_exact(a_start, total_iters=3000, sc_lower=1.0)
        sc = compute_score(a_sg)
        report(sc, a_sg, f"Subgrad restart={restart}")
    
    # Coordinate descent
    if time.monotonic() < phase3_end:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=5, n_top=150)
        sc = compute_score(a_cd)
        report(sc, a_cd, "CoordDescent")
        
        # L-BFGS from CD result
        if time.monotonic() < phase3_end:
            sc2, a_opt = run_lbfgs_lse(a_cd, 2000, maxiter=2000, maxfun=20000)
            report(sc2, a_opt, "CD+LBFGS")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish (10% time)
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*50}")
    
    strategies = [
        ("ER final", lambda: error_reduction_fft(best_a, 100, 0.9)),
        ("LBFGS high beta", lambda: run_lbfgs_lse(best_a, 10000, 2000, 20000)[1]),
        ("Perturb+ER", lambda: error_reduction_fft(
            np.clip(best_a * (1 + 0.02 * rng.standard_normal(len(best_a))), 0, 1000), 100, 0.85)),
        ("Subgrad short", lambda: subgradient_exact(best_a, 2000, 1.0)),
    ]
    
    for label, func in strategies:
        if time.monotonic() > deadline:
            break
        try:
            a_try = func()
            a_try = np.clip(a_try, 0.0, 1000.0)
            s = a_try.sum()
            if s < 1e-6:
                continue
            a_try = a_try / s
            sc = compute_score(a_try)
            report(sc, a_try, label
        )
        except Exception as e:
            print(f"  Strategy {label} failed: {e}")
    
    # Final L-BFGS at very high beta
    if time.monotonic() < deadline - 10:
        try:
            sc, a_opt = run_lbfgs_lse(best_a, 50000, 1000, 10000)
            report(sc, a_opt, "Final LBFGS beta=50000")
        except Exception:
            pass

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
