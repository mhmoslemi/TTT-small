"""
Step function optimization using spectral equalization, LSE continuation,
Nesterov-accelerated subgradient, and active-set quadratic penalty.
Aims to improve upon 1.5093 by using diverse optimization landscapes.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    # Gradient via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad

def lse_obj(a, beta):
    return lse_score_grad(a, beta)

def quad_penalty_score_grad(a, t, rho, active_mask):
    """
    Quadratic penalty score and gradient.
    Minimizes t + rho * sum_{k in active} (c_k - t)^2 / (2 S^2)^2 type objective.
    Actually, let's use: 2n/S^2 * (t + rho * sum_k active_k * (c_k - t)_+^2)
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    S2 = S * S
    if S2 < 1e-24:
        return 1e10, np.zeros(n)
    c_act = c[active_mask]
    excess = np.maximum(c_act - t, 0)
    penalty = float(np.sum(excess * excess))
    score = 2.0 * n * (t + rho * penalty) / S2
    # Gradient
    # d/dt: 2n/S2 * (1 - 2*rho*sum(excess)) ... but t is not a variable here
    # We treat t as fixed (computed from current a).
    # d(score)/da_j = 2n/S2 * rho * sum_k active * 2*(c_k-t)_+ * d(c_k)/da_j
    #                - 4n*(t + rho*penalty)/(S^3) * a_j
    w = np.zeros(2 * n - 1)
    w[active_mask] = 2.0 * rho * excess
    # d(c_k)/da_j = 2 * a_{k-j} for valid k-j
    # So sum_k w_k * 2 * a_{k-j} = conv of w with a (reversed)
    rev_a = a[::-1]
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    RA = np.fft.rfft(rev_a, M_fft)
    W = np.fft.rfft(w, M_fft)
    conv_w = np.fft.irfft(RA * W, M_fft)[n - 1:2 * n - 1]
    d_penalty_da = 2.0 * conv_w
    S3 = S2 * S
    grad = 2.0 * n * (d_penalty_da / S2 - 2.0 * (t + rho * penalty) / S3)
    return score, grad

def quad_penalty_obj(a, t, rho, active_mask):
    return quad_penalty_score_grad(a, t, rho, active_mask)

def spectral_equalize(a, alpha=0.3, iters=50):
    """
    Spectral equalization: iteratively flatten |FFT(a)|^2.
    Works in frequency domain, projects back to non-negative time domain.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    m = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(iters):
        F = np.fft.rfft(a, m)
        mag = np.abs(F)
        phase = np.angle(F)
        # Equalize magnitude
        mean_mag_sq = float(np.mean(mag ** 2))
        if mean_mag_sq < 1e-30:
            break
        # Partial equalization
        new_mag = mag * (mean_mag_sq / (mag ** 2 + 1e-30)) ** alpha
        new_mag = np.maximum(new_mag, 1e-15)
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, m)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_sc, best_a
