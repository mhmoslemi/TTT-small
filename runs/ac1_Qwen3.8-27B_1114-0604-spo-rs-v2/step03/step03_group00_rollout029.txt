"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: adaptive frequency-domain error reduction, L-BFGS with LSE smoothing,
dual-peak coordinate descent, mirror descent, and multi-length exploration.
Key improvements: better gradient computation, adaptive beta schedule,
dual-peak targeting, and more diverse initialization.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Shift for numerical stability
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] = 2 * (rev(a) * p)[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction_v2(a, n_iters=200, alpha_schedule=None):
    """
    Improved error reduction in frequency domain.
    Uses adaptive threshold and soft flattening.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    if alpha_schedule is None:
        alpha_schedule = np.linspace(0.4, 0.9, n_iters)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        # Soft flattening: reduce excess proportionally
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.4 + 0.4 * (it / max(1, n_iters // 3))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def dual_peak_coord_descent(a, n_sweeps=5, n_top=60):
    """
    Coordinate descent targeting top peaks simultaneously.
    Reduces top-K peak contributors and redistributes to low-contributors.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top peaks
        top_k = min(3, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old_j = a[j]
                candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.85, 1.15]]
                candidates.append(0.0)
                best_val = a_old_j
                best_local_sc = best_sc if not (a is best_a) else compute_score(a)
                a[j] = 0  # save/restore
                saved = a_old_j
                for cv in candidates:
                    cv = max(0.0, min(1000.0, cv))
                    if abs(cv - saved) < 1e-10:
                        continue
                    a[j] = cv
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = cv
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish_v2(a, total_iters=3000):
    """Projected subgradient with adaptive step and sum-preserving update."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient: grad[i] = 4n/S^2 * (a[k*-i] - M/S) for 0<=k*-i<n
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Scale to get proper subgradient of the full objective
        grad = (4.0 * n) * (sg / (S * S) - M / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        # Adaptive step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        step = max(step, 1e-6)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def mirror_descent(a, n_iters=2000, eta0=0.1):
    """
    Mirror descent (entropic) on the simplex for minimizing max(conv).
    Scale-invariant, so we work with sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        # Gradient of max(conv) wrt a[i] = 2*a[k*-i] (if 0<=k*-i<n)
        grad_max = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad_max[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Mirror descent: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        eta = eta0 / (1.0 + 0.1 * it)
        log_a = np.log(a) - eta * grad_max
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_initializations(n, rng, parent):

        if len(a) < n:

    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            scale = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak reduction
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 50)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.03
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.03 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Error reduction restart with perturbation
            a_pert = error_reduction_v2(
                best_a * (1 + 0.03 * rng.standard_normal(n_best)),
                n_iters=80
            )
        else:
            # Block rescale
            a_pert = best_a.copy()
            n_blocks = 3 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.7 + 0.6 * rng.random()
                a_pert[s:e] *= factor
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=4000, maxfun=80000)
        report(sc, a_opt, f"local#{trial} s={strategy}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 4: Polish ---")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # Mirror descent polish
        a_md = mirror_descent(best_a, n_iters=int(min(2000, remaining / 0.1)))
        sc = compute_score(a_md)
        report(sc, a_md, "mirror")

    if remaining > 30:
        # Subgradient polish
        a_sp = subgradient_polish_v2(best_a, total_iters=int(min(3000, remaining / 0.05)))
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    if remaining > 20:
        # High-beta L-BFGS
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS b=5000")

    if remaining > 15:
        # Final error reduction
        a_er = error_reduction_v2(best_a, n_iters=150)
        sc = compute_score(a_er)
        report(sc, a_er, "final ER")

    if remaining > 15 and best_score > 1.505:
        # One more round of L-BFGS from ER result
        sc, a_opt = run_lbfgs(a_er, 3000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "ER+LBFGS b=3000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
