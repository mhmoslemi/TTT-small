Optimization of step function heights to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed L-BFGS with continuation, iterative equalization, and subgradient polish.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Top-level helper functions (no closures)

def fast_self_conv(a):
    """Compute linear self-convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_exact_score(a):
    """Compute the exact evaluation score."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))

def lse_objective_and_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient.
    Objective: 2n * LSE_max(conv(a,a), beta) / (sum(a))^2
    """
    n = len(a)
    c = fast_self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    # For numerical stability
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    p = exp_scaled / sum_exp
    L = c_max + (max_scaled + np.log(sum_exp)) / beta
    
    # Gradient of L w.r.t. a
    # dL/da[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation of p and a)
    L_fft = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    cross = np.fft.irfft(P * np.conj(A), L_fft)
    dL_da = 2.0 * cross[:n]
    
    # Objective = 2n * L / S^2
    obj = 2.0 * n * L / (S * S)
    # Gradient
    S2 = S * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / (S2 * S))
    
    return obj, grad

def run_lbfgs(a0, beta, time_limit, maxfun=None):
    """Run L-BFGS-B to minimize LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if np.sum(a0) < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    t_start = time.monotonic()
    if maxfun is None:
        maxfun = int(100 + 200 * time_limit)
    
    def fun(x):
        val, g = lse_objective_and_grad(x, beta)
        return val, g
    
    options = {
        'ftol': 1e-18,
        'gtol': 1e-14,
        'maxiter': max(100, int(50 * time_limit)),
        'maxfun': maxfun,
        'maxls': 50
    }
    
    try:
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 1:
            return compute_exact_score(a0), a0
        res = minimize(fun, a0, method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    
    return compute_exact_score(a_out), a_out

def equalization_step(a, n_steps=50, alpha=0.3, eps=0.01, deadline=None):
    """
    Iterative equalization: make the active convolution peaks equal
    by adjusting a multiplicatively.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S_orig = float(np.sum(a))

    L_fft = 1 << (3 * n - 2).bit_length()
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        M = float(np.max(c))
        if M < 1e-15:

        n_active = int(np.sum(active))
        if n_active == 0:
            active[int(np.argmax(c))] = True
        
        # Compute pressure h[j] = sum_{k in active} p[k] * a[k-j]

        p = np.zeros(2 * n - 1)

        P = np.fft.rfft(p, L_fft)
        A = np.fft.rfft(a, L_fft)
        cross = np.fft.irfft(P * np.conj(A), L_fft)
        h = 2.0 * cross[:n]
        
        # Ensure positive
        h = np.maximum(h, 1e-15)
        h_mean = float(np.mean(h[h > 1e-10])) if np.any(h > 1e-10) else float(np.mean(h))
        
        # Multiplicative update
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S_new = float(np.sum(a))
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S_orig / S_new)
        
        sc = compute_exact_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_sc, best_a

def subgradient_polish(a, n_iters=200, deadline=None):

    n = len(a)

    best_a = a.copy()
    best_sc = compute_exact_score(a)

        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        S = float(np.sum(a))
        if S < 1e-12:
            break
        M = float(np.max(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max at a
        k_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            # d/d a[j] (a*a)[k_star] = 2 * a[k_star - j]
            idx = np.arange(k_star - j_end, k_star - j_start + 1)
            sg[j_start:j_end+1] = 2.0 * a[idx]
        
        S2 = S * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / (S2 * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size (using lower bound of 1.0)
        step = max(0.0, sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize to maintain sum
        S_new = float(np.sum(a))
        if S_new > 1e-12:
            a = a * (S_orig / S_new)
        else:
            a[:] = 0.5
    
    return best_sc, best_a
