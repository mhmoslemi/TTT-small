def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(c.max()) / (S * S)

def adam_continuation(a_init, betas, lrs, n_steps_each, time_limit, t_start):
    """Adam with continuation over beta values."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.99
    eps = 1e-8
    t_global = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for beta, lr in zip(betas, lrs):
        for step in range(n_steps_each):
            t_global += 1
            if time.monotonic() - t_start > time_limit:
                return best_score, best_a
            
            score, grad = lse_smooth_score_grad(a, beta)
            
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            m_hat = m / (1 - b1**t_global)
            v_hat = v / (1 - b2**t_global)
            
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e6)
            
            # Normalize to maintain scale
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def peak_shave(a, n_iterations=100, time_limit=None, t_start=None):

        S = a.sum()
        score = 2.0 * n * c.max() / (S * S)

            best_a = a.copy()

        # Find coordinates contributing most to C[k_star]
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            continue

        a[top_global] -= reduction / len(top_global)
        a[top_global] = np.clip(a[top_global], 0.0, None)

        removed = float(a[top_global].sum() + reduction - a[top_global].sum())  # wasn't removed, clipped
        # Actually let's redo: compute how much was actually clipped
        # Redo:
        old_vals = a[top_global].copy()
        # Already clipped above, so removed = old_vals.sum() - a[top_global].sum()
        # But we already modified a. Let me restructure.
        
        # Better approach:
        a = best_a.copy()  # Reset
        # Reduce top contributors by a fraction
        frac = 0.03
        reduce_amount = frac * a[top_global].sum()
        a[top_global] *= (1 - frac)
        
        # Distribute to coordinates NOT in the contributing set
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[js:je+1] = False
        if non_contrib_mask.sum() > 0:
            a[non_contrib_mask] += reduce_amount / non_contrib_mask.sum()
        
        # Check improvement

def generate_tilted_cosine(n, tilt=0.3, width=1.0):
    """Generate a tilted raised-cosine window."""
    x = np.linspace(0, 1, n)
    # Raised cosine window
    w = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Tilt
    t = 2 * x - 1  # ranges from -1 to 1
    a = w * (1 + tilt * t)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a = w.copy()
    return a

def generate_structured(n, rng):
    """Generate various structured initial sequences."""
    strategy = rng.integers(0, 5)
    if strategy == 0:
        # Tilted cosine
        tilt = rng.uniform(0.1, 0.8)
        a = generate_tilted_cosine(n, tilt=tilt)
    elif strategy == 1:
        # Exponential ramp
        rate = rng.uniform(0.5, 3.0)
        x = np.arange(n) / n
        a = np.exp(rate * (x - 0.5))
    elif strategy == 2:
        # Two-lobe
        center1 = rng.uniform(0.2, 0.4)
        center2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(0.05, 0.2)
        x = np.arange(n) / n
        a = np.exp(-((x - center1)/w)**2) + np.exp(-((x - center2)/w)**2)
    elif strategy == 3:
        # Linear ramp with flat top
        ramp_frac = rng.uniform(0.2, 0.5)
        x = np.arange(n) / n
        a = np.minimum(1.0, x / ramp_frac) * np.minimum(1.0, (1-x) / ramp_frac + 0.5)
    else:
        # Random smooth
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2*np.pi*freq*x + phase)
        a = np.clip(a, 0.01, None)
    
    return a / a.sum()  # Normalize
