Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def active_set_score_grad(a, active_lags, beta):

    Score and gradient using LSE over a subset of lags (active set).
    active_lags: array of lag indices to include
    beta: smoothing parameter

    n = len(a)

    L = c_max + log_sum / beta

    p = exp_scaled / exp_scaled.sum()

def get_active_lags(c, top_frac=0.05, min_count=5):
    """Get the indices of lags that are 'active' (close to max)."""
    c_max = c.max()
    threshold = c_max * (1.0 - top_frac)
    active = np.where(c >= threshold)[0]
    if len(active) < min_count:
        # Include at least min_count largest
        sorted_idx = np.argsort(c)[::-1][:min_count]
        active = sorted_idx
    return active

def run_lbfgs_wrapper(a0, objective_fn, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B with a given objective function."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        objective_fn, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

def subgradient_polish(a, n_iters, time_budget):
    """Exact subgradient descent with adaptive steps."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end+1]
            sg_M[j_start:j_end+1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Occasionally add small noise to escape flat regions
        if it % 50 == 0 and it > 0:
            noise_scale = 0.01 * (1.0 - it / n_iters)
            a = a * (1.0 + noise_scale * np.random.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive active-set LSE smoothing, continuation methods, multi-length exploration,
and hybrid polishing (subgradient + targeted peak reduction + high-precision LSE).
Key innovations: active-set tracking for focused optimization, scale-invariance
exploitation, and aggressive final polishing with very high beta LSE.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global config for objective function (avoids lambdas/closures)
_obj_config = {'beta': 100.0, 'active_lags': None}


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _lse_obj(x):
    """Objective function using global config for beta and active lags."""
    beta = _obj_config['beta']
    active = _obj_config.get('active_lags', None)
    if active is not None:
        return _active_set_score_grad(x, active, beta)
    else:
        return _full_lse_score_grad(x, beta)


def _full_lse_score_grad(a, beta):
    """Full LSE-smoothed score and gradient over all lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:3 * n - 2]
    g = conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def _active_set_score_grad(a, active_lags, beta):
    """LSE score and gradient restricted to active lags."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_active = c[active_lags]
    c_max = float(c_active.max())
    scaled = beta * (c_active - c_max)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    # Build weight vector over all lags
    q = np.zeros(2 * n - 1)
    q[active_lags] = p

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    Q = np.fft.rfft(q, M2_fft)
    conv_res = np.fft.irfft(RA * Q, M2_fft)[:3 * n - 2]
    g = conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def get_active_lags(c, top_frac=0.05, min_count=10, max_frac=0.3):
    """Identify active lags (those close to the maximum)."""
    c_max = float(c.max())
    if c_max < 1e-15:
        return np.arange(len(c))
    threshold = c_max * (1.0 - top_frac)
    active = np.where(c >= threshold)[0]
    # Ensure minimum count
    if len(active) < min_count:
        sorted_idx = np.argsort(c)[::-1][:min_count]
        active = sorted_idx
    # Cap maximum count
    max_count = max(min_count, int(len(c) * max_frac))
    if len(active) > max_count:
        sorted_idx = np.argsort(c)[::-1][:max_count]
        active = sorted_idx
    return active


def run_lbfgs(a0, maxiter=5000, maxfun=10000):
    """Run L-BFGS-B optimization with current _obj_config."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        _lse_obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    return a_out

    for beta in betas:

        if elapsed > total_time:

        remaining = total_time - elapsed

        else:

        _obj_config['beta'] = float(beta)

            sc = compute_score(a_opt)

                best_a = a_opt.copy()
            a = a_opt

def subgradient_polish(a, n_iters, time_budget):
    """Exact subgradient descent with Polyak step size and periodic restart."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(c) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step: Polyak-like with cap
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        step = max(step, 0.001)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)

        # Periodic small perturbation to escape flat regions
        if it % 100 == 99:
            noise_scale = 0.005 * max(0.0, 1.0 - it / float(n_iters))
            if noise_scale > 1e-6:
                a = a * (1.0 + noise_scale * rng.standard_normal(n))
                a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def multi_peak_subgradient(a, n_iters, time_budget):
    """Subgradient using average of top-k peak subgradients."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    k_peaks = 3

    for it in range(n_iters):
        if time.monotonic() - t_start > time_budget:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        # Find top-k peaks
        top_k_indices = np.argsort(c)[::-1][:k_peaks]
        c_max = float(c[top_k_indices[0]])
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Average subgradient over top-k
        sg_avg = np.zeros(n)
        for k_star in top_k_indices:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg_avg[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg_avg /= k_peaks

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_avg / S2 - 2.0 * c_max / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.2)
        step = max(step, 0.001)

        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def targeted_perturb_search(a, n_trials, time_budget, rng):
    """
    Targeted perturbations focused on reducing the peak convolution lag.
    Tries multiple strategies and keeps improvements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t_start > time_budget:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        strategy = trial % 5

        if strategy == 0:
            # Reduce top contributors at peak lag, redistribute
            if j_end >= j_start:
                contrib = a[j_start:j_end + 1][::-1].copy()
                n_reduce = max(5, n // 80)
                top_local = np.argsort(contrib)[::-1][:n_reduce]
                a_new = a.copy()
                total_red = 0.0
                for li in top_local:
                    gi = j_end - li
                    red = a_new[gi] * (0.02 + 0.04 * rng.random())
                    a_new[gi] -= red
                    a_new[gi] = max(a_new[gi], 0.0)
                    total_red += red
                # Redistribute to low-contribution area
                n_add = max(10, n // 40)
                add_idx = rng.integers(0, n, size=n_add)
                a_new[add_idx] += total_red / n_add
        elif strategy == 1:
            # Frequency-domain chirp perturbation
            x = np.arange(n, dtype=np.float64) / n
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            a_new = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif strategy == 2:
            # Mirror swap
            mid = n // 2
            block = max(10, n // 30)
            a_new = a.copy()
            # Swap a block with its mirror
            src = rng.integers(0, mid - block)
            a_new[mid - src - block:mid - src] = a[src:src + block].copy()
            a_new[src:src + block] = a[mid - src - block:mid - src].copy()
        elif strategy == 3:
            # Sparse random perturbation
            a_new = a.copy()
            n_sparse = max(20, n // 30)
            idx = rng.integers(0, n, size=n_sparse)
            a_new[idx] *= (1.0 + 0.1 * rng.standard_normal(n_sparse))
        else:
            # Blend with a structured pattern
            x = np.arange(n) / n
            struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 10 * x + rng.random() * 6.28)
            alpha = 0.05 + 0.1 * rng.random()
            a_new = (1 - alpha) * a + alpha * struct * (a.sum() / struct.sum())

        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new  # Continue from improvement
        elif rng.random() < 0.3:
            a = a_new  # Occasionally accept for exploration

    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    inits.append(0.3 + 0.7 * np.sin(np.pi * x) ** 0.5)

    # Multi-frequency
    for seed_f in range(3):
        n_freqs = 2 + seed_f
        sig = np.zeros(n)
        for fi in range(n_freqs):
            f = 1.0 + (fi + 1) * (1.0 + rng.random() * 3.0)
            p = 2.0 * np.pi * rng.random()
            amp = 0.05 + 0.1 * rng.random()
            sig += amp * np.cos(2.0 * np.pi * f * x + p)
        inits.append(np.clip(0.5 + sig, 0.01, 1000.0))

    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))

    # Block structure
    for nb in (16, 32, 64):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].astype(np.float64).copy())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses multi-phase strategy:
    Phase 1: Multi-start continuation LSE (35% time)
    Phase 2: Active-set deep polishing (30% time)
    Phase 3: Diverse local search (20% time)
    Phase 4: Final high-precision refinement (15% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 25.0
    usable_time = total_budget - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, "
                  f"t={time.monotonic() - t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
        for dl in [-1024, -512, 512, 1024, 2048]:
            cand = parent.size + dl
            if 2048 <= cand <= 32768:
                lengths.append(cand)
    for n in [4096, 5120, 6962, 8192, 10240, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit based on time: roughly 120s per length in phase 1
    max_lengths = max(3, min(6, int(usable_time * 0.35 / 120)))
    lengths = lengths[:max_lengths]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (len=' + str(parent.size) + ')' if parent is not None else 'no'}")
    print(f"Total budget: {budget_s}s, usable: {usable_time:.0f}s")

    # =========================================================================
    # Phase 1: Multi-start continuation optimization (35%)
    # =========================================================================
    p1_time = 0.35 * usable_time
    p1_deadline = t_start + p1_time
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-start continuation ({p1_time:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\n  n={n}, remaining={remaining:.0f}s")

        inits = build_inits(n, rng, parent)
        # Budget per init
        per_init_time = remaining / max(1, len(inits))
        per_init_time = min(per_init_time, 60.0)  # Cap at 60s per init
        n_inits = max(2, min(len(inits), int(remaining / 30)))

        for i, a0 in enumerate(inits[:n_inits]):
            if time.monotonic() > p1_deadline:
                break
            it_time = min(per_init_time, p1_deadline - time.monotonic())
            if it_time < 10:
                break
            sc, a_opt = continuation_optimize(
                a0, it_time,
                beta_start=10, beta_end=5000,
                n_stages=max(3, int(it_time / 15)),
                use_active=True, active_frac=0.05
            )
            report(sc, a_opt, f"P1 n={n} i={i}")

    print(f"\nAfter Phase 1: best = {best_score:.8f} (n={len(best_a)})")

    # =========================================================================
    # Phase 2: Active-set deep polishing (30%)
    # =========================================================================
    p2_time = 0.30 * usable_time
    p2_deadline = t_start + p1_time + p2_time
    print(f"\n{'='*60}")
    print(f"Phase 2: Active-set deep polishing ({p2_time:.0f}s)")
    print(f"{'='*60}")

    # Polish the best solution deeply
    time_per_polish = p2_time / 4.0

    # Deep polish 1: from best
    sc, a_pol1 = continuation_optimize(
        best_a, time_per_polish,
        beta_start=100, beta_end=100000,
        n_stages=max(4, int(time_per_polish / 10)),
        use_active=True, active_frac=0.03
    )
    report(sc, a_pol1, "P2 deep1")

    # Deep polish 2: from perturbed best
    a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(len(best_a)))
    a_pert = np.clip(a_pert, 0, 1000)
    sc, a_pol2 = continuation_optimize(
        a_pert, time_per_polish,
        beta_start=100, beta_end=100000,
        n_stages=max(4, int(time_per_polish / 10)),
        use_active=True, active_frac=0.03
    )
    report(sc, a_pol2, "P2 deep2")

    # Deep polish 3: from best with different active set fraction
    sc, a_pol3 = continuation_optimize(
        best_a, time_per_polish,
        beta_start=50, beta_end=50000,
        n_stages=max(4, int(time_per_polish / 10)),
        use_active=True, active_frac=0.10
    )
    report(sc, a_pol3, "P2 deep3")

    # Deep polish 4: from subgradient result
    sc_sg, a_sg = subgradient_polish(best_a, 3000, min(30.0, time_per_polish))
    report(sc_sg, a_sg, "P2 subgrad")
    sc, a_pol4 = continuation_optimize(
        a_sg, time_per_polish * 0.5,
        beta_start=500, beta_end=50000,
        n_stages=3,
        use_active=True, active_frac=0.03
    )
    report(sc, a_pol4, "P2 deep4")

    print(f"\nAfter Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # =========================================================================
    # Phase 3: Diverse local search (20%)
    # =========================================================================
    p3_time = 0.20 * usable_time
    p3_deadline = t_start + p1_time + p2_time + p3_time
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse local search ({p3_time:.0f}s)")
    print(f"{'='*60}")

    # Targeted perturbation search
    tpr_time = p3_time * 0.4
    sc, a_tpr = targeted_perturb_search(best_a, 50, tpr_time, rng)
    report(sc, a_tpr, "P3 target")

    # Multi-peak subgradient
    mp_time = p3_time * 0.2
    sc, a_mp = multi_peak_subgradient(best_a, 3000, mp_time)
    report(sc, a_mp, "P3 multipeak")

    # Polish the multi-peak result
    if time.monotonic() < p3_deadline:
        rem = p3_deadline - time.monotonic()
        sc, a_mp_pol = continuation_optimize(
            a_mp, min(rem, p3_time * 0.2),
            beta_start=500, beta_end=100000,
            n_stages=3, use_active=True, active_frac=0.03
        )
        report(sc, a_mp_pol, "P3 mp_pol")

    # Try slightly different length
    if time.monotonic() < p3_deadline - 30:
        alt_n = len(best_a) + 1024
        if alt_n <= 32768:
            print(f"  Trying alt length n={alt_n}")
            a_alt = np.ones(alt_n) * 0.5
            idx = np.linspace(0, len(best_a) - 1, alt_n).astype(int)
            a_alt = best_a[idx].copy()
            rem = min(60.0, p3_deadline - time.monotonic())
            if rem > 20:
                sc, a_alt_opt = continuation_optimize(
                    a_alt, rem,
                    beta_start=50, beta_end=20000,
                    n_stages=4, use_active=True, active_frac=0.05
                )
                report(sc, a_alt_opt, f"P3 alt_n={alt_n}")

    print(f"\nAfter Phase 3: best = {best_score:.8f} (n={len(best_a)})")

    # =========================================================================
    # Phase 4: Final high-precision refinement (15%)
    # =========================================================================
    p4_time = 0.15 * usable_time
    print(f"\n{'='*60}")
    print(f"Phase 4: Final refinement ({p4_time:.0f}s)")
    print(f"{'='*60}")

    # Very high beta LSE (approaches exact max)
    stage_time = p4_time / 4.0

    # Stage 1: High beta LSE
    if time.monotonic() < deadline - 30:
        sc, a_f1 = continuation_optimize(
            best_a, stage_time,
            beta_start=5000, beta_end=500000,
            n_stages=3, use_active=True, active_frac=0.01
        )
        report(sc, a_f1, "P4 high_beta")

    # Stage 2: Subgradient with very small steps
    if time.monotonic() < deadline - 25:
        rem = min(stage_time, deadline - time.monotonic() - 25)
        sc, a_f2 = subgradient_polish(best_a, 5000, rem)
        report(sc, a_f2, "P4 subgrad")

    # Stage 3: One more deep LSE from best
    if time.monotonic() < deadline - 10:
        rem = min(stage_time, deadline - time.monotonic() - 10)
        sc, a_f3 = continuation_optimize(
            best_a, rem,
            beta_start=10000, beta_end=1000000,
            n_stages=2, use_active=True, active_frac=0.01
        )
        report(sc, a_f3, "P4 final_lse")

    # Stage 4: Final subgradient micro-steps
    if time.monotonic() < deadline - 3:
        rem = max(5.0, deadline - time.monotonic() - 3)
        sc, a_f4 = subgradient_polish(best_a, 2000, rem)
        report(sc, a_f4, "P4 micro_sg")

    # =========================================================================
    # Final output
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 1024

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
