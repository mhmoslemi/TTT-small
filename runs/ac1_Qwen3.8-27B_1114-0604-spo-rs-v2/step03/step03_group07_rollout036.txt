    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)

    v = np.zeros(n)  # Second moment
    t = 0
    
    best_sc = compute_score(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    # Adaptive learning rate: start with lr, decay over iterations

            break
        t += 1
        sc, grad = lse_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate schedule: cosine decay
        progress = it / max_iter
        lr_t = lr * 0.5 * (1 + math.cos(math.pi * progress))
        
        a -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)
    
    # Final exact score
    exact_sc = compute_score(best_a)
    return exact_sc, best_a

def spectral_init(n, rng):
    """Initialize using spectral method: choose target power spectrum and invert."""
    # We want |A(f)|^2 to be as flat as possible
    # A(f) = DFT(a) for the zero-padded a
    # If we want a flat power spectrum, we can use a random phase
    # and take the inverse DFT, then clip to non-negative
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    # Target: flat power spectrum with some structure
    # Use a random phase
    A = np.zeros(M_fft // 2 + 1, dtype=np.complex128)
    freqs = np.arange(len(A))
    
    # Low-pass filtered flat spectrum
    bandwidth = 0.3 + 0.4 * rng.random()
    mask = freqs < len(A) * bandwidth
    A[mask] = np.exp(2j * np.pi * rng.random(mask.sum()))
    # Normalize
    A[0] = np.abs(A[0])  # DC component non-negative
    
    a = np.real(np.fft.irfft(A * np.sqrt(len(A)), M_fft))[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a

def build_diverse_inits(n, rng):
    """Build diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / (n - 1)
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Linear ramps (anti-palindromic)
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))
    
    # Quadratic
    inits.append(0.01 + 0.99 * x**2)
    inits.append(0.01 + 0.99 * (1-x)**2)
    
    # Sinusoidal (odd harmonics are anti-palindromic around center)
    for freq in [1, 3, 5]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * x), 0.01, None))
    
    # Spectral init
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init(n, rng))
    
    # Random smooth
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(a, kernel, mode='same'), 0.01, None))
    
    # Two-block
    half = n // 2
    ratio = 0.2 + 0.6 * rng.random()
    inits.append(np.concatenate([np.ones(half) * ratio, np.ones(n-half) * (1-ratio)]))
    inits.append(np.concatenate([np.ones(half) * (1-ratio), np.ones(n-half) * ratio]))
    
    return inits

def subgrad_polish(a, max_iters=5000, time_limit=120.0):
    """Subgradient polish with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    
    step_scale = 1.0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of max(c) w.r.t. a:
        # d(max c)/da[j] = dc[k_star]/da[j] = a[k_star-j] (if valid) + a[j-(k_star-(n-1))]... 
        # Actually dc[k]/da[j] = a[k-j] (if 0<=k-j<n) [from the definition c[k]=sum_i a[i]a[k-i]]
        # So d(max c)/da[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
        
        sg = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        if j_max >= j_min:
            sg[j_min:j_max+1] = 2.0 * a[k_star - j_max:k_star - j_min:-1]
            # Wait: dc[k_star]/da[j] = a[k_star - j] for 0 <= k_star-j < n
            # j ranges from max(0, k_star-n+1) to min(k_star, n-1)
            # a[k_star-j] for j in [j_min, j_max] gives a[k_star-j_max] ... a[k_star-j_min]
            # which is a reversed segment
            
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        
        # Polyak step: (f(x) - f*) / ||g||^2, but we don't know f*
        # Use diminishing step
        step = step_scale / (1.0 + 0.01 * it)
        step = min(step, 1.0 / np.sqrt(gn2 + 1e-10))
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        
        # Adapt step scale
        if it % 100 == 0:
            step_scale *= 0.95
    
    return best_sc, best_a

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE-smoothed objective with Adam optimizer, multi-beta annealing,
coordinate descent refinement, subgradient polish, and multi-start exploration
across diverse initializations and sequence lengths.
Key improvements over baseline: Adam (faster than L-BFGS for n>5000),
normalized working space, vectorized coordinate descent, better beta schedule.

import numpy as np
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    log_sum = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * log_sum / (S * S)

    # Gradient via cross-correlation
    L = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    cc = np.fft.irfft(np.conj(A) * P, L)
    dL_da = cc[:n]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * log_sum / S3)
    return score, grad


def adam_optimize(a0, beta, max_iter, time_limit, lr=0.01, betas=(0.9, 0.999), eps=1e-8):
    """Adam optimizer for LSE-smoothed objective with non-negativity."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a[:] = 0.5

    b1, b2 = betas
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_sc = compute_score(a)
    best_a = a.copy()

    t0 = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        t += 1
        sc, grad = lse_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        progress = it / max_iter
        lr_t = lr * 0.5 * (1 + math.cos(math.pi * min(progress, 1.0)))

        a -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

    exact_sc = compute_score(best_a)
    return exact_sc, best_a


def coord_descent_fast(a, max_passes=20, time_limit=120.0, candidates_per_coord=6):
    """Vectorized coordinate descent: try candidate values for each a[j]."""
    n = len(a)
    a = a.astype(np.float64).copy()

        S = float(a.sum())
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:

            j = int(perm[idx])
            aj = a[j]
            if aj < 1e-10:

            # Remove a[j] contribution from c
            a[j] = 0.0
            c_base = c.copy()
            # Subtract the old contribution
            j_start = max(0, j)

            k_range = j_end - j_start + 1
            a_seg = a[0:k_range] if j_start == j else a[max(0, j_start - j):max(0, j_start - j) + k_range]
            # c[k] had contribution 2*aj*a[k-j] (for k in [j, j+n-1]) plus aj^2 at k=2j
            # But a[k-j] is the current a (with a[j]=0), so for k=2j, a[j]=0,
            # the old contribution was aj*a[j]+a[j]*aj = 0 (since a[j]=0 now)
            # Actually we need to subtract what was there before:
            # Old c[k] included: aj*a[k-j] + a[k-j]*aj (if k-j != j) or aj^2 (if k=2j)
            # = 2*aj*a[k-j] (if k!=2j) or aj^2 (if k=2j)
            # But a[k-j] in the old a (before setting a[j]=0) equals a[k-j] now
            # (since k-j != j when k != 2j)

            for k in range(j_start, j_end + 1):
                m_idx = k - j
                if k == 2 * j:
                    c_base[k] -= aj * aj
                else:
                    c_base[k] -= 2.0 * aj * a[m_idx]

            S_base = S - aj
            if S_base < 1e-10:
                a[j] = aj
                continue

            # Try candidate values

                S_full = S_base + x

                sc = 2.0 * n * float(np.max(c_full)) / (S_full * S_full)

            a[j] = best_x

        sc = compute_score(a)
        if sc < best_sc:

            best_a = a.copy()

def subgrad_polish(a, max_iters=10000, time_limit=120.0):
    """Subgradient descent with diminishing step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()
    step_base = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of score w.r.t. a
        sg = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        if j_max >= j_min:
            # dc[k_star]/da[j] = a[k_star - j] for valid j
            # j in [j_min, j_max], k_star - j in [k_star - j_max, k_star - j_min]
            sg[j_min:j_max + 1] = 2.0 * a[k_star - j_min:k_star - j_max - 1:-1] if j_max > j_min else 2.0 * a[k_star - j_min:k_star - j_max:-1]

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break

        step = step_base / (1.0 + 0.02 * it)
        step = min(step, 2.0 / math.sqrt(gn2 + 1e-10))

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def spectral_init(n, rng, bandwidth=0.5):
    """Initialize from random-phase spectral construction."""
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.zeros(M_fft // 2 + 1, dtype=np.complex128)
    freqs = np.arange(len(A))
    mask = freqs < len(A) * bandwidth
    A[mask] = np.exp(2j * np.pi * rng.random(mask.sum()))
    A[0] = np.abs(A[0])
    a = np.real(np.fft.irfft(A * np.sqrt(M_fft), M_fft))[:n]
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))
    inits.append(0.01 + 0.99 * x ** 2)
    inits.append(0.01 + 0.99 * (1 - x) ** 2)

    for freq in [1, 3, 5]:
        inits.append(np.clip(0.5 + 0.4 * np.sin(np.pi * freq * x), 0.01, None))

    inits.append(spectral_init(n, rng, 0.4))
    inits.append(spectral_init(n, rng, 0.6))
    inits.append(spectral_init(n, rng, 0.8))

    a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, None)
    ksize = max(1, n // 100)
    kernel = np.ones(ksize) / ksize
    inits.append(np.clip(np.convolve(a_rand, kernel, mode='same'), 0.01, None))

    half = n // 2
    for ratio in [0.3, 0.5, 0.7]:
        inits.append(np.concatenate([np.ones(half) * ratio, np.ones(n - half) * (1 - ratio)]))
        inits.append(np.concatenate([np.ones(half) * (1 - ratio), np.ones(n - half) * ratio]))

    return inits


def run_lbfgs(a0, beta, maxiter=5000, time_limit=120.0):
    """L-BFGS-B on LSE-smoothed objective."""
    from scipy.optimize import minimize

    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()

    def obj_and_grad(x):
        sc, g = lse_grad(x, beta)
        return sc, g

    try:
        res = minimize(
            obj_and_grad, a0, method='L-BFGS-B', jac=True,
            bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 30}
        )
    except Exception:
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

            if j_max >= j_min:

                for j in range(j_min, j_max + 1):
                    contrib[j] = best_a[j] * best_a[k_star - j]
                top_n = max(5, n_best // 200)
                top_idx = np.argsort(contrib)[::-1][:top_n]
                red_total = 0.0
                for j in top_idx:
                    amt = a_pert[j] * 0.05
                    a_pert[j] = max(0.0, a_pert[j] - amt)
                    red_total += amt
                bot_idx = np.argsort(contrib)[:-top_n]
                if len(bot_idx) > 0:
                    a_pert[bot_idx] += red_total / len(bot_idx)
        elif strategy == 2:
            # Frequency-domain smoothing
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 2).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.15 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        else:
            # Anti-palindromic bump
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 20)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.06 * sign * np.exp(-d * d))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.95

        sc, a_opt = adam_optimize(a_pert, beta=3000, max_iter=2000,
                                   time_limit=25.0, lr=0.01)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Trial {trial} (s={strategy}): {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # ============================================================
    # Final polish
    # ============================================================
    if time.monotonic() < deadline - 10:
        sc, a_opt = subgrad_polish(best_a, max_iters=10000, time_limit=60.0)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final subgrad: {sc:.8f}")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic()-t_start:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
