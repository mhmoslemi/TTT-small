**Improved Multiplicative Equalization:**

The key idea: at the optimum, for the problem min max_k R(k) s.t. a >= 0, sum(a) = 1, the KKT conditions give:

For each j: sum_{k in active} λ_k * dR(k)/da[j] = μ (constant)

where active = {k : R(k) = max R}.

The multiplicative update approximates λ_k by using R(k) values as weights.

Instead of using a simple threshold, I'll use:
λ_k = exp(α * R(k) / R_mean) for all k (not just active ones).

This gives a smooth weighting that emphasizes the "hot" spots more as α increases. As α → ∞, it approaches the hard threshold.

**Even better: Use the actual dual variable estimation.**

The Lagrangian is: L = max_k R(k) + ν * (sum(a) - 1) - sum_j η_j * a[j]

At the optimum, the "subgradient" of max_k R(k) is a convex combination of the gradients of R(k) for active k. The multipliers λ_k determine this combination.

A practical way to estimate λ: use the eigenvector of the "constraint matrix" or use a power iteration.

Actually, let me try a different angle. The multiplicative update:

where h[j] = sum_k λ_k * dR(k)/da[j] and μ = mean(h).

import numpy as np

import time

    n = len(a)

    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed max-convolution score and gradient.
    Score = 2n * LSE(conv(a,a); beta) / (sum(a))^2
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -100, 100))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[j] = sum_k p[k] * dR(k)/da[j]

    p_rev = p[::-1]
    cp = fast_conv2(p, a)
    cp_rev = fast_conv2(p_rev, a)
    g_L = cp[:n] + cp_rev[n - 1:2 * n - 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)

    Smooth sum-of-squared-excesses objective and gradient.
    F(a) = 2n * sum_k se[k]^2 / S^2
    where se[k] = smooth_hinge(R(k) - target), target = S^2/(2n-1)

    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    se = (excess + sqrt_term) / 2.0  # smooth hinge, always >= 0

    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * S * g - 2.0 * sum_se2 * np.ones(n)) / S3

def run_lbfgs(a0, obj_func, maxiter=5000, maxfun=100000, deadline=None, bounds_lo=0.0, bounds_hi=1000.0):
    """
    Run L-BFGS-B on a custom objective function.
    obj_func(x) -> (value, gradient)
    Returns (exact_score, result_array).
    """
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), bounds_lo, bounds_hi).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(bounds_lo, bounds_hi)] * n

    options = {'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': maxiter,
               'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            sc, _ = obj_func(a0)
            return compute_score(a0), a0
        options['maxfun'] = min(maxfun, max(200, int(remaining * 300)))
        options['maxiter'] = min(maxiter, max(200, int(remaining * 80)))

    try:
        res = minimize(obj_func, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds, options=options)
        a_out = np.clip(res.x, bounds_lo, bounds_hi)
    except Exception:
        a_out = a0
    return compute_score(a_out), a_out

        # h[j] = sum_k lam[k] * dR(k)/da[j] = conv(lam,a)[j] + conv(lam_rev,a)[2n-2-j]
        lam_rev = lam[::-1]
        cl = fast_conv2(lam, a)
        cl_rev = fast_conv2(lam_rev, a)
        h = cl[:n] + cl_rev[n - 1:2 * n - 1][::-1]

        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Damped multiplicative update
        factor = np.power(h_mean / h, alpha)
        factor = np.clip(factor, 0.01, 100.0)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        S_new = float(a.sum())
        if S_new < 1e-12:
            a[:] = 1.0 / n
            break
        a *= S / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def multiplicative_equalize_hard(a, n_iters=100, alpha=0.3, threshold_frac=0.95, deadline=None):
    """Hard-threshold multiplicative equalization (faster, for final polish)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        threshold = M * threshold_frac
        lam = (c >= threshold).astype(np.float64)
        if lam.sum() == 0:
            lam[int(np.argmax(c))] = 1.0
        lam = lam / lam.sum()

        lam_rev = lam[::-1]
        cl = fast_conv2(lam, a)
        cl_rev = fast_conv2(lam_rev, a)
        h = cl[:n] + cl_rev[n - 1:2 * n - 1][::-1]
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        factor = np.power(h_mean / h, alpha)
        factor = np.clip(factor, 0.01, 100.0)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a *= S / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def subgradient_polish(a, n_iters=200, deadline=None):
    """Projected subgradient on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = min(0.5, (sc - 1.0) / (gn * gn + 1e-30))
        step *= 1.0 / (1.0 + 0.05 * it)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def build_symmetric_init(n, rng, parent=None):
    """Generate symmetric initializations."""
    inits = []
    half = n // 2

    if parent is not None:
        p = parent.astype(np.float64).copy()
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p) - 1, n), np.arange(len(p)), p)
        # Symmetrize
        p = 0.5 * (p + p[::-1])
        inits.append(np.clip(p, 0.01, 1000.0))

    x = np.arange(n, dtype=np.float64) / n
    # Bell shape (symmetric)
    inits.append(0.5 + 0.4 * np.exp(-4.0 * (x - 0.5) ** 2))
    # Flat with edge taper
    inits.append(np.clip(1.0 - 0.3 * (2.0 * x - 1.0) ** 4, 0.1, 1.0))
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine (symmetric)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * (x - 0.5)))
    # Double hump
    inits.append(0.3 + 0.5 * np.exp(-8.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (x - 0.7) ** 2))
    # Random symmetric
    half_rand = np.clip(0.5 + 0.3 * rng.standard_normal(half + 1), 0.01, 1000.0)
    full = np.concatenate([half_rand, half_rand[:half][-1::-1]])
    inits.append(full)
    inits.append(full * (1.0 + 0.1 * rng.standard_normal(n)))
    inits[-1] = np.clip(inits[-1], 0.01, 1000.0)
    # Symmetrize all
    inits = [np.clip(0.5 * (a + a[::-1]), 0.01, 1000.0) for a in inits]
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses annealed multiplicative equalization + L-BFGS (LSE and excess objectives).
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}" + (f" (len={parent.size})" if parent is not None else ""))

    # Determine lengths to try
    lengths = [8192, 16384]
    if parent is not None:
        pa = parent.size
        if pa not in lengths and pa <= 32768:
            lengths.insert(0, pa)
    lengths = sorted(set(lengths))[:3]
    print(f"Lengths to try: {lengths}")

    # ==================================================================
    # Phase 1: Multiplicative equalization from diverse starts
    # ==================================================================
    phase1_deadline = t_start + 0.35 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Multiplicative Equalization (t < {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  n = {n}")
        inits = build_symmetric_init(n, rng, parent)

        for i, a0 in enumerate(inits[:5]):
            if time.monotonic() > phase1_deadline:
                break
            sc, a_eq = multiplicative_equalize(a0, n_iters=400,
                                               beta_start=0.3, beta_end=60.0,
                                               alpha=0.4, deadline=phase1_deadline)
            report(sc, a_eq, f"ME n={n} init={i}")

    # ==================================================================
    # Phase 2: L-BFGS refinement with LSE objective (continuation)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS LSE Refinement (t < {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    a_cur = best_a.copy()

    beta_schedule = [30, 80, 200, 500, 1200]
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break

        def lse_wrapper(x, b=beta):
            return lse_score_grad(x, b)

        sc, a_opt = run_lbfgs(a_cur, lse_wrapper, maxiter=8000, maxfun=200000,
                              deadline=phase2_deadline)
        report(sc, a_opt, f"LSE b={beta}")
        a_cur = a_opt

    # ==================================================================
    # Phase 3: Alternating ME + L-BFGS (excess objective)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Alternating ME + Excess L-BFGS (t < {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    a_cur = best_a.copy()
    for rnd in range(6):
        if time.monotonic() > phase3_deadline:
            break
        print(f"  Round {rnd+1}:", end=" ")

        # Step A: Hard-threshold ME
        sc, a_eq = multiplicative_equalize_hard(a_cur, n_iters=60, alpha=0.3,
                                                threshold_frac=0.97,
                                                deadline=phase3_deadline)
        report(sc, a_eq, f"rnd{rnd} hardME")
        a_cur = a_eq

        # Step B: Excess L-BFGS
        if time.monotonic() < phase3_deadline - 10:
            excess_beta = 5.0 + 3.0 * rnd

            def excess_wrapper(x, eb=excess_beta):
                return excess_obj_grad(x, eb)

            sc, a_opt = run_lbfgs(a_cur, excess_wrapper, maxiter=5000, maxfun=150000,
                                  deadline=phase3_deadline)
            report(sc, a_opt, f"rnd{rnd} excess b={excess_beta:.1f}")
            a_cur = a_opt

        # Step C: Soft ME with high beta
        if time.monotonic() < phase3_deadline - 10:
            sc, a_eq2 = multiplicative_equalize(a_cur, n_iters=100,
                                                beta_start=20.0, beta_end=100.0,
                                                alpha=0.3, deadline=phase3_deadline)
            report(sc, a_eq2, f"rnd{rnd} softME")
            a_cur = a_eq2

    # ==================================================================
    # Phase 4: High-beta LSE polish + subgradient
    # ==================================================================
    phase4_deadline = deadline - 30.0
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish (t < {phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for beta in [2000, 5000]:
        if time.monotonic() > phase4_deadline:
            break
        def final_lse(x, b=beta):
            return lse_score_grad(x, b)
        sc, a_opt = run_lbfgs(best_a, final_lse, maxiter=5000, maxfun=100000,
                              deadline=phase4_deadline)
        report(sc, a_opt, f"final LSE b={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 20:
        sc, a_pol = subgradient_polish(best_a, n_iters=300, deadline=deadline - 10)
        report(sc, a_pol, "subgrad")

    # Extra ME from best
    if time.monotonic() < deadline - 15:
        sc, a_eq = multiplicative_equalize_hard(best_a, n_iters=50, alpha=0.2,
                                                threshold_frac=0.98, deadline=deadline - 10)
        report(sc, a_eq, "final hardME")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
