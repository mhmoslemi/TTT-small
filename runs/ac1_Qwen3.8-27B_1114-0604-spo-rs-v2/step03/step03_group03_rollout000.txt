"""
Advanced optimizer combining:
1. LP-based direction finding (cutting plane)
2. Adaptive LSE L-BFGS refinement
3. Peak-targeted coordinate descent
4. Multi-start with spectral initialization
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def coordinate_descent_sweep(a, rng=None, n_coords=None, order=None):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    if n_coords is None:
        n_coords = n
    if order is None:
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    final_score = compute_score(a)
    return final_score, a

def hot_coordinate_sweep(a, n_sweeps=1):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:

        order = np.argsort(contrib)[::-1]
        for i in order:

            a_i = a[i]
            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)
            if k_lo > 0 and k_hi < M_conv - 1:

            elif k_lo > 0:

            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))

                outside_max = 0.0
            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()
            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)

            beta_coef[is_2i] = 1.0
            S_rest = S - a_i

            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    final_score = compute_score(a)
    return final_score, a

def lp_find_direction(f, M, max_constraints=500):
    """
    Find direction g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for active constraints.
    Uses cutting plane: start with peak constraint, add violated ones.
    """
    n = len(f)
    fg = None  # cross-correlation f*g, computed incrementally
    
    # Start with the constraint at the peak lag
    f_conv = fast_conv(f)
    active_k = [int(np.argmax(f_conv))]
    
    for cp_iter in range(max_constraints):
        # Build LP with active constraints
        # A[k_row, j] = f[k_active[k_row] - j] if valid
        K = len(active_k)
        A = lil_matrix((K, n))
        for row, k in enumerate(active_k):
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                A[row, j] = f[k - j]
        A = A.tocsr()
        b = np.full(K, M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if not res.success:
                return None
            g_new = res.x
        except Exception:
            return None
        
        # Check for violated constraints
        # Compute f * g_new (cross-correlation)
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid indices
        # = sum_j f[k-j] * g[j] for valid indices
        # Use FFT:
        fg = np.fft.irfft(np.fft.rfft(f, 2*n-1) * np.fft.rfft(g_new, 2*n-1), 2*n-1)[:2*n-1]
        
        violations = np.where(fg > M + 1e-10)[0]
        if len(violations) == 0:
            return g_new
        
        # Add the most violated constraint
        worst = violations[np.argmax(fg[violations] - M)]
        if worst not in active_k:
            active_k.append(int(worst))
    
    return None  # Didn't converge

def lp_iteration(f, max_lp_iterations=50):
    """
    One LP-based iteration: find direction, blend, return new sequence.
    """
    n = len(f)
    f_conv = fast_conv(f)
    M = float(np.max(f_conv))
    S = float(f.sum())
    
    g = lp_find_direction(f, M, max_constraints=max_lp_iterations)
    if g is None:
        return f.copy(), None
    
    g_sum = float(g.sum())
    if g_sum < 1e-12:
        return f.copy(), None
    
    # Blend: new_f = (1-t)*f + t*g for some t
    # We want to maximize the improvement. Try several t values.
    best_t = 0.5
    best_sc = compute_score(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
        new_f = (1 - t) * f + t * g
        new_f = np.clip(new_f, 0, 1000)
        sc = compute_score(new_f)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    new_f = (1 - best_t) * f + best_t * g
    new_f = np.clip(new_f, 0, 1000)
    return new_f, best_sc

def spectral_init(n, rng):
    """
    Create an initial sequence with flat-ish spectrum.
    """
    # Target: autocorrelation as flat as possible
    # In frequency domain: |F|^2 as flat as possible (except DC)
    # So F[k] ≈ constant for k ≠ 0
    # f = IFFT(F) where F[0] = large (DC), F[k] = c for k ≠ 0
    
    # We want f >= 0, so we need the inverse FFT to be non-negative.
    # One approach: start with a flat spectrum, take IFFT, clip negatives, iterate.
    
    F = np.ones(n) * 0.1  # flat spectrum
    F[0] = 0  # we'll add DC later
    
    for _ in range(50):
        f = np.fft.irfft(F, n)
        f = np.clip(f, 0, None)
        if f.sum() < 1e-6:
            break
        f = f / f.sum() * n  # normalize
        F = np.fft.rfft(f)
        F[0] = 0  # remove DC for the flat part
        # Actually, we want to keep the overall shape but flatten the non-DC part
        non_dc = F[1:].copy()
        F_new = np.zeros(n//2 + 1)
        F_new[0] = F[0]
        F_new[1:] = non_dc / np.max(np.abs(non_dc)) * 0.1
        F = F_new
    
    f = np.fft.irfft(F, n)
    f = np.clip(f, 0, None)
    if f.sum() < 1e-6:
        f = np.ones(n) * 0.5
    return f

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def targeted_perturb(a, rng, strength=0.05):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)
