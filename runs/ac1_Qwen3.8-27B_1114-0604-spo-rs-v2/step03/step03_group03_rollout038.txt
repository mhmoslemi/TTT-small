import numpy as np

import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)

    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S

"""
Multi-strategy optimizer for minimum peak autocorrelation sequences.
Combines coordinate descent, LSE-based L-BFGS, LP-inspired block updates,
and ruin-and-recreate perturbations to escape local minima.
Key improvements: adaptive LSE beta schedule, structured perturbations,
frequency-domain moves, and aggressive local refinement.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrap(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_wrap, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None):
    """One full sweep of coordinate descent with incremental conv."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return a


def hot_coordinate_sweep(a, rng=None):
    """CD sweep prioritizing coordinates contributing to the peak."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    # Compute peak contributions
    k_star = int(np.argmax(conv))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    order = np.argsort(contrib)[::-1]

    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()

        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return a


def lp_block_step(a, n_iter=15):
    """
    LP-inspired iterative step: repeatedly identify the peak lag,
    reduce contributing coordinates, and redistribute mass.
    """
    n = len(a)
    a = a.copy()
    conv = fast_conv(a)

    for it in range(n_iter):
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break

        peak_idx = np.arange(j_start, j_end + 1)
        partners = k_star - peak_idx
        contribs = a[peak_idx] * a[partners]

        # Target: reduce the peak by reducing top contributors
        target_reduction = 0.005 * M
        total_red = 0.0

        # Sort by contribution, reduce from largest
        sorted_order = np.argsort(contribs)[::-1]
        n_reduce = max(1, len(peak_idx) // 10)

        for pos in range(min(n_reduce, len(sorted_order))):
            j = peak_idx[sorted_order[pos]]
            partner_val = a[partners[sorted_order[pos]]]
            if partner_val < 1e-12:
                continue
            max_red = a[j] * 0.03
            needed = (target_reduction - total_red) / partner_val
            red_amt = min(max_red, max(0.0, needed))
            if red_amt > 0:
                a[j] -= red_amt
                total_red += red_amt

        # Redistribute to non-peak positions
        if total_red > 1e-12:
            non_peak = np.ones(n, dtype=bool)
            non_peak[peak_idx] = False
            n_non = int(non_peak.sum())
            if n_non > 0:
                a[non_peak] += total_red / n_non

        # Periodically recompute conv
        if it % 5 == 4:
            conv = fast_conv(a)

    conv = fast_conv(a)
    return a


def ruin_recreate(a, rng, fraction=0.05):
    """Zero out top-contributing elements, redistribute."""
    n = len(a)
    a = a.copy()
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))

    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        partners = k_star - np.arange(j_start, j_end + 1)
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1] * a[partners]

    importance = contrib + 0.05 * a * a
    n_ruin = max(1, int(n * fraction))
    top_idx = np.argsort(importance)[::-1][:n_ruin]
    total = a[top_idx].sum()
    a[top_idx] = 0.0

    if total > 0:
        weights = a + 1e-8
        weights /= weights.sum()
        a += total * weights

    return np.clip(a, 0.0, 1000.0)


def freq_perturb(a, rng, strength=0.01):
    """Perturb in frequency domain."""
    n = len(a)
    A = np.fft.rfft(a)
    noise_mag = strength * np.abs(A) * rng.random(A.shape[0])
    phase = rng.random(A.shape[0]) * 2 * np.pi
    A_perturbed = A + noise_mag * np.exp(1j * phase)
    a_out = np.fft.irfft(A_perturbed, n)
    return np.clip(a_out, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 25.0

    # Load parent
    best_a = None
    best_score = np.inf

    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                p = np.clip(p, 0.0, 1000.0)
                p_score = compute_score(p)
                best_a = p
                best_score = p_score
                print(f"Loaded parent: score={p_score:.8f}, len={len(p)}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_a is None:
        n_init = 4096
        best_a = np.ones(n_init) * 0.5
        best_score = compute_score(best_a)
        print(f"Default init: score={best_score:.8f}, len={len(best_a)}")

    n_work = len(best_a)
    print(f"n={n_work}, starting score={best_score:.8f}")

    # ========== Phase 1: CD from parent (25% time) ==========
    p1_dead = t_start + 0.25 * total_budget
    print(f"\n--- Phase 1: CD ({p1_dead - t_start:.0f}s) ---")

    a_cur = best_a.copy()
    no_imp = 0
    sw = 0
    while time.monotonic() < p1_dead:
        sw += 1
        if sw % 3 == 0:
            a_cur = hot_coordinate_sweep(a_cur)
        else:
            a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        sc = compute_score(a_cur)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            no_imp = 0
            if sw <= 3 or sw % 10 == 0:
                print(f"  P1 sw{sw}: {sc:.8f} BEST")
        else:
            no_imp += 1
        if sw % 10 == 0:
            print(f"  P1 sw{sw}: {sc:.8f} (ni={no_imp})")
        if no_imp >= 20:
            a_cur = ruin_recreate(a_cur, rng, fraction=0.01 + 0.02 * rng.random())
            no_imp = 0

    print(f"P1 done: {best_score:.8f}")

    # ========== Phase 2: LSE adaptive beta (25% time) ==========
    p2_dead = t_start + 0.50 * total_budget
    print(f"\n--- Phase 2: LSE ({p2_dead - t_start:.0f}s) ---")

    a_lse = best_a.copy()
    beta_schedule = [20, 50, 120, 300, 700, 1500, 4000, 10000]
    for beta in beta_schedule:
        if time.monotonic() > p2_dead:
            break
        t0 = time.monotonic()
        mi = 4000 if beta < 500 else 2000
        mf = 60000 if beta < 500 else 30000
        sc, a_opt = run_lbfgs(a_lse, beta, maxiter=mi, maxfun=mf)
        el = time.monotonic() - t0
        improved = sc < best_score - 1e-10
        if improved:
            best_score = sc
            best_a = a_opt.copy()
        a_lse = a_opt.copy()
        tag = " BEST" if improved else ""
        print(f"  LSE b={beta}: {sc:.8f} [{el:.1f}s]{tag}")

        # Quick CD after each LSE
        a_cd = a_opt.copy()
        a_cd = coordinate_descent_sweep(a_cd, rng=rng)
        sc_cd = compute_score(a_cd)
        if sc_cd < best_score - 1e-10:
            best_score = sc_cd
            best_a = a_cd.copy()
            a_lse = a_cd.copy()
            print(f"  LSE+CD b={beta}: {sc_cd:.8f} BEST")

    print(f"P2 done: {best_score:.8f}")

    # ========== Phase 3: LP-inspired + CD (20% time) ==========
    p3_dead = t_start + 0.70 * total_budget
    print(f"\n--- Phase 3: LP-block ({p3_dead - t_start:.0f}s) ---")

    a_cur = best_a.copy()
    for it in range(40):
        if time.monotonic() > p3_dead:
            break
        a_cur = lp_block_step(a_cur, n_iter=10)
        a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        sc = compute_score(a_cur)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            if it % 5 == 0:
                print(f"  P3 it{it}: {sc:.8f} BEST")
    print(f"P3 done: {best_score:.8f}")

    # ========== Phase 4: Diverse restarts (20% time) ==========
    p4_dead = t_start + 0.90 * total_budget
    print(f"\n--- Phase 4: Restarts ({p4_dead - t_start:.0f}s) ---")

    n_cur = len(best_a)
    for trial in range(50):
        if time.monotonic() > p4_dead - 3.0:
            break
        s = trial % 8
        if s == 0:
            a_p = ruin_recreate(best_a, rng, fraction=0.03 + 0.04 * rng.random())
        elif s == 1:
            sc_ = 0.008 + 0.015 * rng.random()
            a_p = np.clip(best_a * (1.0 + sc_ * rng.standard_normal(n_cur)), 0.0, 1000.0)
        elif s == 2:
            blk = max(20, n_cur // 40)
            st = int(rng.integers(0, n_cur - blk))
            a_p = best_a.copy()
            sv = a_p[st:st + blk].sum()
            a_p[st:st + blk] = 0.0
            mk = np.ones(n_cur, dtype=bool)
            mk[st:st + blk] = False
            a_p[mk] += sv / max(1, int(mk.sum()))
        elif s == 3:
            seg = max(20, n_cur // 15)
            s1 = int(rng.integers(0, n_cur - seg))
            s2 = int(rng.integers(0, n_cur - seg))
            a_p = best_a.copy()
            tmp = a_p[s1:s1 + seg].copy()
            a_p[s1:s1 + seg] = a_p[s2:s2 + seg]
            a_p[s2:s2 + seg] = tmp
        elif s == 4:
            a_p = freq_perturb(best_a, rng, strength=0.005 + 0.01 * rng.random())
        elif s == 5:
            # Sinusoidal
            fr = 1.0 + 10.0 * rng.random()
            am = 0.015 + 0.03 * rng.random()
            x = np.arange(n_cur) / n_cur
            a_p = np.clip(best_a * (1.0 + am * np.cos(2 * np.pi * fr * x + rng.random() * 6.28)), 0.0, 1000.0)
        elif s == 6:
            # Mixed: half Gaussian, half frequency
            a_p1 = np.clip(best_a * (1.0 + 0.01 * rng.standard_normal(n_cur)), 0.0, 1000.0)
            a_p2 = freq_perturb(best_a, rng, strength=0.008)
            a_p = 0.5 * a_p1 + 0.5 * a_p2
        else:
            # Aggressive ruin
            a_p = ruin_recreate(best_a, rng, fraction=0.08 + 0.07 * rng.random())

        # Quick CD (3 sweeps)
        for _ in range(3):
            if time.monotonic() > p4_dead - 3.0:
                break
            a_p = coordinate_descent_sweep(a_p, rng=rng)

        sc = compute_score(a_p)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_p.copy()
            print(f"  P4 t{trial} (s={s}): {sc:.8f} BEST")
        elif trial % 10 == 0:
            print(f"  P4 t{trial} (s={s}): {sc:.8f}")

    print(f"P4 done: {best_score:.8f}")

    # ========== Phase 5: Final polish (remaining time) ==========
    print(f"\n--- Phase 5: Polish ---")
    a_cur = best_a.copy()
    for _ in range(25):
        if time.monotonic() > deadline - 15.0:
            break
        if _ % 4 == 0:
            a_cur = hot_coordinate_sweep(a_cur)
        else:
            a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        sc = compute_score(a_cur)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()

    # Final LSE
    if time.monotonic() < deadline - 15.0:
        sc, a_opt = run_lbfgs(best_a, 6000, maxiter=3000, maxfun=40000)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} BEST")
        a_cur = best_a.copy()
        a_cur = coordinate_descent_sweep(a_cur, rng=rng)
        sc = compute_score(a_cur)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            print(f"  Final CD: {sc:.8f} BEST")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 50}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Len={len(out)}, Time={elapsed:.1f}s/{total_budget}s")
    print(f"{'=' * 50}")
    return out
