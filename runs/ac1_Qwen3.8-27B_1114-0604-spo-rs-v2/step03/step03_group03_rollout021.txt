"""
Optimized step function sequence minimizer using:
1. Fast exact 1D coordinate descent (analytical solve, no numerical 1D optimization)
2. ILS (Iterative Least Squares) with adaptive weighting for global moves
3. Peak-targeted perturbation for escaping local minima
4. Multi-phase time allocation for robustness

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def exact_coord_descent_sweep(a, rng=None):
    """
    One full sweep of exact coordinate descent.
    For each coordinate i, solve the 1D subproblem analytically.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for i in order:
        i = int(i)
        a_i = a[i]
        
        # Affected lags: k in [i, min(2n-2, i+n-1)]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        inside_len = k_hi - k_lo + 1
        
        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        # base_inside = conv at affected lags minus a[i]'s contribution
        base_inside = conv[k_lo:k_hi + 1].copy()
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]  # a[k-i] for k in affected range
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        
        # Upper bound for t
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        # Analytical 1D solve
        t_opt = solve_1d_exact(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper)
        
        # Update
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    return a


def solve_1d_exact(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper):
    """
    Solve the 1D subproblem exactly.
    f(t) = 2n * max(max_k(base_k + alpha_k*t + beta_k*t^2), outside_max) / (S_rest + t)^2
    
    Strategy: iterate over candidate max lags, solve analytically, verify.
    """
    m = len(base_inside)
    
    # Find current max lag within affected region
    current_vals = base_inside + 0.0 * alpha * 0.0  # at t=0
    max_idx_inside = int(np.argmax(base_inside))
    
    # Candidates: the top few lags that could be the max
    # Sort by current value descending
    sorted_indices = np.argsort(base_inside)[::-1]
    
    best_t = 0.0
    best_f = np.inf
    
    # Try t = 0
    max_at_0 = max(float(base_inside.max()), outside_max)
    best_f = 2.0 * n * max_at_0 / (S_rest * S_rest) if S_rest > 0 else np.inf
    
    # Try the analytical solution for top candidates
    tried = set()
    for trial in range(min(5, m)):
        k = int(sorted_indices[trial])
        if k in tried:
            continue
        tried.add(k)
        
        B = float(base_inside[k])
        A = float(alpha[k])
        C = float(beta_coef[k])
        
        # f(t) = 2n * (B + A*t + C*t^2) / (S+t)^2 [assuming this lag is max]
        # f'(t) numerator: (2C*S - A)*t + (A*S - 2B)
        denom = 2.0 * C * S_rest - A
        numer = 2.0 * B - A * S_rest
        
        if abs(denom) < 1e-15:
            # Linear numerator: if numer < 0, f is decreasing, opt at t_upper
            # If numer > 0, f is increasing, opt at t=0
            # If numer == 0, f is constant
            if numer < 0:
                t_candidate = t_upper
            else:
                t_candidate = 0.0
        else:
            t_candidate = numer / denom
        
        # Clamp
        t_candidate = max(0.0, min(t_candidate, t_upper))
        
        # Evaluate f at t_candidate
        vals = base_inside + t_candidate * alpha + (t_candidate * t_candidate) * beta_coef
        actual_max = max(float(vals.max()), outside_max)
        S_new = S_rest + t_candidate
        if S_new > 0:
            f_val = 2.0 * n * actual_max / (S_new * S_new)
        else:
            f_val = np.inf
        
        if f_val < best_f:
            best_f = f_val
            best_t = t_candidate
    
    # Also try t = t_upper
    vals = base_inside + t_upper * alpha + (t_upper * t_upper) * beta_coef
    actual_max = max(float(vals.max()), outside_max)
    S_new = S_rest + t_upper
    if S_new > 0:
        f_val = 2.0 * n * actual_max / (S_new * S_new)
        if f_val < best_f:
            best_f = f_val
            best_t = t_upper
    
    # Refine: if the optimal t for the active lag gives a point where another lag is higher,
    # the true optimum might be at the crossing. Do one more iteration.
    if best_t > 0 and best_t < t_upper:
        vals_at_best = base_inside + best_t * alpha + (best_t * best_t) * beta_coef
        max_at_best = float(vals_at_best.max())
        if max_at_best > outside_max:
            # Find which lag is actually max at best_t
            actual_max_idx = int(np.argmax(vals_at_best))
            if actual_max_idx not in tried:
                B = float(base_inside[actual_max_idx])
                A = float(alpha[actual_max_idx])
                C = float(beta_coef[actual_max_idx])
                denom = 2.0 * C * S_rest - A
                numer = 2.0 * B - A * S_rest
                if abs(denom) > 1e-15:
                    t2 = numer / denom
                    t2 = max(0.0, min(t2, t_upper))
                    vals2 = base_inside + t2 * alpha + (t2 * t2) * beta_coef
                    actual_max2 = max(float(vals2.max()), outside_max)
                    S_new2 = S_rest + t2
                    if S_new2 > 0:
                        f2 = 2.0 * n * actual_max2 / (S_new2 * S_new2)
                        if f2 < best_f:
                            best_f = f2
                            best_t = t2
    
    return best_t


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def ils_iteration(a, n_iters=5, beta=50):
    """
    ILS (Iterative Least Squares): reweight lags and solve weighted LS.
    minimize sum_k w_k * conv[k] subject to a >= 0, sum(a) = S
    This is equivalent to minimize a^T Q a where Q depends on weights.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    
    for it in range(n_iters):
        c = fast_conv(a)
        # Weights: emphasize high lags
        c_pos = np.maximum(c, 0)
        w = np.exp(beta * (c_pos - c_pos.max()))
        w /= w.sum()
        
        # Solve: minimize sum_k w_k * conv[k] = a^T Q_a a
        # where Q_a is the matrix such that a^T Q_a a = sum_k w_k * (sum_j a[j]a[k-j])
        # Use projected gradient: a_new = max(0, a - lr * grad)
        # grad of sum_k w_k * conv[k] = 2 * Q_a * a
        # Q_a * a = sum_k w_k * (a * reverse(a))[k] (cross-correlation-like)
        
        # Actually, d/d a[i] of conv[k] = a[k-i] + a[i-(k-n+1)] ... 
        # Let's just use the gradient directly
        # grad[i] = sum_k w_k * d conv[k]/d a[i]
        # d conv[k]/d a[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-n+1)] ... 
        # Actually for full convolution: conv[k] = sum_j a[j]*a[k-j]
        # d conv[k]/d a[i] = a[k-i] (if valid index) + a[i] if k-i == i... 
        # No: d/d a[i] sum_j a[j] a[k-j] = a[k-i] + a[k-i] (if i appears twice) 
        # = 2*a[k-i] if k != 2i, = a[i] if k == 2i
        
        # So grad[i] = sum_k w_k * (2*a[k-i] if k != 2i else a[i])
        # = 2 * sum_k w_k * a[k-i] - a[i] * w[2i] (if 2i is valid)
        
        # This is a convolution of w with a (reversed), minus a correction
        M_conv = 2 * n - 1
        M_fft = 1 << (M_conv - 1).bit_length()
        W = np.fft.rfft(w, M_fft)
        A = np.fft.rfft(a, M_fft)
        # cross-correlation: sum_k w[k] * a[k-i] for valid k
        # This is (w * reverse(a))[n-1 + i] in full convolution
        # Actually: sum_k w[k]*a[k-i] where k-i in [0,n-1], i in [0,n-1]
        # Let m = k-i, then k = m+i, sum_m w[m+i]*a[m] = (w lagged by i) dot a
        # = (reverse(w) * a)[n-1+i] in full conv
        
        W_rev = np.fft.rfft(w[::-1], M_fft)
        cross = np.fft.irfft(W_rev * A, M_fft)
        # cross has length M_fft, we want indices [n-1, 2n-2]
        corr = cross[n - 1:2 * n - 1]
        
        grad = 2.0 * corr
        # Correction for diagonal (k=2i where the derivative is a[i] not 2*a[i])
        for i in range(n):
            ki = 2 * i
            if ki < M_conv:
                grad[i] -= w[ki] * a[i]
        
        # Projected gradient step
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        best_lr = 0.01
        best_val = np.inf
        for lr in [0.001, 0.005, 0.01, 0.05, 0.1]:
            a_trial = np.maximum(0, a - lr * grad)
            # Renormalize to maintain sum
            if a_trial.sum() > 0:
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fast_conv(a_trial)
                val = float(np.sum(w * c_trial))
                if val < best_val:
                    best_val = val
                    best_lr = lr
        
        a = np.maximum(0, a - best_lr * grad)
        if a.sum() > 0:
            a = a * (S / a.sum())
    
    return a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()

def feintuch_iter(a, n_iters=10, target_level=None):
    """Feintuch-style iteration for flattening autocorrelation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    for it in range(n_iters):
        # Compute autocorrelation
        c = fast_conv(a)
        
        # Determine target level
        if target_level is None:
            # Use the minimum non-center value as target
            non_center = np.delete(c, n-1)
            target_level = float(np.min(non_center)) * 0.5 + float(c[n-1]) * 0.1
        else:
            target_level = target_level
        
        # Create target autocorrelation
        R_target = c.copy()
        for k in range(M):
            if k != n - 1:  # Don't modify the center
                R_target[k] = max(R_target[k], target_level)
        
        # Spectral factorization
        # FFT of R_target
        C = np.fft.rfft(R_target, M_fft)
        # Take square root of magnitude, keep phase of FFT(a)
        A = np.fft.rfft(a, M_fft)
        mag_A = np.abs(A)
        phase_A = A / (mag_A + 1e-30)
        
        # New spectrum: sqrt(|C|) * phase_A
        mag_new = np.sqrt(np.abs(C))
        A_new = mag_new * phase_A
        
        # IFFT
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Preserve sum
        if a_new.sum() > 0:
            a_new = a_new * (S / a_new.sum())
        
        a = a_new
    
    return a

"""
Optimized step function sequence minimizer using:
1. Fast exact 1D coordinate descent with analytical subproblem solve
2. ILS (Iterative Least Squares) spectral iterations for global structure
3. LSE-smoothed L-BFGS refinement with adaptive beta
4. Peak-targeted perturbation for escaping local minima
Key improvement: analytical 1D solves eliminate minimize_scalar overhead,
enabling 5-10x more CD sweeps in the same time budget.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def solve_1d_coord(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper):
    """
    Solve 1D subproblem: minimize 2n * max(max_k(b_k+a_k*t+c_k*t^2), omit) / (S+t)^2
    Uses analytical critical points for top candidate lags.
    Returns optimal t in [0, t_upper].
    """
    m = len(base_inside)
    best_t = 0.0
    best_f = np.inf

    # Evaluate at t=0
    max_at_0 = max(float(base_inside.max()), outside_max)
    if S_rest > 1e-15:
        best_f = 2.0 * n * max_at_0 / (S_rest * S_rest)
    else:
        best_f = 1e10

    # Sort by current value (at t=0) to find likely max lags
    sorted_idx = np.argsort(base_inside)[::-1]

    # Try analytical solutions for top candidates
    tried = set()
    for trial in range(min(6, m)):
        k = int(sorted_idx[trial])
        if k in tried:
            continue
        tried.add(k)

        B = float(base_inside[k])
        A = float(alpha[k])
        C = float(beta_coef[k])

        # f(t) = 2n*(B + A*t + C*t^2)/(S+t)^2
        # f'(t) = 0 => (2C*S - A)*t + (A*S - 2B) = 0
        denom = 2.0 * C * S_rest - A
        numer = 2.0 * B - A * S_rest

        candidates_t = [0.0, t_upper]

        if abs(denom) > 1e-15:
            t_crit = numer / denom
            candidates_t.append(t_crit)
            # Also try if the function is monotone (no valid critical point in range)
            # which is handled by the boundary checks

        for tc in candidates_t:
            tc = max(0.0, min(tc, t_upper))
            S_new = S_rest + tc
            if S_new < 1e-15:
                continue
            vals = base_inside + tc * alpha + (tc * tc) * beta_coef
            actual_max = max(float(vals.max()), outside_max)
            f_val = 2.0 * n * actual_max / (S_new * S_new)
            if f_val < best_f:
                best_f = f_val
                best_t = tc

    # Second pass: check which lag is actually max at best_t, try its critical point
    if 0.0 < best_t < t_upper:
        vals_at_best = base_inside + best_t * alpha + (best_t * best_t) * beta_coef
        max_at_best = float(vals_at_best.max())
        if max_at_best > outside_max + 1e-15:
            idx_at_best = int(np.argmax(vals_at_best))
            if idx_at_best not in tried:
                tried.add(idx_at_best)
                B = float(base_inside[idx_at_best])
                A = float(alpha[idx_at_best])
                C = float(beta_coef[idx_at_best])
                denom = 2.0 * C * S_rest - A
                numer = 2.0 * B - A * S_rest
                if abs(denom) > 1e-15:
                    tc = numer / denom
                    tc = max(0.0, min(tc, t_upper))
                    S_new = S_rest + tc
                    if S_new > 1e-15:
                        vals = base_inside + tc * alpha + (tc * tc) * beta_coef
                        actual_max = max(float(vals.max()), outside_max)
                        f_val = 2.0 * n * actual_max / (S_new * S_new)
                        if f_val < best_f:
                            best_f = f_val
                            best_t = tc

    return best_t


def exact_cd_sweep(a, rng=None, hot_order=False):
    """
    One full sweep of exact coordinate descent.
    hot_order: if True, prioritize coordinates contributing to the peak.
    Returns updated array.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if hot_order:
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    elif rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for i in order:
        i = int(i)
        a_i = a[i]

        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        inside_len = k_hi - k_lo + 1

        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        # Compute base_inside (conv without a[i])
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        base_inside = conv[k_lo:k_hi + 1].copy()
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i
        t_upper = max(4.0 * a_i + 0.001, S_rest / n * 4.0)
        t_upper = min(t_upper, 1000.0)

        t_opt = solve_1d_coord(base_inside, alpha, beta_coef, outside_max, S_rest, n, t_upper)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def ils_step(a, beta=40, n_iters=3):
    """
    ILS iteration: reweight autocorrelation lags and solve via spectral method.
    Raises below-average lags toward the target and refactors.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    center = n - 1

    for _ in range(n_iters):
        c = fast_conv(a)

        # Target: raise sidelobes that are below a fraction of the peak
        peak = float(c[center])
        target = 0.7 * peak
        R_target = c.copy()
        for k in range(M):
            if k != center and R_target[k] < target:
                R_target[k] = target

        # Ensure R_target has non-negative spectrum (add DC if needed)
        C_spec = np.fft.rfft(R_target, M_fft)
        if np.any(np.real(C_spec) < -1e-10):
            # Add a small delta at center to fix
            R_target[center] += -np.min(np.real(C_spec)) * M / 2 + 1.0
            C_spec = np.fft.rfft(R_target, M_fft)

        # Spectral factorization: A_new = |sqrt(max(Re(C_spec),0))| * phase(A)
        A = np.fft.rfft(a, M_fft)
        mag_A = np.abs(A) + 1e-30
        phase_A = A / mag_A
        mag_new = np.sqrt(np.maximum(np.real(C_spec), 0.0))
        A_new = mag_new * phase_A

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        a = a_new

    return a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 30)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(1, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def antisymmetric_perturb(a, rng, strength=0.03):
    """Perturb to enhance anti-palindromic structure (suppress center peak)."""
    n = len(a)
    a_pert = a.copy()
    # Push first half up and second half down (or vice versa)
    half = n // 2
    sign = 1.0 if rng.random() > 0.5 else -1.0
    shape = np.linspace(-1, 1, n)
    a_pert = a_pert * (1.0 + sign * strength * shape)
    return np.clip(a_pert, 0.0, 1000.0)

    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
