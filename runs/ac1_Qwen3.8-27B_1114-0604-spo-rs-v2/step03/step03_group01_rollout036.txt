    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smooth_max(c, beta):
    """LSE approximation of max with gradient weights."""
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    return L, p


def gradient_lse_a(a, beta):
    """LSE-smoothed score gradient w.r.t. a (direct space)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, p = lse_smooth_max(c, beta)
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def gradient_lse_x(x, beta):
    """LSE-smoothed score gradient w.r.t. x (log-domain param)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    S = a.sum()
    a = a / S  # normalize to probability vector
    
    c = fast_conv(a)
    L, p = lse_smooth_max(c, beta)
    score = 2.0 * n * L / 1.0  # S=1 after normalization, so score = 2n*L
    
    n = len(a)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    
    # score = 2n * L (since S=1)
    # grad_a = 2n * g (since S=1, the -2L/S term vanishes in normalized space)
    # Actually with S=1: grad of 2n*L w.r.t. a is 2n * g
    # Then chain rule for log domain: grad_x = a * (grad_a - a @ grad_a)
    # But since we're working with normalized a, the constraint is handled by softmax.
    grad_a = 2.0 * n * g
    d = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - d)
    return score, grad_x


def multiplicative_update(a, n_iter=100, step=0.5, time_limit_s=None, t_start=None):
    """
    Multiplicative (EM-like) updates to reduce peak convolution.
    For each iteration, reduce elements that contribute to peak lags.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find top-k lags
        top_k = min(10, len(c))
        sorted_idx = np.argsort(c)[::-1][:top_k]
        c_max = float(c.max())
        # Weights: emphasize lags close to max
        weights = np.exp((c[sorted_idx] - c_max) * 5.0)
        weights /= weights.sum()
        
        # Compute pressure on each element
        pressure = np.zeros(n)
        for w, k in zip(weights, sorted_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                pressure[js:je + 1] += w * a[idx]
        
        # Normalize pressure
        avg_pressure = pressure.mean()
        if avg_pressure < 1e-15:
            break
        
        # Multiplicative update: reduce high-pressure elements, increase low-pressure
        ratio = 1.0 - step * (pressure / avg_pressure - 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        a = a * ratio
        a = np.clip(a, 1e-15, 1e10)
        
        # Re-normalize
        a = a / a.sum()
    
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_score, best_a


def log_domain_adam(a_init, beta_schedule=None, lr=0.01, n_steps=50000,
                    time_limit_s=None, t_start=None, verbose_interval=10000):
    """
    Adam optimizer in log-domain with adaptive LSE beta.
    """
    n = len(a_init)
    a_init_safe = np.clip(a_init, 1e-10, 1e10)
    x = np.log(a_init_safe)
    x = x - x.mean()  # center
    
    if beta_schedule is None:
        beta_schedule = [(0, 50), (n_steps//4, 100), (n_steps//2, 200),
                         (3*n_steps//4, 500), (int(0.9*n_steps), 1000)]
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a_init_safe.copy()
    best_score = compute_score(a_init_safe)
    
    # Build beta schedule as (step, beta) pairs
    beta_schedule = sorted(beta_schedule, key=lambda t: t[0])
    beta_idx = 0
    current_beta = beta_schedule[0][1] if beta_schedule else 100
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Update beta
        while beta_idx + 1 < len(beta_schedule) and t >= beta_schedule[beta_idx + 1][0]:
            beta_idx += 1
        current_beta = beta_schedule[beta_idx][1]
        
        # Adaptive learning rate
        lr_t = lr * (1.0 / (1.0 + 0.01 * t))
        
        score, grad_x = gradient_lse_x(x, current_beta)
        
        m = b1 * m + (1.0 - b1) * grad_x
        v = b2 * v + (1.0 - b2) * grad_x * grad_x
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        if t % verbose_interval == 0:
            a_cur = softmax_of(x)
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                print(f"    LogAdam t={t}, beta={current_beta}: score={sc:.8f}")
    
    a_final = softmax_of(x)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_score, best_a


def softmax_of(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def run_lbfgs_log(x0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B in log-domain."""
    res = minimize(
        gradient_lse_x, x0, args=(beta,),
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = softmax_of(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.x


def run_lbfgs_direct(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B in direct a-space with bounds."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        gradient_lse_a, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def peak_focusing_descent(a, n_iter=200, time_limit_s=None, t_start=None):
    """
    Directly target peak lags: for each peak lag k*, find the dominant
    pair (j, k*-j) and transfer mass from that pair to non-peak positions.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = float(a.sum())
        score = 2.0 * n * float(c.max()) / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find the peak lag
        k_star = int(c.argmax())
        
        # Find the dominant contributing pairs for this lag
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            continue
        
        # Contributions: a[j] * a[k_star - j] for j in [js, je]
        idx = k_star - np.arange(js, je + 1)
        contribs = a[js:je+1] * a[idx]
        
        # Find the top contributor
        top_local = int(np.argmax(contribs))
        j_top = js + top_local
        j_other = k_star - j_top
        
        # Reduce the dominant pair, increase non-peak elements
        # Choose target: element furthest from any peak lag contributor
        all_peak_elems = set()
        top_k_lags = np.argsort(c)[::-1][:5]
        for k in top_k_lags:
            k = int(k)
            js2 = max(0, k - n + 1)
            je2 = min(k, n - 1)
            all_peak_elems.update(range(js2, je2 + 1))
        
        # Find elements NOT in peak sets
        non_peak_mask = np.ones(n, dtype=bool)
        for k in top_k_lags:
            k = int(k)
            js2 = max(0, k - n + 1)
            je2 = min(k, n - 1)
            non_peak_mask[js2:je2+1] = False
        
        if non_peak_mask.sum() == 0:
            # All elements are in peak sets; use random target
            target = int(np.random.randint(n))
        else:
            non_peak_indices = np.where(non_peak_mask)[0]
            # Target the largest non-peak element (to create more balance)
            target = non_peak_indices[np.argmax(a[non_peak_indices])]
        
        # Transfer mass
        transfer = 0.01 * max(a[j_top], a[j_other])
        a[j_top] -= transfer * 0.5
        if j_other != j_top:
            a[j_other] -= transfer * 0.5
        a[target] += transfer
        
        a = np.clip(a, 0.0, 1e10)
    
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_score, best_a


def simulated_annealing(a, n_iter=5000, T_init=0.01, T_end=0.0001,
                        time_limit_s=None, t_start=None, verbose_interval=2000):
    """
    Simulated annealing with structure-aware perturbations.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf
    rng = np.random.default_rng(12345)
    
    for it in range(1, n_iter + 1):
        if time.monotonic() > deadline:
            break
        
        T = T_init * (T_end / T_init) ** (it / n_iter)
        score = 2.0 * n * float(fast_conv(a).max()) / (float(a.sum()) ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Generate perturbation
        strategy = rng.integers(4)
        a_new = a.copy()
        
        if strategy == 0:
            # Gaussian perturbation
            scale = T * 0.1
            a_new = a * (1.0 + scale * rng.standard_normal(n))
        elif strategy == 1:
            # Block modification
            block_size = max(1, n // 100)
            start = int(rng.integers(0, n - block_size))
            a_new[start:start+block_size] *= rng.uniform(0.5, 1.5)
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = rng.uniform(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) / n
            a_new = a * (1.0 + T * np.cos(2 * np.pi * freq * x + phase))
        else:
            # Swap mass between random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            amt = T * a.max()
            a_new[i] -= amt
            a_new[j] += amt
        
        a_new = np.clip(a_new, 0.0, 1e10)
        
        new_score = 2.0 * n * float(fast_conv(a_new).max()) / (float(a_new.sum()) ** 2)
        delta = new_score - score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
        
        if it % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    SA it={it}, T={T:.6f}: best={best_score:.8f}")
    
    return best_score, best_a


def geometric_sequence(n, r):
    """Generate a normalized geometric sequence."""
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def sum_of_geometrics(n, params):
    """Generate a sum of weighted geometric sequences. params = [(w_i, r_i), ...]"""
    a = np.zeros(n)
    for w, r in params:
        a += w * r ** np.arange(n, dtype=np.float64)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-15:
        a[:] = 1.0
    return a / a.sum()


def sinusoidal_sequence(n, f1, f2, a1, a2, phase1, phase2):
    """Generate a sequence with sinusoidal modulation."""
    x = np.arange(n, dtype=np.float64) / n
    a = 1.0 + a1 * np.cos(2 * np.pi * f1 * x + phase1) + a2 * np.cos(2 * np.pi * f2 * x + phase2)
    a = np.clip(a, 1e-6, None)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):

                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        parent_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    else:
        parent = geometric_sequence(4096, 0.995)
        parent_score = compute_score(parent)
        print(f"No parent, using geometric n=4096, score={parent_score:.8f}")
    
    best_score = parent_score
    best_a = parent.copy()
    
    def try_update(sc, a):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    def time_remaining():
        return effective_budget - (time.monotonic() - t_start)
    
    # ============================================================
    # Phase 1: Generate diverse starting points (5% time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse starting points")
    print(f"{'='*60}")
    
    candidates = [best_a.copy()]
    n_base = len(best_a)
    
    # Geometric sequences
    for r in [0.99, 0.995, 0.997, 0.999]:
        g = geometric_sequence(n_base, r)

        a_rand = np.convolve(a_rand, np.ones(10)/10, mode='same')

        candidates.append(a_rand / a_rand.sum())
    
    # Evaluate all candidates
    for i, cand in enumerate(candidates):
        sc = compute_score(cand)
        print(f"  Candidate {i}: n={len(cand)}, score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"  -> New best from candidate {i}")
    
    # ============================================================
    # Phase 2: Log-domain Adam with adaptive beta (25% time)

    print(f"Phase 2: Log-domain Adam (adaptive beta)")
    print(f"{'='*60}")

            best_a, n_iter=20000, T_init=0.005, T_end=0.0001,
            time_limit_s=min(p6_budget * 0.3, 60.0), t_start=t_start
        )
        try_update(sc, a_opt)
        print(f"    SA: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Restarts with diverse perturbations
    n_best = len(best_a)
    n_restarts = 15
    
    for r in range(n_restarts):
        if time.monotonic() > restart_deadline:
            break
        
        strategy = r % 5
        if strategy == 0:
            # Gaussian
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Peak reduction: reduce elements at peak lag
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            a_pert[js:je+1] *= rng.uniform(0.4, 0.7)
            # Redistribute
            removed = float(best_a[js:je+1].sum() - a_pert[js:je+1].sum())
            mask = np.ones(n_best, dtype=bool)
            mask[js:je+1] = False
            if mask.sum() > 0:
                a_pert[mask] += removed / mask.sum()
        elif strategy == 2:
            # Multi-frequency sinusoidal perturbation
            f1 = rng.uniform(0.5, 10)
            f2 = rng.uniform(5, 20)
            x = np.arange(n_best) / n_best
            a_pert = best_a * (1.0 + 0.03*np.cos(2*np.pi*f1*x)
                               + 0.02*np.cos(2*np.pi*f2*x + rng.uniform(0, 6.28)))
        elif strategy == 3:
            # Block swap
            a_pert = best_a.copy()
            b1_start = int(rng.integers(0, n_best//2))
            b1_end = b1_start + max(10, n_best//20)
            b2_start = int(rng.integers(n_best//2, n_best - max(10, n_best//20)))
            b2_end = b2_start + (b1_end - b1_start)
            if b2_end < n_best:
                a_pert[b1_start:b1_end], a_pert[b2_start:b2_end] = \
                    a_pert[b2_start:b2_end].copy(), a_pert[b1_start:b1_end].copy()
        else:
            # Contraction/expansion of center
            a_pert = best_a.copy()
            center = n_best // 2
            width = n_best // 8
            a_pert[center-width:center+width] *= rng.uniform(0.5, 1.2)
            factor = best_a.sum() / a_pert.sum()
            a_pert *= np.clip(factor, 0.5, 2.0)
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick optimization: log-domain Adam + L-BFGS
        time_left = restart_deadline - time.monotonic()
        if time_left < 10:
            break
        adam_time = min(15.0, time_left * 0.4)
        sc, a_r = log_domain_adam(
            a_pert, beta_schedule=[(0, 100), (10000, 300)],
            lr=0.003, n_steps=50000,
            time_limit_s=adam_time, t_start=t_start
        )
        if time.monotonic() < restart_deadline - 5:
            sc2, a_r2, _ = run_lbfgs_direct(a_r, 1000, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_r = sc2, a_r2
        try_update(sc, a_r)
    
    # ============================================================
    # Phase 7: Try different lengths (5% time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 7: Different lengths")
    print(f"{'='*60}")
    
    n_parent = len(best_a)
    for factor in [0.7, 1.3, 1.7, 2.0]:
        if time.monotonic() > restart_deadline:
            break
        tl = int(n_parent * factor)
        if tl < 512 or tl > 50000:
            continue
        a_res = resample_sequence(best_a, tl)
        t0 = time.monotonic()
        sc, a_opt = log_domain_adam(
            a_res, beta_schedule=[(0, 100), (20000, 500)],
            lr=0.003, n_steps=50000,
            time_limit_s=min(20.0, max(5, restart_deadline - time.monotonic())),
            t_start=t_start
        )
        if time.monotonic() < restart_deadline - 3:
            sc2, a_opt2, _ = run_lbfgs_direct(a_opt, 1000, maxiter=2000, maxfun=30000)
            if sc2 < sc:
                sc, a_opt = sc2, a_opt2
        try_update(sc, a_opt)
        print(f"    n={tl} (factor={factor}): {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ============================================================
    # Phase 8: Final intensive polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 8: Final polish")
    print(f"{'='*60}")
    
    final_deadline = t_start + effective_budget - 5.0
    
    # High-beta L-BFGS
    for beta in [3000, 8000, 15000]:
        if time.monotonic() > final_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_direct(best_a, beta, maxiter=5000, maxfun=80000)
        try_update(sc, a_opt)
        print(f"    Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final multiplicative update
    if time.monotonic() < final_deadline - 10:
        t0 = time.monotonic()
        sc, a_opt = multiplicative_update(
            best_a, n_iter=1000, step=0.3,
            time_limit_s=20.0, t_start=t_start
        )
        try_update(sc, a_opt)
        print(f"    Final MultUpd: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Final peak-focusing
    if time.monotonic() < final_deadline - 5:
        t0 = time.monotonic()
        sc, a_opt = peak_focusing_descent(
            best_a, n_iter=2000,
            time_limit_s=15.0, t_start=t_start
        )
        try_update(sc, a_opt)
        print(f"    Final PeakFocus: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # One more high-precision L-BFGS
    if time.monotonic() < final_deadline - 3:
        t0 = time.monotonic()
        sc, a_opt, _ = run_lbfgs_direct(best_a, 20000, maxiter=10000, maxfun=100000)
        try_update(sc, a_opt)
        print(f"    Ultra LBFGS: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
