```python
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
```

```python
"""
Hybrid multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2.

Key improvements over previous approach:
- Diverse frequency-domain and structured initializations
- Gradient-guided coordinate ordering (using LSE gradient to prioritize coordinates)
- Adaptive LSE beta schedule with normalization
- Subgradient diversification steps
- Tighter 1D bounds in coordinate descent
- Better time management with early termination on convergence
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient (normalized for stability)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    # Normalize to keep numbers stable
    a_norm = a / S * np.sqrt(2.0 * n)
    c = fast_conv(a_norm)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = L  # In normalized units, score = max_b when sum = sqrt(2n)
    p = exp_scaled / exp_scaled.sum()
    # Gradient with respect to a_norm
    rev_a = a_norm[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    # Unnormalize gradient: da_norm/da = sqrt(2n)/S
    scale_back = np.sqrt(2.0 * n) / S
    grad = g * scale_back
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize (returns (f, grad))."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective with normalization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 1.0
        S0 = n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def cd_sweep(a, mode='random', rng=None):
    """
    One sweep of coordinate descent.
    mode: 'random', 'hot', 'gradient'
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if mode == 'hot':
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    elif mode == 'gradient':
        # Use LSE gradient to determine priority
        try:
            _, grad = lse_score_grad(a, 500.0)
            order = np.argsort(np.abs(grad))[::-1]
        except Exception:
            order = np.arange(n)
    elif rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for i in order:
        i = int(i)
        a_i = a[i]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib

        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i

        # Tighter upper bound based on current state
        if a_i > 1e-8:
            t_upper = max(2.5 * a_i + 0.005, S_rest / n * 2.5)
        else:
            t_upper = S_rest / n * 2.0
        t_upper = min(t_upper, 1000.0)

        # Also set a tighter lower bound if a_i is large
        t_lo = 0.0
        if a_i > S_rest / n * 5.0:
            t_lo = a_i * 0.5

        if t_upper <= t_lo:
            t_upper = t_lo + 0.01

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(t_lo, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    return compute_score(a), a


def subgradient_step(a, lr):
    """One step of projected subgradient descent on the exact max."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    S = float(a.sum())
    if S < 1e-12:
        return a

    # Subgradient of max(conv) w.r.t. a
    grad_max = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if j == i:
                grad_max[i] = 2.0 * a[i]
            else:
                grad_max[i] = a[j] + a[i]

    max_c = float(c[k_star])
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_max / S2 - 2.0 * max_c * a / S3)

    a_new = np.clip(a - lr * grad, 0.0, 1000.0)
    return a_new


def fg_subgradient(a, max_iters=200, time_limit=30.0):
    """Fessler-Golding style subgradient method with adaptive step size."""
    t0 = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()

    # Fessler-Golding parameterization
    # a = c / (1 + d) where d >= -1
    # Initialize d
    S = a.sum()
    d = np.zeros(n)  # start at d=0, so a = c
    c = S  # c is the "total" parameter

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a_curr = c / (1.0 + d)
        a_curr = np.clip(a_curr, 0.0, 1000.0)

        sc = compute_score(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

        # Compute gradient of score w.r.t. a
        conv = fast_conv(a_curr)
        k_star = int(np.argmax(conv))
        S_curr = float(a_curr.sum())
        if S_curr < 1e-12:
            break

        grad_max = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if j == i:
                    grad_max[i] = 2.0 * a_curr[i]
                else:
                    grad_max[i] = a_curr[j] + a_curr[i]

        max_c = float(conv[k_star])
        S2 = S_curr * S_curr
        S3 = S2 * S_curr
        grad_a = (2.0 * n) * (grad_max / S2 - 2.0 * max_c * a_curr / S3)

        # Chain rule: da/dc = 1/(1+d), da/dd = -c/(1+d)^2
        one_plus_d = np.maximum(1.0 + d, 1e-8)
        grad_c = np.sum(grad_a / one_plus_d)
        grad_d = -c * grad_a / (one_plus_d * one_plus_d)

        # Barzilai-Borwein step (simplified: use fixed decay)
        lr_c = c / (1.0 + 0.05 * (it % 10))
        lr_d = lr_c / max(c, 1e-8)

        c = max(c - lr_c * grad_c, 0.01)
        d = d - lr_d * grad_d
        d = np.clip(d, -0.99, 100.0)

    return best_score, best_a


def gen_init_uniform(n, rng):
    """Random uniform positive sequence."""
    return rng.uniform(0.1, 1.0, size=n)


def gen_init_cosine(n, rng):
    """Cosine-modulated sequence."""
    x = np.arange(n) / n
    freq = 1.0 + 2.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x + phase)
    return np.clip(a, 0.01, None)


def gen_init_gaussian(n, rng):
    """Gaussian-windowed sequence."""
    x = np.arange(n)
    center = n / 2.0
    width = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((x - center) / width) ** 2)
    a += 0.1  # avoid zeros
    return a


def gen_init_binary(n, rng):
    """Random binary-like sequence."""
    p = 0.3 + 0.4 * rng.random()
    a = (rng.random(n) < p).astype(float)
    a[rng.random(n) < 0.2] = 0.0  # add some zeros
    return np.maximum(a, 0.01)


def gen_init_flat_spectrum(n, rng):
    """Sequence with approximately flat power spectrum (in frequency domain)."""
    # Generate random phases, uniform magnitude
    N_fft = 1 << int(np.ceil(np.log2(n)))
    A = np.zeros(N_fft // 2 + 1, dtype=complex)
    A[0] = np.sqrt(n)  # DC
    phases = 2.0 * np.pi * rng.random(N_fft // 2)
    mags = np.ones(N_fft // 2)
    # Add some variation
    mags *= (0.7 + 0.6 * rng.random(N_fft // 2))
    A[1:] = mags * np.exp(1j * phases)
    # Make it real: A[k] = conj(A[N-k])
    A_full = np.zeros(N_fft, dtype=complex)
    A_full[:N_fft // 2 + 1] = A
    for k in range(1, N_fft // 2):
        A_full[N_fft - k] = np.conj(A[k])
    a = np.fft.irfft(A_full)[:n]
    # Shift to make non-negative
    a = a - a.min() + 0.1
    return np.clip(a, 0.0, None)


def gen_init_parent_perturb(parent, rng, scale=0.1):
    """Perturb the parent sequence."""
    n = len(parent)
    noise = rng.standard_normal(n) * scale * parent.mean()
    a = parent + noise
    return np.clip(a, 0.0, 1000.0)


def targeted_perturb(a, rng, strength=0.03):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    if parent is not None and parent.sum() > 0.01:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
        n_work = len(parent)
    else:
        print("No parent available, using length 4096.")
        n_work = 4096
        best_a = np.ones(n_work) * 0.5
        best_score = compute_score(best_a)
        print(f"Initial uniform score: {best_score:.8f}")

    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 0: Diverse initialization (5% of time)
    # ============================================================
    phase0_deadline = t_start + 0.05 * total_budget
    print(f"\n=== Phase 0: Initialization Screening ===")

    candidates = []
    if best_a is not None:
        candidates.append((best_score, best_a.copy()))

    inits = [
        ("cosine", gen_init_cosine(n_work, rng)),
        ("gaussian", gen_init_gaussian(n_work, rng)),
        ("binary", gen_init_binary(n_work, rng)),
        ("flat_spec", gen_init_flat_spectrum(n_work, rng)),
        ("perturb1", gen_init_parent_perturb(best_a, rng, 0.05)),
        ("perturb2", gen_init_parent_perturb(best_a, rng, 0.15)),
    ]

    for name, a_init in inits:
        if time.monotonic() > phase0_deadline:
            break
        a_init = np.clip(a_init, 0.0, 1000.0)
        if a_init.sum() < 0.01:
            continue
        # Quick screening: 2 CD sweeps
        sc = compute_score(a_init)
        for _ in range(2):
            if time.monotonic() > phase0_deadline:
                break
            sc, a_init = cd_sweep(a_init, mode='hot', rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_init.copy()
            print(f"  {name}: {sc:.8f} (NEW BEST)")
        else:
            print(f"  {name}: {sc:.8f}")
        candidates.append((sc, a_init.copy()))

    print(f"Phase 0 done: best={best_score:.8f}")

    # ============================================================
    # Phase 1: LSE optimization with decaying beta (35% of time)
    # ============================================================
    phase1_deadline = t_start + 0.40 * total_budget
    print(f"\n=== Phase 1: LSE Optimization ===")

    betas = [5000, 2000, 1000, 500, 200, 100]
    a_cur = best_a.copy()

    for beta in betas:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=3000, maxfun=50000)
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a_cur = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            # Still use a_cur from LSE output for next iteration (it might be closer)
            a_cur = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

        # CD refinement after each LSE
        for _ in range(2):
            if time.monotonic() > phase1_deadline:
                break
            sc_cd, a_cur = cd_sweep(a_cur, mode='hot', rng=rng)
            if sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cur.copy()

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: Intensive CD with gradient-guided ordering (30% of time)
    # ============================================================
    phase2_deadline = t_start + 0.70 * total_budget
    print(f"\n=== Phase 2: Intensive CD ===")

    a_cur = best_a.copy()
    prev_score = best_score
    no_improve = 0
    sweep_count = 0

    while time.monotonic() < phase2_deadline:
        sweep_count += 1
        # Cycle through modes: hot, random, gradient, hot, random, gradient...
        mode = ['hot', 'random', 'gradient', 'hot', 'random', 'gradient'][sweep_count % 6]
        sc, a_cur = cd_sweep(a_cur, mode=mode, rng=rng)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 20 == 0:
                print(f"  CD sweep {sweep_count} ({mode}): {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 10 == 0:
            print(f"  CD sweep {sweep_count} ({mode}): {sc:.8f} (no_improve={no_improve})")

        # If stuck, try subgradient or perturbation
        if no_improve >= 30:
            if rng.random() < 0.5:
                # Subgradient step
                lr = 0.5 / (sweep_count + 10)
                a_cur = subgradient_step(a_cur, lr)
                a_cur = np.clip(a_cur, 0.0, 1000.0)
            else:
                # Targeted perturbation
                a_cur = targeted_perturb(a_cur, rng, strength=0.01 + 0.02 * rng.random())
            no_improve = 0
            print(f"  Stuck at sweep {sweep_count}, diversifying...")

    print(f"Phase 2 done: {sweep_count} sweeps, best={best_score:.8f}")

    # ============================================================
    # Phase 3: Fessler-Golding subgradient (10% of time)
    # ============================================================
    phase3_deadline = t_start + 0.80 * total_budget
    print(f"\n=== Phase 3: Fessler-Golding Subgradient ===")

    if time.monotonic() < phase3_deadline:
        fg_time = min(60.0, phase3_deadline - time.monotonic())
        sc_fg, a_fg = fg_subgradient(best_a, max_iters=500, time_limit=fg_time)
        if sc_fg < best_score:
            best_score = sc_fg
            best_a = a_fg.copy()
            print(f"  FG: {sc_fg:.8f} (NEW BEST)")
        else:
            print(f"  FG: {sc_fg:.8f}")
            # CD from FG result
            a_cur = a_fg.copy()
            for _ in range(3):
                if time.monotonic() > phase3_deadline:
                    break
                sc_cd, a_cur = cd_sweep(a_cur, mode='hot', rng=rng)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_a = a_cur.copy()
                    print(f"  FG+CD: {sc_cd:.8f} (NEW BEST)")

    # ============================================================
    # Phase 4: Diversification cycles (15% of time)
    # ============================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Diversification ===")

    for trial in range(50):
        if time.monotonic() > phase4_deadline - 5.0:
            break

        strategy = trial % 6
        n_cur = len(best_a)

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.02 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_cur))
        elif strategy == 1:
            # Targeted perturbation (attack peak)
            a_pert = targeted_perturb(best_a, rng, strength=0.01 + 0.03 * rng.random())
        elif strategy == 2:
            # Block zeroing + redistribution
            block_size = max(5, n_cur // 200)
            start = int(rng.integers(0, n_cur - block_size))
            a_pert = best_a.copy()
            saved = a_pert[start:start + block_size].sum()
            a_pert[start:start + block_size] = 0.0
            rest_mask = np.ones(n_cur, dtype=bool)
            rest_mask[start:start + block_size] = False
            a_pert[rest_mask] += saved / max(1, int(rest_mask.sum()))
        elif strategy == 3:
            # Segment swap
            seg_len = max(5, n_cur // 50)
            s1 = int(rng.integers(0, max(1, n_cur - seg_len)))
            s2 = int(rng.integers(0, max(1, n_cur - seg_len)))
            a_pert = best_a.copy()
            tmp = a_pert[s1:s1 + seg_len].copy()
            a_pert[s1:s1 + seg_len] = a_pert[s2:s2 + seg_len]
            a_pert[s2:s2 + seg_len] = tmp
        elif strategy == 4:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            x = np.arange(n_cur, dtype=np.float64) / n_cur
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + 2.0 * np.pi * rng.random()))
        else:
            # Mirror + blend
            a_flipped = best_a[::-1].copy()
            mix = 0.3 + 0.4 * rng.random()
            a_pert = (1.0 - mix) * best_a + mix * a_flipped

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 0.01:
            continue

        # Quick CD from perturbed point
        n_sw = 5
        for sw in range(n_sw):
            if time.monotonic() > phase4_deadline - 5.0:
                break
            if sw % 2 == 0:
                sc_pert, a_pert = cd_sweep(a_pert, mode='hot', rng=rng)
            else:
                sc_pert, a_pert = cd_sweep(a_pert, mode='random', rng=rng)

        sc_pert = compute_score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f} (NEW BEST)")
        elif trial % 10 == 0:
            print(f"  P4 trial {trial} (s={strategy}): {sc_pert:.8f}")

    # ============================================================
    # Final polish (5% of time)
    # ============================================================
    print(f"\n=== Final Polish ===")
    a_cur = best_a.copy()

    # Multiple CD sweeps
    for _ in range(30):
        if time.monotonic() > deadline - 5.0:
            break
        sc, a_cur = cd_sweep(a_cur, mode='hot', rng=rng)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

    # One final LSE with very high beta (nearly exact)
    if time.monotonic() < deadline - 20.0:
        sc, a_opt, _ = run_lbfgs(best_a, 8000, maxiter=3000, maxfun=40000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Final LSE: {sc:.8f} (NEW BEST)")
            # CD after LSE
            a_cur = a_opt.copy()
            for _ in range(5):
                if time.monotonic() > deadline - 3.0:
                    break
                sc, a_cur = cd_sweep(a_cur, mode='hot', rng=rng)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
