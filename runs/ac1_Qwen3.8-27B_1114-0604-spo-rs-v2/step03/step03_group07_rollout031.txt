"""
Optimized step function search minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LSE-continuation L-BFGS with correct gradient, Nesterov accelerated
subgradient polish, anti-palindromic post-processing, and diverse inits.
Key improvements: verified gradient formula via cross-correlation, Nesterov
acceleration for faster subgradient convergence, scale-invariance exploitation,
and more aggressive beta continuation schedule.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_value_grad(a, beta):
    """
    Compute value and gradient of f(a) = 2n * LSE_beta(conv(a,a)) / (sum a)^2.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.sum()) / beta

    # Cross-correlation: (p cross a)[m] = sum_k p[k] * a[k-m]
    M = 3 * n - 2
    M_fft = 1 << (M - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    cross_corr = np.fft.irfft(P * np.conj(A), M_fft)[:n]

    # dLSE/da_m = 4n * cross_corr[m]
    dLSE = 4.0 * n * cross_corr

    # f = 2n * L / S^2
    val = 2.0 * n * L / (S * S)
    # df/da_m = (2n/S^2) * (dLSE_m - 2L/S)
    grad = (2.0 * n / (S * S)) * (dLSE - 2.0 * L / S)

    return val, grad


def run_lbfgs(a0, beta, maxfun, ftol=1e-18, gtol=1e-12):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (score, array)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(a):
        return lse_value_grad(a, beta)

    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': maxfun // 5, 'maxfun': maxfun,
                                'ftol': ftol, 'gtol': gtol, 'maxls': 50})
    except Exception:
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a0, max_iters, time_limit):
    """Nesterov accelerated subgradient with Polyak step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    y = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    t_prev = 2.0 / 2.0

    for k in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(y)
        S = float(y.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        # Subgradient: df/da_m = (2n/S^2) * (2*y[k_star-m] - 2M/S)
        sg = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(k_star, n - 1)
        if m_end >= m_start:
            idx_start = k_star - m_end
            idx_end = k_star - m_start
            sg[m_start:m_end + 1] = 2.0 * y[idx_start:idx_end + 1][::-1]
        sg = (2.0 * n / (S * S)) * (sg - 2.0 * M / S)

        gn2 = float(sg @ sg)
        if gn2 < 1e-30:
            break

        # Polyak step with damping
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (np.sqrt(k + 1) * 0.5))
        step = max(step, 1e-10)

        # Nesterov update
        t_k = 2.0 / (k + 2)
        x_new = y - step * sg
        x_new = np.clip(x_new, 0.0, 1000.0)

        if k == 0:
            y = x_new.copy()
        else:
            momentum = t_prev / t_k - 1.0
            y = x_new + momentum * (x_new - a)
            y = np.clip(y, 0.0, 1000.0)
        a = x_new
        t_prev = t_k

    # Also return the Nesterov-averaged point
    sc_end = compute_score(a)
    if sc_end < best_sc:
        best_sc = sc_end
        best_a = a.copy()

    """Create anti-palindromic version: a[j] + a[n-1-j] = C (max pair sum)."""

            width = max(1, n_best // 10)
            for j in range(n_best):
                d = abs(j - center) / width
                boost = 0.05 * np.exp(-d * d) if j < center else -0.05 * np.exp(-d * d)
                a_pert[j] *= (1.0 + boost)
        elif strat == 2:
            # Frequency domain: attenuate DC (make more spread)
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            freqs = np.arange(len(A))
            att = 1.0 - 0.15 * np.exp(-freqs / (len(A) * 0.1))
            att[0] = 1.0  # Keep DC
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]
        elif strat == 3:
            # Add small anti-palindromic noise
            noise = rng.standard_normal(n_best) - rng.standard_normal(n_best)[::-1]
            a_pert = best_a * (1.0 + 0.03 * noise)
        else:
            # Targeted: reduce elements at positions contributing to peak
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            m_start = max(0, k_star - n_best + 1)
            m_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            if m_end >= m_start:
                contrib = np.zeros(n_best)
                for m in range(m_start, m_end + 1):
                    contrib[m] = best_a[m] * best_a[k_star - m]
                top_n = max(5, n_best // 200)
                top_idx = np.argsort(contrib)[::-1][:top_n]
                bot_idx = np.argsort(contrib)[:top_n]
                red = 0.0
                for m in top_idx:
                    amt = a_pert[m] * 0.05
                    a_pert[m] -= amt
                    a_pert[m] = max(a_pert[m], 0.0)
                    red += amt
                if red > 0:
                    for m in bot_idx:
                        a_pert[m] += red / len(bot_idx)

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a.copy()

        # Quick L-BFGS from perturbation
        time_left = deadline - 20 - time.monotonic()
        if time_left < 20:
            break
        maxfun = estimate_maxfun(n_best, min(30, time_left))
        sc, a_opt = run_lbfgs(a_pert, beta=5000, maxfun=maxfun, gtol=1e-10)
        report(sc, a_opt, f"P5 #{trial} s={strat}")

        # Also try direct score without further optimization
        sc_direct = compute_score(a_pert)
        report(sc_direct, a_pert, f"P5 #{trial} direct")

    # ============================================================
    # Final subgradient polish
    # ============================================================
    if time.monotonic() < deadline - 5:
        sc, a_opt = nesterov_subgrad(best_a, max_iters=100000,
                                     time_limit=deadline - time.monotonic() - 5)
        report(sc, a_opt, "final")

    # ============================================================
    # Post-processing: try anti-palindromic and palindromic versions
    # ============================================================
    a_anti = antipalindromic_reflect(best_a)
    sc_anti = compute_score(a_anti)
    report(sc_anti, a_anti, "post anti")

    a_palin = palindromic_average(best_a)
    sc_palin = compute_score(a_palin)
    report(sc_palin, a_palin, "post palin")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
