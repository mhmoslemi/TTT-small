Hybrid optimizer: Adam-based multiplicative LSE + exact coordinate descent + Fourier perturbations.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_grad_logspace(a, beta):
    """
    Compute the gradient of the LSE score w.r.t. log(a) (i.e., a_i * df/da_i).
    Also returns the LSE score.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta

    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_score = compute_score(a)
    
    for it in range(n_iter):
        score, g = lse_grad_logspace(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**(it + 1))
        v_hat = v / (1 - beta2**(it + 1))
        
        # Multiplicative update
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        log_factor = np.clip(-step, -5.0, 5.0)  # prevent overflow
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-15)
        
        # Renormalize
        a = a * (S_target / float(a.sum()))
        
        if (it + 1) % 200 == 0:
            exact = compute_score(a)
            if exact < best_score:
                best_score = exact
    
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
    return best_score, a

def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)

def cd_sweep(a, hot=False, rng=None):
    """One coordinate descent sweep."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if hot:
        k_star = int(np.argmax(conv))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        order = np.argsort(contrib)[::-1]
    elif rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    # Maintain a prefix/suffix max for outside_max computation
    # For efficiency, precompute prefix max and suffix max
    prefix_max = np.maximum.accumulate(conv)
    suffix_max = np.maximum.accumulate(conv[::-1])[::-1]
    
    for idx in range(n):
        i = int(order[idx])
        a_i = a[i]
        
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)
        
        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(prefix_max[k_lo - 1], suffix_max[k_hi + 1])
        elif k_lo > 0:
            outside_max = prefix_max[k_lo - 1]
        elif k_hi < M_conv - 1:
            outside_max = suffix_max[k_hi + 1]
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i
        base_inside -= old_contrib
        
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)
        
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)
        
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt *
