```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S*S)

def lp_refine_step(a, M, method='highs'):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    Returns the optimal b.
    """
    n = len(a)
    num_lags = 2*n - 1
    
    # Build constraint matrix A (sparse)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Row k, column j: a[k-j] if valid
    
    # For efficiency, build in COO format
    rows = []
    cols = []
    data = []
    
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    A_mat = csr_matrix((data, (rows, cols)), shape=(num_lags, n))
    
    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(num_lags, M), 
                  bounds=bounds, method=method)
    
    if res.success:
        return res.x
    return None
```

**Alternative fast approach: "Coordinate LP"**

Instead of solving the full LP, I can do coordinate descent on the LP:
1. Fix all b[j] except b[j0]
2. Find the maximum valid value of b[j0] (given by the tightest constraint)
3. Set b[j0] to that maximum
4. Repeat for all j

This is like a greedy packing algorithm. It's O(n * (2n)) = O(n^2) per sweep, which for n=6962 is about 48M operations per sweep. With 100 sweeps, that's 4.8 billion operations, which should take maybe 50-100 seconds.

The constraint for b[j0] is:
For each k where 0 ≤ k-j0 < n (i.e., k = j0, j0+1, ..., j0+n-1):
  a[k-j0] * b[j0] ≤ M - sum_{j≠j0} a[k-j] * b[j]
  b[j0] ≤ (M - sum_{j≠j0} a[k-j] * b[j]) / a[k-j0]

So the maximum valid b[j0] = min over all valid k of (M - sum_{j≠j0} a[k-j]*b[j]) / a[k-j0].

This can be computed in O(n) per coordinate (checking n constraints). Total: O(n^2) per sweep.

This is much faster than solving the full LP! And it gives a feasible (though not necessarily optimal) solution.

```python
def greedy_lp(a, M, n_sweeps=50):
    """Greedy coordinate descent for the LP: max sum(b) s.t. (a*b) <= M, b >= 0"""
    n = len(a)
    b = np.zeros(n)
    conv = np.zeros(2*n - 1)  # (a*b) at current b
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Find max valid b[j]
            max_bj = np.inf
            k_start = max(0, j)  # k such that k-j >= 0, i.e., k >= j
            k_end = min(2*n-2, j + n - 1)  # k such that k-j < n, i.e., k < j+n
            
            for k in range(k_start, k_end + 1):
                idx = k - j  # this is the index into a
                if a[idx] <= 0:
                    continue
                remaining = M - conv[k] + a[idx] * b[j]  # current conv[k] includes a[idx]*b[j]
                # Wait, I need to be more careful
                # conv[k] = sum_j' a[k-j'] * b[j']
                # If I set b[j] = v, then conv[k] changes by a[k-j]*(v - b[j])
                # Constraint: conv[k] + a[k-j]*(v - b[j]) <= M
                # => v <= b[j] + (M - conv[k]) / a[k-j]
                if a[idx] > 0:
                    max_bj = min(max_bj, b[j] + (M - conv[k]) / a[idx])
            
            max_bj = max(0, max_bj)
            # Update conv
            if max_bj != b[j]:
                for k in range(k_start, k_end + 1):
                    conv[k] += a[k-j] * (max_bj - b[j])
                b[j] = max_bj
    
    return b
```

```python
def build_constraint_matrix(a):
    n = len(a)
    num_lags = 2 * n - 1
    # Row k, Col j: a[k-j] if 0 <= k-j < n
    # For row k: valid j are max(0, k-n+1) to min(k, n-1)
    # a[k-j] for j in that range = a[max(0, k-n+1) to min(k,n-1)] reversed... 
    # Actually a[k-j] for j = j_start, ..., j_end gives a[k-j_start], ..., a[k-j_end]
    # = a[k - j_start], a[k - j_start - 1], ..., a[k - j_end]
    # = a[min(k, n-1)], a[min(k,n-1)-1], ..., a[max(0, k-n+1)]
    
    rows = []
    cols = []
    data = []
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (rows, cols)), shape=(num_lags, n))
```

- W variables
- 2n-1 constraints (but many will be non-binding if the window is small)
- Actually, only the constraints that involve the window coordinates will be affected

For a window of W coordinates starting at position s:
- Variables: b[s], b[s+1], ..., b[s+W-1]
- Constraint (a*b)[k] ≤ M: only the terms involving j in [s, s+W-1] can change
- The "constant" part is sum_{j not in window} a[k-j]*b[j] (fixed)
- The "variable" part is sum_{j in window} a[k-j]*b[j]

So the LP constraint becomes:
sum_{j in window} a[k-j]*b[j] ≤ M - sum_{j not in window} a[k-j]*b[j]

For constraints where the window doesn't contribute (i.e., no j in the window has 0 ≤ k-j < n), the constraint is automatically satisfied.

```python
"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-directed descent (greedy feasible LP), LSE-continuation L-BFGS,
accelerated subgradient, multiplicative (mirror descent) updates, and
targeted local search. Key innovation: greedy LP maintains a feasible
direction with bounded cross-convolution, providing strong descent
directions different from pure subgradient.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, coo_matrix


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_wrapper(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, timeout=60.0):
    """Run L-BFGS-B with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    try:
        res = minimize(
            lse_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-13,
                     'maxiter': maxiter, 'maxfun': maxiter * 20, 'maxls': 40}
        )
    except Exception:
        return compute_score(a0), a0, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out, res.success


def greedy_lp(a, M, max_sweeps=500, max_step_frac=0.1, rng=None):
    """
    Greedy LP: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Maintains feasibility at every step by only increasing b[j]
    to the maximum allowed by all constraints simultaneously.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    b = np.zeros(n)
    conv = np.zeros(2 * n - 1)
    total_inc = 0.0

    for sweep in range(max_sweeps):
        order = rng.permutation(n)
        swept = 0
        for j in order:
            idx = min(n, 2 * n - 1 - j)
            a_seg = a[:idx]
            mask = a_seg > 1e-15
            if not mask.any():
                continue
            conv_seg = conv[j:j + idx]
            remaining = M - conv_seg[mask]
            candidates = remaining / a_seg[mask]
            pot = float(candidates.min())
            if pot <= 1e-12:
                continue
            step = min(pot, max_step_frac * (M / max(a.max(), 1e-15)))
            if step <= 1e-15:
                continue
            b[j] += step
            conv[j:j + idx] += a[:idx] * step
            total_inc += step
            swept += 1
        if swept == 0:
            break
    return b


def greedy_lp_multi_start(a, M, n_starts=5, max_sweeps=300, rng=None):
    """Run greedy LP multiple times with different orderings, keep best sum."""
    if rng is None:
        rng = np.random.default_rng()
    best_b = None
    best_sum = -1
    for _ in range(n_starts):
        b = greedy_lp(a, M, max_sweeps=max_sweeps,
                      max_step_frac=0.05 + 0.15 * rng.random(), rng=rng)
        s = b.sum()
        if s > best_sum:
            best_sum = s
            best_b = b
    return best_b


def subgrad_polish(a, max_iters=5000, sc_lb=None):
    """Accelerated subgradient descent with Polyak step."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    if sc_lb is None:
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
    y = a.copy()
    gamma = 1.0
    for it in range(max_iters):
        a = (1 - 2.0 / (it + 1 + gamma)) * y + (2.0 / (it + 1 + gamma)) * a
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 1.0 / (0.01 + 0.002 * it))
        y = np.clip(a - step * grad, 0.0, 1000.0)
    return best_sc, best_a


def multiplicative_updates(a, max_iters=2000, rate=0.1):
    """Mirror descent with entropic kernel (multiplicative updates)."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 1e-10, 1000.0).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        step = rate / (1.0 + 0.01 * it)
        a = a * np.exp(-step * grad)
        a = np.clip(a, 1e-10, 1000.0)
        a = a / a.sum() * best_a.sum()
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Diverse initializations with anti-palindromic emphasis."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xn = x / (n - 1) if n > 1 else x

    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))
    inits.append(0.1 + 0.9 * xn ** 2)
    inits.append(0.1 + 0.9 * (1 - xn) ** 2)
    for freq in [1, 2, 3]:
        inits.append(np.clip(0.5 + 0.3 * np.sin(np.pi * freq * xn), 0.01, 10))
    inits.append(np.clip(0.5 + 0.2 * np.cos(2 * np.pi * 5 * xn), 0.01, 10))
    inits.append(np.clip(0.3 + 0.3 * xn + 0.2 * np.sin(2 * np.pi * 13 * xn), 0.01, 10))
    a_rand = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 10)
    inits.append(a_rand)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10))
    half = n // 2
    inits.append(np.concatenate([np.ones(half) * 0.9, np.ones(n - half) * 0.3]))
    inits.append(np.concatenate([np.ones(half) * 0.3, np.ones(n - half) * 0.9]))
    return inits


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_count = max(5, n // 80)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_idx = np.argsort(contrib)[:-top_count]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] = max(a_pert[j] - amt, 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            a_pert[bottom_idx] += total_red / len(bottom_idx)
    return np.clip(a_pert, 0.0, 1000.0)


def lp_refine_phase(a, rng, time_budget):
    """
    LP-directed refinement: iteratively solve greedy LP to find descent
    directions, blend with current solution.
    """
    n = len(a)
    t0 = time.monotonic()
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    S = float(a.sum())
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        c = fast_conv(a)
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Solve greedy LP
        b = greedy_lp(a, M, max_sweeps=200, max_step_frac=0.15, rng=rng)
        b_sum = float(b.sum())
        if b_sum < 1e-6:
            break

        # Blend with decreasing step
        t_step = 0.3 / (1.0 + 0.1 * it)
        a_new = (1.0 - t_step) * a + t_step * b * (S / b_sum)
        a_new = np.clip(a_new, 0.0, 1000.0)
        sc_new = compute_score(a_new)
        if sc_new < sc:
            a = a_new
            S = float(a.sum())
        else:
            # Try smaller step
            t_step = t_step * 0.3
            a_new = (1.0 - t_step) * a + t_step * b * (S / b_sum)
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc_new = compute_score(a_new)
            if sc_new < sc:
                a = a_new
                S = float(a.sum())
            else:
                # No progress, try subgradient step instead
                break

        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_sc:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_sc, best_a

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Choose lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [6962, 8192, 4096]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths), key=lambda x: -x)[:3]
    # Put parent length first
    if parent is not None:
        lengths = [parent.size] + [l for l in lengths if l != parent.size]
    lengths = lengths[:3]

    print(f"Lengths: {lengths}, Parent: {'yes (n=%d)' % parent.size if parent is not None else 'no'}")

    # ============ Phase 1: L-BFGS from structured inits (25% time) ============
    phase1_end = t_start + 0.25 * total_budget
    print(f"\n=== Phase 1: L-BFGS (until {phase1_end - t_start:.0f}s) ===")
    beta_schedule = [50, 150, 400, 1000, 3000]

    for n in lengths[:2]:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        n_inits_use = min(len(inits), 6)
        for i in range(n_inits_use):
            if time.monotonic() > phase1_end:
                break
            a_cur = inits[i].copy()
            for beta in beta_schedule[:3]:
                if time.monotonic() > phase1_end:
                    break
                remaining = phase1_end - time.monotonic()
                maxit = max(200, int(remaining / 0.05))
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit)
                a_cur = a_opt
                report(sc, a_opt, f"P1 n={n} i={i} b={beta}")

    print(f"  After P1: best = {best_score:.8f}")

    # ============ Phase 2: LP-directed refinement (40% time) ============
    phase2_end = t_start + 0.65 * total_budget
    print(f"\n=== Phase 2: LP Refinement (until {phase2_end - t_start:.0f}s) ===")
    n_best = len(best_a)
    time_left_p2 = phase2_end - time.monotonic()
    sc, a_opt = lp_refine_phase(best_a, rng, time_budget=max(0, time_left_p2))
    report(sc, a_opt, "P2 LP")

    # If LP didn't help, try L-BFGS more
    if time.monotonic() < phase2_end:
        n_best = len(best_a)
        for beta in [2000, 5000, 10000]:
            if time.monotonic() > phase2_end:
                break
            remaining = phase2_end - time.monotonic()
            maxit = max(500, int(remaining / 0.03))
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit)
            report(sc, a_opt, f"P2b beta={beta}")

    print(f"  After P2: best = {best_score:.8f}")

    # ============ Phase 3: Subgradient + multiplicative (15% time) ============
    phase3_end = t_start + 0.80 * total_budget
    print(f"\n=== Phase 3: Subgrad + Multiplicative ===")

    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        n_iters = max(1000, int(remaining * 80))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "P3 subgrad")

    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        n_iters = max(500, int(remaining * 50))
        sc, a_opt = multiplicative_updates(best_a, max_iters=n_iters, rate=0.05)
        report(sc, a_opt, "P3 multiplicative")

    # ============ Phase 4: Local search (10% time) ============
    phase4_end = min(phase3_end + 0.10 * total_budget, deadline - 30)
    print(f"\n=== Phase 4: Local Search ===")
    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase4_end and trial < 25:
        trial += 1
        strat = trial % 5
        if strat == 0:
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 2:
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 12)
            d = (np.arange(n_best) - center) / width
            sign = np.where(np.arange(n_best) < center, 1.0, -1.0)
            a_pert = a_pert * (1.0 + 0.06 * sign * np.exp(-d * d))
        elif strat == 3:
            # Anti-palindromic nudge
            a_pert = best_a.copy()
            rev = a_pert[::-1]
            a_pert = a_pert + 0.03 * (a_pert - rev)
        else:
            # Random restart blend
            a_rand = np.clip(0.5 + 0.3 * rng.standard_normal(n_best), 0.01, 10)
            a_pert = 0.7 * best_a + 0.3 * a_rand * (best_a.sum() / a_rand.sum())

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        # Quick L-BFGS refinement
        sc, a_opt, _ = run_lbfgs(a_pert, 3000, maxiter=2000)
        report(sc, a_opt, f"P4 #{trial} s={strat}")

        if trial % 5 == 0:
            print(f"    Trial {trial}: best={best_score:.8f}")

    # ============ Phase 5: Final polish (remaining time) ============
    print(f"\n=== Phase 5: Final Polish ===")
    if time.monotonic() < deadline - 5:
        n_iters = max(2000, int((deadline - time.monotonic() - 5) * 60))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "P5 subgrad")

    if time.monotonic() < deadline - 5:
        remaining = max(10, deadline - 5 - time.monotonic())
        for beta in [5000, 20000]:
            if time.monotonic() > deadline - 5:
                break
            maxit = max(1000, int(remaining / 0.02))
            sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit)
            report(sc, a_opt, f"P5 beta={beta}")
            remaining /= 2

    # ============ Output ============
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
