def lp_direction(f, M, n):
    """
    Solve LP: maximize sum(b) subject to (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b.
    """
    # The constraint matrix A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1)

    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)

                  bounds=[(0, None)] * n)
    
    if res.status == 0:
        return res.x
    else:
        return None

            args=(base, alpha, beta_c, outside_max, S_rest, n),
            bounds=(0.0, t_hi),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = float(res.x)

        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base + t_opt * alpha + (t_opt * t_opt) * beta_c

    return a


def build_lp_matrix(f, n):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows.extend(np.full(len(js), k, dtype=np.int32).tolist())
        cols.extend(js.astype(np.int32).tolist())
        vals.extend(f[ks - 0 for ks in [k - j for j in js]].tolist())

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_direction(f, M, n, timeout=30.0):
    """
    Solve LP: maximize sum(b) subject to (f*b)[k] <= M for all k, b >= 0.
    Returns optimal b or None if infeasible/timeout.
    """
    t0 = time.monotonic()

    # Build constraint matrix efficiently
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        fs = f[js]  # f[j] for j in [j_start, j_end]
        # A[k, j] should be f[k-j], not f[j]
        # (f*b)[k] = sum_j f[k-j] * b[j], so A[k,j] = f[k-j]
        idx = k - js
        fs = f[idx]
        rows.extend(np.full(len(js), k, dtype=np.int64).tolist())
        cols.extend(js.astype(np.int64).tolist())
        vals.extend(fs.tolist())

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      method='highs',
                      bounds=[(0, None)] * n,
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res.status == 0 and res.x is not None:
        b = np.clip(res.x, 0, 1000.0)
        if b.sum() > 1e-10:
            return b
    return None


def lp_step(a, M=None, step_size=None):
    """
    Perform one LP-based direction step.
    Returns updated sequence and score.
    """
    n = len(a)
    conv = fast_conv(a)
    S = float(a.sum())
    if M is None:
        M = float(np.max(conv))

    b = lp_direction(a, M, n)
    if b is None or b.sum() < 1e-10:
        return a, compute_score(a)

    # Rescale b to have same sum as a
    S_b = float(b.sum())
    g = b * (S / S_b)

    # Line search: find best t in [0, 1]
    if step_size is None:
        best_sc = compute_score(a)
        best_t = 0.0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.clip(a_try, 0.0, 1000.0)
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            return a, best_sc
        a_new = (1.0 - best_t) * a + best_t * g
        return np.clip(a_new, 0.0, 1000.0), best_sc
    else:
        a_new = (1.0 - step_size) * a + step_size * g
        a_new = np.clip(a_new, 0.0, 1000.0)
        return a_new, compute_score(a_new)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(np.sum(exp_scaled))
    if exp_sum < 1e-300:
        exp_sum = 1e-300
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(x, beta):
    """Wrapper for scipy minimize."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, t_deadline=None):
    """Run L-BFGS-B on LSE-smought objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    class TimeCallback:
        def __init__(self):
            self.stop = False
            self.dept = self
            self.dept = None

    stop_flag = [False]

    def cb_check(k):
        if t_deadline is not None and time.monotonic() > t_deadline:
            stop_flag[0] = True
        return stop_flag[0]

    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            callback=cb_check,
            options={'ftol': 1e-14, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
    except Exception:
        return compute_score(a0), a0

    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def peak_equalize(a, rng, n_peaks=5, strength=0.03):
    """Reduce values that contribute to the top peaks of the convolution."""
    n = len(a)
    c = fast_conv(a)
    peak_vals = np.sort(c)[::-1][:n_peaks]
    M = float(peak_vals[0])
    a_new = a.copy()
    S = float(a.sum())

    for _ in range(3):
        k_star = int(np.argmax(c))
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a_new[j_start:j_end + 1][::-1]
        top = np.argsort(contrib)[::-1][:max(3, n // 50)]
        bottom = np.argsort(contrib)[:n - len(top)]
        red = 0.0
        for j in top:
            amt = a_new[j] * strength
            a_new[j] = max(a_new[j] - amt, 0.0)
            red += amt
        if len(bottom) > 0 and red > 0:
            a_new[bottom] += red / len(bottom)

    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 100:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(512) * 0.5

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent score: {p_score:.8f}, length: {len(parent)}")
        best_score = p_score
        best_a = parent.copy()
    else:
        print("No parent, using structured init.")
        n_init = 2048
        # Use a raised-cosine shape as initial guess
        x = np.linspace(0, np.pi, n_init)
        best_a = np.sin(x) ** 2 + 0.01
        best_score = compute_score(best_a)
        print(f"Init score: {best_score:.8f}")

    n_work = len(best_a)
    print(f"Working length: {n_work}")

    # ============================================================
    # Phase 1: CD + LP hybrid (55% of time)
    # ============================================================
    phase1_deadline = t_start + 0.55 * total_budget
    print(f"\n=== Phase 1: CD + LP Hybrid (budget {phase1_deadline - t_start:.0f}s) ===")

    a_cur = best_a.copy()
    no_improve = 0
    sweep_count = 0
    lp_count = 0

    while time.monotonic() < phase1_deadline:
        # Check if time for LP step (every 8 CD sweeps)
        if sweep_count > 0 and sweep_count % 8 == 0 and time.monotonic() < phase1_deadline - 60.0:
            t0 = time.monotonic()
            a_new, sc = lp_step(a_cur)
            elapsed_lp = time.monotonic() - t0
            lp_count += 1
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
                print(f"  LP step {lp_count}: {sc:.8f} (NEW BEST, {elapsed_lp:.1f}s)")
            else:
                # Still use it as starting point for CD
                a_cur = a_new
                no_improve += 1
                if lp_count <= 5:
                    print(f"  LP step {lp_count}: {sc:.8f} ({elapsed_lp:.1f}s)")
            sweep_count += 1
            continue

        # CD sweep
        if sweep_count % 3 == 0:
            a_cur = coord_sweep(a_cur, hot=True)
        else:
            a_cur = coord_sweep(a_cur, hot=False, rng=rng)

        sc = compute_score(a_cur)
        sweep_count += 1

        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            if sweep_count <= 10 or sweep_count % 25 == 0:
                print(f"  CD sweep {sweep_count}: {sc:.8f} (NEW BEST)")
        else:
            no_improve += 1

        if sweep_count % 20 == 0:
            print(f"  Sweep {sweep_count}: score={sc:.8f}, best={best_score:.8f}, no_improve={no_improve}")

        # Perturb if stuck
        if no_improve >= 35:
            strat = rng.integers(0, 3)
            if strat == 0:
                a_cur = targeted_perturb(a_cur, rng, strength=0.02 + 0.03 * rng.random())
            elif strat == 1:
                a_cur = peak_equalize(a_cur, rng, n_peaks=5, strength=0.02 + 0.02 * rng.random())
            else:
                scale = 0.02 + 0.04 * rng.random()
                a_cur = np.clip(a_cur * (1.0 + scale * rng.standard_normal(n_work)), 0.0, 1000.0)
            no_improve = 0
            print(f"  Perturbed (strategy {strat}) at sweep {sweep_count}")

    print(f"Phase 1 done: {sweep_count} sweeps, {lp_count} LP steps, best={best_score:.8f}")

    # Phase 2: LSE refinement (20% of time)
    # ============================================================
    phase2_deadline = t_start + 0.75 * total_budget
    print(f"\n=== Phase 2: LSE Refinement (budget {phase2_deadline - t_start:.0f}s) ===")

    for beta in [100, 300, 800, 2000, 5000, 10000]:
        if time.monotonic() > phase2_deadline - 30.0:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=4000, maxfun=80000,
                              t_deadline=min(phase2_deadline - 10.0, time.monotonic() + 120.0))
        elapsed = time.monotonic() - t0
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  LSE beta={beta}: {sc:.8f} (NEW BEST, {elapsed:.1f}s)")
        else:
            print(f"  LSE beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    a_cur = best_a.copy()
    for _ in range(3):
        if time.monotonic() > phase2_deadline:
            break
        a_cur = coord_sweep(a_cur, hot=True)
        sc = compute_score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a
