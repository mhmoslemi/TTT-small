def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):

    S = float(a.sum())

def lse_score_grad(a, beta):

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)

    S2 = S * S

"""
Step function optimization: LSE warm start + LP cutting plane + momentum subgradient.
Key innovation over previous: LP-based direction finding via cutting plane for principled
refinement, combined with diverse restarts using structured initializations.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_obj(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, func, args, maxiter=2000, maxfun=60000):
    """Run L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            func, a0, args=args,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
        )
        return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        return a0


def subgradient_at(a, k_star):
    """Compute subgradient of max-conv at peak k_star."""
    n = len(a)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    sg = np.zeros(n)
    i_vals = np.arange(j_start, j_end + 1)
    sg[i_vals] = 2.0 * a[k_star - i_vals]
    return sg


def momentum_subgrad(a, iters=5000, gamma=0.7, use_adaptive=True):
    """Momentum subgradient method with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        sg = subgradient_at(a, k_star)
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        if use_adaptive:
            step = (sc - 1.0) / (gn * gn + 1e-20)
            step = min(step, 0.5 / (S + 1e-10))
            step = max(step, 1e-6 / (gn * S + 1e-20))
        else:
            step = 0.1 / (gn * S + 1e-20)
        
        v = gamma * v + (1.0 - gamma) * grad
        a -= step * v
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a

    cols_f = []
    vals_f = []

        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        rows_f.extend([p_idx] * len(i_range))
        cols_f.extend(i_range.tolist())
        vals_f.extend(a[k - i_range].tolist())
    
    if not rows_f:
        return None
    
    F_mat = csr_matrix((vals_f, (rows_f, cols_f)), shape=(P, n))
    b_ub = np.full(P, M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=F_mat, b_ub=b_ub,
                      bounds=[(0.0, 1000.0)] * n, method='highs',
                      options={'time_limit': 5.0})
        if not res.success:
            return None
        return res.x
    except Exception:
        return None


def lp_cutting_plane(a, time_budget=300.0, max_active=300, verbose=True):
    """LP cutting plane method for refinement."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    t_start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t_start < time_budget:
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                break
        
        # Build active set
        threshold = 0.85 * M
        active_mask = c > threshold
        active_peaks = np.where(active_mask)[0]
        if len(active_peaks) > max_active:
            top_idx = np.argsort(c[active_peaks])[::-1][:max_active]
            active_peaks = active_peaks[top_idx]
        
        if len(active_peaks) < 2:
            active_peaks = np.argsort(c)[::-1][:10]
        
        g = solve_lp_direction(a, active_peaks, M)
        if g is None:
            break
        
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        
        # Normalize and line search
        g_norm = g * (S / g_sum)
        best_t = 0.0
        best_t_sc = sc
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = a * (1.0 - t) + g_norm * t
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            trial_sc = compute_score(a_trial)
            if trial_sc < best_t_sc:
                best_t_sc = trial_sc
                best_t = t
        
        if best_t == 0.0:
            for t in [0.05, 0.1, 0.15, 0.25]:
                a_trial = a * (1.0 - t) + g_norm * t
                a_trial = np.clip(a_trial, 0.0, 1000.0)
                trial_sc = compute_score(a_trial)
                if trial_sc < best_t_sc:
                    best_t_sc = trial_sc
                    best_t = t
        
        if best_t > 0:
            a = a * (1.0 - best_t) + g_norm * best_t
            a = np.clip(a, 0.0, 1000.0)
        else:
            if verbose:
                print(f"    LP stagnant at iter {stagnant}")
            break
    
    return best_sc, best_a


def diverse_init(n, rng, method='gaussian'):
    """Create diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    center = n / 2.0
    if method == 'gaussian':
        sigma = n * (0.1 + 0.3 * rng.random())
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif method == 'triangular':
        a = np.clip(1.0 - np.abs(x - center) / center, 0.0, 1.0)
    elif method == 'multi_peak':
        n_peaks = int(2 + 4 * rng.random())
        a = np.zeros(n)
        for _ in range(n_peaks):
            mu = rng.uniform(0, n)
            sigma = n * (0.05 + 0.15 * rng.random())
            amp = 0.5 + rng.random()
            a += amp * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif method == 'bump':
        w = n * (0.1 + 0.4 * rng.random())
        a = np.clip(1.0 - np.abs(x - center) / (w / 2.0), 0.0, 1.0)
    elif method == 'flat':
        a = np.ones(n)
    elif method == 'cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    else:
        a = np.ones(n) * 0.5
    return np.clip(a, 0.0, 1.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses LSE warm start, LP cutting plane, momentum subgradient, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    deadline = t_start + total_budget - 10.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        sc = compute_score(parent)
        report(sc, parent, "parent")
        print(f"Parent score: {sc:.8f}, length={len(parent)}")
        n_use = len(parent)
    else:
        n_use = 6962
        a_init = diverse_init(n_use, rng, 'gaussian')
        sc = compute_score(a_init)
        report(sc, a_init, "init")

    print(f"Working with n = {n_use}, starting score = {best_score:.8f}")

    # ===================================================================
    # Phase 1: LSE warm start - 15% time
    # ===================================================================
    phase1_end = t_start + 0.15 * (total_budget - 10.0)
    print("\n=== Phase 1: LSE Warm Start ===")
    a_cur = best_a.copy()

    if time.monotonic() < phase1_end:
        for beta in [50, 200, 1000, 5000]:
            if time.monotonic() > phase1_end:
                break
            t0 = time.monotonic()
            a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=1500, maxfun=30000)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"LSE b={beta}")
            print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 2: LP Cutting Plane - 45% time
    # ===================================================================
    phase2_end = t_start + 0.60 * (total_budget - 10.0)
    print("\n=== Phase 2: LP Cutting Plane ===")
    
    if time.monotonic() < phase2_end:
        time_budget_lp = phase2_end - time.monotonic()
        # Split LP time among multiple rounds with subgradient between
        n_rounds = 4
        per_round = time_budget_lp / n_rounds
        
        a_cur = best_a.copy()
        for r in range(n_rounds):
            if time.monotonic() > phase2_end:
                break
            t0 = time.monotonic()
            time_rem = phase2_end - time.monotonic()
            sc, a_lp = lp_cutting_plane(a_cur, time_budget=min(per_round * 0.7, time_rem * 0.7), max_active=300)
            report(sc, a_lp, f"LP round {r}")
            
            if time.monotonic() < phase2_end:
                sc2, a_sub = momentum_subgrad(a_cur, iters=min(2000, int((phase2_end - time.monotonic()) * 50)))
                report(sc2, a_sub, f"subgrad after LP {r}")
            a_cur = best_a.copy()
            print(f"  Round {r} done ({time.monotonic()-t0:.1f}s)")

    # ===================================================================
    # Phase 3: High-beta LSE refinement - 15% time
    # ===================================================================
    phase3_end = t_start + 0.75 * (total_budget - 10.0)
    print("\n=== Phase 3: High-Beta LSE Refinement ===")
    a_cur = best_a.copy()
    
    if time.monotonic() < phase3_end:
        for beta in [5000, 10000, 20000, 50000]:
            if time.monotonic() > phase3_end:
                break
            t0 = time.monotonic()
            a_cur = run_lbfgs(a_cur, lse_obj, (beta,), maxiter=3000, maxfun=50000)
            sc = compute_score(a_cur)
            report(sc, a_cur, f"refine b={beta}")
            print(f"  beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s}")

    # ===================================================================
    # Phase 4: LP again from refined point - 10% time
    # ===================================================================
    phase4_end = t_start + 0.85 * (total_budget - 10.0)
    print("\n=== Phase 4: LP from Refined Point ===")
    
    if time.monotonic() < phase4_end:
        time_budget_lp2 = phase4_end - time.monotonic()
        sc, a_lp2 = lp_cutting_plane(best_a, time_budget=time_budget_lp2 * 0.8, max_active=500)
        report(sc, a_lp2, "LP phase 4")

    # ===================================================================
    # Phase 5: Diverse restarts - 10% time
    # ===================================================================
    phase5_end = deadline
    print("\n=== Phase 5: Diverse Restarts ===")
    
    n_best = len(best_a)
    restart_count = 0
    methods = ['gaussian', 'triangular', 'multi_peak', 'bump', 'cosine', 'flat']
    
    while time.monotonic() < phase5_end and restart_count < 30:
        t0 = time.monotonic()
        method = methods[restart_count % len(methods)]
        
        if restart_count < 6:
            # Start from scratch with diverse initialization
            a_pert = diverse_init(n_best, rng, method)
        else:
            # Perturb the best solution
            if restart_count % 3 == 0:
                scale = 0.05 + 0.1 * rng.random()
                a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif restart_count % 3 == 1:
                # Frequency domain perturbation
                L = 1 << (2 * n_best - 2).bit_length()
                A = np.fft.fft(best_a, L)
                A_mag = np.abs(A) * (1.0 + 0.05 * rng.standard_normal(L))
                A_mag = np.clip(A_mag, 0, None)
                A_phase = np.angle(A)
                a_pert = np.fft.ifft(A_mag * np.exp(1j * A_phase)).real[:n_best]
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            else:
                # Targeted: reduce peak contributors
                c = fast_conv(best_a)
                k_star = int(np.argmax(c))
                j_start = max(0, k_star - n_best + 1)
                j_end = min(k_star, n_best - 1)
                a_pert = best_a.copy()
                contrib = np.zeros(n_best)
                if j_end >= j_start:
                    i_vals = np.arange(j_start, j_end + 1)
                    contrib[i_vals] = best_a[k_star - i_vals]
                top_n = max(10, n_best // 50)
                top_idx = np.argsort(contrib)[::-1][:top_n]
                bot_idx = np.argsort(contrib)[:top_n * 2]
                total_red = 0.0
                for j in top_idx:
                    red = a_pert[j] * 0.3
                    a_pert[j] -= red
                    a_pert[j] = max(a_pert[j], 0.0)
                    total_red += red
                if len(bot_idx) > 0:
                    a_pert[bot_idx] += total_red / len(bot_idx)
                a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        # Quick optimization from perturbed start
        time_rem = phase5_end - time.monotonic()
        if time_rem < 5:
            break
            
        if restart_count < 6:
            # Full LSE from scratch
            a_opt = run_lbfgs(a_pert, lse_obj, (500,), maxiter=500, maxfun=10000)
            a_opt = run_lbfgs(a_opt, lse_obj, (5000,), maxiter=500, maxfun=10000)
        else:
            # Shorter optimization
            a_opt = run_lbfgs(a_pert, lse_obj, (2000,), maxiter=500, maxfun=8000)
        
        sc = compute_score(a_opt)
        report(sc, a_opt, f"restart {restart_count} ({method})")
        restart_count += 1

    # ===================================================================
    # Final polish
    # ===================================================================
    print("\n=== Final Polish ===")
    if time.monotonic() < deadline:
        sc, a_pol = momentum_subgrad(best_a, iters=3000, gamma=0.5)
        report(sc, a_pol, "final subgrad")
        
        if time.monotonic() < deadline:
            a_final = run_lbfgs(best_a, lse_obj, (50000,), maxiter=2000, maxfun=30000)
            sc = compute_score(a_final)
            report(sc, a_final, "final LSE")

    # ===================================================================
    # Output
    # ===================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
