def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)

        s = float(a.sum())

    return a

def _compute_conv(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    m = 2 * len(a) - 1
    return c[:m].copy()

def _score(c, n):
    return 2.0 * n * float(np.max(c))

def _gradient(c, a, T, L, n):
    m = len(c)
    # Softmax weights
    vmax = float(np.max(c))
    z = T * (c - vmax)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    wsum = float(w.sum())
    if wsum <= 0:
        w = np.ones(m) / m
    else:

    A = np.fft.rfft(a, L)

    g = g_full[:n].copy()
    np.maximum(g, 0.0, out=g)
    return 2.0 * g, w, float(T * vmax + math.log(wsum))

"""
Optimize non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based log-sum-exp gradient with adaptive temperature, multi-length search,
and multi-phase strategy (coarse optimization -> fine tuning -> local refinement).
Key improvements over previous: better temperature adaptation, multiple sequence
lengths, structured restarts, and a final local search phase.
"""

import math
import time
import numpy as np


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _compute_conv(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    m = 2 * len(a) - 1
    c = c[:m].copy()
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    return 2.0 * float(n) * float(np.max(c))


def _compute_gradient(c, a, T, L, n):
    m = len(c)
    vmax = float(np.max(c))
    z = T * (c - vmax)
    z = np.clip(z, -60.0, 0.0)
    w = np.exp(z)
    wsum = float(w.sum())
    if wsum <= 0.0 or not np.isfinite(wsum):
        w = np.ones(m, dtype=np.float64) / float(m)
    else:
        w /= wsum

    w_pad = np.zeros(L, dtype=np.float64)
    w_pad[:m] = w

    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w_pad, L)
    g_full = np.fft.irfft(W * np.conjugate(A), L)
    g = g_full[:n].copy()
    np.maximum(g, 0.0, out=g)
    J = float(T * vmax + math.log(max(wsum, 1e-300)))
    return 2.0 * g, J


def _random_init(n, rng):
    method = int(rng.integers(0, 6))
    if method == 0:
        a = rng.uniform(0.05, 1.0, n)
    elif method == 1:
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2 / 0.03) + 0.05
    elif method == 2:
        t = np.linspace(0.0, 1.0, n)
        k = int(rng.integers(2, 15))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * k * t + phase) + rng.normal(0.0, 0.08, n)
    elif method == 3:
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 6))):
            c = float(rng.uniform(0.1, 0.9))
            w = float(rng.uniform(0.02, 0.1))
            a *= np.exp(-((t - c) ** 2) / (2.0 * w * w))
    elif method == 4:
        a = rng.uniform(0.0, 1.0, n)
        kernel = np.ones(7, dtype=np.float64) / 7.0
        a = np.convolve(a, kernel, mode='same')
    else:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t) + 0.2 * np.cos(4.0 * np.pi * t)
        a = a + rng.normal(0.0, 0.05, n)
    return _normalize(a)


def _taper_init(n, rng):
    """Initialize with a smooth taper shape, good for reducing edge peaks."""
    t = np.linspace(0.0, 1.0, n)
    alpha = float(rng.uniform(0.5, 3.0))
    beta = float(rng.uniform(0.5, 3.0))
    window = (t ** alpha) * ((1.0 - t) ** beta)
    a = window + 0.01 + rng.normal(0.0, 0.02, n)
    return _normalize(a)


def _optimize_phase(a, L, n, m, rng, time_limit, T_base=50.0, T_max=500.0, label=""):
    """Main gradient-based optimization loop."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    c = _compute_conv(a, L)
    best_a = a.copy()
    best_score = _score(c, n)

    T = float(T_base)
    eta = 0.08
    no_improve = 0
    total_iters = 0
    print(f"  [{label}] Start: score={best_score:.6f}")

    while time.monotonic() < deadline:
        c = _compute_conv(a, L)
        curr_score = _score(c, n)

        if curr_score < best_score - 1e-13:
            best_score = curr_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature based on convolution dynamic range
        c_max = float(np.max(c))
        c_mean = float(np.mean(c))
        c_std = float(np.std(c))
        if c_std > 0:
            T_target = m / (20.0 * c_std + 1e-30)
            T_target = max(T_base * 0.5, min(T_max, T_target))
        else:
            T_target = T_base
        T = T * 0.97 + T_target * 0.03

        g, J = _compute_gradient(c, a, T, L, n)
        gmax = float(np.max(g))
        if gmax <= 0.0 or not np.isfinite(gmax):
            break
        d = g / gmax

        # Try multiple step sizes
        accepted = False
        steps = (eta, eta * 0.5, eta * 0.25, eta * 0.7)
        for step in steps:
            if step <= 1e-14:
                continue
            log_a = np.log(np.maximum(a, 1e-300))
            a_new = np.exp(log_a - step * d)
            s = float(a_new.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            a_new /= s
            c_new = _compute_conv(a_new, L)
            sc_new = _score(c_new, n)

            if sc_new < curr_score - 1e-14:
                a = a_new
                accepted = True
                break
            else:
                vmax_new = float(np.max(c_new))
                z_new = T * (c_new - vmax_new)
                z_new = np.clip(z_new, -60.0, 0.0)
                e_new = np.exp(z_new)
                s_new = float(e_new.sum())
                if s_new > 0.0:
                    J_new = T * vmax_new + math.log(s_new)
                    if J_new < J - 1e-12:
                        a = a_new
                        accepted = True
                        break

        if accepted:
            eta = min(0.8, eta * 1.08)
        else:
            eta = max(0.01, eta * 0.85)

        total_iters += 1

        # Restart strategy
        if no_improve > 150:
            mix_frac = 0.05 + 0.25 * float(rng.random())
            perturbation = _random_init(n, rng)
            a = _normalize((1.0 - mix_frac) * best_a + mix_frac * perturbation)
            T = float(T_base)
            eta = 0.08
            no_improve = 0

        if total_iters % 500 == 0:
            print(f"    [{label}] iter={total_iters}, score={best_score:.6f}, T={T:.1f}, eta={eta:.4f}")

    return best_a, best_score


def _local_refinement(a, n, time_limit, rng):
    """Fine-grained local search around a good solution."""
    deadline = time.monotonic() + time_limit
    L = 1 << (3 * n - 2).bit_length()
    m = 2 * n - 1
    a = _normalize(a)
    c = _compute_conv(a, L)
    best_a = a.copy()
    best_score = _score(c, n)

    print(f"  [local] Start: score={best_score:.6f}")

    # Try coordinate-wise small perturbations
    n_tried = 0
    while time.monotonic() < deadline:
        # Random subset of indices
        subset_size = min(n, int(rng.integers(3, 20)))
        idx = rng.integers(0, n, subset_size)
        scale = float(rng.uniform(0.005, 0.05))

        for _ in range(10):
            a_try = a.copy()
            for i in idx:
                factor = 1.0 + float(rng.uniform(-scale, scale))
                a_try[i] = max(0.0, a_try[i] * factor)
            s = float(a_try.sum())
            if s <= 0.0:
                continue
            a_try /= s
            c_try = _compute_conv(a_try, L)
            sc_try = _score(c_try, n)
            n_tried += 1
            if sc_try < best_score - 1e-14:
                best_score = sc_try
                best_a = a_try.copy()
                a = a_try.copy()

    print(f"  [local] Done: score={best_score:.6f}, tried={n_tried}")
    return best_a, best_score


def _evaluate_exact(seq):
    """Evaluate using the exact evaluate_sequence function."""
    try:
        val = evaluate_sequence(seq)
    except NameError:
        # Fallback internal evaluation
        if not isinstance(seq, list) or not seq:
            return float('inf')
        for x in seq:
            if isinstance(x, bool) or not isinstance(x, (int, float)):
                return float('inf')
            if np.isnan(x) or np.isinf(x):
                return float('inf')
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        mb = float(np.max(b))
        return float(2.0 * n * mb / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)


def _try_length(base_seq, n, budget, rng):
    """Run optimization for a specific sequence length."""
    if budget < 30.0:
        return None, float('inf')

    m = 2 * n - 1
    L = 1 << (3 * n - 2).bit_length()

    # Prepare initial sequence of length n
    if base_seq is not None:
        base = np.asarray(base_seq, dtype=np.float64).ravel()
        if len(base) >= n:
            a = _normalize(base[:n])
        else:
            a = _normalize(np.pad(base, (0, n - len(base))))
    else:
        a = _random_init(n, rng)

    # Phase 1: Coarse optimization
    t1_start = time.monotonic()
    t1_budget = min(budget * 0.6, 1200.0)
    a, sc = _optimize_phase(a, L, n, m, rng, t1_budget, T_base=30.0, T_max=300.0, label=f"coarse_n{n}")

    # Phase 2: Fine optimization
    remaining = budget - (time.monotonic() - t1_start)
    if remaining > 20.0:
        t2_budget = min(remaining * 0.5, 600.0)
        a2, sc2 = _optimize_phase(a, L, n, m, rng, t2_budget, T_base=100.0, T_max=1000.0, label=f"fine_n{n}")
        if sc2 < sc:
            a, sc = a2, sc2

    # Phase 3: Local refinement
    remaining = budget - (time.monotonic() - t1_start)
    if remaining > 15.0:
        t3_budget = min(remaining * 0.5, 120.0)
        a3, sc3 = _local_refinement(a, n, t3_budget, rng)
        if sc3 < sc:
            a, sc = a3, sc3

    print(f"  [n={n}] Final internal score: {sc:.6f}")
    return a, sc
