"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative
optimization with diverse initializations, multiple sequence lengths,
and coordinate descent polish. The LP at each step finds the optimal
direction within the polytope {g >= 0 : conv(a,g) <= M}, providing
a second-order improvement over subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]


def score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - j_end : k - j_start : -1]
    return A


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(a, g) <= M.
    Returns optimal g (not rescaled).
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(a_init, max_iters=500, time_budget=60.0, t0=None, step_sizes=None):
    """
    LP-based iterative optimizer.
    a_init: initial sequence (will be normalized).
    Returns (best_a, best_score).
    """
    if t0 is None:
        t0 = time.monotonic()
    n = a_init.size
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_sc = score(a)
    
    if step_sizes is None:
        step_sizes = [0.5, 0.3, 0.2, 0.1, 0.5, 0.3, 0.2, 0.1]
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        
        # Solve LP
        g = lp_direction(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Rescale g to sum 1
        g = g / g_sum
        
        # Try different step sizes
        t = step_sizes[it % len(step_sizes)]
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_sc = score(a_new)
        
        # Accept if better, otherwise try smaller step
        if new_sc <= cur_sc:
            a = a_new
        else:
            # Try half step
            a_half = (1 - t/2) * a + (t/2) * g
            a_half = np.maximum(a_half, 0)
            a_half = a_half / a_half.sum()
            half_sc = score(a_half)
            if half_sc <= cur_sc:
                a = a_half
            else:
                # Skip this step
                pass
    
    # Final check
    final_sc = score(a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()
    
    return best_a, best_sc
