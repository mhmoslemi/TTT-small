    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    
    # Top-k
    k = min(5, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    
    # Weights
    spread = max(0.01, top_vals.max() - top_vals.min())
    w = np.exp((top_vals - top_vals.max()) / spread)
    w /= w.sum()
    
    # Gradient
    g = _make_subgradient(a, top_idx, w)
    g = g / np.mean(g)  # normalize
    
    # Update
    a = a * np.exp(-eta * g)
    a = a / a.sum()
    
    # Adapt eta
    eta *= 0.9995  # very slow decay
    # Or: keep eta fixed
    
    it += 1

"""
Optimize non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses fast exact top-k subgradient mirror descent with adaptive step size,
multiple restarts, and local block polish. Key insight: subgradient at argmax
j* of conv(a,a) is simply 2*a[j*-m] (reflected a), enabling O(n) updates
without extra FFT calls.
"""

import numpy as np
import time


def _next_power_of_2(n):
    n = max(1, n)
    return 1 << (n - 1).bit_length()


def _fft_convolve_self(a):
    """Full self-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = _next_power_of_2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def _normalize(a):
    """Ensure non-negative, finite, sum to 1."""
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score(a):
    """Compute exact score of a sequence (normalized to sum 1)."""
    n = len(a)
    c = _fft_convolve_self(a)
    return float(2.0 * n * np.max(c))


def _resample(a, n_target):
    """Resample normalized sequence to target length via linear interpolation."""
    a = _normalize(a)
    n = len(a)
    if n == n_target:
        return a.copy()
    if n == 1:
        return np.ones(n_target) / n_target
    x_old = np.arange(n, dtype=float)
    x_new = np.arange(n_target, dtype=float) * (n - 1.0) / (n_target - 1.0)
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _init_two_block(n, split_frac, ratio):
    """Two-block initialization: first block value 1, second block value ratio."""
    m = max(1, int(n * split_frac))
    a = np.ones(n)
    a[m:] = ratio
    return _normalize(a)


def _init_cosine(n):
    """Cosine window initialization."""
    t = np.arange(n, dtype=float) / n
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a[0] = 0.0  # Make it a proper window
    a[-1] = 0.0
    a = a + 1e-6
    return _normalize(a)


def _init_gaussian(n, width_frac=0.4):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=float) / n
    a = np.exp(-0.5 * ((t - 0.5) / (width_frac / 3.0)) ** 2)
    return _normalize(a)


def _init_random(n, rng):
    """Random positive initialization."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    return _normalize(a)


def _init_sine(n, freq=2.0):
    """Sine-modulated initialization."""
    t = np.arange(n, dtype=float) / n
    a = 1.0 + 0.5 * np.sin(2.0 * np.pi * freq * t)
    a = np.maximum(a, 0.01)
    return _normalize(a)


def _compute_top_k_gradient(a, c, k=5, beta=50.0):
    """
    Compute weighted subgradient using top-k indices of convolution c.
    Returns (gradient, top_k_indices).
    """
    n = len(a)
    k = min(k, len(c))
    # Find top k indices
    if k < len(c):
        top_idx = np.argpartition(c, -k)[-k:]
    else:
        top_idx = np.arange(len(c))
    top_vals = c[top_idx]
    # Sort in descending order
    order = np.argsort(top_vals)[::-1]
    top_idx = top_idx[order]
    top_vals = top_vals[order]

    # Softmax weights
    max_v = top_vals[0]
    range_v = max_v - top_vals[-1]
    if range_v < 1e-15:
        w = np.full(k, 1.0 / k)
    else:
        scaled_beta = beta / range_v
        w = np.exp(scaled_beta * (top_vals - max_v))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.full(k, 1.0 / k)
        else:
            w = w / w_sum

    # Compute gradient: g[m] = 2 * sum_j w_j * a[j-m] (where valid)
    g = np.zeros(n)
    for j, wj in zip(top_idx, w):
        lo = max(0, int(j) - n + 1)
        hi = min(n - 1, int(j))
        if lo <= hi:
            m_range = np.arange(lo, hi + 1)
            a_idx = int(j) - m_range
            g[m_range] += 2.0 * wj * a[a_idx]

    return g, top_idx


def _optimize_mirror(a0, time_limit, rng, k_smooth=5, beta=50.0,
                     eta_init=0.3, verbose=False, tag=""):
    """
    Mirror descent (multiplicative) with exact top-k subgradient.
    Each iteration: 1 FFT for convolution + O(k*n) for gradient.
    """
    a = _normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    eta = eta_init
    eta_min = 0.01
    patience = 0
    max_patience = 3000
    it = 0
    deadline = time.monotonic() + time_limit
    last_print = 0
    momentum_g = np.zeros(n)
    mom_alpha = 0.7  # EMA coefficient

    while time.monotonic() < deadline:
        # Compute convolution
        c = _fft_convolve_self(a)
        cur_score = 2.0 * n * np.max(c)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Compute top-k gradient
        g, _ = _compute_top_k_gradient(a, c, k=k_smooth, beta=beta)
        g_mean = np.mean(g)
        if g_mean < 1e-15:
            break

        # Normalize gradient
        g = g / g_mean

        # Momentum smoothing
        momentum_g = mom_alpha * momentum_g + (1.0 - mom_alpha) * g
        g_eff = momentum_g
        g_eff_mean = np.mean(g_eff)
        if g_eff_mean < 1e-15:
            g_eff = g
        else:
            g_eff = g_eff / g_eff_mean

        # Multiplicative update
        a_new = a * np.exp(-eta * g_eff)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15 or not np.isfinite(s):
            eta *= 0.5
            continue
        a_new = a_new / s
        a = a_new

        # Adaptive step size
        if patience > 100:
            eta = max(eta_min, eta * 0.95)

        if patience > max_patience:
            # Restart with perturbation from best
            std = 0.1 + 0.05 * (patience // max_patience)
            a = _normalize(best_a * np.exp(rng.normal(0, std, n)))
            eta = eta_init
            patent_init = 0
            momentum_g = np.zeros(n)
            patience = 0

        it += 1
        if verbose and (it - last_print >= 2000):
            last_print = it
            print(f"  [{tag}] iter={it}, score={best_score:.6f}, eta={eta:.4f}, patience={patience}")

    return best_a, best_score


def _local_block_search(a0, time_limit, rng, num_factors=7):
    """Local search: multiply random contiguous blocks by factor."""
    a = _normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0

    factors = np.array([0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.3])

    while time.monotonic() < deadline:

        if r < 0.4:
            w = int(rng.integers(1, max(2, n // 100)))

            a_try = _normalize(a_try)

                improved = True
                break

            stagnant = 0
        else:
            stagnant += 1

                cur_score = best_score

def _coordinate_descent(a0, time_limit, rng):
    """
    Coordinate descent: for each coordinate, find best multiplier via
    ternary search on the univariate function.
    """
    a = _normalize(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + time_limit
    it = 0

    while time.monotonic() < deadline:
        # Pick a random coordinate (or sweep all)
        if n <= 512:
            coords = np.arange(n)
            rng.shuffle(coords)
        else:
            coords = rng.integers(0, n, min(n, 64))

        for m in coords:
            if time.monotonic() >= deadline:
                break
            # Try several multipliers for coordinate m
            base = a[m]
            if base < 1e-15:
                continue
            best_mult = 1.0
            best_sc = best_score
            for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 0.33, 3.0]:
                a[m] = base * mult
                # Quick score check (expensive but necessary)
                s = a.sum()
                if s < 1e-15:
                    a[m] = base
                    continue
                c = _fft_convolve_self(a / s)
                sc = 2.0 * n * np.max(c)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_mult = mult
            a[m] = base * best_mult
            a = _normalize(a)
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()

        it += 1

    return best_a, best_score
