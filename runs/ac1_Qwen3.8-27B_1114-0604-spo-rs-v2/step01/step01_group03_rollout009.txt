    c = irfft(fa * fa, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _cross_corr(p, a):
    """Compute h[i] = sum_j p[j]*a[j-i] for i=0..n-1 using FFT."""
    n = len(a)
    m = len(p)
    a_rev = a[::-1]
    L = 1 << (m + n - 2).bit_length()
    P = rfft(p, L)
    Arev = rfft(a_rev, L)
    conv = irfft(P * Arev, L)
    return conv[n - 1: n - 1 + n]


def _smooth_f_grad(x, beta, n, L_fft):
    """Compute smooth objective and gradient in log-parametrization.
    x is free variable, a = softmax(x)."""
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    c = _fft_conv_2(a, n, L_fft)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    S = exp_bc.sum()
    G = bc_max + math.log(S)
    p = exp_bc / S
    f = 2.0 * n * G
    h = _cross_corr_2(p, a, n, L_fft)
    g_a = 2.0 * n * beta * h
    # Natural gradient in log space
    g_x = a * (g_a - np.dot(a, g_a))
    return f, g_x


def _fft_conv_2(a, n, L):
    """FFT convolution with precomputed length."""
    m = 2 * n - 1
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _cross_corr_2(p, a, n, L):
    """Cross correlation with precomputed FFT length."""
    m = 2 * n - 1
    a_rev = a[::-1]
    P = rfft(p, L)
    Arev = rfft(a_rev, L)
    conv = irfft(P * Arev, L)
    return conv[n - 1: n - 1 + n]


def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_exact(a, k=1):
    """Exact (or top-k) subgradient of max(conv(a,a)) wrt a."""
    n = len(a)
    c = _fft_conv(a)
    m = len(c)
    # Get top-k positions
    if k >= m:
        w = np.full(m, 1.0 / m)
    else:
        cmax = c.max()
        # Softmax with temperature for smoothness
        temp = 0.5 * max(cmax, 1e-15)
        w = np.exp((c - cmax) / temp)
        # Zero out everything except top-k
        if k < m:
            thresh = np.sort(w)[::-1][k]
            w[w < thresh] = 0.0
        wsum = w.sum()
        if wsum > 0:
            w /= wsum
        else:
            w = np.zeros(m)
            w[np.argmax(c)] = 1.0
    h = _cross_corr(w, a)
    return 2.0 * h


def _parametric_init(n, rng, shape_type=0):
    """Create parametric initializations."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    if shape_type == 0:
        z = np.zeros(n)
    elif shape_type == 1:
        z = -x * x
    elif shape_type == 2:
        z = x  # linear
    elif shape_type == 3:
        z = -x  # negative linear
    elif shape_type == 4:
        z = 3.0 * (x * x) - 1.0  # U-shaped
    elif shape_type == 5:
        z = -x * x + 0.3 * np.sin(4 * np.pi * t)
    elif shape_type == 6:
        z = -x * x - 0.3 * np.sin(2 * np.pi * t)
    elif shape_type == 7:
        z = -x**4
    elif shape_type == 8:
        z = -0.5 * x * x + 0.4 * np.cos(6 * np.pi * t)
    elif shape_type == 9:
        z = np.exp(-4.0 * x * x)  # Gaussian bump
    else:
        z = rng.normal(0, 1, n)
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    p /= p.sum()
    return p


def _random_init(n, rng):
    """Random smooth initialization."""
    z = rng.normal(0, 1, n)
    # Smooth it
    if n > 20:
        k = max(3, n // 500)
        kernel = np.ones(k) / k
        z = np.convolve(z, kernel, mode='same')
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    p /= p.sum()
    return p


def _exact_score(a):
    """Compute exact score (matching evaluate_sequence logic)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    mb = c.max()
    return 2.0 * n * mb / (s * s)


def _optimize_lbfgs(a0, n, L, max_time=300.0, beta_range=None, seed=42):
    """Run L-BFGS with beta annealing in log-parametrization."""
    a0 = np.asarray(a0, dtype=np.float64)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x = np.log(a0)
    x -= x.max()

    if beta_range is None:
        beta_range = [1.0, 4.0, 16.0, 64.0, 256.0, 1024.0, 4096.0, 16384.0]

    deadline = time.monotonic() + max_time
    best_a = a0.copy()
    best_f = float('inf')

    for beta in beta_range:
        if time.monotonic() > deadline - 5.0:
            break
        phase_time = min(60.0, max(5.0, (deadline - time.monotonic()) / (len(beta_range) - beta_range.index(beta) + 0.5)))

        def obj_grad(xx):
            return _smooth_f_grad(xx, beta, n, L)

        try:
            res = sp_minimize(
                lambda xx: _smooth_f_grad(xx, beta, n, L)[0],
                x,
                jac=lambda xx: _smooth_f_grad(xx, beta, n, L)[1],
                method='L-BFGS-B',
                options={'maxiter': 100, 'maxfun': 300, 'gtol': 1e-8}
            )
            x = res.x
        except Exception:
            pass

        a = np.exp(x - x.max())
        a = a / a.sum()
        # Evaluate exact score for tracking
        f_exact = _exact_score(a)
        if f_exact < best_f:
            best_f = f_exact
            best_a = a.copy()
            print(f"  beta={beta:.0f}, exact_score={f_exact:.6f}", flush=True)

    return best_a, best_f


def _proj_subgrad_search(a0, n, L, max_time=100.0, k=3):
    """Projected subgradient with top-k active set."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    deadline = time.monotonic() + max_time
    best_a = a.copy()
    best_f = _exact_score(a)
    step = 0.1
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        c = _fft_conv_2(a, n, L)
        g = _subgrad_exact(a, k=k)
        gmax = g.max()
        if gmax > 1e-15:
            g = g / gmax

        # Proximal gradient step
        a_new = a - step * g
        a_new = _proj_simplex(a_new, 1.0)

        # Check improvement
        f_new = _exact_score(a_new)
        if f_new < best_f - 1e-10:
            best_f = f_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.5, step * 1.1)
        elif f_new < _exact_score(a) - 1e-10:
            a = a_new
            no_improve += 1
        else:
            step *= 0.8
            no_improve += 1

        if no_improve > 50:
            # Reheat
            a = best_a.copy()
            step = 0.1
            no_improve = 0
            # Small random perturbation
            pert = np.random.default_rng(it).normal(0, 0.01, n)
            a = _proj_simplex(a + pert, 1.0)

        it += 1
        if it % 200 == 0:
            step *= 0.95

    return best_a, best_f


def _multiplicative_update(a0, n, L, max_time=100.0):
    """Multiplicative update with subgradient."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    deadline = time.monotonic() + max_time
    best_a = a.copy()
    best_f = _exact_score(a)
    step = 0.3

    while time.monotonic() < deadline:
        c = _fft_conv_2(a, n, L)
        g = _subgrad_exact(a, k=3)
        gmax = g.max()
        if gmax > 1e-15:
            g = g / gmax

        a_new = a * np.exp(-step * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        f_new = _exact_score(a_new)
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
            step = min(1.0, step * 1.05)
        else:
            step = max(0.05, step * 0.9)
            a_new = a  # don't update

        a = a_new

    return best_a, best_f


def _try_parent_resized(parent, n, rng):
    """Resize parent sequence to new length n."""
    if parent is None:
        return None
    p = np.asarray(parent, dtype=np.float64)
    if p.size < 10:
        return None
    # Interpolate
    old_n = p.size
    t_new = np.linspace(0, 1, n)
    t_old = np.linspace(0, 1, old_n)
    a_new = np.interp(t_new, t_old, p)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 2400.0

    start_time = time.monotonic()
    global_deadline = start_time + budget - 10.0
    rng = np.random.default_rng(seed)

    # Try to access parent
    parent = None
    try:
        parent = height_sequence_1
    except NameError:
        pass

    print(f"Budget: {budget}s, start: {start_time:.1f}", flush=True)
    if parent is not None:
        print(f"Parent length: {len(parent)}", flush=True)

    # Try multiple sequence lengths
    n_candidates = [4096, 6121, 8192, 12288]
    # Filter: don't go too large if budget is small
    if budget < 1200:
        n_candidates = [4096, 6121, 8192]
    if budget < 600:
        n_candidates = [4096, 6121]

    best_seq = None
    best_exact = float('inf')

    for trial_n in n_candidates:
        if time.monotonic() > global_deadline - 60:
            break

        n = trial_n
        m = 2 * n - 1
        L = 1 << (m - 1).bit_length()
        # Cross-correlation needs larger FFT
        L_cross = 1 << (m + n - 2).bit_length()
        print(f"\n=== Trying n={n}, L={L} ===", flush=True)

        # Generate initial candidates
        inits = []

        # From parent
        if parent is not None:
            p_resize = _try_parent_resized(parent, n, rng)
            if p_resize is not None:
                inits.append(p_resize)

        # Parametric shapes
        for st in range(10):
            inits.append(_parametric_init(n, rng, st))

        # Random
        for _ in range(3):
            inits.append(_random_init(n, rng))

        # Evaluate all inits
        best_init = None
        best_init_score = float('inf')
        for init in inits:
            sc = _exact_score(init)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init

        print(f"  Best init score: {best_init_score:.6f}", flush=True)

        # Phase 1: L-BFGS with moderate beta (main optimization)
        time_for_lbfgs = min(600.0, (global_deadline - time.monotonic()) * 0.5)
        if time_for_lbfgs > 30:
            print(f"  L-BFGS phase ({time_for_lbfgs:.0f}s)...", flush=True)
            a_opt, sc_opt = _optimize_lbfgs(
                best_init, n, L_cross,
                max_time=time_for_lbfgs,
                beta_range=[1.0, 4.0, 16.0, 64.0, 256.0, 1024.0, 4096.0]
            )
            if sc_opt < best_exact:
                best_exact = sc_opt
                best_seq = [float(x) for x in a_opt]
                print(f"  After L-BFGS: {sc_opt:.6f}", flush=True)
            current_best = a_opt
        else:
            current_best = best_init

        # Phase 2: Projected subgradient
        time_for_pg = min(120.0, (global_deadline - time.monotonic()) * 0.15)
        if time_for_pg > 10:
            print(f"  Proj subgrad phase ({time_for_pg:.0f}s)...", flush=True)
            a_pg, sc_pg = _proj_subgrad_search(current_best, n, L_cross, max_time=time_for_pg, k=3)
            if sc_pg < best_exact:
                best_exact = sc_pg
                best_seq = [float(x) for x in a_pg]
                print(f"  After proj subgrad: {sc_pg:.6f}", flush=True)
                current_best = a_pg

        # Phase 3: Multiplicative updates
        time_for_mu = min(80.0, (global_deadline - time.monotonic()) * 0.1)
        if time_for_mu > 10:
            print(f"  Multiplicative update phase ({time_for_mu:.0f}s)...", flush=True)
            a_mu, sc_mu = _multiplicative_update(current_best, n, L_cross, max_time=time_for_mu)
            if sc_mu < best_exact:
                best_exact = sc_mu
                best_seq = [float(x) for x in a_mu]
                print(f"  After multiplicative: {sc_mu:.6f}", flush=True)
                current_best = a_mu

        # Phase 4: Final L-BFGS polish with high beta
        time_for_polish = min(180.0, (global_deadline - time.monotonic()) * 0.3)
        if time_for_polish > 20:
            print(f"  Final polish ({time_for_polish:.0f}s)...", flush=True)
            a_polish, sc_polish = _optimize_lbfgs(
                current_best, n, L_cross,
                max_time=time_for_polish,
                beta_range=[1024.0, 4096.0, 16384.0, 65536.0]
            )
            if sc_polish < best_exact:
                best_exact = sc_polish
                best_seq = [float(x) for x in a_polish]
                print(f"  After polish: {sc_polish:.6f}", flush=True)

    # Final exact verification
    if best_seq is None:
        # Fallback: use parent or uniform
        if parent is not None:
            best_seq = [float(max(0, min(1000, x))) for x in parent]
        else:
            best_seq = [1.0] * 4096

    # Verify with evaluate_sequence if available
    try:
        val = evaluate_sequence(best_seq)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
        print(f"\nFinal evaluate_sequence score: {val:.6f}", flush=True)
    except NameError:
        print(f"\nFinal internal score: {best_exact:.6f}", flush=True)
    except Exception as e:
        print(f"  Error in final eval: {e}", flush=True)

    print(f"Total time: {time.monotonic() - start_time:.1f}s", flush=True)
    return best_seq
