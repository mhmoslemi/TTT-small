multiplicative gradient descent with momentum, and coordinate descent polish.
Combines multiple optimization strategies with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))

def build_conv_matrix(p):
    """Build sparse (2n-1) x n matrix A where A[k,j] = p[k-j] if 0<=k-j<n."""
    n = len(p)
    ns = 2 * n - 1
    rows = np.zeros(ns * n, dtype=np.int32)
    cols = np.zeros(ns * n, dtype=np.int32)
    vals = np.zeros(ns * n, dtype=np.float64)
    count = 0
    for k in range(ns):
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        for j in range(jmin, jmax + 1):
            rows[count] = k
            cols[count] = j
            vals[count] = p[k - j]
            count += 1
    A = coo_matrix((vals[:count], (rows[:count], cols[:count])), shape=(ns, n))
    return A.tocsr()

def lp_solve_best_response(p, M):
    """Solve max sum(b) s.t. conv(p,b) <= M, b >= 0. Returns b* or None."""
    n = len(p)
    A = build_conv_matrix(p)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def mult_grad_step(p, b, eta, tau, vel, beta):
    """One multiplicative gradient step with momentum."""
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, p, mode='valid')
    vel = beta * vel + (1.0 - beta) * g
    p_new = p * np.exp(-eta * vel)
    p_new = np.maximum(p_new, 0)
    s_new = p_new.sum()
    if s_new < 1e-300:
        return p.copy(), vel
    p_new /= s_new
    return p_new, vel

def init_uniform(n):
    return np.ones(n) / n

def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()

def init_cosine(n, freq=1):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * i / n)) ** 2
    return a / a.sum()

def init_two_gauss(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.3)
    c2 = n * rng.uniform(0.7, 0.9)
    sigma = n * rng.uniform(0.03, 0.1)
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_power_law(n, rng):
    i = np.arange(n, dtype=np.float64)
    alpha = rng.uniform(0.3, 1.5)
    a = (i + 1.0) ** (-alpha)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_symmetric_u(n, rng):
    """Symmetric U-shape with tunable depth."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    depth = rng.uniform(0.3, 1.0)
    a = 1.0 + depth * ((i - c) / c) ** 2
    return a / a.sum()

def get_inits(n, rng):

def coordinate_descent_polish(p, time_budget, rng):
    """Polish using coordinate-wise optimization."""
    n = p.size
    p = p.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()
    best_p = p.copy()
    best_sc = score_norm(p)
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            old = p[i]
            # Try a few values
            for delta in [0.5, 0.1, 0.01, -0.5, -0.1, -0.01]:
                cand = p.copy()
                cand[i] = max(0, old + delta * old)
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand /= s
                sc = score_norm(cand)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_p = cand.copy()
                    p = cand.copy()
                    improved = True
                    break
        if not improved:
            # Add small perturbation
            p = best_p + 0.01 * rng.standard_normal(n)
            p = np.maximum(p, 0)
            p = p / p.sum()
    
    return best_p, best_sc

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based best-response iteration,
multiplicative gradient descent with momentum, and coordinate descent polish.
Combines multiple optimization strategies with diverse initializations.
"""

from scipy.sparse import coo_matrix

    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) == 1."""
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))


def build_conv_matrix(p):
    """Build sparse (2n-1) x n matrix A where A[k,j] = p[k-j] if 0<=k-j<n."""
    n = len(p)
    ns = 2 * n - 1
    rows_l = []
    cols_l = []
    vals_l = []
    for k in range(ns):
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        for j in range(jmin, jmax + 1):

            cols_l.append(j)
            vals_l.append(p[k - j])
    A = coo_matrix((vals_l, (rows_l, cols_l)), shape=(ns, n))
    return A.tocsr()

    """Solve max sum(b) s.t. conv(p,b)_k <= M for all k, b >= 0."""

    """Smoothed subgradient of max(conv) using softmax weights."""
    shift = b / tau

        w[int(np.argmax(b))] = 1.0

        w /= s
