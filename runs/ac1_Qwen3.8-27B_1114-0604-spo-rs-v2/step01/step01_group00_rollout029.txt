import numpy as np
import time


def _convolve(f):
    """Compute self-convolution using FFT for efficiency."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _gradient_at_k(f, k):
    """Compute gradient of (f*f)[k] with respect to f."""
    n = len(f)
    grad = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start > m_end:
        return grad
    # grad[m] = 2 * f[k-m] for m in [m_start, m_end]
    idx = k - np.arange(m_start, m_end + 1)
    grad[m_start:m_end+1] = 2.0 * f[idx]
    return grad


def _soft_gradient(f, c, T):
    """Compute softmax-weighted average gradient over lags."""
    n = len(f)
    # Softmax over lags
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    total = p.sum()
    if total < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= total
    
    # Weighted average of gradients
    grad = np.zeros(n)
    # For efficiency, only compute gradients for lags with significant weight
    threshold = 1e-8
    active_lags = np.where(p > threshold)[0]
    for k in active_lags:
        grad += p[k] * _gradient_at_k(f, int(k))
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (css - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        progress = it / n_iters
        
        # Temperature for softmax (from peak to very small)
        T = peak0 * max(1e-10, (1 - progress) ** 2)

        step = base_step * max(min_step / base_step, (1 - progress) ** 1.5)
        
        # Compute gradient
        c = _convolve(f)

        k_star = int(np.argmax(c))
        grad_exact = _gradient_at_k(f, k_star)

        grad_soft = _soft_gradient(f, c, T)

"""
Hybrid subgradient optimizer for minimizing 2*n*max(f*f)/(sum f)^2.
Uses Nesterov-accelerated descent with exact peak-targeting gradient,
top-k lag weighting, adaptive step size, multi-start with diverse
initializations, and targeted perturbation for local minimum escape.
"""

import numpy as np
import time


def _convolve_fft(f):
    """Self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _grad_at_k(f, k):
    """Gradient of (f*f)[k] w.r.t. f. Returns vector of length len(f)."""
    n = len(f)
    grad = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start > m_end:
        return grad
    idx = k - np.arange(m_start, m_end + 1)
    grad[m_start:m_end + 1] = 2.0 * f[idx]
    return grad


def _grad_weighted(f, c, top_k=5, temp=1e-4):
    """Weighted gradient over top-k lags using softmax weights."""
    n = len(f)
    num_lags = len(c)
    k_sel = min(top_k, num_lags)
    top_idx = np.argpartition(c, -k_sel)[-k_sel:]
    vals = c[top_idx].copy()
    vals -= vals.max()
    ws = np.exp(vals / max(temp, 1e-10))
    ws /= ws.sum()
    grad = np.zeros(n)
    for w, k in zip(ws, top_idx):
        if w < 1e-10:
            continue
        grad += w * _grad_at_k(f, int(k))
    return grad


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (css - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smart_perturb(f, n, rng):
    """Perturb f by moving mass away from peak-contributing positions."""
    c = _convolve_fft(f)
    k_peak = int(np.argmax(c))
    m_start = max(0, k_peak - n + 1)
    m_end = min(k_peak, n - 1)
    contrib = np.zeros(n)
    if m_start <= m_end:
        idx = np.arange(m_start, m_end + 1)
        contrib[idx] = f[idx] * f[k_peak - idx]
    n_don = max(2, n // 15)
    n_rec = max(2, n // 15)
    top_c = np.argsort(contrib)[::-1][:n_don]
    bot_c = np.argsort(contrib)[:n_rec]
    tot = f[top_c].sum() * 0.15
    if tot < 1e-15:
        f = f + (0.002 / n) * rng.standard_normal(n)
    else:
        f = f.copy()
        f[top_c] -= tot / n_don
        f[bot_c] += tot / n_rec
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s
    return f


def _optimize_core(f, n, deadline, rng):
    """Nesterov-accelerated subgradient descent with adaptive steps."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = _score(f)

    f_mom = f.copy()
    momentum = 0.85
    base_step = 8.0 / n
    min_step = 0.5 / n

    c = _convolve_fft(f)
    peak0 = c.max()

    max_iters = 200000
    stuck = 0
    improve_thresh = 1e-8
    eval_interval = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        step = base_step * max(min_step / base_step, (1.0 - progress) ** 1.2)
        temp = peak0 * max(1e-6, (1.0 - progress) ** 2)

        # Nesterov lookahead
        y = f + momentum * (f - f_mom)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-12:
            y = f.copy()
        else:
            y = y / sy

        # Gradient at lookahead
        c_y = _convolve_fft(y)
        grad = _grad_weighted(y, c_y, top_k=5, temp=temp)

        # Update
        f_mom = f.copy()
        f_new = f - step * grad
        f_new = _project_simplex(f_new)
        f = f_new

        # Evaluate
        if (it % eval_interval == 0) or (it == max_iters - 1):
            sc = _score(f)
            if sc < best_score - improve_thresh:
                best_score = sc
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck >= 15:
                f = _smart_perturb(f, n, rng)
                f_mom = f.copy()
                stuck = 0

            # Occasionally try a smaller perturbation
            if stuck == 7:
                f_try = f + (0.001 / n) * rng.standard_normal(n)
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                sc_try = _score(f_try)
                if sc_try < best_score:
                    f = f_try
                    f_mom = f.copy()
                    best_score = sc_try
                    best_f = f.copy()
                    stuck = 0

    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _polish_local(f, n, deadline, rng):
    """Targeted coordinate transfers to reduce peak."""
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    best_score = _score(f)
    c = _convolve_fft(f)
    peak = c.max()

    while time.monotonic() < deadline:
        c = _convolve_fft(f)
        k_peak = int(np.argmax(c))
        m_start = max(0, k_peak - n + 1)
        m_end = min(k_peak, n - 1)
        contrib = np.zeros(n)
        if m_start <= m_end:
            idx = np.arange(m_start, m_end + 1)
            contrib[idx] = f[idx] * f[k_peak - idx]

        improved = False
        n_tries = 30
        for _ in range(n_tries):
            top = np.argsort(contrib)[::-1]
            bot = np.argsort(contrib)
            jd = int(top[rng.integers(0, min(8, len(top)))])
            jr = int(bot[rng.integers(0, min(8, len(bot)))])
            if jd == jr:
                continue
            for frac in [0.005, 0.01, 0.03, 0.05, 0.1]:
                amt = f[jd] * frac
                f_try = f.copy()
                f_try[jd] -= amt
                f_try[jr] += amt
                if f_try[jd] < -1e-15:
                    continue
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _convolve_fft(f_try)
                if c_try.max() < peak - 1e-15:
                    f = f_try
                    peak = c_try.max()
                    sc = _score(f)
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    improved = True
                    break

        if not improved:
            f_try = f + (0.0005 / n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            c_try = _convolve_fft(f_try)
            if c_try.max() < peak:
                f = f_try
                peak = c_try.max()

    return best_f, best_score
