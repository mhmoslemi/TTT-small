```python
def lp_response(a, M, solver='CLARABEL' or 'ECOS' or 'SCIP'):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build constraint matrix: (a*b)_k <= M for each k
    # (a*b)_k = sum_i A[k,i] * b_i
    # A[k,i] = a[i] if 0 <= k-i <= n-1, else 0
    # i.e., A[k,i] = a[i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    constraints = []
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n-1, k)
        if i_lo <= i_hi:
            row = np.zeros(n)
            row[i_lo:i_hi+1] = a[i_lo:i_hi+1]
            constraints.append(row @ b <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve()
    return b.value
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating optimization
with multiple strategies: LP response iteration, subgradient descent, and local polish.
The LP-based method solves a linear program at each step to find the optimal "response"
to the current sequence, which provides a principled descent direction.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_of(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def normalize(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(v.size) / v.size
    return v / s


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_response(a):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0. Returns normalized b*."""
    n = len(a)
    b_conv = full_conv(a)
    M = float(np.max(b_conv))
    if M < 1e-300:
        return a.copy()
    
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    
    if not res.success or res.x is None:
        return a.copy()
    
    b_star = np.maximum(res.x, 0)
    s = b_star.sum()
    if s < 1e-300:
        return a.copy()
    return b_star / s


def lp_iterate(a, max_iters, rng, time_limit):
    """Iterate the LP response map with mixing and annealing."""
    n = a.size
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = score_of(a)
    
    t_mix = 1.0  # mixing parameter
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating optimization.
The core idea: at each step, solve a linear program to find the optimal "response"
sequence b* that maximizes sum(b) subject to (a*b)_k <= M. This provides a principled
descent direction. Combined with multiple initializations, mixing, simulated annealing,
and subgradient polishing for local refinement.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import correlate as sp_correlate


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_of(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def normalize(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(v.size) / v.size
    return v / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C such that (C @ b)_k = (a*b)_k (full conv)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_response(a):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0. Returns normalized b*."""
    n = len(a)
    b_conv = full_conv(a)
    M = float(np.max(b_conv))
    if M < 1e-300:
        return a.copy()

    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
    except Exception:
        return a.copy()

    if not res.success or res.x is None:
        return a.copy()

    b_star = np.maximum(res.x, 0.0)
    s = b_star.sum()
    if s < 1e-300:
        return a.copy()
    return b_star / s

    else:
        w /= s
    # grad_j = 2 * sum_m w[j+m] * a[m] = 2 * correlate(w, a, 'full')[j]
    size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(np.conj(W) * A, size)
    return 2.0 * c[:n]


def subgrad_optimize(a, time_limit, rng, verbose=True):
    """Projected smoothed subgradient descent with momentum."""
    n = a.size
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_of(a)

    eta = 0.15
    tau_frac = 0.05
    beta = 0.5
    vel = np.zeros(n)
    it = 0
    stagnation = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() > deadline_global:
            break

        b_conv = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b_conv))
        tau = tau_frac * float(np.max(b_conv))
        if tau < 1e-18:
            tau = 1e-18

        g = compute_smooth_grad(a, b_conv, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = normalize(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = normalize(a - 0.4 * eta * vel)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 500:
            a = normalize(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.08 + 0.15 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()
            beta = 0.3 + 0.4 * rng.random()
            stagnation = 0

        it += 1

    return best_a, best_sc


def polish_random(a, time_limit, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_of(a)
    t0 = time.monotonic()
    scale = 0.03
    stuck = 0

    while time.monotonic() - t0 < time_limit and scale > 1e-7:
        if time.monotonic() > deadline_global:
            break
        improved = False
        n_try = max(5, int(50 * (0.02 / scale)))
        n_try = min(n_try, 200)
        for _ in range(n_try):
            # Random perturbation
            cand = a + scale * rng.standard_normal(n)
            cand = normalize(cand)
            sc = score_of(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True

        if not improved:
            stuck += 1
            if stuck > 8:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def polish_subgrad(a, time_limit, rng):
    """Short subgradient polish with smaller steps."""
    n = a.size
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_of(a)

    eta = 0.05
    tau_frac = 0.03
    vel = np.zeros(n)
    beta = 0.4
    it = 0
    stagnation = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() > deadline_global:
            break
        b_conv = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b_conv))
        tau = tau_frac * float(np.max(b_conv))
        if tau < 1e-18:
            tau = 1e-18

        g = compute_smooth_grad(a, b_conv, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = normalize(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.08:
            a = a_new
        else:
            a = normalize(a - 0.3 * eta * vel)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 2000:
            a = normalize(best_a + 0.01 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.03 + 0.05 * rng.random()
            stagnation = 0

        it += 1

    return best_a, best_sc


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_asymmetric(n, rng):
    """Asymmetric init: high on one side, decaying."""
    i = np.arange(n, dtype=np.float64)
    side = rng.random()
    if side < 0.5:
        a = np.exp(-i / (n * 0.3))
    else:
        a = np.exp(-(n - 1 - i) / (n * 0.3))
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_three_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    centers = sorted(rng.uniform(0.1, 0.9, 3) * n)
    sigma = n * (0.03 + 0.05 * rng.random())
    a = sum(np.exp(-((i - c) / sigma) ** 2) for c in centers)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_comet(n, rng):
    """Sharp peak followed by long tail."""
    i = np.arange(n, dtype=np.float64)
    peak = int(rng.integers(0, n))
    a = np.exp(-np.abs(i - peak) / (n * 0.1))
    a[peak] = 5.0
    a += 0.01
    return a / a.sum()


def get_inits(n, rng):
    inits = [
        init_two_cluster(n, rng),
        init_u_shape(n),
        init_random(n, rng),
        init_asymmetric(n, rng),
        init_three_cluster(n, rng),
        init_comet(n, rng),
    ]
    return inits


# Global deadline set in propose_candidate
deadline_global = 0.0


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based alternating optimization as the primary method, combined with
    subgradient descent, random polishing, multiple initializations, and multiple
    sequence lengths.
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s) - 15.0)
    deadline_global = time.monotonic() + total_budget

    best_overall = None
    best_overall_sc = np.inf

    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = score_of(prev_a)
            print(f"[init] Previous best: n={prev_a.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Define n values to try
    n_list = [128, 256, 512, 1024]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 64 and np_ <= 4096:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    print(f"[plan] n_list={n_list}, budget={total_budget:.1f}s")

    # Phase 1: LP-based optimization for each n (primary method)
    # Allocate time proportionally (smaller n gets more time per init since faster)
    phase1_frac = 0.55
    phase1_time = total_budget * phase1_frac
    raw_w = [1.0 / (0.3 + nv / 512.0) for nv in n_list]
    total_w = sum(raw_w)
    time_per_n = [w / total_w * phase1_time for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline_global:
            break

        t_budget_n = min(time_per_n[idx_n], deadline_global - time.monotonic() - 1.0)
        if t_budget_n < 5.0:
            break

        print(f"\n[phase1] n={nv}, time_budget={t_budget_n:.1f}s")

        inits = get_inits(nv, rng)
        # Add previous best if same length
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # Allocate time per init
        t_per_init = t_budget_n / (len(inits) + 1)

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline_global:
                break
            tb = min(t_per_init, deadline_global - time.monotonic() - 2.0)
            if tb < 3.0:
                break

            print(f"  [init {ii}/{len(inits)}] n={nv}, t={tb:.1f}s")
            a_best, sc_best = lp_iterate(a_init, tb, rng, verbose=(ii == 0))

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [init {ii}] NEW GLOBAL BEST: n={nv}, score={sc_best:.6f}")
            else:
                print(f"  [init {ii}] n={nv}, score={sc_best:.6f}")

        print(f"[phase1] n={nv} done. Global best: {best_overall_sc:.6f}")

    # Phase 2: Run LP from the global best with more time and different mixes
    if best_overall is not None and time.monotonic() < deadline_global - 10.0:
        remain = deadline_global - time.monotonic() - 5.0
        t_phase2 = min(remain * 0.35, 300.0)
        if t_phase2 > 5.0:
            print(f"\n[phase2] LP from best, n={best_overall.size}, t={t_phase2:.1f}s")
            # Try multiple perturbed starts from the best
            n_best = best_overall.size
            for trial in range(5):
                if time.monotonic() > deadline_global:
                    break
                if trial == 0:
                    a_start = best_overall.copy()
                else:
                    noise = 0.02 * rng.standard_normal(n_best)
                    a_start = normalize(best_overall + noise)

                tb = t_phase2 / (5 - trial)
                tb = min(tb, deadline_global - time.monotonic() - 2.0)
                if tb < 3.0:
                    break

                a_best, sc_best = lp_iterate(a_start, tb, rng, verbose=False)
                if sc_best < best_overall_sc:
                    best_overall_sc = sc_best
                    best_overall = a_best.copy()
                    print(f"  [phase2 trial {trial}] NEW BEST: {sc_best:.6f}")

    # Phase 3: Subgradient optimization from best
    if best_overall is not None and time.monotonic() < deadline_global - 10.0:
        remain = deadline_global - time.monotonic() - 5.0
        t_phase3 = min(remain * 0.25, 200.0)
        if t_phase3 > 5.0:
            print(f"\n[phase3] Subgrad from best, n={best_overall.size}, t={t_phase3:.1f}s")
            a_best, sc_best = subgrad_optimize(best_overall, t_phase3, rng, verbose=False)
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [phase3] NEW BEST: {sc_best:.6f}")

    # Phase 4: Polish (random + subgradient)
    if best_overall is not None and time.monotonic() < deadline_global - 3.0:
        remain = deadline_global - time.monotonic() - 2.0
        if remain > 5.0:
            t_pol1 = min(remain * 0.5, 60.0)
            t_pol2 = min(remain * 0.4, 60.0)

            print(f"\n[phase4] Polish, n={best_overall.size}")

            if t_pol1 > 3.0:
                a_best, sc_best = polish_subgrad(best_overall, t_pol1, rng)
                if sc_best < best_overall_sc:
                    best_overall_sc = sc_best
                    best_overall = a_best.copy()
                    print(f"  [polish_subgrad] NEW BEST: {sc_best:.6f}")

            if time.monotonic() < deadline_global - 2.0 and t_pol2 > 3.0:
                a_best2, sc_best2 = polish_random(best_overall, t_pol2, rng)
                if sc_best2 < best_overall_sc:
                    best_overall_sc = sc_best2
                    best_overall = a_best2.copy()
                    print(f"  [polish_random] NEW BEST: {sc_best2:.6f}")

    # Phase 5: Try a few more LP iterations if time remains
    if best_overall is not None and time.monotonic() < deadline_global - 3.0:
        remain = deadline_global - time.monotonic() - 2.0
        if remain > 10.0:
            print(f"\n[phase5] Final LP polish, t={remain:.1f}s")
            a_best, sc_best = lp_iterate(best_overall, remain, rng, verbose=False)
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [phase5] NEW BEST: {sc_best:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(512) / 512.0
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
