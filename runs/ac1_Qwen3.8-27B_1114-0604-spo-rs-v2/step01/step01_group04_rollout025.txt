"""

import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project vector v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, tol=1e-8):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (rescaled to sum 1) or None if infeasible.
    """
    n = f.size
    # Build constraint matrix C: (2n-1) x n
    # C[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # We can build this as a banded matrix
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            C[k, j_start:j_end+1] = f[k - j_start:k - j_end + 1]
    
    # LP: max 1^T g s.t. C g <= M*1, g >= 0
    # linprog minimizes, so: min -1^T g s.t. C g <= M*1, -g <= 0
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

        best_new_sc = cur_sc

            new_sc = normed_score(a_new)

def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def coordinate_descent(a, time_budget, rng):
    """Coordinate descent: optimize one coordinate at a time."""
    n = a.size
    a = a.astype(np.float64).copy()
    best_sc = normed_score(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    improved_any = True
    while improved_any and time.monotonic() - t0 < time_budget:
        improved_any = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            # Try to optimize coordinate i
            # Fix all other coordinates, vary a[i]
            # a[i] >= 0, and we want sum(a) = 1, so we're on the simplex
            # When we change a[i], we need to renormalize
            # Actually, let's just try a few values and pick the best
            orig = a[i]
            candidates = [orig * 0.5, orig * 0.8, orig * 1.2, orig * 1.5,
                         orig * 0.3, orig * 2.0, 0.0, orig * 0.1]
            # Also try moving mass from i to neighbors
            if i > 0:
                candidates.append(None)  # placeholder
            best_local_sc = best_sc
            best_local_val = orig
            
            for cv in candidates:
                if cv is None:
                    continue
                if cv < 0 or orig + cv > 1.0:
                    continue
                a[i] = orig + cv
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a = a / s
                sc = normed_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local_val = orig + cv
            
            a[i] = orig  # restore
            if best_local_val != orig:
                a[i] = best_local_val
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
                    improved_any = True
    
    return best_a, best_sc

def init_asymmetric_gaussian(n, rng):
    """Asymmetric: single Gaussian bump off-center."""
    i = np.arange(n, dtype=np.float64)
    c = n * (0.15 + 0.3 * rng.random())  # center between 0.15n and 0.45n
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_two_bump_asym(n, rng):
    """Two bumps of different sizes, asymmetric."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.5 + 0.3 * rng.random())
    s1 = n * (0.03 + 0.05 * rng.random())
    s2 = n * (0.05 + 0.1 * rng.random())
    h1 = 0.5 + rng.random()
    h2 = 0.3 + 0.5 * rng.random()
    a = h1 * np.exp(-((i - c1) / s1) ** 2) + h2 * np.exp(-((i - c2) / s2) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_power_law(n, rng, side='left'):
    """Power law decay from one side."""
    i = np.arange(n, dtype=np.float64)
    if side == 'left':
        a = 1.0 / (i + 1.0) ** (0.5 + rng.random())
    else:
        a = 1.0 / (n - 1 - i + 1.0) ** (0.5 + rng.random())
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_ramp(n, rng):
    """Linear ramp (asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    if rng.random() < 0.5:
        a = i / (n - 1) + 0.1
    else:
        a = (n - 1 - i) / (n - 1) + 0.1
    a += 0.1 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_quadratic_asym(n, rng):
    """Quadratic centered off-center."""
    i = np.arange(n, dtype=np.float64)
    c = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((i - c) / (n * 0.3)) ** 2) + 0.05
    return a / a.sum()

def get_inits(n, rng):

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based direction finding (SOTA approach) with line search
2. Improved smoothed subgradient descent with adaptive multi-scale temperatures
3. Fast coordinate descent polish

5. Multiple sequence lengths with time-aware allocation

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def multi_scale_grad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad_single(a, b, tau)
    return g_total / n_scales


def lp_direction(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g rescaled to sum 1, or None.
    """
    n = f.size
    # Build constraint matrix C: C[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            C[k, j_start:j_end + 1] = f[k - j_start:k - j_end + 1]

    c_obj = -np.ones(n)
    A_ub = C
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_iterate(a, time_budget, rng):
    """
    LP-based iteration with line search.
    At each step: find best direction via LP, then line search.
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        g = lp_direction(a, M)
        if g is None:
            break

        # Line search: try several step sizes
        best_t = 0.0
        best_new_sc = cur_sc
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = project_simplex((1 - t) * a + t * g)
            new_sc = normed_score(a_new)
            if new_sc < best_new_sc:
                best_new_sc = new_sc
                best_t = t

        if best_t > 0:
            a = project_simplex((1 - best_t) * a + best_t * g)

        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = a.copy()
            last_improve = iter_count
            stall = 0
        else:
            stall += 1
            if stall > 30:
                a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
                stall = 0
                last_improve = iter_count
            elif stall > 80:
                break

        iter_count += 1

    return best_a, best_sc

            tau_frac = 0.02 + 0.05 * rng.random()

def coordinate_descent(a, time_budget, rng):
    """
    Coordinate descent: for each coordinate, try several multipliers and
    accept if the score improves. Efficient for n <= 256.
    """
    n = a.size
    a = a.astype(np.float64).copy()
    best_sc = normed_score(a)
    best_a = a.copy()

    multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]
    rng.shuffle(multipliers)

    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_budget:
        improved = False
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                # Try adding a small value
                for mult in [0.01, 0.005]:
                    a_trial = a.copy()
                    a_trial[i] = a.sum() * mult
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = normed_score(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                continue

            for mult in multipliers:
                new_val = orig * mult
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial = a_trial / s
                sc = normed_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break  # next coordinate

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_sc

            elif nv > 192 and ii < 2 and tb > 5.0:

                lp_time = min(tb * 0.1, 5.0)
                a_p, sc_p = lp_iterate(a_best, lp_time, rng)
                if sc_p < sc_best:
                    a_best, sc_best = a_p, sc_p

            # Phase 2: Subgradient descent
            if time.monotonic() < deadline - 2.0:
                sg_time = min(tb * 0.4, deadline - time.monotonic() - 1.0)
                if sg_time > 1.0:
                    a_s, sc_s = subgrad_optimize(a_best, sg_time, rng)
                    if sc_s < sc_best:
                        a_best, sc_best = a_s, sc_s

            # Phase 3: Coordinate descent polish
            if time.monotonic() < deadline - 2.0 and sc_best <= best_overall_sc * 1.01:
                cd_time = min(tb * 0.2, 30.0, deadline - time.monotonic() - 1.0)
                if cd_time > 1.0:
                    a_c, sc_c = coordinate_descent(a_best, cd_time, rng)
                    if sc_c < sc_best:
                        a_best, sc_best = a_c, sc_c

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f} (NEW BEST!)")
            else:
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f}")

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}")

    # Final extended polish phase on the best solution
    if best_overall is not None and time.monotonic() < deadline - 15.0:
        remain = deadline - time.monotonic() - 5.0
        n_best = best_overall.size
        print(f"[polish] remain={remain:.1f}s, n={n_best}, sc={best_overall_sc:.6f}")

        # LP polish (if n is small enough)
        if n_best <= 192 and remain > 20.0:
            lp_time = min(remain * 0.3, 60.0)
            a_p, sc_p = lp_iterate(best_overall, lp_time, rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] LP -> {sc_p:.6f}")

        # Subgradient polish with different params
        if time.monotonic() < deadline - 5.0:
            sg_time = min(remain * 0.3, 45.0)
            # Use a fresh subgradient run with different momentum
            a_s, sc_s = subgrad_optimize(best_overall, sg_time, rng, use_momentum=True)
            if sc_s < best_overall_sc:
                best_overall_sc = sc_s
                best_overall = a_s.copy()
                print(f"[polish] subgrad -> {sc_s:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 3.0:
            cd_time = min(remain * 0.2, 30.0)
            a_c, sc_c = coordinate_descent(best_overall, cd_time, rng)
            if sc_c < best_overall_sc:
                best_overall_sc = sc_c
                best_overall = a_c.copy()
                print(f"[polish] coord -> {sc_c:.6f}")

        # One more round of LP + coord if time remains
        if time.monotonic() < deadline - 5.0 and n_best <= 192:
            lp_time = min(10.0, deadline - time.monotonic() - 3.0)
            if lp_time > 3.0:
                a_p2, sc_p2 = lp_iterate(best_overall, lp_time, rng)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"[polish2] LP -> {sc_p2:.6f}")

            if time.monotonic() < deadline - 2.0:
                cd_time2 = min(10.0, deadline - time.monotonic() - 2.0)
                if cd_time2 > 2.0:
                    a_c2, sc_c2 = coordinate_descent(best_overall, cd_time2, rng)
                    if sc_c2 < best_overall_sc:
                        best_overall_sc = sc_c2
                        best_overall = a_c2.copy()
                        print(f"[polish2] coord -> {sc_c2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
