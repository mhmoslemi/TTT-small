def lse_value(c, T):
    """Compute log-sum-exp of c with temperature T (numerically stable)."""
    c_max = c.max()
    exp_vals = np.exp((c - c_max) / T)
    return c_max + T * np.log(exp_vals.sum())


def lse_gradient(c, a, T):
    """Compute gradient of LSE w.r.t. a using vectorized correlate."""
    c_max = c.max()
    exp_vals = np.exp((c - c_max) / T)
    p = exp_vals / exp_vals.sum()
    # grad_i = 2 * sum_k p_k * a[k-i] = 2 * correlate(p, a, 'valid')[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad


def softmax_from_z(z):
    """Convert z to probability vector (numerically stable softmax)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def make_obj_grad_factory(T, n):
    """Create objective and gradient functions for L-BFGS in z-space."""

    def obj(z):
        a = softmax_from_z(z)
        c = fast_self_conv(a)
        return lse_value(c, T)

    def grad(z):
        a = softmax_from_z(z)
        c = fast_self_conv(a)
        grad_a = lse_gradient(c, a, T)
        # Chain rule through softmax: dLSE/dz_j = a_j * (grad_a_j - a·grad_a)
        return a * (grad_a - np.dot(a, grad_a))

    return obj, grad


def optimize_lse_anneal(a0, n, deadline, T_schedule, label, verbose=True):
    """Run LSE temperature annealing with L-BFGS."""
    a0 = np.clip(a0, 1e-12, None).astype(np.float64)
    S0 = a0.sum()
    if S0 < 1e-12:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S0

    z = np.log(a0 + 1e-15)
    best_local = compute_score(a0)
    best_local_a = a0.copy()
    t_start = time.monotonic()

    for i, T in enumerate(T_schedule):
        if time.monotonic() > deadline - 10.0:
            break

        obj, grad = make_obj_grad_factory(T, n)

        # Cap iterations based on remaining time
        remaining = deadline - time.monotonic()
        max_iter = max(50, min(500, int(remaining * 50)))

        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'iprint': -1})
        except Exception:
            break

        z = result.x
        # Renormalize z to prevent drift
        z = z - z.mean()
        a = softmax_from_z(z)
        s = compute_score(a)
        if s < best_local:
            best_local = s
            best_local_a = a.copy()
            if verbose:
                print(f"  [{label}] T={T:.2e} -> score={best_local:.6f}")

    return best_local, best_local_a


def active_set_polish(a, n, steps=200, verbose=True):
    """Polish using exact subgradient on the active set (peaks near max)."""
    a = np.clip(a, 1e-12, None).astype(np.float64)
    a = a / a.sum()
    best_local = compute_score(a)
    best_local_a = a.copy()

    for step in range(steps):
        c = fast_self_conv(a)
        c_max = c.max()
        S = a.sum()

        # Active set: peaks within 0.1% of max
        threshold = c_max * 0.999
        active = c >= threshold
        n_active = active.sum()
        if n_active == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1

        # Subgradient: average gradient over active peaks
        p = np.zeros_like(c)
        p[active] = 1.0 / n_active
        grad_a = lse_gradient_weighted(c, a, p, n)

        # Normalize gradient for step size
        g_norm = np.linalg.norm(grad_a)
        if g_norm < 1e-15:
            break

        # Polyak-like step size
        current_score = 2.0 * n * c_max / (S * S)
        step_size = 0.01 / (g_norm + 1e-10)

        # Direction: decrease c_k for active peaks, maintain sum
        # Project gradient onto tangent space of simplex (remove mean component)
        grad_tangent = grad_a - np.dot(a, grad_a)

        a_new = a - step_size * grad_tangent
        a_new = np.clip(a_new, 0.0, None)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            break
        a_new = a_new / a_new_sum

        s_new = compute_score(a_new)
        if s_new < best_local:
            best_local = s_new
            best_local_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            step_size *= 0.5

    return best_local, best_local_a


def lse_gradient_weighted(c, a, weights, n):
    """Compute weighted gradient: sum_k w_k * dc_k/da."""
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad


def init_polynomial(n):
    """Initialize with coefficients of (1-x)^{-1/2}: binom(2j,j)/4^j."""
    a = np.zeros(n)
    for j in range(n):
        # binom(2j, j) / 4^j
        # Use recurrence: c_0 = 1, c_j = c_{j-1} * (2j-1)/(2j)
        if j == 0:
            a[j] = 1.0
        else:
            a[j] = a[j-1] * (2*j - 1) / (2*j)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def init_concentrated(n, width_frac=0.5):
    """Concentrate mass in the middle of the sequence."""
    a = np.zeros(n)
    center = n / 2.0
    sigma = n * width_frac / 4.0
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def init_two_lobe(n, gap_frac=0.3, rng=None):
    """Two separated lobes."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.zeros(n)
    center1 = n * 0.25
    center2 = n * 0.75
    sigma = n * 0.1
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + \
        np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = a / a.sum()
    return a


def init_random_smoothed(n, rng, bandwidth=5):
    """Random sequence with Gaussian smoothing."""
    a = rng.random(n)
    # Simple boxcar smoothing
    kernel = np.ones(bandwidth) / bandwidth
    a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0, None)
    a = a / a.sum()
    return a


def init raised(n, alpha, rng):
    """Raised cosine with random phase."""
    x = np.linspace(0, 1, n)
    phase = rng.random() * 2 * np.pi
    a = 1.0 + alpha * np.cos(2 * np.pi * x + phase)
    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a


def build_initializations(n, rng, parent=None):
    """Build a list of diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0.01:
            inits.append(("parent", p / p.sum()))
    inits.append(("poly", init_polynomial(n)))
    inits.append(("uniform", np.ones(n) / n))
    inits.append(("concentrated", init_concentrated(n)))
    inits.append(("two_lobe", init_two_lobe(n, rng=rng)))
    inits.append(("rand_smooth", init_random_smoothed(n, rng)))
    inits.append(("raised", init_raised(n, 0.5, rng)))
    inits.append(("raised2", init_raised(n, 1.5, rng)))
    # Some random
    for k in range(3):
        a = rng.random(n)
        a = np.clip(a, 0, None)
        a = a / a.sum()
        inits.append((f"rand_{k}", a))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total = float(budget_s)
    deadline = start_time + total - 5.0

    # Try to get parent
    parent = None
    n_default = 2048
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = p
                n_default = p.size
    except Exception:
        pass

    # Candidate lengths to try
    n_candidates = [512, 1024, 2048, 4096]
    if n_default in n_candidates:
        n_candidates.remove(n_default)
    n_candidates.insert(0, n_default)
    # Cap at reasonable size
    n_candidates = [n for n in n_candidates if 256 <= n <= 8192]

    best_overall_score = np.inf
    best_overall_a = None

    print(f"Starting search, budget={total:.0f}s, n_candidates={n_candidates}")

    # Phase 1: Quick exploration across n values (15% of time)
    phase1_end = start_time + 0.15 * (total - 5.0)
    n_results = {}  # n -> (best_score, best_a)

    for n in n_candidates:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        time_budget = min(phase1_end - t0, 120.0)  # Max 2 min per n in phase 1
        if time_budget < 20:
            continue

        inits = build_initializations(n, rng, parent)
        # Use only first 4 initializations in phase 1
        inits = inits[:4]

        local_best = np.inf
        local_best_a = None
        time_per_init = max(10.0, time_budget / len(inits))

        for label, a0 in inits:
            if time.monotonic() > phase1_end - 5:
                break
            T_sched = [0.005, 0.002, 0.001, 0.0005]
            sc, aa = optimize_lse_anneal(a0, n, time.monotonic() + time_per_init,
                                         T_sched, f"n={n}/{label}", verbose=False)
            if sc < local_best:
                local_best = sc
                local_best_a = aa.copy()

        if local_best_a is not None and local_best < n_results.get(n, (np.inf,))[0] if n in n_results else local_best < np.inf:
            n_results[n] = (local_best, local_best_a)
            print(f"  Phase1 n={n}: score={local_best:.6f}")

    # Select best n
    if n_results:
        best_n = min(n_results, key=lambda k: n_results[k][0])
        best_n_score, best_n_a = n_results[best_n]
        if best_n_score < best_overall_score:
            best_overall_score = best_n_score
            best_overall_a = best_n_a.copy()
        print(f"Phase1 best: n={best_n}, score={best_n_score:.6f}")
    else:
        best_n = n_default
        best_n_score = np.inf
        best_n_a = None

    # Phase 2: Thorough optimization at best n (65% of time)
    phase2_end = start_time + 0.80 * (total - 5.0)
    n_opt = best_n

    # Build full set of initializations
    inits = build_initializations(n_opt, rng, parent)
    print(f"Phase 2: optimizing at n={n_opt} with {len(inits)} initializations")

    time_per_init = max(5.0, (phase2_end - time.monotonic()) / len(inits))

    # T schedule for thorough optimization
    T_schedule_full = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]

    for label, a0 in inits:
        if time.monotonic() > phase2_end - 10:
            break

        # Skip if we already tried this in phase 1 and got a good result
        alloc = time_per_init
        sc, aa = optimize_lse_anneal(a0, n_opt, time.monotonic() + alloc,
                                     T_schedule_full, f"p2/{label}", verbose=True)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_a = aa.copy()
            print(f"  Phase2 [{label}] NEW BEST: {best_overall_score:.6f}")

    # Phase 3: Active-set polish + restarts (15% of time)
    phase3_end = deadline
    if best_overall_a is not None:
        print(f"Phase 3: polish, current best={best_overall_score:.6f}")

        # Active set polish
        time_budget = (phase3_end - time.monotonic()) * 0.5
        n_steps = max(50, int(time_budget * 20))
        sc, aa = active_set_polish(best_overall_a, n_opt, steps=n_steps)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_a = aa.copy()
            print(f"  After active-set polish: {best_overall_score:.6f}")

        # Small perturbations + re-optimization
        remaining = phase3_end - time.monotonic()
        n_restarts = max(1, int(remaining / 30))
        for r in range(n_restarts):
            if time.monotonic() > phase3_end - 5:
                break
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_overall_a * (1.0 + scale * rng.standard_normal(n_opt))
            a_pert = np.clip(a_pert, 0, None)
            a_pert = a_pert / a_pert.sum()

            T_short = [0.001, 0.0005, 0.0002]
            alloc = max(5.0, (phase3_end - time.monotonic()) / (n_restarts - r))
            sc, aa = optimize_lse_anneal(a_pert, n_opt, time.monotonic() + alloc,
                                         T_short, f"restart_{r}", verbose=False)
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_a = aa.copy()
                print(f"  Restart {r}: {best_overall_score:.6f}")

    # Also try a different n if time remains
    if time.monotonic() < deadline - 60:
        alt_n = [n for n in n_candidates if n != n_opt]
        if alt_n:
            alt_n_opt = alt_n[0]
            print(f"Trying alternate n={alt_n_opt}")
            inits_alt = build_initializations(alt_n_opt, rng, parent)[:3]
            for label, a0 in inits_alt:
                if time.monotonic() > deadline - 10:
                    break
                sc, aa = optimize_lse_anneal(a0, alt_n_opt, deadline - 5,
                                             T_schedule_full[:5], f"alt_n/{label}",
                                             verbose=False)
                if sc < best_overall_score:
                    best_overall_score = sc
                    best_overall_a = aa.copy()
                    print(f"  Alt n={alt_n_opt} [{label}]: {best_overall_score:.6f}")

    # Final: verify and return
    if best_overall_a is None:
        best_overall_a = np.ones(n_default) / n_default
        best_overall_score = compute_score(best_overall_a)

    out = [float(x) for x in best_overall_a]
    # Verify with the actual evaluator
    try:
        verified = evaluate_sequence(out)
        print(f"Final: internal={best_overall_score:.6f}, "
              f"verified={verified:.6f}, len={len(out)}, "
              f"time={time.monotonic()-start_time:.1f}s")
        if verified < best_overall_score:
            best_overall_score = verified
    except Exception as e:
        print(f"Verification issue: {e}")

    return out
