import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-6
_G_BOUND = 30.0


def _set_gn(n):
    global _G_N, _G_NFFT
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()


def _set_gtau(t):
    global _G_TAU
    _G_TAU = float(t)


def _gfftc(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _gsc(a):
    """Score: 2*n*max_conv / sum^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _gfftc(a)
    return float(2 * n * np.max(c) / (s * s))


def _gsm(z):
    """Softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size)
    return e / s


def _gog(z):
    """Objective + gradient for L-BFGS-B (uses _G_N, _G_TAU)."""
    n = _G_N
    N = _G_NFFT
    tau = _G_TAU
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))
    z = np.clip(z, -_G_BOUND, _G_BOUND)
    a = _gsm(z)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))

        return 1e12, np.zeros(n)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    scale = float(n)
    val *= scale
    grad *= scale
    val += 1e-8 * 0.5 * float(np.dot(z, z))
    grad = grad + 1e-8 * z

    if not np.isfinite(val):

    return float(val), grad


def _glb(z0, mf=500, mi=150):

            "ftol": 1e-14, "pgtol": 1e-12, "disp": False}
    try:

                        bounds=bnds, options=opts)
    except Exception:
        return None

"""
Multi-scale, multi-tau gradient optimization for flat autocorrelation sequences.

Algorithm:
1. Coarse optimization at small n to find a good "shape".
2. Upsample to target n and refine with L-BFGS-B using multiple tau values.
3. Aggressive multi-tau gradient polishing with line search.
4. Random escape moves to avoid local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-6
_G_BOUND = 30.0


def _set_gn(n):
    global _G_N, _G_NFFT
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()


def _set_gtau(t):
    global _G_TAU
    _G_TAU = float(t)


def _gfftc(a):
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _gsc(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _gfftc(a)
    return float(2 * n * np.max(c) / (s * s))


def _gsm(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size, dtype=np.float64)
    return e / s


def _gog(z):
    n = _G_N
    N = _G_NFFT
    tau = _G_TAU
    L = 2 * n - 1
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))
    z = np.clip(z, -_G_BOUND, _G_BOUND)
    a = _gsm(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g
    val = m + tau * np.log(se)
    scale = float(n)
    val *= scale
    grad *= scale
    val += 1e-8 * 0.5 * float(np.dot(z, z))
    grad = grad + 1e-8 * z
    if not np.isfinite(val):
        return 1e12, np.zeros(n)
    return float(val), grad


def _glb(z0, mf=500, mi=150):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    if z0.size != _G_N:
        return None
    z0 = np.clip(z0 - np.mean(z0), -_G_BOUND, _G_BOUND)
    bnds = [(-_G_BOUND, _G_BOUND)] * _G_N
    opts = {"maxfun": int(mf), "maxiter": int(mi),
            "ftol": 1e-14, "pgtol": 1e-12, "disp": False}
    try:
        return minimize(_gog, z0, jac=True, method="L-BFGS-B",
                        bounds=bnds, options=opts)
    except Exception:
        return None


def _gpol(a0, dl, rng_seed=54321):
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-12:
        return a0
    a = a / s
    b = np.log(np.maximum(a, 1e-300))
    b = b - np.mean(b)
    b = np.clip(b, -_G_BOUND, _G_BOUND)
    best = a.copy()
    best_F = float(np.max(_gfftc(a)))
    rng = np.random.default_rng(rng_seed)
    ni = 0
    it = 0

    while time.monotonic() < dl:
        c = _gfftc(a)
        F = float(np.max(c))
        if F < best_F - 1e-18:
            best = a.copy()
            best_F = F

        N = _G_NFFT
        fa = np.fft.rfft(a, N)
        best_step_b = None
        best_step_a = None
        best_step_F = F

        for tau_mult in (1e-3, 3e-3, 1e-2, 3e-2, 1e-1):
            tau = max(best_F * tau_mult, 1e-18)
            diff = (c - F) / tau
            diff = np.clip(diff, -745.0, 0.0)
            e = np.exp(diff)
            se = float(np.sum(e))
            if se <= 0 or not np.isfinite(se):
                continue
            p = e / se
            fp = np.fft.rfft(p, N)
            h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
            s_h = float(np.dot(h, a))
            g = a * (h - s_h)
            gn = np.sqrt(float(np.dot(g, g)))
            if gn < 1e-15:
                continue
            g = g / gn
            for eta in (1e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1):
                b_t = np.clip(b - eta * g * n, -_G_BOUND, _G_BOUND)
                a_t = _gsm(b_t)
                F_t = float(np.max(_gfftc(a_t)))
                if F_t < best_step_F:
                    best_step_F = F_t
                    best_step_b = b_t.copy()
                    best_step_a = a_t.copy()

        if best_step_b is not None:
            b = best_step_b
            a = best_step_a
            ni = 0
        else:
            esc = False
            for sc in (0.01, 0.05, 0.15, 0.5, 1.0, 2.0):
                b_t = np.clip(b + sc * rng.standard_normal(n), -_G_BOUND, _G_BOUND)
                a_t = _gsm(b_t)
                F_t = float(np.max(_gfftc(a_t)))
                if F_t < best_F - 1e-16:
                    b, a = b_t, a_t
                    esc = True
                    break
            if not esc:
                ni += 1
                if ni > 300:
                    break
            else:
                ni = 0

        it += 1
        if it % 300 == 0:
            cur_sc = 2.0 * n * float(np.max(_gfftc(a)))
            print(f"    pol it={it} cur={cur_sc:.9f} best={2*n*best_F:.9f} "
                  f"t_left={max(0,dl-time.monotonic()):.0f}s")

    return best


def _upsample_z(z_small, n_large):
    n_small = len(z_small)
    if n_small == n_large:
        return z_small.copy()
    x_s = np.linspace(0.0, 1.0, n_small, endpoint=True)
    x_l = np.linspace(0.0, 1.0, n_large, endpoint=True)
    return np.interp(x_l, x_s, z_small)

        if np.isfinite(sc):

            f += coeff * rng.standard_normal() * np.cos(2*np.pi*m*u + rng.uniform(0, 2*np.pi))
        f = np.maximum(f, 1e-8)
        z = np.log(f) - np.mean(np.log(f))
        z = np.clip(z, -_G_BOUND, _G_BOUND)
        starts.append((_gsc(_gsm(z)), z))

    starts = [(s, z) for s, z in starts if np.isfinite(s)]
    starts.sort(key=lambda t: t[0])
    return starts

            if res is not None and hasattr(res, "x"):
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                z_new = np.clip(z_new, -_G_BOUND, _G_BOUND)
                a_new = _gsm(z_new)
                sc_new = _gsc(a_new)
                if np.isfinite(sc_new) and sc_new < sc_cur:
                    sc_cur = sc_new
                    z_cur = z_new.copy()
                    print(f"    tau={tau:.1e}: {sc_cur:.9f}")

        a_cur = _gsm(z_cur)
        candidates.append((sc_cur, z_cur, a_cur))

    candidates.sort(key=lambda t: t[0])
    print(f"\n  After L-BFGS: best={candidates[0][0]:.9f}")

    # Phase 2: Aggressive polishing
    print(f"\n=== Polishing phase ===")
    pol_time = deadline - time.monotonic()
    print(f"  Time available: {pol_time:.0f}s")

    # Polish the best candidate with most of the time
    if candidates and pol_time > 30:
        sc_best, z_best, a_best = candidates[0]
        pol_dl = deadline
        print(f"  Polishing best (score={sc_best:.9f}) for {pol_time:.0f}s")
        polished = _gpol(a_best, pol_dl, rng_seed=999)
        psc = _gsc(polished)
        print(f"  Polished: {psc:.9f}")
        if psc < sc_best:
            a_best = polished
            sc_best = psc

    # If time remains, polish second candidate too
    if len(candidates) > 1 and time.monotonic() < deadline - 30:
        sc2, z2, a2 = candidates[1]
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"  Polishing 2nd (score={sc2:.9f}) for {remaining:.0f}s")
            polished2 = _gpol(a2, deadline, rng_seed=777)
            psc2 = _gsc(polished2)
            print(f"  Polished 2nd: {psc2:.9f}")
            if psc2 < sc_best:
                a_best = polished2
                sc_best = psc2

    # Final validation
    best_a = np.asarray(a_best, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(target_n) / target_n
    else:
        best_a = best_a / s

    final_sc = _gsc(best_a)
    print(f"\nFinal internal score: {final_sc:.9f}")

    try:
        exact = evaluate_sequence(best_a.tolist())
        print(f"Official evaluator score: {float(exact):.9f}")
    except Exception:
        pass

    return [float(v) for v in best_a.tolist()]
