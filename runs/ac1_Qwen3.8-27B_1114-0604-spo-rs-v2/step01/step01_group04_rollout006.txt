def max_conv_direct(a):
    """Compute max of conv(a,a) directly for small n."""
    n = len(a)

def max_conv_fft(a):

    n = len(a)

    fa = np.fft.rfft(a, fsize)
    b = np.fft.irfft(fa * fa, fsize)[:size]

    b = max_conv_fft(a)
    return 2.0 * n * b / (s * s)

def coord_descent(a, max_sweeps=100, time_budget=60, rng=None):
    """Coordinate descent to minimize max_conv with sum(a)=1 constraint."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a /= s
    
    n = len(a)
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    sweep = 0
    while sweep < max_sweeps and time.monotonic() - t0 < time_budget:
        # Random order for coordinates
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Get current value
            x0 = a[i]
            S = 1.0 - x0  # sum of other elements
            
            # Ternary search for optimal x in [0, S]
            lo = 0.0
            hi = S
            
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                a[i] = m1
                f1 = max_conv_fft(a)
                
                a[i] = m2
                f2 = max_conv_fft(a)
                
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            a[i] = (lo + hi) / 2.0
            
            # Check improvement
            cur_f = max_conv_fft(a)
            if cur_f < best_score / (2.0 * n):  # best_score = 2n*max_f, so max_f_best = best_score/(2n)
                best_score = 2.0 * n * cur_f
                best_a = a.copy()
                improved = True
        
        sweep += 1
        
        if not improved and sweep > 5:
            break
    
    return best_a, best_score

So I can:
1. Precompute the full convolution once: Q_k for all k.
2. For coordinate i, extract the n affected lags.
3. For each affected lag k, determine c_k and d_k (the coefficient and constant of x).
4. The objective is max(max of unaffected Q_k's, max of affected Q_k(x) for x).
5. Do ternary search on this.

This reduces each 1D evaluation from O(n log n) (FFT) to O(n) (just evaluating the n affected lags). And the precomputation is O(n log n) once per sweep.
