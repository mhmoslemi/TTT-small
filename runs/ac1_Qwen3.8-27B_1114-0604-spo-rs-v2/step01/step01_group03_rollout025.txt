"""
Optimize non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth mirror descent (log-sum-exp approximation of max) with adaptive step size
and temperature annealing, followed by coordinate descent polishing for final refinement.
Key improvements over previous: better temperature schedule, adaptive step size with
backtracking, support identification, and coordinate descent polishing.
"""

import math
import time
import numpy as np


def _conv_self(a, L, m):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _soft_gradient(c, a, L, n, temp):
    m = c.size
    cmax = float(np.max(c))
    z = (c - cmax) / temp
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(m, dtype=np.float64) / float(m)
    else:
        p = e / s
    ar = a[::-1]
    p_fft = np.fft.rfft(p, L)
    ar_fft = np.fft.rfft(ar, L)
    conv_full = np.fft.irfft(p_fft * ar_fft, L)
    grad = conv_full[n - 1:2 * n - 1].copy()
    np.maximum(grad, 0.0, out=grad)
    return grad


def _normalize_pos(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def _hard_subgradient(c, a, n):
    k_star = int(np.argmax(c))
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    idx_start = k_star - lo
    idx_end = k_star - hi
    g[lo:hi + 1] = a[idx_start:idx_end - 1:-1]
    return g


def _topk_subgradient(c, a, n, k=5):
    cmax = float(np.max(c))
    top_idx = np.argsort(c)[::-1][:k]
    weights = np.zeros(len(top_idx), dtype=np.float64)
    total = 0.0
    for i, idx in enumerate(top_idx):
        w = math.exp(-(cmax - c[idx]) / max(cmax * 1e-3, 1e-10))
        weights[i] = w
        total += w
    if total < 1e-30:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= total
    g = np.zeros(n, dtype=np.float64)
    for w, k_star in zip(weights, top_idx):
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx_start = k_star - lo
        idx_end = k_star - hi
        g[lo:hi + 1] += w * a[idx_start:idx_end - 1:-1]
    return g

    a = _normalize_pos(a).copy()
    c = _conv_self(a, L, m)
    current_score = float(2.0 * n * np.max(c))
    best_a = a.copy()
    best_score = current_score
    temp = temp0
    eta = eta0
    it = 0
    no_accept = 0
    stall_count = 0

    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        current_score = float(2.0 * n * np.max(c))

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 500:
            noise_scale = 0.05 + 0.1 * rng.random()
            noise = rng.normal(0.0, noise_scale, n)
            a = _normalize_pos(a + noise * a.max() * 0.3)
            temp = max(temp0 * 0.5, temp * 0.5)
            eta = eta0
            no_accept = 0
            stall_count = 0
            continue

        frac = min(1.0, float(it) / max(1, float(int(time_limit * 200))))
        temp = temp0 + (temp1 - temp0) * frac
        temp = max(temp, n * 1e-4)

        grad = _soft_gradient(c, a, L, n, temp)
        gmax = float(np.max(grad))
        if gmax < 1e-30:

        grad = grad / gmax

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = eta * mult
            if step < 1e-10:

            new_a = a * np.exp(-step * grad)
            new_a = _normalize_pos(new_a)
            new_c = _conv_self(new_a, L, m)
            new_score = float(2.0 * n * np.max(new_c))
            if new_score < current_score - 1e-14:

                eta = eta0
                no_accept = 0
                temp = max(temp0 * 0.3, temp * 0.5)

                a = new_a
                accepted = True
                break

        if accepted:
            no_accept = 0
            eta = min(3.0, eta * 1.05)
        else:
            no_accept += 1
            eta = max(0.005, eta * 0.75)
            if no_accept > 80:
                a = best_a.copy()
                eta = 0.5
                no_accept = 0

        it += 1

    return best_a, best_score


def _coordinate_descent(a, L, m, n, time_limit, rng):
    deadline = time.monotonic() + time_limit
    a = _normalize_pos(a).copy()
    c = _conv_self(a, L, m)
    best_score = float(2.0 * n * np.max(c))
    best_a = a.copy()
    order = np.arange(n)

    pass_count = 0
    while time.monotonic() < deadline:
        np.random.shuffle(order)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-12:
                continue
            affected_lo = max(0, i)
            affected_hi = min(m - 1, i + n - 1)
            k2i = 2 * i
            c_prime = {}
            for k in range(affected_lo, affected_hi + 1):
                if k == k2i:
                    c_prime[k] = c[k] - old_val * old_val
                elif 0 <= k - i < n:
                    c_prime[k] = c[k] - 2.0 * old_val * a[k - i]
                else:
                    c_prime[k] = c[k]
            max_unaffected = float(np.max(np.delete(c, range(affected_lo, affected_hi + 1))))
            candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5]
            best_t = old_val
            best_mc = float(2.0 * n * np.max(c))
            for t in candidates:
                max_c = max_unaffected
                for k in range(affected_lo, affected_hi + 1):
                    if k == k2i:
                        val = c_prime[k] + t * t
                    elif 0 <= k - i < n and k != k2i:
                        val = c_prime[k] + 2.0 * t * a[k - i]
                    else:
                        val = c_prime[k]
                    if val > max_c:
                        max_c = val
                sc = 2.0 * n * max_c
                if sc < best_mc:
                    best_mc = sc
                    best_t = t
            if best_t != old_val:
                a[i] = best_t
                a = _normalize_pos(a)
                c = _conv_self(a, L, m)
                current_sc = float(2.0 * n * np.max(c))
                if current_sc < best_score - 1e-12:
                    best_score = current_sc
                    best_a = a.copy()
        pass_count += 1

    return best_a, best_score


def _support_reduction(a, L, m, n, threshold=1e-6):
    c = _conv_self(a, L, m)
    current_score = float(2.0 * n * np.max(c))
    amax = float(np.max(a))
    zeros = np.where(a < threshold * amax)[0]
    if len(zeros) == 0:
        return a, current_score
    a2 = a.copy()
    a2[zeros] = 0.0
    a2 = _normalize_pos(a2)
    c2 = _conv_self(a2, L, m)
    score2 = float(2.0 * n * np.max(c2))
    if score2 < current_score:
        return a2, score2
    return a, current_score


def _try_different_lengths(a_base, L_base, m_base, n_base, L, m, n, rng, time_per_n=30.0):
    results = []
    if n == n_base:
        results.append((a_base.copy(), float(2.0 * n_base * np.max(_conv_self(a_base, L_base, m_base)))))
    remaining = time_limit
    for target_n in [n_base // 2, n_base * 2, int(n_base * 0.75), int(n_base * 1.5)]:
        if target_n < 100 or target_n > 15000:
            continue
        L_t = 1 << (3 * target_n - 2).bit_length()
        m_t = 2 * target_n - 1
        resampled = np.interp(
            np.linspace(0, 1, target_n),
            np.linspace(0, 1, n_base),
            a_base
        )
        resampled = _normalize_pos(resampled)
        lim = min(time_per_n, remaining / 4)
        if lim < 5:
            break
        tmp, sc = _mirror_descent_smooth(resampled, L_t, m_t, target_n, rng, lim, target_n * 0.1, target_n * 0.005)
        results.append((tmp, sc))
    return results


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global height_sequence_1
    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10:
        budget = 2400.0
    deadline = t_start + budget - 3.0

    rng = np.random.default_rng(seed)

    n = 6121
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if base.size != n:
            n = base.size
    except (NameError, Exception):
        base = np.ones(n, dtype=np.float64)

    a0 = np.maximum(base, 0.0)
    s0 = float(a0.sum())
    if s0 < 1e-10:
        a0 = np.ones(n, dtype=np.float64)
    a0 = a0 / s0

    m = 2 * n - 1
    L = 1 << (3 * n - 2).bit_length()
    print(f"n={n}, m={m}, L={L}, budget={budget}s")

    c = _conv_self(a0, L, m)
    score0 = float(2.0 * n * np.max(c))
    print(f"Initial exact score: {score0:.6f}")

    best_a = a0.copy()
    best_score = score0

    # Phase 1: Smooth mirror descent with annealing (55% of time)
    phase1_time = 0.55 * (deadline - t_start)
    print(f"Phase 1: smooth MD, time budget = {phase1_time:.1f}s")

    # Multiple rounds with decreasing base temperature
    num_rounds = 4
    round_time = phase1_time / num_rounds
    for rnd in range(num_rounds):
        frac_base = 0.15 * (1.0 - rnd / num_rounds) + 0.005
        temp_start = n * frac_base
        temp_end = max(n * 0.002, temp_start * 0.1)
        remaining_rnd = min(round_time, deadline - time.monotonic() - 5.0)
        if remaining_rnd < 5.0:
            break
        tmp, sc = _mirror_descent_smooth(
            best_a, L, m, n, rng, remaining_rnd,
            temp_start, temp_end, eta0=1.5
        )
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = tmp
            print(f"  Round {rnd+1}: score improved to {sc:.6f}")
        else:
            print(f"  Round {rnd+1}: no improvement (best={best_score:.6f}, got={sc:.6f})")
            # Try with a perturbed start
            noise = rng.normal(0, 0.05, n)
            perturbed = _normalize_pos(best_a + noise * best_a.max() * 0.3)
            # Force zero some small coordinates
            amax = float(np.max(perturbed))
            small = np.where(perturbed < 0.01 * amax)[0]
            if len(small) > 0:
                perturbed[small[:len(small)//4]] = 0.0
                perturbed = _normalize_pos(perturbed)
            tmp2, sc2 = _mirror_descent_smooth(
                perturbed, L, m, n, rng, remaining_rnd * 0.6,
                temp_start * 1.5, temp_end, eta0=1.0
            )
            if sc2 < best_score - 1e-12:
                best_score = sc2
                best_a = tmp2
                print(f"  Round {rnd+1} (perturbed): score improved to {sc2:.6f}")

    # Phase 2: Hard subgradient (top-k) optimization (25% of time)
    phase2_time = 0.25 * (deadline - t_start - (time.monotonic() - t_start))
    phase2_time = min(phase2_time, deadline - time.monotonic() - 30.0)
    if phase2_time > 10.0:
        print(f"Phase 2: hard subgradient, time budget = {phase2_time:.1f}s")
        # Top-1
        tmp, sc = _mirror_descent_hard(best_a, L, m, n, rng, phase2_time * 0.5, use_topk=1)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = tmp
            print(f"  Top-1: score improved to {sc:.6f}")
        # Top-3
        remaining2 = min(phase2_time * 0.5, deadline - time.monotonic() - 20.0)
        if remaining2 > 5.0:
            tmp, sc = _mirror_descent_hard(best_a, L, m, n, rng, remaining2, use_topk=3)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = tmp
                print(f"  Top-3: score improved to {sc:.6f}")

    # Phase 3: Support reduction + local refinement (10% of time)
    phase3_time = min(0.10 * (deadline - t_start), deadline - time.monotonic() - 20.0)
    if phase3_time > 5.0:
        print(f"Phase 3: support reduction + refinement, time budget = {phase3_time:.1f}s")
        for thresh in (1e-5, 5e-6, 1e-6, 5e-7):
            reduced, sc = _support_reduction(best_a, L, m, n, threshold=thresh)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = reduced
                print(f"  Support reduction (thresh={thresh}): score = {sc:.6f}")
            remaining3 = min(10.0, deadline - time.monotonic() - 15.0)
            if remaining3 > 3.0:
                tmp, sc = _mirror_descent_smooth(
                    best_a, L, m, n, rng, remaining3,
                    n * 0.02, n * 0.002, eta0=0.8
                )
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = tmp
                    print(f"  Refine after reduction: score = {sc:.6f}")

    # Phase 4: Coordinate descent polishing (10% of time)
    phase4_time = min(0.10 * (deadline - t_start), deadline - time.monotonic() - 10.0)
    if phase4_time > 5.0:
        print(f"Phase 4: coordinate descent, time budget = {phase4_time:.1f}s")
        tmp, sc = _coordinate_descent(best_a, L, m, n, phase4_time, rng)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = tmp
            print(f"  Coord descent: score improved to {sc:.6f}")
        # One more round of smooth MD on the CD result
        remaining4 = min(20.0, deadline - time.monotonic() - 8.0)
        if remaining4 > 3.0:
            tmp, sc = _mirror_descent_smooth(
                best_a, L, m, n, rng, remaining4,
                n * 0.03, n * 0.003, eta0=0.5
            )
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = tmp
                print(f"  Final polish: score improved to {sc:.6f}")

    # Final exact evaluation
    print(f"Best internal score: {best_score:.6f}")
    seq = [float(x) for x in best_a]

    # Verify with evaluate_sequence if available
    try:
        exact = evaluate_sequence(seq)
        print(f"Exact evaluate_sequence score: {exact}")
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
    except NameError:
        # Fallback: compute exactly
        arr = np.array(seq)
        conv_arr = np.convolve(arr, arr)
        exact = float(2.0 * n * np.max(conv_arr) / (np.sum(arr) ** 2))
        print(f"Exact computed score: {exact}")

    # Final time check - if we have time, try one more thing
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        print(f"Extra time remaining: {remaining:.1f}s, trying final restart...")
        # Try a few random restarts from the best
        for attempt in range(3):
            if time.monotonic() > deadline - 10.0:
                break
            noise = rng.normal(0, 0.02 + 0.02 * attempt, n)
            perturbed = _normalize_pos(best_a + noise * best_a.max() * 0.1)
            lim = min(15.0, (deadline - time.monotonic() - 8.0) / (3 - attempt))
            if lim < 3.0:
                break
            tmp, sc = _mirror_descent_smooth(
                perturbed, L, m, n, rng, lim,
                n * 0.05, n * 0.003, eta0=1.0
            )
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = tmp
                seq = [float(x) for x in best_a]
                print(f"  Final attempt {attempt+1}: score improved to {sc:.6f}")

    return seq
