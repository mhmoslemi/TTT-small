"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses focused multi-peak gradient descent with mirror steps, spectral
regularization, momentum, and structured perturbations for escape.
"""

import numpy as np
from scipy.signal import correlate
import time


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation without calling evaluate_sequence."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _gradient_at_lags(f, lags, weights):
    """
    Compute weighted gradient of sum_w w_t * conv(f,f)[k_t] w.r.t. f.
    grad[j] = 2 * sum_t w_t * f[k_t - j] (where valid)
    """
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        # f[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # indices into f: k - j for j in [j_start, j_end]
            # = k - j_start down to k - j_end
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    return grad


def _mirror_step(f, grad, step, momentum_vec=None, beta=0.9):
    """Exponentiated gradient (mirror descent) step with momentum."""
    if momentum_vec is not None:
        grad_eff = grad + beta * momentum_vec
        new_momentum = grad + beta * momentum_vec
    else:
        grad_eff = grad
        new_momentum = np.zeros_like(grad)

    # Scale by range for numerical stability
    span = grad_eff.max() - grad_eff.min()
    if span < 1e-30:
        return f, new_momentum
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad_eff
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f, new_momentum
    f_new /= s
    return f_new, new_momentum


def _project_nonneg_sum1(f):
    """Project onto {x >= 0, sum x = 1}."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _spectral_regularizer(f, n_harmonics=5):
    """Penalty for non-flat power spectrum (encourages low autocorrelation)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    power = np.abs(F) ** 2
    # Focus on low-mid frequencies where convolution peaks tend to come from
    mid = size // 2
    power_mid = power[1:min(n, mid)]  # exclude DC
    if len(power_mid) == 0:
        return 0.0
    mean_p = power_mid.mean()
    if mean_p < 1e-30:
        return 0.0
    var_p = np.var(power_mid) / (mean_p * mean_p)
    return var_p


def _structured_perturb(f, rng, mode=0):
    """Apply a structured perturbation and return modified f (non-neg, sum=1)."""
    n = len(f)
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()

    if mode == 0:
        # Exponential tilting
        alpha = rng.uniform(-1.0, 1.0)
        tilt = np.exp(alpha * (j / n - 0.5))
        f_try = f * tilt
    elif mode == 1:
        # Sinusoidal modulation
        freq = rng.integers(1, max(2, n // 4) + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        f_try = f * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        # Windowing (Hamming-like)
        width = rng.uniform(0.3, 0.8)
        center = rng.uniform(0.2, 0.8)
        w = 1.0 - width * ((j / n - center) / (1 - center)) ** 2
        f_try = f * np.maximum(w, 0.1)
    elif mode == 3:
        # Bimodal injection
        c1, c2 = sorted(rng.uniform(0.1, 0.9, size=2))
        bw = rng.uniform(0.05, 0.2)
        boost = np.exp(-((j / n - c1) / bw) ** 2) + np.exp(-((j / n - c2) / bw) ** 2)
        f_try = f * (1.0 + 0.3 * boost)
    elif mode == 4:
        # Random local swap
        i1 = int(rng.integers(0, n // 2))
        i2 = int(rng.integers(n // 2, n))
        amt = f[i1] * rng.uniform(0.05, 0.3)
        f_try[i1] -= amt
        f_try[i2] += amt
    elif mode == 5:
        # Multi-frequency modulation
        f_try = f.copy()
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 3) + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            f_try = f_try * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    else:
        # Pure random
        f_try = f + rng.uniform(-0.05, 0.05, size=n) * f.mean()

    f_try = np.maximum(f_try, 0.0)
    s = f_try.sum()
    if s < 1e-10:
        return f.copy()
    return f_try / s


def _optimize_focused(f0, n, deadline, rng, max_iters=100000, verbose=False, label=""):
    """
    Focused gradient descent with:
    - Top-K peak targeting
    - Momentum
    - Spectral regularizer
    - Adaptive step size
    - Structured perturbations on stagnation
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.85
    K = 4  # number of top peaks to target
    reg_weight = 0.01  # spectral regularizer weight

    step_base = 0.8
    step_min = 0.002
    n_total = max_iters

    stuck_count = 0
    patience = 8
    improvement_thresh = 1e-7
    last_eval_score = local_best

    for it in range(n_total):
        if time.monotonic() > deadline:
            break

        progress = it / n_total

        # Adaptive step size: large early, small late
        step = step_base * (step_min / step_base) ** (progress ** 0.6)

        # Compute convolution
        c = _autocorr_fft(f)
        peak = c.max()

        # Find top K lags
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]  # sort descending
        top_vals = c[top_idx]

        # Weights: emphasize the top peaks more strongly
        # Use inverse-gap weighting
        gaps = top_vals[0] - top_vals
        gaps = np.maximum(gaps, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Compute gradient at top lags
        grad = _gradient_at_lags(f, top_idx, weights)

        # Add spectral regularizer gradient (approximate)
        if reg_weight > 0 and it % 10 == 0:
            # Gradient of spectral variance - approximate by modulating f
            # Simple approach: penalize positions where f is correlated with its own shifts
            # Use a cheap proxy: penalty on f * reversed_f (the middle convolution)
            n_mid = n - 1
            if 0 <= n_mid < 2 * n - 1:
                mid_grad = np.zeros(n)
                for j in range(n):
                    idx = n_mid - j
                    if 0 <= idx < n:
                        mid_grad[j] = f[idx]
                grad += reg_weight * mid_grad

        # Mirror step with momentum
        f, momentum = _mirror_step(f, grad, step, momentum, beta)
        f = _project_nonneg_sum1(f)

        # Periodic evaluation
        eval_interval = max(100, 5000 // max(1, stuck_count))
        if it % eval_interval == 0 or it == n_total - 1:
            score = _score_fast(f)
            if score < local_best - improvement_thresh:
                local_best = score
                local_best_f = f.copy()
                stuck_count = 0
                last_eval_score = score
                if verbose and it % 5000 == 0:
                    print(f"    {label} iter {it}: {score:.7f}")
            else:
                stuck_count += 1

            # If stuck, apply structured perturbation
            if stuck_count >= patience:
                mode = int(rng.integers(0, 6))
                f = _structured_perturb(f, rng, mode)
                momentum *= 0.5  # damp momentum after perturbation
                stuck_count = 0

                # Also try: reduce K temporarily for exploration
                # and increase step
                step *= 2.0

        # Periodically reduce momentum for stability
        if it % 10000 == 0 and it > 0:
            momentum *= 0.5

    # Final evaluation
    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best


def _polish_coordinate(f, n, deadline, rng):
    """
    Coordinate descent polish: iteratively find the peak lag and
    transfer mass from high-contribution positions to low ones.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()

    c = _autocorr_fft(f)
    peak = c.max()

    n_passes = 500
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Find top 3 peak lags
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]

        improved = False
        for k_star in top3:
            # Contribution of each position to c[k_star]
            contrib = np.zeros(n)
            j_start = max(0, k_star - n + 1)
            j_end = min(n - 1, k_star)
            if j_start <= j_end:
                idx = k_star - np.arange(j_start, j_end + 1)
                contrib[j_start:j_end + 1] = f[np.arange(j_start, j_end + 1)] * f[idx]

            # Find best donor (high contrib) and receiver (low total contribution)
            total_contrib = np.zeros(n)
            for k in top3:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js <= je:
                    idx_k = k - np.arange(js, je + 1)
                    total_contrib[js:je + 1] += f[np.arange(js, je + 1)] * f[idx_k]

            # Donor: position with highest total contribution
            j_don = int(np.argmax(total_contrib))
            # Receiver: position with lowest total contribution
            j_rec = int(np.argmin(total_contrib))

            if j_don == j_rec or total_contrib[j_don] < 1e-15:
                continue

            # Try several transfer fractions
            best_local = None
            best_amt = 0
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                amt = f[j_don] * frac
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                if f_try[j_don] < -1e-15:
                    continue
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    if best_local is None or c_try.max() < best_local:
                        best_local = c_try.max()
                        best_amt = amt

            if best_local is not None:
                f[j_don] -= best_amt
                f[j_rec] += best_amt
                f = np.maximum(f, 0)
                f /= f.sum()
                c = _autocorr_fft(f)
                peak = c.max()
                improved = True
                local_best = _score_fast(f)
                local_best_f = f.copy()

        if not improved and p > 50:
            # Try random coordinate moves
            for _ in range(5):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                f_try = f.copy()
                amt = f_try[i] * 0.1
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    local_best = _score_fast(f)
                    local_best_f = f.copy()
                    break

    return local_best_f, local_best


def _finetune(f, n, deadline, rng):
    """
    Final fine-tuning: very small mirror steps focused on the single peak lag.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    K = 2
    step = 0.01

    n_iters = 50000
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        peak = c.max()

        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        grad = _gradient_at_lags(f, top_idx, weights)

        # Decrease step over time
        step = 0.01 * (0.001 / 0.01) ** (it / n_iters)

        f, momentum = _mirror_step(f, grad, step, momentum, beta=0.9)
        f = _project_nonneg_sum1(f)

        if it % 2000 == 0:
            score = _score_fast(f)
            if score < local_best:
                local_best = score
                local_best_f = f.copy()

    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Multiple diverse starting points with spectral awareness.
    2. Focused multi-peak gradient descent with momentum and mirror steps.
    3. Spectral regularizer to encourage flat power spectrum.
    4. Structured perturbations (exponential tilting, sinusoidal, windowing).
    5. Coordinate descent polish.
    6. Ultra-fine final tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    t_start = time.monotonic()

    # Load initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)

    best_score = np.inf
    best_f = f_init.copy()
    s0 = best_f.sum()
    if s0 > 1e-10:
        best_f = best_f / s0
    else:
        best_f = np.ones(n)

    initial_score = _score_fast(best_f)
    print(f"Initial sequence: n={n}, score={initial_score:.7f}")
    best_score = initial_score

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n)
    starts['left_ramp'] = np.linspace(0.01, 1.0, n)
    starts['right_ramp'] = np.linspace(1.0, 0.01, n)
    starts['u_wide'] = 1.0 + 1.5 * ((j - center) / max(center, 1)) ** 2
    starts['u_narrow'] = 1.0 + 8.0 * ((j - center) / max(center, 1)) ** 2
    starts['bimodal'] = (np.exp(-((j / n - 0.25) / 0.1) ** 2) +
                         np.exp(-((j / n - 0.75) / 0.1) ** 2) + 0.01)
    starts['asym_peak'] = np.exp(-((j / n - 0.6) / 0.25) ** 2) + 0.01
    starts['asym_peak2'] = np.exp(-((j / n - 0.4) / 0.2) ** 2) + 0.01
    starts['sine_mod'] = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * j / n)
    starts['sine_abs'] = 1.0 + 0.6 * np.abs(np.sin(np.pi * j / n))
    starts['quad_ramp'] = 1.0 + 2.0 * (j / n) ** 2
    starts['exp_decay_l'] = np.exp(-2.0 * j / n) + 0.01
    starts['exp_decay_r'] = np.exp(-2.0 * (n - 1 - j) / n) + 0.01
    starts['random1'] = rng.exponential(1.0, size=n)
    starts['dirichlet'] = rng.dirichlet(np.ones(n) * 0.1)

    # Evaluate and sort starts
    start_results = []
    f_given = best_f.copy()
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _score_fast(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("\nStarting point scores (sorted):")
    for sc, name, _ in start_results[:8]:
        print(f"  {name}: {sc:.7f}")

    # Update best if any start is better
    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()

    # Phase 1: Optimize from top starts
    # Time allocation: give more time to better starts
    time_for_opt = (budget_s - 300) * 0.7  # leave 300s for polish + finetune
    n_starts_to_try = 5

    print(f"\n=== Phase 1: Optimizing from top {n_starts_to_try} starts ===")
    for idx in range(n_starts_to_try):
        if time.monotonic() > deadline - 300:
            break

        sc, name, f0 = start_results[idx]
        # Time allocation: decreasing for worse starts
        t_alloc = time_for_opt / n_starts_to_try * (1.0 - 0.12 * idx)
        start_deadline = time.monotonic() + t_alloc

        print(f"  Optimizing '{name}' (initial={sc:.6f}, budget={t_alloc:.0f}s)")

        f_opt, sc_opt = _optimize_focused(
            f0, n, start_deadline, rng,
            max_iters=80000, verbose=True, label=name
        )

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> Best from this start: {sc_opt:.7f}")

    # Phase 2: Additional optimization from the current best with different seeds
    print(f"\n=== Phase 2: Additional runs from best (score={best_score:.7f}) ===")
    time_phase2 = min(300.0, (deadline - time.monotonic()) * 0.4)
    if time_phase2 > 60:
        n_extra = 3
        t_each = time_phase2 / n_extra
        for i in range(n_extra):
            if time.monotonic() > deadline - 200:
                break
            # Perturb the best solution
            f_pert = _structured_perturb(best_f, rng, mode=i % 6)
            d2 = time.monotonic() + t_each
            f_opt, sc_opt = _optimize_focused(
                f_pert, n, d2, rng,
                max_iters=50000, verbose=False, label=f"extra{i}"
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"    extra{i}: new best {best_score:.7f}")
            else:
                print(f"    extra{i}: {sc_opt:.7f}")

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate polish (score={best_score:.7f}) ===")
    polish_budget = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.4))
    polish_deadline = time.monotonic() + polish_budget
    f_pol, sc_pol = _polish_coordinate(best_f, n, polish_deadline, rng)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"    Polished to: {best_score:.7f}")
    else:
        print(f"    Polish result: {sc_pol:.7f}")

    # Phase 4: Ultra-fine tuning
    print(f"\n=== Phase 4: Fine tuning (score={best_score:.7f}) ===")
    fine_budget = min(max(60.0, (deadline - time.monotonic()) * 0.5), 180.0)
    fine_deadline = time.monotonic() + fine_budget
    f_fine, sc_fine = _finetune(best_f, n, fine_deadline, rng)
    if sc_fine < best_score:
        best_score = sc_fine
        best_f = f_fine.copy()
        print(f"    Fine-tuned to: {best_score:.7f}")
    else:
        print(f"    Fine tune: {sc_fine:.7f}")

    # Phase 5: Final micro-polish if time remains
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final micro-polish ===")
        micro_deadline = deadline - 5.0
        f_micro, sc_micro = _polish_coordinate(best_f, n, micro_deadline, rng)
        if sc_micro < best_score:
            best_score = sc_micro
            best_f = f_micro.copy()
            print(f"    Micro-polished to: {best_score:.7f}")

    # Validate with the official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation score: {final_score:.7f}")
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback: use the official score
        best_score = final_score if np.isfinite(final_score) else initial_score

    print(f"\n=== FINAL SCORE: {best_score:.7f} ===")
    print(f"Total time used: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_f]
