"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses 1/sqrt(k+1) initialization (theoretical flat-convolution base),
peak-focused mirror descent with adaptive temperature,
equalization polish, and multi-length search.
"""

import numpy as np
import time


def _conv_fft(f):
    """Compute linear convolution f*f via FFT, length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_norm(f):
    """Score assuming sum(f)=1: 2*n*max(conv(f,f))."""
    n = len(f)
    c = _conv_fft(f)
    return 2.0 * n * c.max()


def _grad_fft(p, f):
    """Compute gradient: grad[m] = 2 * sum_k p[k] * f[k-m]."""
    n = len(f)
    f_rev = f[::-1]
    size = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, size)
    FR = np.fft.rfft(f_rev, size)
    conv = np.fft.irfft(P * FR, size)
    return 2.0 * conv[n - 1:2 * n - 1]


def _optimize_peaks(f0, deadline, rng, label="", max_iters=120000):
    """
    Mirror descent optimization with adaptive peak focusing.
    Phase 1: Soft-max over all lags (broad)
    Phase 2: Focus on top-K peaks (narrow)
    Phase 3: Fine tuning with only the single peak
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n)
    else:
        f /= s

    best_score = _score_norm(f)
    best_f = f.copy()

    c = _conv_fft(f)
    peak0 = c.max()

    # Learn rate schedule
    lr_init = 1.5
    lr_end = 0.003
    # Temperature: start broad, end narrow
    T_init = peak0 * 5.0
    T_end = peak0 * 1e-10

    eval_every = 800
    it = 0
    no_improve = 0
    patience = 30

    while it < max_iters and time.monotonic() < deadline:
        progress = min(1.0, it / max_iters)

        # Phase-based temperature
        if progress < 0.3:
            T = T_init * (T_init / peak0) ** (progress / 0.3)
            lr = lr_init * (0.1 / lr_init) ** (progress / 0.3)
        elif progress < 0.7:
            T = (peak0 / T_init) * (peak0 / (peak0 * 1e-5)) ** ((progress - 0.3) / 0.4)
            lr = 0.1 * (lr_end / 0.1) ** ((progress - 0.3) / 0.4)
        else:
            T = (peak0 * 1e-5) * (T_end / (peak0 * 1e-5)) ** ((progress - 0.7) / 0.3)
            lr = lr_end * (0.3 / 1.0) ** ((progress - 0.7) / 0.3)

        # Compute softmax weights
        cs = c / max(T, 1e-300)
        cs -= cs.max()
        p = np.exp(cs)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(2 * n - 1)
            cidx = np.argmax(c)
            # Spread over top few
            topk = np.argsort(c)[-min(10, 2 * n - 1):]
            p[topk] = 1.0 / len(topk)
        else:
            p /= ps

        # Gradient
        grad = _grad_fft(p, f)

        # Mirror descent step
        span = grad.max() - grad.min()
        if span > 1e-14:
            eta = lr / span
            log_f = np.log(f + 1e-30) - eta * grad
            log_f -= log_f.max()
            f = np.exp(log_f)
            f_sum = f.sum()
            if f_sum > 1e-300:
                f /= f_sum
            else:
                f = best_f.copy()

        it += 1

        # Periodic evaluation
        if it % eval_every == 0 or it == max_iters:
            c = _conv_fft(f)
            sc = _score_norm(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            # If stuck, add perturbation
            if no_improve > patience:
                mag = 0.002 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs > 1e-12:
                    f /= fs
                else:
                    f = best_f.copy()
                no_improve = 0
                c = _conv_fft(f)
        else:
            c = _conv_fft(f)

    return best_f, best_score


def _equalize_polish(f, deadline, rng, label="", max_passes=300):
    """
    Polish by direct peak equalization:
    Identify top peaking lags, transfer mass to reduce the highest peak.
    """
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    best_score = _score_norm(f)
    best_f = f.copy()

    peak = c.max()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _conv_fft(f)
        peak = c.max()
        peak_idx = np.argmax(c)

        # Gradient of c[peak_idx] w.r.t. f: grad[m] = 2*f[peak_idx - m]
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * f[peak_idx - hi:peak_idx - lo:-1]

        # We want to move mass from high-gradient positions to low-gradient ones
        # while keeping sum(f) constant
        grad_centered = grad - grad.mean()

            f_try = np.maximum(f_try, 0.0)
            fs = f_try.sum()
            if fs < 1e-12:

            f_try /= fs
            c_try = _conv_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                sc = _score_norm(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                improved = True
                break

        if not improved:

                i = int(rng.integers(0, n))

                        sc = _score_norm(f)

                        improved = True
                        break

        if p % 50 == 0:
            sc = _score_norm(f)
            print(f"  {label} polish pass {p}: peak={peak:.8f}, score={sc:.6f}")

    return best_f, best_score


def _make_start_1sqrt(n):
    """Theoretical flat-convolution starting point: f(k) = 1/sqrt(k+1)."""
    k = np.arange(n, dtype=np.float64)
    return 1.0 / np.sqrt(k + 1.0)


def _make_start_1sqrt_tilted(n, tilt):
    """Tilted version: f(k) = 1/sqrt(k+1) * (1 + tilt * (k/n - 0.5))."""
    k = np.arange(n, dtype=np.float64)
    w = 1.0 + tilt * (k / n - 0.5)
    return (1.0 / np.sqrt(k + 1.0)) * w


def _make_start_power(n, alpha):
    """Power law: f(k) = 1/(k+1)^alpha."""
    k = np.arange(n, dtype=np.float64)
    return 1.0 / (k + 1.0) ** alpha


def _make_start_windowed(n, beta):
    """Windowed 1/sqrt: f(k) = 1/sqrt(k+1) * (1 - beta * cos(pi*k/(n-1))^2)."""
    k = np.arange(n, dtype=np.float64)
    w = 1.0 - beta * np.cos(np.pi * k / (n - 1)) ** 2
    return (1.0 / np.sqrt(k + 1.0)) * np.maximum(w, 0.1)
