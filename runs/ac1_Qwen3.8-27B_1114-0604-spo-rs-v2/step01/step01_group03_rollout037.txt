```python
"""
Search for non-negative step heights minimizing
    2*n*max(conv(a, a)) / (sum(a)**2).

The final program uses:
  - FFT-based exact self-convolution scores,
  - multi-scale starts (small -> larger lengths),
  - a local power-weighted gradient / Frank-Wolfe search,
  - an auxiliary q-side alignment step that keeps a replica q minimizing p*q,
  - occasional random reheats and active-set focusing.

It returns the best sequence found within the requested time budget.
"""

import math
import time
import numpy as np


def _key_score(item):
    return item[0]


def _normalize(a):
    """Stable normalization to the simplex, replacing bad values."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.size)
    a /= s
    return a


def _normalize_fast(a):
    """Fast normalization, assuming finite non-negative input."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.shape, dtype=np.float64) / float(a.size)
    a = a.copy()
    a /= s
    return a


def _resample_sequence(a, n):
    """Resample a non-negative sequence to length n by linear interpolation."""
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0 or n <= 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    if arr.size == n:
        return _normalize(arr)
    x_old = (np.arange(arr.size, dtype=np.float64) + 0.5) / float(arr.size)
    x_new = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    y = np.interp(x_new, x_old, arr)
    return _normalize(y)


def _push_shape(out, arr):
    """Add arr and its reversal to the candidate list after normalization."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        return
    out.append(_normalize(arr))
    out.append(_normalize(arr[::-1].copy()))


def _random_prior(n, rng):
    """Random smooth non-negative prior."""
    z = rng.normal(0.0, 1.0, n)
    if n > 25:
        k = 11
        z = np.convolve(z, np.ones(k, dtype=np.float64) / float(k), mode="same")

    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x = 2.0 * t - 1.0

    z += 0.65 * float(rng.normal()) * x
    z += 0.65 * float(rng.normal()) * (x * x - 1.0 / 3.0)

    kmax = int(rng.integers(3, 7))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.55) / math.sqrt(float(k))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos((2.0 * np.pi * float(k)) * t + ph)

    z -= float(np.max(z))
    p = np.exp(z)
    p = np.maximum(p, 0.0)
    return _normalize(p)


def _shape_candidates(n, rng, n_random=8):
    """Generate a pool of structured starting sequences."""
    out = []
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    eps = 1e-12

    # Power ramps: includes the strong continuous linear ramp limit.
    for a in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        _push_shape(out, (t + eps) ** a)

    # Exponential ramps, both directions.
    for lam in (-3.0, -1.5, -0.7, 0.7, 1.5, 3.0):
        _push_shape(out, np.exp(lam * (2.0 * t - 1.0)))

    # Linear functions 1 + eps*(t-1/2), clipped later.
    for e in (-2.0, -1.5, -1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0):
        _push_shape(out, 1.0 + e * (t - 0.5))

    # Beta-like products.
    betas = (
        (1.0, 0.0), (0.0, 1.0), (0.5, 0.5), (1.0, 1.0),
        (1.5, 0.5), (0.5, 1.5), (2.0, 0.5), (0.5, 2.0), (2.0, 1.0),
    )
    for a, b in betas:
        _push_shape(out, (t + eps) ** a * (1.0 - t + eps) ** b)

    # Modulated exponentials.
    for A in (0.8, 1.6, 2.8):
        for ph in (0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi):
            arr = np.exp(
                A * np.sin(2.0 * np.pi * t + ph)
                + 0.5 * A * np.sin(4.0 * np.pi * t + 1.3 * ph)
            )
            _push_shape(out, arr)

    # Uniform baseline.
    _push_shape(out, np.ones(n, dtype=np.float64))

    # Random priors.
    for _ in range(max(0, int(n_random))):
        r = _random_prior(n, rng)
        out.append(r)
        out.append(r[::-1].copy())

    return out


def _initial_candidates(n, rng, global_seq, stage_seq, max_count=80):
    out = []

    if stage_seq is not None:
        r = _resample_sequence(stage_seq, n)
        if r is not None:
            out.append(r)

    if global_seq is not None:
        r = _resample_sequence(global_seq, n)
        if r is not None:
            out.append(r)
        try:
            gr = _resample_sequence(np.asarray(global_seq, dtype=np.float64)[::-1], n)
            if gr is not None:
                out.append(gr)
        except Exception:
            pass

    out.extend(_shape_candidates(n, rng, n_random=6))

    if len(out) > max_count:
        # Score-free trimming: keep structured and recent starts.
        # The run-length scorer will rank them anyway.
        out = out[:max_count]
    return out


def _fft_length(n):
    m = 2 * int(n) - 1
    L = 1 << (m - 1).bit_length()
    return int(L), int(m)


def _make_phase(n, L):
    ks = np.arange(L // 2 + 1, dtype=np.float64)
    phase = np.exp(-2j * np.pi * ks * (float(n) - 1.0) / float(L))
    phase[0] = 1.0 + 0.0j
    return phase


def _self_conv(a, L, m):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L, n=m)
    np.maximum(c, 0.0, out=c)
    return c


def _self_conv_and_fft(a, L, m):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L, n=m)
    np.maximum(c, 0.0, out=c)
    return c, fa


def _score_from_conv(c, n):
    mc = float(np.max(c)) if c.size else 0.0
    return float(2.0 * float(n) * mc)


def _make_weights(v, q):
    """
    Weights for active lags. For q > 1000, use a hard active set of the
    top lags, which approximates the subgradient of the max.
    """
    m = int(v.size)
    vmax = float(np.max(v)) if m else 0.0
    if not np.isfinite(vmax) or vmax <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)

    q = float(q)
    if q <= 1e-6:
        return np.full(m, 1.0 / m, dtype=np.float64)

    if q > 1000.0:
        K = max(1, min(64, m // 400))
        if K >= m:
            return np.full(m, 1.0 / m, dtype=np.float64)
        lam = np.zeros(m, dtype=np.float64)
        if K == 1:
            lam[int(np.argmax(v))] = 1.0
        else:
            idx = np.argpartition(v, K)[-K:]
            lam[idx] = 1.0 / float(K)
        return lam

    r = v / vmax
    lam = np.power(r, q)
    s = float(lam.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    lam /= s
    return lam


def _gradient_from_weights(lamF, P_rev, L, n):
    """
    Given lambda_f and P_rev, compute g_i = sum_k lam_k p_{k-i}.
    This is the sub-gradient direction (up to a factor) for the weighted
    sum of self-convolution coordinates.
    """
    g = np.fft.irfft(lamF * P_rev, L, n=L)
    g = g[n - 1:n - 1 + int(n)]
    np.maximum(g, 0.0, out=g)
    return g


def _accept_candidate(sc2, dot2, sc, base_dot, best_score):
    """
    Accept if exact score improves. Also allow small score plateaus if the
    weighted surrogate decreases, to help leave flat maxima.
    """
    limit = sc
    if sc > best_score + 1e-6:
        limit = best_score
    return (sc2 < limit - 1e-12) or (sc2 <= limit + 1e-7 and dot2 < base_dot - 1e-9)


def _evaluate_candidate(p2, lam, L, m, n):
    c2 = _self_conv(p2, L, m)
    sc2 = _score_from_conv(c2, n)
    dot2 = float(np.dot(lam, c2))
    return sc2, dot2


def _align_q(p, q, P, P_rev, L, n, steps, eta_q):
    """
    Align q to p by reducing the cross-convolution p*q. This is a heuristic
    side distribution often useful as a mixing direction.
    """
    q = _normalize_fast(q)
    slc = slice(n - 1, n - 1 + int(n))
    for _ in range(max(1, int(steps))):
        Y = np.fft.rfft(q, L)
        U = P * Y
        gq = np.fft.irfft(U * P_rev, L, n=L)
        gq = gq[slc]
        np.maximum(gq, 0.0, out=gq)
        gmax = float(np.max(gq))
        if not np.isfinite(gmax) or gmax <= 0.0:
            break
        dq = gq / gmax
        np.exp(-float(eta_q) * dq, out=dq)
        q *= dq
        q = _normalize_fast(q)
    return q


def _local_search(p0, L, m, n, rng, time_limit, phase, q0=4.0, qmax=2000.0):
    """
    Main local search:
      - power-weighted multiplicative descent on high self-convolution lags,
      - Frank-Wolfe moves toward low-gradient positions,
      - occasional q-side alignment and mixing.
    """
    p = _normalize(p0)
    if time_limit is None or float(time_limit) <= 0.1:
        c = _self_conv(p, L, m)
        return p, _score_from_conv(c, n)

    deadline = time.monotonic() + float(time_limit)

    c0 = _self_conv(p, L, m)
    best = p.copy()
    best_score = _score_from_conv(c0, n)

    eta = 0.55
    fw_lambda = 0.12
    q = float(q0)
    B = max(1, min(int(n), int(n) // 20))

    no_accept = 0
    no_best = 0
    it = 0
    last_align = -1000
    qside = p.copy()

    align_every = 25 if n > 10000 else 16
    align_steps = 3
    reheat_limit = 420
    slc = slice(n - 1, n - 1 + int(n))

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.35:
            break

        c, P = _self_conv_and_fft(p, L, m)
        sc = _score_from_conv(c, n)

        if sc < best_score - 1e-12:
            best_score = sc
            best = p.copy()
            no_best = 0
        else:
            no_best += 1

        v = float(n) * c
        lam = _make_weights(v, q)
        base_dot = float(np.dot(lam, c))

        P_rev = P.conj() * phase
        P_rev[0] = P[0]
        lamF = np.fft.rfft(lam, L)
        g = _gradient_from_weights(lamF, P_rev, L, n)

        gmax = float(np.max(g)) if g.size else 0.0
        if not np.isfinite(gmax) or gmax <= 0.0:
            d = np.full(n, 1.0 / float(n), dtype=np.float64)
        else:
            d = g / gmax

        accepted = False
        action = 0

        # Occasional q-side alignment / mixing.
        if it - last_align >= align_every and rem > 2.0:
            qside = _align_q(p, qside, P, P_rev, L, n, align_steps, 0.8)

            # If qside itself is better, switch to it.
            cq = _self_conv(qside, L, m)
            scq = _score_from_conv(cq, n)
            dotq = float(np.dot(lam, cq))
            if _accept_candidate(scq, dotq, sc, base_dot, best_score):
                p = qside
                accepted = True
                action = 2
            else:
                for mix in (0.25, 0.50, 0.75):
                    p2 = (1.0 - mix) * p + mix * qside
                    p2 = _normalize_fast(p2)
                    sc2, dot2 = _evaluate_candidate(p2, lam, L, m, n)
                    if _accept_candidate(sc2, dot2, sc, base_dot, best_score):
                        p = p2
                        accepted = True
                        action = 2
                        break
            last_align = it

        if not accepted:
            # Frank-Wolfe style move toward low-gradient coordinates.
            if no_accept >= 6 and (it % 4 == 0):
                action = 1
                if B < int(n):
                    idx = np.argpartition(g, B)[:B]
                    cnt = int(idx.size)
                else:
                    idx = np.arange(n)
                    cnt = int(n)
                qdir = np.zeros(n, dtype=np.float64)
                qdir[idx] = 1.0 / float(cnt)

                for lamf in (fw_lambda, min(1.0, 2.0 * fw_lambda + 0.05)):
                    p2 = (1.0 - lamf) * p + lamf * qdir
                    p2 = _normalize_fast(p2)
                    sc2, dot2 = _evaluate_candidate(p2, lam, L, m, n)
                    if _accept_candidate(sc2, dot2, sc, base_dot, best_score):
                        p = p2
                        accepted = True
                        break
            else:
                # Multiplicative power-gradient direction.
                action = 0
                for e in (eta, max(0.03, 0.5 * eta)):
                    np.exp(-e * d, out=d)
                    p2 = p * d
                    p2 = _normalize_fast(p2)
                    sc2, dot2 = _evaluate_candidate(p2, lam, L, m, n)
                    if _accept_candidate(sc2, dot2, sc, base_dot, best_score):
                        p = p2
                        accepted = True
                        d = g / gmax if gmax > 0 else d
                        break

        if accepted:
            no_accept = 0
            if action == 0:
                eta = min(3.0, eta * 1.10)
            elif action == 1:
                fw_lambda = min(0.85, fw_lambda * 1.18)
            else:
                # q-side move accepted; keep steps roughly unchanged.
                pass
        else:
            no_accept += 1
            if action == 0:
                eta = max(0.03, eta * 0.80)
            elif action == 1:
                fw_lambda = max(0.03, fw_lambda * 0.80)

        if no_accept >= 14:
            q = min(float(qmax), q * 1.7)
            no_accept = 0

        if no_best >= reheat_limit:
            p = best.copy()
            prior = _random_prior(n, rng)
            mix = 0.22 if best_score < 1e9 else 0.5
            p = _normalize_fast((1.0 - mix) * p + mix * prior)
            qside = p.copy()
            eta = 0.55
            fw_lambda = 0.12
            q = float(q0)
            no_accept = 0
            no_best = 0
            last_align = it

        it += 1

    return best, float(best_score)


def _run_length(n, inits, time_limit, rng, global_seq=None, stage_seq=None):
    """Run candidate ranking and local searches for one sequence length."""
    if n <= 0 or time_limit is None or float(time_limit) <= 1.0:
        return _make_fallback_sequence(n) if n > 0 else None, float("inf")

    L, m = _fft_length(n)
    phase = _make_phase(n, L)

    scored = []
    seen = set()
    for a in inits:
        if a is None:
            continue
        a = _normalize(a)
        key = (int(n), tuple(a[::max(1, a.size // 97)]))
        if key in seen:
            continue
        seen.add(key)
        c = _self_conv(a, L, m)
        s = _score_from_conv(c, n)
        if np.isfinite(s):
            scored.append((s, a))

    if not scored:
        fallback = _make_fallback_sequence(n)
        c = _self_conv(fallback, L, m)
        return fallback, _score_from_conv(c, n)

    scored.sort(key=_key_score)
    top = scored[:5]

    weights = None
    if len(top) == 1:
        weights = [1.0]
    elif len(top) == 2:
        weights = [0.65, 0.35]
    elif len(top) == 3:
        weights = [0.58, 0.27, 0.15]
    elif len(top) == 4:
        weights = [0.52, 0.26, 0.14, 0.08]
    else:
        weights = [0.48, 0.24, 0.14, 0.09, 0.05]

    best_p, best_s = top[0][1], top[0][0]
    start = time.monotonic()
    total = float(time_limit)

    q0 = 3.0 if n < 4000 else 4.5
    qmax = 2000.0

    for (s0, a), w in zip(top, weights):
        remain = total - (time.monotonic() - start)
        if remain <= 4.0:
            break
        alloc = max(3.0, remain * float(w))
        res_p, res_s = _local_search(a, L, m, n, rng, alloc, phase, q0=q0, qmax=qmax)
        if res_s < best_s:
            best_s = res_s
            best_p = res_p

    return best_p, float(best_s)


def _make_fallback_sequence(n=8192):
    t = (np.arange(int(n), dtype=np.float64) + 0.5) / float(n)
    return _normalize((t + 1e-9) ** 1.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Entry point invoked by the evaluator. Returns a list of non-negative
    heights optimizing the objective.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 2400.0

    start = time.monotonic()
    hard_deadline = start + max(5.0, float(budget) - 5.0)

    # Reserve time at the end for sanity checks / final exact call.
    reserve = max(80.0, 0.04 * budget)
    search_deadline = start + max(10.0, float(budget) - reserve)
    if search_deadline > hard_deadline - 3.0:
        search_deadline = hard_deadline - 3.0

    rng = np.random.default_rng(int(seed) if np.isfinite(float(seed)) else 12345)

    try:
        global_seq = height_sequence_1  # noqa: F821
    except Exception:
        global_seq = None

    best_seq = None
    best_score = float("inf")
    prev_stage_seq = None

    if budget >= 1800.0:
        stages = [
            (1024, 0.040),
            (4096, 0.140),
            (13000, 0.820),
        ]
    elif budget >= 800.0:
        stages = [
            (1024, 0.040),
            (6121, 0.300),
            (11500, 0.660),
        ]
    elif budget >= 400.0:
        stages = [
            (1024, 0.050),
            (7168, 0.950),
        ]
    else:
        stages = [
            (4096, 1.000),
        ]

    search_time_total = max(5.0, search_deadline - start)

    for idx, (n, frac) in enumerate(stages):
        now = time.monotonic()
        if now >= search_deadline - 2.0:
            break
        alloc = max(5.0, search_time_total * float(frac))
        # Do not overrun search deadline.
        alloc = min(alloc, search_deadline - time.monotonic())
        if alloc < 15.0 and idx > 0:
            break

        inits = _initial_candidates(n, rng, global_seq, prev_stage_seq)
        res_p, res_s = _run_length(n, inits, alloc, rng)

        if res_p is not None and np.isfinite(res_s):
            if res_s < best_score:
                best_score = float(res_s)
                best_seq = res_p
            prev_stage_seq = res_p if res_p.size >= n else prev_stage_seq
            print(f"stage n={n} alloc={alloc:.1f}s current_stage_score={res_s:.10f}")
        else:
            print(f"stage n={n} failed")

    if best_seq is None:
        best_seq = _make_fallback_sequence(6121)

    # Final sanitization.
    arr = np.asarray(best_seq, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    arr /= s

    seq = [float(x) for x in arr]

    # Optional exact check near the end, if enough time remains.
    if time.monotonic() < hard_deadline - 20.0:
        try:
            ex = evaluate_sequence(seq)  # noqa: F821
            print(f"final_internal_best={best_score:.10f} final_exact={ex:.10f}")
        except Exception:
            pass

    return seq
```
