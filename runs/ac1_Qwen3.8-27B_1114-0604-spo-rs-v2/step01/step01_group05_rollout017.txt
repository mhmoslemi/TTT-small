def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_grad(b, beta):
    """
    Smooth max of b and gradient weights.
    Returns (G, w) where G ~= max(b) and w_j = dG/db_j.
    """
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast local objective: 2*n*max(p*p) with sum(p)=1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):

    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """
    Return s_i = sum_j w_j * p_{j-i} for i=0..n-1.
    Used for gradient computation.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def compute_gradient(p, beta):
    """Compute gradient of n*smooth_max(p*p) w.r.t. p."""
    n = p.size
    b = fft_convolution(p)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    grad = 2.0 * svec
    return G, grad, w

def block_to_p(n, B, c):
    """Convert relative block log-weights c (length B-1) to full sequence summing to 1."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_grad(b, beta)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)

def adam_projected_gradient(p0, deadline, rng, beta=100000.0, lr=0.005,

    v = np.zeros(n, dtype=np.float64)  # Second moment

    no_improve = 0

    beta_end = beta

        t += 1

        # Anneal beta

        if not np.isfinite(G):
            break

        cur_score = float(2.0 * n * float(np.max(b)))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 5000 == 0:
            print(f"  adam it={t} best={best_score:.8f} beta={cur_beta:.0f} lr={lr:.4e}")

        if no_improve > patience:
            # Try adapting: reduce beta or restart
            lr *= 0.5
            no_improve = 0
            if lr < 1e-8:
                break

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad = grad - float(np.mean(grad))  # Center for simplex projection

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive step: scale by 1/sqrt(v_hat) per coordinate
        step = lr / (np.sqrt(v_hat) + eps)
        q = p - step * m_hat
        q = project_simplex(q, 1.0)

        # Check improvement
        qb = fft_convolution(q)
        q_score = float(2.0 * n * float(np.max(qb)))
        if q_score < cur_score + 1e-12:
            p = q
        else:
            # Try smaller step
            q = p - 0.5 * step * m_hat
            q = project_simplex(q, 1.0)
            qb = fft_convolution(q)
            q_score = float(2.0 * n * float(np.max(qb)))
            if q_score < cur_score + 1e-12:
                p = q
            else:
                lr *= 0.7

        # Occasionally increase lr if things are going well
        if no_improve == 0 and t % 1000 == 0 and lr < 0.02:
            lr *= 1.5

    return best_p, best_score


def projected_gradient_nesterov(p0, deadline, beta=200000.0, max_iter=10000000):
    """Projected gradient with Nesterov momentum."""
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-14:
        p = normalize_simplex(p * (1.0 + 0.001 * np.random.default_rng(12345).standard_normal(n)))

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    momentum = np.zeros(n, dtype=np.float64)

    # Line search step
    step = 1.0 / float(np.linalg.norm(p))
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        G, w = smooth_max_and_grad(b, beta)
        if not np.isfinite(G):
            break

        cur_score = float(2.0 * n * float(np.max(b)))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 3000 == 0:
            print(f"  nesterov it={it} best={best_score:.8f} step={step:.4e}")

        if no_improve > 4000:
            step *= 0.5
            no_improve = 0
            if step < 1e-12:
                break

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Nesterov: look-ahead
        p_look = project_simplex(p + 0.9 * momentum, 1.0)
        b_look = fft_convolution(p_look)
        G_look, w_look = smooth_max_and_grad(b_look, beta)
        if not np.isfinite(G_look):
            w_look = w
            G_look = G
        svec_look = corr_pw(p_look, w_look)
        svec_look = np.maximum(svec_look, 0.0)
        grad_look = 2.0 * svec_look
        grad_look = grad_look - float(np.mean(grad_look))

        # Update
        momentum = 0.9 * momentum - step * grad_look
        p_new = project_simplex(p + momentum, 1.0)

        # Check
        qb = fft_convolution(p_new)
        q_score = float(2.0 * n * float(np.max(qb)))
        if q_score < cur_score + 1e-10:
            p = p_new
            if no_improve == 0:
                step = min(step * 1.1, 1.0)
        else:
            step *= 0.5
            momentum = np.zeros(n, dtype=np.float64)

    return best_p, best_score


def coordinate_descent_refine(p0, deadline, beta=500000.0, max_rounds=1000000):
    """
    Coordinate descent: iterate over coordinates, for each one find the optimal
    value (given others fixed) that minimizes the smooth max.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    rng = np.random.default_rng(777)

    # Sample a subset of coordinates for each round to avoid full O(n^2) cost
    sample_size = min(n, max(64, n // 4))

    round_idx = 0
    while round_idx < max_rounds and time.monotonic() < deadline:
        round_idx += 1

        if round_idx % 1000 == 0:
            print(f"  coord it={round_idx} best={best_score:.8f}")

        if no_improve > 20000:
            break

        # Pick random subset of coordinates
        if sample_size < n:
            idx = rng.choice(n, size=sample_size, replace=False)
        else:
            idx = np.arange(n)
            rng.shuffle(idx)

        improved = False
        for i in idx:
            if time.monotonic() >= deadline:
                break

            # Try changing p[i] while keeping sum = 1
            # Optimal p[i] minimizes max(b) where b = p*p
            # Approximate: try a few values around current
            p_i_old = p[i]
            # The gradient component for coordinate i tells us direction
            b = fft_convolution(p)
            svec = corr_pw(p, np.ones_like(b) / float(b.size))
            svec = np.maximum(svec, 0.0)

            # Try a small adjustment
            for delta_frac in (0.01, 0.005, -0.01, -0.005, 0.001, -0.001):
                delta = delta_frac * p_i_old
                if delta == 0:
                    continue
                p_new = p.copy()
                p_new[i] = max(0.0, p_i_old + delta)
                # Redistribute
                s_new = float(np.sum(p_new))
                if s_new <= 0:
                    continue
                p_new /= s_new

                # Quick score check (only check if this coordinate matters)
                # Full score is expensive, so use a proxy
                # Actually for coordinate descent we need the full score
                # But that's O(n log n) per evaluation, and we're doing many...
                # Let's be more selective: only evaluate if the gradient suggests improvement
                break

            # Use gradient-based single-coordinate update
            b = fft_convolution(p)
            G, w = smooth_max_and_grad(b, beta)
            if not np.isfinite(G):
                break
            svec = corr_pw(p, w)
            svec = np.maximum(svec, 0.0)
            grad_i = 2.0 * svec[i]
            mean_grad = float(np.mean(2.0 * svec))

            # Single coordinate step
            if grad_i > mean_grad:
                # This coordinate contributes more than average to the peak
                # Try to decrease it
                for frac in (0.1, 0.05, 0.02, 0.01):
                    delta = -frac * p_i_old
                    p_new = p.copy()
                    p_new[i] = max(0.0, p_i_old + delta)
                    s_new = float(np.sum(p_new))
                    if s_new <= 0:
                        continue
                    p_new /= s_new
                    new_b = fft_convolution(p_new)
                    new_M = float(np.max(new_b))
                    old_M = float(np.max(b))
                    if new_M < old_M - 1e-15:
                        p = p_new
                        improved = True
                        break
            elif grad_i < mean_grad:
                # This coordinate contributes less than average
                # Try to increase it (shift weight from high to low)
                for frac in (0.01, 0.02, 0.05):
                    delta = frac * (1.0 / n)
                    p_new = p.copy()
                    p_new[i] = p_i_old + delta
                    s_new = float(np.sum(p_new))
                    if s_new <= 0:
                        continue
                    p_new /= s_new
                    new_b = fft_convolution(p_new)
                    new_M = float(np.max(new_b))
                    old_M = float(np.max(b))
                    if new_M < old_M - 1e-15:
                        p = p_new
                        improved = True
                        break

        if improved:
            no_improve = 0
            cur_score = fast_score(p)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_p = p.copy()
        else:
            no_improve += 1

    return best_p, best_score


def multi_restart_optimize(p0, deadline, rng, n_restarts=5, time_per_restart_frac=0.7):
    """
    Multi-restart: perturb the best solution and re-optimize with Adam.
    """
    best_p = normalize_simplex(p0)
    best_score = fast_score(best_p)
    print(f"  multi-restart start: score={best_score:.8f}")

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(remaining * time_per_restart_frac, remaining - 30)
        if alloc < 30:
            break
        sub_deadline = time.monotonic() + alloc

        # Perturb
        noise_scale = 0.05 / (1 + r * 0.5)
        p_perturbed = best_p + noise_scale * rng.standard_normal(best_p.size)
        p_perturbed = normalize_simplex(p_perturbed)

        # Also try a different perturbation: selectively boost low-contributing coords
        if r % 2 == 1:
            b = fft_convolution(best_p)
            M = float(np.max(b))
            # Find coordinates with low contribution
            w = np.ones_like(b) / float(b.size)
            svec = corr_pw(best_p, w)
            low_coords = np.where(svec < float(np.median(svec)))[0]
            if len(low_coords) > 0:
                idx = rng.choice(low_coords, size=min(len(low_coords), n_restarts * 10), replace=False)
                p_perturbed[idx] *= (1 + 0.2 * rng.standard_normal(len(idx)))
                p_perturbed = normalize_simplex(p_perturbed)

        p_fit, sc_fit = adam_projected_gradient(
            p_perturbed, sub_deadline, rng,
            beta=200000.0, lr=0.003, max_iter=5000000
        )
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  restart {r}: improved to {best_score:.8f}")
        else:
            print(f"  restart {r}: no improvement (got {sc_fit:.8f})")

    return best_p, best_score


def param_block_search(n, deadline, rng, max_B=512):
    """Optimize piecewise-constant log-weights at increasing resolution."""
    best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
    best_score = fast_score(best_p)

    if not HAVE_SCIPY or n < 8:
        return best_p, best_score

    candidates = []
    for B in (8, 16, 32, 64, 128, 256, 512):
        if B <= max_B and B < n and (n % B) == 0:
            candidates.append(B)

    for B in candidates:
        if time.monotonic() >= deadline:
            break

        bounds = [(-8.0, 8.0)] * (B - 1)
        best_c = np.zeros(B - 1, dtype=np.float64)
        best_B_score = float("inf")

        # Try several initializations
        inits = [np.zeros(B - 1, dtype=np.float64)]
        x = np.linspace(0.0, 1.0, B - 1, dtype=np.float64)
        for a in (-4.0, -2.0, 2.0, 4.0):
            inits.append(a * (x - 0.5))
        t = np.linspace(0.0, 2.0 * np.pi, B - 1, dtype=np.float64)
        inits.append(np.cos(t))
        inits.append(np.sin(t))
        inits.append(1.0 * rng.normal(0.0, 1.0, B - 1))
        inits.append(2.0 * rng.normal(0.0, 1.0, B - 1))

        # Beta annealing: start coarse, end sharp
        for beta in (5000.0, 20000.0, 100000.0):
            for c0 in inits:
                if time.monotonic() >= deadline:
                    break
                c = run_block_minimize(c0, n, B, beta, bounds, maxiter=60)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()

        # Final sharp pass
        if time.monotonic() < deadline and np.isfinite(best_B_score):
            c = run_block_minimize(best_c, n, B, 200000.0, bounds, maxiter=80)
            p = block_to_p(n, B, c)
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_B_score:
                best_B_score = sc
                best_c = c.copy()
            p = block_to_p(n, B, best_c)
        else:
            p = block_to_p(n, B, best_c)

        if np.isfinite(best_B_score) and best_B_score < best_score:
            best_score = best_B_score
            best_p = p.copy()
            print(f"  block B={B} score={best_score:.8f}")

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function. Returns a list of non-negative floats.
    Uses multi-phase optimization with Adam, Nesterov, and coordinate descent.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Choose sequence length
    if budget_s < 300:
        N = 4096
    elif budget_s < 1200:
        N = 8192
    else:
        N = 16384

    print(f"N={N}, budget={budget_s}s")

    # Initialize from previous best if available
    try:
        prev = height_sequence_1
        if isinstance(prev, list) and len(prev) > 0:
            prev_arr = np.array([float(x) for x in prev], dtype=np.float64)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = float(np.sum(prev_arr))
            if s > 0 and np.isfinite(s):
                prev_arr /= s
                # Resize to N if needed
                if len(prev_arr) == N:
                    best_p = prev_arr.copy()
                else:
                    # Interpolate or truncate
                    if len(prev_arr) > N:
                        best_p = prev_arr[:N].copy()
                        best_p = normalize_simplex(best_p)
                    else:
                        # Upsample
                        new_p = np.interp(
                            np.linspace(0, 1, N),
                            np.linspace(0, 1, len(prev_arr)),
                            prev_arr
                        )
                        best_p = normalize_simplex(new_p)
                best_score = fast_score(best_p)
                print(f"Loaded previous best: score={best_score:.8f}")
            else:
                best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
                best_score = fast_score(best_p)
        else:
            best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
            best_score = fast_score(best_p)
    except (NameError, Exception):
        best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
        best_score = fast_score(best_p)
        print("No previous best available, starting from uniform")

    print(f"Initial score: {best_score:.8f}")

    # Phase 1: Block optimization
    phase1_deadline = time.monotonic() + min(300.0, max(60.0, 0.25 * budget_s))
    if phase1_deadline > time.monotonic() + 30:
        print("Phase 1: Block optimization")
        p_param, sc_param = param_block_search(N, phase1_deadline, rng, max_B=min(512, N // 4))
        if np.isfinite(sc_param) and sc_param < best_score:
            best_score = sc_param
            best_p = p_param.copy()
            print(f"After Phase 1: score={best_score:.8f}")

    # Phase 2: Adam projected gradient
    phase2_deadline = time.monotonic() + min(600.0, max(120.0, 0.35 * budget_s))
    if phase2_deadline > time.monotonic() + 30:
        print("Phase 2: Adam projected gradient")
        p_adam, sc_adam = adam_projected_gradient(
            best_p, phase2_deadline, rng,
            beta=200000.0, lr=0.005, max_iter=10000000
        )
        if np.isfinite(sc_adam) and sc_adam < best_score:
            best_score = sc_adam
            best_p = p_adam.copy()
            print(f"After Phase 2: score={best_score:.8f}")

    # Phase 3: Nesterov momentum
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("Phase 3: Nesterov momentum")
        nesterov_deadline = time.monotonic() + min(remaining * 0.4, 400.0)
        p_nest, sc_nest = projected_gradient_nesterov(
            best_p, nesterov_deadline, beta=300000.0, max_iter=10000000
        )
        if np.isfinite(sc_nest) and sc_nest < best_score:
            best_score = sc_nest
            best_p = p_nest.copy()
            print(f"After Phase 3: score={best_score:.8f}")

    # Phase 4: Multi-restart
    remaining = deadline - time.monotonic()
    if remaining > 180:
        print(f"Phase 4: Multi-restart (remaining={remaining:.0f}s)")
        n_restarts = min(5, max(2, int(remaining // 200)))
        p_ms, sc_ms = multi_restart_optimize(
            best_p, time.monotonic() + max(0, remaining - 60),
            rng, n_restarts=n_restarts, time_per_restart_frac=0.7
        )
        if np.isfinite(sc_ms) and sc_ms < best_score:
            best_score = sc_ms
            best_p = p_ms.copy()
            print(f"After Phase 4: score={best_score:.8f}")

    # Phase 5: Final Adam polish
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("Phase 5: Final Adam polish")
        p_polish, sc_polish = adam_projected_gradient(
            best_p, time.monotonic() + max(0, remaining - 30), rng,
            beta=500000.0, lr=0.002, max_iter=5000000
        )
        if np.isfinite(sc_polish) and sc_polish < best_score:
            best_score = sc_polish
            best_p = p_polish.copy()
            print(f"After Phase 5: score={best_score:.8f}")

    # Final cleanup
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0.0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s

    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        # Fallback to uniform
        print(f"Official score invalid ({off}), using uniform fallback")
        return [1.0] * N

    print(f"Final official score: {off:.8f}")
    return final_seq
