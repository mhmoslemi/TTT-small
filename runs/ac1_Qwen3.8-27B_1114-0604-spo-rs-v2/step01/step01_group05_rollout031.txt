def fft_conv(a):
    """Compute linear convolution a*a via FFT, returning non-negative result."""
    n = a.size

        return np.empty(0)
    if n == 1:
        return np.array([a[0] * a[0]])
    need = 2 * n - 1
    size = 1

        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def corr_pw(p, w):
    """Compute s_i = sum_j p_j * w[i+j] for i=0..len(p)-1."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # C[k] = sum_j p_j * w_flipped[k-j] = sum_j p_j * w[m-1-(k-j)] = sum_j p_j * w[m-1-k+j]
    # We want s_i = sum_j p_j * w[i+j]
    # C[m-1-i] = sum_j p_j * w[m-1-(m-1-i)+j] = sum_j p_j * w[i+j]
    # So s_i = C[m-1-i] for i=0..n-1
    s = C[m-n:m][::-1].copy()
    return s

    # Background:

```python
"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Algorithm:
1. Generate diverse starting points (edge-boosted, Fourier-based, structured).
2. Nesterov-accelerated projected gradient on the simplex with adaptive smooth-max.
3. Coordinate descent polish with O(n) per-evaluation via direct convolution updates.
4. Try multiple sequence lengths, return the best.
"""

import numpy as np
import time


def fft_conv(a):
    """Linear self-convolution via FFT, non-negative clipped."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Return (smooth_max_value, weights) for smooth max of b."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def corr_pw(p, w):
    """s_i = sum_j p_j * w[i+j] for i=0..len(p)-1."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    nv = v.size
    if nv == 0:
        return v
    if target <= 0.0:
        return np.zeros(nv, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(nv, target / nv, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(nv, target / nv, dtype=np.float64)
    q *= target / s
    return q


def normalize_simplex(p):
    """Clip nonneg and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    p *= 1.0 / s
    return p


def fast_score(a):
    """Compute 2*n*max(a*a) for normalized a (sum=1)."""
    n = a.size
    if n == 0:
        return float('inf')
    b = fft_conv(a)
    return 2.0 * n * float(np.max(b))


def gradient_smooth_max(a, w, n):
    """
    Gradient of smooth_max(a*a, beta) w.r.t. a.
    Returns vector of length n.
    """
    s_vec = corr_pw(a, w)
    return 2.0 * s_vec


def nesterov_projected_grad(p0, deadline, beta_start=3000.0, beta_end=300000.0,
                            momentum_start=0.85, momentum_end=0.98,
                            max_iter=200000):
    """
    Nesterov-accelerated projected gradient on the simplex.
    Minimizes n * smooth_max(a*a, beta) with adaptive beta.
    """
    p = np.asarray(p0, dtype=np.float64).copy()
    p = normalize_simplex(p)
    n = p.size
    if n == 0:
        return p, float('inf')

    best_p = p.copy()
    best_score = fast_score(p)
    v = np.zeros(n, dtype=np.float64)
    prev_p = p.copy()
    lr = 5.0
    no_improve = 0
    max_no_improve = 5000

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        frac = min(1.0, it / max(1, int(max_iter * 0.7)))
        beta = beta_start * (beta_end / beta_start) ** frac
        mu = momentum_start + (momentum_end - momentum_start) * frac

        # Nesterov lookahead
        y = p + mu * (p - prev_p)
        y = project_simplex(y, 1.0)
        prev_p = p.copy()

        # Gradient at y
        b = fft_conv(y)
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break
        grad_a = gradient_smooth_max(y, w, n)
        # Project gradient (remove mean component for simplex constraint)
        grad = grad_a - float(np.mean(grad_a))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Backtracking line search (using smooth objective)
        cur_smooth = n * G
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = lr * mult
            p_new = project_simplex(y - alpha * grad, 1.0)
            b_new = fft_conv(p_new)
            G_new, _ = smooth_max_weights(b_new, beta)
            if np.isfinite(G_new) and n * G_new < cur_smooth + 1e-15:
                p = p_new
                accepted = True
                break

        if not accepted:
            lr *= 0.5
            if lr < 1e-8:
                break
        else:
            if lr < 20.0:
                lr = min(20.0, lr * 1.05)

        # Track best using exact score
        sc = fast_score(p)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

        if it % 500 == 0:
            print(f"  Nesterov it={it} score={best_score:.8f} lr={lr:.2e} beta={beta:.0f}")

    return best_p, best_score


def coord_descent_sweep(a, b, S, n, max_passes=1, deadline=None):
    """
    One or more passes of coordinate descent.
    a: normalized sequence (sum=1)
    b: convolution a*a
    S: sum of a (=1)
    """
    a = a.copy()
    b = b.copy()
    S = float(S)
    n = len(a)
    rng = np.random.default_rng(12345)

    for sweep in range(max_passes):
        if deadline is not None and time.monotonic() >= deadline:
            break

        # Randomize order for better convergence
        order = rng.permutation(n)
        improvements = 0

        for i in order:
            if deadline is not None and (i % 1024 == 0) and time.monotonic() >= deadline:
                return a, b

            a_i = a[i]
            if a_i < 1e-15:
                # Try setting to small positive values
                candidates = [a_i, 0.5 * S / n, S / n]
            else:
                # Geometric and linear perturbations around current value
                candidates = [
                    0.0,
                    a_i * 0.5,
                    a_i * 0.75,
                    a_i,
                    a_i * 1.25,
                    a_i * 1.5,
                    a_i * 2.0,
                    0.5 * S / n + 0.5 * a_i,
                ]
                # Add a few more based on the peak structure
                k_peak = int(np.argmax(b))
                if i <= k_peak <= i + n - 1 and k_peak != 2 * i:
                    # The peak involves a[k_peak - i]
                    a_related = a[k_peak - i]
                    candidates.append(a_related)
                    candidates.append(0.5 * (a_i + a_related))
            candidates = [max(0.0, c) for c in candidates]

            best_t = a_i
            best_sc = fast_score(a)  # current score

            for t in candidates:
                if abs(t - a_i) < 1e-15:
                    continue
                # Compute new convolution values (only changed positions)
                k_min = max(0, i)
                k_max = min(i + n - 1, 2 * n - 2)

                # Compute new max efficiently
                # The changed positions are k in [k_min, k_max]
                # For k=2i: delta = t^2 - a_i^2
                # For k != 2i: delta = 2*(t-a_i)*a[k-i]

                # Find current max excluding changed region for comparison
                if k_min > 0 and k_max < 2 * n - 2:
                    bg_max = max(float(np.max(b[:k_min])) if k_min > 0 else 0.0,
                                 float(np.max(b[k_max+1:])) if k_max + 1 < 2 * n - 1 else 0.0)
                elif k_min == 0:
                    bg_max = float(np.max(b[k_max+1:])) if k_max + 1 < 2 * n - 1 else 0.0
                else:
                    bg_max = float(np.max(b[:k_min])) if k_min > 0 else 0.0

                # Evaluate changed region
                ks = np.arange(k_min, k_max + 1)
                a_ki = a[ks - i]
                delta = 2.0 * (t - a_i) * a_ki
                # Fix k=2i
                idx_2i = 2 * i - k_min
                if 0 <= idx_2i < len(ks):
                    delta[idx_2i] = t * t - a_i * a_i
                new_region = b[ks] + delta
                region_max = float(np.max(new_region)) if new_region.size > 0 else 0.0

                new_max = max(bg_max, region_max, 0.0)
                new_S = S - a_i + t
                if new_S < 1e-30:
                    sc = 2.0 * n * new_max
                else:
                    sc = 2.0 * n * new_max / (new_S * new_S)

                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_t = t

            if best_t != a_i:
                # Update a and b
                k_min = max(0, i)
                k_max = min(i + n - 1, 2 * n - 2)
                ks = np.arange(k_min, k_max + 1)
                a_ki = a[ks - i]
                delta = 2.0 * (best_t - a_i) * a_ki
                idx_2i = 2 * i - k_min
                if 0 <= idx_2i < len(ks):
                    delta[idx_2i] = best_t * best_t - a_i * a_i
                b[ks] = np.maximum(b[ks] + delta, 0.0)
                a[i] = best_t
                S = S - a_i + best_t
                improvements += 1

        # Renormalize after each sweep
        if S > 0:
            a = a / S
            b = b / (S * S)
            S = 1.0

        if improvements == 0:
            break

    return a, b


def generate_starts(n, rng, num_starts=10):
    """Generate diverse starting points."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Edge-boosted: larger at endpoints
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        p = 1.0 + alpha * (2.0 * x - 1.0) ** 2
        p = np.maximum(p, 1e-8)
        p /= p.sum()
        starts.append(p.copy())

    # Sine-based
    for k in (1, 2, 3):
        p = 1.0 + 0.3 * np.sin(2.0 * np.pi * k * x)
        p = np.maximum(p, 1e-8)
        p /= p.sum()
        starts.append(p.copy())

    # Cosine taper (Hann window-like)
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    p = np.maximum(p, 1e-8)
    p /= p.sum()
    starts.append(p.copy())

    # Fourier random
    for _ in range(num_starts - len(starts)):
        size = 1
        while size < n:
            size <<= 1
        zc = rng.normal(size=size // 2 + 1) + 1j * rng.normal(size=size // 2 + 1)
        z = np.real(np.fft.irfft(zc, size))[:n]
        z = z - z.mean()
        z = 1.0 * np.tanh(z)
        p = np.exp(0.5 * z)
        p /= p.sum()
        starts.append(p.copy())

    return starts[:max(4, num_starts)]


def try_length(N, budget_s, seed, rng):
    """Run the full optimization pipeline for a given length N."""
    start_time = time.monotonic()
    deadline = start_time + budget_s

    # Generate starts
    starts = generate_starts(N, rng, num_starts=12)
    print(f"  Generated {len(starts)} starts for N={N}")

    # Score all starts, keep top 3
    scored = []
    for p in starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    top_starts = [p for _, p in scored[:3]]

    best_p = top_starts[0]
    best_score = scored[0][0]
    print(f"  Best start score: {best_score:.8f}")

    # Nesterov optimization on top starts
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        n_starts = min(3, len(top_starts))
        time_per_start = remaining * 0.6 / n_starts

        for i in range(n_starts):
            if time.monotonic() >= deadline - 60.0:
                break
            sub_deadline = time.monotonic() + time_per_start
            p_opt, sc_opt = nesterov_projected_grad(
                top_starts[i], sub_deadline,
                beta_start=3000.0, beta_end=400000.0,
                momentum_start=0.80, momentum_end=0.97
            )
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_p = p_opt
                print(f"  Nesterov start {i}: score={best_score:.8f}")

    # Additional Nesterov from best (longer run)
    remaining = deadline - time.monotonic()
    if remaining > 120.0:
        sub_deadline = time.monotonic() + remaining * 0.4
        p_opt, sc_opt = nesterov_projected_grad(
            best_p, sub_deadline,
            beta_start=10000.0, beta_end=500000.0,
            momentum_start=0.90, momentum_end=0.99
        )
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt
            print(f"  Extended Nesterov: score={best_score:.8f}")

    # Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        a = normalize_simplex(best_p)
        b = fft_conv(a)
        S = float(np.sum(a))
        n = a.size

        # Multiple CD passes with time budget
        cd_deadline = time.monotonic() + remaining * 0.85
        sweep_budget = max(1.0, remaining * 0.85 / 40.0)  # ~40 sweeps max

        for sweep_num in range(200):
            if time.monotonic() >= cd_deadline:
                break
            sweep_start = time.monotonic()
            a, b = coord_descent_sweep(a, b, S, n, max_passes=1, deadline=cd_deadline)
            S = float(np.sum(a))
            if S > 0:
                a = a / S
                b = fft_conv(a)
                S = 1.0
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_p = a.copy()
            sweep_time = time.monotonic() - sweep_start
            if sweep_num % 10 == 0:
                print(f"  CD sweep {sweep_num}: score={best_score:.8f} time={sweep_time:.1f}s")
            if sweep_time < 0.5:
                # Converged quickly, do another
                continue
            if sweep_time > sweep_budget * 3:
                # Too slow, break
                break

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best nonnegative sequence minimizing 2n*max(a*a)/sum(a)^2.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 10.0:
        return [1.0] * 256

    total_start = time.monotonic()
    reserve = 60.0
    total_deadline = total_start + max(10.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Try to use the previous best if available
    try:
        prev = height_sequence_1
        if isinstance(prev, list) and len(prev) > 0:
            prev_arr = np.array([float(x) for x in prev], dtype=np.float64)
            prev_arr = np.maximum(prev_arr, 0.0)
            s = float(np.sum(prev_arr))
            if s > 0:
                prev_arr /= s
                prev_score = fast_score(prev_arr)
                print(f"Previous best available: N={len(prev_arr)}, score={prev_score:.8f}")
            else:
                prev_arr = None
        else:
            prev_arr = None
    except Exception:
        prev_arr = None

    # Strategy: try N=8192 mainly, maybe N=4096 as backup
    best_p = None
    best_score = float('inf')

    # Time allocation
    elapsed = time.monotonic() - total_start
    remaining_total = total_deadline - time.monotonic()

    # Primary: N=8192
    N1 = 8192
    time_alloc_1 = min(remaining_total * 0.7, budget_s - reserve - 60.0)
    if time_alloc_1 > 60.0:
        print(f"\n=== Phase 1: N={N1}, budget={time_alloc_1:.0f}s ===")
        rng1 = np.random.default_rng(seed + 1)

        # If we have the previous best at this length, use it as a start
        if prev_arr is not None and len(prev_arr) == N1:

            extra_starts = generate_starts(N1, rng1, num_starts=8)

            for p in starts_N1:
                sc = fast_score(p)
                if np.isfinite(sc):
                    scored.append((sc, p))
            scored.sort(key=lambda x: x[0])

            # Run Nesterov from the best starts
            top_k = min(3, len(scored))
            p_best_local = scored[0][1]
            sc_best_local = scored[0][0]

                    break

                    sc_best_local = sc_opt
                    p_best_local = p_opt
                    print(f"  N={N1} Nesterov {i}: {sc_opt:.8f}")

            # Extended Nesterov
            if time.monotonic() < time.monotonic() + time_alloc_1 * 0.3:
                ext_deadline = time.monotonic() + time_alloc_1 * 0.25
