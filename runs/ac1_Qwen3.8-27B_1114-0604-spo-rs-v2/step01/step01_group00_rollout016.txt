def _compute_conv(f):
    """Compute linear autocorrelation (convolution with self)."""
    return fftconvolve(f, f)


def _compute_score(f):
    """Compute the objective value."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    conv = _compute_conv(f)
    return 2.0 * n * conv.max() / (s * s)


def _soft_gradient_multipeak(conv, f, top_k=None):
    """

    Returns (gradient, peak_values).

    n = len(f)

        # Use all peaks with soft weighting
        T = conv.max() * 0.1
        if T < 1e-15:
            T = conv.max()
        cs = conv / T
        cs -= cs.max()
        p = np.exp(cs)
        p /= p.sum()

        idx = np.argsort(conv)[::-1][:top_k]
