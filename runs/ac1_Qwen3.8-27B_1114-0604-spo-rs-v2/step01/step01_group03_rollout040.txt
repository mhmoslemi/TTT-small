```
def _random_prior(n,rng):
   t=np.linspace(0,1,n)
   i=np.arange(n,dtype=np.float64)
   z=np.zeros(n)
   z += rng.normal(0,0.2,n)
   kmax=int(rng.integers(2,8))
   for _ in range(kmax):
       freq=float(rng.uniform(0.5,8.0))
       amp=float(rng.uniform(0.2,1.2))
       ph=float(rng.uniform(0,2*np.pi))
       z += amp*np.cos(2*np.pi*freq*t+ph)
   # edge
   z += float(rng.uniform(-1,1))*(t-0.5)
   # center
   z += float(rng.uniform(-1,1))*(1-(2*t-1)**2)
   # parity
   amp=float(rng.uniform(0.1,0.8))
   z += amp*np.cos(np.pi*i + float(rng.uniform(0,2*np.pi)))
   # high freq
   freq=float(rng.uniform(10.0, max(20.0,0.3*n)))
   amp=float(rng.uniform(0.05,0.4))
   z += amp*np.cos(2*np.pi*freq*i/n + float(rng.uniform(0,2*np.pi)))
   # smooth noise
   noise=np.zeros(n)
   if n>10:
       k=int(rng.integers(3, min(15,n-1)+1))
       noise=rng.normal(0,0.3,n)
       kernel=np.ones(k)/float(k)
       noise=np.convolve(noise,kernel,mode='same')
   z += noise
   z = z - float(np.max(z))
   p=np.exp(z)
   p=np.maximum(p,0)
   p = _normalize(p)
   return p
```

```python
"""
Search for non-negative step heights minimizing
    2*n*max(conv(a, a)) / (sum(a)**2).

Strategy:
- Keep the sequence length from the previous construction if available.
- Use FFT-based exact linear convolution for fast objective evaluation.
- Interpret the objective as minimizing the largest coefficient of P(z)^2
  where P(z)=sum a_i z^i.
- Main local search:
    * Solve (approximately) the linear subproblem for a fixed current a:
        min_{g>=0, sum(g)=1} max_k (a*g)_k
      using a multiplicative-weights matrix game.  The averaged primal
      distribution gives one or several directions g.
    * Projected line search along mixtures (1-lambda)a + lambda g.
- Auxiliary polishes:
    * Soft log-sum-exp subgradient updates on the simplex.
    * Cheap one-dimensional transformations (shifts, edge/center boosts).
    * Random Fourier-style restarts to escape local minima.
- Final exact evaluation with the supplied evaluate_sequence and return the
  best sequence found.
"""

import math
import time
import numpy as np


def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= eps:
        a.fill(1.0)
        s = float(a.size)
    return a / s


def _score_c(c, n):
    return float(2.0 * float(n) * float(np.max(c)))


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_ab(a, b, L, m):
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _atrev(w, Frev, L, m, n):
    """
    Returns s_i = sum_k w_k * f_{k-i} for i=0..n-1, where f is the original
    sequence whose reversed FFT is Frev.
    """
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L)[m - n:m]
    np.maximum(g, 0.0, out=g)
    return g


def _score_a(a, L, m, n):
    return _score_c(_conv_self(a, L, m), n)


def _softmax_raw(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        return np.full(v.shape, 1.0 / float(v.size), dtype=np.float64)
    z = v - vmax
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(v.shape, 1.0 / float(v.size), dtype=np.float64)
    return e / s


def _softmax_j(v, temp):
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.full(e.shape, 1.0 / float(e.size), dtype=np.float64)
        s = float(e.size)
    else:
        e = e / s
    J = float(temp * vmax + math.log(s))
    return e, J, vmax


def _smooth_vector(v, w=7):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size <= int(w):
        return _normalize(v)
    k = int(w)
    ker = np.ones(k, dtype=np.float64) / float(k)
    y = np.convolve(v, ker, mode="same")
    return _normalize(y)


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    z = np.zeros(n, dtype=np.float64)

    z += float(rng.normal(0.0, 0.25))

    kmax = int(rng.integers(2, 8))
    for _ in range(kmax):
        freq = float(rng.uniform(0.5, 8.0))
        amp = float(rng.uniform(0.2, 1.2))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * freq * t + ph)

    z += float(rng.uniform(-1.0, 1.0)) * (t - 0.5)
    z += float(rng.uniform(-1.0, 1.0)) * (1.0 - (2.0 * t - 1.0) ** 2)

    amp = float(rng.uniform(0.10, 0.80))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    z += amp * np.cos(np.pi * i + ph)

    freq = float(rng.uniform(10.0, max(20.0, 0.3 * n)))
    amp = float(rng.uniform(0.05, 0.40))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    z += amp * np.cos(2.0 * np.pi * freq * i / float(n) + ph)

    if n > 10:
        k = int(rng.integers(3, min(15, n - 1) + 1))
        noise = rng.normal(0.0, 0.30, n)
        ker = np.ones(k, dtype=np.float64) / float(k)
        z += np.convolve(noise, ker, mode="same")

    p = np.exp(z - float(np.max(z)))
    return _normalize(p)


def _initial_candidates(n, rng, parent_arr):
    out = []

    if parent_arr is not None:
        out.append(parent_arr.copy())
        out.append(_smooth_vector(parent_arr, 5))

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)

    for amp in (0.8, 1.5, 2.5):
        for sgn in (-1.0, 1.0):
            z = sgn * amp * (t - 0.5)
            out.append(_normalize(np.exp(z - float(np.max(z)))))

    for amp in (0.8, 1.5):
        for c in (0.2, 0.5, 0.8):
            z = amp * (1.0 - (t - c) ** 2)
            out.append(_normalize(np.exp(z - float(np.max(z)))))

    for amp in (0.3, 0.8, 1.5):
        z = amp * np.cos(np.pi * i)
        out.append(_normalize(np.exp(z - float(np.max(z)))))

    for k in (1, 2, 3, 5, 10, 20):
        for ph in (0.0, 0.25 * np.pi, 0.5 * np.pi):
            for amp in (0.5, 1.0):
                z = amp * np.cos(2.0 * np.pi * k * t + ph)
                out.append(_normalize(np.exp(z - float(np.max(z)))))

    for _ in range(30):
        out.append(_random_prior(n, rng))

    return out


def _add_candidate(arrays, scores, arr, score, maxlen=80):
    if arr is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not np.isfinite(score):
        return
    arr = np.asarray(arr, dtype=np.float64).copy()
    if len(scores) < maxlen:
        arrays.append(arr)
        scores.append(score)
    else:
        worst = max(scores)
        if score < worst:
            idx = scores.index(worst)
            arrays[idx] = arr
            scores[idx] = score


def _add_direction(dirs, arr, max_len=5):
    if arr is None:
        return
    arr = _normalize(arr)
    for d in dirs:
        try:
            if float(np.dot(arr, d)) > 0.9995:
                return
        except Exception:
            pass
    dirs.append(arr)
    if len(dirs) > max_len:
        del dirs[1]


def _local_transforms(a, L, m, n):
    p = _normalize(a)
    best = p.copy()
    best_score = _score_a(p, L, m, n)

    def _try(q):
        nonlocal_token = None  # no closure; this local helper is only for readability? We'll avoid nesting by not using it.
        return None

    # To satisfy the no-nested-function rule, do all loops inline below.
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    for fr in shift_fracs:

            for sign in (-1, 1):
                q = np.roll(p, sign * s)
                sc = _score_a(q, L, m, n)
                if np.isfinite(sc) and sc < best_score:

                    best = q.copy()

    for alpha in (-0.4, -0.2, -0.1, 0.1, 0.2, 0.4):
        q = p * np.exp(alpha * (t - 0.5))
        q = _normalize(q)
        sc = _score_a(q, L, m, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = q.copy()

    for beta in (-0.5, -0.2, 0.2, 0.5):
        q = p * (1.0 + beta * (1.0 - np.abs(2.0 * t - 1.0)))
        q = _normalize(np.maximum(q, 0.0))
        sc = _score_a(q, L, m, n)
        if np.isfinite(sc) and sc < best_score:

def _line_search_mix(a, b, L, m, n, base_score):
    a = _normalize(a)
    b = _normalize(b)
    if not np.isfinite(base_score):
        base_score = _score_a(a, L, m, n)

    try:
        if float(np.dot(a, b)) > 1.0 - 1e-8:
            return a.copy(), base_score
    except Exception:
        pass

    best_a = a.copy()
    best_score = float(base_score)
    best_lam = 0.0
    best_dir = b.copy()

    dirs = [b.copy()]
    bs = _smooth_vector(b, 7)
    try:
        if float(np.dot(b, bs)) < 0.9999:
            dirs.append(bs)
    except Exception:
        pass

    lams = (0.01, 0.02, 0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 1.00)

    for d in dirs:
        for lam in lams:
            h = (1.0 - lam) * a + lam * d
            h = _normalize(h)
            sc = _score_a(h, L, m, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = h
                best_lam = lam
                best_dir = d.copy()

    if best_score < base_score - 1e-12 and 0.005 < best_lam < 0.995:
        lo = max(0.0, best_lam - 0.12)
        hi = min(1.0, best_lam + 0.12)
        for lam in np.linspace(lo, hi, 11):
            h = (1.0 - float(lam)) * a + float(lam) * best_dir
            h = _normalize(h)
            sc = _score_a(h, L, m, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = h
                best_lam = float(lam)

    return best_a, best_score


def _mws_directions(f, L, m, n, prior, inner, eta_start, eta_end, deadline=None):
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))

    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros(n, dtype=np.float64)
    count = 0

    logprior = np.log(np.maximum(prior, 1e-300))
    logprior -= float(np.max(logprior))

    s = np.zeros(n, dtype=np.float64)
    ps = np.zeros(n, dtype=np.float64)
    start_avg = max(0, inner // 3)

    for it in range(inner):
        if deadline is not None and time.monotonic() > deadline:
            break

        if inner <= 1:
            frac = 1.0
        else:
            frac = float(it) / float(inner - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -500.0, 0.0)
        e = np.exp(z)
        se = float(e.sum())
        if np.isfinite(se) and se > 0.0:
            y = e / se
        else:
            y = np.full(m, 1.0 / float(m), dtype=np.float64)

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[m - n:m]
        np.maximum(s, 0.0, out=s)
        ps = n * s
        sma = float(np.max(ps))

        zx = -eta * (ps - sma) + 0.7 * logprior
        zx -= float(np.max(zx))
        zx = np.clip(zx, -500.0, 0.0)
        ex = np.exp(zx)
        sx = float(ex.sum())
        if np.isfinite(sx) and sx > 0.0:
            x = ex / sx
        else:
            x = prior.copy()

        if it >= start_avg:
            avg += x
            count += 1

    dirs = []
    if count > 0:
        _add_direction(dirs, avg / float(count), 6)
    _add_direction(dirs, x, 6)

    # Sparse directions from the smallest-primal coordinates for the last dual.
    if count > 0:
        ps = np.asarray(ps, dtype=np.float64)
        ksmall = max(1, min(n, int(0.02 * n) + 1))
        idx = np.argsort(ps)[:ksmall]
        d = np.zeros(n, dtype=np.float64)
        d[idx] = 1.0 / float(ksmall)
        _add_direction(dirs, d, 6)

        smin = float(np.min(ps))
        smax = float(np.max(ps))
        thr = smin + max(0.05 * (smax - smin), 1e-8)
        active = (ps <= thr).astype(np.int8)
        cnt = int(active.sum())
        if 1 <= cnt <= max(2, int(0.5 * n)):
            d2 = np.zeros(n, dtype=np.float64)
            d2[active.astype(bool)] = 1.0 / float(cnt)
            _add_direction(dirs, d2, 6)

    if not dirs:
        dirs.append(_normalize(prior))

    return dirs


def _mws_search(a, L, m, n, rng, deadline):
    p = _normalize(a)
    best = p.copy()
    best_score = _score_a(p, L, m, n)
    it = 0
    stale = 0

    while time.monotonic() < deadline - 3.0:
        now = time.monotonic()
        rem = deadline - now
        if rem < 15.0:
            break

        use_random = (stale >= 2) or (it % 12 == 11)
        if use_random:
            q = _random_prior(n, rng)
            mix = 0.10 + 0.40 * float(rng.random())
            prior = _normalize((1.0 - mix) * p + mix * q)
        else:
            prior = p.copy()

        if rem > 180.0:
            inner = 250
        elif rem > 70.0:
            inner = 150
        else:
            inner = 60

        dirs = _mws_directions(p, L, m, n, prior, inner, 5.0, 150.0,
                               deadline=max(now, time.monotonic() + 0.1))

        improved = False
        for d in dirs:
            cand, sc = _line_search_mix(p, d, L, m, n, best_score)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best = cand.copy()
                p = cand.copy()
                improved = True
                break

        if not improved:
            stale += 1
            if stale > 6:
                q = _random_prior(n, rng)
                cand, sc = _line_search_mix(p, q, L, m, n, best_score)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best = cand.copy()
                    p = cand.copy()
                    stale = 0

        if improved:
            stale = 0

        it += 1

    return best, best_score

            v2 = n * c2
            vz2 = float(np.max(v2))
            z2 = temp_eff * (v2 - vz2)
            z2 = np.clip(z2, -700.0, 0.0)
            e2 = np.exp(z2)
            s2 = float(e2.sum())
            if np.isfinite(s2) and s2 > 0.0:
                J2 = float(temp_eff * vz2 + math.log(s2))
                if J2 < J - 1e-8:
                    p = a2
                    accepted = True
                    chosen_step = step
                    break

        if accepted:
            no_accept = 0
            if chosen_step == eta:
                eta = min(0.50, eta * 1.08)
            else:
                eta = max(0.02, eta * 0.90)
        else:
            no_accept += 1
            eta = max(0.02, eta * 0.80)

        it += 1
        if it % 70 == 0:
            temp = min(temp1, temp * 1.25)

    return best, best_score


def _cook(a, L, m, n, rng, deadline):
    p = _normalize(a)
    best = p.copy()
    best_score = _score_a(p, L, m, n)

    meth = 0
    stale = 0

    while time.monotonic() < deadline - 8.0:
        now = time.monotonic()
        rem = deadline - now
        if rem < 30.0:
            break

        slice_mws = min(90.0, max(10.0, 0.04 * rem))
        slice_soft = min(55.0, max(10.0, 0.025 * rem))

        if meth % 8 == 0:
            tmp, sc = _mws_search(p, L, m, n, rng, now + slice_mws)
        elif meth % 8 == 1:
            tmp, sc = _mws_search(p, L, m, n, rng, now + slice_mws * 0.75)
        elif meth % 8 == 2:
            tmp, sc = _direct_soft_search(p, L, m, n, rng, now + slice_soft, 5.0, 320.0)
        elif meth % 8 == 3:
            tmp, sc = _local_transforms(p, L, m, n)
        elif meth % 8 == 4:
            tmp, sc = _mws_search(p, L, m, n, rng, now + slice_mws)
        elif meth % 8 == 5:
            tmp, sc = _direct_soft_search(p, L, m, n, rng, now + slice_soft * 0.75, 20.0, 800.0)
        elif meth % 8 == 6:
            q = _random_prior(n, rng)
            tmp, sc = _line_search_mix(p, q, L, m, n, best_score)
        else:
            if stale > 3:
                q = _random_prior(n, rng)
                mix = 0.20 + 0.40 * float(rng.random())
                probe = _normalize((1.0 - mix) * p + mix * q)
                tmp, sc = _mws_search(probe, L, m, n, rng, now + slice_mws)
            else:
                tmp = p.copy()
                sc = best_score

        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best = np.asarray(tmp, dtype=np.float64).copy()
            p = best.copy()
            stale = 0
        else:
            stale += 1
            p = best.copy()

        if meth % 8 == 0:
            print(f"  cook method={meth} score={best_score:.10f} time_left={rem:.1f}s", flush=True)

        meth += 1

    return best, best_score


def _internal_exact(seq):
    a = np.asarray(seq, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    return float(2.0 * float(n) * float(np.max(b)) / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = float("inf")
    if val is None:
        return float("inf")
    try:
        return float(val)
    except Exception:
        return float("inf")


def _sort_indices(scores, limit=None):
    idxs = list(range(len(scores)))
    idxs.sort(key=scores.__getitem__)
    if limit is not None and len(idxs) > limit:
        return idxs[:limit]
    return idxs


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return a non-negative list of heights minimizing the provided score."""
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60.0:
        budget = 2400.0

    hard_deadline = start + max(5.0, budget - 10.0)
    work_deadline = hard_deadline - 30.0 if budget > 120.0 else hard_deadline - 5.0

    rng = np.random.default_rng(seed)

    n = 6121
    parent_arr = None
    try:
        parent = height_sequence_1
    except NameError:
        parent = None

    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64).ravel()
            if arr.size >= 100 and np.all(np.isfinite(arr)) and arr.size <= 9000:
                n = int(arr.size)
                parent_arr = _normalize(arr)
        except Exception:
            parent_arr = None

    if n < 100 or n > 9000:
        n = 6121
        parent_arr = None

    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()

    if parent_arr is not None and parent_arr.size == n:
        parent_arr, _ = _local_transforms(parent_arr, L, m, n)

    inits = _initial_candidates(n, rng, parent_arr)

    arrays = []
    scores = []
    best_a = None
    best_score = float("inf")

    for a in inits:
        a = _normalize(a)
        sc = _score_a(a, L, m, n)
        if np.isfinite(sc):
            _add_candidate(arrays, scores, a, sc, 100)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _normalize(np.ones(n, dtype=np.float64))
        best_score = _score_a(best_a, L, m, n)

    _add_candidate(arrays, scores, best_a, best_score, 100)

    print(f"n={n} L={L} initial_best={best_score:.10f}", flush=True)

    if time.monotonic() < work_deadline:
        tmp, sc = _cook(best_a, L, m, n, rng, work_deadline)
        if np.isfinite(sc):
            _add_candidate(arrays, scores, tmp, sc, 100)
            if sc < best_score:
                best_score = sc
                best_a = tmp.copy()

        print(f"after_main score={best_score:.10f} time_left={hard_deadline - time.monotonic():.1f}s", flush=True)

        # If useful time remains, give a quick chance to the second-best stored
        # point.  This protects against local minima while preserving most of
        # the budget for the main trajectory.
        exact_reserve = 35.0
        if time.monotonic() < hard_deadline - exact_reserve - 60.0 and len(arrays) >= 2:
            order2 = _sort_indices(scores, 3)
            for idx in order2:
                if idx is None:
                    continue
                if scores[idx] >= best_score * 1.02:
                    continue
                if len(arrays) == 1:
                    break
                tmp2, sc2 = _cook(arrays[idx], L, m, n, rng,
                                  min(time.monotonic() + 90.0,
                                      hard_deadline - exact_reserve - 60.0))
                if np.isfinite(sc2):
                    _add_candidate(arrays, scores, tmp2, sc2, 100)
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = tmp2.copy()

    exact_limit = 16 if n <= 7000 else 6
    order = _sort_indices(scores, exact_limit)
    seen_ids = set()
    best_seq = None
    best_exact = float("inf")

    exact_deadline = hard_deadline - 1.0

    for idx in order:
        if time.monotonic() > exact_deadline:
            break
        arr = arrays[idx]
        if arr is None:
            continue
        oid = id(arr)
        if oid in seen_ids:
            continue
        seen_ids.add(oid)
        seq = [float(x) for x in arr]
        val = _exact_score(seq)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq
            print(f"  exact_candidate score={val:.10f}", flush=True)

    if best_seq is None:
        seq = [float(x) for x in _normalize(best_a)]
        val = _exact_score(seq)
        if np.isfinite(val):
            best_seq = seq
        else:
            best_seq = seq

    print(f"final_returned_exact={best_exact:.10f}", flush=True)
    return best_seq
```
