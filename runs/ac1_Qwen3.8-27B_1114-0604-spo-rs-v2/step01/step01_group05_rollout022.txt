    s = float(np.sum(p))
    if s <= 0:
        p = np.full(n, 1.0 / n)
    else:
        p /= s
    return p


def gen_init_chirp(n, rng):
    """Amplitude-modulated chirp pattern."""
    x = np.arange(n, dtype=float) / n
    freq = rng.uniform(2.0, 8.0)
    amp = 0.5 + 0.5 * np.cos(2.0 * np.pi * rng.uniform(0.5, 3.0) * x)
    p = np.abs(np.sin(2.0 * np.pi * freq * x + rng.uniform(0, 2*np.pi))) * amp
    p = np.maximum(p, 1e-10)
    p /= p.sum()
    return p


def generate_starts(n, rng):
    """Generate a diverse set of initial sequences."""
    starts = []
    starts.append(gen_init_uniform(n))
    starts.append(gen_init_sparse_random(n, rng, 0.3))
    starts.append(gen_init_sparse_random(n, rng, 0.5))
    starts.append(gen_init_sparse_random(n, rng, 0.7))
    starts.append(gen_init_periodic_gap(n, rng, 2))
    starts.append(gen_init_periodic_gap(n, rng, 3))
    starts.append(gen_init_periodic_gap(n, rng, 5))
    starts.append(gen_init_smoothed_binary(n, rng, 1.0))
    starts.append(gen_init_cluster(n, rng, 6))
    starts.append(gen_init_cluster(n, rng, 12))
    starts.append(gen_init_chirp(n, rng))
    # Try to use height_sequence_1 if available
    try:
        hs = np.array(height_sequence_1, dtype=float)
        if hs.size == n:
            starts.append(normalize_simplex(hs))
        elif hs.size > 0:
            # Resample to length n
            t = np.linspace(0, hs.size - 1, n)
            resampled = np.interp(t, np.arange(hs.size), hs)
            starts.append(normalize_simplex(resampled))
    except Exception:
        pass
    return starts


def nesterov_projected_gradient(p0, deadline, rng,
                                  beta_start=200.0, beta_end=200000.0,
                                  max_iter=500000,
                                  sparsify_interval=3000,
                                  sparsify_thresh=0.01):
    """
    Nesterov-accelerated projected gradient with adaptive beta.
    Returns (best_p, best_score).
    """
    n = p0.size
    if n == 0:
        return p0, float("inf")

    p = normalize_simplex(p0)
    v = p.copy()  # Nesterov momentum variable
    prev_p = p.copy()
    momentum = 0.9

    # Adaptive beta schedule
    total_iterations = max_iter
    it = 0
    best_p = p.copy()
    best_score = fast_score(p)
    step = 1.0
    no_improve = 0

    # Break symmetry for uniform start
    mean_p = float(np.mean(p))
    if float(np.max(np.abs(p - mean_p))) < 1e-13:
        t = 0.123 * np.arange(n, dtype=float)
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(t)))
        v = p.copy()
        prev_p = p.copy()
        best_score = fast_score(p)
        best_p = p.copy()

    while it < total_iterations and time.monotonic() < deadline:
        it += 1

        # Adaptive beta: ramp from beta_start to beta_end
        frac = min(1.0, it / (total_iterations * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute smooth max and gradient at v (Nesterov lookahead)
        b = fft_convolve(p)
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break

        s_p = float(np.sum(p))
        cur_score = 2.0 * n * float(np.max(b)) / (s_p * s_p)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 2000 == 0:
            print(f"  NPG it={it} beta={beta:.0f} best={best_score:.8f} step={step:.3e} noimp={no_improve}")

        if no_improve > 5000:
            # Try a perturbation to escape
            if rng.random() < 0.5:
                pert = rng.normal(0, 0.01 * float(np.mean(p)), n)
                p_try = normalize_simplex(p + pert)
                sc_try = fast_score(p_try)
                if sc_try < best_score:
                    p = p_try
                    v = p.copy()
                    prev_p = p.copy()
                    no_improve = 0
                else:
                    break
            else:
                break

        # Nesterov step: evaluate gradient at lookahead point
        v = p + momentum * (p - prev_p)
        v = np.maximum(v, 0.0)
        sv = float(np.sum(v))
        if sv > 0:
            v = v / sv
        else:
            v = p.copy()

        # Gradient at v (approximate)
        grad = grad_conv(v, w)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Line search
        accepted = False
        base = step
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01):
            alpha = base * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolve(q)
            G2, _ = smooth_max_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if n * G2 < n * G - 1e-12:
                # Accept
                cur_score_new = 2.0 * n * float(np.max(qb))
                prev_p = p.copy()
                p = q
                if cur_score_new < best_score - 1e-14:
                    best_score = cur_score_new
                    best_p = p.copy()
                    no_improve = 0
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-8:
                momentum = max(0.0, momentum - 0.05)
                if momentum < 0.1:
                    break
            # Decay momentum when stuck
            momentum = max(0.0, momentum * 0.95)
        else:
            if step < 1.5:
                step = min(2.0, step * 1.3)
            # Adapt momentum
            momentum = min(0.95, momentum * 1.01)

        # Periodic sparsification
        if sparsify_interval > 0 and it % sparsify_interval == 0:
            thresh = sparsify_thresh * float(np.max(p))
            mask = p > thresh
            if mask.sum() > n // 4 and mask.sum() < n:
                p_sp = np.where(mask, p, 0.0)
                p_sp = normalize_simplex(p_sp)
                sc_sp = fast_score(p_sp)
                if sc_sp < best_score:
                    p = p_sp
                    v = p.copy()
                    prev_p = p.copy()
                    best_score = sc_sp
                    best_p = p.copy()
                    no_improve = 0

    return best_p, best_score


def peak_shave(p, deadline, max_iter=50000):
    """
    Targeted subgradient method: directly reduce the max lag.
    Returns (best_p, best_score).
    """
    n = p.size
    p = normalize_simplex(p)
    best_p = p.copy()
    best_score = fast_score(p)
    step = 0.5
    no_improve = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1
        b = fft_convolve(p)
        k_star = int(np.argmax(b))
        M = float(b[k_star])
        s_p = float(np.sum(p))
        cur_score = 2.0 * n * M / (s_p * s_p)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        if it % 5000 == 0:
            print(f"  PeakShave it={it} best={best_score:.8f} k*={k_star} M={M:.6e}")

        # Exact subgradient of max at k_star
        # d(p*p)[k*]/dp[j] = (2 - delta(2j,k*)) * p[k*-j] * indicator(0<=k*-j<n)
        grad = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                coeff = 1.0 if (2 * j == k_star) else 2.0
                grad[j] = coeff * p[kj]

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Make gradient sum-zero (for simplex projection invariance)
        grad = grad - float(np.mean(grad))

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolve(q)
            M2 = float(np.max(qb))
            sc2 = 2.0 * n * M2
            if sc2 < 2.0 * n * M - 1e-10:
                p = q
                accepted = True
                if 2.0 * n * M2 < best_score * s_p * s_p - 1e-10:
                    best_score = 2.0 * n * M2 / (float(np.sum(p))**2)
                    best_p = p.copy()
                    no_improve = 0
                break

        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            if step < 1.0:
                step = min(1.5, step * 1.2)

    return best_p, best_score


def block_optimize(n, B, deadline, rng, beta_schedule=(200.0, 50000.0), max_iter=20000):
    """
    Optimize piecewise-constant block weights using projected gradient.
    B blocks, each of length L = n//B.
    Returns (best_block_weights, best_p_full, best_score).
    """
    L = n // B
    if L < 1 or B < 1:
        return None, None, float("inf")

    # Work with block weights (length B), normalized to sum 1
    # Then expand to full sequence by repeating each block element L times
    # Full sequence sum = B * (sum of block weights) * (1/L)... 
    # Actually: p_full = repeat(block_weights / B, L) so sum = B*(sum(bw)/B)*1 = sum(bw)/L... 
    # Let me think: if block_weights[i] is the weight for block i, and we repeat L times,
    # then p_full[j*L:(j+1)*L] = block_weights[j] / L (so sum = sum(block_weights)/L... no)
    # p_full = repeat(bw/L, L) => sum = B * bw[i]/L for each block => total = sum(bw)/L * L / B... 
    # Simpler: p_full[j] = bw[j // L] / n * ... 
    # Let's just: p_full = repeat(bw, L) / (L * sum(bw)) so sum(p_full) = 1
    # Then (p_full * p_full) at lag k has contributions from within-block and cross-block
    
    # Actually, the simplest: p_full = np.repeat(bw, L) and then normalize
    # This makes within-block uniform. The smoothing across block boundaries is what we lose.
    
    # For block optimization, we optimize over bw (length B) on simplex
    bw = np.full(B, 1.0 / B)
    best_bw = bw.copy()
    best_p = np.repeat(bw, L) / (L * bw.sum())
    best_p = normalize_simplex(best_p)
    best_sc = fast_score(best_p)
    
    prev_bw = bw.copy()
    momentum = 0.8
    step = 1.0
    no_improve = 0
    it = 0
    
    beta_start, beta_end = beta_schedule
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        frac = min(1.0, it / (max_iter * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        p_full = np.repeat(bw, L)
        p_full = normalize_simplex(p_full)
        
        b = fft_convolve(p_full)
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break
        
        sc = 2.0 * n * float(np.max(b))
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_bw = bw.copy()
            best_p = p_full.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        if it % 5000 == 0:
            print(f"    Block B={B} it={it} beta={beta:.0f} best={best_sc/n:.8f}")
            
        if no_improve > 3000:
            break
        
        # Gradient for block weights
        # grad_bw[i] = sum of grad over block i
        # Full gradient:
        v = bw + momentum * (bw - prev_bw)
        v = np.maximum(v, 0.0)
        sv = v.sum()
        if sv > 0:
            v = v / sv
        else:
            v = bw.copy()
        
        v_full = np.repeat(v, L)
        v_full = normalize_simplex(v_full)
        
        grad_full = grad_conv(v_full, w)
        grad_bw = grad_full.reshape(B, L).sum(axis=1)
        grad_bw = grad_bw - float(np.mean(grad_bw))
        
        gnorm = float(np.linalg.norm(grad_bw))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break
        
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step * mult
            q_bw = project_simplex(bw - alpha * grad_bw, 1.0)
            q_full = normalize_simplex(np.repeat(q_bw, L))
            qb = fft_convolve(q_full)
            G2, _ = smooth_max_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if B * G2 < B * G - 1e-12:
                prev_bw = bw.copy()
                bw = q_bw
                accepted = True
                break
        
        if not accepted:
            step *= 0.5
            momentum *= 0.95
            if step < 1e-9:
                break
        else:
            if step < 1.0:
                step = min(1.5, step * 1.2)
            momentum = min(0.92, momentum * 1.005)
    
    return best_bw, best_p, best_sc


def lp_inspired_move(p, deadline, max_iter=1000):
    """
    LP-inspired directional step: find direction that maximizes mass
    while keeping convolution below current max (approximately).
    Simplified: use the constraint that gradient dot direction <= 0 for peak lags.
    """
    n = p.size
    p = normalize_simplex(p)
    best_p = p.copy()
    best_score = fast_score(p)
    
    b = fft_convolve(p)
    M = float(np.max(b))
    
    # Identify "active" lags (near the max)
    thresh = M * 0.95
    active_mask = b >= thresh
    active_lags = np.where(active_mask)[0]
    
    if len(active_lags) == 0:
        return best_p, best_score
    
    # Try to find a direction d >= 0 with sum(d) > 0
    # such that for each active lag k: sum_j p[k-j]*d[j] <= 0
    # This is hard to satisfy exactly, so use a relaxed version:
    # Minimize sum of positive violations, subject to d >= 0, ||d|| = 1
    
    # Simplified: use the negative of the average gradient over active lags
    # as the direction, then project.
    total_grad = np.zeros(n)
    for k in active_lags:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                total_grad[j] += p[kj]
    
    total_grad /= max(1, len(active_lags))
    # We want to move in direction -total_grad (to reduce the active lags)
    # But we also want to increase mass where it doesn't create new peaks
    # Compromise: use -total_grad projected to have zero mean
    direction = -total_grad
    direction = direction - float(np.mean(direction))
    
    if float(np.linalg.norm(direction)) < 1e-15:
        return best_p, best_score
    
    # Try a step
    for step in (0.1, 0.05, 0.02, 0.01, 0.005):
        q = project_simplex(p + step * direction, 1.0)
        sc = fast_score(q)
        if sc < best_score:
            best_p = q
            best_score = sc
            return best_p, best_score
    
    return best_p, best_score


def official_evaluate(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of non-negative floats.
    
    Strategy:
    1. Generate diverse initial sequences
    2. Run block optimization for coarse structure (multiple block sizes)
    3. Run Nesterov projected gradient for fine optimization
    4. Peak-shave refinement
    5. LP-inspired moves
    6. Return best sequence found
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    n = SEQ_LEN
    rng = np.random.default_rng(seed)

    print(f"Starting optimization: n={n}, budget={budget_s:.0f}s")

    # Generate diverse starts
    starts = generate_starts(n, rng)
    print(f"Generated {len(starts)} initial sequences")

    # Score all starts
    scored_starts = []
    for i, p in enumerate(starts):
        sc = fast_score(p)
        if np.isfinite(sc):
            scored_starts.append((sc, p))
            print(f"  Start {i}: score={sc:.8f}")

    # Sort by score
    scored_starts.sort(key=lambda x: x[0])

    best_p = scored_starts[0][1].copy()
    best_score = scored_starts[0][0]
    print(f"Best initial score: {best_score:.8f}")

    # ---- Phase 1: Block optimization (coarse structure) ----
    block_deadline = min(deadline, start_time + 0.35 * budget_s)
    print(f"\nPhase 1: Block optimization (deadline at {block_deadline - start_time:.0f}s)")

    for B in (16, 32, 64, 128):
        if time.monotonic() >= block_deadline:
            break
        if n % B != 0:
            continue

        # Use top-3 starts as block initializations
        for idx in range(min(3, len(scored_starts))):
            if time.monotonic() >= block_deadline:
                break
            p_start = scored_starts[idx][1]
            # Downsample to block size
            L = n // B
            bw_init = p_start.reshape(B, L).mean(axis=1)
            bw_init = normalize_simplex(bw_init)
            # Insert into scored_starts as a starting point for block opt
            # Actually just run block optimization from uniform and from the best
            pass

        # Run block optimization from uniform
        t_block_start = time.monotonic()
        alloc = min(block_deadline - t_block_start, 40.0)
        if alloc < 5.0:
            continue
        sub_deadline = t_block_start + alloc
        _, p_block, sc_block = block_optimize(n, B, sub_deadline, rng)

        if p_block is not None and np.isfinite(sc_block):
            sc_full = fast_score(p_block)
            if sc_full < best_score:
                best_score = sc_full
                best_p = p_block.copy()
                print(f"  Block B={B} improved to {best_score:.8f}")

    # ---- Phase 2: Nesterov projected gradient (fine optimization) ----
    print(f"\nPhase 2: NPG refinement (deadline at {deadline - start_time:.0f}s)")

    # Allocate time among top starts
    top_count = min(4, len(scored_starts))
    # Include the current best
    refinement_starts = [best_p]
    for i in range(top_count):
        if scored_starts[i][1] is not best_p:
            refinement_starts.append(scored_starts[i][1])

    # Time allocation: give more time to later (potentially better) starts
    remaining = deadline - time.monotonic() - 10.0
    if remaining < 30:
        refinement_starts = [best_p]

    if len(refinement_starts) == 1:
        allocs = [max(0, remaining)]
    else:
        # Weighted allocation: best start gets less (already good), others get more
        weights = [0.2] + [0.8 / (len(refinement_starts) - 1)] * (len(refinement_starts) - 1)
        if len(refinement_starts) > 1:
            # Actually give best start the most time since it's most likely to improve
            weights = [0.4] + [0.6 / (len(refinement_starts) - 1)] * (len(refinement_starts) - 1)
        allocs = [remaining * w for w in weights]

    for i, p_start in enumerate(refinement_starts):
        alloc = allocs[i]
        if alloc < 20:
            break
        sub_deadline = time.monotonic() + alloc
        print(f"  NPG run {i+1}/{len(refinement_starts)}: alloc={alloc:.0f}s, start_score={fast_score(p_start):.8f}")

        p_fit, sc_fit = nesterov_projected_gradient(
            p_start, sub_deadline, rng,
            beta_start=100.0, beta_end=500000.0,
            max_iter=1000000,
            sparsify_interval=5000,
            sparsify_thresh=0.005
        )

        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  NPG run {i+1} improved to {best_score:.8f}")

        # Also add this result to potential starts for others
        if np.isfinite(sc_fit):
            scored_starts.insert(0, (sc_fit, p_fit))

    # ---- Phase 3: Peak shaving ----
    print(f"\nPhase 3: Peak shaving")
    if time.monotonic() < deadline - 10:
        ps_deadline = time.monotonic() + min(60.0, deadline - time.monotonic() - 10)
        p_shave, sc_shave = peak_shave(best_p, ps_deadline, max_iter=50000)
        if np.isfinite(sc_shave) and sc_shave < best_score:
            best_score = sc_shave
            best_p = p_shave.copy()
            print(f"  Peak shaving improved to {best_score:.8f}")

    # ---- Phase 4: LP-inspired moves ----
    print(f"\nPhase 4: LP-inspired moves")
    if time.monotonic() < deadline - 5:
        for _ in range(3):
            if time.monotonic() >= deadline - 3:
                break
            p_lp, sc_lp = lp_inspired_move(best_p, time.monotonic() + 5.0)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_p = p_lp.copy()
                print(f"  LP move improved to {best_score:.8f}")

    # ---- Final: one more NPG run with very high beta for polish ----
    print(f"\nPhase 5: Final polish")
    if time.monotonic() < deadline - 5:
        polish_deadline = time.monotonic() + min(30.0, deadline - time.monotonic() - 3)
        p_polish, sc_polish = nesterov_projected_gradient(
            best_p, polish_deadline, rng,
            beta_start=50000.0, beta_end=500000.0,
            max_iter=200000,
            sparsify_interval=0,
            sparsify_thresh=0.0
        )
        if np.isfinite(sc_polish) and sc_polish < best_score:
            best_score = sc_polish
            best_p = p_polish.copy()
            print(f"  Polish improved to {best_score:.8f}")

    # ---- Final cleanup ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    print(f"\nFinal internal score: {best_score:.8f}")
    print(f"Sum: {sum(final_seq):.6f}")
    print(f"Max element: {max(final_seq):.6f}")
    print(f"Min element: {min(final_seq):.6f}")
    print(f"Num zeros: {sum(1 for x in final_seq if x == 0.0)}")

    # Official evaluation
    off = official_evaluate(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-9:
        print("Falling back to uniform!")
        return [1.0] * n

    return final_seq
