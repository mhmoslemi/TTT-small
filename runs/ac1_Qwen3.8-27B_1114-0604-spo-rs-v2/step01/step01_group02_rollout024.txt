"""
LSE-smoothed gradient descent with FFT-based gradients for minimizing
the peak-to-average ratio of linear self-autocorrelation.
"""

import numpy as np
import time

def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def lse_gradient(a, temp):
    """Compute (value, gradient) of 2n * LSE_T(a*a) / sum(a)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    S2 = S * S
    
    # LSE value and softmax weights
    m = float(c.max())
    exp_c = np.exp((c - m) / temp)
    sum_exp = float(exp_c.sum())
    w = exp_c / sum_exp
    lse_val = m + temp * np.log(sum_exp)
    
    # d[j] = sum_k w[k] * a[k-j] for valid indices
    # = (a * w_rev)[2n-2-j] where w_rev = w[::-1]
    # conv has length 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, N)
    conv_aw = np.fft.irfft(A * W, N)[:3 * n - 2]
    d = conv_aw[n - 1 : 2 * n - 1][::-1]
    
    val = 2.0 * n * lse_val / S2
    grad = (4.0 * n / S2) * d - (4.0 * n * lse_val / (S2 * S))
    
    return val, grad

def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)
    np.maximum(a, 0.0, out=a)

def optimize_lse(a0, deadline, rng, init_temp, final_temp, init_lr, patience=200):
    """Run LSE-Adam optimization until deadline."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Initial evaluation
    c = fast_self_conv(a)
    S = a.sum()
    score = 2.0 * n * float(c.max()) / (S * S)
    best_score = score
    best_a = a.copy()
    no_improve = 0
    
    # Total duration for temperature scheduling
    t_start = time.monotonic()
    total_dur = max(1.0, deadline - t_start)
    
    while time.monotonic() < deadline:
        # Determine current temperature (exponential annealing)
        frac = min(1.0, (time.monotonic() - t_start) / total_dur)
        temp = init_temp * (final_temp / init_temp) ** (frac * frac)  # quadratic annealing
        
        # Compute gradient
        val, g = lse_gradient(a, temp)
        
        # Adam update
        t += 1
        adam_step(a, g, m, v, t, init_lr)
        
        # Evaluate true score periodically
        if t % 10 == 0 or t == 1:
            c = fast_self_conv(a)
            S = a.sum()
            if S < 0.01:
                # Reset
                a = best_a.copy()
                m[:] = 0
                v[:] = 0
                t = 0
                continue
            score = 2.0 * n * float(c.max()) / (S * S)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= patience:
            # Perturb and restart
            a = best_a * (1.0 + 0.1 * rng.standard_normal(n))
            np.maximum(a, 0.0, out=a)
            m[:] = 0
            v[:] = 0
            t = 0
            no_improve = 0
            patience = max(50, patience // 2)
    
    return best_score, best_a

def objective_and_grad(a, temp):
    """Returns (objective, gradient) for LSE-smoothed objective."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    S2 = S * S
    
    m = float(c.max())
    exp_c = np.exp((c - m) / temp)
    sum_exp = float(exp_c.sum())
    w = exp_c / sum_exp
    lse_val = m + temp * np.log(sum_exp)
    
    val = 2.0 * n * lse_val / S2

    N = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N)
