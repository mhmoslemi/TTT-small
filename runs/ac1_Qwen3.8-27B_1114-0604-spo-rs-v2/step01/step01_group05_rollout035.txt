```python
"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences using
Adam optimization in log-space with beta annealing.
"""

import numpy as np
import time

def fft_convolution(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]**2])
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fast_score(p):
    """Compute 2*n*max(p*p) / (sum(p))^2 efficiently."""
    n = p.size
    S = p.sum()
    if S <= 0:
        return float('inf')
    b = fft_convolution(p)
    M = b.max()
    return 2.0 * n * M / (S * S)

def smooth_max(b, beta):
    """Smooth approximation of max(b) using log-sum-exp."""
    m = b.max()
    if m <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = e.sum()
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for the gradient computation."""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return s
```

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using Adam in log-space with beta annealing.
Uses FFT-based convolution and multiple diverse initializations.
"""

import numpy as np
import time

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]**2])
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fast_score(p):
    n = p.size
    S = p.sum()
    if S <= 0:
        return float('inf')
    b = fft_convolution(p)
    M = b.max()
    return 2.0 * n * M / (S * S)

def smooth_max(b, beta):
    m = b.max()
    if m <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = e.sum()
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j}"""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return s

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / v.size)
    low = v.min() - 1
    high = v.max() + 1
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = np.maximum(v - mid, 0).sum()
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0)
    s = q.sum()
    if s <= 0:
        return np.full(v.shape, target / v.size)
    return q * (target / s)

def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.full(n, 1.0/n))
    
    # 1/sqrt(i+1) profile
    a = 1.0 / np.sqrt(np.arange(n) + 1.0)
    inits.append(a / a.sum())
    
    # sqrt(i+1) profile (increasing)
    a = np.sqrt(np.arange(n) + 1.0)
    inits.append(a / a.sum())
    
    # Cosine taper
    i = np.arange(n)
    a = 0.5 * (1 + np.cos(np.pi * i / (n - 1)))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # Triangle (peak in middle)
    i = np.arange(n)
    c = n / 2
    a = 1 - np.abs(i - c) / c
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    
    # Exponential decay
    a = np.exp(-2.0 * np.arange(n) / n)
    inits.append(a / a.sum())
    
    # Two bumps
    i = np.arange(n)
    a = np.exp(-8*(i/n - 0.25)**2) + np.exp(-8*(i/n - 0.75)**2)
    inits.append(a / a.sum())
    
    # Random smooth (low-pass filtered noise)
    for _ in range(3):
        z = rng.standard_normal(n)
        # Low-pass filter in frequency domain
        Z = np.fft.rfft(z)
        freqs = np.fft.rfftfreq(n)
        Z *= np.exp(-4 * (freqs / 0.5)**2)  # Gaussian low-pass
        z = np.fft.irfft(Z, n)
        a = np.maximum(z, 0)
        if a.sum() > 0:
            inits.append(a / a.sum())
        else:
            inits.append(np.full(n, 1.0/n))
    
    return inits

def adam_logspace(x0, deadline, beta_schedule, n_iter_max=500000):
    """
    Adam optimizer in log-space.
    x0: initial x values (log of a)
    beta_schedule: function or tuple (beta_start, beta_end) for annealing
    """
    n = x0.size
    x = x0.copy().astype(np.float64)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Learning rate schedule
    lr = 0.01
    best_a = softmax(x).copy()
    best_score = fast_score(best_a)
    
    it = 0
    no_improve = 0
    check_interval = 100
    
    while it < n_iter_max and time.monotonic() < deadline:
        # Get current beta
        if isinstance(beta_schedule, tuple):
            frac = min(1.0, it / n_iter_max)
            beta = beta_schedule[0] * (beta_schedule[1] / beta_schedule[0]) ** frac
        else:
            beta = beta_schedule
        
        a = softmax(x)
        b = fft_convolution(a)
        G, w = smooth_max(b, beta)
        
        if not np.isfinite(G):
            # Reduce beta
            beta = max(10, beta * 0.1)
            continue
        
        # Gradient in a-space
        grad_a = 2.0 * corr_pw(a, w)
        grad_a = np.maximum(grad_a, 0)  # clip small negatives from FFT roundoff
        
        # Convert to x-space gradient
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic score check
        if it % check_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += check_interval
            
            if it % 1000 == 0:
                print(f"  adam it={it} beta={beta:.0f} score={fast_score(best_a):.8f} lr={lr:.4f}")
            
            # Learning rate decay
            if no_improve > 5000:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    break
            
            if no_improve > 20000:
                break
    
    final_a = softmax(x)
    final_score = fast_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a
    
    return best_a, best_score

def full_projected_gradient(p0, deadline, beta_start=500, beta_end=500000, max_iter=500000):
    """Projected gradient on the simplex with Nesterov momentum."""
    n = p0.size
    a = p0.copy().astype(np.float64)
    a = project_simplex(a, 1.0)
    
    # Nesteros momentum
    a_prev = a.copy()
    momentum = 0.9
    
    best_a = a.copy()
    best_score = fast_score(a)
    
    lr = 1.0
    it = 0
    no_improve = 0
    
    while it < max_iter and time.monotonic() < deadline:
        # Nesterov step
        a_nested = a_prev + momentum * (a - a_prev)
        a_nested = project_simplex(a_nested, 1.0)
        
        b = fft_convolution(a_nested)
        G, w = smooth_max(b, beta_start * (beta_end/beta_start) ** (it/max_iter))
        
        if not np.isfinite(G):
            break
        
        grad_a = 2.0 * corr_pw(a_nested, w)
        grad_a = np.maximum(grad_a, 0)
        grad = grad_a - np.mean(grad_a)
        
        # Step
        step = lr / (np.sqrt(it + 1))  # 1/sqrt(t) step size
        a_new = project_simplex(a_nested - step * grad, 1.0)
        
        # Check improvement
        if it % 200 == 0:
            cur_score = fast_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 200
        
        a_prev = a.copy()
        a = a_new
        it += 1
        
        if no_improve > 10000:
            break
    
    final_score = fast_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.
Uses Adam in log-space with beta annealing, multiplicative update polish,
and Nesterov-accelerated projected gradient. Combines diverse initialization
with the previous best sequence to avoid local minima.
"""

import numpy as np
import time

def fft_convolution(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0] * a[0]])
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    n = p.size
    if n == 0:
        return float('inf')
    S = float(p.sum())
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(b.max())
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M / (S * S)

def smooth_max(b, beta):
    """Smooth max via log-sum-exp. Returns (value, weights)."""
    m = float(b.max())
    if m <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} (cross-correlation for gradient)."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(max(n, 1))
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return s

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(x) / x.size
    return e / s

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    low = float(v.min()) - 1.0
    high = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.maximum(v - mid, 0.0).sum())
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(q.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n)
    return q * (target / s)

def normalize_simplex(p):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size)
    return p / s

def make_initializations(n, rng, prev_seq=None):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.full(n, 1.0 / n))

    # 1/sqrt(i+1)
    a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    inits.append(a / a.sum())

    # Increasing sqrt
    a = np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    inits.append(a / a.sum())

    # Cosine taper (Hann-like)
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * i / max(n - 1, 1)))
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())

    # Triangle peak center
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / max(c, 1.0)
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())

    # Exponential decay
    a = np.exp(-3.0 * i / n)
    inits.append(a / a.sum())

    # Two Gaussian bumps
    a = np.exp(-12.0 * (i / n - 0.3) ** 2) + np.exp(-12.0 * (i / n - 0.7) ** 2
              ) + 0.01
    inits.append(a / a.sum())

    # Fluid-like: high at edges, low in middle
    a = (1.0 - np.sin(np.pi * i / n)) ** 2 + 0.05
    inits.append(a / a.sum())

    # Random smooth profiles
    for scale in (2.0, 5.0, 10.0):
        z = rng.standard_normal(n)
        Z = np.fft.rfft(z)
        freqs = np.fft.rfftfreq(n)
        Z *= np.exp(-(freqs / (1.0 / (scale * 1.0 / n))) ** 2)
        z = np.fft.irfft(Z, n)
        a = np.maximum(z, 0.0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())
        else:
            a = np.exp(0.5 * z)
            inits.append(a / a.sum())

    # Previous best sequence (resampled to length n if needed)
    if prev_seq is not None and len(prev_seq) > 0:
        prev = np.asarray(prev_seq, dtype=np.float64)
        prev = np.maximum(prev, 0.0)
        ps = float(prev.sum())
        if ps > 0 and np.isfinite(ps):
            prev = prev / ps
            if prev.size == n:
                inits.append(prev.copy())
            else:
                # Resample via linear interpolation
                old_x = np.linspace(0, 1, prev.size)
                new_x = np.linspace(0, 1, n)
                resampled = np.interp(new_x, old_x, prev)
                resampled = np.maximum(resampled, 0.0)
                rs = float(resampled.sum())
                if rs > 0:
                    inits.append(resampled / rs)

    return inits

def adam_logspace(a0, deadline, n_iter_max=800000):
    """
    Adam optimizer in log-space with beta annealing.
    a0: initial non-negative sequence (will be normalized).
    """
    n = a0.size
    a0 = normalize_simplex(a0)
    x = np.log(a0 + 1e-15)

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10
    lr = 0.05

    m_ = np.zeros(n)
    v_ = np.zeros(n)

    best_a = a0.copy()
    best_score = fast_score(a0)

    beta_start = 50.0
    beta_end = 1000000.0

    it = 0
    no_improve = 0
    print_interval = 500

    while it < n_iter_max and time.monotonic() < deadline:
        # Beta annealing (exponential schedule)
        frac = min(1.0, it / (n_iter_max * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        a = softmax(x)
        b = fft_convolution(a)
        G, w = smooth_max(b, beta)

        if not np.isfinite(G):
            beta = max(10.0, beta * 0.01)
            continue

        # Gradient in a-space
        grad_a = 2.0 * corr_pw(a, w)
        grad_a = np.maximum(grad_a, 0.0)

        # Convert to x-space: grad_x = a * (grad_a - dot(grad_a, a))
        dot_val = float(np.dot(grad_a, a))
        grad_x = a * (grad_a - dot_val)

        # Handle non-finite gradient
        grad_x = np.where(np.isfinite(grad_x), grad_x, 0.0)

        # Adam update
        it += 1
        m_ = beta1 * m_ + (1.0 - beta1) * grad_x
        v_ = beta2 * v_ + (1.0 - beta2) * grad_x ** 2
        m_hat = m_ / (1.0 - beta1 ** it)
        v_hat = v_ / (1.0 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if it % print_interval == 0:
            cur_a = softmax(x)
            cur_score = fast_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                no_improve = 0
            else:
                no_improve += print_interval

            if it % (print_interval * 10) == 0:
                print(f"    adam it={it} beta={beta:.0f} best={best_score:.8f} lr={lr:.5f} no_imp={no_improve}")

            # Adaptive learning rate
            if no_improve > 10000:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
            elif no_improve > 5000:
                lr *= 0.7
                no_improve = 0

            if no_improve > 30000:
                break

    # Final check
    final_a = softmax(x)
    fs = fast_score(final_a)
    if fs < best_score:
        best_score = fs
        best_a = final_a.copy()

    return best_a, best_score

def multiplicative_update(a0, deadline, n_iter=50000):
    """
    Multiplicative update inspired by LP dual:
    g[i] proportional to 1/gradient[i], which pushes mass away from
    positions contributing to the peak.
    """
    n = a0.size
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = fast_score(a)

    beta = 10000.0
    t = 0.1  # mixing parameter

    for it in range(n_iter):
        if time.monotonic() >= deadline:
            break

        b = fft_convolution(a)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break

        grad_a = 2.0 * corr_pw(a, w)
        grad_a = np.maximum(grad_a, 1e-15)

        # Multiplicative update: g ∝ a / grad (push away from peak)
        g = a / (grad_a + 1e-10)
        gs = float(g.sum())
        if gs <= 0 or not np.isfinite(gs):
            break
        g /= gs

        # Mix
        a = (1.0 - t) * a + t * g
        a = normalize_simplex(a)

        if it % 1000 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # Adaptive t: decrease over time
        t = max(0.01, 0.1 * (1.0 - it / n_iter) + 0.01)

        # Increase beta
        if it % 5000 == 0 and it > 0:
            beta = min(1000000.0, beta * 2.0)

    fs = fast_score(a)
    if fs < best_score:
        best_score = fs
        best_a = a.copy()

    return best_a, best_score

def nesterov_pgd(a0, deadline, max_iter=300000):
    """Nesterov-accelerated projected gradient on the simplex."""
    n = a0.size
    a = normalize_simplex(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = fast_score(a)

    momentum = 0.9
    beta_start = 1000.0
    beta_end = 1000000.0
    lr = 5.0

    it = 0
    no_improve = 0

    while it < max_iter and time.monotonic() < deadline:
        # Nesterov lookahead
        a_neck = a_prev + momentum * (a - a_prev)
        a_neck = project_simplex(a_neck, 1.0)

        frac = min(1.0, it / max_iter)
        beta = beta_start * (beta_end / beta_start) ** frac

        b = fft_convolution(a_neck)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break

        grad_a = 2.0 * corr_pw(a_neck, w)
        grad_a = np.maximum(grad_a, 0.0)
        grad = grad_a - float(np.mean(grad_a))

        # Step size: O(1/sqrt(t))
        step = lr / np.sqrt(it + 1.0)
        a_new = project_simplex(a_neck - step * grad, 1.0)

        a_prev = a.copy()
        a = a_new
        it += 1

        if it % 500 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 500

            if it % 5000 == 0:
                print(f"    nesterov it={it} best={best_score:.8f} beta={beta:.0f}")

            if no_improve > 10000:
                break

    fs = fast_score(a)
    if fs < best_score:
        best_score = fs
        best_a = a.copy()

    return best_a, best_score

def lp_iteration(a0, deadline, n_iter=20000):
    """
    Simplified LP-based iteration from the state-of-the-art description.
    Finds g maximizing sum(g) s.t. f*g <= M, g >= 0 (approximately),
    then mixes f with g.
    """
    n = a0.size
    a = normalize_simplex(a0)
    best_a = a.copy()
    best_score = fast_score(a)

    # Normalize to sum = sqrt(2n) as in the description
    target_sum = np.sqrt(2.0 * n)
    a_scaled = a * target_sum

    t_mix = 0.15
    beta = 50000.0

    for it in range(n_iter):
        if time.monotonic() >= deadline:
            break

        b = fft_convolution(a_scaled)
        M = float(b.max())
        if M <= 0 or not np.isfinite(M):
            break

        # Compute gradient (proxy for LP solution)
        G, w = smooth_max(b, beta)
        grad_a = 2.0 * corr_pw(a_scaled, w)
        grad_a = np.maximum(grad_a, 1e-15)

        # LP solution approximation: g ∝ 1/max(gradient, mean_gradient)
        mean_grad = float(np.mean(grad_a))
        g = 1.0 / np.maximum(grad_a, mean_grad * 0.5)

        # Also try: g from the dual perspective
        # The dual variable lambda[k] should be high where b[k] exceeds M
        # Then g[i] ∝ sum_k lambda[k] * F[k,i]
        # Approximate with: g[i] ∝ sum_k w[k] * a[k-i] (which is just grad_a/2)
        # So g ∝ 1/grad_a is the natural choice

        gs = float(g.sum())
        if gs <= 0 or not np.isfinite(gs):
            break
        g /= gs
        g_scaled = g * target_sum

        # Mix
        a_new = (1.0 - t_mix) * a_scaled + t_mix * g_scaled
        a_new = np.maximum(a_new, 0.0)
        ns = float(a_new.sum())
        if ns <= 0:
            break
        a_scaled = a_new / ns * target_sum

        # Track best in normalized form
        a_norm = a_scaled / target_sum
        if it % 500 == 0:
            cur_score = fast_score(a_norm)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_norm.copy()

        # Adapt mixing and beta
        t_mix = max(0.05, t_mix * 0.999)
        if it % 2000 == 0 and it > 0:
            beta = min(500000.0, beta * 1.5)

    fs = fast_score(best_a)
    return best_a, fs

def propose_candidate(seed=42, budget_s=2400, **kwargs):

    Search for the best non-negative sequence minimizing the evaluation score.
    Uses diverse initialization + Adam in log-space + multiplicative update
    + Nesterov PGD + LP iteration.
    """

        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Choose sequence length
    if budget_s >= 1800:
        N = 16384
    elif budget_s >= 900:
        N = 8192
    else:
        N = 4096

    print(f"Starting search with N={N}, budget={budget_s:.0f}s")

    # Get previous best if available
    prev_seq = None
    try:
        if 'height_sequence_1' in dir() or True:
            prev_seq = list(height_sequence_1)
            print(f"Using previous best sequence of length {len(prev_seq)}")
    except Exception:
        prev_seq = None

    # Generate initializations
    inits = make_initializations(N, rng, prev_seq)
    print(f"Generated {len(inits)} initializations")

    # Score all initializations
    scored = []
    for i, p in enumerate(inits):
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, i, p))
            print(f"  init {i}: score={sc:.8f}")

    # Sort by score
    scored.sort(key=lambda x: x[0])

    # Phase 1: Adam optimization from top initializations
    best_p = scored[0][2].copy() if scored else np.full(N, 1.0 / N)
    best_score = scored[0][0] if scored else 2.0
    print(f"Best initial score: {best_score:.8f}")

    # Allocate time: 60% Adam, 15% mult update, 15% Nesterov, 10% LP
    total_time = max(10.0, deadline - time.monotonic())
    adam_time = total_time * 0.55
    mult_time = total_time * 0.15

    # Adam from top 3 starts

    n_top = min(3, len(scored))

    for idx in range(n_top):
        if time.monotonic() >= adam_deadline:
            break
        sc, _, p0 = scored[idx]
        sub_dl = min(adam_deadline, time.monotonic() + per_adam * (n_top - idx))
        print(f"Adam from init {idx} (score={sc:.6f}), time_budget={per_adam:.0f}s")
        p_fit, sc_fit = adam_logspace(p0, sub_dl)
        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Adam improved to {best_score:.8f}")

    # Adam restart with perturbation from best
    if time.monotonic() < deadline - 120:
        perturb_time = min(60.0, (deadline - time.monotonic()) * 0.1)
        p_pert = best_p.copy()
        noise = rng.standard_normal(N) * 0.1
        p_pert = normalize_simplex(p_pert + noise)
        sub_dl = time.monotonic() + perturb_time
        print(f"Adam perturb restart, budget={perturb_time:.0f}s")
        p_fit, sc_fit = adam_logspace(p_pert, sub_dl)
        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Perturb improved to {best_score:.8f}")

    # Phase 2: Multiplicative update polish
    if time.monotonic() < deadline - 60:
        mult_deadline = time.monotonic() + mult_time
        print(f"Multiplicative update, budget={mult_time:.0f}s")
        p_fit, sc_fit = multiplicative_update(best_p, mult_deadline)
        if sc_fit < best_score:

            print(f"  Mult update improved to {best_score:.8f}")

    # Phase 3: Nesterov PGD

        print(f"Nesterov PGD, budget={nesterov_time:.0f}s")
        p_fit, sc_fit = nesterov_pgd(best_p, nest_dl)

        print(f"LP iteration, budget={lp_time:.0f}s")
        p_fit, sc_fit = lp_iteration(best_p, lp_dl)
        if sc_fit < best_score:

    # Phase 5: Final Adam polish with high beta

                                       n_iter_max=200000)
        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Final polish improved to {best_score:.8f}")

    # Final cleanup
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-6:
        # Fallback: uniform sequence
        fallback = [1.0] * N
        fb_score = official_score(fallback)
        print(f"Fallback uniform score: {fb_score:.8f}")
        return fallback

    return final_seq
