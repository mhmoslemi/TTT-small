"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using alternating LP (block coordinate descent on the bilinear max-convolution problem).

Key insight: The score equals 2*n*max_k (a*a)[k] when sum(a)=1. We alternate:
  - Fix b, solve LP: min t s.t. (x*b)[k] <= t for all k, sum x = 1, x >= 0
  - Update b = x
This gives monotone non-increasing scores and converges to a fixed point.
"""

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _score_normalized(p):
    m = len(p)
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    return float(2.0 * m * np.max(c))

    p = np.asarray(p, dtype=float)
    p = np.clip(p, 0, 1000)
    s = float(p.sum())
    if s < 0.01:

def _build_lp(a):
    n = len(a)
    n_rows = 2 * n - 1
    n_cols = n + 1
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        cnt = i_end - i_start + 1
        if cnt > 0:
            idx = np.arange(i_start, i_end + 1, dtype=np.int32)
            vals = a[k - idx]
            rows_list.append(np.full(cnt, k, dtype=np.int32))
            cols_list.append(idx)
            vals_list.append(vals)
        rows_list.append(np.array([k], dtype=np.int32))
        cols_list.append(np.array([n], dtype=np.int32))
        vals_list.append(np.array([-1.0]))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n_cols))

    A_ub = _build_lp(a)
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0

                         bounds=[(0, None)] * (n + 1), method='highs')

            x = result.x[:n]
            x = np.maximum(x, 0)
            s = x.sum()
            if s > 0:

            return x
    except Exception:
        pass
    return None

def _resample(arr, m):
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0) + 1e-12
    return _sanitize(out)

        rng = np.random.default_rng((int(seed) % 1000000007) + 1)
    except Exception:
        rng = np.random.default_rng(42)

    # Get parent sequence if available
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except Exception:
        pass

    best_arr = None
    best_score = float('inf')

    def _update_best(arr, sc):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(sc) and sc < best_score:
            best_arr = arr.copy()
            best_score = sc

    # Try multiple lengths with LP alternating
    lengths = [2048, 4096, 8192]

    for n in lengths:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        # Allocate time based on length
        if n == 2048:
            t_alloc = min(400.0, rem * 0.3)
        elif n == 4096:
            t_alloc = min(1000.0, rem * 0.5)
        else:
            t_alloc = min(rem * 0.7, 1200.0)
        t_alloc = max(60.0, t_alloc)

        print(f"Trying n={n}, time_alloc={t_alloc:.0f}s")

        # Generate initial points
        inits = []
        # Two-block seeds
        for frac in (0.3, 0.5, 0.7):
            for ratio in (0.3, 1.0, 3.0):
                ib = _two_block_seed(n, frac, ratio)
                if ib is not None:
                    inits.append(ib)
        # Resample parent
        if parent_arr is not None:
            ip = _resample(parent_arr, n)
            if ip is not None:
                inits.append(ip)
                # Perturbed parent
                ip2 = _sanitize(ip * np.exp(rng.normal(0, 0.05, n)))
                if ip2 is not None:
                    inits.append(ip2)
        # Smooth random inits
        for mode in range(4):
            ir = _smooth_init(n, rng, mode)
            if ir is not None:
                inits.append(ir)

        # Score and sort initial points
        scored = []
        for init in inits:
            sc = _score_normalized(init)
            if np.isfinite(sc):
                scored.append((sc, init))
        scored.sort(key=lambda x: x[0])

        if not scored:
            continue

        # Run LP alternating on top 3 initial points
        n_starts = min(3, len(scored))
        per_start = t_alloc / n_starts

        for si in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            init = scored[si][1]
            t_here = min(per_start, rem - 10)
            if t_here < 20:
                break
            print(f"  Start {si}: init_score={scored[si][0]:.6f}, time={t_here:.0f}s")
            res, sc = _alt_lp(init, t_here, max_iters=300, verbose=False)
            _update_best(res, sc)
            if res is not None:
                # Quick polish with MMD
                rem2 = deadline - time.monotonic()
                if rem2 > 60:
                    t_pol = min(60.0, rem2 * 0.3)
                    rp, scp = _mmd_polish(res, rng, t_pol, 50, 500)
                    _update_best(rp, scp)
            print(f"  After start {si}: best={best_score:.6f}")

    # Polish the best result
    rem = deadline - time.monotonic()
    if rem > 60 and best_arr is not None:
        print(f"Polishing best (score={best_score:.6f}), time remaining={rem:.0f}s")
        n_best = len(best_arr)
        # MMD polish
        t1 = min(rem * 0.4, 300.0)
        if t1 > 30:
            p1, s1 = _mmd_polish(best_arr, rng, t1, 80, 3000)
            _update_best(p1, s1)

        # Block polish
        rem = deadline - time.monotonic()
        if rem > 40:
            t2 = min(rem * 0.4, 200.0)
            p2, s2 = _block_polish(best_arr, rng, t2)
            _update_best(p2, s2)

        # Final MMD
        rem = deadline - time.monotonic()
        if rem > 40:
            t3 = min(rem * 0.5, 150.0)
            p3, s3 = _mmd_polish(best_arr, rng, t3, 150, 5000)
            _update_best(p3, s3)

    # If we still haven't found anything, use two-block
    if best_arr is None or not np.isfinite(best_score):
        n_fb = 2048
        for frac in (0.3, 0.5, 0.7):
            for ratio in (0.3, 1.0, 3.0):
                fb = _two_block_seed(n_fb, frac, ratio)
                if fb is not None:
                    sc = _score_normalized(fb)
                    _update_best(fb, sc)

    if best_arr is None:
        best_arr = np.ones(1024) / 1024

    # Final safety check
    best_arr = np.asarray(best_arr, dtype=float)
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0, 1000)
    s = float(best_arr.sum())
    if s <= 0 or not np.isfinite(s):
        best_arr = np.ones(1024)
    else:
        best_arr = best_arr / s

    elapsed = time.monotonic() - start_time
    final_score = _score_normalized(best_arr)
    print(f"Final: n={len(best_arr)}, score={final_score:.6f}, time={elapsed:.1f}s")

    # Verify with evaluate_sequence if available
    try:
        lst = [float(x) for x in best_arr]
        ev_score = float(evaluate_sequence(lst))
        print(f"Verified score: {ev_score:.6f}")
    except Exception as ex:
        print(f"Verification failed: {ex}")

    return [float(x) for x in best_arr]
