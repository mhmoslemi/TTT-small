from scipy.optimize import linprog

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * np.max(b) / (s * s)

def normalize(a):
    s = a.sum()
    if s <= 0:
        return a
    return a / s

def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_refine(a, M, max_iter=100):

Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.
Core idea: at each step, solve an LP to find a direction g that keeps (a*g) bounded,
then take a convex combination to reduce the peak of a*a.
Combined with gradient-based polishing and multiple initializations.

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def normalize(a):
    """Normalize to sum=1."""
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / a.size
    return a / s


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1} (simplex)."""
    n = v.size

        return v

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)

        shift -= shift.max()
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum
        
        g = 2.0 * np.correlate(w, a, mode='valid')
        
        # Momentum
        vel = beta * vel + (1 - beta) * g
        a_new = project_nonneg_sum1(a - eta * vel)
        
        sc_new = compute_score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = project_nonneg_sum1(a - 0.5 * eta * vel)
            no_improve += 1
        
        if no_improve > 200:
            a = project_nonneg_sum1(best_a + 0.05 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.2 + 0.3 * rng.random()
            no_improve = 0
        
        it += 1
    
    return best_a, best_sc

The optimal solution g* has the property that it's "maximally spread out" while keeping (a*g*) <= M. Intuitively, g* will put mass where a is small (to avoid violating the constraint), which means g* is "anti-correlated" with a in some sense.

((1-t)*a + t*g*) * ((1-t)*a + t*g*) 
= (1-t)^2 (a*a) + 2t(1-t)(a*g*) + t^2 (g* * g*)

The cross term (a*g*) <= M = max(a*a). And (g* * g*) has its own peak, which might be larger or smaller than M.

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.
Core idea: solve LP to find direction g keeping (a*g) bounded, then convex-combine.
Combined with adaptive step sizes, gradient polishing, diverse initializations,
and random restarts to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] (linear convolution)."""
    n = a.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def solve_lp_direction(a, M, top_frac=1.0, time_limit=15):
    """
    Solve LP: max sum(g) s.t. g >= 0, (a*g)[k] <= M for selected k.
    If top_frac < 1.0, only constrain the top fraction of peaks.
    Returns g normalized to sum=1, or None.
    """
    n = a.size
    A = build_conv_matrix(a)
    m = A.shape[0]

    if top_frac < 1.0:
        # Only constrain top peaks
        conv_vals = full_conv(a)
        threshold = np.quantile(conv_vals, 1.0 - top_frac)
        active = conv_vals >= threshold
        A_active = A[active, :]
        b_ub = np.full(A_active.shape[0], M)
    else:
        A_active = A
        b_ub = np.full(m, M)

    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(a, time_budget, rng, verbose=True):
    """
    Iteratively refine a using LP-based direction finding with adaptive steps.
    """
    n = a.size
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    t = 0.5
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    if verbose:
        print(f"    [lp] start: n={n}, score={best_sc:.6f}")

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        # Try LP with full constraints first, fall back to relaxed
        g = solve_lp_direction(a, M * 1.0001, top_frac=1.0,
                               time_limit=min(15, remaining * 0.3))

        if g is None or float(np.dot(a, g)) > 0.9999:
            # Try with relaxed constraints
            g = solve_lp_direction(a, M * 1.01, top_frac=0.8,
                                   time_limit=min(10, remaining * 0.2))

        if g is None:
            a = project_simplex(a + 0.08 * rng.standard_normal(n))
            no_improve += 1
        else:
            # Line search
            best_local_a = a.copy()
            best_local_sc = cur_sc
            for trial_t in [t, 0.6 * t, 0.3 * t, 0.1 * t, 0.05 * t]:
                cand = (1 - trial_t) * a + trial_t * g
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-10:
                    continue
                cand = cand / cs
                sc_c = compute_score(cand)
                if sc_c < best_local_sc:
                    best_local_sc = sc_c
                    best_local_a = cand.copy()

            a = best_local_a
            cur_sc = best_local_sc

            if cur_sc < best_sc - 1e-9:
                best_sc = cur_sc
                best_a = a.copy()
                no_improve = 0
                # Increase step if improving
                t = min(0.8, t * 1.02)
            else:
                no_improve += 1
                t = max(0.1, t * 0.95)

            if verbose and it % 200 == 0:
                print(f"    [lp] it={it}, score={best_sc:.6f}, t={t:.3f}")

        # Periodic restart from best
        if no_improve > 40:
            a = project_simplex(best_a + 0.15 * rng.standard_normal(n))
            t = 0.4 + 0.3 * rng.random()
            no_improve = 0

        it += 1

    if verbose:
        print(f"    [lp] done: it={it}, best={best_sc:.6f}")
    return best_a, best_sc


def gradient_polish(a, time_budget, rng, verbose=False):
    """Polish using smoothed subgradient descent with momentum."""
    n = a.size
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    eta = 0.4
    tau_frac = 0.04
    vel = np.zeros(n)
    beta = 0.7
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        tau = tau_frac * M
        if tau < 1e-18:
            tau = 1e-18

        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws

        g = 2.0 * np.correlate(w, a, mode='valid')
        vel = beta * vel + (1 - beta) * g
        a_new = project_simplex(a - eta * vel)
        sc_new = compute_score(a_new)

        if sc_new < best_sc - 1e-10:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = project_simplex(a - 0.5 * eta * vel)
            no_improve += 1

        if no_improve > 300:
            a = project_simplex(best_a + 0.08 * rng.standard_normal(n))

            eta = 0.2 + 0.4 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()
            beta = 0.5 + 0.3 * rng.random()
            no_improve = 0

        it += 1

        if verbose and it % 500 == 0:
            print(f"    [grad] it={it}, score={best_sc:.6f}")

    return best_a, best_sc

        init_gaussian(n),

        init_sinc(n, rng),
        init_random(n, rng),
        init_u_shape(n),
        init_power_law(n, rng),
        init_periodic(n, rng),
    ]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iterative refinement with adaptive steps, gradient polishing,
    diverse initializations across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_overall = None
    best_overall_sc = np.inf

    # Load previous best
    prev_a = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = normalize(h)
            prev_sc = compute_score(prev_a)
            print(f"[init] Previous: n={h.size}, score={prev_sc:.6f}")
            if prev_sc < best_overall_sc:
                best_overall = prev_a.copy()
                best_overall_sc = prev_sc
    except Exception:
        pass

    # N values to try
    n_list = [64, 128, 256, 512]
    if prev_a is not None and prev_a.size not in n_list and 32 <= prev_a.size <= 2048:
        n_list.append(prev_a.size)
    n_list = sorted(set(n_list))

    remaining = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, time={remaining:.1f}s")

    # Time allocation: weighted toward smaller n (faster LP) but give larger n a chance
    raw_w = [1.0 / (0.5 + nv / 200.0) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * (deadline - time.monotonic()) * 0.85
        if time_n < 30:
            continue

        inits = get_initializations(nv, rng)
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # Score and rank initializations
        scored = []
        for a_init in inits:
            a_n = normalize(a_init)
            sc = compute_score(a_n)
            scored.append((sc, a_n))
        scored.sort(key=lambda x: x[0])

        # Try top 4-5 inits
        n_try = min(len(scored), 5)
        time_per = time_n / n_try

        print(f"[n={nv}] top inits: {[f'{s:.4f}' for s, _ in scored[:n_try]]}")

        for ii in range(n_try):
            if time.monotonic() > deadline:
                break

            sc_i, a_i = scored[ii]
            tb = min(time_per, deadline - time.monotonic() - 10.0)
            if tb < 15:
                break

            # LP refinement (75% of time)
            a_lp, sc_lp = lp_refine(a_i, tb * 0.75, rng, verbose=(ii == 0 and nv <= 256))

            # Gradient polish (25% of time)
            a_gp, sc_gp = gradient_polish(a_lp, tb * 0.25, rng)

            sc_best = min(sc_lp, sc_gp)
            a_best = a_lp if sc_lp <= sc_gp else a_gp

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [n={nv}, init={ii}] *** NEW BEST: {sc_best:.6f} ***")
            else:
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f}")

    # Final polish
    if best_overall is not None and time.monotonic() < deadline - 15.0:
        remain = deadline - time.monotonic() - 10.0
        if remain > 30:
            print(f"[polish] remain={remain:.1f}s, n={best_overall.size}")

            # Gradient polish
            a_p, sc_p = gradient_polish(best_overall, remain * 0.4, rng, verbose=False)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] grad: {sc_p:.6f}")

            # LP polish
            if time.monotonic() < deadline - 15.0:
                remain2 = deadline - time.monotonic() - 10.0
                if remain2 > 20:
                    a_p2, sc_p2 = lp_refine(best_overall, remain2, rng, verbose=False)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"[polish] lp: {sc_p2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128, dtype=np.float64)
        best_overall_sc = compute_score(best_overall)

    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, eval={final_sc:.6f}")
    return result
