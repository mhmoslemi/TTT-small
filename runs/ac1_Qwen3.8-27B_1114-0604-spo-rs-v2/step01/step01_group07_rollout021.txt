Strategy:

import numpy as np

from scipy.optimize import minimize

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _make_multi_harmonic(n, amps, n_harmonics=3):
    """Create a multi-harmonic anti-palindromic sequence."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64) / float(n)
    for k in range(1, n_harmonics + 1):

            a += amps[k-1] * np.sin(k * theta) / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s

    """Score function for Nelder-Mead optimization."""
    amps = x[:n_harmonics]
    a = _make_multi_harmonic(n, amps, n_harmonics)
    return _internal_score(a)

def _optimize_harmonics(n, n_harmonics=3, deadline=None, x0=None):

Multi-harmonic anti-palindromic sequence optimization with extended
active-set mirror descent for minimizing the peak-to-average convolution ratio.

Strategy:
1. Generate multi-harmonic anti-palindromic candidates (sinusoidal basis)
   and optimize amplitues via Nelder-Mead for several lengths.
2. Refine the best candidate using active-set mirror descent on the
   non-negative simplex with adaptive learning rate and random restarts.
3. Validate and return the best sequence found.

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence logic."""
    arr = np.asarray(a, dtype=np.float64)

    s = float(arr.sum())

        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

    Create an anti-palindromic multi-harmonic sequence of length n.
    a_i = (1 + sum_k amps[k] * sin(2*pi*(k+1)*(i+0.5)/n)) / n, clipped >= 0.
    """

    return a / s

        g = g / gn

        # Multiplicative (mirror) update
        exp_arg = np.clip(-lr * g, -3.0, 3.0)
        arr_new = arr * np.exp(exp_arg)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, None)
        ns = float(arr_new.sum())
        if not np.isfinite(ns) or ns < 0.01:
            lr = max(min_lr, lr * 0.5)
            continue
        arr_new = arr_new / ns

        # Evaluate and adapt
        if it % eval_period == 0:
            sc = _internal_score(arr_new)
            if np.isfinite(sc) and sc < best_score:
                best = arr_new.copy()
                best_score = sc
                arr = arr_new
                stall = 0
                lr = min(max_lr, lr * 1.04)
            else:
                stall += 1
                if sc > best_score * 1.002:
                    # Bad step: revert and reduce lr
                    arr = best.copy()
                    lr = max(min_lr, lr * 0.75)
                else:
                    # Neutral step: keep going, slight lr increase
                    arr = arr_new
                    lr = min(max_lr, lr * 1.01)

                if stall > max_stall:
                    # Random perturbation to escape local minimum
                    rng = np.random.default_rng((it * 7919) & 0x7FFFFFFF)
                    pert = rng.standard_normal(n) * 0.003
                    arr_p = arr + pert
                    arr_p = np.clip(arr_p, 0.0, None)
                    ps = float(arr_p.sum())
                    if np.isfinite(ps) and ps > 0.01:
                        arr = arr_p / ps
                    else:
                        arr = best.copy()
                    lr = max(min_lr, lr * 0.5)
                    stall = 0

    if verbose:
        print(f"    [md] done: it={it} best_score={best_score:.8f} time={time.monotonic()-t_start:.1f}s")

    return best, best_score


def _coarse_sine_scan(n, deadline):
    """Quick 1D scan over sinusoidal amplitude for a given n."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    sine = np.sin(theta)
    base = 1.0 / float(n)

    best_sc = np.inf
    best_amp = 0.78

    # Coarse grid
    for amp in np.linspace(0.0, 1.4, 35):
        if time.monotonic() > deadline:
            break
        a = base * (1.0 + amp * sine)
        a = np.clip(a, 0.0, None)
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_amp = float(amp)

    # Local refinement
    if np.isfinite(best_sc):
        step = 1.4 / 34.0
        for _ in range(4):
            lo = max(0.0, best_amp - step)
            hi = min(1.4, best_amp + step)
            if hi - lo < 1e-8:
                break
            for amp in np.linspace(lo, hi, 15):
                if time.monotonic() > deadline:
                    break
                a = base * (1.0 + amp * sine)
                a = np.clip(a, 0.0, None)
                sc = _internal_score(a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_amp = float(amp)
            step *= 0.5

    return best_amp, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low-scoring non-negative sequence.
    Returns a list of floats.
    """
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 25.0

    print(f"[start] budget={total_budget:.0f}s")

    best_seq = None
    best_sc = np.inf

    # --- Phase 1: Multi-harmonic parameter search ---
    print("[phase1] Coarse sinusoidal scan + multi-harmonic Nelder-Mead")

    # Quick sinusoidal scan for good initial amplitude
    n_init = 2048
    init_deadline = min(t0 + 30.0, hard_deadline - 500.0)
    init_amp, init_sc = _coarse_sine_scan(n_init, init_deadline)
    print(f"  coarse scan: amp={init_amp:.4f} score={init_sc:.8f}")

    n_values = [4096, 2048, 8192]
    n_harm_max = 3

    for n in n_values:
        if time.monotonic() > hard_deadline - 1200.0:
            break

        # 1 harmonic
        x0_1 = np.array([init_amp])
        amps1, sc1 = _optimize_amplitudes(n, 1, x0_1, max_iter=200)
        a1 = _make_anti_palindromic(n, amps1)
        sc1 = _internal_score(a1)
        if sc1 < best_sc:
            best_sc = sc1
            best_seq = a1.copy()
            print(f"  n={n} K=1: score={sc1:.8f} A1={amps1[0]:.4f}")

        # 2 harmonics
        x0_2 = np.array([init_amp, 0.15])
        amps2, sc2 = _optimize_amplitudes(n, 2, x0_2, max_iter=300)
        a2 = _make_anti_palindromic(n, amps2)
        sc2 = _internal_score(a2)
        if sc2 < best_sc:
            best_sc = sc2
            best_seq = a2.copy()
            print(f"  n={n} K=2: score={sc2:.8f} A={amps2}")

        # 3 harmonics
        x0_3 = np.array([init_amp, 0.12, 0.06])
        amps3, sc3 = _optimize_amplitudes(n, 3, x0_3, max_iter=400)
        a3 = _make_anti_palindromic(n, amps3)
        sc3 = _internal_score(a3)
        if sc3 < best_sc:
            best_sc = sc3
            best_seq = a3.copy()
            print(f"  n={n} K=3: score={sc3:.8f} A={amps3}")

        # Try a few more initializations for 2 harmonics
        for a2_init in [0.1, -0.1, 0.25, -0.25]:
            if time.monotonic() > hard_deadline - 1000.0:
                break
            x0_2b = np.array([init_amp, a2_init])
            amps2b, sc2b = _optimize_amplitudes(n, 2, x0_2b, max_iter=250)
            a2b = _make_anti_palindromic(n, amps2b)
            sc2b = _internal_score(a2b)
            if sc2b < best_sc:
                best_sc = sc2b
                best_seq = a2b.copy()
                print(f"  n={n} K=2 (init {a2_init}): score={sc2b:.8f} A={amps2b}")

    print(f"[phase1] best internal score: {best_sc:.8f}")

    if best_seq is None:
        best_seq = _make_anti_palindromic(2048, np.array([init_amp, 0.0, 0.0]))
        best_sc = _internal_score(best_seq)

    # --- Phase 2: Mirror descent refinement ---
    md_budget = max(20.0, hard_deadline - time.monotonic() - 60.0)
    if md_budget > 30.0:
        print(f"[phase2] Mirror descent refinement, budget={md_budget:.0f}s")

        # Run on the best candidate
        refined, ref_sc = _mirror_descent(best_seq, md_budget)
        if np.isfinite(ref_sc) and ref_sc < best_sc:
            best_seq = refined
            best_sc = ref_sc
            print(f"  [md] improved: {best_sc:.8f}")
        else:
            print(f"  [md] no improvement, keeping {best_sc:.8f}")

        # If time remains, try mirror descent on a different n
        remaining = hard_deadline - time.monotonic() - 50.0
        if remaining > 120.0:
            alt_n = 4096 if best_seq.size != 4096 else 8192
            alt_seq = _make_anti_palindromic(alt_n, np.array([init_amp, 0.1, 0.05]))
            alt_sc = _internal_score(alt_seq)
            print(f"  [alt] trying n={alt_n} init score={alt_sc:.8f}")
            alt_refined, alt_ref_sc = _mirror_descent(alt_seq, remaining - 20.0)
            if np.isfinite(alt_ref_sc) and alt_ref_sc < best_sc:
                best_seq = alt_refined
                best_sc = alt_ref_sc
                print(f"  [alt] improved: {best_sc:.8f}")
            else:
                print(f"  [alt] no improvement ({alt_ref_sc:.8f})")

    # --- Phase 3: Final validation ---
    print(f"[phase3] Final validation")
    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        arr = np.ones(2048, dtype=np.float64) / 2048.0
        s = 1.0
    arr = arr / s

    try:
        real_score = float(evaluate_sequence(arr.tolist()))
        print(f"[FINAL] actual score = {real_score:.8f}, n = {arr.size}, time = {time.monotonic()-t0:.1f}s")
    except Exception as e:
        print(f"[ERROR during validation] {e}")
        real_score = np.inf

    return [float(x) for x in arr]
