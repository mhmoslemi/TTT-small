def fft_convolution(a):

    if n == 0:
        return np.array([])
    need = 2*n - 1
    size = 1
    while size < need:
        size *= 2
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0)

def compute_convolve_grad(a, active_set):

    theta = (cssv[rho] - target) / (rho + 1.0)

def fft_cross_convolution(a, b):

"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolution(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fft_cross_convolution(a, b):
    """Linear cross-convolution a*b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)[:need]
    return np.maximum(result, 0.0)


def cross_correlation_fft(v, f, size):
    """Compute R[j] = sum_k v[k] * f[k-j] for j=0..n-1 using FFT."""
    V = np.fft.rfft(v, size)
    F = np.fft.rfft(f, size)
    R = np.fft.irfft(np.conj(V) * F, size)
    n = len(f)
    return R[:n]


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 1e-15:
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    return p / s


def fast_score(a):
    """Compute 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return float('inf')
    b = fft_convolution(a)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def flat_spectrum_init(n, rng, n_iter=300, blend=0.4, target_power=None):
    """
    Initialize a sequence with approximately flat magnitude spectrum.
    Iteratively replaces the magnitude in frequency domain.
    """
    x = np.linspace(0, 1, n)
    # Start with a U-shape (large at edges, small in middle)
    a = 0.3 + 0.7 * np.abs(2 * x - 1) ** 0.5
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    if target_power is None:
        target_power = 1.0 / np.sqrt(2 * n - 1)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        new_mag = mag.copy()
        # Blend toward flat target for non-DC bins
        new_mag[1:] = blend * target_power + (1 - blend) * mag[1:]
        # Keep DC
        new_mag[0] = mag[0]
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
    
    return a


def u_shape_init(n, rng, power=1.0):
    """U-shaped initialization: large at edges, small in middle."""
    x = np.linspace(0, 1, n)
    # U-shape: (1 - sin(pi*x))^power gives large at edges
    a = (1.0 - np.sin(np.pi * x)) ** power + 0.01
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def diverse_starts(n, rng, count=12):
    """Generate diverse starting sequences."""
    starts = []
    
    # U-shapes with different powers
    for p in (0.5, 1.0, 2.0, 3.0):
        starts.append(u_shape_init(n, rng, power=p))
    
    # Flat-spectrum inits
    starts.append(flat_spectrum_init(n, rng, n_iter=200, blend=0.3))
    starts.append(flat_spectrum_init(n, rng, n_iter=400, blend=0.5))
    starts.append(flat_spectrum_init(n, rng, n_iter=200, blend=0.7))
    
    # Random smooth
    size = 1
    while size < n:
        size <<= 1
    for _ in range(3):
        z = rng.normal(size=size // 2 + 1)
        cutoff = size // 8
        z[cutoff:] = rng.normal(size=size // 2 + 1 - cutoff) * 0.1
        a = np.fft.irfft(z * np.exp(1j * rng.uniform(0, 2 * np.pi, size // 2 + 1)), size)[:n]
        a = np.maximum(a, 1e-10)
        starts.append(a / a.sum())
    
    # Symmetric U-shape with noise
    base = u_shape_init(n, rng, power=1.5)
    noise = rng.normal(0, 0.1, n)
    a = np.maximum(base + noise, 1e-10)
    a = (a + a[::-1]) / 2  # enforce symmetry
    starts.append(a / a.sum())
    
    return starts[:count]


def best_response_lp_approx(f, M, n_iter=1000, eta0=None, deadline=None, label=""):
    """
    Approximate LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M.
    Uses projected gradient with adaptive step size.
    Returns g with sum = sum(f).
    """
    n = len(f)
    S = float(np.sum(f))
    f_norm = f / S
    M_norm = M / (S * S)
    
    g = f_norm.copy()
    if eta0 is None:
        eta = 1.0 / n
    else:
        eta = eta0
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    best_g = g.copy()
    best_violation = float('inf')
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        fg = fft_cross_convolution(f_norm, g)
        v = np.maximum(fg - M_norm, 0.0)
        violation = float(np.sum(v))
        
        if violation < best_violation:
            best_violation = violation
            best_g = g.copy()
        
        if violation < 1e-12:
            break
        
        # Gradient of 0.5*sum(v^2) w.r.t. g
        grad = cross_correlation_fft(v, f_norm, size)
        grad = np.maximum(grad, 0.0)
        
        g_new = g - eta * grad
        g_new = np.maximum(g_new, 0.0)
        s = float(g_new.sum())
        if s < 1e-15:
            g_new = f_norm.copy()
        else:
            g_new = g_new / s
        
        # Adapt step size
        fg_new = fft_cross_convolution(f_norm, g_new)
        v_new = np.maximum(fg_new - M_norm, 0.0)
        violation_new = float(np.sum(v_new))
        
        if violation_new > 2.0 * violation + 1e-10:
            eta *= 0.5
            g = best_g.copy()  # Revert
        elif violation_new < 0.7 * violation and eta < 1.0 / n * 10:
            eta *= 1.3
            g = g_new
        else:
            if violation_new < violation:
                g = g_new
            # else keep g
        
        if it % 200 == 0:
            print(f"  BR {label} it={it} viol={violation:.3e} eta={eta:.2e}")
    
    return best_g * S


def fista_subgradient(a0, deadline, epsilon=0.1, label="", print_every=500):
    """
    FISTA with exact subgradient for min max(a*a) s.t. a>=0, sum(a)=1.
    """
    n = len(a0)
    a = normalize_simplex(a0)
    y = a.copy()
    t = 1.0
    step = 2.0 / n  # Initial step
    
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(fft_convolution(a)))
    
    size = 1
    while size < 4 * n:
        size <<= 1
    
    it = 0
    no_improve = 0
    last_restart = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        b = fft_convolution(y)
        M = float(np.max(b))
        threshold = M * (1.0 - epsilon)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average of gradients over active set
        m = len(active)
        h = np.zeros(2 * n - 1)
        h[active] = 1.0 / m
        g = cross_correlation_fft(h, y, size)
        g = 2.0 * g
        
        # FISTA step
        a_new = project_simplex(y - step * g, 1.0)
        
        # Track best
        b_new = fft_convolution(a_new)
        M_new = float(np.max(b_new))
        score = 2.0 * n * M_new
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % print_every == 0:
            print(f"  FISTA {label} it={it} best={best_score:.8f} step={step:.2e} |A|={m} no_imp={no_improve}")
        
        # Adapt step
        if no_improve > 500:
            step *= 0.5
            no_improve = 0
        elif no_improve == 0 and step < 10.0:
            step = min(10.0, step * 1.1)
        
        # Periodic restart
        if it - last_restart > 3000:
            y = a_new.copy()
            t = 1.0
            last_restart = it
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y_new = a_new + ((t - 1.0) / t_new) * (a_new - a)
            y_new = np.maximum(y_new, 0.0)
            sy = float(y_new.sum())
            if sy < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / sy
            y = y_new
            t = t_new
        
        a = a_new
    
    return best_a, best_score


def smooth_max_beta(b, beta):
    """Smooth max via log-sum-exp."""
    m = float(np.max(b))
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300:
        return m, np.zeros_like(b)
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w


def full_projected_gradient_smooth(a0, deadline, beta=500000, label="", print_every=500):
    """
    Full projected gradient with smooth max (softmax) approximation.
    Uses backtracking line search.

    n = len(a0)

            no_improve += 1
        
        if it % print_every == 0:
            print(f"  PG {label} it={it} best={best_score:.8f} step={step:.2e} no_imp={no_improve}")
        
        if no_improve > 5000:
            break
        
        # Gradient of smooth max
        svec = cross_correlation_fft(w, p, 4 * n if 4 * n >= 2 * n else 2 * n)
        # Actually need proper size
        size = 1
        while size < 4 * n:
            size <<= 1
        svec = cross_correlation_fft(w, p, size)
        g = 2.0 * np.maximum(svec, 0.0)
        grad = g - float(np.mean(g))
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Backtracking
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult / gnorm
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            G2, _ = smooth_max_beta(qb, beta)
            if n * G2 < cur_smooth - 1e-12:
                p = q
                accepted = True
                break
        
        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            if step < 5.0:
                step = min(5.0, step * 1.2)
    
    return best_p, best_score


def best_response_iteration(f0, deadline, n_br_iters=50, label=""):
    """
    Run the best-response iteration:
    f <- (1-t)*f + t*best_response(f)
    """
    n = len(f0)
    f = normalize_simplex(f0)
    t_blend = 0.5  # Blending parameter
    eta = 1.0 / n  # Step size for LP approximation
    
    best_f = f.copy()
    best_score = fast_score(f)
    
    outer_it = 0
    while time.monotonic() < deadline:
        outer_it += 1
        b = fft_convolution(f)
        M = float(np.max(b))
        S = float(np.sum(f))
        
        # Solve LP approximation
        g = best_response_lp_approx(f, M, n_iter=n_br_iters, eta0=eta, deadline=deadline, label=f"{label}-{outer_it}")
        
        # Blend
        new_f = (1 - t_blend) * f + t_blend * g
        new_f = normalize_simplex(new_f)
        
        sc = fast_score(new_f)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = new_f.copy()
            f = new_f
            # Increase blend slightly
            t_blend = min(0.8, t_blend * 1.05)
        else:
            t_blend = max(0.2, t_blend * 0.8)
            f = new_f  # Still update f (it might help escape)
        
        if outer_it % 5 == 0:
            print(f"  BR-iter {label} outer={outer_it} score={best_score:.8f} t={t_blend:.3f}")
        
        # Decrease blend over time for convergence
        if outer_it % 20 == 0:
            t_blend = max(0.1, t_blend * 0.9)
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function returning a list of nonnegative floats minimizing
    2*n*max(a*a)/(sum(a))^2.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 0:
        return [1.0]
    
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    
    N = 8192
    
    reserve = 60.0
    deadline = start_time + budget_s - reserve
    
    print(f"Starting search: N={N}, budget={budget_s}s, seed={seed}")
    
    # Generate diverse starting points
    print("Generating starts...")
    starts = diverse_starts(N, rng, count=12)
    
    scored = []
    for i, s in enumerate(starts):
        sc = fast_score(s)
        scored.append((sc, i, s))
        print(f"  Start {i}: score={sc:.8f}")
    
    scored.sort(key=lambda x: x[0])
    
    best_a = scored[0][2].copy()
    best_score = scored[0][0]
    print(f"Best initial: {best_score:.8f}")
    
    # Phase 1: Best-response iteration on top starts
    remaining = deadline - time.monotonic()
    if remaining > 60:
        top_k = min(3, len(scored))
        allocs = (0.45, 0.30, 0.25)
        
        for i in range(top_k):
            if time.monotonic() > deadline - 30:
                break
            frac = allocs[i] if i < len(allocs) else 0.2
            alloc = remaining * frac
            sub_dl = time.monotonic() + alloc
            
            print(f"\nPhase 1: Best-response on start {scored[i][1]} (score={scored[i][0]:.6f}), alloc={alloc:.0f}s")
            a_br, sc_br = best_response_iteration(scored[i][2], sub_dl, n_br_iters=30, label=f"br{i}")
            
            if sc_br < best_score:
                best_score = sc_br
                best_a = a_br.copy()
                print(f"  New best after BR: {best_score:.8f}")
            
            remaining = deadline - time.monotonic()
    
    # Phase 2: FISTA refinement
    if time.monotonic() < deadline - 30:
        alloc = (deadline - time.monotonic()) * 0.5
        sub_dl = time.monotonic() + alloc
        print(f"\nPhase 2: FISTA refinement, alloc={alloc:.0f}s")
        a_fista, sc_fista = fista_subgradient(best_a, sub_dl, epsilon=0.1, label="main")
        
        if sc_fista < best_score:
            best_score = sc_fista
            best_a = a_fista.copy()
            print(f"  New best after FISTA: {best_score:.8f}")
    
    # Phase 3: Smooth max projected gradient from best so far
    if time.monotonic() < deadline - 20:
        remaining2 = deadline - time.monotonic() - 10
        alloc = remaining2 * 0.6
        sub_dl = time.monotonic() + alloc
        print(f"\nPhase 3: Smooth PG refinement, alloc={alloc:.0f}s")
        a_pg, sc_pg = full_projected_gradient_smooth(best_a, sub_dl, beta=500000, label="pg")
        
        if sc_pg < best_score:
            best_score = sc_pg
            best_a = a_pg.copy()
            print(f"  New best after PG: {best_score:.8f}")
    
    # Phase 4: Another FISTA pass with smaller epsilon (more focused)
    if time.monotonic() < deadline - 10:
        remaining3 = deadline - time.monotonic() - 5
        alloc = remaining3 * 0.7
        sub_dl = time.monotonic() + alloc
        print(f"\nPhase 4: FISTA (epsilon=0.03), alloc={alloc:.0f}s")
        a_f2, sc_f2 = fista_subgradient(best_a, sub_dl, epsilon=0.03, label="f2")
        
        if sc_f2 < best_score:
            best_score = sc_f2
            best_a = a_f2.copy()
            print(f"  New best after FISTA2: {best_score:.8f}")
    
    # Final normalization and check
    best_a = np.maximum(best_a, 0.0)
    s = float(np.sum(best_a))
    if s > 0:
        best_a = best_a / s
    else:
        best_a = np.ones(N) / N
    
    # Cap at 1000 to be safe
    best_a = np.minimum(best_a, 1000.0)
    
    final_seq = [float(x) for x in best_a]
    
    # Verify with official evaluator
    try:
        off = evaluate_sequence(final_seq)
        print(f"\nFinal official score: {off:.8f}")
    except Exception as e:
        print(f"Official eval error: {e}")
        off = float('inf')
    
    if not np.isfinite(off) or off > 2.0:
        print("Falling back to uniform")
        return [1.0] * N
    
    return final_seq
