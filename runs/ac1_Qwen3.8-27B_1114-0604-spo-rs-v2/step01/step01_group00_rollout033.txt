import numpy as np

def lp_expansion(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)
    # Build constraint matrix C (2n-1 x n)
    # C[k, i] = a[i] if 0 <= k-i < n, else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            C[k, i] = a[i]

    return a  # fallback

Here's the "greedy expansion" idea:
1. Start with b = a (which is feasible: conv(a,a) <= M).
2. For each position j (in some order), increase b[j] as much as possible without violating any constraint conv(a, b)[k] <= M.
3. The amount we can increase b[j] is limited by: for all k, a[j] * b[j] (contribution to conv(a,b)[k] from position j) must not push conv(a,b)[k] over M.
   Specifically, the constraint is: sum_i a[i]*b[k-i] <= M.
   If we increase b[j] by δ, the change in conv(a,b)[k] is a[k-j]*δ (if 0 <= k-j < n).
   So we need: for all k with 0 <= k-j < n: a[k-j]*δ <= M - conv(a,b)[k].
   The maximum δ is: min over k of (M - conv(a,b)[k]) / a[k-j].
4. Add this δ to b[j].
5. Repeat for all j.

This is one "pass" of the greedy algorithm. We can do multiple passes.

The cost per pass is O(n * (2n-1)) = O(n^2) in the worst case, but in practice we can use the fact that conv(a,b) can be maintained incrementally.

Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent with adaptive temperature, exact subgradient refinement,
and coordinate descent polish. Multiple diverse restarts.
"""

from scipy.signal import correlate
import time

def _conv_fft(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def _eval(f):
    return evaluate_sequence(f.tolist())

def _soft_gradient(c, a, T):
    """Softmax-weighted gradient of max(conv)."""
    n = len(a)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= s
    return 2.0 * correlate(p, a, mode='valid')

def _exact_gradient(c, a):
    """Exact subgradient of max(conv) at a."""
    n = len(a)
    # Find peak lags (within 1% of max)
    c_max = c.max()
    # Use all lags that are exactly at the max (or very close)
    peak_mask = c >= c_max * (1 - 1e-10)
    p = peak_mask.astype(np.float64)

    return 2.0 * correlate(p, a, mode='valid')
