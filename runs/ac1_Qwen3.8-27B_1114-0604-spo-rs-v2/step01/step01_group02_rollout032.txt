"""
Hybrid optimizer: log-sum-exp smoothed gradient (Adam) + L-BFGS-B polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 for non-negative sequences.
Uses FFT-based convolution and gradient computation for efficiency.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mfft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mfft)
    return np.fft.irfft(A * A, Mfft)[:M]

def smoothed_obj_and_grad(a, tau):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    c = fast_self_conv(a)
    # Stable log-sum-exp
    c_max = float(c.max())
    shifted = tau * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    L = c_max + np.log(sum_exp) / tau
    w = exp_shifted / sum_exp  # softmax weights, length 2n-1
    
    # R[j] = sum_m a[m] * w[m+j]
    # = (a_rev * w)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1  # = 3n-2
    Mfft2 = 1 << (conv_len - 1).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, Mfft2)
    W_fft = np.fft.rfft(w, Mfft2)
    conv_result = np.fft.irfft(A_rev_fft * W_fft, Mfft2)[:conv_len]
    R = conv_result[n-1 : 2*n-1]  # length n
    
    F = 2.0 * n * L / (S * S)
    grad = 4.0 * n * R / (S * S) - 4.0 * n * L * a / (S * S * S)
    
    return F, grad

def exact_obj(a):
    """Exact (unsmoothed) objective."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam step with projection."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad

    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 1e-6:
        a += 0.1
    return a

"""
Multi-strategy optimizer to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses: (1) Log-sum-exp smoothed Adam with adaptive temperature, (2) L-BFGS-B polish,
(3) LP-based subgradient iteration. FFT-based gradient computation throughout.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    Mfft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mfft)
    return np.fft.irfft(A * A, Mfft)[:M]


def exact_obj(a):
    """Exact objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def smoothed_obj_grad(a, tau):
    """
    Log-sum-exp smoothed objective and gradient.
    F = 2n * (1/tau) * log(sum exp(tau*c)) / S^2
    Grad computed via FFT cross-correlation.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf, np.zeros(n)

    c = fast_self_conv(a)  # length 2n-1

    # Numerically stable log-sum-exp
    c_max = float(c.max())
    shifted = tau * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    if sum_exp < 1e-300:
        return np.inf, np.zeros(n)
    L = c_max + np.log(sum_exp) / tau
    w = exp_shifted / sum_exp  # softmax weights, length 2n-1

    # Compute R[j] = sum_m a[m] * w[m+j] for j=0..n-1
    # R[j] = (a_rev * w_linear)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # 3n-2
    Mfft2 = 1 << (conv_len - 1).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, Mfft2)
    W_fft = np.fft.rfft(w, Mfft2)
    conv_result = np.fft.irfft(A_rev_fft * W_fft, Mfft2)[:conv_len]
    R = conv_result[n - 1:2 * n - 1]  # length n

    # F = 2n * L / S^2
    F = 2.0 * n * L / (S * S)
    # grad = 4n*R/S^2 - 4n*L*a/S^3
    grad = 4.0 * n * R / (S * S) - 4.0 * n * L * a / (S * S * S)

    return F, grad


def adam_update(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update with non-negativity projection."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - t1)
    v_hat = v / (1.0 - t2)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 1e-8:
        a += 0.01
    return a

        try:
            F, _ = smoothed_obj_grad(x, tau)
            if np.isnan(F) or np.isinf(F):
                return 1e10
            return F
        except Exception:
            return 1e10

    def grad(x):
        try:
            _, g = smoothed_obj_grad(x, tau)
            if np.any(np.isnan(g)):
                return np.zeros(len(x))
            return g
        except Exception:
            return np.zeros(len(x))

    bounds = [(0.0, 1000.0)] * n
    res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14,
                            'gtol': 1e-12, 'maxls': 50})
    a_opt = np.clip(res.x, 0.0, 1000.0)
    sc = exact_obj(a_opt)
    return sc, a_opt


def build_lp_matrix(a):
    """
    Build the (2n-1) x n constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_start:k - j_end + 1][::-1]
    return A


def lp_step(a, M):
    """
    One LP iteration: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the normalized update direction g (with same sum as a) or None.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-6:
        return None

    A = build_lp_matrix(a)
    c_obj = np.ones(n)  # maximize sum(b)
    # linprog minimizes, so negate
    c_neg = -c_obj

    # Constraints: A @ b <= M (vector of M's)
    b_ub = np.full(2 * n - 1, M)

    bounds = [(0.0, 1000.0)] * n

    res = linprog(c_neg, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None

    b_star = res.x
    sum_b = float(b_star.sum())
    if sum_b < 1e-10:
        return None

    # Normalize to match sum of a
    g = b_star * (S / sum_b)
    return g


def optimize_lp(a0, end_time, step_size=0.3, seed=42):
    """
    LP-based iteration: repeatedly solve LP and take a step.
    Returns (best_score, best_a).
    """
    rng = np.random.default_rng(seed)
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.1

    best_score = exact_obj(a)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        M = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # Perturb and continue
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            no_improve = 0
            max_no_improve = max(20, max_no_improve // 2)
            if max_no_improve <= 20:
                break
            continue

        g = lp_step(a, M)
        if g is None:
            break

        # Update with step size
        a_new = (1.0 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0.0)
        a_new = np.minimum(a_new, 1000.0)

        # Check if peak decreased; if not, try smaller step
        sc_new = exact_obj(a_new)
        if sc_new > sc * 1.01:
            step_size *= 0.5
            if step_size < 0.001:
                break
            a_new = (1.0 - step_size) * a + step_size * g
            a_new = np.maximum(a_new, 0.0)
        else:
            step_size = min(0.5, step_size * 1.1)
            a = a_new

    return best_score, best_a


def init_uniform(n):
    return np.ones(n)


def init_parent(n):
    """Get parent sequence if available."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size >= n:
            return np.clip(arr[:n], 0.0, 1000.0)
        elif arr.size > 0:
            out = np.ones(n) * 0.1
            out[:arr.size] = arr
            return np.clip(out, 0.0, 1000.0)
    except Exception:
        pass
    return None


def init_cosine(n):
    x = np.linspace(0, n - 1, n)
    a = 0.5 + 0.5 * np.cos(np.pi * (x - n / 2.0) / (n / 2.0 + 1.0))
    return np.clip(a, 0.01, 1000.0)


def init_gaussian(n):
    x = np.linspace(-1.0, 1.0, n)
    return np.exp(-3.0 * x * x)


def init_rand_smooth(n, rng):
    coeffs = rng.standard_normal(n)
    f = np.fft.rfft(coeffs)
    k = max(2, n // 100)
    mask = np.zeros_like(f)
    mask[:k] = 1.0
    f *= mask
    a = np.fft.irfft(f, n)
    a = np.abs(a) + 0.1
    return a


def init_block_symmetric(n, rng):
    """Random symmetric blocks."""
    half = n // 2
    nb = max(2, half // 8)
    per = half // nb
    vals = 0.3 + 0.7 * rng.random(nb)
    a_half = np.repeat(vals, per)
    if len(a_half) < half:
        a_half = np.concatenate([a_half, vals[:half - len(a_half)]])
    a_half = a_half[:half]
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, [a_half[half // 2]], a_half[::-1]])
    return np.clip(a[:n], 0.01, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Uses a hybrid of Adam (log-sum-exp), L-BFGS-B, and LP-based methods.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = float(budget_s)
    deadline = t_start + total - 15.0

    # Determine n
    n = 2048
    parent_arr = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0:
            n = p.size
            parent_arr = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    n = int(min(max(512, n), 8192))

    best_score = np.inf
    best_a = None

    def consider(score, arr, label):
        nonlocal best_score, best_a
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = arr.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {score:.8f} (t={elapsed:.1f}s)")

    print(f"Starting search: n={n}, budget={total:.0f}s")

    # ============================================================
    # Phase 1: LP-based refinement from parent (most promising start)
    # ============================================================
    lp_deadline = t_start + total * 0.30
    if parent_arr is not None and len(parent_arr) == n:
        consider(exact_obj(parent_arr), parent_arr, "parent_init")
        sc, aa = optimize_lp(parent_arr, lp_deadline, step_size=0.3, seed=seed)
        consider(sc, aa, "LP_from_parent")
    else:
        a0 = init_uniform(n)
        sc, aa = optimize_lp(a0, lp_deadline, step_size=0.3, seed=seed)
        consider(sc, aa, "LP_uniform")

    # Also try LP from a few other starts if time remains
    other_lp_end = min(lp_deadline + total * 0.1, deadline - total * 0.3)
    for tag, a0 in [("cos", init_cosine(n)), ("gauss", init_gaussian(n))]:
        if time.monotonic() > other_lp_end:
            break
        sc, aa = optimize_lp(a0, other_lp_end, step_size=0.2, seed=seed + 10)
        consider(sc, aa, f"LP_{tag}")

    # ============================================================
    # Phase 2: Adam + log-sum-exp from best and other initializations
    # ============================================================
    adam_deadline = t_start + total * 0.70
    inits = []
    if parent_arr is not None and len(parent_arr) == n:
        inits.append(("parent", parent_arr.copy()))
    inits.append(("cos", init_cosine(n)))
    inits.append(("gauss", init_gaussian(n)))
    inits.append(("rand", init_rand_smooth(n, rng)))
    inits.append(("block", init_block_symmetric(n, rng)))

    if best_a is not None:
        inits.insert(0, ("best_so_far", best_a.copy()))

    n_inits = len(inits)
    time_per_adam = max(3.0, (adam_deadline - time.monotonic()) / n_inits)

    for idx, (tag, a0) in enumerate(inits):
        if time.monotonic() > adam_deadline:
            break
        a0 = np.clip(a0, 0.0, 1000.0)
        if a0.sum() < 1e-6:
            a0 = np.ones(n) * 0.1
        sc0 = exact_obj(a0)
        consider(sc0, a0, f"{tag}_init")

        # Adam with annealing
        sc, aa = optimize_adam(a0, adam_deadline,
                               tau_start=3.0, tau_end=2000.0,
                               lr_init=0.08, seed=seed + idx,
                               rescale_interval=100)
        # But cap time per init
        consider(sc, aa, f"adam_{tag}")
        # Reset deadline for next: we've used our share
        adam_deadline = t_start + total * 0.70 - time_per_adam * (n_inits - idx - 1)
        adam_deadline = max(adam_deadline, time.monotonic() + 2.0)

    # ============================================================
    # Phase 3: L-BFGS-B polish at various tau values
    # ============================================================
    lbfgs_deadline = t_start + total * 0.90
    if best_a is not None and time.monotonic() < lbfgs_deadline:
        for tau in [50, 100, 300, 1000]:
            if time.monotonic() > lbfgs_deadline:
                break
            sc, aa = optimize_lbfgs(best_a, tau=tau, maxiter=200)
            consider(sc, aa, f"lbfgs_tau{tau}")
        # One more round from the new best
        if best_a is not None:
            sc, aa = optimize_lbfgs(best_a, tau=500, maxiter=300)
            consider(sc, aa, "lbfgs_final")

    # ============================================================
    # Phase 4: Final short Adam polish from the global best
    # ============================================================
    if best_a is not None and time.monotonic() < deadline:
        sc, aa = optimize_adam(best_a, deadline,
                               tau_start=500.0, tau_end=5000.0,
                               lr_init=0.005, seed=seed + 200,
                               rescale_interval=50)
        consider(sc, aa, "final_adam")

    # ============================================================
    # Safety and output
    # ============================================================
    if best_a is None:
        best_a = np.ones(n)

    out = [float(x) for x in best_a]
    verified = evaluate_sequence(out)
    elapsed = time.monotonic() - t_start
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(out)}, time={elapsed:.1f}s")

    return out
