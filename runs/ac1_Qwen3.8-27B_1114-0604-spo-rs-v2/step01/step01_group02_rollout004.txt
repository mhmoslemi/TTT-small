Multi-phase optimizer combining L-BFGS-B, Adam, and spectral methods
to minimize the peak autocorrelation ratio of non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    Nf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Nf)
    return np.fft.irfft(A * A, Nf)[:M]

def obj_val(a):
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10
    return 2.0 * n * float(c.max()) / (S * S)

def soft_obj(a, k, temp):
    """Smooth objective using top-k soft-max approximation."""
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    # Top-k soft-max
    if k >= len(c):
        idx = np.arange(len(c))
        vals = c
    else:
        idx = np.argpartition(c, -k)[-k:]
        vals = c[idx]
    
    top = float(vals.max())
    if top < 1e-12:
        wts = np.ones(len(idx)) / len(idx)
    else:
        wts = np.exp((vals - top) / temp)
        wts /= wts.sum()
    
    M = float(np.dot(wts, vals))
    
    # Gradient of M w.r.t. a
    grad_M = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:

        if j_lo > j_hi:

    grad = 2.0 * n * (grad_M / (S * S) - 2.0 * M / (S ** 3))

    vh = v / (1 - b2 ** t)

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a += 0.01 / len(a)
    return a

def run_lbfgsb(a0, duration, k=30, temp=0.05):
    """L-BFGS-B optimization with smooth objective."""
    n = len(a0)
    a0 = clip_a(a0.astype(np.float64).copy())
    
    def fun(x):
        c = fast_conv(x)
        S = x.sum()
        if S < 0.01:
            return 1e10
        if k >= len(c):
            idx = np.arange(len(c))
            vals = c
        else:
            idx = np.argpartition(c, -k)[-k:]
            vals = c[idx]
        top = float(vals.max())
        if top < 1e-12:
            M = 0.0
        else:
            wts = np.exp((vals - top) / temp)
            wts /= wts.sum()
            M = float(np.dot(wts, vals))
        return 2.0 * n * M / (S * S)
    
    def fun_grad(x):
        c = fast_conv(x)
        S = x.sum()
        if S < 0.01:
            return 1e10, np.zeros(n)
        if k >= len(c):
            idx = np.arange(len(c))
            vals = c
        else:
            idx = np.argpartition(c, -k)[-k:]
            vals = c[idx]
        top = float(vals.max())
        if top > 1e-12:
            wts = np.exp((vals - top) / temp)
            wts /= wts.sum()
        else:
            wts = np.ones(len(idx)) / len(idx)
        M = float(np.dot(wts, vals))
        
        grad_M = np.zeros(n)
        for i in range(len(idx)):
            ki = int(idx[i])
            wi = float(wts[i])
            if wi < 1e-15:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(ki, n - 1)
            if j_lo > j_hi:
                continue
            grad_M[j_lo:j_hi+1] += wi * 2.0 * x[j_lo:j_hi+1][::-1]
        
        g = 2.0 * n * (grad_M / (S * S) - 2.0 * M / (S ** 3))
        val = 2.0 * n * M / (S * S)
        return val, g
    
    # Run L-BFGS-B for a time budget
    # Since we can't easily control the time inside scipy,
    # we'll run it with a maxiter that's reasonable
    t0 = time.monotonic()
    result = sp_minimize(fun_grad, a0, method='L-BFGS-B',
                         bounds=[(0, 1000)] * n,
                         options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    sc = obj_val(result.x)
    return sc, clip_a(result.x)
