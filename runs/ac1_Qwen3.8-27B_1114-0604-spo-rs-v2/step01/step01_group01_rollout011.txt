```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequence a.

Algorithm:
1. Generate diverse initial sequences (analytical chirp seeds + resampled parent).
2. For each promising initial:
   a. Spectral variance minimization (smooth, fast convergence).
   b. Peak-directed multiplicative descent with adaptive softmax temperature.
   c. Coordinate descent with exact 1D line search (final polish).
3. Multi-length exploration (try several n values).
4. Return the best sequence found.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))


def _spectral_variance_grad(p):
    """
    Gradient of sum_{j>=1} (|A[j]|^2 - target)^2 with respect to p,
    where A = FFT(p). Returns the gradient vector (same shape as p).
    We want to minimize this, which flattens the spectrum.
    """
    n = len(p)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(p, N)
    mag2 = np.abs(A[:N // 2 + 1]) ** 2
    # target: mean of non-DC magnitudes
    if N // 2 + 1 > 1:
        target = float(np.mean(mag2[1:]))
    else:
        target = mag2[0]
    diff = mag2 - target
    diff[0] = 0.0  # Don't force DC
    # grad of sum_j diff[j]^2 * 2 * |A[j]|^2 ... 
    # Actually: d(|A[j]|^2)/dp[k] = 2*Re(conj(A[j]) * exp(-2*pi*i*j*k/N))
    # So grad = sum_j 2*diff[j] * 2*Re(conj(A[j]) * exp(-2*pi*i*j*k/N))
    # = 4 * Re(IFFT(diff * conj(A)))  [with appropriate indexing]
    # Let's compute: B[j] = 4 * diff[j] * conj(A[j]), then grad[k] = Re(IFFT(B))
    B = np.zeros(N // 2 + 1, dtype=complex)
    B[:len(diff)] = 4.0 * diff * np.conj(A[:len(diff)])
    # Need full spectrum for IFFT
    B_full = np.zeros(N, dtype=complex)
    B_full[:N // 2 + 1] = B
    if N // 2 + 1 < N:
        B_full[N // 2 + 1:] = np.conj(B_full[N // 2:0:-1][::-1])
    # Simpler: use rfft convention
    grad_full = np.fft.irfft(B * 2, N)  # factor of 2 from autograd of |A|^2
    grad = grad_full[:n].copy()
    return grad


def _spectral_descent(p0, rng, time_limit):
    """
    Gradient descent on spectral variance (smooth objective).
    Works on probability simplex.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    deadline = time.monotonic() + time_limit
    best = p.copy()
    best_score = _exact_score(p)
    lr = 0.01
    stall = 0

    while time.monotonic() < deadline:
        g = _spectral_variance_grad(p)
        # Project gradient to simplex: subtract mean
        g = g - np.mean(g)
        # Step
        p_new = p - lr * g
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()
        
        sc = _exact_score(p_new)
        if sc < best_score:
            best = p_new.copy()
            best_score = sc
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            lr *= 0.8
            if stall > 20:
                # Restart from best with perturbation
                p = best * np.exp(rng.normal(0, 0.02, n))
                p = np.maximum(p, 1e-15)
                p = p / p.sum()
                lr = 0.005
                stall = 0
        p = p_new

    return best, best_score

    it = 0

        else:

                smooth_try = float(np.dot(w, d_try))

                if smooth_try < best_smooth - 1e-12:
                    # Check exact improvement
                    if exact_try < best_score or stall > 0:
                        chosen = p_try
                        best_smooth = smooth_try
                        break
            
            if chosen is not None:
                p = chosen
                sc = _exact_score(p)
                if sc < best_score:
                    best = p.copy()
                    best_score = sc
                stall = 0
                alpha = min(alpha * 1.1, 5.0)
            else:

                    elif stall > 30:

                        alpha = 0.15
                    stall = 0
                if stall > 60:
                    p = best.copy()
                    best_score = _exact_score(p)
                    alpha = 0.15
                    beta = max(beta, beta_start)

        it += 1

def _coord_descent_1d(p0, rng, time_limit):
    """
    Coordinate descent: for each coordinate, find optimal 1D step
    that minimizes the max autocorrelation value.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    if n < 4:
        return p, _exact_score(p)
    
    deadline = time.monotonic() + time_limit
    best = p.copy()
    best_score = _exact_score(p)
    floor = 1e-14
    c = _conv_full(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    cur_max = float(np.max(c))
    
    order = list(range(n))
    stagnant = 0

    while time.monotonic() < deadline:
        # Random.shuffle for diversity
        rp = rng.permutation(n)
        improved = 0
        for i in rp:
            if time.monotonic() > deadline - 2:
                    break
            # Compute gradient direction for coordinate i
            # delta_c[j] = a[j-i] + a[i-j] for the linear part
            g = np.zeros(len(c))
            # g[j] = p[j-i] if 0<=j-i<n, plus p[i-j] if 0<=i-j<n
            j_min = max(0, i)
            j_max = min(len(c) - 1, i + n - 1)
            for j in range(j_min, j_max + 1):
                gj = 0.0
                if 0 <= j - i < n:
                    gj += p[j - i]
                if 0 <= i - j < n:
                    gj += p[i - j]
                g[j] = gj
            # Also the quadratic term: if j == 2*i, there's a delta^2 term
            has_quad = (2 * i < len(c))
            
            # Find delta that minimizes max_j (c[j] + delta*g[j] + delta^2*has_quad*(j==2i))
            # Binary search on T
            # For convergence, try a few candidate deltas
            best_delta = 0.0
            best_T = cur_max
            
            # Try various deltas
            v = p[i]
            candidates = [-0.5*v, -0.2*v, -0.05*v, 0.05*v, 0.2*v, 0.5*v, 0.0]
            # Also try to push towards reducing the max
            peak_j = int(np.argmax(c))
            if g[peak_j] > 0:
                candidates.append(-0.3 * v * g[peak_j] / (np.max(g) + 1e-15))
                candidates.append(-0.5 * v * g[peak_j] / (np.max(g) + 1e-15))
            else:
                candidates.append(0.3 * v * abs(g[peak_j]) / (np.max(g) + 1e-15))
            
            for delta in candidates:
                new_v = v + delta
                if new_v < floor:
                    continue
                # Scale to keep sum = 1: scale all other elements
                # Actually, let's just change p[i] and renormalize
                p[i] = new_v
                s = p.sum()
                if s < 1e-10:
                    p[i] = v
                    break
                p_scaled = p / s
                c_new = _conv_full(p_scaled, p_scaled)
                c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                T = float(np.max(c_new))
                if T < best_T - 1e-14:
                    best_T = T
                    best_delta = delta
                    best_c = c_new
                    best_p = p_scaled.copy()
            
            p[i] = v  # reset
            if best_delta != 0.0:
                p = best_p
                c = best_c
                cur_max = best_T
                improved += 1
                if cur_max < best_score / (2.0 * n):
                    best_score = 2.0 * n * cur_max
                    best = p.copy()
        
        if improved == 0:
            stagnant += 1
            if stagnant > 5:
                # Perturb
                p = _sanitize(best * np.exp(rng.normal(0, 0.02, n)))
                if p is not None:
                    p = np.maximum(p, floor)
                    p = p / p.sum()
                    c = _conv_full(p, p)
                    c = np.where(np.isfinite(c), c, 0.0)
                    cur_max = float(np.max(c))
                stagnant = 0
        else:
            stagnant = 0

    return best, best_score


def _chirp_seed(n, freq=1.0, phase=0.0, window='raised_cosine', depth=0.5):
    """Generate a chirp-like seed sequence."""
    t = (np.arange(n) + 0.5) / n
    # Chirp: quadratic phase
    phase_arr = 2.0 * np.pi * freq * t ** 2 + phase
    base = 1.0 + depth * np.cos(phase_arr)
    # Window
    if window == 'raised_cosine':
        w = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window == 'gaussian':
        w = np.exp(-((t - 0.5) / 0.25) ** 2)
    else:
        w = np.ones(n)
    p = base * w
    p = np.maximum(p, 0.0)
    p = p + 1e-10
    return p / p.sum()


def _bump_seed(n, rng):
    """Generate a smooth bump seed."""
    K = int(rng.integers(2, 6))
    p = np.zeros(n)
    t = (np.arange(n) + 0.5) / n
    for _ in range(K):
        c = float(rng.uniform(0.1, 0.9))
        s = float(rng.uniform(0.05, 0.3))
        w = float(rng.uniform(0.5, 2.0))
        p += w * np.exp(-((t - c) / s) ** 2)
    p = p + 1e-10
    return p / p.sum()


def _resample(arr, m):
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Returns a list of floats.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10.0  # Leave 10s buffer

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 7)

    best_arr = None
    best_score = float('inf')

    # Try to get parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        parent_arr = None

    print(f"Parent available: {parent_arr is not None}")
    if parent_arr is not None:
        ps = _exact_score(parent_arr)
        print(f"Parent score: {ps:.6f}, length: {len(parent_arr)}")
        best_arr = parent_arr.copy()
        best_score = ps

    # Lengths to try
    lengths_to_try = [2048, 3072, 4096, 6144]
    
    # For each length, generate starts and optimize
    for n in lengths_to_try:
        if time.monotonic() > deadline - 300:
            break
        
        n_rem = deadline - time.monotonic()
        if n_rem < 200:
            break
            
        # Generate diverse starts for this length
        starts = []
        
        # From parent
        if parent_arr is not None:
            sp = _resample(parent_arr, n)
            if sp is not None:
                starts.append(sp)
            # Perturbed parent
            sp2 = _resample(parent_arr, n)
            if sp2 is not None:
                sp2 = _sanitize(sp2 * np.exp(rng.normal(0, 0.05, n)))
                if sp2 is not None:
                    starts.append(sp2)
        
        # Chirp seeds
        for freq in [0.5, 1.0, 1.5, 2.0]:
            for depth in [0.3, 0.5, 0.7]:
                for win in ['raised_cosine', 'gaussian']:
                    cs = _chirp_seed(n, freq=freq, depth=depth, window=win)
                    starts.append(cs)
        
        # Bump seeds
        for _ in range(4):
            starts.append(_bump_seed(n, rng))
        
        # Evaluate and sort starts
        scored = []
        for s in starts:
            if s is None:
                continue
            sc = _exact_score(s)
            if np.isfinite(sc):
                scored.append((sc, s))
        
        scored.sort(key=lambda x: x[0])
        print(f"n={n}: best start score = {scored[0][0]:.6f}")
        
        # Take top starts (at most 4)
        top_starts = [s for _, s in scored[:4]]
        
        # Allocate time: give more time to better starts
        t_total = min(n_rem - 100, 2400.0)
        fracs = [0.35, 0.25, 0.20, 0.20]
        
        for idx, sp in enumerate(top_starts):
            if time.monotonic() > deadline - 150:
                break
            frac = fracs[idx] if idx < len(fracs) else 0.1
            t_alloc = t_total * frac
            t_alloc = min(t_alloc, max(30, (deadline - time.monotonic()) - 150))
            if t_alloc < 20:
                break
            
            # Phase A: Spectral descent (short, fast)
            t_spec = min(30.0, 0.15 * t_alloc)
            if t_spec > 10:
                pa, sca = _spectral_descent(sp, rng, t_spec)
                if pa is not None and sca < (best_score if np.isfinite(best_score) else float('inf')):
                    best_arr = pa.copy()
                    best_score = sca
                    print(f"  n={n} start{idx} spectral: {sca:.6f}")
                if pa is not None:
                    sp = pa
            
            # Phase B: Peak multiplicative descent (main work)
            t_mm = 0.5 * t_alloc
            pb, scb = _peak_multiplicative_descent(
                sp, rng, t_mm,
                beta_start=50.0, beta_end=5000.0
            )
            if pb is not None and scb < best_score:
                best_arr = pb.copy()
                best_score = scb
                print(f"  n={n} start{idx} multiplicative: {scb:.6f}")
            if pb is not None:
                sp = pb
            
            # Phase C: Coordinate descent polish
            t_cd = min(0.25 * t_alloc, (deadline - time.monotonic()) - 100)
            if t_cd > 15:
                pc, scc = _coord_descent_1d(sp, rng, t_cd)
                if pc is not None and scc < best_score:
                    best_arr = pc.copy()
                    best_score = scc
                    print(f"  n={n} start{idx} coord: {scc:.6f}")
                if pc is not None:
                    sp = pc

    # Final polish on the best
    if best_arr is not None and time.monotonic() < deadline - 60:
        n_best = len(best_arr)
        t_rem = deadline - time.monotonic() - 20
        
        # Another round of multiplicative descent with higher beta
        t1 = min(120.0, 0.4 * t_rem)
        if t1 > 20:
            p1, s1 = _peak_multiplicative_descent(
                best_arr, rng, t1,
                beta_start=200.0, beta_end=10000.0
            )
            if p1 is not None and s1 < best_score:
                best_arr = p1.copy()
                best_score = s1
                print(f"Final multiplicative: {best_score:.6f}")
        
        # Coordinate descent
        t2 = min(120.0, 0.4 * (deadline - time.monotonic() - 20))
        if t2 > 15:
            p2, s2 = _coord_descent_1d(best_arr, rng, t2)
            if p2 is not None and s2 < best_score:
                best_arr = p2.copy()
                best_score = s2
                print(f"Final coord: {best_score:.6f}")
        
        # Tiny perturbations and re-evaluate
        t3 = deadline - time.monotonic() - 10
        if t3 > 5:
            for _ in range(min(20, int(t3 / 2))):
                if time.monotonic() > deadline - 10:
                    break
                p_try = _sanitize(best_arr * np.exp(rng.normal(0, 0.005, len(best_arr))))
                if p_try is not None:
                    s_try = _exact_score(p_try)
                    if s_try < best_score:
                        best_arr = p_try
                        best_score = s_try
                        print(f"  Perturb improved: {best_score:.6f}")

    # Final verification
    if best_arr is None:
        best_arr = np.ones(3072, dtype=float) / 3072
        best_score = _exact_score(best_arr)

    best_arr = _sanitize(best_arr)
    if best_arr is None:
        best_arr = np.ones(3072, dtype=float) / 3072

    # Verify with official evaluator
    final_list = [float(x) for x in best_arr]
    try:
        official = float(evaluate_sequence(final_list))
        if np.isfinite(official) and official < best_score:
            best_score = official
    except Exception:
        pass

    print(f"FINAL: score={best_score:.6f}, length={len(best_arr)}")
    return final_list
```
