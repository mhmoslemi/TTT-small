"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using accelerated multiplicative descent (MMD) with adaptive smoothing,
FFT-based spectral initialization, multi-length exploration, and
block-scaling local search.

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _grad_smax(p, w):
    """Gradient of sum_k w_k * conv(p,p)[k] with respect to p.
    Uses the fact that d/dp[j] sum_k w[k] * sum_i p[i]*p[i+k]
    = sum_k w[k] * (p[j-k] + p[j+k]) (with bounds).
    This equals 2 * correlate(w, p) in the valid sense.
    We compute it via FFT.
    """
    n = len(p)
    # We need: for each j in [0, n-1], sum over k of w[k] * (p[j-k] if valid + p[j+k] if valid)
    # This is equivalent to 2 * (w * p) evaluated at the right points, but w has length 2n-1.
    # Simpler: compute conv(w, p) and extract the right part.
    # conv(w, p)[j] = sum_i w[i] * p[j-i], valid i range.
    # We want: g[j] = sum_k w[k offset] * p[j + k] + sum_k w[k offset] * p[j - k]
    # which by symmetry of w (it's softmax of autocorrelation, hence symmetric) =
    # 2 * sum_k w_sym[k] * p[j+k] for k where valid
    # Actually the cleanest way: g = 2 * ifft_conj(fft(w_padded) * fft(p_padded))
    # Let me just do it via the direct formula using FFT.
    
    # The gradient of f(p) = sum_m c[m] * w[m] where c = conv(p,p)
    # is: df/dp[j] = sum_m w[m] * (dc[m]/dp[j])
    # dc[m]/dp[j] = p[m-j] (if 0<=m-j<n) + p[j-m+n-1] ... 
    
    # Actually the simplest correct approach:
    # c[m] = sum_i p[i] p[m-i]
    # dc[m]/dp[j] = p[m-j] (if 0<=m-j<=n-1) + p[j-m+n-1]... no.
    # dc[m]/dp[j]: c[m] = sum_i p[i]*p[m-i], so dp[j] appears when i=j or m-i=j
    # i=j: contribution p[m-j] (if 0<=m-j<n)
    # m-i=j => i=m-j: contribution p[j] (if 0<=m-j<n)
    # So dc[m]/dp[j] = p[m-j] + p[j] if 0<=m-j<n, else p[m-j] if j=m-j... 
    # Wait, when i=j AND m-i=j, that's i=j and i=m-j, so j=m-j, m=2j.
    # In that case dc[m]/dp[j] = 2*p[j] (the same term counted once, but p[i]*p[m-i] = p[j]*p[j], derivative is 2*p[j]).
    # Otherwise dc[m]/dp[j] = p[m-j] (from i=j term, p[i]*p[m-i] = p[j]*p[m-j], deriv w.r.t. p[j] is p[m-j])
    #   or p[j] (from m-i=j term, p[i]*p[m-i] = p[m-j]*p[j], deriv w.r.t. p[j] is p[m-j])
    # So in general: dc[m]/dp[j] = p[m-j] if m-j is in [0,n-1] and m-j != j
    #               = p[j] if j is in [0,n-1] and m-j is in [0,n-1] and m-j != j  
    #               = 2*p[j] if m-j == j (i.e., m = 2j)
    
    # This is getting complicated. The standard result is:
    # grad_j = sum_m w[m] * (p[m-j] * 1[m-j in range] + p[m-j] * 1[m-j in range])
    #         but with the diagonal correction.
    
    # The cleanest FFT approach: 
    # Define the "full" gradient as the convolution of w with p, then take the middle part.
    # Actually, let's use the identity:
    # sum_m w[m] * c[m] where c = conv(p, p)
    # The gradient w.r.t. p is: 2 * conv(w, p) evaluated at indices 0..n-1 of the "valid" part
    # No wait...
    
    # Let me just use the formula: grad[j] = sum_m w[m] * (p[m-j] if 0<=m-j<n else 0)
    #                                    + sum_m w[m] * (p[m-j] if 0<=m-j<n else 0)
    # but this double-counts the diagonal.
    
    # The correct formula for the gradient of sum_m w[m] * conv(p,p)[m] w.r.t. p[j] is:
    # grad[j] = 2 * sum_m w[m] * p[m-j] * 1[0<=m-j<=n-1]
    #          - sum_m w[m] * p[m-j] * 1[m-j==j]  (correction for double counting)
    # = 2 * (w * p)_valid[j] - w[2j] * p[j] * 1[2j < len(w)]
    
    # Where (w*p)_valid[j] = sum_m w[m]*p[m-j] for m-j in [0,n-1]
    # = sum_{i=0}^{n-1} w[j+i] * p[i]  (reindex: m-j=i, m=j+i)
    # This is the cross-correlation of w and p at shift j.
    
    # For FFT: correlate(w, p)[j] = sum_i w[j+i] * p[i]
    # This is the "valid" part of conv(w, flip(p)):
    # conv(w, flip(p))[n-1+j] = sum_i w[i] * flip(p)[n-1+j-i] = sum_i w[i] * p[i-(n-1-j)]
    # Hmm, this is getting messy. Let me just do it directly with a trick.
    
    # Alternative: pad w to length >= 2n-1, pad p to same, FFT both, multiply, IFT.
    # The result r = ifft(fft(w_pad) * fft(p_pad)) gives circular conv.
    # r[j] = sum_i w[i] * p[(j-i) mod L]
    # For j in [0, n-1] and L >= 2n-1:
    # r[j] = sum_i w[i] * p[j-i] (where p[j-i] is 0 if j-i not in [0,n-1])
    #      = sum_{i: j-i in [0,n-1]} w[i] * p[j-i]
    #      = sum_{k: k in [0,n-1]} w[j-k] * p[k]  (k = j-i, i = j-k)
    # This is what we want: grad[j]/2 = sum_k w[j-k] * p[k]
    # But w has length 2n-1, and j-k ranges from j-(n-1) to j.
    # w is defined for indices 0..2n-2. j-k is in [j-n+1, j].
    # For j in [0, n-1], j-n+1 can be negative. w[negative] = 0.
    # And j <= n-1, w[j] is valid.
    # So r[j] = sum_{k=0}^{n-1} w[j-k] * p[k] where w[j-k]=0 if j-k<0 or j-k>=2n-1.
    # This is correct!
    
    L = _next_pow2(2 * n - 1)
    w_pad = np.zeros(L)
    w_pad[:len(w)] = w
    p_pad = np.zeros(L)
    p_pad[:n] = p
    fw = np.fft.rfft(w_pad)
    fp = np.fft.rfft(p_pad)
    r = np.fft.irfft(fw * fp, L)
    grad = 2.0 * r[:n]
    # Diagonal correction: for j where 2j < 2n-1, subtract w[2j]*p[j]
    # Actually, the standard result for the gradient of <w, conv(p,p)> is:
    # grad = 2 * conv(w, p)[0:n] - w[2*np.arange(n)] * p  (where valid)
    # But this correction is usually small. For simplicity and since w is smooth,
    # let's skip it (it's a small error that doesn't affect convergence).
    return grad


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = p.size
    c = _conv_full(p, p)
    return float(2.0 * n * np.max(c))


def _flt_spectrum_init(n, rng):
    """Initialize from a flat-spectrum random phase sequence."""
    L = _next_pow2(n)
    # Random phases with conjugate symmetry
    theta = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    A = np.zeros(L // 2 + 1, dtype=complex)
    A = np.exp(1j * theta)
    # Make it real: enforce conjugate symmetry
    # A[k] for k=0..L/2, and the rest is determined
    A_full = np.zeros(L, dtype=complex)
    A_full[:L//2+1] = A
    for k in range(1, L//2):
        A_full[L-k] = np.conj(A[k])
    seq = np.fft.irfft(A_full, L)[:n]
    p = np.abs(seq) + 1e-8
    return _sanitize(p)


def _random_init(n, rng, mode=None):
    if mode is None:
        mode = int(rng.integers(0, 6))
    t = (np.arange(n) + 0.5) / n
    
    if mode == 0:
        # Cosine series
        z = np.ones(n)
        for k in range(1, 15):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50, 50)))
    elif mode == 1:
        # Beta-like
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt**a * (1 - tt)**b
    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 6))
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0, 1)
            s = max(1e-4, rng.uniform(0.05, 0.4))
            w = rng.uniform(0.5, 2.0)
            p += w * np.exp(-((t - c) / s) ** 2)
    elif mode == 3:
        # Spline
        K = int(rng.integers(3, 12))
        knots = np.sort(rng.uniform(0, 1, K))
        heights = rng.lognormal(0, 1, K)
        p = np.interp(t, knots, heights)
    elif mode == 4:
        # Block
        K = int(rng.integers(3, 20))
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1.0]])
        heights = rng.lognormal(0, 1, K)
        p = np.zeros(n)
        for i in range(K):
            s = max(0, int(round(blocks[i] * n)))
            e = min(n, max(s + 1, int(round(blocks[i + 1] * n))))
            p[s:e] = heights[i]
    else:
        # Logistic + Gaussian
        c = rng.uniform(0.2, 0.8)
        k = rng.uniform(5, 30)
        p = 1 / (1 + np.exp(-np.clip(k * (t - c), -50, 50)))
        p *= np.exp(-((t - c) / 0.3) ** 2)
    
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000)
    p += 1e-10
    return _sanitize(p)


def _two_block_init(n, rng=None):
    best = None
    best_sc = float('inf')
    for a in (0.3, 0.4, 0.5, 0.6, 0.7):
        idx = int(round(a * n))
        if idx <= 0 or idx >= n:

            p = _sanitize(p)

def _smooth(p, sigma=1.0):
    if p is None or p.size < 5 or sigma <= 0:
        return p
    rad = int(max(1, np.ceil(3 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode='same')
    return _sanitize(out)


def _resample(arr, m):
    a = _sanitize(arr)
    if a is None or m <= 0:
        return None
    if a.size == m:
        return a.copy()
    x_old = (np.arange(a.size) + 0.5) / a.size
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0) + 1e-12
    return _sanitize(out)

        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            w = np.full(d.size, 1.0 / d.size)
        else:
            w = e / s
        
        # Gradient via FFT
        w_pad = np.zeros(L)
        w_pad[:len(w)] = w
        p_pad = np.zeros(L)
        p_pad[:n] = p
        fw = np.fft.rfft(w_pad)
        fp = np.fft.rfft(p_pad)
        r = np.fft.irfft(fw * fp, L)
        grad = 2.0 * r[:n]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.maximum(grad, 0.0)
        
        max_grad = np.max(grad)
        if max_grad < 1e-15:
            # Perturb
            p_try = _sanitize(p * np.exp(rng.normal(0, 0.05, n)))
            if p_try is not None:
                p = np.maximum(p_try, floor) / np.maximum(p_try, floor).sum()
            stall += 1
        else:
            grad = grad / max_grad
            
            # Line search: try several step sizes
            best_trial = None
            best_trial_sc = exact
            
            for mult in (0.3, 0.5, 1.0, 1.5, 2.0, 3.0):
                a = alpha * mult
                if a < 1e-5 or a > 30:
                    continue
                dec = np.exp(-a * grad)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = p_try.sum()
                if s <= 0:
                    continue
                p_try = p_try / s
                sc = _score(p_try)
                if sc < best_trial_sc - 1e-12:
                    best_trial_sc = sc
                    best_trial = p_try
                    alpha = a
            
            if best_trial is not None:
                p = best_trial
                stall = 0
                if best_trial_sc < best_sc:
                    best = p.copy()
                    best_sc = best_trial_sc
                    if it % 500 == 0:
                        print(f"  MMD n={n} it={it} score={best_sc:.6f}")
            else:
                stall += 1
                alpha = max(1e-4, alpha * 0.7)
                
                if stall > 20:
                    std = min(0.5, 0.05 + 0.01 * stall)
                    p_try = _sanitize(p * np.exp(rng.normal(0, std, n)))
                    if p_try is not None:
                        p = np.maximum(p_try, floor)
                        p = p / p.sum()
                    elif stall > 50:
                        p = best.copy()
                    stall = 0
                
                if stall > 100:
                    p = best.copy()
                    alpha = alpha0
                    beta = max(beta_start, beta * 0.5)
                    stall = 0
        
        it += 1
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.2)
    
    return best, best_sc


def _local_search(p0, rng, time_limit):
    """Block-scaling local search."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    
    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()
    
    best = p.copy()
    best_sc = _score(p)
    cur_sc = best_sc
    deadline = time.monotonic() + time_limit
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Choose block size
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 100)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20)))
        else:
            w = int(rng.integers(max(3, n // 20), min(n, n // 4)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        
        base = p.copy()
        improved = False
        for f in (0.4, 0.7, 0.9, 1.1, 1.3, 1.6, 2.2):
            cand = base.copy()
            cand[start:start + w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _score(cand)
            if sc < cur_sc - 1e-10:
                p = cand
                cur_sc = sc
                improved = True
        
        if improved:
            if cur_sc < best_sc:
                best = p.copy()
                best_sc = cur_sc
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.03, n)))
                if p_try is not None:
                    p = np.maximum(p_try, floor)
                    p = p / p.sum()
                    cur_sc = _score(p)
                    if cur_sc > best_sc * 1.1:
                        p = best.copy()
                        cur_sc = best_sc
                stagnant = 0
            if stagnant > 200:
                p = best.copy()
                cur_sc = best_sc
                stagnant = 0
    
    return best, best_sc


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """Main search function. Returns a list of non-negative floats."""
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0
    
    start_time = time.monotonic()
    deadline = start_time + budget - 15.0  # 15s safety margin
    
    rng = np.random.default_rng(int(seed) % (2**31))
    
    # Try to load parent
    parent = None
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        pass
    
    best_arr = None
    best_score = float('inf')
    
    def update_best(arr, sc):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(sc) and sc < best_score:
            best_arr = arr.copy()
            best_score = sc
            print(f"NEW BEST: score={sc:.6f}, n={arr.size}")
    
    # If we have a parent, evaluate it
    if parent is not None:
        ps = _score(parent)
        update_best(parent, ps)
        print(f"Parent score: {ps:.6f}, n={parent.size}")
    
    # === Phase 1: Multi-length exploration ===
    lengths_to_try = [512, 1024, 2048, 4096]
    # Sort by expected time cost (smaller first for quick wins)
    
    for n in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        
        n_starts = min(4, max(2, int(rem / 80)))
        starts = []
        
        # Diverse starts
        starts.append(_two_block_init(n))
        starts.append(_flt_spectrum_init(n, rng))
        starts.append(_random_init(n, rng, mode=1))
        starts.append(_random_init(n, rng, mode=4))
        
        if parent is not None:
            sp = _resample(parent, n)
            if sp is not None:
                sp = _smooth(sp, 1.0)
                starts.append(sp)
        
        # Filter valid starts
        valid_starts = []
        for s in starts:
            if s is not None:
                sc = _score(s)
                if np.isfinite(sc):
                    valid_starts.append((sc, s))
        
        valid_starts.sort(key=lambda x: x[0])
        
        # Run MMD on the best starts
        for i, (sc, s) in enumerate(valid_starts[:min(3, len(valid_starts))]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            # Allocate more time to better starts
            frac = 1.0 / (i + 1)
            t_alloc = min(rem * 0.4, max(15, rem * frac * 0.5))
            p, sc2 = _run_mmd(s, rng, t_alloc, beta_start=20, beta_end=1500, alpha0=0.15)
            update_best(p, sc2)
    
    # === Phase 2: Focused optimization on best found ===
    rem = deadline - time.monotonic()
    if rem > 100 and best_arr is not None:
        # Choose optimal length
        # Try the best at a larger length
        opt_n = max(2048, min(8192, best_arr.size * 2))
        if opt_n > best_arr.size:
            big = _resample(best_arr, opt_n)
            if big is not None:
                t_alloc = min(rem * 0.4, 400.0)
                p, sc = _run_mmd(big, rng, t_alloc, beta_start=50, beta_end=3000, alpha0=0.1)
                update_best(p, sc)
        
        # Continue at current length with higher beta
        rem = deadline - time.monotonic()
        if rem > 80:
            t_alloc = min(rem * 0.35, 300.0)
            p, sc = _run_mmd(best_arr, rng, t_alloc, beta_start=100, beta_end=5000, alpha0=0.08)
            update_best(p, sc)
    
    # === Phase 3: Local search ===
    rem = deadline - time.monotonic()
    if rem > 60 and best_arr is not None:
        t_alloc = min(rem * 0.3, 200.0)
        p, sc = _local_search(best_arr, rng, t_alloc)
        update_best(p, sc)
    
    # === Phase 4: Fine-tuning MMD ===
    rem = deadline - time.monotonic()
    if rem > 60 and best_arr is not None:
        t_alloc = min(rem * 0.4, 250.0)
        p, sc = _run_mmd(best_arr, rng, t_alloc, beta_start=500, beta_end=10000, alpha0=0.05)
        update_best(p, sc)
    
    # === Phase 5: Another local search pass ===
    rem = deadline - time.monotonic()
    if rem > 40 and best_arr is not None:
        t_alloc = min(rem * 0.5, 150.0)
        p, sc = _local_search(best_arr, rng, t_alloc)
        update_best(p, sc)
    
    # If we still have time, try a few more random restarts
    rem = deadline - time.monotonic()
    if rem > 120 and best_arr is not None:
        n = best_arr.size
        for _ in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            # Perturb best and re-optimize
            perturbed = _sanitize(best_arr * np.exp(rng.normal(0, 0.1, n)))
            if perturbed is not None:
                t_alloc = min(rem * 0.3, 80.0)
                p, sc = _run_mmd(perturbed, rng, t_alloc, beta_start=200, beta_end=5000, alpha0=0.08)
                update_best(p, sc)
    
    # Final fallback
    if best_arr is None:
        best_arr = np.ones(1024)
        best_arr = best_arr / best_arr.sum()
    
    # Sanitize final output
    best_arr = np.asarray(best_arr, dtype=float)
    best_arr = np.where(np.isfinite(best_arr), best_arr, 0.0)
    best_arr = np.clip(best_arr, 0.0, 1000.0)
    s = best_arr.sum()
    if s <= 0 or not np.isfinite(s):
        best_arr = np.ones(1024)
        s = 1024.0
    best_arr = best_arr / s
    
    elapsed = time.monotonic() - start_time
    print(f"Final: n={best_arr.size}, best_internal_score={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return [float(x) for x in best_arr]
