- Minimize `M`

The constraint `F(C) >= 0` is equivalent to `C` being a valid autocorrelation sequence (positive definite). This is a linear constraint in the time domain but infinite in the frequency domain (equivalent to the PSD being non-negative at all frequencies).

    while time.monotonic() - t0 < budget_s and it < max_iters:
        it += 1

        conv = _fft_conv(arr)
        if conv.size == 0:
            break

        # Softmax weights
        mx = float(np.max(conv))
        shifted = T * (conv - mx)
        shifted = np.clip(shifted, -60.0, 60.0)
        exp_s = np.exp(shifted)
        total = float(np.sum(exp_s))
        if total < 1e-30:
            break
        w = exp_s / total

        # Compute weighted gradient from active lags
        g = np.zeros(n, dtype=np.float64)
        threshold = 1e-6
        active = np.where(w > threshold)[0]
        for k_idx in active:
            k = int(k_idx)
            wk = float(w[k])
            g += wk * _grad_lag(k, arr)

        # Project to tangent space of simplex
        g = g - np.mean(g)

        if np.max(np.abs(g)) < 1e-18:
            break

        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 1e-30, 1000.0)
        s_new = float(np.sum(arr_new))
        if not np.isfinite(s_new) or s_new < 0.01:
            lr = max(min_lr, lr * 0.5)
            continue
        arr_new = arr_new / s_new

        # Evaluate
        score_new = _score(arr_new)

        if np.isfinite(score_new) and score_new < best_score:
            arr = arr_new
            best = arr_new.copy()
            best_score = score_new
            last_improve = it
            lr = min(max_lr, lr * 1.03)
        else:
            if np.isfinite(score_new) and score_new > best_score * 1.1:
                lr = max(min_lr, lr * 0.6)
            else:
                lr = max(min_lr, lr * 0.85)
            # Don't update arr (revert)
            # Actually, we should keep arr as is since we only update on improvement
            # But we need arr to be the current working point
            # Let's use a different strategy: always update arr, but keep best separately
            # Actually the above logic is: if improved, update both arr and best
            # If not, don't update arr (so it stays at the last improvement)
            # This is a "best-first" strategy which can get stuck
            # Better: always update arr, but only update best if improved
            arr = arr_new  # always move forward for exploration

        # Anneal temperature
        T = min(max_T, T * T_rate)

        # Stagnation check
        if it - last_improve > no_improve_limit:
            # Restart from best with perturbation
            arr = best.copy()
            noise = rng.dirichlet(np.ones(n) * 0.05)
            mix = rng.uniform(0.02, 0.1)
            arr = (1.0 - mix) * arr + mix * noise
            arr = np.clip(arr, 1e-30, None)
            arr = arr / float(arr.sum())
            lr = 1.0
            T = 1.0
            last_improve = it

        # Periodic: if we've been improving for a while, try a bigger perturbation
        if it % (restart_period * 4) == 0 and it - last_improve > restart_period:
            arr = best.copy()
            noise = rng.dirichlet(np.ones(n) * 0.1)
            mix = rng.uniform(0.05, 0.2)
            arr = (1.0 - mix) * arr + mix * noise
            arr = np.clip(arr, 1e-30, None)
            arr = arr / float(arr.sum())
            lr = 0.5
            T = 0.5
            last_improve = it

    return best, best_score


def _scan_and_refine(budget_s, seed=42):
    """
    Main search: scan initializations, refine the best ones.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0  # leave 20s for validation

    rng = np.random.RandomState(seed)

    candidates = []  # list of (score, array, name)

    # Try to use previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0, 1000)
            s = float(prev.sum())
            if s > 0.01:
                prev = prev / s
                sc = _score(prev)
                if np.isfinite(sc):
                    candidates.append((sc, prev, "prev_best"))
                    print(f"[init] prev_best: n={prev.size}, score={sc:.8f}")
    except (NameError, Exception):
        pass

    # Sinusoidal initializations
    lengths_to_try = [2048, 4096, 8192]

    for n in lengths_to_try:
        if time.monotonic() > deadline:
            break
        # Single mode
        for amp in [0.6, 0.75, 0.85, 0.95, 1.05]:
            a = _make_sine(n, amp, 1)
            if a is not None:
                sc = _score(a)
                if np.isfinite(sc):
                    candidates.append((sc, a, f"sine1_n{n}_a{amp}"))
        # Multi-mode
        for a1 in [0.6, 0.75, 0.85]:
            for a2 in [0.15, 0.25, 0.35]:
                a = _make_sine_multi(n, [a1, a2], [1, 2])
                if a is not None:
                    sc = _score(a)
                    if np.isfinite(sc):
                        candidates.append((sc, a, f"sine2_n{n}_a{a1}_{a2}"))

    # Anti-palindromic linear ramp
    for n in [2048, 4096]:
        i = np.arange(n, dtype=np.float64)
        a = (0.5 + 0.5 * np.sin(np.pi * (i + 0.5) / n))
        # Actually let's use a ramp: high on one side, low on the other
        a = (i + 1.0) / float(np.sum(np.arange(1, n + 1, dtype=np.float64)))
        sc = _score(a)
        if np.isfinite(sc):
            candidates.append((sc, a, f"ramp_n{n}"))

    # Random Dirichlet
    for n in [2048, 4096]:
        for _ in range(3):
            a = rng.dirichlet(np.ones(n) * 1.0)
            sc = _score(a)
            if np.isfinite(sc):
                candidates.append((sc, a, f"rand_n{n}"))

    # Sort by score
    candidates.sort(key=lambda x: x[0])

    if not candidates:
        # Fallback
        a = np.ones(2048, dtype=np.float64) / 2048.0
        return [float(x) for x in a]

    print(f"[scan] {len(candidates)} candidates, best score={candidates[0][0]:.8f} ({candidates[0][2]})")

    # Refine top candidates
    best_overall = None
    best_overall_score = np.inf

    n_to_refine = min(5, len(candidates))
    elapsed = time.monotonic() - t0
    remaining = max(10.0, deadline - time.monotonic())

    for idx in range(n_to_refine):
        sc, arr, name = candidates[idx]
        if time.monotonic() > deadline - 30:
            break

        # Allocate time: better candidates get more time
        if idx == 0:
            alloc = remaining * 0.5
        elif idx == 1:
            alloc = remaining * 0.25
        else:
            alloc = remaining * (0.25 / (n_to_refine - 1))

        alloc = max(10.0, alloc)
        seed_i = seed + idx * 1000

        print(f"[refine] {name} (score={sc:.8f}, n={arr.size}), budget={alloc:.0f}s")
        refined, ref_score = _refine(arr, alloc, seed=seed_i)

        if np.isfinite(ref_score) and ref_score < best_overall_score:
            best_overall_score = ref_score
            best_overall = refined
            print(f"[refine] improved: {name} -> score={ref_score:.8f}")
        else:
            print(f"[refine] no improvement: {name} -> score={ref_score:.8f}")

    if best_overall is None:
        best_overall = candidates[0][1]
        best_overall_score = candidates[0][0]

    return best_overall, best_overall_score


def _sanitize(x):
    """Sanitize before passing to evaluate_sequence."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    return arr
