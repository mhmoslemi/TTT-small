def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

    if not np.isfinite(s) or s < 0.01:
        a = np.ones(n, dtype=np.float64)

def _softmax_gradient_step(a, beta, lr):
    """
    One step of soft-max tempered gradient descent.
    a: current sequence (normalized, sum=1)
    beta: temperature parameter
    lr: learning rate
    Returns: updated sequence
    """
    n = a.size
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    
    # Compute tempered weights
    shifted = c - max_c  # <= 0, for numerical stability
    weights = np.exp(beta * shifted)
    total_w = float(np.sum(weights))
    if total_w < 1e-300:
        # Fallback to hard active set
        active = c >= (max_c - 1e-10)
        weights = active.astype(np.float64)
        total_w = float(np.sum(weights))
    weights /= total_w
    
    # Compute gradient: g[i] = sum_k weights[k] * 2*a[k-i] (where 0 <= k-i < n)
    g = np.zeros(n, dtype=np.float64)
    
    # Efficient computation using FFT or direct summation
    # g[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
    # Let j = k-i, so k = i+j, j in [0, n-1]
    # g[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j] (where i+j < 2n-1)
    # This is a cross-correlation of weights with a!
    
    # weights has length 2n-1, a has length n
    # cross-correlation: (weights * a)[i] = sum_j weights[i+j] * a[j]
    # This is the cross-correlation of weights and a
    
    # Use FFT for efficiency
    N = _next_pow2(2 * n)
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: we want sum_j weights[i+j] * a[j]
    # This is the cross-correlation of a and weights, shifted
    # Actually, let's think more carefully.
    # g[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]  (where i+j < 2n-1)
    # Let w[j] = weights[j] for j=0..2n-2, w[j]=0 otherwise
    # g[i]/2 = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is (cross-correlation of w and a) at lag i
    # cross_corr[i] = sum_j w[j+i] * a[j]
    # In frequency domain: cross_corr = ifft(fft(w) * conj(fft(a)))... 
    # Actually, cross-correlation x⊛y = ifft(conj(X) * Y) or ifft(X * conj(Y)) depending on convention.
    
    # Let me just use the direct approach for correctness, or compute via convolution.
    # sum_j w[i+j] * a[j] = sum_j w[i+j] * a[j]
    # If we reverse w: w_rev[k] = w[2n-2-k]
    # Then sum_j w[i+j] * a[j] = sum_j w_rev[2n-2-i-j] * a[j] = (a * w_rev)[2n-2-i]... 
    
    # Actually, let's just use the fact that this is a convolution:
    # Let b[j] = a[j], c[k] = weights[k]
    # sum_j c[i+j] * b[j] is the cross-correlation (c ⊛ b)[i]
    # = (b * c_rev)[i + (len(c)-1)] where c_rev is reversed c
    # Hmm, this is getting complicated. Let me just use a direct loop for the gradient
    # since it's O(n * num_active) and num_active is small.
    
    # Actually, for efficiency with large n, let me use the FFT approach properly.
    # sum_j w[i+j] * a[j] for j=0..n-1, i=0..n-1
    # = sum_j w_conv[j] * a[j] where w_conv[j] = w[i+j]
    # This is the cross-correlation of w and a evaluated at lags 0, 1, ..., n-1
    
    # Cross-correlation: (w ⊛ a)[i] = sum_j w[i+j] * conj(a[j]) = sum_j w[i+j] * a[j] (real)
    # In frequency domain: FFT(w ⊛ a) = conj(FFT(w)) * FFT(a)
    # Wait, that's not right. Let me use the standard definition.
    # (f ⊛ g)[n] = sum_m f[m+n] * conj(g[m])
    # FFT(f ⊛ g) = conj(FFT(f)) * FFT(g)
    
    # So: cross_corr = ifft(conj(fft(w, N)) * fft(a, N))
    # And we want cross_corr[0:n]
    
    W_full = np.fft.rfft(weights, N)
    A_full = np.fft.rfft(a, N)
    cross_corr = np.fft.irfft(np.conj(W_full) * A_full, N)
    
    # cross_corr[i] = sum_j w[i+j] * a[j] (with appropriate wrap-around)
    # For i = 0..n-1 and j = 0..n-1, we need i+j < 2n-1, which is always true since i+j <= 2n-2
    # And the FFT gives us the circular correlation, but since our sequences are zero-padded to length N >= 2n,
    # the linear cross-correlation is embedded in the result.
    
    g = 2.0 * cross_corr[:n]
    
    # Project gradient to tangent plane (remove mean to preserve sum)
    g_mean = float(np.mean(g))
    g = g - g_mean
    
    # Multiplicative update
    exponent = -lr * g
    exponent = np.clip(exponent, -2.0, 2.0)
    new_a = a * np.exp(exponent)
    new_a = np.where(np.isfinite(new_a), new_a, 0.0)
    new_a = np.clip(new_a, 0.0, 1000.0)
    s = float(new_a.sum())
    if not np.isfinite(s) or s < 0.01:
        return a
    new_a = new_a / s
    return new_a

def _activate_set_optimize(a, total_time):
    """
    Active-set optimization: focus on the top lags and try to equalize them.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size
    
    lr = 1.0
    K = 10  # Number of active lags
    
    while time.monotonic() - start < total_time:
        c = _fft_convolve(arr)
        max_c = float(np.max(c))
        
        # Get active set
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
            active = active[np.argsort(c[active])[::-1]]
        
        # Compute gradient as average over active lags
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                g[idx] += 2.0 * arr[k - idx]
        
        g /= float(len(active))
        
        # Project to tangent plane
        g -= np.mean(g)
        
        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -1.5, 1.5)
        new_a = arr * np.exp(exponent)
        new_a = np.clip(new_a, 0.0, 1000.0)
        ns = float(new_a.sum())
        if ns < 0.01:
            lr *= 0.5
            continue
        new_a /= ns
        
        sc = _internal_score(new_a)
        if np.isfinite(sc) and sc < best_score:
            best = new_a.copy()
            best_score = sc
            arr = new_a
            lr = min(5.0, lr * 1.01)
        else:
            lr = max(0.01, lr * 0.8)
            arr = best.copy()
            if lr < 0.01:
                break
        
        if time.monotonic() - start > total_time - 5:
            break
    
    return best, best_score

"""
Low-peak nonnegative step-function search using soft-max tempered gradient descent.

Strategy:
1. Generate candidates from multi-harmonic anti-palindromic sinusoidal families.
2. Apply phase-based soft-max gradient optimization with proper scaling.
3. Apply active-set refinement for final polish.
4. Validate with evaluate_sequence and return the best.
"""

import numpy as np
import time
import math


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_cand(n, amps, modes):

    if n <= 0:

    a = np.ones(n, dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)

    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(n, dtype=np.float64) / n
    return a / s

def _gradient_step(a, beta, lr):
    """
    One soft-max tempered gradient step.
    a: normalized sequence (sum=1), non-negative
    beta: temperature (higher = more focus on peak)
    lr: step size (dimensionless, applied after gradient normalization)
    Returns updated sequence.
    """
    n = a.size
    c = _fft_convolve(a)
    max_c = float(np.max(c))

    # Tempered weights
    shifted = c - max_c  # all <= 0
    weights = np.exp(beta * shifted)
    total_w = float(np.sum(weights))
    if total_w < 1e-300:
        # Fallback: hard active set (top few)
        K = max(5, n // 100)
        active = np.argpartition(c, -K)[-K:]
        weights = np.zeros_like(c)
        weights[active] = 1.0
        total_w = float(K)
    weights = weights / total_w

    # Compute gradient via cross-correlation
    # g[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is 2 * (weights cross-correlate a)[i]
    N = _next_pow2(3 * n)
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)
    g = 2.0 * cross[:n]

    # Normalize gradient for stable step size
    g_std = float(np.std(g))
    if g_std < 1e-15:
        return a
    g = g / g_std

    # Project to tangent plane (preserve sum)
    g = g - float(np.mean(g))

    # Multiplicative update
    exponent = -lr * g
    exponent = np.clip(exponent, -3.0, 3.0)
    new_a = a * np.exp(exponent)
    new_a = np.where(np.isfinite(new_a), new_a, 0.0)
    new_a = np.clip(new_a, 0.0, 1000.0)
    s = float(new_a.sum())
    if not np.isfinite(s) or s < 0.01:
        return a
    return new_a / s


def _active_set_step(a, K, lr):
    """One active-set gradient step (hard top-K)."""
    n = a.size
    c = _fft_convolve(a)

    if c.size <= K:
        active = np.arange(c.size)
    else:
        active = np.argpartition(c, -K)[-K:]

    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1)
            g[idx] += 2.0 * a[k - idx]

    g /= float(len(active))
    g_std = float(np.std(g))
    if g_std < 1e-15:
        return a
    g = g / g_std
    g = g - float(np.mean(g))

    exponent = -lr * g
    exponent = np.clip(exponent, -3.0, 3.0)
    new_a = a * np.exp(exponent)
    new_a = np.where(np.isfinite(new_a), new_a, 0.0)
    new_a = np.clip(new_a, 0.0, 1000.0)
    s = float(new_a.sum())
    if not np.isfinite(s) or s < 0.01:
        return a
    return new_a / s


def _soft_optimize(a, time_budget):
    """
    Phase-based soft-max gradient optimization.
    Phase 1: Low beta, moderate lr (explore shape)
    Phase 2: Medium beta, medium lr (refine)
    Phase 3: High beta, low lr (peak reduction)
    Phase 4: Active-set polish
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size
    original_score = best_score

    print(f"    [soft] n={n} start_score={best_score:.8f} budget={time_budget:.1f}s")

    # Phase configuration: (time_fraction, beta_range, lr_range, K_for_active)
    phases = [
        (0.25, (1.0, 5.0), (0.3, 0.1), 0),
        (0.25, (5.0, 20.0), (0.2, 0.05), 0),
        (0.25, (20.0, 80.0), (0.1, 0.02), 0),
        (0.25, (0, 0), (0.05, 0.01), 10),  # Active set
    ]

    for phase_idx, (tf, (b_lo, b_hi), (lr_lo, lr_hi), K) in enumerate(phases):
        phase_time = time_budget * tf
        phase_start = time.monotonic()
        if time.monotonic() - start > time_budget - 2:
            break

        patience = 0
        max_patience = 3000
        local_best = arr.copy()
        local_best_score = _internal_score(local_best)
        it = 0

        while time.monotonic() - phase_start < phase_time and it < 100000:
            it += 1
            progress = min(1.0, it / max(1, int(phase_time * 10)))

            if K > 0:
                # Active-set step
                lr = lr_lo + (lr_hi - lr_lo) * progress
                new_a = _active_set_step(arr, K, lr)
            else:
                # Soft-max step
                beta = b_lo + (b_hi - b_lo) * progress
                lr = lr_lo + (lr_hi - lr_lo) * progress
                new_a = _gradient_step(arr, beta, lr)

            # Evaluate periodically
            if it % 30 == 0 or it < 50:
                sc = _internal_score(new_a)
                if np.isfinite(sc) and sc < local_best_score:
                    local_best = new_a.copy()
                    local_best_score = sc
                    patience = 0
                    if sc < best_score:
                        best = new_a.copy()
                        best_score = sc
                else:
                    patience += 30
                    if sc > local_best_score * 1.01:
                        arr = local_best.copy()
                    else:
                        arr = new_a
                    if patience > max_patience:
                        arr = local_best.copy()
                        patience = 0
                        lr_lo *= 0.7
                        lr_hi *= 0.7
                        if lr_lo < 0.001:
                            break
            else:
                arr = new_a

        # End of phase
        sc = _internal_score(arr)
        if np.isfinite(sc) and sc < best_score:
            best = arr.copy()
            best_score = sc
        arr = best.copy()

        if it % 1000 < 300:
            print(f"    [soft] phase{phase_idx} done, score={best_score:.8f}")

    improvement = original_score - best_score
    print(f"    [soft] final score={best_score:.8f} (improved by {improvement:.6f})")
    return best, best_score


def _random_perturb(a, sigma, seed=None):
    """Add Gaussian noise and renormalize (stay non-negative)."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()
    noise = rng.randn(a.size) * sigma
    new_a = a + noise
    new_a = np.clip(new_a, 0.0, None)
    s = float(new_a.sum())
    if s < 0.01:
        return a
    return new_a / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns the best sequence found.
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 40.0

    print(f"=== Search starting, budget={budget_s}s ===")

    candidates = []  # (name, array, score)

    # Phase 1: Generate diverse candidates
    gen_deadline = t0 + min(180.0, total_budget * 0.1)

    # Try different lengths
    for n in [2048, 4096, 1024]:
        if time.monotonic() > gen_deadline:
            break
        a, sc = _scan_family(n, gen_deadline)
        candidates.append((f"sine_n{n}", a, sc))
        print(f"[gen] n={n} sine score={sc:.8f}")

    # Try cos-perturbed candidates
    for n in [2048, 4096]:
        if time.monotonic() > gen_deadline:
            break
        for c_amp in [0.05, -0.05, 0.1, -0.1]:
            a = _make_cos_cand(n, 0.72, c_amp)
            sc = _score_cand(a)
            candidates.append((f"cos_n{n}_c{c_amp}", a, sc))
        print(f"[gen] n={n} cos variants done")

    # Try multi-harmonic
    for n in [2048, 4096]:
        if time.monotonic() > gen_deadline:
            break
        for amps, modes, name in [
            ([0.7, 0.15], [1, 2], "h12"),
            ([0.7, -0.15], [1, 2], "h12n"),
            ([0.7, 0.1], [1, 3], "h13"),
            ([0.7, 0.1, 0.05], [1, 2, 3], "h123"),
        ]:
            a = _make_cand(n, amps, modes)
            sc = _score_cand(a)
            candidates.append((f"multi_{name}_n{n}", a, sc))
    print(f"[gen] multi-harmonic done")

    # Try using height_sequence_1 if available
    try:
        hs1 = None
        if 'height_sequence_1' in globals():
            hs1 = np.array(globals()['height_sequence_1'], dtype=np.float64)
        elif 'height_sequence_1' in dir():
            hs1 = np.array(height_sequence_1, dtype=np.float64)
        if hs1 is not None and hs1.size > 0:
            hs1 = np.clip(hs1, 0.0, 1000.0)
            s = float(hs1.sum())
            if s > 0.01:
                hs1 = hs1 / s
                sc = _score_cand(hs1)
                candidates.append(("hs1", hs1, sc))
                print(f"[gen] hs1 n={hs1.size} score={sc:.8f}")
    except Exception as e:
        print(f"[gen] hs1 error: {e}")

    # Sort by score
    candidates.sort(key=lambda x: x[2])
    print(f"[gen] Best candidate: {candidates[0][0]} score={candidates[0][2]:.8f}")

    # Phase 2: Optimize top candidates
    remaining = hard_deadline - time.monotonic()
    top_k = min(4, len(candidates))
    time_per = remaining / top_k - 5

    for idx in range(top_k):
        if time.monotonic() > hard_deadline - 30:
            break
        name, cand, sc = candidates[idx]
        opt_time = min(time_per, hard_deadline - time.monotonic() - 30)
        if opt_time < 15:
            break

        print(f"\n[optimize] {name} (score={sc:.8f}, time={opt_time:.0f}s)")

        # Main optimization
        opt_a, opt_score = _soft_optimize(cand, opt_time * 0.7)

        if np.isfinite(opt_score):
            candidates.append((f"{name}_opt", opt_a, opt_score))
            best_from_opt = opt_a
            best_opt_score = opt_score
        else:
            best_from_opt = cand
            best_opt_score = sc

        # Additional: try random perturbation + short refine
        remaining2 = hard_deadline - time.monotonic() - 25
        if remaining2 > 20:
            for trial_seed in [seed, seed + 1, seed + 2]:
                if time.monotonic() > hard_deadline - 25:
                    break
                perturbed = _random_perturb(best_from_opt, 0.02, trial_seed)
                pert_score = _score_cand(perturbed)
                if np.isfinite(pert_score) and pert_score < best_opt_score:
                    # Refine the perturbed version briefly
                    brief_time = min(remaining2 * 0.3, 30)
                    if brief_time > 5:
                        p_a, p_score = _soft_optimize(perturbed, brief_time)
                        if np.isfinite(p_score) and p_score < best_opt_score:
                            candidates.append((f"{name}_pert{trial_seed}", p_a, p_score))
                            best_from_opt = p_a
                            best_opt_score = p_score
                        remaining2 = hard_deadline - time.monotonic() - 25

    # Re-sort
    candidates.sort(key=lambda x: x[2])
    print(f"\n[optimize] Best after optimization: {candidates[0][0]} score={candidates[0][2]:.8f}")

    # Phase 3: Final polish on the very best
    remaining = hard_deadline - time.monotonic() - 20
    if remaining > 30 and len(candidates) > 0:
        best_name, best_cand, best_sc = candidates[0]
        print(f"\n[final] Polishing {best_name} (score={best_sc:.8f}, time={remaining:.0f}s)")
        # Use active-set heavily for final polish
        final_a, final_score = _soft_optimize(best_cand, remaining)
        if np.isfinite(final_score) and final_score < best_sc:
            candidates.insert(0, (f"{best_name}_final", final_a, final_score))
            print(f"[final] Improved to {final_score:.8f}")

    # Phase 4: Validate with evaluate_sequence
    print(f"\n[validate] Validating top candidates...")
    candidates.sort(key=lambda x: x[2])

    best_actual = np.inf
    best_out = None
    validated = set()

    for name, cand, sc in candidates:
        if time.monotonic() > hard_deadline:
            break
        if len(validated) >= 6:
            break

        arr = np.asarray(cand, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not np.isfinite(s) or s < 0.01:
            continue
        arr = arr / s

        key = (arr.size, round(sc, 6))
        if key in validated:
            continue
        validated.add(key)

        try:
            real_score = float(evaluate_sequence(arr.tolist()))
        except Exception as e:
            print(f"[validate] Error on {name}: {e}")
            real_score = np.inf

        if np.isfinite(real_score) and real_score < best_actual:
            best_actual = real_score
            best_out = arr
            print(f"[validate] {name} n={arr.size} score={real_score:.8f}")
            if best_actual < 1.50:
                break

    # Fallback
    if best_out is None or not np.isfinite(best_actual):
        print("[validate] Using fallback")
        best_out = _make_cand(2048, [0.72], [1])
        best_actual = _internal_score(best_out)

    print(f"\n=== Done. Best score: {best_actual:.8f} ===")
    return [float(x) for x in best_out]
