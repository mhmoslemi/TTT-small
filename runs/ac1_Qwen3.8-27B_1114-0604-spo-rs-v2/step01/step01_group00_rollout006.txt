    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval(f):
    """Evaluate the sequence."""
    return evaluate_sequence(f.tolist())


def _normalize(f):
    """Normalize to sum=1, clip to [0, inf)."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def _smoothed_max(c, T):
    """Compute smoothed max and its gradient weights."""
    cs = c / T
    cs = cs - cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p /= s
    m = T * (np.log(s) + cs.max())  # This is T * log(sum(exp(c/T)))
    # Actually: smoothed_max = T * log(sum(exp(c/T)))
    # Let me recompute: cs = c/T - max(c)/T = (c - c_max)/T
    # sum(exp(cs)) = sum(exp((c-c_max)/T))
    # T * log(sum(exp((c-c_max)/T))) + c_max = T * log(sum(exp(c/T)))
    m = T * np.log(s) + c.max()
    return m, p


def _grad_at_lags(p, f):
    """Compute gradient: sum_k p[k] * 2*f[k-j] (correlated with f)."""
    n = len(f)
    # grad[j] = 2 * sum_k p[k] * f[k-j] (where k-j is in [0,n-1])
    # This is 2 * correlate(p, f, 'valid')
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.flatnonzero(cond))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _gradient_step(f, c, T, step_size, method='mirror'):
    """Take one optimization step."""
    n = len(f)
    _, p = _smoothed_max(c, T)
    grad = _grad_at_lags(p, f)
    
    if method == 'mirror':
        # Exponentiated gradient (mirror descent)
        span = grad.max() - grad.min()
        if span < 1e-15:
            return f
        eta = step_size / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s = f_new.sum()
        if s < 1e-300:
            return f
        return f_new / s
    else:
        # Euclidean projected gradient
        f_new = f - step_size * grad
        return _project_simplex(f_new)


def _optimize_adaptive(f0, n, deadline, rng, label=""):
    """Adaptive optimization with targeted lag attacks and annealing."""
    f = _normalize(f0.copy())
    local_best = _eval(f)
    local_best_f = f.copy()
    
    c = _autocorr_fft(f)
    peak0 = c.max()
    
    # Temperature schedule: start high (explore all lags), end low (focus on max)
    T_start = peak0 * 0.5
    T_end = peak0 * 1e-10
    
    # Step size schedule
    step_start = 0.5
    step_end = 0.01
    
    n_iters = 40000
    no_improve = 0
    patience = 8
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        # Cosine annealing for temperature
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        # Step size: linear from start to end
        step = step_start + (step_end - step_start) * progress
        
        c = _autocorr_fft(f)
        f = _gradient_step(f, c, T, step, method='mirror')
        
        # Periodic evaluation
        if it % 2000 == 0 or it == n_iters - 1:
            score = _eval(f)
            if score < local_best - 1e-7:
                local_best = score
                local_best_f = f.copy()
                no_improve = 0
                if label and it % 10000 == 0:
                    print(f"  {label} iter {it}: {score:.6f}")
            else:
                no_improve += 1
            
            if no_improve >= patience:
                # Restart with perturbation
                mag = 0.01 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = _normalize(f)
                no_improve = 0
        
        # Every 5000 iters, do a targeted attack on the top lag
        if it % 5000 == 4999:
            c = _autocorr_fft(f)
            k_star = int(np.argmax(c))
            # Gradient at k_star: 2*f[k_star - j]
            grad_star = np.zeros(n)
            valid = (k_star >= np.arange(n)) & (k_star < np.arange(n) + n)
            # f[k_star - j] for j=0..n-1, valid when 0 <= k_star-j < n
            idx = k_star - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad_star[valid] = 2.0 * f[idx[valid]]
            
            # Take a step specifically to reduce this lag
            span = grad_star.max() - grad_star.min()
            if span > 1e-15:
                eta = 0.1 / span
                log_f = np.log(f + 1e-300) - eta * grad_star
                log_f -= log_f.max()
                f_try = np.exp(log_f)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < c.max() - 1e-12:
                    f = f_try
    
    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best


def _coordinate_polish(f, n, deadline, rng):
    """Coordinate descent polish focusing on reducing the peak lag."""
    f = _normalize(f.copy())
    local_best = _eval(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()
    
    n_passes = 50
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        # Find top 3 lags
        top_lags = np.argsort(c)[::-1][:3]
        
        improved = False
        for k_star in top_lags:
            k_star = int(k_star)
            # Try to reduce this lag by moving mass from high-contribution
            # positions to low-contribution positions
            idx = k_star - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            contrib = np.zeros(n)
            contrib[valid] = f[valid] * f[idx[valid]]
            
            j_donate = int(np.argmax(contrib))
            j_receive = int(np.argmin(contrib))
            
            if j_donate == j_receive:
                continue
            
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                f_try = f.copy()
                amt = f_try[j_donate] * frac
                if amt < 1e-15:
                    continue
                f_try[j_donate] -= amt
                f_try[j_receive] += amt
                f_try = _normalize(f_try)
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-13:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    local_best = _eval(f)
                    local_best_f = f.copy()
                    improved = True
                    break
        
        # Random coordinate moves
        for _ in range(20):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            for frac in [0.03, 0.07, 0.12]:
                f_try = f.copy()
                amt = f_try[i] * frac
                if amt < 1e-15:
                    break
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = _normalize(f_try)
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-13:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    local_best = _eval(f)
                    local_best_f = f.copy()
                    break
    
    return local_best_f, local_best


def _simulated_annealing(f_start, n, deadline, rng):
    """Simulated annealing wrapper for global exploration."""
    f = _normalize(f_start.copy())
    best_f = f.copy()
    best_score = _eval(f)
    
    c = _autocorr_fft(f)
    peak = c.max()
    
    T = peak * 0.3  # Initial temperature
    T_min = peak * 1e-8
    cooling = 0.9995
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Generate perturbation
        strategy = rng.random()
        if strategy < 0.4:
            # Small Gaussian perturbation
            mag = T / (peak * np.sqrt(n))
            f_try = f + mag * rng.standard_normal(n)
        elif strategy < 0.7:
            # Swap two coordinates
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            f_try = f.copy()
            amt = 0.05 * max(f[i], f[j])
            f_try[i] -= amt
            f_try[j] += amt
        else:
            # Block move: perturb a random block
            block_size = int(rng.integers(5, min(50, n // 4)))
            start = int(rng.integers(0, n - block_size))
            f_try = f.copy()
            f_try[start:start+block_size] += 0.02 * rng.standard_normal(block_size)
        
        f_try = _normalize(f_try)
        c_try = _autocorr_fft(f_try)
        peak_try = c_try.max()
        
        # Accept/reject
        delta = peak_try - peak
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-30)):
            f = f_try
            c = c_try
            peak = peak_try
        
        # Cool down
        T *= cooling
        
        # Periodic evaluation
        if it % 500 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                print(f"  SA iter {it}: {score:.6f}")
        
        # Reheat if stuck
        if it % 5000 == 0:
            # Check if we're making progress
            pass
    
    return best_f, best_score


def _make_initializations(n, rng, given_f):
    """Create diverse initializations."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    inits = {}
    
    # Given sequence
    inits['given'] = _normalize(np.clip(given_f, 0, 1000).copy())
    
    # Power-law ramps (various exponents)
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f_ramp = (1.0 - j / (n - 1)) ** p + 1e-10
        inits[f'right_{p}'] = _normalize(f_ramp)
        f_ramp_l = (j / (n - 1)) ** p + 1e-10
        inits[f'left_{p}'] = _normalize(f_ramp_l)
    
    # Exponential decay
    for rate in [0.5, 1.0, 2.0, 3.0]:
        f_exp = np.exp(-rate * j / n) + 1e-10
        inits[f'exp_r_{rate}'] = _normalize(f_exp)
        f_exp_l = np.exp(-rate * (n-1-j) / n) + 1e-10
        inits[f'exp_l_{rate}'] = _normalize(f_exp_l)
    
    # Beta-like shapes (concentrated left or right)
    for alpha, beta in [(3, 1), (5, 1), (2, 1), (1, 3), (1, 5), (4, 2), (2, 4)]:
        x = j / (n - 1)
        f_beta = x ** (alpha - 1) * (1 - x) ** (beta - 1) + 1e-10
        inits[f'beta_{alpha}_{beta}'] = _normalize(f_beta)
    
    # Sine-based
    inits['sine_half'] = _normalize(np.sin(np.pi * j / (2 * n)) ** 2)
    inits['sine_shift'] = _normalize(np.sin(np.pi * (j + 0.2 * n) / (2 * n)) ** 2)
    inits['cos_asym'] = _normalize((1 + np.cos(np.pi * j / n)) * (1 - 0.5 * j / n) + 0.01)
    
    # Random
    inits['rand1'] = _normalize(rng.exponential(1.0, size=n))
    inits['rand2'] = _normalize(rng.dirichlet(np.ones(n) * 0.01))
    inits['rand3'] = _normalize(np.abs(rng.standard_normal(n)))
    
    return inits
