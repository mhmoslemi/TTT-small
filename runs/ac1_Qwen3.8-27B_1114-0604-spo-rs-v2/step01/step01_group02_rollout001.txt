"""
Multi-phase optimizer to minimize f(a) = 2n*max_k conv(a,a)[k] / (sum a)^2.

Approach:
  1. Multi-length exploration with diverse initializations
  2. Gradient-based refinement with adaptive soft-peak surrogate
  3. Direct peak-shaving via water-filling heuristic
  4. Coordinate descent for final polish
  5. Random restarts to escape local minima

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_and_conv(a):
    """Compute score and convolution."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf, c, 0.0
    return 2.0 * n * float(np.max(c)) / (S * S), c, S


def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam optimizer step (in-place)."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a[:] = np.clip(a - lr * mh / (np.sqrt(vh) + eps), 0.0, 1000.0)


def gradient_topk(a, c, k, temp_frac=0.1):
    """
    Compute gradient of the weighted-peak surrogate.
    Uses exponential soft-max weights on the top-k peaks.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    
    k_act = min(k, len(c))
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    
    top = float(vals[0])
    if top > 1e-12:
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    
    M = float(np.dot(vals, wts))
    
    grad = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        # d conv(ki)/d a[j] = 2*a[ki-j] for j in [j_start, j_end]
        grad[j_start:j_end+1] += wi * 2.0 * a[j_start:j_end+1][::-1]
    
    # d/d a_j of [2n * M / S^2] = 2n * (grad_c / S^2 - 2*M / S^3)
    grad = (2.0 * n) * (grad / S2 - 2.0 * M / (S2 * S))
    return grad


def peak_shave(a, c, num_peaks=3, strength=0.3):
    """
    Direct peak-shaving: reduce values that contribute to the top peaks,
    redistribute to low-contributing positions.
    """
    n = len(a)
    S = float(a.sum())
    a_new = a.copy()
    
    # Find top peaks
    top_idx = np.argsort(c)[::-1][:num_peaks]
    
    for ki in top_idx:
        # Contributors: j where a[j]*a[ki-j] is large
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        products = a[js] * a[ki - js]
        
        # Reduce the largest contributors
        threshold = np.percentile(products, 70)
        mask = products > threshold
        if not mask.any():
            continue
        
        # Amount to reduce
        total_reduce = strength * products[mask].sum()
        a_new[js[mask]] -= total_reduce / mask.sum()
        
        # Redistribute to non-contributors
        non_contrib_mask = ~mask
        if non_contrib_mask.any() and a_new[js[non_contrib_mask]].sum() > 0:
            a_new[js[non_contrib_mask]] += total_reduce / non_contrib_mask.sum()
    
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new


def coordinate_descent(a, c, max_passes=5, step_frac=0.1):
    """
    Coordinate descent: for each coordinate, try small adjustments
    and keep if they improve the score.
    """
    n = len(a)
    best_score, c, S = score_and_conv(a)
    
    for _ in range(max_passes):
        # Shuffle coordinates
        perm = np.random.permutation(n)
        for j in perm:
            c_old = c.copy()
            best_local = best_score
            a_old = a[j]
            
            # Try a few values
            for frac in (-0.5, -0.25, -0.1, 0.1, 0.25, 0.5):
                a_new_val = a_old * (1.0 + frac)
                if a_new_val < 0:
                    a_new_val = 0.0
                if a_new_val > 1000:
                    continue
                
                a[j] = a_new_val
                c_new = fast_self_conv(a)
                S_new = float(a.sum())
                if S_new < 0.01:
                    a[j] = a_old
                    continue
                score_new = 2.0 * n * float(np.max(c_new)) / (S_new * S_new)
                
                if score_new < best_local:
                    best_local = score_new
                    c = c_new
                    S = S_new
            
            if a[j] == a_old and best_local == best_score:
                a[j] = a_old  # no change
            best_score = best_local
        
        if best_score == best_local and _ > 2:
            break
    
    return a, best_score
