Optimizes non-negative step function heights to minimize
2*n*max(conv(a,a))/(sum(a))^2 using FFT-based gradient descent
with adaptive peak weighting, multiplicative updates, and
coordinate descent polishing.

import time

def _normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.asarray(a, dtype=np.float64).copy()
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s < 1e-300:
        a.fill(1.0 / a.size)
        return a
    a /= s
    return a

def _autocorr(a, L, m):
    """Compute linear autocorrelation (convolution with itself) via FFT.
    Returns array of length m = 2n-1."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _score(c, n):
    """Compute the score from the autocorrelation."""
    return float(2.0 * n * np.max(c))

def _gradient(c, a, L, n, m, peak_frac=0.05, temp=1.0):
    """Compute gradient direction to reduce max autocorrelation.
    
    Uses a combination of top-k peak weighting and softmax.
    Returns gradient vector of length n.
    """
    # Scale c for numerical work
    v = n * c
    
    # Method 1: Softmax weighting
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -50.0, 0.0)
    w_soft = np.exp(z)
    w_soft /= float(w_soft.sum())
    
    # Method 2: Top-k hard weighting
    k = max(1, int(peak_frac * m))
    top_idx = np.argpartition(v, -k)[-k:]
    w_hard = np.zeros(m)
    w_hard[top_idx] = 1.0 / k
    
    # Combine
    alpha = 0.5  # weight for hard vs soft
    w = alpha * w_hard + (1.0 - alpha) * w_soft

    # grad[m] = sum_k w[k] * (a[m+k] + a[m-k]) (with zero-padding)

    # Let me use the FFT approach:
    # Let F = fft(a, L), W = fft(w, L)
    # The "cross-correlation" of w and a is: ifft(conj(W) * F)
    # The "convolution" of w and a is: ifft(W * F)
    # We want: sum_k w[k] * a[m+k] + sum_k w[k] * a[m-k]

def _norm(a):

    a = np.maximum(a, 0.0)

    F = np.fft.rfft(a, L)

    idx = np.argpartition(v, m - k)[-k:]
    w_h = np.zeros(m)
    w_h[idx] = 1.0 / k
    
    # Blend
    return 0.4 * w_h + 0.6 * w_s


def _grad(a, w, L, n, m):
    """Full gradient of weighted autocorrelation w.r.t. a."""
    F = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    p1 = np.fft.irfft(W * F, L)[:n]
    p2 = np.fft.irfft(np.conj(W) * F, L)[:n]
    g = p1 + p2
    np.maximum(g, 0.0, out=g)
    return g


def _step(a, g, eta):
    """Multiplicative update."""
    gm = float(g.max())
    if gm < 1e-300:
        return a.copy()
    return _norm(a * np.exp(-eta * g / gm))


def _ls(a, g, L, m, n, eta0):
    """Line search: try decreasing step sizes."""
    best_a = a.copy()
    best_s = _sc(_acorr(a, L, m), n)
    for i in range(6):
        eta = eta0 * (0.5 ** i)
        if eta < 1e-6:
            break
        cnd = _step(a, g, eta)
        s = _sc(_acorr(cnd, L, m), n)
        if s < best_s:
            best_s = s
            best_a = cnd
    return best_a, best_s


def _kick(a, n, rng, frac=0.4):
    """Random perturbation for diversification."""
    r = rng.random(n)
    mix = frac * (0.5 + 0.5 * rng.random())
    return _norm((1.0 - mix) * a + mix * r)


def _local_search(a, L, m, n, rng, time_limit):
    """Stochastic local search polishing."""
    deadline = time.monotonic() + time_limit
    a = _norm(a.copy())
    sc = _sc(_acorr(a, L, m), n)
    best_a = a.copy()
    best_sc = sc
    scale = 0.1
    it = 0
    
    while time.monotonic() < deadline:
        # Gaussian perturbation
        noise = rng.normal(0.0, scale, n)
        cand = a + noise
        cand = _norm(cand)
        s2 = _sc(_acorr(cand, L, m), n)
        
        if s2 < sc:
            a = cand
            sc = s2
            scale = min(0.3, scale * 1.05)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        else:
            scale = max(0.01, scale * 0.95)
        
        it += 1
        if it % 200 == 0 and sc > best_sc * 1.005:
            # Reset to best and retry with smaller scale
            a = best_a.copy()
            sc = best_sc
            scale = max(0.01, scale * 0.5)
    
    return best_a, best_sc
