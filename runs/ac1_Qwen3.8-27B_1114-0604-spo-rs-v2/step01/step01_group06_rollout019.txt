"""
Algorithm: Multi-phase optimization for minimizing 2n*max(a*a)/(sum(a))^2.
Phase 1: Spectral reshaping - flatten the convolution spectrum A^2 in frequency domain.
Phase 2: L-BFGS in log-space with soft-max approximation to max().
Phase 3: Local peak-shaving coordinate search.
Starts from the previous best (height_sequence_1) if available.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Global state for optimizer callbacks
_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-5
_G_BOUND = 35.0
_G_BOUNDS = []


def _set_state(n, tau=1e-5):
    global _G_N, _G_NFFT, _G_TAU, _G_BOUND, _G_BOUNDS
    _G_N = int(n)
    _G_NFFT = 1 << (2 * _G_N - 1).bit_length()
    _G_TAU = float(tau)
    _G_BOUND = 35.0
    _G_BOUNDS = [(-_G_BOUND, _G_BOUND)] * _G_N


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _fft_conv(a, N=None):
    n = len(a)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1].real
    return c


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * np.max(c) / (S * S))


def _zz_from_a(a):
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z = z - np.mean(z)
    return np.clip(z, -_G_BOUND + 1, _G_BOUND - 1)


def _obj_grad(z):
    n = _G_N
    N = _G_NFFT
    tau = _G_TAU
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_G_BOUND, neginf=-_G_BOUND)
    z = np.clip(z, -_G_BOUND, _G_BOUND)

    a = _softmax(z)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    p = e / se

    # Gradient of soft-max w.r.t. a: dF/da_j = 2 * sum_k p[k]*a[k-j]
    P_full = np.fft.fft(np.concatenate([p, np.zeros(N - len(p))]))
    Af_full = np.fft.fft(np.concatenate([a, np.zeros(N - len(a))]))
    corr = np.fft.ifft(P_full * np.conj(Af_full)).real[:n]
    dF_da = 2.0 * corr

    # Chain rule through softmax: dF/dz_j = sum_i dF/da_i * da_i/dz_j
    # da_i/dz_j = a_i*(delta_ij - a_j)
    # dF/dz_j = dF/da_j * a_j - a_j * sum_i dF/da_i * a_i
    s_d = float(np.dot(dF_da, a))
    grad = dF_da * a - a * s_d

    val = m + tau * np.log(se)
    # Scale for numerical comfort
    scale = float(n)
    val *= scale
    grad *= scale

    # Tiny regularization to fix gauge
    pen = 1e-12
    val += 0.5 * pen * float(np.dot(z, z))
    grad += pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n)
    return float(val), grad


def _run_lbfgs(z0, maxfun=600, maxiter=200):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    if z0.size != _G_N:
        return None
    z0 = np.clip(z0 - np.mean(z0), -_G_BOUND, _G_BOUND)
    opts = {"maxfun": int(maxfun), "maxiter": int(maxiter),
            "ftol": 1e-13, "pgtol": 1e-11, "disp": False}
    try:
        res = minimize(_obj_grad, z0, jac=True, method="L-BFGS-B",
                       bounds=_G_BOUNDS, options=opts)
        return res
    except Exception:
        return None


def _spectral_reshape(a, deadline, n_steps=200):
    """
    Iteratively reshape the spectrum to flatten the convolution.
    C = A^2 is the spectrum of the convolution. We dampen high-magnitude
    frequency components of C to flatten the time-domain convolution.
    """
    n = _G_N
    N = _G_NFFT
    best = a.copy()
    best_sc = _score(a)
    a = a.copy()
    S = np.sum(a)

    it = 0
    while time.monotonic() < deadline and it < n_steps:
        A = np.fft.fft(np.concatenate([a, np.zeros(N - n)]))
        C = A * A  # spectrum of the convolution

        # Target: reduce dynamic range of |C|
        C_mag = np.abs(C)
        C_max = np.max(C_mag)
        if C_max < 1e-30:
            break
        C_mean = np.mean(C_mag)

        # Damping: compress above-mean values
        for beta in (0.1, 0.2, 0.35, 0.5, 0.7):
            C_new = C.copy()
            # Dampen frequencies where |C| is large
            mask = C_mag > C_mean
            C_new[mask] = C[mask] * (1.0 - beta) + C[mask] * beta * (C_mean / C_mag[mask])
            # Take square root to get new A
            A_new = np.sqrt(C_new)
            a_new = np.fft.ifft(A_new).real
            a_new = np.maximum(a_new, 0.0)
            sn = float(np.sum(a_new))
            if sn < 0.01:
                continue
            a_new /= sn * 1.0  # keep sum = S for fair comparison... actually score is scale invariant
            a_new *= S / sn
            a_new = a_new / np.sum(a_new)
            sc = _score(a_new)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = a_new.copy()
                a = a_new.copy()
                break
        else:
            # No improvement with any beta, try inverse operation (boost low freq)
            for beta in (0.05, 0.1, 0.2):
                C_new = C.copy()
                mask = C_mag < C_mean
                if np.sum(mask) == 0:
                    continue
                C_new[mask] = C[mask] * (1.0 + beta)
                A_new = np.sqrt(C_new)
                a_new = np.fft.ifft(A_new).real
                a_new = np.maximum(a_new, 0.0)
                sn = float(np.sum(a_new))
                if sn < 0.01:
                    continue
                a_new /= sn
                sc = _score(a_new)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best = a_new.copy()
                    a = a_new.copy()
                    break

        it += 1
        if it % 40 == 0:
            print(f"  spectral reshape it={it}, best_score={best_sc:.9f}, "
                  f"time_left={max(0, deadline-time.monotonic()):.0f}s")

    return best, best_sc


def _peak_shave(a, deadline, n_steps=5000):
    """
    Local search: identify the peak lag and redistribute weight
    from peak-contributing elements to non-contributing ones.
    """
    n = _G_N
    best = a.copy()
    best_sc = _score(a)
    a = a.copy()
    rng = np.random.default_rng(12345)

    no_improve = 0
    it = 0
    while time.monotonic() < deadline and it < n_steps:
        c = _fft_conv(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        S = float(np.sum(a))

        # Compute contributions to the peak: contrib[i] = a[i]*a[k_star-i]
        # Valid i: 0<=i<n and 0<=k_star-i<n
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        if i_hi < i_lo:
            no_improve += 1
            it += 1
            continue

        contrib = np.zeros(n)
        idx = np.arange(i_lo, i_hi + 1)
        contrib[idx] = a[idx] * a[k_star - idx]

        # Find top contributors and bottom contributors
        total_contrib = float(np.sum(contrib))
        if total_contrib < 1e-30:
            no_improve += 1
            it += 1
            continue

        # Normalize contributions
        frac = contrib / total_contrib

        # Try to move weight from top contributors to bottom ones
        n_move = max(2, n // 512)  # number of elements to try
        top_idx = np.argsort(frac)[::-1][:n_move]
        bot_idx = np.argsort(frac)[:n_move]

        # Ensure bottom indices have some weight to receive (or at least a[k_star-i] is not zero)
        valid_bot = [j for j in bot_idx if a[j] > 1e-10 or a[k_star - j] > 1e-10 if 0 <= k_star - j < n]
        valid_top = [j for j in top_idx if a[j] > 1e-10 if 0 <= k_star - j < n]

        if len(valid_top) == 0 or len(valid_bot) == 0:
            no_improve += 1
            it += 1
            continue

        improved = False
        for eta in (0.001, 0.005, 0.01, 0.02, 0.05):
            a_try = a.copy()
            moved = 0.0
            for j in valid_top[:5]:
                amt = eta * a[j] / len(valid_top[:5])
                a_try[j] -= amt
                moved += amt
                # Distribute to bottom elements
                for k in valid_bot[:5]:
                    a_try[k] += amt / len(valid_bot[:5])
            if np.any(a_try < 0):
                a_try = np.maximum(a_try, 0)
            sc = _score(a_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation
            if no_improve > 50:
                scale = 0.01 * (no_improve / 50.0)
                scale = min(scale, 0.1)
                a_try = a + scale * rng.standard_normal(n)
                a_try = np.maximum(a_try, 0)
                a_try /= np.sum(a_try)
                sc = _score(a_try)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best = a_try.copy()
                    a = a_try.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 500:
                        break
            else:
                no_improve += 1

        it += 1
        if it % 500 == 0:
            print(f"  peak_shave it={it}, best_score={best_sc:.9f}, "
                  f"peak_lag={k_star}, time_left={max(0,deadline-time.monotonic()):.0f}s")

    return best, best_sc

    # Sort by score, keep top 12
    starts.sort(key=lambda x: x[1])
    return starts[:12]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum score sequence.
    Phase 1: Spectral reshaping
    Phase 2: L-BFGS in log-space
    Phase 3: Peak-shaving local search
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()

    # Choose n
    n = 8192
    _set_state(n, tau=1e-5)

    # Get initial best
    best_a = None
    best_score = np.inf

    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            prev = np.asarray(hs, dtype=np.float64)
            if prev.size == n and np.all(np.isfinite(prev)) and np.all(prev >= 0):
                S = float(np.sum(prev))
                if S > 0.01:
                    best_a = prev / S
                    best_score = _score(best_a)
                    print(f"Loaded previous best: score={best_score:.9f}")
    except Exception as ex:
        print(f"Warning: could not load previous best: {ex}")

    if best_a is None:
        # Create a reasonable initial guess
        a = np.ones(n)
        a[n//2:] = 2.0
        a /= np.sum(a)
        best_a = a
        best_score = _score(a)
        print(f"Created initial guess: score={best_score:.9f}")

    # === Phase 1: Spectral reshaping (25% of time) ===
    phase1_deadline = time.monotonic() + 0.25 * (deadline - time.monotonic())
    print(f"\n=== Phase 1: Spectral reshaping (deadline in {phase1_deadline - time.monotonic():.0f}s) ===")
    a1, sc1 = _spectral_reshape(best_a, phase1_deadline, n_steps=300)
    if sc1 < best_score:
        best_score = sc1
        best_a = a1.copy()
        print(f"Phase 1 improved: {best_score:.9f}")

    # === Phase 2: L-BFGS (45% of time) ===
    phase2_deadline = time.monotonic() + 0.45 * (deadline - time.monotonic())
    print(f"\n=== Phase 2: L-BFGS (deadline in {phase2_deadline - time.monotonic():.0f}s) ===")

    starts = _make_starts(best_a, rng)
    candidates = []

    for idx, (z0, sc0) in enumerate(starts):
        elapsed = time.monotonic() - t0
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Allocate iterations based on remaining time
        if idx < 3:
            maxfun = 1200
            maxiter = 400
        elif idx < 7:
            maxfun = 700
            maxiter = 250
        else:
            maxfun = 400
            maxiter = 150

        # Check if we have enough time (rough estimate)
        per_fun_time = 0.005  # ~5ms per function eval
        if maxfun * per_fun_time > remaining * 0.8:
            maxfun = int(remaining * 0.8 / per_fun_time)
            maxiter = int(maxfun * 0.4)
            if maxfun < 50:
                break

        res = _run_lbfgs(z0, maxfun=maxfun, maxiter=maxiter)
        if res is not None and hasattr(res, "x"):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            if z_new.size == n:
                a_new = _softmax(np.clip(z_new, -_G_BOUND, _G_BOUND))
                sc_new = _score(a_new)
                if np.isfinite(sc_new):
                    if sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                        print(f"  L-BFGS start {idx}: score={sc_new:.9f} "
                              f"elapsed={time.monotonic()-t0:.0f}s")
                    candidates.append((sc_new, a_new.copy()))

        if len(candidates) > 6:
            candidates.sort(key=lambda x: x[0])
            candidates = candidates[:6]

    print(f"Phase 2 done: best_score={best_score:.9f}")

    # === Phase 3: Peak shaving + polish (remaining time) ===
    phase3_deadline = deadline - 15.0  # leave 15s for cleanup
    print(f"\n=== Phase 3: Peak shaving (deadline in {phase3_deadline - time.monotonic():.0f}s) ===")

    # First, do a short L-BFGS polish on the best
    if time.monotonic() < phase3_deadline - 120:
        z_best = _zz_from_a(best_a)
        res = _run_lbfgs(z_best, maxfun=800, maxiter=300)
        if res is not None and hasattr(res, "x"):
            a_pol = _softmax(np.clip(res.x, -_G_BOUND, _G_BOUND))
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol.copy()
                print(f"  Polish L-BFGS: {best_score:.9f}")

    # Then peak shaving
    if time.monotonic() < phase3_deadline - 30:
        a3, sc3 = _peak_shave(best_a, phase3_deadline, n_steps=20000)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            print(f"Phase 3 peak shaving: {best_score:.9f}")

    # Final validation
    print(f"\nFinal best score (internal): {best_score:.9f}")

    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator score: {float(exact):.9f}")
            best_score = float(exact)
        else:
            print(f"Official evaluator returned: {exact}")
    except Exception as ex:
        print(f"Validation error: {ex}")

    # Defensive cleanup
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    S = float(np.sum(best_a))
    if S < 0.01 or not np.isfinite(S):
        best_a = np.ones(n) / n
    else:
        best_a = best_a / S

    print(f"Final: n={n}, internal_score={_score(best_a):.9f}")
    return [float(v) for v in best_a.tolist()]
