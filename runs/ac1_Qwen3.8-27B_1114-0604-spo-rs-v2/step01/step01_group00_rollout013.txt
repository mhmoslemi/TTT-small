Minimizes 2*n*max(f*f)/(sum f)^2 for non-negative f.

import numpy as np

def _autocorr_fft(f):
    """Linear autocorrelation (convolution) of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

def _peak(c):
    return c.max()

def _topk_grad(f, c, k=3):
    """Compute average subgradient from top-k lags."""
    n = len(f)
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for idx in top_k:
        lo = max(0, idx - n + 1)
        hi = min(n, idx + 1)
        grad[lo:hi] += 2.0 * f[idx - np.arange(lo, hi)]
    grad /= k
    return grad

def _polish(f, deadline, rng, n):
    """Coordinate descent: try moving mass between pairs."""
    f = f.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    c = _autocorr_fft(f)
    local_best = c.max()
    local_best_f = f.copy()
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Try moving mass from positions that contribute to c[k_star]
        # to positions that don't
        # contrib[i] = 2 * f[k_star - i] (gradient component)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        contrib[lo:hi] = 2.0 * f[k_star - np.arange(lo, hi)]
        
        # Find best donor (high contrib) and receiver (low contrib)
        donor = int(np.argmax(contrib))
        # Receiver should have low contribution AND be outside the "active" range
        receiver_candidates = np.argsort(contrib)[:n//10]
        
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            for recv in receiver_candidates[:5]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                if f_try[donor] < 0:
                    continue
                f_try /= f_try.sum()

"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2 using adaptive mirror descent
with top-k subgradient, simulated annealing acceptance, diverse restarts, and polish.
Key improvements: adaptive temperature with reheating, top-k gradient broadening,
SA acceptance for escaping local minima, and efficient FFT-based inner loop.
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Linear autocorrelation of f with itself via FFT (fast)."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _topk_grad(f, c, k):
    """Compute average subgradient from top-k lag positions."""
    n = len(f)
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for idx in top_k:
        lo = max(0, idx - n + 1)
        hi = min(n, idx + 1)
        grad[lo:hi] += 2.0 * f[idx - np.arange(lo, hi)]
    grad /= k
    return grad


def _mirror_step(f, grad, eta):
    """One mirror descent (exponentiated gradient) step on the simplex."""
    scale = grad.max() - grad.min()
    if scale < 1e-18:
        return f
    log_f = np.log(f + 1e-300) - (eta / scale) * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _optimize_sa(f0, deadline, rng, n):
    """
    Mirror descent with simulated annealing acceptance, adaptive top-k,
    and reheating when stuck.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _autocorr_fft(f)
    peak0 = c.max()
    local_best = peak0
    local_best_f = f.copy()

    eta = 0.8
    T = peak0 * 0.05
    T_floor = peak0 * 1e-14
    last_improve = time.monotonic()
    iter_count = 0
    total_accepted = 0
    total_rejected = 0

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()

        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()
            last_improve = time.monotonic()

        stuck_time = time.monotonic() - last_improve

        # Reheating and perturbation when stuck
        if stuck_time > 15:
            T = peak0 * 0.2
            eta = max(eta, 0.5)
            # Scale perturbation with how stuck we are
            mag = min(0.05, 0.005 + 0.003 * (stuck_time / 60.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
        else:
            # Slow cooling
            T = max(T_floor, T * 0.9995)
            eta *= 0.99995

        # Adaptive top-k: start focused (k=2), broaden to k=7
        progress = min(1.0, iter_count / 20000.0)
        k = 2 + int(5 * progress)
        grad = _topk_grad(f, c, k)

        f_new = _mirror_step(f, grad, eta)
        c_new = _autocorr_fft(f_new)
        new_peak = c_new.max()

            f = f_new
            eta = min(eta * 1.005, 3.0)

    c = _autocorr_fft(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _polish(f, deadline, rng, n):
    """Coordinate descent polish: move mass between specific coordinate pairs."""
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    c = _autocorr_fft(f)
    local_best = c.max()
    local_best_f = f.copy()
    passes = 0

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()

        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()

        k_star = int(np.argmax(c))

        # Contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        contrib[lo:hi] = 2.0 * f[k_star - np.arange(lo, hi)]

        # Try structured moves: high-contrib → low-contrib
        donor = int(np.argmax(contrib))
        low_idx = np.argsort(contrib)[:max(10, n // 50)]

        improved = False
        for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
            for recv in low_idx[:8]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.03, 0.08, 0.15])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)

                    f = f_try
                    break

            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)

    c = _autocorr_fft(f)
    if c.max() < local_best:

        local_best_f = f.copy()

    for _ in range(count - len(starts)):

        if r < 0.3:

        print(f"Fallback uniform: {final_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
