    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s

Low-peak nonnegative step-function search using normalized-gradient mirror descent

learning-rate scheduling.

Key improvements over previous approach:
1. Normalized gradients (scale-free learning rate)

import numpy as np

import math


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))


def _make_multi_sine(n, amps):

    i = np.arange(n, dtype=np.float64)

    a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    return a / s

                grad[idx] += 2.0 * arr[k_int - idx]
        grad /= float(len(active))

        # Project to tangent space of simplex
        grad = grad - np.mean(grad)

        # Normalize gradient (scale-free learning rate)
        grad_norm = float(np.sqrt(np.mean(grad ** 2)))
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Multiplicative update (mirror descent on simplex)
        step = np.clip(eta * grad, -2.0, 2.0)
        arr_new = arr * np.exp(-step)
        arr_new = np.maximum(arr_new, 1e-30)
        s_new = float(arr_new.sum())
        if s_new < 0.01:
            break
        arr = arr_new / s_new

        # Periodic score check
        if it % period == 0:
            sc = _internal_score(arr)
            if sc < best_score - 1e-10:
                best = arr.copy()
                best_score = sc
                last_improve = it
                eta = min(max_eta, eta * 1.03)
                if verbose and it % (period * 20) == 0:
                    elapsed = time.monotonic() - t_start
                    print(f"    iter={it} score={best_score:.8f} eta={eta:.3f} "
                          f"K={K} max_b={max_b:.6e} t={elapsed:.1f}s")
            else:
                # Revert to best
                arr = best.copy()
                eta = max(min_eta, eta * 0.85)
                if it - last_improve > stall_limit:
                    # Random perturbation restart
                    restart_count += 1
                    if restart_count > max_restarts:
                        break
                    noise = np.random.randn(n) * 0.02
                    arr = best * (1.0 + noise)
                    arr = np.maximum(arr, 0.0)
                    s_arr = arr.sum()
                    if s_arr > 0.01:
                        arr = arr / s_arr
                    else:
                        arr = best.copy()
                    eta = 0.5
                    K = max(4, K // 2)
                    last_improve = it
                    stall_limit = max(300, stall_limit // 2)
                    if verbose:
                        print(f"    restart #{restart_count} at iter={it}, "
                              f"reducing K to {K}")

    if verbose:
        elapsed = time.monotonic() - t_start
        print(f"  _optimize done: best_score={best_score:.8f} iters={it} "
              f"time={elapsed:.1f}s restarts={restart_count}")

    return best, best_score


def _scan_init(deadline):
    """Quick scan of multi-harmonic initializations."""
    best_a = None
    best_score = np.inf

    lengths = [2048, 4096, 1024]
    amp_sets = [
        [0.78],
        [0.80, 0.10],
        [0.80, 0.15],
        [0.80, 0.20],
        [0.75, 0.10],
        [0.75, 0.15, 0.05],
        [0.80, 0.12, 0.06],
        [0.70, 0.20],
        [0.65, 0.25],
        [0.85, 0.05],
        [0.90, 0.0],
        [0.78, -0.10],
        [0.78, -0.15],
        [0.70, 0.30, -0.05],
    ]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        for amps in amp_sets:
            if time.monotonic() > deadline:
                break
            a = _make_multi_sine(n, amps)
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _make_multi_sine(2048, [0.78])
        best_score = _internal_score(best_a)

    return best_a, best_score


def _refine_amps(a, n, deadline):
    """Local refinement of the multi-harmonic amplitudes."""
    best_a = a.copy()
    best_score = _internal_score(best_a)

    # Try small perturbations in harmonic space
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)

    # Basis functions (odd harmonics)
    harmonics = []
    for k in range(5):
        harmonics.append(np.sin((2 * k + 1) * theta) / float(n))

    # Try adding small amounts of each harmonic
    for h_idx, h in enumerate(harmonics[:3]):
        for scale in [0.03, 0.06, 0.1]:
            for sign in [1, -1]:
                if time.monotonic() > deadline:
                    break
                candidate = a + sign * scale * h
                candidate = np.maximum(candidate, 0.0)
                s = candidate.sum()
                if s < 0.01:
                    continue
                candidate = candidate / s
                sc = _internal_score(candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = candidate.copy()

    return best_a, best_score


def _safe_resolve(x):
    """Sanitize sequence for evaluate_sequence."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence.
    
    Timeline (for 2400s budget):
    - Phase 1 (60s): Scan multi-harmonic initializations
    - Phase 2 (30s): Refine amplitudes locally
    - Phase 3 (~2200s): Main iterative optimization
    - Phase 4 (60s): Second optimization from perturbed start
    - Phase 5 (40s): Validation and return
    """
    np.random.seed(seed)
    t0 = time.monotonic()
    total = max(120.0, float(budget_s))
    hard_deadline = t0 + total - 20.0

    print(f"=== Search started, budget={total:.0f}s ===")

    # Phase 1: Scan initializations
    scan_deadline = t0 + 60.0
    scan_deadline = min(scan_deadline, hard_deadline - 100.0)
    best_a, best_score = _scan_init(scan_deadline)
    print(f"Phase 1: best init score={best_score:.8f} n={len(best_a)} "
          f"time={time.monotonic()-t0:.1f}s")

    # Phase 2: Refine amplitudes
    refine_deadline = t0 + 90.0
    refine_deadline = min(refine_deadline, hard_deadline - 80.0)
    best_a, best_score = _refine_amps(best_a, len(best_a), refine_deadline)
    print(f"Phase 2: refined score={best_score:.8f} time={time.monotonic()-t0:.1f}s")

    # Phase 3: Main optimization
    # Allocate most of the remaining time
    remaining_to_deadline = hard_deadline - time.monotonic()
    # Reserve 120s for phase 4 and 40s for validation
    opt_budget = remaining_to_deadline - 160.0
    opt_budget = max(opt_budget, 60.0)
    opt_deadline = time.monotonic() + opt_budget
    print(f"Phase 3: main optimization, budget={opt_budget:.0f}s, n={len(best_a)}")

    opt_a, opt_score = _optimize(best_a, opt_deadline, verbose=True)
    if opt_score < best_score:
        best_a = opt_a
        best_score = opt_score
    print(f"Phase 3 done: best_score={best_score:.8f} time={time.monotonic()-t0:.1f}s")

    # Phase 4: Second optimization attempt from a different start
    if time.monotonic() < hard_deadline - 60.0:
        n = len(best_a)
        i = np.arange(n, dtype=np.float64)
        theta = 2.0 * np.pi * (i + 0.5) / float(n)
        # Different perturbation
        pert = 0.15 * np.sin(3.0 * theta) + 0.05 * np.sin(5.0 * theta)
        start2 = best_a * (1.0 + pert)
        start2 = np.maximum(start2, 0.0)
        s2 = start2.sum()
        if s2 > 0.01:
            start2 = start2 / s2
            deadline4 = min(hard_deadline - 50.0, time.monotonic() + 100.0)
            opt2_a, opt2_score = _optimize(start2, deadline4, verbose=True)
            if opt2_score < best_score:
                best_a = opt2_a
                best_score = opt2_score
                print(f"Phase 4: improved to {best_score:.8f}")
            else:
                print(f"Phase 4: no improvement (got {opt2_score:.8f})")
        # Also try a completely different length
        if time.monotonic() < hard_deadline - 60.0:
            alt_n = 4096 if n == 2048 else 2048
            alt_a = _make_multi_sine(alt_n, [0.78, 0.1])
            sc_alt = _internal_score(alt_a)
            if sc_alt < best_score + 0.1:  # Only if competitive
                deadline4b = min(hard_deadline - 50.0, time.monotonic() + 80.0)
                opt3_a, opt3_score = _optimize(alt_a, deadline4b, verbose=False)
                if opt3_score < best_score:
                    best_a = opt3_a
                    best_score = opt3_score
                    print(f"Phase 4b (alt_n={alt_n}): improved to {best_score:.8f}")

    # Phase 5: Validation
    print(f"Phase 5: validating, internal_score={best_score:.8f}")
    final_arr = _safe_resolve(best_a)
    if final_arr is None:
        final_arr = np.ones(256, dtype=np.float64) / 256.0
        print("WARNING: Falling back to constant sequence!")

    try:
        real_score = float(evaluate_sequence(final_arr.tolist()))
        print(f"FINAL SCORE: {real_score:.8f} (n={final_arr.size})")
    except Exception as e:
        print(f"Validation error: {e}")
        final_arr = np.ones(256, dtype=np.float64) / 256.0
        real_score = float(evaluate_sequence(final_arr.tolist()))
        print(f"Fallback score: {real_score:.8f}")

    total_time = time.monotonic() - t0
    print(f"=== Search completed in {total_time:.1f}s ===")

    return [float(x) for x in final_arr]
