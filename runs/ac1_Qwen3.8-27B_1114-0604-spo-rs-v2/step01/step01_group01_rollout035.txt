        # linprog minimizes, so negate objective

            continue

        sum_g = float(np.sum(g))

            continue

        it += 1

    return best, best_score


def _run_mmd(p0, rng, time_limit, beta_start=30.0, beta_end=1000.0, verbose=False):
    p = _sanitize_vec(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _exact_score(p)

    m = int(p.size)
    floor = 1e-18 / float(m)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    beta = float(max(1.0, beta_start))
    beta_max = float(max(beta, beta_end))
    alpha = 0.1
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        d = m * c
        cur_exact = float(2.0 * np.max(d))
        w = _soft_weights(d, beta)
        h = _conv_valid_front(w, p)
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        maxh = float(np.max(h)) if h.size else 0.0

        if not np.isfinite(maxh) or maxh <= 1e-15:
            p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.05, m)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e = _exact_score(p_try)
                if np.isfinite(e) and e <= best_exact * 1.05:
                    p = p_try
                    cur_exact = e
            stall += 1
        else:
            h = h / maxh
            chosen = None
            ch_exact = cur_exact

            for a in (0.5 * alpha, alpha, 2.0 * alpha, 4.0 * alpha):
                if not (1e-4 <= a <= 20.0):
                    continue
                dec = np.exp(-a * h)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = float(p_try.sum())
                if s <= 0.0 or not np.isfinite(s):
                    continue
                p_try = p_try / s
                e = _exact_score(p_try)
                if np.isfinite(e) and e < ch_exact:
                    chosen = p_try
                    ch_exact = e

            if chosen is not None:
                p = chosen
                cur_exact = ch_exact
                stall = 0
                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and it % 100 == 0:
                        print(f"  MMD n={m} it={it} best={best_exact:.6f} beta={beta:.0f}")
                if cur_exact > best_exact * 1.20 and rng.random() < 0.3:
                    p = best.copy()
                    cur_exact = best_exact
            else:
                stall += 1
                alpha = max(1e-4, alpha * 0.6)
                if stall >= 15:
                    std = 0.05 if stall < 50 else min(0.4, 0.05 + 0.005 * stall)
                    p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, std, m)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e = _exact_score(p_try)
                        if np.isfinite(e) and e <= best_exact * 1.05:
                            p = p_try
                            cur_exact = e
                        else:
                            p = best.copy()
                            cur_exact = best_exact
                    stall = 0
                if stall > 80:
                    p = best.copy()
                    cur_exact = best_exact
                    alpha = 0.1
                    beta = max(beta, beta_start)
                    stall = 0

        it += 1
        if it % 20 == 0:
            beta = min(beta_max, beta * 1.15)

    return best, best_exact


def _local_block_search(p0, rng, time_limit):
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    n = int(p.size)
    if n <= 1:
        return p, _exact_score(p)

    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()
    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0
    factors = (0.35, 0.65, 0.85, 1.15, 1.35, 1.75, 2.5)

    while time.monotonic() < deadline:
        r = float(rng.random())
        if r < 0.30:
            w = int(rng.integers(1, max(2, n // 200 + 1)))
        elif r < 0.75:
            w = int(rng.integers(max(2, n // 200), max(3, n // 40 + 1)))
        else:
            low = max(3, n // 40)
            high = max(low + 1, min(n, n // 5 + 1))
            w = int(rng.integers(low, high + 1))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        base = p.copy()
        current_best = base
        current_score = cur_score

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= float(f)
            cand = _sanitize_vec(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best.copy()
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.05, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if np.isfinite(sc) and sc <= best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
            if stagnant > 400:
                p = best.copy()
                cur_score = best_score
                stagnant = 0

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 30.0:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget - 15.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    # Get parent if available
    try:
        parent_raw = height_sequence_1
    except NameError:
        parent_raw = None

    parent_arr = None
    if parent_raw is not None:
        try:
            parent_arr = _sanitize_vec(np.asarray(parent_raw, dtype=float))
        except Exception:
            parent_arr = None

    best_arr = None
    best_score = float('inf')

    def _update_best(arr, score):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(score) and score < best_score:
            best_arr = arr.copy()
            best_score = score

    # ---- Phase 1: Quick exploration (up to 150s) ----
    print(f"Phase 1: Exploration, budget={budget:.0f}s")
    rem = deadline - time.monotonic()
    if rem > 100.0:
        small_n = 512
        starts = []
        tb = _two_block_best(small_n)
        if tb is not None:
            starts.append(tb)
        if parent_arr is not None:
            sp = _resample(parent_arr, small_n)
            if sp is not None:
                starts.append(sp)
        for mode in (1, 2, 4, 5, 0):
            starts.append(_initial_sequence(small_n, rng, mode))

        for p0 in starts:
            rem = deadline - time.monotonic()
            if rem < 20.0:
                break
            talloc = min(15.0, max(3.0, rem - 5.0))
            p, sc = _run_mmd(p0, rng, talloc, beta_start=30.0, beta_end=400.0)
            _update_best(p, sc)

    print(f"Phase 1 done: best={best_score:.6f}, time={time.monotonic()-start_time:.0f}s")

    # ---- Phase 2: LP-based descent (main phase, up to 1200s) ----
    rem = deadline - time.monotonic()
    if rem > 300.0:
        target_n = 4096 if rem > 800.0 else 3072
        print(f"Phase 2: LP descent at n={target_n}, remaining={rem:.0f}s")

        # Build starting points for LP descent
        lp_starts = []
        if parent_arr is not None:
            sp = _resample(parent_arr, target_n)
            if sp is not None:
                lp_starts.append(sp)
        if best_arr is not None:
            sp = _resample(best_arr, target_n)
            if sp is not None:
                lp_starts.append(sp)
        tb = _two_block_best(target_n)
        if tb is not None:
            lp_starts.append(tb)
        for mode in (5, 1, 2):
            lp_starts.append(_initial_sequence(target_n, rng, mode))

        # Pick the best 2 starting points by score
        scored_starts = []
        for sp in lp_starts:
            if sp is not None:
                sc = _exact_score(sp)
                if np.isfinite(sc):
                    scored_starts.append((sc, sp))
        scored_starts.sort(key=lambda x: x[0])

        # Allocate time: best start gets 60%, second gets 40%
        lp_budget = min(rem - 100.0, 1200.0)
        fracs = [0.55, 0.45]

        for idx, (sc, sp) in enumerate(scored_starts[:2]):
            rem = deadline - time.monotonic()
            if rem < 100.0:
                break
            frac = fracs[idx] if idx < len(fracs) else 0.2
            talloc = max(30.0, lp_budget * frac)
            talloc = min(talloc, rem - 60.0)
            if talloc < 20.0:
                break
            print(f"  LP start {idx}: initial_score={sc:.6f}, talloc={talloc:.0f}s")
            p, sc = _lp_descent(sp, rng, talloc, K_init=80, verbose=True)
            _update_best(p, sc)
            print(f"  LP start {idx} done: score={sc:.6f}")

    print(f"Phase 2 done: best={best_score:.6f}, time={time.monotonic()-start_time:.0f}s")

    # ---- Phase 3: MMD polish (up to 500s) ----
    rem = deadline - time.monotonic()
    if rem > 150.0 and best_arr is not None:
        t1 = min(0.4 * rem, 500.0)
        print(f"Phase 3: MMD polish, t={t1:.0f}s")
        p, sc = _run_mmd(best_arr, rng, t1, beta_start=80.0, beta_end=2000.0, verbose=True)
        _update_best(p, sc)

    print(f"Phase 3 done: best={best_score:.6f}, time={time.monotonic()-start_time:.0f}s")

    # ---- Phase 4: Second LP round on the polished result (up to 400s) ----
    rem = deadline - time.monotonic()
    if rem > 200.0 and best_arr is not None:
        t2 = min(0.35 * rem, 400.0)
        print(f"Phase 4: LP round 2, t={t2:.0f}s")
        p, sc = _lp_descent(best_arr, rng, t2, K_init=100, verbose=True)
        _update_best(p, sc)

    print(f"Phase 4 done: best={best_score:.6f}, time={time.monotonic()-start_time:.0f}s")

    # ---- Phase 5: Coordinate search (up to 200s) ----
    rem = deadline - time.monotonic()
    if rem > 80.0 and best_arr is not None:
        t3 = min(0.4 * rem, 200.0)
        print(f"Phase 5: Block search, t={t3:.0f}s")
        p, sc = _local_block_search(best_arr, rng, t3)
        _update_best(p, sc)

    # ---- Phase 6: Final MMD with very high beta ----
    rem = deadline - time.monotonic()
    if rem > 80.0 and best_arr is not None:
        t4 = min(0.3 * rem, 150.0)
        print(f"Phase 6: Final MMD, t={t4:.0f}s")
        p, sc = _run_mmd(best_arr, rng, t4, beta_start=200.0, beta_end=5000.0, verbose=True)
        _update_best(p, sc)

    print(f"All phases done: best_score={best_score:.6f}, time={time.monotonic()-start_time:.0f}s")

    # ---- Final validation with evaluate_sequence ----
    if best_arr is None:
        best_arr = _initial_sequence(3072, rng, 1)
        best_score = _exact_score(best_arr)

    final_list = [float(x) for x in best_arr]
    try:
        sc = float(evaluate_sequence(final_list))
        if np.isfinite(sc):
            print(f"evaluate_sequence confirms: {sc:.6f}")
    except Exception:
        pass

    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float)
    else:
        out = out / s

    return [float(x) for x in out]
