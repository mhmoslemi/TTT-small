```python
import numpy as np
import time

"""
Multi-phase optimizer for minimizing the peak autocorrelation of a non-negative sequence.

Algorithm:
1. Initialize from parent sequence (resampled to target length) or structured starts.
2. Phase 1: Minimize autocorrelation variance (smooth objective) using Adam.
3. Phase 2: Minimize soft-peak (logsumexp) with decreasing temperature using Adam.
4. Phase 3: Local polish via coordinate descent.
5. Return best sequence found within time budget.
"""

# Global state
_N = 0
_NFFT = 0


def _init_state(n):
    global _N, _NFFT
    _N = int(n)
    _NFFT = 1 << (3 * _N - 2).bit_length()


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _fft_conv(a):
    n = _N
    fa = np.fft.rfft(a, _NFFT)
    c = np.fft.irfft(fa * fa, _NFFT)[:2 * n - 1].real
    return c


def _score_normalized(a):
    """Score assuming sum(a)=1: 2n * max(c)."""
    n = _N
    c = _fft_conv(a)
    return float(2.0 * n * np.max(c))


def _grad_variance(z):
    """Gradient of sum(c^2) where a=softmax(z), c=a*a."""
    n = _N
    a = _softmax(z)
    c = _fft_conv(a)
    fc = np.fft.rfft(c, _NFFT)
    fa = np.fft.rfft(a, _NFFT)
    T = np.fft.irfft(fc * np.conj(fa), _NFFT)[:n]
    grad_a = 2.0 * T
    s = np.dot(grad_a, a)
    grad_z = a * (grad_a - s)
    val = float(np.dot(c, c))
    return val, grad_z


def _grad_softpeak(z, tau):
    """Gradient of tau*logsumexp(c/tau) where a=softmax(z), c=a*a."""
    n = _N
    a = _softmax(z)
    c = _fft_conv(a)
    c_max = float(np.max(c))
    shifted = np.clip((c - c_max) / tau, -700.0, 0.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not np.isfinite(se):
        return c_max, np.zeros(n)
    p = e / se
    fp = np.fft.rfft(p, _NFFT)
    fa = np.fft.rfft(a, _NFFT)
    T = np.fft.irfft(fp * np.conj(fa), _NFFT)[:n]
    s = np.dot(T, a)
    grad_z = a * (T - s)
    val = c_max + tau * np.log(se)
    return val, grad_z


def _adam_step(z, grad, m, v, t, lr, beta1, beta2, eps, clip_norm=5.0):
    gnorm = np.linalg.norm(grad)
    if gnorm > clip_norm:
        grad = grad * (clip_norm / gnorm)
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v


def _resample_to_length(a_src, n_target):
    """Resample a_src to length n_target via linear interpolation in log-space."""
    a_src = np.asarray(a_src, dtype=np.float64).ravel()
    a_src = np.clip(a_src, 1e-30, 1000.0)
    log_a = np.log(a_src)
    x_old = np.linspace(0.0, 1.0, len(a_src))
    x_new = np.linspace(0.0, 1.0, n_target)
    log_new = np.interp(x_new, x_old, log_a)
    a_new = np.exp(log_new - np.max(log_new))
    return a_new


def _make_structured_start(n, rng):
    """Generate a collection of structured starting sequences."""
    starts = []
    u = np.linspace(0.0, 1.0, n)
    x = 2.0 * u - 1.0

    # Two-level blocks
    half = n // 2
    for ratio in (1.3, 1.7, 2.2, 3.0, 4.0):
        f = np.empty(n)
        f[:half] = 1.0
        f[half:] = ratio
        starts.append(f)

    # Monotone linear
    for A in (0.3, 0.6, 0.9, 1.2, 1.5):
        starts.append(1.0 + A * x)

    # Smooth step (sigmoid)
    for A in (0.5, 1.0, 1.5, 2.0):
        for w in (0.05, 0.1, 0.2, 0.4):
            sig = 1.0 / (1.0 + np.exp(-(x / w)))
            starts.append(1.0 + A * (sig - 0.5))

    # Cosine hump
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        starts.append(1.0 + alpha * np.cos(np.pi * u))

    # Quadratic
    for A in (0.3, 0.7, 1.2, 2.0):
        starts.append(1.0 + A * (u * u * 4.0 - 1.0))

    # Sinusoidal
    for m in (1, 2, 3, 5):
        for A in (0.2, 0.5, 1.0):
            starts.append(1.0 + A * np.sin(2.0 * np.pi * m * u))

    # Random smooth
    for _ in range(8):
        f = np.ones(n)
        for m in range(1, 10):
            f += 0.15 * rng.standard_normal() * np.sin(2.0 * np.pi * m * u + rng.standard_normal())
        starts.append(f)

    return [s for s in starts if np.min(s) > 0.0]

def _log_space(a):
```
