Optimized step function search using adaptive multiplicative gradient descent,
projected subgradient polishing, and Frank-Wolfe hybrid steps.
Combines FFT-based soft gradients with sharp subgradient refinement and
simplex projection to minimize 2*n*max(conv(a,a))/(sum(a))^2.

import math
import numpy as np


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < eps:
        return np.ones_like(a) / len(a)
    return a / s


def _autocorr_fft(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _subgrad_fft(a, w, Lsg):
    n = len(a)
    W = np.fft.rfft(w, Lsg)
    A = np.fft.rfft(a, Lsg)
    g = np.fft.irfft(W * np.conj(A), Lsg)[:n]
    np.maximum(g, 0.0, out=g)
    return 2.0 * g


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    c = _autocorr_fft(a, L, m)
    return 2.0 * n * float(c.max()), c

    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / s


def _random_init(n, rng, mode="smooth"):
    if mode == "uniform":
        return _normalize(np.ones(n))
    elif mode == "random":
        return _normalize(rng.random(n) + 0.01)
    elif mode == "smooth_random":
        z = rng.normal(0, 0.3, n)
        ksize = max(3, n // 100)
        kernel = np.ones(ksize) / ksize
        z = np.convolve(z, kernel, mode="same")
        z += 0.5 * rng.normal(0, 0.2, n)
        return _normalize(np.exp(z - z.max()))
    elif mode == "bell":
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        z = -((t - center) / width) ** 2
        z += rng.normal(0, 0.1, n)
        return _normalize(np.exp(z - z.max()))
    else:
        return _normalize(rng.random(n) + 0.01)


def _init_from_base(n, rng):
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if len(base) == n and np.all(np.isfinite(base)) and np.all(base >= 0):
            s = base.sum()
            if s > 0.01:
                return _normalize(base.copy())
    except (NameError, Exception):
        pass
    return None


def _mul_step(a, g_norm, eta):
    a_new = a * np.exp(-eta * g_norm)
    return _normalize(a_new)


def _fw_step(a, g, t):
    j_min = int(np.argmin(g))
    a_new = (1.0 - t) * a.copy()
    a_new[j_min] += t
    return _normalize(a_new)


def _proj_subgrad_step(a, g, eta):
    g_max = float(np.max(np.abs(g)))
    if g_max < 1e-300:
        return a
    g = g / g_max
    a_new = a - eta * g
    a_new = _project_simplex(a_new)
    return a_new


def _try_exact(seq):
    try:
        val = evaluate_sequence(seq)
        if val is not None and np.isfinite(val):
            return float(val)
    except Exception:
        pass
    return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if budget < 10.0:
        budget = 10.0

    rng = np.random.default_rng(seed)
    n = 6121
    m = 2 * n - 1
    Lc = _next_pow2(2 * n - 2)
    Lsg = _next_pow2(3 * n - 2)

    # Initialize
    a = _init_from_base(n, rng)
    if a is None:
        a = _random_init(n, rng, "smooth_random")

    best_a = a.copy()
    best_score, _ = _score(a, Lc, m, n)
    print(f"Initial score (internal): {best_score:.6f}")

    # Parameters
    eta = 0.08
    beta = 8.0
    it = 0
    no_improve = 0
    last_exact = start_time
    best_exact = best_score

    phase1_end = start_time + 0.45 * budget
    phase2_end = start_time + 0.75 * budget
    final_deadline = start_time + budget - 40.0

    print(f"Phase 1 starts at {start_time - start_time:.1f}s, ends at {phase1_end - start_time:.1f}s")

    # Phase 1: Soft multiplicative gradient descent
    while time.monotonic() < phase1_end:
        c = _autocorr_fft(a, Lc, m)
        current_score = 2.0 * n * float(c.max())

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if time.monotonic() - last_exact > 90:
            seq = [float(x) for x in best_a]
            val = _try_exact(seq)
            if val < best_exact:
                best_exact = val
                print(f"  Exact: {val:.6f} at t={time.monotonic()-start_time:.0f}s")
            last_exact = time.monotonic()

        w = _softmax_weights(c, beta)
        g = _subgrad_fft(a, w, Lsg)
        g_max = float(g.max())
        if g_max < 1e-300:
            break
        g_norm = g / g_max

        # Try multiplicative step with backtracking
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            step = eta * factor
            if step < 1e-6:
                continue
            a_new = _mul_step(a, g_norm, step)
            c_new = _autocorr_fft(a_new, Lc, m)
            sc_new = 2.0 * n * float(c_new.max())
            if sc_new < current_score - 1e-14:
                a = a_new
                accepted = True
                break

        if not accepted:
            # Try Frank-Wolfe
            t_fw = min(0.3, 2.0 / (it + 2.0))
            a_fw = _fw_step(a, g_norm, t_fw)
            c_fw = _autocorr_fft(a_fw, Lc, m)
            sc_fw = 2.0 * n * float(c_fw.max())
            if sc_fw < current_score:
                a = a_fw
                accepted = True

        if accepted:
            eta = min(eta * 1.03, 0.5)
        else:
            eta = max(eta * 0.92, 0.01)

        # Temperature schedule
        progress = min(1.0, it / 5000.0)
        target_beta = 8.0 + 60.0 * progress
        beta = 0.9 * beta + 0.1 * target_beta

        # Restart if stuck
        if no_improve > 300:
            mix = rng.uniform(0.05, 0.2)
            noise = rng.normal(0, 0.3, n)
            a = _normalize((1 - mix) * best_a + mix * np.exp(noise))
            beta = max(5.0, beta * 0.7)
            eta = 0.08
            no_improve = 0
            it_reset = it

        it += 1

    print(f"Phase 1 done. Best internal: {best_score:.6f}, it={it}")

    # Phase 2: Sharper optimization with mixed strategies
    beta = 50.0
    eta = 0.03
    no_improve = 0
    it2 = 0

    while time.monotonic() < phase2_end:
        c = _autocorr_fft(a, Lc, m)
        current_score = 2.0 * n * float(c.max())

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if time.monotonic() - last_exact > 90:
            seq = [float(x) for x in best_a]
            val = _try_exact(seq)
            if val < best_exact:
                best_exact = val
                print(f"  Exact: {val:.6f} at t={time.monotonic()-start_time:.0f}s")
            last_exact = time.monotonic()

        w = _softmax_weights(c, beta)
        g = _subgrad_fft(a, w, Lsg)
        g_max = float(g.max())
        if g_max < 1e-300:
            break
        g_norm = g / g_max

        # Alternate strategies
        if it2 % 3 == 0:
            # Multiplicative
            a_new = _mul_step(a, g_norm, eta)
        elif it2 % 3 == 1:
            # Projected subgradient
            eta_pg = 0.5 / math.sqrt(it2 + 1.0)
            a_new = _proj_subgrad_step(a, g, eta_pg)
        else:
            # Hybrid: small multiplicative + FW correction
            a_tmp = _mul_step(a, g_norm, eta * 0.5)
            t_fw = 0.5 / (it2 + 1.0)
            a_new = _fw_step(a_tmp, g_norm, t_fw)

        c_new = _autocorr_fft(a_new, Lc, m)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < current_score - 1e-14:
            a = a_new
            eta = min(eta * 1.01, 0.2)
        else:
            eta = max(eta * 0.95, 0.005)

        # Adaptive beta
        if no_improve > 100:
            beta = min(beta * 1.5, 200.0)
            if no_improve > 400:
                mix = rng.uniform(0.02, 0.1)
                noise = rng.normal(0, 0.15, n)
                a = _normalize((1 - mix) * best_a + mix * np.exp(noise))
                beta = 30.0
                eta = 0.03
                no_improve = 0

        it2 += 1

    print(f"Phase 2 done. Best internal: {best_score:.6f}, it2={it2}")

    # Phase 3: Fine polishing with very sharp gradients
    a = best_a.copy()
    beta = 100.0
    eta = 0.01
    no_improve = 0
    it3 = 0

    while time.monotonic() < final_deadline:
        c = _autocorr_fft(a, Lc, m)
        current_score = 2.0 * n * float(c.max())

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Use exact subgradient (sharp)
        k_star = int(np.argmax(c))
        g_exact = np.zeros(n)
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                g_exact[j] = 2.0 * a[k]

        # Also compute soft version for blending
        w = _softmax_weights(c, beta)
        g_soft = _subgrad_fft(a, w, Lsg)

        # Blend exact and soft
        blend = min(1.0, it3 / 500.0)
        g_blend = (1 - blend) * g_soft + blend * g_exact
        g_max = float(np.max(g_blend))
        if g_max < 1e-300:
            break
        g_norm = g_blend / g_max

        # Very small multiplicative step
        step = eta / (1.0 + 0.01 * no_improve)
        a_new = _mul_step(a, g_norm, step)

        # Also try projected subgradient
        eta_pg = 0.3 / math.sqrt(it3 + 1.0)
        a_pg = _proj_subgrad_step(a, g_blend, eta_pg)

        c_new = _autocorr_fft(a_new, Lc, m)
        sc_new = 2.0 * n * float(c_new.max())

        c_pg = _autocorr_fft(a_pg, Lc, m)
        sc_pg = 2.0 * n * float(c_pg.max())

        if sc_new <= sc_pg and sc_new < current_score - 1e-15:
            a = a_new
        elif sc_pg < current_score - 1e-15:
            a = a_pg
        else:
            # Try FW step
            t_fw = 1.0 / (it3 + 1.0)
            a_fw = _fw_step(a, g_norm, t_fw)
            c_fw = _autocorr_fft(a_fw, Lc, m)
            sc_fw = 2.0 * n * float(c_fw.max())
            if sc_fw < current_score - 1e-15:
                a = a_fw
            # else: no improvement, do nothing

        if no_improve > 200:
            # Tiny perturbation
            noise = rng.normal(0, 0.01, n)
            a = _normalize(best_a * np.exp(noise))
            no_improve = 0
            it3 = 0

        it3 += 1

    print(f"Phase 3 done. Best internal: {best_score:.6f}, it3={it3}")

    # Final: Try multiple candidates with exact evaluation
    candidates = [best_a, a]

    # Add perturbations of best
    for amp in (0.005, 0.01, 0.02, 0.05):
        for _ in range(3):
            noise = rng.normal(0, amp, n)
            cand = _normalize(best_a * np.exp(noise))
            candidates.append(cand)

    # Try different n values quickly
    for n_alt in (5000, 7000, 8000):
        if time.monotonic() > final_deadline - 10:
            break
        Lc_alt = _next_pow2(2 * n_alt - 2)
        m_alt = 2 * n_alt - 1
        # Resample best_a to new length
        t_old = np.linspace(0, 1, len(best_a))
        t_new = np.linspace(0, 1, n_alt)
        a_alt = np.interp(t_new, t_old, best_a)
        a_alt = _normalize(a_alt)
        for _ in range(20):
            c_alt = _autocorr_fft(a_alt, Lc_alt, m_alt)
            sc_alt = 2.0 * n_alt * float(c_alt.max())
            if sc_alt < 2.0:
                w_alt = _softmax_weights(c_alt, 50.0)
                # Quick gradient step
                Lsg_alt = _next_pow2(3 * n_alt - 2)
                g_alt = _subgrad_fft(a_alt, w_alt, Lsg_alt)
                g_alt /= max(1e-300, g_alt.max())
                a_alt = _mul_step(a_alt, g_alt, 0.05)
        candidates.append(a_alt)

    best_seq = None
    best_val = float('inf')

    for cand in candidates:
        if time.monotonic() > start_time + budget - 5:
            break
        seq = [float(x) for x in cand]
        val = _try_exact(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"  Candidate evaluated: {val:.6f}")

    if best_seq is None:
        best_seq = [float(x) for x in best_a]
        best_val = _try_exact(best_seq)

    print(f"Final best score: {best_val:.6f}")
    return best_seq
