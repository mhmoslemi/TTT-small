import numpy as np
import time
from scipy.signal import correlate

def _convolve(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()  # Wait, 2*n-1 elements, need size >= 2*n-1
    # Actually: 1 << (2*n-1).bit_length() gives the next power of 2 >= 2*n-1
    # But we need size >= (n-1) + (n-1) + 1 = 2*n-1
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) - (cssv - 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _subgradient_topK(c, f, K=5):
    """Compute subgradient as average of top-K peak gradients."""
    n = len(f)
    # Find top-K lags
    top_k = np.argsort(c)[::-1][:K]
    # For each peak lag k, grad_k[j] = 2*f[k-j] if 0 <= k-j < n
    grad = np.zeros(n)
    for k in top_k:

        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        # f[k-j] for j in [j_min, j_max] is f[k-j_max] to f[k-j_min]
        # i.e., f[k-j_min : k-j_max+1] reversed...

        # This is f[k-j_max : k-j_min+1] in reverse order.
        # In numpy: f[k-j_min : k-j_max-1, -1] or f[k-j_max : k-j_min+1][::-1]
        idx_start = k - j_min  # = k - max(0, k-n+1)
        idx_end = k - j_max + 1  # = k - min(k, n-1) + 1

    return np.fft.irfft(F * F, size)[:target]


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

    candidates['u_shape'] = 1.0 + 5.0 * ((j - center) / center) ** 2
    candidates['bimodal'] = (np.exp(-((j - 0.3 * n)) ** 2 / (0.04 * n * n)) +
                            np.exp(-((j - 0.7 * n)) ** 2 / (0.04 * n * n)) + 0.01)
    candidates['sine'] = 1.0 + 0.3 * np.cos(6.0 * np.pi * j / n)
    candidates['exp_ramp'] = np.exp(0.5 * j / n)

    start_list = []
    for name, f0 in candidates.items():
        f_c = np.maximum(f0, 0.0)
        s_c = f_c.sum()
        if s_c < 1e-10:
            continue
        f_c = f_c / s_c
        sc = 2.0 * n * _convolve(f_c).max()
        start_list.append((sc, name, f_c))

    start_list.sort(key=lambda x: x[0])
    print("Start scores:")
    for sc, name, _ in start_list:
        print(f"  {name}: {sc:.6f}")

    best_f = start_list[0][2].copy()
    best_score = start_list[0][0]

    # Phase 1: Subgradient from top starts with various K
    n_good_starts = min(3, len(start_list))
    p1_total_time = (budget_s - 90.0) * 0.65
    time_per_start = p1_total_time / n_good_starts

    for si in range(n_good_starts):
        sc_s, name_s, f_s = start_list[si]
        if time.monotonic() > deadline - 90.0:
            break
        t_alloc_start = time_per_start * (1.0 - 0.15 * si)
        start_dl = time.monotonic() + t_alloc_start

        print(f"\nPhase 1 from '{name_s}' (score={sc_s:.6f}, alloc={t_alloc_start:.0f}s):")

        K_list = [3, 7, 15, 30]
        t_per_K = t_alloc_start / len(K_list)

        for K in K_list:
            if time.monotonic() > start_dl - 5.0:
                break
            K_dl = min(time.monotonic() + t_per_K, start_dl - 3.0)
            print(f"  K={K}, time={t_per_K:.0f}s")

            f_opt, sc_opt = _phase1_subgradient(
                f_s, n, K_dl, K, alpha_init=1.0, beta=0.9
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"    New best: {best_score:.6f}")

    # Phase 2: L-BFGS polish
    print(f"\nPhase 2: L-BFGS polish (best={best_score:.6f})")
    p2_deadline = min(time.monotonic() + (budget_s - 60.0) * 0.35, deadline)

    T_values = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for T_val in T_values:
        if time.monotonic() > p2_deadline - 5.0:
            break
        f_opt, sc_opt = _phase2_lbfgs(best_f, n, p2_deadline, T_val)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  T={T_val}: new best={best_score:.6f}")
        else:
            # Try from a slightly perturbed version
            if time.monotonic() < p2_deadline - 20.0:
                f_pert = _project_simplex(best_f + 0.01 * rng.standard_normal(n) / np.sqrt(n))
                f_opt, sc_opt = _phase2_lbfgs(f_pert, n, p2_deadline, T_val)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_f = f_opt.copy()
                    print(f"  T={T_val} (perturbed): new best={best_score:.6f}")
                    break

    # Final verification
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback to initial
        best_f = np.clip(f_init, 0.0, 1000.0)
        final_score = evaluate_sequence(best_f.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL SCORE: {best_score:.6f} (elapsed: {elapsed:.0f}s) ===")
    return [float(x) for x in best_f]
