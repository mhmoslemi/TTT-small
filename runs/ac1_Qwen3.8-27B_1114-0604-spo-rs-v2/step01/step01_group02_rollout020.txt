```python
Hybrid optimizer combining LP-based refinement, gradient descent, and structured initialization
to minimize f(a) = 2n * max(convolve(a,a)) / (sum(a))^2.
"""

from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    n = len(a)

        i_end = min(k, n - 1)

def topk_soft_gradient(a, c, k, temp_frac=0.1):
    """Compute soft-max weighted gradient of the max-convolution surrogate."""
    n = len(a)
    S = float(a.sum())
    
    # Get top k convolution values
    k_act = min(k, len(c))
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        vals = c[idx]
        wts = np.array([1.0])
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
        vals = c[idx]
        order = np.argsort(vals)[::-1]
        idx = idx[order]
        vals = vals[order]
        top = float(vals[0])
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    
    M_w = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad, M_w

def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)
```

```python
import numpy as np

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def soft_topk_gradient(a, c, k, temp_frac):

    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    temp = temp_frac * top if top > 1e-12 else 1.0
    wts = np.exp((vals - top) / temp)
    wts = wts / wts.sum()

    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:

        j_start = max(0, ki - n + 1)

def adam_update(a, g, m, v, t, lr):

def lp_solve(a, M, n_constraints=None):
    """
    Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k (or top k).
    Returns g or None.
    """
    n = len(a)
    S = a.sum()
    
    # Build constraint matrix
    if n_constraints is None or n_constraints >= 2 * n - 1:
        # Use all constraints
        k_indices = list(range(2 * n - 1))
    else:
        # Use top n_constraints lags
        c = fast_self_conv(a)
        k_indices = list(np.argsort(c)[-n_constraints:])
    
    n_con = len(k_indices)
    A = np.zeros((n_con, n))
    for row, k in enumerate(k_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A[row, idx] = a[k - idx]
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=np.full(n_con, M),
                     bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        g = np.clip(result.x, 0, None)
        if g.sum() > S * 1.0005:
            return g
    return None
```

**Iterative Proximal Gradient (FISTA) on the max-convolution**:
- Objective: f(a) = max_k (a*a)[k] (scaled appropriately)
- This is the max of convex quadratic functions
- We can use a smoothed approximation: f_ε(a) = ε * log(sum_k exp((a*a)[k]/ε))
- The gradient of f_ε is (1/sum) * sum_k exp(conv[k]/ε) * 2*a[k-j] (weighted sum of individual gradients)
- This is essentially what the top-k soft-max is doing, but with a log-sum-exp instead of top-k

The advantage of log-sum-exp is that it uses ALL the convolution values (not just top-k), giving a smoother approximation. The disadvantage is that computing exp(conv[k]/ε) for all k might overflow.

We can handle this with the standard trick: exp(x/ε) = exp((x - max)/ε) * exp(max/ε), and since we're taking a weighted average, the constant factor cancels.

```python
def logsumexp_gradient(a, c, eps):
    """Gradient of the log-sum-exp approximation to max(c)."""
    n = len(a)
    S = float(a.sum())
    
    # Compute softmax weights (log-sum-exp trick)
    c_shifted = c - c.max()
    wts = np.exp(c_shifted / eps)
    wts = wts / wts.sum()
    
    # Weighted gradient
    grad_c = np.zeros(n)
    for k in range(len(c)):

        if wi < 1e-12:

        j_end = min(k, n - 1)

    M_ls = eps * np.log(wts.sum() * np.exp((c.max() - c.max()) / eps))  # This is just the logsumexp
    # Actually: logsumexp(c/eps) = c.max()/eps + log(sum exp((c - c.max())/eps))
    # d/d a: (1/eps) * sum_k exp((c_k - c_max)/eps) / sum_l exp((c_l - c_max)/eps) * dc_k/da

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad, M_w
```

```python
"""
Hybrid optimizer minimizing f(a) = 2n * max(convolve(a,a)) / (sum(a))^2.
Uses adaptive top-k gradient descent with soft-max smoothing, periodic LP-based
refinement (solving max sum(g) s.t. g>=0, ||a*g||_inf <= M), multi-length search,
and structured initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    """Compute the objective from convolution and sum."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def soft_topk_gradient(a, c, k, temp_frac):
    """Compute gradient using soft-max weighted top-k peaks."""
    n = len(a)
    S = float(a.sum())
    k_act = min(k, len(c))
    if k_act <= 0:
        return np.zeros(n), 0.0
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act

    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]

    M_w = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad, M_w


def adam_update(a, g, m, v, t, lr):
    """Adam optimizer step (in-place on a, m, v)."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def lp_refine(a, M, n_top=None):
    """
    LP refinement: find g >= 0 maximizing sum(g) subject to (a*g)[k] <= M.
    Uses top n_top constraints if specified, else all.
    Returns refined g or None.
    """
    n = len(a)
    S = a.sum()
    total_cons = 2 * n - 1

    if n_top is None or n_top >= total_cons:
        k_list = np.arange(total_cons)
    else:
        c = fast_self_conv(a)
        k_list = np.argsort(c)[-n_top:]

    n_con = len(k_list)
    A = np.zeros((n_con, n))
    for row, k in enumerate(k_list):
        i_start = max(0, int(k) - n + 1)
        i_end = min(int(k), n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A[row, idx] = a[int(k) - idx]

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(n_con, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            g = np.clip(result.x, 0.0, None)
            g_sum = g.sum()
            if g_sum > S * 1.001:
                return g
    except Exception as e:
        print("    LP error: %s" % str(e))
    return None


def make_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform baseline
    inits.append("uniform", np.ones(n))

    # Cosine modulated
    for freq in [1, 2, 4, 7, 13]:
        for amp in [0.3, 0.5, 0.8]:
            x = np.linspace(0.0, 1.0, n)
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
            inits.append(("cos_%d_%s" % (freq, amp), np.clip(a, 0.01, 1000.0)))

    # Quadratic / ramp
    for alpha in [0.5, 1.0, 2.0]:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + alpha * (2.0 * x - 1.0)
        inits.append(("ramp_%s" % alpha, np.clip(a, 0.01, 1000.0)))

    # Gaussian bumps
    for _ in range(3):
        a = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            center = rng.integers(0, n)
            width = rng.uniform(n / 15.0, n / 4.0)
            amp = rng.uniform(-0.6, 0.6)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        inits.append(("gauss_%d" % rng.integers(10000), np.clip(a, 0.01, 1000.0)))

    # Block patterns
    for bl in [4, 8, 16, 64]:
        nb = n // bl + 1
        vals = 0.4 + 0.8 * rng.random(nb)
        a = np.repeat(vals, bl)[:n]
        inits.append(("block_%d" % bl, np.clip(a, 0.01, 1000.0)))

    # Random
    inits.append("rand1", np.clip(0.5 + 0.5 * rng.random(n), 0.01, 1000.0))
    inits.append("rand2", np.clip(1.0 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))

    # Symmetric
    half = n // 2
    a = 1.0 + 0.4 * rng.standard_normal(half)
    a = np.clip(a, 0.1, 5.0)
    full = np.concatenate([a, a[::-1]])[:n]
    inits.append("sym", full)

    return inits


def optimize_one(a0, deadline, rng, init_lr, use_lp, lp_interval, lp_topk,
                 tag=""):
    """
    Run adaptive gradient descent with optional LP refinement on a single sequence.
    Returns (best_score, best_a).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-8:
        a = np.ones(n) * 0.1

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    lr = init_lr

    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    patience = 350
    it = 0

    target_sum = a.sum()

    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic LP refinement
        if use_lp and it > 50 and it % lp_interval == 0:
            if time.monotonic() < deadline - 90:
                M = float(np.max(c))
                t0 = time.monotonic()
                g = lp_refine(a, M, n_top=lp_topk)
                dt = time.monotonic() - t0
                if g is not None:
                    alpha = 0.3 + 0.3 * rng.random()
                    g_rescaled = g * (target_sum / g.sum())
                    a = (1.0 - alpha) * a + alpha * g_rescaled
                    a = np.clip(a, 0.0, 1000.0)
                    m[:] = 0.0
                    v[:] = 0.0
                    t_count = 0
                    no_improve = 0
                    if tag:
                        print("  [%s] LP it=%d t=%.1fs ratio=%.3f" % (
                            tag, it, dt, g.sum() / target_sum))
                else:
                    if tag and it % (lp_interval * 5) == lp_interval:
                        print("  [%s] LP it=%d no improvement" % (tag, it))

        # Restart mechanism
        if no_improve >= patience:
            scale = 0.02 + 0.06 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-6:
                a = best_a.copy()
            m[:] = 0.0
            v[:] = 0.0
            t_count = 0
            lr = max(0.003, lr * 0.5)
            no_improve = 0
            patience = max(80, patience * 3 // 4)
            continue

        # Adaptive parameters
        progress = min(1.0, it / 4000.0)
        topk = int(8 + 72 * progress)
        temp_frac = 0.25 * (1.0 - progress) + 0.05

        g, _ = soft_topk_gradient(a, c, topk, temp_frac)
        t_count += 1
        adam_update(a, g, m, v, t_count, lr)
        a = np.clip(a, 0.0, 1000.0)

        # Rescale to maintain target sum (keeps values in reasonable range)
        cur_sum = a.sum()
        if cur_sum > 1e-8:
            a *= target_sum / cur_sum
            a = np.clip(a, 0.0, 1000.0)

        # LR decay
        if it > 500 and it % 800 == 0:
            lr = max(0.001, lr * 0.75)

        it += 1

    return best_score, best_a

    out = [float(x) for x in best_a]

        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float('inf')

    print("=== FINAL: internal=%.6f, verified=%.6f, n=%d, time=%.1fs ===" % (
        best_score, verified, len(out), time.monotonic() - start))

    return out
```
