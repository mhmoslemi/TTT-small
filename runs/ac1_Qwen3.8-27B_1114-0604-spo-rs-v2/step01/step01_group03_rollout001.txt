"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based evaluation with a sharpened subgradient (top-k focus on peaks),
adaptive multiplicative updates, peak-equalization moves, and multi-n exploration.
"""

import numpy as np
import time
import math


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _fft_conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    return float(2.0 * n * np.max(c))


def _score_a(a, L, m, n):
    return _score(_fft_conv_self(a, L, m), n)


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.4, n)
    z += rng.normal(0.0, 0.5) * x
    kmax = int(rng.integers(2, 9))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.5) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.12, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(7, dtype=np.float64) / 7.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _structured_initials(n, rng):
    out = []
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine
    a = (1.0 + np.cos(2.0 * np.pi * t)) / 2.0
    a = np.maximum(a, 1e-15)
    out.append(_normalize(a))
    # (1+cos)^p for various p
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (1.0 + np.cos(2.0 * np.pi * t)) / 2.0
        a = np.power(np.maximum(a, 1e-15), p)
        out.append(_normalize(a))
    # Gaussian centered
    for sig in (0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        a = np.exp(-0.5 * ((t - 0.5) / sig) ** 2)
        out.append(_normalize(a))
    # Von Mises
    for kappa in (2.0, 4.0, 6.0, 8.0, 12.0, 20.0):
        a = np.exp(kappa * np.cos(2.0 * np.pi * t))
        out.append(_normalize(a))
    # Two peaks
    for c in (0.2, 0.25, 0.3, 0.35):
        a = np.exp(-0.5 * ((t - c) / 0.12) ** 2) + np.exp(-0.5 * ((t - (1-c)) / 0.12) ** 2)
        out.append(_normalize(a))
    # Anti-convex (U-shape)
    for alpha in (1.0, 2.0, 3.0, 5.0):
        a = np.power(np.minimum(t, 1.0 - t) / 0.5, alpha)
        out.append(_normalize(a))
    # Random priors
    for _ in range(8):
        out.append(_random_prior(n, rng))
    return out


def _topk_gradient(b, n, k_top, temp):
    """Compute gradient direction using top-k positions with softmax weighting."""
    m = len(b)
    v = n * b.copy()
    vmax = float(np.max(v))
    # Use softmax with temperature to weight positions
    z = temp * (v - vmax)
    z = np.clip(z, -30.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        w = np.full(m, 1.0 / m)
    else:
        w /= ws
    # Additional concentration: boost top-k weights
    if k_top < m and k_top >= 1:
        idx = np.argsort(v)[-k_top:]
        boost = 5.0
        w[idx] *= boost
        w /= float(w.sum())
    # Gradient: g_i = 2 * sum_k w_k * b_contribution
    # The contribution of a_i to b_k is a_{k-i} (for valid k-i)
    # So g = 2 * (w * a_reversed) correlated appropriately
    # g_i = 2 * sum_k w_k * a_{k-i} for 0<=k-i<n
    # This is a cross-correlation of w with reversed a
    # Using FFT: g = 2 * ifft(rfft(a_reversed) * rfft(w))
    return w, vmax


def _compute_direction(a, w, L, n):
    """Compute the gradient direction from weights w on convolution positions."""
    m = 2 * n - 1
    # g_i = 2 * sum_k w_k * a_{k-i} for 0<=k-i<n
    # = 2 * (w convolved with reversed a), taking positions [n-1 : 2n-1]
    Frev = np.fft.rfft(a[::-1], L)
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):2 * n - 1]
    np.maximum(g, 0.0, out=g)
    return 2.0 * g


def _peak_equalize(a, b, L, m, n, top_k):
    """Try to explicitly reduce the top peaks by redistributing mass away from contributors."""
    idx = np.argsort(b)[-top_k:]
    peak_vals = b[idx]
    peak_mean = float(np.mean(peak_vals))
    # For each peak position k, the contributors are a_i * a_{k-i}
    # We want to reduce a_i for i that contribute to peaks
    pressure = np.zeros(n, dtype=np.float64)
    for k in idx:
        w_k = (b[k] / peak_mean) if peak_mean > 0 else 1.0
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            pressure[i] += w_k * a[j]
    # Normalize pressure
    pmax = float(pressure.max())
    if pmax <= 0 or not np.isfinite(pmax):
        return a
    pressure /= pmax
    # Apply: reduce where pressure is high
    a_new = a * np.exp(-0.3 * pressure)
    a_new = _normalize(a_new)
    return a_new


def _direct_search(a, L, m, n, rng, time_limit, top_k_frac=0.05, temp0=1.0, temp1=200.0):
    """Main iterative optimization loop."""
    if time_limit is None or time_limit <= 0.0:
        a = _normalize(a)
        return a, _score_a(a, L, m, n)

    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    c = _fft_conv_self(a, L, m)
    current_score = _score(c, n)
    best_a = a.copy()
    best_score = current_score

    eta = 0.08
    temp = float(temp0)
    no_accept = 0
    it = 0
    it_since_best = 0
    top_k = max(3, int(top_k_frac * m))

    while time.monotonic() < deadline:
        c = _fft_conv_self(a, L, m)
        current_score = _score(c, n)

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_accept = 0
            it_since_best = 0
        else:
            it_since_best += 1

        # Reset if stuck
        if it_since_best > 200:
            a = best_a.copy()
            temp = max(float(temp0), temp * 0.3)
            eta = 0.08
            no_accept = 0
            it_since_best = 0
            continue

        # Random reheat if very stuck
        if no_accept > 50:
            prior = _random_prior(n, rng)
            mix = 0.1 + 0.3 * float(rng.random())
            a = _normalize((1.0 - mix) * a + mix * prior)
            c = _fft_conv_self(a, L, m)
            current_score = _score(c, n)
            if current_score < best_score - 1e-13:
                best_score = current_score
                best_a = a.copy()
            no_accept = 0
            it_since_best = 0
            temp = max(float(temp0), temp * 0.5)
            eta = 0.08
            continue

        # Compute gradient direction
        w, vmax = _topk_gradient(c, n, top_k, temp)
        g = _compute_direction(a, w, L, n)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 0:
            break
        d = g / gmax

        # Line search for step size
        accepted = False
        for step in (eta, 0.5 * eta, 0.25 * eta):
            if step <= 1e-12:
                continue
            a2 = a * np.exp(-step * d)
            # Ensure non-negativity
            a2 = np.maximum(a2, 1e-300)
            a2 = _normalize(a2)
            c2 = _fft_conv_self(a2, L, m)
            sc2 = _score(c2, n)
            if sc2 < current_score - 1e-15:
                a = a2
                accepted = True
                break

        if accepted:
            no_accept = 0
            eta = min(0.5, eta * 1.1)
        else:
            no_accept += 1
            eta = max(0.01, eta * 0.8)

        # Periodic peak equalization
        if it % 40 == 0 and it > 0:
            a_pe = _peak_equalize(a, c, L, m, n, top_k)
            sc_pe = _score_a(a_pe, L, m, n)
            if sc_pe < best_score - 1e-14:
                a = a_pe
                best_score = sc_pe
                best_a = a.copy()
                no_accept = 0
                it_since_best = 0

        it += 1
        if it % 60 == 0:
            temp = min(float(temp1), temp * 1.3)

    return best_a, best_score


def _mws_direction(f, L, m, n, prior, inner=20, eta_start=3.0, eta_end=60.0):
    """Multiplicative weights saddle point direction."""
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros_like(x)
    count = 0
    logprior = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -40.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if not np.isfinite(ys) or ys <= 0:
            y.fill(1.0 / float(m))
        else:
            y /= ys
        avg += x
        count += 1

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)
        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma) + 0.6 * logprior
        zx -= float(np.max(zx))
        zx = np.clip(zx, -40.0, 0.0)
        x = np.exp(zx)
        xs = float(x.sum())
        if not np.isfinite(xs) or xs <= 0:
            x = prior.copy()
        else:
            x /= xs

    if count == 0:
        return x
    return _normalize(0.5 * (avg / float(count)) + 0.5 * x)


def _mws_search(a, L, m, n, rng, time_limit, inner=20):
    """MWS-based search: find direction via multiplicative weights, then line search."""
    if time_limit is None or time_limit <= 0.0:
        a_norm = _normalize(a)
        return a_norm, _score_a(a_norm, L, m, n)

    deadline = time.monotonic() + float(time_limit)
    best_a = _normalize(a)
    best_score = _score_a(best_a, L, m, n)

    while time.monotonic() < deadline:
        prior = _random_prior(n, rng)
        x = _mws_direction(best_a, L, m, n, prior, inner, 3.0, 60.0)
        # Line search
        found = False
        for lam in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
            h = _normalize((1.0 - lam) * best_a + lam * x)
            sc = _score_a(h, L, m, n)
            if sc < best_score:
                best_a = h
                best_score = sc
                found = True
                break
        if not found:
            # Try smoothed direction
            xs = _normalize(np.convolve(x, np.ones(9) / 9.0, mode="same"))
            for lam in (0.05, 0.1, 0.2, 0.3):
                h = _normalize((1.0 - lam) * best_a + lam * xs)
                sc = _score_a(h, L, m, n)
                if sc < best_score:
                    best_a = h
                    best_score = sc
                    found = True
                    break

    return best_a, best_score


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        # Fallback internal evaluation
        if not isinstance(seq, list) or not seq:
            return float(np.inf)
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float(np.inf)
        b = np.convolve(a, a)
        mb = float(np.max(b))
        return float(2.0 * n * mb / (s * s))
    except Exception:
        return float(np.inf)
    if val is None:
        return float(np.inf)
    return float(val)


def _try_n(n, budget, rng, base_seq=None):
    """Run the search for a specific n value."""
    L = 1 << (2 * n - 2).bit_length()
    m = 2 * n - 1

    # Initial candidates
    inits = _structured_initials(n, rng)
    if base_seq is not None:
        try:
            arr = np.asarray(base_seq, dtype=np.float64).ravel()
            if arr.size == n and np.all(np.isfinite(arr)) and np.sum(arr) > 0:
                inits.insert(0, _normalize(arr))
        except Exception:
            pass

    best_a = None
    best_score = float(np.inf)
    for a in inits:
        sc = _score_a(a, L, m, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is None:
        return None, float(np.inf)

    # Phase 1: Direct search
    remaining = budget - (time.monotonic() - _try_n_start)
    if remaining > 30:
        lim = min(0.35 * remaining, 400.0)
        tmp, sc = _direct_search(best_a, L, m, n, rng, lim, 0.05, 1.0, 150.0)
        if sc < best_score:
            best_score = sc
            best_a = tmp

    # Phase 2: MWS search
    remaining = budget - (time.monotonic() - _try_n_start)
    if remaining > 30:
        lim = min(0.25 * remaining, 250.0)
        tmp, sc = _mws_search(best_a, L, m, n, rng, lim, inner=22)
        if sc < best_score:
            best_score = sc
            best_a = tmp

    # Phase 3: More direct search with lower temp
    remaining = budget - (time.monotonic() - _try_n_start)
    if remaining > 30:
        lim = min(0.3 * remaining, 300.0)
        tmp, sc = _direct_search(best_a, L, m, n, rng, lim, 0.03, 5.0, 400.0)
        if sc < best_score:
            best_score = sc
            best_a = tmp

    # Phase 4: Final MWS
    remaining = budget - (time.monotonic() - _try_n_start)
    if remaining > 20:
        lim = min(0.15 * remaining, 150.0)
        tmp, sc = _mws_search(best_a, L, m, n, rng, lim, inner=25)
        if sc < best_score:
            best_score = sc
            best_a = tmp

    return best_a, best_score

        pass
    except Exception:
        pass

    n_values.sort(key=lambda x: (x != (base_n or -1), abs(x - (base_n or 6000))))

    total_n = len(n_values)
    time_allocs = []
    if base_n is not None:

            time_allocs.append((budget - 30.0) / total_n)

    best_overall_seq = None

    for i, (nv, t_alloc) in enumerate(zip(n_values, time_allocs)):

            except Exception:
                pass

    # Final: also try a quick refinement on the best found
