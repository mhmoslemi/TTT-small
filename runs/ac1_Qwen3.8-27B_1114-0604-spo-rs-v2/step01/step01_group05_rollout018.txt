    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fft_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:

    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return m, w

        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    G = m + float(np.log(Z)) / beta
    return float(G), w


def corr_pw(p, w):

    w = np.asarray(w, dtype=np.float64)

    size = 1

    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

def fft_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)

def corr_pw(p, w):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return s


def normalize_simplex(p):

    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size))

        else:

            best_score = best_B_score
            best_p = p.copy()
            print(f"  block B={B} score={best_score:.8f}")

    return best_p, best_score


def get_prior_best(n):
    try:
        hs = globals().get('height_sequence_1', None)
        if hs is not None and len(hs) > 0:
            arr = np.array(hs, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = float(np.sum(arr))
            if s > 0 and np.isfinite(s):
                arr /= s
                if len(arr) == n:
                    return arr.copy()
                else:
                    x_old = np.linspace(0.0, 1.0, len(arr))
                    x_new = np.linspace(0.0, 1.0, n)
                    resampled = np.interp(x_new, x_old, arr)
                    resampled = np.maximum(resampled, 0.0)
                    rs = float(np.sum(resampled))
                    if rs > 0:
                        resampled /= rs
                    return resampled
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage beta annealing with Nesterov PG and multiplicative weights.
    Warm-starts from prior best when available.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(0.0, budget_s - reserve)

    N = 8192
    rng = np.random.default_rng(seed)

    best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
    best_score = fast_score(best_p)
    print(f"Initial score (uniform N={N}): {best_score:.8f}")

    # Try prior best
    prior = get_prior_best(N)
    if prior is not None:
        prior_score = fast_score(prior)
        print(f"Prior best score: {prior_score:.8f}")
        if np.isfinite(prior_score) and prior_score < best_score:
            best_p = prior.copy()
            best_score = prior_score

    # Phase 1: Block optimization
    phase1_dl = time.monotonic() + min(90.0, 0.12 * budget_s)
    if time.monotonic() < deadline:
        p_block, sc_block = param_block_search(N, phase1_dl, rng, max_B=256)
        if np.isfinite(sc_block) and sc_block < best_score:
            best_score = sc_block
            best_p = p_block.copy()
            print(f"Phase 1 (block) score: {best_score:.8f}")

    # Phase 2: Multi-beta annealing
    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    remaining = deadline - time.monotonic()

    if remaining > 30.0:
        weights = [float(beta + 100) for beta in beta_schedule]
        total_w = float(sum(weights))
        time_fracs = [w / total_w for w in weights]

        p_work = best_p.copy()
        sc_work = best_score

        for i, beta in enumerate(beta_schedule):
            sub_time = max(5.0, remaining * time_fracs[i])
            sub_dl = time.monotonic() + sub_time

            if time.monotonic() >= deadline:
                break

            print(f"Phase 2 beta={beta}, t={sub_time:.0f}s, best={best_score:.8f}")

            if i < len(beta_schedule) - 4:
                p_new, sc_new = nesterov_projected_gradient(p_work, sub_dl, beta)
            else:
                p_new, sc_new = multiplicative_weights_refine(p_work, sub_dl, beta)

            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_p = p_new.copy()
                print(f"  improved to {best_score:.8f}")
            p_work = p_new.copy()

    # Phase 3: Final high-beta polish
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        print(f"Phase 3: polish, t={remaining:.0f}s, score={best_score:.8f}")
        p_fin, sc_fin = nesterov_projected_gradient(
            best_p, time.monotonic() + remaining * 0.6, 1000000
        )
        if np.isfinite(sc_fin) and sc_fin < best_score:
            best_score = sc_fin
            best_p = p_fin.copy()
            print(f"  NG polish: {best_score:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 5.0:
            p_fin2, sc_fin2 = multiplicative_weights_refine(
                best_p, time.monotonic() + remaining, 1000000
            )
            if np.isfinite(sc_fin2) and sc_fin2 < best_score:
                best_score = sc_fin2
                best_p = p_fin2.copy()
                print(f"  MW polish: {best_score:.8f}")

    # Phase 4: Perturbation restarts
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        print(f"Phase 4: perturbations, t={remaining:.0f}s")
        for trial in range(5):
            if time.monotonic() >= deadline - 5.0:
                break
            p_pert = best_p + rng.normal(0.0, 0.02, N) * best_p
            p_pert = normalize_simplex(p_pert)
            sub_dl = time.monotonic() + min(remaining * 0.15, 25.0)
            p_tr, sc_tr = nesterov_projected_gradient(p_pert, sub_dl, 100000)
            if np.isfinite(sc_tr) and sc_tr < best_score:
                best_score = sc_tr
                best_p = p_tr.copy()
                print(f"  trial {trial}: {best_score:.8f}")

    # Final
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    try:
        off_val = float(evaluate_sequence(final_seq))
    except Exception:
        off_val = float("inf")

    print(f"Final official score: {off_val:.8f}")

    if not np.isfinite(off_val) or off_val > 2.0:
        fallback = [1.0 / N] * N
        try:
            off_fb = float(evaluate_sequence(fallback))
            if np.isfinite(off_fb) and off_fb <= off_val:
                return fallback
        except Exception:
            pass
        return fallback

    return final_seq
