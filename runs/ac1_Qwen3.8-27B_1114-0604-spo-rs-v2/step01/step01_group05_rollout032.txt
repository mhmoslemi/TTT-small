def fft_convolution(a):
    """Linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def corr_pw(p, w):
    """Return s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s

    M = float(np.max(b))
    return 2.0 * n * M / (S * S)

    """
    Compute gradient of the smooth max (or top-K average) of the self-convolution.
    Returns (gradient, max_value).

    n = p.size

    M = float(np.max(b))

    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    return g, M, b

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences of length 8192.
Uses adaptive smooth-max gradient descent with increasing sharpness (beta),
momentum, and periodic perturbations. Starts from previous best if available.
Key improvements: top-K weighting blends, momentum, multi-cycle annealing.
"""

import numpy as np
import time


def fft_convolution(a):
    """Linear self-convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def corr_pw(p, w):
    """Return s_i = sum_j p_j * w_{i+j} for i=0..n-1 (cross-correlation)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target} via bisection on threshold."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    q *= target / s
    return q


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / float(p.size), dtype=np.float64)
    return p / s


def fast_score(p):
    """Compute 2n*max(a*a)/(sum a)^2."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def compute_weights(b, beta, K):
    """
    Compute weights on the convolution values.
    Blends smooth-max (softmax) with top-K uniform for robustness.
    """
    M = float(np.max(b))
    if M <= 0.0:
        return np.ones_like(b) / b.size, M

    x = beta * (b - M)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return w, M
    w_smooth = e / Z

    if K is not None and K < b.size and K > 1:
        K = min(K, b.size)
        idx = np.argpartition(b, -K)[-K:]
        w_topk = np.zeros_like(b)
        w_topk[idx] = 1.0 / K
        # Blend: 50% smooth, 50% top-K
        w = 0.5 * w_smooth + 0.5 * w_topk
    else:
        w = w_smooth

    return w, M


def gradient_descent_adaptive(p0, deadline, rng,
                              betas=(5000, 20000, 80000, 300000, 1000000),
                              Ks=(None, 4, 16, 64, 256),
                              max_iter_per_beta=200000,
                              momentum=0.9,
                              perturb_every=50000,
                              perturb_scale=0.002):
    """
    Adaptive smooth-max gradient descent with increasing beta and top-K blend.
    Uses momentum and periodic perturbations to escape local minima.
    """
    p = normalize_simplex(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    velocity = np.zeros(n, dtype=np.float64)
    it = 0

    for beta, K in zip(betas, Ks):
        step = 1.0
        no_improve = 0

        for _ in range(max_iter_per_beta):
            if time.monotonic() >= deadline:
                return best_p, best_score

            b = fft_convolution(p)
            w, M = compute_weights(b, beta, K)
            cur_score = 2.0 * n * M  # sum(p)=1 so (sum)^2 = 1

            if cur_score < best_score - 1e-15:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 20000:
                break

            # Gradient: 2 * corr_pw(p, w)
            svec = corr_pw(p, w)
            svec = np.maximum(svec, 0.0)
            g = 2.0 * svec
            grad = g - float(np.mean(g))

            gnorm = float(np.linalg.norm(grad))
            if not np.isfinite(gnorm) or gnorm < 1e-16:
                break

            # Momentum update
            velocity = momentum * velocity + (1.0 - momentum) * grad / gnorm

            accepted = False
            cur_obj = 2.0 * n * M

            for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
                alpha = step * mult
                q = project_simplex(p - alpha * velocity, 1.0)
                qb = fft_convolution(q)
                qM = float(np.max(qb))
                q_obj = 2.0 * n * qM

                if q_obj < cur_obj - 1e-14:
                    p = q
                    cur_obj = q_obj
                    accepted = True
                    break

            if not accepted:
                step *= 0.5
                if step < 1e-10:
                    break
            else:
                step = min(step * 1.2, 3.0)

            it += 1

            # Periodic perturbation to escape local minima
            if it % perturb_every == 0 and no_improve > 5000:
                noise = rng.normal(0, perturb_scale, n)
                p = normalize_simplex(p + noise)
                velocity = np.zeros(n, dtype=np.float64)
                step = 0.5

            if it % 50000 == 0:
                print(f"    beta={beta} K={K} it={it} best={best_score:.8f} "
                      f"cur={2.0*n*M:.8f} step={step:.3e} no_imp={no_improve}")

    return best_p, best_score


def structured_starts(n, rng, count=6):
    """Generate diverse structured starting points."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear ramps (up and down)
    x = np.linspace(0, 1, n)
    for a in (-3.0, -1.5, 1.5, 3.0):
        p = np.exp(a * x)
        p /= p.sum()
        starts.append(p.copy())

    # Quadratic / U-shape
    for c in (2.0, 5.0):
        p = 1.0 + c * (2.0 * x - 1.0) ** 2
        p = np.maximum(p, 1e-8)
        p /= p.sum()
        starts.append(p.copy())

    # Sinusoidal
    for freq in (1, 2, 3):
        p = 1.0 + 0.5 * np.sin(2.0 * np.pi * freq * x)
        p = np.maximum(p, 1e-8)
        p /= p.sum()
        starts.append(p.copy())

    # Random smooth
    for _ in range(3):
        z = rng.normal(0, 1, n)
        # Smooth via FFT lowpass
        Z = np.fft.rfft(z)
        freqs = np.fft.rfftfreq(n)
        cutoff = 3.0 / n
        Z[freqs > cutoff] *= 0.1
        z = np.fft.irfft(Z, n)
        z = z - z.mean()
        p = np.exp(1.0 * z)
        p /= p.sum()
        starts.append(p.copy())

    return starts[:count]
