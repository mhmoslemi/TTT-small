"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.
Uses smooth-max gradient descent with adaptive step size, multiple starts,
and both block and full-length parameterizations.
"""

import numpy as np
import time

def fft_conv(a):
    """Compute linear self-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]*a[0]])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max(b, beta):
    """Smooth approximation of max(b) and its 'gradient weights'."""
    m = np.max(b)
    if m <= 0:
        w = np.ones_like(b) / len(b)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = np.sum(e)
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def score(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    b = fft_conv(a)
    return 2.0 * n * np.max(b) / (s*s)

def corr_full(p, w):
    """
    Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1.
    This gives the derivative of smooth_max(p*p) w.r.t. p_i (up to factor 2).
    """
    n = len(p)
    m = len(w)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = C[m-1-i] for i = 0..n-1
    s = C[m-n:m][::-1].copy()
    return s

def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target/n)
    low = np.min(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(60):
        mid = (low + high) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > target:
            low = mid
        else:
            high = mid
    tau = (low + high) / 2
    q = np.maximum(v - tau, 0)
    s = np.sum(q)
    if s < 1e-300:
        q = np.full(n, target/n)
    else:
        q *= target / s
    return q

def normalize(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def full_gradient_step(p, beta, step):
    """One projected gradient step. Returns (new_p, accepted, new_score_smooth)."""
    n = len(p)
    b = fft_conv(p)
    G, w = smooth_max(b, beta)
    svec = corr_full(p, w)
    svec = np.maximum(svec, 0)
    grad = 2.0 * svec
    grad -= np.mean(grad)  # project onto sum-preserving subspace
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return p, False, G
    
    for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.003906, 0.001953):
        alpha = step * mult

        qb = fft_conv(q)
        G2, _ = smooth_max(qb, beta)

def block_objective_and_grad(c, n, B, beta):
    """
    Objective and gradient for block-constant parameterization.
    c has length B-1 (relative log weights for first B-1 blocks, last is 0).
    """
    L = n // B
    z = np.zeros(B)
    if B > 1:
        z[:B-1] = c
    z = z - np.max(z)
    wb = np.exp(z)
    s = np.sum(wb)
    if s < 1e-300:
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    p = np.repeat(H, L)
    
    b = fft_conv(p)
    G, w = smooth_max(b, beta)
    if not np.isfinite(G):
        return np.inf, np.zeros(B-1)
    
    svec = corr_full(p, w)
    svec = np.maximum(svec, 0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = np.dot(wb, Gsum)
    gradz = (wb / L) * (Gsum - Wavg)
    grad = n * gradz[:B-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return n * G, grad

def block_refine_scipy(c0, n, B, beta, deadline):
    """Use scipy L-BFGS-B for block optimization."""
    try:
        from scipy.optimize import minimize
    except ImportError:
        return c0
    
    bounds = [(-8.0, 8.0)] * (B-1)
    best_c = c0.copy()
    best_val, _ = block_objective_and_grad(c0, n, B, beta)
    
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return best_c
    
    maxiter = max(30, int(remaining * 5))
    maxiter = min(maxiter, 200)
    
    try:
        res = minimize(
            block_objective_and_grad,
            c0,
            args=(n, B, beta),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-6, 'maxls': 30}
        )
        if np.isfinite(res.fun) and res.fun < best_val:
            best_c = res.x
    except Exception:
        pass
    
    return np.clip(best_c, -8, 8)

def generate_starts(n, rng, count=12):

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.

Approach:
- The self-convolution peak is minimized by making the convolution profile flat.
- Use smooth-max (log-sum-exp) as a differentiable surrogate for the max.

import numpy as np
import time


def fft_conv(a):
    """Compute linear self-convolution a*a via FFT (clipped to non-negative)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max(b, beta):
    """Return (smooth_max_value, normalized_weights) for the log-sum-exp approximation."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / Z
    G = m + np.log(Z) / beta
    return G, w


def fast_score(a):
    """Compute 2*n*max(conv)/(sum)^2 using FFT."""
    n = a.size
    s = float(np.sum(a))
    if s < 1e-300:
        return float('inf')
    b = fft_conv(a)
    M = float(np.max(b))
    return 2.0 * n * M / (s * s)


def corr_full(p, w):
    """
    Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1 via FFT.
    Used for gradient of smooth_max(p*p) w.r.t. p.
    """
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return s


def project_simplex_naive(v, target=1.0):
    """Fast clip-and-rescale projection onto {x>=0, sum(x)=target}."""
    q = np.maximum(v, 0.0)
    s = float(np.sum(q))
    n = q.size
    if s < 1e-300 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def project_simplex_exact(v, target=1.0):
    """Exact Euclidean projection onto {x>=0, sum(x)=target} via bisection."""
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s < 1e-300 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def block_to_seq(n, B, c):
    """Convert log-difference block weights to full sequence (sum=1)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s < 1e-300:
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_obj_grad(c, n, B, beta):
    """Objective and gradient for block-constant parameterization."""
    L = n // B
    p = block_to_seq(n, B, c)
    b = fft_conv(p)
    G, w = smooth_max(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)
    svec = corr_full(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    wb = (np.repeat(((p * L)), L) / L).reshape(B, L).sum(axis=1)  # block sums
    # Actually, block sum = p.reshape(B,L).sum(axis=1) which equals wb (normalized)
    block_sums = p.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(block_sums, Gsum))
    gradz = block_sums * (Gsum - Wavg)
    grad = n * gradz[:B - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad


def run_block_lbfgs(c0, n, B, beta, deadline):
    """Run L-BFGS-B on block parameters."""
    try:
        from scipy.optimize import minimize
    except ImportError:
        return c0

    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    expected_len = max(0, B - 1)
    if c0.size != expected_len:
        tmp = np.zeros(expected_len, dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)

    remaining = deadline - time.monotonic()
    if remaining < 5.0:
        return c0

    maxiter = max(20, min(150, int(remaining * 3)))
    bounds = [(-8.0, 8.0)] * expected_len

    try:
        res = minimize(
            block_obj_grad, c0, args=(n, B, beta),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-5, 'maxls': 30}
        )
        if np.isfinite(res.fun):
            return np.clip(res.x, -8.0, 8.0)
    except Exception:
        pass
    return c0

def generate_diverse_starts(n, rng, count=15):
    """Generate diverse non-negative sequences summing to 1."""
    starts = []
    x = np.linspace(0.0, 1.0, n)

    # Uniform
    starts.append(np.ones(n) / n)

    # U-shaped / bathtub profiles (concentrated at edges)
    for a in [0.5, 1.0, 2.0, 3.0, 4.0, 6.0, 8.0]:
        p = np.exp(a * (2.0 * x - 1.0) ** 2)
        starts.append(p / p.sum())

    # Cosine window (concentrated at center)
    p = np.cos(np.pi * (x - 0.5)) ** 2 + 0.001
    starts.append(p / p.sum())

    # Power-law edge concentration
    for pwr in [0.5, 1.0, 2.0]:
        dist = np.minimum(x, 1.0 - x) + 1e-10
        p = dist ** (-pwr)
        starts.append(p / p.sum())

    # Sinusoidal patterns
    for freq in [1, 2, 3, 5]:
        p = 1.0 + 0.7 * np.sin(2.0 * np.pi * freq * x + 0.5)
        p = np.maximum(p, 0.01)
        starts.append(p / p.sum())

    # Random smooth (interpolated noise)
    for _ in range(4):
        n_coarse = max(4, n // 64)
        coeffs = rng.normal(0, 1, n_coarse)
        x_c = np.linspace(0, 1, n_coarse)
        smooth = np.interp(x, x_c, coeffs)
        p = np.exp(1.0 * smooth)
        starts.append(p / p.sum())

    # Two-lobe profile
    for peak_pos in [0.25, 0.35]:
        p = np.exp(-20.0 * (x - peak_pos) ** 2) + np.exp(-20.0 * (x - (1 - peak_pos)) ** 2) + 0.01
        starts.append(p / p.sum())

    return starts[:count]
