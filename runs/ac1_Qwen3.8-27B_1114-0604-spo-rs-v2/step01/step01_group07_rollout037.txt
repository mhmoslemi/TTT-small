Low-peak nonnegative step-function search using gradient-based optimization
with softmax-approximated max, diverse initializations, and active-set refinement.

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _grad_cross_correlate(p, a):
    """Compute grad[j] = sum_{m=0}^{n-1} p[j+m] * a[m]."""
    n = a.size
    N = _next_pow2(p.size + n - 1)
    fp = np.fft.rfft(p, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(fp * fa_rev, N)
    return conv[n-1:n-1+n]


def _compute_score(a):
    """Compute the evaluation score for a normalized non-negative sequence."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s)


def _compute_score_and_conv(a):
    """Compute score and the full convolution."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf, np.array([])
    c = _fft_convolve(a, a)
    mx = float(np.max(c))
    return 2.0 * n * mx / (s * s), c

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())

Low-peak nonnegative step-function search using gradient-based optimization
with softmax-approximated max, multi-harmonic sinusoidal initializations,
and active-set LP-inspired refinement.

Key ideas:
- The anti-palindromic multi-harmonic family a_i = (1 + sum_m A_m sin(2*pi*m*(i+0.5)/n))/n
  is a rich starting point since ALL integer harmonics are anti-palindromic.
- Smooth optimization via softmax-approximated max with multiplicative weight updates
  on the simplex, using FFT-based gradient computation (O(n log n) per iteration).
- Active-set refinement inspired by the LP dual structure.

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, b=None):
    """Linear convolution a*b using FFT."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _grad_correlate(p, a):
    """
    Compute grad[j] = sum_{m=0}^{n-1} p[j+m] * a[m] for j=0..n-1,
    where p has length 2n-1 and a has length n.
    Uses FFT-based cross-correlation.
    """
    n = a.size
    N = _next_pow2(p.size + n - 1)
    fp = np.fft.rfft(p, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(fp * fa_rev, N)
    return conv[n - 1:n - 1 + n]


def _normalize(a):
    """Normalize to sum=1, ensure non-negative and bounded."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _compute_score(a):
    """Compute the evaluation score (assumes valid non-negative input)."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))


def _compute_score_conv(a):
    """Compute score and full convolution simultaneously."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf, np.array([], dtype=np.float64)
    c = _fft_convolve(a, a)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s)), c


def _make_multiharmonic(n, amps, phases=None):
    """
    Create a multi-harmonic anti-palindromic sinusoidal candidate.
    amps: list of (harmonic_mode, amplitude) pairs.
    phases: optional list of phase offsets (radians).
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for idx, (mode, amp) in enumerate(amps):
        phase = 0.0
        if phases is not None and idx < len(phases):
            phase = phases[idx]
        a += amp * np.sin(2.0 * np.pi * mode * (i + 0.5) / float(n) + phase)
    return _normalize(a)

    n = a.size
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = _compute_score(best_a)
    last_improve = start
    
    it = 0
    total_iters_expected = max(100, int(budget_s / 0.005))  # rough estimate
    n_iters_done = 0
    
    while time.monotonic() < min(start + budget_s, deadline):
        t_now = time.monotonic()
        if t_now - start > budget_s:
            break
        
        # Progress-based annealing
        progress = min(1.0, n_iters_done / max(1, total_iters_expected))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        # Compute convolution
        _, c = _compute_score_conv(a)
        if c.size == 0:
            break
        
        c_max = float(np.max(c))
        if c_max <= 0:
            break
        
        # Score check
        if n_iters_done % check_every == 0:
            score = 2.0 * n * c_max / float(a.sum()) ** 2
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = t_now
        
        # Compute softmax probabilities (temperature-scaled)
        scaled = T * c / c_max
        scaled = scaled - np.max(scaled)
        exp_s = np.exp(scaled)
        p = exp_s / np.sum(exp_s)
        
        # Compute gradient
        grad = 2.0 * _grad_correlate(p, a)
        
        # Apply multiplicative weight update
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-30:
            break
        
        log_upd = -eta * grad / grad_max
        log_upd = np.clip(log_upd, -2.5, 2.5)
        a = a * np.exp(log_upd)
        
        # Floor and normalize
        a = np.where(a > 1e-25, a, 1e-25)
        a = _normalize(a)
        
        it += 1
        n_iters_done += 1
        
        # Stagnation detection and restart
        if n_iters_done % (check_every * 6) == 0:
            cur_score = _compute_score(a)
            if t_now - last_improve > budget_s * 0.25 and cur_score > best_score * 1.05:
                # Perturb from best and restart
                rng = np.random.RandomState(it)
                a = best_a * (1.0 + 0.1 * rng.randn(n))
                a = _normalize(a)
                last_improve = t_now
                n_iters_done = 0  # reset annealing
    
    # Final score check
    final_score = _compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _active_set_refine(a, budget_s, deadline, K=None):
    """
    Active-set LP-inspired refinement.
    Uses only the top K autocorrelation lags to compute a focused gradient.
    """
    a = _normalize(a.copy())
    n = a.size
    if K is None:
        K = min(32, max(8, n // 128))
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = _compute_score(best_a)
    lr = 0.3
    it = 0
    last_improve = start
    
    while time.monotonic() < min(start + budget_s, deadline):
        it += 1
        t_now = time.monotonic()
        if t_now - start > budget_s:
            break
        
        _, c = _compute_score_conv(a)
        if c.size == 0:
            break
        
        c_max = float(np.max(c))
        if c_max <= 0:
            break
        
        score = 2.0 * n * c_max / float(a.sum()) ** 2
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = t_now
            lr = min(1.0, lr * 1.05)
        elif it % 30 == 0 and score > best_score * 1.02:
            a = best_a.copy()
            lr = max(0.05, lr * 0.8)
        
        # Select top K lags
        if c.size <= K:
            active_idx = np.arange(c.size)
        else:
            active_idx = np.argpartition(c, -K)[-K:]
        
        # Build probability
        w = c[active_idx]
        w = w / np.sum(w)
        
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[active_idx.astype(int)] = w
        
        # Compute focused gradient
        grad = 2.0 * _grad_correlate(p, a)
        
        # Update
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-30:
            break
        log_upd = -lr * grad / grad_max
        log_upd = np.clip(log_upd, -2.0, 2.0)
        a = a * np.exp(log_upd)
        a = np.where(a > 1e-25, a, 1e-25)
        a = _normalize(a)
    
    final_score = _compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _scan_multiharmonic(deadline):
    """
    Scan multi-harmonic sinusoidal family for good initial points.
    """
    best_a = None
    best_score = np.inf
    
    lengths = [2048, 4096, 8192, 16384]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        
        # Single harmonic scan
        for amp1 in np.linspace(0.5, 1.0, 12):
            if time.monotonic() > deadline:
                break
            a = _make_multiharmonic(n, [(1, float(amp1))])
            sc = _compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Two harmonics
        for amp1 in [0.7, 0.8, 0.9]:
            for amp2 in [0.05, 0.1, 0.15, 0.2, 0.3]:
                if time.monotonic() > deadline:
                    break
                # Try both signs for second harmonic
                for sign in [1, -1]:
                    a = _make_multiharmonic(n, [(1, amp1), (2, sign * amp2)])
                    sc = _compute_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
        
        # Three harmonics
        for amp1 in [0.75, 0.85]:
            for amp2 in [0.1, 0.2]:
                for amp3 in [0.05, 0.1]:
                    if time.monotonic() > deadline:
                        break
                    for s2 in [1, -1]:
                        for s3 in [1, -1]:
                            a = _make_multiharmonic(n, [
                                (1, amp1), (2, s2 * amp2), (3, s3 * amp3)
                            ])
                            sc = _compute_score(a)
                            if sc < best_score:
                                best_score = sc
                                best_a = a.copy()
        
        print(f"  n={n}, best so far: {best_score:.6f}")
    
    if best_a is None:
        best_a = _make_multiharmonic(2048, [(1, 0.8)])
        best_score = _compute_score(best_a)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score using:
    1. Multi-harmonic sinusoidal scan for diverse initializations
    2. Gradient-based optimization with softmax-approximated max
    3. Active-set LP-inspired refinement
    4. Multiple restarts with different temperatures
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 25.0
    
    rng = np.random.RandomState(seed)
    
    # Get provided starting point
    provided_seq = None
    try:
        provided_seq = height_sequence_1
    except NameError:
        pass
    
    # === Phase 1: Diverse initializations ===
    print("[Phase 1] Multi-harmonic scan & initializations...")
    p1_deadline = t0 + min(200.0, total_budget * 0.08)
    
    candidates = []
    
    # Use provided sequence
    if provided_seq is not None:
        try:
            prov = np.asarray(provided_seq, dtype=np.float64).ravel()
            if prov.size > 0:
                prov = _normalize(prov)
                sc = _compute_score(prov)
                candidates.append((sc, prov.copy(), "provided_n%d" % prov.size))
                print("  provided: n=%d, score=%.6f" % (prov.size, sc))
        except Exception as e:
            print("  provided failed: %s" % str(e))
    
    # Multi-harmonic scan
    scan_a, scan_sc = _scan_multiharmonic(p1_deadline)
    candidates.append((scan_sc, scan_a, "scan_n%d" % scan_a.size))
    print("  scan best: %.6f (n=%d)" % (scan_sc, scan_a.size))
    
    # A few random perturbations of the best
    for i in range(4):
        if time.monotonic() > p1_deadline:
            break
        pert = scan_a * (1.0 + 0.2 * rng.randn(scan_a.size))
        pert = _normalize(pert)
        sc = _compute_score(pert)
        candidates.append((sc, pert, "pert%d" % i))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print("  Top candidate: %.6f (%s)" % (candidates[0][0], candidates[0][2]))
    
    # Keep top 4 for refinement
    top_cands = []
    for sc, arr, name in candidates[:4]:
        if arr.size != 0:
            top_cands.append((sc, arr, name))
    
    if not top_cands:
        top_cands = [(2.0, np.ones(2048, dtype=np.float64) / 2048.0, "const")]
    
    # === Phase 2: Gradient optimization ===
    print("[Phase 2] Gradient optimization...")
    p2_deadline = t0 + total_budget * 0.65
    
    best_overall = top_cands[0][1].copy()
    best_score = top_cands[0][0]
    
    n_refine = len(top_cands)
    remaining_for_refine = max(0, p2_deadline - time.monotonic() - 30)
    
    for idx in range(n_refine):
        if time.monotonic() > p2_deadline:
            break
        sc0, arr0, name = top_cands[idx]
        remaining = p2_deadline - time.monotonic()
        if remaining < 40:
            break
        
        # Allocate time: best candidates get more time
        frac = 1.0 / (n_refine - idx) * 0.55
        alloc = min(remaining - 20, max(60, remaining_for_refine * frac))
        
        print("  Optimizing '%s' (init=%.6f), budget=%.0fs" % (name, sc0, alloc))
        
        a_opt, sc_opt = _gradient_optimize(
            arr0, alloc, p2_deadline,
            T_start=2.0, T_end=20.0,
            eta_start=0.6, eta_end=4.0,
            check_every=40
        )
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_overall = a_opt.copy()
            print("    -> improved to %.6f" % sc_opt)
        else:
            print("    -> no improvement (got %.6f)" % sc_opt)
    
    # === Phase 3: Active-set refinement ===
    print("[Phase 3] Active-set refinement...")
    p3_deadline = t0 + total_budget * 0.82
    
    remaining = p3_deadline - time.monotonic()
    if remaining > 30:
        a_act, sc_act = _active_set_refine(
            best_overall, min(remaining, 500), p3_deadline
        )
        if sc_act < best_score:
            best_score = sc_act
            best_overall = a_act.copy()
            print("  -> improved to %.6f" % sc_act)
        else:
            print("  -> no improvement (got %.6f)" % sc_act)
    
    # === Phase 4: Second gradient pass with different params ===
    print("[Phase 4] Second gradient pass...")
    p4_deadline = t0 + total_budget * 0.93
    
    remaining = p4_deadline - time.monotonic()
    if remaining > 30:
        a2, sc2 = _gradient_optimize(
            best_overall, min(remaining, 400), p4_deadline,
            T_start=4.0, T_end=35.0,
            eta_start=0.4, eta_end=3.0,
            check_every=30
        )
        if sc2 < best_score:
            best_score = sc2
            best_overall = a2.copy()
            print("  -> improved to %.6f" % sc2)
        else:
            print("  -> no improvement (got %.6f)" % sc2)
    
    # === Phase 5: Final active-set polish ===
    p5_deadline = hard_deadline - 10
    remaining = p5_deadline - time.monotonic()
    if remaining > 20:
        a5, sc5 = _active_set_refine(
            best_overall, min(remaining, 200), p5_deadline
        )
        if sc5 < best_score:
            best_score = sc5
            best_overall = a5.copy()
            print("  Final polish -> improved to %.6f" % sc5)
    
    # === Validation ===
    print("[Validation] Checking with evaluate_sequence...")
    final_seq = _normalize(best_overall)
    
    best_valid_score = np.inf
    best_valid_seq = None
    
    # Try the best candidate
    try:
        s = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(s):
            best_valid_score = s
            best_valid_seq = final_seq.copy()
            print("  Best: %.8f (n=%d)" % (s, final_seq.size))
        else:
            print("  Best: invalid score")
    except Exception as e:
        print("  Best: error %s" % str(e))
    
    # If best didn't validate, try other candidates
    if best_valid_seq is None:
        for sc, arr, name in top_cands[:3]:
            try:
                seq = _normalize(arr).tolist()
                s = float(evaluate_sequence(seq))
                if np.isfinite(s) and s < best_valid_score:
                    best_valid_score = s
                    best_valid_seq = _normalize(arr)
                    print("  Fallback '%s': %.8f" % (name, s))
            except Exception:
                pass
    
    # Also try the scan candidate directly
    try:
        seq = _normalize(scan_a).tolist()
        s = float(evaluate_sequence(seq))
        if np.isfinite(s) and s < best_valid_score:
            best_valid_score = s
            best_valid_seq = _normalize(scan_a)
            print("  Scan fallback: %.8f" % s)
    except Exception:
        pass
    
    # Ultimate fallback
    if best_valid_seq is None:
        best_valid_seq = np.ones(2048, dtype=np.float64) / 2048.0
        best_valid_score = 2.0
        print("  Ultimate fallback: 2.0")
    
    print("[Done] Final score: %.8f, n=%d, time=%.1fs" % (
        best_valid_score, best_valid_seq.size, time.monotonic() - t0))
    
    return [float(x) for x in best_valid_seq]
