import numpy as np
import time

def _next_pow2(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """
    Fast internal evaluation matching evaluate_sequence, assuming no clipping.
    """
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _score_and_conv(a):
    """Return (score, convolution) for a sequence."""
    arr = np.asarray(a, dtype=np.float64)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf, np.array([])
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    sc = float(2.0 * arr.size * mx / (s * s))
    return sc, c


def _subgradient_from_peaks(arr, conv, K):
    """
    Compute averaged subgradient of the top-K peaks of the convolution.
    Returns gradient array of same shape as arr.
    """
    n = arr.size
    conv_len = len(conv)
    K = min(K, conv_len)
    
    # Get indices of top K peaks
    if K < conv_len:
        top_idx = np.argpartition(conv, -K)[-K:]
    else:
        top_idx = np.arange(conv_len)
    
    # Sort by value to weight properly
    top_vals = conv[top_idx]
    top_idx_sorted = top_idx[np.argsort(top_vals)[::-1]]
    top_vals_sorted = np.sort(top_vals)[::-1]
    
    # Weight by peak value (larger peaks get more gradient)
    # Use a soft weighting to smooth
    weight_scale = top_vals_sorted[0] if top_vals_sorted[0] > 0 else 1.0
    weights = np.maximum(top_vals_sorted, 0.0) / weight_scale
    
    grad = np.zeros(n, dtype=np.float64)
    
    for idx, pos in enumerate(top_idx_sorted):
        if weights[idx] < 0.01:
            continue
        k = int(pos)
        # grad[m] += 2 * arr[k - m] for valid m
        # (handling the case where k-m == m, i.e., 2m=k)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max < i_min:
            continue
        idx_range = np.arange(i_min, i_max + 1)
        # For each i in range, contribution to grad[i] is:
        # 2*arr[k-i] if i != k-i, else arr[k-i]
        contrib = 2.0 * arr[k - idx_range]
        # Where 2*i == k, the contribution is arr[k-i] not 2*arr[k-i]
        midpoint = k // 2
        if k % 2 == 0 and i_min <= midpoint <= i_max:
            pos_in_range = midpoint - i_min
            contrib[pos_in_range] = arr[midpoint]
        grad += weights[idx] * contrib
    
    # Normalize by sum of weights
    wsum = weights.sum()
    if wsum > 0:
        grad /= wsum
    
    return grad


def _mirror_descent_refine(arr, budget_s, K_list, verbose=True):
    """
    Aggressive mirror descent refinement on the probability simplex.
    Uses backtracking line search and adaptive K.
    """
    start = time.monotonic()
    arr = np.asarray(arr, dtype=np.float64).copy()
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score, _ = _score_and_conv(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    lr = 0.5 * n  # initial learning rate
    max_lr = 2.0 * n
    min_lr = 0.1
    it = 0
    max_iters = 100000
    last_improve = 0
    stagnation_limit = 200
    
    # Start with smaller K, increase over time
    cur_K = K_list[0]
    K_increase_interval = 50
    
    if verbose:
        print(f"    [refine] start score={best_score:.8f} n={n} lr={lr:.1f}")
    
    while time.monotonic() - start < budget_s and it < max_iters:
        if time.monotonic() - start > budget_s:
            break
        it += 1
        
        # Update K periodically
        if it % K_increase_interval == 0 and cur_K < K_list[-1]:
            cur_K = min(cur_K + K_list[-1] // len(K_list), K_list[-1])
        
        conv = _fft_convolve(arr)
        maxc = float(np.max(conv))
        if not np.isfinite(maxc):
            break
        
        # Compute subgradient
        grad = _subgradient_from_peaks(arr, conv, cur_K)
        
        # Project to simplex tangent: subtract mean
        g = grad - np.mean(grad)
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break
        
        # Mirror descent step: a_new = a * exp(-lr * g / g_norm)
        step = lr / g_norm
        step = min(step, 2.0)  # cap the exponent
        exponent = -step * g / g_norm
        exponent = np.clip(exponent, -3.0, 3.0)
        
        new_arr = arr * np.exp(exponent)
        new_arr = np.where(np.isfinite(new_arr), new_arr, 0.0)
        new_arr = np.clip(new_arr, 0.0, 1000.0)
        ns = float(new_arr.sum())
        if not np.isfinite(ns) or ns < 0.01:
            lr = max(min_lr, lr * 0.5)
            continue
        new_arr = new_arr / ns
        
        # Evaluate new point
        new_score, _ = _score_and_conv(new_arr)
        
        if np.isfinite(new_score) and new_score < best_score:
            # Accept
            arr = new_arr
            if new_score < best_score * (1 - 1e-6):
                best = new_arr.copy()
                best_score = new_score
                last_improve = it
                # Increase lr slightly when improving
                lr = min(max_lr, lr * 1.02)
            else:
                # Marginal improvement, keep lr
                pass
        else:
            # Reject: backtracking
            lr = max(min_lr, lr * 0.7)
            if it - last_improve > stagnation_limit:
                if verbose:
                    print(f"    [refine] stagnating at it={it}, lr={lr:.2f}, score={best_score:.8f}")
                break
        
        if it % 500 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    [refine] it={it} score={best_score:.8f} lr={lr:.1f} K={cur_K} time={elapsed:.1f}s")
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"    [refine] done it={it} best_score={best_score:.8f} time={elapsed:.1f}s")
    
    return best, best_score


def _perturb(arr, rng, scale=0.05):
    """Add multiplicative noise and renormalize."""
    n = arr.size
    noise = 1.0 + scale * rng.standard_normal(n)
    new_arr = arr * np.maximum(noise, 0.0)
    new_arr = np.clip(new_arr, 0.0, 1000.0)
    s = float(new_arr.sum())
    if s < 0.01:
        return arr.copy()
    return new_arr / s


def _make_sine_candidate(n, amp, mode=1):
    """Create a nonnegative normalized sinusoidal anti-palindromic sequence."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s

        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64) / float(n)
    for amp, mode in zip(amps, modes):
        a += (float(amp) * np.sin(float(mode) * theta)) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _scan_candidates(deadline, verbose=True):
    """Scan sinusoidal family for good starting points."""
    best_a = None
    best_score = np.inf
    rng = np.random.default_rng(123)
    
    # Primary lengths to try
    lengths = [2048, 4096, 1024, 8192, 512]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        
        # Scan amplitude
        amps = np.linspace(0.0, 1.1, 30)
        local_best_a = None
        local_best_score = np.inf
        
        for amp in amps:
            if time.monotonic() > deadline:
                break
            a = _make_sine_candidate(n, amp, 1)
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < local_best_score:
                local_best_score = sc
                local_best_a = a
        
        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_a = local_best_a
            if verbose:
                print(f"  [scan] n={n} best_amp~{float(amps[np.argmin([_internal_score(_make_sine_candidate(n, a_, 1)) for a_ in amps])]):.3f} score={best_score:.8f}")
        
        # Also try multi-harmonic
        for a2, a3 in [(0.2, 0.1), (0.3, 0.15), (0.15, 0.05), (0.25, -0.1)]:
            if time.monotonic() > deadline:
                break
            for amp1 in [0.6, 0.7, 0.8, 0.9]:
                a = _make_multi_sine_candidate(n, [amp1, a2, a3], [1, 3, 5])
                sc = _internal_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a
                    if verbose:
                        print(f"  [scan-multi] n={n} amps=({amp1},{a2},{a3}) score={best_score:.8f}")
    
    if best_a is None:
        best_a = _make_sine_candidate(2048, 0.78, 1)
        best_score = _internal_score(best_a)
    
    return best_a, best_score


def _safe_separate(x):
    """Sanitize a sequence before returning."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.
    
    Strategy:
    1. Scan sinusoidal family for good starting points (~100s)
    2. Aggressive mirror descent refinement (~1600s)
    3. Multi-restart with perturbation (~400s)
    4. Final validation with evaluate_sequence (~50s)
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: Scan for good starting points
    scan_deadline = t0 + 100.0
    if total_budget < 300:
        scan_deadline = t0 + max(10.0, total_budget * 0.1)
    
    print(f"[Phase 1] Scanning candidates (budget {scan_deadline - t0:.0f}s)")
    scan_a, scan_score = _scan_candidates(scan_deadline, verbose=True)
    print(f"[Phase 1] Best scan score: {scan_score:.8f} at t={time.monotonic()-t0:.1f}s")
    
    # Phase 2: Primary refinement
    refine_budget = max(30.0, total_budget - 550.0)
    if total_budget < 300:
        refine_budget = max(10.0, total_budget * 0.5)
    
    K_list = [10, 20, 40, 80, 150]
    
    print(f"[Phase 2] Primary refinement (budget {refine_budget:.0f}s)")
    refine_start = time.monotonic()
    best_a, best_score = _mirror_descent_refine(
        scan_a, refine_budget, K_list, verbose=True
    )
    elapsed = time.monotonic() - refine_start
    print(f"[Phase 2] Done: score={best_score:.8f} time={elapsed:.1f}s")
    
    # Phase 3: Multi-restart refinement
    remaining = hard_deadline - time.monotonic()
    if remaining > 100.0:
        multirestart_budget = min(remaining - 50.0, 400.0)
        print(f"[Phase 3] Multi-restart (budget {multirestart_budget:.0f}s)")
        
        # Try several perturbations of the best solution
        n_restarts = max(1, int(multirestart_budget / 60.0))
        per_restart = multirestart_budget / n_restarts
        
        for ri in range(n_restarts):
            if time.monotonic() > hard_deadline - 40.0:
                break
            # Create perturbed version
            scale = 0.03 + 0.02 * ri
            perturbed = _perturb(best_a, rng, scale=scale)
            
            sc_perturbed, _ = _score_and_conv(perturbed)
            if not np.isfinite(sc_perturbed):
                continue
            
            # Brief refinement from perturbed start
            sub_budget = min(per_restart, remaining - 50.0)
            if sub_budget < 10.0:
                break
            
            K_list_short = [10, 30, 60]
            new_a, new_score = _mirror_descent_refine(
                perturbed, sub_budget, K_list_short, verbose=False
            )
            
            if np.isfinite(new_score) and new_score < best_score:
                best_a = new_a
                best_score = new_score
                print(f"  [restart {ri}] improved to {best_score:.8f}")
            else:
                # Also try continuing from best
                pass
        
        # One more refinement from the overall best
        remaining2 = hard_deadline - time.monotonic()
        if remaining2 > 30.0:
            print(f"[Phase 3b] Final polish (budget {remaining2 - 20:.0f}s)")
            K_list_final = [20, 50, 100, 200]
            best_a, best_score = _mirror_descent_refine(
                best_a, remaining2 - 20.0, K_list_final, verbose=True
            )
    
    # Phase 4: Validation
    print(f"[Phase 4] Validation at t={time.monotonic()-t0:.1f}s")
    
    candidates = []
    candidates.append(("primary", best_a))
    candidates.append(("scan", scan_a))
    
    # Add a fallback
    fallback = _make_sine_candidate(2048, 0.78, 1)
    candidates.append(("fallback", fallback))
    
    best_actual = np.inf
    best_out = None
    
    for name, cand in candidates:
        if time.monotonic() > hard_deadline:
            break
        arr = _safe_separate(cand)
        if arr is None:
            continue
        try:
            real_score = float(evaluate_sequence(arr.tolist()))
        except Exception:
            real_score = np.inf
        if np.isfinite(real_score) and real_score < best_actual:
            best_actual = real_score
            best_out = arr
            print(f"  [validate] {name} n={arr.size} actual_score={real_score:.8f}")
        else:
            print(f"  [validate] {name} n={arr.size} actual_score={real_score}")
    
    if best_out is None:
        best_out = np.ones(256, dtype=np.float64) / 256.0
        print("[fallback] using constant sequence")
    
    print(f"[DONE] Final score: {best_actual:.8f} at t={time.monotonic()-t0:.1f}s")
    return [float(x) for x in best_out]
