def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def softmax(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def log_param_score(params):
    """Score when sequence is softmax(params)."""
    a = softmax(params)
    return compute_score(a)


def project_nonneg_sum1(v):
    """Project v to {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s

def structured_init(n, rng, method='uniform'):
    """Create structured initial sequences."""
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'bimodal':
        i = np.arange(n, dtype=np.float64)
        c1 = n * (0.2 + 0.1 * rng.random())
        c2 = n * (0.7 + 0.1 * rng.random())
        sigma = n * 0.08
        a = np.exp(-((i - c1) / sigma)**2) + np.exp(-((i - c2) / sigma)**2)
        a += 0.01 * np.abs(rng.standard_normal(n))
        return a / a.sum()
    elif method == 'u_shape':
        i = np.arange(n, dtype=np.float64)
        m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
        a = 1.0 / np.sqrt(m)
        return a / a.sum()
    elif method == 'cosine':
        i = np.arange(n, dtype=np.float64)
        freq = 1 + int(rng.integers(0, 4))
        a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1)))**2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif method == 'random':
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    elif method == 'sparse':
        a = np.zeros(n)
        k = max(2, n // 4)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = 1.0
        return a / a.sum()
    else:
        return np.ones(n) / n
