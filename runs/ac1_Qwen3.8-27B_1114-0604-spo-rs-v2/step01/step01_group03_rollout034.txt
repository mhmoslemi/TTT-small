```python
"""
Search for a non-negative coefficient sequence minimizing the oracle score
evaluate_sequence().

The implementation is deliberately adaptive:
  * It probes several simple families with the real oracle.
  * It keeps a pool of candidates (parent, constant, sparse, periodic,
    sinusoidal, randomized priors).
  * It runs an FFT-based multiplicative/soft-max optimizer for several
    plausible weighted-convolution objectives.
  * It then performs oracle-guided refinement, estimating lag weights from
    the oracle score of the current best sequence.
  * Finally it evaluates a final candidate pool and returns the best sequence
    found.

The code always returns a list of finite, non-negative Python floats.
"""

import math
import time
import numpy as np


# ----------------------------------------------------------------------------
# Basic array utilities
# ----------------------------------------------------------------------------

def _uniform1(n):
    n = int(max(1, n))
    a = np.ones(n, dtype=np.float64)
    return a / float(n)


def _spike1(n, idx=0):
    n = int(max(1, n))
    a = np.zeros(n, dtype=np.float64)
    idx0 = int(idx) % n
    a[idx0] = 1.0
    return a


def _two_spike1(n):
    n = int(max(1, n))
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    if n > 1:
        a[-1] = 1.0
    return a / float(a.sum())


def _normalize1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    a = np.where(a > 1e-300, a, 1e-300)
    a = a / float(a.sum())
    return a


def _to_final_list(a, target_sum=10.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return [1.0]
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64)
    else:
        a = _normalize1(a)
    a = a * float(target_sum)
    # Tiny positive floor so the final sequence is strictly positive, while
    # keeping zeros effectively irrelevant.
    a = np.maximum(a, 1e-15)
    a = a / float(a.sum())
    a = a * float(target_sum)
    a = np.minimum(a, 1000.0)
    a = np.maximum(a, 0.0)
    s2 = float(a.sum())
    if s2 < 0.01:
        a = np.full(n, 0.1 / max(1, n), dtype=np.float64)
        a = a * (10.0 / max(float(a.sum()), 1e-300))
    return [float(x) for x in a]


def _fft_len(n):
    n = int(max(1, n))
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L


def _autocorr_from_fft(a_fft, L, n):
    c = np.fft.irfft(a_fft * a_fft, L)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def _autocorr(a, L=None, n=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size) if n is None else int(n)
    if n <= 0:
        return np.zeros(1, dtype=np.float64)
    if L is None:
        L = _fft_len(n)
    A = np.fft.rfft(a, L)
    return _autocorr_from_fft(A, L, n)


def _wm_obj(v, W):
    mask = W > 0.0
    if not np.any(mask):
        return 0.0
    return float(np.max(v[mask] * W[mask]))


def _compute_hint(a, L=None, n=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size) if n is None else int(n)
    if n <= 0:
        return 1e300
    if L is None:
        L = _fft_len(n)
    try:
        C = _autocorr(a, L, n)
        v = float(n) * C
        full = 2.0 * float(np.max(v)) if v.size else 0.0
        off = 2.0 * float(np.max(v[1:])) if n > 1 else 0.0
        hint = min(full, off)
        if not math.isfinite(hint):
            hint = 1e300
        return float(hint)
    except Exception:
        return 1e300

# Candidate families

def _period2_array(n, r):
    n = int(max(1, n))
    r = float(r)
    r = min(0.5, max(0.0, r))
    a = np.empty(n, dtype=np.float64)
    a[0::2] = 1.0 - r
    a[1::2] = r
    return _normalize1(a)


def _sin_sequence(n, freq, amp, phase):
    n = int(max(1, n))
    i = np.arange(n, dtype=np.float64)
    a = np.exp(float(amp) * np.cos(2.0 * math.pi * float(freq) * i + float(phase)))
    return _normalize1(a)


def _pattern_repeat(n, pattern):
    n = int(max(1, n))
    p = np.asarray(pattern, dtype=np.float64).ravel()
    if p.size == 0:
        p = np.ones(2, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    if p.sum() <= 0.0:
        p = np.ones(p.size, dtype=np.float64)
    reps = n // p.size + 1
    a = np.tile(p, reps)[:n]
    return _normalize1(a)


def _sparse_random(n, rng, K):
    n = int(max(1, n))
    K = int(max(1, min(K, n)))
    idx = rng.choice(n, size=K, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.exponential(1.0, K) + 0.1
    return _normalize1(a)


def _sparse_equal(n, K):
    n = int(max(1, n))
    K = int(max(1, min(K, n)))
    step = max(1, n // K)
    a = np.zeros(n, dtype=np.float64)
    i = 0
    count = 0
    while i + 1 <= n and count < K:
        a[i] = 1.0
        i += step
        count += 1
    if count == 0:
        a[0] = 1.0
    return _normalize1(a)


def _random_prior(n, rng):
    n = int(max(1, n))
    base = rng.exponential(1.0, n)
    freqs = (0.0, 0.25, 1.0 / 3.0, 0.5)
    idx = int(rng.integers(0, len(freqs)))
    ph = float(rng.uniform(0.0, 2.0 * math.pi))
    amp = float(0.5 + 2.0 * rng.random())
    t = np.linspace(0.0, 1.0, n)
    mod = np.exp(amp * np.cos(2.0 * math.pi * freqs[idx] * t + ph))
    lam = float(np.clip(np.exp(float(rng.normal(math.log(0.95), 0.15))), 0.55, 0.999))
    geo = lam ** t
    geo = 0.5 * geo + 0.5 * geo[::-1]
    a = 0.35 * base / float(n) + 0.35 * mod + 0.30 * geo
    return _normalize1(a)


def _shape_candidates(n, rng):
    n = int(max(1, n))
    out = []
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0

    out.append(_uniform1(n))
    out.append(_normalize1(np.maximum(0.0, 1.0 - np.abs(x))))
    out.append(_normalize1(np.maximum(0.0, 1.0 - np.abs(x)) ** 2))
    out.append(_normalize1(np.sin(math.pi * t) ** 2))

    for lam in (0.70, 0.85, 0.90, 0.95, 0.98, 0.995):
        g = (float(lam) ** t)
        out.append(_normalize1(g))
        out.append(_normalize1(g[::-1]))

    frq = (0.25, 1.0 / 3.0, 0.40, 0.50)
    amp_list = (0.5, 1.0, 2.0, 4.0, 8.0)
    for amp in amp_list:
        for f in frq:
            for ph in (0.0, math.pi / 4.0, math.pi / 2.0):
                out.append(_sin_sequence(n, f, amp, ph))

    for p in (2, 3, 4, 5):
        for _ in range(6):
            pat = rng.exponential(1.0, p) + 0.1
            out.append(_pattern_repeat(n, pat))

    for r in (0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.45):
        out.append(_period2_array(n, r))

    for K in (1, 2, 5, 10, 20, 50):
        out.append(_sparse_random(n, rng, K))
        out.append(_sparse_equal(n, K))

    return out

def _oracle_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        # Fallback only if the oracle is unavailable in this environment.
        try:
            a = np.asarray(seq, dtype=np.float64).ravel()
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S < 0.01:
                return float("inf")
            n = int(a.size)
            L = _fft_len(n)
            C = _autocorr(a, L, n)
            return float(2.0 * n * float(np.max(C)) / (S * S))
        except Exception:
            return float("inf")
    except Exception:
        return float("inf")
    try:
        val = float(val)
    except Exception:
        return float("inf")
    if not math.isfinite(val):
        return float("inf")
    return val


def _make_state():
    return {
        "best_val": float("inf"),
        "best_seq": None,
        "best_tag": "",
        "oracle_calls": 0
    }


def _update_state(state, val, seq, tag=""):
    try:
        val = float(val)
    except Exception:
        return
    if math.isfinite(val) and val < state["best_val"]:
        state["best_val"] = val
        state["best_seq"] = seq
        state["best_tag"] = str(tag)


def _add_candidate(pool, arr, tag, rank=5, L=None, n=None):
    a = _normalize1(arr)
    n = int(a.size) if n is None else int(n)
    if n <= 0:
        return -1
    if L is None:
        L = _fft_len(n)
    hint = _compute_hint(a, L, n)
    if not math.isfinite(hint):
        hint = 1e300
    entry = {
        "arr": a.copy(),
        "tag": str(tag),
        "rank": int(rank),
        "hint": float(hint),
        "oracle": float("inf")
    }
    if len(pool) < 1200:
        pool.append(entry)
        return len(pool) - 1
    # Replace the worst unevaluated hint if this one is better.
    worst = -1
    worst_hint = -1.0
    for i, e in enumerate(pool):
        if e["oracle"] >= float("inf") and e["hint"] > worst_hint:
            worst_hint = e["hint"]
            worst = i
    if worst >= 0 and hint < worst_hint:
        pool[worst] = entry
        return worst
    return -1


def _pool_best_index(pool):
    best_o = float("inf")
    best_i = -1
    for i, e in enumerate(pool):
        if e["oracle"] < best_o:
            best_o = e["oracle"]
            best_i = i
    if best_i >= 0:
        return best_i
    best_h = float("inf")
    best_i = -1
    for i, e in enumerate(pool):
        if e["hint"] < best_h:
            best_h = e["hint"]
            best_i = i
    return best_i


def _evaluate_entry(state, pool, idx, hard, force=False):
    if idx < 0 or idx >= len(pool):
        return float("inf")
    entry = pool[idx]
    if entry["oracle"] < float("inf"):
        return float(entry["oracle"])
    if not force and state["oracle_calls"] >= 400:
        return float("inf")
    if time.monotonic() > hard - 5.0:
        return float("inf")
    seq = _to_final_list(entry["arr"])
    val = _oracle_score(seq)
    entry["oracle"] = val
    state["oracle_calls"] += 1
    _update_state(state, val, seq, entry["tag"])
    return val


# ----------------------------------------------------------------------------
# Soft-max / multiplicative optimizer for weighted convolution objectives
# ----------------------------------------------------------------------------

def _softmax_q(v, W, temp):
    n = int(v.size)
    mask = W > 1e-300
    if not np.any(mask):
        q = np.full(n, 1.0 / max(1, n), dtype=np.float64)
        return q, 0.0
    vm = v[mask]
    logw = np.log(W[mask])
    vmax = float(np.max(vm))
    logwmax = float(np.max(logw))
    z = float(temp) * (vm - vmax) + logw - logwmax
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 0.0:
        e = np.ones_like(vm)
        s = float(e.sum())
    qm = e / s
    q = np.zeros(n, dtype=np.float64)
    q[mask] = qm
    J = float(temp) * vmax + logwmax + math.log(s)
    return q, J


def _gradient_from_fft(A, q, L, n):
    r = np.zeros(L, dtype=np.float64)
    r[0] = 2.0 * q[0]
    if n > 1:
        r[1:n] = q[1:n]
        r[L - (n - 1):L] = q[n - 1:0:-1]
    R = np.fft.rfft(r, L)
    G = np.fft.irfft(A * R, L)[:n]
    np.maximum(G, 0.0, out=G)
    return G


def _smooth_weights(W, w=7):
    W = np.asarray(W, dtype=np.float64).ravel().copy()
    n = int(W.size)
    if n <= 1:
        return W
    w = int(max(3, min(2 * n - 1, w)))
    k = np.ones(w, dtype=np.float64) / float(w)
    y = np.convolve(W, k, mode="same")
    y = np.maximum(y, 0.0)
    m = float(y.max())
    if m > 0.0:
        y = y / m
    else:
        y = np.ones(n, dtype=np.float64)
    return y


def _estimate_weights(a, score, n, L, floor=0.0):
    a = _normalize1(a)
    n = int(a.size)
    C = _autocorr(a, L, n)
    S2 = float(a.sum() * a.sum())
    if S2 <= 1e-300:
        return np.ones(n, dtype=np.float64)
    raw = (2.0 * float(n) * C) / S2
    W = np.zeros(n, dtype=np.float64)
    if not math.isfinite(score) or score <= 1e-12:
        W.fill(1.0)
        return W

    floor = float(max(0.0, floor))

    # Center weight.  If the center contribution alone would exceed the oracle
    # score, discount it proportionally.
    if raw[0] > score * 1.02:
        W[0] = max(floor, score / max(float(raw[0]), 1e-300))
    else:
        W[0] = 1.0

    rs = raw[1:]
    Ws = np.zeros(n - 1, dtype=np.float64)
    if n > 1:
        high = rs > score
        if np.any(high):
            Ws[high] = np.minimum(1.0, score / rs[high])
        med = (~high) & (rs > 0.4 * score)
        Ws[med] = 1.0
        W[1:] = np.maximum(Ws, floor)

    wm = float(W.max())
    if wm <= 0.0:
        W.fill(1.0)
    else:
        W = W / wm
    return W


def _soft_search(a_init, W, time_limit, rng, temp0=10.0, temp_end=500.0, eta_start=0.10):
    a_init = _normalize1(a_init)
    n = int(a_init.size)
    if n <= 1:
        return a_init, 0.0
    if time_limit is None or time_limit <= 0.1:
        return a_init, 0.0

    W = np.asarray(W, dtype=np.float64).ravel()
    if W.size != n:
        W = np.ones(n, dtype=np.float64)
    W = np.maximum(W, 0.0)
    wm = float(W.max())
    if wm <= 0.0:
        W = np.ones(n, dtype=np.float64)
    else:
        W = W / wm

    L = _fft_len(n)

    # If center weight dominates, the surrogate is exactly the l2 penalty and
    # the uniform vector is optimal.
    if n > 1:
        wside = float(np.max(W[1:]))
        if W[0] >= wside - 1e-12 and W[0] > 0.0:
            return _uniform1(n), 2.0 * float(W[0])
        if W[0] <= 1e-12:
            return _spike1(n, 0), 0.0

    a = _normalize1(a_init)
    A = np.fft.rfft(a, L)
    C = _autocorr_from_fft(A, L, n)
    v = float(n) * C
    cur_obj = 2.0 * _wm_obj(v, W)
    best_obj = cur_obj
    best_a = a.copy()

    temp = float(temp0)
    temp_end = float(max(temp0, temp_end))
    eta = float(eta_start)
    no_accept = 0
    since_best = 0
    it = 0

    deadline = time.monotonic() + float(time_limit)

    while time.monotonic() < deadline:
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_accept = 0
            since_best = 0
        else:
            since_best += 1

        if since_best > 350:
            a = best_a.copy()
            A = np.fft.rfft(a, L)
            C = _autocorr_from_fft(A, L, n)
            v = float(n) * C
            cur_obj = 2.0 * _wm_obj(v, W)
            temp = max(float(temp0), min(temp, temp * 0.5))
            eta = float(eta_start)
            no_accept = 0
            since_best = 0
            continue

        if no_accept > 90:
            probe = a.copy()
            probe_obj = cur_obj

            uni = _uniform1(n)
            for mix in (0.10, 0.25, 0.45):
                p = _normalize1((1.0 - mix) * a + mix * uni)
                Ap = np.fft.rfft(p, L)
                Cp = _autocorr_from_fft(Ap, L, n)
                op = 2.0 * _wm_obj(float(n) * Cp, W)
                if op < probe_obj:
                    probe_obj = op
                    probe = p

            prior = _random_prior(n, rng)
            for mix in (0.15, 0.30, 0.55):
                p = _normalize1((1.0 - mix) * a + mix * prior)
                Ap = np.fft.rfft(p, L)
                Cp = _autocorr_from_fft(Ap, L, n)
                op = 2.0 * _wm_obj(float(n) * Cp, W)
                if op < probe_obj:
                    probe_obj = op
                    probe = p

            a = probe
            A = np.fft.rfft(a, L)
            C = _autocorr_from_fft(A, L, n)
            v = float(n) * C
            cur_obj = probe_obj
            no_accept = 0
            since_best = 0
            temp = max(float(temp0), temp * 0.6)
            eta = float(eta_start)
            continue

        q, cur_soft = _softmax_q(v, W, temp)
        G = _gradient_from_fft(A, q, L, n)
        gmax = float(np.max(G))
        if gmax <= 1e-300:
            return best_a, best_obj

        d = G / gmax
        accepted = False

        for step in (eta, 0.5 * eta, 0.25 * eta):
            if step <= 1e-8:
                continue
            fac = np.exp(np.clip(-step * d, -700.0, 0.0))
            a2 = _normalize1(a * fac)
            A2 = np.fft.rfft(a2, L)
            C2 = _autocorr_from_fft(A2, L, n)
            v2 = float(n) * C2
            obj2 = 2.0 * _wm_obj(v2, W)
            q2, soft2 = _softmax_q(v2, W, temp)
            if obj2 < cur_obj - 1e-12 or soft2 < cur_soft - 1e-8:
                a = a2
                A = A2
                C = C2
                v = v2
                cur_obj = obj2
                cur_soft = soft2
                accepted = True
                break

        if accepted:
            eta = min(0.80, eta * 1.12)
        else:
            no_accept += 1
            eta = max(0.010, eta * 0.72)

        it += 1
        if it % 50 == 0:
            temp = min(temp_end, temp * 1.25)

    return best_a, best_obj

def _diagnostics_evaluate(pool, state, hard, diag_deadline):
    order = []
    for i, e in enumerate(pool):
        order.append((e["rank"], e["hint"], i))
    order.sort()
    eval_count = 0
    for _r, _h, i in order:
        now = time.monotonic()
        if now > diag_deadline or eval_count >= 35 or state["oracle_calls"] >= 60:
            break
        if hard - now < 5.0:
            break
        val = _evaluate_entry(state, pool, i, hard)
        eval_count += 1
        if state["best_val"] < float("inf"):
            print("[diag] eval=%d best=%.10f tag=%s calls=%d" %
                  (eval_count, state["best_val"], state["best_tag"], state["oracle_calls"]),
                  flush=True)


def _profile_search(pool, state, arr_best, score_best, deadline, rng, n, L, hard):
    if time.monotonic() >= deadline - 10.0:
        return
    n = int(max(1, n))
    L = _fft_len(n)
    uniform = _uniform1(n)
    period = _period2_array(n, 0.35)
    sparse = _sparse_equal(n, max(1, min(30, n // 12 if n > 12 else 1)))
    random = _random_prior(n, rng)

    tasks = []

    if math.isfinite(score_best) and score_best > 0.0:
        W_est = _estimate_weights(arr_best, score_best, n, L, floor=0.0)
        tasks.append((W_est, arr_best, "est_best"))
        tasks.append((_smooth_weights(W_est, 9), arr_best, "est_best_sm"))
        tasks.append((_estimate_weights(arr_best, score_best, n, L, floor=0.02),
                      arr_best, "est_best_floor"))

    for alpha in (0.25, 0.50, 0.75):
        W = np.ones(n, dtype=np.float64)
        W[0] = float(alpha)
        tasks.append((W, period, "mix_a=%.2f" % alpha))

    k = np.arange(n, dtype=np.float64)
    denom = max(1.0, float(n) / 20.0)
    for alpha in (0.50, 0.75):
        W = np.ones(n, dtype=np.float64)
        if n > 1:
            W[1:] = 1.0 / (1.0 + k[1:] / denom)
        W[0] = float(alpha)
        m = float(W.max())
        if m > 0:
            W = W / m
        tasks.append((W, uniform, "decay_a=%.2f" % alpha))

    K = min(max(2, n // 10 if n > 10 else 2), 220)
    W = np.zeros(n, dtype=np.float64)
    W[0] = 0.70
    if K > 1:
        W[1:K] = 1.0
    tasks.append((W, uniform, "lowlags"))

    W = np.ones(n, dtype=np.float64)
    W[0] = 0.05
    tasks.append((W, sparse, "lowcenter_sparse"))

    W = np.ones(n, dtype=np.float64)
    W[0] = 0.50
    tasks.append((W, random, "random_mix"))

    if not math.isfinite(score_best):
        W = np.ones(n, dtype=np.float64)
        W[0] = 0.50
        tasks.append((W, uniform, "fallback_mix"))

    ntasks = max(1, len(tasks))
    per_task = max(12.0, min(80.0, float(max(1.0, (deadline - time.monotonic()))) / float(ntasks)))

    for tag_base, init, W in tasks:
        now = time.monotonic()
        rem = min(deadline, hard - 5.0) - now
        if rem < 8.0:
            break
        t = min(per_task, rem - 4.0)
        if t < 6.0:
            continue
        try:
            cand, _obj = _soft_search(init, W, t, rng)
        except Exception:
            cand = init
        if cand is None:
            cand = init
        tag = tag_base
        idx = _add_candidate(pool, cand, tag, rank=6, L=L, n=n)
        now = time.monotonic()
        if idx >= 0 and hard - now > 25.0:
            _evaluate_entry(state, pool, idx, hard)
            if state["best_val"] < float("inf"):
                print("[profile] best=%.10f tag=%s calls=%d" %
                      (state["best_val"], state["best_tag"], state["oracle_calls"]),
                      flush=True)


def _oracle_adaptive_search(pool, state, a_init, init_score, deadline, rng, n, L, hard):
    if not math.isfinite(init_score) or init_score <= 1e-8:
        return
    n = int(max(1, n))
    L = _fft_len(n)
    best_a = _normalize1(a_init)
    best_s = float(init_score)

    idx = _add_candidate(pool, best_a, "adaptive_init", rank=1, L=L, n=n)
    if idx >= 0:
        _evaluate_entry(state, pool, idx, hard)

    no_improve = 0
    while time.monotonic() < deadline - 25.0:
        if best_s <= 1e-8:
            break
        now = time.monotonic()
        chunk = min(45.0, max(5.0, (min(deadline, hard - 10.0) - now) * 0.35))
        if chunk < 5.0:
            break

        try:
            W = _estimate_weights(best_a, best_s, n, L, floor=0.0)
            Wsm = _smooth_weights(W, 9)
            cand, _ = _soft_search(best_a, W, chunk, rng)
        except Exception:
            cand = best_a

        idx = _add_candidate(pool, cand, "adaptive_direct", rank=1, L=L, n=n)
        if idx >= 0:
            val = _evaluate_entry(state, pool, idx, hard)
        else:
            seq = _to_final_list(cand)
            val = _oracle_score(seq)
            state["oracle_calls"] += 1
            _update_state(state, val, seq, "adaptive_direct")

        if math.isfinite(val) and val < best_s - 1e-12:
            best_s = float(val)
            seq_c = _to_final_list(cand)
            best_a = _normalize1(seq_c)
            no_improve = 0
            print("[adaptive] improve %.10f calls=%d" % (best_s, state["oracle_calls"]), flush=True)
        else:
            no_improve += 1

        if no_improve >= 1 and time.monotonic() < deadline - 25.0:
            chunk2 = min(30.0, max(5.0, (min(deadline, hard - 10.0) - time.monotonic()) * 0.30))
            if chunk2 >= 5.0:
                try:
                    cand2, _ = _soft_search(best_a, Wsm, chunk2, rng)
                except Exception:
                    cand2 = best_a
                idx2 = _add_candidate(pool, cand2, "adaptive_smooth", rank=1, L=L, n=n)
                if idx2 >= 0:
                    val2 = _evaluate_entry(state, pool, idx2, hard)
                else:
                    seq2 = _to_final_list(cand2)
                    val2 = _oracle_score(seq2)
                    state["oracle_calls"] += 1
                    _update_state(state, val2, seq2, "adaptive_smooth")
                if math.isfinite(val2) and val2 < best_s - 1e-12:
                    best_s = float(val2)
                    seq_c = _to_final_list(cand2)
                    best_a = _normalize1(seq_c)
                    no_improve = 0
                    print("[adaptive-smooth] improve %.10f calls=%d" %
                          (best_s, state["oracle_calls"]), flush=True)

        if no_improve >= 3 and time.monotonic() < deadline - 25.0:
            prior = _random_prior(n, rng)
            for mix in (0.15, 0.30, 0.50):
                probe = _normalize1((1.0 - mix) * best_a + mix * prior)
                idxp = _add_candidate(pool, probe, "adaptive_probe", rank=2, L=L, n=n)
                if idxp >= 0:
                    vp = _evaluate_entry(state, pool, idxp, hard)
                else:
                    vp = _oracle_score(_to_final_list(probe))
                    state["oracle_calls"] += 1
                    _update_state(state, vp, _to_final_list(probe), "adaptive_probe")
                if math.isfinite(vp) and vp < best_s - 1e-12:
                    best_s = float(vp)
                    seq_c = _to_final_list(probe)
                    best_a = _normalize1(seq_c)
                    no_improve = 0
                    print("[adaptive-probe] improve %.10f calls=%d" %
                          (best_s, state["oracle_calls"]), flush=True)
                    break
            else:
                no_improve = 1

    # Finalize the adaptive best in the pool.
    idxf = _add_candidate(pool, best_a, "adaptive_final", rank=1, L=L, n=n)
    if idxf >= 0:
        _evaluate_entry(state, pool, idxf, hard, force=True)


def _final_pool_evaluation(pool, state, hard):
    order = []
    for i, e in enumerate(pool):
        if e["oracle"] < float("inf"):
            order.append((0, e["oracle"], e["hint"], i))
        else:
            order.append((1, e["hint"], e["hint"], i))
    order.sort()
    evaluated = 0
    for _kind, _val, _hint, i in order:
        if time.monotonic() > hard - 5.0:
            break
        if evaluated >= 250:
            break
        if pool[i]["oracle"] >= float("inf"):
            _evaluate_entry(state, pool, i, hard, force=True)
            evaluated += 1

def _get_parent_raw():
    try:
        p = height_sequence_1
    except NameError:
        return None
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    return arr

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Return the best non-negative sequence found within the time budget."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not math.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = max(30.0, budget)

    t0 = time.monotonic()
    hard = t0 + max(10.0, budget - 2.0)
    work_deadline = t0 + max(20.0, budget - 120.0)

    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    state = _make_state()
    pool = []

    parent = _get_parent_raw()

    # Choose primary search length.  The known parent length is usually ideal.
    if parent is not None and 8 <= parent.size <= 20000:
        n = int(parent.size)
    else:
        n = 6121
    L = _fft_len(n)

    # Add parent-derived candidates.
    if parent is not None and parent.size > 0:
        pos = np.maximum(parent, 0.0)
        absP = np.abs(parent)
        for tag, tag_arr, rank in (
            ("parent_clip", pos, 1),
            ("parent_abs", absP, 2),
            ("parent_raw_clipped", parent, 1)
        ):
            if tag_arr.size <= 20000:
                _add_candidate(pool, tag_arr, tag, rank=rank, L=None, n=None)

    # Main-length special/schedule candidates.
    _add_candidate(pool, _uniform1(n), "constant_n", rank=0, L=L, n=n)
    _add_candidate(pool, _spike1(n, 0), "spike_n", rank=2, L=L, n=n)
    _add_candidate(pool, _two_spike1(n), "two_n", rank=3, L=L, n=n)

    shapes = _shape_candidates(n, rng)
    for i, s in enumerate(shapes):
        tag = "shape_%03d" % i
        _add_candidate(pool, s, tag, rank=4, L=L, n=n)

    # A few alternative lengths, mainly for cheap special candidates.
    alt_lengths = []
    for m in (4096, 8192, 12242, 16384):
        if m != n and 16 <= m <= 20000:
            alt_lengths.append(m)
        if len(alt_lengths) >= 3:
            break
    for m in alt_lengths:
        Lm = _fft_len(m)
        _add_candidate(pool, _uniform1(m), "constant_%d" % m, rank=0, L=Lm, n=m)
        _add_candidate(pool, _spike1(m, 0), "spike_%d" % m, rank=2, L=Lm, n=m)
        _add_candidate(pool, _period2_array(m, 0.35), "period2_%d" % m, rank=4, L=Lm, n=m)

    # Diagnostics: evaluate a small ordered subset with the real oracle.
    diag_deadline = min(work_deadline, t0 + 45.0)
    _diagnostics_evaluate(pool, state, hard, diag_deadline)

    if state["best_seq"] is not None and state["best_val"] < 1e-4:
        print("[early] near-zero score %.12f tag=%s" % (state["best_val"], state["best_tag"]), flush=True)
        return state["best_seq"]

    # Pick a primary best candidate and possibly switch to its length.
    best_idx = _pool_best_index(pool)
    if best_idx < 0:
        best_arr = _uniform1(n)
        best_score = float("inf")
    else:
        best_arr = pool[best_idx]["arr"]
        best_score = pool[best_idx]["oracle"] if pool[best_idx]["oracle"] < float("inf") else state["best_val"]
        n = int(best_arr.size)
        L = _fft_len(n)

    # Decide how much heavy search to run.  If the constant vector is already
    # near the theoretical full-convolution optimum, keep the search modest.
    now = time.monotonic()
    if state["best_tag"] in ("constant_n", "constant") and state["best_val"] < 3.0:
        profile_time = min(420.0, max(0.0, work_deadline - now))
        adaptive_time = min(240.0, max(0.0, work_deadline - now - profile_time))
    else:
        profile_time = min(1250.0, max(0.0, work_deadline - now))
        adaptive_time = min(900.0, max(0.0, work_deadline - now - profile_time))

    profile_deadline = min(work_deadline, now + profile_time)
    if profile_time > 20.0:
        _profile_search(pool, state, best_arr, best_score, profile_deadline, rng, n, L, hard)

    # Refresh best candidate for adaptive phase, preferably from oracle pool.
    best_idx = _pool_best_index(pool)
    if best_idx >= 0:
        best_arr = pool[best_idx]["arr"]
        best_score = pool[best_idx]["oracle"] if pool[best_idx]["oracle"] < float("inf") else state["best_val"]
        n = int(best_arr.size)
        L = _fft_len(n)

    adaptive_deadline = min(work_deadline, now + adaptive_time)
    if adaptive_time > 30.0 and math.isfinite(state["best_val"]):
        _oracle_adaptive_search(pool, state, best_arr, best_score, adaptive_deadline, rng, n, L, hard)

    # If something very good was found, stop early but safely.
    if state["best_seq"] is not None and state["best_val"] < 1e-4:
        print("[early-final] near-zero score %.12f tag=%s" %
              (state["best_val"], state["best_tag"]), flush=True)
        return state["best_seq"]

    # Final sweep over remaining top candidates, using the actual oracle if time permits.
    _final_pool_evaluation(pool, state, hard)

    fallback = _to_final_list(_uniform1(n))
    if state["best_seq"] is None:
        return fallback
    return state["best_seq"]
```
