import numpy as np
from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score(a):
    n = len(a)
    c = fast_self_conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def solve_lp(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M elementwise."""
    n = len(a)
    # conv(a,b)[k] = sum_{i} a[i] * b[k-i] for valid i
    # This is a linear function of b.
    # Build the constraint matrix A_ub such that A_ub @ b <= M * ones(2n-1)
    
    # For each k (0 to 2n-2), conv(a,b)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * b[k-i]
    # Let j = k-i, so j ranges from max(0,k-n+1) to min(k, n-1), and i = k-j
    # conv(a,b)[k] = sum_j a[k-j] * b[j] for j in [max(0,k-n+1), min(k,n-1)]
    
    # This is a Toeplitz-like matrix. Let's build it.
    # A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # i.e., A_ub[k, j] = a[k-j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    # For efficiency, we can use a sparse matrix.
    from scipy.sparse import csr_matrix
    
    # Actually for n ~ 2000, the dense matrix is 4095 x 2048 which is about 8M entries.
    # That's manageable but sparse would be faster.
    
    # Let me use the Toeplitz structure. The matrix is a semi-infinite Toeplitz matrix.
    # Row k has non-zeros at positions j where 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k.
    
    # For k=0: only j=0, value a[0]
    # For k=1: j=0,1, values a[1], a[0]
    # For k=2: j=0,1,2, values a[2], a[1], a[0]
    # ...
    # For k=n-1: j=0,...,n-1, values a[n-1], a[n-2], ..., a[0]
    # For k=n: j=1,...,n-1, values a[n-1], a[n-2], ..., a[1]
    # ...
    # For k=2n-2: j=n-1, value a[n-1]
    
    # This is the "convolution matrix" or "Toeplitz matrix" for the convolution.
    
    # Let me just build it explicitly for n up to ~4096.
    # Actually, for n=2048, the matrix is (2n-1) x n = 4095 x 2048, which is fine.
    
    rows = []
    cols = []
    vals = []

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(b) = -minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:

from scipy.sparse import diags
import time

    n, m = len(a), len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()

    B = np.fft.rfft(b, M_fft)

    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

    """Build sparse (2n-1) x n matrix for conv(a, .)"""
    diags_list = []
    offsets = []
    for d in range(n):
        diags_list.append(np.full(n, a[d]))
        offsets.append(d)
    return diags(diags_list, offsets, shape=(2*n-1, n), format='csr')

def lp_solve(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)
    A = build_conv_A(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None

    a = a / S
    best_score = compute_score(a)
    best_a = a.copy()

    for iteration in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        S_curr = a.sum()
        curr_score = 2.0 * n * M / (S_curr * S_curr)

            best_score = curr_score
            best_a = a.copy()

        if b is None:

            b = lp_solve(a, M * 0.9)

LP-based optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Core algorithm (from state-of-the-art):
  1. Given current sequence a, compute M = max(conv(a,a)).
  2. Solve LP: max sum(b) s.t. b >= 0, conv(a,b) <= M.
  3. Rescale b to match sum(a), update a = (1-t)a + tb.
  4. Repeat with adaptive step size and periodic restarts.

This avoids the soft-max approximation used in gradient methods and
directly solves the relevant relaxation at each step.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags
import time


def fast_conv(a, b=None):
    """Linear convolution of 1-D arrays via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute the evaluation score (internal, fast version)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_A_sparse(n, a):
    """Build the (2n-1) x n sparse matrix A where A[k,j] = a[k-j] (valid indices).
    This matrix satisfies: conv(a, b)[k] = sum_j A[k,j] * b[j]."""
    # The matrix has n superdiagonals (offsets 0 to n-1), each of length n.
    # The d-th superdiagonal (offset d) has constant value a[d].
    diags_list = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    return diags(diags_list, offsets, shape=(2 * n - 1, n), format='csr')


def solve_lp_step(a, M, deadline_check=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b or None if infeasible/failed."""
    n = len(a)
    A = build_conv_A_sparse(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    
    # Use HiGHS for speed
    options = {
        'presolve': True,
        'dual_feasibility_tolerance': 1e-9,
        'primal_feasibility_tolerance': 1e-9,
        'time_limit': 60,
    }
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs', options=options)
    
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_iteration(a, t_step, rng, max_iters=5000):
    """Run the LP-based iteration from sequence a."""
    n = len(a)
    S0 = a.sum()
    a = a / S0  # Normalize to sum = 1
    best_score = compute_score(a)
    best_a = a.copy()
    no_improve = 0
    t = t_step
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline_check:
            break
        
        c = fast_conv(a)
        M = float(np.max(c))
        curr_score = 2.0 * n * M  # Since sum(a) = 1
        
        if curr_score < best_score - 1e-9:
            best_score = curr_score
            best_a = a.copy()
            no_improve = 0
            # Gradually increase step size when making progress
            t = min(0.3, t * 1.02)
        else:
            no_improve += 1
            # Reduce step size when stalling
            if no_improve % 50 == 0:
                t = max(0.01, t * 0.8)
        
        if no_improve > 300:
            # Restart with perturbation
            perturbation = 0.02 * rng.standard_normal(n)
            a = best_a + perturbation
            a = np.maximum(a, 0)
            a = a / a.sum()
            t = t_step
            no_improve = 0
            stall_count += 1
            if stall_count > 20:

            continue

        b = solve_lp_step(a, M, deadline_check)
        if b is None:

            b = solve_lp_step(a, M * 1.01, deadline_check)

        if S_b < 1e-10:

        if S_new > 1e-10:
            a = a_new / S_new
        else:
            a = best_a
    
    return best_score, best_a, t

        M = float(np.max(c))
        curr_score = 2.0 * n * M
        iter_count += 1

        if curr_score < best_score - 1e-10:
            best_score = curr_score
            best_a = a.copy()
            no_improve = 0
            # Slightly increase step when making progress
            t = min(0.25, t * 1.01)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                t = max(0.005, t * 0.85)

        if no_improve > 200:
            stall_rounds += 1
            if stall_rounds > 30:
                break
            # Perturb and restart
            noise_scale = 0.01 + 0.02 * stall_rounds * 0.1
            pert = noise_scale * rng.standard_normal(n)
            a = best_a + pert
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa < 1e-10:
                a = best_a.copy()
            else:
                a = a / sa
            t = t_init
            no_improve = 0
            continue

        # Solve LP
        b = solve_lp(a, M)
        if b is None:
            # Try slightly relaxed constraint
            b = solve_lp(a, M * 1.001, time_limit=10)
            if b is None:
                t *= 0.5
                no_improve += 5
                continue

        S_b = b.sum()
        if S_b < 1e-10:
            break

        # Rescale b to sum=1
        b_norm = b / S_b

        # Convex combination update
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new > 1e-10:
            a = a_new / S_new
        else:
            a = best_a.copy()

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"    iter {iter_count}: score={curr_score:.8f}, "
                  f"best={best_score:.8f}, t={t:.4f}")

    if verbose:
        print(f"  LP opt done: {iter_count} iters, best={best_score:.8f}")

    return best_score, best_a


def make_init_sinusoidal(n, freqs, amplitudes, rng):
    """Create a sinusoidal mix initialization."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f, amp in zip(freqs, amplitudes):
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * f * x + phase)
    return np.maximum(a, 0.01)


def make_init_ramp(n, alpha, rng):
    """Ramp-like initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * x - 1.0)
    return np.maximum(a, 0.01)


def make_init_bell(n, center, width):
    """Gaussian bell initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + np.exp(-((x - center) / width) ** 2)
    return np.maximum(a, 0.01)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Uses LP-based iterative method: at each step, solve an LP to find
    the best direction b satisfying conv(a,b) <= max(conv(a,a)),
    then take a convex combination step.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    global_deadline = start + total - 15.0  # Safety buffer

    # --- Get initial sequence ---
    n = 2048
    a_init = None
    parent_n = 0

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            parent = np.clip(parent, 0, 1000)
            if parent.sum() > 0.01 and parent.size >= 64:
                a_init = parent
                parent_n = parent.size
                n = parent.size
        except Exception:
            pass

    if a_init is None:
        a_init = make_init_sinusoidal(n, [1, 3, 7], [0.4, 0.3, 0.2], rng)

    # Cap n for LP feasibility
    n_max = 4096
    if n > n_max:
        # Take a centered window of length n_max
        center = n // 2
        half = n_max // 2
        a_init = a_init[center - half: center + half]
        n = n_max
    n = len(a_init)
    print(f"Starting: n={n}, parent_n={parent_n}")

    S0 = a_init.sum()
    a_norm = a_init / S0
    init_score = compute_score(a_norm)
    print(f"Initial score: {init_score:.8f}")

    best_score = init_score
    best_a = a_norm.copy()

    def try_update(sc, arr):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # --- Phase 1: Main LP optimization from parent (50% time) ---
    phase1_end = start + 0.50 * (total - 15.0)
    print(f"Phase 1: LP optimization (t=0.1), deadline at +{phase1_end - start:.0f}s")
    sc, aa = lp_optimize(a_norm, 0.1, phase1_end, max_iters=500000, verbose=True)
    try_update(sc, aa)

    # --- Phase 2: Refine with smaller step (20% time) ---
    phase2_end = start + 0.70 * (total - 15.0)
    if best_score < 2.0:  # Only refine if we're in a reasonable range
        print(f"Phase 2: Refine (t=0.04), deadline at +{phase2_end - start:.0f}s")
        sc, aa = lp_optimize(best_a, 0.04, phase2_end, max_iters=500000, verbose=True)
        try_update(sc, aa)

    # --- Phase 3: Try alternative initializations (15% time) ---
    phase3_end = start + 0.85 * (total - 15.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Alternative inits, deadline at +{phase3_end - start:.0f}s")
        alternates = [
            make_init_sinusoidal(n, [2, 5, 11], [0.5, 0.3, 0.2], rng),
            make_init_ramp(n, 1.2, rng),
            make_init_bell(n, 0.3, 0.15),
            make_init_sinusoidal(n, [4, 9], [0.4, 0.3], rng),
        ]
        time_per = max(5.0, (phase3_end - time.monotonic()) / len(alternates))
        for i, alt in enumerate(alternates):
            if time.monotonic() > phase3_end - 5:
                break
            alt_norm = alt / alt.sum()
            alt_score = compute_score(alt_norm)
            if alt_score >= best_score:
                # Blend with best
                alt_norm = 0.5 * best_a + 0.5 * alt_norm
                alt_norm = np.maximum(alt_norm, 0)
                alt_norm = alt_norm / alt_norm.sum()
            end_i = start + (time.monotonic() - start) + time_per
            end_i = min(end_i, phase3_end - 2)
            sc, aa = lp_optimize(alt_norm, 0.08, end_i, max_iters=100000,
                                 verbose=False)
            try_update(sc, aa)

    # --- Phase 4: Final polish (remaining time) ---
    if time.monotonic() < global_deadline:
        print(f"Phase 4: Final polish (t=0.015)")
        sc, aa = lp_optimize(best_a, 0.015, global_deadline,
                             max_iters=500000, verbose=True)
        try_update(sc, aa)

    # --- Final verification ---
    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - start
    print(f"\nFinal result: internal={best_score:.8f}, verified={verified:.8f}, "
          f"n={len(out)}, time={elapsed:.1f}s/budget={total:.0f}s")

    # If verification failed, try the parent sequence as fallback
    if not np.isfinite(verified) or verified > best_score + 0.1:
        print("WARNING: verification mismatch, trying parent fallback")
        if parent_n > 0 and a_init is not None:
            # a_init was normalized, so rescale
            parent_scale = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            out = [float(max(0, x)) for x in parent_scale]
            out = [min(1000.0, x) for x in out]
            verified2 = float(evaluate_sequence(out))
            if np.isfinite(verified2):
                print(f"Parent fallback score: {verified2:.8f}")
                return out
        # Ultimate fallback
        out = [1.0] * n
        return out

    return out
