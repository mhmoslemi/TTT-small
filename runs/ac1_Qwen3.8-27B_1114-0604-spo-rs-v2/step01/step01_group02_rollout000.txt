"""
Optimized search for non-negative sequences minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses log-domain gradient descent with adaptive peak weighting, peak deflation,
and structured restarts. Starts from parent sequence if available.
"""

import numpy as np
import time

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_fn(n, c, S):
    """Objective: 2n * max(c) / S^2"""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def softmax_weights(c, k, temp_factor):
    """Get soft-max weights for top-k peaks."""
    M = max(1, min(k, len(c)))
    idx = np.argpartition(c, -M)[-M:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = temp_factor * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(M) / M
    return idx, wts, float(np.dot(vals, wts))

def compute_grad(a, c, idx, wts, M_w):
    """Gradient of the soft-max weighted peak objective w.r.t. a.
    
    f = 2n * M_w / S^2
    df/d a_m = 2n/S^2 * sum_i w_i * 2*a[k_i - m] - 4n*M_w/S^3
    """
    n = len(a)
    S = float(a.sum())
    grad = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        # d c_ki / d a_m = 2 * a[ki - m] for valid m
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        # a[ki - m] for m in [j_start, j_end]
        # When m = j_start, ki - m = ki - j_start; when m = j_end, ki - m = ki - j_end
        # ki - j_start = ki - max(0, ki-n+1) = min(ki, n-1) = j_end
        # ki - j_end = ki - min(ki, n-1) = max(0, ki-n+1) = j_start
        # So a[ki-m] for m in [j_start, j_end] is a reversed slice of a[j_start:j_end+1]
        grad[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad / S2) - (4.0 * n * M_w) / S3
    return grad, S

def adam_step(params, grads, m, v, t, lr):
    """Adam update in log-domain."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    m[:] = b1 * m + (1.0 - b1) * grads
    v[:] = b2 * v + (1.0 - b2) * grads * grads
    m_hat = m / bc1
    v_hat = v / bc2
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Optimized search minimizing 2n*max(conv(a,a))/(sum a)^2 for non-negative sequences.
Uses direct-domain gradient descent with adaptive soft-max peak weighting,
Barzilai-Borwein-inspired step sizing, peak deflation restarts, and
multi-length exploration. Starts from parent sequence when available.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Return 2n * max(c) / S^2."""
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def get_topk_weights(c, k, temp_frac):
    """Return (indices, weights, weighted_max) for top-k peaks."""
    M = min(k, len(c))
    if M <= 0:
        return np.array([]), np.array([]), 0.0
    idx = np.argpartition(c, -M)[-M:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-14:
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        s = wts.sum()
        if s > 0:
            wts /= s
        else:
            wts = np.ones(M) / M
    else:
        wts = np.ones(M) / M
    return idx, wts, float(np.dot(vals, wts))


def gradient_weighted(a, c, idx, wts, M_w, n):
    """Compute gradient of 2n*M_w/S^2 w.r.t. a using weighted peaks."""
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    grad = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(ki, n - 1)
        if j_lo > j_hi:
            continue
        grad[j_lo:j_hi + 1] += wi * 2.0 * a[j_lo:j_hi + 1][::-1]
    return (2.0 * n) * (grad / S2) - (4.0 * n * M_w) / S3


def adam_update(a, g, m, v, t, lr):
    """Standard Adam in-place update."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a -= lr * mh / (np.sqrt(vh) + eps)


def peak_deflation(a, c, strength, rng):
    """Reduce top peak by decreasing its largest contributors."""
    n = len(a)
    peak_idx = int(np.argmax(c))
    j_lo = max(0, peak_idx - n + 1)
    j_hi = min(peak_idx, n - 1)
    if j_lo > j_hi:
        return a
    contrib = a[j_lo:j_hi + 1] * a[j_lo:j_hi + 1][::-1]
    n_top = max(1, min(10, (j_hi - j_lo + 1) // 4))
    top_local = np.argpartition(contrib, -n_top)[-n_top:]
    for tl in top_local:
        j = j_lo + tl
        a[j] *= (1.0 - strength * rng.random())
    np.clip(a, 0.0, None, out=a)
    return a


def run_optimization(a0, end_time, rng, lr_start, temp_start, k_peaks,
                     patience, deflate_every, deflate_strength):
    """Run gradient descent with restarts until end_time."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 1e-8, 1000.0).copy()
    # Normalize to sum=1 for stability
    S0 = a.sum()
    if S0 < 1e-6:
        a = np.ones(n) / n
    else:
        a /= S0

    c = fast_self_conv(a)
    score = compute_score(n, c, a.sum())
    best_score = score if np.isfinite(score) else np.inf
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_start
    no_imp = 0
    temp = temp_start
    steps_since_deflate = 0

    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = a.sum()
        if S < 1e-8:
            a = best_a.copy()
            c = fast_self_conv(a)
            S = a.sum()
        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= patience:
            # Restart strategy
            r = rng.random()
            if r < 0.4:
                # Small perturbation from best
                scale = 0.02 + 0.04 * rng.random()
                a = best_a * (1.0 + scale * rng.standard_normal(n))
                np.clip(a, 1e-8, None, out=a)
            elif r < 0.7:
                # Peak deflation
                c_best = fast_self_conv(best_a)
                a = best_a.copy()
                peak_deflation(a, c_best, deflate_strength, rng)
                if a.sum() < 1e-8:
                    a = best_a.copy()
            else:
                # Larger perturbation
                scale = 0.1 + 0.1 * rng.random()
                a = best_a * (1.0 + scale * rng.standard_normal(n))
                np.clip(a, 1e-8, None, out=a)

            # Renormalize
            a /= max(a.sum(), 1e-8)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.01, lr * 0.65)
            no_imp = 0
            temp = max(0.01, temp * 0.95)
            steps_since_deflate = 0
            continue

        # Compute gradient
        k_use = min(k_peaks, 2 * n - 1)
        idx, wts, M_w = get_topk_weights(c, k_use, temp)
        g = gradient_weighted(a, c, idx, wts, M_w, n)

        # Adam step
        t += 1
        lr_t = lr / (1.0 + 0.003 * t)
        adam_update(a, g, m, v, t, lr_t)
        np.clip(a, 0.0, None, out=a)

        # Renormalize to keep sum=1
        S = a.sum()
        if S > 1e-8:
            a /= S

        # Periodic peak deflation
        steps_since_deflate += 1
        if deflate_every > 0 and steps_since_deflate >= deflate_every:
            c_cur = fast_self_conv(a)
            peak_deflation(a, c_cur, 0.02 * deflate_strength, rng)
            S = a.sum()
            if S > 1e-8:
                a /= S
            steps_since_deflate = 0

        # Cool temperature slowly
        if t % 500 == 0:
            temp = max(0.01, temp * 0.997)

    return best_score, best_a
