"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def _conv_fft(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = rfft(f, size)
    conv = irfft(F * F, size)
    return conv[:2 * n - 1]


def _score(f):
    """Compute the objective: 2*n*max(conv)/(sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    conv = _conv_fft(f)
    return 2.0 * n * conv.max() / (s * s)


def _init_uniform(n):
    return np.ones(n)


def _init_cosine(n):
    j = np.arange(n)
    f = 1.0 + 0.5 * np.cos(2.0 * np.pi * j / n)
    return np.maximum(f, 0.1)


def _init_anti_palindrome(n):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # High at edges, low in center (anti-palindromic tendency)
    f = 0.5 + 0.5 * ((j - center) / center) ** 2
    return f


def _init_bump(n, pos=0.3, width=0.15):
    j = np.arange(n, dtype=np.float64) / n
    f = np.exp(-((j - pos) ** 2) / (2 * width ** 2)) + 0.05
    return f


def _init_two_bumps(n):
    j = np.arange(n, dtype=np.float64) / n
    f = (np.exp(-((j - 0.25) ** 2) / 0.01) +
         np.exp(-((j - 0.75) ** 2) / 0.01) + 0.05)
    return f


def _init_sine_mod(n, freq=3):
    j = np.arange(n, dtype=np.float64)
    f = 1.0 + 0.7 * np.sin(2.0 * np.pi * freq * j / n)
    return np.maximum(f, 0.01)


def _init_random(rng, n):
    return rng.exponential(1.0, size=n) + 0.01


def _normalize(f):
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _gradient_at_peak(conv, f, k_star, n):
    """Compute gradient of conv[k_star] w.r.t. f.
    grad[j] = f[k_star - j] if 0 <= k_star - j < n, else 0."""
    grad = np.zeros(n)
    # k_star - j must be in [0, n-1], so j in [k_star - (n-1), k_star]
    j_min = max(0, k_star - (n - 1))
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        src_indices = k_star - indices
        grad[indices] = f[src_indices]
    return grad


def _multi_peak_gradient(conv, f, n, top_k=5, temperature=0.01):
    """Weighted gradient from top-K peak lags."""
    # Get top K lag indices
    n_conv = len(conv)
    top_indices = np.argsort(conv)[::-1][:top_k]
    top_values = conv[top_indices]
    
    # Softmax weights
    vals = top_values / temperature
    vals -= vals.max()
    weights = np.exp(vals)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, k_star in enumerate(top_indices):
        if weights[i] < 1e-10:
            continue
        g = _gradient_at_peak(conv, f, k_star, n)
        grad += weights[i] * g
    return grad


def _backtrack_step(f, grad, conv_peak, n, max_iter=50):
    """Backtracking line search to reduce the peak."""
    f0 = f.copy()
    peak0 = conv_peak
    step = 0.5
    best_f = f0
    best_peak = peak0
    
    for _ in range(max_iter):
        f_try = f0 - step * grad
        f_try = np.maximum(f_try, 0.0)
        f_try = _normalize(f_try)
        
        conv_try = _conv_fft(f_try)
        peak_try = conv_try.max()
        
        if peak_try < best_peak:
            best_peak = peak_try
            best_f = f_try.copy()
            step *= 1.5  # Try larger step
        else:
            step *= 0.5
        
        if step < 1e-10:
            break
    
    return best_f, best_peak


def _optimize(f, deadline, rng, verbose=False, top_k=5, n_iter_max=500000):
    """Main optimization loop using exact peak gradient with backtracking."""
    f = np.maximum(f, 0.0)
    f = _normalize(f)
    n = len(f)
    
    conv = _conv_fft(f)
    current_score = 2.0 * n * conv.max() / (f.sum() ** 2)
    best_score = current_score
    best_f = f.copy()
    
    stuck = 0
    improvement_thresh = 1e-8
    
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        conv = _conv_fft(f)
        peak = conv.max()
        current_score = 2.0 * n * peak / (f.sum() ** 2)
        
        if current_score < best_score:
            if current_score < best_score - improvement_thresh:
                best_score = current_score
                best_f = f.copy()
                stuck = 0
                if verbose and it % 100000 == 0:
                    print(f"    iter {it}: score={best_score:.8f}")
            else:
                stuck += 1
        else:
            stuck += 1
        
        # Find peak lag
        k_star = int(np.argmax(conv))
        
        # Compute gradient
        if it % 3 == 0:
            # Use multi-peak gradient every few iterations
            grad = _multi_peak_gradient(conv, f, n, top_k=top_k, temperature=peak * 0.1)
        else:
            grad = _gradient_at_peak(conv, f, k_star, n)
        
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm
        
        # Backtracking step
        f, new_peak = _backtrack_step(f, grad, peak, n)
        
        # Periodic perturbation to escape local minima
        if stuck > 30:
            magnitude = 0.02 / np.sqrt(n)
            f = f + magnitude * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f = _normalize(f)
            stuck = 0
    
    # Final evaluation
    conv = _conv_fft(best_f)
    final_score = 2.0 * n * conv.max() / (best_f.sum() ** 2)
    if final_score < best_score:
        best_score = final_score
    
    return best_f, best_score


def _polish(f, deadline, rng, n_passes=5000):
    """Coordinate-level polish: find peak lag and do targeted adjustments."""
    f = f.copy()
    n = len(f)
    conv = _conv_fft(f)
    best_score = 2.0 * n * conv.max() / (f.sum() ** 2)
    best_f = f.copy()
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        conv = _conv_fft(f)
        peak_val = conv.max()
        k_star = int(np.argmax(conv))
        
        # Gradient at peak
        grad = _gradient_at_peak(conv, f, k_star, n)
        
        # Find the index j where grad is largest (most contributing to peak)
        j_donor = int(np.argmax(grad * f))
        
        # Find a good receiver: low gradient contribution
        contrib = grad * f
        # Try to move mass from high-contribution to low-contribution
        sorted_idx = np.argsort(contrib)
        
        moved = False
        for _ in range(5):
            j_recv = int(sorted_idx[rng.integers(0, len(sorted_idx))])
            if j_recv == j_donor:
                continue
            
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = f[j_donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_donor] -= amt
                f_try[j_recv] += amt
                f_try = np.maximum(f_try, 0.0)
                f_try = _normalize(f_try)
                
                conv_try = _conv_fft(f_try)
                if conv_try.max() < peak_val * (1 - 1e-10):
                    f = f_try
                    peak_val = conv_try.max()
                    moved = True
                    break
        
        if not moved:
            # Random perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = f[i] * 0.05
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0.0)
                f_try = _normalize(f_try)
                conv_try = _conv_fft(f_try)
                if conv_try.max() < peak_val:
                    f = f_try
        
        # Track best
        conv = _conv_fft(f)
        sc = 2.0 * n * conv.max() / (f.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(f,f))/(sum f)^2.
    
    Strategy:
    1. Explore multiple (n, initialization) pairs quickly.
    2. Focus optimization on the best candidates using exact peak gradient
       descent with backtracking line search.
    3. Polish with coordinate-level adjustments.
    """

    best_score = np.inf
    best_f = None
    
    # Phase 1: Quick exploration of different n and initializations (300s)
    print("Phase 1: Exploration...")
    explore_deadline = time.monotonic() + 300.0
    
    n_candidates = [3000, 5000, 6121, 8000, 10000, 12000, 15000, 20000]
    if n_given:
        n_candidates.insert(0, n_given)
    
    # Remove duplicates while preserving order
    seen = set()
    n_unique = []
    for n in n_candidates:
        if n not in seen and n >= 100:
            seen.add(n)
            n_unique.append(n)
    
    # Initialize functions
    inits = [
        ("uniform", _init_uniform),
        ("cosine", _init_cosine),
        ("anti_pal", _init_anti_palindrome),
        ("bump", lambda n: _init_bump(n, 0.35, 0.12)),
        ("two_bump", _init_two_bumps),
        ("sine3", lambda n: _init_sine_mod(n, 3)),
        ("sine7", lambda n: _init_sine_mod(n, 7)),
        ("random", _init_random),
    ]
    
    candidates = []  # (score, n, f)
    
    # Quick evaluation of initializations for each n
    for n in n_unique:
        if time.monotonic() > explore_deadline:
            break
        for name, init_fn in inits:
            if time.monotonic() > explore_deadline:
                break
            try:
                f0 = init_fn(n) if init_fn is not _init_random else _init_random(rng, n)
                f0 = _normalize(f0)
                sc = _score(f0)
                candidates.append((sc, n, f0, name))
            except Exception:
                continue
    
    if f_given is not None:
        f0 = _normalize(f_given)
        sc = _score(f0)
        candidates.append((sc, n_given, f0, "given"))
    
    candidates.sort(key=lambda x: x[0])
    
    print("Top 10 initial candidates:")
    for sc, n, _, name in candidates[:10]:
        print(f"  n={n}, {name}: {sc:.6f}")
    
    # Phase 2: Optimize top candidates (most of the time)
    print("\nPhase 2: Optimization...")
    opt_deadline = time.monotonic() + (budget_s - 180).0
    
    # Give more time to better candidates
    top_candidates = candidates[:5]
    time_per_candidate = (opt_deadline - time.monotonic()) / len(top_candidates)
    
    for idx, (sc0, n, f0, name) in enumerate(top_candidates):
        if time.monotonic() > opt_deadline:
            break
        
        t_alloc = time_per_candidate * (1.0 - 0.1 * idx)
        cand_deadline = time.monotonic() + t_alloc
        
        print(f"\n  Optimizing candidate {idx} (n={n}, {name}, initial_score={sc0:.6f}, time={t_alloc:.0f}s)")
        
        f_opt, sc_opt = _optimize(f0.copy(), cand_deadline, rng, verbose=True, top_k=5)
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.8f}")
    
    # Phase 3: Polish the best solution
    print(f"\nPhase 3: Polishing (current best: {best_score:.8f})...")
    polish_deadline = min(time.monotonic() + 120.0, total_deadline)
    
    if best_f is not None:
        f_polished, sc_polished = _polish(best_f.copy(), polish_deadline, rng)
        if sc_polished < best_score:
            best_score = sc_polished
            best_f = f_polished.copy()
            print(f"  After polish: {best_score:.8f}")
    
    # Phase 4: Final optimization round from polished solution
    if best_f is not None and time.monotonic() < total_deadline - 10:
        print("\nPhase 4: Final optimization...")
        final_deadline = total_deadline
        f_final, sc_final = _optimize(best_f.copy(), final_deadline, rng, verbose=True, top_k=3)
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()
            print(f"  Final: {best_score:.8f}")
    
    # Ensure valid output
    if best_f is None:
        best_f = np.ones(6121) / 6121
    
    best_f = np.maximum(best_f, 0.0)

        best_score = final_eval
