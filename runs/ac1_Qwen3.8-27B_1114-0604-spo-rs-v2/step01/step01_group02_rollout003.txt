Now, for the mirror descent, the objective (with sum(a) = 1 normalization) is:

The subgradient w.r.t. a (before normalization) is:
g[j] = 2n * sum_weights w_i * 2*a[ki - j]

But we need to account for the normalization. If we parameterize with unconstrained variables and project, it's different from direct mirror descent on the simplex.

For direct mirror descent on the simplex (a >= 0, sum a = 1):
- The mirror map is Φ(a) = sum a[i] * log(a[i]) (negative entropy)
- The Bregman divergence is D(a, b) = sum a[i] log(a[i]/b[i]) - sum a[i] + sum b[i]
- The update is: a_new = argmin_{a in simplex} <g, a> + (1/η) * D(a, a_old)
- Solution: a_new[i] = a_old[i] * exp(-η * g[i]) / sum_j a_old[j] * exp(-η * g[j])

This is the Standard Exponential Family update, also known as the multiplicative weights update.

The step size η should be chosen carefully. A common choice is η = sqrt(2 * (log n) / (T * L²)) where L is the Lipschitz constant of the gradient and T is the number of iterations. But for non-convex problems, there's no clean theory, so we'll use a heuristic.

Let me also think about the "deflation" approach more carefully.

After running mirror descent for a while, we have a sequence with a few dominant peaks. We can try to reduce these peaks by:
1. Identifying the pairs (i, k-i) that contribute most to the peak at lag k
2. Reducing a[i] or a[k-i] for those pairs
3. Redistributing the energy to positions that have low "impact" on all peaks

The "impact" of position j on peak k is proportional to a[k-j]. So we want to add energy to positions j where a[k-j] is small for all k near the peak.

This is essentially a local search / coordinate descent.

```python
"""
Subgradient + peak deflation optimizer for minimizing 2*n*max(conv(a,a))/(sum a)^2.

Uses:
- Normalized averaged subgradient with simplex projection (fast, robust)
- Adaptive top-k smooth-max subgradient with cooling
- Peak deflation: targeted reduction of worst autocorrelation peaks
- Multi-length sweep and restarts
- Mirror descent (multiplicative weights) as an alternative local optimizer
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    """Objective value given n, convolution c and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgrad_topk(a, c, k, temperature=None):
    """Compute subgradient of weighted-peak surrogate of max(c).
    Returns (subgrad, weighted_max)."""
    n = len(a)
    k_act = min(k, len(c))
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        wts = np.array([1.0])
    else:
        top_idx = np.argpartition(c, -k_act)[-k_act:]
        vals = c[top_idx]
        top_val = float(vals.max())
        if temperature is None:
            temperature = 0.05 * top_val if top_val > 1e-12 else 1.0
        wts = np.exp((vals - top_val) / max(temperature, 1e-30))
        wts = wts / wts.sum()
        idx = top_idx

    subgrad = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        # Gradient of c[ki] w.r.t. a[j] = 2*a[ki-j] for j in [j_start, j_end]
        subgrad[j_start:j_end + 1] += wi * 2.0 * a[ki - j_start:ki - j_end + 1][::-1]

    M_weighted = float(np.dot(c[idx], wts))
    return subgrad, M_weighted


def peak_deflation(a, c, n_peaks=3, strength=0.3):
    """Reduce the strongest peaks by adjusting a locally.
    For each peak at lag k, reduce a[j] where a[k-j] is large, and
    redistribute to positions with low cross-contribution."""
    n = len(a)
    a = a.copy()
    # Find top n_peaks lags
    top_lags = np.argpartition(c, -n_peaks)[-n_peaks:]
    top_lags = top_lags[np.argsort(c[top_lags])[::-1]]

    for ki in top_lags:
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        partners = a[ki - js][::-1]  # a[ki-j] for j in js
        # Reduce a[j] proportional to partner value (where both are large)
        reduction = strength * partners * a[js]
        # Find where to redistribute: positions with low max partner across all top lags
        # Simple approach: add evenly to positions outside the affected range
        total_reduction = reduction.sum()
        if total_reduction < 1e-15:
            continue
        a[js] -= reduction
        a[js] = np.maximum(a[js], 0)
        # Redistribute to other positions
        mask = np.zeros(n, dtype=bool)
        mask[js] = False
        if mask.sum() > 0:
            a[mask] += total_reduction / mask.sum()
        else:
            a[js] += total_reduction

    # Re-normalize
    S = a.sum()
    if S > 0:
        a = a / S
    return a


def mirror_descent_optimize(a, n, max_iters, lr_start, lr_end, topk, cooling):
    """Run mirror descent (multiplicative weights) starting from a (sum=1)."""
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = np.inf
    best_a = a.copy()
    temp_scale = 0.1  # Initial temperature scale

    for it in range(max_iters):
        lr = lr_start * (lr_end / lr_start) ** (it / max(1, max_iters - 1))
        temp = temp_scale * (1.0 + it / max(1, max_iters))
        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        subgrad, M_w = subgrad_topk(a, c, topk, temperature=0.05 * M_w if M_w > 0 else 1.0)
        # Scale gradient: we're optimizing 2n * max_c with sum(a)=1
        g = 2.0 * n * subgrad

        # Multiplicative update
        log_ratio = -lr * g
        log_ratio -= log_ratio.max()  # stability
        a = a * np.exp(log_ratio)
        S = a.sum()
        if S > 0:
            a = a / S
        else:
            a = np.ones(n) / n

    return best_score, best_a


def subgrad_optimize(a, n, budget, lr_init, topk_init, temp_scale_init, patience=500):
    """Normalized averaged subgradient method with simplex projection.
    Runs until budget (time) expires, returns best (score, a)."""
    deadline = time.monotonic() + budget
    a = a.astype(np.float64).copy()
    a = a / a.sum()

    best_score = np.inf
    best_a = a.copy()

    g_sum = np.zeros(n)
    temp = temp_scale_init
    k = topk_init
    no_improve = 0
    step_base = lr_init
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            # Restart from best with perturbation
            a = best_a * (1.0 + 0.1 * np.random.standard_normal(n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            g_sum[:] = 0
            no_improve = 0
            patience = max(100, patience // 2)
            step_base *= 0.7
            temp *= 0.5
            continue

        # Cool temperature and topk
        temp *= 0.999
        if it % 500 == 0 and k > 2:
            k = max(2, k - 1)

        subgrad, M_w = subgrad_topk(a, c, k, temperature=0.05 * M_w if M_w > 0 else 1.0)
        g_sum += 2.0 * n * subgrad

        # Polyak step: eta_t = f(a_1) - f* / ||g_avg||^2 (approximate)
        # Use fixed diminishing step
        eta = step_base / (1.0 + 0.01 * it)
        v = a - eta * g_sum / (it ** 0.5)
        a = project_simplex(v)

    return best_score, best_a


def coordinate_descent(a, n, max_rounds=100):
    """Local coordinate descent: for each coordinate, find optimal value
    given others fixed (to reduce the max convolution)."""
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = np.inf
    best_a = a.copy()

    c = fast_self_conv(a)
    max_c = float(c.max())

    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Try reducing a[j] and see if max_c decreases
            # c[k] involves a[j] in terms: a[j]*a[k-j] for valid k-j
            # Reducing a[j] reduces c[k] for all k where k-j is valid
            # We want to find the largest reduction in a[j] such that no c[k] 
            # that is currently below max_c becomes greater than max_c
            # Simplified: just try a few candidate values
            for frac in [0.8, 0.5, 0.2, 0.0]:
                a[j] = old_val * frac
                c_new = fast_self_conv(a)
                max_new = float(c_new.max())
                if max_new <= max_c:
                    max_c = max_new
                    improved = True
                    break
                a[j] = old_val
            # Also try increasing if a[j] is at 0
            if old_val < 1e-10:
                for frac in [0.1, 0.5, 1.0, 2.0]:
                    candidate = a.copy()
                    candidate[j] = frac / n
                    c_new = fast_self_conv(candidate)
                    if float(c_new.max()) <= max_c:
                        a = candidate
                        max_c = float(c_new.max())
                        improved = True
                        break

        S = a.sum()
        a = a / S
        score = score_from_conv(n, fast_self_conv(a), a.sum())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if not improved:
            break

    return best_score, best_a


def build_diverse_inits(n, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n))
    # Gaussian
    x = np.arange(n) / (n - 1)
    inits.append(np.exp(-2 * (x - 0.5) ** 2).copy())
    inits.append(np.exp(-8 * (x - 0.5) ** 2).copy())
    # Linear ramps
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))
    # Cosine modulated
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * f * x), 0.01))
    # Random with structure
    for _ in range(5):
        a = 0.5 + 0.5 * rng.random(n)
        # Apply a low-pass filter to create smooth structure
        kernel = np.ones(5) / 5
        a_f = np.convolve(a, kernel, mode='same')
        inits.append(np.maximum(a_f, 0.01))
    # Binary-like
    for _ in range(3):
        a = (rng.random(n) > 0.3).astype(float) * 0.8 + 0.2
        inits.append(a)
    # Exponential decay from center
    mid = (n - 1) / 2.0
    inits.append(np.exp(-np.abs(x - 0.5) * 4).copy())
    # Two-bump
    inits.append(np.exp(-5 * (x - 0.3) ** 2) + np.exp(-5 * (x - 0.7) ** 2))
    # Normalize all to sum = 1
    return [a / a.sum() for a in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    deadline = start_time + total_budget - 5.0

    best_score = np.inf
    best_a = None

    def update_best(s, a):
        nonlocal best_score, best_a
        if s < best_score:
            best_score = s
            best_a = a.copy()
            print("  [NEW BEST] score = %.6f at t = %.1fs" % (s, time.monotonic() - start_time))

    # Try to load parent
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 10 and np.all(np.isfinite(arr)) and np.all(arr >= 0):
            parent = np.clip(arr, 0, 1000)
            parent = parent / parent.sum()
            print("Loaded parent with n=%d" % parent.size)
    except Exception:
        pass

    # Determine which n values to try
    n_candidates = [1024, 2048, 4096]
    if parent is not None:
        n_p = parent.size
        if n_p not in n_candidates:
            n_candidates.insert(0, n_p)

    print("Trying n values: %s" % n_candidates)
    print("Budget: %d seconds" % total_budget)

    # Phase 1: Quick sweep over n values with diverse inits (30% of budget)
    phase1_time = 0.30 * (total_budget - 5.0)
    phase1_end = start_time + phase1_time

    for n in n_candidates:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break

        inits = build_diverse_inits(n, rng)
        # Add parent if it matches this n
        if parent is not None and parent.size == n:
            inits.insert(0, parent.copy())

        # Allocate time per init
        time_per_init = max(10.0, remaining / (len(inits) + 2))

        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            t_alloc = min(time_per_init, phase1_end - time.monotonic())
            if t_alloc < 5:
                break

            # Use subgradient method
            sc, aa = subgrad_optimize(a0, n, t_alloc, lr_init=0.02,
                                      topk_init=8, temp_scale_init=0.1, patience=300)
            update_best(sc, aa)

    print("Phase 1 done. Best = %.6f" % best_score)

    # Phase 2: Focus on best n, longer optimization (40% of budget)
    phase2_time = 0.40 * (total_budget - 5.0)
    phase2_end = start_time + phase1_time + phase2_time

    if best_a is not None:
        n_best = best_a.size
        # Run mirror descent from best
        t_alloc = min(phase2_end - time.monotonic(), 0.5 * phase2_time)
        if t_alloc > 20:
            max_iters = int(t_alloc * 80)  # ~80 iterations/sec estimate
            sc, aa = mirror_descent_optimize(best_a, n_best, max_iters,
                                             lr_start=0.1, lr_end=0.001,
                                             topk=8, cooling=0.999)
            update_best(sc, aa)
            best_a_for_next = aa if sc < best_score else best_a
        else:
            best_a_for_next = best_a.copy()

        # Run subgradient from best with more time
        t_alloc = phase2_end - time.monotonic()
        if t_alloc > 20:
            sc, aa = subgrad_optimize(best_a_for_next, n_best, t_alloc,
                                      lr_init=0.005, topk_init=4,
                                      temp_scale_init=0.05, patience=400)
            update_best(sc, aa)

    # Phase 3: Peak deflation + coordinate descent (20% of budget)
    phase3_time = 0.20 * (total_budget - 5.0)
    phase3_end = start_time + phase1_time + phase2_time + phase3_time

    if best_a is not None:
        n_best = best_a.size
        while time.monotonic() < phase3_end:
            a_cur = best_a.copy()
            # Apply peak deflation
            c = fast_self_conv(a_cur)
            a_deflated = peak_deflation(a_cur, c, n_peaks=5, strength=0.1)
            a_deflated = a_deflated / a_deflated.sum()
            sc = score_from_conv(n_best, fast_self_conv(a_deflated), a_deflated.sum())
            update_best(sc, a_deflated)

            # Coordinate descent on the best
            max_rounds = min(50, int((phase3_end - time.monotonic()) / 5))
            if max_rounds > 5:
                sc_cd, aa_cd = coordinate_descent(best_a, n_best, max_rounds=max_rounds)
                update_best(sc_cd, aa_cd)

            if time.monotonic() > phase3_end:
                break

    # Phase 4: Final polish with very careful local search
    if best_a is not None and time.monotonic() < deadline:
        n_best = best_a.size
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Try a few random perturbations + rapid subgradient
            for attempt in range(5):
                if time.monotonic() > deadline:
                    break
                pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                t_alloc = min(30.0, deadline - time.monotonic())
                if t_alloc < 5:
                    break
                sc, aa = subgrad_optimize(pert, n_best, t_alloc,
                                          lr_init=0.002, topk_init=3,
                                          temp_scale_init=0.02, patience=200)
                update_best(sc, aa)

    # Final verification
    if best_a is None:
        # Fallback: uniform
        n = 2048
        best_a = np.ones(n)
        best_a = best_a / best_a.sum()

    # Evaluate with the official function
    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print("Final: internal best = %.6f, verified = %.6f, n = %d, time = %.1fs"
          % (best_score, verified, len(out), time.monotonic() - start_time))

    return out
```
