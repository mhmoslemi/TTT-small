def _local_optimize(arr, deadline):
    """Simple multiplicative gradient descent on the simplex to minimize max autocorrelation."""
    arr = np.asarray(arr, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return arr
    arr /= s
    
    best = arr.copy()
    best_score = _safe_score(best)
    n = arr.size
    lr = 0.5
    K = min(16, 2 * n - 1)
    
    for it in range(500):
        if time.monotonic() > deadline:
            break
        
        # Compute autocorrelation via FFT
        size = 2 * n - 1
        N = 1
        while N < size:
            N <<= 1
        fa = np.fft.rfft(arr, N)
        c = np.fft.irfft(fa * fa, N)[:size]
        
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        # Find top K active lags
        if size <= K:
            active = np.arange(size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient: grad[i] = 2 * sum_{k in active} a[k-i] (with bounds)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * arr[k - idx]
        
        grad /= float(len(active))
        
        # Project to tangent plane (remove mean for simplex constraint)
        g = grad - grad.mean()
        if np.max(np.abs(g)) < 1e-12:
            break
        
        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -1.0, 1.0)
        new_arr = arr * np.exp(exponent)
        new_arr = np.clip(new_arr, 0.0, 1000.0)
        ns = new_arr.sum()
        if ns < 0.01:
            lr *= 0.5
            continue
        new_arr /= ns
        
        new_score = _safe_score(new_arr)
        if new_score < best_score:
            best = new_arr.copy()
            best_score = new_score
            arr = new_arr
            lr = min(5.0, lr * 1.1)
        else:
            lr = max(0.05, lr * 0.8)
            if lr < 0.06:
                break
    
    return best

    This function verifies empirically and explores local alternatives.
    """

    hard_deadline = t0 + budget_s - 10.0
    
    # Try to access the provided best sequence
    try:
        h1 = np.array(height_sequence_1, dtype=np.float64)
    except (NameError, TypeError):
        h1 = None

    candidates = []
    
    # Constant sequences (theoretical optimum)
    for n in [256, 512, 1024, 2048, 4096, 8192]:
        candidates.append((_make_constant(n), f'const_n{n}'))
    
    # Provided sequence
    if h1 is not None and h1.size > 0:
        candidates.append((h1, 'height_sequence_1'))
    
    # Some structured alternatives
    for n in [1024, 2048, 4096]:
        for amp in [0.5, 0.78, 0.9]:
            candidates.append((_make_sine(n, amp, 1), f'sine_n{n}_a{amp}'))
    
    # Evaluate candidates
    best_score = np.inf
    best_seq = None
    
    for seq, name in candidates:
        if time.monotonic() > hard_deadline:
            break
        score = _safe_score(seq)
        print(f"{name}: n={seq.size}, score={score:.10f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    
    # Local optimization from the best candidate if we have time
    remaining = hard_deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"[optimize] Starting local optimization with {remaining:.1f}s remaining")
        opt_end = min(t0 + budget_s - 15.0, time.monotonic() + min(remaining * 0.6, 600.0))
        optimized = _local_optimize(best_seq, opt_end)
        opt_score = _safe_score(optimized)
        print(f"[optimize] Result: n={optimized.size}, score={opt_score:.10f}")
        if opt_score < best_score:
            best_score = opt_score
            best_seq = optimized
    
    # Also try optimizing from the provided sequence if different
    remaining = hard_deadline - time.monotonic()
    if remaining > 30 and h1 is not None and (best_seq is None or not np.allclose(h1, best_seq)):
        opt_end = min(t0 + budget_s - 15.0, time.monotonic() + min(remaining * 0.5, 300.0))
        h1_work = h1.copy()
        h1_work = np.clip(h1_work, 0, 1000)
        s = h1_work.sum()
        if s > 0.01:
            h1_work /= s
            optimized2 = _local_optimize(h1_work, opt_end)
            opt_score2 = _safe_score(optimized2)
            print(f"[optimize_h1] Result: n={optimized2.size}, score={opt_score2:.10f}")
            if opt_score2 < best_score:
                best_score = opt_score2
                best_seq = optimized2
    
    # Final verification
    if best_seq is not None:
        final_score = _safe_score(best_seq)
        print(f"\n[FINAL] n={best_seq.size}, score={final_score:.10f}")
        print(f"[FINAL] total_time={time.monotonic() - t0:.2f}s")
    
    if best_seq is None:
        best_seq = _make_constant(2048)
    
    return [float(x) for x in best_seq]
