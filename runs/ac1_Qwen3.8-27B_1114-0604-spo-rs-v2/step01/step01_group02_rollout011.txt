import numpy as np
import time

def fast_self_conv(a):
    n = a.size
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(v.shape, 1.0 / v.size)
    return v / s

def compute_score_simplex(n, conv):
    """Score when sum(a)=1: f = 2n * max(conv)."""
    return 2.0 * n * float(np.max(conv))

def grad_peaks(a, conv, topk, temperature):
    """Compute weighted gradient of top-k peaks. Sum(a)=1 assumed."""
    n = a.size
    M = conv.size
    k_eff = min(topk, M)
    if k_eff == 1:
        idx = np.array([np.argmax(conv)])
        wts = np.array([1.0])
    else:
        threshold = np.partition(conv, M - k_eff)[M - k_eff]
        idx_all = np.where(conv >= threshold)[0]
        # If too many tied, take first k_eff
        if idx_all.size > k_eff:
            # break ties by taking the largest
            vals = conv[idx_all]
            order = np.argsort(vals)[::-1][:k_eff]
            idx_all = idx_all[order]
        idx = idx_all
        vals = conv[idx]
        top = vals[0]
        if top > 1e-12:
            wts = np.exp((vals - top) / (temperature * top))
            wts = wts / wts.sum()
        else:
            wts = np.ones(idx.size) / idx.size
    
    grad = np.zeros(n)
    for ii in range(idx.size):
        k = int(idx[ii])
        w = float(wts[ii])
        if w < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end+1] += w * 2.0 * a[j_start:j_end+1][::-1]
    return grad, float(np.dot(conv[idx], wts))

def adam_step(a, g, m, v, t, lr):
    b1, b2 = 0.9, 0.999
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mc = m / (1.0 - b1**t)
    vc = v / (1.0 - b2**t)
    a -= lr * mc / (np.sqrt(vc) + 1e-8)

    v = np.zeros(n)
    t = 0
    lr = init_lr
    
    conv = fast_self_conv(a)
    score = compute_score_simplex(n, conv)

    temp = 0.1  # initial temperature for soft-max

        score = compute_score_simplex(n, conv)
        
        if score < best_score - 1e-9:

    """Gradient of 2n * max(conv) / S^2 using soft-max approximation."""
    n = a.size
    S = float(a.sum())
    M = conv.size
    k = min(topk, M)
    
    # Get top-k peaks
    if k == 1:
        idx = np.array([np.argmax(conv)])
        wts = np.array([1.0])
    else:

        vals = conv[threshold_idx]

def lp_step(f, M, solver='highs'):
    """Solve the LP: max sum(b) s.t. f*b <= M, b >= 0."""
    n = f.size

    c = -np.ones(n)

import numpy as np

from scipy.optimize import linprog

def fast_self_conv(a):
    n = a.size
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def obj(a):
    """Objective: 2n * max(conv) / S^2"""
    n = a.size
    conv = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)

def build_conv_matrix(n, f):
    """Build the matrix A where (A @ g)[k] = (f*g)[k]."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            A[k, js:je+1] = f[k-je:k-js+1][::-1]
    return A

def solve_lp(f, M):
    """Solve: max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    n = f.size
    # Build constraint matrix
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n))
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            # A[k, j] = f[k-j] for j in [js, je]
            # f[k-j] for j from js to je: f[k-je], f[k-je+1], ..., f[k-js]
            A[k, js:je+1] = f[k-je:k-js+1][::-1]
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})

    return None
