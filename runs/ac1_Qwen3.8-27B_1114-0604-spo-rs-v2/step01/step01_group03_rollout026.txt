def softmax_obj_and_grad(x, beta, L1, L2, n, m):
    """
    x: parameter vector in R^n
    a = softmax(x)
    Returns: objective value, gradient w.r.t. x
    """
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    
    # Compute conv(a,a)
    fa = np.fft.rfft(a, L1)
    c = np.fft.irfft(fa * fa, L1)[:m]
    c = np.maximum(c, 0)
    
    # Smooth max
    bc = beta * c
    mc = bc.max()
    e = np.exp(bc - mc)
    se = e.sum()
    val = (mc + np.log(se)) / beta
    
    # Weights
    w = e / se
    
    # Gradient w.r.t. a
    a_rev = a[::-1]
    W = np.fft.rfft(w, L2)
    AR = np.fft.rfft(a_rev, L2)
    conv_wr = np.fft.irfft(W * AR, L2)
    ga = 2.0 * conv_wr[n-1:n-1+n]
    
    # Gradient w.r.t. x: grad_x[i] = a[i] * (ga[i] - dot(a, ga))
    dot_ag = a @ ga
    gx = a * (ga - dot_ag)
    
    return val, gx

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using progressive smooth-max
optimization with L-BFGS in softmax parameterization, multiple restarts,
and active-set refinement.
"""
import time
import numpy as np
from scipy.optimize import minimize as scipy_minimize

def _conv_self_fft(a, L, m):
    """Compute conv(a,a) using FFT, returning first m elements."""
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:m]
    c = np.maximum(c, 0.0)
    return c

def _smooth_max_val(c, beta):
    """Compute logsumexp(beta*c)/beta."""
    bc = beta * c
    mc = float(bc.max())
    e = np.exp(bc - mc)
    se = float(e.sum())
    return float(mc + np.log(se)) / beta

def _smooth_obj_and_grad(x, beta, L1, L2, n, m):
    """
    Objective and gradient of smooth max of conv(softmax(x), softmax(x)).
    """
    xs = x - x.max()
    ex = np.exp(xs)
    a = ex / ex.sum()
    
    c = _conv_self_fft(a, L1, m)
    
    bc = beta * c
    mc = float(bc.max())
    e = np.exp(bc - mc)
    se = float(e.sum())
    val = float(mc + np.log(se)) / beta
    
    w = e / se
    
    # Gradient w.r.t. a
    a_rev = a[::-1]
    W = np.fft.rfft(w, L2)
    AR = np.fft.rfft(a_rev, L2)
    conv_wr = np.fft.irfft(W * AR, L2)
    ga = 2.0 * np.ascontiguousarray(conv_wr[n-1:n-1+n])
    
    # Gradient w.r.t. x
    dot_ag = float(a @ ga)
    gx = a * (ga - dot_ag)
    
    return val, gx

def _exact_score(a):
    """Compute exact objective 2*n*max(conv(a,a))/(sum(a))^2."""
    m = 2 * len(a) - 1
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return float(2.0 * len(a) * c.max() / (s * s))

def _lbgfs_smooth(a_init, beta, L1, L2, n, m, max_iter=500, time_limit=60.0):
    """Run L-BFGS on smooth max objective."""
    # Convert a to x (log of a, with offset)
    a = np.maximum(a_init, 1e-300)
    x0 = np.log(a / a.sum())  # softmax(x0) = a
    
    start = time.monotonic()
    
    res = scipy_minimize(
        _smooth_obj_and_grad,
        x0,
        args=(beta, L1, L2, n, m),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': 2000}
    )

    xs = res.x - res.x.max()
    ex = np.exp(xs)
    a_out = ex / ex.sum()
    a_out = np.maximum(a_out, 0)
    a_out /= a_out.sum()
    
    elapsed = time.monotonic() - start
    return a_out, elapsed

def _active_set_refine(a, L1, m, n, top_k=10, iters=100):
    """
    Refine by trying to equalize the top-k lags of conv(a,a).
    Use gradient descent on the variance of top-k values.
    """
    a = a.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    
    for it in range(iters):
        c = _conv_self_fft(a, L1, m)
        
        # Find top-k indices
        top_idx = np.argsort(c)[-top_k:]
        top_vals = c[top_idx]
        target = top_vals.mean()

        # = sum_{k in top} 2*(c[k]-target) * grad(c[k])
        # grad(c[k])[i] = 2*a[k-i] (for valid k-i)
        
        g = np.zeros(n)
        for k in top_idx:
            coeff = 2.0 * (c[k] - target)
            # grad(c[k])[i] = 2*a[k-i] for 0<=k-i<n
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            for i in range(lo, hi+1):
                g[i] += coeff * 2.0 * a[k-i]
        
        # Normalize and step
        g_norm = np.sqrt(np.sum(g**2))
        if g_norm < 1e-15:
            break
        g /= g_norm
        
        # Line search
        best_score = _exact_score(a)
        for step in [0.01, 0.005, 0.001, 0.0005]:
            a_try = a - step * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            sc = _exact_score(a_try)
            if sc < best_score:
                best_score = sc
                a = a_try
                break
    
    return a

    ex = np.exp(xs)
    a = ex / ex.sum()

    c = np.fft.irfft(fa * fa, L1)[:m]

    ga = 2.0 * np.ascontiguousarray(conv_wr[n-1:n-1+n])

def _fft_conv(a, L, m):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa*fa, L)[:m]
    np.maximum(c, 0, out=c)
    return c

    """g[i] = 2*sum_j a[j]*w[i+j] = 2*conv(w, a_rev)[n-1+i]"""
    ar = a[::-1]
    W = np.fft.rfft(c_w, L2)
    AR = np.fft.rfft(ar, L2)
    cv = np.fft.irfft(W*AR, L2)
    return 2.0 * np.ascontiguousarray(cv[n-1:n-1+n])

# --- Objective ---
def _obj(a):
    n = a.size
    s = a.sum()
    if s < 1e-15: return 1e30
    c = np.convolve(a, a)
    return 2.0*n*c.max()/(s*s)

    ex = np.exp(xs)
    a = ex / ex.sum()
    c = _fft_conv(a, L1, m)
    bc = beta * c
    mc = bc.max()
    e = np.exp(bc - mc)
    se = e.sum()

    val = (mc + math.log(se)) / beta
    w = e / se
    ga = _fft_grad(w, a, L2, n)
    dt = a @ ga
    gx = a * (ga - dt)
    return val, gx

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using progressive log-sum-exp
smoothing with L-BFGS, active-set equalization, and coordinate polishing.
Multiple restarts with adaptive budget allocation.
"""
import time, math
import numpy as np
from scipy.optimize import minimize as spm

def _fft_conv(a, L, m):
    """Full linear convolution a*a via FFT, first m elements."""
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_grad_from_w(w, a, L2, n):
    """Compute g[i] = 2*sum_j a[j]*w[i+j] via FFT."""
    ar = a[::-1]
    W = np.fft.rfft(w, L2)
    AR = np.fft.rfft(ar, L2)
    cv = np.fft.irfft(W * AR, L2)
    return 2.0 * np.ascontiguousarray(cv[n-1:n-1+n])

def _fast_obj(a, L1, m):
    """Fast objective using FFT."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return 1e30
    c = _fft_conv(a, L1, m)
    return 2.0 * float(c.max()) * n / (s * s)

def _exact_obj(a):
    """Exact objective using direct convolution."""
    n = a.size
    af = a.astype(np.float64)
    s = float(af.sum())
    if s < 1e-15:
        return 1e30
    c = np.convolve(af, af)
    return float(2.0 * n * c.max()) / (s * s)
