Optimizes a non-negative sequence to minimize 2n*max(conv(a,a))/(sum a)^2.
Uses mirror descent (multiplicative updates) with multi-peak exact subgradient,
combined with random coordinate moves for escape from local minima.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = a.size
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def multipeak_gradient(a, conv, num_peaks):
    """Compute gradient using exact subgradients from top-k peaks."""
    n = a.size
    g = np.zeros(n)
    if num_peaks >= 2 * n - 1:
        peaks = np.arange(2 * n - 1)
    else:
        peaks = np.argpartition(conv, -num_peaks)[-num_peaks:]
    for ki in peaks:
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            g[j_start:j_end + 1] += 2.0 * a[ki - j_end:ki - j_start + 1][::-1]
    return g


def mul_update(b, g, eta, S):
    """Mirror descent: b_new = b * exp(-eta*g) / Z * (S / sum)."""
    log_b = np.log(np.maximum(b, 1e-300))
    exponent = log_b - eta * g
    exponent -= exponent.max()
    new_b = np.exp(exponent)
    total = new_b.sum()
    if total < 1e-300:
        return np.full(b.size, S / b.size)
    return new_b * S / total


def update_conv_move(conv, a, j, jp, delta, n):
    """In-place update of conv after a[j]-=delta, a[jp]+=delta."""
    # a[j] decreases by delta
    ks = max(0, j)
    ke = min(2 * n - 2, j + n - 1)
    ms = ks - j
    me = ke - j + 1
    conv[ks:ke + 1] -= 2.0 * delta * a[ms:me]
    # a[jp] increases by delta
    ks = max(0, jp)
    ke = min(2 * n - 2, jp + n - 1)
    ms = ks - jp
    me = ke - jp + 1
    conv[ks:ke + 1] += 2.0 * delta * a[ms:me]


def normalize_to_sum(a, S):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(a.size, S / a.size)
    else:
        a = a * S / s
    return a


def gradient_phase(a0, deadline, rng, eta0, topk, patience):
    """Run mirror descent with multi-peak gradient until deadline."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, S)
    conv = fast_self_conv(a)
    score = float(conv.max())
    best_score = score
    best_a = a.copy()
    t = 0
    stale = 0
    
    while time.monotonic() < deadline:
        g = multipeak_gradient(a, conv, topk)
        t += 1
        eta = eta0 / (1.0 + t * 1e-5)
        a = mul_update(a, g, eta, S)
        conv = fast_self_conv(a)
        score = float(conv.max())
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1
        
        if stale >= patience:
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = normalize_to_sum(a, S)
            conv = fast_self_conv(a)
            score = float(conv.max())
            t = 0
            stale = 0
            patience = max(200, patience // 3)
    
    return best_score, best_a


def random_walk_phase(a0, conv0, deadline, rng, max_iters):
    """Random coordinate moves to escape local minima."""
    n = a0.size
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    conv = conv0.copy()
    score = float(conv.max())
    best_score = score
    best_a = a.copy()
    
    # Temperature for simulated annealing
    T0 = score * 0.01
    T_min = score * 0.0001
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        T = T0 * (T_min / T0) ** (it / max(1, max_iters))
        
        # Pick random pair
        j = rng.integers(n)
        jp = rng.integers(n)
        if j == jp:
            continue
        
        # Small transfer
        frac = 0.005 + 0.01 * rng.random()
        delta = a[j] * frac
        if delta < 1e-12:
            continue
        if a[jp] + delta > 1000:
            continue
        
        # Apply move
        a[j] -= delta
        a[jp] += delta
        update_conv_move(conv, a, j, jp, delta, n)
        new_score = float(conv.max())
        
        # Accept or reject
        if new_score < score:
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            # Boltzmann acceptance
            d = new_score - score
            if d < T:
                # Accept with probability exp(-d/T)
                if np.log(rng.random()) > -d / max(T, 1e-15):
                    score = new_score
                else:
                    # Reject
                    a[j] += delta
                    a[jp] -= delta
                    update_conv_move(conv, a, j, jp, -delta, n)
            else:
                # Reject
                a[j] += delta
                a[jp] -= delta
                update_conv_move(conv, a, j, jp, -delta, n)
        
        # Normalize periodically to prevent drift
        if it % 10000 == 0 and it > 0:
            a = normalize_to_sum(a, S)
            conv = fast_self_conv(a)
            score = float(conv.max())
    
    return best_score, best_a


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        inits.append(("parent", normalize_to_sum(parent, np.sqrt(2.0 * n))))
    
    inits.append(("uniform", np.full(n, 1.0)))
    
    # Random smooth (low-pass filtered)
    for _ in range(3):
        r = rng.random(n)
        # Simple low-pass: moving average
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        smooth = np.convolve(r, kernel, mode='same')
        inits.append(("smooth", smooth))
    
    # Asymmetric ramp
    x = np.linspace(0, 1, n)
    inits.append(("ramp_up", 0.5 + x))
    inits.append(("ramp_down", 1.5 - x))
    
    # Cosine shapes
    for freq in (1, 2, 3):
        inits.append(("cos%d" % freq, 1.0 + 0.5 * np.cos(2 * np.pi * freq * x)))
    
    # Random
    inits.append(("random", 0.5 + rng.random(n)))
    
    # Exponential (heavy at one end)
    inits.append(("exp", np.exp(-3.0 * x)))
    inits.append(("exp2", np.exp(3.0 * x)))
    
    return inits
