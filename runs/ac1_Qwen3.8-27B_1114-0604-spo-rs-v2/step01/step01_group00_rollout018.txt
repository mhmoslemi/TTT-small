```python
"""
Step function optimizer using mirror descent with exact multi-peak subgradient.
Minimizes 2*n*max(conv(f,f))/(sum(f))^2 for non-negative f.
Strategies: diverse initialization, temperature-annealed peak targeting,
adaptive step size, coordinate descent polish.
"""

import numpy as np
import time


def _autocorr(f):
    """Full linear autocorrelation (self-convolution) of f."""
    return np.convolve(f, f, mode='full')


def _compute_score(f):
    """Compute the objective value 2*n*max_c/(sum_f)^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    c = _autocorr(f)
    return 2.0 * n * c.max() / (s * s)


def _normalize(f):
    """Normalize f to sum=1, non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / len(f)
    return f / s


def _peak_subgrad(f, c, K=10, T=0.005):
    """
    Compute weighted subgradient of max(c) using top-K peaks.
    Returns gradient vector g of length n.
    """
    n = len(f)
    # Find top-K peaks
    # For efficiency, we can look at all 2n-1 positions
    # By symmetry c[k] = c[2n-2-k], so we only need k=0..n-1
    # But let's just use all for simplicity
    if 2 * n - 1 <= 1000:
        topk_idx = np.argsort(c)[::-1][:K]
    else:
        # For large n, use partial sort
        topk_idx = np.argpartition(c, -K)[-K:]
        topk_idx = topk_idx[np.argsort(c[topk_idx])[::-1]]
    
    # Softmax weights based on peak values
    peak_vals = c[topk_idx]
    # Temperature-scaled softmax
    scaled = (peak_vals - peak_vals.max()) / T
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for w, k in zip(weights, topk_idx):
        # grad c[k][i] = f[k-i] if 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        idx = k - np.arange(i_lo, i_hi + 1)
        g[i_lo:i_hi + 1] += w * f[idx]
    
    return g


def _mirror_step(f, g, eta):
    """Mirror descent step on the simplex."""
    log_f = np.log(f + 1e-300)
    log_f -= eta * g
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _optimize(f0, deadline, rng, label=""):
    """
    Optimize starting from f0 using mirror descent with subgradient.
    Returns (best_f, best_score).
    """
    n = len(f0)
    f = _normalize(f0.copy())
    
    # Initial evaluation
    c = _autocorr(f)
    curr_peak = c.max()
    best_peak = curr_peak
    best_f = f.copy()
    
    # Hyperparameters
    T0 = curr_peak * 0.1  # initial temperature for peak selection
    T_end = curr_peak * 1e-6
    eta0 = 0.5  # initial step size
    eta_end = 0.01
    max_iters = 500000
    
    it = 0
    no_improve = 0
    improve_thresh = 1e-8
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        
        # Annealing schedule
        progress = min(1.0, it / max_iters)
        T = T0 * (T_end / T0) ** (progress ** 0.5)
        eta = eta0 * (eta_end / eta0) ** (progress ** 0.3)
        
        # Compute subgradient
        g = _peak_subgrad(f, c, K=10, T=T)
        
        # Mirror descent step
        f_new = _mirror_step(f, g, eta)
        
        # Evaluate
        c_new = _autocorr(f_new)
        peak_new = c_new.max()
        
        if peak_new < curr_peak - improve_thresh:
            f = f_new
            c = c_new
            curr_peak = peak_new
            no_improve = 0
            
            if peak_new < best_peak:
                best_peak = peak_new
                best_f = f.copy()
                if it % 100000 == 0:
                    score = 2.0 * n * best_peak
                    print(f"  {label} iter {it}: peak={best_peak:.8f}, score={score:.6f}")
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 50:
                f_new = _mirror_step(f, g, eta * 0.1)
                c_new = _autocorr(f_new)
                if c_new.max() < curr_peak:
                    f = f_new
                    c = c_new
                    curr_peak = c_new.max()
                    no_improve = 0
                else:
                    # Random perturbation
                    f = f + 0.01 / n * rng.standard_normal(n)
                    f = _normalize(f)
                    c = _autocorr(f)
                    curr_peak = c.max()
                    no_improve = 0
        
        # Every 5000 iterations, try a "jump"
        if it % 5000 == 0 and rng.random() < 0.3:
            f_pert = f.copy()
            # Randomly add some mass from low-contrib to high-contrib positions
            n_pert = min(n, 100)
            idx = rng.choice(n, size=n_pert, replace=False)
            f_pert[idx] *= np.exp(0.1 * rng.standard_normal(n_pert))
            f_pert = _normalize(f_pert)
            c_pert = _autocorr(f_pert)
            if c_pert.max() < curr_peak:
                f = f_pert
                c = c_pert
                curr_peak = c_pert.max()
                if curr_peak < best_peak:
                    best_peak = curr_peak
                    best_f = f.copy()
    
    # Final evaluation
    final_score = 2.0 * n * best_peak / 1.0  # since sum = 1
    if time.monotonic() < deadline:
        # One more polish
        f = best_f.copy()
        c = _autocorr(f)
        curr_peak = c.max()
        for _ in range(1000):

            if c_new.max() < curr_peak:
                f = f_new
                c = c_new
                curr_peak = c_new.max()
                if curr_peak < best_peak:
                    best_peak = curr_peak
                    best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak


def _polish_cd(f0, deadline, rng):
    """
    Coordinate descent polish.
    Transfers mass from high-contribution to low-contribution positions
    relative to the current peak.
    """
    n = len(f0)
    f = _normalize(f0.copy())
    c = _autocorr(f)
    curr_peak = c.max()
    best_peak = curr_peak
    best_f = f.copy()
    
    # Multiple passes with decreasing step sizes
    step_sizes = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for step in step_sizes:
        for outer in range(50):
            if time.monotonic() > deadline:
                break
            
            c = _autocorr(f)
            curr_peak = c.max()
            if curr_peak < best_peak:
                best_peak = curr_peak
                best_f = f.copy()
            
            # Find the peak
            k_star = int(np.argmax(c))
            
            # Compute contributions to the peak
            # c[k_star] = sum_i f[i]*f[k_star-i]
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            idx = np.arange(i_lo, i_hi + 1)
            mirror = k_star - idx
            contrib = f[idx] * f[mirror]
            
            # Total is c[k_star]
            # Find the pair with highest contribution
            max_local = np.argmax(contrib)
            min_local = np.argmin(contrib)
            i_donate = idx[max_local]
            i_receive = idx[min_local]
            
            if i_donate == i_receive:
                continue
            
            # Try transferring mass
            amt = f[i_donate] * step
            if amt <= 0:
                continue
            
            f_try = f.copy()
            f_try[i_donate] -= amt
            f_try[i_receive] += amt
            if f_try[i_donate] < 0:
                continue
            
            c_try = _autocorr(f_try)
            if c_try.max() < curr_peak - 1e-12:
                f = f_try
                c = c_try
                curr_peak = c_try.max()
            
            # Also try random transfers
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                amt = f[i] * step * 0.5
                if amt <= 0:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0:
                    continue
                c_try = _autocorr(f_try)
                if c_try.max() < curr_peak - 1e-12:
                    f = f_try
                    c = c_try
                    curr_peak = c_try.max()
    
    return best_f, 2.0 * n * best_peak

    f_clump[:n//2] = 1.0
    starts.append(('clump', _normalize(f_clump + 1e-6)))

    starts.append(('binary', _normalize(f_bin)))

    start_scores = []
    for name, f0 in starts:
        c0 = _autocorr(f0)
        peak0 = c0.max()
        score0 = 2.0 * n * peak0  # since sum = 1
        start_scores.append((score0, name, f0))
        print(f"  {name}: score={score0:.6f}, peak={peak0:.8f}")
    
    # Sort by score
    start_scores.sort(key=lambda x: x[0])
    
    # Track global best
    best_score = start_scores[0][0]
    best_f = start_scores[0][2].copy()

        if time.monotonic() > deadline - 60:
            break
        
        score0, name, f0 = start_scores[idx]
        t_alloc = time_per_start * (1.0 - 0.1 * idx)
        start_deadline = time.monotonic() + t_alloc
        
        print(f"\nOptimizing from '{name}' (budget: {t_alloc:.0f}s, initial score: {score0:.6f})")
        
        f_opt, score_opt = _optimize(f0, start_deadline, rng, label=name)
        
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print(f"  New best from '{name}': {best_score:.6f}")
        else:
            print(f"  Best from '{name}': {score_opt:.6f} (not new best)")

        f_pert = best_f + 0.01 / n * rng.standard_normal(n)
