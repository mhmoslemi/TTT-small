def full_conv(a):

def lp_direction(f, M):

    Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M.

    return f.copy(), 1.0

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based multiplicative
weight updates (solving max sum(b) s.t. (f*b)[k]<=M, b>=0) combined with
adaptive steps, hybrid gradient fallback, multi-start, and polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import correlate as _correlate


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_norm(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))


def normalize(v):
    """Clip to non-negative and normalize to sum 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(v.size) / v.size
    return v / s


def build_A(f):
    """Build convolution matrix A such that (f*b) = A @ b. A is (2n-1) x n."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M. Returns (b_opt, sum_b)."""
    n = f.size
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x, float(np.sum(res.x))
    except Exception:
        pass
    return f.copy(), 1.0


def smooth_gradient(a, b):
    """Compute smoothed subgradient of max(b) using log-sum-exp with adaptive tau."""
    n = a.size
    tau = 0.03 * float(np.max(b))
    if tau < 1e-18:
        tau = 1e-18
    shifted = b / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    # grad[j] = 2 * sum_m w[j+m] * a[m] = 2 * correlate(w, a, 'valid')[j]
    grad = 2.0 * _correlate(w, a, mode='valid')
    return grad


def optimize_lp(a, time_budget, rng, verbose=False):
    """
    Main LP-based optimization with hybrid gradient fallback.
    a: normalized non-negative array (sum=1).
    Returns (best_a, best_score).
    """
    n = a.size
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = score_norm(a)

    t0 = time.monotonic()
    step = 0.25
    it = 0
    last_imp = 0
    stall = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        # Try LP direction
        b_opt, sum_b = lp_direction(a, M)
        use_lp = sum_b > 1.0 + 1e-6

        if use_lp:
            direction = b_opt / sum_b
            alpha = (sum_b - 1.0)  # "gain" from LP
            # Scale step by gain: larger gain allows larger step
            eff_step = step * min(2.0, 1.0 + 0.5 * alpha)
            eff_step = min(eff_step, 0.7)
            a_new = normalize((1.0 - eff_step) * a + eff_step * direction)
            new_sc = score_norm(a_new)
            lp_fail_count = 0
        else:
            # Fall back to smoothed gradient
            g = smooth_gradient(a, b)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                stall += 1
                it += 1
                continue
            g = g / g_norm
            a_new = normalize(a - step * g)
            new_sc = score_norm(a_new)
            lp_fail_count += 1

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a_new.copy()
            last_imp = it
            stall = 0
            a = a_new
            step = min(0.6, step * 1.03)
        elif new_sc < cur_sc:
            a = a_new
            stall += 1
        else:
            # Backtrack: try half step
            s2 = step * 0.4
            if use_lp:
                a2 = normalize((1.0 - s2) * a + s2 * direction)
            else:
                a2 = normalize(a - s2 * g)
            sc2 = score_norm(a2)
            if sc2 < cur_sc:
                a = a2
                step = s2
            else:
                step *= 0.75
            stall += 1

        # Stall handling
        if stall > 80:
            # Choose restart strategy
            r = rng.random()
            if r < 0.4:
                # Small perturbation of best
                pert = 0.03 + 0.05 * rng.random()
                a = normalize(best_a + pert * rng.standard_normal(n))
            elif r < 0.7:
                # Medium perturbation
                pert = 0.1 + 0.1 * rng.random()
                a = normalize(best_a + pert * rng.standard_normal(n))
            else:
                # Random restart (keep some structure)
                a = normalize(best_a * 0.5 + 0.5 * np.abs(rng.standard_normal(n)))
            step = 0.2 + 0.2 * rng.random()
            stall = 0
            last_imp = it

        # Periodic reheat
        if it % 500 == 0 and it > 0:
            step = 0.15 + 0.25 * rng.random()

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            tag = "LP" if use_lp else "GR"
            print(f"    [{tag} it={it}] cur={cur_sc:.6f} best={best_sc:.6f} "
                  f"step={step:.3f} stall={stall} t={elapsed:.1f}s")

    return best_a, best_sc


def optimize_gradient(a, time_budget, rng, verbose=False):
    """Pure smoothed-subgradient descent with momentum (for polishing)."""
    n = a.size
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = score_norm(a)

    t0 = time.monotonic()
    eta = 0.08
    beta = 0.4
    vel = np.zeros(n)
    tau_frac = 0.02
    it = 0
    last_imp = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        shifted = b / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        g = 2.0 * _correlate(w, a, mode='valid')

        vel = beta * vel + (1.0 - beta) * g
        a_new = normalize(a - eta * vel / (np.linalg.norm(vel) + 1e-15))

        new_sc = score_norm(a_new)
        if new_sc <= cur_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = normalize(a - 0.5 * eta * vel / (np.linalg.norm(vel) + 1e-15))

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_imp = it

        it += 1
        if it - last_imp > 2000:
            a = normalize(best_a + 0.02 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.05 * rng.random()
            last_imp = it

        if it % 5000 == 0:
            eta *= 0.85
            tau_frac = 0.01 + 0.03 * rng.random()

    return best_a, best_sc


def polish_random(a, time_budget, rng):
    """Fine-grained random perturbation polish."""
    n = a.size
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = score_norm(a)

    t0 = time.monotonic()
    scale = 0.005
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-8:
        improved = False
        for _ in range(30):
            cand = normalize(a + scale * rng.standard_normal(n))
            sc = score_norm(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 10:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.15 * rng.random())
    c2 = n * (0.75 + 0.15 * rng.random())
    sigma = n * (0.05 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 3))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_power_law(n, rng):
    i = np.arange(n, dtype=np.float64)
    p = 0.3 + 0.7 * rng.random()
    a = 1.0 / np.power(i + 1.0, p)
    a = a + a[::-1]  # make symmetric
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()
