```python
"""
Low-peak nonnegative sequence search.

This program refines a set of structured starting sequences using a
multiplicative projected-gradient style subgradient method for the exact
objective

    2 n max_k (a * a)_k / (sum a)^2,

where a >= 0.  The gradient is computed from weighted active convolution
constraints using FFTs.  Several structured starts are generated, including
the provided previous sequence, one-sine phases, anti-palindromic projections,
and spectral square-root-of-flat starts.  The best starts are then given
dedicated refinement budgets.  The program always returns a valid nonnegative
Python list.
"""

import numpy as np
import time

_RNG = np.random.RandomState(12345)


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _make_uniform(n):
    """Uniform normalized sequence of length n."""
    n = max(1, int(n))
    return np.ones(n, dtype=np.float64) / float(n)


def _sanitize_sequence(x, floor=0.0):
    """Convert x to a finite nonnegative normalized numpy sequence."""
    if x is None:
        return None
    try:
        arr = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    if floor > 0.0:
        arr = np.maximum(arr, float(floor))

    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        arr = np.ones(arr.size, dtype=np.float64) / float(arr.size)
    else:
        arr = arr / s

    return arr


def _fft_convolve(a):
    """Linear self-convolution a * a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    L = _next_pow2(need)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return c[:need]


def _score_and_conv(a):
    """Return (score, convolution, max_conv) for a nonnegative sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf"), None, float("inf")

    c = _fft_convolve(a)
    if c.size == 0:
        return float("inf"), None, float("inf")

    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0

    score = float(2.0 * n * m / (s * s))
    if not np.isfinite(score):
        return float("inf"), c, m
    return score, c, m


def _logsumexp_loss(c, m, beta):
    """Smooth surrogate logsumexp(beta * c)."""
    if c is None or c.size == 0:
        return float("inf")
    if beta <= 0.0:
        return float(np.max(c))

    x = float(beta) * (np.asarray(c, dtype=np.float64) - float(m))
    x = np.clip(x, -50.0, 50.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    return float(beta * float(m) + np.log(s))


def _mult_update(a, direction, step):
    """Multiplicative update on the simplex: normalize(exp(log(a)+step*direction))."""
    a = np.asarray(a, dtype=np.float64)
    loga = np.log(np.maximum(a, 1e-300))
    z = loga + float(step) * direction
    if np.max(np.abs(z)) > 40.0:
        z = z - float(np.max(z))
    z = np.clip(z, -50.0, 50.0)
    b = np.exp(z)
    s = float(b.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a.copy()
    return b / s


def _grad_from_weights(w, a):
    """
    Gradient of sum_k w_k (a*a)_k with respect to a.

    Uses the identity
        grad_j = 2 * coeff_{j} of P(z) A(z^{-1}),
    computed as 2 * conv(w, reversed(a))[n-1 : n-1+n].
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    need = 3 * n - 2
    L = _next_pow2(max(1, need))
    ar = a[::-1]
    W = np.fft.rfft(w, L)
    AR = np.fft.rfft(ar, L)
    conv = np.fft.irfft(W * AR, L)
    return 2.0 * conv[n - 1:n + n - 1]


def _weights_from_c(c, m, thr_frac, q=1.0):
    """Create normalized nonnegative weights concentrated near max(c)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return None
    if m <= 1e-300:
        return np.ones(c.size, dtype=np.float64) / float(c.size)

    thr = float(m) - max(0.0, float(thr_frac)) * float(m)
    v = c - thr
    v = np.maximum(v, 0.0)
    if q != 1.0:
        v = v ** float(q)

    s = float(v.sum())
    if not np.isfinite(s) or s <= 1e-300:
        k = min(32, int(c.size))
        if k <= 0:
            return np.ones(c.size, dtype=np.float64) / float(c.size)
        if k < c.size:
            idx = np.argpartition(c, c.size - k)[-k:]
        else:
            idx = np.arange(c.size, dtype=np.int64)
        v = np.zeros(c.size, dtype=np.float64)
        v[idx] = 1.0
        s = float(idx.size)

    v = v / s
    return v


def _random_perturb(a, magnitude):
    """Small randomized multiplicative perturbation."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a.copy()
    noise = _RNG.normal(0.0, float(magnitude), size=a.size)
    noise = np.clip(noise, -3.0, 3.0)
    return _mult_update(a, noise, 1.0)


def _local_refine(start, budget_s, name="refine"):
    """
    Long local search on the simplex.

    The direction is a weighted sum of gradients of the most active convolution
    coefficients.  A multiplicative line search accepts candidates that improve
    the exact objective or, cautiously, improve a log-sum-exp surrogate.
    """
    budget_s = float(budget_s)
    if budget_s <= 0.0:
        return start, _score_and_conv(start)[0]

    a = _sanitize_sequence(start, floor=1e-12)
    if a is None:
        a = _make_uniform(256)

    current_score, current_c, current_m = _score_and_conv(a)
    if not np.isfinite(current_score):
        return a, float("inf")

    current = a.copy()
    best = a.copy()
    best_score = current_score

    thr_frac = 0.15
    step_factor = 0.6
    no_imp = 0
    it = 0
    start_t = time.monotonic()
    end_t = start_t + budget_s
    last_print = start_t

    while time.monotonic() < end_t:
        it += 1

        if it % 80 == 0 and time.monotonic() - last_print > 20.0:
            print(
                f"[{name}] it={it} current={current_score:.8f} "
                f"best={best_score:.8f} M={current_m:.6e} thr={thr_frac:.3f}"
            )
            last_print = time.monotonic()

        if best_score < 1.0006:
            break

        w = _weights_from_c(current_c, current_m, thr_frac, 1.0)
        if w is None:
            break

        grad = _grad_from_weights(w, current)
        g = grad - float(np.mean(grad))
        maxg = float(np.max(np.abs(g)))
        if not np.isfinite(maxg) or maxg < 1e-15:
            break

        step = step_factor / maxg
        accepted = False
        accepted_factor = 1.0

        current_tol = 1e-12 + 1e-10 * max(1.0, abs(current_score))
        best_tol = 1e-12 + 1e-10 * max(1.0, abs(best_score))

        loss_beta = 8.0 / max(float(current_m), 1e-300)
        cur_loss = _logsumexp_loss(current_c, current_m, loss_beta)

        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = _mult_update(current, -g, step * factor)
            sc, cc, mm = _score_and_conv(cand)
            if not np.isfinite(sc):
                continue

            if sc < best_score - best_tol:
                current = cand
                current_score = sc
                current_c = cc
                current_m = mm
                best = cand.copy()
                best_score = sc
                accepted = True
                accepted_factor = factor
                no_imp = 0
                step_factor = min(1.0, step_factor * 1.04)
                if thr_frac > 0.025:
                    thr_frac *= 0.87
                break

            if sc < current_score - current_tol:
                current = cand
                current_score = sc
                current_c = cc
                current_m = mm
                accepted = True
                accepted_factor = factor
                no_imp = max(0, no_imp - 1)
                if factor >= 0.5:
                    step_factor = min(0.9, step_factor * 1.02)
                break

            cand_loss = _logsumexp_loss(cc, mm, loss_beta)
            if (
                np.isfinite(cand_loss)
                and cand_loss < cur_loss - 1e-8
                and sc < current_score * (1.0 + 1e-4)
            ):
                current = cand
                current_score = sc
                current_c = cc
                current_m = mm
                accepted = True
                accepted_factor = factor
                no_imp = min(no_imp, 50)
                break

        if not accepted:
            step_factor = max(0.02, step_factor * 0.70)
            no_imp += 1

            if current_score > best_score * 1.002 and no_imp > 20:
                current = best.copy()
                current_score, current_c, current_m = _score_and_conv(current)
                no_imp = 0

            if no_imp % 25 == 0:
                if thr_frac < 0.45:
                    thr_frac = min(0.50, thr_frac * 1.25)
                else:
                    thr_frac = 0.12

                if no_imp > 70:
                    mag = 0.02 + 0.0005 * min(200, no_imp / 10)
                    cand = _random_perturb(best, mag)
                    sc, cc, mm = _score_and_conv(cand)
                    if np.isfinite(sc) and sc < best_score * 1.02:
                        current = cand
                        current_score = sc
                        current_c = cc
                        current_m = mm
                        no_imp = 30

            if no_imp > 2500:
                break

    # If floating point noise made best very close, return best.
    return best, best_score


def _to_anti_palindromic(a):
    """Project onto anti-palindromic nonnegative normalized sequences."""
    a = _sanitize_sequence(a, floor=0.0)
    if a is None:
        return None
    n = a.size
    mean = 1.0 / float(n)
    rev = a[::-1]
    h = 0.5 * ((a - mean) - (rev - mean))
    b = mean + h
    b = np.clip(b, 0.0, None)
    s = float(b.sum())
    if not np.isfinite(s) or s < 0.01:
        return _make_uniform(n)
    return b / s


def _make_sine(n, amp, phase):
    """One-sine candidate: normalize(1 + amp * sin(theta + phase))."""
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + float(amp) * np.sin(theta + float(phase))
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return _make_uniform(n)
    return a / s


def _quick_sine_candidates(n):
    """Quick scan over one-sine amplitudes and phases."""
    n = max(1, int(n))
    best_a = _make_uniform(n)
    best_sc = _score_and_conv(best_a)[0]

    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)

    amps = (
        -1.0, -0.9, -0.8, -0.7, -0.6, -0.5, -0.4, -0.3,
        0.0,
        0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0
    )
    phases = (
        0.0,
        np.pi / 4.0,
        np.pi / 2.0,
        3.0 * np.pi / 4.0,
        np.pi,
        5.0 * np.pi / 4.0,
        3.0 * np.pi / 2.0,
        7.0 * np.pi / 4.0,
    )

    for ph in phases:
        base = np.sin(theta + ph)
        for amp in amps:
            a = 1.0 + float(amp) * base
            a = np.clip(a, 0.0, None)
            s = float(a.sum())
            if not np.isfinite(s) or s < 0.01:
                continue
            a = a / s
            sc, _, _ = _score_and_conv(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _spectral_candidates(n):
    """
    Construct candidates from spectral square roots of a flat target
    convolution, followed by clipping and normalization.
    """
    n = max(1, int(n))
    best_a = _make_uniform(n)
    best_sc = _score_and_conv(best_a)[0]

    try:
        need = 2 * n - 1
        L = _next_pow2(need)
        target = np.ones(need, dtype=np.float64) / float(need)
        C = np.fft.rfft(target, L)
        Abase = np.sqrt(C)
        k = np.arange(L // 2 + 1, dtype=np.float64)

        shifts = (-n // 2, -n // 4, -n // 8, 0, n // 8, n // 4, n // 2)
        for sh in shifts:
            try:
                phase = np.exp(-1j * (2.0 * np.pi * k * float(sh)) / float(L))
                A = Abase * phase
                full = np.fft.irfft(A, L)

                slices = (full[:n], full[n + 1 - n:n + 1] if L > n else full[:n])
                for raw in slices:
                    b = np.clip(raw, 0.0, None)
                    s = float(b.sum())
                    if not np.isfinite(s) or s < 0.01:
                        continue
                    b = b / s
                    sc, _, _ = _score_and_conv(b)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = b.copy()
            except Exception:
                pass
    except Exception:
        pass

    return best_a, best_sc


def _register_candidate(cands, name, arr):
    """Register a candidate and its anti-palindromic projection."""
    a = _sanitize_sequence(arr, floor=0.0)
    if a is None:
        return

    sc, _, _ = _score_and_conv(a)
    if np.isfinite(sc):
        cands.append((float(sc), name, a, int(a.size)))

    b = _to_anti_palindromic(a)
    if b is not None:
        sc2, _, _ = _score_and_conv(b)
        if np.isfinite(sc2):
            cands.append((float(sc2), name + "_anti", b, int(b.size)))


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Search for a nonnegative sequence with low evaluate_sequence score.

    Returns a Python list of floats.
    """
    global _RNG

    try:
        _RNG.seed(int(seed))
    except Exception:
        _RNG.seed(12345)

    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    total_budget = max(30.0, total_budget)

    t0 = time.monotonic()
    deadline = t0 + total_budget - 30.0

    cands = []

    # Previous state of the art, if present.
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            _register_candidate(cands, "prev", hs)
    except Exception:
        pass

    # Structured starts.
    for n in (2048, 4096):
        try:
            if time.monotonic() > deadline - 60.0:
                break
            sa, ss = _quick_sine_candidates(n)
            _register_candidate(cands, f"sine_{n}", sa)
        except Exception:
            pass

        try:
            if time.monotonic() > deadline - 60.0:
                continue
            pa, ps = _spectral_candidates(n)
            _register_candidate(cands, f"spectral_{n}", pa)
        except Exception:
            pass

    # Final guaranteed fallback candidate.
    fallback = _make_uniform(2048)
    _register_candidate(cands, "uniform2048", fallback)

    if not cands:
        return [float(x) for x in fallback]

    # Sort by internal score without comparing arrays.
    scored = []
    for idx, (sc, name, arr, n) in enumerate(cands):
        if np.isfinite(sc):
            scored.append((float(sc), idx, name, arr, int(n)))
    scored.sort()

    best_a = scored[0][3].copy()
    best_internal = scored[0][0]

    # Refinement budgets: concentrate most time on the best starts.
    refined = []
    if time.monotonic() < deadline - 45.0:
        remaining = deadline - time.monotonic() - 25.0
        n_top = min(3, len(scored))
        if n_top == 1:
            budgets = (remaining,)
        elif n_top == 2:
            budgets = (0.62 * remaining, 0.38 * remaining)
        else:
            budgets = (0.52 * remaining, 0.32 * remaining, 0.16 * remaining)

        for j in range(n_top):
            if time.monotonic() > deadline - 25.0:
                break
            if remaining <= 5.0:
                break

            sc, idx, name, arr, n = scored[j]
            b = max(5.0, min(budgets[j], deadline - time.monotonic() - 25.0))
            if b <= 1.0:
                break

            try:
                ra, rsc = _local_refine(arr, b, name)
            except Exception:
                ra, rsc = arr, sc

            if ra is not None:
                ra = _sanitize_sequence(ra, floor=0.0)
                if ra is None:
                    ra = arr
                    rsc = sc
                else:
                    rsc, _, _ = _score_and_conv(ra)
                    if not np.isfinite(rsc):
                        rsc = sc
                        ra = arr

                refined.append((float(rsc), f"{name}_refined", ra, int(ra.size)))
                if np.isfinite(rsc) and rsc < best_internal:
                    best_internal = rsc
                    best_a = ra.copy()

            if best_internal < 1.45:
                break

    # Combine initial and refined candidates, validate with the official scorer.
    pool = []
    for sc, idx, name, arr, n in scored[:4]:
        pool.append((float(sc), idx, name, arr, int(n)))
    for sc, name, arr, n in refined:
        pool.append((float(sc), idx=len(pool), name, arr, int(n)))

    final_scored = []
    for idx, (sc, _, name, arr, n) in enumerate(pool):
        if np.isfinite(sc):
            final_scored.append((float(sc), idx, name, arr, int(n)))
    final_scored.sort()

    best_actual = float("inf")
    best_out = None

    for _, _, name, arr, n in final_scored[:4]:
        if time.monotonic() > deadline - 5.0:
            break
        out = _sanitize_sequence(arr, floor=0.0)
        if out is None:
            continue
        try:
            real = float(evaluate_sequence(out.tolist()))
        except Exception:
            real = float("inf")

        if np.isfinite(real) and real < best_actual:
            best_actual = real
            best_out = out
            print(f"[validate] {name} n={n} actual={real:.8f}")
            if best_actual < 1.45:
                break

    if best_out is None:
        best_out = best_a
        best_actual = best_internal

    if best_out is None or not np.isfinite(float(best_out.sum())) or best_out.size == 0:
        best_out = fallback

    best_out = _sanitize_sequence(best_out, floor=0.0)
    if best_out is None:
        best_out = fallback

    print(f"[final] best_actual={float(best_actual) if np.isfinite(float(best_actual)) else 'inf'}")
    return [float(x) for x in best_out]
```
