    """Minimize log(sum(exp(beta * c))) using gradient descent."""
    a = a.copy()
    n = len(a)
    for _ in range(n_iter):
        c = fft_convolve(a)
        # Compute gradient
        # dc[k]/da[j] = a[k-j] + a[j-k] (with valid indices)
        # grad[j] = sum_k c_k_weight * dc[k]/da[j]
        # where c_k_weight = exp(beta*c[k]) / sum(exp(beta*c)))
        
        weights = np.exp(beta * c)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(len(c)):
            w = weights[k]
            if w < 1e-10:
                continue
            # dc[k]/da[j] for each j
            # c[k] = sum_i a[i]*a[k-i]
            # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            for j in range(lo, hi+1):
                g = 0
                if 0 <= k-j < n:
                    g += a[k-j]
                if 0 <= j-k < n:
                    g += a[j-k]
                grad[j] += w * g
        
        # Update with projection onto nonnegativity
        a = a - lr * grad
        a = np.maximum(a, 0)
        # Renormalize
        s = a.sum()
        if s > 0:
            a /= s
    
    return a

```python
def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Low-peak nonnegative step-function search using gradient-based optimization
with log-sum-exp smoothing, FFT-based gradient computation, and active-set
refinement. Explores multiple parametric initializations (sinusoidal, chirp,
multi-harmonic) and refines via projected gradient descent on the simplex.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_init_sine(n, amp):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    a = 1.0 + amp * np.sin(theta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _make_init_chirp(n, amp):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i * i / n
    a = 1.0 + amp * np.cos(theta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _make_init_multi(n, amp1, amp2):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i / n
    a = 1.0 + amp1 * np.sin(theta) + amp2 * np.sin(2 * theta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _make_init_symmetric_sine(n, amp):
    """Anti-palindromic: a[i] + a[n-1-i] = constant."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    a = 1.0 + amp * np.sin(theta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _gradient_descent(a, n_iter, beta_start=20.0, beta_end=200.0,
                      lr_start=1.0, lr_end=0.01, deadline=None,
                      verbose=False, label=""):
    """
    Minimize max(conv(a,a)) via projected gradient descent on the simplex
    with log-sum-exp smoothing and annealing.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _internal_score(best_a)
    N = _next_pow2(4 * n)  # enough for all FFT operations
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = _fft_convolve(a)
        
        # Compute weights (softmax of beta * c)
        # Animate beta
        if n_iter > 1:
            frac = it / max(1, n_iter - 1)
        else:
            frac = 1.0
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Numerically stable softmax
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient via FFT
        W = np.fft.rfft(w, N)
        A_fft = np.fft.rfft(a, N)
        
        # term1[j] = sum_d w[j+d]*a[d] = ifft(FFT(w) * conj(FFT(a)))[j]
        term1 = np.fft.irfft(W * np.conj(A_fft), N)
        # term2[j] = sum_d w[j-d]*a[d] = ifft(FFT(w) * FFT(a))[j]
        term2 = np.fft.irfft(W * A_fft, N)
        
        grad = beta * (term1[:n] + term2[:n])
        
        # Gradient step with simplex projection
        a_new = a - lr * grad
        a = _project_simplex(a_new)
        
        # Check improvement periodically
        if it % 50 == 0 or it == n_iter - 1:
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_a = a.copy()
                best_score = sc
                if verbose:
                    print(f"  [{label}] iter={it} score={sc:.8f} beta={beta:.1f} lr={lr:.4f}")
    
    # Final score
    sc = _internal_score(a)
    if np.isfinite(sc) and sc < best_score:
        best_a = a.copy()
        best_score = sc
    
    return best_a, best_score


def _active_set_refine(a, budget_s=60.0, K=12):
    """
    Active-set refinement: focus gradient on the lags where c[k] is largest.
    """
    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _internal_score(best_a)
    
    N = _next_pow2(4 * n)
    lr = max(1.0, 0.02 * n)
    max_lr = 0.3 * n
    it = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() - start < budget_s:
        it += 1
        
        c = _fft_convolve(a)
        max_c = c.max()
        
        # Select active lags
        if len(c) <= K:
            active = np.arange(len(c))
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient from active lags only
        # Use the same FFT trick but with "sparse" weights
        w = np.zeros(len(c))
        w[active] = 1.0 / K
        w /= w.sum()
        
        W = np.fft.rfft(w, N)
        A_fft = np.fft.rfft(a, N)
        term1 = np.fft.irfft(W * np.conj(A_fft), N)
        term2 = np.fft.irfft(W * A_fft, N)
        grad = term1[:n] + term2[:n]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        a_new = a - lr * grad / gnorm * 0.01  # scale down
        a = _project_simplex(a_new)
        
        # Check score
        if it % 20 == 0:
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_a = a.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.1)
                stall_count = 0
            else:
                if sc > best_score * 1.05:
                    a = best_a.copy()
                    lr = max(1.0, lr * 0.7)
                stall_count += 1
                if stall_count > 10:
                    break
    
    sc = _internal_score(a)
    if np.isfinite(sc) and sc < best_score:
        best_a = a.copy()
        best_score = sc
    
    return best_a, best_score


def _safe_resolve(x):
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def propose_candidate(seed=42,
