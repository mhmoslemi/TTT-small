6. Iterate

This is essentially what the previous algorithm does, but with the smooth-max instead of the hard max.

**Key improvement I'll try: Hydraulic pressure model**

Instead of using smooth-max, I'll use a "hydraulic" model:
- Imagine each `b_k` as a "bucket" that can hold up to `t` units
- Any excess flows back as a force on the `a_i` values
- The force on `a_j` from bucket k is proportional to `max(0, b_k - t) * (a_{k-j} + a_{j-k})` (if valid)
- This creates a natural "pressure equalization" that tries to flatten the convolution

This is equivalent to projected gradient descent on `sum_k max(0, b_k - t)^2` with `t` updated to be the current max.

Let me now write the code. I'll combine several ideas:

1. Start from `height_sequence_1` if available
2. Use a hybrid approach:
   - First, do a few iterations of block-level optimization to get a good "shape"
   - Then, do full-resolution gradient descent with the "hydraulic" sub-gradient
   - Use adaptive step sizes and momentum
   - Periodically try different "modes" (e.g., focus on reducing the single max vs. balancing the top-5 maxes)

def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0:
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)

        return v
    if target <= 0.0:
        return np.zeros_like(v)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size))
    
    # Bisection on the threshold
    lo = float(np.min(v))
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0:
        q = np.full(v.shape, target / float(v.size))
    else:
        q *= target / s
    return q


def normalize_simplex(p):

    s = float(np.sum(p))
    if s <= 0.0:
        return np.full(p.shape, 1.0 / float(p.size))
    return p / s


def smooth_max_weights(b, beta):

def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    if L == 0:
        return np.ones(n) / n
    z = np.zeros(B)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).flatten()
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - z.max()
    wb = np.exp(z)
    s = wb.sum()
    if s <= 0:
        return np.ones(n) / n
    wb = wb / s / L
    return np.repeat(wb, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    if L == 0:
        return float("inf"), np.zeros(max(0, B-1))
    p = block_to_p(n, B, c)
    b = fft_convolution(p)
    G, w = smooth_max_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(max(0, B-1))
    s = corr_vector(p, w)
    g = 2.0 * s
    Gsum = g.reshape(B, L).sum(axis=1)
    # gradient w.r.t. log-weights z (where p = repeat(exp(z - max(z)) / (B*L) / L... ))
    # Actually, let's use the chain rule properly.
    # p_i = wb[block(i)] where wb = softmax(z) / L (approximately, up to normalization)
    # The gradient w.r.t. z is: (softmax(z) * (Gsum - mean(softmax(z) * Gsum)))
    # This is the standard softmax gradient.
    z = np.zeros(B)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).flatten()
        m = min(B-1, c.size)
        z[:m] = c[:m]
    z = z - z.max()
    wb = np.exp(z)
    wb = wb / wb.sum()
    grad_z = wb * (Gsum - np.dot(wb, Gsum))
    grad = n * grad_z  # scale by n since objective is n * G
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad[:B-1] if B > 1 else np.zeros(0)


def phased_optimization(p_init, deadline, rng, beta_start=5000.0, beta_end=500000.0):
    """
    Main optimization loop with adaptive parameters.
    Uses projected gradient/sub-gradient descent with momentum.
    """
    p = normalize_simplex(p_init)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Break symmetry
    if n > 1 and np.max(np.abs(p - p.mean())) < 1e-14:
        t = np.arange(n) * 1e-4
        p = normalize_simplex(p * (1.0 + 0.01 * np.sin(2*np.pi*t*3.7)))
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    step = 0.5
    it = 0
    no_improve = 0
    beta = beta_start
    
    # Schedule: gradually increase beta
    total_iters = 50000
    annealing_period = 2000
    
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        
        # Update beta (annealing schedule)
        progress = min(1.0, it / (total_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute convolution and gradient
        b = fft_convolution(p)
        M = float(np.max(b))
        cur_score = 2.0 * n * M / float(np.sum(p)**2)
        S = float(np.sum(p))
        cur_score = 2.0 * n * M / (S * S)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 500 == 0:
            print(f"  iter={it} beta={beta:.0f} score={best_score:.8f} cur={cur_score:.8f} step={step:.4f}")
        
        if no_improve > 5000:
            # Try a perturbation
            pert = rng.normal(0, 0.01, n)
            p = normalize_simplex(p + pert)
            no_improve = 0
            step = 0.5
            continue
        
        # Compute gradient
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break
        grad = gradient_from_weights(p, w)
        
        # Center the gradient (remove mean to respect the sum constraint)
        grad = grad - grad.mean()
        
        # Momentum
        momentum = mu * momentum + grad
        direction = -momentum
        
        # Adaptive step size (backtracking)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        
        cur_smooth = n * G
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult / dnorm * 0.1
            q = project_simplex(p + alpha * direction, 1.0)
            qb = fft_convolution(q)
            G2, _ = smooth_max_weights(qb, beta)
            if np.isfinite(G2) and n * G2 < cur_smooth - 1e-14:
                p = q
                cur_smooth = n * G2
                accepted = True
                break
        
        if not accepted:
            step *= 0.7
            if step < 1e-10:
                break
        else:
            step = min(5.0, step * 1.1)
    
    return best_p, best_score


def subgradient_step(p, b, n, top_k=5):
    """
    Compute a sub-gradient of max(b) using the top-k largest values.
    This is more "democratic" than using just the single max.
    """
    # Find top-k indices
    k = min(top_k, b.size)
    top_idx = np.argpartition(b, -k)[-k:]
    top_vals = b[top_idx]
    
    # Weights proportional to how close to max
    max_val = b[top_idx].max()
    weights = np.ones(k) / k  # uniform weights for simplicity
    # Or: weights = (top_vals / max_val) ** 10  # sharp weights
    # weights /= weights.sum()
    
    w = np.zeros(b.size)
    w[top_idx] = weights
    
    return gradient_from_weights(p, w)


def block_search(n, deadline, rng):
    """Block-level L-BFGS optimization."""
    if not HAVE_SCIPY or n < 16:
        return None, float("inf")
    
    best_p = None
    best_score = float("inf")
    
    for B in (16, 32, 64, 128, 256):
        if B >= n or n % B != 0:
            continue
        if time.monotonic() >= deadline:
            break
        
        bounds = [(-5.0, 5.0)] * (B - 1)
        best_c = np.zeros(B - 1)
        best_B_score = float("inf")
        
        # Multiple initializations
        x = np.linspace(0, 1, B-1)
        inits = [
            np.zeros(B-1),
            2.0 * (x - 0.5),
            -2.0 * (x - 0.5),
            3.0 * np.sin(np.pi * x),
            -3.0 * np.sin(np.pi * x),
            2.0 * np.cos(2*np.pi*x),
            rng.normal(0, 1, B-1) * 1.5,
        ]
        
        for c0 in inits:
            if time.monotonic() >= deadline:
                break
            c0 = np.clip(c0, -5.0, 5.0)
            try:
                res = minimize(
                    block_objective, c0,
                    args=(n, B, 5000.0),
                    method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-5}
                )
                c = np.clip(res.x, -5.0, 5.0)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
            except Exception:
                pass
        
        # Sharper pass
        if np.isfinite(best_B_score) and time.monotonic() < deadline:
            try:
                res = minimize(
                    block_objective, best_c,
                    args=(n, B, 100000.0),
                    method='L-BFGS-B', jac=True,
                    bounds=bounds,
                    options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-7}
                )
                c = np.clip(res.x, -5.0, 5.0)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
            except Exception:
                pass
        
        p = block_to_p(n, B, best_c)
        sc = fast_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            print(f"  block B={B}: score={sc:.8f}")
    
    return best_p, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):

def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size

            lo = mid
        else:

    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):

    return q * (target / s)


def normalize_simplex(p):

    s = float(np.sum(p))

    return p / s

def corr_vector(p, w):
    """Compute corr[i] = sum_j p[j] * w[i+j] for i=0..n-1 via FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)

    m = w.size

    size = 1

    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return np.maximum(s, 0.0)

    """Gradient of sum_k w[k]*b[k] where b = p*p. Result: 2*corr(p,w)."""
    s = corr_vector(p, w)

def block_to_p(n, B, c):
    """Convert block log-weights (length B-1) to full sequence of length n."""
    L = n // B
    if L == 0 or B <= 0:
        return np.ones(n) / max(n, 1)
    z = np.zeros(B)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).flatten()
        m = min(B - 1, c.size)
        if m > 0:

    z = z - z.max()
    wb = np.exp(z)
    s = float(wb.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n) / n
    wb = (wb / s) / L
    return np.repeat(wb, L)

            inits.extend([
                3.0 * (x - 0.5),
                -3.0 * (x - 0.5),
                4.0 * np.sin(np.pi * x),
                -4.0 * np.sin(np.pi * x),
                2.0 * np.cos(2 * np.pi * x),
                2.0 * np.sin(2 * np.pi * x),
                rng.normal(0, 1.2, B - 1),
            ])
        
        for c0 in inits:
            if time.monotonic() >= deadline:
                break
            c0 = np.clip(c0, -5.0, 5.0)
            try:
                res = minimize(block_objective, c0, args=(n, B, 5000.0),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 120, 'ftol': 1e-10, 'gtol': 1e-5, 'maxcor': 10})
                c = np.clip(res.x, -5.0, 5.0)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
            except Exception:
                pass
        
        # Sharper pass with high beta
        if np.isfinite(best_B_score) and time.monotonic() < deadline:
            try:
                res = minimize(block_objective, best_c, args=(n, B, 100000.0),
                               method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-7, 'maxcor': 10})
                c = np.clip(res.x, -5.0, 5.0)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
            except Exception:
                pass
        
        p = block_to_p(n, B, best_c)
        sc = fast_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            print(f"    block B={B}: score={sc:.8f}")
    
    return best_p, best_score


def hybrid_refine(p0, deadline, rng, beta_start=3000.0, beta_end=500000.0,
                  max_iter=200000, top_k_mode_every=4, top_k_size=5):
    """
    Full-resolution projected gradient with:
    - Smooth-max gradient (majority of steps)
    - Top-k subgradient (periodic, for balancing multiple peaks)
    - Momentum
    - Adaptive step size with backtracking
    - Periodic perturbation to escape local minima
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")
    
    # Break symmetry
    if n > 1 and float(np.max(np.abs(p - p.mean()))) < 1e-14:
        t = np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(3.7 * 2 * np.pi * t / n)))
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.8
    it = 0
    no_improve = 0
    total_iters = min(max_iter, 200000)
    perturb_period = 3000
    
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        
        # Progressive beta increase
        progress = min(1.0, it / (0.85 * total_iters))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute current state
        b = fft_convolution(p)
        M = float(np.max(b))
        S = float(np.sum(p))
        cur_score = 2.0 * n * M / (S * S)
        
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 1000 == 0:
            print(f"      iter={it} beta={beta:.0f} best={best_score:.8f} "
                  f"cur={cur_score:.8f} step={step:.4f} no_imp={no_improve}")
        
        # Perturbation to escape local minima
        if no_improve > perturb_period:
            scale = 0.005 + 0.02 * (no_improve - perturb_period) / perturb_period
            pert = rng.normal(0, scale, n)
            p = normalize_simplex(p + pert)
            no_improve = 0
            step = 0.8
            momentum *= 0.3
            continue
        
        if no_improve > perturb_period * 3:
            break
        
        # Choose gradient mode
        use_topk = (it % top_k_mode_every == 0) and it > 500
        
        if use_topk:
            # Top-k subgradient: equal weight on top-k peaks
            k = min(top_k_size, b.size)
            if k > 1:
                top_idx = np.argpartition(b, -k)[-k:]
                w = np.zeros(b.size)
                w[top_idx] = 1.0 / k
            else:
                w = np.zeros(b.size)
                w[int(np.argmax(b))] = 1.0
        else:
            # Smooth-max gradient
            G, w = smooth_max_weights(b, beta)
            if not np.isfinite(G):
                break
        
        grad = compute_gradient(p, w)
        
        # Center gradient (orthogonalize to all-ones direction)
        grad = grad - float(np.mean(grad))
        
        # Momentum
        momentum = mu * momentum + grad
        
        # Direction
        direction = -momentum
        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-16:
            break
        
        direction = direction / dnorm
        
        # Backtracking line search
        cur_obj = n * M  # proxy; we'll use smooth max for acceptance
        G_cur, _ = smooth_max_weights(b, beta)
        cur_smooth = n * G_cur
        
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.01):
            alpha = step * mult
            q = project_simplex(p + alpha * direction, 1.0)
            qb = fft_convolution(q)
            M2 = float(np.max(qb))
            S2 = float(np.sum(q))
            if S2 <= 0:
                continue
            sc2 = 2.0 * n * M2 / (S2 * S2)
            G2, _ = smooth_max_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            
            # Accept if either score or smooth objective improves
            if (n * G2 < cur_smooth - 1e-14) or (sc2 < cur_score - 1e-12):
                p = q
                cur_smooth = n * G2
                cur_score = sc2
                accepted = True
                break
        
        if accepted:
            step = min(5.0, step * 1.15)
        else:
            step *= 0.6
            if step < 1e-9:
                step = 0.3  # reset to moderate step
                momentum *= 0.5
    
    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator safely."""
    try:
        val = float(evaluate_sequence(seq))
        return val if np.isfinite(val) else float("inf")
    except Exception:
        return float("inf")


def generate_candidates(n, rng):
    """Generate diverse initial candidates."""
    candidates = []
    x = np.linspace(0.0, 1.0, n)
    
    # Sine-squared (bell shape, zero at both ends)
    p = np.sin(np.pi * x) ** 2
    p = np.maximum(p, 0)
    candidates.append(("sine2", normalize_simplex(p)))
    
    # Sine (zero at left, small at right)
    p = np.sin(np.pi * x)
    p = np.maximum(p, 0)
    candidates.append(("sine", normalize_simplex(p)))
    
    # Various powers of sine
    for pw in (1.5, 3.0, 4.0):
        p = np.sin(np.pi * x) ** pw
        p = np.maximum(p, 0)
        candidates.append((f"sine{pw}", normalize_simplex(p)))
    
    # Skewed sine-squared
    for skew in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        p = np.sin(np.pi * x) ** 2 * np.exp(skew * (x - 0.5))
        p = np.maximum(p, 0)
        candidates.append((f"skew{skew}", normalize_simplex(p)))
    
    # Gaussian-like bumps (off-center)
    for center, width in [(0.3, 0.1), (0.4, 0.15), (0.6, 0.15), (0.7, 0.1),
                          (0.35, 0.2), (0.65, 0.2)]:
        p = np.exp(-((x - center) / width) ** 2)
        p = np.maximum(p, 0)
        candidates.append((f"gauss_{center}_{width}", normalize_simplex(p)))
    
    # Double hump
    for c1, c2 in [(0.3, 0.7), (0.25, 0.75), (0.35, 0.65)]:
        for w in (0.08, 0.12, 0.18):
            p = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
            p = np.maximum(p, 0)
            candidates.append((f"dgh_{c1}_{c2}", normalize_simplex(p)))
    
    # Cosine window (Hann-like)
    p = 0.5 * (1 + np.cos(2 * np.pi * x))
    p = np.maximum(p, 0)
    candidates.append(("hann", normalize_simplex(p)))
    
    # Flat top with tapered edges
    for taper in (0.1, 0.15, 0.2):
        p = np.ones(n)
        m = int(taper * n)
        if m > 0:
            p[:m] = 0.5 * (1 - np.cos(np.pi * np.arange(m) / m))
            p[-m:] = 0.5 * (1 - np.cos(np.pi * np.arange(m) / m))[::-1]
        p = np.maximum(p, 0)
        candidates.append((f"flat_{taper}", normalize_simplex(p)))
    
    # Random smooth (via random Fourier coefficients)
    for _ in range(4):
        size = 1
        while size < n:
            size <<= 1
        # Random low-frequency signal
        freqs = np.fft.rfftfreq(size)
        mask = freqs < 5.0 / size * n  # low frequency mask
        zc = np.zeros(size // 2 + 1, dtype=complex)
        for i in range(size // 2 + 1):
            if mask[i]:
                zc[i] = rng.normal() + 1j * rng.normal()
        z = np.real(np.fft.irfft(zc, size))[:n]
        z = z - z.mean()
        z = 2.0 * np.tanh(z * 0.5)
        p = np.exp(z)
        p = np.maximum(p, 0)
        candidates.append(("rand_smooth", normalize_simplex(p)))
    
    return candidates


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal nonnegative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    
    Multi-phase: diverse starts -> block L-BFGS -> full hybrid gradient refine.
    """
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    
    try:
        budget_s = float(budget_s)
    except (ValueError, TypeError):
        budget_s = 2400.0
    
    if budget_s <= 0:
        return [1.0]
    
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    
    # Time budget allocation
    reserve = 60.0
    total_time = max(10.0, budget_s - reserve)
    deadline = t_start + total_time
    
    N = 8192
    print(f"=== propose_candidate: budget={budget_s:.0f}s, N={N} ===")
    
    # --- Phase 0: Gather and score candidates ---
    scored = []
    
    # Try loading previous best
    try:
        prev = height_sequence_1
        if isinstance(prev, (list, np.ndarray)) and len(prev) > 0:
            prev_arr = np.asarray(prev, dtype=np.float64)
            prev_arr = np.clip(prev_arr, 0.0, 1000.0)
            prev_arr = normalize_simplex(prev_arr)
            sc = fast_score(prev_arr)
            if np.isfinite(sc):
                scored.append((sc, "previous", prev_arr))
                print(f"Previous best: score={sc:.8f}, len={len(prev_arr)}")
    except (NameError, Exception) as e:
        print(f"No previous sequence: {e}")
    
    # Generate diverse candidates
    candidates = generate_candidates(N, rng)
    for name, p in candidates:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, name, p))
    
    scored.sort(key=lambda x: x[0])
    print(f"Total candidates: {len(scored)}")
    print(f"Top 5: {[f'{n}={s:.6f}' for s, n, _ in scored[:5]]}")
    
    # --- Phase 1: Block search ---
    block_budget = max(30.0, 0.25 * total_time)
    block_deadline = t_start + block_budget
    if time.monotonic() < block_deadline:
        print(f"\n--- Phase 1: Block search (budget={block_budget:.0f}s) ---")
        p_block, sc_block = run_block_search(N, block_deadline, rng)
        if p_block is not None and np.isfinite(sc_block):
            scored.insert(0, (sc_block, "block", p_block))
            scored.sort(key=lambda x: x[0])
            print(f"Block best: {sc_block:.8f}")
    
    # --- Phase 2: Full hybrid refinement from top starts ---
    top_n = min(4, len(scored))
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 2: Full refinement (remaining={remaining:.0f}s) ---")
    
    if remaining > 60:
        # Allocate time proportionally to top candidates
        weights = []
        for i in range(top_n):
            if i == 0:
                weights.append(0.40)
            elif i == 1:
                weights.append(0.25)
            elif i == 2:
                weights.append(0.20)
            else:
                weights.append(0.15)
        wsum = sum(weights)
        weights = [w / wsum for w in weights]
        
        best_p = scored[0][2].copy()
        best_score = scored[0][0]
        
        for i in range(top_n):
            alloc = remaining * weights[i] - 20.0  # small reserve per run
            if alloc < 15:
                continue
            sub_deadline = time.monotonic() + alloc
            p0 = scored[i][2].copy()
            name = scored[i][1]
            sc0 = scored[i][0]
            print(f"  Refining '{name}' (start={sc0:.8f}, alloc={alloc:.0f}s)")
            
            p_fit, sc_fit = hybrid_refine(
                p0, sub_deadline, rng,
                beta_start=3000.0, beta_end=500000.0,
                max_iter=100000, top_k_mode_every=4, top_k_size=5
            )
            
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  *** Improved to {sc_fit:.8f} ***")
            else:
                print(f"  No improvement (got {sc_fit:.8f})")
            
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
    
    # --- Phase 3: Final high-beta polish ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n--- Phase 3: Final polish (budget={remaining:.0f}s) ---")
        p_fit, sc_fit = hybrid_refine(
            best_p, deadline, rng,
            beta_start=50000.0, beta_end=2000000.0,
            max_iter=50000, top_k_mode_every=3, top_k_size=3
        )
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Polish improved to {sc_fit:.8f}")
    
    # --- Final: validate and return ---
    final_p = normalize_simplex(best_p)
    final_p = np.clip(final_p, 0.0, 1000.0)
    final_seq = [float(v) for v in final_p]
    
    off = official_score(final_seq)
    print(f"\n=== Final official score: {off:.8f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    
    if not np.isfinite(off) or off > 2.0 + 1e-6:
        print("WARNING: Falling back to uniform sequence!")
        return [1.0 / N] * N
    
    return final_seq
