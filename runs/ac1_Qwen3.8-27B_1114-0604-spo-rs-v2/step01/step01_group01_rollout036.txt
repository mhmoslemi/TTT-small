```python
from scipy.optimize import linprog

def lp_direction(a, M):
    """Find b that maximizes sum(b) subject to convolve(a, b) <= M, b >= 0"""
    n = len(a)
    # convolve(a, b)[k] = sum_i a[i] * b[k-i] for valid k
    # Full convolution has 2n-1 elements
    # We need convolve(a, b)[k] <= M for all k = 0, ..., 2n-2

    # A[k, j] = a[j] if k >= j and k-j < n, else 0
    # Actually: convolve(a,b)[k] = sum_j a[j]*b[k-j] where 0<=j<n and 0<=k-j<n
    # So A[k,j] = a[j] if 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)

    # For n=3072, A is (6143, 3072) - manageable

        for j in range(j_start, j_end + 1):

    return None
```

```python
"""
LP-based iterative refinement for minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Algorithm:
1. Start from initial sequence (parent or random)
2. Iteratively solve LP: max sum(b) s.t. b>=0, convolve(a,b) <= M
3. Normalize and mix back
4. Polish with MMD and local search
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def _convolve(a, b):
    """Compute full linear convolution using FFT for large inputs."""
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b)
    L = 1
    while L < nout:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]

def _score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = _convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    return 2.0 * n * np.max(c) / (S * S)

def _solve_lp(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, convolve(a, b)[k] <= M for all k.
    Returns the optimal b (unnormalized).
    """
    n = len(a)
    # Build constraint matrix A (sparse)
    # A[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            if a[j] > 0:  # skip zeros for sparsity
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```

LP-based iterative refinement for minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Key idea: iteratively solve the LP
    max sum(b) s.t. b >= 0, convolve(a, b)[k] <= M for all k
where M = max(convolve(a,a)). The optimal b, normalized to sum 1, gives a
direction that "spreads out" the convolution. Mixing back and repeating
converges to a sequence with a flattening autocorrelation profile.

Combined with MMD (multiplicative smoothed-max descent) polish and local
block search for final refinement.

import numpy as np
from scipy.optimize import linprog

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _convolve_full(a, b):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = np.asarray(p, dtype=float)
    m = p.size
    c = _convolve_full(p, p)
    sm = float(p.sum())
    if sm < 0.01:
        return float('inf')
    return float(2.0 * m * np.max(c) / (sm * sm))


def _build_lp_arrays(n):
    """Build (rows, cols) for the LP constraint matrix structure."""
    rows_list = []
    cols_list = []
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        cnt = k_end - k_start + 1
        rows_list.append(np.arange(k_start, k_end + 1, dtype=np.int32))
        cols_list.append(np.full(cnt, j, dtype=np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _solve_lp(a, M, structure):
    """Solve max sum(b) s.t. b>=0, convolve(a,b) <= M elementwise."""
    n = len(a)
    rows, cols = structure
    vals = a[cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

            h = h / maxh
            chosen = None
            for a_mult in (0.5, 1.0, 2.0, 4.0):
                a_val = alpha * a_mult
                if 1e-4 <= a_val <= 20.0:
                    dec = np.exp(-a_val * h)
                    p_try = p * dec
                    p_try = np.maximum(p_try, floor)
                    ss = float(p_try.sum())
                    if ss > 0 and np.isfinite(ss):
                        p_try = p_try / ss
                        # Use smooth score for step selection
                        smooth_cur = float(np.dot(w, d))
                        c_try = _convolve_full(p_try, p_try)
                        d_try = n * c_try
                        e_try_shift = np.exp(np.clip(beta * (d_try - np.max(d_try)), -500, 500))
                        s_try_shift = float(e_try_shift.sum())
                        if s_try_shift > 0:
                            w_try = e_try_shift / s_try_shift
                            smooth_try = float(np.dot(w_try, d_try))
                        else:
                            smooth_try = float(np.max(d_try))
                        if smooth_try + 1e-12 < smooth_cur:
                            chosen = p_try
                            break
            if chosen is not None:
                p = chosen
                stall = 0
                sc = _exact_score(p)
                if sc < best_score:
                    best = p.copy()
                    best_score = sc
            else:
                stall += 1
                alpha = max(1e-4, alpha * 0.7)
                if stall >= 20:
                    p_try = _sanitize(p * np.exp(rng.normal(0, 0.1, n)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor) / p_try.sum()
                        sc = _exact_score(p_try)
                        if sc < best_score * 1.05:
                            p = p_try
                        else:
                            p = best.copy()
                    stall = 0
                if stall > 80:
                    p = best.copy()
                    alpha = 0.1
                    stall = 0

        it += 1
        if it % 15 == 0:
            beta = min(beta_end, beta * 1.12)

    return best, best_score


def _local_block_search(p0, rng, time_limit):
    """Random block scaling local search."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = p.size
    floor = 1e-18 / n
    p = np.maximum(p, floor) / np.maximum(p, floor).sum()
    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = (0.4, 0.7, 0.9, 1.1, 1.3, 1.7, 2.5)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 300 + 1)))
        elif r < 0.8:
            w = int(rng.integers(max(2, n // 300), max(3, n // 50 + 1)))
        else:
            lo = max(3, n // 50)
            hi = max(lo + 1, min(n, n // 10 + 1))
            w = int(rng.integers(lo, hi + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        improved = False
        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, floor)
            ss = float(cand.sum())
            if ss <= 0:
                continue
            cand = cand / ss
            sc = _exact_score(cand)
            if sc < cur_score - 1e-12:
                cur_score = sc
                p = cand
                improved = True
                break
        if improved:
            stagnant = 0
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.05, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor) / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
            if stagnant > 300:
                p = best.copy()
                cur_score = best_score
                stagnant = 0

    return best, best_score


def _resample(arr, m):
    """Linear interpolation resampling to length m."""
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _initial_sequence(m, rng, mode):
    """Create a diverse non-negative normalized start."""
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    if mode == 0:
        # Beta-like
        a = float(rng.uniform(0.1, 3.0))
        b = float(rng.uniform(0.1, 3.0))
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)
    elif mode == 1:
        # Gaussian bumps
        K = int(rng.integers(1, 5))
        p = np.zeros(m)
        for _ in range(K):
            c = float(rng.uniform(0, 1))
            s = float(rng.uniform(0.05, 0.4))
            w = float(rng.uniform(0.5, 2.0))
            p += w * np.exp(-((t - c) / s) ** 2)
    elif mode == 2:
        # Two-block (asymmetric)
        frac = float(rng.uniform(0.3, 0.7))
        idx = int(round(frac * m))
        p = np.ones(m)
        p[idx:] = float(rng.uniform(0.5, 3.0))
        p = p * np.exp(-((t - 0.5) / 0.3) ** 2 * 0.5)
    else:
        # Cosine series
        z = np.ones(m)
        for k in range(1, min(10, m // 20 + 1)):
            amp = float(rng.normal(0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0, 2 * np.pi))
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -30, 30)))
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000.0) + 1e-10
    s = float(p.sum())
    if s <= 0:
        p = np.ones(m)
        s = float(m)
    return p / s

    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 30:
        budget = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget - 30.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    # Try to get parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
        if parent_arr is not None:
            print(f"Parent loaded, n={parent_arr.size}, score={_exact_score(parent_arr):.6f}")
    except NameError:
        pass
    except Exception:
        pass

    # Choose working length
    n = 4096
    if budget < 600:
        n = 2048

    # Build LP structure (expensive but done once)
    print(f"Building LP structure for n={n}...")
    t0 = time.monotonic()
    structure = _build_lp_arrays(n)
    print(f"  Done in {time.monotonic()-t0:.1f}s, nnz={len(structure[0])}")

    # Build starting points
    starts = []
    if parent_arr is not None:
        sp = _resample(parent_arr, n)
        if sp is not None:
            starts.append(sp)
            print(f"  Parent resampled to n={n}, score={_exact_score(sp):.6f}")

    for mode in (0, 1, 2, 3):
        starts.append(_initial_sequence(n, rng, mode))

    scored = []

    print(f"Start scores: {[f'{s:.4f}' for s, _ in scored]}")

        if talloc < 30:
