        for amp1, amp2, m1, m2 in [(0.5, 0.3, 1, 2), (0.6, 0.2, 1, 3), (0.4, 0.4, 2, 3)]:
            a = _make_two_sine(n, amp1, m1, amp2, m2)
            sc = _score(a)
            candidates.append((sc, a, f"2sine_n{n}_{amp1}_{amp2}_{m1}_{m2}"))
            print(f"[init] 2sine n={n} score={sc:.6f}")

    # Try height_sequence_1 if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            h1 = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            h1 = np.clip(np.where(np.isfinite(h1), h1, 0.0), 0.0, 1000.0)
            s1 = h1.sum()
            if s1 > 0.01:
                h1 = h1 / s1
                sc1 = _score(h1)
                candidates.append((sc1, h1, "height_sequence_1"))
                print(f"[init] height_sequence_1 n={h1.size} score={sc1:.6f}")
        except Exception as e:
            print(f"[init] error loading height_sequence_1: {e}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    best_init_score = candidates[0][0]
    best_init = candidates[0][1].copy()
    print(f"[init] best: {candidates[0][2]} score={best_init_score:.6f}")

    # Also keep second best for diversity
    second_init = candidates[1][1].copy() if len(candidates) > 1 else None

    # Phase 2: Mirror descent optimization (most of remaining time)
    remaining = hard_deadline - time.monotonic()
    opt_budget = remaining * 0.85
    opt_deadline = time.monotonic() + opt_budget

    print(f"[opt] starting mirror descent, budget={opt_budget:.0f}s")
    opt_a, opt_score = _mirror_descent(best_init, opt_budget, deadline=opt_deadline)
    print(f"[opt] mirror descent done: score={opt_score:.6f}")

    best_a = opt_a.copy()
    best_score = opt_score

    # Phase 3: If time remains, try optimizing the second initialization
    remaining = hard_deadline - time.monotonic()
    if remaining > 120 and second_init is not None:
        opt_budget2 = min(remaining * 0.6, 300.0)
        opt2_a, opt2_score = _mirror_descent(second_init, opt_budget2, deadline=hard_deadline - 30)
        print(f"[opt2] second init score={opt2_score:.6f}")
        if np.isfinite(opt2_score) and opt2_score < best_score:
            best_a = opt2_a.copy()
            best_score = opt2_score

    # Phase 4: Final block refinement if time remains
    remaining = hard_deadline - time.monotonic()
    if remaining > 30:
        block_budget = min(remaining * 0.5, 60.0)
        bl_a, bl_score = _refine_block(best_a, block_budget, deadline=hard_deadline - 10)
        print(f"[block] score={bl_score:.6f}")
        if np.isfinite(bl_score) and bl_score < best_score:
            best_a = bl_a.copy()
            best_score = bl_score

    # Phase 5: Validate with evaluate_sequence
    print(f"[validate] best internal score={best_score:.8f}")
    final_arr, final_score = _safe_validate(best_a)

    if final_arr is None or not np.isfinite(final_score):
        # Fallback: validate the initial best
        final_arr, final_score = _safe_validate(best_init)
        print(f"[validate] fallback score={final_score:.8f}")

    if final_arr is None or not np.isfinite(final_score):
        # Ultimate fallback: uniform
        final_arr = np.ones(256, dtype=np.float64) / 256.0
        final_score = 2.0
        print("[validate] using uniform fallback, score=2.0")

    print(f"[done] final score={final_score:.8f}, n={final_arr.size}, time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in final_arr]
