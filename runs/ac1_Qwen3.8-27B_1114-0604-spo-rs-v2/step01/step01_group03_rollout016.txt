```python
def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    v = v.copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - s))[0][-1]
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def _project_simplex(v, s_target=1.0):
    """Project v onto {x : x >= 0, sum(x) = s_target} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - s_target))[0]
    if len(rho) == 0:
        return np.full(n, s_target / n)
    theta = (cv[rho[-1]] - s_target) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        n = len(seq)
        s = sum(seq)
        if s < 0.01:
            return float('inf')
        b = np.convolve(seq, seq)
        return float(2.0 * n * max(b) / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)

    if remaining > 30:
        print(f"\nLocal refinement, time={remaining:.0f}s")
        t_alloc = min(120.0, remaining - 10)
        a_ref, sc_ref = _local_refine(best_a, fft_size, N, t_alloc, rng)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"  Refined: {best_score:.6f}")

    # Phase 5: Coordinate refinement
    remaining = hard_deadline - time.monotonic()
    if remaining > 15:
        t_alloc = min(60.0, remaining - 5)
        a_coord, sc_coord = _coordinate_refine(best_a, fft_size, N, t_alloc, rng)
        if sc_coord < best_score:
            best_score = sc_coord
            best_a = a_coord.copy()
            print(f"  Coordinate refined: {best_score:.6f}")

    # Final: evaluate with exact function
    print(f"\nFinal internal score: {best_score:.6f}")
    seq = [float(x) for x in best_a]
    exact = _exact_score(seq)
    print(f"Exact score: {exact:.6f}")

    # If exact is worse (due to numerical differences), try a few alternatives
    if exact > best_score + 0.01:
        # Try slightly different normalizations
        for scale in [0.9, 0.95, 1.0, 1.05, 1.1]:
            seq_try = [float(x * scale) for x in best_a]
            sc_try = _exact_score(seq_try)
            if sc_try < exact:
                exact = sc_try
                seq = seq_try
        print(f"Adjusted exact score: {exact:.6f}")

    return seq
```
