```python
"""
Search for non-negative step-function coefficients minimizing

    evaluate_sequence(a) = 2 * len(a) * max(conv(a, a)) / sum(a)^2

The implementation uses:
  - FFT-based exact scoring and gradient computation.
  - Multiplicative mirror descent on the probability simplex, with an
    adaptive smoothed-max / active-peak weight vector and exact line search.
  - Random perturbations for escaping plateaus.
  - Random block scaling and deterministic block sweeps for local polishing.
  - A staged length schedule: small basin search, main length (usually 4096),
    optional larger length (usually 8192), and final official-evaluator checks.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _sanitize_vec(x):
    try:
        arr = np.asarray(x, dtype=float).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return arr / s


def _conv_unit(a, L):
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=float)
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)
    out = out[:2 * n - 1]
    out = np.where(out > 0.0, out, 0.0)
    return out


def _score_unit(a, L):
    n = int(a.size)
    if n <= 0:
        return float("inf")
    c = _conv_unit(a, L)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def _front_conv(w, p, L):
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    out = out[:p.size]
    out = np.where(out > 0.0, out, 0.0)
    return out


def _weights_from_c(c, beta, n):
    m = int(c.size)
    if m == 0:
        return np.array([0.0], dtype=float)
    d = c * float(n)
    d = np.where(np.isfinite(d), d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax):
        return np.full(m, 1.0 / m, dtype=float)
    beta = max(0.0, float(beta))
    if beta < 1e-8:
        return np.full(m, 1.0 / m, dtype=float)

    e = np.exp(beta * (d - dmax))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        e = np.zeros(m, dtype=float)
        e[int(np.argmax(d))] = 1.0
        s = 1.0
    w = e / s

    if beta > 60.0 and m > 8:
        k = int(max(4, min(128, n // 8)))
        if k < m:
            thr = float(np.partition(d, m - k)[-1])
            active = (d >= thr).astype(float)
            asum = float(np.sum(active))
            if asum > 0.0:
                active = active / asum
                mix = (beta - 60.0) / 400.0
                if mix > 0.0:
                    if mix > 1.0:
                        mix = 1.0
                    w = (1.0 - mix) * w + mix * active
    return w


def _beta_start(n, a, b, eps=0.25):
    if n <= 0:
        return None
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    lp = -float(a) * np.log(t + float(eps)) - float(b) * np.log(1.0 - t + float(eps))
    lp = np.where(np.isfinite(lp), lp, 0.0)
    lp = lp - float(np.max(lp))
    p = np.exp(lp)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    p = p + 1e-12
    return _sanitize_vec(p)


def _smooth_gaussian(p, samples=1.0):
    p = _sanitize_vec(p)
    if p is None:
        return None
    s = float(samples)
    if s <= 0.0 or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3.0 * s)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / s) ** 2)
    kern = kern / kern.sum()
    out = np.convolve(p, kern, mode="same")
    return _sanitize_vec(out)


def _resample(arr, m):
    a = _sanitize_vec(arr)
    if a is None:
        return None
    m = int(m)
    if m <= 0:
        return None
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n, dtype=float) + 0.5) / float(n)
    x_new = (np.arange(m, dtype=float) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize_vec(out)


def _two_block_start(n, frac, ratio, samples=0.75):
    p = np.ones(n, dtype=float)
    idx = int(round(float(frac) * n))
    if idx <= 0:
        idx = 1
    if idx >= n:
        idx = n - 1
    p[idx:] = float(ratio)
    p = _smooth_gaussian(p, samples)
    return _sanitize_vec(p)


def _initial_sequence(n, rng, mode=None):
    if n <= 0:
        return None
    if mode is None:
        mode = int(rng.integers(0, 6))
    t = (np.arange(n, dtype=float) + 0.5) / float(n)

    if mode == 0:
        z = np.ones(n, dtype=float)
        for kk in range(1, 9):
            amp = float(rng.normal(0.0, 1.0 / (0.5 + kk)))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * kk * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        a = float(rng.uniform(-0.6, 0.6))
        b = float(rng.uniform(-0.6, 0.6))
        p = _beta_start(n, a, b, 0.25)

    elif mode == 2:
        K = int(rng.integers(1, 5))
        p = np.zeros(n, dtype=float)
        centers = np.sort(rng.uniform(0.0, 1.0, K))
        sigmas = rng.uniform(0.02, 0.30, K)
        weights = rng.uniform(0.5, 2.0, K)
        for ii in range(K):
            c = float(centers[ii])
            s = max(1e-4, float(sigmas[ii]))
            w = float(weights[ii])
            p = p + w * np.exp(-((t - c) / s) ** 2)

    elif mode == 3:
        K = int(rng.integers(3, 9))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        K = int(rng.integers(2, 8))
        edges = np.sort(rng.uniform(0.0, 1.0, K - 1))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.zeros(n, dtype=float)
        bounds = [0.0]
        bounds.extend([float(x) for x in edges])
        bounds.append(1.0)
        for ii in range(K):
            s = max(0, int(round(bounds[ii] * n)))
            e = min(n, max(s + 1, int(round(bounds[ii + 1] * n))))
            if e > s:
                p[s:e] = float(heights[ii])

    else:
        c = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(5.0, 20.0))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50.0, 50.0)))
        p = p * np.exp(-((t - c) / 0.25) ** 2)

    if p is None:
        return None
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    p = p + 1e-12
    return _sanitize_vec(p)


def _cand_key(item):
    return item[0]


def _prune_candidates(cands, keep):
    if len(cands) <= keep:
        return cands
    cands.sort(key=_cand_key)
    cands[:] = cands[:keep]
    return cands


def _add_candidate(cands, arr, score, keep=12):
    a = _sanitize_vec(arr)
    if a is None:
        return
    if score is None or not np.isfinite(score):
        L = _next_pow2(2 * a.size)
        score = _score_unit(a, L)
    if not np.isfinite(score):
        return
    cands.append((float(score), int(a.size), a.copy()))
    _prune_candidates(cands, keep)


def _get_parent():
    try:
        raw = height_sequence_1
    except Exception:
        return None
    return _sanitize_vec(raw)


def _official_score(lst):
    try:
        val = float(evaluate_sequence(lst))
        return val
    except Exception:
        return float("inf")


def _structured_starts(n, rng, parent, cands, keep=12):
    if parent is not None:
        base = _resample(parent, n)
        if base is not None:
            _add_candidate(cands, base, None, keep)
            sm = _smooth_gaussian(base, 0.75)
            if sm is not None:
                _add_candidate(cands, sm, None, keep)
            for pert in (0.02, 0.05, 0.10):
                p_try = base * np.exp(rng.normal(0.0, float(pert), n))
                _add_candidate(cands, p_try, None, keep)

    for frac in (0.30, 0.40, 0.50, 0.60, 0.70):
        for ratio in (0.2, 0.5, 1.0, 2.0, 4.0):
            p = _two_block_start(n, frac, ratio, 0.75)
            _add_candidate(cands, p, None, keep)

    exps = (-0.5, -0.25, 0.0, 0.25, 0.5)
    for a in exps:
        for b in exps:
            p = _beta_start(n, a, b, 0.25)
            _add_candidate(cands, p, None, keep)

    for mode in (0, 1, 2, 3, 4, 5):
        p = _initial_sequence(n, rng, mode)
        _add_candidate(cands, p, None, keep)

    _prune_candidates(cands, keep)


def _perturb_unit(p, rng, floor, n, L, base_score, attempts=6):
    best_p = p.copy()
    best_sc = _score_unit(p, L)
    if not np.isfinite(best_sc):
        best_sc = float("inf")

    stds = (0.03, 0.07, 0.15, 0.30, 0.50, 0.80)
    attempts = int(max(1, min(len(stds), attempts)))
    for ii in range(attempts):
        std = float(stds[ii])
        q = p * np.exp(rng.normal(0.0, std, n))
        q = np.maximum(q, floor)
        s = float(np.sum(q))
        if s <= 0.0 or not np.isfinite(s):
            continue
        q = q / s
        sc = _score_unit(q, L)
        if np.isfinite(sc) and sc < best_sc - 1e-14:
            best_p = q
            best_sc = sc
    return best_p, best_sc


def _run_mmd(p0, rng, time_limit, beta_start=30.0, beta_max=1500.0,
             verbose=False, name="mmd"):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))

    p = _sanitize_vec(p0)
    if p is None:
        return None, float("inf")

    n = int(p.size)
    L = _next_pow2(2 * n)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_s = _score_unit(p, L)
    if not np.isfinite(best_s):
        best_s = float("inf")

    if time_limit <= 0.0:
        return best, best_s

    eta = 1.0
    beta = float(beta_start)
    it = 0
    since_best = 0
    stall = 0
    last_print = -10000
    cur_s = best_s

    while time.monotonic() < deadline - 0.5:
        c = _conv_unit(p, L)
        if c.size == 0:
            break
        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            p = best.copy()
            c = _conv_unit(p, L)
            cmax = float(np.max(c))
        cur_s = float(2.0 * n * cmax)

        improved = np.isfinite(cur_s) and (cur_s < best_s - 1e-12)
        if improved:
            best = p.copy()
            best_s = cur_s
            since_best = 0
        else:
            since_best += 1

        w = _weights_from_c(c, beta, n)
        d = c * float(n)
        cur_smooth = float(np.dot(w, d))

        h = _front_conv(w, p, L)
        h = np.where(np.isfinite(h), h, 0.0)
        hmax = float(np.max(h)) if h.size else 0.0
        if not np.isfinite(hmax) or hmax <= 1e-18:
            p_try, sc_try = _perturb_unit(p, rng, floor, n, L, best_s, 4)
            p = p_try
            cur_s = sc_try
            it += 1
            if verbose and (it - last_print >= 200):
                print("mmd %s n=%d it=%d best=%.6f cur=%.6f beta=%.1f"
                      % (name, n, it, best_s, cur_s, beta))
                last_print = it
            continue
        h = h / hmax

        alphas = [0.05, 0.10, 0.25, 0.50, 1.00, 2.00, 4.00]
        for mult in (0.25, 0.50, 1.00, 2.00, 4.00):
            a = max(1e-3, eta * float(mult))
            alphas.append(a)

        uniq = []
        for a in alphas:
            bad = False
            for b in uniq:
                tol = 0.25 * max(1e-6, abs(float(a)), abs(float(b)))
                if abs(float(a) - float(b)) <= tol:
                    bad = True
                    break
            if not bad:
                uniq.append(float(a))
        uniq.sort()
        if len(uniq) > 8:
            uniq = uniq[:8]

        best_exact_q = None
        best_exact_s = cur_s
        best_exact_sm = cur_smooth
        best_exact_alpha = 0.0

        best_sur_q = None
        best_sur_s = cur_s
        best_sur_sm = cur_smooth
        best_sur_alpha = 0.0

        limit_exact = max(cur_s * 1.08, best_s * 1.25)

        for a in uniq:
            dec = np.exp(-float(a) * h)
            q = p * dec
            q = np.maximum(q, floor)
            s = float(np.sum(q))
            if s <= 0.0 or not np.isfinite(s):
                continue
            q = q / s

            cq = _conv_unit(q, L)
            cqmax = float(np.max(cq)) if cq.size else 0.0
            if not np.isfinite(cqmax):
                continue
            sc = float(2.0 * n * cqmax)
            dq = cq * float(n)
            sm = float(np.dot(w, dq))

            if sc < best_exact_s - 1e-12:
                best_exact_s = sc
                best_exact_q = q
                best_exact_sm = sm
                best_exact_alpha = a

            if sc <= limit_exact + 1e-12 and sm < best_sur_sm - 1e-10:
                best_sur_s = sc
                best_sur_q = q
                best_sur_sm = sm
                best_sur_alpha = a

        accepted = False
        if best_exact_q is not None and best_exact_s < cur_s - 1e-12:
            p = best_exact_q
            cur_s = best_exact_s
            accepted = True
            if cur_s < best_s - 1e-12:
                best = p.copy()
                best_s = cur_s
            eta = max(0.05, min(10.0, max(eta * 0.9, best_exact_alpha * 0.8)))
        elif best_sur_q is not None and best_sur_sm < cur_smooth - 1e-10:
            p = best_sur_q
            cur_s = best_sur_s
            accepted = True
            if cur_s < best_s - 1e-12:
                best = p.copy()
                best_s = cur_s
            eta = max(0.05, min(10.0, best_sur_alpha * 0.8))
        else:
            eta = max(0.05, eta * 0.55)
            stall += 1
            if stall > 12:
                p_try, sc_try = _perturb_unit(p, rng, floor, n, L, best_s, 4)
                if sc_try <= max(best_s * 1.05, cur_s * 1.02):
                    p = p_try
                    cur_s = sc_try
                else:
                    p = best.copy()
                    cur_s = best_s
                stall = 0
                eta = 1.0

        if accepted:
            stall = 0
            if since_best <= 5:
                eta = min(10.0, eta * 1.15)

        if since_best < 5:
            beta = min(float(beta_max), beta * 1.10)
        elif since_best > 80:
            if beta >= 0.9 * float(beta_max):
                beta = max(10.0, beta * 0.5)
            else:
                beta = min(float(beta_max), beta * 1.25)
        else:
            beta = min(float(beta_max), beta * 1.02)

        if cur_s > best_s * 1.25 and (rng.random() < 0.25):
            p = best.copy()
            cur_s = best_s

        it += 1
        if verbose and (it - last_print >= 200):
            print("mmd %s n=%d it=%d best=%.6f cur=%.6f beta=%.1f eta=%.3f stall=%d"
                  % (name, n, it, best_s, cur_s, beta, eta, stall))
            last_print = it

    if np.isfinite(cur_s) and cur_s < best_s:
        best = p.copy()
        best_s = cur_s
    return best, best_s


def _local_block_search(p0, rng, time_limit, verbose=False):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))
    p = _sanitize_vec(p0)
    if p is None:
        return None, float("inf")

    n = int(p.size)
    L = _next_pow2(2 * n)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_s = _score_unit(p, L)
    cur_s = best_s
    stall = 0
    factors = (0.0, 0.25, 0.50, 0.75, 1.30, 1.70, 2.50)

    size2 = max(2, n // 200)
    size3 = max(size2 + 1, min(n, n // 20))
    size4 = max(size3 + 1, min(n, n // 5))

    while time.monotonic() < deadline - 1.0:
        r = float(rng.random())
        if r < 0.12:
            start = int(rng.integers(n))
            size = 1
        elif r < 0.75:
            u = float(rng.random())
            if u < 0.5:
                size = int(rng.integers(1, size2 + 1))
            elif u < 0.9:
                size = int(rng.integers(size2, size3 + 1))
            else:
                size = int(rng.integers(size3, size4 + 1))
            size = max(1, min(n, size))
            start = int(rng.integers(n - size + 1))
        else:
            c = _conv_unit(p, L)
            m = int(c.size)
            if m <= 0:
                start = int(rng.integers(n))
                size = 1
            else:
                top_count = min(m, max(1, min(32, m // 50 + 1)))
                idx_top = np.argpartition(c, m - top_count)[-top_count:]
                k = int(idx_top[int(rng.integers(len(idx_top)))])
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if hi < lo:
                    start = int(rng.integers(n))
                    size = 1
                else:
                    rev = p[k - hi:k - lo + 1][::-1]
                    seg = p[lo:hi + 1] * rev
                    wsum = float(np.sum(seg))
                    pos = np.nonzero(seg > 0.0)[0]
                    if pos.size == 0 or wsum <= 0.0:
                        i = int(rng.integers(lo, hi + 1))
                    else:
                        probs = seg[pos] / wsum
                        ps = float(np.sum(probs))
                        if ps > 0.0 and np.isfinite(ps):
                            probs = probs / ps
                        i = lo + int(pos[int(rng.choice(pos.size, p=probs))])
                    maxsize = max(2, min(n, n // 500 + 3))
                    size = int(rng.integers(1, maxsize + 1))
                    start = max(0, min(n - size, i - size // 2))

        best_q = None
        best_sc = cur_s
        for f in factors:
            q = p.copy()
            if f == 0.0:
                q[start:start + size] = floor
            else:
                q[start:start + size] *= float(f)
            q = np.maximum(q, floor)
            s = float(np.sum(q))
            if s <= 0.0 or not np.isfinite(s):
                continue
            q = q / s
            sc = _score_unit(q, L)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_q = q

        if best_q is not None:
            p = best_q
            cur_s = best_sc
            if cur_s < best_s - 1e-12:
                best = p.copy()
                best_s = cur_s
                stall = 0
                if verbose:
                    print("local %d best=%.6f" % (n, best_s))
        else:
            stall += 1
            if stall > 40:
                p_try, sc_try = _perturb_unit(p, rng, floor, n, L, best_s, 4)
                if sc_try <= max(best_s * 1.05, cur_s * 1.01):
                    p = p_try
                    cur_s = sc_try
                else:
                    p = best.copy()
                    cur_s = best_s
                stall = 0
            if stall > 300:
                p = best.copy()
                cur_s = best_s
                stall = 0

    if np.isfinite(cur_s) and cur_s < best_s:
        best = p.copy()
        best_s = cur_s
    return best, best_s


def _sweep_blocks_once(p, block_size, L, factors, deadline=None):
    cur_s = _score_unit(p, L)
    n = int(p.size)
    floor = 1e-18 / float(n)
    block_size = max(1, int(block_size))

    for start in range(0, n, block_size):
        if deadline is not None and time.monotonic() > deadline:
            break
        end = min(n, start + block_size)
        best_q = None
        best_sc = cur_s
        for f in factors:
            if abs(float(f) - 1.0) < 1e-12:
                continue
            q = p.copy()
            if float(f) == 0.0:
                q[start:end] = floor
            else:
                q[start:end] *= float(f)
            q = np.maximum(q, floor)
            s = float(np.sum(q))
            if s <= 0.0 or not np.isfinite(s):
                continue
            q = q / s
            sc = _score_unit(q, L)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_q = q
        if best_q is not None:
            p = best_q
            cur_s = best_sc
    return p, cur_s


def _local_sweeps(p0, rng, time_limit, verbose=False):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))
    p = _sanitize_vec(p0)
    if p is None:
        return None, float("inf")
    n = int(p.size)
    L = _next_pow2(2 * n)
    best_s = _score_unit(p, L)

    sizes = []
    for bs in (16, 32, 64, 128, 256, 512):
        if bs < n:
            sizes.append(bs)
    sizes = sorted(set(sizes))
    if len(sizes) > 3:
        sizes = sizes[-3:]

    factors = (0.55, 0.80, 1.25, 1.65)
    last_s = best_s
    for bs in sizes:
        if time.monotonic() >= deadline - 10.0:
            break
        p, sc = _sweep_blocks_once(p, bs, L, factors, deadline - 1.0)
        if np.isfinite(sc) and sc < last_s:
            last_s = sc
            if verbose:
                print("sweep bs=%d best=%.6f" % (bs, last_s))

    if np.isfinite(best_s) and last_s < best_s:
        best_s = last_s
    return p, best_s


def propose_candidate(seed=12345, budget_s=2400):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60.0:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget - 45.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    parent = _get_parent()
    cands = []
    best_arr = None
    best_score = float("inf")

    # Small basin search.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 260.0:
        small_n = 1024
        small_cands = []
        _structured_starts(small_n, rng, parent, small_cands, keep=10)
        small_cands.sort(key=_cand_key)
        for item in small_cands[:4]:
            ln = deadline - time.monotonic()
            if ln < 45.0:
                break
            talloc = min(25.0, max(8.0, ln / 3.0))
            p, sc = _run_mmd(item[2], rng, talloc, 30.0, 500.0, False, "small")
            if p is not None and np.isfinite(sc):
                _add_candidate(cands, p, sc, 12)
                if sc < best_score:
                    best_arr = p.copy()
                    best_score = sc

    # Main length candidate generation.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 1200.0:
        main_n = 4096
    elif rem > 600.0:
        main_n = 2048
    else:
        main_n = 1024

    main_cands = []

    if parent is not None:
        base = _resample(parent, main_n)
        if base is not None:
            _add_candidate(main_cands, base, None, 12)
            sm = _smooth_gaussian(base, 0.75)
            if sm is not None:
                _add_candidate(main_cands, sm, None, 12)
            for pert in (0.03, 0.08):
                p_try = base * np.exp(rng.normal(0.0, float(pert), main_n))
                _add_candidate(main_cands, p_try, None, 12)

    if small_cands:
        small_cands.sort(key=_cand_key)
        for item in small_cands[:3]:
            sp = _resample(item[2], main_n)
            if sp is None:
                continue
            sp = _smooth_gaussian(sp, 0.50)
            sp = _sanitize_vec(sp * np.exp(rng.normal(0.0, 0.03, main_n)))
            _add_candidate(main_cands, sp, None, 12)

    rem = max(0.0, deadline - time.monotonic())
    if rem > 120.0:
        _structured_starts(main_n, rng, parent, main_cands, keep=12)

    if len(main_cands) == 0:
        one = np.ones(main_n, dtype=float) / float(main_n)
        main_cands.append((2.0, main_n, one))

    main_cands.sort(key=_cand_key)
    for item in main_cands[:3]:
        if np.isfinite(item[0]) and item[0] < best_score:
            best_score = item[0]
            best_arr = item[2].copy()

    if best_arr is None:
        best_arr = main_cands[0][2].copy()
        best_score = main_cands[0][0]

    # Staged budget allocation. This preserves time for a larger final
    # refinement stage whenever the runtime is large enough.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 1800.0:
        big_budget = min(900.0, 0.38 * rem)
        mid_budget = min(350.0, 0.16 * rem)
    elif rem > 1000.0:
        big_budget = min(450.0, 0.30 * rem)
        mid_budget = min(200.0, 0.15 * rem)
    else:
        big_budget = 0.0
        mid_budget = min(250.0, 0.20 * rem)

    main_budget = max(60.0, rem - big_budget - mid_budget)

    if main_budget > 30.0 and main_cands:
        t1 = min(main_budget * 0.62, deadline - big_budget - mid_budget - 20.0)
        t1 = max(20.0, t1)
        p, sc = _run_mmd(main_cands[0][2], rng, t1, 30.0, 1500.0, True, "main1")
        if p is not None and np.isfinite(sc):
            _add_candidate(cands, p, sc, 12)
            if sc < best_score:
                best_arr = p.copy()
                best_score = sc

    rem = max(0.0, deadline - time.monotonic())
    usable_main2 = rem - max(0.0, big_budget) - max(0.0, mid_budget) - 20.0
    if usable_main2 > 80.0 and len(main_cands) > 1:
        t2 = min(main_budget * 0.38, usable_main2)
        t2 = max(20.0, t2)
        p, sc = _run_mmd(main_cands[1][2], rng, t2, 30.0, 1500.0, True, "main2")
        if p is not None and np.isfinite(sc):
            _add_candidate(cands, p, sc, 12)
            if sc < best_score:
                best_arr = p.copy()
                best_score = sc

    # Mid-range polishing at the main length.
    rem = max(0.0, deadline - time.monotonic())
    if rem > max(120.0, big_budget + 60.0) and best_arr is not None:
        tloc = min(160.0, 0.55 * mid_budget, max(0.0, rem - big_budget - 80.0))
        if tloc > 25.0:
            p, sc = _local_block_search(best_arr, rng, tloc, True)
            if p is not None and np.isfinite(sc):
                _add_candidate(cands, p, sc, 12)
                if sc < best_score:
                    best_arr = p.copy()
                    best_score = sc

    rem = max(0.0, deadline - time.monotonic())
    if rem > max(120.0, big_budget + 60.0) and best_arr is not None:
        tmmd = min(220.0, 0.45 * mid_budget, max(0.0, rem - big_budget - 50.0))
        if tmmd > 25.0:
            p, sc = _run_mmd(best_arr, rng, tmmd, 50.0, 2500.0, True, "mainpol")
            if p is not None and np.isfinite(sc):
                _add_candidate(cands, p, sc, 12)
                if sc < best_score:
                    best_arr = p.copy()
                    best_score = sc

    # Optional larger-length stage.
    rem = max(0.0, deadline - time.monotonic())
    if rem > 500.0 and best_arr is not None:
        big_n = 8192
        if best_arr.size >= big_n:
            big_n = max(8192, int(best_arr.size))
            if big_n > 12288:
                big_n = 12288

        big = _resample(best_arr, big_n)
        if big is not None:
            big = _smooth_gaussian(big, 1.0)
            if big is not None:
                big = _sanitize_vec(big * np.exp(rng.normal(0.0, 0.02, big_n)))

            rem = max(0.0, deadline - time.monotonic())
            if rem > 350.0 and big is not None:
                tb1 = min(800.0, 0.60 * rem)
                p, sc = _run_mmd(big, rng, tb1, 50.0, 3000.0, True, "big")
                if p is not None and np.isfinite(sc):
                    _add_candidate(cands, p, sc, 12)
                    if sc < best_score:
                        best_arr = p.copy()
                        best_score = sc

            rem = max(0.0, deadline - time.monotonic())
            if rem > 220.0 and best_arr is not None:
                tb2 = min(130.0, 0.20 * rem)
                p, sc = _local_block_search(best_arr, rng, tb2, True)
                if p is not None and np.isfinite(sc):
                    _add_candidate(cands, p, sc, 12)
                    if sc < best_score:
                        best_arr = p.copy()
                        best_score = sc

            rem = max(0.0, deadline - time.monotonic())
            if rem > 220.0 and best_arr is not None:
                tb3 = min(200.0, 0.20 * rem)
                p, sc = _run_mmd(best_arr, rng, tb3, 80.0, 4000.0, True, "bigpol")
                if p is not None and np.isfinite(sc):
                    _add_candidate(cands, p, sc, 12)
                    if sc < best_score:
                        best_arr = p.copy()
                        best_score = sc

    # Final validation with the official evaluator, if available.
    final_cands = []
    if best_arr is not None:
        final_cands.append(best_arr)

    cands.sort(key=_cand_key)
    for item in cands[:6]:
        if item[2] is not None:
            final_cands.append(item[2])
    if parent is not None:
        final_cands.append(parent)

    selected = best_arr
    selected_score = best_score
    if selected is None:
        selected = np.ones(64, dtype=float)

    max_evals = 3 if int(np.asarray(selected).size) > 6000 else 6
    evaluated = 0
    for arr in final_cands:
        if evaluated >= max_evals:
            break
        if deadline - time.monotonic() < 5.0:
            break
        a = _sanitize_vec(arr)
        if a is None or a.size == 0:
            continue
        lst = [float(x) for x in a]
        sc = _official_score(lst)
        if np.isfinite(sc) and sc < selected_score:
            selected_score = sc
            selected = a
        evaluated += 1

    out = np.asarray(selected, dtype=float).reshape(-1)
    if out.size == 0:
        out = np.ones(64, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(np.sum(out))
    if s <= 0.0 or not np.isfinite(s):
        out = np.ones(64, dtype=float)
    else:
        out = out / s
    return [float(x) for x in out]
```
