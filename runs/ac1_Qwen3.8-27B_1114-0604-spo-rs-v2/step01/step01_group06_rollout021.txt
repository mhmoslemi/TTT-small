```python
    """
    At each iteration:
    1. Compute c = a*a
    2. Find active set: positions where c[k] > max(c) - tau_thresh * sum(c)/(2n-1)
    3. Compute gradient at active positions
    4. Take a gradient step (with line search)
    """
    n = a.size
    a = a.copy()
    a = a / a.sum()
    
    for it in range(max_iter):
        c = _fft_conv(a)
        F = c.max()
        mean_c = c.sum() / (2*n-1)
        
        # Active set: positions close to the max
        active = c > F - 0.01 * F  # within 1% of max
        p = np.zeros_like(c)
        p[active] = 1.0 / active.sum()
        
        # Gradient: dF/da[i] = 2 * sum_k p[k] * a[k-i]
        # Using FFT:
        N = 1 << (3*n-2).bit_length()
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        
        # Project gradient (remove mean component for the normalization)
        g = a * (h - np.dot(h, a))
        
        # Line search

                a_new = a_new / a_new.sum()
```

```python
import numpy as np
import time
from scipy.optimize import minimize

"""
Algorithm for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences:

1. Multi-strategy initialization: multi-level block sequences (2-6 levels),
   smooth profiles, resampled previous best, random Fourier combinations.
2. Adam optimizer in log-space (softmax parameterization) with annealed
   smooth-max temperature (start smooth for good gradient, end sharp for accuracy).
3. Active-set gradient polish for final refinement.
4. Multiple sequence lengths tried (8192 primary, 16384 if time allows).
5. Coordinate descent for very local refinement.

Key improvements over previous:
- Adam instead of L-BFGS-B (more robust for non-convex landscape)
- Annealed temperature (tau decreases over iterations)
- Multi-level block starting points (not just 2-level)
- Coordinate descent polish
- Better learning rate scheduling
"""

# Global state
_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-3
_G_BOUND = 25.0


def _init_n(n, tau=1e-3):
    global _G_N, _G_NFFT, _G_TAU
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(np.sum(a))
    if s < 1e-12 or not np.isfinite(s):
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1 (faster, no re-normalization)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    c = _fft_conv(a)
    return float(2.0 * a.size * np.max(c))


def _compute_grad(a, tau):
    """
    Compute gradient of smooth max objective w.r.t. a (before softmax chain rule).
    Returns (value, grad_wrt_a) where grad is dF/da with F = tau*logsumexp(c/tau).
    """
    n = a.size
    N = _G_NFFT
    c = _fft_conv(a)
    m = float(np.max(c))
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return m, np.zeros(n)
    p = e / se
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    g = 2.0 * a * h  # dF/da (before normalization constraint)
    val = m + tau * np.log(se)
    return val, g


def _obj_and_grad_logspace(z, tau):
    """
    Objective and gradient in z-space (log-space), where a = softmax(z).
    We minimize F(a) = tau * logsumexp(c/tau) with a = softmax(z), sum(a)=1.
    Since sum(a)=1 automatically, the score is 2*n*F.
    """
    n = _G_N
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))
    z = np.clip(z, -_G_BOUND, _G_BOUND)
    a = _softmax(z)
    val, grad_a = _compute_grad(a, tau)
    # Chain rule: dz->da gives grad_z = a * (grad_a - sum(a*grad_a))
    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
    # Small penalty for gauge invariance of softmax
    pen = 1e-12
    val += 0.5 * pen * float(np.dot(z, z))
    grad_z += pen * z
    if not np.isfinite(val):
        val = 1e12
        grad_z = np.zeros(n)
    return float(val), grad_z


class _Adam:
    """Adam optimizer with learning rate decay."""

    def __init__(self, n, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.n = n
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0

    def step(self, z, val, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad * grad)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        z_new = z - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return np.clip(z_new, -_G_BOUND, _G_BOUND)


def _run_adam(z0, deadline, tau_start=0.1, tau_end=1e-5, lr_start=0.05,
              lr_end=0.001, min_interval=2.0):
    """
    Run Adam optimization with annealed temperature.
    Returns best (score, z) found during the run.
    """
    n = _G_N
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z = np.clip(z - np.mean(z), -_G_BOUND, _G_BOUND)
    adam = _Adam(n, lr=lr_start)
    
    best_score = _score_normalized(_softmax(z))
    best_z = z.copy()
    
    n_iter = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        # Anneal parameters based on fraction of time elapsed
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1.0, deadline - t0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        adam.lr = lr

        val, grad = _obj_and_grad_logspace(z, tau)
        z = adam.step(z, val, grad)
        n_iter += 1

        # Periodic evaluation
        if n_iter % 200 == 1:
            a = _softmax(z)
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
                if n_iter % 2000 == 1:
                    elapsed2 = time.monotonic() - t0
                    print(f"    adam it={n_iter} score={2*n*sc/(1.0):.9f} "
                          f"(norm={sc:.9f}) tau={tau:.2e} lr={lr:.2e} "
                          f"t={elapsed2:.0f}s")
            # Reset if score explodes
            if sc > best_score * 3.0 + 1.0:
                z = best_z + 0.01 * np.random.randn(n)
                adam.m = np.zeros(n)
                adam.v = np.zeros(n)

    return best_score, best_z


def _active_set_polish(a, deadline, max_rounds=2000):
    """
    Polish using active-set gradient method.
    At each step, identify positions where c is near the max,
    compute the effective gradient, and take a step.
    """
    n = _G_N
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a
    a /= s

    best = a.copy()
    best_sc = _score_normalized(a)
    no_improve = 0

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        F = float(np.max(c))

        if F < best_sc - 1e-14:
            best = a.copy()
            best_sc = F
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        # Broad active set: positions within a small fraction of max
        threshold = F - 0.005 * F
        active = c >= threshold
        if not active.any():
            active[np.argmax(c)] = True
        p = np.zeros_like(c)
        p[active] = 1.0 / float(active.sum())

        N = _G_NFFT
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

        # Gradient of F w.r.t. a, projected to maintain sum=1
        g = a * (2.0 * h - 2.0 * float(np.dot(a, h)))

        # Line search
        accepted = False
        for eta in (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1):
            a_new = a - eta * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new /= sn
            F_new = float(np.max(_fft_conv(a_new)))
            if F_new < F - 1e-15:
                a = a_new
                accepted = True
                break

        if not accepted:
            # Small random perturbation to escape
            a_pert = a + 0.002 * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            F_pert = float(np.max(_fft_conv(a_pert)))
            if F_pert < F:
                a = a_pert

    return best


def _coordinate_descent(a, deadline, max_rounds=5000):
    """
    Coordinate descent: optimize small groups of coordinates at a time.
    """
    n = _G_N
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a
    a /= s

    best_sc = _score_normalized(a)
    rng = np.random.default_rng(12345)
    no_improve = 0
    group_size = max(4, n // 2048)

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        F = float(np.max(c))
        if F < best_sc - 1e-14:
            best = a.copy()
            best_sc = F
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:

        # Pick a random group of coordinates

def _make_multilevel_block(n, levels, ratios, rng):
    """Create a multi-level block sequence."""
    a = np.empty(n, dtype=np.float64)
    boundaries = np.cumsum(levels) / np.sum(levels) * n
    boundaries = np.round(boundaries).astype(int)
    boundaries = np.minimum(boundaries, n)
    start = 0
    for i in range(len(ratios)):
        end = boundaries[i] if i < len(boundaries) else n
        a[start:end] = ratios[i]
        start = end
    a[n:] = 0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def _make_smooth_profile(n, profile_type, params):
    """Create a smooth profile sequence."""
    x = np.linspace(0, 1, n)
    if profile_type == "linear":
        a = 1.0 + params * (2 * x - 1)
    elif profile_type == "quadratic":
        a = 1.0 + params * (3 * (x - 0.5) ** 2 - 0.25)
    elif profile_type == "sigmoid":
        w = params.get("w", 0.1)
        a = 1.0 + params.get("amp", 1.0) * (1.0 / (1.0 + np.exp(-(x - 0.5) / w)) - 0.5)
    elif profile_type == "cosine":
        a = 1.0 + params * np.cos(2 * np.pi * x)
    elif profile_type == "raised_cos":
        a = params + (1.0 - params) * 0.5 * (1 + np.cos(2 * np.pi * x))
    elif profile_type == "two_sigmoid":
        w = params.get("w", 0.05)
        amp = params.get("amp", 0.5)
        a = 1.0 + amp * (1.0 / (1.0 + np.exp(-(x - 0.3) / w))
                         - 1.0 / (1.0 + np.exp(-(x - 0.7) / w)) - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s > 0:
        a /= s
    return a

    unique = []
    for sc, z in starts:
        if len(unique) >= 30:
            break
        is_dup = False
        for _, z2 in unique:
            if np.linalg.norm(z - z2) < 0.5:
                is_dup = True
                break
        if not is_dup:
            unique.append((sc, z))
    return unique


def _optimize_single_n(n, rng, deadline):
    """Run the full optimization for a given n. Returns best (score, a)."""
    _init_n(n)
    print(f"  Starting optimization for n={n}")

    starts = _generate_starts(n, rng)
    if not starts:
        a = np.ones(n) / n
        return _score(a), a

    best_score = np.inf
    best_a = None

    # Sort starts by initial quality
    starts.sort(key=lambda x: x[0])

    # Phase 1: Adam optimization from top starts
    n_starts_to_try = min(len(starts), 12)
    time_per_start = max(10.0, (deadline - time.monotonic() - 120) / max(1, n_starts_to_try))

    for i in range(n_starts_to_try):
        if time.monotonic() > deadline - 120:
            break
        sc0, z0 = starts[i]
        start_time = time.monotonic()
        start_deadline = start_time + min(time_per_start, deadline - time.monotonic() - 90)

        if start_deadline < start_time + 10:
            continue

        print(f"  Start {i}: initial_score={2*n*sc0:.9f}, budget={start_deadline-start_time:.0f}s")

        # Run Adam with different hyperparameter settings
        settings = [
            {"tau_start": 0.1, "tau_end": 1e-5, "lr_start": 0.05, "lr_end": 0.001},
            {"tau_start": 0.05, "tau_end": 1e-5, "lr_start": 0.03, "lr_end": 0.0005},
            {"tau_start": 0.2, "tau_end": 1e-4, "lr_start": 0.08, "lr_end": 0.002},
        ]
        # Use only 1-2 settings per start depending on time
        n_settings = min(len(settings), max(1, int((start_deadline - start_time) / 30)))

        for si in range(n_settings):
            if time.monotonic() > start_deadline:
                break
            s = settings[si]
            sub_deadline = min(start_deadline, time.monotonic() + (start_deadline - time.monotonic()) / max(1, n_settings - si))
            sc, z = _run_adam(z0, sub_deadline, **s)
            a = _softmax(z)
            sc_norm = _score_normalized(a)
            sc_full = 2.0 * n * sc_norm
            if sc_full < best_score:
                best_score = sc_full
                best_a = a.copy()
                print(f"    New best from start {i}, setting {si}: score={sc_full:.9f}")

    if best_a is None:
        best_a = _softmax(starts[0][1])
        best_score = _score(best_a)

    # Phase 2: Active-set polish
    if time.monotonic() < deadline - 30:
        print(f"  Polishing best (score={best_score:.9f})...")
        pol_deadline = min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        a_polished = _active_set_polish(best_a, pol_deadline)
        sc_polished = _score(a_polished)
        if sc_polished < best_score:
            best_score = sc_polished
            best_a = a_polished
            print(f"  After active-set polish: score={best_score:.9f}")

    # Phase 3: Coordinate descent
    if time.monotonic() < deadline - 15:
        cd_deadline = min(deadline - 5, time.monotonic() + 60)
        a_cd = _coordinate_descent(best_a, cd_deadline)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd
            print(f"  After coordinate descent: score={best_score:.9f}")

    # Phase 4: Another polish round on the very best
    if time.monotonic() < deadline - 10:
        pol2_deadline = deadline - 5
        a_pol2 = _active_set_polish(best_a, pol2_deadline)
        sc_pol2 = _score(a_pol2)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_a = a_pol2
            print(f"  After final polish: score={best_score:.9f}")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient sequence minimizing the evaluation function.
    Returns the best sequence found as a list of floats.
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 5.0
    rng = np.random.default_rng(int(seed))

    print(f"Total budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    best_score = np.inf
    best_a = None

    # Try primary length
    n_primary = 8192
    try:
        sc, a = _optimize_single_n(n_primary, rng, deadline)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"Primary n={n_primary}: score={best_score:.9f}")
    except Exception as e:
        print(f"Error with n={n_primary}: {e}")

    # Try larger length if time remains
    time_left = deadline - time.monotonic()
    if time_left > 300:
        n_large = 16384
        print(f"\nTrying n={n_large} with {time_left:.0f}s remaining...")
        try:
            deadline2 = time.monotonic() + time_left - 30
            sc, a = _optimize_single_n(n_large, rng, deadline2)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"Large n={n_large}: score={best_score:.9f}")
        except Exception as e:
            print(f"Error with n={n_large}: {e}")

    # Try smaller length if time remains (sometimes smaller is better)
    time_left = deadline - time.monotonic()
    if time_left > 200 and best_score > 1.51:
        n_small = 4096
        print(f"\nTrying n={n_small} with {time_left:.0f}s remaining...")
        try:
            deadline3 = time.monotonic() + time_left - 20
            sc, a = _optimize_single_n(n_small, rng, deadline3)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"Small n={n_small}: score={best_score:.9f}")
        except Exception as e:
            print(f"Error with n={n_small}: {e}")

    # Fallback
    if best_a is None:
        n = 8192
        best_a = np.ones(n) / n
        best_score = _score(best_a)

    # Final normalization and safety
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(8192) / 8192
    else:
        best_a = best_a / s

    # Validate with official evaluator
    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"\nOfficial evaluator score: {float(exact):.9f}")
        else:
            print(f"\nWARNING: Official evaluator returned non-finite: {exact}")
            # Fallback to safe sequence
            best_a = np.ones(8192) / 8192
    except Exception as e:
        print(f"Evaluator check failed: {e}")

    result = [float(v) for v in best_a.tolist()]
    print(f"Final sequence length: {len(result)}, score: {best_score:.9f}")
    return result
```
