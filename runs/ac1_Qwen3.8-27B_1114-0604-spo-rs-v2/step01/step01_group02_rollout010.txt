def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-15):
    """In-place Adam update."""
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    m_hat = m / (1.0 - b1**t)
    v_hat = v / (1.0 - b2**t)
    a[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Minimize f(a) = 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

Strategy:
- Exploit scale invariance: normalize sum to constant, minimize max convolution.
- FFT-based convolution for O(n log n) speed.
- Log-sum-exp smooth max with temperature annealing for stable gradients.
- Adam optimizer with adaptive learning rate and periodic restarts.
- Multi-scale: try several sequence lengths, keep best result.
- Dedicated parent refinement + diverse explorations + final polish phase.
"""

import numpy as np
import time


def fft_self_conv(a):
    M = 2 * len(a) - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def compute_score(a, c):
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return float(2.0 * len(a) * c.max() / (S * S))


def compute_logsumexp_grad(a, c, T):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n), np.inf
    c_max = c.max()
    shifted = (c - c_max) / T
    exp_c = np.exp(shifted)
    L = c_max / T + np.log(exp_c.sum())
    w = exp_c / exp_c.sum()
    N = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    g = np.fft.irfft(A.conj() * W, N)[:n].real
    grad = (2.0 * n / T) * (2.0 * g / (S * S) - 2.0 * L / (S ** 3))
    return grad, L


def compute_topk_grad(a, c, k=8):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n), c.max()
    k_act = min(k, 2 * n - 1)
    top = np.argpartition(c, -k_act)[-k_act:]
    vals = c[top]
    order = np.argsort(vals)[::-1]
    top = top[order]
    vals = vals[order]
    peak = vals[0]
    if peak > 1e-12:
        wts = np.exp((vals - peak) / (0.1 * peak))
        wts /= wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    M_w = float(np.dot(vals, wts))
    grad = np.zeros(n)
    for i in range(k_act):
        ki = int(top[i])
        wi = wts[i]
        if wi < 1e-12:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            grad[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    grad = (2.0 * n) * (grad / (S * S) - 2.0 * M_w / (S ** 3))
    return grad, M_w


def anti_peak_step(a, c, S):
    n = len(a)
    peaks = np.where(c > 0.98 * c.max())[0]
    if len(peaks) == 0:
        return a
    a_new = a.copy()
    for ki in peaks[:20]:
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        seg = a[js:je + 1].copy()
        rev = seg[::-1]
        prods = seg * rev
        if prods.max() <= 0:
            continue
        reduce = 0.3 * prods / prods.sum()
        a_new[js:je + 1] -= reduce
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    a_new *= S / s_new
    return a_new


def adam_update(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-15):
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    a[:] -= lr * (m / bc1) / (np.sqrt(v / bc2) + eps)


def cos_lr(t, T, lr0, lr_min):
    if t >= T:
        return lr_min
    return lr_min + 0.5 * (lr0 - lr_min) * (1.0 + np.cos(np.pi * t / T))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0 / len(a)
        return a
    a *= target / s
    return a


def get_parent(n):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()

    except Exception:

def perturbed_copy(a, rng, scale=0.1):
    a2 = a * (1.0 + scale * rng.standard_normal(len(a)))

def make_inits(n, rng):
    inits = []
    p = get_parent(n)
    if p is not None and p.sum() > 0:
        inits.append(("parent", p.copy()))
        inits.append(("parent_pert", perturbed_copy(p, rng, 0.15)))
    inits.append(("uni", np.ones(n)))
    for f in [1, 2, 3, 5]:
        x = np.linspace(0, 1, n)
        inits.append(("cos%d" % f, np.maximum(1.0 + 0.6 * np.cos(2 * np.pi * f * x), 0.01)))
    inits.append(("ramp", 1.0 + 2.0 * np.linspace(0, 1, n)))
    inits.append(("rand", 0.3 + 0.7 * rng.random(n)))
    for nb in [4, 8, 16, 32]:
        per = n // nb
        blk = 0.3 + 0.7 * rng.random(nb)
        a = np.repeat(blk, per)
        if len(a) < n:
            a = np.concatenate([a, rng.random(n - len(a)) * 0.5 + 0.3])
        inits.append(("blk%d" % nb, a[:n]))
    gap = max(2, n // 512)
    ga = 0.5 + 0.5 * rng.random(n)
    ga[::gap] = 0.0
    inits.append(("gap", ga))
    return inits


def optimize_stream(a0, end_time, rng, T_init, T_final, lr_init, lr_min,
                    phase_iters, anti_peak_every, use_topk, topk_k):
    n = len(a0)
    a = np.array(a0, dtype=np.float64)
    normalize_sum(a, 1.0)
    S = 1.0
    c = fft_self_conv(a)
    sc = compute_score(a, c)
    best = sc
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    patience = 150
    no_imp = 0
    t0 = time.monotonic()
    while time.monotonic() < end_time:
        if t > 0 and t % anti_peak_every == 0:
            a = anti_peak_step(a, c, S)
        g, _ = compute_logsumexp_grad(a, c, T_init + (T_final - T_init) * min(1.0, t / phase_iters))
        if use_topk and t % 3 == 0:
            g2, _ = compute_topk_grad(a, c, topk_k)
            g = 0.5 * g + 0.5 * g2
        t += 1
        lr = cos_lr(t, phase_iters, lr_init, lr_min)
        adam_update(a, g, m, v, t, lr)
        normalize_sum(a, S)
        if t % 2 == 0 or t < 10:
            c = fft_self_conv(a)
            sc = compute_score(a, c)
            if np.isfinite(sc) and sc < best:
                best = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
            if no_imp >= patience:
                a = perturbed_copy(best_a, rng, 0.1 + 0.1 * rng.random())
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                no_imp = 0
                patience = max(60, patience // 2)
                c = fft_self_conv(a)
        if time.monotonic() - t0 > 0:
            pass
    return best, best_a
