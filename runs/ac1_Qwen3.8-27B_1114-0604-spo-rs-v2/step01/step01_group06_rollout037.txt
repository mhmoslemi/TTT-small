```python
import numpy as np
import time

def _fft_conv(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()  # Next power of 2 >= 3n-2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    return np.maximum(c, 0.0)

def _score(a):
    """Compute 2*n*max(c)/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s <= 0:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * np.max(c) / (s * s)

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _active_set_subgradient(a, frac=0.1):
    """Compute the average subgradient of max(c) at a, using the active set."""
    n = len(a)
    c = _fft_conv(a)
    max_c = np.max(c)
    threshold = (1.0 - frac) * max_c
    s = (c >= threshold).astype(np.float64)  # Active set indicator
    active_count = np.sum(s)
```

```python
        # Adaptive learning rate
        eff_lr = lr * (1.0 - 0.5 * (t / max_iterations))
        z = z - eff_lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -25.0, 25.0)

        if t % print_interval == 0:
            a = _softmax(z)
            sc = _obj(a)
            print(f"  adam it={t} obj={sc:.9f} best_obj={_obj(_softmax(best_z)):.9f} "
                  f"time={time.monotonic()-t0:.0f}s")

    return best_z


def _active_set_polish(a0, deadline, frac_schedule=None, max_iter=20000):
    """Active-set subgradient method with line search in original space."""
    n = _N
    a = np.array(a0, dtype=np.float64, copy=True)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a0
    a = a / s

    best_a = a.copy()
    best_maxc = _obj_maxc(a)[1]

    if frac_schedule is None:
        frac_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.0, 0.0]

    frac_idx = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _conv(a)
        max_c = c.max()

        if max_c < best_maxc - 1e-18:
            best_maxc = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Random perturbation to escape
            z = np.log(np.maximum(a, 1e-300))
            z += 0.1 * rng.standard_normal(n)
            z = z - z.mean()
            z = np.clip(z, -25.0, 25.0)
            a = _softmax(z)
            no_improve = 0
            if frac_idx < len(frac_schedule) - 1:
                frac_idx += 1
            continue

        frac = frac_schedule[min(frac_idx, len(frac_schedule) - 1)]
        threshold = (1.0 - frac) * max_c
        s_active = (c >= threshold).astype(np.float64)
        cnt = s_active.sum()
        if cnt < 1:
            s_active = c == max_c
            cnt = s_active.sum()

        h = np.correlate(s_active, a, mode='valid')[:n]
        g = h / cnt + a

        # Line search
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Normalized gradient direction (we want to decrease max_c)
        direction = -g / gn

        accepted = False
        # Try a range of step sizes
        base_step = max_c / gn  # Scale-aware step
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            eta = base_step * mult
            a_new = np.maximum(a + eta * direction, 0.0)
            ss = a_new.sum()
            if ss < 1e-12:
                continue
            a_new = a_new / ss
            max_c_new = _conv(a_new).max()
            if max_c_new < max_c - 1e-15:
                a = a_new
                accepted = True
                break

        if not accepted:
            # Try multiplicative update
            for lr in [0.01, 0.005, 0.001, 0.0005]:
                a_new = a * np.exp(-lr * g / a.clip(1e-300, None))
                a_new = np.maximum(a_new, 0.0)
                ss = a_new.sum()
                if ss < 1e-12:
                    continue
                a_new = a_new / ss
                max_c_new = _conv(a_new).max()
                if max_c_new < max_c - 1e-15:
                    a = a_new
                    accepted = True
                    break

        if it % 200 == 0:
            sc = 2.0 * _N * max_c / (a.sum() ** 2)
            best_sc = 2.0 * _N * best_maxc / (best_a.sum() ** 2)
            print(f"  polish it={it} cur={sc:.9f} best={best_sc:.9f} "
                  f"frac={frac:.3f} time={time.monotonic()-t0:.0f}s")

    return best_a


def _make_exponential_start(n, alpha):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i / n)
    return a / a.sum()


def _make_block_start(n, ratio):
    a = np.ones(n)
    half = n // 2
    a[half:] = ratio
    return a / a.sum()


def _make_sine_start(n, A, m, phase=0.0):
    u = np.linspace(0, 1, n)
    a = 1.0 + A * np.sin(2 * np.pi * m * u + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_cosine_basis_start(n, coeffs):
    u = np.linspace(0, 1, n)
    a = np.ones(n)
    for k, c in enumerate(coeffs, 1):
        a += c * np.cos(2 * np.pi * k * u)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_gaussian_start(n, center, width):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - center * n) / (width * n)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_clip = np.clip(a, 1e-20, None)
    log_a = np.log(a_clip)
    log_new = np.interp(x_new, x_old, log_a)
    new = np.exp(log_new - log_new.max())
    return new / new.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for optimal non-negative step function coefficients.
    Returns a list of non-negative floats minimizing the evaluation function.
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 8.0
    t_start = time.monotonic()

    # Choose sequence length
    n = 8192
    _init(n)
    print(f"n={n}, budget={budget_s:.0f}s")

    rng = np.random.default_rng(int(seed))

    # ---- Generate diverse starts ----
    starts = []

    # Try to use the previous best solution
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 10:
                g = np.clip(g, 1e-10, 1000.0)
                if np.all(np.isfinite(g)) and g.sum() > 0:
                    starts.append(("prev_best", _resample(g, n)))
    except Exception:
        pass

    # Exponential starts
    for alpha in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0]:
        starts.append((f"exp_{alpha}", _make_exponential_start(n, alpha)))

    # Block starts
    for ratio in [1.2, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 7.0, 10.0]:
        starts.append((f"block_{ratio}", _make_block_start(n, ratio)))

    # Sine-modulated starts
    for A in [0.3, 0.5, 0.7, 0.9]:
        for m in [1, 2, 3, 5, 8]:
            for phase in [0.0, np.pi / 2, np.pi]:
                s = _make_sine_start(n, A, m, phase)
                starts.append((f"sine_A{A}_m{m}_p{phase:.1f}", s))

    # Cosine basis starts
    for scale in [0.2, 0.4, 0.6, 0.8]:
        coeffs = [scale * (0.5 + 0.5 * rng.standard_normal()) * (-1) ** k for k in range(1, 6)]
        starts.append((f"cosb_{scale}", _make_cosine_basis_start(n, coeffs)))

    # Gaussian-shaped starts
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            starts.append((f"gauss_c{center}_w{width}", _make_gaussian_start(n, center, width)))

    # Random smooth starts
    for trial in range(8):
        coeffs = 0.3 * rng.standard_normal(10)
        starts.append((f"rand_{trial}", _make_cosine_basis_start(n, coeffs)))

    # Evaluate all starts
    print(f"Total starts: {len(starts)}")
    scored_starts = []
    for name, a in starts:
        sc = _obj(a)
        if np.isfinite(sc):
            scored_starts.append((sc, name, a))

    scored_starts.sort(key=lambda x: x[0])
    print(f"Best start: {scored_starts[0][1]} with score={scored_starts[0][0]:.9f}")
    for i, (sc, name, a) in enumerate(scored_starts[:10]):
        print(f"  {i}: {name} score={sc:.9f}")

    # ---- Phase 2: Adam optimization from best starts ----
    best_a = scored_starts[0][2].copy()
    best_score = scored_starts[0][0]

    # Reserve time for polishing
    polish_reserve = 300.0
    adam_reserve = 120.0

    # Run Adam on top starts
    top_k = min(6, len(scored_starts))
    for idx in range(top_k):
        sc, name, a = scored_starts[idx]
        time_left = deadline - time.monotonic()
        if time_left < polish_reserve + 60:
            break

        z = np.log(np.maximum(a, 1e-20))
        z = z - z.mean()
        z = np.clip(z, -20.0, 20.0)

        # Time allocation: more time for better starts
        if idx == 0:
            adam_deadline = time.monotonic() + min(time_left - polish_reserve - adam_reserve, 600.0)
            tau = 1e-5
            lr = 0.01
        elif idx < 3:
            adam_deadline = time.monotonic() + min(time_left - polish_reserve - adam_reserve, 300.0)
            tau = 1e-5
            lr = 0.01
        else:
            adam_deadline = time.monotonic() + min(time_left - polish_reserve - adam_reserve, 150.0)
            tau = 1e-4
            lr = 0.02

        print(f"Adam start {idx} ({name}): tau={tau}, lr={lr}")
        z_best = _adam_z(z, adam_deadline, tau=tau, lr=lr,
                         max_iterations=3000)
        a_new = _softmax(z_best)
        sc_new = _obj(a_new)
        print(f"  Adam done: score={sc_new:.9f} (was {sc:.9f})")

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  New best: {sc_new:.9f}")

    # ---- Phase 3: Active-set subgradient polishing ----
    time_left = deadline - time.monotonic()
    print(f"\nStarting active-set polish, time left={time_left:.0f}s")

    # Multiple polish passes with different frac schedules
    polish1_deadline = time.monotonic() + min(time_left * 0.6, 400.0)
    a_polished = _active_set_polish(best_a, polish1_deadline,
                                    frac_schedule=[0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.0, 0.0])
    sc_polished = _obj(a_polished)
    if sc_polished < best_score:
        best_score = sc_polished
        best_a = a_polished.copy()
    print(f"After polish 1: {best_score:.9f}")

    # Second polish with tighter active set
    time_left = deadline - time.monotonic()
    if time_left > 60:
        polish2_deadline = time.monotonic() + min(time_left * 0.7, 300.0)
        a_polished2 = _active_set_polish(best_a, polish2_deadline,
                                         frac_schedule=[0.05, 0.02, 0.01, 0.005, 0.0, 0.0, 0.0, 0.0])
        sc_polished2 = _obj(a_polished2)
        if sc_polished2 < best_score:
            best_score = sc_polished2
            best_a = a_polished2.copy()
        print(f"After polish 2: {best_score:.9f}")

    # ---- Phase 4: Random restarts if time remains ----
    time_left = deadline - time.monotonic()
    if time_left > 120:
        print(f"\nPhase 4: Random restarts, time left={time_left:.0f}s")
        for trial in range(20):
            time_left = deadline - time.monotonic()
            if time_left < 30:
                break

            # Perturb the best solution
            z = np.log(np.maximum(best_a, 1e-20))
            z = z - z.mean()
            perturbation_scale = max(0.05, 0.5 / (1 + trial * 0.5))
            z += perturbation_scale * rng.standard_normal(n)
            z = np.clip(z, -20.0, 20.0)

            # Quick Adam
            adam_d = time.monotonic() + min(60.0, time_left * 0.4)
            z_new = _adam_z(z, adam_d, tau=1e-5, lr=0.01, max_iterations=800)
            a_new = _softmax(z_new)

            # Quick polish
            pol_d = time.monotonic() + min(40.0, time_left * 0.3)
            a_new = _active_set_polish(a_new, pol_d,
                                       frac_schedule=[0.1, 0.05, 0.02, 0.01, 0.0, 0.0])

            sc_new = _obj(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                print(f"  Restart {trial}: new best {sc_new:.9f}")
            else:
                print(f"  Restart {trial}: {sc_new:.9f} (no improvement)")

    # ---- Final validation ----
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s

    final_score = _obj(best_a)
    print(f"\nFinal score (internal): {final_score:.9f}")
    print(f"Total time: {time.monotonic() - t_start:.0f}s")

    # Validate with official evaluator
    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator score: {float(exact):.9f}")
    except Exception as e:
        print(f"Evaluator check failed: {e}")

    return [float(v) for v in best_a.tolist()]
```
