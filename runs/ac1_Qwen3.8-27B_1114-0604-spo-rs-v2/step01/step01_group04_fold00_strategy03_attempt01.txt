def project_simplex(v):
    # Projects onto the simplex: a>=0, a.sum()=1.
    # Based on standard algorithm from Duchi et al.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    mask = u * idx > cssv - 1.0
    idx2 = np.max(np.where(mask))
    if idx2 == 0:
        return np.ones(n)/n
    else:
        thres = (cssv[idx2] - 1.0) / idx2
        return np.maximum(v - thres, 0.0)
