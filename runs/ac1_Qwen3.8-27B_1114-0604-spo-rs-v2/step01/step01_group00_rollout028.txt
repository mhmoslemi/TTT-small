Step function optimizer using combination of:
1. Smoothed subgradient descent with FFT-based autocorrelation
2. Subsampled LP direction finding for periodic corrections
3. Multiple diverse starting points
4. Final coordinate-descent polish

from scipy.signal import correlate

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval(f):
    return evaluate_sequence(f.tolist())


def _gradient_from_autocorr(c, f, T):
    """Compute smoothed gradient from autocorrelation c and sequence f."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(c))
        p[n - 1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p


def _softmax_gradient(c, f, T, top_k=None):
    """Compute gradient using top-k lags."""
    n = len(f)
    c_len = len(c)
    if top_k is not None and top_k < c_len:
        # Use top-k lags
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_masked = np.full(c_len, -np.inf)
        c_masked[idx] = c[idx]
    else:
        c_masked = c
    return _gradient_from_autocorr(c_masked, f, T)


def _mirror_descent_step(f, grad, eta):
    """Mirror descent step on the simplex."""
    log_f = np.log(f + 1e-30) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _solve_subsampled_lp(f, lags):
    """Solve LP: min t s.t. (f*g)[k] <= t for k in lags, sum g = 1, g >= 0."""
    n = len(f)
    K = len(lags)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for idx, k in enumerate(lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        g_idx = np.arange(k - i_end, k - i_start + 1)
        A[idx, g_idx] = f[i_start:i_end + 1]
    
    # Full A_ub: [A | -1]
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    return result.x[:n]


def _optimize_gradient(f, n, deadline, rng, label=""):
    """Smoothed subgradient descent with adaptive parameters."""
    f = f.astype(np.float64).copy()

    else:

    local_best = _eval(f)
    local_best_f = f.copy()
    
    c = _autocorr_fft(f)
    peak0 = c.max()
    
    T = peak0 * 0.05
    T_min = peak0 * 1e-10
    
    # Adaptive step size
    base_eta = 1.0
    momentum = np.zeros(n)
    mu = 0.8
    
    it = 0
    max_iters = 1000000
    stuck = 0
    eval_interval = max(1, int(0.1 * n / 10))  # Evaluate every ~0.1n/10 iterations
    
    while time.monotonic() < deadline and it < max_iters:
        # Compute autocorrelation and gradient
        c = _autocorr_fft(f)
        
        # Adaptive temperature

        T = max(T_min, peak0 * 0.05 * (1 - progress) ** 2)
        
        grad, p = _softmax_gradient(c, f, T, top_k=min(20, 2*n-1))
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        
        # Step size: adapt based on gradient magnitude
        grad_norm = np.max(np.abs(momentum))
        if grad_norm < 1e-15:
            break
        eta = base_eta / grad_norm
        
        # Mirror descent step

        # Sometimes mix with linear interpolation for stability

            f = 0.5 * f + 0.5 * f_new
        else:
            f = f_new
        
        it += 1

                local_best_f = f.copy()
                stuck = 0

                stuck = 0

def _optimize_lp(f, n, deadline, rng, label=""):
    """LP-based optimization (subsampled)."""
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    local_best = _eval(f)
    local_best_f = f.copy()
    
    K = 30  # Number of lags to constrain

    alpha_idx = 0
    
    it = 0
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        # Get top K lags
        top_k = np.argpartition(c, -K)[-K:]

        if g is None:
            break

        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Update f
        alpha = alpha_schedule[min(alpha_idx, len(alpha_schedule) - 1)]
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        
        # Check improvement

                f = f_new

        if score < local_best:

            local_best_f = f.copy()
    
    return local_best_f, local_best


def _polish(f, n, deadline, rng):
    """Coordinate descent polish."""
    f = f.copy()
    local_best = _eval(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()
    
    while time.monotonic() < deadline:
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Try moving mass from positions that contribute to the peak
        # to positions that don't
        # Contribution of position j to lag k_star: f[j] * f[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[j] * f[idx]
        
        # Find best donor (high contribution) and recipient (low contribution)
        j_donors = np.argsort(contrib)[-5:]  # Top 5 donors
        j_recipients = np.argsort(contrib)[:5]  # Bottom 5 recipients
        
        improved = False
        for jd in j_donors:
            for jr in j_recipients:
                if jd == jr:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    f_try = f.copy()
                    amt = f_try[jd] * frac
                    if amt < 1e-15:
                        continue
                    f_try[jd] -= amt
                    f_try[jr] += amt
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        peak = c_try.max()
                        c = c_try
                        local_best = _eval(f)
                        local_best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        # Random perturbations
        for _ in range(20):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = f_try[i] * 0.03
            if amt < 1e-15:
                continue
            f_try[i] -= amt
            f_try[jx] += amt
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                c = c_try
        
        if not improved:
            # Add noise and try again
            f = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr_fft(f)
            peak = c.max()
    
    return local_best_f, local_best

"""
Step function optimizer combining:
1. Smoothed subgradient descent (top-k softmax) with momentum and mirror descent
2. Subsampled LP direction finding for periodic corrections
3. Multiple diverse starting points across different sequence lengths
4. Final coordinate-descent polish
Key insight: asymmetric sequences can beat the symmetric lower bound of 2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time


def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval(f):
    return evaluate_sequence(f.tolist())


def _compute_gradient(c, f, T, top_k=None):
    n = len(f)
    c_len = len(c)
    if top_k is not None and top_k < c_len:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = np.full(c_len, -np.inf)
        c_work[idx] = c[idx]
    else:
        c_work = c.copy()
    cs = c_work / T
    cs -= cs[np.isfinite(cs)].max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(c_len)
        p[n - 1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p


def _mirror_step(f, grad, eta):
    log_f = np.log(f + 1e-30) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _solve_subsampled_lp(f, lags):
    n = len(f)
    K = len(lags)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for idx, k in enumerate(lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        g_idx = np.arange(k - i_end, k - i_start + 1)
        A[idx, g_idx] = f[i_start:i_end + 1]
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status != 0:
            return None
        return np.maximum(result.x[:n], 0)
    except Exception:
        return None


def _gradient_optimize(f, n, deadline, rng, label=""):
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _eval(f)
    local_best_f = f.copy()

    c = _autocorr_fft(f)
    peak0 = c.max()
    T_start = peak0 * 0.05
    T_end = peak0 * 1e-10

    momentum = np.zeros(n)
    mu = 0.85

    it = 0
    max_iters = 2000000
    stuck = 0
    eval_interval = 500

    while time.monotonic() < deadline and it < max_iters:
        progress = min(1.0, it / max_iters)
        T = max(T_end, T_start * (1 - progress) ** 3)

        c = _autocorr_fft(f)
        top_k = min(30, 2 * n - 1)
        grad, p = _compute_gradient(c, f, T, top_k=top_k)

        momentum = mu * momentum + (1 - mu) * grad
        grad_max = np.max(np.abs(momentum))
        if grad_max < 1e-18:
            break
        eta = 1.0 / grad_max

        f_new = _mirror_step(f, momentum, eta)

        if it % 500 == 0:
            f = 0.7 * f + 0.3 * f_new
        else:
            f = f_new

        it += 1

        if it % eval_interval == 0:
            score = _eval(f)
            if score < local_best - 1e-7:
                local_best = score
                local_best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 15:
                f = local_best_f + 0.003 / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                stuck = 0
                momentum = np.zeros(n)

    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best


def _lp_optimize(f, n, deadline, rng, label=""):
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    local_best = _eval(f)
    local_best_f = f.copy()

    K = 50
    alphas = [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
    alpha_idx = 0

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        top_k_idx = np.argpartition(c, -K)[-K:]
        g = _solve_subsampled_lp(f, top_k_idx.tolist())
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        alpha = alphas[min(alpha_idx, len(alphas) - 1)]
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-10:
            break
        f_new = f_new / f_new_sum

        c_new = _autocorr_fft(f_new)
        if c_new.max() < c.max() - 1e-12:
            f = f_new
            alpha_idx = max(0, alpha_idx - 1)
        else:
            alpha_idx = min(alpha_idx + 1, len(alphas) - 1)
            if alpha_idx >= len(alphas) - 1:
                break

        score = _eval(f)
        if score < local_best:
            local_best = score
            local_best_f = f.copy()

    return local_best_f, local_best


def _polish(f, n, deadline, rng):
    f = f.astype(np.float64).copy()
    local_best = _eval(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()

    iter_count = 0
    while time.monotonic() < deadline:
        k_star = int(np.argmax(c))
        contrib = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        for i in range(i_start, i_end + 1):
            contrib[i] = f[i] * f[k_star - i]

        top_donors = np.argsort(contrib)[-8:]
        low_recipients = np.argsort(contrib)[:8]

        improved = False
        for jd in top_donors:
            if improved:
                break
            for jr in low_recipients:
                if jd == jr:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    f_try = f.copy()
                    amt = f_try[jd] * frac
                    if amt < 1e-15:
                        continue
                    f_try[jd] -= amt
                    f_try[jr] += amt
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        peak = c_try.max()
                        c = c_try
                        local_best = _eval(f)
                        local_best_f = f.copy()
                        improved = True
                        break

        for _ in range(30):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = f_try[i] * 0.02
            if amt < 1e-15:
                continue
            f_try[i] -= amt
            f_try[jx] += amt
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                c = c_try

        iter_count += 1
        if iter_count % 100 == 0 and not improved:
            f = local_best_f + 0.002 / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr_fft(f)
            peak = c.max()

    return local_best_f, local_best


def _make_starts(n, rng):
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = {}
    starts['uniform'] = np.ones(n)
    starts['exp_right'] = np.exp(-j / (n * 0.5)) + 0.001
    starts['exp_left'] = np.exp(-(n - 1 - j) / (n * 0.5)) + 0.001
    starts['exp_sym'] = np.exp(-np.abs(j - center) / (n * 0.3)) + 0.001
    starts['ramp'] = (j / n) + 0.01
    starts['ramp2'] = (1 - j / n) + 0.01
    starts['bimodal'] = (np.exp(-(j - 0.3 * n) ** 2 / (0.05 * n * n)) +
                         np.exp(-(j - 0.7 * n) ** 2 / (0.05 * n * n)) + 0.01)
    starts['sine'] = 1.0 + 0.7 * np.sin(2.0 * np.pi * j / n)
    starts['sine2'] = 1.0 + 0.5 * np.sin(4.0 * np.pi * j / n + 0.5)
    starts['random'] = rng.exponential(1.0, size=n)
    starts['power'] = (j / n) ** 3 + 0.01
    starts['power2'] = (j / n) ** 0.5 + 0.01
    for key in starts:
        v = starts[key]
        v = np.maximum(v, 0)
        v = v / v.sum()
        starts[key] = v
    return starts
