def _autocorr_fft(f):

    n = len(f)

def _eval(f):
    """Evaluate sequence using the provided function."""

def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    f_new = f_nesterov - step * grad

def _multi_peak_gradient_from_weights(p, f):
    """Compute gradient given explicit weights p."""
    n = len(f)
    fr = f[::-1]
    conv_size = 1 << (len(p) + n - 1).bit_length()

    FR = np.fft.rfft(fr, conv_size)
    conv = np.fft.irfft(P * FR, conv_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _gradient_from_weights(p, f):
    """Compute gradient of weighted sum of autocorrelation values.
    grad[i] = 2 * sum_k p[k] * f[k-i] for valid k-i (0 <= k-i < n)
    Uses FFT-based convolution for efficiency.
    """
    n = len(f)
    fr = f[::-1]
    conv_size = 1 << (len(p) + n - 1).bit_length()
    P = np.fft.rfft(p, conv_size)
    FR = np.fft.rfft(fr, conv_size)
    conv = np.fft.irfft(P * FR, conv_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def _get_gradient(f, c, top_k=3, T=None):
    """Get gradient using either top-k peaks or soft-max weighting."""
    if T is not None:
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        p /= p.sum()
    else:
        p = np.zeros(len(c))
        indices = np.argsort(c)[::-1][:top_k]
        p[indices] = 1.0 / top_k
    return _gradient_from_weights(p, f)


def _spectral_flatten(f, strength=0.05):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    power = np.abs(F) ** 2
    # Compute target: pull non-DC bins toward their mean
    if len(power) > 1:
        mean_pw = np.mean(power[1:])
        new_power = power.copy()
        new_power[1:] = power[1:] + strength * (mean_pw - power[1:])
        new_power = np.maximum(new_power, 0)
    else:
        new_power = power
    # Preserve phase
    mag = np.abs(F)
    phase = np.ones_like(F)
    mask = mag > 1e-15
    phase[mask] = F[mask] / mag[mask]
    new_mag = np.sqrt(new_power)
    F_new = new_mag * phase
    f_new = np.fft.irfft(F_new, size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def _nesterov_iter(f, f_prev, grad, step, momentum_t):
    """One Nesterov accelerated gradient step with simplex projection."""
    t = (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t)) / 2.0
    beta = (momentum_t - 1.0) / t
    f_lookahead = f + beta * (f - f_prev)
    f_new = f_lookahead - step * grad
    f_new = _project_simplex(f_new)
    return f_new, t


def _optimize_main(f_init, n, deadline, rng, top_k_start=5, momentum_reset_interval=5000, label=""):
    """Main Nesterov optimization loop with adaptive strategies."""
    f = f_init.copy()
    f_prev = f.copy()
    local_best = _eval(f)
    local_best_f = f.copy()
    momentum_t = 1.0
    stuck_count = 0
    
    iteration = 0
    max_iters = 1000000
    patience = 12
    
    # Adaptive parameters
    base_step = 0.5
    step = base_step
    
    while time.monotonic() < deadline and iteration < max_iters:
        c = _autocorr_fft(f)
        
        # Adaptive top_k: start broad, narrow down
        progress = min(iteration / 50000.0, 1.0)
        top_k = max(1, top_k_start - int(progress * (top_k_start - 1)))
        
        # Adaptive step size: decrease over time
        step = base_step * (1.0 - 0.7 * progress) + 0.1
        step = max(step, 0.1)
        
        # Get gradient
        grad = _get_gradient(f, c, top_k=top_k)
        
        # Nesterov step
        f_new, momentum_t = _nesterov_iter(f, f_prev, grad, step, momentum_t)
        f_prev = f
        f = f_new
        iteration += 1
        
        # Reset momentum periodically to avoid overshooting
        if iteration % momentum_reset_interval == 0:
            f_prev = f.copy()
            momentum_t = 1.0
        
        # Evaluate and check for improvement
        if iteration % 3000 == 0:
            score = _eval(f)
            if score < local_best - 1e-8:
                local_best = score
                local_best_f = f.copy()
                stuck_count = 0
                if label and iteration % 30000 == 0:
                    print(f"    {label} iter {iteration}: {score:.6f}")
            else:
                stuck_count += 1
            
            # Escape strategies
            if stuck_count >= patience:
                strategy = rng.random()
                if strategy < 0.4:
                    # Spectral flattening
                    f = _spectral_flatten(f, strength=0.03 + 0.07 * rng.random())
                elif strategy < 0.7:
                    # Gaussian perturbation
                    f = f + (0.005 / n) * rng.standard_normal(n)
                    f = _project_simplex(f)
                else:
                    # Targeted perturbation: boost low-contrib positions
                    c_now = _autocorr_fft(f)
                    peak_lag = int(np.argmax(c_now))
                    contrib = np.zeros(n)
                    for i in range(n):
                        j_idx = peak_lag - i
                        if 0 <= j_idx < n:
                            contrib[i] = f[i] * f[j_idx]
                    # Boost the lowest contributing positions
                    low_idx = np.argsort(contrib)[:n // 10]
                    f[low_idx] += 0.002 / n
                    f = _project_simplex(f)
                
                f_prev = f.copy()
                momentum_t = 1.0
                stuck_count = 0
    
    # Final evaluation
    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best


def _optimize_lbfgs_smooth(f_init, n, deadline, T_start=0.1):
    """L-BFGS with log-sum-exp smoothing, decreasing T."""
    f = f_init.copy()
    best_f = f.copy()
    best_score = _eval(f)
    
    current_T = T_start
    min_T = 0.005
    
    while time.monotonic() < deadline and current_T > min_T:
        T = current_T
        
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-15:
                return 2 * n, np.zeros(n)
            x = x / s
            c = _autocorr_fft(x)
            cs = c / T
            cs -= cs.max()
            val = 2 * n * T * np.log(np.sum(np.exp(cs)))
            p = np.exp(cs)
            p /= p.sum()
            grad = _gradient_from_weights(p, x)
            # Project gradient to simplex tangent space
            grad = grad - np.mean(grad)
            return val, grad
        
        x0 = best_f.copy()
        try:
            res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 200, 'maxfun': 1000,
                                   'ftol': 1e-14, 'gtol': 1e-14,
                                   'iprint': -1})
            f_new = np.maximum(res.x, 0)
            s = f_new.sum()
            if s > 1e-10:
                f_new = f_new / s
                score = _eval(f_new)
                if score < best_score:
                    best_score = score
                    best_f = f_new.copy()
        except Exception:
            pass
        
        # Decrease T
        current_T *= 0.6
        
        if time.monotonic() > deadline - 30:
            break
    
    return best_f, best_score


def _polish_refined(f, n, deadline, rng):
    """Refined polish: exact line search in coordinate pairs."""
    f = f.copy()
    local_best = _eval(f)
    local_best_f = f.copy()
    
    iter_count = 0
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() < deadline and no_improve < max_no_improve:
        c = _autocorr_fft(f)
        peak_val = c.max()
        peak_lag = int(np.argmax(c))
        
        # Compute contributions to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j_idx = peak_lag - i
            if 0 <= j_idx < n:
                contrib[i] = f[i] * f[j_idx]
        
        # Try moving mass from high-contrib to low-contrib positions
        sorted_idx = np.argsort(contrib)
        improved_this_round = False
        
        for attempt in range(20):
            # Source: high contribution
            src_candidates = sorted_idx[sorted_idx >= n // 4]
            # Destination: low contribution
            dst_candidates = sorted_idx[sorted_idx < n // 4]
            
            if len(src_candidates) == 0 or len(dst_candidates) == 0:
                break
            
            src = src_candidates[rng.integers(0, len(src_candidates))]
            dst = dst_candidates[rng.integers(0, len(dst_candidates))]
            
            if src == dst:
                continue
            
            # Binary search for best fraction
            lo, hi = 0.001, 0.5
            best_frac = 0.01
            best_c_max = peak_val
            
            for _ in range(5):
                mid = (lo + hi) / 2
                f_try = f.copy()
                amt = f_try[src] * mid
                f_try[src] -= amt
                f_try[dst] += amt
                if f_try[src] < 0:
                    hi = mid
                    continue
                f_try = f_try / f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < best_c_max:
                    best_c_max = c_try.max()
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            # Apply best fraction
            f_try = f.copy()
            amt = f_try[src] * best_frac
            f_try[src] -= amt
            f_try[dst] += amt
            if f_try[src] >= 0:
                f_try = f_try / f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak_val - 1e-15:
                    f = f_try
                    local_best = _eval(f)
                    local_best_f = f.copy()
                    improved_this_round = True
                    break
        
        iter_count += 1
        if not improved_this_round:
            no_improve += 1
            # Perturb to escape
            f = f + (0.002 / n) * rng.standard_normal(n)
            f = _project_simplex(f)
        else:
            no_improve = 0
    
    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(autocorr)/(sum f)^2.
    
    Strategy:
    1. Multiple diverse structural starting points
    2. Nesterov accelerated projected gradient with adaptive multi-peak gradient
    3. Spectral domain flattening for diversity
    4. L-BFGS with log-sum-exp smoothing for refinement
    5. Refined coordinate pair polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    
    # Initialize from given sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    
    # Normalize to simplex
    s = f_init.sum()
    if s < 1e-10:
        f_init = np.ones(n)
    else:
        f_init = f_init / s
    
    best_f = f_init.copy()
    best_score = _eval(f_init)
    print(f"Initial: score={best_score:.6f}, n={n}")
    
    # === Generate diverse starting points ===
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    starts = {}
    starts['given'] = f_init.copy()
    
    # Various structural templates
    starts['u_wide'] = _project_simplex(1.0 + 1.0 * ((j - center) / center) ** 2)
    starts['u_narrow'] = _project_simplex(1.0 + 5.0 * ((j - center) / center) ** 2)
    starts['u_very_narrow'] = _project_simplex(1.0 + 20.0 * ((j - center) / center) ** 2)
    starts['bimodal'] = _project_simplex(
        np.exp(-(j - 0.2 * n) ** 2 / (0.04 * n * n)) +
        np.exp(-(j - 0.8 * n) ** 2 / (0.04 * n * n)) + 0.005)
    starts['trimodal'] = _project_simplex(
        np.exp(-(j - 0.15 * n) ** 2 / (0.02 * n * n)) +
        np.exp(-(j - 0.5 * n) ** 2 / (0.02 * n * n)) +
        np.exp(-(j - 0.85 * n) ** 2 / (0.02 * n * n)) + 0.005)
    
    # Structured random (smooth)
    for ri in range(3):
        x = rng.standard_normal(n)
        # Low-pass filter to make it smooth
        size = 1 << (2 * n - 1).bit_length()
        X = np.fft.rfft(x, size)
        freq = np.arange(len(X))
        cutoff = int(n * 0.1)
        X[cutoff:] *= 0.1
        x_smooth = np.fft.irfft(X, size)[:n]
        starts[f'smooth_rand_{ri}'] = _project_simplex(np.abs(x_smooth) + 0.01)
    
    # Random with structure
    starts['random_exp'] = _project_simplex(rng.exponential(1.0, size=n))
    starts['random_bimodal'] = _project_simplex(
        rng.exponential(1.0, size=n) * 
        (1.0 + 2.0 * np.exp(-(j - 0.3 * n) ** 2 / (0.05 * n * n)) +
         2.0 * np.exp(-(j - 0.7 * n) ** 2 / (0.05 * n * n))))
    
    # Cosine-modulated
    for freq in [2, 3, 5]:
        starts[f'cos_{freq}'] = _project_simplex(
            1.0 + 0.3 * np.cos(2 * np.pi * freq * j / n) +
            0.2 * ((j - center) / center) ** 2)
    
    # Evaluate all starts
    print("\nEvaluating starting points:")
    start_results = []
    for name, f0 in starts.items():
        f0 = np.maximum(f0, 0)
        s0 = f0.sum()
        if s0 < 1e-10:
            f0 = np.ones(n)
        else:
            f0 = f0 / s0
        sc = _eval(f0)
        start_results.append((sc, name, f0.copy()))
        if sc < best_score:
            best_score = sc
            best_f = f0.copy()
    
    start_results.sort(key=lambda x: x[0])
    for sc, name, _ in start_results[:5]:
        print(f"  {name}: {sc:.6f}")
    
    # === Phase 1: Nesterov from top 4 starts ===
    print(f"\n=== Phase 1: Nesterov optimization ===")
    phase1_deadline = time.monotonic() + (budget_s - 400) * 0.45
    time_per_start = (phase1_deadline - time.monotonic()) / 4.0
    
    for idx, (sc, name, f0) in enumerate(start_results[:4]):
        if time.monotonic() > phase1_deadline:
            break
        t_alloc = time_per_start * (1.0 - 0.05 * idx)
        s_deadline = time.monotonic() + t_alloc
        print(f"  Starting '{name}' (score={sc:.6f}, time={t_alloc:.0f}s)")
        
        f_opt, sc_opt = _optimize_main(f0, n, s_deadline, rng,
                                        top_k_start=5, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.6f}")
    
    # === Phase 2: Focused Nesterov (top_k=1) on current best ===
    print(f"\n=== Phase 2: Focused optimization ===")
    phase2_deadline = min(time.monotonic() + (budget_s - 400) * 0.25, deadline - 300)
    if time.monotonic() < phase2_deadline:
        f_opt, sc_opt = _optimize_main(best_f, n, phase2_deadline, rng,
                                        top_k_start=2, label="focused")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Focused best: {best_score:.6f}")
    
    # === Phase 3: Spectral-shaping focused optimization ===
    print(f"\n=== Phase 3: Spectral shaping optimization ===")
    phase3_deadline = min(time.monotonic() + 200, deadline - 200)
    if time.monotonic() < phase3_deadline:
        f_cur = best_f.copy()
        f_prev = f_cur.copy()
        mom_t = 1.0
        local_best = best_score
        local_best_f = best_f.copy()
        
        iteration = 0
        while time.monotonic() < phase3_deadline and iteration < 100000:
            # Alternate between gradient step and spectral shaping
            if iteration % 3 == 0:
                c = _autocorr_fft(f_cur)
                # Use soft-max with small T for focus
                peak = c.max()
                T = max(peak * 0.01, 1e-6)
                grad = _get_gradient(f_cur, c, top_k=1, T=T)
                step = 0.3
                f_new, mom_t = _nesterov_iter(f_cur, f_prev, grad, step, mom_t)
                f_prev = f_cur
                f_cur = f_new
            else:
                # Spectral flattening
                strength = 0.02 + 0.03 * (iteration / 100000.0)
                f_cur = _spectral_flatten(f_cur, strength=strength)
                f_prev = f_cur.copy()
                mom_t = 1.0
            
            iteration += 1
            
            if iteration % 5000 == 0:
                score = _eval(f_cur)
                if score < local_best:
                    local_best = score
                    local_best_f = f_cur.copy()
                    print(f"    Spectral iter {iteration}: {local_best:.6f}")
        
        score = _eval(f_cur)
        if score < local_best:
            local_best = score
            local_best_f = f_cur.copy()
        
        if local_best < best_score:
            best_score = local_best
            best_f = local_best_f.copy()
            print(f"  Phase 3 best: {best_score:.6f}")
    
    # === Phase 4: L-BFGS refinement ===
    print(f"\n=== Phase 4: L-BFGS refinement ===")
    phase4_deadline = min(time.monotonic() + 200, deadline - 100)
    if time.monotonic() < phase4_deadline:
        f_opt, sc_opt = _optimize_lbfgs_smooth(best_f, n, phase4_deadline, T_start=0.05)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  L-BFGS best: {best_score:.6f}")
    
    # === Phase 5: Refined polish ===
    print(f"\n=== Phase 5: Polish ===")
    polish_time = min(180.0, max(30.0, deadline - time.monotonic()))
    if polish_time > 20:
        polish_deadline = time.monotonic() + polish_time
        f_polished, sc_polished = _polish_refined(best_f, n, polish_deadline, rng)
        if sc_polished < best_score:
            best_score = sc_polished
            best_f = f_polished.copy()
            print(f"  Polished: {best_score:.6f}")
    
    # === Final: one more focused Nesterov from best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Final pass ({remaining:.0f}s remaining) ===")
        final_deadline = time.monotonic() + min(remaining - 10, 120)
        f_opt, sc_opt = _optimize_main(best_f, n, final_deadline, rng,
                                        top_k_start=1, label="final")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Final: {best_score:.6f}")
    
    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score
    
    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
