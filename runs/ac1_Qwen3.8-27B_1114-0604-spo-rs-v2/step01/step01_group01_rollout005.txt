"""
Game-theoretic (max-min) approach to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses multiplicative weights on the dual variable with warm-started QP solves.

import numpy as np
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    nout = len(a) + len(b) - 1
    if nout <= 0:
        return np.array([], dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _hankel_matvec(q, a):
    """(H(q) @ a)[i] = sum_j q[i+j] * a[j]"""
    n = len(a)
    a_rev = a[::-1]
    L = _next_pow2(len(q) + n - 1)
    fq = np.fft.rfft(q, L)
    fa = np.fft.rfft(a_rev, L)
    conv = np.fft.irfft(fq * fa, L)
    return conv[n-1:2*n-1].astype(float, copy=False)

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _solve_qp_fista(q, a_init, max_iter=300, tol=1e-10):
    """Solve min a^T H(q) a s.t. a in simplex using FISTA."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Estimate L (Lipschitz constant of gradient = 2 * lambda_max(H))
    # Use power iteration for a rough estimate
    v = a.copy()
    for _ in range(10):
        w = _hankel_matvec(q, v)
        norm = np.linalg.norm(w)
        if norm < 1e-15:
            L = 1.0
            break
        v = w / norm
    lam_max = v @ _hankel_matvec(q, v)
    if lam_max <= 0:
        lam_max = 1.0
    L = 2.0 * lam_max * 1.1  # small safety factor
    
    # FISTA
    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    obj_prev = a @ _hankel_matvec(q, a)
    
    for k in range(max_iter):
        grad = 2.0 * _hankel_matvec(q, y)
        a_new = _project_simplex(y - (1.0/L) * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y / y_sum
        
        a_prev = a
        a = a_new
        t = t_new
        
        obj_curr = a @ _hankel_matvec(q, a)
        if abs(obj_curr - obj_prev) < tol * max(1.0, abs(obj_prev)):
            obj_prev = obj_curr
            break
        obj_prev = obj_curr
    
    return a
