import numpy as np
import time
from scipy.optimize import minimize

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def logsumexp(x, tau):
    """Compute (1/tau) * log(sum(exp(tau * x)))"""
    x_max = np.max(x)
    if x_max > 1e100:
        return np.inf
    return x_max + np.log(np.sum(np.exp(tau * (x - x_max)))) / tau

def objective_and_grad(b, tau, n):
    """Objective and gradient for softmax parameterization.
    b is the free variable, a = softmax(b)."""
    # Compute softmax
    b_max = np.max(b)
    exp_b = np.exp(b - b_max)
    a = exp_b / exp_b.sum()
    
    # Compute convolution
    c = fast_self_conv(a)
    
    # Objective: (1/tau) * log(sum(exp(tau * c)))
    c_max = np.max(c)
    logits = tau * (c - c_max)
    exp_logits = np.exp(logits)
    obj = c_max + np.log(np.sum(exp_logits)) / tau
    
    # Gradient of objective w.r.t. c
    w = exp_logits / np.sum(exp_logits)  # weights for each convolution peak
    
    # Gradient of conv w.r.t. a
    # d(conv[k])/d(a[m]) = 2*a[k-m] if k-m in [0,n-1]
    grad_a = np.zeros(n)
    for i in range(len(c)):
        if w[i] < 1e-15:
            continue
        ki = i
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            grad_a[j_start:j_end+1] += w[i] * 2.0 * a[j_start:j_end+1][::-1]
    
    # Gradient w.r.t. b: chain rule through softmax
    # da_i/db_j = a_i * (delta_ij - a_j)
    # dF/db_j = sum_i dF/da_i * a_i * (delta_ij - a_j)
    #          = a_j * (dF/da_j) - a_j * sum_i dF/da_i * a_i
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)
    
    return obj, grad_b

    grad_a = np.zeros(n)
    threshold = 1e-12
    for ki in range(len(c)):
        wi = w[ki]
        if wi < threshold:
            continue
        j_lo = max(0, ki - n + 1)
        j_hi = min(ki, n - 1)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            grad_a[indices] += wi * 2.0 * a[ki - indices]
    
    # Chain rule through softmax: dF/db = a * (dF/da - dot(dF/da, a))
    dot = float(np.dot(grad_a, a))
    grad_b = a * (grad_a - dot)
    
    return obj, grad_b


def _scipy_obj(b):
    """Wrapper for scipy (uses global tau and n)."""
    return lf_obj_grad(b, _G_tau, _G_n)[0]


def _scipy_grad(b):
    """Wrapper for scipy (uses global tau and n)."""
    return lf_obj_grad(b, _G_tau, _G_n)[1]


def run_lf_lbfgs(b0, tau, n, maxiter=300, time_limit=60.0):
    """
    Run L-BFGS-B on the smoothed problem.
    Returns (a_optimized, b_optimized, obj_value).
    """
    global _G_tau, _G_n
    _G_tau = tau
    _G_n = n
    
    t0 = time.monotonic()
    
    # Initial evaluation
    obj0, _ = lf_obj_grad(b0, tau, n)
    
    result = minimize(_scipy_obj, b0, jac=_scipy_grad, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                               'ftol': 1e-12, 'gtol': 1e-8})
    
    b_opt = result.x

    obj_opt, _ = lf_obj_grad(b_opt, tau, n)

    return a_opt, b_opt, obj_opt, elapsed


def init_uniform(n):
    return np.ones(n) / n


def init_random(n, rng):
    x = rng.random(n) + 0.01
    return x / x.sum()


def init_cosine(n, freq, alpha, rng):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x)
    a = np.maximum(a, 0.01) + 0.01 * rng.random(n)
    return a / a.sum()


def init_ramp(n, alpha):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * x - 1.0)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_block(n, nb, rng):
    per = n // nb
    vals = 0.3 + 0.7 * rng.random(nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    a = a[:n]
    a = a / a.sum()
    return a


def init_bipolar(n, rng):
    """Two-block structure."""
    a = np.zeros(n)
    a[:n//3] = 1.0
    a[n//2:] = 1.0
    a += 0.01 * rng.random(n)
    return a / a.sum()


def init_parent_scaled(parent, n, rng):
    """Resample parent to length n (with interpolation or truncation)."""
    if len(parent) == n:
        a = parent.copy()
    elif len(parent) > n:
        # Truncate
        a = parent[:n].copy()
    else:
        # Upsample via linear interpolation
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def perturb(a, scale, rng):
    """Perturb a and renormalize."""
    b = a * (1.0 + scale * rng.standard_normal(len(a)))
    b = np.maximum(b, 1e-10)
    return b / b.sum()


def peak_reduce(a, n_steps=100, step_size=0.02, rng=None):
    """
    Local peak reduction: find the max peak of conv, identify contributing
    pairs, and reduce them slightly while increasing others.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    best = a.copy()
    best_score = score_from_conv(n, fast_self_conv(best), best.sum())
    
    for step in range(n_steps):
        c = fast_self_conv(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        s = a.sum()
        score = score_from_conv(n, c, s)
        if score < best_score:
            best = a.copy()
            best_score = score
        
        # Find which a[j] contribute most to the peak
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(peak_idx, n - 1)
        # Contribution of a[j] to conv[peak_idx] is 2*a[peak_idx-j]
        contributions = 2.0 * a[peak_idx - np.arange(j_lo, j_hi + 1)]
        
        # Reduce the top contributors, increase random others
        top_k = max(1, (j_hi - j_lo + 1) // 10)
        top_indices = np.argsort(contributions)[-top_k:] + j_lo
        bottom_indices = np.setdiff1d(np.arange(n), top_indices)
        # Pick a few to increase
        n_inc = max(1, len(bottom_indices) // 20)
        inc_indices = rng.choice(bottom_indices, size=n_inc, replace=False)
        
        # Reduce top, increase bottom
        total_reduce = step_size * a[top_indices].sum()
        a[top_indices] -= (total_reduce / len(top_indices)) * a[top_indices] / max(a[top_indices].max(), 1e-10)
        a[inc_indices] += total_reduce / len(inc_indices) * a[inc_indices].mean() / max(a[inc_indices].mean(), 1e-10)
        
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    
    return best, best_score


def anneal_optimize(a0, n, rgs, time_limit, tau_start=5.0, tau_end=2000.0, n_tau_steps=8):
    """
    Continuation method: anneal tau from tau_start to tau_end,
    running L-BFGS-B at each level.
    """
    a = a0.copy()
    b = np.log(np.maximum(a, 1e-30))
    b -= np.mean(b)  # center for numerical stability
    
    # Log-spaced tau values
    taus = np.geomspace(tau_start, tau_end, n_tau_steps)
    
    elapsed_total = 0.0
    for tau in taus:
        if time.monotonic() > time_limit:
            break
        remaining = time_limit - time.monotonic() - 2.0
        if remaining < 3.0:
            break
        time_alloc = min(remaining / (n_tau_steps - int(np.searchsorted(taus, tau)) ), 60.0)
        time_alloc = max(3.0, time_alloc)
        
        a, b, obj_lf, elapsed = run_lf_lbfgs(b, tau, n, maxiter=200, time_limit=time_alloc)
        elapsed_total += elapsed
    
    # Final evaluation
    c = fast_self_conv(a)
    score = score_from_conv(n, c, a.sum())
    return a, score
