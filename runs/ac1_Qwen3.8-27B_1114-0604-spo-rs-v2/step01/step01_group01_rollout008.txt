"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using peak-targeted multiplicative descent
with mirror-masking, multi-scale refinement, and swap-based local search.
"""

import numpy as np
import time

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    result = np.fft.irfft(fa * fb, L)
    return result[:n]

def _convolve_sq(a):
    """Compute a convolved with a using FFT."""
    return _fft_convolve(a, a)

def _score(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    r = _convolve_sq(a)
    return 2.0 * n * np.max(r) / (s * s)

def _peak_mask(a, k, n):
    """Compute the mirror mask for peak at position k."""
    # mask[j] = a[k-j] if 0 <= k-j < n, else 0
    mask = np.zeros(n)
    valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
    idx = k - np.arange(n)
    mask[valid] = a[idx[valid]]
    return mask

def _soft_peak_mask(a, r, top_k=5, beta=10.0, n=None):
    """Weighted combination of masks for top peaks."""
    if n is None:
        n = len(a)
    # Get top_k peak positions
    indices = np.argsort(r)[::-1][:top_k]
    weights = np.exp(beta * (r[indices] - r[indices[0]]))
    weights = weights / weights.sum()
    
    mask = np.zeros(n)
    for w, k in zip(weights, indices):
        mask += w * _peak_mask(a, k, n)
    return mask

def _normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def _peak_descent(a, rng, time_limit, beta=10.0, top_k=5):
    """Peak-targeted multiplicative descent."""
    n = len(a)
    a = _normalize(a.copy())
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    
    deadline = time.monotonic() + time_limit
    alpha = 0.5
    stagnant = 0
    
    while time.monotonic() < deadline:
        r = _convolve_sq(a)
        mask = _soft_peak_mask(a, r, top_k=top_k, beta=beta, n=n)
        max_mask = mask.max()
        
        if max_mask < 1e-15:
            break
        
        mask_norm = mask / max_mask
        
        # Multiplicative update
        a_new = a * np.exp(-alpha * mask_norm)
        a_new = _normalize(a_new)
        
        new_score = _score(a_new)
        
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            stagnant = 0
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
        else:
            stagnant += 1
            alpha *= 0.7
            if alpha < 0.01:
                # Reset alpha and perturb
                alpha = 0.5
                noise = np.exp(rng.normal(0, 0.1, n))
                a = _normalize(a * noise)
                cur_score = _score(a)
                if cur_score < best_score:
                    best = a.copy()
                    best_score = cur_score
                stagnant = 0
        
        # Adaptively increase beta (sharpen focus on peaks)
        if stagnant % 10 == 0:
            beta = min(beta * 1.5, 100.0)
        
        if stagnant > 200:
            break
    
    return best, best_score

def _swap_search(a, rng, time_limit):
    """Try swapping pairs of coordinates to reduce the peak."""
    n = len(a)
    a = _normalize(a.copy())
    best = a.copy()
    best_score = _score(a)

        # Products are largest where both a[i] and a[k_star-i] are large

        # and try to reduce one of them

        j = int(rng.integers(0, n))
        if i == j:
            continue

        a_new[i], a_new[j] = a[j], a[i]

        else:

Key insight: The gradient of r[k] w.r.t. a[j] is 2*a[k-j] (if valid), so

import numpy as np

def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    L = 1
    while L < n_out:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    result = np.fft.irfft(fa * fb, L)
    return result[:n_out]


def _conv_sq(a):
    """Compute a convolved with a."""
    return _fft_convolve(a, a)


def _score_fast(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    r = _conv_sq(a)
    return 2.0 * n * float(np.max(r)) / (s * s)


def _normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _peak_mask(a, k, n):
    """Mirror mask: mask[j] = a[k-j] if 0<=k-j<n else 0."""
    j = np.arange(n)
    mirror = k - j
    valid = (mirror >= 0) & (mirror < n)
    mask = np.zeros(n)
    mask[valid] = a[mirror[valid]]
    return mask


def _soft_mask(a, r, top_k=5, beta=10.0):
    """Weighted combination of peak masks for top_k peaks."""
    n = len(a)
    # Get top peaks
    if len(r) > top_k:
        top_idx = np.argpartition(r, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(r))
    top_vals = r[top_idx]
    weights = np.exp(beta * (top_vals - top_vals.max()))
    weights /= weights.sum()

    mask = np.zeros(n)
    for w, k in zip(weights, top_idx):
        mask += w * _peak_mask(a, k, n)
    return mask

    # Sine-modulated
    for eps in [0.3, 0.6]:
        a = 1.0 + eps * np.cos(2 * np.pi * t)
        inits.append(_normalize(a))

    # Random smooth
    for _ in range(3):
        a = np.exp(rng.normal(0, 1.5, n))
        # Smooth with Gaussian kernel
        sigma = 5
        k = np.arange(-3 * sigma, 3 * sigma + 1)
        kern = np.exp(-0.5 * (k / sigma) ** 2)
        kern /= kern.sum()
        a = np.convolve(a, kern, mode='same')
        inits.append(_normalize(a))

    # Asymmetric bell
    c = 0.35
    s = 0.2
    a = np.exp(-((t - c) / s) ** 2)
    inits.append(_normalize(a))

    # Random piecewise constant
    K = int(rng.integers(3, 7))
    edges = np.sort(rng.uniform(0, 1, K - 1))
    edges = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1, K)
    a = np.zeros(n)
    for i in range(K):
        s_idx = int(edges[i] * n)
        e_idx = int(edges[i + 1] * n)
        a[s_idx:e_idx] = heights[i]
    inits.append(_normalize(a))

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses peak-targeted multiplicative descent, block scaling, and peak surgery.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 40  # Reserve time for finalization

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.array(height_sequence_1, dtype=float)
        parent = _normalize(parent)
        print(f"Parent loaded: n={len(parent)}, score={_score_fast(parent):.6f}")
    except:
        parent = None

    best_arr = None
    best_score = float('inf')

    def update_best(arr, score, tag=""):
        nonlocal best_arr, best_score
        if score < best_score:
            best_arr = arr.copy()
            best_score = score
            t = time.monotonic() - start_time
            print(f"  *** NEW BEST: {score:.6f} (n={len(arr)}) {tag} at t={t:.0f}s")

    # === Phase 1: Quick exploration at n=1024 ===
    print("\n=== Phase 1: Exploration (n=1024) ===")
    n1 = 1024
    starts = []

    if parent is not None:
        starts.append(_resample(parent, n1))

    inits = _make_initializations(n1, rng)
    starts.extend(inits)

    # Score all starts
    scored = []
    for s in starts:
        sc = _score_fast(s)
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    print(f"  Best initial: {scored[0][0]:.6f}")

    # Short peak descent on top 4 starts
    for i in range(min(4, len(scored))):
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        t_alloc = min(30.0, rem * 0.2)
        if t_alloc < 5:
            break
        result, sc = _peak_descent(scored[i][1], rng, t_alloc, beta_init=3.0, top_k=5)
        update_best(result, sc, f"phase1_{i}")

    # === Phase 2: Main optimization at n=2048 ===
    print("\n=== Phase 2: Main optimization (n=2048) ===")
    n2 = 2048
    main_starts = []

    if parent is not None:
        main_starts.append(_resample(parent, n2))

    for sc, arr in scored[:3]:
        main_starts.append(_resample(arr, n2))

    # Also add some fresh inits at n2
    for s in _make_initializations(n2, rng)[:3]:
        main_starts.append(s)

    # Score and sort
    main_scored = []
    for s in main_starts:
        sc = _score_fast(s)
        main_scored.append((sc, s))
    main_scored.sort(key=lambda x: x[0])
    print(f"  Best main start: {main_scored[0][0]:.6f}")

    # Run peak descent on best starts with more time
    time_per_run = max(60.0, min(180.0, (deadline - time.monotonic()) / 5.0))
    for i in range(min(3, len(main_scored))):
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        t_alloc = min(time_per_run, rem * 0.3)
        if t_alloc < 20:
            break
        print(f"  Run {i}: start_score={main_scored[i][0]:.6f}, t_alloc={t_alloc:.0f}s")
        result, sc = _peak_descent(main_scored[i][1], rng, t_alloc, beta_init=5.0, top_k=5)
        update_best(result, sc, f"phase2_{i}")

    # Block descent on best
    if best_arr is not None and time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        t_alloc = min(120.0, rem * 0.2)
        if t_alloc > 20:
            print(f"  Block descent: t_alloc={t_alloc:.0f}s")
            result, sc = _block_descent(best_arr, rng, t_alloc)
            update_best(result, sc, "phase2_block")

    # === Phase 3: Refinement ===
    print("\n=== Phase 3: Refinement ===")
    if best_arr is not None:
        # Peak surgery
        rem = deadline - time.monotonic()
        t_alloc = min(120.0, rem * 0.15)
        if t_alloc > 20:
            print(f"  Peak surgery: t_alloc={t_alloc:.0f}s")
            result, sc = _peak_surgery(best_arr, rng, t_alloc)
            update_best(result, sc, "phase3_surgery")

        # More peak descent with higher beta
        rem = deadline - time.monotonic()
        t_alloc = min(150.0, rem * 0.2)
        if t_alloc > 20:
            print(f"  Peak descent (beta=20): t_alloc={t_alloc:.0f}s")
            result, sc = _peak_descent(best_arr, rng, t_alloc, beta_init=20.0, top_k=3)
            update_best(result, sc, "phase3_hibeta")

        # Block descent again
        rem = deadline - time.monotonic()
        t_alloc = min(100.0, rem * 0.1)
        if t_alloc > 20:
            result, sc = _block_descent(best_arr, rng, t_alloc)
            update_best(result, sc, "phase3_block")

    # === Phase 4: Try larger size ===
    print("\n=== Phase 4: Larger size ===")
    if best_arr is not None and time.monotonic() < deadline - 200:
        n3 = 4096
        big_a = _resample(best_arr, n3)
        rem = deadline - time.monotonic()
        t_alloc = min(200.0, rem * 0.3)
        if t_alloc > 30:
            print(f"  Upscale to n={n3}: t_alloc={t_alloc:.0f}s")
            result, sc = _peak_descent(big_a, rng, t_alloc, beta_init=5.0, top_k=5)
            update_best(result, sc, "phase4_big")

            rem = deadline - time.monotonic()
            if rem > 60:
                t_alloc2 = min(60.0, rem * 0.3)
                result2, sc2 = _block_descent(result, rng, t_alloc2)
                update_best(result2, sc2, "phase4_big_block")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final polish ===")
    if best_arr is not None:
        # Very focused peak descent
        rem = deadline - time.monotonic()
        t_alloc = min(100.0, rem * 0.3)
        if t_alloc > 15:
            print(f"  Final peak descent: t_alloc={t_alloc:.0f}s")
            result, sc = _peak_descent(best_arr, rng, t_alloc, beta_init=30.0, top_k=2)
            update_best(result, sc, "phase5_focus")

        # Peak surgery
        rem = deadline - time.monotonic()
        t_alloc = min(80.0, rem * 0.25)
        if t_alloc > 15:
            result, sc = _peak_surgery(best_arr, rng, t_alloc)
            update_best(result, sc, "phase5_surgery")

        # One more block descent
        rem = deadline - time.monotonic()
        t_alloc = min(60.0, rem * 0.2)
        if t_alloc > 10:
            result, sc = _block_descent(best_arr, rng, t_alloc)
            update_best(result, sc, "phase5_block")

    # === Finalize ===
    print("\n=== Finalizing ===")
    if best_arr is None:
        # Fallback
        best_arr = np.ones(n2) / n2
        best_score = 2.0
        print("  Using uniform fallback!")

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_arr]
    try:
        eval_sc = evaluate_sequence(final_list)
        print(f"  Internal score: {best_score:.6f}")
        print(f"  Evaluated score: {eval_sc:.6f}")
    except Exception as e:
        print(f"  Evaluation error: {e}")

    # Ensure all values are valid
    final_list = [max(0.0, min(1000.0, float(x))) for x in final_list]
    s = sum(final_list)
    if s < 0.01:
        final_list = [1.0 / 2048] * 2048

    print(f"\nFinal: n={len(final_list)}, sum={sum(final_list):.4f}")
    return final_list
