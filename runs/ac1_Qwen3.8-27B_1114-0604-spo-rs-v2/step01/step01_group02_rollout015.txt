"""
Adaptive gradient optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses a blend of exact subgradient and soft-max gradient with adaptive weighting
based on peak separation. Multiple refinement phases with decreasing learning rate.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def grad_adaptive(a, c, topk, temp_scale=0.1):
    """
    Compute adaptive gradient blending exact subgradient (for dominant peak)
    with soft-max gradient (for nearby peaks).
    Returns (gradient, weighted_max, sum).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)

    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]

    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0

    if top1 > 1e-15:
        ratio = top2 / top1
    else:
        ratio = 1.0

    # If top peak dominates, use more exact gradient
    exact_w = float(np.clip((0.9 - ratio) / 0.4, 0.0, 1.0))

    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_scale * top1, 1e-15)
        wts_smooth = np.exp((vals - top1) / temp)
        wts_smooth = wts_smooth / wts_smooth.sum()
        if exact_w < 0.01:
            wts = wts_smooth
        else:
            wts_exact = np.zeros(k_act)
            wts_exact[0] = 1.0
            wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
            wts = wts / wts.sum()
        M = float(np.dot(vals, wts))

    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S

    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
