```
def fft_convolution(a):
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.empty(0, dtype=np.float64)
    need = 2*n - 1
    if n == 1: return np.array([a[0]*a[0]], dtype=np.float64)
    size=1
    while size < need: size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size, n=need)
    b = np.ascontiguousarray(b, dtype=np.float64)
    # Clip small negative; do not clip large negative (shouldn't)
    b = np.maximum(b, 0.0)
    return b
```

```
def corr_pw(p,w):
```

```
def random_smear_z(n,rng,strength=1.0):
    if n<=1: return np.zeros(n)
    half=n//2+1
    freq=np.fft.rfftfreq(n)
    K=max(16.0, float(np.sqrt(n))*1.5)
    decay=np.exp(-np.square(freq*n/K))
    amp_r=rng.normal(0.0,1.0,half)*decay
    amp_i=rng.normal(0.0,1.0,half)*decay
    amp_r[0]=0.0
    zc=amp_r+1j*amp_i
    z=np.fft.irfft(zc,n)
    z -= float(np.mean(z))
    z *= strength
    z = 0.8*np.tanh(z)
    return z
```

```python
"""
Search non-negative sequences minimizing
    2*n*max(p*p)/(sum(p))**2
Using FFT-based smooth-max gradient flow in log coordinates.

Main ideas:
- Normalize by the simplex with a softmax/log representation, so non-negativity
  and sum constraints are automatic.
- Minimize a smooth logistic approximation of max(p*p) with L-BFGS-B in the
  log variables.  Several beta values are used as annealing.
- Generate many asymmetric starts, especially endpoint-power profiles around
  p_i ~ 1/sqrt(i+1), which is a useful continuous-optimum shape.
- Keep multiple final candidate lengths (including down-/upsampled versions)
  and return the one with best fast FFT score.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

CLIP_Z = 50.0
INF = float("inf")


def safe_normalize(arr):
    """Return a finite, non-negative vector normalized to sum 1."""
    a = np.asarray(arr, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)

    a = a / s
    a = np.minimum(a, 1000.0)

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)

    a = a / s
    a = np.maximum(a, 0.0)
    return np.ascontiguousarray(a, dtype=np.float64)


def fft_convolution(a):
    """Linear convolution a*a, using FFT.  Tiny negative roundoff is clipped."""
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size, n=need)
    b = np.ascontiguousarray(b, dtype=np.float64)
    b = np.maximum(b, 0.0)
    return b


def smooth_max_and_weights(b, beta):
    """
    Smooth maximum G = beta^{-1} log sum exp(beta b_k), and softmax weights.
    G approximates max(b) when beta is large.
    """
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    if b.size == 0:
        return 0.0, np.empty(0, dtype=np.float64)

    m = float(np.max(b))
    if not np.isfinite(m):
        return m, np.zeros(b.size, dtype=np.float64)

    if m <= 0.0:
        w = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
        return 0.0, w

    x = beta * (b - m)
    x = np.minimum(x, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros(b.size, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    w = e / Z
    G = m + float(np.log(Z)) / beta
    if not np.isfinite(G):
        G = m

    return float(G), w


def corr_pw(p, w):
    """
    Return s_i = sum_j p_j w_{i+j}, for i=0..n-1.
    w has length 2n-1.
    """
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([float(p[0] * w[0])], dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size, n=need)

    # C_{m-1-i} = s_i
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def fast_score_raw(a):
    """
    Fast objective for a non-negative vector.  It is scale invariant, so
    normalization is not required.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return INF

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return INF

    b = fft_convolution(a)
    if b.size == 0:
        return INF

    m = float(np.max(b))
    if not np.isfinite(m):
        return INF

    return float(2.0 * a.size * m / (s * s))


def score_candidate(arr):
    """Score after safe sanitization/normalization."""
    a = safe_normalize(arr)
    return fast_score_raw(a)


def canonical_z(z):
    """Canonical log coordinates: finite, max zero, clipped."""
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size == 0:
        return z

    z = np.where(np.isfinite(z), z, 0.0)
    mx = float(np.max(z))
    if not np.isfinite(mx):
        return z

    z = z - mx
    z = np.clip(z, -CLIP_Z, 0.0)
    return z


def make_p_from_z(z):
    """Convert log coordinates to a probability vector."""
    n = np.size(z)
    if n == 0:
        return np.empty(0, dtype=np.float64)

    z = canonical_z(z)
    p = np.exp(z)
    s = float(np.sum(p))

    if s <= 0.0 or not np.isfinite(s):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        p = p / s

    return np.ascontiguousarray(p, dtype=np.float64)


def canonical_log(p):
    """Convert a probability vector to canonical log coordinates."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)

    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        p = p / s

    p = np.maximum(p, np.finfo(np.float64).tiny)
    z = np.log(p)
    return canonical_z(z)


def objective_log(c, n, beta):
    """
    Smooth objective and gradient in log coordinates.

    Minimize q(p) = n * G(p), where G is smooth max of p*p.
    Gradient returns (value, grad_wrt_log_coords).
    """
    z = canonical_z(c)
    if z.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(z.size, n)
        tmp[:m] = z[:m]
        z = tmp

    p = np.exp(z)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        p = p / s

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return 1e100, np.zeros(n, dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.where(np.isfinite(svec), svec, 0.0)

    C = float(np.dot(p, svec))
    if not np.isfinite(C):
        C = G

    # dG/dp_i = 2 * sum_j w_{i+j} p_j
    grad_p = 2.0 * p * (svec - C)
    # d/dz = p * (grad_p - sum(p*grad_p)) = p*(2s - 2C)
    grad_z = p * (svec - C)
    grad_z *= 2.0

    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)

    val = float(n) * G
    grad = float(n) * grad_z
    return val, grad


def run_lbfgs(z0, n, beta, max_iter):
    """Run L-BFGS-B on log coordinates, robustly."""
    if not HAVE_SCIPY or max_iter <= 0:
        return canonical_z(z0)

    c0 = canonical_z(z0)
    try:
        res = minimize(
            objective_log,
            c0,
            args=(n, beta),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": int(max_iter),
                "ftol": 1e-12,
                "gtol": 1e-12,
                "maxcor": 10,
                "maxls": 40,
                "disp": False,
            },
        )
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0

    c = canonical_z(c)
    return c


def beta_schedule(n):
    """Smooth_max beta values and L-BFGS iteration counts."""
    nf = float(n)
    betas = [
        max(100.0, nf / 10.0),
        max(500.0, nf / 2.0),
        nf,
        5.0 * nf,
        20.0 * nf,
        80.0 * nf,
        240.0 * nf,
    ]

    # Ensure strictly increasing.
    for i in range(1, len(betas)):
        if betas[i] <= betas[i - 1]:
            betas[i] = betas[i - 1] * 1.5

    if n >= 24576:
        iters = [56, 50, 44, 38, 30, 24, 20]
    elif n >= 8192:
        iters = [70, 62, 54, 46, 38, 30, 24]
    else:
        iters = [90, 80, 70, 60, 50, 40, 32]

    return betas, iters


def resample_sequence(arr, new_n):
    """Linearly resample a non-negative sequence to a new length."""
    a = safe_normalize(arr)
    old_n = a.size
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if new_n == 1:
        return np.array([1.0], dtype=np.float64)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)

    s = float(np.sum(b))
    if s <= 0.0 or not np.isfinite(s):
        b = np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    else:
        b = b / s

    return np.ascontiguousarray(b, dtype=np.float64)


def append_start(potes, arr):
    """Append a safe non-negative normalized start if valid."""
    a = np.asarray(arr, dtype=np.float64).reshape(-1)
    if a.size <= 0:
        return

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return

    a = a / s
    potes.append(np.ascontiguousarray(a, dtype=np.float64))


def random_smear_z(n, rng, strength=1.0):
    """A smooth random log-coordinate perturbation."""
    if n <= 1:
        return np.zeros(n, dtype=np.float64)

    half = n // 2 + 1
    freq = np.fft.rfftfreq(n)

    K = max(16.0, 1.5 * float(np.sqrt(n)))
    decay = np.exp(-np.square(freq * float(n) / K))

    amp_r = rng.normal(0.0, 1.0, half) * decay
    amp_i = rng.normal(0.0, 1.0, half) * decay
    amp_r[0] = 0.0

    zc = amp_r.astype(np.complex128) + 1j * amp_i.astype(np.complex128)
    z = np.fft.irfft(zc, n)

    z -= float(np.mean(z))
    z = 0.8 * np.tanh(z * strength)
    return z.astype(np.float64)


def generate_starts(n, rng, prev_arr):
    """Generate a diverse pool of starts, especially asymmetric power profiles."""
    starts = []

    idx = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Uniform fallback.
    append_start(starts, np.full(n, 1.0 / float(n), dtype=np.float64))

    # Core one-sided power profile.
    base = 1.0 / np.sqrt(idx + 1.0)
    append_start(starts, base)

    # Endpoint exponent scan.
    for alpha in (0.48, 0.50, 0.52, 0.54, 0.56, 0.58):
        append_start(starts, np.power(idx + 1.0, -alpha))

    # Offset variants.
    for c in (0.5, 1.0, 2.0, 4.0, 8.0):
        append_start(starts, 1.0 / np.sqrt(idx + c))

    # Exponential tilt of the power profile.
    for eta in (-2.0, -1.0, -0.5, -0.2, 0.2, 0.5, 1.0, 2.0):
        append_start(starts, base * np.exp(eta * x))

    # Quadratic modulation.
    for a, b in (
        (-1.0, 2.0),
        (-2.0, 4.0),
        (1.0, -2.0),
        (2.0, -4.0),
        (-1.0, -2.0),
        (1.0, 2.0),
    ):
        y = x - 0.5
        append_start(starts, base * np.exp(a * x + b * y * y))

    # Soft cutoffs for the tail.
    for lam in (0.2, 0.35, 0.5, 0.65):
        for cut in (1.0, 3.0):
            append_start(starts, base * np.exp(-cut * np.maximum(x - lam, 0.0)))

    # Tiny right-endpoint component.
    inv_right = 1.0 / np.sqrt(n - idx + 1.0)
    for eps in (0.03, 0.08, 0.15, 0.35):
        append_start(starts, base + eps * inv_right)

    # Low-frequency trigonometric modulation.
    for m in (1, 2, 3, 5):
        ang = 2.0 * np.pi * float(m) * idx / float(n)
        for amp in (0.2, 0.5, 0.9):
            append_start(starts, base * np.exp(amp * np.sin(ang)))
            append_start(starts, base * np.exp(amp * np.cos(ang)))

    # Smooth random perturbations of the power profile.
    for _ in range(12):
        z = random_smear_z(n, rng, strength=0.8)
        append_start(starts, base * np.exp(z))
        append_start(starts, np.exp(z))

    # Some noisier starts for diversity.
    for sig in (0.08, 0.2, 0.35):
        append_start(starts, base * (1.0 + sig * rng.normal(0.0, 1.0, n)))
    append_start(starts, base * (1.0 + 0.3 * random_smear_z(n, rng, strength=1.2)))
    append_start(starts, 1.0 + 0.35 * rng.normal(0.0, 1.0, n))
    append_start(starts, base + 0.5 * inv_right + 0.25 * rng.normal(0.0, 1.0, n))

    # Previous best construction, if available.
    if prev_arr is not None:
        prev = safe_normalize(prev_arr)
        append_start(starts, prev)

        prev_res = resample_sequence(prev, n)
        append_start(starts, prev_res)

        for sig in (0.05, 0.12, 0.25):
            noise = rng.normal(0.0, sig, n)
            append_start(starts, prev_res * np.exp(0.6 * noise))

    return starts


def diverse_top(scored, K):
    """Select at most K diverse sequences from scored candidates."""
    if K <= 0:
        return []

    selected = []
    for sc, _, p in scored:
        if len(selected) >= K:
            break

        dup = False
        for q in selected:
            if np.max(np.abs(p - q)) < 1e-7:
                dup = True
                break

        if not dup:
            selected.append(p)

    # Fill if not enough diverse items.
    if len(selected) < K:
        for sc, _, p in scored:
            if len(selected) >= K:
                break
            selected.append(p)

    return selected


def subgradient_polish(z, p, n, deadline, beta, rng):
    """
    Backtracking normalized log-gradient polish.  Used as a fallback and
    final micro-polisher.
    """
    if n <= 0:
        return z, np.empty(0, dtype=np.float64), np.array([1.0], dtype=np.float64), INF

    z = canonical_z(z)
    p = make_p_from_z(z)

    best_p = p.copy()
    b = fft_convolution(p)
    best_sc = fast_score_raw(p)
    if b.size == 0 or not np.isfinite(best_sc):
        best_sc = INF

    target_step = 0.25
    stall = 0
    it = 0
    max_iters = 1000000

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        if it % 400 == 0:
            print(f"subgradient polish it={it} best={best_sc:.10f} step={target_step:.3e}")

        b = fft_convolution(p)
        if b.size == 0:
            break

        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        Gcur = float(n) * G
        mcur = float(np.max(b))
        cur_sc = float(2.0 * n * mcur) if np.isfinite(mcur) else INF
        if np.isfinite(cur_sc) and cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p.copy()

        svec = corr_pw(p, w)
        svec = np.where(np.isfinite(svec), svec, 0.0)

        C = float(np.dot(p, svec))
        if not np.isfinite(C):
            C = G

        grad = p * (svec - C)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        maxg = float(np.max(np.abs(grad)))

        if not np.isfinite(maxg) or maxg < 1e-16:
            break

        alpha = target_step / maxg
        accepted = False

        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005):
            z2 = z - (alpha * mult) * grad
            z2 = canonical_z(z2)
            p2 = make_p_from_z(z2)

            b2 = fft_convolution(p2)
            G2, _ = smooth_max_and_weights(b2, beta)
            if not np.isfinite(G2):
                continue

            if float(n) * G2 < Gcur - 1e-13:
                z = z2
                p = p2
                G = G2
                m2 = float(np.max(b2))
                accepted_sc = float(2.0 * n * m2) if np.isfinite(m2) else INF

                if np.isfinite(accepted_sc) and accepted_sc < best_sc:
                    best_sc = accepted_sc
                    best_p = p.copy()

                accepted = True
                break

        if accepted:
            target_step = min(1.0, target_step * 1.05)
            stall = 0
        else:
            target_step *= 0.5
            if target_step < 1e-4:
                # Small smooth noise kick.
                noise = random_smear_z(n, rng, strength=0.25)
                z = canonical_z(z + 0.05 * noise)
                p = make_p_from_z(z)
                stall += 1
                if stall > 30:
                    break

    p = make_p_from_z(z)
    return z, p, best_p, best_sc


def continuation_refine(p0, n, deadline, rng, iter_scale=1.0, high_only=False):
    """
    Continue beta annealing / L-BFGS refinement from p0 until deadline.
    Returns best sequence and its fast FFT score.
    """
    p0 = safe_normalize(p0)
    if p0.size != n:
        p0 = resample_sequence(p0, n)

    best_p = p0.copy()
    best_sc = fast_score_raw(p0)
    if not np.isfinite(best_sc):
        best_sc = INF

    z = canonical_log(p0)
    betas, iters = beta_schedule(n)

    start_idx = max(0, len(betas) // 2) if high_only else 0
    iter_cost = 0.035 if n <= 16384 else 0.07

    for idx in range(start_idx, len(betas)):
        now = time.monotonic()
        if now >= deadline:
            break

        beta = float(betas[idx])
        rem = float(deadline - now)

        max_allowed = int(rem / max(1e-3, iter_cost))
        if max_allowed < 10:
            break

        maxit = int(float(iters[idx]) * float(iter_scale))
        if maxit <= 0:
            continue
        maxit = min(maxit, max_allowed)

        if HAVE_SCIPY:
            z = run_lbfgs(z, n, beta, maxit)
            p = make_p_from_z(z)
            sc = fast_score_raw(p)

            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p.copy()
                print(f"  lbfgs beta={beta:.3g} score={best_sc:.10f}")
        else:
            # Fallback without scipy.
            sub_deadline = time.monotonic() + max(1.0, min(rem * 0.9, 80.0))
            z, p, psub, scsub = subgradient_polish(z, p, n, sub_deadline, beta, rng)

            if np.isfinite(scsub) and scsub < best_sc:
                best_sc = scsub
                best_p = psub.copy()

    # Final sharpening stage.
    now = time.monotonic()
    if HAVE_SCIPY and now < deadline:
        rem = float(deadline - now)
        max_allowed = int(rem / max(1e-3, iter_cost))
        max_polish = min(int(70.0 * float(iter_scale)), max_allowed)
        if max_polish >= 10:
            beta = float(betas[-1])
            z = run_lbfgs(z, n, beta, max_polish)
            p = make_p_from_z(z)
            sc = fast_score_raw(p)

            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p.copy()
                print(f"  final lbfgs polish score={best_sc:.10f}")

    return best_p, best_sc

def propose_candidate(seed=42, budget_s=2400.0):
    """
    Search function.  Returns a list of non-negative floats.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if not np.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 2400.0

    if budget_s < 5.0:
        return [1.0]

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 60.0)

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.RandomState(seed)

    # Previous best, if exposed by the environment.
    prev_raw = globals().get("height_sequence_1", None)
    prev_arr = None
    if prev_raw is not None:
        try:
            prev_arr = safe_normalize(prev_raw)
            if prev_arr.size == 0:
                prev_arr = None
        except Exception:
            prev_arr = None

    # Choose main working length.
    if budget_s < 300.0:
        N = 4096
    elif budget_s < 900.0:
        N = 8192
    elif budget_s < 1800.0:
        N = 16384
    else:
        N = 32768

    print(f"budget={budget_s:.1f}s N={N} deadline_reserve=60s")

    starts = generate_starts(N, rng, prev_arr)
    scored = []

    for i, p in enumerate(starts):
        sc = fast_score_raw(p)
        if np.isfinite(sc):
            scored.append((sc, i, p))

    scored.sort()

    if len(scored) >= 3:
        print("best static starts:")
        for sc, _, _ in scored[:5]:
            print(f"  {sc:.10f}")

    if N >= 24576:
        K = 8
    elif N >= 8192:
        K = 10
    else:
        K = 12

    if budget_s >= 2000.0:
        K += 2
    K = min(K, len(scored))

    top = diverse_top(scored, K)
    if not top:
        top = [np.full(N, 1.0 / float(N), dtype=np.float64)]

    global_best = None
    global_best_sc = INF

    fracs = [0.32, 0.18, 0.12, 0.10, 0.08, 0.07, 0.06, 0.05, 0.04, 0.04, 0.03, 0.03]

    for i, p in enumerate(top):
        now = time.monotonic()
        remaining = float(deadline - now)
        if remaining < 30.0:
            break

        frac = fracs[i] if i < len(fracs) else 0.03
        alloc = remaining * frac

        # Keep at least a small search if there is total time.
        if remaining > 300.0 and alloc < 25.0:
            alloc = 25.0

        sub_deadline = now + max(2.0, alloc)

        if i == 0:
            scale = 1.15
        elif i < 4:
            scale = 1.0
        else:
            scale = 0.8

        p_fit, sc_fit = continuation_refine(p, N, sub_deadline, rng, iter_scale=scale)

        if np.isfinite(sc_fit) and sc_fit < global_best_sc:
            global_best_sc = sc_fit
            global_best = p_fit.copy()
            print(f"start {i} improved global: {global_best_sc:.10f}")

    if global_best is None:
        global_best = top[0].copy()
        global_best_sc = fast_score_raw(global_best)

    # Extra continuation on the global best, using high betas.
    if time.monotonic() < deadline:
        p_fit, sc_fit = continuation_refine(
            global_best, N, deadline, rng, iter_scale=1.0, high_only=True
        )
        if np.isfinite(sc_fit) and sc_fit < global_best_sc:
            global_best_sc = sc_fit
            global_best = p_fit.copy()
            print(f"high beta global polish: {global_best_sc:.10f}")

    # Final micro-polish.
    if time.monotonic() < deadline:
        betas, _ = beta_schedule(N)
        z0 = canonical_log(global_best)
        _z, _p, p_polish, sc_polish = subgradient_polish(
            z0, global_best, N, deadline, float(betas[-1]), rng
        )
        if np.isfinite(sc_polish) and sc_polish < global_best_sc:
            global_best_sc = sc_polish
            global_best = p_polish.copy()
            print(f"subgradient final polish: {global_best_sc:.10f}")

    # Build final candidate set.  Include previous construction and some
    # down-/upsampled versions, because the best length can differ from N.
    candidates = []
    candidates.append((global_best_sc if np.isfinite(global_best_sc) else INF, 0, global_best))

    if prev_arr is not None:
        sc_prev = fast_score_raw(prev_arr)
        candidates.append((sc_prev if np.isfinite(sc_prev) else INF, 1, prev_arr))

    max_final_len = 8192
    if budget_s >= 900.0:
        max_final_len = 16384
    if budget_s >= 1600.0:
        max_final_len = 32768

    target_lengths = (4096, 8192, 16384, 32768)
    cnt = 2
    for L in target_lengths:
        if L > max_final_len:
            continue
        if L == N:
            continue
        # Avoid extremely long final objects for small budgets.
        if L <= max(4096, 2 * N):
            try:
                cand = resample_sequence(global_best, L)
                sc = fast_score_raw(cand)
                if np.isfinite(sc):
                    candidates.append((sc, cnt, cand))
                    cnt += 1
            except Exception:
                pass

    best_seq = None
    best_val = INF
    for sc, _, arr in candidates:
        if not np.isfinite(sc):
            continue
        if sc < best_val:
            best_val = sc
            best_seq = arr

    if best_seq is None:
        best_seq = global_best if global_best is not None else np.array([1.0], dtype=np.float64)

    final_arr = safe_normalize(best_seq)

    # Safety fallback.
    if final_arr.size == 0:
        return [1.0]

    # Keep values finite and within evaluator caps.
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    final_arr = np.maximum(final_arr, 0.0)
    final_arr = np.minimum(final_arr, 1000.0)

    s = float(np.sum(final_arr))
    if s <= 0.0 or not np.isfinite(s):
        return [1.0]

    final_arr = final_arr / s
    final_arr = np.maximum(final_arr, 0.0)

    print(f"final length={final_arr.size} fast_score={best_val:.10f}")

    return [float(x) for x in final_arr]
```
