def _norm1(a):

    s = a.sum()
    if s < 1e-300:

        return np.ones(n) / n
    return a / s

def _fft_size(n):

def _conv_self(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c  # length L, first 2n-1 are valid

def _soft_grad(a, c, T, n, L):
    """
    Compute gradient of 2n*logsumexp(T*c)/T wrt a (for sum(a)=1).
    grad[j] = 4n * sum_k w[k] * a[k-j] where w = softmax(T*c)
    Computed via FFT: w cross_corr a
    """
    m = 2 * n - 1
    c_valid = c[:m]
    v = T * c_valid
    v -= v.max()
    e = np.exp(v)
    s = e.sum()
    w = e / s  # length m
    
    # Compute w cross_corr a: result[j] = sum_k w[k] * a[k-j] for valid indices
    # DFT: rfft(cross_corr) = conj(rfft(w)) * rfft(a)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    grad = np.fft.irfft(np.conj(W) * A, L)[:n]
    return 4.0 * n * grad

    m = 2 * n - 1
    return 2.0 * n * np.max(c[:m])

    v = T * cv
    v -= v.max()
    e = np.exp(v)

A proper gradient step would be `a <- normalize(a - eta * grad)`. The multiplicative version is more like a "mirror descent" step with the negative entropy as the mirror map.

For mirror descent, the update is:
`a_new = argmin_a { <grad, a> + D_KL(a || a_old) / eta }`
where `D_KL(a||b) = sum a[i] log(a[i]/b[i])`.

The solution is `a_new[i] = a_old[i] * exp(-eta * grad[i]) / Z`.

This is exactly the multiplicative update! So it's a valid (and in some sense superior) optimization method.

"""
Search for nonnegative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated mirror descent (multiplicative updates) with adaptive
log-sum-exp temperature, simulated annealing restarts, and multi-start strategy.
Key improvements: aggressive temperature annealing, backtracking line search,
periodic large jumps to escape local minima, and final fine-tuning phase.
"""

import numpy as np
import time
import math


def _norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _next_pow2(x):
    return 1 << (x - 1).bit_length()


def _conv_self(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c


def _softmax_w(c_valid, T):
    v = T * c_valid
    v -= v.max()
    e = np.exp(np.clip(v, -700.0, 0.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(c_valid) / len(c_valid)
    return e / s


def _grad_mult(a, w, n, L):
    """4n * (w cross_corr a) via FFT. cross_corr[j] = sum_k w[k]*a[k-j]."""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    return 4.0 * n * np.fft.irfft(W * np.conj(A), L)


def _score_max(c, n, m):
    return 2.0 * n * np.max(c[:m])


def _soft_score(c, T, n, m):
    cv = c[:m]
    v = T * cv
    vmax = v.max()
    e = np.exp(np.clip(v - vmax, -700.0, 0.0))
    return 2.0 * n * (vmax + math.log(max(e.sum(), 1e-300))) / T


def _gen_shapes(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    shapes = []
    # U-shapes
    for alpha in [0.3, 0.7, 1.2, 2.0, 3.5, 5.0]:
        shapes.append(1.0 + alpha * x * x)
    # Bimodal
    for c in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        shapes.append(np.exp(-30.0 * (t - c) ** 2) + np.exp(-30.0 * (t - (1.0 - c)) ** 2))
    # Sine harmonics
    for k in [1, 2, 3, 4]:
        for ph in [0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi]:
            shapes.append(1.0 + 0.6 * np.sin(2.0 * np.pi * k * t + ph))
    # Cosine hills
    for w in [0.08, 0.15, 0.25, 0.4]:
        shapes.append(np.cos(np.pi * (t - 0.5) / w) ** 2)
    # Random smooth
    for _ in range(12):
        z = rng.normal(0, 1, n)
        k = max(3, n // 40)
        z = np.convolve(z, np.ones(k) / k, mode='same')
        sd = z.std()
        if sd > 1e-10:
            z = 1.0 + 0.4 * (z - z.mean()) / sd
        else:
            z = np.ones(n)
        shapes.append(z)
    # Damped oscillation
    for freq in [3, 5, 8]:
        shapes.append(np.exp(-3.0 * (t - 0.5) ** 2) * (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t)))
    out = []
    for s in shapes:
        s = np.maximum(s, 0.005)
        out.append(s / s.sum())
    return out


def _perturb(a, rng, frac):
    noise = rng.standard_normal(len(a)) * frac
    a2 = a + noise
    return _norm1(a2)


def _optimize(a_init, L, m, n, rng, time_limit, T0=8.0, T1=0.2):
    """Core mirror descent optimizer with adaptive temperature and annealing."""
    a = _norm1(a_init.astype(np.float64).copy())
    deadline = time.monotonic() + time_limit

    c = _conv_self(a, L)
    best_a = a.copy()
    best_score = _score_max(c, n, m)

    eta = 0.8
    no_improve = 0
    it = 0
    est_total = int(time_limit * 80)  # rough
    est_total = max(est_total, 500)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.5:
            break

        frac = min(1.0, it / float(est_total))
        T = T0 * (T1 / T0) ** frac
        T = max(T, 0.05)

        c = _conv_self(a, L)
        cur = _score_max(c, n, m)

        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Stagnation: jump
        if no_improve > 150:
            jump_size = 0.03 + 0.12 * rng.random()
            a = _perturb(best_a, rng, jump_size)
            no_improve = 0
            eta = 0.8
            it = min(it, est_total // 3)
            continue

        # Compute gradient via soft weights
        w = _softmax_w(c[:m], T)
        g = _grad_mult(a, w, n, L)
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-300:
            break
        g = g / gmax

        # Line search (multiplicative)
        accepted = False
        for step in (eta, 0.5 * eta, 0.25 * eta, 0.1 * eta):
            if step < 1e-10:
                continue
            a2 = a * np.exp(-step * g)
            a2 = _norm1(a2)
            c2 = _conv_self(a2, L)
            sc2 = _score_max(c2, n, m)
            if sc2 < cur:
                a = a2
                accepted = True
                break

        if accepted:
            eta = min(3.0, eta * 1.03)
        else:
            eta = max(0.02, eta * 0.65)
            # Simulated annealing: occasionally accept worse
            if T > 1.0 and rng.random() < 0.05:
                a2 = a * np.exp(-eta * g)
                a = _norm1(a2)

        it += 1

    # Final: check best_a
    c_final = _conv_self(best_a, L)
    sc_final = _score_max(c_final, n, m)
    if sc_final < best_score:
        best_score = sc_final
    return best_a, best_score


def _fine_tune(a, L, m, n, rng, time_limit):
    """Fine-tuning with very low temperature."""
    a = _norm1(a.astype(np.float64).copy())
    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_score = _score_max(_conv_self(a, L), n, m)
    eta = 0.1
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.3:
            break
        T = 0.1 + 0.2 * max(0.0, 1.0 - it / 2000.0)
        c = _conv_self(a, L)
        cur = _score_max(c, n, m)
        if cur < best_score - 1e-13:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            a = best_a.copy()
            # Small perturbation
            a = _perturb(a, rng, 0.01)
            no_improve = 0
            eta = 0.1
            continue
        w = _softmax_w(c[:m], T)
        g = _grad_mult(a, w, n, L)
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-300:
            break
        g = g / gmax
        accepted = False
        for step in (eta, 0.5 * eta, 0.25 * eta):

                accepted = True
                break
        if accepted:
            eta = min(0.5, eta * 1.02)
        else:
            eta = max(0.005, eta * 0.7)
        it += 1
