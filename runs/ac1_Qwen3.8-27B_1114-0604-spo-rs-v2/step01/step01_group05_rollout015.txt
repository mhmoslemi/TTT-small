    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_and_weights(b, beta):
    """Smooth max of b with softmax weights for gradient computation."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w

def fast_score(p):
    """Fast local objective: 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))

def corr_pw(p, w):
    """
    Compute s_i = sum_j p[j] * w[j+i] for i=0..n-1.
    This gives the subgradient direction of max(p*p) with weights w.
    Uses FFT-based cross-correlation.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = C[m-1-i] for i = 0..n-1
    s = C[m - n : m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)

def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q

def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p

def compute_gradient(p, w):
    """Compute the projected gradient direction (zero-mean by construction)."""
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    grad = g - float(np.mean(g))
    return grad

def momentum_refine(p0, deadline, beta_start=5000.0, beta_end=2000000.0,
                    momentum=0.85, max_iter=10**8):
    """
    Projected gradient with heavy-ball momentum and beta annealing.
    Returns (best_p, best_score).
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)
    p_prev = p.copy()

    step = 1.0
    no_improve = 0
    it = 0
    start_t = time.monotonic()
    anneal_iters = 50000

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Beta annealing: exponential from beta_start to beta_end
        if it < anneal_iters:
            frac = it / anneal_iters
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_max = float(np.max(b))
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            elapsed = time.monotonic() - start_t
            print(f"  iter={it} beta={beta:.0f} best={best_score:.8f} "
                  f"step={step:.3e} noimp={no_improve} t={elapsed:.0f}s")

        if no_improve > 8000 and it > anneal_iters:
            break
        if no_improve > 3000 and beta >= beta_end * 0.9:
            break

        grad = compute_gradient(p, w)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Heavy-ball momentum: p_new = project(p - lr*grad + momentum*(p - p_prev))
        delta = p - p_prev
        p_new = project_simplex(p - step * grad + momentum * delta, 1.0)

        # Evaluate the new point
        b_new = fft_convolution(p_new)
        score_new = 2.0 * n * float(np.max(b_new))

        # Accept if not too much worse (allow some increase for momentum)
        if score_new < cur_score + 0.05 * (best_score - 1.0):
            p_prev = p
            p = p_new
            # Slowly increase step if making progress
            if no_improve == 0 and step < 3.0:
                step = min(3.0, step * 1.02)
        else:
            # Rejected: reduce step and dampen momentum effect
            p_prev = p
            step *= 0.5
            if step < 1e-9:
                step = 1e-9
                break

    # Final check
    final_score = fast_score(p)
    if final_score < best_score:
        best_score = final_score
        best_p = p.copy()

    return best_p, best_score

def block_to_p(n, B, c):
    """Convert block log-weights to full sequence normalized to sum 1."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1 and c is not None:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)

def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    p = block_to_p(n, B, c)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(max(0, B - 1), dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(np.exp(z_local(n, B, c)) / float(L), Gsum)) if False else 0.0

    # Recompute wb for the gradient
    z = np.zeros(B, dtype=np.float64)
    if B > 1 and c is not None:
        c_arr = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c_arr.size)
        if m > 0:
            z[:m] = c_arr[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    sb = float(np.sum(wb))
    if sb <= 0 or not np.isfinite(sb):
        wb = np.ones(B, dtype=np.float64) / B
    else:
        wb /= sb

    Wavg = float(np.dot(wb, Gsum))
    H = wb / L
    gradz = H * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)

def z_local(n, B, c):
    """Helper to get z vector (not actually used, kept for clarity)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1 and c is not None:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]
    return z

def run_block_minimize(c0, n, B, beta, bounds, maxiter=100):
    """Run L-BFGS-B on block parameters."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return np.asarray(c0, dtype=np.float64)

    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    m_needed = max(0, B - 1)
    if c0.size != m_needed:
        tmp = np.zeros(m_needed, dtype=np.float64)
        m = min(tmp.size, c0.size)
        if m > 0:
            tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)

    try:
        res = minimize(
            block_objective, c0, args=(n, B, beta),
            method="L-BFGS-B", jac=True, bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-6, "maxcor": 12}
        )
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0

    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)

def structured_starts(n, rng, max_count=12):
    """Generate diverse structured starting sequences."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear ramps
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    # Two-peak (split)
    for frac in (0.3, 0.4, 0.5):
        p = np.ones(n, dtype=np.float64)
        p[: int(n * frac)] = 2.0
        p[int(n * frac):] = 0.5
        p = np.maximum(p, 1e-10)
        p /= p.sum()
        starts.append(p)

    # Gaussian bump (centered)
    xc = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for sigma in (0.1, 0.15, 0.2):
        p = np.exp(-0.5 * ((xc - 0.5) / sigma) ** 2)
        p /= p.sum()
        starts.append(p)

    # One-sided decay
    for rate in (1.0, 2.0, 3.0):
        p = np.exp(-rate * x)
        p /= p.sum()
        starts.append(p)

    # Raised cosine
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    p = np.maximum(p, 1e-10)
    p /= p.sum()
    starts.append(p)

    # Random smooth (via Fourier)
    for _ in range(2):
        size = 1
        while size < n:
            size <<= 1
        zc = rng.normal(size=size // 2 + 1) + 1j * rng.normal(size=size // 2 + 1)
        zc = np.concatenate([zc, np.conj(zc[:0:-1])])
        z = np.fft.irfft(zc)
        z = z[:n]
        z = z - z.mean()
        z = 1.5 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    return starts[:max_count]

def get_previous_solution(n):
    """Try to get height_sequence_1 and resize to n if needed."""
    try:
        hs = height_sequence_1
        if isinstance(hs, (list, np.ndarray)):
            hs = np.asarray(hs, dtype=np.float64)
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0 and np.isfinite(s):
                if hs.size == n:
                    return hs / s
                else:
                    # Resize via interpolation
                    old_n = hs.size
                    x_old = np.linspace(0, 1, old_n)
                    x_new = np.linspace(0, 1, n)
                    resized = np.interp(x_new, x_old, hs / s)
                    resized = np.maximum(resized, 0.0)
                    ss = resized.sum()
                    if ss > 0:
                        return resized / ss
    except Exception:
        pass
    return None

def official_score(seq):
    """Call the authoritative evaluator safely."""
    try:
        val = float(evaluate_sequence(seq))
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")

def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Returns a list of non-negative floats.
    Uses multi-phase optimization: block init -> momentum gradient with annealing.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(0.0, budget_s - reserve)

    N = 16384
    rng = np.random.default_rng(seed)

    print(f"Starting search: N={N}, budget={budget_s:.0f}s")

    # ---- Collect initial candidates ----
    candidates = []

    # Previous best solution (if available)
    prev = get_previous_solution(N)
    if prev is not None:
        candidates.append(("prev", prev))
        print(f"Loaded previous solution, score={fast_score(prev):.8f}")

    # Structured starts
    for i, p0 in enumerate(structured_starts(N, rng, max_count=12)):
        candidates.append((f"struct_{i}", p0))

    # Small perturbations of the previous solution
    if prev is not None:
        for scale in (0.01, 0.05, 0.1):
            perturbation = scale * rng.normal(size=N)
            p_pert = prev * (1.0 + perturbation)
            p_pert = normalize_simplex(p_pert)
            candidates.append((f"pert_{scale}", p_pert))

    # Score all candidates and pick the best few
    scored = []
    for name, p0 in candidates:
        sc = fast_score(p0)
        if np.isfinite(sc):
            scored.append((sc, name, p0))
            print(f"  init {name}: score={sc:.8f}")

    scored.sort(key=lambda x: x[0])

    # ---- Phase 1: Block optimization (quick, 60s) ----
    block_deadline = start_time + min(60.0, 0.1 * budget_s)
    best_p = scored[0][2].copy() if scored else np.full(N, 1.0 / N)
    best_score = scored[0][0] if scored else float("inf")

    if time.monotonic() < block_deadline and N >= 256:
        from scipy.optimize import minimize

        best_block_score = best_score
        for B in (32, 64, 128, 256):
            if B >= N or N % B != 0:
                continue
            if time.monotonic() >= block_deadline:
                break

            bounds = [(-8.0, 8.0)] * (B - 1)

            # Try a few initial directions
            inits = [np.zeros(B - 1, dtype=np.float64)]
            x = np.linspace(0, 1, B - 1)
            inits.append(2.0 * (x - 0.5))
            inits.append(-2.0 * (x - 0.5))
            inits.append(np.sin(2 * np.pi * x))
            inits.append(0.5 * rng.normal(size=B - 1))

            for c0 in inits:
                if time.monotonic() >= block_deadline:
                    break
                c = run_block_minimize(c0, N, B, 20000.0, bounds, maxiter=60)
                p = block_to_p(N, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_block_score:
                    best_block_score = sc
                    best_p = p.copy()
                    best_score = sc
                    print(f"  block B={B}: score={sc:.8f}")

    print(f"After block phase: best_score={best_score:.8f}")

    # ---- Phase 2: Momentum gradient refinement (bulk of time) ----
    refine_deadline = deadline - 30.0

    # Use the best candidate from scoring and block phase
    start_p = best_p.copy()

    # Try the top 3 initial candidates with momentum refinement
    top_candidates = []
    used = set()
    for sc, name, p0 in scored:
        if p0 is start_p or name in used:
            continue
        top_candidates.append(p0)
        used.add(name)
        if len(top_candidates) >= 2:
            break

    # Allocate time: 60% to best, 25% to second, 15% to third
    total_refine_time = max(60.0, refine_deadline - time.monotonic())
    fracs = (0.60, 0.25, 0.15)

    all_starts = [start_p] + top_candidates
    for i, p0 in enumerate(all_starts):
        if time.monotonic() >= refine_deadline:
            break
        remaining = refine_deadline - time.monotonic()
        if remaining < 30.0:
            break
        frac = fracs[i] if i < len(fracs) else 0.1
        alloc = remaining * frac
        sub_deadline = time.monotonic() + alloc

        print(f"\nPhase 2, start {i}: allocating {alloc:.0f}s")
        p_fit, sc_fit = momentum_refine(p0, sub_deadline,
                                         beta_start=5000.0,
                                         beta_end=2000000.0,
                                         momentum=0.85)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Improved! Score: {best_score:.8f}")

    # ---- Phase 3: Fine-tuning with very high beta ----
    remaining = deadline - time.monotonic()
    if remaining > 20.0 and best_score < 1.6:
        print(f"\nPhase 3: fine-tuning with {remaining:.0f}s remaining")
        p_fit, sc_fit = momentum_refine(best_p, deadline,
                                         beta_start=500000.0,
                                         beta_end=5000000.0,
                                         momentum=0.7)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Phase 3 improved! Score: {best_score:.8f}")

    # ---- Final cleanup and validation ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0 and np.isfinite(final_s):
        final_p = final_p / final_s

    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"\nFinal: local_score={fast_score(final_p):.8f}, official={off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback: uniform sequence
        print("WARNING: Falling back to uniform sequence")
        return [1.0] * N

    print(f"Success! Score = {off:.8f} (target: 1.5027)")
    return final_seq
