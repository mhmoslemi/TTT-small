def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

"""
Low-peak nonnegative step-function search using multi-harmonic initialization
and aggressive mirror-descent refinement with adaptive learning rate and restarts.

Strategy:
1. Generate diverse initial candidates (anti-palindromic multi-harmonic shapes)
2. Select best candidate via fast scoring
3. Apply heavy mirror-descent refinement with top-K active sets, adaptive LR,
   periodic restarts, and smooth exploration perturbations
4. Return best validated sequence
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    """Fast internal score computation (matches evaluate_sequence)."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _make_anti_pal(n, amplitudes):
    """Create anti-palindromic multi-harmonic sequence normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    nf = float(n)
    shape = np.ones(n, dtype=np.float64)
    for m, A in enumerate(amplitudes, start=1):
        if A != 0.0:
            shape += A * np.sin(2.0 * np.pi * float(m) * (i + 0.5) / nf)
    a = shape / nf
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 0.01:
        return np.ones(n, dtype=np.float64) / nf
    return a / s


def _compute_gradient_lags(a, top_lags):
    """
    Compute the averaged gradient of max(b) over top_lags.
    For lag k: dB[k]/da[i] = 2*a[k-i] for valid i.
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for kk in top_lags:
        k = int(kk)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= float(len(top_lags))
    return grad


def _refine_mirror(a, budget_s, verbose=True):
    """
    Mirror descent refinement with top-K active sets, adaptive learning rate,
    smooth exploration restarts, and convergence detection.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    start_time = time.monotonic()
    lr = float(n) / 30.0
    min_lr = 0.5
    max_lr = float(n)

    K = min(16, 2 * n - 1)
    check_interval = 5 if n <= 4096 else 10
    it = 0
    last_improve = 0
    no_improve_limit = 300
    restart_count = 0
    max_restarts = 20

    if verbose:
        print(f"  [refine] n={n} initial_score={best_score:.8f} lr={lr:.1f} K={K}")

    while time.monotonic() - start_time < budget_s:
        it += 1
        remaining = budget_s - (time.monotonic() - start_time)
        if remaining < 2.0:
            break

        b = _fft_convolve(a)
        if b.size == 0:
            break

        # Top-K active lags
        if K < b.size:
            top_k_idx = np.argpartition(b, -K)[-K:]
        else:
            top_k_idx = np.arange(b.size)

        # Compute gradient
        grad = _compute_gradient_lags(a, top_k_idx)

        # Center gradient (preserve sum to first order)
        g = grad - np.mean(grad)
        g_norm = np.sqrt(np.mean(g * g))
        if g_norm < 1e-15:
            break

        # Normalize gradient for stable learning rate
        g = g / g_norm

        # Mirror descent update
        exp_arg = -lr * g
        exp_arg = np.clip(exp_arg, -1.5, 1.5)
        new_a = a * np.exp(exp_arg)
        new_a = np.clip(new_a, 1e-30, 1000.0)
        ns = float(new_a.sum())
        if not np.isfinite(ns) or ns < 0.01:
            new_a = a.copy()
            lr = max(min_lr, lr * 0.7)
            continue
        new_a = new_a / ns

        # Accept step (always in mirror descent)
        a = new_a

        # Periodic score check
        if it % check_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_a = a.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.03)
            else:
                lr = max(min_lr, lr * 0.92)

            # Convergence / restart logic
            if it - last_improve > no_improve_limit:
                restart_count += 1
                if restart_count > max_restarts:
                    break
                if verbose:
                    print(f"    restart #{restart_count} at iter={it} score={best_score:.8f} lr={lr:.2f}")
                # Smooth exploration: add low-frequency perturbation
                freq = np.random.randint(1, 5)
                phase = np.random.uniform(0, 2 * np.pi)
                perturb = 0.05 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
                a = best_a + perturb * best_a
                a = np.clip(a, 1e-10, None)
                s2 = float(a.sum())
                a = a / s2
                lr = float(n) / 40.0
                last_improve = it
                no_improve_limit = max(100, no_improve_limit // 2)

    if verbose:
        print(f"  [refine] done iters={it} best_score={best_score:.8f} restarts={restart_count}")

    return best_a, best_score


def _refine_coordinate(a, budget_s, verbose=True):
    """
    Coordinate descent refinement: optimize pairs of coefficients at a time.
    Good for fine-tuning after mirror descent.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 0.01:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    start_time = time.monotonic()
    if verbose:
        print(f"  [coord] n={n} initial_score={best_score:.8f}")

    # Precompute b
    b = _fft_convolve(a)
    if b.size == 0:
        return best_a, best_score

    # Find the peak lag
    peak_lag = int(np.argmax(b))
    peak_val = b[peak_lag]

    # Coordinate descent: try moving mass from high-gradient to low-gradient positions
    it = 0
    stall = 0

    while time.monotonic() - start_time < budget_s:
        it += 1
        if time.monotonic() - start_time > budget_s - 1.0:
            break

        b = _fft_convolve(a)
        if b.size == 0:
            break
        peak_lag = int(np.argmax(b))
        peak_val = float(b[peak_lag])

        # Compute gradient at peak lag
        n_val = n
        k = peak_lag
        lo = max(0, k - (n_val - 1))
        hi = min(n_val - 1, k)
        if lo > hi:
            break
        idx = np.arange(lo, hi + 1)
        grad_vals = 2.0 * a[k - idx]

        # Find positions with high and low gradient contribution
        grad_high = np.max(grad_vals)
        grad_low = np.min(grad_vals)

        if grad_high - grad_low < 1e-12:
            # Gradient is uniform, try a different approach
            break

        # Move a small amount of mass from high-grad to low-grad position
        hi_pos = idx[np.argmax(grad_vals)]
        lo_pos = idx[np.argmin(grad_vals)]

        # Try a small transfer
        eps = 0.001 * float(a.sum())
        trial = a.copy()
        trial[hi_pos] -= eps
        trial[lo_pos] += eps
        if trial[hi_pos] < 0:
            trial[hi_pos] = 0
            trial[lo_pos] += eps

        ts = float(trial.sum())
        trial = trial / ts

        sc = _score(trial)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            a = trial
            best_a = a.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                break

    if verbose:
        print(f"  [coord] done iters={it} best_score={best_score:.8f}")

    return best_a, best_score
