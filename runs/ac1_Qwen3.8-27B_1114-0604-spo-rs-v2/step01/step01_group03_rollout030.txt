Uses FFT-based autocorrelation, softmax-smoothed gradient descent, and

def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, 0.0)
    s2 = float(a.sum())
    if s2 > 0:
        a /= s2
    return a


def _fft_autocorr(a, L):
    """Compute aperiodic autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    return c[:2 * n - 1]


def _soft_gradient(a, L, n, temp):
    """Compute gradient of softmax-smoothed max autocorrelation.
    Returns (gradient, soft_max_value, max_autocorr_value)."""
    R = _fft_autocorr(a, L)
    R = np.maximum(R, 0.0)
    vmax = float(np.max(R))
    z = R / temp
    z -= float(np.max(z))
    z = np.clip(z, -500.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = e / s
    J = temp * (float(np.max(R)) - math.log(s))
    # Compute v[i] = sum_k w[k] * a[k-i] via FFT
    # v[i] = (w * rev_a)[n-1+i] where rev_a = reversed a
    rev_a = a[::-1]
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(rev_a, L)
    conv = np.fft.irfft(W * A, L)
    v = conv[n - 1:2 * n - 1]
    grad = 2.0 / temp * v
    return grad, J, vmax


def _mws_qp_step(a, y, L, n, step=0.5, num_steps=3):
    """One projected gradient step for QP: min a^T H_y a s.t. a>=0, sum(a)=1.
    H_y[i,j] = y[i+j]. Gradient of a^T H_y a is 2*H_y*a = 2*(y * rev_a)[n-1:2n-1]."""
    for _ in range(num_steps):
        rev_a = a[::-1]
        Y = np.fft.rfft(y, L)
        A = np.fft.rfft(rev_a, L)
        conv = np.fft.irfft(Y * A, L)
        Ha = conv[n - 1:2 * n - 1]
        grad = 2.0 * Ha
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s > 1e-15:
            a_new /= s
        else:
            break
        a = a_new
    return a

    f[:kmax + 1] = spectrum
    f[kmax + 1:] = np.conj(f[1:kmax + 1][::-1])
    sig = np.fft.irfft(f, L)[:n]
    a = np.maximum(sig, 0.0)
    if a.sum() < 1e-10:
        a = np.abs(sig)
    return _normalize(a)


def _initial_candidates(n, L, rng, seed_base=42):
    out = []
    # From height_sequence_1 if available
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n and np.all(np.isfinite(arr)):
            out.append(_normalize(arr))
        elif arr.size > 0:
            # Resample to length n
            idx = (np.arange(n) * arr.size / n).astype(int)
            idx = np.clip(idx, 0, arr.size - 1)
            out.append(_normalize(arr[idx]))
    except (NameError, Exception):
        pass
    # Uniform
    out.append(_normalize(np.ones(n, dtype=np.float64)))
    # Cosine windows
    for _ in range(4):
        out.append(_cosine_window(n, rng))
    # Gaussian windows
    for _ in range(3):
        out.append(_gaussian_window(n, rng))
    # FFT-based priors
    for _ in range(4):
        out.append(_fft_based_prior(n, L, rng))
    # Random priors
    for _ in range(5):
        out.append(_random_prior(n, rng))
    # Semitic: high at edges
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    out.append(_normalize(a))
    # Two-hump
    a = 0.5 + 0.5 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.0)
    out.append(_normalize(a))
    # Flat with noise
    a = 1.0 + 0.3 * rng.normal(0, 1, n)
    out.append(_normalize(a))
    return out


def _exact_score_fast(a, n):
    """Fast exact score using FFT."""
    L = 1 << (2 * n - 1).bit_length()
    R = _fft_autocorr(a, L)
    max_b = float(np.max(R))
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def _evaluate_seq(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        a = np.asarray([max(0.0, min(1000.0, float(x))) for x in seq], dtype=np.float64)
        n = len(a)
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        L = 1 << (2 * n - 1).bit_length()
        R = _fft_autocorr(a, L)
        return float(2.0 * n * np.max(R) / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)


def _soft_gradient_search(a, L, n, rng, time_limit, temp0=0.5, temp_max=200.0, eta0=0.3):
    """Softmax-smoothed gradient descent with adaptive temperature and step size."""
    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_exact = _exact_score_fast(a, n)
    
    temp = float(temp0)
    eta = float(eta0)
    no_improve = 0
    it = 0
    last_exact_check = time.monotonic()
    
    while time.monotonic() < deadline:
        grad, J, vmax = _soft_gradient(a, L, n, temp)
        
        # Adaptive temperature: if softmax is too peaked (low entropy), increase temp
        R = _fft_autocorr(a, L)
        R = np.maximum(R, 0.0)
        z = R / temp
        z -= float(np.max(z))
        e = np.exp(z)
        s_e = float(e.sum())
        w = e / s_e
        entropy = -float(np.sum(w * np.log(np.maximum(w, 1e-300))))
        max_entropy = math.log(2 * n - 1)
        entropy_frac = entropy / max_entropy if max_entropy > 0 else 1.0
        
        # If too peaked, increase temperature; if too flat, decrease
        if entropy_frac < 0.3:
            temp = min(temp * 1.5, temp_max)
        elif entropy_frac > 0.7 and temp > temp0 * 2:
            temp = max(temp * 0.9, temp0)
        
        # Multiplicative update
        d = np.maximum(grad, 0.0)
        d_max = float(np.max(d))
        if d_max > 1e-15:
            d = d / d_max
        else:
            break
        
        # Try multiple step sizes
        accepted = False
        for step in (eta, eta * 0.5, eta * 0.25, eta * 2.0):
            if step <= 1e-12:
                continue
            a_new = a * np.exp(-step * d)
            a_new = _normalize(a_new)
            sc = _exact_score_fast(a_new, n)
            cur_sc = _exact_score_fast(a, n)
            if sc < cur_sc - 1e-12:
                a = a_new
                accepted = True
                eta = min(1.0, eta * 1.1)
                break
            # Also accept if soft objective improved
            grad_new, J_new, _ = _soft_gradient(a_new, L, n, temp)
            if J_new < J - 1e-8:
                a = a_new
                accepted = True
                eta = min(1.0, eta * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            eta = max(0.05, eta * 0.8)
            if no_improve > 50:
                # Random restart / mixing
                prior = _random_prior(n, rng)
                mix = rng.uniform(0.2, 0.5)
                a = _normalize((1.0 - mix) * a + mix * prior)
                temp = min(temp_max, temp * 1.3)
                eta = eta0
                no_improve = 0
        
        it += 1
        # Periodic exact score check
        if time.monotonic() - last_exact_check > 2.0:
            sc = _exact_score_fast(a, n)
            if sc < best_exact - 1e-10:
                best_exact = sc
                best_a = a.copy()
            last_exact_check = time.monotonic()
        
        # If time is running low, do a final check
        if time.monotonic() > deadline - 5.0:
            sc = _exact_score_fast(a, n)
            if sc < best_exact:
                best_exact = sc
                best_a = a.copy()
            break
    
    # Final check
    sc = _exact_score_fast(a, n)
    if sc < best_exact:
        best_exact = sc
        best_a = a.copy()
    
    return best_a, best_exact


def _mws_search(a, L, n, rng, time_limit, inner_steps=5):
    """MWS game-theoretic optimization with FFT-based QP best response."""
    deadline = time.monotonic() + float(time_limit)
    best_a = _normalize(a)
    best_exact = _exact_score_fast(best_a, n)
    
    # Initialize lag distribution
    m = 2 * n - 1
    y = np.ones(m, dtype=np.float64) / m
    eta_mws = 1.0  # MWS learning rate
    last_exact_check = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute payoff for current a
        R = _fft_autocorr(best_a, L)
        R = np.maximum(R, 0.0)
        max_R = float(np.max(R))
        
        # Update lag distribution (exponential weighting)
        z = eta_mws * R / max(max_R, 1e-15)
        z -= float(np.max(z))
        z = np.clip(z, -50.0, 0.0)
        y_new = np.exp(z)
        s_y = float(y_new.sum())
        if s_y > 0:
            y_new /= s_y
        else:
            y_new = np.ones(m) / m
        y = 0.7 * y + 0.3 * y_new  # Smoothed update
        y /= float(y.sum())
        
        # QP best response: min a^T H_y a s.t. a>=0, sum=1
        # Use projected gradient with FFT
        a_qp = best_a.copy()
        L_qp = L  # FFT size
        
        # Adaptive step for QP: estimate Lipschitz constant
        # ||H_y||_2 <= 1 for probability distribution y
        step_qp = 0.4
        for _ in range(inner_steps):
            rev_a = a_qp[::-1]
            Y = np.fft.rfft(y, L_qp)
            A = np.fft.rfft(rev_a, L_qp)
            conv = np.fft.irfft(Y * A, L_qp)
            Ha = conv[n - 1:2 * n - 1]
            a_qp = a_qp - step_qp * 2.0 * Ha
            a_qp = np.maximum(a_qp, 0.0)
            s = float(a_qp.sum())
            if s > 1e-15:
                a_qp /= s
            else:
                a_qp = best_a.copy()
                break
        
        # Mix QP result with current best
        mix = rng.uniform(0.3, 0.8)
        a_cand = _normalize((1.0 - mix) * best_a + mix * a_qp)
        
        sc_cand = _exact_score_fast(a_cand, n)
        if sc_cand < best_exact - 1e-12:
            best_a = a_cand
            best_exact = sc_cand
        
        # Adaptive MWS learning rate
        eta_mws = min(5.0, eta_mws * 1.02)
        
        # Periodic check
        if time.monotonic() - last_exact_check > 3.0:
            last_exact_check = time.monotonic()
            if time.monotonic() > deadline - 10.0:
                break
    
    # Final check
    sc = _exact_score_fast(best_a, n)
    if sc < best_exact:
        best_exact = sc
    return best_a, best_exact


def _sparsify(a, n, rng, frac=0.1):
    """Set smallest elements to zero and renormalize."""
    a = a.copy()
    k = max(1, int(frac * n))
    idx = np.argsort(a)
    a[idx[:k]] = 0.0
    s = float(a.sum())
    if s > 1e-15:
        a /= s
    else:
        a = _normalize(np.ones(n))
    return a


def _symmetrize(a, n):
    """Sy
mmetrize the sequence (make it palindromic)."""
    a_sym = 0.5 * (a + a[::-1])
    return _normalize(a_sym)


def _post_process(a, n, L, rng, time_limit):
    """Post-processing: try sparsification, symmetrization, small perturbations."""
    deadline = time.monotonic() + float(time_limit)
    best_a = a.copy()
    best_score = _exact_score_fast(a, n)
    
    # Try symmetrization
    a_sym = _symmetrize(a, n)
    sc = _exact_score_fast(a_sym, n)
    if sc < best_score:
        best_a = a_sym
        best_score = sc
    
    # Try sparsification at different levels
    for frac in (0.05, 0.1, 0.15, 0.2, 0.3):
        a_sp = _sparsify(a, n, rng, frac)
        sc = _exact_score_fast(a_sp, n)
        if sc < best_score:
            best_a = a_sp
            best_score = sc
        # Re-optimize after sparsification
        if time.monotonic() < deadline - 5.0:
            a_opt, sc_opt = _soft_gradient_search(a_sp, L, n, rng, 5.0, temp0=0.5, temp_max=50.0, eta0=0.2)
            if sc_opt < best_score:
                best_a = a_opt
                best_score = sc_opt
    
    # Small random perturbations
    for _ in range(30):
        if time.monotonic() > deadline - 2.0:
            break
        a_p = best_a + rng.normal(0, 0.01, n)
        a_p = _normalize(a_p)
        sc = _exact_score_fast(a_p, n)
        if sc < best_score:
            best_a = a_p
            best_score = sc
    
    # Final soft optimization from best
    if time.monotonic() < deadline - 10.0:
        a_opt, sc_opt = _soft_gradient_search(best_a, L, n, rng, 
                                              min(30.0, deadline - time.monotonic() - 5.0),
                                              temp0=0.3, temp_max=100.0, eta0=0.15)
        if sc_opt < best_score:
            best_a = a_opt
            best_score = sc_opt
    
    return best_a, best_score


def _choose_n():
    """Choose sequence length."""
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == 6121:
            return 6121
    except (NameError, Exception):
        pass
    return 6121


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns the best sequence found."""
    try:
        budget = float(budget_s)
    except (Exception,):
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 2400.0

    start = time.monotonic()
    hard_deadline = start + budget - 3.0
    rng = np.random.default_rng(seed)
    
    print(f"Starting search with budget={budget:.1f}s")
    
    n = _choose_n()
    L = 1 << (2 * n - 1).bit_length()
    print(f"Using n={n}, FFT size L={L}")
    
    best_a = None
    best_score = float('inf')
    
    def _update_best(a, score):
        nonlocal best_a, best_score
        if score < best_score:
            best_a = a.copy()
            best_score = score
            print(f"  New best: {best_score:.6f} at t={time.monotonic()-start:.1f}s")
    
    # Phase 1: Evaluate initial candidates
    print("Phase 1: Initial candidates...")
    candidates = _initial_candidates(n, L, rng, seed)
    for i, a in enumerate(candidates):
        sc = _exact_score_fast(a, n)
        _update_best(a, sc)
        if i < 5:
            print(f"    Candidate {i}: score={sc:.6f}")
    
    if time.monotonic() > hard_deadline - 60.0:
        seq = [float(x) for x in best_a]
        return seq
    
    # Phase 2: Soft gradient search from best candidate
    print(f"Phase 2: Soft gradient search (score={best_score:.6f})...")
    remaining = hard_deadline - time.monotonic()
    lim = min(400.0, 0.25 * remaining)
    a_opt, sc_opt = _soft_gradient_search(best_a, L, n, rng, lim, temp0=0.5, temp_max=300.0, eta0=0.3)
    _update_best(a_opt, sc_opt)
    
    # Phase 3: MWS search
    if time.monotonic() < hard_deadline - 100.0:
        print(f"Phase 3: MWS search (score={best_score:.6f})...")
        remaining = hard_deadline - time.monotonic()
        lim = min(300.0, 0.20 * remaining)
        a_mws, sc_mws = _mws_search(best_a, L, n, rng, lim, inner_steps=4)
        _update_best(a_mws, sc_mws)
    
    # Phase 4: Multiple restarts with soft gradient
    if time.monotonic() < hard_deadline - 80.0:
        print(f"Phase 4: Multiple restarts (score={best_score:.6f})...")
        restarts = 0
        while time.monotonic() < hard_deadline - 80.0:
            if restarts % 3 == 0:
                # Start from best with perturbation
                prior = _random_prior(n, rng)
                mix = rng.uniform(0.1, 0.4)
                a_init = _normalize((1.0 - mix) * best_a + mix * prior)
            elif restarts % 3 == 1:
                # Fresh random start
                a_init = _random_prior(n, rng)
            else:
                # FFT-based start
                a_init = _fft_based_prior(n, L, rng)
            
            remaining = hard_deadline - time.monotonic() - 80.0
            lim = min(120.0, max(10.0, remaining / 3.0))
            a_opt, sc_opt = _soft_gradient_search(a_init, L, n, rng, lim, 
                                                  temp0=rng.uniform(0.3, 2.0),
                                                  temp_max=rng.uniform(100.0, 400.0),
                                                  eta0=rng.uniform(0.1, 0.5))
            _update_best(a_opt, sc_opt)
            restarts += 1
    
    # Phase 5: MWS from best (second round)
    if time.monotonic() < hard_deadline - 60.0:
        print(f"Phase 5: MWS refinement (score={best_score:.6f})...")
        remaining = hard_deadline - time.monotonic() - 60.0
        lim = min(150.0, 0.5 * remaining)
        a_mws, sc_mws = _mws_search(best_a, L, n, rng, lim, inner_steps=6)
        _update_best(a_mws, sc_mws)
    
    # Phase 6: Post-processing
    if time.monotonic() < hard_deadline - 20.0:
        print(f"Phase 6: Post-processing (score={best_score:.6f})...")
        remaining = hard_deadline - time.monotonic() - 20.0
        a_pp, sc_pp = _post_process(best_a, n, L, rng, remaining)
        _update_best(a_pp, sc_pp)
    
    # Phase 7: Final polishing
    if time.monotonic() < hard_deadline - 10.0:
        print(f"Phase 7: Final polish (score={best_score:.6f})...")
        remaining = hard_deadline - time.monotonic() - 10.0
        lim = max(5.0, remaining)
        a_final, sc_final = _soft_gradient_search(best_a, L, n, rng, lim,
                                                   temp0=0.2, temp_max=100.0, eta0=0.1)
        _update_best(a_final, sc_final)
    
    # Verify with evaluate_sequence
    print(f"\nFinal internal score: {best_score:.6f}")
    seq = [float(x) for x in best_a]
    exact_val = _evaluate_seq(seq)
    print(f"Exact evaluation: {exact_val:.6f}")
    
    if not np.isfinite(exact_val):
        # Fallback
        seq = [1.0 / n] * n
        exact_val = _evaluate_seq(seq)
        print(f"Fallback score: {exact_val:.6f}")
    
    # Try all recent candidates with exact evaluation
    # (already done internally, but double-check the best)
    
    print(f"Total time: {time.monotonic()-start:.1f}s")
    return seq
