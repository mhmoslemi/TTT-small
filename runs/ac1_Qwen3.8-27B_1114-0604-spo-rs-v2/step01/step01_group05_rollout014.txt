"""

2. Use Nesterov-accelerated projected gradient with adaptive smooth-max beta.

import numpy as np

def fft_convolution(a):
    """Linear convolution a*a using FFT, clipped to remove tiny negatives."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_grad(b, beta):
    """
    Compute smooth max of b and the gradient weights softmax(w).
    Returns (G, w, argmax_idx) where G ≈ max(b), w is softmax of beta*(b-max).
    """
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0), 0

    m = float(np.max(b))
    ai = int(np.argmax(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b), ai

    if beta <= 0.0 or m <= 0.0:
        w = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
        return m, w, ai

    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[ai] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w, ai


def fast_score(p):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """
    Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1.
    This is the gradient of max(p*p) weighted by softmax w.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([p[0] * w[0]], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = sum_j p_j * w_{i+j}
    # C[k] = sum_j p_j * w[::-1]_{k-j} = sum_j p_j * w[m-1-k+j]
    # We want s_i = sum_j p_j * w_{i+j}
    # C[m-1-i] = sum_j p_j * w[m-1-(m-1-i)+j] = sum_j p_j * w_{i+j} = s_i
    # So s_i = C[m-1-i], for i=0..n-1
    # That means s = C[m-1], C[m-2], ..., C[m-n] reversed
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size))
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size))
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size))
    p *= 1.0 / s
    return p


def diverse_starts(n, rng, height_seq_1=None, max_starts=6):
    """Generate diverse starting sequences."""
    starts = []

    if height_seq_1 is not None:
        hs = np.asarray(height_seq_1, dtype=np.float64)
        if hs.size == n and float(np.sum(hs)) > 0:
            starts.append(normalize_simplex(hs))

    p = np.full(n, 1.0 / float(n))
    starts.append(p)

    x = np.linspace(0.0, 1.0, n)
    for a in (-2.0, 2.0, -1.0, 1.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        starts.append(normalize_simplex(p))

    # Two-level structure
    p = np.ones(n)
    p[:n//2] = 1.5
    p[n//2:] = 0.5
    starts.append(normalize_simplex(p))

    p = np.ones(n)
    p[:n//3] = 0.5
    p[n//3:2*n//3] = 1.5
    p[2*n//3:] = 0.5
    starts.append(normalize_simplex(p))

    # Random smooth perturbation
    zc = rng.normal(size=1) + 1j * rng.normal(size=1)
    freqs = np.fft.rfftfreq(n)
    mask = np.abs(freqs) < 0.1
    zc_full = np.zeros(n//2 + 1, dtype=complex)
    zc_full[mask] = rng.normal(size=int(np.sum(mask))) + 1j * rng.normal(size=int(np.sum(mask)))
    z = np.fft.irfft(zc_full, n)
    z = 0.8 * np.tanh(z)
    starts.append(normalize_simplex(np.exp(z)))

    return starts[:max_starts]


def nesterov_refine(p0, deadline, rng, beta_start=500.0, beta_end=500000.0,
                    max_iter=2000000, shake_every=800, shake_mag=0.001):
    """
    Nesterov-accelerated projected gradient on the simplex with adaptive beta.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)
    print(f"  nesterov start: score={best_score:.8f}")

    # Nesterov momentum
    y = p.copy()
    momentum = 0.9
    step = 0.5
    it = 0
    no_improve = 0

    total_iters_est = max_iter
    beta_now = beta_start

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Adaptive beta: geometric increase over the run
        frac = min(1.0, it / float(max_iter))
        beta_now = beta_start * (beta_end / beta_start) ** frac

        b = fft_convolution(y)
        G, w, amax = smooth_max_and_grad(b, beta_now)
        if not np.isfinite(G):
            break

        cur_smooth = n * G
        cur_exact = 2.0 * n * float(np.max(b))

        if cur_exact < best_score - 1e-15:
            best_score = cur_exact
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 500 == 0:
            print(f"  nesterov it={it} beta={beta_now:.0f} best={best_score:.8f} step={step:.3e}")

        if no_improve > 3000:
            # Shake: add random perturbation
            noise = rng.normal(0, shake_mag, n)
            p = normalize_simplex(p + noise)
            y = p.copy()
            no_improve = 0
            step = 0.5
            continue

        # Compute gradient at y
        svec = corr_pw(y, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec - float(np.mean(2.0 * svec))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Line search with backtracking
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)

            qb = fft_convolution(q)
            G2, _, _ = smooth_max_and_grad(qb, beta_now)
            if not np.isfinite(G2):
                continue
            if n * G2 < cur_smooth - 1e-14:
                p = q
                cur_smooth = n * G2
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.3, 5.0)

        # Nesterov update for y
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y = p + momentum * (p - best_p + (p - p))  # placeholder
        # Standard Nesterov: y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k)
        # Simplified: y = (1+m)*p - m*prev_p
        # For stability, use: y = p (if momentum is too aggressive)
        # Actually let's use: y_k = x_k (no momentum in y, just in step direction)
        # This is equivalent to heavy-ball on the projected problem
        y = p.copy()

    return best_p, best_score


def heavy_ball_refine(p0, deadline, rng, beta_start=200.0, beta_end=1000000.0,
                      max_iter=2000000, shake_every=600, shake_mag=0.002):
    """
    Heavy-ball momentum projected gradient with adaptive beta and shakes.
    More stable than Nesterov for non-smooth objectives.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    p_prev = p.copy()
    best_p = p.copy()
    best_score = fast_score(p)
    print(f"  heavy_ball start: score={best_score:.8f}")

    momentum = 0.85
    step = 1.0
    it = 0
    no_improve = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        frac = min(1.0, it / float(max_iter))
        beta_now = beta_start * (beta_end / beta_start) ** frac

        b = fft_convolution(p)
        G, w, amax = smooth_max_and_grad(b, beta_now)
        if not np.isfinite(G):
            break

        cur_smooth = n * G
        cur_exact = 2.0 * n * float(np.max(b))

        if cur_exact < best_score - 1e-15:
            best_score = cur_exact
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"  heavy_ball it={it} beta={beta_now:.0f} best={best_score:.8f} "
                  f"step={step:.3e} mom={momentum:.3f}")

        if no_improve > shake_every:
            # Shake to escape local minimum
            noise = rng.normal(0, shake_mag, n)
            p = normalize_simplex(p + noise)
            p_prev = p.copy()
            no_improve = 0
            step = 0.5
            momentum = 0.85
            continue

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad -= float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Heavy ball: effective direction = grad + momentum * (grad - prev_grad)
        # Simplified: p_new = project(p - step * (grad + momentum * (p - p_prev)))
        # This is equivalent to: velocity update in the simplex
        vel = (p - p_prev)
        direction = grad + momentum * vel

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * direction, 1.0)

            qb = fft_convolution(q)
            G2, _, _ = smooth_max_and_grad(qb, beta_now)
            if not np.isfinite(G2):
                continue
            if n * G2 < cur_smooth - 1e-14:
                p_prev = p
                p = q
                cur_smooth = n * G2
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.2, 5.0)

    return best_p, best_score


def exact_max_polish(p0, deadline, max_iter=50000):
    """
    Directly target the worst convolution coefficient.
    Pick argmax of b, compute its exact gradient, and do a local step.
    """
    p = normalize_simplex(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)

    it = 0
    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        cur_max = float(np.max(b))
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()

        if it % 1000 == 0:
            print(f"  polish it={it} score={best_score:.8f}")

        # Find all positions within 1% of max
        thresh = cur_max * 0.995
        active = np.where(b >= thresh)[0]

        if len(active) == 0:
            break

        # Gradient of b_k at the active positions (averaged)
        # grad_i = sum_{k in active} (2/n_active) * a_{k-i} (for valid k-i)
        # Use the correlation formula with uniform weights on active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        svec = corr_pw(p, w)
        grad = 2.0 * svec
        grad -= float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Line search
        accepted = False
        for step in (0.1, 0.05, 0.02, 0.01, 0.005, 0.002):
            q = project_simplex(p - step * grad, 1.0)
            qb = fft_convolution(q)
            qmax = float(np.max(qb))
            if qmax < cur_max - 1e-15:
                p = q
                cur_max = qmax
                accepted = True
                break

        if not accepted:
            break

    return best_p, best_score


def block_refine(p_full, n, B, deadline, rng):
    """
    Optimize at block level then upsample.
    p_full: full sequence of length n (sum=1)
    Returns refined full sequence.
    """
    L = n // B
    if L < 1 or B < 2:
        return p_full

    # Coarse: average within blocks
    coarse = p_full[:B * L].reshape(B, L).sum(axis=1)
    coarse_sum = float(np.sum(coarse))
    if coarse_sum <= 0:
        return p_full
    coarse = coarse / coarse_sum

    # Optimize coarse sequence with simple gradient
    c = coarse.copy()
    best_c = c.copy()
    best_cs = 2.0 * B * float(np.max(fft_convolution(c)))  # approximate

    step = 0.5
    for it in range(2000):
        if time.monotonic() >= deadline:
            break
        b = fft_convolution(c)
        G, w, _ = smooth_max_and_grad(b, 500.0)
        if not np.isfinite(G):
            break
        cur = B * G
        if B * float(np.max(b)) < best_cs - 1e-10:
            best_cs = B * float(np.max(b))
            best_c = c.copy()

        svec = corr_pw(c, w)
        grad = 2.0 * svec - float(np.mean(2.0 * svec))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        alpha = step
        q = project_simplex(c - alpha * grad, 1.0)
        qb = fft_convolution(q)
        G2, _, _ = smooth_max_and_grad(qb, 500.0)
        if B * G2 < cur - 1e-12:
            c = q
            step = min(step * 1.1, 3.0)
        else:
            step *= 0.5
            if step < 1e-8:
                break

    # Upsample: repeat each block value L times, then blend with original
    upsampled = np.repeat(best_c, L)
    # Blend to preserve fine structure
    p_refined = 0.5 * upsampled + 0.5 * p_full[:B * L]
    p_refined = normalize_simplex(p_refined)
    return p_refined

    if deadline - time.monotonic() > 60:
        t3 = time.monotonic()
        p_block = block_refine(best_p, N, 64, time.monotonic() + min(40, deadline - time.monotonic() - 30), rng)
        sc_block = fast_score(p_block)
        if sc_block < best_score:
            best_score = sc_block
            best_p = p_block
            print(f"After block refine: {best_score:.8f}")
        # Do a short heavy-ball from the block-refined
        if deadline - time.monotonic() > 30:
            p3, sc3 = heavy_ball_refine(p_block, time.monotonic() + min(30, deadline - time.monotonic() - 20),
                                         rng, beta_start=10000.0, beta_end=500000.0,
                                         max_iter=200000, shake_every=300, shake_mag=0.001)
            if sc3 < best_score:
                best_score = sc3
                best_p = p3
        print(f"After phase 3: {best_score:.8f} ({time.monotonic()-t3:.1f}s)")

    # Phase 4: Exact max polishing
    if deadline - time.monotonic() > 20:
        t4 = time.monotonic()
        p4, sc4 = exact_max_polish(best_p, time.monotonic() + min(30, deadline - time.monotonic() - 10),
                                    max_iter=20000)
        if sc4 < best_score:
            best_score = sc4
            best_p = p4
        print(f"After exact polish: {best_score:.8f} ({time.monotonic()-t4:.1f}s)")

    # Phase 5: Final short refinement with high beta
    if deadline - time.monotonic() > 15:
        t5 = time.monotonic()
        p5, sc5 = heavy_ball_refine(best_p, time.monotonic() + min(20, deadline - time.monotonic() - 5),
                                     rng, beta_start=100000.0, beta_end=2000000.0,
                                     max_iter=100000, shake_every=200, shake_mag=0.0005)
        if sc5 < best_score:
            best_score = sc5
            best_p = p5
        print(f"After final refine: {best_score:.8f} ({time.monotonic()-t5:.1f}s)")

    # Final: normalize and validate
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s

    final_seq = [float(x) for x in final_p]

    # Validate with official evaluator
    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        return [1.0] * N

    return final_seq
