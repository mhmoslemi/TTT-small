import numpy as np
import time

def _fft_convolve(a, N=None):
    n = len(a)
    if N is None:
        N = 1 << (3*n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2*n-1]

def _simplex_project(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cv - target) / rho > 0
    if not np.any(cond):
        return np.full(n, target/n)
    rho_star = np.max(rho[cond])
    tau = (cv[rho_star-1] - target) / rho_star
    return np.maximum(v - tau, 0.0)

def _score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf

            v = momentum * v - g_unit
            a_new = _proj_simplex(a + eta_cur * v, 1.0)

            c_new = _fft_conv(a_new, N)
            m_new = float(np.max(c_new))

            if m_new < best_cmax - 1e-18:
                best_cmax = m_new
                best = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                # If stuck, add small noise
                if no_improve % 2000 == 1999:
                    v = v + 0.01 * rng.standard_normal(n)
                if no_improve > 15000:
                    break

            a = a_new
            it += 1

            if verbose and it % 50000 == 0:
                sc = 2.0 * n * m_new
                print(f"  tau_level={li}, tau={tau:.2e}, it={it}, score={sc:.6f}, best={2*n*best_cmax:.6f}")

    return best, 2.0 * n * best_cmax


def _exact_polish(a0, N, deadline, verbose=False):
    """Polish using exact subgradient (single peak or small active set)."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best = a.copy()
    best_cmax = float(np.max(_fft_conv(a, N)))
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() < deadline:
        c = _fft_conv(a, N)
        m = float(np.max(c))

        # Active set: positions within 0.1% of max
        threshold = m * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient: g[j] = 2 * mean_{k in active} a[k-j]
        g = np.zeros(n)
        for k in active[:30]:  # cap active set size
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    g[j] += a[idx]
        g *= 2.0 / len(active[:30])

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-18:
            break

        g_unit = g / g_norm
        eta = 1.0 / (n * g_norm * 0.5 + 1e-10)
        eta = min(eta, 1.0)

        a_new = _proj_simplex(a - eta * g_unit, 1.0)
        c_new = _fft_conv(a_new, N)
        m_new = float(np.max(c_new))

        if m_new < best_cmax - 1e-18:
            best_cmax = m_new
            best = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = _proj_simplex(a - 0.1 * eta * g_unit, 1.0)
            m_try = float(np.max(_fft_conv(a_try, N)))
            if m_try < best_cmax - 1e-18:
                best_cmax = m_try
                best = a_try.copy()
                a = a_try
            else:
                # Random perturbation
                a_pert = a + 0.005 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                m_pert = float(np.max(_fft_conv(a_pert, N)))
                if m_pert < best_cmax:
                    best_cmax = m_pert
                    best = a_pert.copy()
                    a = a_pert
                it += 1
                if it % 1000 == 0 and verbose:
                    print(f"  polish it={it}, best_score={2*n*best_cmax:.6f}")
                if it > 50000:
                    break
        it += 1

    return best, 2.0 * n * best_cmax


def _make_starts(a_prev, n, N):
    """Generate diverse initial candidates."""
    starts = []

    # Resampled previous best
    if a_prev is not None and len(a_prev) > 10:
        a_prev = np.asarray(a_prev, dtype=np.float64)
        a_prev = np.clip(a_prev, 0, None)
        s = float(np.sum(a_prev))
        if s > 0:
            a_res = a_prev / s
            if len(a_res) != n:
                x_old = np.linspace(0, 1, len(a_res))
                x_new = np.linspace(0, 1, n)
                a_res = np.interp(x_new, x_old, a_res)
            a_res = np.maximum(a_res, 0)
            a_res /= a_res.sum()
            starts.append(a_res)

    x = np.linspace(0, 1, n)

    # Sigmoid profiles
    for A in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        for w in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
            a = 1.0 + A / (1.0 + np.exp(-(x - 0.5) / w))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            starts.append(a)

    # Two-level blocks
    for r in [1.5, 2.0, 3.0, 5.0, 10.0]:
        a = np.empty(n)
        a[: n // 2] = 1.0
        a[n // 2 :] = r
        a /= a.sum()
        starts.append(a)

    # Linear profiles
    for A in [0.5, 1.0, 2.0]:
        a = 1.0 + A * x
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        starts.append(a)

    # Cosine window
    a = 0.5 + 0.5 * np.cos(np.pi * (2 * x - 1))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    starts.append(a)

    # Uniform
    starts.append(np.full(n, 1.0 / n))

    # Score and sort
    scored = []
    for a in starts:
        sc = _score_fast(a, N)
        scored.append((sc, a))
    scored.sort(key=lambda t: t[0])
    return scored


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative coefficient sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 15.0
    rng = np.random.default_rng(int(seed))

    n = 8192
    N = _next_pow2(3 * n - 2)

    print(f"Starting search: n={n}, N={N}, budget={budget_s}s")

    # Get initial candidates
    a_prev = None
    try:
        a_prev = height_sequence_1
    except NameError:
        pass

    starts = _make_starts(a_prev, n, N)
    print(f"Generated {len(starts)} initial candidates, best initial score = {starts[0][0]:.6f}")

    best_a = starts[0][1].copy()
    best_score = starts[0][0]

    # Temperature levels for annealing
    tau_levels = [
        5.0 / n,
        2.0 / n,
        1.0 / n,
        0.5 / n,
        0.2 / n,
        0.1 / n,
        0.05 / n,
        0.02 / n,
        0.01 / n,
        0.005 / n,
        0.002 / n,
        0.001 / n,
    ]

    # Optimize from top candidates
    n_tries = min(5, len(starts))
    for i in range(n_tries):
        time_left = deadline - time.monotonic()
        if time_left < 90:
            break

        if i == 0:
            alloc = min(time_left * 0.4, 600.0)
        elif i == 1:
            alloc = min(time_left * 0.25, 400.0)
        else:
            alloc = min(time_left * 0.15, 250.0)

        t_end = time.monotonic() + alloc
        a_opt, sc_opt = _run_smooth_opt(
            starts[i][1], N, tau_levels, t_end, momentum=0.82, verbose=(i == 0)
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            print(f"  Candidate {i}: score improved to {sc_opt:.9f} (time_left={deadline - time.monotonic():.0f}s)")

    # Try perturbed versions of the best
    time_left = deadline - time.monotonic()
    n_pert = min(8, int(time_left // 60))
    for pi in range(n_pert):
        if time.monotonic() > deadline - 30:
            break
        scale = 0.005 + 0.01 * pi
        a_pert = best_a + scale * rng.standard_normal(n)
        a_pert = np.maximum(a_pert, 0)
        a_pert /= a_pert.sum()

        t_end = min(deadline - 20, time.monotonic() + min(120, time_left / (n_pert - pi + 1) - 5))
        a_opt, sc_opt = _run_smooth_opt(a_pert, N, tau_levels[:6], t_end, momentum=0.80)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            print(f"  Perturbation {pi} (scale={scale:.3f}): score={sc_opt:.9f}")

    # Final exact polishing
    time_left = deadline - time.monotonic()
    if time_left > 20:
        polish_deadline = min(deadline - 5, time.monotonic() + time_left * 0.7)
        a_final, sc_final = _exact_polish(best_a, N, polish_deadline, verbose=True)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final
            print(f"  After exact polish: score={sc_final:.9f}")

    # Final normalization
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    # Validate with official evaluator
    try:
        official = evaluate_sequence(best_a.tolist())
        if np.isfinite(official):
            print(f"Official evaluator score: {official:.9f}")
        else:
            print("WARNING: Official evaluator returned non-finite! Using fallback.")
            best_a = np.full(n, 1.0 / n)
    except Exception as e:
        print(f"Evaluation warning: {e}")

    print(f"Final score (internal): {2.0*n*float(np.max(_fft_conv(best_a, N))):.9f}")
    return [float(v) for v in best_a.tolist()]
