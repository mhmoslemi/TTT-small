For active k: a^T Q_k a = t.
For inactive k: a^T Q_k a <= t.

            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
                print("  Jump improvement -> %.6f" % best_score)
            if time.monotonic() >= deadline:
                break

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print("FINAL: internal=%.6f verified=%.6f len=%d time=%.1fs" %
          (best_score, verified, len(out), time.monotonic() - start))
    return out
