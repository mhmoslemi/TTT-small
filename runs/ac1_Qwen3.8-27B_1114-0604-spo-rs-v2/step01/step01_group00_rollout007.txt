- Moving mass from j_donate to j_receive doesn't increase any other convolution value too much

"""
Step function optimizer: minimizes 2*n*max(f*f)/(sum f)^2 using
exponentiated gradient descent with adaptive top-K softmax peak targeting,
backtracking line search, and periodic peak surgery.
"""

import numpy as np
from scipy.signal import correlate
import time


def _full_conv(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _obj(c, s):
    """Objective: 2n * max_c / s^2 where s = sum(f)."""
    n = len(c)
    # c has length 2n_f - 1 where n_f = (len(c)+1)//2
    n_f = (len(c) + 1) // 2
    return 2.0 * n_f * c.max() / (s * s)


def _topk_gradient(c, f, K):
    """Compute gradient using top-K softmax weights."""
    m = len(c)
    if K >= m:
        # Full softmax
        cs = c - c.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(m)
            p[(m - 1) // 2] = 1.0
        else:
            p /= s
    else:
        # Top-K softmax
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx].copy()
        vals -= vals.max()
        p_k = np.exp(vals)
        p_k /= p_k.sum()
        p = np.zeros(m)
        p[idx] = p_k
    
    # grad[j] = 2 * sum_k p[k] * f[k-j] for valid (k,j)
    # This is 2 * correlate(p, f, mode='valid')
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _mirror_descend(f, grad, eta):
    """One exponentiated gradient (mirror descent) step."""
    log_ratio = -eta * grad
    # Shift for numerical stability
    log_ratio -= log_ratio.max()
    f_new = f * np.exp(log_ratio)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def _peak_surgery(f, c, k_star, n, rng):
    """Try to reduce the peak at lag k_star by moving mass."""
    best_f = f.copy()
    best_peak = c.max()
    
    # Positions contributing to peak at k_star
    contrib = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            contrib[j] = f[j] * f[idx]
    
    # Top contributors (to reduce)
    top_idx = np.argsort(contrib)[::-1]
    # Bottom contributors (to receive)
    bot_idx = np.argsort(contrib)
    
    # Try several moves
    for i in range(min(10, n)):
        j_d = top_idx[i]
        for j_r in bot_idx[:5]:
            if j_d == j_r:
                continue
            for frac in [0.02, 0.05, 0.1]:
                amt = f[j_d] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_d] -= amt
                f_try[j_r] += amt
                if f_try[j_d] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _full_conv(f_try)
                if c_try.max() < best_peak - 1e-12:
                    best_peak = c_try.max()
                    best_f = f_try.copy()
    
    return best_f, best_peak
