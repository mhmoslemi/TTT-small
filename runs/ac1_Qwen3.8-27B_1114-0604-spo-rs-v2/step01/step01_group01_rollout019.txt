from scipy.optimize import linprog
import time

def sla...p_step(a0, method='highs'):
    """One SLP iteration."""
    n = len(a0)
    c0 = np.convolve(a0, a0)  # length 2n-1
    
    # Build LP: minimize M
    # Variables: a[0..n-1], M (index n)
    # Constraints:
    #   For each k in [0, 2n-2]:
    #     2 * sum_i a0[k-i] * a[i] - M <= c0[k]
    #   sum(a) = 1
    #   a[i] >= 0, M >= 0
    #
    # linprog form: minimize c^T x
    # subject to A_ub @ x <= b_ub
    #            A_eq @ x = b_eq
    #            x >= 0 (by bounds)
    
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: for each k
    # 2 * sum_{i: 0<=k-i<n, 0<=i<n} a0[k-i] * a[i] - M <= c0[k]
    # Coefficients: for a[i]: 2 * a0[k-i] (if valid), for M: -1
    A_ub = np.zeros((2*n - 1, n + 1))
    b_ub = np.zeros(2*n - 1)
    
    for k in range(2*n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = 2.0 * a0[j]
        A_ub[k, n] = -1.0
        b_ub[k] = c0[k]
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a[i] >= 0 (default), M >= 0 (default, since x >= 0)
    # Actually, M can be any non-negative value, but since we're minimizing it, it'll be >= 0 anyway.
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=[(0, None)] * (n+1), method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

Hybrid SLP + smoothed-max optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
1. SLP (Successive Linear Programming) with constraint generation for the main optimization.
   This directly handles the max constraint via linearized quadratic constraints.
2. Smoothed-max multiplicative descent for refinement.
3. Block scaling for final local search.
4. Optional up-sampling to larger sequence lengths.

import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog
import time


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s <= 0:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0, 1000)
    a = np.where(np.isfinite(a), a, 0)
    s = a.sum()
    if s <= 0:
        return np.ones(len(a)) / len(a)
    return a / s


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    L = next_fast_len(n)
    fa = rfft(a, L)
    fb = rfft(b, L)
    out = irfft(fa * fb, L)
    return out[:n]


def _slp_optimize(a0, n_constraints_max=500, max_iters=200, tol=1e-8, time_limit=600.0):
    """
    SLP with constraint generation.
    
    At each step:
    1. Solve LP with current active constraints (linearized at a0)
    2. Check if solution satisfies all original constraints
    3. Add violated constraints to active set
    4. Update a0 and repeat
    """
    start = time.monotonic()
    a = _normalize(a0.copy())
    n = len(a)
    c0 = _fft_convolve(a, a)
    current_score = 2.0 * n * c0.max()
    best_a = a.copy()
    best_score = current_score
    
    # Start with the center constraint (most likely to be active)
    active_set = [n - 1]
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Build and solve LP
        # Variables: a[0..n-1], M (total n+1)
        # Minimize M
        # s.t. for k in active_set: 2 * sum_i a0[k-i] * a[i] - M <= c0[k]
        #      sum(a) = 1
        #      a >= 0, M >= 0
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize M
        
        A_ub = []
        b_ub = []
        
        for k in active_set:
            row = np.zeros(n + 1)
            # 2 * sum_{i valid} a0[k-i] * a[i]
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                if 0 <= j < n:
                    row[i] = 2.0 * a[j]
            row[n] = -1.0  # coefficient of M
            A_ub.append(row)
            b_ub.append(c0[k])
        
        A_ub = np.array(A_ub) if A_ub else np.zeros((0, n + 1))
        b_ub = np.array(b_ub) if b_ub else np.zeros(0)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * (n + 1)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - start))})
        
        if not res.success:
            break
        
        a_new = res.x[:n]
        M_new = res.x[n]
        
        # Check feasibility for ALL constraints
        c_new = _fft_convolve(a_new, a_new)
        
        # Find violated constraints
        violations = []
        for k in range(2 * n - 1):
            if k in active_set:
                continue
            if c_new[k] > M_new + tol * max(1.0, M_new):
                violations.append((c_new[k] - M_new, k))
        
        if len(violations) == 0:
            # Feasible! Update and check if we improved
            a_new = _normalize(a_new)
            c_new = _fft_convolve(a_new, a_new)
            score_new = 2.0 * n * c_new.max()
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
            c0 = c_new
            current_score = score_new
            # Reset active set to just the max
            active_set = [int(np.argmax(c0))]
            break
        else:
            # Add most violated constraints
            violations.sort(reverse=True)
            n_add = min(len(violations), n_constraints_max - len(active_set), 50)
            for _, k in violations[:n_add]:
                if k not in active_set:
                    active_set.append(k)
            
            # Update a0 for next linearization
            a = a_new
            c0 = c_new
    
    return best_a, best_score

                best_a = a.copy()
                if verbose:
                    print(f"  SLP it={it} max={best_max:.8f} score={2*n*best_max:.6f}")
        else:
            # No improvement from LP solution; add constraints and continue
            pass
        
        # Update active set: add most violated constraints
        violations = []
        for k in range(2 * n - 1):
            if k not in active_set:
                viol = c_cur[k] - 2.0 * n * best_max * a.sum()**2 / (2.0 * n)
                # Actually just check c_cur[k] vs current max
                if c_cur[k] > max_cur * 0.99:
                    violations.append((c_cur[k], k))
        
        # Keep top constraints that are near the max
        if len(active_set) < 300:
            violations.sort(reverse=True)
            for _, k in violations[:30]:
                if k not in active_set:
                    active_set.append(k)
        
        # Also always keep the argmax
        am = int(np.argmax(c_cur))
        if am not in active_set:
            active_set.append(am)
        
        # Convergence check
        if max_cur < best_max * (1 - 1e-10):
            best_max = max_cur
            best_a = a.copy()
        
        if max_cur <= best_max * (1 + 1e-12) and len(active_set) > 50:
            # Nearly converged
            if verbose and it > 5:
                print(f"  SLP converged at it={it}, score={2*n*best_max:.6f}")
            break
    
    return best_a, float(2.0 * n * best_max)


def _smoothed_max_optimize(a0, rng, time_limit=300.0, beta_start=50.0, beta_end=3000.0, verbose=True):
    """
    Smoothed max multiplicative gradient descent on the simplex.
    Uses softmax smoothing of the max of convolution coefficients.
    """
    start_time = time.monotonic()
    a = _normalize(a0.copy())
    n = len(a)
    floor = 1e-20 / n
    
    c = _fft_conv(a, a)
    best_a = a.copy()
    best_val = float(c.max())
    
    beta = beta_start
    alpha = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() - start_time < time_limit:
        # Compute softmax weights on convolution
        shift = float(c.max())
        e = np.exp(beta * (c - shift))
        s = e.sum()
        if s <= 0 or not np.isfinite(s):
            w = np.ones(c.size) / c.size
        else:
            w = e / s
        
        # Gradient direction: w convolved with a (valid correlation)
        # h[i] = sum_j w[j] * a[j] where j = i (correlation)
        # Actually we need: grad of sum_k w[k] * c[k] w.r.t. a[i]
        # c[k] = sum_i a[i]*a[k-i], so d(c[k])/da[i] = a[k-i] (if valid) + a[i-(k-i)]... 
        # Actually d(c[k])/da[i] = a[k-i] + a[i] * d(k-i)/da[i]... 
        # c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i)
        # But if k-i == i (i.e., 2i=k), then both terms give a[i], so dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if k=2i, else a[k-i].
        # Wait no. c[k] = sum_j a[j]*a[k-j]. d/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
        # = a[k-i] + a[i] (if both k-i and i are valid indices)
        # So dc[k]/da[i] = a[k-i] + a[i-k]... no.
        # Let me be more careful. c[k] = sum_{j=0}^{n-1} a[j]*a[k-j] (with a out of range = 0).
        # dc[k]/da[i] = a[k-i] (setting j=i) + a[i] * d(a[k-j])/da[i] (setting k-j=i, j=k-i)
        # = a[k-i] + a[k-i] (if k-i != i) or a[k-i]*2 (if k-i == i)
        # Actually: dc[k]/da[i] = a[k-i] + delta(i, k-i) * a[i] ... no.
        # c[k] = sum_j a[j] * a[k-j]. Two terms involve a[i]:
        #   j = i: gives a[i] * a[k-i]
        #   k-j = i, i.e., j = k-i: gives a[k-i] * a[i]
        # If i != k-i: dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i]
        # If i == k-i (i = k/2): dc[k]/da[i] = 2*a[i] (same thing)
        # So in all cases: dc[k]/da[i] = 2*a[k-i] (if 0 <= k-i < n)
        # Wait, that's not right either when i = k-i.
        # If i = k/2: the term a[j]*a[k-j] with j=i gives a[i]*a[i] = a[i]^2.
        #   d/da[i] of a[i]^2 = 2*a[i]. And 2*a[k-i] = 2*a[i]. OK consistent.
        # If i != k/2: two distinct terms a[i]*a[k-i] and a[k-i]*a[i], total = 2*a[i]*a[k-i].
        #   d/da[i] = 2*a[k-i]. And 2*a[k-i] from formula. OK.
        # So dc[k]/da[i] = 2 * a[k-i] if 0 <= k-i < n, else 0.
        
        # Therefore grad = sum_k w[k] * 2*a[k-i] = 2 * (w * a_reversed)[i] (cross-correlation)
        # More precisely: h[i] = 2 * sum_k w[k] * a[k-i] (where a[k-i] is valid)
        # This is 2 * cross-correlation of w and a.
        
        # Compute h[i] = sum_{k} w[k] * a[k-i] for valid k-i
        # = sum_{j >= 0} w[i+j] * a[j] ... this is a correlation.
        # Use FFT: h = 2 * ifft(conj(fft(w)) * fft(a))[0:n]
        
        L = next_fast_len(max(2 * n, len(w)))
        fw = rfft(w, L)
        fa = rfft(a, L)
        # Cross-correlation: sum_k w[k]*a[k-i] = ifft(conj(fw)*fa)
        h = 2.0 * irfft(np.conjugate(fw) * fa, L)[:n]
        h = np.maximum(h, 0.0)
        
        if h.max() < 1e-18:
            # Degenerate; perturb
            a_try = _normalize(a * np.exp(rng.normal(0, 0.05, n)))
            c_try = _fft_conv(a_try, a_try)
            if c_try.max() < best_val * 1.05:
                a = a_try
                c = c_try
            stall += 1
        else:
            h = h / h.max()
            
            # Try multiple step sizes
            best_t = None
            best_c_val = float(c.max())
            for mult in (0.25, 0.5, 1.0, 2.0, 4.0):
                a_try = alpha * mult
                dec = np.exp(-a_try * h)
                a_new = a * dec
                a_new = np.maximum(a_new, floor)
                s_new = a_new.sum()
                if s_new <= 0:
                    continue
                a_new = a_new / s_new
                c_new = _fft_conv(a_new, a_new)
                m_new = float(c_new.max())
                if m_new < best_c_val:
                    best_c_val = m_new
                    best_t = (a_new, c_new)
            
            if best_t is not None:
                a, c = best_t
                stall = 0
                if c.max() < best_val:
                    best_val = float(c.max())
                    best_a = a.copy()
                    if verbose and it % 20 == 0:
                        print(f"  SmoothMax it={it} score={2*n*best_val:.6f} beta={beta:.0f}")
            else:
                stall += 1
                alpha = max(0.01, alpha * 0.7)
                if stall > 30:
                    # Perturb and reset
                    a_try = _normalize(a * np.exp(rng.normal(0, 0.1, n)))
                    c_try = _fft_conv(a_try, a_try)
                    if c_try.max() < best_val * 1.1:
                        a = a_try
                        c = c_try
                    else:
                        a = best_a.copy()
                        c = _fft_conv(a, a)
                    alpha = 0.1
                    stall = 0
        
        it += 1
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.1)
    
    return best_a, float(2.0 * n * best_val)


def _block_scale(a0, rng, time_limit=200.0, verbose=True):
    """Block multiplicative scaling local search."""
    start_time = time.monotonic()
    a = _normalize(a0.copy())
    n = len(a)
    floor = 1e-20 / n
    
    c = _fft_conv(a, a)
    best_a = a.copy()
    best_val = float(c.max())
    
    factors = (0.4, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.5)
    stagnant = 0
    
    while time.monotonic() - start_time < time_limit:
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            w = int(rng.integers(n // 20, min(n, n // 5 + 1)))
        w = max(1, min(n, w))
        
        if w >= n:
            start = 0
        else:
            start = int(rng.integers(0, n - w + 1))
        
        improved = False
        for f in factors:
            cand = a.copy()
            cand[start:start + w] *= f
            cand = _normalize(cand)
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            c_cand = _fft_conv(cand, cand)
            m_cand = float(c_cand.max())
            if m_cand < best_val - 1e-14:
                a = cand
                c = c_cand
                best_val = m_cand
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            stagnant += 1
            if stagnant > 100:
                a_try = _normalize(a * np.exp(rng.normal(0, 0.05, n)))
                c_try = _fft_conv(a_try, a_try)
                if c_try.max() < best_val * 1.02:
                    a = a_try
                    c = c_try
                else:
                    a = best_a.copy()
                    c = _fft_conv(a, a)
                stagnant = 0
        
        if verbose:
            elapsed = time.monotonic() - start_time
            if int(elapsed) % 30 == 0 and int(elapsed) > 0:
                print(f"  BlockScale t={int(elapsed)}s score={2*n*best_val:.6f}")
    
    return best_a, float(2.0 * n * best_val)


def _make_starts(n, rng, parent=None):
    """Generate diverse starting points."""
    starts = []
    t = np.arange(n, dtype=float) / n
    
    # 1. Two-block (asymmetric)
    for frac in (0.4, 0.5, 0.6):
        for ratio in (1.5, 2.0, 3.0, 5.0):
            p = np.ones(n)
            idx = int(frac * n)
            p[idx:] = ratio
            starts.append(_normalize(p))
    
    # 2. Cosine window
    starts.append(_normalize(0.5 * (1 + np.cos(2 * np.pi * t))))
    
    # 3. Raised cosine variant
    starts.append(_normalize(0.5 * (1 + np.cos(4 * np.pi * t))))
    
    # 4. Gaussian bump
    starts.append(_normalize(np.exp(-((t - 0.5) / 0.2) ** 2)))
    
    # 5. Two Gaussian bumps
    starts.append(_normalize(np.exp(-((t - 0.3) / 0.15) ** 2) + 0.7 * np.exp(-((t - 0.7) / 0.15) ** 2)))
    
    # 6. Random smooth
    for mode in range(4):
        K = int(rng.integers(2, 7))
        coeffs = rng.normal(0, 1, K)
        p = np.ones(n)
        for k in range(K):
            p += coeffs[k] * np.cos(2 * np.pi * (k + 1) * t + rng.uniform(0, 2 * np.pi))
        p = np.abs(p) + 0.1
        starts.append(_normalize(p))
    
    # 7. Block levels
    K = int(rng.integers(3, 8))
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks_p = [0.0] + list(edges) + [1.0]
    heights = rng.lognormal(0, 0.8, K)
    p = np.zeros(n)
    for i in range(K):
        s = int(blocks_p[i] * n)
        e = max(s + 1, int(blocks_p[i + 1] * n))
        p[s:e] = heights[i]
    starts.append(_normalize(p))
    
    # 8. Parent (if available)
    if parent is not None:
        starts.append(_normalize(parent.copy()))
    
    return starts


def _resample(a, n_new):
    """Resample a normalized sequence to new length."""
    a = _normalize(a)
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):

    Returns a list of non-negative floats.

    try:
        iseed = int(seed)
    except (ValueError, TypeError):
        iseed = 42
    rng = np.random.default_rng(iseed)

    try:
        parent = _normalize(np.asarray(height_sequence_1, dtype=float))
    except (NameError, Exception):
        parent = None
    
    n = 3072
    
    # Phase 1: Generate and evaluate starts
    print(f"Phase 1: Generating starts (n={n})")
    starts = _make_starts(n, rng, parent)
    
    scored_starts = []
    for s in starts:
        sc = _score(s)
        if np.isfinite(sc):
            scored_starts.append((sc, s))
    
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Best start score: {scored_starts[0][0]:.6f}")
    
    best_a = scored_starts[0][1].copy()
    best_score = scored_starts[0][0]

    print(f"Phase 2: SLP optimization")
    elapsed = time.monotonic() - total_start
    remaining = deadline - time.monotonic()
    
    top_starts = [s for sc, s in scored_starts[:3]]
    slp_results = []
    
    for i, s in enumerate(top_starts):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(200.0, max(30.0, rem * 0.3))
        print(f"  SLP start {i}, score={_score(s):.6f}, time={t_alloc:.0f}s")
        sa, ss = _slp_optimize(s, max_iters=200, time_limit=t_alloc, verbose=True)

            print(f"  New best: {best_score:.6f}")
    
    # Also try SLP from a random start different from the top ones
    rem = deadline - time.monotonic()
    if rem > 100 and len(scored_starts) > 3:
        extra_start = scored_starts[int(rng.integers(3, min(8, len(scored_starts))))][1]
        t_alloc = min(150.0, max(30.0, rem * 0.2))

        sa, ss = _slp_optimize(extra_start, max_iters=200, time_limit=t_alloc, verbose=True)
        if ss < best_score:
            best_score = ss
            best_a = sa.copy()
            print(f"  New best: {best_score:.6f}")
    
    # Phase 3: Smoothed max refinement
    print(f"Phase 3: Smoothed max refinement")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_alloc = min(400.0, max(60.0, rem * 0.35))

        sm_a, sm_s = _smoothed_max_optimize(

        )
        if sm_s < best_score:

    # Phase 4: Block scaling

        if ssb < best_score:

            best_a = sb.copy()
            print(f"  Big SLP best: {best_score:.6f}")
        
        rem = deadline - time.monotonic()
        if rem > 100:
            t2 = min(150.0, rem * 0.4)
            sbb, ssbb = _smoothed_max_optimize(
                best_a, rng, time_limit=t2,
                beta_start=100.0, beta_end=5000.0, verbose=True
            )
            if ssbb < best_score:
                best_score = ssbb
                best_a = sbb.copy()
                print(f"  Big SmoothMax best: {best_score:.6f}")
    
    # Final: verify and return
    rem = deadline - time.monotonic()
    best_a = _normalize(best_a)
    
    # Verify with official evaluator
    final_list = [float(x) for x in best_a]

    if rem > 5:

                final_score = ev_score
        except Exception as e:

    print(f"Final: n={len(best_a)}, score={final_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")
    
    return final_list
