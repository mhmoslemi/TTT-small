def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        a.fill(1.0)
        return a / len(a)
    return a / s


def _autocorr(a, L):
    """Compute (a*a)[:n] via FFT."""
    n = len(a)
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(A * A, L)[:n]
    np.maximum(r, 0.0, out=r)
    return r


def _score_from_autocorr(r, n):
    return 2.0 * n * r.max()


def _gradient(a, r, beta, L, n):
    """
    Compute gradient of the smoothed max objective.
    beta is the inverse temperature (higher = sharper max approximation).
    """
    # w[k] = softmax(beta * n * r[k])
    v = beta * n * r
    v_max = v.max()
    z = v - v_max
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    w = e / e.sum()

    W = np.fft.rfft(w, L)

    backward = np.fft.irfft(np.conj(A) * W, L)[:n]
    
    np.maximum(forward, 0.0, out=forward)
    np.maximum(backward, 0.0, out=backward)
    
    grad = n * (forward + backward)
    return grad

def _normalize(a):
    """Normalize a to sum to 1, ensuring non-negativity."""
    a = np.asarray(a, dtype=np.float64).copy()
    a[np.isnan(a) | np.isinf(a)] = 0.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def _autocorr(a, L):
    """Compute autocorrelation (a*a)[:n] via FFT."""
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(A * A, L)
    np.maximum(r, 0.0, out=r)
    return r[:len(a)]


def _score(r, n):
    """Score from autocorrelation."""
    return 2.0 * n * float(r.max())


def _compute_gradient(a, r, w, L, n):
    """Compute gradient of smoothed max objective."""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    first = np.fft.irfft(np.conj(W) * A, L)[:n]
    second = np.fft.irfft(W * A, L)[:n]
    np.maximum(first, 0.0, out=first)
    np.maximum(second, 0.0, out=second)
    return n * (first + second)

        frac = min(1.0, elapsed / max(time_limit, 0.1))
        beta = beta_start * (beta_end / beta_start) ** frac

                a = _normalize((1 - mix) * best_a + mix * noise)
            else:
                a = _normalize(best_a * (1 + rng.normal(0, 0.03, n)))
            patience = 0

        # Compute softmax weights

        if v_max > 700.0 / beta:
            # Extreme case: effectively hard max

            w[mask] = 1.0 / mask.sum()
        else:

            e = np.exp(z)
            w = e / e.sum()
        
        # Gradient
        grad = _compute_gradient(a, r, w, L, n)
        
        # Centered multiplicative update
        g_mean = grad.mean()
        g_std = grad.std() + 1e-15
        exponent = -step * (grad - g_mean) / g_std
        exponent = np.clip(exponent, -5.0, 5.0)

            patience = 0

        else:
            step = max(0.001, step * 0.7)
    
    return best_a, best_score

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 via FFT-based smoothed gradient descent.
Uses log-sum-exp approximation of max with adaptive temperature, centered
multiplicative updates, cross-correlation-based O(n log n) gradients,
and multi-restart strategy with diverse initializations.
"""

import math
import time
import numpy as np


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a[~np.isfinite(a)] = 0.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _autocorr(a, L):
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(A * A, L)
    np.maximum(r, 0.0, out=r)
    n = len(a)
    return r[:n]


def _score(r, n):
    return 2.0 * n * float(r.max())


def _compute_gradient(a, r, w, L, n):
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    first = np.fft.irfft(np.conj(W) * A, L)[:n]
    second = np.fft.irfft(W * A, L)[:n]
    np.maximum(first, 0.0, out=first)
    np.maximum(second, 0.0, out=second)
    return n * (first + second)


def _softmax_weights(r, beta, n):
    v = beta * n * r
    v_max = float(v.max())
    if v_max <= 0:
        return np.ones(n, dtype=np.float64) / n
    z = v - v_max
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(n, dtype=np.float64)
        w[v == v_max] = 1.0 / np.sum(v == v_max)
        return w
    return e / s


def _smoothed_optimize(a, L, n, time_limit, beta_start, beta_end,
                       step_init, rng, print_prefix=""):
    t0 = time.monotonic()
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(_autocorr(a, L), n)
    step = step_init
    patience = 0
    it = 0
    min_step = 0.002
    max_step = 0.5

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 0.1))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        r = _autocorr(a, L)
        cur = _score(r, n)

        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 600:
            if patience > 1500:
                noise = np.abs(rng.normal(0, 0.5, n))
                mix = 0.3 + 0.4 * rng.random()
                a = _normalize((1.0 - mix) * best_a + mix * noise)
            else:
                a = _normalize(best_a * (1.0 + rng.normal(0, 0.02, n)))
            patience = 0
            step = step_init
            it = 0
            continue

        w = _softmax_weights(r, beta, n)
        grad = _compute_gradient(a, r, w, L, n)

        g_mean = float(grad.mean())
        g_std = float(grad.std())
        if g_std < 1e-15:
            g_std = max(g_mean, 1e-10)

        best_new = a
        best_new_score = cur
        best_trial_step = 0.0

        trials = (step, step * 0.5, step * 1.5, step * 0.25)
        for ts in trials:
            if ts < 1e-10 or ts > 5.0:
                continue
            exp_t = -ts * (grad - g_mean) / g_std
            exp_t = np.clip(exp_t, -8.0, 8.0)
            a_new = a * np.exp(exp_t)
            a_new = _normalize(a_new)
            r_new = _autocorr(a_new, L)
            sc = _score(r_new, n)
            if sc < best_new_score:
                best_new_score = sc
                best_new = a_new
                best_trial_step = ts

        if best_new_score < cur - 1e-14:
            a = best_new
            if best_trial_step == step:
                step = min(max_step, step * 1.08)
            else:
                step = max(min_step, step * 0.85)
            patience = 0
        else:
            step = max(min_step, step * 0.65)
            if step <= min_step * 1.1:
                a = _normalize(best_a * (1.0 + rng.normal(0, 0.01, n)))
                step = step_init * 0.5

        it += 1
        if it % 500 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if print_prefix and rem > 0:
                print(f"  {print_prefix} it={it} score={best_score:.7f} rem={rem:.0f}s")

    return best_a, best_score
