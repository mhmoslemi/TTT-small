Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based descent and 
smoothed-max gradient optimization with multiple scales.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _sanitize(a):
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return None
    return a / s

def _exact_score(a):
    a = _sanitize(a)
    if a is None:
        return float('inf')
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c))

def _conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    return A

def _lp_descent_step(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    A = _conv_matrix(f)
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_descent(f0, rng, time_limit, verbose=False):
    """LP-based descent: iteratively solve LP and blend."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    
    n = len(f)
    best = f.copy()
    best_score = _exact_score(f)
    f = f.copy()
    
    deadline = time.monotonic() + time_limit
    t = 0.3  # initial blend parameter
    stagnant = 0
    
    while time.monotonic() < deadline:
        cur_score = _exact_score(f)
        if cur_score < best_score:
            best = f.copy()
            best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
        
        M = np.max(np.convolve(f, f))
        g = _lp_descent_step(f, M)
        
        if g is None:
            # LP failed, use gradient step instead
            c = np.convolve(f, f)
            j_star = np.argmax(c)
            # gradient of c[j_star] w.r.t. f
            grad = np.zeros(n)
            for i in range(n):
                j = j_star - i
                if 0 <= j < n:

            f = f * np.exp(-0.1 * grad / max(1e-15, np.max(grad)))

            if f is None:

            continue
        
        g = _sanitize(g)

            t *= 0.5

smoothed-max gradient optimization, multi-scale search, and local block polishing.
Combines the theoretical LP approach with practical gradient heuristics.

import time
from scipy.optimize import linprog
from scipy import sparse as sp_sparse


def _sanitize(a):
    """Clip to valid non-negative vector normalized to sum 1."""
    a = np.asarray(a, dtype=float)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:

def _exact_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    a = _sanitize(a)
    if a is None:

    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c))


def _conv_full(a, b):
    """Full convolution using FFT for large inputs."""
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b)
    L = 1
    while L < nout:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]

        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)

        remaining = np.setdiff1d(np.arange(len(c_full)), top_idx)
        bottom_idx = remaining[np.random.permutation(len(remaining))][:bottom_k]
        active_idx = np.concatenate([top_idx, bottom_idx])
    
    A = _conv_matrix_active(f, active_idx)
    if A.shape[0] == 0:
        return None
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def _lp_descent(f0, rng, time_limit, verbose=False):
    """Iterative LP-based descent with adaptive blending."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_score = _exact_score(f)
    f = f.copy()
    deadline = time.monotonic() + time_limit
    t_blend = 0.25
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        if time.monotonic() - deadline > 5:
            break
        cur_score = _exact_score(f)
        if cur_score < best_score:
            best = f.copy()
            best_score = cur_score
            stagnant = 0
            if verbose and it % 50 == 0:
                print(f"  LP iter {it}: score={best_score:.6f}, t={t_blend:.4f}")
        else:
            stagnant += 1

        c = np.convolve(f, f)
        M = float(np.max(c))

        # Try LP step
        g = _lp_step(f, M, n_active=min(400, 2 * len(c)))
        if g is not None:
            g = _sanitize(g)
            if g is not None:
                f_new = (1.0 - t_blend) * f + t_blend * g
                f_new = _sanitize(f_new)
                if f_new is not None:
                    new_score = _exact_score(f_new)
                    if new_score < cur_score + 1e-9:
                        f = f_new
                        t_blend = min(0.4, t_blend * 1.05)
                        it += 1
                        continue
        # If LP didn't help or failed, try gradient step
        c = np.convolve(f, f)
        j_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = j_star - i
            if 0 <= j < n:
                grad[i] = f[j]
        grad_max = np.max(grad)
        if grad_max > 1e-15:
            lr = 0.15
            f = f * np.exp(-lr * grad / grad_max)
            f = _sanitize(f)
            if f is None:
                f = best.copy()
        
        # Adaptive step size
        if stagnant > 10:
            t_blend = max(0.02, t_blend * 0.8)
        if stagnant > 30:
            f = best.copy()
            t_blend = 0.15
            stagnant = 0
            # Perturb
            f = f * np.exp(rng.normal(0, 0.03, n))
            f = _sanitize(f)
        it += 1

    return best, best_score


def _smoothed_max_gradient(f, beta):
    """Compute gradient of smoothed max of conv(f,f)."""
    n = len(f)
    c = np.convolve(f, f)
    if len(c) == 0:
        return None, float('inf')
    # Smoothed max weights
    c_shifted = c - np.max(c)
    w = np.exp(beta * c_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum
    # Gradient: d/d f[i] of sum_k w_k * (f*f)[k]
    # (f*f)[k] = sum_j f[j]*f[k-j], so d/d f[i] = sum_k w_k * (f[k-i] + f[i-k])
    # For full convolution: (f*f)[k] = sum_{j} f[j]*f[k-j] (valid j)
    # d(df*f)/df[i] at position k = f[k-i] (if valid)
    # So gradient[i] = sum_k w_k * f[k-i] (for valid k-i)
    grad = np.zeros(n)
    for i in range(n):
        # (f*f)[k] depends on f[i] when k-i is in [0, n-1] or k is in [i, i+n-1]
        # grad[i] = sum_{k} w[k] * f[k-i] for k where 0 <= k-i < n and 0 <= k < 2n-1
        # Also by symmetry: grad[i] = sum_{k} w[k] * f[i-(k-n+1)] ... 
        # Actually: (f*f)[k] = sum_j f[j] f[k-j]
        # d/d f[i] (f*f)[k] = f[k-i] (if 0<=k-i<n) + f[i-k+n-1]... no
        # Simply: d/d f[i] (f*f)[k] = f[k-i] if 0 <= k-i <= n-1
        k_lo = i
        k_hi = i + n - 1
        k_lo = max(0, k_lo)
        k_hi = min(2 * n - 2, k_hi)
        if k_hi >= k_lo:
            grad[i] = np.dot(w[k_lo:k_hi + 1], f[0:n - k_lo + k_hi - k_lo + 1][:n])
            # More carefully: sum_{k=k_lo}^{k_hi} w[k] * f[k-i]
            indices = np.arange(k_lo, k_hi + 1)
            f_indices = indices - i  # these are in [0, n-1]
            grad[i] = np.dot(w[indices], f[f_indices])
    return grad, float(2.0 * n * np.max(c))


def _gradient_descent(f0, rng, time_limit, beta_start=50.0, beta_end=2000.0, verbose=False):
    """Smoothed max gradient descent with adaptive step size."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_score = _exact_score(f)
    f = f.copy()
    deadline = time.monotonic() + time_limit
    beta = beta_start
    lr = 0.5
    stagnant = 0
    it = 0
    floor = 1e-18 / n
    f = np.maximum(f, floor)
    f = f / f.sum()

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        c_shifted = c - np.max(c)
        w = np.exp(beta * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30 or not np.isfinite(w_sum):
            w = np.ones(len(c)) / len(c)
        else:
            w = w / w_sum

        # Compute gradient efficiently using convolution
        # grad[i] = sum_k w[k] * f[k-i] for valid indices
        # This is a cross-correlation of w with f
        # grad = ifft(conj(fft(f)) * fft(w)) ... need to be careful with lengths
        # Simpler: use the fact that grad[i] = (w * f_reversed)[i + n - 1] style
        # Actually: grad[i] = sum_{k} w[k] f[k-i] = (w cross- correlate f)[i]
        # = conv(w, f_reversed)[i + (len(f)-1)] ... let me just use direct for small n
        if n <= 2048:
            grad = np.zeros(n)
            for i in range(n):
                k_vals = np.arange(i, i + n)
                k_vals = k_vals[k_vals < len(w)]
                if len(k_vals) > 0:
                    grad[i] = np.dot(w[k_vals], f[k_vals - i])
        else:
            # FFT-based
            L = 1
            while L < 2 * n + len(w) - 1:
                L <<= 1
            fw = np.fft.rfft(w, L)
            ff = np.fft.rfft(f, L)
            # We want grad[i] = sum_k w[k] f[k-i] = (w * f')[i] where f' = f reversed
            # Actually this is the cross-correlation
            # corr[i] = sum_k w[k] f[k-i]
            # = conv(w, flip(f))[i + n - 1]
            f_flip = f[::-1]
            L2 = 1
            while L2 < n + n - 1:
                L2 <<= 1
            fw2 = np.fft.rfft(w, L2)
            ff2 = np.fft.rfft(f_flip, L2)
            conv_wf = np.fft.irfft(fw2 * ff2, L2)
            grad = conv_wf[n - 1:n - 1 + n]

        grad = np.maximum(grad, 0.0)
        gmax = np.max(grad)
        if gmax < 1e-15:
            # Degenerate: perturb
            f = f * np.exp(rng.normal(0, 0.05, n))
            f = np.maximum(f, floor)
            f = f / f.sum()
            stagnant += 1
        else:
            grad_normed = grad / gmax
            # Try to decrease: f -> f * exp(-lr * grad_normed)
            best_step = None
            for trial_lr in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
                f_try = f * np.exp(-trial_lr * grad_normed)
                f_try = np.maximum(f_try, floor)
                s = f_try.sum()
                if s < 1e-15:
                    continue
                f_try = f_try / s
                sc = 2.0 * n * np.max(np.convolve(f_try, f_try))
                if best_step is None or sc < best_step[1]:
                    best_step = (f_try, sc, trial_lr)

            if best_step is not None:
                f, sc, best_step[2] = best_step
                lr = best_step[2]
                if sc < best_score:
                    best = f.copy()
                    best_score = sc
                    stagnant = 0
                    if verbose and it % 100 == 0:
                        print(f"  Grad iter {it}: score={best_score:.6f}, beta={beta:.0f}")
                else:
                    stagnant += 1
            else:
                stagnant += 1
                lr = max(0.01, lr * 0.5)

        if stagnant > 15:
            lr = max(0.01, lr * 0.5)
        if stagnant > 40:
            f = best.copy()
            lr = 0.5
            beta = max(beta_start, beta * 0.5)
            stagnant = 0
            f = f * np.exp(rng.normal(0, 0.05, n))
            f = np.maximum(f, floor)
            f = f / f.sum()

        beta = min(beta_end, beta * 1.08)
        it += 1

    return best, best_score


def _block_search(f0, rng, time_limit, verbose=False):
    """Local search: multiply random contiguous blocks by factors."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = len(f)
    best = f.copy()
    best_score = _exact_score(f)
    f = f.copy()
    deadline = time.monotonic() + time_limit
    floor = 1e-18 / n
    f = np.maximum(f, floor)
    f = f / f.sum()
    stagnant = 0
    factors = [0.4, 0.65, 0.85, 1.15, 1.35, 1.8, 2.5]

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 200 + 1)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 200), max(3, n // 30 + 1)))
        else:
            lo = max(3, n // 40)
            hi = max(lo + 1, min(n, n // 8 + 1))
            w = int(rng.integers(lo, hi + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = f.copy()
        cur_best = base
        cur_score = _exact_score(base)

        for fac in factors:
            cand = base.copy()
            cand[start:start + w] *= fac
            cand = np.maximum(cand, floor)
            s = cand.sum()
            if s < 1e-15:
                continue
            cand = cand / s
            sc = 2.0 * n * np.max(np.convolve(cand, cand))
            if sc < cur_score - 1e-12:
                cur_score = sc
                cur_best = cand.copy()

        if cur_best is not base:
            f = cur_best
            if cur_score < best_score:
                best = f.copy()
                best_score = cur_score
                stagnant = 0
                if verbose and stagnant == -1:
                    pass
        else:
            stagnant += 1
            if stagnant > 50:
                f_try = f * np.exp(rng.normal(0, 0.05, n))
                f_try = np.maximum(f_try, floor)
                f_try = f_try / f_try.sum()
                sc = 2.0 * n * np.max(np.convolve(f_try, f_try))
                if sc < best_score * 1.05:
                    f = f_try
                else:
                    f = best.copy()
                stagnant = 0
            if stagnant > 300:
                f = best.copy()
                stagnant = 0

    return best, best_score


def _resample_to(a, m):
    """Resample normalized vector to length m."""
    a = _sanitize(a)
    if a is None:
        return None
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _random_start(n, rng, mode=0):
    """Generate a diverse random non-negative normalized sequence."""
    t = (np.arange(n) + 0.5) / n
    if mode == 0:
        # Skewed exponential
        rate = rng.uniform(0.5, 5.0)
        direction = rng.choice([-1, 1])
        p = np.exp(direction * rate * (t - 0.5))
    elif mode == 1:
        # Beta-like
        a_param = rng.uniform(0.3, 3.0)
        b_param = rng.uniform(0.3, 3.0)
        p = np.power(np.clip(t, 1e-8, 1 - 1e-8), a_param) * np.power(1 - np.clip(t, 1e-8, 1 - 1e-8), b_param)
    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 4))
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.03, 0.3)
            w = rng.uniform(0.5, 2.0)
            p += w * np.exp(-((t - c) / s) ** 2)
    elif mode == 3:
        # Two-block asymmetric
        split = rng.uniform(0.2, 0.8)
        ratio = rng.uniform(0.2, 5.0)
        p = np.ones(n)
        idx = int(split * n)
        p[idx:] *= ratio
    else:
        # Smooth random spectrum
        p = np.ones(n)
        for k in range(1, 8):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(p, -50, 50)))

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0) + 1e-10
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return p / s


def _two_block_seed(n):
    """Deterministic two-block seed that tends to be good."""
    p = np.ones(n)
    # Try several splits and ratios, return best
    best_p = None
    best_sc = float('inf')
    for a in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [0.3, 0.5, 0.8, 1.5, 2.5]:
            p = np.ones(n)
            idx = int(a * n)
            p[idx:] *= ratio
            p = _sanitize(p)
            if p is not None:
                sc = _exact_score(p)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p.copy()
    return best_p


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses LP-based descent, gradient methods, and local search.
    """
    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget_s - 30.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng((iseed % 1000000007) + 7)

    best_arr = None
    best_score = float('inf')

    # Load parent if available
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    if parent_arr is not None:
        s = _exact_score(parent_arr)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_arr = parent_arr.copy()
        print(f"Parent score: {best_score:.6f}, n={len(parent_arr)}")

    # ---- Phase 1: Quick exploration (60s) ----
    t1_deadline = start_time + 60.0
    small_n = 1024

    starts = []
    if parent_arr is not None:
        rs = _resample_to(parent_arr, small_n)
        if rs is not None:
            starts.append(rs)
        # Also try perturbed version
        rs2 = rs * np.exp(rng.normal(0, 0.1, small_n))
        rs2 = _sanitize(rs2)
        if rs2 is not None:
            starts.append(rs2)

    tb = _two_block_seed(small_n)
    if tb is not None:
        starts.append(tb)
        # Smoothed version
        from scipy.ndimage import gaussian_filter1d
        tb_smooth = gaussian_filter1d(tb, 2.0)
        tb_smooth = _sanitize(tb_smooth)
        if tb_smooth is not None:
            starts.append(tb_smooth)

    for mode in range(5):
        starts.append(_random_start(small_n, rng, mode))

    # Quick gradient on each start
    scored_starts = []
    for p0 in starts:
        if time.monotonic() > t1_deadline:
            break
        p0 = _sanitize(p0)
        if p0 is None:
            continue
        sc0 = _exact_score(p0)
        # Brief gradient descent (10s each)
        p, sc = _gradient_descent(p0, rng, 12.0, beta_start=50.0, beta_end=500.0)
        if p is not None and np.isfinite(sc) and sc < sc0:
            scored_starts.append((sc, p))
        else:
            scored_starts.append((sc0, p0))

    scored_starts.sort(key=lambda x: x[0])
    print(f"Phase 1 best small-n score: {scored_starts[0][0]:.6f}" if scored_starts else "Phase 1: no starts")

    # Update global best
    if scored_starts:
        sc_best_sm, p_best_sm = scored_starts[0]
        if sc_best_sm < best_score:
            best_score = sc_best_sm
            best_arr = p_best_sm.copy()

    # ---- Phase 2: LP-based optimization at moderate n (700s) ----
    t2_deadline = start_time + 760.0
    mid_n = 2048
    # Upsample best to mid_n
    mid_starts = []
    if parent_arr is not None:
        rs = _resample_to(parent_arr, mid_n)
        if rs is not None:
            mid_starts.append(rs)
    if scored_starts:
        for sc, p in scored_starts[:3]:
            rs = _resample_to(p, mid_n)
            if rs is not None:
                mid_starts.append(rs)
    if not mid_starts:
        mid_starts.append(_random_start(mid_n, rng, 0))

    # Also add a fresh two-block seed at mid_n
    tb_mid = _two_block_seed(mid_n)
    if tb_mid is not None:
        mid_starts.append(tb_mid)

    for idx, p0 in enumerate(mid_starts[:4]):
        if time.monotonic() > t2_deadline - 60:
            break
        remaining = t2_deadline - time.monotonic()
        t_alloc = min(180.0, max(30.0, remaining / (len(mid_starts[:4]) - idx)))
        print(f"Phase 2: LP descent start {idx}/{len(mid_starts[:4])}, budget={t_alloc:.0f}s")
        p, sc = _lp_descent(p0, rng, t_alloc, verbose=True)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  New best (LP): {best_score:.6f}")
        # Also try gradient after LP
        if p is not None and time.monotonic() < t2_deadline - 30:
            t_grad = min(60.0, t2_deadline - time.monotonic() - 10)
            p2, sc2 = _gradient_descent(p, rng, t_grad, beta_start=100.0, beta_end=1000.0)
            if p2 is not None and np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_arr = p2.copy()
                print(f"  New best (post-LP grad): {best_score:.6f}")

    # ---- Phase 3: Gradient optimization at best n (500s) ----
    t3_deadline = start_time + 1260.0
    if best_arr is not None:
        # Try at original best length and at 3072
        for target_n in [len(best_arr), 3072]:
            if time.monotonic() > t3_deadline - 100:
                break
            if target_n == len(best_arr):
                p0 = best_arr.copy()
            else:
                p0 = _resample_to(best_arr, target_n)
            if p0 is None:
                continue
            remaining = t3_deadline - time.monotonic()
            t_alloc = min(250.0, max(60.0, remaining * 0.6))
            print(f"Phase 3: gradient n={target_n}, budget={t_alloc:.0f}s")
            p, sc = _gradient_descent(p0, rng, t_alloc, beta_start=80.0, beta_end=3000.0, verbose=True)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = p.copy()
                print(f"  New best (grad n={target_n}): {best_score:.6f}")

    # ---- Phase 4: Local block search (400s) ----
    t4_deadline = start_time + 1660.0
    if best_arr is not None:
        remaining = t4_deadline - time.monotonic()
        t_alloc = min(350.0, max(60.0, remaining * 0.7))
        print(f"Phase 4: block search, budget={t_alloc:.0f}s")
        p, sc = _block_search(best_arr, rng, t_alloc, verbose=True)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  New best (block): {best_score:.6f}")

    # ---- Phase 5: Final LP polish + upsample (300s) ----
    t5_deadline = deadline
    if best_arr is not None:
        # LP polish at current length
        remaining = t5_deadline - time.monotonic()
        if remaining > 120:
            t_alloc = min(120.0, remaining * 0.5)
            print(f"Phase 5: LP polish, budget={t_alloc:.0f}s")
            p, sc = _lp_descent(best_arr, rng, t_alloc)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = p.copy()
                print(f"  New best (polish): {best_score:.6f}")

        # Try upsample to larger n
        remaining = t5_deadline - time.monotonic()
        if remaining > 100 and len(best_arr) < 4096:
            big_n = min(4096, 2 * len(best_arr))
            big = _resample_to(best_arr, big_n)
            if big is not None:
                t_alloc = min(90.0, remaining * 0.5)
                print(f"Phase 5: upsample to n={big_n}, budget={t_alloc:.0f}s")
                p, sc = _gradient_descent(big, rng, t_alloc, beta_start=100.0, beta_end=2000.0)
                if p is not None and np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_arr = p.copy()
                    print(f"  New best (big n={big_n}): {best_score:.6f}")

    # ---- Final verification ----
    if best_arr is None:
        best_arr = np.ones(1024) / 1024.0

    best_arr = _sanitize(best_arr)
    if best_arr is None:
        best_arr = np.ones(1024) / 1024.0

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_arr]
    try:
        eval_score = float(evaluate_sequence(final_list))
        if np.isfinite(eval_score):
            print(f"Final evaluate_sequence score: {eval_score:.6f}")
            if eval_score < best_score:
                best_score = eval_score
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"Final best score: {best_score:.6f}, n={len(best_arr)}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    return final_list
