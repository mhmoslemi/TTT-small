_G_TAU = 1e-5
_G_SCALE = 1.0
_G_COST = 0.01
_G_COST_INIT = 0
_G_BOUND = 40.0
_G_CALL_DEADLINE = None
_G_LAST_Z = None


# ---------------------------------------------------------------------------
# Basic helpers
# ---------------------------------------------------------------------------

def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _set_state(n, tau=None):
    global _G_N, _G_NL, _G_NFFT, _G_TAU, _G_SCALE, _G_COST, _G_COST_INIT

    n = int(n)
    if n != _G_N:
        _G_N = n
        L = 2 * n - 1
        _G_NL = L
        _G_NFFT = _next_pow2(L)
        _G_SCALE = float(n)

        # Rough cost estimate for one FFT convolution.  The gradient objective
        # uses several FFTs, so inflate the estimate.
        try:
            a = np.ones(n, dtype=np.float64)
            t0 = time.monotonic()
            _fft_conv(a)
            dt = time.monotonic() - t0
            _G_COST = max(1e-4, dt * 10.0)
            _G_COST_INIT = n
        except Exception:
            _G_COST = 0.01
            _G_COST_INIT = n

    if tau is not None:
        _G_TAU = float(tau)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)

    z = np.clip(z, -100.0, 100.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))

    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    return e / s


def _normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def _z_from_a(a):
    a = _normalize_sequence(a)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    a = np.maximum(a, 1e-300)
    z = np.log(a)
    z = z - float(np.mean(z))
    z = np.clip(z, -_G_BOUND + 1.0, _G_BOUND - 1.0)
    return z.astype(np.float64)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

    if n == 0:
        return np.array([], dtype=np.float64)

    if n != _G_N or _G_NFFT <= 1:
        L = 2 * n - 1
        N = _next_pow2(L)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:L].real
        return np.maximum(c, 0.0)

    fa = np.fft.rfft(a, _G_NFFT)
    c = np.fft.irfft(fa * fa, _G_NFFT)[:_G_NL].real
    c = np.maximum(c, 0.0)

    if _G_NL > 1:
        c = 0.5 * (c + c[::-1])

    return c


def _score_q(a):
    """
    Returns max_k k_coeff * n for a normalized sequence, i.e. score / 2.
    """
    a = _normalize_sequence(a)
    n = a.size
    if n == 0:
        return float("inf")

    if n == _G_N and _G_NFFT > 1:
        c = _fft_conv(a)
        qmax = float(np.max(c)) * float(n)
    else:
        L = 2 * n - 1
        N = _next_pow2(L)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:L].real
        c = np.maximum(c, 0.0)
        qmax = float(np.max(c)) * float(n)

    if not np.isfinite(qmax):
        return float("inf")
    return float(qmax)


def _resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if n <= 0:
        return np.array([], dtype=np.float64)

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))

    if a.size == 0 or not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    a = a / s

    if a.size == n:
        return a.astype(np.float64)

    if a.size == 1:
        return np.full(n, a[0], dtype=np.float64)

    z = np.log(np.maximum(a, 1e-300))
    z = z - float(np.mean(z))
    z = np.clip(z, -_G_BOUND + 1.0, _G_BOUND - 1.0)

    x_old = np.linspace(0.0, 1.0, a.size, endpoint=True)
    x_new = np.linspace(0.0, 1.0, n, endpoint=True)
    z_new = np.interp(x_new, x_old, z)

    z_new = z_new - float(np.mean(z_new))
    z_new = np.clip(z_new, -_G_BOUND + 1.0, _G_BOUND - 1.0)

    return _softmax(z_new)


def _random_smooth_z(n, scale, rng, max_modes=8):
    n = int(n)
    if n <= 1:
        return np.zeros(max(1, n), dtype=np.float64)

    freq = np.fft.rfftfreq(n)
    fmax = max(1e-12, float(max(1, int(max_modes))) / float(n))
    w = np.exp(-(freq / fmax) ** 2)
    w[0] = 0.0

    spec = (rng.standard_normal(w.size) + 1j * rng.standard_normal(w.size)) * w
    z = np.fft.irfft(spec, n)

    z = z - float(np.mean(z))
    st = float(np.std(z))
    if st > 1e-18:
        z = float(scale) * z / st
    else:
        z.fill(0.0)

    return np.clip(z, -_G_BOUND, _G_BOUND).astype(np.float64)


# ---------------------------------------------------------------------------
# Start collection
# ---------------------------------------------------------------------------

def _start_score_key(item):
    return item[0]


def _add_candidate(starts, z, limit=1000):
    if len(starts) >= int(limit):
        return

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != _G_N or _G_N <= 0:
        return

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_G_BOUND, neginf=-_G_BOUND)

    z = z - float(np.mean(z))
    z = np.clip(z, -_G_BOUND + 1.0, _G_BOUND - 1.0)

    a = _softmax(z)
    q = _score_q(a)

    if np.isfinite(q):
        starts.append((2.0 * float(q), z.copy()))


def _add_reference_start(starts, ref, n, rng, limit=1000):
    ref = np.asarray(ref, dtype=np.float64).ravel()
    if ref.size < 2:
        return

    ref = np.nan_to_num(ref, nan=0.0, posinf=0.0, neginf=0.0)
    ref = np.maximum(ref, 0.0)
    if not np.all(np.isfinite(ref)) or float(np.sum(ref)) <= 0.0:
        return

    a = _resample_sequence(ref, n)
    z = _z_from_a(a)

    _add_candidate(starts, z, limit)
    _add_candidate(starts, z[::-1].copy(), limit)

    # A few low-frequency escapes around the reference.
    for sc in (0.2, 0.7, 2.0):
        dz = _random_smooth_z(n, sc, rng, max_modes=8)
        _add_candidate(starts, z + dz, limit)
        _add_candidate(starts, z[::-1].copy() + dz, limit)


def _make_starts(n, rng, prev_a, count=60):
    _set_state(n)
    starts = []
    count = int(max(12, count))
    limit = max(220, 6 * count)

    # Uniform sequence.
    _add_candidate(starts, np.zeros(n, dtype=np.float64), limit)

    # Known previous solution, if available.
    if prev_a is not None:
        _add_reference_start(starts, prev_a, n, rng, limit)

    # The earlier best solution exposed by the harness, if available.
    global_prev = globals().get("height_sequence_1")
    if global_prev is not None:
        _add_reference_start(starts, global_prev, n, rng, limit)

    u = np.linspace(0.0, 1.0, n, endpoint=True)
    v = 2.0 * u - 1.0

    # Monotone linear/log-odds profiles.
    for A in (0.3, 0.7, 1.2, 2.0, 3.5, 6.0, 10.0):
        _add_candidate(starts, A * v, limit)
        _add_candidate(starts, -A * v, limit)

    log_odd = np.log(np.maximum(u, 1e-4) / np.maximum(1.0 - u, 1e-4))
    for A in (0.2, 0.5, 1.0, 2.0, 4.0):
        _add_candidate(starts, A * log_odd, limit)
        _add_candidate(starts, -A * log_odd, limit)

    # Symmetric quadratic / cubic-ish profiles.
    quad = v * v - 0.5
    for A in (0.5, 1.5, 4.0, 10.0, 20.0):
        _add_candidate(starts, A * quad, limit)
        _add_candidate(starts, -A * quad, limit)

    cubic = v * v * v - 0.5 * v
    for A in (1.0, 3.0, 8.0, 20.0):
        _add_candidate(starts, A * cubic, limit)
        _add_candidate(starts, -A * cubic, limit)

    # Low-frequency sinusoids.
    for m in (1, 2, 3, 4):
        cm = np.cos(2.0 * np.pi * float(m) * u)
        sm = np.sin(2.0 * np.pi * float(m) * u)
        for A in (0.5, 1.5, 4.0, 8.0):
            _add_candidate(starts, A * cm, limit)
            _add_candidate(starts, A * sm, limit)

    # Smooth monotone step profiles in log space.
    for cpos in (0.3, 0.5, 0.7):
        for wdt in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7):
            s = 1.0 / (1.0 + np.exp(-(u - cpos) / wdt))
            s = s - float(np.mean(s))
            for A in (0.5, 1.5, 4.0, 10.0, 20.0, 40.0):
                _add_candidate(starts, A * s, limit)

    # Piecewise low-resolution random levels, optionally monotone.
    for K in (2, 3, 4, 6, 8, 12, 16, 24, 32):
        for rep in range(3):
            if len(starts) >= limit:
                break
            scale = float(rng.choice([0.5, 1.2, 2.5, 5.0]))
            levels = rng.standard_normal(K) * scale
            if rep == 1:
                levels = np.sort(levels)
            elif rep == 2:
                levels = levels - float(np.mean(levels))

            xk = np.linspace(0.0, 1.0, K, endpoint=True)
            z = np.interp(u, xk, levels)
            _add_candidate(starts, z, limit)

    # Random smooth log-density escapes.
    n_rand = max(24, count)
    for _ in range(n_rand):
        if len(starts) >= limit:
            break
        scale = float(rng.choice([0.3, 0.7, 1.5, 3.0, 6.0, 12.0]))
        maxm = int(rng.choice([3, 5, 8, 16, 32, 64, 128]))
        z = _random_smooth_z(n, scale, rng, maxm)
        _add_candidate(starts, z, limit)

    return starts


def _tau_schedule(n, is_final):
    if n <= 2048:
        taus = [0.25, 0.10, 0.04, 0.015, 0.006, 0.0025, 0.001, 0.0004]
    elif n <= 8192:
        taus = [0.15, 0.06, 0.025, 0.01, 0.004, 0.0015, 0.0006, 0.00025]
    else:
        taus = [0.12, 0.05, 0.02, 0.008, 0.003, 0.0012, 0.0005, 0.0002]

    if is_final:
        taus += [0.00008, 0.00003, 0.000015, 0.000008, 0.000004]

    return taus


# ---------------------------------------------------------------------------
# Objective and gradient for L-BFGS-B
# ---------------------------------------------------------------------------

def _objective(z):
    """
    Returns (value, grad) for the scaled smooth objective.
    """
    global _G_LAST_Z

    n = _G_N
    z = np.asarray(z, dtype=np.float64).ravel()

    if n <= 0:
        return 1e12, np.zeros(1, dtype=np.float64)

    if z.size != n:
        return 1e12, np.zeros(n, dtype=np.float64)

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_G_BOUND, neginf=-_G_BOUND)

    z = np.clip(z, -_G_BOUND, _G_BOUND)

    a = _softmax(z)
    fa = np.fft.rfft(a, _G_NFFT)
    c = np.fft.irfft(fa * fa, _G_NFFT)[:_G_NL].real
    c = np.maximum(c, 0.0)
    c = 0.5 * (c + c[::-1])

    q = float(n) * c
    qmax = float(np.max(q))

    if not np.isfinite(qmax):
        return 1e12, np.zeros(n, dtype=np.float64)

    tau = float(_G_TAU)
    if tau <= 0.0 or not np.isfinite(tau):
        tau = 1e-8

    diff = (q - qmax) / tau
    diff = np.clip(diff, -745.0, 0.0)
    w = np.exp(diff)
    sw = float(np.sum(w))

    if sw <= 0.0 or not np.isfinite(sw):
        return 1e12, np.zeros(n, dtype=np.float64)

    p = w / sw
    p = 0.5 * (p + p[::-1])
    psum = float(np.sum(p))
    if psum > 0.0 and np.isfinite(psum):
        p /= psum
    else:
        return 1e12, np.zeros(n, dtype=np.float64)

    fp = np.fft.rfft(p, _G_NFFT)
    h = np.fft.irfft(fp * np.conjugate(fa), _G_NFFT)[:n].real

    dh = float(np.dot(h, a))

    # For F = tau * logsumexp(q / tau), with q = n * conv(a,a),
    # dF/dz_i = 2 n a_i * (h_i - sum_j a_j h_j).
    grad_z = 2.0 * float(n) * a * (h - dh)

    F = qmax + tau * float(np.log(sw))
    scale = _G_SCALE

    val = 2.0 * scale * F
    grad = scale * grad_z

    # Tiny ridge improves gauge control for L-BFGS.
    ridge = 1e-10 * scale
    val += 0.5 * ridge * float(np.dot(z, z))
    grad = grad + ridge * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)

    return float(val), np.asarray(grad, dtype=np.float64)


def _timeout_callback(xk, *args):
    global _G_LAST_Z

    if _G_CALL_DEADLINE is not None and time.monotonic() > _G_CALL_DEADLINE:
        try:
            _G_LAST_Z = np.asarray(xk, dtype=np.float64).ravel().copy()
        except Exception:
            _G_LAST_Z = None
        raise _BudgetExceeded()


def _run_lbfgs(z, tau, deadline, maxfun_cap=1500, max_seconds=120.0):
    global _G_TAU, _G_CALL_DEADLINE, _G_LAST_Z

    if tau is not None:
        _G_TAU = float(tau)
    if _G_TAU <= 0.0 or not np.isfinite(_G_TAU):
        _G_TAU = 1e-8

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != _G_N or _G_N <= 0:
        return z

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_G_BOUND, neginf=-_G_BOUND)

    z = np.clip(z, -_G_BOUND, _G_BOUND)

    now = time.monotonic()
    if deadline is None:
        end = now + 5.0
    else:
        end = float(deadline)

    if end <= now:
        return z

    call_end = min(end, now + float(max(2.0, max_seconds)))
    if call_end <= now + 1.0:
        return z

    _G_CALL_DEADLINE = call_end - 0.2
    _G_LAST_Z = z.copy()

    time_left = call_end - now
    cost = max(1e-4, _G_COST)
    maxfun = int(max(30, min(int(maxfun_cap), time_left / cost)))
    maxiter = int(max(30, min(2000, maxfun // 4)))

    opts = {
        "maxfun": int(maxfun),
        "maxiter": int(maxiter),
        "ftol": 1e-11,
        "pgtol": 1e-9,
        "maxls": 50,
        "disp": False,
    }

    z_up = z
    try:
        res = minimize(
            _objective,
            z,
            jac=True,
            method="L-BFGS-B",
            options=opts,
            callback=_timeout_callback,
        )
        if res is not None and hasattr(res, "x"):
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if x.size == _G_N and np.all(np.isfinite(x)):
                z_up = np.clip(x, -_G_BOUND, _G_BOUND)
    except _BudgetExceeded:
        z_up = _G_LAST_Z if _G_LAST_Z is not None else z
    except Exception:
        z_up = _G_LAST_Z if _G_LAST_Z is not None else z

    _G_CALL_DEADLINE = None

    if z_up is None:
        z_up = z

    z_up = np.asarray(z_up, dtype=np.float64).ravel()
    if z_up.size != _G_N or not np.all(np.isfinite(z_up)):
        z_up = z

    return np.clip(z_up, -_G_BOUND, _G_BOUND)


# ---------------------------------------------------------------------------
# Stage / final refiners
# ---------------------------------------------------------------------------

def _optimize_start(z, tau_list, start_end):
    z = np.asarray(z, dtype=np.float64).ravel().copy()
    if z.size != _G_N or _G_N <= 0:
        z = np.zeros(_G_N, dtype=np.float64) if _G_N > 0 else np.array([], dtype=np.float64)

    a = _softmax(z)
    q = _score_q(a)
    if not np.isfinite(q):
        a = np.full(_G_N, 1.0 / float(_G_N), dtype=np.float64)
        z = _z_from_a(a)
        q = _score_q(a)

    best_z = z.copy()
    best_a = a.copy()
    best_q = float(q)

    ntau = max(1, len(tau_list))

    for i, tau in enumerate(tau_list):
        now = time.monotonic()
        if now >= start_end - 3.0:
            break

        rem = start_end - 4.0 - now
        if rem <= 2.0:
            break

        tau_deadline = min(start_end - 3.0, now + max(2.0, rem / max(1, ntau - i)))

        stall = 0
        while time.monotonic() < tau_deadline - 1.0:
            call_end = min(tau_deadline - 0.5, time.monotonic() + 120.0)
            if call_end <= time.monotonic() + 1.0:
                break

            z2 = _run_lbfgs(z, tau, deadline=call_end, maxfun_cap=2000, max_seconds=120.0)
            a2 = _softmax(z2)
            q2 = _score_q(a2)

            if not np.isfinite(q2):
                break

            if q2 < best_q - 1e-15:
                best_q = float(q2)
                best_z = z2.copy()
                best_a = a2.copy()

            if q2 < q - 1e-14:
                z = z2.copy()
                a = a2.copy()
                q = float(q2)
                stall = 0
            else:
                stall += 1
                if q2 <= q + 1e-12:
                    z = z2.copy()
                    a = a2.copy()
                    q = float(q2)
                else:
                    if best_q < q:
                        z = best_z.copy()
                        a = best_a.copy()
                        q = float(best_q)
                    break

                if stall > 2:
                    break

        if best_q < q - 1e-12:
            z = best_z.copy()
            a = best_a.copy()
            q = float(best_q)

    return best_z, best_a, float(best_q)


def _stage_optimize(n, prev_a, stage_end, rng, is_final):
    _set_state(n)

    now = time.monotonic()
    if now >= stage_end - 15.0:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        return a, 2.0

    if n <= 1024:
        count = 256
    elif n <= 2048:
        count = 120
    elif n <= 4096:
        count = 80
    elif n <= 8192:
        count = 56
    else:
        count = 40

    starts = _make_starts(n, rng, prev_a, count)

    if not starts:
        a = np.full(n, 1.0 / float(n), dtype=np.float64)
        return a, 2.0

    starts.sort(key=_start_score_key)

    if n <= 1024:
        topk = 24
    elif n <= 2048:
        topk = 20
    elif n <= 4096:
        topk = 16
    elif n <= 8192:
        topk = 14
    else:
        topk = 12

    selected = starts[:topk]
    tau_list = _tau_schedule(n, is_final)

    best_a = None
    best_q = float("inf")

    for idx, (_, z) in enumerate(selected):
        now = time.monotonic()
        if now >= stage_end - 20.0:
            break

        rem = max(0.0, stage_end - 20.0 - now)
        remaining = max(1, len(selected) - idx)
        base = rem / float(remaining)

        if idx < 2:
            factor = 3.0
        elif idx < 6:
            factor = 1.6
        else:
            factor = 1.0

        budget = min(max(15.0, base * factor), 900.0)
        start_end = min(stage_end - 20.0, now + budget)

        zout, aout, qout = _optimize_start(z, tau_list, start_end)

        if np.isfinite(qout) and qout < best_q:
            best_q = float(qout)
            best_a = np.asarray(aout, dtype=np.float64).copy()

    if best_a is None:
        best_a = np.full(n, 1.0 / float(n), dtype=np.float64)
        best_q = _score_q(best_a)

    # Use leftover stage time on the local best.
    if time.monotonic() < stage_end - 5.0:
        zbest = _z_from_a(best_a)
        small_taus = tau_list[-2:] if len(tau_list) >= 2 else tau_list
        for tau in small_taus:
            if time.monotonic() >= stage_end - 4.0:
                break
            call_end = min(stage_end - 4.0, time.monotonic() + 120.0)
            z2 = _run_lbfgs(zbest, tau, deadline=call_end, maxfun_cap=1500, max_seconds=120.0)
            a2 = _softmax(z2)
            q2 = _score_q(a2)
            if np.isfinite(q2) and q2 < best_q:
                best_q = float(q2)
                best_a = np.asarray(a2, dtype=np.float64).copy()
                zbest = z2.copy()

    return best_a, 2.0 * float(best_q)


def _exact_polish(a0, deadline):
    _set_state(_G_N)
    a = _normalize_sequence(a0)
    n = a.size
    if n <= 0:
        return a0, float("inf")

    z = _z_from_a(a)
    best_a = a.copy()
    best_q = _score_q(a)

    rng = np.random.default_rng(987654321)
    no_improve = 0
    it = 0
    tau_active = 1e-5

    while time.monotonic() < deadline - 3.0:
        c = _fft_conv(a)
        q = float(n) * c
        m = float(np.max(q))

        if np.isfinite(m) and m < best_q - 1e-16:
            best_q = float(m)
            best_a = a.copy()

        diff = (q - m) / tau_active
        diff = np.clip(diff, -745.0, 0.0)
        w = np.exp(diff)
        sw = float(np.sum(w))

        if sw > 0.0 and np.isfinite(sw):
            p = w / sw
            p = 0.5 * (p + p[::-1])
            psum = float(np.sum(p))
            if psum > 0.0:
                p /= psum

                fa = np.fft.rfft(a, _G_NFFT)
                fp = np.fft.rfft(p, _G_NFFT)
                h = np.fft.irfft(fp * np.conjugate(fa), _G_NFFT)[:n].real
                dh = float(np.dot(h, a))
                d = a * (h - dh)
                dn = float(np.linalg.norm(d))

                accepted = False
                if dn > 1e-14:
                    step_scale = 1.0 / dn
                    for mult in (0.8, 0.4, 0.2, 0.1, 0.05, 0.025, 0.0125,
                                 0.00625, 0.003125, 0.00156):
                        dz = -mult * step_scale * d
                        dz = np.clip(dz, -3.0, 3.0)
                        zt = np.clip(z + dz, -_G_BOUND, _G_BOUND)
                        at = _softmax(zt)
                        qt = _score_q(at)

                        if np.isfinite(qt) and qt < m - 1e-15:
                            z = zt
                            a = at
                            m = float(qt)
                            if qt < best_q:
                                best_q = float(qt)
                                best_a = at.copy()
                            accepted = True
                            no_improve = 0
                            break

                if not accepted:
                    escaped = False
                    for sc in (0.05, 0.15, 0.4, 1.0, 2.5):
                        dz = _random_smooth_z(n, sc, rng, max_modes=8)
                        zt = np.clip(z + dz, -_G_BOUND, _G_BOUND)
                        at = _softmax(zt)
                        qt = _score_q(at)

                        if np.isfinite(qt) and qt < best_q - 1e-15:
                            z = zt
                            a = at
                            best_q = float(qt)
                            best_a = at.copy()
                            escaped = True
                            no_improve = 0
                            break

                    if not escaped:
                        no_improve += 1
                        if no_improve > 100:
                            break
                else:
                    no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        it += 1
        if it % 50 == 0:
            print(
                f"exact_polish it={it} q={float(best_q):.9f} "
                f"time_left={max(0.0, deadline - time.monotonic()):.0f}s"
            )

    return best_a, float(best_q)


def _final_polish(a, deadline):
    n = int(np.asarray(a).size)
    if n <= 0:
        a = np.ones(1024, dtype=np.float64) / 1024.0
        n = 1024

    _set_state(n)
    a = _normalize_sequence(a)

    z = _z_from_a(a)
    best_a = a.copy()
    best_q = _score_q(a)
    current_q = float(best_q)

    taus = [
        1e-4, 5e-5, 2e-5, 1e-5, 5e-6,
        2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8
    ]

    no_tau_improve = 0

    for tau in taus:
        if time.monotonic() >= deadline - 25.0:
            break

        tau_improved = False
        stall = 0

        while time.monotonic() < deadline - 25.0:
            call_end = min(deadline - 24.0, time.monotonic() + 30.0)
            if call_end <= time.monotonic() + 3.0:
                break

            z2 = _run_lbfgs(z, tau, deadline=call_end, maxfun_cap=2500, max_seconds=30.0)
            a2 = _softmax(z2)
            q2 = _score_q(a2)

            if not np.isfinite(q2):
                break

            if q2 < best_q - 1e-15:
                best_q = float(q2)
                best_a = np.asarray(a2, dtype=np.float64).copy()
                z = z2.copy()
                current_q = float(q2)
                stall = 0
                tau_improved = True
                no_tau_improve = 0
            else:
                if q2 <= current_q + 1e-12:
                    z = z2.copy()
                    current_q = float(q2)
                else:
                    if best_q < current_q:
                        z = _z_from_a(best_a)
                        current_q = float(best_q)

                stall += 1
                if stall > 2:
                    break

        if not tau_improved:
            no_tau_improve += 1
            if no_tau_improve >= 5:
                break

        print(f"final_lbfgs tau={tau:.3e} best_q={float(best_q):.9f} "
              f"time_left={max(0.0, deadline - time.monotonic()):.0f}s")

    if time.monotonic() < deadline - 10.0:
        a3, q3 = _exact_polish(best_a, max(deadline - 6.0, time.monotonic() + 30.0))
        if np.isfinite(q3) and q3 < best_q:
            best_q = float(q3)
            best_a = np.asarray(a3, dtype=np.float64).copy()

    return best_a, float(best_q)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative coefficient sequence under the given
    wall-clock budget and return it as a Python list of floats.
    """
    budget_s = max(60.0, float(budget_s))
    t0 = time.monotonic()
    total_deadline = t0 + budget_s - 8.0

    reserve = max(60.0, min(220.0, 0.08 * budget_s))
    work_deadline = max(t0 + 60.0, total_deadline - reserve)

    rng = np.random.default_rng(int(seed))

    if budget_s >= 1800.0:
        n_final = 32768
    elif budget_s >= 900.0:
        n_final = 16384
    else:
        n_final = 8192

    base_lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    lengths = [int(x) for x in base_lengths if int(x) <= int(n_final)]
    if not lengths:
        lengths = [1024]

    if len(lengths) == 6:
        fr = [0.05, 0.06, 0.10, 0.14, 0.19, None]
    elif len(lengths) == 5:
        fr = [0.08, 0.11, 0.16, 0.22, None]
    elif len(lengths) == 4:
        fr = [0.12, 0.18, 0.26, None]
    elif len(lengths) == 3:
        fr = [0.15, 0.30, None]
    else:
        fr = [None]

    total_time = max(60.0, work_deadline - t0)

    best_a = None
    best_score = float("inf")
    prev_a = None

    print(f"search budget={budget_s:.0f}s, final_length_candidates={n_final}, "
          f"lengths={lengths}")

    for i, n in enumerate(lengths):
        stage_start = time.monotonic()

        if i == len(lengths) - 1:
            stage_end = work_deadline
        else:
            f = fr[i]
            if f is None:
                stage_end = min(work_deadline - 45.0, stage_start + 60.0)
            else:
                stage_end = min(
                    work_deadline - 45.0,
                    max(stage_start + 25.0, stage_start + total_time * float(f))
                )

        if stage_start >= stage_end - 15.0:
            continue

        a, score = _stage_optimize(n, prev_a, stage_end, rng, is_final=(i == len(lengths) - 1))

        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_a = np.asarray(a, dtype=np.float64).copy()

        # Feed the better of (global best, stage best) into the next length.
        if best_a is not None and best_score < float(score):
            prev_a = best_a.copy()
        else:
            prev_a = np.asarray(a, dtype=np.float64).copy()

        print(f"stage n={n} stage_score={float(score):.9f} "
              f"global_best={float(best_score):.9f} "
              f"elapsed={time.monotonic() - t0:.0f}s")

    # If nothing survived, construct a safe fallback from any known sequence.
    if best_a is None:
        global_prev = globals().get("height_sequence_1")
        if global_prev is not None:
            g = np.asarray(global_prev, dtype=np.float64).ravel()
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g = np.maximum(g, 0.0)
            if g.size >= 2 and float(np.sum(g)) > 0.0:
                best_a = _resample_sequence(g, 8192)
                _set_state(8192)
                best_score = 2.0 * _score_q(best_a)
                prev_a = best_a.copy()
        if best_a is None:
            best_a = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
            _set_state(1024)
            best_score = 2.0
        prev_a = best_a

    # Final refinement on the best found so far.
    final_deadline = max(time.monotonic() + 20.0, total_deadline - 12.0)
    try:
        best_a, best_q = _final_polish(best_a, final_deadline)
        if np.isfinite(best_q):
            best_score = 2.0 * float(best_q)
    except Exception:
        pass

    # Defensive normalization before returning.
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    if best_a.size == 0:
        best_a = np.full(1024, 1.0 / 1024.0, dtype=np.float64)

    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)

    s = float(np.sum(best_a))
    if not np.isfinite(s) or s < 0.01:
        best_a = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
    else:
        best_a = best_a / s

    seq = [float(v) for v in best_a.tolist()]

    print(f"final: length={len(seq)}, internal_score={float(best_score):.9f}, "
          f"total_elapsed={time.monotonic() - t0:.0f}s")

    return seq
