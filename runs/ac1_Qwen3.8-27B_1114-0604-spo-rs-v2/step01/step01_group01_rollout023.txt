def _lp_direction(a, t):
    """Solve LP to find direction b that reduces max convolution."""
    n = len(a)
    # Maximize sum(b) = -minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)[k] <= t for k = 0, ..., 2n-2
    # (a*b)[k] = sum_i a[i] * b[k-i]
    # For each k, this is a linear constraint in b.
    
    # A_ub * b <= b_ub
    # Row k: sum_i a[i] * b[k-i] <= t
    # This means: for b[j], the coefficient in row k is a[k-j] (if 0 <= k-j < n)
    
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # b_ub = [t] * (2n-1)
    
    # This matrix has size (2n-1) x n = 6143 x 3072 for n=3072
    # That's about 19 million elements - might be too large for dense
    
    # Use sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):

            if 0 <= i < n:
                rows.append(k)
                cols.append(j)

The terms involving `a[i]` are:

- `a[k-i] * a[i]` (when k-j=i, so k-i must be in [0,n-1])

(for k where both i and k-i are in [0, n-1] and i ≠ k-i)

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

Strategy:
- Multiplicative majorization descent (MMD) with power-sum smoothing
- Adam-style adaptive per-coordinate step sizes
- LP-based direction finding for occasional refinement
- 1D coordinate descent for final refinement (exact per-coordinate optimization)
- Diverse initializations: two-block, Gaussian, cosine-tapered, provided sequence
- Multi-scale: optimize at smaller n, upscale to larger n

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000:
        return np.convolve(a, b).astype(float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    n = len(p)
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    return 2.0 * n * float(np.max(c))


def _score_with_sum(p):
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    n = len(p)
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    return 2.0 * n * float(np.max(c)) / (s * s)


def _resample(a, m):
    if m <= 0:
        return None
    a = _sanitize(a)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _init_two_block(n, rng):
    best = None
    best_sc = float('inf')
    for alpha in np.linspace(0.2, 0.8, 13):
        for ratio in np.logspace(-1, 1.5, 15):
            k = max(1, min(n - 1, int(round(alpha * n))))
            p = np.ones(n)
            p[k:] = ratio
            p = _sanitize(p)
            if p is None:
                continue
            sc = _score(p)
            if sc < best_sc:
                best_sc = sc
                best = p.copy()
    return best


def _init_gaussian(n, rng):
    modes = [
        lambda t: np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2),
        lambda t: np.exp(-((t - 0.5) / 0.2) ** 2) + 0.3 * np.exp(-((t - 0.1) / 0.1) ** 2),
        lambda t: (1 + np.cos(2 * np.pi * t)) ** 2,
        lambda t: np.maximum(0, np.sin(np.pi * t) ** 3),
    ]
    f = modes[int(rng.integers(0, len(modes)))]
    t = (np.arange(n) + 0.5) / n
    p = f(t) + 1e-10
    return _sanitize(p)


def _init_cosine_taper(n, rng):
    t = np.arange(n) / (n - 1)
    asym = float(rng.uniform(0.3, 0.7))
    p = np.cos(np.pi * (t - asym)) ** 2
    p = np.maximum(p, 0) + 1e-10
    return _sanitize(p)

    """MMD with power-sum smoothing and optional Adam steps."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _score(p)

    n = len(p)
    floor = 1e-15 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit

    cur_convs = _conv_full(p, p)
    cur_score = 2.0 * n * float(np.max(cur_convs))

    # Adam moments
    m_avg = np.zeros(n)
    v_avg = np.zeros(n)
    adam_t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.5

    stall = 0
    it = 0
    p_power = float(p_power)

    while time.monotonic() < deadline:
        if time.monotonic() - deadline > 2.0:
            break

        # Compute power-sum smoothing weights
        c = cur_convs
        c_clipped = np.maximum(c, 1e-15)
        pw = c_clipped ** (p_power - 1)
        pw_sum = pw.sum()
        if pw_sum < 1e-30 or not np.isfinite(pw_sum):
            w = np.ones(len(c)) / len(c)
        else:
            w = pw / pw_sum

        # Gradient direction: correlation of w with p
        L = _next_pow2(max(len(c), 2 * n))
        fw = np.fft.rfft(w, L)
        fp = np.fft.rfft(p, L)
        grad = np.fft.irfft(fw * np.conjugate(fp), L)
        grad = grad[:n].astype(float)
        grad = np.maximum(grad, 0.0)
        g_max = float(grad.max()) if grad.size else 0.0

        if g_max < 1e-15:
            # Perturb
            p_try = _sanitize(p * np.exp(rng.normal(0, 0.03, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try /= p_try.sum()
                sc = _score(p_try)
                if np.isfinite(sc) and sc < best_score * 1.05:
                    p = p_try
                    cur_convs = _conv_full(p, p)
                    cur_score = 2.0 * n * float(np.max(cur_convs))
            stall += 1
            it += 1
            continue

        grad_norm = grad / g_max

        # Adam update
        if use_adam:
            adam_t += 1
            m_avg = beta1 * m_avg + (1 - beta1) * grad_norm
            v_avg = beta2 * v_avg + (1 - beta2) * grad_norm ** 2
            m_hat = m_avg / (1 - beta1 ** adam_t)
            v_hat = v_avg / (1 - beta2 ** adam_t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            if not np.all(np.isfinite(step)):
                step = grad_norm
        else:
            step = lr * grad_norm

        # Try multiple step sizes
        chosen = None
        best_try_score = cur_score
        for mult in [1.0, 0.5, 0.25, 2.0, 0.1]:
            a_try = float(mult)
            dec = np.exp(-a_try * step)
            p_try = p * dec
            p_try = np.maximum(p_try, floor)
            s = p_try.sum()
            if s < 1e-10 or not np.isfinite(s):
                continue
            p_try = p_try / s
            sc = _score(p_try)
            if sc < best_try_score:
                best_try_score = sc
                chosen = p_try

        if chosen is not None:
            p = chosen
            cur_convs = _conv_full(p, p)
            cur_score = 2.0 * n * float(np.max(cur_convs))
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
                stall = 0
                if verbose and it % 200 == 0:
                    print(f"  MMD n={n} it={it} score={best_score:.6f}")
            else:
                stall += 1
        else:
            stall += 1
            lr = max(0.01, lr * 0.9)

        if stall > 100:
            # Perturb and restart
            std = 0.02 + 0.01 * min(stall / 100, 5)
            p_try = _sanitize(p * np.exp(rng.normal(0, std, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try /= p_try.sum()
            else:
                p_try = best.copy()
            p = p_try
            cur_convs = _conv_full(p, p)
            cur_score = 2.0 * n * float(np.max(cur_convs))
            lr = 0.5
            m_avg = np.zeros(n)
            v_avg = np.zeros(n)
            adam_t = 0
            stall = 0

        if stall > 500:
            p = best.copy()
            cur_convs = _conv_full(p, p)
            cur_score = 2.0 * n * float(np.max(cur_convs))
            lr = 0.3
            stall = 0

        it += 1
        if it % 500 == 0 and verbose:
            print(f"  MMD n={n} it={it} cur={cur_score:.6f} best={best_score:.6f} stall={stall}")

    return best, best_score


def _lp_direction(a, t, timeout=10):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=t for all k. Returns b or None."""
    n = len(a)
    # Build sparse constraint matrix
    # Row k: sum_i a[i] * b[k-i] <= t for k=0..2n-2
    # A[k,j] = a[k-j] if 0<=k-j<n, else 0

    rows_list = []
    cols_list = []
    vals_list = []

    for j in range(n):
        for i in range(n):
            k = i + j
            if k < 2 * n - 1:
                rows_list.append(k)
                cols_list.append(j)
                vals_list.append(a[i])

    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, t)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def _coordinate_descent(p0, rng, time_limit, verbose=False):
    """Exact 1D coordinate descent: optimize each coordinate in sequence."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _score(p)

    n = len(p)
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    no_improve = 0
    passes = 0

    while time.monotonic() < deadline:
        if time.monotonic() - deadline > 5.0:
            break
        passes += 1
        score_before = _score(p)

        for i in range(n):
            if time.monotonic() > deadline - 5.0:
                break
            old_val = p[i]
            S = p.sum() - old_val
            if S < 1e-12:
                continue

            # Precompute base convolution with p[i]=0
            p_zero = p.copy()
            p_zero[i] = 0.0
            c_base = _conv_full(p_zero, p_zero)

            # Changed indices: k where 0 <= k-i < n, i.e., k in [i, i+n-1]
            k_idx = np.arange(max(0, i), min(2 * n - 1, i + n - 1) + 1)
            # For each k, the coefficient of t in c_k(t)
            # c_k(t) = c_base[k] + 2*a[k-i]*t (for k != 2i)
            # c_k(t) = c_base[k] + t^2 (for k = 2i)

            if 2 * i < 2 * n - 1:
                # There's a quadratic term
                quad_idx = 2 * i
                quad_base = c_base[quad_idx]
            else:
                quad_idx = -1
                quad_base = 0.0

            # Linear coefficients for non-quadratic changed indices
            lin_mask = np.ones(len(k_idx), dtype=bool)
            if quad_idx >= 0:
                lin_mask[k_idx == quad_idx] = False
            lin_k = k_idx[lin_mask]
            lin_coeff = 2.0 * a_vals = 2.0 * p[np.clip(lin_k - i, 0, n - 1)]
            lin_base = c_base[lin_k]

            # Unchanged max
            unchanged_mask = np.ones(2 * n - 1, dtype=bool)
            unchanged_mask[k_idx] = False
            if unchanged_mask.any():
                unchanged_max = float(c_base[unchanged_mask].max())
            else:
                unchanged_max = 0.0

            # 1D objective
            def obj_1d(t_val):
                if t_val < -1e-12:
                    return 1e10
                t_val = max(0.0, t_val)
                # Changed values
                vals = lin_base + lin_coeff * t_val
                mx = float(vals.max()) if vals.size else 0.0
                if quad_idx >= 0:
                    mx = max(mx, quad_base + t_val * t_val)
                mx = max(mx, unchanged_max)
                denom = (S + t_val) ** 2
                if denom < 1e-20:
                    return 1e10
                return 2.0 * n * mx / denom

            # Bounds for t
            t_lo = 0.0
            t_hi = max(2.0 * old_val, S * 0.5, 1e-6)

            try:
                res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                      options={'xatol': 1e-12})
                if res.success:
                    p[i] = float(res.x)
            except Exception:
                pass

        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-12:
            break
        p = p / s

        cur_score = _score(p)
        if cur_score < best_score - 1e-10:
            best = p.copy()
            best_score = cur_score
            no_improve = 0
            if verbose and passes % 5 == 0:
                print(f"  CoordDescent pass={passes} score={best_score:.8f}")
        else:
            no_improve += 1
            if no_improve >= 15:
                break

    return best, best_score


def _local_block_search(p0, rng, time_limit, verbose=False):
    """Block scaling local search."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = (0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0)

    while time.monotonic() < deadline:
        if time.monotonic() - deadline > 3.0:
            break
        r = rng.random()
        if r < 0.4:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.8:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            w = int(rng.integers(max(3, n // 20), max(4, n // 5 + 1)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        best_f = None
        best_fsc = _score(p)
        for f in factors:
            cand = p.copy()
            cand[start:start + w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < best_fsc:
                best_fsc = sc
                best_f = cand

        if best_f is not None:
            p = best_f
            if best_fsc < best_score:
                best = p.copy()
                best_score = best_fsc
                stagnant = 0
                if verbose and best_score < 1.52:
                    print(f"  BlockSearch score={best_score:.8f}")
        else:
            stagnant += 1
            if stagnant > 200:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.03, n)))
                if p_try is not None:
                    sc = _score(p_try)
                    if sc < best_score * 1.02:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0
            if stagnant > 1000:
                p = best.copy()
                stagnant = 0

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of non-negative floats
    minimizing the evaluation score.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # 30s safety buffer

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng((iseed % 10**9) + 7)

    # Get parent sequence if available
    parent = None
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        pass

    best_arr = None
    best_score = float('inf')

    def _update_best(arr, sc):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(sc) and sc < best_score:
            best_arr = arr.copy()
            best_score = sc

    # ======== PHASE 1: Quick initialization (0-60s) ========
    print(f"Phase 1: Initialization, deadline in {deadline - time.monotonic():.1f}s")

    n_small = 1024
    starts_small = []

    tb = _init_two_block(n_small, rng)
    if tb is not None:
        starts_small.append(tb)

    starts_small.append(_init_gaussian(n_small, rng))
    starts_small.append(_init_cosine_taper(n_small, rng))

    if parent is not None:
        sp = _resample(parent, n_small)
        if sp is not None:
            starts_small.append(sp)

    # Quick MMD at small scale
    for p0 in starts_small[:4]:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        talloc = min(12.0, max(3.0, rem - 10.0))
        p, sc = _run_mmd(p0, rng, talloc, p_power=4.0, use_adam=True)
        _update_best(p, sc)
        print(f"  Small n={n_small}: score={sc:.6f}")

    # ======== PHASE 2: Main optimization at target size (60s - 1800s) ========
    rem = deadline - time.monotonic()
    if rem > 600:
        n_main = 4096
    elif rem > 300:
        n_main = 3072
    else:
        n_main = 2048

    print(f"Phase 2: Main optimization at n={n_main}, remaining {rem:.1f}s")

    # Build starts at main size
    starts_main = []

    tb_big = _init_two_block(n_main, rng)
    if tb_big is not None:
        starts_main.append(tb_big)

    if parent is not None:
        sp = _resample(parent, n_main)
        if sp is not None:
            # Add small perturbation
            sp2 = _sanitize(sp * np.exp(rng.normal(0, 0.02, n_main)))
            if sp2 is not None:
                starts_main.append(sp2)
            starts_main.append(sp)

    # Up-resample from small results
    if best_arr is not None and best_arr.size != n_main:
        up = _resample(best_arr, n_main)
        if up is not None:
            starts_main.append(up)
            up2 = _sanitize(up * np.exp(rng.normal(0, 0.03, n_main)))
            if up2 is not None:
                starts_main.append(up2)

    starts_main.append(_init_gaussian(n_main, rng))
    starts_main.append(_init_cosine_taper(n_main, rng))

    # Score and sort starts
    scored_starts = []
    for sp in starts_main:
        if sp is not None:
            sc = _score(sp)
            if np.isfinite(sc):
                scored_starts.append((sc, sp))
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Best start score: {scored_starts[0][0]:.6f}" if scored_starts else "  No valid starts")

    # Run MMD on top starts
    n_starts = min(3, len(scored_starts))
    main_t0 = time.monotonic()
    main_budget = deadline - main_t0 - 300.0  # leave 300s for phases 3-5
    main_budget = max(60.0, main_budget)

    fracs = [0.5, 0.3, 0.2] if n_starts == 3 else [0.6, 0.4] if n_starts == 2 else [0.8]

    for idx in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        talloc = fracs[idx] * main_budget
        talloc = min(talloc, max(30.0, rem - 250.0))
        if talloc < 20:
            talloc = max(10.0, rem - 100.0)

        sp = scored_starts[idx][1]
        print(f"  MMD start {idx+1}/{n_starts}: init_score={scored_starts[idx][0]:.6f}, talloc={talloc:.0f}s")

        # Run with different power settings
        p, sc = _run_mmd(sp, rng, talloc * 0.7, p_power=4.0, use_adam=True, verbose=True)
        _update_best(p, sc)

        # Try with different power
        rem2 = deadline - time.monotonic()
        if rem2 > 100 and talloc * 0.7 > 30:
            p2, sc2 = _run_mmd(p if p is not None else sp, rng, min(talloc * 0.3, rem2 - 80),
                               p_power=10.0, use_adam=True, verbose=True)
            _update_best(p, sc)
            _update_best(p2, sc2)

        print(f"  After MMD {idx+1}: best={best_score:.6f}")

    # ======== PHASE 3: LP-based refinement ========
    rem = deadline - time.monotonic()
    print(f"Phase 3: LP refinement, remaining {rem:.1f}s")
    if rem > 120 and best_arr is not None:
        # Ensure we're at the right size
        if best_arr.size != n_main:
            best_arr = _resample(best_arr, n_main)
            best_score = _score(best_arr)

        lp_iters = 0
        lp_budget = min(200.0, rem - 100.0)
        p_cur = best_arr.copy()
        p_cur_score = best_score

        while time.monotonic() < deadline - 100 and lp_iters < 8:
            t = float(np.max(_conv_full(p_cur, p_cur)))
            b = _lp_direction(p_cur, t, timeout=15)
            if b is None:
                break
            # Mix
            for alpha in [0.3, 0.5, 0.7]:
                p_try = (1 - alpha) * p_cur + alpha * b
                p_try = _sanitize(p_try)
                if p_try is None:
                    continue
                sc = _score(p_try)
                if sc < p_cur_score:
                    p_cur = p_try
                    p_cur_score = sc
                    break
            _update_best(p_cur, p_cur_score)
            lp_iters += 1
            print(f"  LP iter {lp_iters}: score={best_score:.6f}")

    # ======== PHASE 4: Coordinate descent refinement ========
    rem = deadline - time.monotonic()
    print(f"Phase 4: Coordinate descent, remaining {rem:.1f}s")
    if rem > 100 and best_arr is not None:
        # Do coordinate descent on a subsampled version first for speed, then full
        n_cd = min(n_main, 2048)
        p_cd = _resample(best_arr, n_cd)
        if p_cd is not None:
            t_cd = min(rem - 60, 200.0)
            p_cd, sc_cd = _coordinate_descent(p_cd, rng, t_cd, verbose=True)
            if p_cd is not None:
                # Up-resample back
                p_full = _resample(p_cd, n_main)
                if p_full is not None:
                    sc_full = _score(p_full)
                    _update_best(p_full, sc_full)

        # Full coordinate descent if time remains
        rem = deadline - time.monotonic()
        if rem > 80 and best_arr is not None and best_arr.size == n_main:
            t_cd2 = min(rem - 40, 150.0)
            p_cd2, sc_cd2 = _coordinate_descent(best_arr, rng, t_cd2, verbose=True)
            _update_best(p_cd2, sc_cd2)

    # ======== PHASE 5: Final polish with MMD + block search ========
    rem = deadline - time.monotonic()
    print(f"Phase 5: Final polish, remaining {rem:.1f}s")
    if rem > 60 and best_arr is not None:
        # MMD with high power for precision
        t_mmd = min(rem * 0.4, 120.0)
        p_mmd, sc_mmd = _run_mmd(best_arr, rng, t_mmd, p_power=20.0, use_adam=True, verbose=True)
        _update_best(p_mmd, sc_mmd)

        # Block search
        rem = deadline - time.monotonic()
        if rem > 40:
            t_bs = min(rem * 0.6, 100.0)
            p_bs, sc_bs = _local_block_search(best_arr, rng, t_bs, verbose=True)
            _update_best(p_bs, sc_bs)

        # One more MMD pass
        rem = deadline - time.monotonic()
        if rem > 30:
            p_final, sc_final = _run_mmd(best_arr, rng, min(rem - 10, 60.0),
                                         p_power=10.0, use_adam=True)
            _update_best(p_final, sc_final)

    # ======== PHASE 6: Final evaluation ========
    print(f"Phase 6: Final evaluation. Best score so far: {best_score:.8f}")

    # Try to evaluate with the official function
    final_list = [float(x) for x in best_arr]
    try:
        official_sc = float(evaluate_sequence(final_list))
        print(f"  Official score: {official_sc:.8f}")
        if not np.isfinite(official_sc):
            # Try to fix
            best_arr = np.clip(best_arr, 0, 1000)
            best_arr = best_arr / best_arr.sum()
            final_list = [float(x) for x in best_arr]
            official_sc = float(evaluate_sequence(final_list))
            print(f"  Fixed official score: {official_sc:.8f}")
    except Exception as e:
        print(f"  Evaluation error: {e}")
        official_sc = best_score

    # Ensure valid output
    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(n_main, dtype=float) / n_main
    else:
        out = out / s

    elapsed = time.monotonic() - start_time
    print(f"Done in {elapsed:.1f}s. Final: n={len(out)}, score={best_score:.8f}")

    return [float(x) for x in out]
