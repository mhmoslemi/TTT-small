"""
Hybrid LP-guided and gradient-based optimizer for minimizing
f(a) = 2n * max(conv(a,a)) / (sum a)^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    next_p2 = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, next_p2)
    return np.fft.irfft(A * A, next_p2)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via real FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    next_p2 = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, next_p2)
    B = np.fft.rfft(b, next_p2)
    return np.fft.irfft(A * B, next_p2)[:M]


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = fast_self_conv(a)
    return float(2.0 * n * max(c) / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0.
    A @ g gives conv(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None if fails."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search(f, g, n_evals=20):
    """Find best blend t in [0,1] for (1-t)*f + t*g.
    Returns (best_t, best_obj, best_a)."""
    best_t = 0.0
    best_obj = compute_objective(f)
    best_a = f.copy()
    
    # Coarse search
    ts = np.linspace(0.0, 1.0, n_evals)
    for t in ts:
        if t == 0.0:
            continue
        a_t = (1.0 - t) * f + t * g
        obj = compute_objective(a_t)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_t.copy()
    
    # Fine search around best
    if best_t > 0.0:
        for _ in range(3):
            step = (ts[1] - ts[0]) * 0.3
            t_lo = max(0.0, best_t - step)
            t_hi = min(1.0, best_t + step)
            for t in np.linspace(t_lo, t_hi, 10):
                if t == 0.0:
                    continue
                a_t = (1.0 - t) * f + t * g
                obj = compute_objective(a_t)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
                    best_a = a_t.copy()
            step *= 0.5
    
    return best_t, best_obj, best_a


def grad_power(a, c, p=4):
    """Gradient of the power-method approximation to max(c).
    Uses (sum c^p)^(1/p) as a smooth surrogate for max(c)."""
    n = len(a)
    S = float(a.sum())
    
    # Compute surrogate and its gradient
    cp = np.power(np.maximum(c, 0.0), p)
    surrogate = float(np.sum(cp)) ** (1.0 / p)
    
    if surrogate < 1e-30:
        return np.zeros(n), surrogate
    
    # d(surrogate)/d(c_k) = (1/p) * (sum c^p)^(1/p - 1) * p * c_k^(p-1)
    #                      = surrogate^(1-p) * c_k^(p-1)
    dc = np.zeros_like(c)
    mask = c > 0
    dc[mask] = np.power(c[mask], p - 1)
    dc *= surrogate ** (1 - p)
    
    # Weight for the max surrogate
    w = dc / max(dc.sum(), 1e-30)
    
    # Gradient of weighted convolution peak w.r.t. a
    grad_c = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # d conv[k] / d a[j] = a[k-j] for j in [j_start, j_end]
        grad_c[j_start:j_end+1] += w[k] * 2.0 * a[j_start:j_end+1][::-1]
    
    # Full gradient of 2n * surrogate / S^2
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_c / S2 - 2.0 * surrogate / S3)
    return grad, surrogate


def adam_step(a, g, m, v, t, lr):
    """In-place Adam update."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)
    np.clip(a, 0.0, None, out=a)


def init_from_parent(n, rng):

            out = np.zeros(n)
            out[:arr.size] = arr
            out[arr.size:] = pad_val
            return np.clip(out, 0.0, 1000.0)

    # Use a "front-loaded" profile

def init_msequence_like(n, rng):
    """Create an initialization based on a binary m-sequence pattern, then smooth."""
    # Generate a pseudo-random binary pattern with good autocorrelation properties
    # Use a simple LFSR-like pattern
    a = np.ones(n) * 0.5
    # Add some structure: alternating blocks with varied heights
    block_size = max(4, n // 64)
    for i in range(0, n, block_size):
        end = min(i + block_size, n)
        a[i:end] = 0.3 + 0.7 * rng.random()
    # Smooth
    kernel = np.ones(5) / 5.0
    a = np.convolve(a, kernel, mode='same')
    return np.clip(a, 0.01, 1000.0)


def init_front_loaded(n, alpha, rng):
    """Front-loaded initialization: large values at start, small at end."""
    x = np.linspace(0, 1, n)
    a = (1.0 - x) ** alpha + 0.1
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, 1000.0)


def init_two_sided(n, alpha, rng):

    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2*x - 1)**2
    return np.clip(a, 0.01, 1000.0)


def optimize_lp_guided(a0, phase_end, rng, tag, lp_time=3.0):
    """Main LP-guided descent loop."""
    n = len(a0)

        M = float(np.max(c))

        if g is None:

        if new_score < best_score:
            best_score = new_score

            no_improve = 0
            a = new_a
        elif new_score < current_score:
            # Accept even if not globally best (helps escape plateaus)
            a = new_a
            no_improve += 1
        else:
            no_improve += 1
            # Small random nudge to escape
            a = a + 0.001 * rng.standard_normal(n)
            np.clip(a, 0.0, None, out=a)
        
        if no_improve >= max_no_improve:
            # Restart from best with perturbation

            a = best_a * (1.0 + scale * rng.standard_normal(n))
            np.clip(a, 0.0, None, out=a)
            no_improve = 0
            max_no_improve = max(20, max_no_improve // 2)
            print(f"  [{tag}] Restart, best={best_score:.6f}")
        
        # Progress report
        if no_improve % 20 == 0:
            print(f"  [{tag}] iter {no_improve}, best={best_score:.6f}")
    
    return best_score, best_a


def optimize_gradient(a0, phase_end, rng, tag, lr=0.1, p=4):

    n = len(a0)

                m[:] = 0
                v[:] = 0

The LP is: max sum(g) s.t. conv(f, g) <= M, g >= 0.

The solution g might not be a good direction to move. For example, if f is already good, the LP might return g = f (since conv(f,f) <= M is satisfied with equality at the peak). In that case, moving in the direction g-f = 0 does nothing.

Actually, if M = max(conv(f,f)), then conv(f,f) <= M is satisfied (with equality at the peak). So g = f is always feasible. The LP maximizes sum(g), so it will try to make g as large as possible while keeping conv(f,g) <= M.

If f is uniform (all 1's), then M = n, and the constraint conv(f,g) <= n means sum of g's "spread" is limited. The optimal g would concentrate mass where f is small... but f is uniform, so the optimal g is also uniform (or close to it). This means the LP won't help improve a uniform sequence.

The LP is more useful when f has structure (non-uniform), because then the LP can find directions that exploit the structure.

For the line search: if g is very different from f, the line search might find a good improvement. If g is close to f, there's no improvement.
