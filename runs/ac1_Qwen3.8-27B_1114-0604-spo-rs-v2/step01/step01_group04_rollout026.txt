def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using exact coordinate descent
with vectorized 1D subproblem solving, multiple sequence lengths, diverse
initializations, and iterated local search (ILS) with random perturbations.
The 1D subproblem is solved analytically by computing candidate optima from
the active-set conditions, then verifying consistency.
"""

import numpy as np
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def solve_coord_1d(base, slopes, quad_mask, S_others, a_i, n, buf):
    """
    Solve the 1D subproblem for one coordinate.
    Find x >= 0 minimizing 2*n*max_k(base_k + slope_k*x + quad_k*x^2) / (S_others+x)^2.
    Uses analytic candidate generation + vectorized evaluation.
    """
    # Generate candidates
    candidates = [0.0, a_i]

    # Linear active-set candidates: x = S_others - 2*base_k/slope_k
    pos = slopes > 1e-14
    if np.any(pos):
        x_c = S_others - 2.0 * base[pos] / slopes[pos]
        x_c = x_c[x_c > 1e-10]
        if x_c.size > 0:
            candidates.extend(x_c.tolist())

    # Quadratic active-set candidate
    qi = np.where(quad_mask > 0.5)[0]
    if qi.size > 0:
        xq = base[qi[0]] / S_others
        if xq > 1e-10:
            candidates.append(xq)

    # Clip candidates to reasonable range
    hi = max(2.0 * a_i + 0.1, S_others + 0.1)
    cand_arr = np.array(candidates)
    cand_arr = cand_arr[(cand_arr >= 0.0) & (cand_arr <= hi)]
    if cand_arr.size == 0:
        cand_arr = np.array([0.0, a_i])

    # Deduplicate (rough)
    if cand_arr.size > 1:
        cand_arr = np.unique(np.round(cand_arr, 10))

    # Vectorized evaluation
    m = cand_arr.size
    L = base.size  # 2n-1
    # buf shape: (m, L)
    buf[:m, :] = base[None, :] + slopes[None, :] * cand_arr[:, None] + quad_mask[None, :] * cand_arr[:, None] ** 2
    max_conv = buf[:m, :].max(axis=1)
    new_S = S_others + cand_arr
    f_vals = 2.0 * n * np.maximum(max_conv, 0.0) / (new_S * new_S)

    idx_best = int(np.argmin(f_vals))
    return cand_arr[idx_best], f_vals[idx_best]


def coord_descent_pass(a, conv, rng, n, buf, do_renormalize=True):
    """One pass of coordinate descent."""
    S = a.sum()
    order = rng.permutation(n)

    for i in order:
        S_others = S - a[i]
        if S_others < 1e-14:
            continue

        # Compute base, slopes, quad_mask for coordinate i
        base = conv.copy()
        slopes = np.zeros(2 * n - 1)
        quad_mask = np.zeros(2 * n - 1)

        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        ks = np.arange(k_min, k_max + 1)
        kis = ks - i

        # Subtract linear contribution
        base[ks] -= 2.0 * a[i] * a[kis]
        slopes[ks] = 2.0 * a[kis]

        # Fix k=2i (quadratic)
        quad_k = 2 * i
        if k_min <= quad_k <= k_max:
            # We subtracted 2*a[i]*a[i], need to subtract only a[i]^2
            # Difference: add back a[i]^2
            base[quad_k] += a[i] * a[i]
            slopes[quad_k] = 0.0
            quad_mask[quad_k] = 1.0

        # Solve 1D subproblem
        x_opt, f_opt = solve_coord_1d(base, slopes, quad_mask, S_others, a[i], n, buf)

        # Update if better
        old_score_factor = conv.max() / (S * S)
        new_score_factor = f_opt / (2.0 * n)
        if f_opt < 2.0 * n * old_score_factor * (1.0 + 1e-12):
            old_val = a[i]
            a[i] = x_opt
            S = S_others + x_opt
            delta = x_opt - old_val
            for k in range(k_min, k_max + 1):
                ki = k - i
                if k == quad_k:
                    conv[k] += x_opt * x_opt - old_val * old_val
                else:
                    conv[k] += 2.0 * delta * a[kis[k - k_min]]

    if do_renormalize:
        a = a / a.sum()
        conv = full_conv(a)

    return a, conv


def coord_descent_optimize(a_init, time_budget, rng):
    """Run coordinate descent optimization with ILS restarts."""
    n = a_init.size
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-14:
        a = np.ones(n) / n
    a = a / a.sum()
    conv = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(conv.max())

    # Pre-allocate buffer for 1D solver (max candidates ~ n+2)
    max_cand = n + 4
    buf = np.zeros((max_cand, 2 * n - 1))

    t0 = time.monotonic()
    stall = 0
    pass_count = 0
    converged = False

    while time.monotonic() - t0 < time_budget:
        a, conv = coord_descent_pass(a, conv, rng, n, buf, do_renormalize=False)
        # Renormalize
        S = a.sum()
        if S < 1e-14:
            break
        a = a / S
        conv = full_conv(a)
        sc = 2.0 * n * float(conv.max())

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        pass_count += 1

        # Periodic ILS restart
        if stall > 30 and time.monotonic() - t0 < time_budget * 0.9:
            # Random perturbation
            scale = 0.03 + 0.04 * (stall / 50.0)
            a = best_a + scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = full_conv(a)
            stall = 0

        # Stronger restart if stuck longer
        if stall > 80:
            # Larger perturbation
            n_pert = max(1, n // 4)
            a = best_a.copy()
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= rng.uniform(0.1, 2.0, size=n_pert)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = full_conv(a)
            stall = 0

    return best_a, best_sc


def init_uniform(n, rng):
    return np.ones(n) / n


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.01
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.2 * rng.random())
    c2 = n * (0.65 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n, rng):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_sided(n, rng):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.2
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + float(rng.integers(0, 3))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-4
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.03 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_plateau(n, rng):
    """Plateau: high in a middle block, low elsewhere."""
    a = np.ones(n) * 0.1
    w = int(n * (0.3 + 0.3 * rng.random()))
    start = int((n - w) * rng.random())
    a[start:start + w] = 1.0
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def get_inits(n, rng):
    """Return list of initialization arrays for given n."""
    inits = []
    inits.append(init_two_cluster(n, rng))
    inits.append(init_u_shape(n, rng))
    inits.append(init_sided(n, rng))
    inits.append(init_cosine(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_plateau(n, rng))
    inits.append(init_sqrt_decay(n, rng))
    return inits


def polish_local(a, time_budget, rng):
    """Local polish: very small random perturbations accepted if they help."""
    n = a.size
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    scale = 0.01
    stuck = 0

    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        for _ in range(20):
            cand = best_a + scale * rng.standard_normal(n)
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-14:
                continue
            cand = cand / cs
            sc = normed_score(cand)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 10:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc
