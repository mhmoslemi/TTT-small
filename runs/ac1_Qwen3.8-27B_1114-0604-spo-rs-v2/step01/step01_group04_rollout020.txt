"""
Minimizes 2*n*max(conv(p,p))/sum(p)^2 using smooth-max gradient descent
with adaptive beta annealing, simplex projection, multiple initializations,
FFT-based gradient computation, and targeted polishing moves.
"""

import numpy as np
import time


def convolve_fft(a, b):
    """Full linear convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    fsize = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:n_out]


def self_conv(a):
    """Full self-convolution a*a using FFT."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:2 * n - 1]


def score_of(p):
    """Compute 2*n*max(conv(p,p)) assuming sum(p)=1."""
    n = len(p)
    C = self_conv(p)
    return 2.0 * n * float(np.max(C))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(p, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*C_k))) w.r.t. p.
    where C = conv(p,p). Returns (gradient, smooth_obj).
    """
    n = len(p)
    C = self_conv(p)
    C_shift = C - C.max()
    exp_vals = np.exp(beta * C_shift)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        # Degenerate case
        w = np.zeros(len(C))
        w[int(np.argmax(C))] = 1.0
    else:
        w = exp_vals / w_sum
    smooth_obj = np.log(w_sum) / beta + C.max()
    
    # Gradient: g[i] = 2 * sum_j p[j] * w[i+j]
    # = 2 * conv(p_reversed, w)[i + n - 1]
    conv_pw = convolve_fft(p[::-1], w)
    g = 2.0 * conv_pw[n - 1:2 * n - 1]
    
    return g, smooth_obj

    """

    """
    n = len(p0)
    p = project_simplex(p0.astype(np.float64).copy())
    best_p = p.copy()
    best_sc = score_of(p)
    
    if verbose:
        print(f"  [opt] n={n}, initial score={best_sc:.6f}")
    
    # Parameters
    beta = 50.0
    lr = 0.3
    momentum = 0.3
    vel = np.zeros(n)
    it = 0
    last_improve = 0
    stall = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        # Adaptive beta: increase over time
        frac = min(1.0, it / 3000)
        beta_eff = 30.0 + 200.0 * frac ** 2
        
        # Compute gradient and current score
        g, _ = smooth_max_grad(p, beta_eff)
        C = self_conv(p)
        cur_sc = 2.0 * n * float(np.max(C))
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        else:
            break
        
        # Momentum
        vel = momentum * vel + (1.0 - momentum) * g
        direction = vel / (np.linalg.norm(vel) + 1e-15)
        
        # Step
        p_new = project_simplex(p - lr * direction)
        
        # Evaluate new point
        C_new = self_conv(p_new)
        new_sc = 2.0 * n * float(np.max(C_new))
        
        # Accept/reject (simulated annealing)
        if new_sc <= cur_sc or rng.random() < 0.15 * np.exp(-frac):
            p = p_new
            cur_sc = new_sc
        
        # Track best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 1000:
            p = project_simplex(best_p + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            lr = 0.1 + 0.3 * rng.random()
            momentum = 0.2 + 0.4 * rng.random()
            stall = 0
            last_improve = it
        
        # Periodic parameter refresh
        if it % 5000 == 0 and it > 0:
            lr = 0.1 + 0.3 * rng.random()
            momentum = 0.2 + 0.4 * rng.random()
        
        it += 1
    
    if verbose:
        print(f"  [opt] n={n}, final score={best_sc:.6f} ({it} iters)")
    
    return best_p, best_sc


def targeted_polish(p0, time_budget, rng):
    """
    Polish using targeted moves: specifically reduce the max convolution position.
    """
    n = len(p0)
    p = project_simplex(p0.astype(np.float64).copy())
    best_p = p.copy()
    best_sc = score_of(p)
    
    t0 = time.monotonic()
    step_size = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        C = self_conv(p)
        k_star = int(np.argmax(C))
        cur_sc = 2.0 * n * float(C[k_star])
        
        # Find partners for position k_star
        # C[k_star] = sum_i p[i]*p[k_star-i] for valid i
        # Partner of i is j = k_star - i (if 0 <= j < n)
        partners = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                partners[i] = p[j]
        
        # Move: decrease p where partner is large, increase where partner is small
        median_partner = np.median(partners[partners > 0]) if np.any(partners > 0) else 0
        
        # Find top and bottom contributors
        active = partners > 1e-10
        if np.sum(active) < 4:
            break
        
        sorted_idx = np.argsort(partners[active])
        n_move = max(1, n // 10)
        decrease_idx = np.where(active)[0][sorted_idx[-n_move:]]  # high partners
        increase_idx = np.where(active)[0][sorted_idx[:n_move]]  # low partners
        
        # Also consider inactive positions for increase
        inactive = ~active
        if np.sum(inactive) > 0:
            rand_inactive = rng.choice(np.where(inactive)[0], size=min(n_move, np.sum(inactive)), replace=False)
            increase_idx = np.concatenate([increase_idx, rand_inactive])
        
        # Make the move
        total_decrease = np.sum(p[decrease_idx]) * step_size
        if total_decrease < 1e-15:
            break
        
        p_new = p.copy()
        p_new[decrease_idx] -= total_decrease * p[decrease_idx] / (np.sum(p[decrease_idx]) + 1e-15)
        p_new[increase_idx] += total_decrease * p[increase_idx] / (np.sum(p[increase_idx]) + 1e-15)
        
        # Ensure non-negative
        p_new = np.maximum(p_new, 0)
        p_new = project_simplex(p_new)
        
        # Check all convolution positions (not just k_star)
        C_new = self_conv(p_new)
        new_sc = 2.0 * n * float(np.max(C_new))
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_p = p_new.copy()
            p = p_new.copy()
        elif rng.random() < 0.1:
            p = p_new.copy()
        
        # Decay step size
        step_size *= 0.995
        if step_size < 1e-6:
            step_size = 0.005
            p = project_simplex(best_p + 0.02 * rng.standard_normal(n))
        
        it += 1
    
    return best_p, best_sc


def init_linear_ramp(n, direction=1):
    """Linear ramp: high on one end, low on the other."""
    i = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = i + 1
    else:
        a = (n - i) + 1
    return a / a.sum()


def init_anti_sym(n, rng):
    """Anti-symmetric: p[i] * p[n-1-i] should be small."""
    i = np.arange(n, dtype=np.float64)
    # Use a logistic-like function that's high on one side
    mid = n / 2
    a = 1.0 / (1.0 + np.exp(-0.1 * (i - mid)))
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_gaussian_mix(n, rng):
    """Mixture of 2-3 Gaussians at random positions."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    n_bumps = 2 + int(rng.integers(0, 2))
    for _ in range(n_bumps):
        c = rng.uniform(0.1 * n, 0.9 * n)
        s = rng.uniform(0.05 * n, 0.2 * n)
        a += np.exp(-((i - c) / s) ** 2)
    a += 0.02
    return a / a.sum()


def init_cosine_mod(n, rng):
    """Cosine-modulated: p[i] = (1 + a*cos(w*i + phi))^2."""
    i = np.arange(n, dtype=np.float64)
    a = rng.uniform(0.3, 0.9)
    w = rng.uniform(0.5, 3.0) * 2 * np.pi / n
    phi = rng.uniform(0, 2 * np.pi)
    p = (1.0 + a * np.cos(w * i + phi)) ** 2
    p = np.maximum(p, 0.01)
    return p / p.sum()


def init_symmetric_lobe(n, rng):
    """Two symmetric lobes (anti-symmetric in the sense that center is low)."""
    i = np.arange(n, dtype=np.float64)
    c = n * rng.uniform(0.2, 0.4)
    s = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c) / s) ** 2) + np.exp(-((i - (n - 1 - c)) / s) ** 2)
    a += 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def get_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(init_linear_ramp(n, 1))
    inits.append(init_linear_ramp(n, -1))
    inits.append(init_anti_sym(n, rng))
    inits.append(init_gaussian_mix(n, rng))
    inits.append(init_cosine_mod(n, rng))
    inits.append(init_symmetric_lobe(n, rng))
    inits.append(init_random(n, rng))
    return inits
