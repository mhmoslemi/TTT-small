"""
Minimize 2*n*max(a*a)/(sum(a))^2 using LP-based best-response iterations
with multiplicative weight updates, followed by gradient refinement.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize

    HAVE_SCIPY = False


def fft_conv(a, b):
    """Linear convolution of a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na, nb = a.size, b.size
    if na == 0 or nb == 0:
        return np.empty(0)
    need = na + nb - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)[:need]
    return np.maximum(result, 0.0)  # clip tiny negatives


def fft_autoconv(a):
    """Guadratic auto-convolution a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]**2])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A * A, size)[:need]
    return np.maximum(result, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    S = p.sum()
    if S <= 0:
        return float('inf')
    b = fft_autoconv(p)
    M = b.max()
    return 2.0 * n * M / (S * S)


def normalize_simplex(p):
    """Normalize to nonnegative with sum=1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0:
        return np.full(p.shape, 1.0/p.shape[0])
    return p / s


def mwu_best_response(f, M, n_inner, eta):
    """
    Approximate LP: max sum(g) s.t. g>=0, (f*g)_k <= M.
    Using multiplicative weight updates.
    Returns g (sum=1).
    """
    n = len(f)
    g = np.ones(n) / n
    f_rev = f[::-1].copy()
    
    for it in range(n_inner):
        # v = f * g (cross-convolution, length 2n-1)
        v = fft_conv(f, g)
        
        # Excess
        v_excess = np.maximum(0.0, v - M)
        
        if v_excess.max() < 1e-15:
            break  # Already feasible
        
        # grad_i = (A^T v_excess)_i = conv(f_rev, v_excess)[i + n - 1]
        # conv(f_rev, v_excess) has length n + (2n-1) - 1 = 3n - 2
        cd = fft_conv(f_rev, v_excess)
        # cd has length 3n-2
        gr = cd[n-1 : n-1+n]  # length n
        
        # Multiplicative update
        gr = np.maximum(gr, 0.0)
        scale = eta * gr
        scale = np.clip(scale, -30.0, 30.0)
        g = g * np.exp(-scale)
        
        s = g.sum()
        if s > 0 and np.isfinite(s):
            g /= s
        else:
            g = np.ones(n) / n
    
    return g


def best_response_iteration(f, n_inner, eta, alpha):
    """
    One outer iteration: compute best response g, update f.
    """
    b = fft_autoconv(f)
    M = b.max()
    
    g = mwu_best_response(f, M, n_inner, eta)
    
    f_new = (1 - alpha) * f + alpha * g
    f_new = normalize_simplex(f_new)
    
    return f_new


def smooth_max_and_weights(b, beta):
    """Smooth max and softmax weights for gradient computation."""
    m = b.max()
    if m <= 0:
        w = np.ones_like(b) / b.size
        return m, w
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = e.sum()
    if Z <= 0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return float(G), w


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} via FFT."""
    n = len(p)
    m = len(w)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n : m][::-1].copy()
    return np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Project v onto simplex {x>=0, sum(x)=target}."""
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target/n)
    v = v.copy()
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    q = np.maximum(v - tau, 0.0)
    s = q.sum()
    if s <= 0:
        return np.full(n, target/n)
    return q * (target / s)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multiplicative peak-shaving (best-response)
iterations with adaptive parameters, strategic restarts, and gradient polishing.

The core idea: at each step, identify where the convolution a*a peaks, compute the
"pressure" on each element (how much reducing it would lower the peaks), and apply
a multiplicative update. This is equivalent to an approximate solution of the LP
"max sum(g) s.t. g>=0, a*g <= M" applied as a self-consistent iteration.
"""

import numpy as np
import time

def fft_conv(a, b):
    """Linear convolution a*b via FFT, clipping tiny negatives."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na, nb = a.size, b.size
    if na == 0 or nb == 0:
        return np.empty(0, dtype=np.float64)
    need = na + nb - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)[:need]
    return np.maximum(result, 0.0)


def fft_autoconv(a):
    """Auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A * A, size)[:need]
    return np.maximum(result, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_autoconv(p)
    if b.size == 0:
        return float('inf')
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def normalize_simplex(p):
    """Normalize to nonnegative with sum=1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.shape[0]), dtype=np.float64)
    return p / s


def peak_shaving(f, deadline, n_inner_eta=1.0, decay=0.9999, peak_band=0.15,
                 min_eta=0.01, check_interval=50):
    """
    Multiplicative peak-shaving iteration.
    
    At each step:
    1. Compute b = f*f
    2. Identify peak band (elements near the max)
    3. Compute pressure[i] = sum_k excess[k] * f[k-i] (via FFT)
    4. Update: f_i *= exp(-eta * pressure_i)
    5. Renormalize
    
    Parameters:
    - n_inner_eta: initial step size eta
    - decay: multiplicative decay of eta per iteration
    - peak_band: fraction of max to define "peaks" (0.15 means top 15% of range)
    - min_eta: minimum eta
    - check_interval: how often to compute exact score
    """
    n = len(f)
    f = normalize_simplex(f)
    best_f = f.copy()
    best_score = fast_score(f)
    eta = n_inner_eta
    f_rev = f[::-1].copy()
    stall_count = 0
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        b = fft_autoconv(f)
        M = float(b.max())
        if M <= 0:
            break
        
        # Define peak band
        threshold = M * (1.0 - peak_band)
        excess = np.where(b > threshold, b - threshold, 0.0)
        
        if excess.max() < 1e-15:
            break
        
        # Update f_rev for pressure computation
        f_rev = f[::-1].copy()
        
        # pressure[i] = conv(f_rev, excess)[i + n - 1]
        cd = fft_conv(f_rev, excess)
        # cd has length n + (2n-1) - 1 = 3n-2
        pressure = cd[n - 1: n - 1 + n].copy()
        pressure = np.maximum(pressure, 0.0)
        
        # Normalize pressure to have reasonable scale
        pmax = pressure.max()
        if pmax > 0:
            pressure = pressure / pmax
        
        # Multiplicative update
        scale = eta * pressure
        scale = np.clip(scale, -20.0, 20.0)
        f = f * np.exp(-scale)
        
        s = float(np.sum(f))
        if s > 0 and np.isfinite(s):
            f /= s
        else:
            f = normalize_simplex(f)
            break
        
        # Periodic score check
        if it % check_interval == 0:
            sc = fast_score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            if it % 500 == 0:
                print(f"    peak_shave it={it} cur={sc:.8f} best={best_score:.8f} eta={eta:.4f}")
            
            # If stalled a lot, add small perturbation
            if stall_count > 200:
                rng_local = np.random.default_rng(it)
                f = f * (1.0 + 0.01 * rng_local.normal(0, 1, n))
                f = normalize_simplex(f)
                stall_count = 0
        
        # Decay eta
        eta = max(min_eta, eta * decay)
    
    # Final check
    sc = fast_score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    return best_f, best_score


def full_best_response(f, deadline, n_inner=100, eta=0.3, alpha=0.2,
                       check_interval=5):
    """
    Best-response iteration: at each step, solve (approximately) the LP
    max sum(g) s.t. g>=0, f*g <= M, then mix f with g.
    """
    n = len(f)
    f = normalize_simplex(f)
    best_f = f.copy()
    best_score = fast_score(f)
    f_rev_base = None
    stall = 0
    
    outer = 0
    while time.monotonic() < deadline:
        outer += 1
        
        b = fft_autoconv(f)
        M = float(b.max())
        
        # Inner MWU to find g
        g = np.ones(n) / n
        for inner in range(n_inner):
            v = fft_conv(f, g)
            v_excess = np.maximum(0.0, v - M)
            if v_excess.max() < 1e-15:
                break
            f_rev = f[::-1]
            cd = fft_conv(f_rev, v_excess)
            gr = cd[n - 1: n - 1 + n]
            gr = np.maximum(gr, 0.0)
            gr_max = gr.max()
            if gr_max > 0:
                gr = gr / gr_max
            scale = eta * gr
            scale = np.clip(scale, -20.0, 20.0)
            g = g * np.exp(-scale)
            gs = float(np.sum(g))
            if gs > 0 and np.isfinite(gs):
                g /= gs
            else:
                g = np.ones(n) / n
        
        # Mix
        a_t = alpha * max(0.1, 1.0 - outer / 300.0)
        f_new = (1.0 - a_t) * f + a_t * g
        f_new = normalize_simplex(f_new)
        
        if outer % check_interval == 0:
            sc = fast_score(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
                stall = 0
                if outer % 100 == 0:
                    print(f"    best_resp outer={outer} score={sc:.8f} eta={eta:.3f}")
            else:
                f = f_new  # keep moving even without improvement
                stall += 1
                if stall > 100:
                    # Perturb to escape
                    rng_p = np.random.default_rng(outer * 7)
                    f = f * (1.0 + 0.02 * rng_p.normal(0, 1, n))
                    f = normalize_simplex(f)
                    stall = 0
        else:
            f = f_new
        
        # Slowly decrease eta
        if outer % 50 == 0:
            eta = max(0.05, eta * 0.95)
    
    sc = fast_score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    return best_f, best_score


def smooth_max_and_weights(b, beta):
    """Smooth max and softmax weights."""
    m = float(np.max(b))
    if m <= 0:
        w = np.ones_like(b) / b.size
        return m, w
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} via FFT."""
    n = len(p)
    m = len(w)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Project onto {x>=0, sum(x)=target}."""
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    v = v.copy()
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0:
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def gradient_refine(p0, deadline, beta=300000.0, max_iter=20000):
    """Projected gradient with smooth-max."""
    p = normalize_simplex(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    step = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        
        b = fft_autoconv(p)
        G, w = smooth_max_and_weights(b, beta)
        M = float(b.max())
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        
        svec = corr_pw(p, w)
        grad = 2.0 * svec
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_autoconv(q)
            qM = float(qb.max())
            qscore = 2.0 * n * qM
            if qscore < cur_score - 1e-12:
                p = q
                cur_score = qscore
                accepted = True
                break
        
        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(3.0, step * 1.15)
        
        if it % 1000 == 0:
            print(f"    grad_refine it={it} score={best_score:.8f} step={step:.2e}")
    
    return best_p, best_score


def structured_starts(n, seed=0):
    """Generate diverse starting profiles."""
    rng = np.random.default_rng(seed)
    starts = []
    x = np.linspace(0.0, 1.0, n)
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Linear ramps
    for a in (-3.0, -1.5, 1.5, 3.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)
    
    # Two-step
    for eps in (0.3, 0.5, 0.7):
        p = np.ones(n)
        p[:n//3] = 1.0 + eps
        p[2*n//3:] = 1.0 + eps
        p[n//3:2*n//3] = 1.0 - eps * 0.5
        p = np.maximum(p, 0.01)
        p /= p.sum()
        starts.append(p)
    
    # Sinusoidal
    for freq in (2, 3, 5):
        z = 0.8 * np.sin(2 * np.pi * freq * x)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)
    
    # Random smooth
    for _ in range(3):
        zc = rng.normal(size=2*n) + 1j * rng.normal(size=2*n)
        zc[int(n):] = 0  # low-pass filter
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - z.mean()
        z = 1.0 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)
    
    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function combining peak-shaving, best-response, and gradient methods.
    """
    start_time = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 0:
        return [1.0]
    
    reserve = 90.0
    deadline = start_time + max(0, budget_s - reserve)
    
    N = 8192
    
    # Get initial sequence
    p = None
    try:
        h = globals().get('height_sequence_1', None)
        if h is not None and len(h) == N:
            p = np.array(h, dtype=np.float64)
            p = normalize_simplex(p)
    except Exception:
        pass
    
    if p is None:
        p = np.ones(N) / N
    
    best_p = p.copy()
    best_score = fast_score(p)
    print(f"Initial N={N} score={best_score:.8f}")
    
    # ===== Phase 1: Peak shaving (main workhorse) =====
    print(f"\nPhase 1: Peak shaving")
    phase1_deadline = deadline - 120.0
    p1, s1 = peak_shaving(p, phase1_deadline, n_inner_eta=2.0, decay=0.99995,
                          peak_band=0.2, min_eta=0.05, check_interval=30)
    if s1 < best_score:
        best_score = s1
        best_p = p1.copy()
        print(f"  After Phase 1: {best_score:.8f}")
    
    # ===== Phase 2: Peak shaving with different parameters =====
    if time.monotonic() < deadline - 120.0:
        print(f"\nPhase 2: Peak shaving (narrower band, smaller eta)")
        phase2_deadline = deadline - 90.0
        p2, s2 = peak_shaving(best_p, phase2_deadline, n_inner_eta=0.5,
                              decay=0.99998, peak_band=0.05, min_eta=0.01,
                              check_interval=20)
        if s2 < best_score:
            best_score = s2
            best_p = p2.copy()
            print(f"  After Phase 2: {best_score:.8f}")
    
    # ===== Phase 3: Best-response iteration =====
    if time.monotonic() < deadline - 90.0:
        print(f"\nPhase 3: Best-response (LP approximation)")
        phase3_deadline = deadline - 60.0
        p3, s3 = full_best_response(best_p, phase3_deadline, n_inner=80,
                                    eta=0.4, alpha=0.25, check_interval=3)
        if s3 < best_score:
            best_score = s3
            best_p = p3.copy()
            print(f"  After Phase 3: {best_score:.8f}")
    
    # ===== Phase 4: Multi-start exploration =====
    if time.monotonic() < deadline - 120.0:
        print(f"\nPhase 4: Multi-start exploration")
        starts = structured_starts(N, seed=seed)
        # Score all starts
        scored = []
        for si in starts:
            sc = fast_score(si)
            if np.isfinite(sc):
                scored.append((sc, si))
        scored.sort(key=lambda x: x[0])
        
        # Try top starts that are different from current best
        tried = 0
        for sc_start, p_start in scored:
            if time.monotonic() >= deadline - 90.0:
                break
            if tried >= 4:
                break
            # Skip if too similar to current best
            if np.allclose(p_start, best_p, atol=1e-6):
                continue
            tried += 1
            
            sub_deadline = deadline - 90.0
            # Quick peak-shaving from this start
            pp, ss = peak_shaving(p_start, sub_deadline, n_inner_eta=1.5,
                                  decay=0.9999, peak_band=0.15, min_eta=0.03,
                                  check_interval=20)
            if ss < best_score:
                best_score = ss
                best_p = pp.copy()
                print(f"  Multi-start {tried}: {best_score:.8f}")
    
    # ===== Phase 5: Gradient refinement =====
    if time.monotonic() < deadline - 30.0:
        print(f"\nPhase 5: Gradient refinement")
        phase5_deadline = deadline - 20.0
        p5, s5 = gradient_refine(best_p, phase5_deadline, beta=500000.0,
                                 max_iter=20000)
        if s5 < best_score:
            best_score = s5
            best_p = p5.copy()
            print(f"  After Phase 5: {best_score:.8f}")
    
    # ===== Phase 6: Final peak shaving with very small steps =====
    if time.monotonic() < deadline - 20.0:
        print(f"\nPhase 6: Final polishing")
        phase6_deadline = deadline - 10.0
        p6, s6 = peak_shaving(best_p, phase6_deadline, n_inner_eta=0.1,
                              decay=0.99999, peak_band=0.03, min_eta=0.001,
                              check_interval=10)
        if s6 < best_score:
            best_score = s6
            best_p = p6.copy()
            print(f"  After Phase 6: {best_score:.8f}")
    
    # Finalize
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]
    
    # Verify with official evaluator
    try:
        off = evaluate_sequence(final_seq)
        print(f"\nOfficial score: {off:.8f}")
        if not np.isfinite(off) or off > 2.0 + 1e-9:
            print("Falling back to uniform")
            return [1.0] * N
    except Exception as e:
        print(f"Evaluation error: {e}")
        return [1.0] * N
    
    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s, Final score: {off:.8f}")
    return final_seq
