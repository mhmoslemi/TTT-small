def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):

    arr = np.clip(arr, 0.0, 1000.0)

def _optimize_mirror(a0, time_budget, K=40, verbose=True):
    """Mirror descent with backtracking on the simplex."""
    t_start = time.monotonic()
    n = a0.size
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_sc = _score(best_a)
    if verbose:
        print(f"  [optimize] start score={best_sc:.8f}")

    lr = 2.0
    min_lr = 0.01
    no_improve = 0
    max_no_improve = 200
    it = 0
    stagnation_count = 0

    while time.monotonic() - t_start < time_budget:
        it += 1
        c = _fft_convolve(a)
        max_c = float(np.max(c))
        if not np.isfinite(max_c):
            break

        # Active set: top K lags
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        # Compute gradient
        grad = _gradient_active(a, active)
        # Project to simplex tangent
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Backtracking line search
        found = False
        trial_lr = lr
        for _ in range(12):
            if trial_lr < min_lr:
                break
            exponent = -trial_lr * grad
            exponent = np.clip(exponent, -2.0, 2.0)
            new_a = a * np.exp(exponent)
            new_a = np.clip(new_a, 0.0, None)
            ns = new_a.sum()
            if ns < 1e-10:
                trial_lr *= 0.5
                continue
            new_a = new_a / ns
            new_sc = _score(new_a)
            if np.isfinite(new_sc) and new_sc < best_sc:
                a = new_a
                best_a = a.copy()
                best_sc = new_sc
                lr = min(lr * 1.02, 50.0)
                found = True
                no_improve = 0
                stagnation_count = 0
                break
            else:
                trial_lr *= 0.5

        if not found:
            no_improve += 1
            lr = max(min_lr, lr * 0.7)
            if no_improve > 30:
                stagnation_count += 1
                # Perturbation to escape local minimum
                if stagnation_count > 5:
                    rng = np.random.default_rng(it)
                    pert = rng.standard_normal(n) * 0.05
                    new_a = a * (1.0 + pert)
                    new_a = np.clip(new_a, 0.0, None)
                    ns = new_a.sum()
                    if ns > 1e-10:
                        a = new_a / ns
                    no_improve = 0
                    lr = 1.0
                    stagnant

                if no_improve > max_no_improve:
                    break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  [optimize] it={it} score={best_sc:.8f} lr={lr:.3f} elapsed={elapsed:.1f}s")

    return best_a, best_sc


def _optimize_gradient_proj(a0, time_budget, K=30, verbose=True):

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _scan_sine(n, deadline, mode=1):
    """Find best amplitude for sine family."""

    best_amp = 0.7

            if sc < best_sc:

def _make_anti_palin_random(n, seed, spikiness=0.8):
    """Random anti-palindromic sequence with controlled spikiness."""
    rng = np.random.default_rng(seed)
    half = n // 2
    # Generate random values for first half, mirror for second half
    d = rng.standard_normal(half)
    # Make it "spiky" by clipping
    d = np.tanh(d * spikiness)
    # a[i] = c/2 + d[i], a[n-1-i] = c/2 - d[i]
    c = 2.0 / n
    a = np.zeros(n)
    # Scale d so max |d| <= c/2
    dmax = np.max(np.abs(d))
    if dmax > 0:
        d = d / dmax * (c / 2.0) * 0.95
    for i in range(half):
        a[i] = c / 2.0 + d[i]
        a[n - 1 - i] = c / 2.0 - d[i]
    if n % 2 == 1:
        a[half] = c / 2.0
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def _make_two_level(n, frac_high, seed=None):

    half = n // 2
    c = 2.0 / n
    a = np.zeros(n)
    # First half: high, second half: low (anti-palindromic)

    return a

        if isinstance(hs, list) and len(hs) > 0:
            prev_best = _safe_sequence(np.array(hs, dtype=np.float64))
            if prev_best is not None:
                prev_sc = _score(prev_best)
                print(f"[main] loaded height_sequence_1, n={prev_best.size}, score={prev_sc:.8f}")
    except Exception:
        pass

    # Generate initial candidates
    candidates = []

    # 1. Sinusoidal candidates at various lengths
    for n in [2048, 4096, 1024]:
        if time.monotonic() > deadline - 100:
            break
        a, sc = _scan_sine(n, deadline - 60)
        candidates.append((a, sc, f"sine_n{n}"))
        print(f"[main] sine n={n}: score={sc:.8f}")

    # 2. Anti-palindromic random candidates
    for seed_val in [1, 7, 42, 99]:
        for n in [2048, 4096]:
            if time.monotonic() > deadline - 80:
                break
            for spik in [0.5, 0.8, 1.0]:
                a = _make_anti_palin_random(n, seed_val * 100 + int(spik * 10) + n, spik)
                sc = _score(a)
                candidates.append((a, sc, f"rand_n{n}_sp{spik}_s{seed_val}"))

    # 3. Two-level
    for n in [2048, 4096]:
        a = _make_two_level(n, 0.5)
        sc = _score(a)
        candidates.append((a, sc, f"twolevel_n{n}"))

    # 4. Previous best
    if prev_best is not None:
        prev_sc = _score(prev_best)
        candidates.append((prev_best, prev_sc, "prev_best"))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[1])
    print(f"[main] best initial: {candidates[0][2]} score={candidates[0][1]:.8f}")

    # Optimize top candidates
    best_overall_a = None
    best_overall_sc = np.inf
    best_overall_name = ""

    # Allocate time: give more time to better candidates
    top_candidates = candidates[:6]
    remaining_time = deadline - time.monotonic()
    if remaining_time < 30:
        # Not enough time, just return best initial
        for a, sc, name in top_candidates:
            if sc < best_overall_sc:
                best_overall_a = a.copy()
                best_overall_sc = sc
                best_overall_name = name
    else:
        # Give the top candidate most time
        times = []
        for i, (a, sc, name) in enumerate(top_candidates):
            if i == 0:
                times.append(remaining_time * 0.45)
            elif i == 1:
                times.append(remaining_time * 0.20)
            elif i == 2:
                times.append(remaining_time * 0.15)
            else:
                times.append(remaining_time * 0.08)
            if time.monotonic() > deadline - 20:
                times[-1] = 5  # minimal time
                break

        for i, (a, sc, name) in enumerate(top_candidates):
            if time.monotonic() > deadline - 20:
                break
            tb = min(times[i], deadline - time.monotonic() - 10)
            if tb < 5:
                break
            print(f"[main] optimizing {name} (score={sc:.8f}) with budget={tb:.1f}s")

            # Use mirror descent
            opt_a, opt_sc = _optimize_mirror(a, tb, K=50, verbose=True)
            if opt_sc < best_overall_sc:
                best_overall_a = opt_a
                best_overall_sc = opt_sc
                best_overall_name = f"mirror_{name}"
                print(f"[main] NEW BEST: {best_overall_name} score={best_overall_sc:.8f}")

            # Also try projected gradient with remaining time
            if time.monotonic() < deadline - 30:
                tb2 = min(60.0, deadline - time.monotonic() - 20)
                if tb2 > 10:
                    opt_a2, opt_sc2 = _optimize_gradient_proj(opt_a, tb2, K=40, verbose=False)
                    if opt_sc2 < best_overall_sc:
                        best_overall_a = opt_a2
                        best_overall_sc = opt_sc2
                        best_overall_name = f"gradproj_{name}"
                        print(f"[main] NEW BEST: {best_overall_name} score={opt_sc2:.8f}")

    # Final validation with evaluate_sequence
    print(f"[main] best candidate: {best_overall_name} internal_score={best_overall_sc:.8f}")

    # Validate top candidates with the actual evaluator
    validation_list = []
    if best_overall_a is not None:
        validation_list.append((best_overall_a, best_overall_sc, best_overall_name))
    for a, sc, name in top_candidates[:3]:
        validation_list.append((a, sc, name))

    # Deduplicate by name
    seen = set()
    unique_validation = []
    for a, sc, name in validation_list:
        if name not in seen:
            seen.add(name)
            unique_validation.append((a, sc, name))

    best_actual = np.inf
    best_out = None
    for a, sc, name in unique_validation:
        if time.monotonic() > deadline:
            break
        arr = _safe_sequence(a)
        if arr is None:
            continue
        try:
            real_sc = float(evaluate_sequence(arr.tolist()))
        except Exception:
            real_sc = np.inf
        if np.isfinite(real_sc) and real_sc < best_actual:
            best_actual = real_sc
            best_out = arr
            print(f"[validate] {name}: actual_score={real_sc:.8f}")

    if best_out is None:
        # Fallback: best internal score candidate
        if best_overall_a is not None:
            best_out = _safe_sequence(best_overall_a)
        if best_out is None and prev_best is not None:
            best_out = prev_best
        if best_out is None:
            best_out = _safe_sequence(_make_sine(2048, 0.78, 1))
        if best_out is None:
            best_out = np.ones(256, dtype=np.float64) / 256.0

    elapsed = time.monotonic() - t0
    print(f"[main] done in {elapsed:.1f}s, best actual score={best_actual:.8f}")

    return [float(x) for x in best_out]
