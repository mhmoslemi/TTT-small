"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase approach:
1. Global search via differential evolution on parameterized families 
   (trigonometric exponentials, sum of Gaussians)
2. Local refinement via softmax-parameterized subgradient descent
3. Simulated annealing / random perturbation polish
4. Multiple sequence lengths explored
"""

import numpy as np
import time
from scipy.optimize import differential_evolution


def project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def trig_exp_sequence(params, n):
    """
    Generate sequence from trigonometric exponential parameterization.
    a_i proportional to exp(sum_k params[k] * cos(2*k*pi*i/n))
    params has K elements for K harmonics (excluding DC).
    """
    i = np.arange(n, dtype=np.float64)
    log_a = np.zeros(n)
    for k, c in enumerate(params, start=1):
        log_a += c * np.cos(2.0 * k * np.pi * i / n)
    # Normalize (subtract max for numerical stability)
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def trig_exp_objective(params, n):
    a = trig_exp_sequence(params, n)
    return normed_score(a)


def gauss_sequence(params, n):
    """
    Generate sequence from sum of Gaussians.
    params: [w1, mu1, sig1, w2, mu2, sig2, ...]
    """
    K = len(params) // 3
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for k in range(K):
        w = abs(params[3*k]) + 0.01
        mu = params[3*k+1]
        sig = abs(params[3*k+2]) + 0.5
        a += w * np.exp(-((i - mu)**2) / (2.0 * sig**2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def gauss_objective(params, n):
    a = gauss_sequence(params, n)
    return normed_score(a)


def subgrad_optimize(a, time_budget, rng, use_softmax=True):
    """
    Subgradient descent with optional softmax parameterization.
    a is assumed to sum to 1 and be non-negative.
    """
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    if use_softmax:
        # Initialize v such that softmax(v) ≈ a
        v = np.log(a + 1e-30)
        v -= v.mean()  # center for numerical stability
    
    eta = 0.3
    tau_frac = 0.05
    n_scales = 3
    beta = 0.3
    vel = np.zeros(n)
    
    iters = 0
    last_improve = 0
    stall = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        
        # Multi-scale subgradient
        g = np.zeros(n)
        for s in range(n_scales):
            t = tau * (2.0 ** s)
            shift = b / t
            shift -= shift.max()
            w = np.exp(shift)
            sw = w.sum()
            if sw < 1e-300:
                w = np.zeros(b.size)
                w[int(np.argmax(b))] = 1.0
            else:
                w /= sw
            g += 2.0 * np.correlate(w, a, mode='valid')
        g /= n_scales
        
        if use_softmax:
            # Gradient in v-space
            grad_v = a * g
            # Remove mean for numerical stability (softmax is shift-invariant)
            grad_v -= grad_v.mean()
            if beta > 0:
                vel = beta * vel + (1 - beta) * grad_v
                dv = vel
            else:
                dv = grad_v
            v = v - eta * dv
            # Numerical stability
            v -= v.max()
            a_new = np.exp(v)
            a_new /= a_new.sum()
        else:
            if beta > 0:
                vel = beta * vel + (1 - beta) * g
                direction = vel
            else:
                direction = g
            a_new = project_simplex(a - eta * direction)
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        # Accept/reject
        if new_sc <= cur_sc or rng.random() < 0.05:
            if use_softmax:
                v = v - eta * dv  # already done above
            a = a_new
        else:
            # Try smaller step
            if use_softmax:
                v = v - 0.5 * eta * dv
                v -= v.max()
                a = np.exp(v)
                a /= a.sum()
            else:
                a = project_simplex(a - 0.5 * eta * direction)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
        elif iters - last_improve > 3000:
            stall += 1
        
        iters += 1
        
        if stall > 5:
            # Restart
            if use_softmax:
                v = np.log(best_a + 1e-30) + 0.02 * rng.standard_normal(n)
                v -= v.mean()
            else:
                a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.2 + 0.2 * rng.random()
            tau_frac = 0.03 + 0.03 * rng.random()
            last_improve = iters
            stall = 0
        
        # Adaptive step size
        if iters % 5000 == 0 and iters > 0:
            eta *= 0.85
            if eta < 0.01:
                eta = 0.1 + 0.2 * rng.random()
    
    return best_a, best_sc


def anneal_optimize(a, time_budget, rng):
    """Simulated annealing with random perturbations."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    scale = 0.05
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        cand = a + scale * rng.standard_normal(n)
        cand = project_simplex(cand)
        sc = normed_score(cand)
        
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            a = cand.copy()
            no_improve = 0
        elif sc < normed_score(a) or rng.random() < np.exp(-(sc - normed_score(a)) / scale):
            a = cand.copy()
            no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            scale *= 0.7
            no_improve = 0
            if scale < 1e-5:
                break
    
    return best_a, best_sc


def coordinate_descent(a, time_budget, rng):
    """
    Coordinate descent: for each coordinate, try to optimize it.
    """
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        for j in range(n):
            # Try different values for a[j]
            old_val = a[j]
            # Try nearby values
            for factor in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0]:
                new_val = old_val * factor
                a[j] = new_val
                # Renormalize (keep sum = 1)
                s = a.sum()
                if s > 0:
                    a /= s
                sc = normed_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            a[j] = old_val
            # Renormalize back
            s = a.sum()
            if s > 0:
                a /= s
        
        iters += 1
        if not improved and iters > 10:
            break
    
    return best_a, best_sc
