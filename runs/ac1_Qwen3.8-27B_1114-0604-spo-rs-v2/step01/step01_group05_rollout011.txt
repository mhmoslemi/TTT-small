def fft_convolution(a):
    """Linear convolution a*a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast local objective: 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j w_{i+j} for gradient computation."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:

        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)

        c0 = tmp

        "ftol": 1e-12,
        "gtol": 1e-6,
        "maxls": 50,
        "maxcor": 12,
    }
    try:
        res = minimize(
            block_objective, c0, args=(n, B, beta),

    except Exception:
        c = c0

def structured_starts(n, rng, max_count=12):

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.
Multi-phase optimization: adaptive-beta projected gradient with annealing,
block-level L-BFGS, and targeted perturbation around correlation hot spots.
Uses previous best sequence as warm start when available.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b and corresponding softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for gradient of smooth max."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)

        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)
    options = {

        "gtol": 1e-7,
        "maxls": 50,
        "maxcor": 15,

    try:
        res = minimize(

def structured_starts(n, rng, max_count=16):

        p = np.exp(z)
        p /= p.sum()
        starts.append(p)
    # Gaussian bells
    for center in (0.3, 0.5, 0.7):
        for width in (0.1, 0.2, 0.4):
            p = np.exp(-((x - center) ** 2) / (2.0 * width * width))
            p /= p.sum()
            starts.append(p)
    # Double peak
    p = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01)
    p /= p.sum()
    starts.append(p)
    # M-sequence-like (pseudo-binary)
    for s in range(4):
        rng2 = np.random.default_rng(s * 7919 + 13)
        p = rng2.random(n)
        p = np.where(p > 0.5, 1.0, 0.2)
        p /= p.sum()
        starts.append(p)
    # Smooth random (low-pass filtered noise)
    size = 1
    while size < n:
        size <<= 1
    for s in range(4):

        z = 2.5 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()

    return starts[:max_count]


def choose_best_starts(starts, count):
    """Return up to count starts with lowest fast_score."""
    scored = []
    for p in starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda t: t[0])
    return [s[1] for s in scored[:count]]


def full_refine_adaptive(p0, deadline, rng, max_iter=10000000):
    """Full-length projected gradient with logarithmic beta annealing."""
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    # Break symmetry
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-13:
        t = 0.123 * np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(t)))

    best_p = p.copy()
    best_score = fast_score(p)

    step = 1.0
    no_improve = 0
    it = 0

    beta_start = 2000.0
    beta_end = 5000000.0
    anneal_iters = min(max_iter, max(1000, int(0.2 * max_iter)))

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Logarithmic beta annealing
        frac = min(1.0, it / anneal_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_smooth = float(n) * G
        m = float(np.max(b))
        cur_score = float(2.0 * n * m)

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"    it={it} best={best_score:.8f} beta={beta:.0f} step={step:.3e}")

        if no_improve > 8000:
            # Try to restart with perturbation
            if time.monotonic() < deadline:
                pert = p + 0.01 * rng.normal(size=n)
                pert = normalize_simplex(pert)
                pscore = fast_score(pert)
                if pscore < best_score:
                    p = pert
                    best_p = pert.copy()
                    best_score = pscore
                    no_improve = 0
                    step = 1.0
                else:
                    break
            else:
                break

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        base = step

        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001):
            alpha = base * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if float(n) * G2 < cur_smooth - 1e-13:
                p = q
                cur_smooth = float(n) * G2
                cur_score = float(2.0 * n * m2)
                accepted = True
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_p = p.copy()
                    no_improve = 0
                break

        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            if step < 1.0:
                step = min(1.0, step * 1.15)

    return best_p, best_score


def targeted_perturbation(p, deadline, rng, max_ops=20000):
    """Targeted perturbation: try swaps that reduce hot-lag correlations."""
    p = normalize_simplex(p)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    ops = 0
    stall = 0

    while ops < max_ops and time.monotonic() < deadline:
        ops += 1
        b = fft_convolution(p)
        m = float(np.max(b))
        cur_score = 2.0 * n * m

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            break

        if ops % 500 == 0:
            print(f"    targeted ops={ops} score={best_score:.8f}")

        # Identify hot lags
        threshold = 0.92 * m
        hot_lags = np.where(b >= threshold)[0]
        if len(hot_lags) == 0:
            hot_lags = np.array([int(np.argmax(b))])

        # Pick a hot lag and try targeted swaps
        lag = int(hot_lags[rng.integers(0, min(len(hot_lags), 30))])
        n_pairs = n - lag
        if n_pairs <= 0:
            continue

        # Sample a few positions
        n_try = min(20, n_pairs)
        indices = rng.integers(0, n_pairs, size=n_try)

        improved = False
        for idx in indices:
            delta = p[idx] * 0.08
            if delta < 1e-15 or delta >= p[idx]:
                continue

            # Move mass from idx to idx+lag
            q = p.copy()
            q[idx] -= delta
            q[idx + lag] += delta
            s = float(np.sum(q))
            if s > 0:
                q /= s
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            if m2 < m - 1e-14:
                p = q
                m = m2
                improved = True
                break

            # Move mass from idx+lag to idx
            q = p.copy()
            q[idx] += delta
            q[idx + lag] -= delta
            q = np.maximum(q, 0.0)
            s = float(np.sum(q))
            if s > 0:
                q /= s
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            if m2 < m - 1e-14:
                p = q
                m = m2
                improved = True
                break

        if not improved:
            # Random pair swap as fallback
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 == i2:
                continue
            delta = min(p[i1], p[i2]) * 0.05
            if delta < 1e-15:
                continue
            direction = rng.integers(0, 2)
            q = p.copy()
            if direction == 0:
                q[i1] -= delta
                q[i2] += delta
            else:
                q[i1] += delta
                q[i2] -= delta
            q = np.maximum(q, 0.0)
            s = float(np.sum(q))
            if s > 0:
                q /= s
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            if m2 < m - 1e-14:
                p = q
                m = m2

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    return val if np.isfinite(val) else float("inf")


def get_previous_sequence(n):
    """Try to load previous best sequence from global variable."""
    try:
        prev = globals().get('height_sequence_1', None)
        if prev is not None:
            prev = np.asarray(prev, dtype=np.float64).flatten()
            prev = np.maximum(prev, 0.0)
            s = float(np.sum(prev))
            if s > 0 and np.isfinite(s):
                prev = prev / s
                if prev.size == n:
                    return prev
                else:
                    # Interpolate
                    from scipy.interpolate import interp1d
                    x_old = np.linspace(0, 1, prev.size)
                    x_new = np.linspace(0, 1, n)
                    f = interp1d(x_old, prev, kind='cubic', fill_value='extrapolate')
                    new = np.maximum(f(x_new), 0.0)
                    ns = float(np.sum(new))
                    if ns > 0:
                        return new / ns
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function minimizing 2*n*max(a*a)/(sum(a))^2.

    Phases:
    1. Warm start from previous best if available
    2. Block-level L-BFGS for coarse structure
    3. Full refinement with adaptive beta annealing (primary phase)
    4. Targeted perturbation around correlation hot spots
    5. Final short refinement
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    reserve = 50.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    # Choose N based on budget
    if budget_s < 200.0:
        N = 4096
    else:
        N = 8192

    rng = np.random.default_rng(seed)
    print(f"N={N}, budget={budget_s:.0f}s, deadline in {deadline - start_time:.0f}s")

    # Load previous sequence
    prev_p = get_previous_sequence(N)
    if prev_p is not None:
        prev_score = fast_score(prev_p)
        print(f"Previous sequence loaded: score={prev_score:.8f}")
        best_p = prev_p.copy()
        best_score = prev_score
    else:
        best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
        best_score = fast_score(best_p)
        print(f"Starting from uniform: score={best_score:.8f}")

    # Phase 1: Block optimization (brief, for coarse structure)
    t1 = time.monotonic()
    remaining = deadline - t1
    if remaining > 60.0 and prev_p is None:
        alloc = min(remaining * 0.2, 300.0)
        print(f"Phase 1: Block search, alloc={alloc:.0f}s")
        p_block, sc_block = param_block_search(N, t1 + alloc, rng, max_B=128)
        if np.isfinite(sc_block) and sc_block < best_score:
            best_score = sc_block
            best_p = p_block.copy()
            print(f"  After block: {best_score:.8f}")
    elif remaining > 60.0 and prev_p is not None:
        # Quick verification block search to confirm coarse structure is good
        alloc = min(remaining * 0.05, 60.0)
        print(f"Phase 1: Quick block verification, alloc={alloc:.0f}s")
        p_block, sc_block = param_block_search(N, t1 + alloc, rng, max_B=64)
        if np.isfinite(sc_block) and sc_block < best_score:
            best_score = sc_block
            best_p = p_block.copy()
            print(f"  After block: {best_score:.8f}")

    # Phase 2: Full refinement with adaptive beta (main workhorse)
    t2 = time.monotonic()
    remaining = deadline - t2
    if remaining > 60.0:
        # Build starting pool
        starts = [best_p]
        if prev_p is not None:
            starts.append(prev_p)
        for p0 in structured_starts(N, rng, max_count=10):
            starts.append(p0)

        top_starts = choose_best_starts(starts, 4)

        # Allocate time: 60% to best, 25% to second, 15% to third
        fracs = (0.55, 0.25, 0.15, 0.05)
        for i, p0 in enumerate(top_starts):
            rem = deadline - time.monotonic() - 30.0
            if rem < 30.0:
                break
            frac = fracs[i] if i < len(fracs) else 0.05
            alloc = rem * frac
            sub_deadline = time.monotonic() + alloc
            print(f"  Full refine start {i}, alloc={alloc:.0f}s, initial_score={fast_score(p0):.8f}")
            p_fit, sc_fit = full_refine_adaptive(p0, sub_deadline, rng)
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  After refine {i}: {best_score:.8f}")

    # Phase 3: Targeted perturbation
    t3 = time.monotonic()
    remaining = deadline - t3
    if remaining > 30.0:
        max_ops = int(remaining * 1500)  # ~1500 ops per second target
        max_ops = min(max_ops, 50000)
        print(f"Phase 3: Targeted perturbation, max_ops={max_ops}")
        p_target, sc_target = targeted_perturbation(best_p, deadline - 10.0, rng, max_ops=max_ops)
        if np.isfinite(sc_target) and sc_target < best_score:
            best_score = sc_target
            best_p = p_target.copy()
            print(f"  After targeted: {best_score:.8f}")

    # Phase 4: Final short refinement
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 20.0:
        sub_deadline = deadline - 5.0
        print(f"Phase 4: Final refinement, alloc={remaining - 5.0:.0f}s")
        p_final_r, sc_final = full_refine_adaptive(best_p, sub_deadline, rng, max_iter=500000)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_p = p_final_r.copy()
            print(f"  Final: {best_score:.8f}")

    # Cleanup and return
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0.0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s

    final_seq = [float(x) for x in final_p]

    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        alt = [1.0] * N
        off_alt = official_score(alt)
        if np.isfinite(off_alt) and off_alt < off:
            return alt

    return final_seq
