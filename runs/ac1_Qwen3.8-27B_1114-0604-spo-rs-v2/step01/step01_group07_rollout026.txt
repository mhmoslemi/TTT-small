```python
"""
Minimum peak autocorrelation nonnegative sequence search.
Uses subgradient mirror descent on the nonnegative simplex,
with adaptive active-set selection and multiple initializations.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.inf
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))

def _make_sine(n, amp, mode=1):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _optimize_1d(f, low, high, tol=1e-6, max_iter=50):
    """Golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = low, high
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x = (a + b) / 2.0
    return x, f(x)
```

```python
    arr = arr / s

        progress = min(1.0, float(it) / 10000.0)
```

```python
                if time.monotonic() > deadline:
                    break
                a = _two_sine_seq(n, a1, a2, 2)
                sc = _score_fast(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"  [scan] n={n} two_sine a1={a1} a2={a2} score={sc:.6f}")
        # Cosine shape
        for amp in [0.5, 0.8, 1.0]:
            a = _cosine_seq(n, amp)
            sc = _score_fast(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _sine_seq(2048, 0.78, 1)
        best_sc = _score_fast(best_a)
    return best_a, best_sc


def _subgrad_optimize(a0, budget_s, K_init=4, K_max=64, lr_scale=1.0, seed=0,
                      momentum_beta=0.8, verbose=True):
    """
    Subgradient mirror descent on the nonnegative simplex.
    Minimizes max convolution value (the peak of a*a).
    """
    rng = np.random.RandomState(seed)
    arr = np.asarray(a0, dtype=np.float64).copy()
    n = len(arr)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        arr = np.ones(n) / n
        s = 1.0
    arr /= s

    best = arr.copy()
    best_sc = _score_fast(best)

    lr = lr_scale * n
    t0 = time.monotonic()
    it = 0
    last_improve = 0
    mom = np.zeros(n, dtype=np.float64)
    stall_count = 0
    eval_period = 2

    while time.monotonic() - t0 < budget_s:
        it += 1

        # Adaptive K schedule
        frac = min(1.0, float(it) / 3000.0)
        K = int(K_init + frac * (K_max - K_init))

        # Compute convolution
        c = _fft_conv(arr)

        # Get top-K positions
        if K >= c.size:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        # Compute subgradient (average gradient at active positions)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                idx = np.arange(lo, hi + 1)
                grad[idx] += arr[idx] + arr[k - idx]
        grad /= max(1, len(active))

        # Project to tangent space of simplex
        g = grad - grad.mean()
        gmax = np.abs(g).max()
        if gmax < 1e-15:
            break

        # Momentum
        mom = momentum_beta * mom + (1.0 - momentum_beta) * g
        g_use = mom

        # Mirror (multiplicative weight) update
        step = lr * g_use
        step = np.clip(step, -4.0, 4.0)
        arr_new = arr * np.exp(-step)
        arr_new = np.clip(arr_new, 1e-300, 1000.0)
        sn = arr_new.sum()
        if sn < 0.01 or not np.isfinite(sn):
            lr *= 0.5
            if lr < 1e-10:
                break
            continue
        arr_new /= sn
        arr = arr_new

        # Periodic evaluation
        if it % eval_period == 0:
            sc = _score_fast(arr)
            if sc < best_sc:
                best = arr.copy()
                best_sc = sc
                last_improve = it
                lr = min(lr * 1.02, 5.0 * n)
                stall_count = 0
            else:
                stall_count += 1
                if sc > best_sc * 1.003:
                    arr = best.copy()
                    lr *= 0.75
                if stall_count > 500:
                    # Perturbed restart
                    noise_scale = 0.03 + 0.02 * (stall_count // 1000)
                    arr = best + noise_scale * rng.randn(n)
                    arr = np.clip(arr, 1e-10, None)
                    arr /= arr.sum()
                    lr = 0.5 * n
                    mom = np.zeros(n)
                    stall_count = 0
                    last_improve = it
                    if verbose:
                        print(f"    [restart] it={it} noise={noise_scale:.3f} score={best_sc:.8f}")

        # Adaptive eval frequency
        if it > 3000:
            eval_period = 3
        if it > 10000:
            eval_period = 5

    return best, best_sc


def _polish(a0, budget_s, seed=777):
    """Final polish with large K and small learning rate."""
    rng = np.random.RandomState(seed)
    arr = np.asarray(a0, dtype=np.float64).copy()
    n = len(arr)
    arr = np.clip(arr, 0.0, 1000.0)
    arr /= arr.sum()

    best = arr.copy()
    best_sc = _score_fast(best)

    n_conv = 2 * n - 1
    t0 = time.monotonic()
    it = 0
    last_improve = 0
    stall = 0
    lr = 0.3 * n
    mom = np.zeros(n)
    beta = 0.9

    # Use all positions (or top 256)
    K = min(256, n_conv)

    while time.monotonic() - t0 < budget_s:
        it += 1

        c = _fft_conv(arr)

        if K >= c.size:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                idx = np.arange(lo, hi + 1)
                grad[idx] += arr[idx] + arr[k - idx]
        grad /= max(1, len(active))

        g = grad - grad.mean()
        if np.abs(g).max() < 1e-16:
            break

        mom = beta * mom + (1.0 - beta) * g
        step = lr * mom
        step = np.clip(step, -2.0, 2.0)
        arr_new = arr * np.exp(-step)
        arr_new = np.clip(arr_new, 1e-300, 1000.0)
        sn = arr_new.sum()
        if sn < 0.01 or not np.isfinite(sn):
            lr *= 0.5
            continue
        arr_new /= sn
        arr = arr_new

        if it % 5 == 0:
            sc = _score_fast(arr)
            if sc < best_sc:
                best = arr.copy()
                best_sc = sc
                last_improve = it
                lr = min(lr * 1.01, 1.0 * n)
                stall = 0
            else:
                stall += 1
                if sc > best_sc * 1.001:
                    arr = best.copy()
                    lr *= 0.8
                if stall > 300:
                    break

    return best, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low-peak nonnegative sequence.
    Returns the best valid sequence found within the time budget.
    """
    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    hard_deadline = t0 + total - 25.0

    print(f"Starting search with budget={total:.0f}s")

    # Phase 1: Find good initialization (up to 90s)
    scan_deadline = t0 + min(90.0, total * 0.04)
    init_a, init_sc = _scan_init(scan_deadline)
    print(f"Phase 1 done: best init score={init_sc:.8f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Main optimization
    remaining = hard_deadline - time.monotonic()
    p2_budget = remaining * 0.50
    p3_budget = remaining * 0.25
    p4_budget = remaining * 0.15

    best_a = init_a.copy()
    best_sc = init_sc

    # Run 2a: Aggressive (small K, large LR)
    t_start = time.monotonic()
    a2a, sc2a = _subgrad_optimize(init_a, p2_budget,
                                   K_init=2, K_max=48,
                                   lr_scale=1.5, seed=42,
                                   momentum_beta=0.7)
    if sc2a < best_sc:
        best_a, best_sc = a2a, sc2a
        print(f"Phase 2a: improved to {best_sc:.8f} (time={time.monotonic()-t0:.1f}s)")
    else:
        print(f"Phase 2a: no improvement (best stays {best_sc:.8f}, got {sc2a:.8f})")

    # Run 2b: Balanced (medium K, medium LR) from best so far
    if time.monotonic() < hard_deadline - 60:
        remain = hard_deadline - time.monotonic()
        b2b = min(p3_budget, remain * 0.5)
        a2b, sc2b = _subgrad_optimize(best_a, b2b,
                                       K_init=8, K_max=96,
                                       lr_scale=0.8, seed=123,
                                       momentum_beta=0.8)
        if sc2b < best_sc:
            best_a, best_sc = a2b, sc2b
            print(f"Phase 2b: improved to {best_sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        else:
            print(f"Phase 2b: no improvement (best stays {best_sc:.8f})")

    # Run 2c: Different initialization (mode 2 or two-sine)
    if time.monotonic() < hard_deadline - 60:
        remain = hard_deadline - time.monotonic()
        b2c = min(p4_budget, remain * 0.4)
        # Try mode 2 initialization
        n_tr = len(best_a)
        alt_init = _sine_seq(n_tr, 0.8, 2)
        alt_sc = _score_fast(alt_init)
        if alt_sc < best_sc + 0.1:  # Only if competitive
            a2c, sc2c = _subgrad_optimize(alt_init, b2c,
                                           K_init=4, K_max=64,
                                           lr_scale=1.0, seed=456,
                                           momentum_beta=0.75)
            if sc2c < best_sc:
                best_a, best_sc = a2c, sc2c
                print(f"Phase 2c (alt init): improved to {best_sc:.8f}")

    # Phase 3: Polish (large K, small LR)
    if time.monotonic() < hard_deadline - 30:
        remain = hard_deadline - time.monotonic()
        p3_actual = min(remain * 0.5, 300.0)
        a3, sc3 = _polish(best_a, p3_actual, seed=777)
        if sc3 < best_sc:
            best_a, best_sc = a3, sc3
            print(f"Phase 3 (polish): improved to {best_sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        else:
            print(f"Phase 3 (polish): no further improvement (best={best_sc:.8f})")

    # Final: try one more polish if time allows
    if time.monotonic() < hard_deadline - 20:
        remain = hard_deadline - time.monotonic()
        if remain > 30:
            a4, sc4 = _polish(best_a, remain * 0.7, seed=888)
            if sc4 < best_sc:
                best_a, best_sc = a4, sc4
                print(f"Final polish: improved to {best_sc:.8f}")

    # Prepare output
    arr = np.asarray(best_a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    arr_sum = arr.sum()
    if arr_sum < 0.01:
        arr = np.ones(2048) / 2048.0
    else:
        arr = arr / arr_sum

    out = [float(x) for x in arr]

    # Validate with the real evaluator
    try:
        real_sc = float(evaluate_sequence(out))
        print(f"FINAL: score={real_sc:.8f}, n={len(out)}, time={time.monotonic()-t0:.1f}s")
    except Exception as e:
        print(f"Validation error: {e}, using fallback")
        out = [1.0 / 2048.0] * 2048

    return out
```
