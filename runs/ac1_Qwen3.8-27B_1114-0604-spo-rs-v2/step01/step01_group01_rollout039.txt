```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def solve_lp_direction(a, M):
    """Solve: max 1^T g s.t. A@
 g <= M, g >= 0.
    Returns the optimal g (normalized to sum 1) or None."""
    n = len(a)
    # A is (2n-1) x n, A[j, i] = a[j-i] if 0 <= j-i < n
    # Build A as sparse
    rows = []
    cols = []
    vals = []
    for j in range(2*n - 1):
        for i in range(max(0, j-n+1), min(n, j+1)):
            rows.append(j)
            cols.append(i)
            vals.append(a[j-i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T g s.t. A@g <= M, g >= 0
    # linprog minimizes, so: min -1^T g s.t. A@g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 0:
            return g / s
    return None
```

```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def normalize(a):
    """Normalize to sum 1, clip to [0, 1000]."""
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0, 1000)
    a = np.where(np.isfinite(a), a, 0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

def score_fn(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = normalize(a)
    if a is None:
        return np.inf
    n = len(a)
    c = np.convolve(a, a)
    return float(2 * n * c.max() / (a.sum()**2))

def conv_full(a, b):
    """Full convolution using FFT for speed."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L *= 2
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n]

def lp_direction(a, M, K=None):
    """
    Find the LP direction: max sum(g) s.t. conv(a,g)[j] <= M, g >= 0.
    If K is given, only use the top K tightest constraints.
    Returns normalized g (sum = 1) or None.
    """
    n = len(a)
    # Build the constraint matrix A_con: shape (2n-1, n)
    # A_con[j, i] = a[j-i] if 0 <= j-i < n
    rows = []
    cols = []
    for j in range(2*n-1):
        i_lo = max(0, j-n+1)
        i_hi = min(n, j+1)
        for i in range(i_lo, i_hi):
            rows.append(j)
            cols.append(i)
    rows = np.array(rows)
    cols = np.array(cols)
    vals = a[rows - cols]
    
    # If K is given, select the top K rows
    if K is not None and K < 2*n-1:
        c_val = conv_full(a, a)
        # Select rows where c_val is largest (most likely to be tight)
        top_idx = np.argsort(-c_val)[:K]
        # Filter rows/cols/vals to only include those rows
        mask = np.zeros(2*n-1, dtype=bool)
        mask[top_idx] = True
        # For each non-zero (row, col), check if row is in top_idx
        keep = mask[rows]
        rows = rows[keep]
        cols = cols[keep]
        vals = vals[keep]
    
    A_con = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    # If using subset, adjust shape
    if K is not None and K < 2*n-1:
        A_con = A_con[top_idx]
    
    b = np.full(A_con.shape[0], M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_con, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 0:
            return g / s
    return None
```

def _gradient_descent(p0, rng, time_limit, beta_start=50.0, beta_end=2000.0):
    """
    Multiplicative gradient descent on smoothed max of convolution.
    Returns (best_sequence, best_score).
    """
    p = _normalize(p0)
    if p is None:
        return None, np.inf
    n = len(p)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()
    
    best = p.copy()
    best_score = _score(p)
    beta = float(beta_start)
    step = 0.5
    deadline = time.monotonic() + time_limit
    stall = 0
    
    while time.monotonic() < deadline:
        exact, smooth, grad = _gradient_smooth(p, beta)
        if not np.isfinite(exact) or grad is None:
            break
        
        if exact < best_score:
            best = p.copy()
            best_score = exact
            stall = 0
        else:
            stall += 1
        
        # Normalize gradient (project onto tangent space of simplex)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.maximum(grad, 0.0)  # Only decrease where gradient is positive
        gmax = float(np.max(grad)) if grad.size else 0
        if gmax < 1e-15:
            # Stuck: perturb
            p = _normalize(p * np.exp(rng.normal(0, 0.1, n)))
            if p is None:
                break
            p = np.maximum(p, floor)
            p = p / p.sum()
            stall = 0
            continue
        
        grad_n = grad / gmax
        
        # Try a few step sizes
        best_try = None
        best_try_score = best_score
        for t in [step, step * 0.5, step * 0.25, step * 2.0]:
            if t < 1e-4 or t > 10:
                continue
            p_try = p * np.exp(-t * grad_n)
            p_try = np.maximum(p_try, floor)
            s = float(p_try.sum())
            if s <= 0:
                continue
            p_try = p_try / s
            sc = _score(p_try)
            if sc < best_try_score:
                best_try_score = sc
                best_try = p_try.copy()
        
        if best_try is not None:
            p = best_try
            step = min(step * 1.1, 5.0)
        else:
            step = max(step * 0.6, 0.01)
            if stall > 30:
                p = _normalize(best * np.exp(rng.normal(0, 0.05, n)))
                if p is not None:
                    p = np.maximum(p, floor)
                    p = p / p.sum()
                stall = 0
            if stall > 100:
                break
        
        if time.monotonic() % 10 < time_limit / max(1, time_limit):  # every ~10 iterations
            beta = min(beta_end, beta * 1.1)
    
    # Final check
    final_score = _score(best)
    if final_score < best_score:
        best_score = final_score
    return best, best_score


def _build_conv_indices(n):
    """Precompute row/col indices for sparse convolution matrix (2n-1) x n."""
    total = n * n  # exact count
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for j in range(2 * n - 1):
        i_lo = max(0, j - n + 1)
        i_hi = min(n, j + 1)
        for i in range(i_lo, i_hi):
            rows[idx] = j
            cols[idx] = i
            idx += 1
    return rows, cols


def _lp_direction(a, M, rows, cols, K=None):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[j] <= M for all j (or top K), g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    vals = a[rows - cols]
    
    if K is not None and K < 2 * n - 1:
        # Only use top K constraints
        c_val = _conv_full(a, a)
        top_idx = np.argsort(-c_val)[:K]
        top_set = set(top_idx.tolist())
        mask = np.array([r in top_set for r in rows.tolist() if r < 2*n-1])
        # More efficient: use searchsorted or direct indexing
        keep_mask = np.isin(rows, top_idx)
        rows_s = rows[keep_mask]
        cols_s = cols[keep_mask]
        vals_s = vals[keep_mask]
        A = csr_matrix((vals_s, (rows_s, cols_s)), shape=(2 * n - 1, n))
        A = A[top_idx]
    else:
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    b = np.full(A.shape[0], M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = float(g.sum())
        if s > 0:
            return g / s
    return None


def _lp_iterate(a, rows, cols, time_limit, K=None):
    """
    Iterate: compute M, solve LP for direction g, blend a with g.
    Returns (best_sequence, best_score).
    """
    p = _normalize(a)
    if p is None:
        return None, np.inf
    n = len(p)
    
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        M = float(np.max(c))
        
        # Solve LP
        g = _lp_direction(p, M, rows, cols, K=K)
        if g is None:
            stall += 10
            if stall > 50:
                break
            # Try without constraint limit
            g = _lp_direction(p, M, rows, cols, K=None)
            if g is None:
                break
        
        # Line search: find best t in (0,1]
        best_t = 0
        best_t_score = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
            p_try = (1 - t) * p + t * g
            p_try = _normalize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < best_t_score:
                best_t_score = sc
                best_t = t
                best_t_seq = p_try
        
        if best_t > 0:
            p = best_t_seq
            if best_t_score < best_score:
                best = p.copy()
                best_score = best_t_score
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb and try again
                p = _normalize(p * np.exp(np.random.default_rng(it).normal(0, 0.05, n)))
                if p is None:
                    break
                stall = 0
            if stall > 100:
                break
        
        it += 1
        if it % 20 == 0:
            print(f"  LP iter {it}: score={best_score:.6f}, stall={stall}")
    
    return best, best_score


def _block_search(p0, rng, time_limit):
    """Coordinate block scaling search."""
    p = _normalize(p0)
    if p is None:
        return None, np.inf
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() < deadline:
        # Choose block size
        r = rng.random()
        if r < 0.4:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.8:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            w = int(rng.integers(max(3, n // 20), max(4, min(n, n // 5 + 1))))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        
        base_score = best_score
        improved = False
        for f in factors:
            cand = best.copy()
            cand[start:start + w] *= f
            cand = _normalize(cand)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < best_score - 1e-12:
                best = cand
                best_score = sc
                improved = True
        
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                p_try = _normalize(best * np.exp(rng.normal(0, 0.08, n)))
                if p_try is not None:
                    sc = _score(p_try)
                    if sc <= best_score * 1.1:
                        best = p_try
                        best_score = sc
                stagnant = 0
            if stagnant > 300:
                break
    
    return best, best_score


def _resample(arr, m):
    """Resample normalized vector to length m."""
    a = _normalize(arr)
    if a is None or m <= 0:
        return None
    n = a.size
    if n == m:
        return a
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    return _normalize(out)


def _initial_sequence(n, rng, mode=0):
    """Generate a diverse initial sequence."""
    t = (np.arange(n, dtype=float) + 0.5) / n
    if mode == 0:
        # Cosine series
        p = np.ones(n)
        for k in range(1, 10):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(p, -30, 30)))
    elif mode == 1:
        # Beta-like
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt**a * (1 - tt)**b
    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 5))
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0, 1)
            s = rng.uniform(0.02, 0.3)
            w = rng.uniform(0.5, 2.0)
            p += w * np.exp(-((t - c) / s)**2)
    elif mode == 3:
        # Periodic pattern
        freq = int(rng.integers(1, 10))
        p = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        p = p + 0.3 * np.cos(4 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    else:
        p = np.ones(n)
    
    p = np.where(np.isfinite(p), p, 0)
    p = np.clip(p, 0, 1000)
    p = p + 1e-10
    s = p.sum()
    if s <= 0:
        return np.ones(n) / n
    return p / s


def _smooth_array(p, sigma):
    """Apply Gaussian smoothing."""
    p = _normalize(p)
    if p is None or sigma <= 0 or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma)**2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode='same')
    return _normalize(out)
