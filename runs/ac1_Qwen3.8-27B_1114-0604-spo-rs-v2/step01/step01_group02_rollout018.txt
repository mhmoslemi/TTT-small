    return np.fft.irfft(A * A, M_fft)[:M]

    S = a.sum()
    if S < 1e-10:
        return np.inf, c
    return 2.0 * n * c.max() / (S * S), c

def soft_max_objective(a, n, temp, topk):
    """Smooth approximation of 2n*max(c)/(sum a)^2 using log-sum-exp."""
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    # Use top-k for efficiency
    k = min(topk, len(c))
    top_vals = np.sort(c)[-k:][::-1]
    # log-sum-exp
    m = top_vals[0]
    lse = m + temp * np.log(np.sum(np.exp((top_vals - m) / temp)))
    score = 2.0 * n * lse / (S * S)
    
    # Gradient
    weights = np.exp((top_vals - m) / temp)
    weights = weights / weights.sum()
    
    grad_c = np.zeros(n)
    sorted_idx = np.argsort(c)[::-1][:k]
    for i in range(k):
        ki = sorted_idx[i]
        wi = weights[i]
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        # Correct derivative: ∂c[ki]/∂a[j] = a[ki-j] (or 2a[ki-j] if ki=2j)
        for j in range(j_start, j_end + 1):
            if ki == 2 * j:

            else:
                grad_c[j] += wi * a[ki - j]
    
    # d/d a[j] of 2n * lse / S^2 = 2n * (∂lse/∂a[j] / S^2 - 2 * lse / S^3)
    grad = (2.0 * n) * (grad_c / (S * S) - 2.0 * lse / (S * S * S))
    
    return score, grad

import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def lse_score_and_grad(a, n, topk, temp):
    """Compute the LSE-smoothed score and its gradient."""
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    k = min(topk, len(c))
    # Get top-k values and indices
    if k < len(c):
        top_idx = np.argpartition(c, -k)[-k:]
    else:
        top_idx = np.arange(len(c))
    top_vals = c[top_idx]
    
    # Log-sum-exp
    m = top_vals.max()
    lse = m + temp * np.log(np.sum(np.exp((top_vals - m) / temp)))
    
    score = 2.0 * n * lse / (S * S)
    
    # Weights
    weights = np.exp((top_vals - m) / temp)
    weights = weights / weights.sum()
    
    # Gradient of lse w.r.t. a
    grad_c = np.zeros(n)
    for i in range(k):
        ki = int(top_idx[i])
        wi = weights[i]
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        # Vectorized: a[ki-j] for j in [j_start, j_end] = a[j_start:j_end+1][::-1]
        grad_c[j_start:j_end+1] += wi * a[j_start:j_end+1][::-1]
        # Fix the double-counting at j = ki/2
        if ki % 2 == 0:
            mid = ki // 2
            if j_start <= mid <= j_end:
                grad_c[mid] += wi * a[mid]
    
    # Full gradient: d/d a[j] [2n * lse / S²] = 2n * (grad_c[j]/S² - 2*lse/(S³))
    grad = (2.0 * n) * (grad_c / (S * S) - 2.0 * lse / (S ** 3))
    
    return score, grad
