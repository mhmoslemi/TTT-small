"""
Low-peak nonnegative step-function search using active-set gradient descent
with momentum, multiple initializations, and adaptive learning rate.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _normalize(a):
    arr = np.asarray(a, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.ones_like(arr) / arr.size
    return arr / s

def _make_sine_candidate(n, amp, mode):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _make_tapered_candidate(n, taper_power):
    """Create a sequence with power-law tapering at edges."""
    i = np.arange(n, dtype=np.float64)
    # Distance from nearest edge, normalized to [0, 0.5]
    d = np.minimum(i + 0.5, n - i - 0.5) / (n / 2.0)
    a = np.power(np.clip(d, 0.0, 1.0), taper_power)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _gradient_step(a, K, lr, momentum_buffer=None, momentum_beta=0.5):
    """
    Perform one gradient step on the simplex to reduce max autocorrelation.
    Returns updated a and optional new momentum buffer.
    """
    n = a.size
    c = _fft_convolve(a)
    maxc = float(np.max(c))
    
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]
    
    grad = np.zeros(n, dtype=np.float64)
    cnt = 0
    for kk in active:
        k = int(kk)
        cnt += 1
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

Low-peak nonnegative step-function search.
Uses active-set gradient descent with momentum, adaptive learning rate,
soft-max gradient weighting, and multiple initializations.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score_fast(arr):
    """Fast internal score computation."""
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))

def _normalize(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 1e-15, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return arr / s

    i = np.arange(n, dtype=np.float64)

    Compute the gradient of max(R) projected onto the simplex tangent space.
    Uses top-K lags, optionally with soft-max weighting.
    Returns (gradient, max_c).
    """
    n = a.size
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
        weights = np.ones(len(active))
    else:
        if soft_beta is not None and soft_beta > 0:
            # Use soft-max weighting for top 2K lags
            top_2k = np.argpartition(c, -2*K)[-2*K:]
            top_c = c[top_2k]
            logits = soft_beta * (top_c - top_c.max())
            weights = np.exp(logits)
            weights /= weights.sum()
            active = top_2k
        else:

            weights = np.ones(len(active)) / len(active)

            grad[idx] += w * 2.0 * a[k - idx]

Low-peak nonnegative step-function search using active-set gradient descent

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT (zero-padded)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score_fast(arr):
    """Fast internal evaluation of the score (assumes arr is normalized, non-negative)."""
    arr = np.asarray(arr, dtype=np.float64)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _normalize(a):
    """Normalize to sum=1, enforce non-negativity and finiteness."""
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return arr / s


def _make_sine(n, amp, mode):
    """Create sinusoidal anti-palindromic candidate."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + amp * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)

    """Create cosine candidate."""

    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + amp * np.cos(float(mode) * theta)
    a = np.clip(a, 0.0, 1000.0)

    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.exp(amp * np.sin(float(mode) * theta))
    return _normalize(a)

    """Create dual-frequency sinusoidal candidate."""
    i = np.arange(n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)

def _compute_grad_active(a, K, soft_beta):
    """
    Compute gradient of max autocorrelation projected to simplex tangent space.
    Uses top-K lags with optional soft-max weighting.
    Returns (gradient_vector, current_max).
    """
    n = a.size
    c = _fft_convolve(a)
    max_c = float(np.max(c))
    
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
        weights = np.full(len(active), 1.0 / len(active))
    elif soft_beta is not None and soft_beta > 0:
        # Soft-max weighting over top 2K
        K2 = min(2 * K, c.size)
        top_idx = np.argpartition(c, -K2)[-K2:]
        top_c = c[top_idx]
        logits = soft_beta * (top_c - top_c.max())
        exp_logits = np.exp(logits)
        weights = exp_logits / exp_logits.sum()
        active = top_idx
    else:
        active = np.argpartition(c, -K)[-K:]
        weights = np.full(len(active), 1.0 / len(active))
    
    grad = np.zeros(n, dtype=np.float64)
    for w, kk in zip(weights, active):
        k = int(kk)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += w * 2.0 * a[k - idx]
    
    grad -= np.mean(grad)
    return grad, max_c

    max_lr = 0.4 * n
    min_lr = 0.5
    momentum = None
    beta = 0.4
    start = time.monotonic()
    it = 0
    last_improve = 0
    stagnation = 0
    max_stag = 400
    period = max(3, n // 1024)
    
    while time.monotonic() < deadline:
        it += 1
        
        grad, _ = _compute_grad_active(arr, K, soft_beta)
        
        if momentum is None:
            momentum = grad.copy()
        else:
            momentum = beta * momentum + (1.0 - beta) * grad

        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s = float(arr_new.sum())
        if s < 0.01 or not np.isfinite(s):
            arr_new = arr.copy()
            lr = max(min_lr, lr * 0.5)
            momentum = None
        else:
            arr_new = arr_new / s
            arr = arr_new

            sc = _score_fast(arr)

Low-peak nonnegative step-function search using active-set gradient descent
with momentum, soft-max gradient weighting, adaptive learning rate, and
multiple diverse initializations. Key improvements over prior approach:
- Soft-max weighted gradient over top-2K lags (smoother than hard top-K)

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score_fast(arr):
    """Fast internal score (assumes non-negative finite array)."""
    arr = np.asarray(arr, dtype=np.float64)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _normalize(a):
    """Normalize to sum=1, clip to [0,1000], ensure finite."""
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(arr.size, dtype=np.float64) / max(1, arr.size)
    return arr / s


def _make_sine(n, amp, mode):
    """Sinusoidal anti-palindromic candidate."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + amp * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)


def _make_cosine(n, amp, mode):
    """Cosine candidate (peaked at center or edges)."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + amp * np.cos(float(mode) * theta)
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)


def _make_exp_sine(n, amp, mode):
    """Exponential-sinusoidal (always positive, different contrast)."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.exp(amp * np.sin(float(mode) * theta))
    return _normalize(a)


def _make_dual_sine(n, a1, a2, m1, m2):
    """Dual-frequency sinusoidal."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + a1 * np.sin(float(m1) * theta) + a2 * np.sin(float(m2) * theta)
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)


def _make_taper(n, power):
    """Power-law edge taper (reduces boundary effects)."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i + 0.5, n - i - 0.5) / (n * 0.5)
    a = np.power(np.maximum(d, 1e-10), power)
    return _normalize(a)


def _compute_grad_and_max(a, K, soft_beta):
    """
    Compute gradient of max autocorrelation (projected to tangent space)
    and the current max value. Uses top-K (or soft-max over top-2K) lags.
    Returns (grad, max_c).
    """
    n = a.size
    c = _fft_convolve(a)
    max_c = float(np.max(c))

    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
        weights = np.full(c.size, 1.0 / c.size)
    elif soft_beta is not None and soft_beta > 0:
        K2 = min(2 * K, c.size)
        top_idx = np.argpartition(c, -K2)[-K2:]
        top_c = c[top_idx]
        logits = soft_beta * (top_c - top_c.max())
        exp_l = np.exp(logits)
        weights = exp_l / exp_l.sum()
        active = top_idx
    else:
        active = np.argpartition(c, -K)[-K:]
        weights = np.full(len(active), 1.0 / len(active))

    grad = np.zeros(n, dtype=np.float64)
    for w, kk in zip(weights, active):
        k = int(kk)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += w * 2.0 * a[k - idx]

    grad -= np.mean(grad)
    return grad, max_c
