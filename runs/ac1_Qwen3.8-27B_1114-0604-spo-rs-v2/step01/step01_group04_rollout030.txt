import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, diags

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score_normalized(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                pass

            # Check improvement
            cur_max = max(float(np.max(c[i:i + n])), unchanged_max)
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if verbose and p % 10 == 0:
            print(f"    CD pass {p}: max_c={best_max:.8f}")

        if no_improve > 3 * n:
            break

    a = project_simplex(best_a)
    c = full_conv(a)
    return a, c, float(np.max(c))


def subgrad_step(a, c, tau, eta, vel, beta, rng):
    """One step of smoothed subgradient descent with momentum."""
    n = a.size
    # Smoothed subgradient at temperature tau
    shift = c / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s

    # Gradient: for each lag k with weight w[k], gradient of c[k] w.r.t. a is 2*a shifted
    # Subgradient = sum_k w[k] * grad(c[k])
    # grad(c[k])_i = 2*a[k-i] if 0<=k-i<n else 0
    # So subgrad[i] = 2 * sum_k w[k] * a[k-i] * 1_{0<=k-i<n}
    # = 2 * (w * a)[i] (convolution of w with a, truncated)
    # Using np.correlate or manual FFT
    n_conv = 2 * n - 1
    size = 1 << (n_conv - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    conv_wa = np.fft.irfft(fw * fa, size)[:n_conv]
    # conv_wa[k] = sum_i w[i]*a[k-i], we want for each i: sum_k w[k]*a[k-i]
    # This is actually the cross-correlation: corr[i] = sum_k w[k]*a[k-i]
    # = conv_wa[2*(n-1)-i] ... let me think
    # conv_wa[k] = sum_j w[j]*a[k-j]
    # What we want: g[i] = sum_k w[k]*a[k-i] (for valid k-i)
    # = conv_wa[2*(n-1)-i]? Let's check: conv_wa[2n-2-i] = sum_j w[j]*a[2n-2-i-j]
    # Let k = 2n-2-i-j, then j = 2n-2-i-k. Hmm, not quite.
    # Actually: g[i] = sum_k w[k] * a[k-i] = (w * a_flipped)[i] where a_flipped[j] = a[n-1-j]
    # Or: g[i] = sum_k w[k] * a[k-i]. Let m = k-i, k = m+i: g[i] = sum_m w[m+i]*a[m]
    # = (w shifted by i) dot a = cross_correlation(w, a) at lag i
    # In numpy: np.correlate(w, a, mode='full')[n-1+i] gives sum_k w[k]*a[k-i]
    # But that's O(n^2). Let's use FFT.
    # np.correlate(w, a, 'full')[k] = sum_j w[j] * a[j+k-n+1] ... this is getting confusing.
    # Let me just use the formula directly with FFT:
    # g[i] = sum_k w[k]*a[k-i] for 0<=k-i<n
    # = sum_m w[m+i]*a[m] for m=0..n-1-i (when i>=0)
    # This is the cross-correlation of w and a at lag i.
    # Using FFT: cross_corr = ifft(conj(fft(w)) * fft(a)) ... let me just do it directly.

    # Simpler approach: use the fact that the subgradient is 2 * cross_correlation(w, a)
    # where cross_correlation(w,a)[i] = sum_k w[k]*a[k-i]
    # In terms of convolution: cross_correlation(w,a) = convolve(flip(w), a)
    # flip(w) has length 2n-1, a has length n, so convolve has length 3n-2
    # cross_correlation[i] = convolve(flip(w), a)[i + (2n-2)] ... this is getting messy.

    # Let me just do it the straightforward O(n^2) way for correctness,
    # or use a simpler FFT-based approach.

    # Actually, the simplest correct approach:
    # g[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # = 2 * sum_{m: 0<=m<n, 0<=m+i<2n-1} w[m+i] * a[m]
    # For i in [0, n-1]: g[i] = 2 * sum_{m=0}^{n-1-i} w[m+i] * a[m]

    # This is the cross-correlation. Let me compute it via FFT:
    # cross_corr[i] = sum_m w[m+i]*a[m] = (w circ convolve flip(a))[i] ... no.
    # Let's use: cross_corr = ifft(fft(w, N) * conj(fft(a, N))) where N >= 2n-1
    # Actually: ifft(fft(w) * conj(fft(a)))[i] = sum_k w[k] * a[k-i] (circular)
    # With N >= 2n-1, this gives the linear cross-correlation.

    N = 1 << (2 * n - 1 - 1).bit_length()
    Fw = np.fft.rfft(w, N)
    Fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(Fw * np.conj(Fa), N)[:2 * n - 1]
    # cross[i] = sum_k w[k] * a[(k-i) mod N]
    # For 0 <= i < n: cross[i] = sum_k w[k]*a[k-i] (linear, since N >= 2n-1)
    g = 2.0 * cross[:n]

    # Momentum
    vel_new = beta * vel + (1.0 - beta) * g
    a_new = project_simplex(a - eta * vel_new)
    return a_new, vel_new


def subgrad_optimize(a, time_budget, rng, verbose=False):
    """Smoothed subgradient descent with adaptive parameters."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    c = full_conv(a)
    best_a = a.copy()
    best_sc = float(np.max(c))

    eta = 0.15
    tau_frac = 0.05
    beta = 0.5
    vel = np.zeros(n)
    iters = 0
    last_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        c = full_conv(a)
        cur_max = float(np.max(c))

        tau = tau_frac * cur_max
        if tau < 1e-18:
            tau = 1e-18

        a_new, vel = subgrad_step(a, c, tau, eta, vel, beta, rng)
        c_new = full_conv(a_new)
        new_max = float(np.max(c_new))

        if new_max <= cur_max or (rng is not None and rng.random() < 0.05):
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)

        if new_max < best_sc:
            best_sc = new_max
            best_a = a.copy()
            last_improve = iters

        iters += 1

        # Adaptive reheat
        if iters - last_improve > 5000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.2 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()
            beta = 0.3 + 0.4 * rng.random()
            last_improve = iters

        if iters % 3000 == 0:
            eta = 0.08 + 0.2 * rng.random()
            tau_frac = 0.02 + 0.06 * rng.random()

    return best_a, best_sc * n * 2.0  # Return actual score


def init_u_shape(n):
    """U-shaped: high at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided_quadratic(n):
    """Quadratic: high at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def init_two_peak(n, rng):
    """Two Gaussian peaks."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.7 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + float(rng.integers(0, 4))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_power_law(n, rng):
    """a_i ~ 1/(i+1)^alpha for random alpha."""
    i = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 1.0 * rng.random()
    a = 1.0 / (i + 1.0) ** alpha
    a_sym = 1.0 / (n - i) ** alpha
    a = a + a_sym
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def get_inits(n, rng):
    inits = []
    inits.append(init_u_shape(n))
    inits.append(init_sided_quadratic(n))
    inits.append(init_two_peak(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_cosine(n, rng))
    inits.append(init_power_law(n, rng))
    return inits


def lp_iterate(a, time_budget, rng, verbose=False):
    """
    LP-based iteration: repeatedly solve LP to find direction, update a.
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    c = full_conv(a)
    best_a = a.copy()
    best_max = float(np.max(c))

    t0 = time.monotonic()
    iteration = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        M = float(np.max(c))

        # Solve LP
        lp_time = min(10.0, (time_budget - (time.monotonic() - t0)) * 0.3)
        g = lp_direction(a, M, time_limit=lp_time)

        if g is None:
            # Fallback: use subgradient step
            a_new, _ = subgrad_step(a, c, 0.05 * M, 0.1, np.zeros(n), 0.0, rng)
            g = a_new
        else:
            # Rescale g to sum = 1
            g_sum = g.sum()
            if g_sum < 1e-15:
                g = a.copy()
            else:
                g = g / g_sum
            # Blend
            t_blend = 0.3 + 0.4 * rng.random()
            a_new = project_simplex((1 - t_blend) * a + t_blend * g)

        c_new = full_conv(a_new)
        new_max = float(np.max(c_new))

        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            stall = 0
            if verbose:
                print(f"      LP iter {iteration}: max_c improved to {new_max:.8f}")
        else:
            stall += 1
            # Try smaller step
            a_new2 = project_simplex(0.5 * (1 - t_blend) * a + 0.5 * t_blend * g)
            c_new2 = full_conv(a_new2)
            if float(np.max(c_new2)) < new_max:
                a_new = a_new2
                new_max = float(np.max(c_new2))
                if new_max < best_max:
                    best_max = new_max
                    best_a = a_new.copy()
                    stall = 0

        a = a_new
        c = c_new
        iteration += 1

        if stall > 10:
            # Perturb and restart
            a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
            c = full_conv(a)
            stall = 0

    return best_a, 2.0 * n * best_max


def try_length(n, budget, rng, verbose=False):
    """Try to optimize for a given length n."""
    inits = get_inits(n, rng)
    best_a = None
    best_score = np.inf

    # Time per init
    time_per = budget / (len(inits) + 2)

    for ii, a_init in enumerate(inits):
        if time.monotonic() > budget:
            break

        tb_sub = time_per * 0.4
        tb_lp = time_per * 0.5
        tb_cd = time_per * 0.1

        # Subgradient phase
        a_sg, sc_sg = subgrad_optimize(a_init, tb_sub, rng)

        # LP phase
        a_lp, sc_lp = lp_iterate(a_sg, tb_lp, rng, verbose=verbose)

        # Coordinate descent polish
        a_cd = project_simplex(a_lp)
        c_cd = full_conv(a_cd)
        tb_cd_actual = min(tb_cd, max(2.0, budget - (time.monotonic() - (budget - budget))))
        a_cd2, c_cd2, max_cd = coordinate_descent(a_cd, c_cd, max_passes=50,
                                                   time_budget=tb_cd, rng=rng)
        sc_cd = 2.0 * n * max_cd

        # Track best
        candidates = [(a_sg, sc_sg), (a_lp, sc_lp), (a_cd2, sc_cd)]
        for a_c, sc_c in candidates:
            if sc_c < best_score:
                best_score = sc_c
                best_a = a_c.copy()

        if verbose:
            print(f"    init {ii}: sg={sc_sg:.6f}, lp={sc_lp:.6f}, cd={sc_cd:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iteration, coordinate descent, and subgradient methods
    across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 3.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Lengths to try
    n_list = [64, 128, 256, 512, 1024, 2048]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 4096:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: give more time to lengths near the previous best
    if prev_a is not None:
        np_ = prev_a.size
        raw_w = [1.0 / (1.0 + abs(nv - np_) / 256.0) for nv in n_list]
    else:
        raw_w = [1.0 / (0.5 + nv / 512.0) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.80
        time_n = max(5.0, time_n)
        print(f"[n={nv}] allocating {time_n:.1f}s")

        # If previous best has this length, start with it
        if prev_a is not None and prev_a.size == nv:
            # Do a quick coordinate descent polish on the previous best
            a_prev = prev_a.copy()
            c_prev = full_conv(a_prev)
            tb_quick = min(30.0, time_n * 0.2)
            a_pol, c_pol, max_pol = coordinate_descent(a_prev, c_prev,
                                                       max_passes=50,
                                                       time_budget=tb_quick,
                                                       rng=rng)
            sc_pol = 2.0 * nv * max_pol
            if sc_pol < best_overall_sc:
                best_overall_sc = sc_pol
                best_overall = a_pol.copy()
                print(f"  [polish prev] improved to {sc_pol:.6f}")

            # Continue optimization from the polished version
            tb_remaining = min(time_n * 0.8, deadline - time.monotonic() - 2.0)
            a_res, sc_res = lp_iterate(a_pol, tb_remaining, rng, verbose=True)
            if sc_res < best_overall_sc:
                best_overall_sc = sc_res
                best_overall = a_res.copy()
                print(f"  [LP from prev] improved to {sc_res:.6f}")

            # Coordinate descent polish
            if time.monotonic() < deadline - 5.0:
                a_cd = project_simplex(best_overall)
                c_cd = full_conv(a_cd)
                tb_cd = min(30.0, deadline - time.monotonic() - 3.0)
                a_cd2, c_cd2, max_cd2 = coordinate_descent(a_cd, c_cd,
                                                           max_passes=100,
                                                           time_budget=tb_cd,
                                                           rng=rng)
                sc_cd2 = 2.0 * nv * max_cd2
                if sc_cd2 < best_overall_sc:
                    best_overall_sc = sc_cd2
                    best_overall = a_cd2.copy()
                    print(f"  [CD polish] improved to {sc_cd2:.6f}")
        else:
            # Try multiple inits
            inits = get_inits(nv, rng)
            time_per_init = (time_n - 5.0) / len(inits)
            time_per_init = max(3.0, time_per_init)

            for ii, a_init in enumerate(inits):
                if time.monotonic() > deadline:
                    break

                tb_sg = time_per_init * 0.35
                tb_lp = time_per_init * 0.45
                tb_cd = time_per_init * 0.20

                # Subgradient
                a_sg, sc_sg = subgrad_optimize(a_init, tb_sg, rng)

                # LP iteration
                a_lp, sc_lp = lp_iterate(a_sg, tb_lp, rng)

                # Coordinate descent
                a_cd = project_simplex(a_lp)
                c_cd = full_conv(a_cd)
                tb_cd_actual = min(tb_cd, max(2.0, deadline - time.monotonic() - 3.0))
                a_cd2, c_cd2, max_cd2 = coordinate_descent(a_cd, c_cd,
                                                           max_passes=50,
                                                           time_budget=tb_cd_actual,
                                                           rng=rng)
                sc_cd2 = 2.0 * nv * max_cd2

                # Track best
                for a_c, sc_c in [(a_sg, sc_sg), (a_lp, sc_lp), (a_cd2, sc_cd2)]:
                    if sc_c < best_overall_sc:
                        best_overall_sc = sc_c
                        best_overall = a_c.copy()
                        print(f"  [n={nv}, init={ii}] NEW BEST: {sc_c:.6f}")

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}")

    # Final global polish
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        n_best = best_overall.size
        print(f"[final polish] n={n_best}, time_left={remain:.1f}s")

        # Multiple rounds of coordinate descent
        a_p = project_simplex(best_overall.copy())
        c_p = full_conv(a_p)
        tb_cd1 = min(remain * 0.4, 60.0)
        a_p2, c_p2, max_p2 = coordinate_descent(a_p, c_p, max_passes=200,
                                                 time_budget=tb_cd1, rng=rng)
        sc_p2 = 2.0 * n_best * max_p2
        if sc_p2 < best_overall_sc:
            best_overall_sc = sc_p2
            best_overall = a_p2.copy()
            print(f"  [final CD] {sc_p2:.6f}")

        # LP polish
        if time.monotonic() < deadline - 10.0:
            tb_lp2 = min(remain * 0.3, 30.0)
            a_p3, sc_p3 = lp_iterate(best_overall, tb_lp2, rng)
            if sc_p3 < best_overall_sc:
                best_overall_sc = sc_p3
                best_overall = a_p3.copy()
                print(f"  [final LP] {sc_p3:.6f}")

        # More CD
        if time.monotonic() < deadline - 5.0:
            a_p4 = project_simplex(best_overall.copy())
            c_p4 = full_conv(a_p4)
            tb_cd2 = min(remain * 0.2, 30.0)
            a_p5, c_p5, max_p5 = coordinate_descent(a_p4, c_p4, max_passes=300,
                                                     time_budget=tb_cd2, rng=rng)
            sc_p5 = 2.0 * n_best * max_p5
            if sc_p5 < best_overall_sc:
                best_overall_sc = sc_p5
                best_overall = a_p5.copy()
                print(f"  [final CD2] {sc_p5:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
