from scipy.optimize import minimize

"""
Algorithm: Multi-scale log-space optimization with tau-annealing and peak equalization.
Minimizes score = 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

_global_n = 0
_global_Nfft = 0
_global_bounds = []

def _init(n):
    global _global_n, _global_Nfft, _global_bounds
    _global_n = n
    _global_Nfft = 1 << (2*n - 1).bit_length()
    _global_bounds = [(-50.0, 50.0)] * n

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _fft_conv(a):
    N = _global_Nfft
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*_global_n-1].real
    return np.maximum(c, 0.0)

def _score(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * _global_n * c.max() / (s*s)

def _obj_grad(z, tau):

    N = _global_Nfft
    L = 2*n - 1
    
    z = np.asarray(z, dtype=np.float64).ravel()

    z = np.clip(z, -50.0, 50.0)

    fa = np.fft.rfft(a, N)

    c = np.maximum(c, 0.0)
    
    m = c.max()
    if not np.isfinite(m):
        return 1e12, np.zeros(n)
    
    # Smooth max via log-sum-exp
    d = (c - m) / tau
    d = np.clip(d, -745.0, 0.0)
    e = np.exp(d)
    se = e.sum()
    if se < 1e-300:
        return 1e12, np.zeros(n)
    p = e / se
    
    # Gradient of smooth max w.r.t. a: p^T (shift-a)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    s_h = h @ a
    grad_a = 2.0 * a * (h - s_h)
    
    # Chain rule: d/dz = grad_a * diag(a) (since d a_i / d z_i = a_i * (delta_ij - a_j))

    # = a_j * (sum_i grad_a_i * a_i - sum_i grad_a_i * a_i * a_j)
    # = a_j * (sum_i grad_a_i * a_i) * (1 - sum_i a_i)  ... no that's wrong
    
    # Let me redo: sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i)

    grad_z = a * (grad_a - s_ga)

    val = m + tau * np.log(se)
    val *= scale
    grad_z *= scale
    
    # Small quadratic penalty to break gauge symmetry
    pen = 1e-10
    val += 0.5 * pen * (z @ z)

def _run_lbfgs(z0, tau_schedule, maxfun=400):
    """Run L-BFGS with a tau schedule."""
    n = _global_n
    current_tau = [tau_schedule[0]]
    
    def wrapped(z):
        # Use current tau
        t = current_tau[0]
        return _obj_grad(z, t)
    
    # Use a custom callback to update tau
    it = [0]
    def callback(zk):
        it[0] += 1
        # Update tau based on iteration count
        frac = min(1.0, it[0] / maxfun)
        idx = min(len(tau_schedule) - 1, int(frac * len(tau_schedule)))
        current_tau[0] = tau_schedule[idx]
        return False
    
    opts = {
        "maxfun": maxfun,
        "maxiter": 200,
        "ftol": 1e-14,
        "pgtol": 1e-10,
        "disp": False,
        "callback": callback,
    }
    
    try:
        res = minimize(wrapped, z0, jac=True, method="L-BFGS-B",
                      bounds=_global_bounds, options=opts)
        return res
    except:
        return None

                        vec[lo:hi+1] = 2.0 * a[ki - np.arange(lo, hi+1)]
                        vec -= 2.0 * c[ki]  # subtract g@a
                        delta_z = a * vec * 0.01  # small step
                        b_try = np.clip(b + delta_z, -50, 50)
                        a_try = _softmax(b_try)
                        F_try = _fft_conv(a_try).max()
                        if F_try < best_F - 1e-15:
                            b = b_try
                            a = a_try
                            improved = True
                            no_improve = 0
                        elif F_try < F - 1e-15:
                            b = b_try
                            a = a_try
                            improved = True
                            no_improve = 0
            if not improved:
                # Random kick
                for scale in [0.01, 0.05, 0.1, 0.3]:
                    b_try = np.clip(b + scale * rng.standard_normal(n), -50, 50)
                    a_try = _softmax(b_try)
                    F_try = _fft_conv(a_try).max()
                    if F_try < best_F - 1e-15:
                        b = b_try
                        a = a_try
                        no_improve = 0
                        break
                else:
                    no_improve += 1
                    if no_improve > 150:
                        break
        else:
            no_improve = 0
        
        it += 1
        if it % 100 == 0:
            print(f"  polish it={it} F={F:.10f} best_F={best_F:.10f} "
                  f"score={2*n*F:.6f} best_score={2*n*best_F:.6f} "
                  f"tau={tau:.2e} t_left={deadline-time.monotonic():.0f}s")
    
    return best_a

    n_candidates = [4096, 8192, 16384]
    
    # Allocate time: give more time to larger n (better potential)
    # But also allow smaller n a chance
    time_alloc = {4096: 0.15, 8192: 0.50, 16384: 0.35}

    overall_best_score = np.inf
