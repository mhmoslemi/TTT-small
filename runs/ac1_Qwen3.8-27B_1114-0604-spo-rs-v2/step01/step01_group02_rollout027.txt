def binomial_init(n):
    """Generate the binomial coefficient sequence C(2j,j)/4^j for j=0,...,n-1."""
    a = np.zeros(n)
    for j in range(n):
        a[j] = math.comb(2*j, j) / (4**j)
    return a

def fast_self_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_lp(a, M, solver):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b = cp.Variable(n)
    
    # Build the convolution constraint matrix
    # conv(a,b)[k] = sum_j a[j]*b[k-j] for valid (j, k-j)
    # This is a linear function of b with coefficients from a.
    
    constraints = [b >= 0]
    for k in range(2*n - 1):
        # conv(a,b)[k] = sum_j a[j]*b[k-j] where 0<=j<n and 0<=k-j<n
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            # sum_{j=j_min}^{j_max} a[j] * b[k-j]
            # = sum_{j=j_min}^{j_max} a[j] * b[k-j]
            # Let i = k-j, then i ranges from k-j_max to k-j_min
            # = sum_{i=k-j_max}^{k-j_min} a[k-i] * b[i]
            coeffs = np.zeros(n)
            for j in range(j_min, j_max + 1):
                i = k - j
                coeffs[i] = a[j]
            constraints.append(coeffs @ b <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if b.value is None:
        return a.copy()  # fallback
    return np.maximum(b.value, 0)

Strategy:
1. Initialize with binomial coefficient sequence (gives score ~ pi/2).

import scipy.sparse as sp
import scipy.signal

def binomial_sequence(n):
    """a[j] = C(2j, j) / 4^j for j = 0, ..., n-1."""
    a = np.ones(n)
    for j in range(1, n):
        a[j] = a[j-1] * (4*j - 2) / (4*j)  # C(2j,j)/4^j = C(2(j-1),j-1)/4^{j-1} * (4j-2)/(4j)
    return a


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv(a, b):
    """Linear convolution of a and b via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    """Build the sparse matrix C such that C @ b = conv(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(k - j)  # b[k-j]
            vals.append(a[j])
    C = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return C


def solve_lp_fw(a, M):
    """Solve the Frank-Wolfe LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)

    C = build_conv_matrix(a)
    constraints = [C @ b <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    # Try solvers in order of preference
    for solver in [cp.GLOP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:

            if b.value is not None:
                return np.maximum(b.value, 0.0)
        except Exception:
            continue

    return a.copy()

    return best_score, best_a


def fw_loop(a, time_limit, tag=""):
    """Run Frank-Wolfe iterations until time_limit. Returns (best_score, best_a)."""
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()
    stagnant = 0
    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        a_new, s = fw_step(a)
        if s < best_score - 1e-10:
            best_score = s
            best_a = a_new.copy()
            stagnant = 0
            if tag:
                print("  [%s] FW: %.6f (n=%d)" % (tag, best_score, n))
        else:
            stagnant += 1
        a = a_new

        if stagnant > 30:
            # Perturb and restart
            scale = 0.05 + 0.1 * np.random.uniform()
            a = best_a * (1.0 + scale * np.random.standard_normal(n))
            a = np.clip(a, 0, 1000)
            stagnant = 0

        if time.monotonic() - start > time_limit:
            break

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 15.0

    best_score = np.inf
    best_a = None

    # Try multiple lengths
    lengths = [512, 1024, 2048, 4096]
    time_per_n = max(30.0, (budget_s - 30.0) / len(lengths))

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        phase_end = min(deadline - 30, start + (lengths.index(n) + 1) * time_per_n)
        remaining = phase_end - time.monotonic()
        if remaining < 20:
            continue

        print("Starting n=%d, time remaining for this phase: %.0fs" % (n, remaining))

        # Binomial initialization
        a = binomial_sequence(n).astype(np.float64)
        a = a / a.sum()
        s0 = compute_score(a)
        print("  Binomial init score: %.6f" % s0)

        # Phase 1: Frank-Wolfe
        fw_time = 0.55 * remaining
        if fw_time > 10:
            s_fw, a_fw = fw_loop(a, fw_time, tag="FW_n%d" % n)
            if s_fw < best_score:
                best_score = s_fw
                best_a = a_fw.copy()

        # Phase 2: Gradient polish
        polish_time = min(0.3 * remaining, deadline - time.monotonic() - 10)
        if polish_time > 5 and best_a is not None:
            s_grad, a_grad = grad_polish(a_fw, polish_time, rng, lr_init=0.03, topk=50)
            if s_grad < best_score:
                best_score = s_grad
                best_a = a_grad.copy()
                print("  Grad polish n=%d: %.6f" % (n, best_score))

        # Phase 3: Short FW burst on best
        burst_end = min(deadline - 10, time.monotonic() + 0.1 * remaining)
        if burst_end - time.monotonic() > 10 and best_a is not None:
            s_burst, a_burst = fw_loop(best_a, burst_end - time.monotonic(), tag="burst_n%d" % n)
            if s_burst < best_score:
                best_score = s_burst
                best_a = a_burst.copy()

    # Final fallback
    if best_a is None:
        best_a = binomial_sequence(512)
        best_score = compute_score(best_a)

    out = [float(x) for x in best_a]
    verified = evaluate_sequence(out)
    print("FINAL: internal=%.6f, verified=%.6f, len=%d, time=%.1fs" %
          (best_score, verified, len(out), time.monotonic() - start))
    return out
