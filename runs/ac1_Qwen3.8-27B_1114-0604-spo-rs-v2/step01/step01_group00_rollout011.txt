def _convolve_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2*n - 2).bit_length()  # need at least 2n-1 points
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _grad_top_k(f, c, k=5):
    """Compute gradient from top-k peaks of convolution c."""
    n = len(f)
    # Softmax weights over c
    T = c.max() * 0.05
    cs = c / T
    cs -= cs.max()
    w = np.exp(cs)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= s
    
    # Take top-k
    idx = np.argsort(w)[::-1][:k]
    wk = w[idx].copy()
    wk /= wk.sum()
    
    grad = np.zeros(n)
    for t in range(k):
        kv = int(idx[t])
        wt = wk[t]
        lo = max(0, kv - n + 1)
        hi = min(kv, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * wt * f[kv - (np.arange(lo, hi+1))]
    
    return grad

def _mirror_step(f, grad, eta):

    f_new = np.exp(log_f)

    if s < 1e-300:
