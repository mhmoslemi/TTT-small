def mirror_descent(p0, deadline, beta0, beta1, max_iter):
    """
    Mirror descent (multiplicative update) in log-space.
    p is maintained with sum=1, p >= 0.
    """
    p = p0.copy().astype(np.float64)
    p = np.maximum(p, 1e-30)
    p /= p.sum()
    n = p.size
    
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_convolution(p)))
    
    eta = 1.0  # step size
    no_improve = 0
    it = 0
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        # Adaptive beta: increase over iterations
        frac = it / max(it, max_iter)
        beta = beta0 + (beta1 - beta0) * frac
        
        b = fft_convolution(p)
        G, w = smooth_max_weights(b, beta)
        cur_smooth = n * G  # This is n * smooth_max(b), proportional to score
        cur_max = float(np.max(b))
        cur_score = 2.0 * n * cur_max
        
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
        
        # Compute gradient
        g = pressure_field(p, w)
        g = np.maximum(g, 0.0)
        
        # Mirror descent update: p_i' = p_i * exp(-eta * g_i) / Z
        # For numerical stability, subtract max
        exponents = -eta * g
        exponents -= np.max(exponents)  # shift for stability
        weights = np.exp(exponents)
        Z = float(np.sum(p * weights))
        
        if Z <= 0 or not np.isfinite(Z):
            break
        
        p_new = p * weights / Z
        p_new = np.maximum(p_new, 1e-30)
        p_new /= p_new.sum()
        
        # Check if this is an improvement (using smooth max as proxy)
        b_new = fft_convolution(p_new)
        G_new, _ = smooth_max_weights(b_new, beta)
        
        if n * G_new < cur_smooth:
            p = p_new
            eta = min(eta * 1.1, 50.0)  # increase step size
        else:

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-phase optimization.

Algorithm:
1. Block-parameterized L-BFGS for global structure (piecewise-constant sequences)
2. Mirror descent (multiplicative updates) in log-space for fine optimization
3. Adaptive beta schedule for smooth-max approximation of the max
4. Multiple diverse starting points (raised cosine, Gaussian, two-bump, random)

Key properties:
- Scale-invariant objective, normalized to sum=1
- FFT-based convolution for speed
- Mirror descent naturally enforces a >= 0
- Backtracking step size with momentum
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Return (smooth_max_value, softmax_weights) for b with temperature beta."""
    b = np.asarray(b, dtype=np.float64)
    m = b.size
    if m == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    mx = float(np.max(b))
    if mx <= 0.0:
        w = np.full(m, 1.0 / m, dtype=np.float64)
        return mx, w
    x = beta * (b - mx)
    x = np.clip(x, -50.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros(m, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = mx + float(np.log(Z)) / beta
    return G, w


def pressure_field(p, w):
    """
    Compute g_i = sum_{j=0}^{n-1} p_j * w_{i+j} for i=0..n-1.
    This is the gradient of smooth_max(a*a) with respect to a_i.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size  # should be 2n-1
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    
    # g_i = C[m-1-i] for i=0..n-1
    # So g = C[m-n : m][::-1]
    g = C[m - n : m][::-1].copy()
    return np.ascontiguousarray(g, dtype=np.float64)


def fast_score(p):
    """Compute 2*n*max(a*a)/(sum(a))^2 for normalized p (sum=1)."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p / s

def make_starts(n, rng, count=10):
    """Generate diverse starting points."""
    starts = []
    
    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))
    
    # Raised cosine (Hann-like)
    x = np.arange(n, dtype=np.float64)
    p = np.sin(np.pi * (x + 0.5) / n) ** 2
    starts.append(normalize(p))
    
    # Various Gaussians
    center = 0.5 * (n - 1)
    for sigma_frac in (0.15, 0.25, 0.35, 0.5):
        sigma = sigma_frac * n
        p = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        starts.append(normalize(p))
    
    # Two-bump
    if n >= 64:
        for frac in (0.2, 0.25, 0.3):
            c1 = frac * (n - 1)
            c2 = (1.0 - frac) * (n - 1)
            sigma = 0.12 * n
            p = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
            starts.append(normalize(p))
    
    # Linear taper
    for a in (0.3, 0.7):
        x_norm = x / (n - 1)
        p = (a + (1 - 2*a) * x_norm) if a < 0.5 else (1 - a - (1 - 2*a) * x_norm)
        p = np.maximum(p, 1e-6)
        starts.append(normalize(p))
    
    # Random low-pass
    for k in range(2):
        zc = np.zeros(n, dtype=complex)
        nfreq = n // 4
        for f in range(nfreq):
            zc[f] = rng.normal() + 1j * rng.normal()
            if f < n - 1:
                zc[n - f] = np.conj(zc[f])
        z = np.real(np.fft.ifft(zc))
        p = np.exp(2.0 * z)
        starts.append(normalize(p))
    
    # Truncation pattern (high in middle, low at edges)
    p = np.ones(n, dtype=np.float64)
    edge = n // 6
    p[:edge] = 0.3
    p[-edge:] = 0.3
    starts.append(normalize(p))
    
    return starts[:count]


def mirror_descent(p0, deadline, beta_schedule, max_iter=10000000, verbose=True):
    """
    Mirror descent (multiplicative update) for minimizing max(a*a) on the simplex.
    
    p0: initial point (will be normalized)
    deadline: time limit
    beta_schedule: function of iteration fraction returning beta
    """
    p = normalize(p0)
    n = p.size
    
    if n <= 1:
        return p, fast_score(p)
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    eta = 0.5  # step size
    momentum = 0.7
    v = np.zeros(n, dtype=np.float64)  # momentum buffer
    no_improve = 0
    it = 0
    last_print = 0
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        frac = min(1.0, it / 50000.0)
        beta = beta_schedule(frac)
        
        b = fft_convolution(p)
        G, w = smooth_max_weights(b, beta)
        cur_smooth = float(n) * G
        cur_max = float(np.max(b))
        cur_score = float(2.0 * n * cur_max)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it - last_print > 5000:
            if verbose:
                print(f"  MD it={it} beta={beta:.0f} score={cur_score:.8f} best={best_score:.8f} eta={eta:.4f} noimp={no_improve}")
            last_print = it
        
        if no_improve > 10000:
            # Try to escape: reset momentum, reduce step
            v = np.zeros(n, dtype=np.float64)
            eta = max(eta * 0.1, 0.01)
            no_improve = 0
            if no_improve > 20000:
                break
        
        # Compute gradient
        g = pressure_field(p, w)
        g = np.maximum(g, 0.0)
        
        # Momentum
        v = momentum *
