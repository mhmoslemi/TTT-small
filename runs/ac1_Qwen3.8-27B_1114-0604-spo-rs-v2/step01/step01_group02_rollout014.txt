```python
def spectral_project(a, target_mag, N):
    a_padded = np.zeros(N)
    a_padded[:len(a)] = a
    A = np.fft.rfft(a_padded)
    mag = np.abs(A)
    # Avoid division by zero
    safe_mag = np.where(mag > 1e-15, mag, 1e-15)
    phase = A / safe_mag
    # New magnitude: blend current with target
    alpha = 0.5  # blending parameter
    new_mag = (mag ** (1-alpha)) * (target_mag ** alpha)
    new_A = new_mag * phase
    a_new = np.fft.irfft(new_A, N)[:len(a)]
    return np.maximum(a_new, 0)
```

import numpy as np
import time

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    N = 1 << (2*n-2).bit_length() if 2*n > 2 else 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A*A, N)[:2*n-1]

def compute_target_mag(n, N):
    """Target |FFT| magnitude for flat linear autocorrelation."""
    m = np.arange(N // 2 + 1)
    with np.errstate(divide='ignore', invalid='ignore'):
        mag = np.abs(np.sin(np.pi * m * n / N) / np.sin(np.pi * m / N))
    mag = mag / n
    # Handle m=0: limit is n/n = 1
    mag[0] = 1.0  # Actually lim_{m->0} |sin(pi*m*n/N)/sin(pi*m/N)| = n, so /n = 1
    return mag

def compute_gradient(a, w, N_grad):
    """Compute gradient of sum_k w[k]*conv_k(a) w.r.t. a, via FFT cross-correlation."""
    n = len(a)
    a_flipped = a[::-1]
    F_w = np.fft.rfft(w, N_grad)
    F_a = np.fft.rfft(a_flipped, N_grad)
    conv = np.fft.irfft(F_w * F_a, N_grad)
    xc = conv[n-1:2*n-1]
    return 2.0 * xc

def spectral_project(a, target_mag, N, alpha=0.5):
    """Project a toward target spectrum."""
    a_padded = np.zeros(N)
    a_padded[:len(a)] = a
    A = np.fft.rfft(a_padded)
    mag = np.abs(A)
    safe = np.where(mag > 1e-15, mag, 1e-15)
    phase = A / safe
    new_mag = np.power(mag, 1.0 - alpha) * np.power(target_mag, alpha)
    new_A = new_mag * phase
    a_new = np.fft.irfft(new_A, N)[:len(a)]
    return np.maximum(a_new, 0.0)

def softmax_weights(c, tau):
    """Compute soft-max weights over convolution values."""
    c_max = np.max(c)
    if tau < 1e-15:
        w = np.zeros_like(c)
        w[c == c_max] = 1.0
        w[c == c_max] /= np.sum(w[c == c_max])
        return w
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    s = np.sum(w)
    if s > 0:
        w /= s
    else:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    return w

        S = a.sum()
        score = 2.0 * n * max_c / (S * S) if S > 1e-10 else np.inf
        
        # Update best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve >= patience_current:
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.clip(a, 0, None)
            S = a.sum()
            if S > 1e-10:

            else:
                a = best_a.copy()
            np.random... # use rng
            no_improve = 0

            lr = max(0.001, lr * 0.5)
            continue
        
        # Soft-max weights

`= (1/n) * exp(-πi*m*(n-1)/N) * sin(π*m*n/N) / sin(π*m/N)`

So `|A[m]|^2 = (1/n^2) * sin^2(π*m*n/N) / sin^2(π*m/N)`

And `|A[m]| = (1/n) * |sin(π*m*n/N) / sin(π*m/N)|`

At m=0: `sin(0)/sin(0)` is 0/0. The limit is n (by L'Hôpital or the geometric series: sum of n ones = n, times 1/n = 1). So `|A[0]| = 1`.

Wait: `|A[0]|^2 = (1/n) * sum_{k=0}^{n-1} 1 = 1`. So `|A[0]| = 1`. And the formula gives `(1/n) * |sin(0)/sin(0)|`. The limit of `sin(π*m*n/N)/sin(π*m/N)` as m→0 is n. So `(1/n) * n = 1`. Correct.

Now, the target magnitude should be real and non-negative (it's a magnitude). The formula gives `|A[m]| = (1/n) * |sin(π*m*n/N) / sin(π*m/N)|`. This is always non-negative due to the absolute value. Good.

One issue: when `sin(π*m/N) = 0` (i.e., m is a multiple of N), we have a 0/0 situation. This only happens at m=0 (since m ranges from 0 to N/2 and N/2 < N for N > 1), which we handle separately.

Another issue: when `sin(π*m*n/N) = 0` but `sin(π*m/N) ≠ 0`, the target magnitude is 0. This happens when `m*n` is a multiple of N but m is not. For example, if N = 2n (which is what we get when n is a power of 2), then `sin(π*m*2n/(2n)) = sin(π*m) = 0` for all integer m. So the target magnitude is 0 for all m > 0!

That can't be right. Let me reconsider. If N = 2n, the target autocorrelation is `r[k] = 1/n` for k=0,...,n-1 and `r[k] = 0` for k=n,...,2n-1`.

`|A[m]|^2 = (1/N) * sum_{k=0}^{2n-1} r[k] * exp(-2πi*m*k/(2n))`

Wait, I need to be careful about the DFT convention. Let me use:
`A[m] = sum_{k=0}^{N-1} r[k] * exp(-2πi*m*k/N)` (no 1/N factor, for the non-unitary DFT)

Then `|A[m]|^2` is not what I want. What I want is the DFT of the autocorrelation, which directly gives the power spectrum.

Actually, the relation is: if `x` is a signal, its circular autocorrelation is `r[k] = sum_j x[j]*x[j+k]` (circular), and `DFT(r) = |DFT(x)|^2`.

So if I want `r_target[k]` to be the circular autocorrelation, then `|A_target[m]|^2 = DFT(r_target)[m]`.

With `r_target[k] = 1/n` for k=0,...,n-1 and 0 otherwise (in a period-N sequence):

`DFT(r_target)[m] = sum_{k=0}^{n-1} (1/n) * exp(-2πi*m*k/N) = (1/n) * sin(π*m*n/N) / sin(π*m/N) * exp(-πi*m*(n-1)/N)`

The magnitude is: `|DFT(r_target)[m]| = (1/n) * |sin(π*m*n/N) / sin(π*m/N)|`

And `|A_target[m]| = sqrt(|DFT(r_target)[m]|)`... 

Wait no. If `r` is the circular autocorrelation of `x`, then `DFT(r) = |DFT(x)|^2`. So `|DFT(x)| = sqrt(DFT(r))` (assuming DFT(r) is non-negative, which it is for a valid autocorrelation).

But is `r_target` a valid autocorrelation? A valid autocorrelation must have `DFT(r) >= 0` for all m. Let's check:

`DFT(r_target)[m] = (1/n) * sum_{k=0}^{n-1} exp(-2πi*m*k/N)`

This is a complex number in general (it has a phase factor). Its magnitude is `(1/n) * |sin(π*m*n/N) / sin(π*m/N)|` which is non-negative. But the value itself is complex (has a phase). So `DFT(r_target)[m]` is not necessarily non-negative (as a real number). It's complex!

This means `r_target` is NOT a valid circular autocorrelation (because the DFT of a valid autocorrelation must be real and non-negative).

So we can't directly use `r_target` as an autocorrelation. The "target spectrum" approach needs to be modified.

What we can do instead:
- Use the magnitude `|DFT(r_target)[m]|` as a "guiding" target for the power spectrum.
- The actual power spectrum `|A[m]|^2` should be real and non-negative. We want it to be "close" to `|DFT(r_target)[m]|` in some sense.

Alternatively, we can think of it differently: we want the power spectrum `P[m] = |A[m]|^2` such that `ifft(P)` has its energy concentrated in the first n lags (with the rest as small as possible). This is equivalent to making `ifft(P)[0..n-1]` as uniform as possible while `ifft(P)[n..N-1]` is as close to 0 as possible.

This is a different problem from what I had before. Let me think about it as an optimization:

Minimize `sum_{k=n}^{N-1} (ifft(P)[k])^2` subject to `P[m] >= 0` and `sum(P) = P[0] = (sum a)^2 = 1` (if normalized).

This is a quadratically constrained optimization. The solution is to "project" the ideal P onto the feasible set.

But this is getting too complex. Let me simplify:

**Simplified spectral projection**: Instead of a hard target, use a softer approach:
- Compute the current power spectrum `P = |FFT(a_padded)|^2`
- Compute `r = IFFT(P)` (this is the circular autocorrelation of the padded sequence)
- The "leakage" is the energy in `r[n..N-1]` (which should be 0 ideally)
- Modify P to reduce the leakage: e.g., `P_new = P * (1 - beta * r[n..N-1] mapped back to frequency domain)`

This is getting complicated. Let me instead use a simpler spectral modification:

**Simple spectral flattening**: 
- Compute `A = FFT(a_padded)`
- The power spectrum is `|A|^2`
- To make the autocorrelation flatter, we want `|A|^2` to be flatter
- Apply a "compressor": `|A_new| = |A|^{1/(1+beta)}` for some beta > 0 (this compresses the dynamic range)
- Reconstruct

This is a simple and effective heuristic. Let me use this instead.

Actually, I realize there's an even simpler and more principled approach. Let me think about it from the perspective of optimization.

The problem is: min `max_k conv_k(a)` subject to `a >= 0, sum(a) = 1`.

This is equivalent to: min `t` subject to `conv_k(a) <= t` for all k, `a >= 0`, `sum(a) = 1`.

For a given `t`, the feasible set is defined by quadratic inequalities. We can use a bisection on `t` combined with a feasibility check.

But the feasibility check (does there exist `a >= 0, sum(a)=1` with `conv_k(a) <= t` for all k?) is a quadratic feasibility problem, which is NP-hard in general.

However, we can use a relaxation: instead of checking exact feasibility, we can try to find a feasible point using iterative methods (like projected gradient or alternating projections).

**Alternating Projections Approach:**
1. Start with `a >= 0, sum(a) = 1`
2. For each constraint `conv_k(a) <= t`:
   - If `conv_k(a) > t`, project `a` to satisfy `conv_k(a) = t`
   - The projection onto `conv_k(a) <= t` is solved by minimizing `||a - a_projected||^2` subject to `conv_k(a_projected) = t` (if the current point violates) and `a_projected >= 0`
3. Also project onto `sum(a) = 1` and `a >= 0`
4. Iterate

The projection onto `conv_k(a) = t` (with `a >= 0`) is a quadratic optimization:
min `||a - b||^2` subject to `sum_j a[j]*a[k-j] = t` and `a >= 0`

This is a QCQP (quadratically constrained quadratic program) with one quadratic equality constraint. It can be solved in closed form using Lagrange multipliers (if the non-negativity constraints are inactive).

Without non-negativity: 
min `||a - b||^2` s.t. `a^T Q_k a = t` where `Q_k` is the matrix representing the quadratic form `conv_k(a)`.

The solution is `a = b - lambda * Q_k * b` for some lambda chosen so that `a^T Q_k a = t`.

With non-negativity, it's more complex (active set method).

This approach is interesting but complex to implement. Let me stick with the gradient-based approach but make it more efficient.

**Final Algorithm Design:**

1. Use FFT for both convolution and gradient computation (O(N log N) per iteration).
2. Soft-max annealing: start with high temperature (smooth), decrease over time.
3. Spectral regularization: periodically apply a magnitude compression to the FFT.
4. Multiple lengths (4096, 8192, 16384).
5. Multiple restarts with different initializations.
6. The parent sequence is used as one of the initializations.

Key improvements over the previous code:
- FFT-based gradient (O(N log N) instead of O(n * n) = O(n^2) in the worst case)
- Spectral regularization step
- Better annealing schedule
- Multiple sequence lengths

- Gradient: 3 FFTs of size N_grad (2 forward + 1 inverse) but N_grad ≈ 2*N (since we need 3n vs 2n), so similar cost
- Spectral projection: 2 FFTs of size N
- Total per iteration: maybe 7-8 FFTs of size ~N

For n = 8192, N = 16384, each FFT takes about 0.1-0.5 ms on a modern CPU. So each iteration is about 1-4 ms. In 2400 seconds, we can do about 600,000 - 2,400,000 iterations. That's plenty.

For n = 16384, N = 32768, each FFT takes about 0.3-1 ms. Each iteration is about 3-8 ms. We can do about 300,000 - 800,000 iterations.

Let me also think about the learning rate schedule. The objective is `2n * max_c / S^2`. With `S = 1` (normalized), it's `2n * max_c`. The gradient magnitude is proportional to `2n * |a|` (roughly). So the learning rate should be on the order of `1 / (2n * max|a|)` to make reasonable progress. With `a` normalized to sum 1, `max|a|` is at most 1 (when one element is 1 and the rest are 0). Typically `max|a| ≈ 1/sqrt(n)` for a "spread out" sequence. So the gradient is about `2n * 1/sqrt(n) = 2*sqrt(n)`. The learning rate should be about `1/(2*sqrt(n))`. But this is a rough estimate. I'll use a learning rate schedule that starts a bit higher and decays.

import numpy as np
import time

def fast_self_conv(a):
    n = len(a)
    N = 1 << (2*n-2).bit_length() if 2*n > 2 else 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A*A, N)[:2*n-1]

def compute_gradient(a, w, N_grad):
    """2 * cross_correlate(w, a) via FFT."""
    n = len(a)
    a_flipped = a[::-1]
    F_w = np.fft.rfft(w, N_grad)
    F_a = np.fft.rfft(a_flipped, N_grad)
    result = np.fft.irfft(F_w * F_a, N_grad)
    return 2.0 * result[n-1:n-1+n]

def softmax_weights(c, tau):
    c_max = np.max(c)
    if tau < 1e-18:
        idx = np.argmax(c)
        # All mass on the max
        n_max = np.sum(c == c_max)
        w = np.zeros_like(c)
        w[c == c_max] = 1.0 / n_max
        return w
    w = np.exp((c - c_max) / tau)
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w
    w /= s
    return w

def spectral_compress(a, N, beta=0.3):
    """Compress the dynamic range of the power spectrum."""
    a_padded = np.zeros(N)
    a_padded[:len(a)] = a
    A = np.fft.rfft(a_padded)
    mag = np.abs(A)
    safe = np.where(mag > 1e-15, mag, 1e-15)
    phase = A / safe
    # Compress: new_mag = mag^(1/(1+beta)) which compresses peaks
    new_mag = np.power(mag, 1.0 / (1.0 + beta))
    new_mag = new_mag / np.max(new_mag) * np.max(mag)  # preserve total energy roughly
    new_A = new_mag * phase
    a_new = np.fft.irfft(new_A, N)[:len(a)]
    return np.maximum(a_new, 0.0)

def adam_step(a, grad, m, v, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)

    t_step = 0
    
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    patience_current = 200
    
    # Annealing parameters
    tau_start = fast_self_conv(a).max() * 0.5
    tau_min = fast_self_conv(a).max() * 0.01
    tau = tau_start
    lr = 0.05
    lr_decay = 0.9995
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < max_time:
        # Compute convolution
        c = fast_self_conv(a)
        max_c = float(np.max(c))
        S = a.sum()
        score = 2.0 * n * max_c / (S * S) if S > 1e-10 else np.inf
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:

        # Restart
        if no_improve >= patience_current:
            a = best_a * (1.0 + 0.03 * rng.standard_normal(n))

            S = a.sum()
            if S > 1e-10:

        grad_c = compute_gradient(a, w, N_grad)

        Wc = float(np.dot(w, c))
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (grad_c / S2 - 2.0 * Wc * a / S3)
        
        # Adam step

                    a /= S
        
        # Anneal temperature
        tau = max(tau_min, tau * 0.998)
        # Slowly decay learning rate
        lr *= lr_decay
