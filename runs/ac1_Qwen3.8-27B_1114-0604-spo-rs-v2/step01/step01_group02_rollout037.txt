def h_topk(p,y,k,tau):
    n=p.size; L=y.size
    raw=np.zeros(n,float)
    if k<=0: return raw
    if k>=L:
        return 1.0/L * np.ones(n) # constant no gradient; but not used
    idx=np.argpartition(y, -k)[-k:]
    vals=y[idx]
    order=np.argsort(vals)[::-1]
    idx=idx[order]
    top=float(vals[0])
    # if top very small: uniform topk
    if top <= 1e-15:

"""
Search for low-peak non-negative step functions.

Minimize:
    2*n*max_k conv(a,a)_k / (sum a)^2
by normalizing sum(a)=1 and then minimizing 2*n*max_k conv(a,a)_k.

The optimizer combines:
  * log-sum-exp / multiplicative projected gradient steps on the simplex,
  * top-k and full-softmax approximations of the max-convolution,
  * inverse-curvature candidates,
  * periodic linear-programming directions as in the referenced state-of-art,
  * symmetry and restarts.
"""

import numpy as np
import time
import math
from operator import itemgetter

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


# -------------------------------------------------------------------------
# Basic normalization and scoring
# -------------------------------------------------------------------------

def positive_normalize(a, min_eps=1e-8):
    """Clip to [0, 1000], make all entries strictly positive, normalize sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return None
    floor = min_eps / float(a.size)
    a = np.maximum(a, floor)
    s = float(a.sum())
    if s <= 0.0:
        return None
    return a / s


def fast_self_conv(a):
    """Linear self-convolution using real FFT."""
    n = a.size
    out_len = 2 * n - 1
    N = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:out_len]
    return out


def fast_score(p):
    """Score assuming sum(p)=1: 2*n*max(conv(p,p))."""
    n = p.size
    if n == 0:
        return np.inf
    c = fast_self_conv(p)
    c = np.maximum(c, 0.0)
    return 2.0 * float(n) * float(np.max(c))


def score_and_y(p):
    """Return score and y = n*conv(p,p)."""
    n = p.size
    c = fast_self_conv(p)
    c = np.maximum(c, 0.0)
    score = 2.0 * float(n) * float(np.max(c))
    y = float(n) * c
    return score, y


def fft_conv(a, b):
    """Linear convolution of two 1D arrays using real FFT."""
    out_len = a.size + b.size -1
    N = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    out = np.fft.irfft(A * B, N)[:out_len]
    return out


# -------------------------------------------------------------------------
# Gradient / field construction
# -------------------------------------------------------------------------

def h_from_indices(p, y, indices, tau):
    """
    For peak weights q_k supported at indices, compute
    raw[j] = sum_k q_k p_{k-j}.
    This is proportional to the derivative of the weighted peak objective.
    """
    n = p.size
    raw = np.zeros(n, dtype=np.float64)
    if indices.size == 0:
        return raw

    vals = y[indices]
    k = vals.size
    top = float(np.max(vals))
    if top <= 1e-15:
        w = np.ones(k, dtype=np.float64) / float(k)
    else:
        w = np.exp((vals - top) / max(float(tau), 1e-12))
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(k, dtype=np.float64) / float(k)
        else:
            w = w / s

    for ii in range(k):
        wi = float(w[ii])
        if wi <= 1e-16:
            continue
        ki = int(indices[ii])
        j0 = max(0, ki - n + 1)
        j1 = min(ki, n - 1)
        if j0 <= j1:
            raw[j0:j1 + 1] += wi * p[j0:j1 + 1][::-1]

    return raw


def h_topk(p, y, k, tau):
    """Peak-weighted gradient field using the k largest y-values."""
    n = p.size
    L = y.size
    if k <= 0:
        return np.zeros(n, dtype=np.float64)
    k = int(min(k, L))
    if k >= L:
        return np.ones(n, dtype=np.float64) / float(L)
    idx = np.argpartition(y, -k)[-k:]
    return h_from_indices(p, y, idx, tau)


def h_softmax(p, y, tau):
    """Dense softmax approximation of the max, using FFT convolution."""
    n = p.size
    L = y.size
    m = float(np.max(y)) if y.size else 0.0
    if m <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(max(1, L))
    logits = (y - m) / max(float(tau), 1e-12)
    q = np.exp(np.clip(logits, -700.0, 0.0))
    s = float(q.sum())
    if s <= 0.0:
        q = np.ones(L, dtype=np.float64) / float(L)
    else:
        q = q / s
    out = fft_conv(q, p[::-1])
    raw = out[n - 1:2 * n - 1]
    return np.maximum(raw, 0.0)


def sample_active_indices(y, k, tau, rng):
    """Sample active peak indices stochastically, always including the global maximum."""
    L = y.size
    k = int(min(k, L))
    if k <= 0:
        return np.array([], dtype=int)
    top = int(np.argmax(y))
    if k == 1:
        return np.array([top], dtype=int)

    m = float(np.max(y))
    logits = (y - m) / max(float(tau), 1e-12)
    probs = np.exp(np.clip(logits, -700.0, 0.0))
    s = probs.sum()
    if s <= 0.0:
        probs = np.ones(L, dtype=np.float64) / float(L)
    else:
        probs = probs / s

    idxs = rng.choice(L, size=k - 1, replace=False, p=probs)
    idxs = np.append(idxs, top)
    return idxs.astype(int)


def prepare_gradient(p, raw):
    """Convert raw peak field into a centered, scale-normalized gradient vector."""
    n = p.size
    if raw.size != n:
        return np.zeros(n, dtype=np.float64)
    g = float(n) * raw
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=np.float64)
    mean = float(np.dot(p, g))
    g = g - mean
    var = float(np.dot(p, g * g))
    sd = math.sqrt(var) if var > 0.0 else 0.0
    if sd > 1e-15:
        g = g / sd
    return np.clip(g, -8.0, 8.0)

def multiplicative_candidates(p, g, base_lr, current_score, lrs=None):
    """Try a few multiplicative log-field updates and return the best improving one."""
    if lrs is None:
        lrs = (float(base_lr), float(base_lr) * 0.35)

    best_p = None
    best_sc = np.inf

    if not np.all(np.isfinite(g)):
        return None, np.inf

    for lr in lrs:
        if lr <= 0.0 or not np.isfinite(lr):
            continue
        ratio = np.exp(-lr * g)
        p_try = p * ratio
        s = float(p_try.sum())
        if s <= 0.0 or not np.isfinite(s):
            continue
        p_try = p_try / s
        sc = fast_score(p_try)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p_try

    if best_p is not None and best_sc < current_score - 1e-12:
        return best_p, best_sc
    return None, np.inf


def mix_candidates(p, b, current_score, t_values):
    """Try convex mixes (1-t)p + t*b and return the best improving one."""
    if b is None:
        return None, np.inf
    best_p = None
    best_sc = np.inf

    for tt in t_values:
        t = float(tt)
        if t <= 0.0 or not np.isfinite(t):
            continue
        p_try = (1.0 - t) * p + t * b
        s = float(p_try.sum())
        if s <= 0.0 or not np.isfinite(s):
            continue
        p_try = p_try / s
        sc = fast_score(p_try)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p_try

    if best_p is not None and best_sc < current_score - 1e-12:
        return best_p, best_sc
    return None, np.inf


def inverse_candidate(p, raw):
    """Candidate b ~ 1/(n*raw+floor), with capping and normalization."""
    n = p.size
    rho = float(n) * raw
    rm = float(np.max(rho))
    if not np.isfinite(rm) or rm <= 1e-15:
        return None

    mean = float(np.mean(rho))
    floor = max(1e-10, 0.01 * mean)
    b = 1.0 / (rho + floor)
    cap = 200.0 / float(n)

    if b.max() > cap:
        b = np.minimum(b, cap)
    s = float(b.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    b = b / s
    return positive_normalize(b)


def lp_direction(p, y, k_active=32, rng=None):
    """
    LP direction:
        max sum(b)
        s.t. (p*b)_k <= max(p*p)_k for active k,
             b >= 0.
    Active set includes top peaks, the center (for coverage), and optional sampled peaks.
    The returned b is normalized to sum 1.
    """
    if linprog is None or p.size == 0:
        return None

    n = p.size
    L = y.size
    mx = float(np.max(y))
    if not np.isfinite(mx) or mx <= 0.0:
        return None

    Mc = mx / float(n)

    k_top = int(max(1, min(k_active, L)))
    idx_top = np.argpartition(y, -k_top)[-k_top:]
    idx_top = idx_top[np.argsort(y[idx_top])[::-1]]

    idx_list = [int(x) for x in idx_top]
    central = n - 1
    if int(central) not in idx_list:
        idx_list.append(int(central))

    k_extra = int(max(0, min(8, k_active // 4)))
    if rng is not None and k_extra > 0:
        sampled = sample_active_indices(y, k_extra, 0.18, rng)
        for si in sampled:
            si = int(si)
            if si not in idx_list:
                idx_list.append(si)

    # Deduplicate while preserving order.
    seen = set()
    idx_unique = []
    for x in idx_list:
        if x not in seen:
            seen.add(x)
            idx_unique.append(x)
    idx_list = idx_unique
    K = len(idx_list)

    A = np.zeros((K, n), dtype=np.float64)
    for r, ki in enumerate(idx_list):
        j0 = max(0, ki - n + 1)
        j1 = min(ki, n - 1)
        if j0 <= j1:
            A[r, j0:j1 + 1] = p[j0:j1 + 1][::-1]

    col_sums = A.sum(axis=0)
    if np.any(col_sums <= 1e-300):
        # The central row should make this true for positive p; if not, LP is not well-posed here.
        return None

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    for slack in (1.0 + 1e-7, 1.0 + 1e-5, 1.0 + 1e-3):
        b_ub = np.full(K, Mc * slack, dtype=np.float64)
        t0 = time.monotonic()
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": 3.0},
            )
        except Exception:
            try:
                res = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                )
            except Exception:
                return None

        if res.success and hasattr(res, "x") and res.x is not None:
            b = np.asarray(res.x, dtype=np.float64)
            if np.all(np.isfinite(b)):
                b = np.maximum(b, 0.0)
                s = float(b.sum())
                if s > 1e-12:
                    b = b / s
                    cap = 200.0 / float(n)
                    if b.max() > cap:
                        b = np.minimum(b, cap)
                        s = float(b.sum())
                        if s > 0.0:
                            b = b / s
                    return positive_normalize(b)

        # If the first solve hit the time limit, do not retry larger slacks.
        if not res.success and (time.monotonic() - t0) > 2.5:
            break

    return None


def smooth_random_init(n, rng):
    """Low-frequency random smooth positive sequence."""
    if n <= 0:
        return None
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    sig = np.zeros(n, dtype=np.float64)
    max_f = min(32, max(2, n // 2))
    for f in range(1, max_f + 1):
        amp = rng.uniform(0.0, 1.0) / (f + 1.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        sig += amp * np.cos(2.0 * math.pi * f * t + phase)
    p = np.exp(0.7 * sig)
    s = p.sum()
    if s <= 0.0:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = p / s
    return positive_normalize(p)


def cosine_init(n, freq, alpha, rng):
    """Single cosine modulation."""
    if n <= 0:
        return None
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    phase = rng.uniform(0.0, 2.0 * math.pi)
    p = np.exp(float(alpha) * np.cos(2.0 * math.pi * float(freq) * t + phase))
    s = p.sum()
    if s <= 0.0:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = p / s
    return positive_normalize(p)


def resize_average(p, target):
    """Resize a sequence by mass-preserving linear interpolation."""
    p = positive_normalize(p)
    if p is None:
        return None
    target = int(target)
    if target <= 0:
        return None
    n = p.size
    if target == n:
        return p.copy()

    cum = np.concatenate((np.zeros(1, dtype=np.float64), np.cumsum(p, dtype=np.float64)))
    x_old = np.arange(n + 1, dtype=np.float64)
    x_new = (np.arange(target, dtype=np.float64) + 0.5) * (float(n) / float(target))
    if x_new.max() > x_old[-1]:
        x_new = np.clip(x_new, x_old[0], x_old[-1])
    y_new = np.interp(x_new, x_old, cum)
    q = np.diff(y_new)
    q = np.maximum(q, 0.0)
    return positive_normalize(q)


def downsample(p, factor):
    p = positive_normalize(p)
    if p is None:
        return None
    target = max(1, p.size // max(1, int(factor)))
    return resize_average(p, target)


def symmetrize(p):
    p = positive_normalize(p)
    if p is None:
        return None
    q = 0.5 * (p + p[::-1])
    return positive_normalize(q)


def perturb_log(best, sigma, rng):
    """Log-normal perturbation in log domain."""
    best = positive_normalize(best)
    if best is None:
        return None
    x = np.log(np.maximum(best, 1e-300))
    x = x - x.max()
    x = x + float(sigma) * rng.standard_normal(best.size)
    x = np.clip(x, -25.0, 0.0)
    p = np.exp(x)
    s = p.sum()
    if s <= 0.0:
        return best.copy()
    return positive_normalize(p)

# Main local search

def local_search(p0, n, end_time, rng, tag, verbose=False):
    """Local refiner with gradient, inverse, symmetry, restarts, and LP moves."""
    start = time.monotonic()
    p = positive_normalize(p0)
    if p is None or p.size != int(n):
        p = np.ones(max(1, int(n)), dtype=np.float64)
    n = p.size

    best_p = p.copy()
    best_score = fast_score(p)
    cur_score = best_score

    it = 0
    stuck = 0
    last_lp = start - 1.0e9
    total = max(1e-6, end_time - start)

    while time.monotonic() < end_time - 0.05:
        now = time.monotonic()
        frac = min(1.0, max(0.0, (now - start) / total))
        remaining = end_time - now

        score, y = score_and_y(p)
        if not np.isfinite(score):
            p_new = positive_normalize(best_p)
            if p_new is None:
                p_new = np.ones(n, dtype=np.float64) / float(n)
            p = p_new
            score = best_score
            cur_score = best_score
            c = fast_self_conv(p)
            y = float(n) * np.maximum(c, 0.0)
        else:
            cur_score = score

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            stuck = 0
        else:
            stuck += 1

        # Occasional symmetry improvement.
        if it % 300 == 150:
            sym = symmetrize(best_p)
            if sym is not None:
                sc_sym = fast_score(sym)
                if np.isfinite(sc_sym) and sc_sym < best_score - 1e-12:
                    best_score = sc_sym
                    best_p = sym.copy()
                    if cur_score > best_score:
                        p = best_p.copy()
                        cur_score = best_score
                    stuck = 0

        # Do not wander too far from the current best.
        if cur_score > best_score + 0.02:
            p = best_p.copy()
            cur_score = best_score
            stuck = 0

        # ----- Linear-programming move -----
        if n <= 6000 and remaining > 8.0:
            if frac < 0.25:
                lp_interval = 15.0
            elif frac < 0.60:
                lp_interval = 35.0
            else:
                lp_interval = 70.0

            if (now - last_lp >= lp_interval) or (stuck > 120 and remaining > 15.0):
                t0 = time.monotonic()
                k_lp = int(max(16, min(64, n // 40)))
                if stuck > 100:
                    k_lp = int(max(k_lp, min(80, k_lp + 10)))

                b = lp_direction(p, y, k_active=k_lp, rng=rng)
                dt = time.monotonic() - t0
                last_lp = now

                if b is not None:
                    p_try, sc = mix_candidates(
                        p, b, cur_score,
                        t_values=(1.0, 0.7, 0.5, 0.3, 0.15, 0.05),
                    )
                    if p_try is not None:
                        p_new = positive_normalize(p_try)
                        if p_new is None:
                            p_new = best_p.copy()
                        p = p_new
                        cur_score = sc
                        if cur_score < best_score - 1e-12:
                            best_score = cur_score
                            best_p = p.copy()
                        stuck = 0
                        it += 1
                        if verbose and it % 5000 == 0:
                            print("[%s] LP it=%d best=%.6f cur=%.6f" %
                                  (tag, it, best_score, cur_score))
                        continue

                if dt > 5.0:
                    # Slow LP: rely more on cheap moves by advancing last_lp.
                    last_lp = now

        # ----- Cheap gradient / multiplicative move -----
        if stuck > 60 and (it % 60 == 30):
            raw = h_softmax(p, y, 0.12)
        elif stuck > 30 and (it % 9 == 5):
            k_rand = int(max(16, min(64, n // 50)))
            inds = sample_active_indices(y, k_rand, 0.22, rng)
            raw = h_from_indices(p, y, inds, 0.22)
        else:
            if frac < 0.30:
                k_grad = int(max(12, min(32, int(n // 100) + 8)))
            elif frac < 0.70:
                k_grad = int(max(16, min(48, int(n // 100) + 16)))
            else:
                k_grad = int(max(24, min(64, int(n // 100) + 24)))
            tau = 0.25 if frac < 0.30 else 0.15
            raw = h_topk(p, y, k_grad, tau)

        rmax = float(np.max(raw)) if raw.size else 0.0
        if not np.isfinite(rmax) or rmax <= 1e-15:
            p_new = perturb_log(best_p, 0.10, rng)
            if p_new is None:
                p_new = best_p.copy()
            p = p_new
            cur_score = fast_score(p)
            it += 1
            stuck += 1
            continue

        g = prepare_gradient(p, raw)
        base_lr = 0.12 if frac < 0.50 else 0.06
        if stuck > 40:
            base_lr *= 0.6

        p_try, sc = multiplicative_candidates(p, g, base_lr, cur_score)
        if p_try is not None:
            p_new = positive_normalize(p_try)
            if p_new is None:
                p_new = best_p.copy()
            p = p_new
            cur_score = sc
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
            stuck = 0
            it += 1
            if verbose and it % 5000 == 0:
                print("[%s] grad it=%d best=%.6f cur=%.6f" %
                      (tag, it, best_score, cur_score))
            continue

        # Inverse field move, used periodically when gradient fails.
        if it % 6 == 3:
            b = inverse_candidate(p, raw)
            if b is not None:
                p_try, sc = mix_candidates(p, b, cur_score, t_values=(0.35, 0.20, 0.10))
                if p_try is not None:
                    p_new = positive_normalize(p_try)
                    if p_new is None:
                        p_new = best_p.copy()
                    p = p_new
                    cur_score = sc
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_p = p.copy()
                    stuck = 0
                    it += 1
                    continue

        stuck += 2

        if stuck > 50:
            best_pert = None
            best_sc = cur_score
            for sigma in (0.05, 0.10, 0.20, 0.40, 0.80):
                p_try = perturb_log(best_p, sigma, rng)
                if p_try is None:
                    continue
                sc = fast_score(p_try)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_pert = p_try

            if best_pert is not None and best_sc < cur_score:
                p_new = positive_normalize(best_pert)
                if p_new is None:
                    p_new = best_p.copy()
                p = p_new
                cur_score = best_sc
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_p = p.copy()
            else:
                p_new = perturb_log(best_p, 0.05, rng)
                if p_new is None:
                    p_new = best_p.copy()
                p = p_new
                cur_score = fast_score(p)

            stuck = 0
            last_lp = now - 1.0e9  # encourage an LP attempt next time
            it += 1
            continue

        it += 1

        if verbose and it % 5000 == 0:
            print("[%s] it=%d best=%.6f cur=%.6f stuck=%d n=%d" %
                  (tag, it, best_score, cur_score, stuck, n))

    return best_p, best_score

def append_candidate(cand_list, arr, tag):
    """Add a candidate if valid, optionally resizing overly long sequences."""
    q = positive_normalize(arr)
    if q is None:
        return
    if q.size > 12000:
        q = resize_average(q, 8192)
        if q is None:
            return
    sc = fast_score(q)
    if np.isfinite(sc):
        cand_list.append((q, float(sc), int(q.size), tag))

# Entry point
# -------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for the best non-negative sequence within the time budget."""
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    verify_reserve = min(60.0, 0.10 * total)
    hard_deadline = start + total - verify_reserve
    rng = np.random.default_rng(seed)

    parent_raw = globals().get("height_sequence_1")
    parent = positive_normalize(parent_raw) if parent_raw is not None else None

    cands = []

    if parent is not None:
        append_candidate(cands, parent, "parent")
        append_candidate(cands, symmetrize(parent), "parent_sym")

        if parent.size > 128:
            append_candidate(cands, downsample(parent, 2), "parent_half")
        if parent.size > 256:
            append_candidate(cands, downsample(parent, 4), "parent_quarter")

        if parent.size <= 20000:
            base_shifts = [1, 2, 5, 16, 64, max(1, parent.size // 10), max(1, parent.size // 4)]
            seen_shifts = set()
            shifts = []
            for s in base_shifts:
                s = int(s)
                if 0 < s < parent.size and s not in seen_shifts:
                    seen_shifts.add(s)
                    shifts.append(s)
            for s in shifts[:6]:
                append_candidate(cands, np.roll(parent, s), "parent_shift_%d" % s)

        for sig in (0.05, 0.15):
            append_candidate(cands, perturb_log(parent, sig, rng), "parent_pert_%d" % int(sig * 100))

    if parent is not None:
        n_parent = parent.size
        if 256 <= n_parent <= 10000:
            n_prim = n_parent
        elif n_parent > 10000:
            n_prim = 8192
        else:
            n_prim = max(512, int(n_parent * 2) if n_parent < 512 else n_parent)
    else:
        n_prim = 2048

    if parent is not None and n_parent != n_prim:
        append_candidate(cands, resize_average(parent, n_prim), "parent_resized")

    # Generic initializations at the primary length.
    append_candidate(cands, np.ones(n_prim, dtype=np.float64), "uniform")

    for i in range(3):
        append_candidate(cands, smooth_random_init(n_prim, rng), "smooth_%d" % i)

    for f in (1, 2, 3):
        for a in (0.5, 1.0):
            append_candidate(cands, cosine_init(n_prim, f, a, rng), "cos_%d_%d" % (f, int(a * 10)))

    append_candidate(cands, 0.2 + 0.8 * rng.random(n_prim), "random_pos")

    if not cands:
        append_candidate(cands, np.ones(max(64, n_prim), dtype=np.float64), "fallback")

    if len(cands) > 1:
        cands.sort(key=itemgetter(1))

    # Keep a small, high-quality set of starts.
    max_cands = min(6, len(cands))
    cands = cands[:max_cands]

    global_best = None
    global_score = np.inf

    candidate_end = start + max(0.0, min(0.80 * total - verify_reserve, hard_deadline - start))
    if candidate_end <= start + 30.0:
        candidate_end = 0.5 * (start + hard_deadline)

    main_budget = max(1.0, candidate_end - start)

    if len(cands) == 1:
        end_times = [candidate_end]
    elif len(cands) == 2:
        end_times = [start + 0.65 * main_budget, candidate_end]
    elif len(cands) == 3:
        end_times = [start + 0.50 * main_budget,
                     start + 0.80 * main_budget,
                     candidate_end]
    else:
        frac = [0.50, 0.25, 0.15]
        rem = max(0.0, 1.0 - sum(frac))
        per = rem / float(len(cands) - 3)
        for _ in range(len(cands) - 3):
            frac.append(per)
        end_times = []
        cum = 0.0
        for f in frac:
            cum += f
            if cum > 1.0:
                cum = 1.0
            end_times.append(start + cum * main_budget)
        end_times[-1] = candidate_end

    prev_end = start
    for idx, (p0, sc0, n0, tag) in enumerate(cands):
        if end_times[idx] <= prev_end + 5.0:
            break

        p0n = positive_normalize(p0)
        if p0n is None:
            p_start = np.ones(n0, dtype=np.float64) / float(n0)
        else:
            p_start = p0n

        # Blend slightly with the current global best for later same-length candidates,
        # preserving some diversity from p_start.
        if idx > 0 and global_best is not None and global_best.size == p_start.size:
            blended = 0.65 * p_start + 0.35 * global_best
            p_blend = positive_normalize(blended)
            if p_blend is not None:
                p_start = p_blend

        print("starting candidate %d/%d tag=%s n=%d initial=%.6f" %
              (idx + 1, len(cands), tag, p_start.size, fast_score(p_start)))

        bp, bsc = local_search(p_start, p_start.size, end_times[idx], rng, tag, verbose=True)

        if bp is not None and np.isfinite(bsc) and bsc < global_score:
            global_best = bp.copy()
            global_score = float(bsc)
            print("global best after candidate %d: %.6f n=%d" %
                  (idx + 1, global_score, int(global_best.size)))

        prev_end = end_times[idx]
        if time.monotonic() >= candidate_end:
            break

    # Final polish on the best sequence found.
    polish_end = hard_deadline - 5.0
    if global_best is not None and time.monotonic() < polish_end - 10.0:
        print("final polish start best=%.6f" % global_score)
        bp, bsc = local_search(global_best.copy(), global_best.size, polish_end, rng, "final_polish", verbose=True)
        if bp is not None and np.isfinite(bsc) and bsc < global_score:
            global_best = bp.copy()
            global_score = float(bsc)

    # Final verification and choice among a few simple transforms.
    final_out = global_best.copy() if global_best is not None else np.ones(n_prim, dtype=np.float64)
    final_score = global_score

    if global_best is not None:
        final_cands = [global_best.copy()]
        sym = symmetrize(global_best)
        if sym is not None:
            final_cands.append(sym)

        if global_best.size > 128:
            half = downsample(global_best, 2)
            if half is not None:
                final_cands.append(half)

        if global_best.size <= 8000:
            for s in (1, 2, 5, 16):
                if 0 < s < global_best.size:
                    final_cands.append(np.roll(global_best, s))

        if global_best.size <= 4096:
            max_evals = 6
        elif global_best.size <= 8192:
            max_evals = 3
        else:
            max_evals = 1

        for q in final_cands[:max_evals]:
            if time.monotonic() > hard_deadline - 3.0:
                break
            qn = positive_normalize(q)
            if qn is None:
                continue
            out_list = [float(x) for x in qn]
            try:
                val = float(evaluate_sequence(out_list))
            except Exception:
                val = np.inf
            if np.isfinite(val):
                print("verified candidate n=%d score=%.8f" % (qn.size, val))
                if val < final_score:
                    final_score = val
                    final_out = qn.copy()

    # Ensure a valid Python list of floats.
    out = positive_normalize(final_out)
    if out is None:
        out = np.ones(n_prim, dtype=np.float64)

    return [float(x) for x in out]
