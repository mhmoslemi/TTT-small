"""
Search for non-negative step heights minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2).

The implementation uses FFT-based scores and gradients, a log-sum-exp
continuation applied with L-BFGS-B, a multi-start scheme, first-order
multiplicative / active-set gradient polishing, and an LP-style Frank-Wolfe
direction computed by multiplicative weights.  All heavy internal evaluation
is FFT-based; the official evaluator is only called on a handful of final
candidates.
"""

import math
import time

import numpy as np

try:
    from scipy.optimize import minimize
    _HAS_SCIPY = True
except Exception:
    _minimize = None
    _HAS_SCIPY = False


# -----------------------------------------------------------------------
# Global time-stop mechanism for SciPy calls.  This avoids closures and
# lambda functions while still being able to stop long SciPy iterations.
# -----------------------------------------------------------------------

_OPT_STOP = {"active": False, "deadline": 0.0}


def _optim_callback(xk):
    if _OPT_STOP["active"] and time.monotonic() >= _OPT_STOP["deadline"]:
        return 1
    return 0

# Basic linear-algebra / FFT helpers.
# -----------------------------------------------------------------------

def _fft_len_needed(n):
    need = max(2 * n, 3 * n - 2, 1)
    L = 1
    while L < need:
        L <<= 1
    return L


def _normalize(a, floor=0.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    else:
        a = a / s
    if floor > 0.0:
        a = np.maximum(a, floor)
        s2 = float(a.sum())
        if s2 > 0.0:
            a = a / s2
    return a


def _softmax(z, floor=0.0):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(np.clip(z, -745.0, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    else:
        a = e / s
    if floor > 0.0:
        a = np.maximum(a, floor)
        s2 = float(a.sum())
        if s2 > 0.0:
            a = a / s2
    return a


def _log_theta(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    e = np.maximum(a, 1e-300)
    z = np.log(e)
    z = z - float(np.mean(z))
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.mean(z))
    return z


def _conv_self_with_F(a, L, m):
    """Linear self-convolution and its real FFT.  Returns (b, F)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    F = np.fft.rfft(a, L)
    b = np.fft.irfft(F * F, L)[:m]
    np.maximum(b, 0.0, out=b)
    return b, F


def _cross_w_a_with_F(w, F, L, n):
    """
    r[j] = sum_i a_i w[i+j], j=0..n-1, computed from F = FFT(a).
    """
    w = np.asarray(w, dtype=np.float64).ravel()
    Wr = np.fft.rfft(w[::-1], L)
    c = np.fft.irfft(F * Wr, L)
    r = c[n - 1:2 * n - 1][::-1]
    np.maximum(r, 0.0, out=r)
    return r


def _score_internal(a, L, m, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    b, _ = _conv_self_with_F(a, L, m)
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * n * mb / (s * s))


def _logsumexp_weights(b, beta):
    """
    For v=b and parameter beta, return softmax(beta*b), log(sum exp(beta*(b-max))),
    and max(b).
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    mm = b.size
    if mm == 0:
        return np.array([], dtype=np.float64), 0.0, 0.0
    vmax = float(np.max(b))
    z = beta * (b - vmax)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.full(mm, 1.0 / float(mm), dtype=np.float64)
        log_s = math.log(float(mm))
    else:
        w = e / s
        log_s = math.log(s)
    return w, log_s, vmax

def _add_candidate(arrays, scores, a, score, max_len=64):
    if a is None:
        return
    score = float(score)
    if not np.isfinite(score):
        return
    a = _normalize(a)
    if len(arrays) < max_len:
        arrays.append(a.copy())
        scores.append(score)
    else:
        mn = min(scores)
        if score < mn:
            idx = scores.index(mn)
            arrays[idx] = a.copy()
            scores[idx] = score


def _smooth_vector(v, w=7):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size <= int(w):
        return _normalize(v)
    k = np.ones(int(w), dtype=np.float64) / float(int(w))
    y = np.convolve(v, k, mode="same")
    return _normalize(y)


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    amp = rng.uniform(0.5, 3.0)
    if rng.random() < 0.5:
        z = amp * t
    else:
        z = -amp * t

    kmax = int(rng.integers(2, 8))
    for k in range(1, kmax + 1):
        ak = rng.uniform(0.05, 0.45) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z = z + ak * np.cos((2.0 * np.pi * k) * t + ph)

    noise = rng.normal(0.0, 0.20, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z = z + noise
    return _softmax(z)


def _initial_candidates(n, base, rng):
    out = []

    if base is not None and base.size == n:
        b = _normalize(base)
        out.append(b)
        out.append(b[::-1])
        out.append(_smooth_vector(b, 5))
        out.append(_smooth_vector(b, 15))

    t = np.linspace(0.0, 1.0, n)

    for p in (-1.0, -0.5, -0.25, 0.0, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0):
        arr = (t + 1e-9) ** p
        out.append(_normalize(arr))
        out.append(_normalize(arr[::-1]))

    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
        out.append(_normalize(np.exp(alpha * t)))
        out.append(_normalize(np.exp(-alpha * t)))

    for aa in (0.5, 1.0, 2.0):
        for bb in (0.5, 1.0, 2.0):
            arr = (t + 1e-9) ** aa * (1.0 - t + 1e-9) ** bb
            out.append(_normalize(arr))

    for _ in range(8):
        out.append(_random_prior(n, rng))

    if not out:
        out.append(_normalize(np.ones(n, dtype=np.float64)))

    return out


def _line_search(a, g, L, m, n, base_score):
    base = _normalize(a)
    a0 = base.copy()
    best_score = float(base_score)
    best_a = a0.copy()

    directions = []
    g1 = _normalize(g)
    directions.append(g1)
    directions.append(_smooth_vector(g1, 5))
    directions.append(_smooth_vector(g1, 15))

    lams = (0.02, 0.05, 0.10, 0.20, 0.35, 0.55, 0.80, 1.00)

    for d in directions:
        d = _normalize(d)
        for lam in lams:
            h = (1.0 - lam) * a0 + lam * d
            h = _normalize(h)
            sc = _score_internal(h, L, m, n)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = h
    return best_a, best_score


def _sparsify(a, frac):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a
    th = float(np.max(a)) * float(frac)
    b = np.where(a > th, a, 0.0)
    sb = float(b.sum())
    if sb <= 0.0:
        return a
    return b / sb

def _smooth_fun_grad(theta, beta, n, L, m):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    if theta.size != n:
        theta = np.resize(theta, n)

    # The objective is invariant to adding a constant to theta; keep it centered.
    theta = theta - float(np.mean(theta))
    theta = np.clip(theta, -80.0, 80.0)
    theta = theta - float(np.mean(theta))

    a = _softmax(theta)

    b, F = _conv_self_with_F(a, L, m)
    w, log_s, vmax = _logsumexp_weights(b, beta)

    # Smooth objective: (1/beta) * logsumexp(beta*b)
    J = vmax + log_s / float(beta)

    r = _cross_w_a_with_F(w, F, L, n)
    g = 2.0 * r
    gt = g - float(np.dot(g, a)) * a
    gt = gt - float(np.mean(gt))

    if not np.all(np.isfinite(gt)):
        gt = np.zeros(n, dtype=np.float64)

    return float(J), gt


def _lbfgs_call(theta, beta, n, L, m, deadline, maxiter):
    if not _HAS_SCIPY or time.monotonic() >= deadline - 5.0:
        return theta

    per_call = min(90.0, max(15.0, (deadline - time.monotonic()) * 0.55))
    _OPT_STOP["active"] = True
    _OPT_STOP["deadline"] = time.monotonic() + per_call
    try:
        res = minimize(
            _smooth_fun_grad,
            theta,
            method="L-BFGS-B",
            jac=True,
            args=(beta, n, L, m),
            callback=_optim_callback,
            options={
                "maxiter": int(maxiter),
                "maxfun": int(max(200, 6 * maxiter)),
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxls": 20,
                "maxcor": 20,
                "disp": False,
            },
        )
        x = res.x
        if x is None:
            return theta
        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n or not np.all(np.isfinite(x)):
            return theta
        x = x - float(np.mean(x))
        x = np.clip(x, -100.0, 100.0)
        x = x - float(np.mean(x))
        return x
    except Exception:
        return theta
    finally:
        _OPT_STOP["active"] = False


def _lbfgs_optimize(a, deadline, n, L, m, betas, maxiter, label="LB"):
    a = _normalize(a)
    best_a = a.copy()
    best_s = _score_internal(a, L, m, n)
    theta_best = _log_theta(best_a)

    for beta in betas:
        if time.monotonic() >= deadline - 8.0:
            break

        theta_try = _lbfgs_call(theta_best, float(beta), n, L, m, deadline, maxiter)
        a_try = _softmax(theta_try)
        s_try = _score_internal(a_try, L, m, n)

        print(f"{label} beta={beta:9.0f} new={s_try:.10f} best={best_s:.10f}", flush=True)

        if np.isfinite(s_try) and s_try < best_s + 1e-7:
            best_a = a_try.copy()
            best_s = s_try
            theta_best = theta_try

    return best_a, best_s

def _soft_gradient_search(a, deadline, rng, n, L, m, label="SG"):

    eta = 0.30
    no_accept = 0
    it = 0

    betas = (30000.0, 150000.0, 600000.0, 2000000.0)

    while time.monotonic() < deadline:
        if no_accept > 70 and it > 0:
            prior = _random_prior(n, rng)
            mix = 0.20 + 0.30 * float(rng.random())
            a = _normalize((1.0 - mix) * a + mix * prior)
            current_s = _score_internal(a, L, m, n)
            if current_s < best_s:
                best_a = a.copy()
                best_s = current_s
            no_accept = 0
            it = 0
            eta = 0.30

        b, F = _conv_self_with_F(a, L, m)
        beta = betas[min(beta_idx, len(betas) - 1)]
        w, _, _ = _logsumexp_weights(b, beta)

        g = 2.0 * _cross_w_a_with_F(w, F, L, n)
        scale = float(np.max(g))
        if not np.isfinite(scale) or scale <= 0.0:
            no_accept += 1
            continue

        accepted = False
        for step in (eta, eta * 0.5, eta * 0.25, eta * 1.5, eta * 2.5):
            h = a * np.exp(-step * g / scale)
            h = _normalize(h)
            sc = _score_internal(h, L, m, n)
            if np.isfinite(sc) and (sc < current_s - 1e-12 or sc < best_s - 1e-12):
                a = h
                current_s = sc
                if sc < best_s:
                    best_a = h.copy()
                    best_s = sc
                accepted = True
                no_accept = 0
                break

        if accepted:
            eta = min(3.0, eta * 1.20)
        else:
            no_accept += 1
            eta = max(0.03, eta * 0.75)

        it += 1
        if it % 90 == 0 and beta_idx < len(betas) - 1:
            beta_idx += 1
        if it % 250 == 0:
            print(f"{label} it={it} best={best_s:.10f} current={current_s:.10f}", flush=True)

    return best_a, best_s


# -----------------------------------------------------------------------
# LP-style Frank-Wolfe direction by multiplicative weights.
# -----------------------------------------------------------------------

def _mws_direction(F, Frev, f, prior, L, m, n, inner, eta_start, eta_end, gamma):
    x = _normalize(prior)
    if x.size != n:
        x = _normalize(np.ones(n))

    # Initialize dual weights from current self-convolution.
    bf = np.fft.irfft(F * F, L)[:m]
    np.maximum(bf, 0.0, out=bf)
    y, _, _ = _logsumexp_weights(n * bf, 12.0)

    avg_x = np.zeros(n, dtype=np.float64)
    avg_y = np.zeros(m, dtype=np.float64)
    count = 0
    log_prior = np.log(np.maximum(x, 1e-300))

    for it in range(int(inner)):
        frac = 0.0 if int(inner) <= 1 else float(it) / float(int(inner) - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        y, _, _ = _logsumexp_weights(n * u, eta)

        avg_y += y
        count += 1

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[n - 1:m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        pmax = float(np.max(ps)) if ps.size else 0.0
        zx = -eta * (ps - pmax) + gamma * log_prior
        zx = zx - float(np.max(zx))
        zx = np.clip(zx, -50.0, 0.0)
        e = np.exp(zx)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            x = _normalize(prior)
        else:
            x = e / se

        avg_x += x

    if count > 0:
        g = avg_x / float(count)
        yavg = avg_y / float(count)
    else:
        g = x
        yavg = np.full(m, 1.0 / float(m), dtype=np.float64)

    return _normalize(g), _normalize(yavg)


def _mws_search(a, deadline, rng, n, L, m, label="MWS"):
    a = _normalize(a)
    best_a = a.copy()
    best_s = _score_internal(a, L, m, n)

    it = 0
    stale = 0

    inner_base = 22 if n > 4000 else 16
    configs = (
        (inner_base, 8.0, 60.0, 0.4),
        (inner_base + 8, 15.0, 90.0, 0.7),
        (inner_base + 16, 30.0, 140.0, 1.0),
    )

    while time.monotonic() < deadline:
        f = _normalize(best_a)
        F = np.fft.rfft(f, L)
        Frev = np.fft.rfft(f[::-1], L)

        mod = it % 4
        if mod == 0:
            prior = _normalize(f)
        elif mod == 1:
            prior = _smooth_vector(f, 7)
        elif mod == 2:
            prior = _random_prior(n, rng)
        else:
            noise = rng.normal(0.0, 1.0, n)
            prior = _normalize(f * np.maximum(1.0 + 0.10 * noise, 0.0))

        cfg = configs[it % len(configs)]
        g, ym = _mws_direction(
            F, Frev, f, prior, L, m, n,
            int(cfg[0]), float(cfg[1]), float(cfg[2]), float(cfg[3])
        )

        c, sc = _line_search(f, g, L, m, n, best_s)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_a = c.copy()
            best_s = sc
            stale = 0
            print(f"{label} line-search it={it} best={best_s:.10f}", flush=True)
        else:
            stale += 1

        # Also try a direct gradient step using the LP dual weights.
        if time.monotonic() < deadline:
            fy = np.fft.rfft(f, L)
            gy = 2.0 * _cross_w_a_with_F(ym, fy, L, n)
            scaley = float(np.max(gy))
            if np.isfinite(scaley) and scaley > 0.0:
                for step in (0.05, 0.10, 0.20, 0.40, 0.80):
                    h = f * np.exp(-step * gy / scaley)
                    h = _normalize(h)
                    sh = _score_internal(h, L, m, n)
                    if np.isfinite(sh) and sh < best_s - 1e-12:
                        best_a = h.copy()
                        best_s = sh
                        stale = 0
                        print(f"{label} dual-grad it={it} best={best_s:.10f}", flush=True)
                        break

        if stale > 7:
            prior = _random_prior(n, rng)
            mix = 0.70 * best_a + 0.30 * prior
            mix = _normalize(mix)
            scmix = _score_internal(mix, L, m, n)
            if np.isfinite(scmix) and scmix < best_s:
                best_a = mix.copy()
                best_s = scmix
            stale = 0

        it += 1
        if it % 12 == 0:
            print(f"{label} it={it} best={best_s:.10f}", flush=True)

    return best_a, best_s


# -----------------------------------------------------------------------
# Final validation / selection.
# -----------------------------------------------------------------------

def _safe_exact(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = float("inf")
    except Exception:
        val = float("inf")
    if val is None:
        return float("inf")
    try:
        return float(val)
    except Exception:
        return float("inf")


def _final_evaluate(pool_arrays, pool_scores, best_a, best_score, deadline, n, L, m):
    cands = [best_a]

    if pool_arrays:
        order = list(range(len(pool_arrays)))
        order.sort(key=pool_scores.__getitem__)
        for i in order[:10]:
            cands.append(pool_arrays[i])

    top = []
    for a in cands:
        if len(top) >= 14:
            break
        a = _normalize(a)
        top.append(a)
        sym = 0.5 * (a + a[::-1])
        top.append(_normalize(sym))

    for a in top[:4]:
        for frac in (1e-3, 1e-2):
            sp = _sparsify(a, frac)
            sc_sp = _score_internal(sp, L, m, n)
            sc_a = _score_internal(a, L, m, n)
            if sc_sp <= sc_a + 1e-8:
                top.append(sp)

    # Rank by internal score without lambda.
    idxs = list(range(len(top)))
    top_scores = [float(_score_internal(top[i], L, m, n)) for i in idxs]
    order = list(range(len(top)))
    order.sort(key=top_scores.__getitem__)
    top_sorted = [top[i] for i in order[:14]]

    best_seq = None
    best_val = float("inf")

    for a in top_sorted:
        if time.monotonic() > deadline - 8.0:
            break
        a = _normalize(a)
        seq = [float(10.0 * float(x)) for x in a]
        val = _safe_exact(seq)
        if not np.isfinite(val):
            val = _score_internal(a, L, m, n)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_seq = seq

    if best_seq is None:
        best_seq = [float(10.0 * float(x)) for x in _normalize(best_a)]

    return best_seq

# Main entry point.

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = min(budget, 2400.0)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    # Base sequence from previous state-of-art if available.
    base = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        base = None

    if base is not None and base.size > 0:
        if 200 <= base.size <= 12000:
            n = int(base.size)
        else:
            n = 6121
    else:
        n = 6121

    m = 2 * n - 1
    L = _fft_len_needed(n)

    print(f"Starting search: n={n}, L={L}, budget={budget:.1f}s", flush=True)

    # Very small budgets: return a normalized quick construction.
    if budget < 90.0:
        if base is not None and base.size == n:
            a = _normalize(base)
        else:
            t = np.linspace(0.0, 1.0, n)
            a = _normalize(1.0 + t)
        return [float(10.0 * float(x)) for x in a]

    # Reserve final time for exact evaluation.
    hard_deadline = t_start + max(30.0, budget - 5.0)
    work_deadline = t_start + max(60.0, budget - 70.0)
    if work_deadline > hard_deadline - 35.0:
        work_deadline = hard_deadline - 35.0

    pool_arrays = []
    pool_scores = []

    inits = _initial_candidates(n, base, rng)
    for a in inits:
        sc = _score_internal(a, L, m, n)
        _add_candidate(pool_arrays, pool_scores, a, sc, 80)

    if not pool_arrays:
        a = _normalize(np.arange(1, n + 1, dtype=np.float64))
        pool_arrays.append(a)
        pool_scores.append(_score_internal(a, L, m, n))

    idx_best = 0
    best_score = float("inf")
    for i, sc in enumerate(pool_scores):
        if sc < best_score:
            best_score = sc
            idx_best = i
    best_a = pool_arrays[idx_best].copy()

    print(f"Initial best internal score = {best_score:.10f}", flush=True)

    if time.monotonic() > work_deadline - 20.0:
        return _final_evaluate(pool_arrays, pool_scores, best_a, best_score, hard_deadline, n, L, m)

    # ------------------------------------------------------------------
    # Phase A: main L-BFGS continuation from the best initial candidate.
    # ------------------------------------------------------------------
    total_work = max(60.0, work_deadline - t_start)
    phaseA_deadline = t_start + min(0.45 * total_work, 1000.0)
    phaseA_deadline = min(phaseA_deadline, work_deadline - 120.0)

    if phaseA_deadline > time.monotonic() + 60.0:
        betas_main = (
            1000.0, 3000.0, 10000.0, 30000.0,
            100000.0, 300000.0, 1000000.0, 3000000.0
        )
        a2, s2 = _lbfgs_optimize(best_a, phaseA_deadline, n, L, m, betas_main, 140, "LB-main")
        _add_candidate(pool_arrays, pool_scores, a2, s2, 80)
        if np.isfinite(s2) and s2 < best_score:
            best_a = a2.copy()
            best_score = s2
            print(f"Phase A improved best score = {best_score:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Phase B: run moderate L-BFGS on the next-best initial candidates.
    # ------------------------------------------------------------------
    if pool_arrays:
        order = list(range(len(pool_arrays)))
        order.sort(key=pool_scores.__getitem__)
        for i in order[:3]:
            if time.monotonic() > work_deadline - 500.0:
                break
            lim = min(time.monotonic() + 100.0, work_deadline - 480.0)
            if lim <= time.monotonic() + 30.0:
                continue
            cand = pool_arrays[i]
            a3, s3 = _lbfgs_optimize(
                cand, lim, n, L, m,
                (30000.0, 100000.0, 400000.0, 1500000.0),
                90, "LB-alt"
            )
            _add_candidate(pool_arrays, pool_scores, a3, s3, 80)
            if np.isfinite(s3) and s3 < best_score:
                best_a = a3.copy()
                best_score = s3
                print(f"Phase B improved best score = {best_score:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Phase C: alternating MWS direction search, high-beta L-BFGS refinement,
    # and exact-gradient polishing.
    # ------------------------------------------------------------------
    phaseC_deadline = work_deadline - 180.0
    stale = 0
    alt_idx = 1
    loop = 0

    while time.monotonic() < phaseC_deadline:
        rem = phaseC_deadline - time.monotonic()
        if rem < 80.0:
            break

        # MWS / Frank-Wolfe direction.
        slice_mws = min(160.0, max(40.0, 0.38 * rem))
        am, scm = _mws_search(best_a, time.monotonic() + slice_mws, rng, n, L, m, "MWS")
        _add_candidate(pool_arrays, pool_scores, am, scm, 80)
        if np.isfinite(scm) and scm < best_score - 1e-12:
            best_a = am.copy()
            best_score = scm
            stale = 0
            print(f"Phase C MWS improved best score = {best_score:.10f}", flush=True)
        else:
            stale += 1

        if time.monotonic() >= phaseC_deadline - 80.0:
            break

        # If we are stale, polish a decent alternate candidate.
        if stale >= 3 and len(pool_arrays) > 1:
            lim = min(time.monotonic() + 90.0, phaseC_deadline - 120.0)
            if lim > time.monotonic() + 20.0:
                j = alt_idx % max(1, len(pool_arrays))
                alt_idx += 1
                a_alt, s_alt = _lbfgs_optimize(
                    pool_arrays[j], lim, n, L, m,
                    (50000.0, 200000.0, 800000.0, 2500000.0),
                    80, "LB-stale"
                )
                _add_candidate(pool_arrays, pool_scores, a_alt, s_alt, 80)
                if np.isfinite(s_alt) and s_alt < best_score - 1e-12:
                    best_a = a_alt.copy()
                    best_score = s_alt
                    stale = 0
                    print(f"Phase C stale-alt improved best score = {best_score:.10f}", flush=True)

        # High-beta L-BFGS refinement of the current best.
        slice_lb = min(110.0, max(30.0, 0.24 * rem))
        lim_lb = time.monotonic() + slice_lb
        al, scl = _lbfgs_optimize(
            best_a, lim_lb, n, L, m,
            (100000.0, 400000.0, 1500000.0, 4000000.0),
            90, "LB-refine"
        )
        _add_candidate(pool_arrays, pool_scores, al, scl, 80)
        if np.isfinite(scl) and scl < best_score - 1e-12:
            best_a = al.copy()
            best_score = scl
            stale = 0
            print(f"Phase C LB improved best score = {best_score:.10f}", flush=True)
        else:
            stale += 1

        if time.monotonic() >= phaseC_deadline - 60.0:
            break

        # Exact-gradient polishing.
        slice_sg = min(80.0, max(20.0, 0.18 * rem))
        asg, scg = _soft_gradient_search(
            best_a, time.monotonic() + slice_sg, rng, n, L, m, "SG"
        )
        _add_candidate(pool_arrays, pool_scores, asg, scg, 80)
        if np.isfinite(scg) and scg < best_score - 1e-12:
            best_a = asg.copy()
            best_score = scg
            stale = 0
            print(f"Phase C SG improved best score = {best_score:.10f}", flush=True)

        loop += 1
        if loop % 3 == 0:
            print(f"Phase C loop={loop} best={best_score:.10f} stale={stale}", flush=True)

    # ------------------------------------------------------------------
    # Phase D: one final high-precision L-BFGS push if time remains.
    # ------------------------------------------------------------------
    final_deadline = work_deadline - 30.0
    if time.monotonic() < final_deadline - 60.0:
        af, scf = _lbfgs_optimize(
            best_a, final_deadline, n, L, m,
            (200000.0, 800000.0, 3000000.0),
            120, "LB-final"
        )
        _add_candidate(pool_arrays, pool_scores, af, scf, 80)
        if np.isfinite(scf) and scf < best_score:
            best_a = af.copy()
            best_score = scf
            print(f"Final refined score = {best_score:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Final exact evaluation and return.
    # ------------------------------------------------------------------
    seq = _final_evaluate(pool_arrays, pool_scores, best_a, best_score, hard_deadline, n, L, m)

    t_end = time.monotonic()
    print(f"Finished in {t_end - t_start:.1f}s. Best internal score ~ {best_score:.10f}", flush=True)

    return seq
