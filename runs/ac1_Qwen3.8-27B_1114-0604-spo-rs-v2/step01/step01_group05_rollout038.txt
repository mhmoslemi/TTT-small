"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-phase optimization with power-norm
smooth max, Nesterov acceleration, and sinusoidal parametric starts.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_conv(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fast_score(p):
    n = p.size
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_conv(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S*S))

def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target/n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5*(lo+hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(n, target/n, dtype=np.float64)
    else:
        q *= target/s
    return q

def normalize(p):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0/float(p.size), dtype=np.float64)
    return p/s

def cross_corr(p, w):
    """Compute s[i] = sum_j p[j]*w[i+j] for i=0..len(p)-1 using FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)

def power_norm_grad(p, power):
    """
    Compute gradient of n * (sum(b^power))^(1/power) where b = p*p.
    Returns (value, gradient) where gradient is w.r.t. p (before normalization).
    Assumes sum(p) = 1.
    """
    b = fft_conv(p)
    n = p.size
    b_pos = np.maximum(b, 1e-30)
    b_p = b_pos ** power
    B = float(np.sum(b_p))
    if B <= 0 or not np.isfinite(B):
        return float('inf'), np.zeros(n, dtype=np.float64)
    
    val = n * B ** (1.0/power)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-phase optimization:
1. Sinusoidal parametric L-BFGS (global coarse search, low dimension).
2. Block-constant L-BFGS (medium resolution).
3. Full-length Nesterov-accelerated projected gradient with power-norm smooth max.
4. Adaptive power-norm refinement (increasing p for sharper max approximation).
5. Direct peak shaving for final polish.
Primary N=16384, secondary N=32768 if time permits.

import time
from scipy.optimize import minimize


def fft_conv(a):
    """Linear convolution a*a via FFT, clipped non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(a*a)/(sum(a))^2 using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_conv(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize(p):
    """Clip to non-negative, normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p / s


def cross_corr(p, w):
    """s[i] = sum_j p[j] * w[i+j] for i=0..len(p)-1 via FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def power_norm_obj_grad(p, power):
    """
    Objective and gradient of n*(sum(b^power))^(1/power) where b = p*p, sum(p)=1.
    Returns (value, gradient_vector).
    """
    b = fft_conv(p)
    n = p.size
    b_pos = np.maximum(b, 1e-30)
    b_p = b_pos ** power
    B = float(np.sum(b_p))
    if B <= 0 or not np.isfinite(B):
        return float('inf'), np.zeros(n, dtype=np.float64)
    val = n * B ** (1.0 / power)
    u = b_pos ** (power - 1.0)
    g = cross_corr(p, u)
    scale = 2.0 * n * B ** (1.0 / power - 1.0)
    g *= scale
    return val, g


def softmax_obj_grad(p, beta):
    """Objective and gradient using softmax smooth max."""
    b = fft_conv(p)
    n = p.size
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros(n, dtype=np.float64)
    if m <= 0 or beta <= 0:
        w = np.ones(2 * n - 1, dtype=np.float64)
        w /= (2 * n - 1)
        return m, np.full(n, 2.0 * np.dot(w, p), dtype=np.float64)
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        e = np.zeros(2 * n - 1, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    svec = cross_corr(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    return float(n * G), g

    best_p = p.copy()

    while t_iter < max_iter and time.monotonic() < deadline:

        g_s = g - float(np.mean(g))

def block_to_p(n, B, c):
    """Convert block log-weights to full sequence (sum=1)."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        z[:-1] = tmp
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_obj_grad(c, n, B, power):
    """Objective and gradient for block-constant sequences using power-norm."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        z[:-1] = tmp
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float('inf'), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / B
    else:
        wb /= s
    H = wb / L
    p = np.repeat(H, L)
    b = fft_conv(p)
    b_pos = np.maximum(b, 1e-30)
    b_p = b_pos ** power
    Bval = float(np.sum(b_p))
    if Bval <= 0 or not np.isfinite(Bval):
        return float('inf'), np.zeros(B - 1, dtype=np.float64)
    val = n * Bval ** (1.0 / power)
    u = b_pos ** (power - 1.0)
    g_full = cross_corr(p, u)
    g_full *= 2.0 * n * Bval ** (1.0 / power - 1.0)
    Gsum = g_full.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / L) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(val), grad.astype(np.float64)


def run_block_lbfgs(c0, n, B, power, deadline):
    """L-BFGS-B on block representation."""
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)
    options = {"maxiter": 200, "ftol": 1e-12, "gtol": 1e-6, "maxls": 50, "maxcor": 15}
    try:
        res = minimize(block_obj_grad, c0, args=(n, B, power),
                       method="L-BFGS-B", jac=True,
                       bounds=[(-8.0, 8.0)] * (B - 1), options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def sinusoidal_to_p(params, n, max_freq):
    """Convert sinusoidal params to normalized sequence."""
    k = np.arange(n, dtype=np.float64)
    log_a = np.zeros(n, dtype=np.float64)
    for m in range(1, max_freq + 1):
        c = params[2 * (m - 1)]
        d = params[2 * (m - 1) + 1]
        angle = 2.0 * np.pi * m * k / n
        log_a += c * np.cos(angle) + d * np.sin(angle)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = float(np.sum(a))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def sinusoidal_obj_grad(params, n, max_freq, power):
    """Objective and gradient for sinusoidal parameterization."""
    k = np.arange(n, dtype=np.float64)
    log_a = np.zeros(n, dtype=np.float64)
    for m in range(1, max_freq + 1):
        c = params[2 * (m - 1)]
        d = params[2 * (m - 1) + 1]
        angle = 2.0 * np.pi * m * k / n
        log_a += c * np.cos(angle) + d * np.sin(angle)
    log_a -= np.max(log_a)
    u = np.exp(log_a)
    Z = float(np.sum(u))
    if Z <= 0 or not np.isfinite(Z):
        return float('inf'), np.zeros(2 * max_freq, dtype=np.float64)
    p = u / Z
    b = fft_conv(p)
    b_pos = np.maximum(b, 1e-30)
    b_p = b_pos ** power
    Bval = float(np.sum(b_p))
    if Bval <= 0 or not np.isfinite(Bval):
        return float('inf'), np.zeros(2 * max_freq, dtype=np.float64)
    val = n * Bval ** (1.0 / power)
    u_conv = b_pos ** (power - 1.0)
    g_full = cross_corr(p, u_conv)
    g_full *= 2.0 * n * Bval ** (1.0 / power - 1.0)
    # dF/dtheta_a = sum_k g_full[k] * p[k] * (dlog_a[k]/dtheta_a - sum_j p[j]*dlog_a[j]/dtheta_a)
    grad = np.zeros(2 * max_freq, dtype=np.float64)
    gp = g_full * p
    for m in range(1, max_freq + 1):
        angle = 2.0 * np.pi * m * k / n
        ca = np.cos(angle)
        sa = np.sin(angle)
        h_c = float(np.dot(p, ca))
        h_s = float(np.dot(p, sa))
        grad[2 * (m - 1)] = float(np.dot(gp, ca - h_c))
        grad[2 * (m - 1) + 1] = float(np.dot(gp, sa - h_s))
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(val), grad


def peak_shave(p0, deadline, max_iter=50000):
    """Direct peak shaving: target the max of b=a*a directly."""
    p = normalize(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    eta = 0.01
    no_improve = 0
    for t in range(max_iter):
        if time.monotonic() >= deadline:
            break
        b = fft_conv(p)
        M = float(np.max(b))
        # Weight: top 5% of b values
        threshold = np percentile(b, 95) if b.size > 10 else M * 0.8
        w = np.where(b >= threshold, (b / max(M, 1e-30)) ** 3, 0.0)
        w_sum = float(np.sum(w))
        if w_sum <= 0 or not np.isfinite(w_sum):
            break
        w = w / w_sum
        # Pressure on each element
        pressure = cross_corr(p, w)
        pressure = np.maximum(pressure, 0.0)
        # Update: reduce high-pressure elements
        q = project_simplex(p - eta * (pressure - float(np.mean(pressure))), 1.0)
        sc = fast_score(q)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()  # p (before update) is still best
            p = q
            no_improve = 0
            if eta < 0.05:
                eta *= 1.1
        else:
            p = q  # still move, but track best
            no_improve += 1
            eta *= 0.9
        if no_improve > 5000:
            break
    # Final check
    sc = fast_score(p)
    if sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def structured_starts(n, rng, max_count=10):
    """Generate diverse starting profiles."""
    starts = []
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p.copy())
    # Two-level: high in middle, low at edges (or vice versa)
    for eps in (0.3, 0.5, 0.7):
        p = np.ones(n, dtype=np.float64)
        half = n // 2
        p[:half] = 1.0 + eps
        p[half:] = max(1e-4, 1.0 - eps)
        p = normalize(p)
        starts.append(p.copy())
        p2 = np.ones(n, dtype=np.float64)
        p2[:half] = max(1e-4, 1.0 - eps)
        p2[half:] = 1.0 + eps
        p2 = normalize(p2)
        starts.append(p2.copy())
    # Random smooth
    size = 1
    while size < n * 2:
        size <<= 1
    for _ in range(3):
        zc = rng.normal(size=size // 2 + 1) + 1j * rng.normal(size=size // 2 + 1)
        z = np.real(np.fft.irfft(zc, size))[:n]
        z = z - float(np.mean(z))
        z = 2.0 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p.copy())
    return starts[:max_count]


def run_search_for_N(N, deadline, rng, previous_p=None):
    """Run the full search pipeline for a given N. Returns (best_p, best_score)."""
    print(f"  Starting search for N={N}")
    best_p = np.full(N, 1.0 / N, dtype=np.float64)
    best_score = fast_score(best_p)
    print(f"    uniform: score={best_score:.8f}")
    
    # If we have a previous sequence, resample to this N
    if previous_p is not None:
        prev = np.asarray(previous_p, dtype=np.float64)
        prev = normalize(prev)
        # Resample by interpolation
        from numpy import interp
        old_n = prev.size
        new_n = N
        old_x = np.linspace(0, 1, old_n + 1, dtype=np.float64)
        old_p = np.concatenate([prev, [prev[0]]])
        new_x = np.linspace(0, 1, new_n + 1, dtype=np.float64)
        resampled = interp(new_x[:-1], old_x, old_p)
        resampled = normalize(resampled)
        sc = fast_score(resampled)
        print(f"    resampled prev: score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_p = resampled.copy()
    
    remaining = lambda: deadline - time.monotonic()
    
    # --- Phase 1: Sinusoidal parametric L-BFGS ---
    if remaining() > 60.0:
        print("    Phase 1: Sinusoidal parametric optimization")
        phase1_deadline = min(deadline, time.monotonic() + max(30.0, 0.08 * 2400.0))
        max_freq = min(12, N // 32)
        power_param = 50.0
        best_sin_score = float('inf')
        best_sin_p = best_p.copy()
        for restart in range(4):
            if time.monotonic() >= phase1_deadline:
                break
            if restart == 0:
                c0 = np.zeros(2 * max_freq, dtype=np.float64)
            else:
                scale = 1.5 * (restart - 1)
                c0 = scale * rng.standard_normal(2 * max_freq)
            options = {"maxiter": 300, "ftol": 1e-14, "gtol": 1e-7, "maxls": 50, "maxcor": 20}
            try:
                res = minimize(sinusoidal_obj_grad, c0, args=(N, max_freq, power_param),
                               method="L-BFGS-B", jac=True, options=options)
                c_opt = np.asarray(res.x, dtype=np.float64)
            except Exception:
                c_opt = c0
            p_sin = sinusoidal_to_p(c_opt, N, max_freq)
            sc = fast_score(p_sin)
            print(f"    sinusoidal restart {restart}: score={sc:.8f}")
            if sc < best_sin_score:
                best_sin_score = sc
                best_sin_p = p_sin.copy()
        if best_sin_score < best_score:
            best_score = best_sin_score
            best_p = best_sin_p.copy()
            print(f"    after sinusoidal: score={best_score:.8f}")
    
    # --- Phase 2: Block-constant L-BFGS ---
    if remaining() > 60.0:
        print("    Phase 2: Block-constant optimization")
        phase2_deadline = min(deadline, time.monotonic() + max(30.0, 0.12 * 2400.0))
        power_block = 50.0
        best_block_score = float('inf')
        best_block_p = best_p.copy()
        B_candidates = [16, 32, 64, 128, 256, 512]
        B_candidates = [B for B in B_candidates if B < N and N % B == 0]
        for B in B_candidates:
            if time.monotonic() >= phase2_deadline:
                break
            power_b = 30.0 if B <= 128 else 50.0
            # Initialize from current best (extract block averages)
            L = N // B
            if L >= 2:
                p_best = best_p
                block_avgs = p_best.reshape(B, L).mean(axis=1)
                c0 = np.log(np.maximum(block_avgs, 1e-15))
                c0 = c0 - c0[-1]  # fix last block to 0
                c0 = c0[:-1]
                c0 = np.clip(c0, -8.0, 8.0)
            else:
                c0 = np.zeros(B - 1, dtype=np.float64)
            c_opt = run_block_lbfgs(c0, N, B, power_b, phase2_deadline)
            p_blk = block_to_p(N, B, c_opt)
            sc = fast_score(p_blk)
            print(f"    block B={B}: score={sc:.8f}")
            if sc < best_block_score:
                best_block_score = sc
                best_block_p = p_blk.copy()
        if best_block_score < best_score:
            best_score = best_block_score
            best_p = best_block_p.copy()
            print(f"    after block: score={best_score:.8f}")
    
    # --- Phase 3: Nesterov gradient with increasing power ---
    if remaining() > 120.0:
        print("    Phase 3: Nesterov gradient refinement")
        # Adaptive power schedule
        power_schedule = [10, 25, 50, 100, 200, 500]
        phase3_deadline = min(deadline, time.monotonic() + max(120.0, 0.55 * 2400.0))
        for power in power_schedule:
            if time.monotonic() >= phase3_deadline:
                break
            remaining_phase = phase3_deadline - time.monotonic()
            if remaining_phase < 30.0:
                break
            p_cur, sc_cur = nesterov_refine(best_p, phase3_deadline, power,
                                            max_iter=min(500000, int(remaining_phase * 2000)))
            if sc_cur < best_score:
                best_score = sc_cur
                best_p = p_cur.copy()
                print(f"    after nesterov power={power}: score={best_score:.8f}")
    
    # --- Phase 4: Softmax refinement (sharper) ---
    if remaining() > 60.0:
        print("    Phase 4: Softmax refinement")
        phase4_deadline = min(deadline, time.monotonic() + max(30.0, 0.10 * 2400.0))
        for beta in (50000, 200000, 1000000):
            if time.monotonic() >= phase4_deadline:
                break
            p_cur, sc_cur = softmax_refine(best_p, phase4_deadline, beta, max_iter=200000)
            if sc_cur < best_score:
                best_score = sc_cur
                best_p = p_cur.copy()
                print(f"    after softmax beta={beta}: score={best_score:.8f}")
    
    # --- Phase 5: Peak shaving ---
    if remaining() > 30.0:
        print("    Phase 5: Peak shaving")
        phase5_deadline = min(deadline, time.monotonic() + max(20.0, 0.05 * 2400.0))
        p_cur, sc_cur = peak_shave(best_p, phase5_deadline)
        if sc_cur < best_score:
            best_score = sc_cur
            best_p = p_cur.copy()
            print(f"    after peak shave: score={best_score:.8f}")
    
    # Final Nesterov polish
    if remaining() > 30.0:
        p_cur, sc_cur = nesterov_refine(best_p, deadline, power=100, max_iter=200000)
        if sc_cur < best_score:
            best_score = sc_cur
            best_p = p_cur.copy()
            print(f"    final polish: score={best_score:.8f}")
    
    print(f"  Final for N={N}: score={best_score:.8f}")
    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    if np.isfinite(val):
        return val
    return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function. Returns the best sequence found within the time budget.
    Uses multi-phase optimization with power-norm and softmax smooth max,
    Nesterov acceleration, and peak shaving.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    if budget_s <= 0:
        return [1.0]
    
    start_time = time.monotonic()
    reserve = 60.0 if budget_s > 1000 else 30.0
    deadline = start_time + max(0.0, budget_s - reserve)
    
    rng = np.random.default_rng(seed)
    
    # Choose primary N based on budget
    if budget_s < 300:
        N_primary = 2048
        N_secondary = None
    elif budget_s < 800:
        N_primary = 4096
        N_secondary = 8192
    elif budget_s < 1500:
        N_primary = 8192
        N_secondary = 16384
    else:
        N_primary = 16384
        N_secondary = 32768
    
    # Check for previous solution
    prev_p = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            prev_seq = list(height_sequence_1)
            if len(prev_seq) > 0:
                prev_p = np.asarray(prev_seq, dtype=np.float64)
                prev_p = normalize(prev_p)
                print(f"Using previous sequence of length {prev_p.size}, score={fast_score(prev_p):.8f}")
    except Exception:
        prev_p = None
    
    print(f"Budget: {budget_s:.0f}s, Primary N={N_primary}, Secondary N={N_secondary}")
    
    best_p = None
    best_score = float('inf')
    
    # Run search for primary N
    p1, s1 = run_search_for_N(N_primary, deadline, rng, previous_p=prev_p)
    best_p = p1
    best_score = s1
    
    # Run search for secondary N if time allows
    if N_secondary is not None and time.monotonic() < deadline - 120.0:
        remaining = deadline - time.monotonic()
        if remaining > 200.0:
            alloc = min(remaining - 60.0, 0.4 * budget_s)
            sub_deadline = time.monotonic() + alloc
            print(f"  Secondary search with N={N_secondary}, alloc={alloc:.0f}s")
            p2, s2 = run_search_for_N(N_secondary, sub_deadline, rng, previous_p=best_p)
            if s2 < best_score:
                best_score = s2
                best_p = p2
                print(f"  New best from N={N_secondary}: {best_score:.8f}")
    
    # Final official evaluation
    final_p = normalize(best_p)
    final_seq = [float(x) for x in final_p]
    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")
    
    if not np.isfinite(off) or off > 2.0:
        # Fallback: uniform sequence
        fallback_n = N_primary
        fallback = [1.0] * fallback_n
        fb_off = official_score(fallback)
        print(f"Fallback uniform score: {fb_off:.8f}")
        return fallback
    
    return final_seq
