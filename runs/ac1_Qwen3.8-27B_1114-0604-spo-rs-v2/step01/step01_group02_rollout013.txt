```python
import numpy as np

def compute_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_value(c, T):
    """Compute smooth max (log-sum-exp normalized by T)."""
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp((c - c_max) * T))) / T

def smooth_max_weights(c, T):
    """Compute softmax weights for the smooth max."""
    c_max = np.max(c)
    w = np.exp((c - c_max) * T)
    return w / np.sum(w)

    """Compute cross-correlation of a with w via FFT.
    Result[j] = sum_i a[i] * w[i+j] (valid ranges only)
    Equivalently, this is the correlation (not convolution) of a and w.
    """
    n = len(a)
    m = len(w)  # should be 2n-1
    # We want: result[j] = sum_{i} a[i] * w[i+j] for i+j in [0, m-1]
    # This is equivalent to: flip a, convolve with w, then appropriate slice
    # Actually: cross-corr(a, w)[j] = sum_i a[i] w[i+j] = conv(flip(a), w)[n-1+j]
    # Hmm, let me think more carefully.
    
    # We want h[j] = sum_i a[i] * w[i+j] where both i and i+j must be valid indices.
    # i ranges from max(0, 0) to min(n-1, m-1-j) = min(n-1, 2n-2-j)
    
    # This is the "cross-correlation" which in signal processing is:
    # h[j] = (a ⋆ w)[j] = sum_i a[i] w[i+j]
    
    # To compute via FFT: reverse a, then convolve with w, then appropriate slice.
    # conv(flip(a), w)[k] = sum_i flip(a)[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
    # Let j' = n-1-i: = sum_{j'} a[j'] * w[k-n+1+j']
    # So conv(flip(a), w)[k] = h[k-n+1] = sum_{j'} a[j'] * w[j' + (k-n+1)]
    
    # Therefore: h[j] = conv(flip(a), w)[j+n-1]
    
    fft_size = 1 << (n + m - 2).bit_length()
    A_flipped = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A_flipped * W, fft_size)
    return conv_result[n-1 : n-1+n]  # h[0..n-1]

def objective_and_gradient(a, T):

    n = len(a)
    c = compute_conv(a)
    S = np.sum(a)
    
    # Smooth max
    c_max = np.max(c)
    shifted = (c - c_max) * T
    exp_shifted = np.exp(shifted)
    log_sum = np.sum(exp_shifted)
    LSE = c_max + np.log(log_sum) / T
    
    # Weights
    w = exp_shifted / log_sum
    
    # Gradient of smooth max w.r.t. c: just w
    # Gradient of c w.r.t. a: cross_correlation(a, w) * 2 (approximately)
    # More precisely: dc[k]/da[j] ≈ 2*a[k-j] (for k-j valid)
    # So grad_c[j] = sum_k w[k] * 2*a[k-j] = 2 * cross_correlation(a, w)[j]
    
    grad_c = 2.0 * cross_correlation(a, w)

    # Correction[j] = -w[2j] * a[j] for 2j < 2n-1

        if k < 2 * n - 1:

    grad_c += correction
    
    # Full objective: f = 2n * LSE / S^2
    # df/da[j] = 2n * grad_c[j] / S^2 - 2n * LSE * 2 / S^3
    obj = 2.0 * n * LSE / (S * S)
    grad = 2.0 * n * grad_c / (S * S) - 4.0 * n * LSE / (S * S * S)
    
    return obj, grad
```

For alpha < 1/2: let's check alpha = 0: LHS = 1, RHS = 1. So alpha = 0 is a solution (trivial, uniform).

So 1-2alpha ≈ 1+alpha → -2alpha ≈ alpha → alpha ≈ 0.

So the minimum of f is at alpha → 0 (uniform sequence), giving f → 2. The exponential decay never beats the uniform sequence!

This makes sense: the uniform sequence already achieves f = 2, and the exponential decay makes the convolution less flat (peaked), so it's worse.

```python
def cross_correlation(a, w):
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(A * np.conj(W), fft_size)
    # corr[k] = sum_i a[i] * w[i+k-n+m-1]... 
    # Actually np.correlate(a, w, 'full')[k] = sum_i a[i] * w[i + k - (m-1)]
    # So to get h[j] = sum_i a[i]*w[i+j], we need k - (m-1) = j, i.e., k = j + m - 1
    # corr has length n+m-1, indexed 0 to n+m-2
    # h[j] = corr[j + m - 1] for j = 0, ..., n-1
    return corr[m-1 : m-1+n]
```
