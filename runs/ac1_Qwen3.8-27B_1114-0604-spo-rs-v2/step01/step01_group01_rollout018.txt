    n = len(a)

    if s < 1e-15:
        return float('inf')

    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = a.sum()

        # Softmax weights on peaks
        shift = peak_vals.max()
        exps = np.exp(beta * (peak_vals - shift))
        s_exp = exps.sum()
        if s_exp < 1e-30:
            weights = np.full(k, 1.0 / k)
        else:
            weights = exps / s_exp

        # Compute gradient
        grad = _grad_from_peaks(p, peaks, weights)
        gmax = grad.max()

        if gmax < 1e-15:
            p = _normalize(p * np.exp(rng.normal(0, 0.05, n)))
            stale += 1
            continue

        grad_n = grad / gmax

        # Line search: try a range of step sizes
        candidates = []
        for mult in [0.1, 0.25, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
            a_try = alpha * mult
            if a_try < 1e-4 or a_try > 50.0:
                continue
            candidates.append(a_try)
        if not candidates:
            candidates = [alpha]

        best_new_score = cur_score
        best_p_try = None
        best_a_try = alpha

        for a_try in candidates:
            dec = np.exp(-a_try * grad_n)
            p_try = p * dec
            p_try = np.maximum(p_try, floor)
            s_try = p_try.sum()
            if s_try < 1e-15 or not np.isfinite(s_try):
                continue
            p_try = p_try / s_try
            sc_try = _score_normalized(p_try)
            if np.isfinite(sc_try) and sc_try < best_new_score - 1e-12:
                best_new_score = sc_try
                best_p_try = p_try
                best_a_try = a_try

        total_iters += 1

        if best_p_try is not None:
            p = best_p_try
            alpha = best_a_try
            stale = 0
            if best_new_score < best_score:
                best = p.copy()
                best_score = best_new_score
            # Progress: increase beta slightly, narrow k
            beta = min(beta_end, beta * 1.02)
            k_frac = max(0.01, k_frac * 0.999)
        else:
            stale += 1
            alpha = max(0.01, alpha * 0.7)
            if stale > 20:
                # Medium perturbation
                std = 0.05 + 0.01 * min(stale, 50)
                p = _normalize(p * np.exp(rng.normal(0, std, n)))
                stale = 0
            if stale > 80:
                # Reset to best, reduce beta (anneal)
                p = best.copy()
                beta = max(beta_start, beta * 0.2)
                alpha = 0.5
                k_frac = min(0.15, k_frac * 3)
                stale = 0
            if stale > 200:
                # Large structural perturbation
                p = _normalize(p * np.exp(rng.normal(0, 0.3, n)))
                beta = beta_start
                alpha = 0.5
                k_frac = 0.15
                stale = 0

    return best, _score_normalized(best) if best is not None else float('inf')


def _coord_descend(p0, rng, time_limit):
    """
    Pairwise coordinate descent: pick two positions, transfer mass between them.
    Returns (best_p, best_score).
    """
    p = _normalize(p0.copy())
    n = len(p)
    best = p.copy()
    best_score = _score_normalized(p)
    deadline = time.monotonic() + time_limit
    stale = 0

    while time.monotonic() < deadline:
        # Pick random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        vi = p[i]
        vj = p[j]
        if vi < 1e-18 and vj < 1e-18:
            continue

        # Try several transfer amounts
        max_t = 0.4 * min(vi, vj)
        tried = 0
        improved = False

        for frac in [0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
            t = frac * max_t
            # Try both directions: i gains, j loses; and vice versa
            for sign in [1, -1]:
                if sign > 0:
                    if vi - t < 0:
                        continue
                else:
                    if vj - t < 0:
                        continue

                p_try = p.copy()
                if sign > 0:
                    p_try[i] = vi + t
                    p_try[j] = vj - t
                else:
                    p_try[i] = vi - t
                    p_try[j] = vj + t

                p_try = _normalize(p_try)
                sc = _score_normalized(p_try)
                tried += 1
                if sc < best_score - 1e-12:
                    best = p_try.copy()
                    best_score = sc
                    p = p_try.copy()
                    improved = True

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 500:
                # Perturb and reset
                p = _normalize(p * np.exp(rng.normal(0, 0.05, n)))
                stale = 0
                if _score_normalized(p) > best_score * 1.01:
                    p = best.copy()

        # Periodic print
        if tried % 5000 < 100:
            rem = deadline - time.monotonic()
            if rem < 0:
                break

    return best, best_score


def _init_uniform(n):
    return np.ones(n) / n


def _init_two_block(n, rng):
    """Two-level block sequence."""
    split = int(rng.uniform(0.2, 0.8) * n)
    ratio = rng.uniform(0.3, 3.0)
    a = np.ones(n)
    a[split:] = ratio
    return _normalize(a)


def _init_cosine(n, rng):
    """Cosine-modulated amplitude."""
    f = rng.uniform(0.5, 5.0)
    A = rng.uniform(0.3, 0.95)
    phi = rng.uniform(0, 2 * np.pi)
    t = np.arange(n) / n
    a = 1.0 + A * np.cos(2 * np.pi * f * t + phi)
    a = np.maximum(a, 0.01)
    return _normalize(a)


def _init_gaussian_bumps(n, rng):
    """Sum of random Gaussian bumps."""
    K = int(rng.integers(2, 6))
    a = np.zeros(n)
    t = np.arange(n) / n
    for _ in range(K):
        center = rng.uniform(0, 1)
        sigma = rng.uniform(0.02, 0.4)
        weight = rng.uniform(0.3, 2.0)
        a += weight * np.exp(-((t - center) / max(sigma, 1e-4)) ** 2)
    if a.max() < 1e-10:
        a = np.ones(n)
    return _normalize(a)


def _init_window(n, rng):
    """Various window functions."""
    mode = int(rng.integers(0, 4))
    t = np.arange(n) / n
    if mode == 0:
        a = np.sin(np.pi * t) ** rng.uniform(0.5, 3.0)
    elif mode == 1:
        a = np.cos(np.pi * (t - 0.5)) ** rng.uniform(0.3, 2.0)
        a = np.abs(a)
    elif mode == 2:
        a = 1.0 - rng.uniform(0.3, 0.9) * (2 * t - 1) ** 2
    else:
        s1 = rng.uniform(0.2, 0.5)
        s2 = rng.uniform(0.5, 0.8)
        a = np.exp(-((t - 0.5) / 0.15) ** 2)
    return _normalize(np.maximum(a, 0.001))


def _init_fibonacci_like(n, rng):
    """Sequence based on fractional parts of golden ratio (low-discrepancy)."""
    golden = (1 + math.sqrt(5)) / 2
    a = np.zeros(n)
    for i in range(n):
        a[i] = 1.0 + rng.uniform(0.2, 0.8) * (0.5 + 0.5 * math.sin(2 * math.pi * (i * golden) % 1.0))
    return _normalize(a)


def _resample(a, n_new):
    """Resample a normalized sequence to a new length via interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    if n_old == 1:
        return np.ones(n_new) / n_new
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    out = np.interp(x_new, x_old, a)
    return _normalize(out)


def _structural_perturb(p, rng):
    """Apply a structural perturbation: reshape via window, split, or merge."""
    n = len(p)
    mode = int(rng.integers(0, 4))
    a = p.copy()
    if mode == 0:
        # Apply a random window on top
        t = np.arange(n) / n
        w = 1.0 + rng.uniform(-0.5, 0.5) * np.cos(2 * np.pi * rng.uniform(0.5, 3) * t)
        a = a * np.maximum(w, 0.01)
    elif mode == 1:
        # Scale a random contiguous block
        start = int(rng.integers(0, n // 2))
        length = int(rng.integers(n // 10, n // 3))
        factor = rng.uniform(0.3, 2.5)
        a[start:start + length] *= factor
    elif mode == 2:
        # Mirror and blend
        mirror = a[::-1]
        alpha = rng.uniform(0.3, 0.7)
        a = alpha * a + (1 - alpha) * mirror
    else:
        # Low-pass filter (smooth) or high-pass (sharpen)
        k = int(rng.integers(2, max(3, n // 50)))
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        if rng.random() < 0.3:
            a = a - a * a.sum() / n  # Detrend slightly
    return _normalize(a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of non-negative floats
    minimizing evaluate_sequence.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + budget - 15.0  # Reserve 15s at end for safety

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31) + 1)

    # Try to load parent sequence
    parent = None
    try:
        parent_raw = height_sequence_1
        parent = _normalize(np.asarray(parent_raw, dtype=float))
        print(f"Loaded parent sequence of length {len(parent)}, score={_exact_score(parent):.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Could not load parent: {e}")

    # ---- PHASE 1: Diverse exploration ----
    print("\n=== Phase 1: Exploration ===")
    candidates = []  # List of (score, length, array)

    # Add parent if available
    if parent is not None:
        parent_score = _exact_score(parent)
        candidates.append((parent_score, len(parent), parent.copy()))
        print(f"Parent score: {parent_score:.6f}")

    # Try multiple lengths and initializations
    lengths_to_try = [2048, 3072, 4096]
    if budget > 2000:
        lengths_to_try.append(1024)

    init_modes = [
        lambda n, r: _init_two_block(n, r),
        lambda n, r: _init_cosine(n, r),
        lambda n, r: _init_gaussian_bumps(n, r),
        lambda n, r: _init_window(n, r),
        lambda n, r: _init_fibonacci_like(n, r),
        lambda n, r: _init_uniform(n),
    ]

    # Short exploration runs
    time_per_run = min(25.0, max(8.0, (budget * 0.12) / (len(lengths_to_try) * 4)))
    explored = 0
    for n in lengths_to_try:
        if time.monotonic() > deadline - 600:
            break
        for mode_idx in range(min(4, len(init_modes))):
            if time.monotonic() > deadline - 600:
                break
            p0 = init_modes[mode_idx](n, rng)
            if parent is not None and mode_idx == 0:
                # Also try resampled parent
                p0 = _resample(parent, n)
                p0 = _normalize(p0 * np.exp(rng.normal(0, 0.05, n)))

            t_rem = deadline - time.monotonic() - 600
            t_alloc = min(time_per_run, max(5.0, t_rem * 0.1))
            p, sc = _mul_descend(p0, rng, t_alloc, beta_start=20.0, beta_end=500.0)
            if p is not None and np.isfinite(sc):
                candidates.append((sc, len(p), p.copy()))
                explored += 1
                print(f"  Explored n={n}, mode={mode_idx}: score={sc:.6f}")

    # Sort and keep best
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best score: {candidates[0][0]:.6f} (length {candidates[0][1]})")

    # ---- PHASE 2: Focused deep optimization ----
    print("\n=== Phase 2: Deep Optimization ===")
    best_arr = candidates[0][2].copy()
    best_score = candidates[0][0]

    # Take top 3 distinct candidates for deep optimization
    top_cands = []
    seen_lengths = set()
    for sc, ln, arr in candidates:
        if len(top_cands) >= 3:
            break
        top_cands.append((sc, ln, arr))

    # Allocate time proportionally (best gets most time)
    rem = deadline - time.monotonic()
    if rem < 100:
        print("Not enough time for Phase 2, jumping to polish.")
    else:
        # Reserve time for Phase 3
        phase3_time = min(500.0, 0.2 * rem)
        phase2_time = rem - phase3_time - 30

        n_runs = len(top_cands)
        fracs = [0.45, 0.30, 0.25][:n_runs]
        if n_runs == 1:
            fracs = [1.0]
        elif n_runs == 2:
            fracs = [0.6, 0.4]

        for idx, (sc0, ln, arr) in enumerate(top_cands):
            if time.monotonic() > deadline - phase3_time - 30:
                break
            t_alloc = fracs[idx] * phase2_time
            t_alloc = max(60.0, t_alloc)
            t_alloc = min(t_alloc, deadline - time.monotonic() - phase3_time - 10)

            print(f"  Deep opt {idx+1}/{n_runs}: n={ln}, start_score={sc0:.6f}, time={t_alloc:.0f}s")
            p, sc = _mul_descend(arr, rng, t_alloc, beta_start=30.0, beta_end=5000.0)
            if p is not None and np.isfinite(sc):
                print(f"    Result: score={sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_arr = p.copy()
                candidates.append((sc, len(p), p.copy()))

    print(f"After Phase 2: best_score={best_score:.6f}")

    # ---- PHASE 3: Polishing ----
    print("\n=== Phase 3: Polishing ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Coordinate descent polishing
        t_coord = min(rem * 0.4, 400.0)
        print(f"  Coordinate descent: {t_coord:.0f}s")
        p, sc = _coord_descend(best_arr, rng, t_coord)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"    After coord descent: {sc:.6f}")

        rem = deadline - time.monotonic()
        if rem > 60:
            # High-beta gradient polishing
            t_grad = min(rem * 0.4, 300.0)
            print(f"  High-beta gradient: {t_grad:.0f}s")
            p, sc = _mul_descend(best_arr, rng, t_grad, beta_start=200.0, beta_end=10000.0)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = p.copy()
                print(f"    After high-beta: {sc:.6f}")

            rem = deadline - time.monotonic()
            if rem > 60:
                # Another round of coordinate descent
                t_coord2 = min(rem * 0.5, 200.0)
                p, sc = _coord_descend(best_arr, rng, t_coord2)
                if p is not None and np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_arr = p.copy()
                    print(f"    After 2nd coord descent: {sc:.6f}")

    # ---- Phase 4: Try upscaling if time remains ----
    rem = deadline - time.monotonic()
    if rem > 200 and best_arr is not None:
        n_cur = len(best_arr)
        n_big = min(8192, 2 * n_cur)
        if n_big > n_cur:
            print(f"\n=== Phase 4: Upscale to n={n_big} ===")
            big = _resample(best_arr, n_big)
            t_big = min(rem * 0.5, 300.0)
            p, sc = _mul_descend(big, rng, t_big, beta_start=50.0, beta_end=5000.0)
            if p is not None and np.isfinite(sc):
                # Convert back to unnormalized for comparison
                sc_full = _exact_score(p)
                if sc_full < best_score:
                    best_score = sc_full
                    best_arr = p.copy()
                    print(f"  Upscaled result: {sc_full:.6f}")
                else:
                    print(f"  Upscale didn't help: {sc_full:.6f} vs {best_score:.6f}")

    # ---- Final validation ----
    print(f"\nFinal best score (internal): {best_score:.6f}, length={len(best_arr)}")

    # Validate with the official evaluator
    final_list = [float(x) for x in best_arr]
    try:
        official = float(evaluate_sequence(final_list))
        print(f"Official evaluate_sequence: {official:.6f}")
        if not np.isfinite(official):
            # Fallback: try a uniform sequence
            print("WARNING: official score is inf, using uniform fallback")
            final_list = [1.0] * len(best_arr)
    except Exception as e:
        print(f"Could not validate: {e}")

    # Ensure output is valid
    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s < 1e-15:
        out = np.ones(2048)
    # Don't normalize here - the evaluator handles scaling
    # But ensure all values are valid
    return [float(x) for x in out]
