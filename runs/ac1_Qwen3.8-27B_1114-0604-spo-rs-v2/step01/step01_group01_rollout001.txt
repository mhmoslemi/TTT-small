    n = len(a)

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_correlation_front(w, p):
    """Compute v[j] = sum_i p[i] * w[j+i] for j=0..len(p)-1."""
    lenp = len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(2 * lenp)
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize(p):

    if p.ndim != 1 or p.size == 0:

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):

    return p / s

def _gradient_topk(p, beta, top_k=10):
    """
    Compute gradient using top-K softmax-weighted peaks.
    Returns (exact_score, smooth_value, gradient_array, max_grad).
    """
    m = p.size
    c = _conv_full(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c  # scale so max of d relates directly to score/2
    exact = float(2.0 * np.max(d)) if d.size else float("inf")

    # Softmax weights over d with temperature controlled by beta
    shift = float(np.max(d))
    e = np.exp(-beta * (shift - d))  # inverse: higher d gets higher weight
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.full(d.size, 1.0 / d.size)
    else:
        w = e / s

    smooth = float(np.dot(w, d))

    # Gradient: h[j] = sum_k w[k] * a[k-j] (valid correlation)
    h = _conv_correlation_front(w, p)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0

    return exact, smooth, h, maxh

def _resample(arr, m):
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-10)
    return _sanitize(out)


def _smooth_array(p, samples=1.0):
    p = _sanitize(p)
    if p is None or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3.0 * samples)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / samples) ** 2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode="same")
    return _sanitize(out)


def _run_mmd(p0, rng, time_limit, beta_start=20.0, beta_end=3000.0, verbose=False):
    """Improved mirror descent with adaptive line search."""
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")
    if time_limit <= 0:
        return p, _exact_score(p)

    m = int(p.size)
    floor = 1e-20 / m
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    beta = float(beta_start)
    alpha = 0.5
    stall = 0
    it = 0
    last_print = -10000

    _, _, h, maxh = _gradient_topk(p, beta, top_k=10)
    cur_smooth = float(np.max(m * _conv_full(p, p)))

    while time.monotonic() < deadline:
        cur_exact, cur_smooth, h, maxh = _gradient_topk(p, beta, top_k=10)

        if not np.isfinite(maxh) or maxh < 1e-15:
            p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.05, m)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor) / p_try.sum()
                e = _exact_score(p_try)
                if np.isfinite(e) and e < best_exact * 1.05:
                    p = p_try
            stall += 1
        else:
            h = np.maximum(h, 0.0)
            h = h / maxh

            # Adaptive line search: try range of step sizes
            step_candidates = []
            for mult in (0.25, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
                a = alpha * mult
                if 1e-5 <= a <= 50.0:
                    step_candidates.append(a)
            if not step_candidates:
                step_candidates = [0.1, 0.5]

            chosen = None
            ch_smooth = cur_smooth
            ch_exact = cur_exact

            for a in sorted(step_candidates, key=lambda x: abs(x - alpha)):
                dec = np.exp(-a * h)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = float(p_try.sum())
                if s <= 0 or not np.isfinite(s):
                    continue
                p_try = p_try / s
                e, s2, _, _ = _gradient_topk(p_try, beta, top_k=10)
                if not np.isfinite(e) or not np.isfinite(s2):
                    continue
                if s2 < ch_smooth - 1e-10:
                    chosen = p_try
                    ch_smooth = s2
                    ch_exact = e
                    alpha = a

            if chosen is not None:
                p = chosen
                stall = 0
                if ch_exact < best_exact:
                    best = p.copy()
                    best_exact = ch_exact
                    if verbose and (it - last_print >= 100):
                        last_print = it
                        elapsed = time.monotonic() - (deadline - time_limit)
                        print(f"  MMD it={it} best={best_exact:.6f} beta={beta:.0f} elapsed={elapsed:.1f}s")
            else:
                stall += 1
                alpha = max(1e-4, alpha * 0.7)
                if stall >= 10:
                    std = min(0.3, 0.05 + 0.01 * stall)
                    p_try = _sanitize(p * np.exp(rng.normal(0.0, std, m)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor) / p_try.sum()
                        e = _exact_score(p_try)
                        if np.isfinite(e) and e < best_exact * 1.08:
                            p = p_try
                        else:
                            p = best.copy()
                    stall = 0

                if stall > 60:
                    p = best.copy()
                    alpha = max(0.1, alpha * 0.5)
                    stall = 0

        it += 1
        if it % 15 == 0:
            beta = min(beta_end, beta * 1.12)

    return best, best_exact


def _coordinate_refine(p0, rng, time_limit):
    """Fine coordinate-wise and block-wise local search."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = int(p.size)
    floor = 1e-20 / n
    p = np.maximum(p, floor) / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    # Precompute factors
    factors_small = (0.8, 0.9, 1.1, 1.2)
    factors_med = (0.6, 0.8, 1.25, 1.5, 2.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.4:
            # Small block
            w = int(rng.integers(1, max(2, n // 500 + 1)))
            factors = factors_small
        elif r < 0.8:
            # Medium block
            w = int(rng.integers(max(2, n // 500), max(3, n // 50 + 1)))
            factors = factors_med
        else:
            # Larger block
            w = int(rng.integers(max(3, n // 50), max(4, n // 10 + 1)))
            factors = (0.7, 0.85, 1.15, 1.3, 1.6)

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        current_best = base
        current_score = best_score if p is best else _exact_score(p)

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, floor)
            s = cand.sum()
            if s <= 0:
                continue
            cand = cand / s
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best.copy()
            if current_score < best_score:
                best = p.copy()
                best_score = current_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.03, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor) / p_try.sum()
                    sc = _exact_score(p_try)
                    if np.isfinite(sc) and sc < best_score * 1.03:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0

        if stagnant > 200:
            p = best.copy()
            stagnant = 0

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Returns best sequence found within time budget.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 60.0:
        budget = 120.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20.0  # leave 20s safety margin

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 42)

    # Target length
    target_n = 4096

    # Collect initial candidates
    candidates = []  # list of (score, array)

    # Try to use previous best
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        parent_arr = None

    if parent_arr is not None:
        sc = _exact_score(parent_arr)
        candidates.append((sc, parent_arr.copy()))
        print(f"Parent sequence: n={parent_arr.size}, score={sc:.6f}")
        # Also resample to target length
        if parent_arr.size != target_n:
            pr = _resample(parent_arr, target_n)
            if pr is not None:
                prs = _exact_score(pr)
                candidates.append((prs, pr.copy()))
                # Also add smoothed version
                ps = _smooth_array(pr, 1.0)
                if ps is not None:
                    pss = _exact_score(ps)
                    candidates.append((pss, ps.copy()))

    # Two-block grid search
    tb = _two_block_grid(target_n)
    if tb is not None:
        sc = _exact_score(tb)
        candidates.append((sc, tb.copy()))
        print(f"Two-block: score={sc:.6f}")
        # Smoothed version
        tbs = _smooth_array(tb, 2.0)
        if tbs is not None:
            scs = _exact_score(tbs)
            candidates.append((scs, tbs.copy()))

    # Diverse random initializations
    for mode in range(8):
        p0 = _initial_sequence(target_n, rng, mode)
        if p0 is not None:
            sc = _exact_score(p0)
            candidates.append((sc, p0.copy()))

    # Sort by score and keep top
    candidates.sort(key=lambda x: x[0])
    print(f"Initial candidates: best={candidates[0][0]:.6f}, {len(candidates)} total")

    # Select top starts (up to 5)
    n_starts = min(5, len(candidates))
    starts = [candidates[i][1] for i in range(n_starts)]

    best_arr = candidates[0][1].copy()
    best_score = candidates[0][0]

    # Phase 1: Main MMD optimization on top starts
    rem = deadline - time.monotonic()
    if rem > 100:
        # Allocate time: give most to the best start
        fracs = [0.35, 0.20, 0.15, 0.15, 0.10][:n_starts]
        # Normalize fracs
        fsum = sum(fracs)
        fracs = [f / fsum for f in fracs]

        for idx in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            talloc = min(fracs[idx] * (rem + 10), rem - 30)
            talloc = max(20.0, talloc)

            p, sc = _run_mmd(
                starts[idx], rng, talloc,
                beta_start=15.0, beta_end=2000.0,
                verbose=(idx == 0)
            )
            if p is not None and np.isfinite(sc):
                if sc < best_score:
                    best_arr = p.copy()
                    best_score = sc
                    print(f"Phase 1 start {idx}: improved to {sc:.6f}")
                candidates.append((sc, p.copy()))

    print(f"After Phase 1: best_score={best_score:.6f}, elapsed={time.monotonic()-start_time:.1f}s")

    # Phase 2: Coordinate refinement
    rem = deadline - time.monotonic()
    if rem > 60:
        t1 = min(0.4 * rem, 600.0)
        p, sc = _coordinate_refine(best_arr, rng, t1)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"After coord refine: {sc:.6f}")

    # Phase 3: Higher-precision MMD polish
    rem = deadline - time.monotonic()
    if rem > 40:
        t2 = min(0.3 * rem, 400.0)
        p, sc = _run_mmd(
            best_arr, rng, t2,
            beta_start=100.0, beta_end=5000.0,
            verbose=True
        )
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"After Phase 3 polish: {sc:.6f}")

    # Phase 4: Final coordinate refinement
    rem = deadline - time.monotonic()
    if rem > 30:
        t3 = min(0.5 * rem, 300.0)
        p, sc = _coordinate_refine(best_arr, rng, t3)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"After final coord refine: {sc:.6f}")

    # Phase 5: Try up-polishing to larger length
    rem = deadline - time.monotonic()
    if rem > 120 and best_arr is not None:
        big_n = 8192
        big = _resample(best_arr, big_n)
        if big is not None:
            sc_big = _exact_score(big)
            print(f"Up-sampled to n={big_n}: score={sc_big:.6f}")
            if sc_big < best_score * 1.01:  # only if competitive
                t4 = min(0.4 * rem, 200.0)
                pb, scb = _run_mmd(big, rng, t4, beta_start=50.0, beta_end=3000.0, verbose=False)
                if pb is not None and np.isfinite(scb) and scb < best_score:
                    best_arr = pb.copy()
                    best_score = scb
                    print(f"After big MMD: {scb:.6f}")

                rem = deadline - time.monotonic()
                if rem > 30:
                    pl, scl = _coordinate_refine(best_arr, rng, min(60.0, 0.3 * rem))
                    if pl is not None and np.isfinite(scl) and scl < best_score:
                        best_arr = pl.copy()
                        best_score = scl
                        print(f"After big coord: {scl:.6f}")

    # Final validation and cleanup
    rem = deadline - time.monotonic()
    if not np.isfinite(best_score):
        best_score = _exact_score(best_arr)

    # Verify with evaluate_sequence if time permits
    final_list = [float(x) for x in best_arr]
    if rem > 5:
        try:
            sc_eval = float(evaluate_sequence(final_list))
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
                print(f"Final evaluate_sequence score: {sc_eval:.6f}")
        except Exception:
            pass

    # Ensure valid output
    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(target_n, dtype=float) / target_n
    else:
        out = out / s  # normalize to sum 1 (doesn't change score)

    print(f"FINAL: n={out.size}, score={_exact_score(out):.6f}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in out]
