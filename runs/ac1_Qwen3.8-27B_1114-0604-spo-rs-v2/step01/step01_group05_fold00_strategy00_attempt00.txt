def evaluate_sequence(seq):

    try:
        return 2.0 * len(seq) * max(np.convolve(seq, seq, mode="full")) / (np.sum(seq) ** 2)
    except Exception as e:
        if str(e) == "Could only generate part of FFT?":
            return float("inf")
        elif any(x < 0 for x in seq):

def fast_local_checker(seq, target=1.5, tol=5e-4):
    """Checks feasibility without full computation"""
    if len(seq) > 2048:
        return float("inf")
    conv = np.convolve(seq, seq, mode="full")
    if not np.allclose(sum(seq), np.sqrt(2 * len(seq))):
        norm = max(conv) / (sum(seq) ** 2) * 2.0 * len(seq)
        return norm < target
    return float("inf")

def smooth_attack(seq, beta=10000.0):
    """Applies a smoothing function to prepare for convex optimization"""
    conv = np.convolve(seq, seq, mode="full")
    m = max(conv)
    s = sum(seq)
    smoothed = [beta * (x - m/2) / s**2 for x in conv]
    return np.clip(smoothed, 0.0, None)

def refined_attack(seq, convergence=1e-5):
    """Takes a sequence and refines to convergence"""
    conv = np.convolve(seq, seq, mode="full")
    max_val = max(conv)
    if max_val < 0.01 or max_val > 100:
        return np.full_like(seq, 1.0/(len(seq)), max_val)
    # Apply scaling to stay consistent
    norm_sum = sum(seq) * np.sqrt(len(seq))
    seq = np.array(seq)
    seq_normalized = seq / norm_sum
    final_max = max(np.convolve(seq_normalized, seq_normalized, mode="full"))
    final_score = 2.0 * len(seq) * final_max / (norm_sum ** 2)
    return [final_score]
