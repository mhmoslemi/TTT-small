"""
Algorithm: Minimize peak autocorrelation of a non-negative sequence.

Strategy:
1. Direct-space Adam optimization with power-mean smooth max.
2. Multi-phase: start with smooth (small p) and sharpen (large p).

def _fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N, N)[:L]
    return np.maximum(c, 0.0)


def _cross_corr(p, a):
    """
    Compute h[j] = sum_k p[k] * a[k-j] for j in [0, len(a)-1].
    This is the cross-correlation of p with a.
    """
    n = len(a)
    L = len(p)
    N = 1 << (L + n - 1 - 1).bit_length()
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N, N)[:n]
    return h


def _score_direct(a):
    """Compute the score 2*n*max_b/sum_a^2."""
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    mb = float(np.max(b))
    return float(2.0 * len(a) * mb / (s * s))


def _power_mean_grad(a, p):
    """
    Compute gradient of F_p(a) = (sum_k b_k^p)^(1/p) w.r.t. a.
    Returns (F_p_value, grad_a).
    """
    n = len(a)
    b = _fft_convolve(a)
    # Clip b to avoid issues with 0^(p-1) when p < 1 (not our case, p >= 2)
    b_pos = np.maximum(b, 1e-30)
    wp = b_pos ** (p - 1)  # weights
    sum_b_p = float(np.sum(wp * b_pos))
    if sum_b_p < 1e-30:
        return 0.0, np.zeros(n)
    F_p = sum_b_p ** (1.0 / p)
    # grad_a[j] = 2 * C * sum_k wp[k] * a[k-j] where C = F_p^(1-p)
    # For Adam, we can drop C (it's a positive scalar that Adam normalizes away)
    h = _cross_corr(wp, a)
    grad = 2.0 * h
    return F_p, grad


def _normalize(a):
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

(p ⋆ a)[j] = sum_m p[j+m] * conj(a[m])

For real signals, this is sum_m p[j+m] * a[m].

The FFT relation: if we zero-pad p to length N and a to length N (N >= L + n - 1), then:
FFT(p ⋆ a) = conj(FFT(p)) * FFT(a)
