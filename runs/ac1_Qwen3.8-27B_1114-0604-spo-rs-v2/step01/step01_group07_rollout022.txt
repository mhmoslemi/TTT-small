"""
Low-peak nonnegative step-function search using mirror descent with 
adaptive active-set subgradient and backtracking line search.
Initializes from multiple parametric families (sinusoidal, half-sine, 
raised cosine) and refines with projected mirror descent.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))

def _make_sine(n, amp, mode):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:

def _mirror_descent(a_init, budget_s, lr_init=5.0, max_active=40, verbose=False):
    """Mirror descent with adaptive active set and backtracking line search."""
    arr = np.asarray(a_init, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr /= s
    
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    start = time.monotonic()
    lr = lr_init
    it = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() - start < budget_s:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        max_c = float(np.max(c))
        if not np.isfinite(max_c):
            break
        
        # Determine active set: lags where b_k is close to max
        threshold = max_c * 0.90  # 90% of max
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) > max_active:
            # Use top max_active
            top_idx = np.argsort(c)[-max_active:]
            active_lags = top_idx
        
        if len(active_lags) == 0:
            break
        
        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active_lags:
            k = int(k)
            m_min = max(0, k - n + 1)
            m_max = min(n - 1, k)
            if m_max >= m_min:
                idx_in_a = np.arange(k - m_max, k - m_min + 1)
                idx_in_grad = np.arange(m_min, m_max + 1)
                grad[idx_in_grad] += 2.0 * arr[idx_in_a]
        grad /= float(len(active_lags))
        
        # Project to tangent plane of simplex
        g = grad - np.mean(grad)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Normalize gradient for stable learning rate
        g = g / g_norm
        
        # Backtracking line search
        found = False
        for attempt in range(10):
            a_new = arr * np.exp(-lr * g)
            a_new = np.where(np.isfinite(a_new), a_new, 0.0)
            a_new = np.clip(a_new, 0.0, 1000.0)
            s_new = float(a_new.sum())
            if not np.isfinite(s_new) or s_new < 0.01:
                lr *= 0.5
                continue
            a_new /= s_new
            sc = _internal_score(a_new)
            if np.isfinite(sc) and sc < best_score:
                arr = a_new
                best = arr.copy()
                best_score = sc
                last_improve = it
                stall_count = 0
                # Slightly increase lr
                lr = min(lr * 1.02, 50.0)
                found = True
                break
            else:
                lr *= 0.5
                if lr < 0.01:
                    break
        
        if not found:
            stall_count += 1
            if stall_count > 200:
                # Restart with perturbation
                arr = best.copy()
                noise = np.random.randn(n) * 0.01
                arr = arr + noise
                arr = np.clip(arr, 0.0, None)
                arr /= arr.sum()
                lr = lr_init
                stall_count = 0
        
        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"  MD it={it} score={best_score:.8f} lr={lr:.4f} active={len(active_lags)} time={elapsed:.1f}s")
    
    return best, best_score

"""
Low-peak nonnegative step-function search using mirror descent with
adaptive active-set subgradient and backtracking line search.
Initializes from multiple parametric families and refines with
projected mirror descent in the simplex.
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _internal_score_and_conv(a):
    """Return (score, convolution, max_val) for a normalized nonneg array."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return np.inf, np.array([]), 0.0
    normed = arr / s
    c = _fft_convolve(normed)
    mx = float(np.max(c))
    sc = float(2.0 * n * mx / (s * s))
    return sc, c, mx


def _make_sine(n, amp, mode):
    """Anti-palindromic sinusoidal shape."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _make_half_sine(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / float(n))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_raised_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (i + 0.5) / float(n)))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(2.0 * (i + 0.5) / float(n) - 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_burns(n):
    """JW Burns window."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / float(n)
    a = (1.0 - x) ** 2 + 0.038376 * ((1.0 - x) ** 2) ** 0.7
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _compute_subgradient(a, active_lags):
    """
    Compute the average subgradient of max_k b_k w.r.t. a.
    grad[m] = (1/|A|) * sum_{k in A} 2*a[k-m] (if 0<=k-m<n, else 0)
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    cnt = 0
    for k in active_lags:
        k = int(k)
        # m ranges such that 0 <= k-m < n, i.e., k-n+1 <= m <= k
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_max >= m_min:
            idx_a = np.arange(k - m_max, k - m_min + 1)
            idx_g = np.arange(m_min, m_max + 1)
            grad[idx_g] += 2.0 * a[idx_a]
            cnt += 1
    if cnt > 0:
        grad /= float(cnt)
    return grad


def _mirror_descent(a_init, budget_s, lr_init=3.0, max_active=50,
                    threshold_frac=0.92, verbose=True):
    """
    Mirror descent (multiplicative update) with adaptive active set
    and backtracking line search. Minimizes max(b_k) subject to
    sum(a)=1, a>=0.
    """
    arr = np.asarray(a_init, dtype=np.float64).copy()
    n = arr.size
    arr = np.clip(arr, 0.0, None)
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr /= s

    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, np.inf

    start = time.monotonic()
    lr = lr_init
    it = 0
    stall = 0
    best_lr = lr

    while time.monotonic() - start < budget_s:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        max_c = float(np.max(c))
        if not np.isfinite(max_c) or max_c <= 0:
            break

        # Adaptive active set
        thresh = max_c * threshold_frac
        active = np.where(c >= thresh)[0]
        if len(active) > max_active:
            active = np.argsort(c)[-max_active:]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient
        grad = _compute_subgradient(arr, active)
        g = grad - np.mean(grad)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        # Backtracking line search
        improved = False
        for _ in range(20):
            exponent = -lr * g
            exponent = np.clip(exponent, -5.0, 5.0)
            a_new = arr * np.exp(exponent)
            a_new = np.where(np.isfinite(a_new), a_new, 0.0)
            a_new = np.clip(a_new, 0.0, 1000.0)
            s_new = float(a_new.sum())
            if not np.isfinite(s_new) or s_new < 0.01:
                lr *= 0.5
                continue
            a_new /= s_new
            sc = _internal_score(a_new)
            if np.isfinite(sc) and sc < best_score:
                arr = a_new
                best = a_new.copy()
                best_score = sc
                improved = True
                best_lr = lr
                lr = min(lr * 1.05, 50.0)
                stall = 0
                break
            else:
                lr *= 0.5
                if lr < 0.001:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Perturbation restart from best
                arr = best.copy()
                noise = np.random.randn(n) * 0.003
                arr = arr + noise
                arr = np.clip(arr, 0.0, None)
                s2 = arr.sum()
                if s2 > 0.01:
                    arr /= s2
                else:
                    arr = best.copy()
                lr = best_lr * 0.5
                stall = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it} score={best_score:.8f} lr={lr:.4f} "
                  f"|A|={len(active)} t={elapsed:.1f}s")

    return best, best_score


def _optimize_sine_params(n, deadline):
    """Optimize sinusoidal amplitude and mode for a given length."""
    best_score = np.inf
    best_a = None

    for mode in [1, 2, 3]:
        for amp in np.linspace(0.2, 1.4, 25):
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, amp, mode)
            sc = _internal_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # Local refinement of best amplitude
    if best_a is not None and time.monotonic() < deadline:
        # We know it's a sine, so scan amplitude more finely
        for mode in [1, 2]:
            for amp in np.linspace(0.3, 1.3, 60):
                if time.monotonic() > deadline:
                    break
                a = _make_sine(n, amp, mode)
                sc = _internal_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    np.random.seed(seed)

        # Various shapes

        ]
        for name, a in shapes:

    # Try global variable if available
    try:

    # Sort by score

    print(f"[Phase 1] {len(candidates)} candidates. Best: "

    # --- Phase 2: Refine top candidates with mirror descent ---

    best_arr = candidates[0][1].copy()

    # Allocate time budget among top candidates
    # Give the best candidate the most time
    num_refine = min(5, len(candidates))
    # Time fractions: 0.45, 0.20, 0.10, 0.08, 0.05
    fractions = [0.45, 0.20, 0.10, 0.08, 0.05]

            budget = remaining - 60

        print(f"  [{i+1}/{num_refine}] Refining '{name}' "
              f"(init_score={sc_init:.8f}) budget={budget:.0f}s")

        t_start = time.monotonic()
        refined, r_score = _mirror_descent(
            a_init, budget,
            lr_init=3.0,
            max_active=50,
            threshold_frac=0.92,

        elapsed = time.monotonic() - t_start

        if np.isfinite(r_score) and r_score < best_sc:
            best_arr = refined.copy()
            best_sc = r_score
            print(f"  -> Improved! score={r_score:.8f} "

        else:
            print(f"  -> No improvement (best stays {best_sc:.8f})")

    print(f"[Phase 2] Done. Best internal score={best_sc:.8f}")

    # --- Phase 3: Final polish with smaller learning rate ---

    if remaining > 120:
        polish_budget = min(remaining - 60, 200.0)
        print(f"[Phase 3] Final polish with budget={polish_budget:.0f}s")
        polished, p_score = _mirror_descent(
            best_arr, polish_budget,
            lr_init=1.0,
            max_active=30,
            threshold_frac=0.95,
            verbose=True
        )
        if np.isfinite(p_score) and p_score < best_sc:
            best_arr = polished.copy()
            best_sc = p_score
            print(f"[Phase 3] Final polish improved to {p_score:.8f}")

    # --- Phase 4: Validate ---
    print("[Phase 4] Validation...")
    final = np.clip(best_arr, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final > 0.01:
        final /= s_final
    else:
        final = _make_sine(2048, 0.85, 1)

    try:
        real_score = float(evaluate_sequence(final.tolist()))
        print(f"[Phase 4] Validated: {real_score:.8f} "
              f"(n={final.size})")
    except Exception as e:
        print(f"[Phase 4] Validation error: {e}")
        real_score = np.inf

    if not np.isfinite(real_score):
        # Fallback
        final = _make_sine(2048, 0.85, 1)
        try:
            real_score = float(evaluate_sequence(final.tolist()))
            print(f"[fallback] score={real_score:.8f}")
        except Exception:
            pass

    total_time = time.monotonic() - t0
    print(f"[done] total_time={total_time:.1f}s, "
