```python
from scipy.optimize import linprog

def lp_step(p, M):
    """
    Given p (non-negative, sum=1) and M = max(conv(p,p)),
    find q that maximizes sum(q) subject to q>=0 and conv(p,q)_k <= M for all k.
    """
    n = len(p)
    # conv(p, q)_k = sum_j p_j * q_{k-j}
    # For k = 0, ..., 2n-2:
    # conv(p,q)_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} p_j * q_{k-j}
    #
    # Let's write this as A @ q <= M * ones(2n-1)
    # where A[k, j] = p_{k-j} if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = p[idx]
    
    c = -np.ones(n)  # minimize -sum(q)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        q = res.x
        return q
    return None
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    n = a.size
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa*fa, fsize)[:size]

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def conv_matrix(p):
    """Build the matrix A such that (A @ q)[k] = conv(p, q)[k]."""
    n = len(p)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = p[idx]
    return A

def lp_step(p, M):
    n = len(p)
    A = conv_matrix(p)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), 
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def lp_iteration(p, max_iters=10000, blend=1.0, time_budget=60.0):
    """Run LP-based iteration to improve p."""
    n = len(p)
    p = p.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()
    best_p = p.copy()
    best_sc = normed_score(p)
    
    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = full_conv(p)
        M = float(np.max(b))
        q = lp_step(p, M)
        if q is None or q.sum() <= 0:
            break
        
        q = q / q.sum()
        p_new = (1-blend)*p + blend*q
        p_new = np.maximum(p_new, 0)
        p_new = p_new / p_new.sum()
        
        sc = normed_score(p_new)
        if sc < best_sc:
            best_sc = sc
            best_p = p_new.copy()
        p = p_new
    
    return best_p, best_sc
```

```python
    n = a.size
    size = 2 * n - 1

    fa = np.fft.rfft(a, fsize)

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(p):
    """Build A such that A @ q = conv(p, q) (linear convolution)."""
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    pf = p[::-1]
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        A[k, lo:hi + 1] = pf[n - 1 - hi:n - 1 - lo][::-1]
    return A

        shift -= shift.max()

            w[int(np.argmax(b))] = 1.0
        else:
            w /= s

        vel = beta * vel + (1.0 - beta) * g

def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_bell(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = n * 0.2
    a = np.exp(-0.5 * ((i - c) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.1 * rng.random())
    c2 = n * (0.65 + 0.1 * rng.random())
    sigma = n * (0.04 + 0.06 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 4))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_decreasing(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0)
    return a / a.sum()


def init_medium_peak(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - c) / (n * 0.15)) ** 2) + 0.3 * np.exp(-0.5 * ((i - c) / (n * 0.4)) ** 2)
    a += 0.01
    return a / a.sum()


def get_inits(n, rng):
    inits = []
    inits.append(init_uniform(n))
    inits.append(init_u_shape(n))
    inits.append(init_bell(n))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_sided(n))
    inits.append(init_cosine(n, rng))
    inits.append(init_decreasing(n))
    inits.append(init_medium_peak(n))
    return inits


def polish_local(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    scale = 0.03
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        n_trials = max(5, int(30 * time_budget / max(time_budget, 1)))
        for _ in range(n_trials):
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 4:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Primary method: LP-based fixed-point iteration (convex relaxation).
    Secondary: smoothed subgradient descent with momentum.
    Combines multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_overall = None
    best_overall_sc = np.inf

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Define n values to try
    n_list = [64, 96, 128, 160, 192, 256, 320, 384, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 2048:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time proportionally to 1/sqrt(n) (faster per-iter for small n)
    raw_w = [1.0 / np.sqrt(nv) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    # Phase 1: LP iterations
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.75
        inits = get_inits(nv, rng)

        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        n_inits = len(inits)
        inits_per = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            tb = min(inits_per, deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break

            # Run LP iteration
            a_gp, sc_gp = run_lp_iteration(a_init, tb * 0.7, blend=1.0, verbose=(nv <= 128))

            if sc_gp < best_overall_sc:
                best_overall_sc = sc_gp
                best_overall = a_gp.copy()
                print(f"[LP] n={nv}, init={ii}, score={sc_gp:.6f} (BEST)")

            # If time remains, do a short subgradient polish from LP result
            remain = deadline - time.monotonic() - 2.0
            if remain > 3.0 and sc_gp < best_overall_sc + 0.1:
                sg_time = min(remain * 0.2, tb * 0.3)
                if sg_time > 1.0:
                    a_sg, sc_sg = subgrad_optimize(a_gp, sg_time, rng)
                    if sc_sg < best_overall_sc:
                        best_overall_sc = sc_sg
                        best_overall = a_sg.copy()
                        print(f"[SG] n={nv}, init={ii}, score={sc_sg:.6f} (BEST)")

            if ii % 3 == 0:
                print(f"  n={nv}, init={ii}/{n_inits}, best_sc={best_overall_sc:.6f}, "
                      f"time_left={deadline - time.monotonic():.0f}s")

        print(f"[phase1] n={nv} done. Overall best: {best_overall_sc:.6f}")

    # Phase 2: Focused optimization on best n
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        best_n = best_overall.size
        print(f"[phase2] Focused opt on n={best_n}, remain={remain:.1f}s")

        # Multiple restarts around best solution with LP
        for rx in range(5):
            if time.monotonic() > deadline - 5.0:
                break
            tb = min(remain / (5 - rx), 120.0)
            if tb < 3.0:
                break
            # Perturb the best solution
            pert = best_overall + 0.05 * rng.standard_normal(best_n)
            pert = project_simplex(pert)
            a_r, sc_r = run_lp_iteration(pert, tb, blend=1.0)
            if sc_r < best_overall_sc:
                best_overall_sc = sc_r
                best_overall = a_r.copy()
                print(f"  [restart {rx}] score={sc_r:.6f} (BEST)")
            else:
                print(f"  [restart {rx}] score={sc_r:.6f}")

    # Phase 3: Final polish
    if best_overall is not None and time.monotonic() < deadline - 3.0:
        remain = deadline - time.monotonic() - 2.0
        if remain > 5.0:
            print(f"[phase3] Polish, remain={remain:.1f}s")

            # Subgradient polish
            sg_pol_time = min(remain * 0.5, 60.0)
            if sg_pol_time > 3.0:
                a_p, sc_p = subgrad_optimize(best_overall, sg_pol_time, rng)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [polish-SG] {sc_p:.6f}")

            # Random perturbation polish
            if time.monotonic() < deadline - 3.0:
                rp_time = min(deadline - time.monotonic() - 2.0, 30.0)
                if rp_time > 2.0:
                    a_p2, sc_p2 = polish_local(best_overall, rp_time, rng)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"  [polish-RP] {sc_p2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
