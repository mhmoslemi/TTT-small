def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a, target_sum=1.0):
    s = a.sum()
    if s < 1e-10:
        return a
    return a * (target_sum / s)

def ift_flatten(a, alpha=0.5, n_iter=50):
    """Iteratively flatten the magnitude spectrum to reduce autocorrelation peak."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    a = a.copy()

        s = a_new.sum()
        if s > 1e-10:

        a = a_new
    return a

    n = len(a)
    S = a.sum()

    idx = np.argpartition(c, -top_k)[-top_k:]
    vals = c[idx]
    # Weight by proximity to max
    max_val = vals.max()
    temp = 0.1 * max_val if max_val > 1e-15 else 1.0
    wts = np.exp((vals - max_val) / temp)

    # Gradient of weighted max

        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue

        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue

        # a[ki-j] for j in [j_start, j_end] is a segment of a, reversed
        grad[j_start:j_end+1] += wi * 2.0 * a[j_start:j_end+1][::-1]

    grad_score = (2.0 * n) * (grad / (S * S) - 2.0 * M * a / (S * S * S))
    
    a_new = a - lr * grad_score
    a_new = np.maximum(a_new, 0.0)
    return a_new

    if best_a is not None and time.monotonic() < deadline:
        sc, aa = run_adam(best_a, deadline, lr_init=0.01, topk_init=1,
                          topk_final=1, kick_interval=500, kick_gamma=1.05,
                          patience=500)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print("Phase4 (polish) best=%.6f" % best_score)

    # Fallback
    if best_a is None:
        best_a = np.ones(1024)

    out = [float(x) for x in best_a]
    verified = evaluate_sequence(out)
    elapsed = time.monotonic() - start
    print("Final: internal=%.6f verified=%.6f len=%d time=%.1fs" %
          (best_score, verified, len(out), elapsed))
    return out
