```python
import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    """Full convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Score assuming sum(a)=1."""
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g (unnormalized)."""
    n = a.size
    # Constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    # We can build it efficiently using the Toeplitz structure
    
    # Build A: for each k (0..2n-2) and j (0..n-1):
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # This is a banded matrix.
    
    # Actually, let's build it as a sparse matrix for efficiency
    # For each j, the non-zero entries in column j are at rows k = j, j+1, ..., j+n-1
    # A[j+i, j] = a[i] for i = 0, ..., n-1
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        for i in range(n):
            A[j+i, j] = a[i]
    A = A.tocsr()
    
    # c = -ones(n) (minimize -sum(g))
    c = -np.ones(n)
    
    # Constraints: A @ g <= M * ones(2n-1)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x

def lp_iterate(a, n_iters, t, time_budget, rng):
    """Run LP-based iterations."""
    a = a / a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = full_conv(a)
        M = float(np.max(b))
        
        g = solve_lp(a, M)
        if g is None or g.sum() <= 0:
            break
        
        g = g / g.sum()
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = normed_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_sc
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))

def build_lp_matrix(a):
    """Build the LP constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = a.size
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A[k,j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = a[k-j]
    return A

def solve_lp(a, M):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0. Returns g."""
    n = a.size
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_method(a, time_budget, rng, t_init=0.2, verbose=False):
    """LP-based iterative optimization."""
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    n = a.size
    
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t = t_init
    t0 = time.monotonic()
    last_improve = 0
    iters = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_budget:
        b_conv = full_conv(a)
        M = float(np.max(b_conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum <= 1e-10:
            break
        
        g = g / g_sum
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = normed_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            last_improve = iters
            stall_count = 0
            if verbose:
                print(f"  iter {iters}: score={sc:.6f} M={M:.6f}")
        else:
            stall_count += 1
        
        a = a_new
        iters += 1
        
        if stall_count > 50:
            # Reduce step size
            t *= 0.8
            if t < 0.01:
                # Try perturbation
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                t = t_init * (0.5 + 0.5 * rng.random())
                stall_count = 0
        
        if iters > 0 and iters % 200 == 0 and verbose:
            print(f"  iter {iters}: best={best_sc:.6f} current_M={M:.6f}")
    
    return best_a, best_sc
```

```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP direction-finding (maximize sum(g) s.t. (a*g)_k <= M),
line search, multiple sequence lengths, diverse initializations,
and local random polish. The LP leverages the banded Toeplitz structure
of the convolution constraint matrix for efficiency.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))


def build_A(a):
    """Build LP constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0.
    Uses the banded structure: column j is a shifted version of a."""
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(A, M, n):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0.0)

                a = cand.copy()
                improved = True

                break

            # Use standard LP for most, aggressive for the first init
            if ii == 0 and 'prev' in str(a_init[:3]) or (prev_a is not None and prev_a.size == nv and ii == 0):
                a_best, sc_best = lp_optimize_aggressive(a_init, tb, rng)
            else:
                a_best, sc_best = lp_optimize(a_init, tb, rng)

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"[BEST] n={nv}, init={ii}, score={sc_best:.6f}")
            else:
                print(f"       n={nv}, init={ii}, score={sc_best:.6f}")

        print(f"[n_done] n={nv} complete. Overall best: {best_overall_sc:.6f}")

    # Polish phase: refine the best solution found
    if best_overall is not None and time.monotonic() < deadline - 40:
        remain = deadline - time.monotonic() - 15.0
        if remain > 30:
            print(f"\n[polish] Starting polish phase. n={best_overall.size}, remaining={remain:.0f}s")

            # LP polish (short)
            lp_polish_time = min(remain * 0.4, 100.0)
            if lp_polish_time > 10:
                a_p, sc_p = lp_optimize(best_overall, lp_polish_time, rng)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"[polish] LP: {sc_p:.6f}")

            # Aggressive LP polish
            if time.monotonic() < deadline - 20:
                ag_time = min(remain * 0.25, 60.0)
                if ag_time > 10:
                    a_ag, sc_ag = lp_optimize_aggressive(best_overall, ag_time, rng)
                    if sc_ag < best_overall_sc:
                        best_overall_sc = sc_ag
                        best_overall = a_ag.copy()
                        print(f"[polish] Aggressive: {sc_ag:.6f}")

            # Random polish
            if time.monotonic() < deadline - 15:
                r_time = min(remain * 0.2, 50.0)
                if r_time > 5:
                    a_r, sc_r = random_polish(best_overall, r_time, rng)
                    if sc_r < best_overall_sc:
                        best_overall_sc = sc_r
                        best_overall = a_r.copy()
                        print(f"[polish] Random: {sc_r:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(64) / 64
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
