import numpy as np
import time

def _fft_convolve(a):
    n = a.size
    need = 2*n - 1
    N = 1 << (need - 1).bit_length()
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:need]

def _score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * a.size * b.max() / (s*s)

def _local_optimize(a0, budget_s):
    # Normalize
    a = a0.copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    best = a.copy()
    best_score = _score(best)
    
    start = time.monotonic()
    K = 32  # Number of active constraints
    eta = 0.5  # Learning rate
    
    while time.monotonic() - start < budget_s:
        b = _fft_convolve(a)
        bmax = b.max()
        
        # Get active set (top K values)
        idx = np.argpartition(b, -K)[-K:]
        
        # Compute gradient: for each active k, grad_i += 2*a[k-i]
        n = a.size
        grad = np.zeros(n)
        for k in idx:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                # grad[i] += 2 * a[k-i] for i in [i_min, i_max]
                j = k - np.arange(i_min, i_max + 1)
                grad[i_min:i_max+1] += 2.0 * a[j]
        
        # Normalize gradient
        grad /= K
        
        # Project to simplex tangent (remove mean)
        grad -= grad.mean()

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]
