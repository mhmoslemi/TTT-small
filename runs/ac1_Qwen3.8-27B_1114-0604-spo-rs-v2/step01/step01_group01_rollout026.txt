```python
import numpy as np
import time
from scipy.optimize import linprog

def _conv_full(a, b):
    """Full convolution using FFT for efficiency."""
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([])
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b)
    # FFT-based
    L = 1
    while L < nout:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:nout]

def _score(p):
    """Compute the score for a normalized non-negative sequence."""
    n = len(p)
    c = _conv_full(p, p)
    if len(c) == 0:
        return float('inf')
    max_b = np.max(c)
    sum_a = np.sum(p)
    if sum_a < 0.01:
        return float('inf')
    return 2 * n * max_b / (sum_a ** 2)

def _normalize(p):
    """Normalize to sum 1, ensuring non-negativity."""
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return None
    return p / s
```

```python
def _lp_step(a, active_lags, d_bound=0.5):
    """
    Solve the LP to find a step direction d that minimizes the
    linearized max convolution over active lags.
    
    Returns the step d (or None if LP fails).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    
    # Variables: d_0, ..., d_{n-1}, t
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1) For each active lag k: 2*sum_j a_{k-j} d_j - t <= -c_k
    #    (i.e., t >= c_k + 2*(conv(a,d))_k)
    rows = []
    cols = []
    vals = []
    b_list = []
    
    for k in active_lags:
        # (conv(a,d))_k = sum_j a_{k-j} d_j where k-j in [0,n-1]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            ai = a[k - j]
            if ai > 1e-15:
                rows.append(len(b_list))
                cols.append(j)
                vals.append(2.0 * ai)
        # -t term
        rows.append(len(b_list))
        cols.append(n)
        vals.append(-1.0)
        b_list.append(-c[k])
    
    # 2) d_i >= -a_i  =>  -d_i <= a_i
    for i in range(n):
        rows.append(len(b_list))
        cols.append(i)
        vals.append(-1.0)
        b_list.append(a[i])
    
    # 3) d_i <= d_bound * a_i (upper bound, optional)
    for i in range(n):
        rows.append(len(b_list))
        cols.append(i)
        vals.append(1.0)
        b_list.append(d_bound * a[i])
    
    A_ub = sp.coo_matrix((vals, (rows, cols)), shape=(len(b_list), n + 1)).tocsr()
    b_ub = np.array(b_list)
    
    # Equality: sum d_i = 0
    A_eq = sp.coo_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1)).tocsr()
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        return d
    return None
```

```python
"""
LP-based linearized descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses sparse LP with scipy HiGHS, multiple starting points, and line search.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    nout = len(a) + len(b) - 1
    L = 1
    while L < nout:
        L <<= 1
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:nout]


def _autoconv(a):
    """Auto-convolution of a with itself."""
    return _fft_convolve(a, a)


def _compute_score(p):
    """Score: 2*n*max(conv(p,p))/sum(p)^2. Assumes p >= 0 and sum(p) >= 0.01."""
    n = len(p)
    c = _autoconv(p)
    s = float(p.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * c.max() / (s * s))


def _normalize(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 0.0:
        return None
    return p / s


def _make_two_block(n, p_frac, ratio):
    """Two-block sequence: 1 for first p_frac*n elements, ratio for rest."""
    a = np.ones(n)
    split = int(round(p_frac * n))
    a[split:] = ratio
    return _normalize(a)
```

```python
def _solve_lp_step(a, active_lags, max_step=1.0):
    """
    Solve LP to find step direction d minimizing linearized max auto-conv
    over active lags.
    
    min t
    s.t. c_k + 2*(conv(a,d))_k <= t  for k in active_lags
         -a_i <= d_i <= max_step * a_i
         sum(d_i) = 0
    """
    n = len(a)
    c = _autoconv(a)
    n_act = len(active_lags)
    
    # Build sparse A_ub
    row_list = []
    col_list = []
    val_list = []
    b_list = []
    
    # Constraint 1: active lag constraints
    for idx, k in enumerate(active_lags):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            coeff = 2.0 * a[k - j]
            if coeff > 1e-14:
                row_list.append(idx)
                col_list.append(j)
                val_list.append(coeff)
        # -t
        row_list.append(idx)
        col_list.append(n)
        val_list.append(-1.0)
        b_list.append(-c[k])
    
    # Constraint 2: d_i >= -a_i  =>  -d_i <= a_i
    for i in range(n):
        row_list.append(len(b_list))
        col_list.append(i)
        val_list.append(-1.0)
        b_list.append(a[i])
    
    # Constraint 3: d_i <= max_step * a_i
    for i in range(n):
        row_list.append(len(b_list))
        col_list.append(i)
        val_list.append(1.0)
        b_list.append(max_step * a[i])
    
    n_rows = len(b_list)
    A_ub = sp.coo_matrix((val_list, (row_list, col_list)), shape=(n_rows, n + 1)).tocsr()
    b_ub = np.array(b_list, dtype=float)
    
    # Equality: sum d_i = 0
    A_eq = sp.coo_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1)).tocsr()
    b_eq = np.array([0.0])
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

```python
def _lp_descent(a0, time_left=600):
    a = _normalize(a0).copy()
    n = len(a)
    best = a.copy()
    best_score = _compute_score(a)
    deadline = time.monotonic() + time_left
    
    stall_count = 0
    max_step = 1.0
    active_frac = 0.95
    
    while time.monotonic() < deadline:
        c = _autoconv(a)
        c_max = c.max()
        
        # Find active lags
        threshold = active_frac * c_max
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        
        # Also include the top few lags even if below threshold
        top_k = min(50, len(c))
        top_indices = np.argsort(c)[-top_k:]
        active_lags = np.unique(np.concatenate([active_lags, top_indices]))
        
        if len(active_lags) > 100:
            # Keep only top 100
            c_at_active = c[active_lags]
            top_100 = np.argsort(c_at_active)[-100:]
            active_lags = active_lags[top_100]
        
        # Solve LP
        d = _solve_lp_step_fast(a, active_lags, max_step)
        
        if d is None:
            stall_count += 1
            continue
        
        # Check if the step is actually useful
        d_norm = np.abs(d).max()
        if d_norm < 1e-12:
            stall_count += 1
            continue
        
        # Line search
        best_alpha = 0.0
        cur_score = best_score
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = a + alpha * d
            if (a_try < 0).any():
                continue
            a_try = _normalize(a_try)
            s_try = _compute_score(a_try)
            if s_try < cur_score - 1e-12:
                best_alpha = alpha
                cur_score = s_try
        
        if best_alpha > 0:
            a = _normalize(a + best_alpha * d)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
            stall_count = 0
        else:
            stall_count += 1
            # Try reducing step size
            if stall_count % 5 == 0:
                max_step = max(0.1, max_step * 0.5)
            if stall_count > 30:
                # Random restart
                d_rand = np.random.randn(n) * 0.05
                a_try = _normalize(a + d_rand)
                if a_try is not None:
                    s_try = _compute_score(a_try)
                    if s_try < best_score * 1.02:
                        a = a_try
                    else:
                        a = best.copy()
                stall_count = 0
                max_step = 0.5
        
        # Adaptively increase active_frac (include more lags as we converge)
        if stall_count == 0:
            active_frac = max(0.8, active_frac - 0.01)
    
    return best, best_score
```

```python
        best_score = _score(best_seq)
    
    print(f"Final best: {best_score:.6f}")
    return [float(x) for x in best_seq]
```
