= (if m in S(k)) f[k-m] + (if k-m in S(k)) f[k-m]

Now, m in S(k) means max(0,k-n+1) <= m <= min(k, n-1), which means 0 <= m < n and 0 <= k-m < n.
k-m in S(k) means max(0,k-n+1) <= k-m <= min(k, n-1), which means 0 <= k-m < n and 0 <= m < n.

These are the same condition! So both terms are present iff 0 <= k-m < n.

Therefore: ∂c[k]/∂f[m] = 2*f[k-m] if 0 <= k-m < n, else 0.

(When k=2m, we get 2*f[m], which is correct as verified above.)

So the gradient of the peak c[k*] with respect to f is:
`grad[m] = 2 * f[k*-m]` if 0 <= k*-m < n, else 0.

This is essentially 2 times a shifted (and possibly reflected) copy of f.

**Steepest descent on the peak**: To decrease c[k*], we want to move f in the direction -grad, i.e., decrease f[m] where f[k*-m] is large. This means we want to reduce f at positions that are "correlated" with the peak lag.

But we need to maintain sum(f) = 1 and f >= 0. So we need a projected gradient step or a mirror descent step.

def _autocorr(f):
    """Compute full convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval_local(f):
    """Local evaluation (assumes f is numpy array, non-negative, sum=1)."""
    c = _autocorr(f)
    return 2.0 * len(f) * c.max()


def _gradient_peak(f, k_star):
    """Gradient of c[k_star] w.r.t. f. grad[m] = 2*f[k_star - m] if valid."""
    n = len(f)
    grad = np.zeros(n)
    # k_star - m is in [0, n-1] means m is in [k_star-n+1, k_star]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        # grad[m] = 2 * f[k_star - m] for m in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * f[k_star - indices]
    return grad


def _soft_gradient(f, fT, T):
    """
    Soft-max gradient: weighted sum of peak gradients.
    w_k = exp(c[k]/T) / sum(exp(c[j]/T))
    grad = sum_k w_k * grad_k
    """
    n = len(f)
    c = _autocorr(f)
    cs = c / T
    cs -= cs.max()
    w = np.exp(cs)
    w /= w.sum()
    
    # grad[m] = 2 * sum_k w[k] * f[k-m] (for valid k-m)
    # This is 2 * correlate(w, f, mode='valid')... let me check
    # sum_k w[k] * f[k-m]: this is a cross-correlation of w and f
    # Using np.correlate(w, f, 'valid') gives sum_j w[j] * f[j + len(w) - len(f) - 1 + m]
    # Hmm, this is getting complicated. Let me just do it directly.
    
    # Actually, grad[m] = 2 * sum_{k: 0<=k-m<n} w[k] * f[k-m]
    # Let j = k-m, so k = j+m: grad[m] = 2 * sum_{j: 0<=j<n, 0<=j+m<2n-1} w[j+m] * f[j]
    # = 2 * sum_j f[j] * w[j+m]
    # This is 2 * (f convolved with w)[m]... no wait.
    # = 2 * sum_j f[j] * w[m+j]
    # This is the cross-correlation of f and w at lag m.
    # np.correlate(f, w, 'full')[m + len(w) - 1] ... hmm
    # Actually, cross-correlation R[m] = sum_j f[j] * conj(w[j-m]) for complex
    # For real: R[m] = sum_j f[j] * w[j-m]
    # We want sum_j f[j] * w[j+m] = R[-m] = R[2n-2-m] (by symmetry if w is symmetric)
    # But w is not necessarily symmetric.
    
    # Let me just compute it via FFT for efficiency.
    # grad[m] = 2 * sum_j f[j] * w[j+m]
    # This is the convolution of f (reversed) with w, evaluated at specific points.
    # Or: if we define f_rev[j] = f[n-1-j], then
    # sum_j f[j] * w[j+m] = sum_j f_rev[n-1-j] * w[j+m]
    # Let i = n-1-j: sum_i f_rev[i] * w[n-1-i+m]
    # = (f_rev convolved with w)[n-1+m]
    # Hmm, this is getting messy. Let me just use the FFT approach.
    
    # Alternative: grad = 2 * ifft(fft(f) * fft(w_reversed))... 
    
    # Actually, the simplest way: grad[m] = 2 * sum_j f[j] * w[j+m]
    # Let's compute this as: for each m, it's a dot product of f with a shifted w.
    # We can compute all m at once using convolution:
    # sum_j f[j] * w[j+m] = (f * w_reversed)[m + n - 1] where w_reversed[j] = w[2n-2-j]
    # Actually: (f conv w)[k] = sum_j f[j] * w[k-j]
    # We want sum_j f[j] * w[j+m] = sum_j f[j] * w[(n-1) - (n-1-j-m)] 
    # Let me define w_flip[j] = w[2n-2-j]
    # sum_j f[j] * w[j+m] = sum_j f[j] * w_flip[2n-2-j-m]
    # Let k' = 2n-2-j-m, so j = 2n-2-k'-m
    # = sum_{k'} f[2n-2-k'-m] * w_flip[k']
    # = (f_shifted conv w_flip)[k'] where f_shifted depends on m...
    # This is circular. Let me just use a direct FFT computation.
    
    # The cleanest way: grad[m] = 2 * sum_j f[j] * w[j+m]
    # Define g[m] = sum_j f[j] * w[j+m] for m = 0, ..., 2n-2
    # This is the cross-correlation of f and w.
    # In numpy: np.correlate(f, w, mode='full') gives:
    # result[k] = sum_j f[j] * w[j - k + (len(w)-1)]  ... I need to check the convention.
    
    # Actually, let me just use the FFT:
    # g[m] = sum_j f[j] * w[j+m]
    # = ifft(fft(f, 2n-1) * conj(fft(w, 2n-1)))[m]  ... not quite

    # Cross-correlation: (f ⋆ w)[m] = sum_j conj(f[j]) * w[j+m] (for real: sum_j f[j] * w[j+m])
    # This equals ifft(conj(fft(f)) * fft(w)) ... let me verify.
    # 
    # FFT of cross-correlation: FFT(f ⋆ w) = conj(FFT(f)) * FFT(w)
    # So f ⋆ w = ifft(conj(FFT(f)) * FFT(w))
    
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    grad = 2.0 * np.fft.irfft(np.conj(F) * W, size)[:n]
    
    return grad, c


def _mirror_step(f, grad, step):
    """Exponentiated gradient step on the simplex (f >= 0, sum f = 1)."""
    log_f = np.log(f + 1e-300) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _optimize_peak_shaving(f, deadline, rng, label=""):
    """
    Optimize f by targeting the peak of the convolution.
    Uses soft-max gradient with annealing temperature.
    """
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    c = _autocorr(f)
    peak0 = c.max()
    T = peak0
    T_min = peak0 * 1e-8
    
    best_score = float('inf')
    best_f = f.copy()
    
    n_iters = 50000
    stuck_count = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        # Temperature annealing
        T = peak0 * (T_min / peak0) ** (progress ** 0.5)
        # Step size: larger early, smaller late
        step = 0.5 * (0.01 / 0.5) ** (progress ** 0.6)
        
        grad, c = _soft_gradient(f, None, T)
        f = _mirror_step(f, grad, step)
        
        # Evaluate every 1000 iterations
        if it % 1000 == 0 or it == n_iters - 1:
            c_now = _autocorr(f)
            score = 2.0 * n * c_now.max()
            if score < best_score:
                best_score = score
                best_f = f.copy()
                stuck_count = 0
                if label and it % 5000 == 0:
                    print(f"    {label} iter {it}: score={score:.6f}, peak={c_now.max():.2e}")
            else:
                stuck_count += 1
            
            if stuck_count > 5:
                # Perturbation
                magnitude = 0.01 / np.sqrt(n)
                f = f + magnitude * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck_count = 0
    
    # Final evaluation
    c_final = _autocorr(f)
    score = 2.0 * n * c_final.max()
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _polish_coordinate_descent(f, deadline, rng):
    """
    Polish using coordinate descent: find the peak lag, then transfer mass
    from high-contribution positions to low-contribution ones.
    """
    n = len(f)
    f = f.copy()
    c = _autocorr(f)
    peak = c.max()
    best_score = 2.0 * n * peak
    best_f = f.copy()
    
    iterations = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        iterations += 1
        c = _autocorr(f)
        peak = c.max()
        k_star = int(np.argmax(c))
        
        # Gradient at peak lag: grad[m] = 2*f[k_star - m]
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] = 2.0 * f[k_star - indices]
        
        # We want to decrease c[k_star] = sum_j f[j]*f[k_star-j]
        # Strategy: move mass from positions where f[j]*f[k_star-j] is large
        # to positions where it's small
        contrib = f * np.zeros(n)  # contribution of each position
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[j] * f[idx]
            else:
                contrib[j] = 0.0
        
        # Find best (donor, receiver) pair
        # Donor: high contrib, Receiver: low contrib
        # Try the top few donors and bottom few receivers
        
        donor_candidates = np.argsort(contrib)[::-1][:5]
        receiver_candidates = np.argsort(contrib)[:5]
        
        improved = False
        for d in donor_candidates:
            for r in receiver_candidates:
                if d == r:
                    continue
                for frac in [0.1, 0.05, 0.02]:
                    f_try = f.copy()
                    amt = f_try[d] * frac
                    if amt < 1e-15:
                        continue
                    f_try[d] -= amt
                    f_try[r] += amt
                    c_try = _autocorr(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        peak = c_try.max()
                        best_score = 2.0 * n * peak
                        best_f = f.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Try random perturbation
            if no_improve > 50:
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i != jx:
                    f_try = f.copy()
                    amt = f_try[i] * 0.01
                    f_try[i] -= amt
                    f_try[jx] += amt
                    c_try = _autocorr(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        peak = c_try.max()
                        best_score = 2.0 * n * peak
                        best_f = f.copy()
                        no_improve = 0
                        improved = True
                if not improved:
                    # Larger random kick
                    if no_improve > 200:
                        f = f + (0.005 / np.sqrt(n)) * rng.standard_normal(n)
                        f = np.maximum(f, 0.0)
                        f = f / f.sum()
                        no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_f, best_score


def _generate_initializations(n, rng):
    """Generate diverse initial candidates."""
    candidates = {}
    j = np.arange(n, dtype=np.float64)
    
    # One-sided exponentials
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f = np.exp(-alpha * j / n)
        f = f / f.sum()
        candidates[f'exp_a{alpha}'] = f
    
    # Polynomial decay
    for p in [1, 2, 3, 5]:
        f = (1.0 - j / n) ** p
        f = np.maximum(f, 0)
        f = f / f.sum()
        candidates[f'poly_p{p}'] = f
    
    # Two-sided with gap (avoid center)
    for gap in [0.1, 0.15, 0.2]:
        center = n / 2
        f = np.exp(-((j - n * (0.5 - gap)) / (n * 0.15))**2) + \
            np.exp(-((j - n * (0.5 + gap)) / (n * 0.15))**2)
        f = f / f.sum()
        candidates[f'gap_{gap}'] = f
    
    # Single off-center bump
    for pos in [0.25, 0.35, 0.65, 0.75]:
        f = np.exp(-((j - n * pos) / (n * 0.1))**2)
        f = f / f.sum()
        candidates[f'bump_{pos}'] = f
    
    # Sinusoidal
    for p in [1, 2, 4]:
        f = np.abs(np.sin(np.pi * j / n)) ** p
        f = f / (f.sum() + 1e-300)
        candidates[f'sin_p{p}'] = f
    
    # Frequency domain: squared magnitude of sum of random phases
    for n_freq in [3, 5, 10]:
        phases = rng.uniform(0, 2 * np.pi, n_freq)
        freqs = rng.choice(range(1, n // 2), n_freq, replace=False)
        g = np.zeros(n)
        for k in range(n_freq):
            g += np.cos(2 * np.pi * freqs[k] * j / n + phases[k])
        f = g ** 2
        f = np.maximum(f, 0)
        if f.sum() > 1e-10:
            f = f / f.sum()
            candidates[f'freq_{n_freq}_{int(freqs[0])}'] = f
    
    # Random
    f = rng.exponential(1.0, n)
    f = f / f.sum()
    candidates['random'] = f
    
    f = rng.dirichlet(np.ones(n) * 0.1)
    candidates['dirichlet'] = f
    
    return candidates


def _simulated_annealing(f, deadline, rng):
    """Simulated annealing polish."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    c = _autocorr(f)
    current_score = 2.0 * n * c.max()
    best_score = current_score
    best_f = f.copy()
    
    T_start = 0.01 / n
    T_end = 1e-10 / n
    n_total = 100000
    
    for it in range(n_total):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_total
        T = T_start * (T_end / T_start) ** progress
        
        # Random move: transfer mass between two random positions
        i = int(rng.integers(0, n))
        jx = int(rng.integers(0, n))
        if i == jx:
            continue
        
        f_try = f.copy()
        amt = f_try[i] * 0.05 * (1 + rng.standard_normal() * 0.5)
        if amt > f_try[i]:
            amt = f_try[i] * 0.5
        f_try[i] -= amt
        f_try[jx] += amt
        
        c_try = _autocorr(f_try)
        new_score = 2.0 * n * c_try.max()
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            f = f_try
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
    
    return best_f, best_score

def _autocorr(f):

def _peak_gradient(f, c):

    grad = np.zeros(n)

        grad[indices] = 2.0 * f[k_star - indices]

def _mirror_step(f, grad, step):
    """Exponentiated gradient (mirror descent) step on the simplex."""
    log_f = np.log(f + 1e-300) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _optimize_gradient(f, deadline, rng, label=""):
    """
    Main gradient-based optimizer with annealing.
    Combines soft-max gradient (exploration) with peak gradient (exploitation).
    """
    n = len(f)
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    c = _autocorr(f)
    peak0 = c.max()
    T = peak0 * 0.5
    T_min = peak0 * 1e-8
    
    best_score = 2.0 * n * peak0
    best_f = f.copy()
    stuck_count = 0
    
    n_iters = 80000
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        
        # Temperature: geometric annealing
        T = peak0 * (T_min / peak0) ** (progress ** 0.4)
        
        # Step size: anneal from 0.8 to 0.005
        step = 0.8 * (0.005 / 0.8) ** (progress ** 0.5)
        
        # Compute gradient
        grad, c = _soft_gradient(f, T)
        
        # For late iterations, blend with exact peak gradient
        if progress > 0.5:
            peak_grad, k_star = _peak_gradient(f, c)
            alpha = (progress - 0.5) / 0.5  # 0 to 1
            grad = (1 - alpha) * grad + alpha * peak_grad
        
        # Mirror descent step
        f = _mirror_step(f, grad, step)
        
        # Evaluate periodically
        if it % 2000 == 0 or it == n_iters - 1:
            c_now = _autocorr(f)
            score = 2.0 * n * c_now.max()
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                stuck_count = 0
                if label and it % 10000 == 0:
                    print(f"      {label} iter {it}: {score:.6f}")
            else:
                stuck_count += 1
            
            if stuck_count > 5:
                # Perturbation to escape local minimum
                mag = 0.005 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck_count = 0
    
    c_final = _autocorr(f)
    score = 2.0 * n * c_final.max()
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _coordinate_polish(f, deadline, rng):
    """
    Intensive coordinate descent polish.
    Targets the current peak lag and transfers mass to reduce it.
    """
    n = len(f)
    f = f.copy()
    c = _autocorr(f)
    peak = c.max()
    best_score = 2.0 * n * peak
    best_f = f.copy()
    
    no_improve = 0
    max_no_improve = 2000
    
    while time.monotonic() < deadline and no_improve < max_no_improve:
        c = _autocorr(f)
        peak = c.max()
        k_star = int(np.argmax(c))
        
        # Compute contribution of each position to the peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            contrib[indices] = f[indices] * f[k_star - indices]
        
        # Find best donor (high contrib) and receiver (low contrib)
        # Sort by contribution
        order = np.argsort(contrib)
        
        improved = False
        # Try top 10 donors with bottom 10 receivers
        donors = order[::-1][:10]
        receivers = order[:10]
        
        for d in donors:
            if improved:
                break
            for r in receivers:
                if d == r:
                    continue
                if f[d] < 1e-12:
                    continue
                for frac in [0.15, 0.08, 0.04, 0.02]:
                    amt = f[d] * frac
                    f_try = f.copy()
                    f_try[d] -= amt
                    f_try[r] += amt
                    c_try = _autocorr(f_try)
                    if c_try.max() < peak - 1e-13:
                        f = f_try
                        peak = c_try.max()
                        best_score = 2.0 * n * peak
                        best_f = f.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            # Try random moves
            for _ in range(5):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx or f[i] < 1e-12:
                    continue
                f_try = f.copy()
                amt = f_try[i] * 0.03
                f_try[i] -= amt
                f_try[jx] += amt
                c_try = _autocorr(f_try)
                if c_try.max() < peak - 1e-13:
                    f = f_try
                    peak = c_try.max()
                    best_score = 2.0 * n * peak
                    best_f = f.copy()
                    no_improve = 0
                    break
            
            # Periodic perturbation
            if no_improve % 100 == 0 and no_improve > 0:
                mag = 0.003 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
    
    return best_f, best_score


def _generate_inits(n, rng):
    """Generate diverse initial candidates exploiting one-sided structure."""
    candidates = {}
    j = np.arange(n, dtype=np.float64)
    
    # One-sided exponentials (key insight: suppresses center peak)
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        f = np.exp(-alpha * j / n)
        f = f / f.sum()
        candidates[f'exp_{alpha}'] = f
        # Also try reversed
        f_rev = np.exp(-alpha * (n - 1 - j) / n)
        f_rev = f_rev / f_rev.sum()
        candidates[f'exprev_{alpha}'] = f_rev
    
    # Polynomial one-sided
    for p in [1, 2, 3, 4]:
        f = np.maximum(1.0 - j / n, 0.0) ** p
        if f.sum() > 1e-10:
            f = f / f.sum()
            candidates[f'poly_{p}'] = f
    
    # Two-bump (gap at center to reduce center peak)
    for gap in [0.08, 0.12, 0.15, 0.2]:
        f = np.exp(-((j / n - 0.5 + gap) / 0.12) ** 2) + \
            np.exp(-((j / n - 0.5 - gap) / 0.12) ** 2)
        f = f / f.sum()
        candidates[f'gap_{gap}'] = f
    
    # Off-center single bump
    for pos in [0.2, 0.3, 0.7, 0.8]:
        f = np.exp(-((j / n - pos) / 0.08) ** 2)
        f = f / f.sum()
        candidates[f'bump_{pos}'] = f
    
    # Sinusoidal (zero at both ends)
    for p in [1, 2, 3]:
        f = np.abs(np.sin(np.pi * j / n)) ** p
        f = f / (f.sum() + 1e-300)
        candidates[f'sin_{p}'] = f
    
    # Asymmetric: one-sided with oscillation
    for alpha in [1.0, 2.0]:
        f = np.exp(-alpha * j / n) * (1.0 + 0.3 * np.sin(6.0 * np.pi * j / n))
        f = np.maximum(f, 0.0)
        if f.sum() > 1e-10:
            f = f / f.sum()
            candidates[f'asym_{alpha}'] = f
    
    # Frequency-domain construction: |sum of complex exponentials|^2
    for n_freq in [3, 5, 7]:
        for trial in range(3):
            freqs = rng.choice(np.arange(1, n // 2), n_freq, replace=False)
            phases = rng.uniform(0, 2 * np.pi, n_freq)
            g = np.zeros(n)
            for k in range(n_freq):
                g += np.cos(2 * np.pi * freqs[k] * j / n + phases[k])
            f = g ** 2
            f = np.maximum(f, 0.0)
            if f.sum() > 1e-10:
                f = f / f.sum()
                key = f'fd_{n_freq}_{trial}_{int(freqs[0])}'
                candidates[key] = f
    
    # Random
    f = rng.exponential(1.0, n)
    f = f / f.sum()
    candidates['rand_exp'] = f
    
    f = rng.dirichlet(np.ones(n) * 0.05)
    candidates['rand_dir'] = f
    
    return candidates


def _simulated_annealing(f, deadline, rng):
    """Simulated annealing for final polish."""
    n = len(f)
    f = f.copy()
    f = f / f.sum()
    c = _autocorr(f)
    current_score = 2.0 * n * c.max()
    best_score = current_score
    best_f = f.copy()
    
    T_start = 0.005 / n
    T_end = 1e-12 / n
    n_total = 200000
    eval_count = 0
    
    for it in range(n_total):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_total
        T = T_start * (T_end / T_start) ** progress
        
        # Move: transfer small amount between two positions
        i = int(rng.integers(0, n))
        jx = int(rng.integers(0, n))
        if i == jx:
            continue
        
        # Scale: smaller moves for lower T
        scale = 0.02 * (1.0 + T / T_start)
        f_try = f.copy()
        amt = f_try[i] * scale * max(0.1, rng.standard_normal())
        if abs(amt) > f_try[i] * 0.5:
            amt = f_try[i] * 0.3 * (1 if amt > 0 else -1)
        f_try[i] -= amt
        f_try[jx] += amt
        if f_try[i] < 0 or f_try[jx] < 0:
            continue
        
        c_try = _autocorr(f_try)
        new_score = 2.0 * n * c_try.max()
        eval_count += 1
        
        delta = new_score - current_score
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            f = f_try
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(f,f))/(sum f)^2.
    
    Multi-phase strategy:
    1. Diverse initialization exploiting one-sided structure.
    2. Gradient-based peak shaving with annealing from top starts.
    3. Intensive gradient optimization on best solution.
    4. Coordinate descent polish targeting the peak lag.
    5. Simulated annealing for final micro-optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    # Get given sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    
    s0 = f_init.sum()
    if s0 > 1e-10:
        f_norm = f_init / s0
    else:
        f_norm = np.ones(n) / n
    
    # Evaluate given
    given_score = evaluate_sequence(f_norm.tolist())
    c_given = _autocorr(f_norm)
    local_given = 2.0 * n * c_given.max()
    print(f"n={n}, Given score (evaluator): {given_score:.6f}, local: {local_given:.6f}")
    
    best_score = local_given
    best_f = f_norm.copy()
    
    # Generate and evaluate initializations
    print("\nGenerating initializations...")
    candidates = _generate_inits(n, rng)
    candidates['given'] = f_norm.copy()
    
    scored = []
    for name, f0 in candidates.items():
        f0 = np.maximum(f0, 0.0)
        s = f0.sum()
        if s < 1e-10:
            continue
        f0 = f0 / s
        c = _autocorr(f0)
        score = 2.0 * n * c.max()
        scored.append((score, name, f0))
    
    scored.sort(key=lambda x: x[0])
    print("Top 10 initializations:")
    for i, (sc, name, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name}: {sc:.6f}")
    
    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_f = scored[0][2].copy()
        print(f"  Best init: {best_score:.6f}")
    
    # Phase 1: Optimize top diverse starts
    print(f"\n=== Phase 1: Gradient optimization from top starts ===")
    # Pick top candidates but ensure diversity (not all too similar)
    selected = [scored[0]]
    for item in scored[1:]:
        if len(selected) >= 6:
            break
        # Check diversity: at least some difference
        selected.append(item)
    
    # Evaluate with actual evaluator for the best init
    eval_best = evaluate_sequence(best_f.tolist())
    print(f"  Evaluator confirms best init: {eval_best:.6f}")
    if eval_best < best_score:
        best_score = eval_best
    
    time_per_opt = (budget_s * 0.35) / len(selected)
    
    for idx, (sc, name, f0) in enumerate(selected):
        if time.monotonic() > deadline - budget_s * 0.4:
            break
        
        t_alloc = min(time_per_opt, deadline - time.monotonic() - budget_s * 0.35)
        if t_alloc < 10:
            break
        opt_deadline = time.monotonic() + t_alloc
        print(f"\n  Optimizing '{name}' (score={sc:.6f}, time={t_alloc:.0f}s)")
        
        f_opt, sc_opt = _optimize_gradient(f0, opt_deadline, rng, label=name)
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.6f}")
    
    # Phase 2: Intensive optimization on current best
    print(f"\n=== Phase 2: Intensive optimization (best={best_score:.6f}) ===")
    phase2_time = budget_s * 0.25
    phase2_deadline = min(time.monotonic() + phase2_time, deadline - budget_s * 0.2)
    f_opt, sc_opt = _optimize_gradient(best_f, phase2_deadline, rng, label="intensive")
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  After phase 2: {best_score:.6f}")
    
    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate polish (best={best_score:.6f}) ===")
    phase3_time = budget_s * 0.2
    phase3_deadline = min(time.monotonic() + phase3_time, deadline - budget_s * 0.08)
    f_pol, sc_pol = _coordinate_polish(best_f, phase3_deadline, rng)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"  After polish: {best_score:.6f}")
    
    # Phase 4: One more gradient round from polished solution
    if time.monotonic() < deadline - budget_s * 0.08:
        print(f"\n=== Phase 4: Final gradient round (best={best_score:.6f}) ===")
        phase4_deadline = min(time.monotonic() + budget_s * 0.07, deadline - 20)
        f_opt, sc_opt = _optimize_gradient(best_f, phase4_deadline, rng, label="final_grad")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  After phase 4: {best_score:.6f}")
    
    # Phase 5: Simulated annealing
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Simulated annealing (best={best_score:.6f}) ===")
        phase5_deadline = min(time.monotonic() + budget_s * 0.06, deadline - 10)
        f_sa, sc_sa = _simulated_annealing(best_f, phase5_deadline, rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_f = f_sa.copy()
            print(f"  After SA: {best_score:.6f}")
    
    # Final coordinate polish with remaining time
    if time.monotonic() < deadline - 5:
        phase6_deadline = deadline - 5
        f_pol2, sc_pol2 = _coordinate_polish(best_f, phase6_deadline, rng)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_f = f_pol2.copy()
            print(f"  Final polish: {best_score:.6f}")
    
    # Final evaluation with actual evaluator
    final_f = np.clip(best_f, 0.0, 1000.0)
    final_f = final_f / final_f.sum()
    final_score = evaluate_sequence(final_f.tolist())
    print(f"\n  Evaluator final score: {final_score:.6f}")
    
    if final_score < best_score:
        best_score = final_score
    
    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in final_f]
