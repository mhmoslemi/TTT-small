def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n

            theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n)
            a += amp * np.sin(theta) / n

    else:

    """Build the convolution matrix A where (A @ b)_k = (f*b)_k"""
    n = len(f)
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for j in range(n):

        # k-j must be in [0, n-1], so k in [j, j+n-1]
        k_start = j
        k_end = min(2*n-1, j+n)
        for k in range(k_start, k_end):
            A[k, j] = f[k-j]
    return A.tocsr()

    return None

    f = f.copy()
    n = len(f)

    f = f / s  # normalize to sum 1
    t = t_init
    best_f = f.copy()
    best_score = _internal_score(best_f)
    
    for it in range(max_iters):

        M = float(np.max(conv))
        
        # Solve LP

        # Normalize g to have same sum as f
        g = g / g.sum()
        
        # Update

        f_new = np.clip(f_new, 0, 1000.0)
        s_new = f_new.sum()
        if s_new < 0.01:

        f_new = f_new / s_new

            best_score = sc
            t = min(0.5, t * 1.02)
        else:
            t = max(0.01, t * 0.8)
        
        f = f_new
        
        if it % 10 == 0:
            print(f"  LP iter {it}: M={M:.6f}, score={best_score:.8f}, t={t:.4f}")

"""
Low-peak nonnegative step-function search using multi-phase optimization.
Combines sinusoidal initialization, mirror descent, LP-based Frank-Wolfe updates,
and coordinate descent polishing to minimize 2*n*max(f*f)/(sum(f))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c[:n] += 0.0  # ensure real
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine(n, amp, mode):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n)
    a = (1.0 + amp * np.sin(theta)) / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _make_multi_sine(n, amps):
    """amps is a list of (mode, amplitude) pairs."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / float(n)
    for mode, amp in amps:
        if abs(amp) > 1e-15:
            theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n)
            a += (amp / float(n)) * np.sin(theta)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _optimize_amplitude(n, mode, deadline):
    """Find best amplitude for a single mode."""
    best_score = np.inf
    best_amp = 0.7
    base = 1.0 / n
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n)
    delta = np.sin(theta)
    
    # Coarse scan
    for amp in np.linspace(0.0, 1.4, 29):
        if time.monotonic() > deadline:
            break
        a = base * (1.0 + amp * delta)
        a = np.clip(a, 0.0, 1000.0)
        sc = _internal_score(a)
        if sc < best_score:
            best_score = sc
            best_amp = float(amp)
    
    # Refine
    step = 1.4 / 28.0
    for _ in range(6):
        lo = max(0.0, best_amp - step)
        hi = min(1.4, best_amp + step)
        if hi - lo < 1e-8:
            break
        for amp in np.linspace(lo, hi, 15):
            if time.monotonic() > deadline:
                break
            a = base * (1.0 + amp * delta)
            a = np.clip(a, 0.0, 1000.0)
            sc = _internal_score(a)
            if sc < best_score:
                best_score = sc
                best_amp = float(amp)
        step *= 0.4
    
    return best_amp, best_score


def _initialize(deadline):
    """Phase 1: Find good initial sequence via sinusoidal scanning."""
    n = 2048
    best_a = None
    best_score = np.inf
    
    # Single mode 1
    amp, sc = _optimize_amplitude(n, 1, deadline)
    if sc < best_score:
        best_score = sc
        best_a = _make_sine(n, amp, 1)
    
    # Try mode 1 + mode 3 combination
    # For the mode 3 amplitude, do a small scan
    for amp3 in [0.0, 0.1, 0.2, -0.1, -0.2]:
        if time.monotonic() > deadline:
            break
        a = _make_multi_sine(n, [(1, amp), (3, amp3)])
        sc = _internal_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    if best_a is None:
        best_a = _make_sine(n, 0.78, 1)
        best_score = _internal_score(best_a)
    
    return best_a, best_score


def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    # Standard simplex projection
    if target_sum <= 0:
        return np.zeros(n)
    
    # We want to project onto {x >= 0, sum(x) = target_sum}
    # This is: x = proj(v) = max(v - tau, 0) where tau is chosen so sum = target_sum
    
    # Use bisection on tau
    v_sorted = np.sort(v)[::-1]
    # Find the right threshold
    cumsum = np.cumsum(v_sorted)
    # We need sum(max(v - tau, 0)) = target_sum
    # equivalently: sum of (v_i - tau) for v_i > tau = target_sum
    
    lo, hi = 0.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    # Adjust for numerical error
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-zero entries
        nz = result > 0
        if nz.sum() > 0:
            result[nz] += diff / nz.sum()
            result = np.maximum(result, 0.0)
    return result


def _mirror_descent(a, deadline, phase_time):
    """Phase 2: Mirror descent with multiplicative weight updates."""
    start = time.monotonic()
    phase_end = start + min(phase_time, deadline - start)
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s  # normalize to sum 1
    
    best = arr.copy()
    best_score = _internal_score(best)
    K = 16  # number of active lags
    
    lr = 5.0
    min_lr = 0.1
    it = 0
    stagnation = 0
    
    while time.monotonic() < phase_end:
        it += 1
        
        conv = _fft_convolve(arr)
        if conv.size == 0:
            break
        
        maxc = float(np.max(conv))
        
        # Find top K lags
        if conv.size <= K:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(conv, -K)[-K:]
        
        # Compute subgradient (average over active lags)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # d(f*f)_k / df_j = f_{k-j} + f_{j-k} (with f_x = 0 if x out of [0,n-1])
            # For vec: grad += arr[k::-1] (reversed, clipped) conceptually
            # Actually: (f*f)_k = sum_i f_i * f_{k-i}
            # d/d f_j: f_{k-j} (from first term when i=j) + f_{j-k} (from second term when k-i=j, i.e., i=k-j)
            # But we need 0 <= k-j < n and 0 <= j-k < n
            # f_{k-j}: valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
            # f_{j-k}: valid when 0 <= j-k <= n-1, i.e., k <= j <= k+n-1
            
            # For the first term: jac_j += f_{k-j} for j in [max(0,k-n+1), min(n-1,k)]
            # For the second term: jac_j += f_{j-k} for j in [max(0,k), min(n-1,k+n-1)]
            # But the second term is just the first term reflected, so:
            # jac_j = f_{|k-j|} ... no that's not right.
            
            # Let me be more careful:
            # (f*f)_k = sum_{i: 0<=i<n, 0<=k-i<n} f_i * f_{k-i}
            # = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f_i * f_{k-i}
            
            # d/d f_j for j in range:
            # If j is in the sum range [max(0,k-n+1), min(n-1,k)]:
            #   contribution from i=j: f_{k-j}
            #   contribution from i=k-j (if k-j != j and k-j is in range): f_j
            # So the derivative is f_{k-j} + f_j (if k-j != j) or 2*f_j (if k-j == j, i.e., k=2j)
            
            # More simply: d(f*f)_k / df_j = f_{k-j} if 0<=k-j<n (first sum) + f_{j-k} if 0<=j-k<n (second sum)
            # But f_{j-k} = f_{-(k-j)}, and since convolution is symmetric, f_{j-k} = 0 unless j >= k, in which case f_{j-k}.
            
            # Actually the cleanest way:
            # (f*f)_k = sum_i f_i * f_{k-i}
            # d/d f_j = f_{k-j} * 1_{0 <= k-j < n} + f_{j-k} * 1_{0 <= j-k < n}
            #         = f_{k-j} * 1_{k-n+1 <= j <= k} + f_{j-k} * 1_{k <= j <= k+n-1}
            
            # For the first part: j ranges from max(0, k-n+1) to min(n-1, k)
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_hi1 >= j_lo1:
                # grad[j] += f[k-j] for j in [j_lo1, j_hi1]
                # f[k-j] for j in [j_lo1, j_hi1] = f[k-j_hi1] ... f[k-j_lo1]
                # = f[k-j_hi1 : k-j_lo1+1] reversed
                idx = np.array([k - j for j in range(j_lo1, j_hi1 + 1)])
                grad[j_lo1:j_hi1+1] += arr[idx]
            
            # For the second part: j ranges from max(0, k) to min(n-1, k+n-1)
            # But this is the same as the first part reflected around j = k/2
            # Actually, let's just note that (f*f)_k is symmetric: (f*f)_k = (f*f)_{k} 
            # and d/d f_j (f*f)_k = d/d f_{n-1-j} (f*f)_{2n-2-k} ... no that's for anti-palindromic.
            
            # The second term: f_{j-k} for j in [k, min(n-1, k+n-1)]
            # = f[j-k] for j in [max(0,k), min(n-1, k+n-1)]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                idx = np.array([j - k for j in range(j_lo2, j_hi2 + 1)])
                grad[j_lo2:j_hi2+1] += arr[idx]
        
        # Average the gradient
        grad /= float(len(active))
        
        # Project gradient to tangent space (remove mean to preserve sum)
        grad = grad - np.mean(grad)
        
        # Check if gradient is meaningful
        if np.max(np.abs(grad)) < 1e-15:
            break
        
        # Multiplicative weight update
        exponent = -lr * grad
        # Clip to avoid overflow
        exponent = np.clip(exponent, -5.0, 5.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        
        # Project to simplex (sum = 1)
        arr_new = _project_simplex(arr_new, 1.0)
        
        sc = _internal_score(arr_new)
        if sc < best_score:
            arr = arr_new
            best = arr_new.copy()
            best_score = sc
            stagnation = 0
            lr = min(50.0, lr * 1.01)
        else:
            stagnation += 1
            if sc > best_score * 1.05:
                lr = max(min_lr, lr * 0.7)
            else:
                lr = max(min_lr, lr * 0.95)
            
            if stagnation > 200:
                break
        
        if it % 500 == 0:
            print(f"  Mirror iter {it}: score={best_score:.8f}, lr={lr:.3f}")
    
    return best, best_score


def _build_A_sparse(f):
    """Build sparse convolution matrix A where (A @ b)_k = (f*b)_k."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # A[k, j] = f[k-j] for k in [j, min(j+n-1, 2n-2)]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, M, time_limit=10):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = _build_A_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'time_limit': time_limit})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def _lp_refine(a, deadline, phase_time):
    """Phase 3: LP-based Frank-Wolfe refinement."""
    start = time.monotonic()
    phase_end = start + min(phase_time, deadline - start)
    
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    t = 0.05  # blending parameter
    min_t = 0.005
    it = 0
    stagnation = 0
    
    print(f"  LP refine starting: n={n}, initial score={best_score:.8f}")
    
    # Pre-build A matrix (will be updated as f changes)
    while time.monotonic() < phase_end:
        it += 1
        
        # Compute current convolution and M
        conv = _fft_convolve(arr)
        M = float(np.max(conv))
        
        # Solve LP
        g = _lp_solve(arr, M, time_limit=15)
        if g is None or g.sum() < 1e-10:
            print(f"  LP failed at iter {it}")
            break
        
        # Normalize g to sum 1
        g = g / g.sum()
        
        # Blend
        new_arr = (1.0 - t) * arr + t * g
        new_arr = np.clip(new_arr, 0.0, 1000.0)
        ns = new_arr.sum()
        if ns < 0.01:
            break
        new_arr = new_arr / ns
        
        sc = _internal_score(new_arr)
        
        if sc < best_score:
            arr = new_arr
            best = new_arr.copy()
            best_score = sc
            stagnation = 0
            t = min(0.3, t * 1.01)
        else:
            stagnation += 1
            t = max(min_t, t * 0.85)
            if sc > best_score * 1.01:
                t = max(min_t, t * 0.7)
        
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {it}: score={best_score:.8f}, M={M:.6e}, t={t:.4f}, time={elapsed:.0f}s")
        
        if stagnation > 100:
            # Try to revive with a larger step
            t = 0.1
            stagnation = 0
            # Or break if we've been stuck too long
            if best_score < 1.51:
                break
    
    return best, best_score


def _coordinate_descent(a, deadline):
    """Phase 4: Coordinate descent polishing."""
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    # Try random coordinate perturbations
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        # Pick a few random coordinates
        n_coords = min(10, n)
        idxs = rng.choice(n, size=n_coords, replace=False)
        
        improved = False
        for idx in idxs:
            idx = int(idx)
            # Try increasing and decreasing this coordinate
            for frac in [0.9, 0.95, 1.05, 1.1, 1.2, 0.8, 0.7]:
                trial = best.copy()
                trial[idx] *= frac
                trial = np.clip(trial, 0.0, 1000.0)
                ts = trial.sum()
                if ts < 0.01:
                    continue
                trial = trial / ts
                sc = _internal_score(trial)
                if sc < best_score:
                    best = trial
                    best_score = sc
                    improved = True
                    break
        
        if not improved:
            break
    
    return best, best_score


def _validate(a):

The standard algorithm (from Wang & Carreira-Perpinan 2013):
1. Sort v in descending order: u_1 >= u_2 >= ... >= u_n
2. Find rho = max{t : u_t - (sum_{i=1}^t u_i - target_sum)/t > 0}

```python
def _project_simplex(v, target_sum):
    """Project onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = v.size
    if n == 0:
        return v.copy()
    if target_sum <= 0:
        return np.zeros(n)
    
    u = np.sort(v)[::-1]  # descending
    cumsum = np.cumsum(u)
    # Find rho: max t such that u[t] - (cumsum[t] - target_sum) / t > 0
    # i.e., t * u[t] > cumsum[t] - target_sum
    # i.e., t * u[t] + target_sum > cumsum[t]
    
    t = np.arange(1, n + 1)
    cond = (u * t > (cumsum - target_sum))
    rho = np.sum(cond)  # number of True values
    
    if rho == 0:
        return np.zeros(n)
    
    theta = (cumsum[rho - 1] - target_sum) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        nz = result > 0
        if nz.sum() > 0:
            result[nz] += diff / nz.sum()
            result = np.maximum(result, 0.0)
    return result
```

Actually, let me double-check the formula. We want to project v onto {x >= 0, sum(x) = S} (where S = target_sum).

The KKT conditions give: x_i = max(v_i - theta, 0) where theta is chosen so sum(x) = S.

Let rho = #{i : v_i > theta} (the number of positive components).
Then sum(x) = sum_{i in support} (v_i - theta) = S
=> sum_{i in support} v_i - rho * theta = S
=> theta = (sum_{i in support} v_i - S) / rho

The support is {i : v_i > theta}, which is the first rho elements when v is sorted in decreasing order.

So the algorithm is:
1. Sort v in decreasing order: u_1 >= u_2 >= ... >= u_n
2. For each possible rho = 1, ..., n, check if u_rho > (sum_{i=1}^{rho} u_i - S) / rho
3. The correct rho is the largest one satisfying this
4. theta = (sum_{i=1}^{rho} u_i - S) / rho
5. x_i = max(v_i - theta, 0)

rho * u_rho > cumsum[rho] - S
=> cumsum[rho] < S + rho * u_rho

Since cumsum[rho] - rho * u_rho is increasing in rho (because u is decreasing), we can binary search.

    # We can construct it using block operations

    # row j has f[0]
    # row j+1 has f[1]
    # ...
    # row j+n-1 has f[n-1] (if j+n-1 < 2n-1, i.e., j < n)
    
    # Total non-zeros: for column j, count = min(n, 2n-1-j+1) = min(n, 2n-j)
    # For j = 0: n non-zeros (rows 0 to n-1)
    # For j = 1: n non-zeros (rows 1 to n)
    # ...
    # For j = n-1: n non-zeros (rows n-1 to 2n-2)
    # So each column has exactly n non-zeros!
    # Total: n^2 non-zeros
    
    # Actually wait: for column j, rows range from j to min(j+n-1, 2n-2)
    # Count = min(j+n-1, 2n-2) - j + 1
    # For j <= n-1: count = min(j+n-1, 2n-2) - j + 1
    #   If j + n - 1 <= 2n-2 (i.e., j <= n-1, which is always true for j < n):
    #   count = j + n - 1 - j + 1 = n
    # So yes, each column has exactly n non-zeros. Total: n^2.
    
    # Build using structured arrays
    j_idx = np.repeat(np.arange(n), n)
    # For column j, the rows are j, j+1, ..., j+n-1
    # offset within column: 0, 1, ..., n-1
    offset = np.tile(np.arange(n), n)
    row_idx = j_idx + offset
    
    # Filter out rows >= 2n-1 (shouldn't happen for j < n, but just in case)
    # Actually for j = n-1, max row = n-1 + n-1 = 2n-2, which is fine.
    # So no filtering needed.

    vals = f[val_idx]
    
    A = csr_matrix((vals, (row_idx, j_idx)), shape=(2*n-1, n))

def _next_pow2(m):
    n = 1
    while n < m:

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):

            active = np.argpartition(conv, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_hi1 >= j_lo1:
                idx = np.arange(k - j_hi1, k - j_lo1 + 1)
                grad[j_lo1:j_hi1 + 1] += arr[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                idx = np.arange(j_lo2 - k, j_hi2 - k + 1)
                grad[j_lo2:j_hi2 + 1] += arr[idx]

        grad /= float(len(active))
        grad -= np.mean(grad)

        if np.max(np.abs(grad)) < 1e-16:
            break

        exponent = -lr * grad
        exponent = np.clip(exponent, -4.0, 4.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        arr_new = _project_simplex(arr_new, 1.0)

        sc = _internal_score(arr_new)
        if sc < best_score:
            arr = arr_new
            best = arr_new.copy()
            best_score = sc
            stagnation = 0
            lr = min(30.0, lr * 1.005)
        else:
            stagnation += 1
            if sc > best_score * 1.03:
                lr = max(min_lr, lr * 0.75)
            else:
                lr = max(min_lr, lr * 0.95)
            if stagnation > 300:
                break

        if it % 500 == 0:
            print(f"  Mirror iter {it}: score={best_score:.8f}, lr={lr:.3f}, t={time.monotonic()-start:.0f}s")

    return best, best_score


def _precompute_A_structure(n):
    j_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    offset = np.tile(np.arange(n, dtype=np.int32), n)
    row_idx = (j_idx + offset).astype(np.int32)
    return row_idx, j_idx, offset


def _make_conv_matrix(f, structure):
    n = len(f)
    row_idx, j_idx, offset = structure
    vals = f[offset.astype(np.int64)]
    return csr_matrix((vals, (row_idx, j_idx)), shape=(2 * n - 1, n))


def _lp_solve(f, M, structure, time_limit=20):
    n = len(f)
    A = _make_conv_matrix(f, structure)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_time):
    start = time.monotonic()
    phase_end = min(deadline, start + max_time)
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    structure = _precompute_A_structure(n)

    t = 0.04
    min_t = 0.003
    max_t = 0.25
    it = 0
    stagnation = 0
    lp_fail_count = 0

    print(f"  LP phase start: n={n}, score={best_score:.8f}")

    while time.monotonic() < phase_end:
        it += 1
        conv = _fft_convolve(arr)
        M = float(np.max(conv))

        g = _lp_solve(arr, M, structure, time_limit=25)
        if g is None or g.sum() < 1e-12:
            lp_fail_count += 1
            if lp_fail_count > 10:
                print(f"  LP repeatedly failing, stopping at iter {it}")
                break
            continue
        lp_fail_count = 0

        g_sum = float(g.sum())
        g = g / g_sum

        new_arr = (1.0 - t) * arr + t * g
        new_arr = np.clip(new_arr, 0.0, 1000.0)
        ns = new_arr.sum()
        if ns < 0.01:
            t = max(min_t, t * 0.8)
            continue
        new_arr = new_arr / ns

        sc = _internal_score(new_arr)
        if sc < best_score:
            arr = new_arr
            best = new_arr.copy()
            best_score = sc
            stagnation = 0
            t = min(max_t, t * 1.008)
        else:
            stagnation += 1
            if sc > best_score * 1.005:
                t = max(min_t, t * 0.7)
            else:
                t = max(min_t, t * 0.9)

        if it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {it}: score={best_score:.8f}, M={M:.4e}, t={t:.5f}, elapsed={elapsed:.0f}s")

        if stagnation > 200:
            if best_score < 1.52:
                print(f"  LP converged (stagnation) at iter {it}, score={best_score:.8f}")
                break
            t = 0.08
            stagnation = 0

    return best, best_score


def _subgradient_polish(a, deadline, max_time):
    start = time.monotonic()
    phase_end = min(deadline, start + max_time)
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    K = 30
    lr = 0.5
    min_lr = 0.01
    it = 0
    stagnation = 0

    while time.monotonic() < phase_end:
        it += 1
        conv = _fft_convolve(arr)
        if conv.size <= K:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(conv, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_hi1 >= j_lo1:
                idx = np.arange(k - j_hi1, k - j_lo1 + 1)
                grad[j_lo1:j_hi1 + 1] += arr[idx]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_hi2 >= j_lo2:
                idx = np.arange(j_lo2 - k, j_hi2 - k + 1)
                grad[j_lo2:j_hi2 + 1] += arr[idx]

        grad /= float(len(active))
        grad -= np.mean(grad)

        if np.max(np.abs(grad)) < 1e-16:
            break

        lr_eff = lr / float(np.sqrt(it + 1))
        arr_new = arr - lr_eff * grad
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        arr_new = _project_simplex(arr_new, 1.0)
        arr = arr_new

        if it % 100 == 0:
            sc = _internal_score(arr)
            if sc < best_score:
                best = arr.copy()
                best_score = sc
                stagnation = 0
            else:
                stagnation += 1
                lr = max(min_lr, lr * 0.9)
            if stagnation > 50:
                break
            print(f"  Subgrad iter {it}: score={best_score:.8f}, lr={lr:.4f}")

    return best, best_score


def _validate_and_return(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return [1.0] * 128
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return [1.0] * 128
    arr = arr / s
    return [float(x) for x in arr]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(f*f)/(sum(f))^2.
    Phase 1: Sinusoidal initialization (~15s)
    Phase 2: Mirror descent with multiplicative weights (~400s)
    Phase 3: LP-based Frank-Wolfe refinement (~1600s)
    Phase 4: Subgradient polish (~300s)
    """
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 10.0

    # Phase 1: Initialization
    init_deadline = t0 + 12.0
    best_a, best_score = _initialize(init_deadline)
    print(f"[Phase 1] score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Mirror descent
    if time.monotonic() < hard_deadline - 200:
        mirror_deadline = t0 + 380.0
        best_a, best_score = _mirror_descent(best_a, mirror_deadline, 360.0)
        print(f"[Phase 2] score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # Phase 3: LP refinement (main workhorse)
    if time.monotonic() < hard_deadline - 100:
        lp_deadline = t0 + 2050.0
        best_a, best_score = _lp_refine(best_a, lp_deadline, 1650.0)
        print(f"[Phase 3] score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # Phase 4: Subgradient polish
    if time.monotonic() < hard_deadline - 30:
        sg_deadline = hard_deadline - 5.0
        best_a2, best_score2 = _subgradient_polish(best_a, sg_deadline, 300.0)
        if best_score2 < best_score:
            best_a = best_a2
            best_score = best_score2
            print(f"[Phase 4] improved to score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")
        else:
            print(f"[Phase 4] no improvement, time={time.monotonic()-t0:.1f}s")

    # Final
    result = _validate_and_return(best_a)
    try:
        real_score = float(evaluate_sequence(result))
        print(f"[Final] validated score: {real_score:.8f}")
    except Exception as e:
        print(f"[Final] validation error: {e}")

    return result
