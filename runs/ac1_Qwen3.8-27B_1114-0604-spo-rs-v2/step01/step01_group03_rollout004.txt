"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using adaptive-temperature log-sum-exp
gradient descent with top-k peak weighting, multi-scale initialization,
and structured random restarts. Key improvements over baseline: tighter
temperature range, top-k focused gradients, line search on step size,
and multiple n exploration.
"""

import math

import numpy as np


def _norm(a, eps=1e-30):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones_like(a, dtype=np.float64) / len(a)
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _conv_fft(a, L, m):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    return 2.0 * n * float(np.max(c))


def _grad_from_weights(w, a_rev_fft, L, m, n):
    """Compute gradient given weights w over convolution positions.
    grad[j] = 2 * sum_l w[j+l] * a[l]  (cross-correlation)
    Using: cc[j] = conv(a_rev, w)[j+n-1]
    """
    F_w = np.fft.rfft(w, L)
    cc = np.fft.irfft(a_rev_fft * F_w, L)
    grad = 2.0 * cc[n - 1:n - 1 + n]
    np.maximum(grad, 0.0, out=grad)
    return grad


def _soft_weights(c, temp, top_k=None):
    """Compute softmax weights over convolution values."""
    v = c.copy()
    vmax = float(v.max())
    if top_k is not None and top_k < len(v):
        # Zero out entries below the top_k threshold to focus
        thresh = np.sort(v)[-top_k]
        v[v < thresh] = -1e30
        vmax = thresh
    z = temp * (v - vmax)
    z = np.clip(z, -60.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        w = np.zeros_like(v)
        w[v >= vmax - 1e-12] = 1.0 / np.sum(v >= vmax - 1e-12)
        return w
    return e / s


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.5)
    z += rng.normal(0.0, 0.8) * x
    z += rng.normal(0.0, 0.8) * (x * x - 1.0 / 3.0)
    z += rng.normal(0.0, 0.3) * (x**3 - x)

    kmax = int(rng.integers(2, 10))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.03, 0.6) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)

    noise = rng.normal(0.0, 0.2, n)
    if n > 10:
        w = max(3, n // 200)
        noise = np.convolve(noise, np.ones(w) / w, mode="same")
    z += noise

    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _norm(p)

def _optimize(a, L, m, n, rng, deadline, phase="refine"):
    """Core optimization loop with adaptive temperature and step size."""
    a = _norm(a)
    c = _conv_fft(a, L, m)
    cur_score = _score(c, n)

    best_a = a.copy()
    best_score = cur_score
    best_time = time.monotonic()

    # Parameters
    if phase == "explore":
        temp = 3.0
        temp_max = 30.0
        lr = 0.15
        top_k = None
        patience = 80
    elif phase == "refine":
        temp = 1.0
        temp_max = 20.0
        lr = 0.08
        top_k = 50
        patience = 150
    else:  # polish
        temp = 0.5
        temp_max = 5.0
        lr = 0.03
        top_k = 20
        patience = 200

    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        a_rev_fft = np.fft.rfft(a[::-1], L)

        w = _soft_weights(c, temp, top_k)
        grad = _grad_from_weights(w, a_rev_fft, L, m, n)
        gmax = float(grad.max())
        if not np.isfinite(gmax) or gmax < 1e-14:
            break
        grad /= gmax

        # Line search: try several step sizes
        accepted = False
        for step in (lr, 0.7 * lr, 0.4 * lr, 0.2 * lr):
            if step < 1e-6:
                continue
            a_new = a * np.exp(-step * grad)
            a_new = _norm(a_new)
            c_new = _conv_fft(a_new, L, m)
            sc_new = _score(c_new, n)

            if sc_new < cur_score - 1e-15:
                a = a_new
                c = c_new
                cur_score = sc_new
                accepted = True
                break

        if accepted:
            if no_improve == 0:
                lr = min(0.5, lr * 1.08)
            no_improve = 0
        else:
            no_improve += 1
            lr = max(0.005, lr * 0.7)
            # Restore to best if we've been stuck
            if no_improve > patience:
                a = best_a.copy()
                c = _conv_fft(a, L, m)
                cur_score = _score(c, n)
                lr = 0.15
                temp = min(temp_max, temp * 2.0 + 1.0)
                no_improve = 0

                # Random perturbation to escape
                if rng.random() < 0.6:
                    prior = _random_prior(n, rng)
                    mix = 0.1 + 0.3 * rng.random()
                    a = _norm((1.0 - mix) * a + mix * prior)
                    c = _conv_fft(a, L, m)
                    cur_score = _score(c, n)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            best_time = time.monotonic()

        # Adaptive temperature
        it += 1
        if it % 30 == 0:
            # Increase temp gradually, but cap it
            temp = min(temp_max, temp * 1.05)
            # Occasionally reset temp for exploration
            if no_improve > patience // 2 and rng.random() < 0.3:
                temp = min(temp_max, temp * 1.5 + 2.0)

        # Check time
        if time.monotonic() > deadline - 2.0:
            break

    return best_a, best_score
