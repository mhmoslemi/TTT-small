def lp_update(a, M, step=0.1):
    """
    LP-based update: find b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0.
    Then update a = (1-step)*a + step*b_normalized.
    """
    from scipy.optimize import linprog
    
    n = a.size
    # Build constraint matrix T: (2n-1) x n
    # T[k, j] = a[j] if 0 <= k-j < n, else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            T[k, j] = a[j]
    
    # Maximize sum(b) => Minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: T @ b <= M
    A_ub = T
    b_ub = np.full(2*n-1, M)
    
    # b >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return a, 0.0
    
    b_opt = res.x
    b_sum = b_opt.sum()
    if b_sum < 1e-10:
        return a, 0.0
    
    b_norm = b_opt / b_sum
    
    # Update
    a_new = (1 - step) * a + step * b_norm
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    
    return a_new, b_sum

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based updates (solving a linear
program to find optimal target sequences), coordinate descent, and adaptive
subgradient steps. Focuses on n=128 with diverse initializations and intensive
final polishing.

import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def compute_score(a):
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))

def proj_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_for_lag(a, k):
    n = a.size
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return grad
    indices = np.arange(lo, hi + 1)
    grad[indices] = 2.0 * a[np.abs(k - indices)]
    return grad

def lp_based_update(a, M, T_matrix):
    """Solve LP to find target sequence b."""
    n = a.size
    c = -np.ones(n)
    A_ub = T_matrix
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    b = res.x
    s = b.sum()
    if s < 1e-10:
        return None
    return b / s

def build_T_matrix(a):
    """Build the (2n-1) x n Toeplitz matrix for conv(a, b)."""
    n = a.size
    T = np.zeros((2*n-1, n))
    for j in range(n):
        lo = j  # k starts at j (since k-j >= 0 means k >= j)
        hi = min(2*n-2, j + n - 1)  # k-j < n means k < j+n
        T[lo:hi+1, j] = a[j]
    return T

def coord_descent(a, max_rounds=3, n_factors=8):
    """Coordinate descent with multiplicative perturbations."""
    n = a.size
    best_sc = compute_score(a)
    factors = np.array([0.3, 0.5, 0.7, 1.0, 1.4, 1.7, 2.0, 3.0])
    
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            orig = a[i]
            if orig < 1e-14:
                continue
            best_f = 1.0
            best_f_sc = best_sc
            for f in factors:
                if f == 1.0:
                    continue
                cand = a.copy()
                cand[i] = orig * f
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand /= s
                sc = compute_score(cand)
                if sc < best_f_sc:
                    best_f_sc = sc
                    best_f = f
            if best_f != 1.0:
                a[i] = orig * best_f
                a = np.maximum(a, 0.0)
                a /= a.sum()
                best_sc = best_f_sc
                improved = True
        if not improved:
            break
    
    return a, best_sc

    a = proj_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    T = build_T_matrix(a)
    step_lp = 0.3
    step_sg = 0.1

    iteration = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        M = float(np.max(b))
        
        if cur_sc < best_sc:

            best_a = a.copy()
            last_improve = iteration

        T = build_T_matrix(a)
        b_target = lp_based_update(a, M, T)

            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            
            b_new = full_conv(a_new)
            new_sc = 2.0 * n * float(np.max(b_new))
            
            if new_sc <= cur_sc * 1.01:  # allow slight increase
                a = a_new
                step_lp = min(step_lp * 1.02, 0.5)
            else:
                step_lp *= 0.7
                if step_lp < 0.01:
                    step_lp = 0.3
        else:

            max_b = M
            active_mask = b >= 0.85 * max_b
            active_lags = np.where(active_mask)[0]
            if len(active_lags) < 3:

            grad = np.zeros(n)

            a_new = proj_simplex(a - step_sg * grad)

        # Coordinate descent every 20 iterations

    return best_a, best_sc
