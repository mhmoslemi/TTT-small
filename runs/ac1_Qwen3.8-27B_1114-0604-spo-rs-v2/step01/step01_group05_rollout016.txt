def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fft_cross_convolution(a, b):
    """Linear convolution a*b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)
    FB = np.fft.rfft(b, size)
    c = np.fft.irfft(FA * FB, size)[:need]
    return c


def smooth_max_and_weights(b, beta):
    """Smooth max (log-sum-exp) and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float('inf'), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast local score: 2*n*max(p*p) (assumes sum(p)=1)."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j w_{i+j} for gradient computation."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


# ============================================================
# LP-based blending (the key innovation)
# ============================================================

def solve_blending_lp(p, max_n=4096, time_budget=45.0):
    """
    Solve the LP: min t s.t. (p*q)_k <= t, sum(q)=1, q>=0.
    Returns the optimal q or None.
    Uses the dual formulation for smaller problem size.
    """
    if not HAS_SCIPY:
        return None

    n = len(p)
    if n > max_n:
        # Downsample: average blocks
        block = n // max_n
        n_eff = max_n // block * block
        if n_eff < 16:
            return None
        p_eff = p[:n_eff].reshape(n_eff // block, block).mean(axis=1)
        p_eff = normalize_simplex(p_eff)
    else:
        p_eff = p
        n = len(p_eff)

    if n < 8:
        return None

    # Dual: max lambda
    # s.t. sum_k y_k = 1
    #      (P^T y)_j >= lambda for j = 0,...,n-1
    #      y_k >= 0 for k = 0,...,2n-2
    #
    # Variables: [y_0,...,y_{2n-2}, lambda]  (2n-1 + 1 = 2n variables)
    # Minimize: -lambda
    #
    # Constraints:
    #   Equality: sum y = 1
    #   Inequality j: -(P^T y)_j + lambda <= 0, i.e.
    #     -sum_k p[k-j]*y_k + lambda <= 0  (where 0<=k-j<n)

    nvar = 2 * n  # 2n-1 y's + 1 lambda
    n_cons = n  # n inequality constraints

    c = np.zeros(nvar)
    c[-1] = -1.0

    # Build sparse A_ub (n x nvar)
    row_idx = []
    col_idx = []
    vals = []

    for j in range(n):
        # Column for lambda
        row_idx.append(j)
        col_idx.append(nvar - 1)
        vals.append(1.0)
        # Columns for y_k: P[k,j] = p[k-j] if 0 <= k-j < n
        # (P^T y)_j = sum_k p[k-j] * y_k
        # Constraint: -sum_k p[k-j] y_k + lambda <= 0
        k_lo = j
        k_hi = j + n - 1
        for k in range(k_lo, k_hi + 1):
            if k < 2 * n - 1:
                row_idx.append(j)
                col_idx.append(k)
                vals.append(-p_eff[k - j])

    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(n_cons, nvar))
    b_ub = np.zeros(n_cons)

    A_eq = np.zeros((1, nvar))
    A_eq[0, :2 * n - 1] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (2 * n - 1) + [(None, None)]

    start_t = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, time_budget - (time.monotonic() - start_t))})
        if not res.success:
            return None
        y = np.maximum(res.x[:2 * n - 1], 0.0)
        lam = float(res.x[-1])
    except Exception:
        return None

    # Recover primal q from dual solution using "active set" heuristic:
    # q_j > 0 only where (P^T y)_j ≈ lambda
    # Compute (P^T y)_j = sum_k p[k-j] y_k
    Py = np.zeros(n, dtype=np.float64)
    # Use FFT for efficiency
    need = n + (2 * n - 1) - 1
    size = 1
    while size < need:
        size <<= 1
    # (P^T y)_j = sum_{k=j}^{j+n-1} p[k-j] y[k] = cross-correlation
    # Let's compute it as: flip y, convolve with p
    y_flip = y[::-1]
    prod = fft_cross_convolution(p_eff, y_flip)
    # (P^T y)_j corresponds to prod[2n-2-j + (n-1)]... let me just do it directly
    # Actually (P^T y)_j = sum_{i=0}^{n-1} p[i] y[j+i]
    # This is a "sliding inner product" = correlation
    # Let's use: for each j, Py[j] = dot(p_eff, y[j:j+n])
    # For efficiency with FFT: Py = ifft(conj(fft(p)) .* fft(y, n-1)) ... complex
    # Let me just do it with a loop for small n, or FFT for large n
    if n <= 2000:
        for j in range(n):
            Py[j] = float(np.dot(p_eff, y[j:j + n]))
    else:
        # Use FFT: Py[j] = sum_i p[i] y[j+i] = (p * y_reversed)[j + len(y) - 1]... 
        # Simpler: Py = real(ifft(conj(FFT(p, 2n-1)) .* FFT(y, 2n-1))) shifted
        size2 = 1
        while size2 < 2 * n - 1:
            size2 <<= 1
        FP = np.fft.fft(p_eff, size2)
        FY = np.fft.fft(y, size2)
        # sum_i p[i] y[j+i] = sum_i p[i] y[j+i]
        # = ifft(conj(FP) * FY)[j] ... let me verify
        # ifft(conj(FP) * FY)[k] = sum_i p[i] y[k+i] (circular, but zero-padded so linear)
        Py_circ = np.real(np.fft.ifft(np.conj(FP) * FY, size2))[:n]
        Py = Py_circ

    Py = np.maximum(Py, 0.0)
    tol = 1e-6 * max(1.0, abs(lam)) + 1e-12

    # Active set: where Py[j] >= lam - tol
    active = Py >= (lam - tol)
    n_active = int(active.sum())

    if n_active == 0:
        # Fallback: use top few
        top_k = max(1, n // 4)
        idx = np.argsort(Py)[-top_k:]
        active = np.zeros(n, dtype=bool)
        active[idx] = True
        n_active = top_k
    elif n_active > n // 2:
        # Too many active: keep top half
        top_k = n // 2
        idx = np.argsort(Py)[-top_k:]
        active = np.zeros(n, dtype=bool)
        active[idx] = True
        n_active = top_k

    q_eff = np.zeros(n, dtype=np.float64)
    q_eff[active] = 1.0 / n_active

    # Verify: check that (p*q_eff)_k <= lam + small_tol
    pqe = fft_convolution(p_eff)  # just checking, not used
    # Actually check cross-conv
    pq_cross = fft_cross_convolution(p_eff, q_eff)
    max_pq = float(np.max(pq_cross))
    if max_pq > lam * (1 + 1e-3) + 1e-10:
        # The active set constraint is violated; use a softer approach
        # Weight by softplus of (Py - lam)
        diff = Py - lam
        w = np.exp(np.clip(diff / (tol + 1e-15), -30, 30))
        sw = float(w.sum())
        if sw > 0:
            q_eff = w / sw
        else:
            q_eff = 1.0 / n
            q_eff[~active] = 0.0
            q_eff = q_eff / q_eff.sum()

    # Upsample back to full length
    if n < len(p):
        q = np.interp(np.linspace(0, 1, len(p)), np.linspace(0, 1, n), q_eff)
        q = np.maximum(q, 0.0)
        s = q.sum()
        if s > 0:
            q /= s
        else:
            q = np.ones(len(p)) / len(p)
    else:
        q = q_eff

    return normalize_simplex(q)

# Gradient descent at full resolution
# ============================================================

def full_gradient_descent(p0, deadline, beta_start=5000.0, beta_end=500000.0,
                          max_iter=200000, step_init=1.0):
    """Projected gradient descent with adaptive beta and step size."""
    p = normalize_simplex(p0)
    n = p.size
    if n <= 1:
        return p, fast_score(p)

    best_p = p.copy()
    best_score = fast_score(p)
    step = step_init
    no_improve = 0
    it = 0

    # Break symmetry if needed
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-13:
        t_arr = np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 1e-5 * np.sin(0.3 * t_arr)))

    total_iter_target = max_iter
    while it < total_iter_target and time.monotonic() < deadline:
        it += 1

        # Adaptive beta: linearly interpolate in log space
        frac = min(1.0, it / max(1, total_iter_target))
        beta = beta_start * (beta_end / beta_start) ** frac

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_score = fast_score(p)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 2000 == 0:
            print(f"  GD it={it} beta={beta:.0f} best={best_score:.8f} step={step:.3e}")

        if no_improve > 5000:
            break

        # Compute gradient
        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Line search (backtracking)
        cur_smooth = n * G
        accepted = False
        base = step
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078):
            alpha = base * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if n * G2 < cur_smooth - 1e-12:
                p = q
                cur_smooth = n * G2
                accepted = True
                new_sc = fast_score(p)
                if new_sc < best_score - 1e-14:
                    best_score = new_sc
                    best_p = p.copy()
                break

        if accepted:
            if step < 3.0:
                step = min(3.0, step * 1.15)
        else:
            step *= 0.4
            if step < 1e-8:
                break

    return best_p, best_score

# Block-level optimization
# ============================================================

def block_to_p(n, B, c):
    """Convert block log-weights c to full sequence."""
    L = n // B
    if B <= 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    tmp = np.zeros(B - 1, dtype=np.float64)
    m = min(tmp.size, c.size)
    tmp[:m] = c[:m]
    c = tmp
    z = np.zeros(B, dtype=np.float64)
    z[:-1] = c
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    if B <= 1:
        return float('inf'), np.zeros(0, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    tmp = np.zeros(B - 1, dtype=np.float64)
    m = min(tmp.size, c.size)
    tmp[:m] = c[:m]
    c = tmp
    z = np.zeros(B, dtype=np.float64)
    z[:-1] = c
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(B - 1, dtype=np.float64)
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds, max_iter=150):
    """Run L-BFGS-B on block representation."""
    if not HAS_SCIPY:
        return c0
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    c0 = np.clip(c0, -8.0, 8.0)
    options = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 50, "maxcor": 12}
    try:
        res = minimize(block_objective, c0, args=(n, B, beta),
                       method='L-BFGS-B', jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)

# Structured starts
# ============================================================

def structured_starts(n, rng, max_count=10):
    """Generate diverse starting profiles."""
    starts = []
    p = np.full(n, 1.0 / float(n), dtype=np.float64)
    starts.append(p)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        z = a * (x - 0.5)
        q = np.exp(z)
        q /= q.sum()
        starts.append(q)

    # Two-level
    for eps in (0.5, 0.7, 0.85):
        q = np.ones(n, dtype=np.float64)
        half = n // 2
        q[:half] = 1.0 + eps
        q[half:] = max(1e-3, 1.0 - eps)
        q = np.maximum(q, 1e-8)
        q /= q.sum()
        starts.append(q)

    # Periodic patterns
    for freq in (3, 5, 8):
        t = 2 * np.pi * freq * x
        q = 1.0 + 0.5 * np.cos(t)
        q = np.maximum(q, 1e-6)
        q /= q.sum()
        starts.append(q)

    # Random with structure
    for scale in (1.0, 2.0, 4.0):
        z = rng.normal(size=n)
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        z = gaussian_filter1d(z, sigma=n // 20)
        z = scale * z - float(np.mean(z))
        q = np.exp(z)
        q = np.maximum(q, 1e-10)
        q /= q.sum()
        starts.append(q)

    return starts[:max_count]

# Main search function

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best sequence of non-negative heights minimizing
    2*n*max(a*a)/(sum(a))^2. Uses LP-based blending + gradient descent.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)

    # Sequence lengths to try (ordered by preference)
    lengths_to_try = [8192, 4096, 16384]

    # Get initial point from height_sequence_1 if available
    init_seq = None
    try:
        hs1 = height_sequence_1
        if isinstance(hs1, list) and len(hs1) > 0:
            init_seq = np.array(hs1, dtype=np.float64)
            init_seq = np.maximum(init_seq, 0.0)
            s = init_seq.sum()
            if s > 0:
                init_seq /= s
    except Exception:
        init_seq = None

    best_overall = None
    best_score_overall = float('inf')

    for N in lengths_to_try:
        if time.monotonic() >= deadline:
            break

        print(f"\n=== Length N={N} ===")
        t_start = time.monotonic()

        # Get initial point for this length
        if init_seq is not None and len(init_seq) == N:
            p0 = init_seq.copy()
        elif init_seq is not None and len(init_seq) > 0:
            # Interpolate
            p0 = np.interp(np.linspace(0, 1, N), np.linspace(0, 1, len(init_seq)), init_seq)
            p0 = normalize_simplex(p0)
        else:
            p0 = np.full(N, 1.0 / float(N), dtype=np.float64)

        p0 = normalize_simplex(p0)
        score0 = fast_score(p0)
        print(f"  Initial score: {score0:.8f}")

        best_p = p0.copy()
        best_score = score0

        # --- Phase 1: Block optimization (coarse to fine) ---
        if deadline - time.monotonic() > 120.0 and N >= 64:
            print("  Phase 1: Block optimization")
            block_candidates = []
            for B in (16, 32, 64, 128, 256, 512, 1024):
                if B < N and N % B == 0:
                    block_candidates.append(B)

            for B in block_candidates:
                if time.monotonic() >= deadline - 60.0:
                    break
                bounds = [(-8.0, 8.0)] * (B - 1)
                best_c = np.zeros(B - 1, dtype=np.float64)
                best_B_score = float('inf')

                # Multiple starts
                inits = [np.zeros(B - 1, dtype=np.float64)]
                x_arr = np.linspace(0.0, 1.0, B - 1, dtype=np.float64)
                for a in (-2.0, 2.0, -4.0, 4.0):
                    inits.append(a * (x_arr - 0.5))
                inits.append(np.cos(2 * np.pi * np.linspace(0, 2, B - 1)))
                inits.append(np.sin(2 * np.pi * np.linspace(0, 2, B - 1)))
                inits.append(1.0 * rng.normal(0, 1, B - 1))
                inits.append(2.0 * rng.normal(0, 1, B - 1))

                for c0 in inits:
                    if time.monotonic() >= deadline - 30.0:
                        break
                    c = run_block_minimize(c0, N, B, 20000.0, bounds, max_iter=80)
                    pp = block_to_p(N, B, c)
                    sc = fast_score(pp)
                    if np.isfinite(sc) and sc < best_B_score:
                        best_B_score = sc
                        best_c = c.copy()

                # Sharpen
                if time.monotonic() < deadline - 30.0 and np.isfinite(best_B_score):
                    c = run_block_minimize(best_c, N, B, 200000.0, bounds, max_iter=100)
                    pp = block_to_p(N, B, c)
                    sc = fast_score(pp)
                    if np.isfinite(sc) and sc < best_B_score:
                        best_B_score = sc
                        best_c = c.copy()
                    pp = block_to_p(N, B, best_c)

                if np.isfinite(best_B_score) and best_B_score < best_score:
                    best_score = best_B_score
                    best_p = pp.copy()
                    print(f"    Block B={B}: {best_score:.8f}")

        # --- Phase 2: Full gradient descent ---
        if deadline - time.monotonic() > 60.0:
            print("  Phase 2: Gradient descent")
            alloc = min(deadline - time.monotonic() - 30.0, 0.5 * (deadline - start_time))
            sub_deadline = time.monotonic() + alloc
            p_gd, sc_gd = full_gradient_descent(best_p, sub_deadline,
                                                beta_start=3000.0, beta_end=300000.0,
                                                max_iter=100000, step_init=0.8)
            if np.isfinite(sc_gd) and sc_gd < best_score:
                best_score = sc_gd
                best_p = p_gd.copy()
                print(f"    After GD: {best_score:.8f}")

        # --- Phase 3: LP blending ---
        if deadline - time.monotonic() > 60.0 and HAS_SCIPY:
            print("  Phase 3: LP blending")
            lp_iterations = 0
            consecutive_no_improve = 0

            while time.monotonic() < deadline - 30.0 and consecutive_no_improve < 5:
                lp_iterations += 1
                t_lp_start = time.monotonic()
                q = solve_blending_lp(best_p, max_n=4096, time_budget=30.0)
                t_lp = time.monotonic() - t_lp_start

                if q is None:
                    consecutive_no_improve += 2
                    break

                # Try blending with several t values
                improved = False
                for t_blend in (0.1, 0.2, 0.3, 0.15, 0.25, 0.4, 0.5):
                    p_blend = (1.0 - t_blend) * best_p + t_blend * q
                    p_blend = normalize_simplex(p_blend)
                    sc_blend = fast_score(p_blend)
                    if np.isfinite(sc_blend) and sc_blend < best_score - 1e-12:
                        best_score = sc_blend
                        best_p = p_blend.copy()
                        improved = True
                        print(f"    LP iter {lp_iterations} t={t_blend}: {best_score:.8f} (LP took {t_lp:.1f}s)")
                        break

                if improved:
                    consecutive_no_improve = 0
                    # Short GD polish after successful blend
                    if deadline - time.monotonic() > 40.0:
                        p_polish, sc_polish = full_gradient_descent(
                            best_p, min(deadline - 10.0, time.monotonic() + 20.0),
                            beta_start=50000.0, beta_end=200000.0,
                            max_iter=20000, step_init=0.5)
                        if np.isfinite(sc_polish) and sc_polish < best_score:
                            best_score = sc_polish
                            best_p = p_polish.copy()
                            print(f"    After polish: {best_score:.8f}")
                else:
                    consecutive_no_improve += 1

            print(f"  LP phase done: {lp_iterations} iterations")

        # --- Phase 4: Final gradient descent polish ---
        if deadline - time.monotonic() > 20.0:
            print("  Phase 4: Final polish")
            sub_deadline = deadline - 5.0
            p_final, sc_final = full_gradient_descent(best_p, sub_deadline,
                                                      beta_start=50000.0, beta_end=500000.0,
                                                      max_iter=50000, step_init=0.5)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_p = p_final.copy()
                print(f"    Final: {best_score:.8f}")

        # Update overall best
        if best_score < best_score_overall:
            best_score_overall = best_score
            best_overall = best_p.copy()
            print(f"  NEW OVERALL BEST: {best_score_overall:.8f} (N={N})")

        print(f"  N={N} done in {time.monotonic()-t_start:.1f}s, best={best_score:.8f}")

    # --- Finalize ---
    if best_overall is None:
        # Fallback: uniform
        N = 4096
        best_overall = np.full(N, 1.0 / float(N), dtype=np.float64)

    final_p = normalize_simplex(best_overall)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Quick official check
    try:
        off_val = evaluate_sequence(final_seq)
        if not np.isfinite(off_val) or off_val > 2.0:
            print(f"  WARNING: official score {off_val}, using fallback")
            N_fb = 4096
            final_seq = [1.0 / float(N_fb)] * N_fb
    except Exception:
        pass

    print(f"\nFinal: N={len(final_seq)}, score={best_score_overall:.8f}")
    return final_seq
