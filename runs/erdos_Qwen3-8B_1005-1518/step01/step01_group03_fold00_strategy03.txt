```python
def project_to_feasible(h):
    h_clipped = np.clip(h, 0, 1)
    sum_h = np.sum(h_clipped)
    if sum_h == 0:
        return h_clipped
    scaling = (n_points / 2) / sum_h
    return h_clipped * scaling

def mutate(h, rate):
    return h + np.random.normal(0, rate, h.shape)
```
