import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values as the starting point
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        # Compute the cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Extract maximum value and scale by dx
        return np.max(corr) * dx

    # Define the constraint: sum(h) == n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    # Bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimization with SLSQP
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Ensure the solution is valid
    h_values = result.x
    c5_bound = result.fun

    if not (np.allclose(h_values, h_values, atol=1e-8) and 
            np.all(0 <= h_values) and np.all(h_values <= 1) and 
            np.isclose(np.sum(h_values), n_points / 2, atol=1e-8)):
        # Fallback to initial values if optimization fails
        h_values = h_initial
        c5_bound = objective(h_initial)

    return (h_values, c5_bound, n_points)
