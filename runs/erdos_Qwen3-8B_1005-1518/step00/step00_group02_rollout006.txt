import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Initialize h with provided initial values or a uniform guess
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h = initial_h_values.copy()

        h = np.zeros(45)

        h = h_flat.copy()
        h = np.clip(h, 0, 1)

        h = h * scale
        cross_corr = correlate(h, 1 - h, mode='full')
