            next_pop.append(t[best])
        for i in range(0, 50, 2):
            p1 = next_pop[i]
            p2 = next_pop[i+1] if i+1 < 50 else next_pop[i]
            if np.random.rand() < 0.7:
                split = np.random.randint(1, n_points)
                c1 = np.concatenate([p1[:split], p2[split:]])
                c2 = np.concatenate([p2[:split], p1[split:]])
            else:
                s1, s2 = np.random.randint(1, n_points-1, 2)
                c1 = np.concatenate([p1[:s1], p2[s1:s2], p1[s2:]])
                c2 = np.concatenate([p2[:s1], p1[s1:s2], p2[s2:]])
            next_pop[i] = c1
            next_pop[i+1] = c2
        for i in range(50):
            if np.random.rand() < 0.05:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        next_pop[i][j] = np.random.rand()
                next_pop[i] = repair(next_pop[i])
        pop = next_pop
        for h in pop:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    return (best_h, best_c5, n_points)
