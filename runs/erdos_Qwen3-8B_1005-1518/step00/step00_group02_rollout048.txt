```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h0 = h_initial.copy()

    start_time = time.time()
    result = minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        return (h_initial, objective(h_initial), n_points)

    if result.success:
        h_values = result.x
        return (h_values, result.fun, n_points)
    else:
        return (h_initial, objective(h_initial), n_points)
```
