        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'ftol': 1e-8, 'maxiter': 1000})

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_initial, np.max(np.correlate(h_initial, (1 - h_initial), mode='full')) * dx, n_points)

    if result.success:
        h_values = result.x
        cross_corr = np.correlate(h_values, (1 - h_values), mode='full')
        c5_bound = np.max(cross_corr) * dx
        return (h_values, c5_bound, n_points)
    else:
        h_values = h_initial
        cross_corr = np.correlate(h_values, (1 - h_values), mode='full')
        c5_bound = np.max(cross_corr) * dx
        return (h_values, c5_bound, n_points)
