import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = initial_h_values  # Global input
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def repair_feasibility(h):
        h = np.clip(h, 0.0, 1.0)
        curr_sum = h.sum()
        if curr_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / curr_sum
        return h * scaling

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

    pop_size = 50
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        curr_sum = h.sum()
        if curr_sum == 0:
            h = np.zeros_like(h)
        else:
            h = h * (target_sum / curr_sum)
        population.append(h)

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    time_left = budget_s
    gen = 0

    while time_left > 0:
        start = time.time()
        new_pop = []
        for _ in range(pop_size):
            tourn = random.sample(population, 3)
            new_pop.append(min(tourn, key=compute_c5))
        for i in range(0, pop_size, 2):
            if random.random() < 0.7:
                pt = random.randint(1, n_points - 1)
                p1, p2 = new_pop[i], new_pop[i+1]
                c1 = np.concatenate([p1[:pt], p2[pt:]])
                c2 = np.concatenate([p2[:pt], p1[pt:]])
                new_pop[i], new_pop[i+1] = c1, c2
        for i in range(pop_size):
            if random.random() < 0.05:
                idx = random.randint(0, n_points - 1)
                new_pop[i][idx] += np.random.uniform(-0.1, 0.1)
                new_pop[i] = repair_feasibility(new_pop[i])
        new_pop = [repair_feasibility(p) for p in new_pop]
        for h in new_pop:
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
        population = new_pop
        time_left -= (time.time() - start)

    return best_h, best_c5, n_points
