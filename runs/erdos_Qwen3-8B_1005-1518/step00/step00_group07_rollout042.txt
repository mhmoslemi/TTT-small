import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Default n_points
    n_points = 80
    dx = 2.0 / n_points
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values
    else:
        h = np.zeros(n_points)
        h[n_points//4:-n_points//4] = 0.5
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h *= scale

    # Function to compute C5
    def compute_c5(h_val):
        overlap = np.correlate(h_val, 1 - h_val, mode='full') * dx
        return np.max(overlap)

    # Objective function to minimize C5
    def objective(h_val):
        return compute_c5(h_val)

    # Constraints
    def constraint_sum(h_val):
        return np.sum(h_val) - n_points / 2

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    # Optimization
    elapsed = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    while elapsed < budget_s:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-5)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
            elapsed = time.time() - start_time
        else:
            break

    return (best_h, best_c5, n_points)
