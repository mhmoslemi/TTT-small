import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
        s = np.sum(h)
        h = h * (n / 2) / s if abs(s - n / 2) > 1e-6 else h
        h = np.clip(h, 0, 1)
    else:
        n = 42
        h = np.full(n, 0.5)
        s = np.sum(h)
        h = h * (n / 2) / s
        h = np.clip(h, 0, 1)
    T, alpha, iters, sig = 1.0, 0.99, 10000, 0.1
    best_h = h.copy()
    best_c5 = 0.0

    def calc_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * (2.0 / n)

    best_c5 = calc_c5(best_h)

    for _ in range(iters):
        delta = np.random.randn(n) * sig
        candidate = h + delta
        candidate = np.clip(candidate, 0, 1)
        s_candidate = np.sum(candidate)
        candidate = candidate * (n / 2) / s_candidate
        c5_candidate = calc_c5(candidate)
        c5_current = calc_c5(h)
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h[:] = candidate[:]
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h[:] = candidate[:]
        T *= alpha
        sig *= 0.95

    s_best = np.sum(best_h)
    best_h = best_h * (n / 2) / s_best if abs(s_best - n / 2) > 1e-6 else best_h
    best_h = np.clip(best_h, 0, 1)
    best_c5 = calc_c5(best_h)
    return (best_h, best_c5, n)
