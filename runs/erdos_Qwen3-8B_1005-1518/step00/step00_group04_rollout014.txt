    # Compute initial best_overlap
    dx = 2.0 / n_points
    current_overlap = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
    best_overlap = current_overlap

    # SA parameters
    T = 1.0
    alpha = 0.99
    n_iterations = 10000
    sigma = 0.1

    # Start timer
    start_time = time.time()

    for iteration in range(n_iterations):
        # Compute current_overlap
        current_overlap = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx

        # Generate candidate_h
        delta = np.random.randn(n_points) * sigma
        candidate_h = current_h + delta
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_candidate = np.sum(candidate_h)
        if sum_candidate == 0:
            # Avoid division by zero
            candidate_h = current_h.copy()
        else:
            scale_factor = (n_points / 2) / sum_candidate
            candidate_h = candidate_h * scale_factor

        # Compute candidate_overlap
        candidate_overlap = np.max(np.correlate(candidate_h, 1 - candidate_h, mode='full')) * dx

        # Acceptance decision
        delta_c = candidate_overlap - current_overlap
        if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
            current_h = candidate_h.copy()

        # Update best_overlap
        if candidate_overlap < best_overlap:
            best_overlap = candidate_overlap

        # Cool down
        T *= alpha

        # Adjust sigma
        if iteration % 1000 == 0:
            sigma *= 0.95

        # Check time budget
        if time.time() - start_time > budget_s:
            break

    return (current_h, best_overlap, n_points)
