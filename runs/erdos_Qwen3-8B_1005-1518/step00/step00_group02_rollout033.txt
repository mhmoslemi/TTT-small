import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h0 = initial_h_values
    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time.time()
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'eps': 1e-8, 'maxiter': 500})
    elapsed_time = time.time() - start_time

    if elapsed_time > budget_s:
        return (h0, objective(h0), n_points)
    else:
        if result.success:
            h_values = result.x
            c5_bound = objective(h_values)
        else:

            c5_bound = objective(h0)
        return (h_values, c5_bound, n_points)
