        h[mid - 10:mid + 10] = 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    final_h = h.copy()
    start = time.time()
    for _ in range(1000):
        if time.time() - start > budget_s:
            break
        ov = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(ov)
        h_shifted = np.roll(h, k)
        grad = 1 - h_shifted
        s = np.zeros_like(h)
        s[grad < 0] = 1
        s_sum = np.sum(s)
        if s_sum > 0:
            s = s * (n_points / 2) / s_sum
        else:
            s = np.zeros_like(h)
            s[0] = n_points / 2
        eta = 0.5
        improved = False
        for _ in range(5):
            new_h = (1 - eta) * h + eta * s
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
            if new_c5 < best_c5:
                h = new_h.copy()
                best_c5 = new_c5
                improved = True
                break
            eta *= 0.5
        if not improved:
            break
        final_h = h.copy()
    return final_h, best_c5, n_points
