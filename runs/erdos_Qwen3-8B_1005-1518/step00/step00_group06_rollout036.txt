import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if available, otherwise construct a starting guess
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        # Construct a random starting vector with n/2 ones (for even n)
        n = 95  # Default n
        h_initial = np.zeros(n)
        num_ones = n // 2
        h_initial[:num_ones] = 1.0
        # Adjust for odd n to ensure sum is n/2
        if n % 2 == 1:
            h_initial[-1] += 0.5
    n = len(h_initial)
    dx = 2.0 / n

    # Objective function to minimize max overlap integral
    def objective(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx

    # Constraints: sum of h must be n / 2
    def constraint(h):
        return np.sum(h) - n / 2

    # Bounds for each variable [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{'type': 'eq', 'fun': constraint}]

    # Start optimization
    t0 = time.time()
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )
    elapsed = time.time() - t0

    # Early termination if time budget is exceeded
    if elapsed >= budget_s:
        return (h_initial, objective(h_initial), n)

    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n)
