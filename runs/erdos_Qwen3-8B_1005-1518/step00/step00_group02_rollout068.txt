    dx = 2.0 / n_points
    # Compute cross-correlation between h and 1 - h, mode 'full'
    corr = np.correlate(h, 1 - h, mode='full')
    # Find the maximum value and multiply by dx
    max_corr = np.max(corr) * dx
    return max_corr

def normalize_h(h, n_points):
    # Clip to [0, 1]
    h_clipped = np.clip(h, 0.0, 1.0)
    # Compute the total sum

        scaled = np.ones_like(h_clipped) * (n_points / 2) / n_points
    else:

        scaled = h_clipped * scale
    return scaled

def compute_c5(h, n_points):
