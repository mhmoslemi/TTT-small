        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    
    def generate_population(size):
        pop = [np.random.rand(n_points) * 2 for _ in range(size)]
        for h in pop:
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        return pop
    
    population = [initial_h]
    for _ in range(49):
        population.append(generate_population(1)[0])
    
    best_h = population[0]
    best_c5 = compute_c5(best_h, n_points)
    
    start_time = time.time()
    remaining_time = budget_s
    POP_SIZE = 50
    CROSSOVER_RATE = 0.7
    MUTATION_RATE = 0.05
    
    while remaining_time > 0.1:
        new_population = [np.copy(best_h)]
        for _ in range(POP_SIZE - 1):
            parent1 = select_parent(population)
            parent2 = select_parent(population)
            child = crossover(parent1, parent2)
            child = mutate(child, MUTATION_RATE)
            child = repair(child, n_points / 2)
            new_population.append(child)
        population = new_population
        for h in population:
            current_c5 = compute_c5(h, n_points)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        remaining_time = budget_s - (time.time() - start_time)
    
    return best_h, best_c5, n_points
