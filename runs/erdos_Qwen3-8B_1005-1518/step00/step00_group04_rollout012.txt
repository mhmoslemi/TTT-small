import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Assume initial_h_values is available
    # n_points is the length of initial_h_values
    n_points = len(initial_h_values)
    h_initial = initial_h_values
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if not np.isclose(sum_h, n_points / 2):
        raise ValueError("Initial h sum is not correct.")

    def objective(x):
        h = x.reshape(n_points)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    x0 = np.array(h_initial)

    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-7, 'maxiter': 200, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Early return if time is up
        return (x0, objective(x0), n_points)

    # Check constraint feasibility
    if not np.isclose(np.sum(result.x), n_points / 2):
        result.x = np.clip(result.x, 0, 1)
        result.x = result.x + (n_points / 2 - np.sum(result.x)) / n_points
    h_opt = result.x
    c5_bound = result.fun

    return (h_opt, c5_bound, n_points)
