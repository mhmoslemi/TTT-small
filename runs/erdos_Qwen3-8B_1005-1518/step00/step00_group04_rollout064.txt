import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Determine initial_h and n_points
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 42
        initial_h = np.full(n_points, 0.5)
        # Ensure sum is n_points/2
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    # Compute dx
    dx = 2.0 / n_points
    # Compute initial overlap
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_overlap = compute_overlap(best_h)
    current_h = best_h.copy()
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    max_iterations = 10000
    start_time = time.time()
    for i in range(max_iterations):
        # Generate candidate
        delta = np.random.randn(n_points) * sigma
        candidate_h = current_h + delta
        # Clip to [0, 1]
        candidate_h = np.clip(candidate_h, 0, 1)
        # Rescale to meet sum constraint
        sum_candidate = np.sum(candidate_h)
        if sum_candidate == 0:
            # handle this case, but unlikely
            continue
        scale = (n_points / 2) / sum_candidate
        candidate_h *= scale
        # Compute overlap
        candidate_overlap = compute_overlap(candidate_h)
        # Acceptance probability
        delta_C = candidate_overlap - best_overlap
        if delta_C < 0 or np.random.rand() < np.exp(-delta_C / T):
            current_h = candidate_h
            if candidate_overlap < best_overlap:
                best_overlap = candidate_overlap
                best_h = candidate_h.copy()
        # Cool temperature
        T *= alpha
        # Decrease sigma
        if i % 1000 == 0:
            sigma *= 0.95
        # Check time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    # Post-processing
    # Ensure best_h is within [0,1] and sum is correct
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        # Rescale again
        scale = (n_points / 2) / sum_best
        best_h *= scale
        best_h = np.clip(best_h, 0, 1)
    return (best_h, best_overlap, n_points)
