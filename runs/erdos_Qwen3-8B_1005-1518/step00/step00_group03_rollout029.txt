    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        overlaps = compute_overlaps(h)
        max_overlap = np.max(overlaps)
        current_C = max_overlap * dx

        current_k_list = [k for k in range(-n_points + 1, n_points) if overlaps[k + n_points - 1] == max_overlap]

        total_grad = np.zeros(n_points)
        for k in current_k_list:
            if k >= 0:
                pos = np.zeros(n_points)
                pos[0:n_points - k] = 1 - h[k:n_points]
                neg = np.zeros(n_points)
                neg[0:n_points - k] = -h[0:n_points - k]
            else:
                pos = np.zeros(n_points)
                pos[-k:n_points] = 1 - h[0:n_points + k]
                neg = np.zeros(n_points)
                neg[-k:n_points] = -h[-k:n_points]
            total_grad += pos + neg

        h_new = h - alpha * total_grad
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h = np.sum(h_new)
        if sum_h != 0:
            scale = (n_points / 2.0) / sum_h
            h_new = h_new * scale
        h_new = np.clip(h_new, 0.0, 1.0)

        overlaps_new = compute_overlaps(h_new)
        max_overlap_new = np.max(overlaps_new)
        new_C = max_overlap_new * dx

        if new_C < current_C:
            h = h_new.copy()
            current_C = new_C
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
        else:
            alpha *= backtracking_factor
            if alpha < 1e-10:
                break

    return (best_h.tolist(), best_C, n_points)
