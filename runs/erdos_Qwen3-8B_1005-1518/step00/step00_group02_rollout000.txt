import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h_initial = np.sum(initial_h_values)
    
    # Normalize initial_h_values to meet the sum constraint
    if abs(sum_h_initial - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h_initial
        initial_h_values = initial_h_values * scale
    
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Calculate time for each start
    time_per_start = budget_s // 30
    n_starts = min(10, int(budget_s // time_per_start))
    
    for _ in range(n_starts):
        # Generate noise and perturb
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = initial_h_values + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        scale = (n_points / 2) / sum_perturbed
        h_perturbed = h_perturbed * scale
        
        # Objective function
        def obj(h):
            cross_corr = np.correlate(h, 1 - h, mode='full')
            return np.max(cross_corr) * dx
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = opt.minimize(
                obj,
                h_perturbed,
                method='SLSQP',
                constraints=cons,
                bounds=bounds,
                options={'max_iter': 200, 'ftol': 1e-8, 'eps': 1e-8}
            )
            if res.success:
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = res.x.copy()
        except:
            pass
    
    # Final check with initial_h_values
    cross_corr_initial = np.correlate(initial_h_values, 1 - initial_h_values, mode='full')
    initial_c5 = np.max(cross_corr_initial) * dx
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h_values.copy()
    
    return best_h, best_c5, n_points
