import time

def run(seed=42, budget_s=1100, **kwargs):

    n = len(h)

    best_c5 = evaluate_erdos_solution(h)
    alpha = 0.1
    beta = 0.5
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        eps = 1e-5
        contrib = np.where(np.abs(corr - current_c5) < eps)[0]
        
        G = np.zeros_like(h)
        for j in contrib:
            k = j - (n - 1)
            if k >= 0:
                m = k
                G[0:n - m] += 1 - h[m:]
                G[m:] += -h[0:n - m]
            else:
                m = -k
                G[m:] += 1 - h[0:n - m]
                G[0:n - m] += -h[m:]
        
        h_new = np.clip(h - alpha * G, 0, 1)
        sum_h = np.sum(h_new)
        scale = (n / 2) / sum_h
        h_new *= scale
