        # Compute overlaps

        max_corr = np.max(corr)
        current_c = max_corr * dx
        # Find contributing shifts
        index_list = np.where(corr == max_corr)[0]
        s_list = [i - (n - 1) for i in index_list]

                if 0 <= m - s < n:
