        corr = np.correlate(h, 1 - h, 'full')
        return np.max(corr) * dx

    def repair(h):
        s = np.sum(h)
        target = n_points / 2
        if abs(s - target) < 1e-8:
            return h
        scale = target / s
        h = h * scale
        h = np.clip(h, 0, 1)
        chg = target - np.sum(h)
        if chg:
            h[0] += chg
        return h

    population = [initial_h.copy()]
    np.random.seed(seed)
    for _ in range(49):
        perturb = np.random.rand(n_points) * 0.2 - 0.1
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = repair(h)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    crossover_rate = 0.7
    mutation_rate = 0.05
    tournament_size = 3
    time_start = time.time()
    for _ in range(1000):
        if time.time() - time_start >= budget_s:
            break
        selected = []
        for _ in range(len(population)):
            tournament = np.random.choice(population, tournament_size)
            selected.append(min(tournament, key=compute_c5))
        selected = np.array(selected)
        children = []
        for i in range(0, len(selected), 2):
            a, b = selected[i], selected[i+1]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([a[:split], b[split:]])
                child2 = np.concatenate([b[:split], a[split:]])
                children.extend([child1, child2])
            else:
                children.extend([a, b])
        new_population = []
        for h in children:
            if np.random.rand() < mutation_rate:
                noise = np.random.rand(n_points) * 0.2 - 0.1
                h_mut = h + noise
                h_mut = np.clip(h_mut, 0, 1)
                h_mut = repair(h_mut)
                new_population.append(h_mut)
            else:
                new_population.append(h)
        population = new_population
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        if best_c5 == compute_c5(best_h):
            mutation_rate = min(mutation_rate + 0.05, 0.2)

    return best_h, best_c5, n_points
