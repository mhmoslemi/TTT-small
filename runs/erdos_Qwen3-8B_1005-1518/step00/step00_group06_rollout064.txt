def run(seed=42, budget_s=1100, **kwargs):

        h = initial_h_values

        n_points = 95
        h = np.zeros(n_points)

        idx = np.random.choice(n_points, cnt, replace=False)
        h[idx] = 1.0
        h = h.astype(float)
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    while time.time() - start_time < budget_s:
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(c)
        max_idx = np.argmax(c)
        shift = max_idx - (n - 1)
        h_shifted = np.roll(h, shift)
        grad = (1 - h_shifted) * dx
        grad_mean = np.mean(grad)
        grad = grad - grad_mean * np.ones_like(grad)
        alpha = np.inf
        for i in range(n):
            g = grad[i]
            if abs(g) < 1e-10:
                continue
            if g > 0:
                alpha_i = h[i] / g
            else:
                alpha_i = (1 - h[i]) / (-g)
            alpha = min(alpha, alpha_i)
        h -= alpha * grad
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-8:
            h = np.clip(h, 0.0, 1.0)
            scaling_factor = 1.0 / np.sum(h) * (n / 2.0) / dx
            h = h * scaling_factor
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
    return (best_h, best_c5, n)
