import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        corr = correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        return c5

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {
        'maxiter': 500,
        'ftol': 1e-7,
        'xtol': 1e-7,
        'disp': False
    }

    try:
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options=options
        )
    except:
        result = minimize(
            objective,
            h_initial,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=constraints,
            options=options
        )

    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
