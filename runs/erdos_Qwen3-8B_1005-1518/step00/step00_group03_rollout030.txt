import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    initial_guess = initial_h_values.copy()

    st = time()
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8,
                      options={'maxiter': 1000})
    et = time()

    h_values = result.x
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx

    # Early exit if time budget is exhausted
    if et - st > budget_s:
        return (h_values, c5_bound, n_points)

    return (h_values, c5_bound, n_points)
