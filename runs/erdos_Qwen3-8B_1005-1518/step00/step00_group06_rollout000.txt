import numpy as np
import numpy.fft as fft
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in locals() else None
    n_points = 95
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.concatenate([h, np.zeros(n_points//2)])
        h = np.sin(np.pi * np.arange(n_points) / n_points) * 0.5 + 0.5
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = np.interp(h, (0, 1), (n_points/2, n_points/2))
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    for _ in range(10000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Find the shift with maximum overlap
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        k = k % n_points
        # Compute gradient vector for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] = (1 - h[j]) * dx
        # Adjust h in the direction to reduce the overlap
        step_size = 0.01
        h_new = h - step_size * grad
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize sum to n_points / 2
        sum_h_new = np.sum(h_new)
        scale = (n_points / 2) / sum_h_new if sum_h_new != 0 else 1.0
        h_new *= scale
        h = h_new
    return (best_h, best_c5, n_points)
