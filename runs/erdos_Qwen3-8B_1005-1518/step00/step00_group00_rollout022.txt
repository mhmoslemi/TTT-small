import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    # Predefined initial values if available
    initial_h_values = kwargs.get('initial_h_values', [0.5] * 45)
    n_points = 45
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize h
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    # Parameters
    max_iter = 1000
    step_size = 0.1
    decay_rate = 0.995
    eval_freq = 10
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for iter in range(max_iter):
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr * dx)

        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Early termination if budget is exhausted
        if time.time() - start_time > budget_s:
            break

        # Find the k with maximum overlap
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Compute gradient for this k
        grad = np.zeros_like(h)
        for j in range(n_points):
            i_plus_k = (j + k) % n_points
            i_minus_k = (j - k) % n_points
            grad[j] = 1.0 - h[i_plus_k] - h[i_minus_k]

        # Update h
        h -= step_size * grad

        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        # Normalize to maintain target sum
        h_sum = np.sum(h) * dx
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        else:
            h = h.copy()

        # Decay step size
        step_size *= decay_rate

        # Optionally evaluate periodically
        if iter % eval_freq == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr * dx)
            print(f"Iter {iter}, C5: {current_c5}")

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h.tolist(), final_c5, n_points)
