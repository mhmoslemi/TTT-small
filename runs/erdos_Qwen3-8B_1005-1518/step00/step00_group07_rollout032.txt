            grad[i] = (c5_perturb - c5_perturb2) / (2 * epsilon)
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), 0):
            h = h * (n_points / 2) / np.sum(h)
        
        if iter_count % 100 == 0:
            h += np.random.normal(0, perturb_strength, n_points)
            h = np.clip(h, 0, 1)
            if not np.isclose(np.sum(h), 0):
                h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
        iter_count += 1
    
    return best_h, best_c5, n_points
