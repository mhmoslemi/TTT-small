                      bounds=bounds, constraints={'type': 'eq', 'fun': constraint_func},
                      options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8})
    
    elapsed = time.time() - start_time
    if result.success:
        h_opt = result.x
    else:
        h_opt = initial_h
    
    c5 = objective(h_opt)
    return h_opt, c5, n_points
