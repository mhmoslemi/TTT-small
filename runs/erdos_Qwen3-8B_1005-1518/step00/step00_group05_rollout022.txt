import numpy as np
import random
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_gen = 100
    best_c5 = float('inf')
    best_h = initial_h.copy()
    start_time = time.time()
    
    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    def normalize(h):
        s = np.sum(h)
        return np.clip(h * (n_points / 2) / s, 0, 1)
    
    def crossover(a, b):
        split = np.random.randint(1, n_points - 1)
        return np.concatenate([a[:split], b[split:]])
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
        return normalize(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    def constraints(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Initialize population with variations
    population = [normalize(initial_h + np.random.uniform(-0.1, 0.1, n_points)) for _ in range(100)]
    
    for gen in range(max_gen):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        scores = [compute_c5(h) for h in population]
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h = population[best_idx].copy()
        parents = [population[i] for i in np.argsort(scores)[:int(len(population)*0.2)]]
        offspring = []
        for _ in range(len(population) - len(parents)):
            a, b = random.sample(parents, 2)
            child = crossover(a, b)
            child = mutate(child)
            offspring.append(child)
        population = parents + offspring
    
    # Gradient refinement
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraints})
    optimal_h = result.x
    optimal_c5 = compute_c5(optimal_h)
    
    return (optimal_h, optimal_c5, n_points)
