import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import time
    
    # Start timing
    start_time = time.time()
    
    # Assume initial_h_values is available
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Define objective function
    def obj(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    # Constraints: sum(h) = n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(
        fun=obj,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_initial, obj(h_initial), n_points)
    
    # Return the best solution found
    best_h = result.x
    best_c5 = obj(best_h)
    
    return (best_h, best_c5, n_points)
