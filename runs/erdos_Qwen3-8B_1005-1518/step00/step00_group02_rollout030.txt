import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h_initial = initial_h_values
    dx = 2.0 / n_points

    def objective(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )

    h_opt = result.x
    c5_bound = objective(h_opt)

    # Feasibility check
    if not (np.all(h_opt >= 0) and np.all(h_opt <= 1) and np.isclose(np.sum(h_opt), n_points / 2)):
        h_opt = h_initial
        c5_bound = objective(h_opt)

    return h_opt, c5_bound, n_points
