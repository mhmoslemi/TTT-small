    def compute_c5(h):

        return np.max(corr) * dx

    def repair_feasibility(h):
        current_sum = h.sum()
        target_sum = n_points / 2
        if abs(current_sum - target_sum) < 1e-6:
            return h
        delta = target_sum - current_sum
        if delta > 0:
            for i in range(n_points):
                if h[i] < 1:
                    add = min(delta, 1 - h[i])
                    h[i] += add
                    delta -= add
                    if delta == 0:

        else:
            for i in range(n_points):
                if h[i] > 0:
                    rem = min(-delta, h[i])
                    h[i] -= rem
                    delta += rem
                    if delta == 0:

        return h
