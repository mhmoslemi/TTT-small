    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    time_remaining = budget_s

    max_iter = 1000
    step_size = 0.01
    perturb_rate = 0.1
    perturb_strength = 0.05
    stagnation = 0
    max_stagnation = 50

    for _ in range(max_iter):
        start = time.time()
        cor = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cor)
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for m in range(n_points):
            i_plus_k = m + k
            if 0 <= i_plus_k < n_points:
                grad[m] += dx * (1 - h[i_plus_k])
            i_minus_k = m - k
            if 0 <= i_minus_k < n_points:
                grad[m] -= dx * h[i_minus_k]
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h *= (n_points / 2) / total
        c5 = evaluate_erdos_solution(h, n_points, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            stagnation += 1
            if stagnation >= max_stagnation:
                indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
                h[indices] += np.random.uniform(-perturb_strength, perturb_strength, size=len(indices))
                h = np.clip(h, 0, 1)
                total = np.sum(h)
                if total > 0:
                    h *= (n_points / 2) / total
                stagnation = 0
        time_remaining -= (time.time() - start)
        if time_remaining < 0:
            break
    return (best_h, best_c5, n_points)
