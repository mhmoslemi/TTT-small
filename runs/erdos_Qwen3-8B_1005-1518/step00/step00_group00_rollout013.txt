        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return overlaps

        overlaps = compute_overlaps(h)
        max_overlap = overlaps.max()

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.full(n_points, n_points / 2.0 / n_points)
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def compute_overlaps(h):
        return np.correlate(h, 1 - h, mode='full') * dx
    
    while time.time() - start_time < budget_s:
        overlaps = compute_overlaps(h)
        max_overlap = overlaps.max()
        max_lag = np.argmax(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        grad = np.zeros(n_points)
        k = max_lag
        if k >= 0:
            m = k
            for i in range(n_points - m):
                grad[i] += (1 - h[i + m])
                grad[i + m] -= h[i]
        else:
            m = -k
            for i in range(n_points - m):
                grad[i] += (1 - h[i + k])
                grad[i + k] -= h[i]
        
        alpha = 0.5
        rho = 0.5
        c = 0.1
        while True:
            h_next = h - alpha * grad
            h_next = np.clip(h_next, 0, 1)
            current_sum = h_next.sum()
            target_sum = n_points / 2
            if current_sum > target_sum + 1e-6:
                scale = target_sum / current_sum
                h_next *= scale
            else:
                deficit = target_sum - current_sum
                available = np.sum(1 - h_next)
                if available > 1e-6:
                    h_next += deficit * (1 - h_next) / available
            overlaps_next = compute_overlaps(h_next)
            if overlaps_next.max() < overlaps.max() + c * alpha * grad.dot(grad):
                break
            alpha *= rho
        
        h = h_next
        
        if time.time() - start_time >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
