        n_points = len(h_initial)
    else:
        h_initial = np.zeros(45)
        n_points = 45
    population_size = 50
    population = []
    if h_initial is not None:
        population.append(h_initial.copy())
        for _ in range(population_size - 1):
            h_rand = np.random.rand(n_points)
            current_sum = np.sum(h_rand)
            scale = (n_points / 2.0) / current_sum if current_sum != 0 else 1.0
            h_rand = h_rand * scale
            h_rand = np.clip(h_rand, 0, 1)
            population.append(h_rand)
    else:
        for _ in range(population_size):
            h_rand = np.random.rand(n_points)
            current_sum = np.sum(h_rand)
            scale = (n_points / 2.0) / current_sum if current_sum != 0 else 1.0
            h_rand = h_rand * scale
            h_rand = np.clip(h_rand, 0, 1)
            population.append(h_rand)
    best_h = None
    best_c5 = float('inf')
    for h in population:
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        values = corr * dx
        c5 = np.max(values)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    start_time = time.time()
    mutation_rate = 0.05
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(population_size):
            tournament = np.random.choice(population, size=3, replace=False)
            parent = min(tournament, key=lambda x: compute_c5(x, n_points))
            if np.random.rand() < 0.7:
                cross_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent[:cross_point], parent[cross_point:]])
            else:
                cross_points = np.sort(np.random.choice(n_points, size=2, replace=False))
                child = np.concatenate([parent[:cross_points[0]], parent[cross_points[0]:cross_points[1]], parent[cross_points[1]:]])
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        delta = np.random.uniform(-0.1, 0.1)
                        child[i] += delta
                        child[i] = np.clip(child[i], 0, 1)
            required_sum = n_points / 2.0
            current_sum = np.sum(child)
            if current_sum != 0:
                scale = required_sum / current_sum
                child *= scale
                child = np.clip(child, 0, 1)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        population = new_population
        for h in population:
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode='full')
            values = corr * dx
            c5 = np.max(values)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)
