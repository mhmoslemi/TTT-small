    import time

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 80
        h = np.zeros(n_points)
        h[n_points//4 : 3*n_points//4] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)

    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    max_iter = int(budget_s / 50)

    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < 1e-10:
            break

        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k] * dx

        h = np.clip(h - 0.01 * grad, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)

        if time.time() - start_time > budget_s:
            break

    corr = correlate(h, 1 - h, mode='full')
    return (h, np.max(corr) * dx, n_points)
