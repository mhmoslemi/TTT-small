import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        h = np.round(h).astype(int)
    else:
        n_points = 70
        h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, 35, replace=False)
        h[indices] = 1
    
    assert np.sum(h) == 35
    dx = 2.0 / n_points

    def compute_c5(h_array):
        corr = np.correlate(h_array, 1.0 - h_array, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()
    temp = 1.0

    while True:
        current_time = time.time()
        if current_time - time_start > budget_s:
            break

        h_candidate = h.copy()
        zeros = np.where(h_candidate == 0)[0]
        ones = np.where(h_candidate == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            continue
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]

        new_c5 = compute_c5(h_candidate)

        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_candidate.copy()
                best_c5 = new_c5

        temp *= 0.999

    return (h.tolist(), best_c5, n_points)
