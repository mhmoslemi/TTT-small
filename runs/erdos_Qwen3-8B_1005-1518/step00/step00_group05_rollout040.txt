        optimized_h = result.x
        optimized_c5 = compute_c5(optimized_h)
        return optimized_h, optimized_c5

    best_h = h
    best_c5 = current_c5
    max_iter = 5

    for _ in range(max_iter):
        h, c5 = optimize_once(best_h, best_c5, n_points)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        else:
            break

    return (best_h, best_c5, n_points)
