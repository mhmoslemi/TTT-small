def run(seed=42, budget_s=1100, **kwargs):

def evaluate_particle(h, n_points):
    h_neg = 1 - h
    corr = np.correlate(h, h_neg, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx
