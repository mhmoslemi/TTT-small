import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n_points = h.size
    else:
        n_points = 42
        h = np.full(n_points, 0.5)

    dx = 2.0 / n_points

    def objective(h):
        correl = np.correlate(h, 1 - h, mode='full')
        return np.max(correl) * dx

    # Constraint function for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        # Initial evaluation to get baseline
        initial_c5 = objective(h)

        # Time budget estimation: set maximum iterations based on budget
        max_iterations = int((budget_s / 1100) * 2000)
        result = optimize.minimize(
            objective, h, method='SLSQP',
            constraints=constraints,
            bounds=bounds,
            options={'maxiter': max_iterations, 'disp': False}
        )

        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = h
        best_c5 = initial_c5

    # Ensure constraints are satisfied after optimization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, n_points / 2, rtol=1e-6, atol=1e-6):
        scaling_factor = (n_points / 2) / sum_h
        best_h = best_h * scaling_factor

    # Final evaluation
    current_c5 = objective(best_h)
    if current_c5 < best_c5:
        best_c5 = current_c5

    return (best_h, best_c5, n_points)
