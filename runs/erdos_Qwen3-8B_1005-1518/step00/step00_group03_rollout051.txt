import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    
    # Check for initial_h_values
    if 'initial_h_values' in vars() or 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback to a simple initial guess (unused if initial_h_values is available)
        h = np.zeros(70, dtype=np.float64)
        h[:35] = 1.0
        h = h / 35 * 35  # Ensure sum is correct
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    alpha = 0.1
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_mask = np.isclose(corr, max_corr, atol=1e-8)
        indices = np.where(max_mask)[0]
        shifts = [i - (n_points - 1) for i in indices]
        
        total_grad = np.zeros(n_points)
        for k in shifts:
            grad = np.zeros(n_points)
            # Compute contributions from j + k
            j_plus_k = np.arange(n_points) + k
            valid_plus = j_plus_k < n_points
            if np.any(valid_plus):
                grad[valid_plus] += 1.0 - h[j_plus_k[valid_plus]]
            # Compute contributions from j - k
            j_minus_k = np.arange(n_points) - k
            valid_minus = j_minus_k >= 0
            if np.any(valid_minus):
                grad[valid_minus] -= h[j_minus_k[valid_minus]]
            total_grad += grad
        
        # Update h
        h_new = h - alpha * total_grad
        h_new = np.clip(h_new, 0, 1)
        s = np.sum(h_new)
        scale = (n_points / 2.0) / s if s != 0 else 1.0
        h_new *= scale
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            alpha = 0.1
        else:
            alpha *= 0.5
        
        h = h_new
    
    return best_h, best_c5, n_points
