import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def objective(h):
        return compute_c5(h)

    def optimize_start(start_h):
        h0 = np.array(start_h)
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0., 1.) for _ in range(n_points)]
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            return result.x, result.fun
        else:
            return None, np.inf

    start_time = time.time()
    best_h = np.array(initial_h)
    best_c5 = compute_c5(best_h)

    # Initial optimization
    h_opt, c5_opt = optimize_start(best_h)
    if h_opt is not None and c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    num_starts = min(5, int((budget_s - (time.time() - start_time)) / 100))
    for _ in range(num_starts):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        noise = np.random.normal(0, 0.1, size=n_points)
        perturbed = initial_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        h_opt, c5_opt = optimize_start(perturbed)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_best
        best_h = np.clip(best_h * scale, 0, 1)
        sum_best = np.sum(best_h)
    return best_h, best_c5, n_points
