import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default starting guess with 45 points
        n_points = 45
        h = np.zeros(n_points)
        # Fill half with 1.0 to meet the integral constraint
        h[:n_points // 2] = 1.0
        # Ensure the sum is correct
        req_sum = n_points / 2
        if abs(np.sum(h) - req_sum) > 1e-6:
            scale = req_sum / np.sum(h)
            h *= scale
    
    n_points = len(h)
    if n_points == 0:
        return (np.array([]), 0.0, 0)
    
    dx = 2.0 / n_points
    
    # Objective function to minimize the maximum overlap
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx
    
    # Constraints: sum of h must be n_points / 2
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    
    # Bounds: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with bounds and constraints
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Compute C5 with initial guess if time runs out
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        return (h, c5, n_points)
    
    best_h = result.x
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
