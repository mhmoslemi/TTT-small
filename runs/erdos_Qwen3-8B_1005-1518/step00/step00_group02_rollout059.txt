        initial_h = np.zeros(n_points, dtype=np.float64)
        initial_h[:22] = 1.0
        initial_h[22] = 0.5
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def cons(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0., 1.) for _ in range(n_points)]
    
    # Run optimization within time budget
    result = opt.minimize(obj, initial_h, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': cons}],
                          bounds=bounds,
                          options={'ftol': 1e-6, 'maxiter': 1000})
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, obj(initial_h), n_points)
    
    if result.success:
        h_best = result.x
    else:
        h_best = initial_h
    
    corr_best = np.correlate(h_best, 1 - h_best, mode='full')
    c5_bound = np.max(corr_best) * dx
    return (h_best, c5_bound, n_points)
