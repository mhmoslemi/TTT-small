import numpy as np
import random
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

    sum_val = np.sum(h)
    required_sum = n_points / 2.0
    scaling = required_sum / sum_val if sum_val != 0 else 1.0
    h = h * scaling

    sum_val = np.sum(h)
    delta = required_sum - sum_val
    if delta != 0:
        h[0] += delta
        h[0] = np.clip(h[0], 0, 1)
    return h
