import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    h = h * (n_points / 2) / (np.sum(h) * dx)
    
    def evaluate_overlap(h):
        l = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(l)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        l = np.correlate(h, 1 - h, mode='full') * dx
        k_val = np.argmax(l)
        k_shift = k_val - (n_points - 1)
        
        C = cp.Variable()
        h_vars = cp.Variable(n_points)
        constraints = [cp.sum(h_vars) == n_points / 2, 0 <= h_vars, h_vars <= 1]
        sum_overlap = 0
        for i in range(n_points):
            j = (i + k_shift) % n_points
            sum_overlap += h_vars[i] * (1 - h_vars[j]) * dx
        constraints.append(sum_overlap <= C)
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal' and prob.value < best_c5:
                best_c5 = prob.value
                best_h = h_vars.value
                h = best_h
        except:
            break
    
    return best_h, best_c5, n_points
