```python
            available = np.sum(1 - h)
            if available >= deficit:
                h += deficit * (1 - h) / available
            else:
                pass  # Infeasible, skip

    # Check time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        break
```

c. Backtracking line search:
```python
def backtracking_line_search(h, grad, max_lag, current_overlaps, dx):
    alpha = 0.5
    rho = 0.5
    c = 0.1

    while True:
        h_next = h - alpha * grad
        h_next = np.clip(h_next, 0, 1)
        sum_next = h_next.sum()
        if sum_next < (n_points / 2 - 1e-6):
            deficit = (n_points / 2) - sum_next
            available = np.sum(1 - h_next)
            if available > 0:
                h_next += deficit * (1 - h_next) / available
        overlaps_next = np.correlate(h_next, 1 - h_next, mode='full') * dx
        max_overlap_next = overlaps_next.max()
        if max_overlap_next < current_overlaps.max() + c * alpha * grad.dot(grad):
            break
        alpha *= rho
    return alpha
```
