        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    elapsed = time.time() - start_time

    best_h = result.x
    best_c5 = objective(best_h)

    # Ensure feasibility
    if not (np.all(best_h >= 0) and np.all(best_h <= 1) and np.isclose(np.sum(best_h), n_points / 2)):
        best_h = h_initial
        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
