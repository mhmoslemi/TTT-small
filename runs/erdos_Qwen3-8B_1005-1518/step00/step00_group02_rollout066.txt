    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    def constraint_func(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    perturbations = 3
    for _ in range(perturbations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        perturbed_h = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        perturbed_h += noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        if current_sum != 0:
            perturbed_h = perturbed_h * (target_sum / current_sum)
        else:
            perturbed_h = np.zeros(n_points)

        result = opt.minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'maxiter': 100, 'ftol': 1e-7}
        )

        if result.success:
            current_h = result.x
        else:
            current_h = perturbed_h

        corr = np.correlate(current_h, 1 - current_h, mode='full')
        dx = 2.0 / n_points
        current_c5 = np.max(corr * dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    dx = 2.0 / n_points
    c5_bound = np.max(corr * dx)

    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if current_sum != 0:
        best_h = best_h * (target_sum / current_sum)
    else:
        best_h = np.zeros(n_points)

    return (best_h, c5_bound, n_points)
