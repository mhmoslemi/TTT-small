import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    def compute_c5(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    num_runs = 50
    for _ in range(num_runs):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        perturbed = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale = (n_points / 2.0) / sum_perturbed
        perturbed *= scale
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        def objective(x):
            return compute_c5(x)
        
        result = minimize(objective, perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, n_points)
