    else:
        n_points = initial_h_values.shape[0]
        target_sum = n_points / 2.0
        pop_size = 100
        population = [initial_h_values.copy()]

        best_c5 = np.inf
        def compute_c5(h):

        best_c5 = compute_c5(best_h)

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(pop_size):
            idxs = np.random.choice(len(population), size=3, replace=False)
            candidates = [population[i] for i in idxs]
            c5s = [compute_c5(h) for h in candidates]
            best_idx = np.argmin(c5s)
            parent1 = candidates[best_idx]
            while True:
                idx = np.random.randint(len(population))
                if idx != idxs[best_idx]:
                    parent2 = population[idx]
                    break
            point = np.random.randint(1, n_points)
            child = np.zeros(n_points)
            child[:point] = parent1[:point]
            child[point:] = parent2[point:]
            if np.random.rand() < 0.05:
                idx = np.random.randint(n_points)
                child[idx] = np.random.rand()
            current_sum = np.sum(child)
            if abs(current_sum - target_sum) > 1e-6:
                scale = target_sum / current_sum
                child = np.clip(child * scale, 0, 1)
            new_population.append(child)
        population = new_population
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    return best_h, best_c5, n_points
