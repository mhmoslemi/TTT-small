import time

def run(seed=42, budget_s=1100, **kwargs):

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    if not np.isclose(np.sum(h), 0.5 * n_points):
        h = h * (0.5 * n_points) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    time_start = time.time()
    h_vars = cp.Variable(n_points, name='h')
    C = cp.Variable()
    constraints = [cp.sum(h_vars) == 0.5 * n_points]
    constraints += [0 <= h_vars <= 1]
    while time.time() - time_start < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr * dx
        c5_current = np.max(overlap)
        if c5_current <= best_c5:
            break
        best_c5 = c5_current
        best_h = h.copy()
        k_idx = np.argmax(overlap)
        k = k_idx - (n_points - 1)
        if k < 0:
            k += 2 * n_points
        overlap_expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap_expr += h_vars[i] * (1 - h_vars[j])
        overlap_expr *= dx
        constraints.append(overlap_expr <= C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
