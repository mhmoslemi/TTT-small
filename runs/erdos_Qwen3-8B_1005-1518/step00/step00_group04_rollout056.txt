import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    # Use initial_h_values if available, else create a uniform starting guess
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(42, 0.5)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Objective function: max of cross-correlation with (1 - h)
    def objective(h_vec):
        corr = np.correlate(h_vec, (1 - h_vec), mode='full')
        c5 = corr.max() * dx
        return c5
    
    # Constraints: sum(h) must equal n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds: each element must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Setup optimization
    res = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    
    best_h = res.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
