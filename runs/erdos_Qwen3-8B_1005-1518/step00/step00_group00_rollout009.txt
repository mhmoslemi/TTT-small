    while time_remaining > 0:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = overlaps.max()
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find the shift with maximum overlap
        k_max = overlaps.argmax()
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k_max < n_points:
                grad[j] += (1 - h[j + k_max]) * dx
            if 0 <= j - k_max < n_points:
                grad[j] -= h[j - k_max] * dx
            
        # Adjust the most influential index
        j_max = np.argmax(np.abs(grad))
        step = 0.05 * grad[j_max]
        h[j_max] += step
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Check time
        elapsed = time.time() - start_time
        time_remaining = budget_s - elapsed
        if time_remaining <= 0:
            break

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = np.correlate(h, 1 - h, mode="full") * dx
    final_c5 = final_c5.max()
    return (h, final_c5, n_points)
