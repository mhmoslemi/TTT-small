import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    # Normalize to ensure sum(h) * dx == 1
    scale = (n_points / 2) / np.sum(h)
    h *= scale

    def compute_overlap(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_overlap(h)
    best_h = h.copy()
    best_c5 = current_c5

    max_iter = 100
    iter_count = 0
    start_time = time.time()

    while iter_count < max_iter and time.time() - start_time < budget_s:
        def objective(x):
            return compute_overlap(x)

        def constraint(x):
            return np.sum(x) - n_points / 2

        constraints = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Use previous h as starting point
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        if res.success:
            best_h = res.x
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
        iter_count += 1

    return best_h.tolist(), best_c5, n_points
