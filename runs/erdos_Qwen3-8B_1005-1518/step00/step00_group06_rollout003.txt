            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            if sum_minus > 0:
                scaled_minus = h_minus * (target_sum / sum_minus)
                c5_minus = objective(scaled_minus.flatten())
            else:
                c5_minus = float('inf')
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    result = minimize(
        fun=objective,
        x0=h.flatten(),
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        scale_factor = target_sum / final_sum
        final_h *= scale_factor
        c5_value = evaluate_erdos_solution(final_h)
        if c5_value < best_c5:
            best_h = final_h.copy()
            best_c5 = c5_value
    else:
        final_h = best_h.copy()
        c5_value = best_c5

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        final_h = best_h.copy()
        c5_value = best_c5

    return (best_h.tolist(), c5_value, n_points)
