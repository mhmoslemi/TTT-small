import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n_points = len(initial_h_values) if initial_h_values is not None else 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = initial_h_values.copy() if initial_h_values is not None else np.random.rand(n_points) * 2 - 1
    best_c5 = float('inf')
    start_time = time.time()
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.1
    gen = 0
    max_generations = int(budget_s / 0.5)
    
    def evaluate_erdos_solution(h):
        h = np.clip(h, 0, 1)
        non_zero = np.sum(h) * dx
        if np.isclose(non_zero, target_sum):
            h = h / np.sum(h) * target_sum
        else:
            h = h / non_zero * target_sum
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = target_sum / total
        h = h * scale
        return h
    
    population = [best_h.copy() for _ in range(population_size)]
    
    while time.time() - start_time < budget_s:
        if gen >= max_generations:
            break
        gen += 1
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:int(population_size * elite_ratio)]]
        parents = [population[i] for i in sorted_indices[:population_size]]
        
        offspring = []
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                parent1, parent2 = random.sample(parents, 2)
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                parent = random.choice(parents)
                child = parent.copy()
            mutation = np.random.normal(0, mutation_rate, size=n_points)
            child += mutation
            child = project_h(child)
            offspring.append(child)
        
        population = elites + offspring
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
    
    best_h = project_h(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
