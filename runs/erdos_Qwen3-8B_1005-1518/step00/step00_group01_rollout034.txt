    total_mass = n_points / 2.0  # Constraint: sum(h) = n_points / 2
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        return np.max(overlaps)
    
    # Initial evaluation
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    # Optimization loop
    max_iter = 200  # Adjust based on budget
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Evaluate current overlaps
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find the worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        max_idx = np.argmax(overlaps)
        k = max_idx - (n_points - 1)
    
        # Build the objective for this shift
        def objective(x):
            # x is a vector of h values
            h_new = x
            # Compute the overlap for the worst shift
            # j + k is within the array
            # Compute the overlap contribution for this shift k
            # O = sum_{i} h[i]*(1 - h[i +k])
            # Due to numpy's mode 'full' correlate, the shift k is as calculated
            j_plus_k = (np.arange(n_points) + k) % n_points
            j_minus_k = (np.arange(n_points) - k) % n_points
            o = np.sum(h_new * (1 - h_new[j_plus_k]))
            return o * dx
        
        # Set bounds and constraints
        bounds = Bounds(0.0, 1.0)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - total_mass})
        
        # Perform a local optimization to reduce current maximum
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            h = result.x
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
