import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize the random seed
    np.random.seed(seed)
    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:

    best_h = h.copy()

    dx = 2.0 / n_points
    
    # Define a helper function to compute C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Initialize SA parameters
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    max_iter = 10000
    current_h = h.copy()

    for iter in range(max_iter):

            candidate_h = current_h.copy()
