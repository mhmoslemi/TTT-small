import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Use provided initial_h_values if available
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    # Feasibility check and scaling
    if np.isclose(np.sum(h_initial), n_points / 2):
        initial_guess = h_initial.copy()
    else:
        initial_guess = (h_initial * (n_points / 2) / np.sum(h_initial)).clip(0, 1)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimization settings
    options = {
        'maxiter': 1000,
        'ftol': 1e-8,
        'disp': False,
        'eps': 1e-8,
        'tol': 1e-8
    }
    
    start_time = time.time()
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints, options=options)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Early termination if time is exceeded
        return initial_guess, objective(initial_guess), n_points
    
    # Post-processing
    h_best = np.clip(result.x, 0, 1)
    # Ensure sum constraint is satisfied
    h_best = (h_best * (n_points / 2) / np.sum(h_best)).clip(0, 1)
    c5_bound = objective(h_best)
    return h_best, c5_bound, n_points
