def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 50
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    target_sum = n_points / 2
    if not np.isclose(sum_h, target_sum):
        scale = target_sum / sum_h
        h *= scale
        h = np.clip(h, 0, 1)
    
    def evaluate_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def clip_rescale(h, n_points):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        target_sum = n_points / 2
        scale = target_sum / sum_h
        h *= scale
        return np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_C5(best_h)
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        candidates = [best_h]
        for _ in range(19):
            mutated = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0, 1)
            mutated = clip_rescale(mutated, n_points)
            candidates.append(mutated)
        
        for cand in candidates:
