    n_points = len(initial_h) if initial_h is not None else 45  # default to 45 if not provided
    if initial_h is not None:
        h = np.array(initial_h, dtype=np.float64)
    else:
        h = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    # Ensure integral constraint is satisfied
    sum_h = np.sum(h) * dx
    scale = 1.0 / sum_h if sum_h != 0 else 1.0
    h *= scale

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    start_time = time.time()
    max_iterations = int(budget_s / 1)  # Simplified iteration heuristic
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Compute gradient for overlap at k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = dx * (1 - h[j])
                grad[j] = -dx * h[i]
        
        # Update h
        step_size = 0.1
        h_new = h - step_size * grad
        h_clipped = np.clip(h_new, 0, 1)
        scaled_h = h_clipped * (1.0 / (np.sum(h_clipped) * dx))
        h = scaled_h

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
