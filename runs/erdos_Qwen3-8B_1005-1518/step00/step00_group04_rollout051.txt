import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 42
        initial_h = 0.5 * np.ones(n_points)
    
    sum_initial = np.sum(initial_h)
    if abs(sum_initial - n_points / 2) > 1e-6:
        initial_h = initial_h * (n_points / 2) / sum_initial
    initial_h = np.clip(initial_h, 0, 1)
    current_h = initial_h.copy()
    best_h = current_h.copy()
    dx = 2.0 / n_points
    best_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
    
    T = 1.0
    alpha = 0.99
    total_iter = 10000
    sigma = 0.1
    start_time = time.time()
    
    for iter in range(total_iter):
        if time.time() - start_time > budget_s:
            break
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        
        delta_h = np.random.randn(n_points) * sigma
        candidate_h = current_h + delta_h
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_candidate = np.sum(candidate_h)
        if sum_candidate == 0:
            candidate_h = current_h.copy()
        else:
            candidate_h = candidate_h * (n_points / 2) / sum_candidate
        
        candidate_c5 = np.max(np.correlate(candidate_h, 1 - candidate_h, mode='full')) * dx
        
        delta_c = candidate_c5 - current_c5
        if delta_c < 0 or np.exp(-delta_c / T) > np.random.rand():
            current_h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
        
        T *= alpha
        if iter % 1000 == 0 and iter != 0:
            sigma *= 0.95
    
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_best
    
    return (best_h, best_c5, n_points)
