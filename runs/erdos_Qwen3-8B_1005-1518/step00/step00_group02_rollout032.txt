import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h_values = np.zeros(n_points)
        half = n_points / 2
        h_values[:int(half)] = 1.0
        sum_h = np.sum(h_values)
        scaling = (n_points / 2) / sum_h
        h_values *= scaling
    n_points = len(h_values)
    dx = 2.0 / n_points
    start_time = time.time()
    
    def objective(h):
        h = np.array(h, dtype=float)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h_values, method='SLSQP', bounds=bounds, constraints=constraints,
                      tol=1e-6, options={'maxiter': 1000, 'ftol': 1e-6, 'eps': 1e-6})
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:

    else:
        best_h = result.x
        best_c5 = objective(best_h)
