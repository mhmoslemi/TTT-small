        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    population = [initial_h_values.copy()]
    for _ in range(49):
        h = initial_h_values.copy()
        for _ in range(5):
            idx = np.random.randint(len(h))
            delta = np.random.uniform(-0.05, 0.05) * h[idx]
            h[idx] += delta
            idx2 = np.random.randint(len(h))
            while idx2 == idx:
                idx2 = np.random.randint(len(h))
            h[idx2] -= delta
            h = np.clip(h, 0, 1)
        population.append(h)

    for _ in range(20):
        fitness = [compute_c5(p) for p in population]
        top_indices = np.argsort(fitness)[:int(0.2 * len(population))]
        parents = [population[i] for i in top_indices]
        offspring = []
        for _ in range(len(population) - len(parents)):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                point = np.random.randint(1, len(p1) - 1)
                child = p1.copy()
                child[:point] = p2[:point]
            else:
                p = np.random.choice(parents)
                child = p.copy()
                idx = np.random.randint(len(child))
                delta = np.random.uniform(-0.05, 0.05) * child[idx]
                child[idx] += delta
                idx2 = np.random.randint(len(child))
                while idx2 == idx:
                    idx2 = np.random.randint(len(child))
                child[idx2] -= delta
                child = np.clip(child, 0, 1)
            if np.random.rand() < 0.1:
                idx = np.random.randint(len(child))
                delta = np.random.uniform(-0.05, 0.05) * child[idx]
                child[idx] += delta
                idx2 = np.random.randint(len(child))
                while idx2 == idx:
                    idx2 = np.random.randint(len(child))
                child[idx2] -= delta
                child = np.clip(child, 0, 1)
            offspring.append(child)
        combined = population + offspring
        combined.sort(key=lambda x: compute_c5(x))
        population = combined[:len(population)]

    best_h = min(population, key=compute_c5)

    def objective(x):
        return compute_c5(x)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = opt.minimize(
        fun=objective,
        x0=best_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 200}
    )

    h_opt = result.x
    c5_bound = compute_c5(h_opt)
    return h_opt, c5_bound, n_points
