    total = n_points / 2
    h = (h * total) / np.sum(h)  # Ensure sum(h) == total

    def evaluate_c5(h_val):
        corr = scipy.signal.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start = time.time()

    while time.time() - start < budget_s:
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        w, h_shift = np.unravel_index(max_idx, (n_points, n_points))
        w -= n_points - 1
        shift = w

        # Local QP optimization on worst shift
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(dx * cp.sum(h_var * (1 - h_var[(np.arange(n_points) + shift) % n_points])))
        constr = [cp.sum(h_var) == total, 0 <= h_var, h_var <= 1]
        try:
            prob = cp.Problem(obj, constr)
            prob.solve(max_iter=100, solver=cp.SCS)
            if prob.status == 'optimal':
                h = h_var.value
        except:
            pass

        # Random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        h = (h * total) / np.sum(h)

        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
