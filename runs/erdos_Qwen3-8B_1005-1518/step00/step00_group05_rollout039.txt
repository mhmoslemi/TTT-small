import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    def compute_c5(h):

        return np.max(conv) * dx

    def adjust_sum(h, target_sum):
        delta = target_sum - np.sum(h)
        if delta == 0:
            return h
        idx = np.random.choice(n_points)
        h[idx] += delta
        h = np.clip(h, 0, 1)
        return h

    population_size = 100
    population = []
    base = initial_h_values.copy()
    population.append(base)
    for _ in range(population_size - 1):
        h = base.copy()
        rand = np.random.uniform(-0.1, 0.1, n_points)
        h += rand
        h = np.clip(h, 0, 1)
        h = adjust_sum(h, n_points / 2)
        population.append(h)

    start_time = time.time()
    generations = 100
    for gen in range(generations):
        fitness = [compute_c5(h) for h in population]
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        parents = sorted_pop[:int(len(sorted_pop) * 0.2)]
        offspring = []
        for _ in range(len(population) - len(parents)):
            if np.random.rand() < 0.8:

                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                p = random.choice(parents)

                    child[i] += np.random.uniform(-0.05, 0.05)

            child = adjust_sum(child, n_points / 2)
