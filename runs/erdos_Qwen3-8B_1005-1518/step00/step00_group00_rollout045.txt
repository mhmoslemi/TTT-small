    best_h = h
    best_val = compute_c5(h)
    pop_size = 50

    start_time = time.time()
    population = [h.copy() for _ in range(pop_size)]

    while time.time() - start_time < budget_s:
        evals = np.array([compute_c5(p) for p in population])
        idx = np.argsort(evals)
        new_pop = [population[i] for i in idx[:int(pop_size*0.5)]]

        for _ in range(pop_size - len(new_pop)):
            a, b = np.random.choice(population, 2, replace=False)
            p = np.random.rand()
            c = np.concatenate([a[:int(n_points*p)], b[int(n_points*p):]])
            if np.random.rand() < 0.05:
                i, j = np.random.choice(n_points, 2, replace=False)
                c[i] += np.random.uniform(-0.1, 0.1)
                c[j] -= np.random.uniform(-0.1, 0.1)
                c = np.clip(c, 0, 1)
                s = n_points / 2.0
                d = s - np.sum(c)
                if d != 0:
                    k = np.random.randint(n_points)
                    c[k] += d
                    c = np.clip(c, 0, 1)
            new_pop.append(c)

        population = new_pop
        for h in population:
            v = compute_c5(h)
            if v < best_val:
                best_val = v
                best_h = h.copy()

    return best_h, best_val, n_points

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))
