```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    elapsed = time.time() - start
    if elapsed > budget_s:
        return (result.x, objective(result.x), n_points)
    return (result.x, objective(result.x), n_points)
```
