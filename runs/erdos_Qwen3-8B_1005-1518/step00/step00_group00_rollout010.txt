    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    max_iterations = int(100000 * (budget_s / 1100))
    for _ in range(max_iterations):
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while i == j:
            j = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        temp_h = h.copy()
        temp_h[i] += delta
        temp_h[j] -= delta
        temp_h = np.clip(temp_h, 0, 1)
        c5 = evaluate_erdos_solution(temp_h)
        if c5 < best_c5:
            best_h = temp_h
            best_c5 = c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return best_h, best_c5, n_points
