    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        scale = (len(h) / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
    else:
        n = 80
        h = np.zeros(n)
        h[n//4:-n//4] = 0.5
        h = np.clip(h, 0, 1)
        scale = (n / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
    
    def compute_c5(h_vec):
        dx = 2.0 / len(h_vec)
        overlap = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(overlap) * dx
    
    def objective(h_vec):
        return compute_c5(h_vec)
    
    n = len(h)
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-9})
    
    if result.success:
        best_h = result.x
    else:
        best_h = h
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (best_h, compute_c5(best_h), len(best_h))
    
    for _ in range(5):
        if elapsed > budget_s - 10:
            break
        new_h = best_h + np.random.uniform(-0.05, 0.05, size=best_h.shape)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n/2) / np.sum(new_h)
        current_c5 = compute_c5(best_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5:
            best_h = new_h
        elapsed = time.time() - start_time
    
    return (best_h, compute_c5(best_h), len(best_h))
