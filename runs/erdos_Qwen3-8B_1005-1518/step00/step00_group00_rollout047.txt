import numpy as np
import random
import time

def repair(h, n_points):
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-6:
        return h
    scale = target_sum / current_sum
    h_scaled = h * scale
    h_scaled = np.clip(h_scaled, 0.0, 1.0)
    return h_scaled

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    return np.max(corr) * dx
