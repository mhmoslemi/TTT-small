        conv = np.correlate(h, 1 - h, mode='full')
        max_conv = np.max(conv) * dx
        return max_conv
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B for faster convergence
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
