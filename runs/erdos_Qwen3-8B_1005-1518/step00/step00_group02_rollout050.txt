import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    h_start = initial_h_values.copy()

    best_h = h_start
    best_c5 = compute_c5(best_h)

    for _ in range(3):
        perturb = np.random.normal(0, 0.15, n_points)
        h_start = initial_h_values + perturb
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        result = minimize(compute_c5, h_start, method='SLSQP',
                          constraints={'type': 'eq', 'fun': constraint_sum},
                          bounds=bounds, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
