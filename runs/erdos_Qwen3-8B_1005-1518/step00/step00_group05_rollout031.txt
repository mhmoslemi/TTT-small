import numpy as np
import scipy.optimize as opt
import random
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Efficient choice for time budget
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Normalize initial_h_values
    initial_h_values = np.clip(initial_h_values, 0, 1)
    total = np.sum(initial_h_values)
    if abs(total - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values / total * (n_points / 2)
    
    # Initialize population
    population = np.zeros((100, n_points))
    for i in range(100):
        population[i] = initial_h_values.copy()
        # Add small random perturbations
        population[i] += np.random.uniform(-0.1, 0.1, n_points)
        population[i] = np.clip(population[i], 0, 1)
        # Enforce sum constraint
        population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
    
    def fitness(h):
        padded_h = np.pad(h, (0, n_points), 'constant')
        kernel = 1 - h
        conv = correlate(padded_h, kernel, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def evaluate(h):
        return fitness(h)
    
    # Genetic algorithm parameters
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    for gen in range(generations):
        # Evaluate population
        fitnesses = np.array([evaluate(h) for h in population])
        # Select top 20% as parents
        parent_indices = np.argsort(fitnesses)[:int(len(population)*0.2)]
        parents = population[parent_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(len(population) - len(parents)):
            # Crossover
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, int(n_points * 0.05), replace=False)
                child[mutation_indices] += np.random.uniform(-0.1, 0.1, size=len(mutation_indices))
                child = np.clip(child, 0, 1)
                # Enforce sum constraint
                child = child / np.sum(child) * (n_points / 2)
            offspring.append(child)
        
        # Replace population
        population[parent_indices] = offspring
        # Update best
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Gradient refinement
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine top 5 candidates
    candidates = population[:5]
    for h in candidates:
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
    
    return best_h, best_c5, n_points
