import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    np.random.seed(seed)
    
    # Preprocess initial_h to ensure feasibility
    h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        scale = (n_points / 2) / sum_h
        h *= scale
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    # Define objective function
    def objective(h_vec):
        dx = 2.0 / len(h_vec)
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    # Define constraints
    def eq_cons(h_vec):
        return np.sum(h_vec) - len(h_vec) / 2
    
    def ineq_cons_low(h_vec):
        return h_vec
    
    def ineq_cons_high(h_vec):
        return 1 - h_vec
    
    cons = (
        {'type': 'eq', 'fun': eq_cons},
        {'type': 'ineq', 'fun': ineq_cons_low},
        {'type': 'ineq', 'fun': ineq_cons_high}
    )

    num_starts = 10

        noise = np.random.normal(0, sigma, n_points)
