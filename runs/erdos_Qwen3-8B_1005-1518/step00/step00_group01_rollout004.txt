    mutation_rate = 0.1
    while time.time() - start_time < budget_s:
        fitness = [evaluate_c5(ind, n_points) for ind in population]
        sorted_indices = np.argsort(fitness)
        elite_size = int(0.5 * pop_size)
        elites = population[sorted_indices[:elite_size]]
        offspring = []
        for _ in range(pop_size - elite_size):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            mut = np.random.normal(0, mutation_rate, len(child))
            child = np.clip(child + mut, 0, 1)
            child = project_h(child, T)
            offspring.append(child)
        population = elites + offspring
        current_fitness = [evaluate_c5(ind, n_points) for ind in population]
        best_idx = np.argmin(current_fitness)
        current_best_h = population[best_idx]
        current_best_c5 = current_fitness[best_idx]
        if current_best_c5 < best_c5:
            best_h = current_best_h
            best_c5 = current_best_c5
        mutation_rate *= 0.995
    best_h = project_h(best_h, T)
    best_c5 = evaluate_c5(best_h, n_points)
    return (best_h, best_c5, n_points)
