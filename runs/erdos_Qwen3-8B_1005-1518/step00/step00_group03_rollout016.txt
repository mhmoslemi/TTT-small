def run(seed=42, budget_s=1100, **kwargs):

        if time.time() - start_time > budget_s:
            break
        np.random.seed(seed + cycle)
        shifts = np.correlate(h, 1 - h, mode='full')
        current_shifts = shifts * dx
        max_c = np.max(current_shifts)
        idx_max = np.where(np.isclose(current_shifts, max_c, atol=1e-6))[0]
        
        grad = np.zeros(n)
        for k_idx in idx_max:
            k_val = k_idx - (n-1)

                j_plus_k = j + k_val
                if 0 <= j_plus_k < n:
                    grad[j] += (1 - h[j_plus_k])

                if 0 <= j_minus_k < n:
