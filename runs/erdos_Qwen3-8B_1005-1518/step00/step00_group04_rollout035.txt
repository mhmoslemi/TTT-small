import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
    n_points = len(h)
    sum_h = np.sum(h)

    current_h = h.copy()
    best_h = h.copy()

    best_overlap = current_overlap

            break
        delta = np.random.randn(n_points) * sigma
        candidate = current_h + delta
        candidate = np.clip(candidate, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate == 0:
