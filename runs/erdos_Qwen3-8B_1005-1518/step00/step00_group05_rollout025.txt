    res = minimize(obj, h_upscaled, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 50})
    if elapsed < budget_s:
        if res.success:
            optimized_h = res.x
            optimized_c5 = res.fun
        else:
            optimized_h = h_upscaled
            optimized_c5 = compute_c5(optimized_h, dx)
    else:
        optimized_h = h_upscaled
        optimized_c5 = compute_c5(optimized_h, dx)
    return (optimized_h, optimized_c5, 140)
