import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    from scipy import correlate
    import sys
    
    # Assume initial_h_values is accessible
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlaps(h):
        return correlate(h, 1 - h, mode='full') * dx
    
    overlaps = compute_overlaps(h)
    current_max = overlaps.max()
    best_c5 = current_max
    best_h = h.copy()
    
    alpha = 0.1
    backtracking_factor = 0.5
    max_iter = int(budget_s / 20)  # ~50 iterations
    
    start_time = time.time()
    
    for _ in range(max_iter):
        overlaps = compute_overlaps(h)
        current_max = overlaps.max()
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            alpha = 0.1
        else:
            alpha *= backtracking_factor
        
        mask = (overlaps == current_max)
        k_list = np.where(mask)[0] - n_points + 1
        
        gradient = np.zeros(n_points)
        for k in k_list:
            if k >= 0:
                max_i = n_points - k
                indices = np.arange(max_i)
                gradient[indices] += 1.0 - h[indices + k]
            else:
                m = -k
                indices = np.arange(m, n_points)
                gradient[indices] += 1.0 - h[indices + k]
            start_second = max(k, 0)
            indices_second = np.arange(start_second, n_points)
            gradient[indices_second] += -h[indices_second - k]
        
        h_new = h - alpha * gradient
        clipped = np.clip(h_new, 0.0, 1.0)
        sum_clipped = np.sum(clipped)
        scaling = target_sum / sum_clipped if sum_clipped != 0 else 1.0
        scaled = clipped * scaling
        scaled = np.clip(scaled, 0.0, 1.0)
        h = scaled.copy()
        
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, n_points
