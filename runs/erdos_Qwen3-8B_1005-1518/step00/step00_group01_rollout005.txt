    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 47
        initial_h = np.random.rand(n_points)
        target_sum = n_points / 2
        s = np.sum(initial_h)
        if s != 0:
            initial_h = initial_h * (target_sum / s)
        else:
            initial_h[:] = 0
        initial_h = np.clip(initial_h, 0, 1)
        s = np.sum(initial_h)
        if s != 0:
            initial_h = initial_h * (target_sum / s)
        else:
            initial_h[:] = 0
    pop_size = 100
    population = [initial_h.copy() for _ in range(pop_size)]
    best_h = initial_h
    best_c5 = float('inf')
    start_time = time.time()
    mutation_rate = 0.1
    crossover_rate = 0.8
    while True:
        fitness = []
        for h in population:
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            fitness.append(c5)
        top_indices = np.argsort(fitness)[:int(pop_size * 0.5)]
        top_population = [population[i] for i in top_indices]
        new_population = []
        while len(new_population) < pop_size - len(top_population):
            parent1, parent2 = np.random.choice(top_population, 2)
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1 + np.random.normal(0, mutation_rate, n_points)
            child = np.clip(child, 0, 1)
            target_sum = n_points / 2
            current_sum = np.sum(child)
            if current_sum == 0:
                child = np.zeros_like(child) + target_sum / n_points
            else:
                scale = target_sum / current_sum
                child = child * scale
                child = np.clip(child, 0, 1)
            new_population.append(child)
        population = top_population + new_population
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        for h in population:
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    one_minus_h = 1.0 - best_h
    corr = np.correlate(best_h, one_minus_h, mode='full')
    best_c5 = np.max(corr) * (2.0 / n_points)
    return (best_h, best_c5, n_points)
