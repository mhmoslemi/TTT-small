import numpy as np
import random

def run(seed=42, budget_s=1100):
    if 'initial_h_values' in globals() and initial_h_values is not None:

        target = n_points / 2

            return h
        scale = target / s

        clipped = np.clip(scaled, 0, 1)

    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(1 - h, h, mode='full')

        new_pop = []
