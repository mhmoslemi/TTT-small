def run(seed=42, budget_s=1100, **kwargs):
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 95
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        h = h.astype(np.float64)
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    time_start = time.time()
    max_iter = int(budget_s / (n_points * 10))
    
    for _ in range(max_iter):
        c = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c)
        k = max_idx - (n_points - 1)
        h_shifted = np.roll(h, k)
        grad = (1 - h_shifted) * dx
        grad_avg = np.mean(grad)
        grad_proj = grad - grad_avg * np.ones_like(grad)
        
        alpha = np.inf
        for i in range(n_points):
            if grad_proj[i] == 0:
                continue
            if grad_proj[i] > 0:
                alpha_i = h[i] / grad_proj[i]
            else:
                alpha_i = (1 - h[i]) / (-grad_proj[i])
            alpha = min(alpha, alpha_i)
        
        h_new = h - alpha * grad_proj
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new + (target_sum - np.sum(h_new)) / n_points
        
        new_c5 = c[max_idx] * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        if time.time() - time_start > budget_s:
            break
    
    return (best_h, best_c5, n_points)
