import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Constants
    n_points = 200  # Tuned for performance and quality balance
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if i % 3 == 0 else 0.0 for i in range(n_points)])
    # Initialize with provided or generated values
    if 'initial_h_values' in kwargs:
        h = np.array(kwargs['initial_h_values']).copy()
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * (n_points / 2)
        h = np.clip(h, 0, 1)

    # PSO parameters
    swarm_size = 30
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_scale = 1.0
    max_iter = int(budget_s * 1000 / (n_points**2))
    early_stop = 50
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [compute_c5(p) for p in particles]
    global_best = best_h.copy()
    global_best_c5 = best_c5

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            # Compute current C5
            current_c5 = compute_c5(particles[i])
            # Update personal best
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = current_c5
            # Update velocity and position
            v = velocities[i] * w
            r1, r2 = np.random.rand(), np.random.rand()
            v += c1 * r1 * (personal_best[i] - particles[i])
            v += c2 * r2 * (global_best - particles[i])
            velocities[i] = np.clip(v, -velocity_scale, velocity_scale)
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            # Update global best
            if current_c5 < global_best_c5:
                global_best = particles[i].copy()
                global_best_c5 = current_c5
        # Early stopping check
        if iter > 0 and abs(global_best_c5 - best_c5) < 1e-6:
            if iter > early_stop:
                break
        best_c5 = global_best_c5
    # Final check
    final_c5 = compute_c5(global_best)
    return (global_best, final_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / len(h))
