    h *= scale_factor
    n_points = n_initial
    best_h = h.copy()
    best_c5 = compute_C5(h, 2.0 / n_points)
    start_time = time.time()
    
    population_size = 20
    max_generations = 500
    mutation_rate = 0.1
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.uniform(-0.1, 0.1, len(h))
        new_h = h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        scale_factor = (n_points / 2) / sum_new
        new_h *= scale_factor
        population.append(new_h)
    
    for gen in range(max_generations):
        candidates = []
        for candidate in population:
            c5 = compute_C5(candidate, 2.0 / n_points)
            candidates.append((c5, candidate))
        candidates.sort()
        next_pop = [c[1] for c in candidates[:int(len(candidates)*0.2)]]
        while len(next_pop) < population_size:
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = parent1.copy()
            for i in range(len(child)):
                if np.random.rand() < 0.5:
                    child[i] = parent2[i]
            mask = np.random.rand(len(child)) < mutation_rate
            delta = np.random.uniform(-0.05, 0.05, np.sum(mask))
            child[mask] += delta
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            scale_factor = (n_points / 2) / sum_child
            child *= scale_factor
            next_pop.append(child)
        population = next_pop
        for c in candidates:
            if c[0] < best_c5:
                best_c5 = c[0]
                best_h = c[1].copy()
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return best_h, best_c5, n_points
