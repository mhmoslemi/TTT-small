    h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = h * scale

    # Objective function
    def obj(x):
        cross_corr = np.correlate(x, 1 - x, mode='full')
        return np.max(cross_corr) * dx

    # Constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    num_perturbations = int(budget_s // 100)
    noise_scale = 0.1 * np.sqrt(1.0 / n_points)

    for _ in range(num_perturbations):
        perturbed = h + np.random.normal(0, noise_scale, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        scale_perturbed = (n_points / 2) / sum_perturbed if sum_perturbed != 0 else 1.0
        perturbed = perturbed * scale_perturbed

        res = opt.minimize(obj, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return (best_h, best_c5, n_points)
