import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h_values = initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    bounds = [(0., 1.) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}]
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    np.random.seed(seed)
    start_time = time.time()
    
    for _ in range(50):
        if time.time() - start_time >= budget_s:
            break
        perturbed = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        perturbed *= (n_points/2) / sum_p if sum_p > 1e-8 else 1
        def obj(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1-h, mode='full') * dx
            return np.max(corr)
        res = minimize(obj, x0=perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    def obj_initial(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1-h, mode='full') * dx
        return np.max(corr)
    initial_c5 = obj_initial(initial_h_values)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h_values
    
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h = best_h * (n_points/2) / sum_best if sum_best > 1e-8 else best_h
    
    return best_h, best_c5, n_points
