import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 42
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.full(n_points, 0.5)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    current_h = h.copy()
    best_h = h.copy()
    dx = 2.0 / n_points
    best_overlap = compute_overlap(current_h, dx)
    T = 1.0
    alpha = 0.99
    sigma = 0.1
    max_iter = 10000
    start_time = time.time()
    
    for iter in range(max_iter):
        delta = np.random.randn(n_points) * sigma
        candidate = current_h + delta
        candidate = np.clip(candidate, 0, 1)
        sum_c = np.sum(candidate)
        if sum_c != 0:
            candidate = candidate * (n_points / 2) / sum_c
        curr_val = compute_overlap(current_h, dx)
        cand_val = compute_overlap(candidate, dx)
        delta_c = cand_val - curr_val
        if delta_c < 0 or np.random.rand() < np.exp(-delta_c / T):
            current_h[:] = candidate[:]
            if cand_val < best_overlap:
                best_h[:] = candidate[:]
                best_overlap = cand_val
        T *= alpha
        if iter % 1000 == 0 and T > 1e-8:
            sigma *= 0.95
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    dx = 2.0 / n_points
    final_overlap = compute_overlap(best_h, dx)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_best
    return (best_h, final_overlap, n_points)

def compute_overlap(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
