    return max_corr * dx

def repair_feasibility(h, n_points):
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    scaling_factor = target_sum / current_sum
    h_scaled = h * scaling_factor
    h_clamped = np.clip(h_scaled, 0, 1)
    return h_clamped
