def run(seed=42, budget_s=1100, **kwargs):
    import initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    h = np.array(h_initial, dtype=float)
    
    def objective(x):
        h_val = x
        corr = np.correlate(h_val, 1 - h_val, mode="full")
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - (n_points / 2)})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
    
    if result.success:
        best_h = result.x
        c5_bound = result.fun
    else:
        best_h = h
        c5_bound = objective(h)
    
    return (best_h, c5_bound, n_points)
