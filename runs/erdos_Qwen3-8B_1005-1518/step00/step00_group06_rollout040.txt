import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    if initial_h_values is None:
        n_points = 95
        h = np.zeros(n_points)
        np.random.seed(seed)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = 1.0
        initial_h_values = h
    else:
        n_points = len(initial_h_values)
    
    dx = 2.0 / n_points
    
    def objective(h):
        cross_corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
        if result.success:
            best_h, best_c5 = result.x, result.fun
    
    return (best_h, best_c5, n_points)
