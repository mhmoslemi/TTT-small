        h0[:n_points//2] = 1.0
        if n_points % 2 == 1:
            h0[-1] = 0.5
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    def objective(h):
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (result.x, result.fun, n_points)
    
    best_h = result.x
    c5 = result.fun
    return (best_h, c5, n_points)
