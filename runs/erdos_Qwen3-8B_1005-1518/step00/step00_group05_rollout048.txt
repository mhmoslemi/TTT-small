    pop_size = 100
    cross_rate = 0.8
    mut_rate = 0.05
    
    for gen in range(max_gen):
        fits = []
        for h in population:
            fits.append((evaluate_erdos_solution(h, n_points), h))
        fits.sort()
        parents = [f[1] for f in fits[:int(pop_size * 0.2)]]
        
        offspring = []
        for _ in range(pop_size - len(parents)):
            p1, p2 = random.choice(parents), random.choice(parents)
            if random.random() < cross_rate:
                split = np.random.randint(1, n_points)
                child = np.concatenate([p1[:split], p2[split:]])
            else:
                child = p1.copy()
            
            if random.random() < mut_rate:
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.1, 0.1)
                new_val = child[idx] + delta
                new_val = max(0.0, min(1.0, new_val))
                sum_diff = new_val - child[idx]
                j = (idx + 1) % n_points
                child[j] -= sum_diff
                child[j] = max(0.0, min(1.0, child[j]))
            
            offspring.append(child)
        
        combined = parents + offspring
        fits_combined = [(evaluate_erdos_solution(h, n_points), h) for h in combined]
        fits_combined.sort()
        population = [f[1] for f in fits_combined[:pop_size]]
    
    best = min(population, key=lambda h: evaluate_erdos_solution(h, n_points))
    
    def objective(x):
        h = x.reshape(n_points)
        return evaluate_erdos_solution(h, n_points)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    res = opt.minimize(objective, best, method='L-BFGS-B', bounds=bounds, constraints=cons)
    optimized = res.x
    
    return (optimized, evaluate_erdos_solution(optimized, n_points), n_points)
