    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    n_points_current = len(best_h)
    start_time = time.time()
    max_iter = 100
    pop_size = 20
    mutation_rate = 0.1
    mutation_step = 0.1
    crossover_rate = 0.5
    time_budget = budget_s
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        candidates = []
        # Mutate best_h
        new_h = best_h.copy()
        for i in range(len(new_h)):
            if random.random() < mutation_rate:
                new_h[i] += np.random.uniform(-mutation_step, mutation_step)
                new_h[i] = np.clip(new_h[i], 0.0, 1.0)
        total = np.sum(new_h)
        scale = (len(new_h)/2) / total
        new_h *= scale
        c5 = evaluate_erdos_solution(new_h)
        candidates.append((c5, new_h))
        
        # Generate crossover candidates
        if pop_size > 1:
            for _ in range(10):
                p1, p2 = random.choices([best_h] + candidates, k=2)
                child = p1.copy()
                for i in range(len(child)):
                    if random.random() < crossover_rate:
                        child[i] = p2[i]
                total = np.sum(child)
                scale = (len(child)/2) / total
                child *= scale
                c5 = evaluate_erdos_solution(child)
                candidates.append((c5, child))
        
        # Select best candidate
        best_candidate = min(candidates, key=lambda x: x[0])
        if best_candidate[0] < best_c5:
            best_c5 = best_candidate[0]
            best_h = best_candidate[1]
    
    return (best_h, best_c5, n_points_current)
