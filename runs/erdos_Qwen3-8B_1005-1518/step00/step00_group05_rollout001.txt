            offspring.append(child)
        population += offspring
        population = population[:pop_size]
    
    top_candidates = population[:int(0.1 * pop_size)]
    best_h = top_candidates[0]
    best_c5 = compute_C5(best_h)
    for h in top_candidates[1:]:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(lambda x: compute_C5(x), h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=cons)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    return (best_h, best_c5, n_points)
