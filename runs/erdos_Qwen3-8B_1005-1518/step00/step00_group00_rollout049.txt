    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h  # Enforce integral constraint
    
    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    remaining_time = budget_s
    
    # Optimization parameters
    learning_rate = 0.01
    max_iterations = 10000
    iteration = 0
    
    while remaining_time > 0 and iteration < max_iterations:
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        corr_values = corr * dx
        max_c5 = np.max(corr_values)
        idx = np.argmax(corr_values)
        
        # Gradient computation for the max lag
        k = idx
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] = (1 - h[i + k]) * dx
        
        # Gradient descent step
        h -= learning_rate * grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
        else:
            scale = 1
        h *= scale
        
        # Evaluate current C5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        
        # Update time
        elapsed_time = time.time() - start_time
        remaining_time = budget_s - elapsed_time
        iteration += 1
        
        # Decay learning rate
        learning_rate *= 0.99
    
    return (best_h, best_c5, n_points)
