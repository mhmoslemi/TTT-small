        h_test[i], h_test[j] = new_h_i, new_h_j
        if compute_c5(h_test) < best_c5:
            best_c5 = compute_c5(h_test)
            best_h = h_test.copy()
            h = h_test.copy()
    
    return (best_h, best_c5, n_points)
