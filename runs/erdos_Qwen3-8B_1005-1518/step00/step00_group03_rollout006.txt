import numpy as np
from scipy.optimize import minimize
import time
import copy

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    elapsed = 0.0
    n_points = 70
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    def obj_func(h):
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = copy.deepcopy(h_initial)
    best_c5 = obj_func(best_h)
    
    result = minimize(obj_func, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
    
    final_h = result.x
    final_c5 = result.fun
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h, best_c5, n_points)
    return (best_h, final_c5, n_points)
