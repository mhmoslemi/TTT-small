import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gen_random_binary(n, rng):
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def gen_structured(n, rng):
    k = n // 2
    nblocks = rng.randint(2, max(3, n // 6 + 1))
    base = n // nblocks
    rem = n % nblocks
    h = np.zeros(n)
    pos = 0
    for bi in range(nblocks):
        sz = base + (1 if bi < rem else 0)
        if bi % 2 == 0:
            h[pos:pos + sz] = 1.0
        pos += sz
    ones = int(h.sum())
    if ones > k:
        oi = np.where(h == 1.0)[0]
        for i in rng.choice(oi, size=ones - k, replace=False):
            h[i] = 0.0
    elif ones < k:
        zi = np.where(h == 0.0)[0]
        for i in rng.choice(zi, size=k - ones, replace=False):
            h[i] = 1.0
    return h

def subgrad_search(h, target, deadline, step_init=0.02, lr_decay=0.999, rng=None):
    n = len(h)
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step = step_init
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        active = np.argmax(corr)
        shift = (n - 1) - active
        grad = np.zeros(n)
        for i in range(n):
            j = i + shift
            if 0 <= i < n and 0 <= j < n:
                grad[i] += (1.0 - h[j])
                grad[j] += (-h[i])

                best_c5 = c5
                best_h = h.copy()

            step *= lr_decay
    return best_h, best_c5
