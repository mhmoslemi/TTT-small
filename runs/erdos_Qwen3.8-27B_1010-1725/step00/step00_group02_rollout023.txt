            for _ in range(rng.integers(1, 4)):
                i = rng.integers(0, n)
                h_new[i] = np.clip(h_new[i] + 0.3 * (1 if h[i] > 0.5 else -1), 0, 1)

        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            T = max(T * 0.999, 0.0001)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            prob = np.exp(-(c5_new - c5_cur) / max(T, 1e-10))
            if rng.random() < prob:
                h = h_new
                c5_cur = c5_new
            T = max(T * 0.9997, 0.0001)

    # Phase 3: Coordinate descent local search
    deadline3 = t0 + budget_s * 0.92
    h = best_h.copy()
    c5_cur = best_c5
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    iters = 0
    while time.time() < deadline3 and iters < 20000:
        iters += 1
        i = rng.integers(0, n)
        improved = False
        for d in deltas:
            h_test = h.copy()
            h_test[i] = np.clip(h[i] + d, 0, 1)
            h_test = project_h(h_test, n)
            c5_test = compute_c5(h_test)
            if c5_test < c5_cur - 1e-12:
                h = h_test
                c5_cur = c5_test
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                improved = True
                break
        if not improved and iters % 5000 == 0:
            # Small random kick to escape
            i2 = rng.integers(0, n)
            h[i2] = np.clip(h[i2] + rng.normal(0, 0.03), 0, 1)
            h = project_h(h, n)
            c5_cur = compute_c5(h)

    # Phase 4: Higher resolution
    deadline4 = t0 + budget_s * 0.97
    for n_new in [60, 90]:
        if time.time() > deadline4:
            break
        x_old = (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h)
        x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_h(h_int, n_new)

        h_cur = h_int.copy()
        c5_cur = compute_c5(h_cur)
        T4 = 0.0012
        for _ in range(3000):
            if time.time() > deadline4:
                break
            h_new = h_cur.copy()
            i = rng.integers(0, n_new)
            h_new[i] += rng.normal(0, 0.04)
            h_new = project_h(h_new, n_new)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T4):
                h_cur = h_new
                c5_cur = c5_new
            T4 = max(T4 * 0.999, 0.0002)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
