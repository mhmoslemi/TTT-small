    return np.max(corr) * dx

def corr_gradient(h, lag, n):
    """Gradient of corr at given lag."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-lag] += 1.0
        grad[:n-lag] -= h[lag:]
        grad[lag:] -= h[:n-lag]
    else:
        k = -lag
        grad[k:] += 1.0
        grad[k:] -= h[:n-k]
        grad[:n-k] -= h[k:]
    return grad

def obj_val(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx)

    corr = np.correlate(h, 1.0 - h, mode="full")
