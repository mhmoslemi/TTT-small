        best_h, best_c5 = local_search(best_h.copy(), target, rng, 15000, deadline * 0.92)
        best_n = len(best_h)

    # Phase: Coordinate descent
    if time.time() < deadline * 0.97:
        best_h, best_c5 = coord_descent(best_h, best_n / 2.0, deadline * 0.97)
        best_n = len(best_h)

    # Phase: One more L-BFGS-B polish
    if time.time() < deadline * 0.99:
        target = best_n / 2.0
        obj_final = C5Obj(best_n, target, k=3, beta=500)
        res_f = minimize(obj_final, best_h.copy(), method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * best_n,
                         options={'maxiter': 3000, 'ftol': 1e-18, 'maxfun': 200000})
        h_f = fix_sum(np.clip(res_f.x, 0.0, 1.0), target)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final validation
    best_h = fix_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5 or not np.isfinite(best_c5):
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
