```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.clip(h, 0, 1)
    # Sort by value, set top n/2 to 1, rest to 0
    n_half = n // 2
    thresh = np.sort(h)[-n_half]
    h_bin = (h >= thresh).astype(float)
    # If we have more than n_half ones, reduce
    while h_bin.sum() > n_half:
        idx = np.random.choice(np.where(h_bin == 1)[0])
        h_bin[idx] = 0
    while h_bin.sum() < n_half:
        idx = np.random.choice(np.where(h_bin == 0)[0])
        h_bin[idx] = 1
    return h_bin

def local_search(h, n, num_iters, rng):
    """Random swap local search for binary h."""
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(num_iters):
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            # Update sets
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
```

```python
def project_feasible(h, n):
```

```python
    h_gauss = h_gauss / h_gauss.sum() * (n / 2.0)
    h_gauss = np.clip(h_gauss, 0, 1)
    h_gauss = make_binary(h_gauss, n)
    candidates.append(h_gauss)
    
    # Evaluate all candidates
    for cand in candidates:
        c5_c = compute_c5(cand)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = cand.copy()
    
    # Phase 1: SA on best binary candidate (45% budget)
    if time.time() - t0 < 0.45 * budget_s:
        h_bin = make_binary(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin > best_c5:
            h_bin = best_h.copy()
            c5_bin = best_c5
        b_h, b_c5 = sa_binary(h_bin, n, 0.42 * (budget_s - (time.time() - t0)), rng)
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b_h
    
    # Phase 2: Continuous refinement (25% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_cont = best_h.copy()
        c5_cont = compute_c5(h_cont)
        b_h, b_c5 = continuous_optimize(h_cont, n, 0.25 * remaining, rng)
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b_h
    
    # Phase 3: Higher resolution (25% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n2 = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0, 1)
        h_up = fix_sum(h_up, n2)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        
        remaining2 = budget_s - (time.time() - t0)
        if remaining2 > 40:
            b_h, b_c5 = continuous_optimize(h_up, n2, 0.7 * remaining2 - 5, rng)
            if b_c5 < best_c5:
                best_c5 = b_c5
                best_h = b_h
    
    # Final: ensure feasibility
    fn = len(best_h)
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = fix_sum(final_h, fn)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, fn
```
