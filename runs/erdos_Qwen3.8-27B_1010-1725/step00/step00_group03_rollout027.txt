    """Compute all correlation values r_k = sum valid h[n+k]*(1-h[n])"""

    corr = compute_corr(h)
    return np.max(corr) * dx

def smooth_obj(h, dx, tau):
    corr = compute_corr(h) * dx
    scaled = corr / tau
    mx = np.max(scaled)
    if mx > 500:
        return np.max(corr) * dx
    return tau * (mx + np.log(np.sum(np.exp(scaled - mx))))

def smooth_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr(h) * dx
    scaled = corr / tau
    mx = np.max(scaled)
    if mx > 500:

        k_max = np.argmax(corr)
        w = np.zeros(len(corr))
        w[k_max] = 1.0

        w = exp_s / np.sum(exp_s)

    n_total = len(corr)

        grad[n_lo + j:n_hi + j] += c * (1.0 - h[n_lo:n_hi])
        grad[n_lo:n_hi] += c * (-h[n_lo + j:n_hi + j])
    return grad

def project_box_sum(h, S):
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def optimize_phase(h_init, dx, S, tau0, tau1, lr0, lr1, max_iter, t_limit):
    n = len(h_init)
    h = h_init.copy()
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    lr = lr0
    for it in range(max_iter):
        if time.time() >= t_limit:
            break
        p = (it + 1) / max_iter
        tau = tau0 * (tau1 / tau0) ** p
        h = np.clip(h, 0.0, 1.0)
        grad = smooth_grad(h, dx, tau)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, S)
        c5_new = compute_c5(h_new, dx)
        if c5_new > best_c5 * 1.05 + 0.05:
            lr *= 0.5
            h_new = h - lr * grad
            h_new = project_box_sum(h_new, S)
            c5_new = compute_c5(h_new, dx)
        else:
            lr = min(lr * 1.05, lr1 * 5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h = h_new
    return best, best_c5

def local_search(h_init, dx, S, n_rounds, t_limit):
    h = h_init.copy()
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    n = len(h)
    rng = np.random.RandomState(0)
    delta = 0.05
    r = rng.permutation(n)
    for rnd in range(n_rounds):
        if time.time() >= t_limit:
            break
        for idx in r:
            if time.time() >= t_limit:
                break
            for sign in [1, -1]:
                h_try = h.copy()
                h_try[idx] += sign * delta
                h_try = project_box_sum(h_try, S)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-10:
                    h = h_try
                    best = h.copy()
                    best_c5 = c5_try
                    break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError):
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    S = n / 2.0

    best_h = np.clip(h0, 0.0, 1.0).copy()
    best_h = project_box_sum(best_h, S)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: coarse optimization
    tl1 = t0 + budget_s * 0.35
    best_h, best_c5 = optimize_phase(best_h, dx, S, 0.08, 0.008, 1.5, 0.05, 4000, tl1)

    # Phase 2: fine optimization
    tl2 = t0 + budget_s * 0.60
    best_h, best_c5 = optimize_phase(best_h, dx, S, 0.008, 0.001, 0.5, 0.005, 4000, tl2)

    # Phase 3: local coordinate search
    tl3 = t0 + budget_s * 0.75
    best_h, best_c5 = local_search(best_h, dx, S, 5, tl3)

    # Phase 4: random restarts
    n_rest = 6
    t_rem = t0 + budget_s * 0.90 - time.time()
    if t_rem > 10:
        per_t = t_rem / n_rest
        for r in range(n_rest):
            if time.time() > t0 + budget_s * 0.90:
                break
            h_r = best_h + rng.randn(n) * 0.15
            h_r = project_box_sum(h_r, S)
            t_lim = time.time() + per_t
            h_try, c5_try = optimize_phase(h_r, dx, S, 0.04, 0.003, 0.8, 0.01, 1500, t_lim)
            h_try, c5_try = local_search(h_try, dx, S, 2, min(t_lim, time.time() + 3))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 5: higher resolution attempt
    tl5 = t0 + budget_s * 0.95
    if time.time() < tl5:
        for n_h in [140, 210]:
            if time.time() > tl5 - 5:
                break
            dx_h = 2.0 / n_h
            S_h = n_h / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_h, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_box_sum(np.clip(h_interp, 0, 1), S_h)
            t_lim = tl5 - 2
            h_try, c5_try = optimize_phase(h_interp, dx_h, S_h, 0.05, 0.003, 1.0, 0.01, 2000, t_lim)
            h_try, c5_try = local_search(h_try, dx_h, S_h, 2, min(t_lim, time.time() + 4))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Final polish
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, n_final
