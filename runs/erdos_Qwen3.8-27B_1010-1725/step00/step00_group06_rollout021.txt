import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max(vals, alpha):
    m = np.max(vals)
    a = min(alpha, 300.0 / max(m, 0.005))
    shifted = a * (vals - m)
    return m + np.log(np.sum(np.exp(shifted))) / a

def fix_sum(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lbfgs(x, n, alpha, pen):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = soft_max(corr * dx, alpha)
    dev = np.sum(h) - n / 2.0
    return sm + pen * dev * dev

def slsqp_obj(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return soft_max(corr * dx, 400)

def slsqp_con(x, n):
    return np.sum(x) - n / 2.0

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(95, 0.5)

    n = len(h_best)
    h_best = fix_sum(h_best, n)
    c5_best = compute_c5(h_best)
    best_h = h_best.copy()

    alphas = [15, 30, 60, 120, 250, 400]

    # Phase 1: Optimize at original resolution with L-BFGS-B
    for alpha in alphas:
        if time.time() > deadline - 60:
            break
        pen = 80.0
        try:
            res = minimize(obj_lbfgs, best_h, args=(n, alpha, pen),
                           method='L-BFGS-B', bounds=[(0, 1)] * n,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
            h_t = fix_sum(res.x, n)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best:
                c5_best = c5_t
                best_h = h_t.copy()
        except Exception:
            pass

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 80:
        try:

            if res.x is not None:

                    c5_best = c5_s
                    best_h = h_s

            pass
