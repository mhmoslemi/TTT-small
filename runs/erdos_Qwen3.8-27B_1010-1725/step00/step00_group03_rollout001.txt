import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, T=40.0):
    m = float(np.max(x))
    if T < 1e-6:
        return m
    return m + np.log(np.mean(np.exp(T * (x - m)))) / T

def project_to_sum(h, n):
    tgt = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
