The objective is max_k (A_k * t + B_k) * dx, which is a piecewise-linear convex function of t. Its minimum is at the intersection of the two "active" lines (the two with the maximum at the optimum), or at a boundary.

I can find this by:
1. Evaluating all c_k at t=0 and t=1 (bounded by feasibility)
2. The minimum of max is where the upper envelope is minimized
3. This can be found by checking all pairs of k

For O(n) lines, finding the minimum of the max takes O(n²) or O(n log n). For n=70, this is fine.

def coordinate_minimize(h, i, j, dx, ts):
    """Minimize max_k corr[k]*dx by varying h[i]=t, h[j]=h[j]_0-(t-h[i]_0)"""
    h_i0, h_j0 = h[i], h[j]
    
    # Range of t: 0 ≤ t ≤ 1 and 0 ≤ h_j0 - (t - h_i0) ≤ 1
    # → h_j0 + h_i0 - 1 ≤ t ≤ h_j0 + h_i0
    t_min = max(0, h_j0 + h_i0 - 1)
    t_max = min(1, h_j0 + h_i0)
    
    if t_min >= t_max:
        return t_min  # Degenerate
    
    # At t, h[i]=t, h[j]=h_j0-(t-h_i0)
    # Compute corr for this h: it's expensive to do for many t values
    # Instead, compute coefficients A_k, B_k for each k
    
    # corr[k] = sum_n h[n+k]*(1-h[n]) (for valid indices)
    # When h[i] changes from h_i0 to t: contribution to terms involving i
    # When h[j] changes from h_j0 to h_j0-(t-h_i0): contribution to terms involving j
    
    # For efficiency, compute base_corr (with h[i]=h_i0, h[j]=h_j0)
    # Then for each k, compute the change due to h[i] and h[j]
    
    # Actually, for n=70, let's just evaluate at a few t values and interpolate
    # Or better: compute the piecewise linear function exactly
    
    # For each k (0 to 2n-2 in the full correlation array):
    # corr_full[k] = sum of products
    # The terms involving h[i]: 
    #   - if n=i in the sum: h[n+k]*(1-h[i]) = h[i+k]*(1-h[i]) → contributes (1-h[i])*h[i+k]
    #   - if n+k=i: h[i]*(1-h[n]) → contributes h[i]*(1-h[n])
    # Similarly for h[j]
    
    # This is getting complex. Let me just do a golden section or ternary search
    # over t in [t_min, t_max]
    
    # Since the function is convex (max of linear), ternary search works
    
    def f(t):
        h_new = h.copy()
        h_new[i] = t
        h_new[j] = h_j0 - (t - h_i0)
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            return 1e10
        return compute_c5(h_new, dx)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = t_min, t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    
    for _ in range(50):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = f(d)
    
    t_opt = (a + b) / 2
    h[i] = t_opt
    h[j] = h_j0 - (t_opt - h_i0)
    return t_opt

Actually, I realize I should be more careful about the golden section search. The function f(t) = max_k corr[k(t)] * dx is convex in t (it's the max of linear functions of t). But it's NOT necessarily strictly convex, so there might be a flat region. Golden section still works for convex functions.

But wait: is it truly convex in t? Let me verify.
corr[k](t) = (terms not involving t) + (terms involving h[i]=t) + (terms involving h[j]=hj0-(t-hi0))

The terms involving h[i]=t:
- From n=i: h[i+k]*(1-t) = h[i+k] - t*h[i+k] (if i+k < n)
- From n+k=i, i.e., n=i-k: t*(1-h[i-k]) (if 0 ≤ i-k < n)

The terms involving h[j]=hj0-(t-hi0) = (hj0-hi0)+t... wait, that's not right.
h[j] = hj0 - (t - hi0) = hj0 - t + hi0

- From n=j: h[j+k]*(1-h[j]) = h[j+k]*(1-hj0+t-hi0)
- From n+k=j: h[j]*(1-h[n]) = (hj0-t+hi0)*(1-h[j-k])

So corr[k](t) is a quadratic in t! (Because of the h[j+k]*(1-h[j]) term which has h[j+k] * (constant + t).)

Wait no: h[j+k] is fixed (we're only changing h[i] and h[j]), and (1-h[j]) = 1-hj0+t-hi0 is linear in t. So the product h[j+k]*(1-hj0+t-hi0) is linear in t. Similarly, (hj0-t+hi0)*(1-h[j-k]) is linear in t.

And for h[i]=t:
- h[i+k]*(1-t) = h[i+k] - t*h[i+k]: linear in t
- t*(1-h[i-k]): linear in t

So corr[k](t) IS linear in t! (All terms are products of a fixed value with a linear function of t.)

Great, so the max of linear functions is indeed convex, and golden section works.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, ts):
    h = h.copy()
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 0) & (h < 1)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += d / nf
    return h

def smooth_max_val(corr_vec, T):
    m = np.max(corr_vec)
    shifted = np.clip((corr_vec - m) / T, -300, 300)
    return m + T * np.log(np.sum(np.exp(shifted)))

def coord_descend(h, dx, ts, n, max_passes=200, golden_iters=45):
    gr = (np.sqrt(5) - 1) / 2
    for p in range(max_passes):
        max_change = 0.0
        order = np.arange(n)
        for i in order:
            j = (i + 1) % n
            hi0, hj0 = h[i], h[j]
            t_lo = max(0.0, hj0 + hi0 - 1.0)
            t_hi = min(1.0, hj0 + hi0)
            if t_hi - t_lo < 1e-13:
                continue
            a, b = t_lo, t_hi
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h[i] = c
            h[j] = hj0 - (c - hi0)
            fc = compute_c5(h, dx)
            h[i] = d
            h[j] = hj0 - (d - hi0)
            fd = compute_c5(h, dx)
            for _ in range(golden_iters):
                if fc < fd:
                    b, d, fd = d, c, fc
                    c = b - gr * (b - a)
                    h[i] = c
                    h[j] = hj0 - (c - hi0)
                    fc = compute_c5(h, dx)
                else:
                    a, c, fc = c, d, fc
                    d = a + gr * (b - a)
                    h[i] = d
                    h[j] = hj0 - (d - hi0)
                    fd = compute_c5(h, dx)
            t_opt = 0.5 * (a + b)
            h[i] = t_opt
            h[j] = hj0 - (t_opt - hi0)
            max_change = max(max_change, abs(t_opt - hi0))
        h = project_sum(h, ts)
        if max_change < 1e-12:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.ones(70) * 0.5
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0

    h = project_sum(h0, ts)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with smooth max
    for T in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if rem() < 40:
            break
        def obj(x):
            xc = np.clip(x, 0, 1)
            c = np.correlate(xc, 1 - xc, mode="full") * dx
            return smooth_max_val(c, T)
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0, 1)] * n,
                           options={'maxiter': 150, 'ftol': 1e-15, 'gtol': 1e-9})
            h2 = project_sum(res.x, ts)
            c52 = compute_c5(h2, dx)
            if c52 < best_c5:
                best_c5, best_h = c52, h2.copy()
                h = h2
        except Exception:
            pass

    # Phase 2: Coordinate descent from best
    if rem() > 30:
        h = coord_descend(best_h.copy(), dx, ts, n, max_passes=300)
        h = project_sum(h, ts)
        c5_cd = compute_c5(h, dx)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h.copy()

    # Phase 3: SLSQP with equality constraint
    if rem() > 60:
        T_sl = 0.002
        def obj_sl(x):
            xc = np.clip(x, 0, 1)
            c = np.correlate(xc, 1 - xc, mode="full") * dx
            return smooth_max_val(c, T_sl)
        try:
            res = minimize(obj_sl, best_h, method='SLSQP',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: x.sum() - ts,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': 300, 'ftol': 1e-12})
            h2 = project_sum(res.x, ts)
            c52 = compute_c5(h2, dx)
            if c52 < best_c5:
                best_c5, best_h = c52, h2.copy()
        except Exception:
            pass

    # Phase 4: Multi-restart coordinate descent
    n_restarts = 0
    while rem() > 30 and n_restarts < 60:
        scale = 0.08 * (1.0 + 0.5 * rng.random())
        h_try = best_h + rng.standard_normal(n) * scale
        h_try = project_sum(h_try, ts)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 + 0.01:
            h_try = coord_descend(h_try, dx, ts, n, max_passes=150)
            h_try = project_sum(h_try, ts)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
        n_restarts += 1

    # Phase 5: Fine random search around best
    if rem() > 15:
        step = 0.02
        stall = 0
        h_cur = best_h.copy()
        while rem() > 10 and stall < 2000:
            d = rng.standard_normal(n)
            d -= d.mean()
            h_try = project_sum(h_cur + step * d, ts)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
                h_cur = h_try
                step = min(step * 1.03, 0.15)
                stall = 0
            elif c5_try < compute_c5(h_cur, dx):
                h_cur = h_try
                stall += 1
            else:
                step *= 0.97
                stall += 1
                if stall > 500:
                    h_cur = project_sum(best_h + rng.standard_normal(n) * 0.03, ts)
                    step = 0.03
                    stall = 0

    # Phase 6: Final coordinate descent polish
    if rem() > 5:
        h_final = coord_descend(best_h.copy(), dx, ts, n, max_passes=200)
        h_final = project_sum(h_final, ts)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final

    # Ensure exact constraint
    best_h = project_sum(best_h, ts)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
