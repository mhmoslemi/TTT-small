def repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        if s > 0:
            h = np.clip(h * target / s, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            space = 1.0 - h
            i = int(np.argmax(space))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + diff)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h[int(n // 2)] += diff
        h = np.clip(h, 0.0, 1.0)

        # spread remaining

            mask = (h < 1.0) if diff2 > 0 else (h > 0.0)

import numpy as np
import time

def repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > 0:
            h = np.clip(h * target / s, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            space = 1.0 - h
            i = int(np.argmax(space))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + diff)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mid = n // 2
        h[mid] = np.clip(h[mid] + diff, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-14:
            mask = (h < 1.0) if diff2 > 0 else (h > 0.0)
            cnt = mask.sum()
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff2 / cnt, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def overlap_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def upsample_h(h, n_to):
    n = len(h)
    return np.interp(np.arange(n_to) * n / n_to, np.arange(n), h)

    best_c = eval_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    T = 0.015
    t_start = time.time()
    it = 0
    bszs = (2, 3, 4, 5, 6)
    while time.time() - t_start < budget:
        it += 1
        r = rng.random()
        h_new = cur_h.copy()
        if r < 0.35:
            i = int(rng.integers(n))
            h_new[i] += rng.normal(0, lr)
        elif r < 0.60:
            b = bszs[int(rng.integers(len(bszs)))]
            start = int(rng.integers(0, n - b))
            h_new[start:start + b] += rng.normal(0, lr, size=b)
        elif r < 0.80:
            b = bszs[int(rng.integers(len(bszs)))]
            start = int(rng.integers(0, n - b))
            h_new[start:start + b] = 1.0 - h_new[start:start + b]
        elif r < 0.90:
            b = 2 + int(rng.integers(4))
            s1 = int(rng.integers(0, n - b))
            s2 = int(rng.integers(0, n - b))
            if s1 != s2:
                tmp = h_new[s1:s1 + b].copy()
                h_new[s1:s1 + b] = h_new[s2:s2 + b]
                h_new[s2:s2 + b] = tmp
        else:
            # Directed: target worst shift
            ov = overlap_vec(cur_h)
            k_star = int(np.argmax(ov))
            shift = k_star - (n - 1)
            n_pairs = n - abs(shift)

def make_start(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        # Alternating blocks
        h[::2] = 0.5
        h[1::2] = 0.5
    elif mode == 1:
        # Random binary
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif mode == 2:
        # Cosine banded
        x = np.linspace(0, 2 * np.pi, n)
        h = 0.5 + 0.5 * np.cos(3 * x)
    elif mode == 3:
        # Multi-scale
        x = np.linspace(0, 2 * np.pi, n)
        h = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(5 * x)
    else:
        h = np.full(n, 0.5)
        h[:n // 2] = 0.0
        h[n // 2:] = 1.0
    h = repair(h, n)
    return h
