    idx = np.clip(idx, 0, n_old - 1)
    h_new = h_old[idx].astype(float)
    return project_feasible(h_new, n_new / 2.0)


def local_refine(h, target, rng, n_trials):
    h = h.copy()
    n = len(h)
    best_c = compute_c5(h)
    for _ in range(n_trials):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        orig_i, orig_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        h = project_feasible(h, target)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            h[i], h[j] = orig_i, orig_j
            h = project_feasible(h, target)
    return h, best_c


def optimize_run(h0, target, max_time, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    momentum = np.zeros(n)
    lr = 0.05
    beta = 20.0
    it = 0
    t0 = time.time()
    while (time.time() - t0) < max_time:
        it += 1
        sm, grad = soft_max_grad(h, beta)
        momentum = 0.85 * momentum - lr * grad
        h_new = h + momentum
        h = project_feasible(h_new, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-10:
            best_c5 = c5_now
            best_h = h.copy()
        if it % 80 == 0:
            lr *= 0.94
            beta = min(beta + 2.0, 120.0)
        if it % 250 == 0:
            h, bc = local_refine(h, target, rng, max(20, n))
            if bc < best_c5:
                best_c5 = bc
                best_h = h.copy()
        if it % 600 == 0 and lr < 0.005:
            h = best_h + rng.normal(0, 0.15, n)
            h = project_feasible(h, target)
            momentum = np.zeros(n)
            lr = 0.04
            beta = 20.0
    return best_h, best_c5


def try_binary_snap(h, target, rng):
    """Try to snap to near-binary values."""
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    if n % 2 == 0:
        h_bin = (h >= 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        c = compute_c5(h_bin)
        if c < best_c:
            best_c = c
            best_h = h_bin.copy()
    else:
        for pos in range(n):
            h_try = (h >= 0.5).astype(float)
            h_try[pos] = 0.0
            h_try = project_feasible(h_try, target)
            # Force one element to 0.5
            h_try2 = np.where(np.abs(h_try - 0.5) < 0.01, 0.5, np.where(h_try > 0.5, 1.0, 0.0))
            h_try2 = project_feasible(h_try2, target)
            if abs(h_try2.sum() - target) < 1e-6:
                c = compute_c5(h_try2)
                if c < best_c:
                    best_c = c
                    best_h = h_try2.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Get initial solution
    init = globals().get('initial_h_values', None)
    if init is not None:
        h_init = np.array(init, dtype=float)
    else:
        h_init = np.ones(50) * 0.5

    n_init = len(h_init)
    target_init = n_init / 2.0
    h_init = project_feasible(h_init, target_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization at initial resolution
    t_alloc1 = min(time_left() * 0.45, 350.0)
    if t_alloc1 > 10:
        h1, c5_1 = optimize_run(h_init, target_init, t_alloc1, rng)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Try binary snap
    if time_left() > 20:
        hb, cb = try_binary_snap(best_h, len(best_h) / 2.0, rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb

    # Phase 2: Try different resolutions
    n_candidates = [50, 60, 70, 80]
    n_candidates = [x for x in n_candidates if x != n_init]
    for n_try in n_candidates:
        if time_left() < 40:
            break
        t_sub = min(time_left() * 0.18, 150.0)
        if t_sub < 10:
            break
        h_try = resample_to(best_h, n_try)
        target_try = n_try / 2.0
        h2, c5_2 = optimize_run(h_try, target_try, t_sub, rng)
        # Also try binary snap at this resolution
        h2b, c5_2b = try_binary_snap(h2, target_try, rng)
        if c5_2b < c5_2:
            h2, c5_2 = h2b, c5_2b
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Final refinement at best resolution
    if time_left() > 15:
        n_best = len(best_h)
        target_best = n_best / 2.0
        h3, c5_3 = optimize_run(best_h, target_best, min(time_left() * 0.7, 200.0), rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
        hb, cb = try_binary_snap(best_h, target_best, rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb

    # Final projection and validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-5:
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
