import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_grad(h):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    k_idx = np.argmax(corr)
    k = k_idx - (n - 1)
    
    c5_val = corr[k_idx]
    
    grad = np.zeros(n)
    if k >= 0:
        # i from k to n-1: (1-h[i-k])
        # i from 0 to n-1-k: -h[i+k]
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        sp = -k
        # i from 0 to n-1-sp: (1-h[i+k]) = (1-h[i-sp])
        # i from sp to n-1: -h[i+k] = -h[i-sp]
        grad[:n-sp] += (1.0 - h[sp:])
        grad[sp:] -= h[:n-sp]
    
    return c5_val, grad

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    val, grad = compute_c5_grad(h)
    sum_dev = h.sum() - target_sum
    val += penalty_weight * sum_dev**2
    grad += 2.0 * penalty_weight * sum_dev * np.ones(n)
    # Also add bounds penalty for smoothness
    return val, grad

    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    def fun(x):
        val, _ = compute_c5_grad(x)
        diff = x.sum() - target_sum
        return val + 1000.0 * diff**2
    
    def jac(x):
        val, grad = compute_c5_grad(x)
        diff = x.sum() - target_sum
        grad = grad + 2000.0 * diff * np.ones(n)
        return grad
    
    bounds = [(0, 1)] * n

        if n_new > 600 or time.time() - t0 > DEAD - 20:
            break
        h_u = project(upsample(best_h, n_new), n_new / 2.0)
        h3 = opt_lbfgs(h_u, n_new, 400)
        c3 = c5_val(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

        # SLSQP for smaller upsamples
        if n_new <= 120 and time.time() - t0 < DEAD - 10:
            h4 = opt_slsqp(best_h, n_new, 150)
            c4 = c5_val(h4)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

    # Phase 4: Perturbation local search at best resolution
    n_best = len(best_h)
    if time.time() - t0 < DEAD - 30:
        for attempt in range(6):
            if time.time() - t0 > DEAD - 30:
                break
            scale = 0.04 / (attempt + 1)
            best_h, best_c5 = local_search(best_h, n_best, rng,
                                           n_trials=4, scale=scale)

    # Phase 5: Final L-BFGS polish
    if time.time() - t0 < DEAD - 5:
        h_pol = opt_lbfgs(best_h, n_best, 300, lam=10000.0)
        c_pol = c5_val(h_pol)
        if c_pol < best_c5:
            best_c5, best_h = c_pol, h_pol.copy()

    # Ensure constraint is satisfied
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = c5_val(best_h)

    try:
        evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
