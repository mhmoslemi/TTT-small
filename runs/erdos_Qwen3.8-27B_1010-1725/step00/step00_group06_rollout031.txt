    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    fn = (max_corr + np.log(np.sum(exp_corr))) / beta  # = logsumexp(β*corr)/β

def fix_feasibility(h, target_sum):
