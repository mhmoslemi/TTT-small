def project_to_constraint(y, n):

    target = n / 2.0

    lo, hi = y.min() - 1, y.max() + 1

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def sg_step(h, n):
    """One subgradient step at worst lag"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    s = k_star - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        if 0 <= i - s < n:
            g[i] += (1.0 - h[i - s])
        if 0 <= i + s < n:
            g[i] -= h[i + s]
    g *= dx
    return g, c[k_star]

    target = n / 2.0
    best_h = h.copy()
    best_c5 = c5_of(h)
    alpha = 1.0 / np.sqrt(n)
    for it in range(n_iter):
        g, _ = sg_step(h, n)
        h_new = h - alpha * g
        h_new = project_sum(h_new, target)
        c5 = c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        alpha = alpha * 0.999 + 0.01 / (it + 1)

    return np.interp(np.linspace(0.0, n0 - 1, n), np.arange(n0, dtype=float), h0)

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_obj(x, eps):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return eps * logsumexp(c / eps)

def smooth_grad(x, eps):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    lc = logsumexp(c / eps)
    P = np.exp(c / eps - lc)
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        for k in range(2 * n - 1):
            p = P[k]
            if p < 1e-14:
                continue
            s = k - (n - 1)
            if 0 <= i - s < n:
                gi += p * (1.0 - x[i - s])
            if 0 <= i + s < n:
                gi -= p * x[i + s]
        grad[i] = dx * gi
    return grad

def constr_eq(x):
    return x.sum() - len(x) * 0.5

def subgrad_method(h, n, n_iter=1500):
    target = n * 0.5
    best_h = h.copy()
    best_c5 = c5_of(h)
    alpha = 0.5
    for it in range(n_iter):
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i - s < n:
                g[i] += (1.0 - h[i - s])
            if 0 <= i + s < n:
                g[i] -= h[i + s]
        g *= dx
        h_new = h - alpha * g
        h_new = project_sum(h_new, target)
        c5_new = c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        alpha = 0.3 / (0.5 + 0.001 * (it + 1))
    return best_h, best_c5

def slsqp_opt(h, n, max_it=250):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_eq}]
    best_h = h.copy()
    best_c5 = c5_of(h)
    eps_list = [0.04, 0.015, 0.006, 0.002]
    for eps in eps_list:
        res = minimize(smooth_obj, h, jac=smooth_grad, args=(eps,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_it, 'ftol': 1e-12})
        h = project_sum(np.clip(res.x, 0.0, 1.0), target)
        c = c5_of(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h_init = np.asarray(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = c5_of(h_init)

    grid_sizes = sorted(set([n_init, 2 * n_init, 4 * n_init]), key=lambda x: x)
    grid_sizes = [x for x in grid_sizes if x <= 1000]

    for n in grid_sizes:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.65:
            break

        starts = []
        if n == n_init:
            starts.append(h_init.copy())
        else:
            starts.append(project_sum(upsample(h_init, n), n * 0.5))
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        starts.append(project_sum(blk.copy(), n * 0.5))
        rnd = np.random.rand(n)
        starts.append(project_sum(rnd.copy(), n * 0.5))

        sg_iters = 800 if n <= 100 else 400
        slsqp_iters = 250 if n <= 100 else 150

        for si, h0 in enumerate(starts):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.85:
                break
            time_left = budget_s * 0.85 - elapsed
            if time_left < 10:
                break
            h_sg, c_sg = subgrad_method(h0.copy(), n, n_iter=sg_iters)
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.85:
                if c_sg < best_c5:
                    best_c5 = c_sg
                    best_h = h_sg.copy()
                continue
            h_opt, c_opt = slsqp_opt(h_sg, n, max_it=slsqp_iters)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    elapsed = time.time() - t0
    if elapsed < budget_s * 0.9:
        for _ in range(3):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.92:
                break
            n = best_h.shape[0]
            pert = best_h + 0.05 * np.random.randn(n)
            h_r = project_sum(pert, n * 0.5)
            h_r, c_r = slsqp_opt(h_r, n, max_it=200)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()

    final_c5 = c5_of(best_h)
    return (best_h, final_c5, len(best_h))
