def full_correlation(h):

    dx = 2.0/n
    return np.max(full_correlation(h)) * dx

def gradient_at_max(h):
    """Compute gradient of the max correlation w.r.t. h"""
    n = len(h)
    corrs = full_correlation(h)
    k_star = np.argmax(corrs)  # index in full correlation, 0 to 2n-2
    
    grad = np.zeros(n)
    shift = k_star - (n-1)  # actual shift, -(n-1) to n-1

            # d/dh[j] of h[j+shift]*(1-h[j]) = -h[j+shift]

            grad[j-s] += (1 - h[j])
            # d/dh[j] of h[j-s]*(1-h[j]) = -h[j-s]

def full_correlation(h):

    """Gradient of correlation at given shift w.r.t. h"""
    n = len(h)
    s = abs(shift)
    if s == 0:

    grad = np.zeros(n)
    if shift > 0:
        grad[:n-s] -= h[s:]
        grad[s:] += 1.0 - h[:n-s]
    else:
        grad[:n-s] += 1.0 - h[s:]
        grad[s:] -= h[:n-s]
    return grad

    """Project h to {x : 0<=x_i<=1, sum(x)=target}"""

            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - (lo+hi)/2, 0, 1)

import numpy as np
import time

def full_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(full_correlation(h)) * dx)

def grad_at_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if shift > 0:
        grad[:n - s] -= h[s:]
        grad[s:] += 1.0 - h[:n - s]
    else:
        grad[:n - s] += 1.0 - h[s:]
        grad[s:] -= h[:n - s]
    return grad

def project_simplex(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def optimize_pgd(h0, target, n_iters, beta=50.0, lr_init=0.008, lr_decay=0.9995):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    lr = lr_init
    for it in range(n_iters):
        corrs = full_correlation(h)
        mc = np.max(corrs)
        exps = np.exp(beta * (corrs - mc))
        wsum = exps.sum()
        if wsum < 1e-30:
            wsum = 1.0
        weights = exps / wsum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-6:
                continue
            shift = k - (n - 1)
            grad += w * grad_at_shift(h, shift)
        h = h - lr * grad
        h = project_simplex(h, target)
        c5 = c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= lr_decay
    return best_h, best_c5

def thue_morse(n, target):
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    if t.sum() > target:
        t = 1.0 - t
    if t.sum() != target:
        t = project_simplex(t, target)
    return t

def binary_local_search(h, n_iters, target):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    on = np.where(h_bin == 1)[0]
    off = np.where(h_bin == 0)[0]
    diff = len(on) - int(target)
    if diff > 0:
        for idx in on[-diff:]:
            h_bin[idx] = 0
    elif diff < 0:
        for idx in off[:(-diff)]:
            h_bin[idx] = 1
    best_c5 = c5_value(h_bin)
    best = h_bin.copy()
    rng = np.random.default_rng(12345)
    for it in range(n_iters):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5 = c5_value(h_bin)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best, best_c5

def interpolate_h(h, new_n):
    n = len(h)
    x_old = np.arange(n) * 2.0 / n
    x_new = np.arange(new_n) * 2.0 / new_n
    return np.interp(x_new, x_old, np.append(h, h[-1]), right=h[-1], left=h[0])

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = thue_morse(100, 50.0)

    n = len(h_init)
    target = n / 2.0
    h_init = project_simplex(h_init, target)
    best_h = h_init.copy()
    best_c5 = c5_value(h_init)

    # Phase 1: PGD from initial with various params
    lrs = [0.008, 0.004, 0.002, 0.001]
    for lr in lrs:
        if time.time() - t0 > budget_s * 0.25:
            break
        h, c5 = optimize_pgd(h_init, target, 4000, beta=50, lr_init=lr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        h_init = h

    # Phase 2: Softmax beta variations
    for beta in [20, 100, 200]:
        if time.time() - t0 > budget_s * 0.45:
            break
        h, c5 = optimize_pgd(h_init, target, 3000, beta=beta, lr_init=0.003)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        h_init = h

    # Phase 3: Random restarts
    for trial in range(40):
        if time.time() - t0 > budget_s * 0.70:
            break
        h_rand = project_simplex(np.random.rand(n), target)
        h, c5 = optimize_pgd(h_rand, target, 2500, beta=50, lr_init=0.0
06)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 4: Thue-Morse and structured starts
    for start_n in [n, 100, 128]:
        if time.time() - t0 > budget_s * 0.78:
            break
        ht = thue_morse(start_n, start_n / 2.0)
        if start_n != n:
            ht = interpolate_h(ht, n)
            ht = project_simplex(ht, target)
        h, c5 = optimize_pgd(ht, target, 3000, beta=50, lr_init=0.005)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 5: Binary local search at even n
    n_even = n if n % 2 == 0 else n + 1
    target_e = n_even / 2.0
    h_even = interpolate_h(best_h, n_even)
    h_even = project_simplex(h_even, target_e)
    hb, cb = binary_local_search(h_even, 80000, target_e)
    if cb < best_c5 and n_even <= 1000:
        best_c5 = cb
        best_h = hb
        n = n_even
        target = target_e

    # Phase 6: Fine-tune PGD on best
    if time.time() - t0 < budget_s * 0.88:
        n_f = len(best_h)
        t_f = n_f / 2.0
        for lr in [0.001, 0.0005, 0.0002]:
            if time.time() - t0 > budget_s * 0.95:
                break
            h, c5 = optimize_pgd(best_h.copy(), t_f, 5000, beta=300, lr_init=lr, lr_decay=0.9998)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Final check
    best_c5 = c5_value(best_h)
    return best_h, best_c5, len(best_h)
