import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h):
    """Convert to nearest binary solution with correct sum"""
    n = len(h)
    target = n // 2
    # Sort by value, assign top n/2 to 1
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def interp_to(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, [h[0]]])  # wrap for interpolation
    h_new = np.interp(x_new, x_old, h_ext)
    # Fix sum
    diff = n_new / 2.0 - np.sum(h_new)
    h_new += diff / n_new
    h_new = np.clip(h_new, 0, 1)
    # Exact sum fix
    diff = n_new / 2.0 - np.sum(h_new)
    if abs(diff) > 1e-12 and n_new > 0:
        h_new[0] += diff
        h_new[0] = np.clip(h_new[0], 0, 1)
        diff = n_new / 2.0 - np.sum(h_new)
        # Spread remaining
        for i in range(1, n_new):
            if abs(diff) < 1e-12:
                break
            room = min(abs(diff), (1.0 - h_new[i]) if diff > 0 else h_new[i])
            h_new[i] += np.sign(diff) * room
            diff -= np.sign(diff) * room
    return h_new

def binary_sa(h_bin, n, max_iters=20000, seed=42, time_limit=None, start_time=None):

    rng = np.random.RandomState(seed)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_bin.copy()
    cur_c5 = best_c5
    temp = 0.01
    cooling = 0.9995
    
    ones_list = list(np.where(cur_h >= 0.5)[0])
    zeros_list = list(np.where(cur_h < 0.5)[0])
    
    for it in range(max_iters):

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:

        ones_list.remove(i); ones_list.append(j)
        zeros_list.remove(j); zeros_list.append(i)

            if new_c5 < best_c5:

                best_c5 = new_c5

            ones_list.remove(j); ones_list.append(i)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        sign = 1.0 if diff > 0 else -1.0
        for i in idx:
            if abs(diff) < 1e-12:
                break
            room = (1.0 - h[i]) if sign > 0 else h[i]
            adj = min(abs(diff), room) * sign
            h[i] += adj
            diff -= adj
    return h

def interp_to(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, [h[0]]])
    h_new = np.interp(x_new, x_old, h_ext)
    return fix_sum(h_new, n_new)

def make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def binary_sa(h_bin, n, max_iters, seed, time_limit, start_time):

    ones_list = list(np.where(cur_h >= 0.5)[0])
    zeros_list = list(np.where(cur_h < 0.5)[0])
    for it in range(max_iters):
        if time.time() - start_time > time_limit:

        i = ones_list[rng.randint(len(ones_list))]

        cur_h[i] = 0.0

        ones_list.append(j)
        zeros_list.remove(j)
        zeros_list.append(i)
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-15)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = new_c5
        else:

            cur_h[j] = 0.0
            ones_list.remove(j)
            ones_list.append(i)
            zeros_list.remove(i)
            zeros_list.append(j)
        temp *= cooling
    return best_h, best_c5
