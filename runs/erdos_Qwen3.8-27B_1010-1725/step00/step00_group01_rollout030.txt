def compute_c5(h):
    """Compute C5 value for a given h array."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")

def repair_sum(h, target):
    """Repair h to satisfy sum(h) = target while keeping [0,1]."""
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    n = len(h)
    # Redistribute diff across all elements
    step = diff / n
    h_new = h + step
    # Clip and iterate
    for _ in range(100):
        h_new = np.clip(h_new, 0, 1)
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-12:
            break
        # Adjust only elements not at bounds
        free = (h_new > 0) & (h_new < 1)
        if not np.any(free):
            break
        h_new[free] += diff / np.sum(free)
    return h_new

    return temperature * (np.max(x) + np.log(np.sum(np.exp(x))))
