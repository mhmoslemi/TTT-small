import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = c * dx
    m = np.max(vals)
    shifted = (vals - m) * T
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    # Objective value
    obj = m + np.log(np.sum(exps)) / T
    # Gradient: weighted sum of gradients of each vals[k]
    # vals[k] = c_full[k] * dx
    # Need dc_full[k]/dh[i] for the full correlate
    # np.correlate(h, 1-h, mode='full'):
    # output length is 2n-1
    # Let's compute gradient directly
    # 
    # For the full correlate of a and b (both length n):
    # output[k] = sum_j a[j] * b[j - k + n - 1] for valid j
    # Equivalently, if we define lag l = k - (n-1):
    # output[n-1+l] = sum_j a[j+l] * b[j] for valid j
    # (where valid means 0<=j<=n-1 and 0<=j+l<=n-1)
    #
    # Here a=h, b=1-h
    # So c_full[n-1+l] = sum_j h[j+l] * (1-h[j])
    #
    # dc_full[n-1+l]/dh[i]:
    # h[i] appears as h[j+l] when i=j+l, i.e., j=i-l (contributes (1-h[j]) = (1-h[i-l]))
    # h[i] appears as h[j] in (1-h[j]) when j=i (contributes -h[i+l])
    # So dc/dh[i] = [0<=i-l<=n-1] * (1-h[i-l]) + [0<=i+l<=n-1] * (-h[i+l])
    
    # For the max output index n-1 (lag 0):
    # dc[0]/dh[i] = (1-h[i]) + (-h[i]) = 1 - 2*h[i]
    
    # Let's compute the gradient for all active lags using the weights
    grad = np.zeros(n)
    for l_idx in range(2*n-1):
        if weights[l_idx] < 1e-12:
            continue
        l = l_idx - (n - 1)  # actual lag
        w = weights[l_idx] * dx  # weight including dx
        
        # dc/dh[i] for this lag:
        # Term 1: h[i] in h[j+l] position → j = i-l, need 0<=i-l<=n-1
        #   contribution: (1-h[i-l])
        # Term 2: h[i] in (1-h[j]) position → j = i, need 0<=i+l<=n-1
        #   contribution: -h[i+l]
        
        # Vectorized:
        # For term 1: valid i where 0<=i-l<=n-1 → l<=i<=n-1+l
        i_min1 = max(0, l)
        i_max1 = min(n-1, n-1+l)
        if i_min1 <= i_max1:
            j_range = np.arange(i_min1, i_max1+1)
            grad[j_range] += w * (1.0 - h[j_range - l])
        
        # For term 2: valid i where 0<=i+l<=n-1 → -l<=i<=n-1-l
        i_min2 = max(0, -l)
        i_max2 = min(n-1, n-1-l)
        if i_min2 <= i_max2:
            j_range = np.arange(i_min2, i_max2+1)
            grad[j_range] += w * (-h[j_range + l])
    
    return obj, grad

def project_feasible(h, n):
    """Project h onto {x: sum x = n/2, 0<=x<=1}"""
    h = np.clip(h, 0.0, 1.0)
    # Simple: add constant and re-clip (iterate a few times)
    for _ in range(10):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Only adjust elements that have room
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if np.sum(mask) == 0:
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_method(h, n, eta0, n_iters, t_deadline):
    """Subgradient method with projection"""
    best_local = compute_c5(h)
    best_h_local = h.copy()
    h_curr = h.copy()
    
    for it in range(n_iters):
        if time.time() > t_deadline:
            break
        obj, grad = smooth_obj_and_grad(h_curr, T=200)
        h_next = h_curr - eta0 / (1.0 + it * 0.01) * grad
        h_next = project_feasible(h_next, n)
        score = compute_c5(h_next)
        if score < best_local:
            best_local = score
            best_h_local = h_next.copy()
        h_curr = h_next
    return best_h_local, best_local

def slsqp_optimize(h0, n, T, maxiter, t_deadline):
    """SLSQP optimization"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = smooth_obj_and_grad(x, T)
        return v
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, T)
        return g
    
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                  'jac': lambda x: np.ones(n)}
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=[constraint],
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h0

def make_feasible(h, n):

        if abs(diff) < 1e-10:

        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = c * dx
    m = np.max(vals)
    shifted = (vals - m) * T
    exps = np.exp(shifted - np.max(shifted))
    weights = exps / np.sum(exps)
    obj = m + np.log(np.sum(exps)) / T
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        l = idx - (n - 1)
        ww = w * dx
        i_min1 = max(0, l)
        i_max1 = min(n - 1, n - 1 + l)
        if i_min1 <= i_max1:
            js = np.arange(i_min1, i_max1 + 1)
            grad[js] += ww * (1.0 - h[js - l])
        i_min2 = max(0, -l)
        i_max2 = min(n - 1, n - 1 - l)
        if i_min2 <= i_max2:
            js = np.arange(i_min2, i_max2 + 1)
            grad[js] += ww * (-h[js + l])
    return obj, grad

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if np.sum(mask) == 0:
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, T, maxiter, t_deadline):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v, _ = smooth_obj_and_grad(x, T)
        return v

    def jac(x):
        _, g = smooth_obj_and_grad(x, T)
        return g

    ones = np.ones(n)
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: ones}
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgradient_refine(h, n, t_deadline):
    best_h = h.copy()
    best_s = compute_c5(h)
    h_c = h.copy()
    for it in range(2000):
        if time.time() > t_deadline:
            break
        _, grad = smooth_obj_and_grad(h_c, T=500)
        eta = 0.005 / (1.0 + 0.005 * it)
        h_next = project_feasible(h_c - eta * grad, n)
        s = compute_c5(h_next)
        if s < best_s:
            best_s = s
            best_h = h_next.copy()
        h_c = h_next
    return best_h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    best_score = float('inf')
    best_h = None

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()

    n_list = [47, 56, 64, 80]

    for n in n_list:
        if time.time() > t_end - 40:
            break
        starts = []

        if parent is not None:
            if n == len(parent):
                starts.append(parent.copy())
            else:
                xp = np.linspace(0.0, 2.0, len(parent))
                xn = np.linspace(0.0, 2.0, n)
                starts.append(make_feasible(np.interp(xn, xp, parent), n))

        for _ in range(5):
            starts.append(make_feasible(rng.uniform(0.15, 0.85, n), n))

        x = np.linspace(0.0, 2.0, n)
        for freq in [1, 2, 3, 4, 5]:
            for amp in [0.08, 0.12, 0.18, 0.25]:
                h = 0.5 + amp * np.sin(np.pi * freq * x)
                starts.append(make_feasible(h, n))

        for nb in [3, 4, 5, 6]:
            h = np.zeros(n)
            for bi in range(nb * 2):
                s_idx = int(bi * n / (nb * 2))
                e_idx = int((bi + 1) * n / (nb * 2))
                h[s_idx:e_idx] = 1.0 if bi % 2 == 0 else 0.0
            starts.append(make_feasible(h, n))

        for period in [3, 4, 5, 7]:
            pat = (np.arange(period) < period // 2).astype(float)
            starts.append(make_feasible(np.tile(pat, n // period + 1)[:n], n))

        scored = []
        for h0 in starts:
            if time.time() > t_end - 15:
                break
            s = compute_c5(h0)
            if s < best_score:
                best_score = s
                best_h = h0.copy()
            scored.append((s, h0))

        scored.sort(key=lambda t: t[0])
        for s, h0 in scored[:4]:
            if time.time() > t_end - 20:
                break
            if s > best_score + 0.15:
                break
            h1 = slsqp_optimize(h0, n, T=60, maxiter=200, t_deadline=t_end - 10)
            s1 = compute_c5(h1)
            if s1 < best_score:
                best_score = s1
                best_h = h1.copy()
            h2 = slsqp_optimize(h1, n, T=200, maxiter=300, t_deadline=t_end - 10)
            s2 = compute_c5(h2)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()
            if time.time() < t_end - 15:
                h3, s3 = subgradient_refine(h2, n, t_deadline=t_end - 5)
                if s3 < best_score:
                    best_score = s3
                    best_h = h3.copy()

    if best_h is not None and time.time() < t_end - 15:
        nb = len(best_h)
        h_final = slsqp_optimize(best_h, nb, T=300, maxiter=400, t_deadline=t_end - 3)
        sf = compute_c5(h_final)
        if sf < best_score:
            best_score = sf
            best_h = h_final
        if time.time() < t_end - 5:
            h_sg, s_sg = subgradient_refine(best_h, nb, t_deadline=t_end - 2)
            if s_sg < best_score:
                best_score = s_sg
                best_h = h_sg

    if best_h is None:
        n = 47
        best_h = make_feasible(np.full(n, 0.5), n)
        best_score = compute_c5(best_h)

    return (best_h, best_score, len(best_h))
