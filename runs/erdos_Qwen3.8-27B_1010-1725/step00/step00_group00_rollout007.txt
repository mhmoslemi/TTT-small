import numpy as np
from scipy.optimize import minimize
import time

_G_N = 45

def _cons(x):
    return x.sum() - _G_N / 2.0

def _objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _objective_smooth(h, temp=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr - np.max(corr)
    return (np.max(corr) + temp * np.log(np.sum(np.exp(c / temp)))) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(h_src, n_dst):
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def _slsqp_optimize(h_start, n, maxiter, ftol):
    global _G_N
    _G_N = n
    h_start = _project(h_start.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons}]
    try:
        res = minimize(_objective, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = _project(res.x, n)
            return h_new, _objective(h_new)
    except Exception:
        pass
    return None, None

def _slsqp_smooth(h_start, n, maxiter, ftol, temp):
    global _G_N
    _G_N = n
    h_start = _project(h_start.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons}]
    try:
        res = minimize(_objective_smooth, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       args=(temp,),
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = _project(res.x, n)
            return h_new, _objective(h_new)
    except Exception:
        pass
    return None, None
