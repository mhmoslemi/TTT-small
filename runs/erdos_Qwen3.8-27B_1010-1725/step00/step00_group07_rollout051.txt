import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode="full")
    return float(np.max(r) * dx)

def project_bounded_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0.0, 1.0)

def subgradient_step(h, lr):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode="full")
    k_star = int(np.argmax(r))
    offset = n - 1
    i_min = max(0, offset - k_star)
    i_max = min(n - 1, 2 * n - 2 - k_star)
    grad = np.zeros(n)
    for i in range(i_min, i_max + 1):
        j = i + k_star - offset
        grad[i] += (1.0 - h[j])
        grad[j] += (-h[i])
    grad *= dx
    h_new = h - lr * grad
    target = n / 2.0
    h_new = project_bounded_simplex(h_new, target)
    return h_new, compute_c5(h_new)

def softmax_subgradient_step(h, lr, tau, n_active):
    """Use top n_active shifts for smoother gradient."""
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode="full")
    # Use top shifts weighted by softmax
    r_sorted = np.sort(r)[::-1]
    top = r_sorted[:n_active]
    weights = np.exp(tau * (top - top[0]))
    weights /= weights.sum()
    
    # Get indices of top shifts
    # Actually, let's just use the top n_active shifts
    top_indices = np.argsort(r)[::-1][:n_active]
    
    offset = n - 1

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode="full")
    return float(np.max(r) * dx)

def project_bounded_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if projected.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def compute_topk_subgrad(h, topk=10):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode="full")
    # Get top k shifts
    k_indices = np.argsort(r)[::-1][:topk]
    # Compute weights (normalized)
    r_top = r[k_indices]
    weights = r_top / r_top.sum() if r_top.sum() > 0 else np.ones(topk) / topk
    
    offset = n - 1
    grad = np.zeros(n)
    for idx in range(topk):
        k = k_indices[idx]
        w = weights[idx]
        i_min = max(0, offset - int(k))
        i_max = min(n - 1, 2 * n - 2 - int(k))
        for i in range(i_min, i_max + 1):
            j = i + int(k) - offset
            grad[i] += w * (1.0 - h[j])
            grad[j] += w * (-h[i])
    return grad * dx

    return float(np.max(r) * dx)

def project_bounded_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def topk_subgrad(h, topk):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    ki = np.argsort(r)[::-1][:topk]
    rv = r[ki]
    wsum = rv.sum()
    w = rv / wsum if wsum > 1e-12 else np.full(topk, 1.0 / topk)
    off = n - 1
    g = np.zeros(n)
    for t in range(topk):
        k = int(ki[t])
        wt = float(w[t])
        lo = max(0, off - k)
        hi = min(n - 1, 2 * n - 2 - k)
        vi = np.arange(lo, hi + 1)
        vj = vi + k - off
        g[vi] += wt * (1.0 - h[vj])
        g[vj] -= wt * h[vi]
    return g * dx

def subgrad_loop(h, tgt, n, lr_init, n_steps, topk, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(n_steps):
        if time.time() >= deadline:
            break
        g = topk_subgrad(h, topk)
        h = h - lr * g
        h = project_bounded_simplex(h, tgt)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c5
        if it % 2000 == 1999:
            lr = max(0.005, lr * 0.93)
        if it % 8000 == 7999:
            h = best_h + rng.standard_normal(n) * 0.06
            h = project_bounded_simplex(h, tgt)
            lr = max(lr, 0.06)
    return best_h, best_c5

def refine_round(h, tgt, n, deadline):
    """Try rounding to binary then local pair-swaps."""
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    diff = tgt - h_bin.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                h_bin[idx] = 1.0
                diff -= 1.0
            elif diff < 0:
                h_bin[idx] = 0.0
                diff += 1.0
            if abs(diff) < 1e-9:
                break
    h_bin = project_bounded_simplex(h_bin, tgt)
    c5_bin = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5_bin
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in ones:
            for j in zeros:
                h_new = best_h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                h_new = project_bounded_simplex(h_new, tgt)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-10:
                    best_h = h_new
                    best_c5 = c5
                    improved = True
                    ones = np.where(best_h > 0.5)[0]
                    zeros = np.where(best_h < 0.5)[0]
                    break
            if improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(80, 0.5)

    n = len(h)
    tgt = n / 2.0
    h = project_bounded_simplex(h, tgt)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Main subgradient descent
    d1 = t0 + budget_s * 0.55
    best_h, best_c5 = subgrad_loop(h, tgt, n, 0.12, 500000, 10, d1, rng)

    # Phase 2: Second run from slightly perturbed best with different topk
    h2 = best_h + rng.standard_normal(n) * 0.04
    h2 = project_bounded_simplex(h2, tgt)
    d2 = t0 + budget_s * 0.75
    h2b, c5_2 = subgrad_loop(h2, tgt, n, 0.08, 300000, 15, d2, rng)
    if c5_2 < best_c5:
        best_h, best_c5 = h2b, c5_2

    # Phase 3: Try binary rounding + local search on top candidate
    d3 = t0 + budget_s * 0.92
    h_bin, c5_bin = refine_round(best_h, tgt, n, d3)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    # Phase 4: Final polishing from best
    h4 = best_h + rng.standard_normal(n) * 0.01
    h4 = project_bounded_simplex(h4, tgt)
    d4 = t0 + budget_s * 0.98
    hr, c5_r = subgrad_loop(h4, tgt, n, 0.02, 100000, 5, d4, rng)
    if c5_r < best_c5:
        best_h, best_c5 = hr, c5_r

    # Final projection and verification
    best_h = project_bounded_simplex(best_h, tgt)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-9) and np.all(best_h <= 1.0 + 1e-9)
    assert abs(best_h.sum() - tgt) < 1e-6
    return best_h, best_c5, n
