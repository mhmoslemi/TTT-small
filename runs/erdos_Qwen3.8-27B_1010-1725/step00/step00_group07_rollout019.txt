def optimize_one_coordinate(h, i, n, corr_current):

    t_cur = h[i]
    
    # Compute a_k for each lag
    # a_k = [i+k-n+1 in [0,n-1]] * (1-h[i+k-n+1]) - [i-k+n-1 in [0,n-1]] * h[i-k+n-1]
    a = np.zeros(2*n-1)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        # Fine adjust
        diff = target - np.sum(h)
        if abs(diff) > 1e-15:
            # Add to elements not at boundary

            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

    """Coordinate descent with exact 1D optimization"""
    start = time.time()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in range(n):
            t_cur = h[i]
            
            # Compute a_k
            a = np.zeros(2*n-1)
            k_min1 = max(0, n - 1 - i)
            k_max1 = min(2*n-2, n - 1 + i)
            if k_min1 <= k_max1:
                idx = np.arange(k_min1, k_max1 + 1)
                pos = i + idx - n + 1
                a[idx] += (1.0 - h[pos])
            
            k_min2 = max(0, i + 1)
            k_max2 = min(2*n-2, i + n)
            if k_min2 <= k_max2:
                idx = np.arange(k_min2, k_max2 + 1)
                pos = i - idx + n - 1
                a[idx] -= h[pos]
            
            c = corr - t_cur * a
            
            # Minimize max(c + a*t) over t in [0,1]
            ts = np.linspace(0, 1, 100)
            vals = np.max(c[None, :] + np.outer(ts, a), axis=1)
            best_t = ts[np.argmin(vals)]
            
            h[i] = best_t
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        current_c5 = float(np.max(corr) * 2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_sweeps=500, time_limit=30):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()

            ts = np.linspace(0.0, 1.0, 50)
