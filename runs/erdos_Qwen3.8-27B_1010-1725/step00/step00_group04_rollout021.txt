from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def smooth_max(vals, tau=50.0):
    m = np.max(vals)
    if tau * (vals - m).max() < -50:
        return m
    return m + (1.0/tau) * np.log(np.sum(np.exp(tau * (vals - m))))

def objective_and_grad(h, tau=50.0, penalty=500.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full') * dx
    obj = smooth_max(corr, tau)
    # Penalty for sum constraint
    sum_err = np.sum(h) - N/2.0
    obj += penalty * sum_err * sum_err
    
    # Finite difference gradient
    eps = 1e-5
    grad = np.zeros(N)

        h_pert[j] += eps
        corr_p = np.correlate(h_pert, 1-h_pert, mode='full') * dx

        obj_p += penalty * sum_err_p * sum_err_p
        grad[j] = (obj_p - obj) / eps
    return obj, grad

  - For j in [-s, N-1]: grad[j] += w_s * (1 - h[j+s])
  - For j in [0, N-1+s]: grad[j] -= w_s * h[j-s]
- If s = 0:
  - For all j: grad[j] += w_s * (1 - 2*h[j])

And then add 2*penalty*(sum(h)-N/2) to all components.

This is O(N²) total, which is great.
