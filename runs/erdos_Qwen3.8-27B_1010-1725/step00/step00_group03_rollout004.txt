import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

try:
    from scipy.signal import correlate as _corr
except Exception:
    _corr = None


def project_to_box_sum(z, target=None):
    v = np.asarray(z, dtype=float).ravel()
    n = v.size
    if target is None:
        target = 0.5 * n
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / float(n))
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        m = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - m, 0.0, 1.0)))
        if s > target:
            lo = m
        else:
            hi = m
    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - float(np.sum(out))
    idx = np.where((out > 1e-9) & (out < 1.0 - 1e-9))[0]
    if idx.size:
        for _ in range(4):
            if abs(d) < 1e-14:
                break
            idx = np.where((out > 1e-9) & (out < 1.0 - 1e-9))[0]
            if idx.size == 0:
                break
            if d > 0:
                st = min(d / idx.size, float(np.max(1.0 - out[idx])))
            else:
                st = max(d / idx.size, -float(np.max(out[idx])))
            out[idx] += st
            d = target - float(np.sum(out))
    d = target - float(np.sum(out))
    if abs(d) > 1e-13:
        if d > 0:
            idx = np.where(out < 1.0)[0]
            if idx.size:
                out[idx[0]] = min(1.0, out[idx[0]] + d)
        else:
            idx = np.where(out > 0.0)[0]
            if idx.size:
                out[idx[0]] = max(0.0, out[idx[0]] + d)
    return np.clip(out, 0.0, 1.0)


def full_correlations(h):
    n = h.size
    if _corr is not None:
        try:
            return np.asarray(_corr(h, 1.0 - h, mode="full", method="auto"), dtype=float)
        except Exception:
            pass
    out = np.empty(2 * n - 1, dtype=float)
    vm = 1.0 - h
    for d in range(n):
        l = n - d
        out[n - 1 - d] = float(np.dot(h[:l], vm[d:]))
        if d:
            out[n - 1 + d] = float(np.dot(h[d:], vm[:l]))
    return out


def eval_C(h):
    return float(np.max(full_correlations(h))) * (2.0 / h.size)


def lag_gradient(h, idx):
    n = h.size
    center = n - 1
    d = abs(idx - center)
    g = np.zeros(n, dtype=float)
    if d == 0:
        return 1.0 - 2.0 * h
    if idx > center:
        g[d:n] += 1.0
        g[d:n] -= h[:n - d]
        g[:n - d] -= h[d:n]
    else:
        g[:n - d] += 1.0
        g[:n - d] -= h[d:n]
        g[d:n] -= h[:n - d]
    return g


def smooth_objective(x, tau):
    c = full_correlations(x)
    cmax = float(np.max(c))
    dx = 2.0 / x.size
    if tau <= 1e-10:
        return cmax * dx
    z = np.clip((c - cmax) / tau, -30.0, 0.0)
    w = np.exp(z)
    return (cmax + tau * float(np.log(float(np.sum(w))))) * dx


def smooth_gradient(x, tau):
    c = full_correlations(x)
    cmax = float(np.max(c))
    dx = 2.0 / x.size
    n = x.size
    if tau <= 1e-10:
        return lag_gradient(x, int(np.argmax(c))) * dx
    z = np.clip((c - cmax) / tau, -30.0, 0.0)
    w = np.exp(z)
    s = float(np.sum(w))
    g = np.zeros(n, dtype=float)
    if s <= 0.0:
        return g
    thr = 1e-6 * s
    for i in range(c.size):
        wi = float(w[i])
        if wi > thr:
            g += (wi / s) * lag_gradient(x, i)
    return g * dx


def sum_constraint(x):
    return float(np.sum(x)) - 0.5 * x.size


def sum_constraint_jac(x):
    return np.ones(x.size, dtype=float)


def maxiter_for_n(n):
    if n <= 80:
        return 70
    if n <= 160:
        return 50
    if n <= 240:
        return 35
    return 25


def run_slsqp(x, tau, maxiter):
    n = x.size
    try:
        res = minimize(
            smooth_objective,
            x,
            jac=smooth_gradient,
            args=(tau,),
            bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac}],
            method="SLSQP",
            options={"maxiter": int(max(1, maxiter)), "ftol": 1e-11},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=float)
    except Exception:
        pass
    return x


def line_obj(t, h, src, dest):
    x = h.copy()
    a = x[src] + t
    b = x[dest] - t
    if a < 0.0:
        a = 0.0
    elif a > 1.0:
        a = 1.0
    if b < 0.0:
        b = 0.0
    elif b > 1.0:
        b = 1.0
    x[src] = a
    x[dest] = b
    return eval_C(x)


def pair_refine(h, deadline, seed):
    h = project_to_box_sum(h)
    best = float(eval_C(h))
    best_h = h.copy()
    n = h.size
    rng = np.random.RandomState(seed)
    sweep = 0
    while time.time() < deadline and sweep < 80:
        c = full_correlations(h)
        order = np.argsort(c)[::-1]
        K = min(8, c.size)
        g = np.zeros(n, dtype=float)
        for i in order[:K]:
            g += lag_gradient(h, int(i))
        gp = g - float(np.mean(g))
        pos = np.where(gp > 1e-12)[0]
        neg = np.where(gp < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            if sweep > 1:
                break
            z = h + 0.02 * (rng.rand(n) - 0.5)
            z = project_to_box_sum(z)
            fz = eval_C(z)
            if fz < best - 1e-13:
                best = float(fz)
                best_h = z.copy()
                h = z.copy()
            sweep += 1
            continue
        pos_order = pos[np.argsort(gp[pos])[::-1]][:10]
        neg_order = neg[np.argsort(gp[neg])][:10]
        kp = 50 if n <= 120 else 25
        scored = []
        for s in pos_order:
            for d in neg_order:
                if s != d:
                    scored.append((float(gp[s] - gp[d]), int(s), int(d)))
        added = 0
        while len(scored) < kp and added < kp * 3:
            s = int(rng.randint(n))
            d = int(rng.randint(n))
            if s != d:
                scored.append((float(gp[s] - gp[d]), s, d))
                added += 1
        scored.sort(reverse=True)
        improved = False
        for _score, s, d in scored[:kp]:
            if time.time() >= deadline:
                break
            lo = -float(h[d])
            hi = 1.0 - float(h[s])
            if hi - lo < 1e-10:
                continue
            try:
                r = minimize_scalar(
                    line_obj,
                    args=(h, s, d),
                    bounds=(lo, hi),
                    method="bounded",
                    options={"maxiter": 28},
                )
                t = float(r.x)
                x = h.copy()
                x[s] += t
                x[d] -= t
                x = project_to_box_sum(x)
                f = float(eval_C(x))
                if f < best - 1e-13:
                    best = f
                    best_h = x.copy()
                    h = x.copy()
                    improved = True
            except Exception:
                pass
        sweep += 1
        if not improved:
            if sweep > 2:
                break
            z = h + 0.01 * (rng.rand(n) - 0.5)
            z = project_to_box_sum(z)
            fz = eval_C(z)
            if fz < best - 1e-13:
                best = float(fz)
                best_h = z.copy()
                h = z.copy()
    return best_h, best


def optimize_single(h0, seed, deadline):

    best = float(eval_C(h))
    best_h = h.copy()
    n = h.size

    rem = deadline - time.time()

        taus = (1.2, 0.6, 0.25, 0.1)
        x = h.copy()
        xc = float(eval_C(x))
        for tau in taus[:ntau]:
            if time.time() >= deadline:
                break
            y = project_to_box_sum(run_slsqp(x, tau, mi))
            yc = float(eval_C(y))
            if yc < best:
                best = yc
                best_h = y.copy()
            if yc <= xc + 1e-8:
                x = y
                xc = yc

    rem = deadline - time.time()
    if rem > 8.0:
        pd = min(deadline, time.time() + max(5.0, 0.5 * rem))
        hp, cp = pair_refine(best_h.copy(), pd, seed)
        if cp < best:
            best = float(cp)
            best_h = hp.copy()

    rem = deadline - time.time()
    if rem > 10.0:
        mi = int(max(5, min(maxiter_for_n(n), 0.30 * rem + 5.0)))
        x = best_h.copy()
        xc = float(eval_C(x))
        for tau in (0.3, 0.1):
            if time.time() >= deadline:
                break
            y = project_to_box_sum(run_slsqp(x, tau, mi))
            yc = float(eval_C(y))
            if yc < best:
                best = yc
                best_h = y.copy()
            if yc <= xc + 1e-8:
                x = y
                xc = yc

    return best_h, float(best)


def resample_nearest(h, n):
    m = h.size
    if m == n:
        return h.copy()
    idx = np.floor((np.arange(n) + 0.5) * (m / float(n))).astype(int)
    idx = np.clip(idx, 0, m - 1)
    return project_to_box_sum(h[idx], n / 2.0)


def resample_linear(h, n):
    m = h.size
    if m == n:
        return h.copy()
    x0 = (np.arange(m) + 0.5) * (2.0 / m)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    v = np.interp(x1, x0, h, left=h[0], right=h[-1])
    return project_to_box_sum(v, n / 2.0)


def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    if n > 0:
        k = n // 2
        if k > 0:
            idx = rng.choice(n, size=k, replace=False)
            h[idx] = 1.0
    return project_to_box_sum(h, n / 2.0)
