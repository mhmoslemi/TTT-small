```python
def swap_delta(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0 and h[j]:0->1"""
    corr = np.zeros(2*n-1)
    # h[i]: 1->0
    # a-array change: -(1-h[i-k]) at lag k where i-k is valid (0<=i-k<n)
    # v-array change: +h[i+k] at lag k where i+k is valid (0<=i+k<n)
    for k in range(-(n-1), n):
        dk = k + (n-1)  # index in full correlation
        delta = 0.0
        # h[i] 1->0 in a-array: contributes -(1-h[i-k]) if 0<=i-k<n
        if 0 <= i - k < n:
            delta -= (1.0 - h[i-k])
        # h[i] 1->0 in v-array: contributes +h[i+k] if 0<=i+k<n
        if 0 <= i + k < n:
            delta += h[i+k]
        # h[j] 0->1 in a-array: contributes +(1-h[j-k]) if 0<=j-k<n
        if 0 <= j - k < n:
            delta += (1.0 - h[j-k])
        # h[j] 0->1 in v-array: contributes -h[j+k] if 0<=j+k<n
        if 0 <= j + k < n:
            delta -= h[j+k]
        corr[dk] = delta
    return corr
```

def swap_delta_vec(h, i, j, n):
    """Vectorized change in all correlations for swap (i:1->0, j:0->1)"""
    k = np.arange(-(n-1), n)  # lags
    dk = k + (n-1)  # output indices 0..2n-2
    
    # h[i] 1->0 in a-array: -(1-h[i-k]) where 0<=i-k<n i.e. i-n+1<=k<=i
    mask_a_i = (k >= i - (n-1)) & (k <= i) & (k >= -(n-1)) & (k <= n-1)
    idx_a_i = i - k  # index into h for (1-h)
    
    # h[i] 1->0 in v-array: +h[i+k] where 0<=i+k<n i.e. -i<=k<=n-1-i
    mask_v_i = (k >= -i) & (k <= n-1-i) & (k >= -(n-1)) & (k <= n-1)
    idx_v_i = i + k
    
    # h[j] 0->1 in a-array: +(1-h[j-k]) where 0<=j-k<n
    mask_a_j = (k >= j - (n-1)) & (k <= j) & (k >= -(n-1)) & (k <= n-1)
    idx_a_j = j - k
    
    # h[j] 0->1 in v-array: -h[j+k] where 0<=j+k<n
    mask_v_j = (k >= -j) & (k <= n-1-j) & (k >= -(n-1)) & (k <= n-1)
    idx_v_j = j + k
    
    delta = np.zeros(2*n-1)
    delta[mask_a_i] -= (1.0 - h[idx_a_i[mask_a_i]])
    delta[mask_v_i] += h[idx_v_i[mask_v_i]]
    delta[mask_a_j] += (1.0 - h[idx_a_j[mask_a_j]])
    delta[mask_v_j] -= h[idx_v_j[mask_v_j]]
    
    return delta

    delta = np.zeros(2*n-1)
    
    # h[i]: 1->0
    # a-array: -（1-h[i-k]) for valid i-k

Now, if we change h[i] from 1 to 0:
- In the term h[n+k](1-h[n]), h[i] appears as h[n+k] when n+k=i, i.e., n=i-k.

- h[i] also appears as h[n] in (1-h[n]) when n=i.

  Wait, the original term is h[i+k]*(1-h[i]) = h[i+k]*0 = 0 (since h[i]=1).

If we change h[j] from 0 to 1:
- h[j] appears as h[n+k] when n=j-k.

```python
def get_swap_delta(h, i, j, n):
    """Returns 2n-1 array of changes to full correlation for swap i(1->0), j(0->1)"""
    size = 2*n - 1
    delta = np.zeros(size)
    k = np.arange(-(n-1), n)  # all possible lags
    
    # Term 1: -(1-h[i-k]) for 0 <= i-k < n, i.e., i-n+1 <= k <= i
    m1 = (i - k >= 0) & (i - k < n)
    delta[m1] -= (1.0 - h[i - k[m1]])
    
    # Term 2: +h[i+k] for 0 <= i+k < n, i.e., -i <= k <= n-1-i
    m2 = (i + k >= 0) & (i + k < n)
    delta[m2] += h[i + k[m2]]
    
    # Term 3: +(1-h[j-k]) for 0 <= j-k < n
    m3 = (j - k >= 0) & (j - k < n)
    delta[m3] += (1.0 - h[j - k[m3]])
    
    # Term 4: -h[j+k] for 0 <= j+k < n
    m4 = (j + k >= 0) & (j + k < n)
    delta[m4] -= h[j + k[m4]]
    
    return delta
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def get_swap_delta(h, i, j, n):
    size = 2 * n - 1
    delta = np.zeros(size)
    k = np.arange(-(n - 1), n)
    m1 = (i - k >= 0) & (i - k < n)
    delta[m1] -= (1.0 - h[i - k[m1]])
    m2 = (i + k >= 0) & (i + k < n)
    delta[m2] += h[i + k[m2]]
    m3 = (j - k >= 0) & (j - k < n)
    delta[m3] += (1.0 - h[j - k[m3]])
    m4 = (j + k >= 0) & (j + k < n)
    delta[m4] -= h[j + k[m4]]
    return delta

def to_binary(h, n):
    hb = np.round(h).astype(float)
    target = n // 2
    diff = int(hb.sum()) - target
    if diff > 0:
        idx = np.where(hb == 1)[0]
        order = idx[np.argsort(h[idx])]
        for p in order[:diff]:
            hb[p] = 0.0
    elif diff < 0:
        idx = np.where(hb == 0)[0]
        order = idx[np.argsort(-h[idx])]
        for p in order[:(-diff)]:
            hb[p] = 1.0
    c5, _ = compute_c5(hb)
    if c5 > 1.0:
        return None
    return hb

def local_search(h, n, max_iters=8000, n_tries=80, rng=None, time_limit=50.0):
    dx = 2.0 / n
    t0 = time.time()
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(corr)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    stall_count = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        k_max = np.argmax(corr)
        k_lag = k_max - (n - 1)
        best_new_c5 = c5
        best_i, best_j = -1, -1
        candidates = []
        lo = max(0, -k_lag)
        hi = min(n, n - k_lag)
        for n_pos in range(lo, hi):
            if h[n_pos + k_lag] == 1 and h[n_pos] == 0:
                candidates.append((n_pos + k_lag, n_pos))
        if candidates:
            a = candidates[rng.integers(len(candidates))]
            for _ in range(15):
                z = zeros[rng.integers(len(zeros))]
                candidates.append((a[0], z))
        for _ in range(n_tries):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            candidates.append((i, j))
        seen = set()
        for i, j in candidates:
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            if h[i] != 1 or h[j] != 0 or i == j:
                continue
            d = get_swap_delta(h, i, j, n) * dx
            new_c5 = np.max(corr + d)
            if new_c5 < best_new_c5 - 1e-14:
                best_new_c5 = new_c5
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            c5 = np.max(corr)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 40 and stall_count % 20 == 0:
                nk = int(rng.integers(2, 6))
                for _ in range(nk):
                    if not ones or not zeros:
                        break
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                c5 = np.max(corr)
    return c5, h

def continuous_refine(h, n, rng, time_limit=30.0):
    t0 = time.time()
    c5, corr = compute_c5(h)
    dx = 2.0 / n
    for it in range(2000):
        if time.time() - t0 > time_limit:
            break
        if abs(c5 - np.min(corr)) < 1e-6:
            break
        k_max = np.argmax(corr)
        k_lag = k_max - (n - 1)
        io = np.where(h > 0.01)[0]
        iz = np.where(h < 0.99)[0]
        if len(io) == 0 or len(iz) == 0:
            break
        found = False
        for _ in range(50):
            i = io[rng.integers(len(io))]
            j = iz[rng.integers(len(iz))]
            if i == j or h[i] <= 0 or 1 - h[j] <= 0:
                continue
            d = get_swap_delta(h, i, j, n) * dx
            d_per_unit = d / 1.0
            lags = d_per_unit
            curr = corr.copy()
            best_eps = 0.0
            best_val = c5
            max_eps = min(h[i], 1.0 - h[j])
            for eps in np.linspace(0.01, max_eps, 20):
                new_corr = curr + eps * lags
                val = np.max(new_corr)
                if val < best_val:
                    best_val = val
                    best_eps = eps
            if best_eps > 0.001 and best_val < c5 - 1e-8:
                h[i] -= best_eps
                h[j] += best_eps
                c5, corr = compute_c5(h)
                found = True
                break
        if not found:
            break
    return c5, h

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def periodic_pattern(n, p, ones_in_period, rng):
    base = np.zeros(p)
    base[ones_in_period] = 1.0
    h = np.tile(base, n // p)
    if n % p != 0:
        h = np.concatenate([h, base[:n % p]])
    extra = n - len(h)
    if extra > 0:
        h = np.concatenate([h, np.zeros(extra)])
    return h[:n]

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    def elapsed():
        return time.time() - start_time

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        c5_0, _ = compute_c5(h0)
        best_c5 = c5_0
        best_h = h0.copy()
        best_n = n0
        hb = to_binary(h0, n0)
        if hb is not None:
            c5b, _ = compute_c5(hb)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hb.copy()
        if elapsed() < budget_s * 0.2:
            h_try = hb if hb is not None else h0.copy()
            if n0 % 2 == 0:
                c5l, h_ls = local_search(h_try.copy(), n0, max_iters=5000, n_tries=80, rng=rng, time_limit=budget_s * 0.15)
                if c5l < best_c5:
                    best_c5 = c5l
                    best_h = h_ls.copy()
                    best_n = n0
            else:
                c5c, h_c = continuous_refine(h_try.copy(), n0, rng, time_limit=budget_s * 0.15)
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h_c.copy()
                    best_n = n0

    n_candidates = [94, 96, 98, 100, 102, 104, 106, 108, 110, 112, 114, 116, 118, 120, 122, 124]
    for n in n_candidates:
        if n % 2 != 0:
            continue
        if elapsed() > budget_s - 20:
            break
        time_left = budget_s - 20 - elapsed()
        n_restarts = max(2, min(12, int(time_left / max(1.0, n * 0.08))))
        for r in range(n_restarts):
            if elapsed() > budget_s - 10:
                break
            if r == 0 and n % 4 == 0:
                p = n // 25 if n >= 100 else 4
                p = max(4, min(p, n // 2))
                while p > 2 and n % p != 0:

                else:
                    h = random_binary(n, rng)
            else:
                h = random_binary(n, rng)
            c5, h = local_search(h, n, max_iters=3000, n_tries=60, rng=rng, time_limit=min(25, time_left / max(1, n_restarts - r)))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    if elapsed() < budget_s * 0.8 and best_h is not None:
        nb = len(best_h)
        if nb % 2 == 0 and np.all((best_h == 0) | (best_h == 1)):
            c5t, ht = local_search(best_h.copy(), nb, max_iters=5000, n_tries=120, rng=rng, time_limit=budget_s * 0.15)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = ht.copy()
    return best_h, best_c5, len(best_h)
