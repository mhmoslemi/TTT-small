def c5_value(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr * dx)

def fix_sum(h, target):
    """Project h onto the sum constraint (approximately)."""
    diff = target - h.sum()
    h = h + diff / len(h)
    return np.clip(h, 0, 1)

def upsample(h, new_n):
    """Upsample h to new_n points."""
    old_n = len(h)
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    # Piecewise linear interpolation
    x_old = np.arange(old_n) * dx_old
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # Fix sum
    h_new = fix_sum(h_new, new_n / 2.0)
    return h_new

def local_improve(h, n_iter=100, seed=42):

    np.random.seed(seed)

overlap = sum_i h[i] * (1-h[i + (n-1-k*)]) * dx

To reduce this, I want to:
- Decrease h[i] where (1-h[i+(n-1-k*)]) is large, i.e., where h[i+(n-1-k*)] is small

But I need to maintain the sum constraint, so any decrease in h[i] must be compensated by an increase in h[j] for some j.

The most effective move: pick i where h[i] is large AND h[i+s_shift] is small (where s_shift = n-1-k*). Then decrease h[i] by δ and increase h[i+s_shift] by δ (if in range). This directly reduces the overlap at shift s*.
