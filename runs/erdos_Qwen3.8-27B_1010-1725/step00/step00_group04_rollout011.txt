    corr = np.correlate(h, 1.0 - h, mode='full')

    sm = m + eps * np.log(np.sum(np.exp((corr - m) / eps)))
    n = len(h)
    sp = beta * (np.sum(h) - n / 2.0) ** 2
    return sm + sp

def _direct_obj(h):
    return _compute_c5(h)

def _sum_con(h):
    return np.sum(h) - _OPT['n_target']

def _sum_jac(h):
    return np.ones(len(h))

def _project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            remaining = diff
            for i in order:
                if remaining <= 1e-12:
                    break
                room = 1.0 - h[i]
                add = min(remaining, room)
                h[i] += add
                remaining -= add
        else:
            order = np.argsort(-h)
            remaining = -diff
            for i in order:
                if remaining <= 1e-12:
                    break
                room = h[i]
                sub = min(remaining, room)
                h[i] -= sub
                remaining -= sub
    return h

def _upsample_h(h, factor):
    n = len(h)
    new_n = n * factor
    old_x = (np.arange(n) + 0.5) / n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    return _project_h(h_new)

def _optimize_h(h_start, n, maxiter, time_limit):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    _OPT['n_target'] = n / 2.0

    eps_list = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    h_curr = h_start.copy()
    c5_curr = _compute_c5(h_curr)

    for eps in eps_list:
        if time.time() - t0 > time_limit:
            break
        _OPT['eps'] = eps
        _OPT['beta'] = 50.0
        res = minimize(_smooth_obj, h_curr.copy(), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-7, 'maxls': 40})
        h_opt = _project_h(res.x)
        c5_opt = _compute_c5(h_opt)
        if c5_opt < c5_curr:
            c5_curr = c5_opt
            h_curr = h_opt

    # SLSQP refinement
    if time.time() - t0 < time_limit - 10:
        res = minimize(_direct_obj, h_curr.copy(), method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}],
                       options={'maxiter': 150, 'ftol': 1e-9})
        h_opt = _project_h(res.x)
        c5_opt = _compute_c5(h_opt)
        if c5_opt < c5_curr:
            c5_curr = c5_opt
            h_curr = h_opt

    return h_curr, c5_curr

def _perturb_optimize(h_best, n, seed, time_limit):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_curr = h_best.copy()
    c5_curr = _compute_c5(h_curr)
    bounds = [(0.0, 1.0)] * n
    _OPT['n_target'] = n / 2.0

    for attempt in range(4):
        if time.time() - t0 > time_limit:
            break
        h_pert = h_curr + rng.normal(0, 0.08, n)
        h_pert = _project_h(h_pert)
        _OPT['eps'] = 0.02
        _OPT['beta'] = 100.0
        res = minimize(_smooth_obj, h_pert.copy(), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-7, 'maxls': 40})
        h_opt = _project_h(res.x)
        c5_opt = _compute_c5(h_opt)
        if c5_opt < c5_curr:
            c5_curr = c5_opt
            h_curr = h_opt

    return h_curr, c5_curr

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def _tleft():
        return budget_s - (time.time() - t_start)

    # Initialize from parent
    h_initial = None
    try:
        h_initial = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        pass

    if h_initial is None or len(h_initial) == 0:
        h_initial = np.full(42, 0.5)

    h_best = _project_h(h_initial)
    n_best = len(h_best)
    c5_best = _compute_c5(h_best)

    # Phase 1: Optimize at initial resolution
    if _tleft() > 60:
        h_opt, c5_opt = _optimize_h(h_best, n_best, maxiter=400, time_limit=min(200, _tleft() - 30))
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt
            n_best = len(h_best)

    # Phase 2: Multi-resolution refinement
    for round_idx in range(5):
        if _tleft() < 120:
            break
        new_n = n_best * 2
        if new_n > 2000:
            break
        h_up = _upsample_h(h_best, 2)
        time_alloc = min(_tleft() - 30, 180)
        h_opt, c5_opt = _optimize_h(h_up, new_n, maxiter=250, time_limit=time_alloc)
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt
            n_best = new_n

    # Phase 3: Perturbation restarts at best resolution
    if _tleft() > 40:
        h_opt, c5_opt = _perturb_optimize(h_best, n_best, seed, time_limit=min(_tleft() - 10, 200))
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt

    # Final refinement at best resolution
    if _tleft() > 30:
        h_opt, c5_opt = _optimize_h(h_best, n_best, maxiter=200, time_limit=min(_tleft() - 5, 80))
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt

    # Final projection and evaluation
    h_best = _project_h(h_best)
    c5_final = _compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return (h_best, c5_final, len(h_best))
