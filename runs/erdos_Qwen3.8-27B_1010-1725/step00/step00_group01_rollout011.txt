    # Final fix
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in range(n):
                if h[i] < 1.0:
                    adj = min(diff, 1.0 - h[i])
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    adj = max(diff, h[i] - 1.0)
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
    return h

def project_to_feasible(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    # Use bisection on a water-filling approach
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0, 1)
        s = np.sum(h_proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + mid, 0, 1)
    # Fine-tune
    diff = target - np.sum(h_proj)
    h_proj += diff / n
    return np.clip(h_proj, 0, 1)

def projected_gradient_step(h, n, step_size, alpha=100):
    """One step of projected gradient on smooth maximin"""
    c = compute_correlations(h)
    # Weight: softmax of c/alpha gives approximate subgradient of max
    c_shifted = c - np.max(c)
    weights = np.exp(alpha * c_shifted) / np.sum(np.exp(alpha * c_shifted))
    # Gradient of weighted sum of correlations
    # c_k = (2/n) * sum_i h[i] * (1-h[i-k])
    # Need gradient of sum_k w_k * c_k w.r.t. h[j]
    # This requires knowing which h[j] contributes to which c_k
    # For efficiency, compute gradient numerically for small n
    grad = np.zeros_like(h)
    eps = 1e-7
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] = min(1.0, h[j] + eps)
        c_plus = compute_correlations(h_plus)
        grad[j] = np.sum(weights * c_plus) / eps  # wrong, need (f+ - f-)/2eps
        h_minus = h.copy()
        h_minus[j] = max(0.0, h[j] - eps)
        c_minus = compute_correlations(h_minus)
        grad[j] = (np.sum(weights * c_plus) - np.sum(weights * c_minus)) / (2 * eps)
    
    h_new = h - step_size * grad
    h_new = project_to_feasible(h_new, n)
    return h_new, compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx


def compute_c5(h):
    return float(np.max(compute_correlations(h)))


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(-h):
                if h[i] < 1.0:
                    adj = min(diff, 1.0 - h[i])
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-13:
                        break
        else:
            for i in np.argsort(h):
                if h[i] > 0.0:
                    adj = max(diff, h[i] - 1.0)
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-13:
                        break
    return h


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_p) < target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return fix_sum(h_p, n)


def smooth_max_obj(h, alpha):
    c = compute_correlations(h)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(c - m))) / alpha


def slsqp_optimize(h0, n, alpha, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_obj(x, alpha)

    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    if np.isfinite(res.fun):
        h_opt = fix_sum(res.x, n)
        return h_opt, compute_c5(h_opt)
    return None, float('inf')


def interpolate_h(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_dst = (np.arange(n_dst) + 0.5) / n_dst
    h_dst = np.interp(x_dst, x_src, h_src, period=n_src)
    return fix_sum(h_dst, n_dst)


def subgradient_step(h, n, lr, alpha=80):
    c = compute_correlations(h)
    c_shifted = c - np.max(c)
    w = np.exp(alpha * c_shifted)
    w /= w.sum()
    n_pts = len(h)
    grad = np.zeros(n_pts)
    eps = 1e-8
    for j in range(n_pts):
        hp = h.copy()
        hp[j] = min(1.0, h[j] + eps)
        cm = h.copy()
        cm[j] = max(0.0, h[j] - eps)
        fp = np.dot(w, compute_correlations(hp))
        fm = np.dot(w, compute_correlations(cm))
        grad[j] = (fp - fm) / (2.0 * eps)
    h_new = project_feasible(h - lr * grad, n)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = float('inf')

    parent_score = compute_c5(h_parent)

        if n == n_parent and h_parent is not None:

        else:

        cur_h = h0.copy()
        cur_score = compute_c5(cur_h)

        if cur_score < best_score:

            best_n = n

                best_h = h_opt.copy()
                best_n = n
                cur_h = h_opt.copy()

        for attempt in range(8):

            h_pert = cur_h.copy()
            n_pert = max(3, n // 8)
            idx = rng.choice(n, size=n_pert, replace=False)
            h_pert[idx] += rng.normal(0, 0.08, size=n_pert)
            h_pert = project_feasible(h_pert, n)

            if h_opt is not None and sc < best_score:

                sc = compute_c5(cur_h)
                if sc < best_score:
                    best_score = sc
                    best_h = cur_h.copy()
                    best_n = n
                if it % 40 == 39:
                    lr *= 0.7

    return (best_h, best_score, best_n)
