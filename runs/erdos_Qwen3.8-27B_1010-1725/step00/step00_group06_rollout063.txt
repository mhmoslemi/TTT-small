        hc = fix_sum(h)
    return hc, compute_c5(hc)

def grad_desc(h, lr, n_it, n):
    dx = 2.0 / n
    h = h.copy()
    for _ in range(n_it):
        corr = np.correlate(h, 1.0 - h, mode='full')
        js = int(np.argmax(corr))
        s = js - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
        else:
            t = -s
            g[:n - t] += (1.0 - h[t:])
            g[t:] -= h[:n - t]
        g *= dx
        h = project_feasible(h - lr * g, n)
    return h

def get_top_lags(h, n, k):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return np.argsort(corr)[-k:][::-1]

def multi_lag_grad(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top = np.argsort(corr)[-top_k:][::-1]
    g = np.zeros(n)
    for js in top:
        s = js - (n - 1)
        if s >= 0:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
        else:
            t = -s
            g[:n - t] += (1.0 - h[t:])
            g[t:] -= h[:n - t]
    return g * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    global _gn
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(95, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)
    _gn = n

    # Phase 1: SLSQP with multiple alpha values
    for alpha in [300.0, 150.0, 80.0, 40.0]:
        if time.time() - t0 > budget_s * 0.25:
            break
        hc, c5c = slsqp_opt(best_h, n, alpha, 500)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc

    # Phase 2: Multi-lag gradient descent
    if time.time() - t0 < budget_s * 0.40:
        lr = 0.02
        for it in range(40):
            if time.time() - t0 > budget_s * 0.40:
                break
            g = multi_lag_grad(best_h, n, 5)
            ht = project_feasible(best_h - lr * g, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-10:
                best_c5 = c5t
                best_h = ht
                lr = min(lr * 1.1, 0.08)
            else:
                lr *= 0.6

    # Phase 3: SLSQP again to polish
    if time.time() - t0 < budget_s * 0.50:
        hc, c5c = slsqp_opt(best_h, n, 200.0, 400)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc

    # Phase 4: Multi-start exploration
    for trial in range(80):
        if time.time() - t0 > budget_s * 0.78:
            break
        h_try = best_h.copy()
        if trial % 4 == 0:
            idx = rng.choice(n, max(3, n // 10), replace=False)
            h_try[idx] += rng.uniform(-0.08, 0.08, len(idx))
        elif trial % 4 == 1:
            freq = rng.randint(1, n // 3 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.03, 0.12)
            h_try += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif trial % 4 == 2:
            n_swap = rng.randint(2, 5)
            i_idx, j_idx = rng.choice(n, 2 * n_swap, replace=False)
            for a, b in zip(i_idx[::2], j_idx[::2]):
                d = h_try[a] - h_try[b] + rng.uniform(-0.05, 0.05)
                h_try[a] += d
                h_try[b] -= d
        else:
            n_perm = rng.randint(2, 6)
            perm = rng.permutation(n)
            h_try = h_try[perm]
        h_try = fix_sum(h_try)
        hc, c5c = slsqp_opt(h_try, n, 100.0, 250)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc

    # Phase 5: Upscale
    if time.time() - t0 < budget_s * 0.88:
        n2 = min(2 * n, 400)
        h_up = np.zeros(n2)
        for i in range(n2):
            pos = i * n / n2
            i0 = int(pos) % n
            i1 = (i0 + 1) % n
            fr = pos - int(pos)
            h_up[i] = (1 - fr) * best_h[i0] + fr * best_h[i1]
        h_up = fix_sum(h_up)
        hc, c5c = slsqp_opt(h_up, n2, 100.0, 500)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc
            n = n2

    # Phase 6: Final polish
    if time.time() - t0 < budget_s - 10:
        n_fin = len(best_h)
        hc, c5c = slsqp_opt(best_h, n_fin, 400.0, 300)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc
        # Tiny Nelder-Mead on top pair swaps
        if time.time() - t0 < budget_s - 5:
            for _ in range(200):
                if time.time() - t0 > budget_s - 5:
                    break
                i, j = rng.choice(n_fin, 2, replace=False)
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    ht = best_h.copy()
                    d = frac * (ht[i] - ht[j])
                    if abs(d) < 1e-12:
                        continue
                    ht[i] += d
                    ht[j] -= d
                    ht = np.clip(ht, 0, 1)
                    ht += (n_fin * 0.5 - np.sum(ht)) / n_fin
                    ht = np.clip(ht, 0, 1)
                    c5t = compute_c5(ht)
                    if c5t < best_c5:
                        best_c5 = c5t
                        best_h = ht

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
