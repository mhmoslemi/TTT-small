def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    scaled = T * corr * dx

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr, dx

def project_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def smooth_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    mx = np.max(scaled)
    exp_s = np.exp(scaled - mx)
    w = exp_s / np.sum(exp_s)
    obj = (mx + np.log(np.sum(exp_s))) / T
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-12:
            continue
        lag = m - (n - 1)

            grad[:n - lag] += wm * (1.0 - h[lag:])
            grad[lag:] += -wm * h[:n - lag]
        else:
            a = -lag
            grad[a:] += wm * (1.0 - h[:n - a])
            grad[:n - a] += -wm * h[a:]
    return obj, grad
