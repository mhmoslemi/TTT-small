We want to minimize max_k (1 - R(k)) = 1 - min_k R(k)

Total energy: ∫ h^2 dx = R(0) (with period 2)
Parseval: sum of squared Fourier coefficients = R(0)

If h has mean 1/2 (since ∫_0^2 h = 1, mean = 1/2):

For R(k) to be uniformly high, we want all Fourier modes to contribute to all lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective(h, temp, n, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    smooth_max = m + temp * np.log(np.sum(np.exp(shifted)))
    sum_pen = penalty * (np.sum(h) - n / 2.0) ** 2
    return smooth_max + sum_pen

def fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_point(h, n, temp, maxiter=3000, ftol=1e-14, penalty=1e6):
    res = minimize(objective, h, args=(temp, n, penalty),
                   method='L-BFGS-B', bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    h_new = fix_sum(res.x, n / 2.0)
    return h_new
