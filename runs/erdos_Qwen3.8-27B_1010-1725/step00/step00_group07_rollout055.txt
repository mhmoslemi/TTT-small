```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h, dx):
    """Compute subgradient of max_k corr[k] * dx at h."""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    cmax = np.max(corr)
    active = corr > cmax - 1e-8
    n_active = max(1, int(np.sum(active)))
    
    grad = np.zeros(n)
    # For each active shift, add the gradient contribution
    # corr[k] = sum_i h[i+k-(n-1)] * (1-h[i])
    # Let shift = k - (n-1)
    # dc/dh[j] = (1-h[j-shift])*I(0<=j-shift<n) - h[j+shift]*I(0<=j+shift<n)
    
    for shift in range(-(n-1), n):
        k = shift + (n - 1)
        if not active[k]:
            continue
        g = np.zeros(n)
        if shift > 0:
            # (1-h[j-shift]) for j=shift..n-1
            g[shift:] += (1.0 - h[:n-shift])
            # -h[j+shift] for j=0..n-1-shift
            g[:n-shift] -= h[shift:]
        elif shift < 0:
            s = -shift
            # (1-h[j+shift]) = (1-h[j-s]) for j=s..n-1
            g[s:] += (1.0 - h[:n-s])
            # -h[j+shift] = -h[j-s] for j=0..n-s-1
            g[:n-s] -= h[s:]
        else:
            # (1-h[j]) - h[j] = 1 - 2h[j]
            g[:] = 1.0 - 2.0 * h
        grad += g
    
    return grad / n_active * dx
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_at(h):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    cmax = corr.max()
    active = np.where(corr >= cmax - 1e-7)[0]
    na = len(active)
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        elif s < 0:
            t = -s
            grad[t:] += (1.0 - h[:n - t])
            grad[:n - t] -= h[t:]
        else:
            grad += (1.0 - 2.0 * h)
    return grad / (na * n)

def local_search(h, max_iters, step_size, rng, tol=1e-10, use_swap=False):
    n = len(h)
    ts = float(n) / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_iters):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        hn = best.copy()
        if use_swap and rng.random() < 0.3:
            hn[i], hn[j] = hn[j], hn[i]
        else:
            d = (rng.random() - 0.5) * 2.0 * step_size
            hn[i] = max(0.0, min(1.0, hn[i] - d))
            hn[j] = max(0.0, min(1.0, hn[j] + d))
            if abs(hn.sum() - ts) > 1e-8:
                hn = project_onto_constraint(hn, ts)
        c5 = compute_c5(hn)
        if c5 < best_c5 - tol:
            best_c5 = c5
            best = hn
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = 0.5 * np.ones(80)

    n = len(h0)
    ts = float(n) / 2.0
    h0 = project_onto_constraint(h0, ts)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient descent (35% budget)
    t_end1 = t0 + budget_s * 0.35
    h = h0.copy()
    lr = 0.5
    for it in range(20000):
        if time.time() > t_end1:
            break
        g = subgradient_at(h)
        h = project_onto_constraint(h - lr * g, ts)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 300 == 0:
            lr *= 0.92
        if it % 800 == 400:
            h = project_onto_constraint(h + 0.06 * rng.standard_normal(n), ts)

    # Phase 2: Targeted local search (30% budget)
    t_end2 = t0 + budget_s * 0.65
    while time.time() < t_end2:
        batch = min(50000, max(1000, int((t_end2 - time.time()) / 0.0002)))
        h2, c52 = local_search(best_h, batch, 0.12, rng)
        if c52 >= best_c5:
            break
        best_c5 = c52
        best_h = h2

    # Phase 3: Multi-start exploration (22% budget)
    t_end3 = t0 + budget_s * 0.87
    attempt = 0
    while time.time() < t_end3:
        attempt += 1
        scale = 0.12 + 0.2 * rng.random()
        if attempt % 4 == 0:
            hp = 0.5 + 0.4 * (2 * rng.random(n) - 1)
        else:
            hp = best_h + scale * rng.standard_normal(n)
        hp = project_onto_constraint(hp, ts)
        for it in range(200):
            if time.time() > t_end3:
                break
            hp = project_onto_constraint(hp - 0.07 * subgradient_at(hp), ts)
        hp, c5p = local_search(hp, 2000, 0.05, rng)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

    # Phase 4: Fine-tuning (10% budget)
    t_end4 = t0 + budget_s * 0.97
    if time.time() < t_end4:
        while time.time() < t_end4:
            batch = min(30000, max(500, int((t_end4 - time.time()) / 0.0002)))
            hf, c5f = local_search(best_h, batch, 0.03, rng, tol=5e-11, use_swap=True)
            if c5f >= best_c5:
                break
            best_c5 = c5f
            best_h = hf

    best_h = np.clip(best_h, 0.0, 1.0)
    # Final sum fix
    best_h = project_onto_constraint(best_h, ts)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
```
