import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        delta = target - s
        if delta > 0:
            mask = h < 1.0
            room = (1.0 - h[mask]).sum()
            if room < 1e-15:
                break
            h[mask] += (1.0 - h[mask]) * (delta / room)
        else:
            mask = h > 0.0
            room = h[mask].sum()
            if room < 1e-15:
                break
            h[mask] -= h[mask] * ((-delta) / room)
    h = np.clip(h, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-14:
        j = int(np.argmin(np.abs(h - 0.5)))
        h[j] += resid
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h0, n, budget_s, rng, T0=0.025, T_min=0.0008):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T0
    start = time.time()
    total_perturb_amp = 0.25
    no_improve_count = 0
    while time.time() - start < budget_s:
        r = rng.random()
        h_new = h.copy()
        if r < 0.5:
            n_pert = rng.randint(2, max(3, n // 7))
            idx = rng.choice(n, size=n_pert, replace=False)
            for i in idx:
                h_new[i] += rng.uniform(-0.35, 0.35) * total_perturb_amp
        elif r < 0.75:
            L = rng.randint(max(4, n // 6), max(5, n // 2))
            s0 = rng.randint(0, n - L)
            freq = rng.randint(1, 4)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.08, 0.25) * total_perturb_amp
            t = np.arange(L) / L * 2.0 * np.pi * freq + phase
            h_new[s0:s0 + L] += amp * np.sin(t)
        else:
            h_new += rng.uniform(-0.04, 0.04, size=n) * total_perturb_amp
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        d = c5_new - cur_c5
        if d < 0 or (T > T_min and np.log(rng.random() + 1e-300) < -d / T):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        T = max(T_min, T * 0.9985)
        if no_improve_count > 500:
            T = max(T, T0 * 0.3)
            no_improve_count = 0
        if no_improve_count > 2000:
            total_perturb_amp = max(0.1, total_perturb_amp * 0.5)
            no_improve_count = 0
    return best_h, best_c5

def nm_optimize(h0, n, budget_s, rng):
    start = time.time()
    x0 = project_feasible(h0.copy(), n)
    c5_start = compute_c5(x0)
    def obj(x):
        xp = project_feasible(x, n)
        return compute_c5(xp)
    max_iter = int(min(300, max(50, (budget_s - (time.time() - start)) * 15)))
    result = minimize(obj, x0, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-5, 'fatol': 1e-7,
                               'adaptive': True})
    if time.time() - start > budget_s:
        return x0, c5_start
    h_res = project_feasible(result.x, n)
    c5_res = compute_c5(h_res)
    if c5_res < c5_start:
        return h_res, c5_res
    return x0, c5_start

def upscale_h(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def make_sinusoid(n, k, phase, amp, rng):
    x = np.arange(n) * 2.0 / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * k * x + phase)
    h = project_feasible(h, n)
    return h
