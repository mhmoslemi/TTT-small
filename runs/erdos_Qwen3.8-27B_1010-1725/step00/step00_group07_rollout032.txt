    dx = 2.0 / n

    return np.max(corr), corr

    for _ in range(200):

            break

        else:
            mask = h > 0.0
        if not np.any(mask):
            break

    diff = target - np.sum(h)

        h[idx] -= diff

    best_h = h.copy()

    max_no_improve = 5000

        step = step_sizes[si]

            h_trial[j] += rng.normal(0, step)

def compute_c5_grad(h):
    """Returns c5 and subgradient."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(corr)
    active = np.argmax(corr)
    l = active - (n - 1)
    grad = np.zeros(n)
    if l > 0:
        grad[:n-l] += (1.0 - h[l:])
        grad[l:] -= h[:n-l]
    elif l < 0:
        s = -l
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    else:
        grad[:] = 1.0 - 2.0 * h
    grad *= dx
    # Project to sum=0 for constraint
    grad -= np.mean(grad)
    return c5, grad
