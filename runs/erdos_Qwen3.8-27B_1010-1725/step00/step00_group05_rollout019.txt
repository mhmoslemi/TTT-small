    """Project h onto {0<=h<=1, sum(h)=n_sum}"""
    h = np.clip(h, 0, 1)
    diff = n_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        # If need to add: add to elements < 1
        # If need to subtract: subtract from elements > 0
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
        # Iterate to fix
        for _ in range(100):
            diff = n_sum - np.sum(h)
            if abs(diff) < 1e-12:
                break
            if diff > 0:

            else:

            if not np.any(mask):

            else:

                h = np.clip(h, 0, 1)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(corr.argmax())

    grad = np.zeros(n)

        i1 = np.arange(k, n)
        grad[i1] += (1.0 - h[i1 - k])
