import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(full) * dx)

def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(300):
        diff = x.sum() - target_sum
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = x > 1e-15
        else:
            mask = x < 1.0 - 1e-15
        if not np.any(mask):
            break
        x[mask] -= diff / np.sum(mask)
        x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max(v, t):
    vmax = np.max(v)
    shifted = np.minimum(t * (v - vmax), 0.0)
    return float(vmax + np.log(np.sum(np.exp(shifted))) / t)

def subgradient(h, k_star, n):
    """Subgradient of full[k_star]*w.r.t. h (then scale by dx)"""
    dx = 2.0 / n
    s = k_star - (n - 1)
    grad = np.zeros(n)

            grad[j] += 1.0 - h[j + s]
        for j in range(s, n):
            grad[j] -= h[j - s]

        for j in range(sp, n):
            grad[j] += 1.0 - h[j - sp]
        for j in range(n - sp):
            grad[j] -= h[j + sp]
    return grad * dx

            res = minimize(compute_c5, x0, method='SLSQP', bounds=bl,
                          constraints=[sum_eq],
                          options={'maxiter': max_iter, 'ftol': 1e-13})
            x = project_feasible(res.x, target_sum)
            return x, compute_c5(x)
        except Exception:
            x = project_feasible(x0, target_sum)
            return x, compute_c5(x)

    # Phase 1: Progressive smooth SLSQP
    for ts in [5, 10, 25, 50, 100, 200, 500]:
        if tl() < 15:
            break
        x, c5 = do_slsqp_smooth(best_h, ts, 250)
        if c5 < best_c5:
            best_c5, best_h = c5, x

    # Phase 2: Exact SLSQP
    if tl() > 20:
        x, c5 = do_slsqp_exact(best_h, 250)
        if c5 < best_c5:
            best_c5, best_h = c5, x

    # Phase 3: Subgradient descent with backtracking
    if tl() > 80:
        h_sg = best_h.copy()
        c5_sg = best_c5
        it = 0
        while tl() > 30 and it < 500000:
            full = np.correlate(h_sg, 1.0 - h_sg, mode='full')
            k_star = int(np.argmax(full))
            g = subgrad_vec(h_sg, k_star, n)
            lr = 0.08 / (1.0 + it * 0.0005)
            h_new = project_feasible(h_sg - lr * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_sg:
                h_sg, c5_sg = h_new, c5_new
                if c5_sg < best_c5:
                    best_c5, best_h = c5_sg, h_sg.copy()
            it += 1

    # Phase 4: Coordinate descent
    if tl() > 50:
        h_cd = best_h.copy()
        c5_cd = best_c5
        stall = 0
        for sweep in range(5000):
            if tl() < 20:
                break
            improved = False
            for j in range(n):
                for d in [0.04, -0.04, 0.015, -0.015, 0.005, -0.005, 0.08, -0.08]:
                    xn = h_cd.copy()
                    xn[j] += d
                    xn = project_feasible(xn, target_sum)
                    c = compute_c5(xn)
                    if c < c5_cd - 1e-11:
                        h_cd, c5_cd = xn, c
                        improved = True
                        break
            if not improved:
                stall += 1
                if stall > 50:
                    break
            else:
                stall = 0
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 5: SLSQP refinement after CD
    if tl() > 20:
        for ts in [100, 300]:
            if tl() < 15:
                break
            x, c5 = do_slsqp_smooth(best_h, ts, 200)
            if c5 < best_c5:
                best_c5, best_h = c5, x
        if tl() > 15:
            x, c5 = do_slsqp_exact(best_h, 200)
            if c5 < best_c5:
                best_c5, best_h = c5, x

    # Phase 6: Random restarts + sine init
    tries = 0
    while tl() > 30 and tries < 50:
        tries += 1
        scale = 0.02 + 0.06 * np.random.rand()
        hp = project_feasible(best_h + np.random.randn(n) * scale, target_sum)
        x, c5 = do_slsqp_smooth(hp, 50, 120)
        if c5 < best_c5:
            best_c5, best_h = c5, x
        if tries % 4 == 0:
            xv = np.linspace(0, 2, n)
            for f in [2, 3, 5]:
                ph = 2.0 * np.pi * np.random.rand()
                hs = 0.5 + 0.2 * np.sin(f * np.pi * xv + ph)
                hs = project_feasible(hs, target_sum)
                x2, c52 = do_slsqp_smooth(hs, 40, 100)
                if c52 < best_c5:
                    best_c5, best_h = c52, x2

    # Final polish
    if tl() > 15:
        x, c5 = do_slsqp_exact(best_h, 150)
        if c5 < best_c5:
            best_c5, best_h = c5, x

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
