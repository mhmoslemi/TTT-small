            c5_2 = compute_c5(h2_try)
            if c5_2 < best_c5:
                best_h = h2_try.copy()
                best_c5 = c5_2
                h2 = h2_try
                step2 = min(step2 * 1.03, 0.15)
            else:
                step2 *= 0.98

    # Final verification
    final_c5 = compute_c5(best_h)
    assert abs(best_h.sum() * (2.0 / len(best_h)) - 1.0) < 1e-6
    return best_h, best_c5, len(best_h)
