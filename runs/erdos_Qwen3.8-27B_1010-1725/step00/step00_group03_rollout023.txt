import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_t = 0.01
_g_lambda = 500.0

def _smooth_obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = (vals - m) / _g_t
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = m + _g_t * np.log(sum_exp)
    obj += _g_lambda * (np.sum(h) - _g_target) ** 2
    return obj

def _smooth_grad(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = (vals - m) / _g_t
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-7:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_vals = np.arange(lo, hi + 1)
        j_vals = i_vals - k + n - 1
        grad[i_vals] += dx * w * (1.0 - h[j_vals])
        grad[j_vals] -= dx * w * h[i_vals]
    grad += 2.0 * _g_lambda * (np.sum(h) - _g_target)
    return grad

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _optimize_start(h_start, t_smooth, maxiter):
    global _g_t, _g_lambda
    _g_t = t_smooth
    _g_lambda = 500.0
    n = _g_n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h_start, jac=_smooth_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-12})
        h_opt = _project_feasible(res.x, _g_target)
        return h_opt
    except Exception:
        return h_start.copy()

def _local_refine(h, rng, steps=30, step_scale=0.03):
    best = h.copy()
    best_c5 = _compute_c5(best)
    n = _g_n
    for s in range(steps):
        scale = step_scale * (0.85 ** (s // 10))
        h_trial = best + rng.normal(0, scale, n)
        h_trial = _project_feasible(h_trial, _g_target)
        c5_t = _compute_c5(h_trial)
        if c5_t < best_c5 - 1e-9:
            best = h_trial.copy()
            best_c5 = c5_t
    return best, best_c5

def _try_binarize(h):
    global _g_target
    h_bin = np.round(h)
    h_bin = np.clip(h_bin, 0, 1)
    diff = _g_target - np.sum(h_bin)
    if diff > 0:
        frac = np.where(h_bin < 1)[0]
        frac = frac[np.argsort(h[frac])]
        h_bin[frac[:int(np.ceil(diff))]] = 1.0
    elif diff < 0:

        frac = frac[np.argsort(-h[frac])]
        h_bin[frac[:int(np.ceil(-diff))]] = 0.0
    h_bin = _project_feasible(h_bin, _g_target)
    return h_bin

    # Time management

    final_c5 = _compute_c5(best_h)
    return (best_h, final_c5, n)
