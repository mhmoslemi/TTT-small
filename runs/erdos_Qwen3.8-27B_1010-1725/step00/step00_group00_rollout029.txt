            grad[j] -= h[j + kk]
    
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj_g(g, n):
    h = g + 0.5
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def constraint_sum(x):
    return float(np.sum(x))


def constraint_sum_jac(x):
    return np.ones(len(x), dtype=float)


def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible(n, rng):
    h = rng.random(n)
    return make_feasible(h, n)


def block_start(n, rng):
    n_blocks = int(rng.integers(3, 10))
    h = np.zeros(n)
    for b in range(n_blocks):
        s = b * n // n_blocks
        e = (b + 1) * n // n_blocks
        h[s:e] = rng.uniform(0.15, 0.85)
    return make_feasible(h, n)


def sinusoidal_start(n, rng):
    freq1 = int(rng.integers(2, 7))
    freq2 = int(rng.integers(2, 5))
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(freq1 * np.pi * x + p1) + 0.15 * np.cos(freq2 * np.pi * x + p2)
    return make_feasible(h, n)


def do_slsqp(h0, n, maxiter):
    g0 = h0 - 0.5
    g0 = np.clip(g0, -0.5, 0.5)
    g0 = g0 - np.sum(g0) / n
    bounds = [(-0.5, 0.5)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_jac}]
    try:
        res = minimize(obj_g, g0, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_try = make_feasible(res.x + 0.5, n)
        c_try = c5_value(h_try)
        if np.isfinite(c_try):
            return h_try, c_try
    except Exception:
        pass
    return None, float('inf')

    t0 = time.time()
