import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    return h


def make_E(h, j, n):
    K = 2 * n - 1
    k = np.arange(K)
    off = n - 1
    idx1 = j + k - off
    v1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - k + off
    v2 = (idx2 >= 0) & (idx2 < n)
    E = np.zeros(K)
    E[v1] += 1.0 - h[idx1[v1]]
    E[v2] -= h[idx2[v2]]
    return E


def solve_1d(D, E, n_grid=72):
    xg = np.linspace(0.0, 1.0, n_grid)
    V = np.max(D[None, :] + xg[:, None] * E[None, :], axis=1)
    bi = int(np.argmin(V))
    xb = xg[bi]
    if 0 < bi < n_grid - 1:
        lo = float(xg[bi - 1])
        hi = float(xg[bi + 1])
        xf = np.linspace(lo, hi, 27)
        Vf = np.max(D[None, :] + xf[:, None] * E[None, :], axis=1)
        xb = xf[int(np.argmin(Vf))]
    return float(xb)


def coord_descent(h, n, max_sweeps=40, tol=1e-10, seed=0, t_deadline=1e18):
    target = n / 2.0
    r = np.random.RandomState(seed)
    h = project_sum(h.astype(float).copy(), target)
    for sw in range(max_sweeps):
        if time.time() > t_deadline:
            break
        order = np.arange(n)
        r.shuffle(order)
        md = 0.0
        for j in order:
            E = make_E(h, j, n)
            corr = np.correlate(h, 1.0 - h, mode="full")
            D = corr - h[j] * E
            xn = solve_1d(D, E)
            md = max(md, abs(xn - h[j]))
            h[j] = xn
        h = project_sum(h, target)
        if md < tol:
            break
    return h, evaluate_c5(h)


def rand_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def bin_local_search(h, n, max_iters=2000, seed=0, t_deadline=1e18):
    r = np.random.RandomState(seed)
    h = h.copy()
    best = evaluate_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for _ in range(max_iters):
        if time.time() > t_deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(r.choice(zeros))
        j = int(r.choice(ones))
        h[i] = 1.0
        h[j] = 0.0
        c5 = evaluate_c5(h)
        if c5 <= best:
            best = c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 0.0
            h[j] = 1.0
    return h, best


def fix_binary(h, n, rng):

    return h
