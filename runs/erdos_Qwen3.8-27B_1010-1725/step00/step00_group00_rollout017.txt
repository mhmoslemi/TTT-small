    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max
    max_val = np.max(corr)
    scaled = alpha * (corr - max_val)
    exp_scaled = np.exp(scaled)
    return max_val + np.log(np.sum(exp_scaled)) / alpha

    dx = 2.0 / n

    best_c5 = evaluate_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, len(best_h))
