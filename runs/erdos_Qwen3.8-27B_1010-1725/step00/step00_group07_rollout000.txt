```
import numpy as np

def repair_sum(h,n):
    h = np.asarray(h, dtype=float).reshape(n).copy()
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.5)
    np.clip(h,0.0,1.0,out=h)
    target=0.5*n
    for _ in range(60):
        diff=target-h.sum()
        if abs(diff)<=1e-13: return h
        if diff>0:
            mask=h<1.0
            if not mask.any(): break
            h[mask]+=diff/mask.sum()
        else:
            mask=h>0.0
            if not mask.any(): break
            h[mask]+=diff/mask.sum()
        np.clip(h,0.0,1.0,out=h)
    diff=target-h.sum()
    if abs(diff)>1e-14:
        if diff>0: i=int(np.argmin(h))
        else: i=int(np.argmax(h))
        h[i]+=diff
        np.clip(h,0.0,1.0,out=h)
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def repair_sum(h, n):
    h = np.asarray(h, dtype=float).reshape(n).copy()
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.5)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) <= 1e-13:
            return h
        if diff > 0.0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / float(mask.sum())
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / float(mask.sum())
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            i = int(np.argmin(h))
        else:
            i = int(np.argmax(h))
        h[i] += diff
        np.clip(h, 0.0, 1.0, out=h)
    return h


def upsample_to(h, n):
    h = np.asarray(h, dtype=float).ravel()
    old = h.size
    if old == 0 or n <= 0:
        return None
    if old == n:
        return repair_sum(h, n)
    x0 = (np.arange(old) + 0.5) * 2.0 / old
    x1 = (np.arange(n) + 0.5) * 2.0 / n
    y = np.interp(x1, x0, h)
    return repair_sum(y, n)


def safe_feasible(h, n):
    if h is None:
        return None
    try:
        arr = np.asarray(h, dtype=float).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    if arr.size == n:
        return repair_sum(arr, n)
    return upsample_to(arr, n)


def eval_c5(h):
    n = h.shape[0]
    if n <= 1:
        return 0.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def cost_key(item):
    return item[0]


def make_candidates(parent, n, rng):
    out = []
    base = safe_feasible(parent, n)
    if base is not None:
        out.append(base)
        for sig in (0.01, 0.02, 0.05):
            out.append(repair_sum(base + sig * rng.normal(size=n), n))

    t = (np.arange(n) + 0.5) / float(n)
    for m in (1, 2, 3):
        for ph in (0.0, 0.25, 0.5, 0.75):
            for amp in (0.5, 0.4):
                out.append(repair_sum(0.5 + amp * np.cos(2.0 * np.pi * m * t + ph * np.pi), n))

    for _ in range(6):
        h = np.zeros(n)
        sel = rng.choice(n, size=n // 2, replace=False)
        h[sel] = 1.0
        out.append(h)

    for _ in range(3):
        out.append(repair_sum(np.clip(0.5 + 0.25 * rng.normal(size=n), 0.0, 1.0), n))

    out.append(repair_sum(1.0 - t, n))
    out.append(repair_sum(t, n))
    return out


def _safe_linprog(c_obj, A_ub, b_ub, A_eq, b_eq, bounds):
    for method in ("highs", "interior-point", "revised simplex"):
        try:
            return linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method=method,
                options={"disp": False},
            )
        except Exception:
            pass
    return None


def gradient_matrix(h, n):
    G = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(-(n - 1), n):
        p = n - 1 + k
        if k >= 0:
            G[p, k:] += 1.0 - h[: n - k]
            G[p, : n - k] -= h[k:]
        else:
            s = -k
            G[p, : n - s] += 1.0 - h[s:]
            G[p, s:] -= h[: n - s]
    return G


def lp_step_delta(h, n):
    h = repair_sum(h, n)
    dx = 2.0 / n
    G = gradient_matrix(h, n) * dx
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.shape[0]

    A_ub = np.empty((m, n + 1), dtype=float)
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = []
    for val in h:
        if val <= 0.0:
            lo, up = 0.0, 1.0
        elif val >= 1.0:
            lo, up = -1.0, 0.0
        else:
            lo, up = -val, 1.0 - val
        bounds.append((lo, up))
    bounds.append((0.0, None))

    res = _safe_linprog(c_obj, A_ub, b_ub, A_eq, b_eq, bounds)
    if res is not None and getattr(res, "success", False) and hasattr(res, "x"):
        delta = np.asarray(res.x[:n], dtype=float)
        return h + delta
    return h.copy()


def lp_step_vfix(h, n):
    h = repair_sum(h, n)
    v = 1.0 - h
    A = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(-(n - 1), n):
        p = n - 1 + k
        if k >= 0:
            A[p, k:] = v[: n - k]
        else:
            s = -k
            A[p, : n - s] = v[s:]
    A *= 2.0 / n
    m = A.shape[0]

    A_ub = np.empty((m, n + 1), dtype=float)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = _safe_linprog(c_obj, A_ub, b_ub, A_eq, b_eq, bounds)
    if res is not None and getattr(res, "success", False) and hasattr(res, "x"):
        return repair_sum(res.x[:n], n)
    return h.copy()


def try_step(h, cand, current_c):
    n = h.size
    delta = cand - h
    best_q = None
    best_q_c = current_c
    for a in (1.0, 0.6, 0.4, 0.25, 0.125, 0.0625, 0.03125):
        if a >= 1.0:

        c = eval_c5(q)
        if c < best_q_c - 1e-13:

            best_q = q
    return best_q, best_q_c


def optimize_lp(h, n, time_limit, rng):
    h = repair_sum(h, n)
    cur_c = eval_c5(h)
    best_h = h.copy()
    best_c = cur_c

    if time_limit <= 0.0:
        return best_h, best_c

    start = time.perf_counter()
    it = 0
    stag = 0

    while timeLimit_remaining(time_limit, start) > 12.0:

        step = try_step(h, cand, cur_c)
        if step is not None:

                best_h = h.copy()

                    cur_c = eval_c5(h)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                stag = 0
            if stag > 35:
                break

    best_h = repair_sum(best_h, n)
    best_c = eval_c5(best_h)
    return best_h, best_c


def timeLimit_remaining(time_limit, start):
    return max(0.0, float(time_limit) - (time.perf_counter() - start))


def polish_pairs(h, n, time_limit, rng):

    cur_c = eval_c5(h)
    if time_limit <= 0.0:
        return h, cur_c

    nd = 3 if n <= 100 else (2 if n <= 200 else 1)

        improved = False

        for ai in range(n - 1):
            i = int(order[ai])
            for aj in range(ai + 1, n):
                j = int(order[aj])
                pair_count += 1

                oi = h[i]
                oj = h[j]
                hi_ = min(1.0 - oi, oj)
                lo_ = -min(oi, 1.0 - oj)
                if hi_ - lo_ <= 1e-12:
                    continue

                deltas = []
                for f in (1.0, 0.5, 0.25)[:nd]:
                    d = hi_ * f
                    if d > 1e-12:
                        deltas.append(d)
                    dm = -lo_ * f
                    if dm > 1e-12:
                        deltas.append(-dm)

                sm = min(small, 0.5 * (hi_ - lo_))
                if sm > 1e-8:
                    deltas.append(sm)
                    deltas.append(-sm)

                accepted = False
                for d in deltas:
                    if d > hi_ + 1e-12 or d < lo_ - 1e-12:
                        continue
                    h[i] = oi + d
                    h[j] = oj - d
                    c2 = eval_c5(h)
                    if c2 < cur_c - 1e-13:
                        cur_c = c2
                        improved = True
                        accepted = True
                        break
                    h[i] = oi
                    h[j] = oj

                if not accepted:
                    h[i] = oi
                    h[j] = oj

                if pair_count % 512 == 0 and timeLimit_remaining(time_limit, start) <= 2.0:
                    break
            if timeLimit_remaining(time_limit, start) <= 2.0:
                break

        stale = 0 if improved else stale + 1

    h = repair_sum(h, n)
    cur_c = eval_c5(h)
    return h, cur_c


def run(seed=42, budget_s=1100, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = initial_h_values
    except NameError:
        parent = None

    if parent is not None:
        parent = np.asarray(parent, dtype=float).ravel()

    if parent is not None and parent.size > 0 and parent.size % 2 == 0 and 8 <= parent.size <= 1000:
        n_base = int(parent.size)
    else:
        n_base = 80

    best_h = None
    best_c = np.inf

    elapsed = time.perf_counter() - start
    phase1 = min(max(0.0, budget - elapsed - 60.0), 450.0)
    phase1 = min(phase1, budget * 0.45)

    cands = make_candidates(parent, n_base, rng)
    scored = []
    for hh in cands:
        if hh is None:
            continue
        scored.append((eval_c5(hh), hh))
    scored.sort(key=cost_key)

    if scored:
        best_c, best_h = scored[0]
        best_h = best_h.copy()

    opt_list = []
    if cands:
        opt_list.append(cands[0])
    for _, hh in scored:
        already = False
        for oh in opt_list:
            if hh is oh:
                already = True
                break
        if not already:
            opt_list.append(hh)
            if len(opt_list) >= 3:
                break

    phase1_start = time.perf_counter()
    oi = 0
    while oi < len(opt_list) and budget - (time.perf_counter() - start) > 70.0:
        rem_phase = phase1 - (time.perf_counter() - phase1_start)
        rem_total = budget - (time.perf_counter() - start) - 25.0
        reserve = 25.0 * (len(opt_list) - oi - 1) + 10.0
        limit = rem_phase - reserve
        if limit > rem_total:
            limit = rem_total
        if limit < 8.0:
            limit = 8.0
        if limit <= 0.0:
            break
        hopt, cval = optimize_lp(opt_list[oi], n_base, limit, rng)
        if cval < best_c:
            best_c = cval
            best_h = hopt.copy()
        oi += 1

    rem = budget - (time.perf_counter() - start) - 60.0
    if rem > 20.0 and best_h is not None:
        hp, cp = polish_pairs(best_h, best_h.size, min(rem * 0.15, 90.0), rng)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    for mult in (2, 4):
        n_next = n_base * mult
        if n_next > 1000:
            break
        rem = budget - (time.perf_counter() - start) - 90.0
        if rem < 200.0:
            break

        cands2 = []
        if best_h is not None:
            hseed = upsample_to(best_h, n_next)
            if hseed is not None:
                cands2.append(hseed)
            L = best_h.size
            if L > 0 and n_next % L == 0:
                cands2.append(repair_sum(np.repeat(best_h, n_next // L), n_next))
        cands2.extend(make_candidates(parent, n_next, rng)[:8])

        scored2 = []
        cnt = 0
        for hh in cands2:
            if hh is None:
                continue
            scored2.append((eval_c5(hh), hh))
            cnt += 1
            if cnt >= 10:
                break
        if not scored2:
            continue
        scored2.sort(key=cost_key)

        sp_start = time.perf_counter()
        phase2 = min(rem * 0.4, 260.0)
        maxseeds = 2 if n_next <= 160 else 1
        si = 0
        while si < maxseeds and si < len(scored2) and budget - (time.perf_counter() - start) > 90.0:
            rem_phase = phase2 - (time.perf_counter() - sp_start)
            rem_total = budget - (time.perf_counter() - start) - 50.0
            limit = min(rem_phase, rem_total)
            if limit < 20.0:
                limit = 20.0
            if limit <= 0.0:
                break
            hopt, cval = optimize_lp(scored2[si][1], n_next, limit, rng)
            if cval < best_c:
                best_c = cval
                best_h = hopt.copy()
            si += 1

        rem = budget - (time.perf_counter() - start) - 60.0
        if rem > 20.0 and best_h is not None and best_h.size == n_next:
            hp, cp = polish_pairs(best_h, n_next, min(rem * 0.1, 40.0), rng)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()

    if best_h is None:
        best_h = repair_sum(np.full(n_base, 0.5, dtype=float), n_base)
        best_c = eval_c5(best_h)

    rem = budget - (time.perf_counter() - start) - 5.0
    if rem > 10.0:
        hp, cp = polish_pairs(best_h, best_h.size, min(rem, 120.0), rng)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    best_h = repair_sum(best_h, best_h.size)
    best_c = eval_c5(best_h)

    return best_h, float(best_c), int(best_h.size)
```
