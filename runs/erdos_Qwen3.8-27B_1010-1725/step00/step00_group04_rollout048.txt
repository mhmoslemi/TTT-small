    x_new = np.arange(n_new, dtype=float) * (n_old / float(n_new))
    return np.interp(x_new, x_old, h)

def do_minimize(h_init, n, beta, lam, maxiter):
    _obj_state['beta'] = beta
    _obj_state['lam'] = lam
    _obj_state['n'] = n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective_fn, h_init.copy(), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'maxls': 40, 'ftol': 1e-14})
        return repair(res.x)
    except Exception:
        return h_init.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_best = initial_h_values.copy().astype(float)
    h_best = repair(h_best)
    c5_best = compute_c5(h_best)

    h_cur = h_best.copy()

    resolutions = [42, 84, 168, 336]

    for n_pts in resolutions:
        if time.time() - t0 > budget_s - 10:
            break

        if n_pts > len(h_cur):
            h_cur = upsample(h_cur, n_pts)
            h_cur = repair(h_cur)

        n = len(h_cur)
        c5_cur = compute_c5(h_cur)
        maxiter = 400 if n <= 84 else (200 if n <= 168 else 100)

        for beta in [25.0, 60.0, 120.0, 250.0]:
            if time.time() - t0 > budget_s - 5:
                break
            h_try = do_minimize(h_cur, n, beta, 1e5, maxiter)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h_cur = h_try
                c5_cur = c5_try

        # Perturbation-based local search
        n_pert = 5 if n <= 84 else 3
        for attempt in range(n_pert):
            if time.time() - t0 > budget_s - 5:
                break
            h_pert = h_cur.copy()
            k = max(2, n // 6)
            idx = rng.choice(n, size=k, replace=False)
            h_pert[idx] += rng.uniform(-0.2, 0.2, size=k)
            h_pert = repair(h_pert)
            for beta in [60.0, 120.0]:
                if time.time() - t0 > budget_s - 5:
                    break
                h_try = do_minimize(h_pert, n, beta, 1e5, maxiter // 2)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur:
                    h_cur = h_try
                    c5_cur = c5_try

        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h_cur.copy()

    # Final polish at best resolution
    if time.time() - t0 < budget_s - 5:
        n = len(h_best)
        for beta in [100.0, 300.0]:
            if time.time() - t0 > budget_s - 3:
                break
            maxiter = 100 if n > 84 else 200
            h_try = do_minimize(h_best, n, beta, 1e6, maxiter)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                h_best = h_try
                c5_best = c5_try

    h_final = repair(h_best)
    c5_final = compute_c5(h_final)
    return (h_final, c5_final, len(h_final))
