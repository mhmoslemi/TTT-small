import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    active = np.where(corr >= max_val - 1e-10)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    grad = np.zeros(n)
    for idx in active:
        lag = int(idx - (n - 1))
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        for j in range(n):
            val = 0.0
            jl = j - lag
            if i_min <= jl <= i_max:
                val += 1.0 - h[jl]
            jpl = j + lag
            if i_min <= j <= i_max and 0 <= jpl <= n - 1:
                val -= h[jpl]
            grad[j] += val
    grad /= float(len(active))
    return grad * dx

def sum_constraint(h):
    n = len(h)
    return float(h.sum() - n / 2.0)

def sum_constraint_jac(h):
    return np.ones(len(h))

def obj_fun(h):
    return compute_c5(h)

def obj_jac(h):
    return compute_gradient(h)

def make_bounds(n):
    return [(0.0, 1.0)] * n

def project_clamp(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run_slsqp(h0, n, max_iter, ftol):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    bnds = make_bounds(n)
    try:
        res = minimize(obj_fun, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, jac=obj_jac,
                      options={'maxiter': max_iter, 'ftol': ftol})
        return np.asarray(res.x, dtype=float)
    except Exception:
        return h0.copy()

def gen_random_h(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return project_clamp(h, n / 2.0)

def gen_binary_h(n, rng):
    perm = rng.permutation(n)
    h = np.zeros(n)
    half = n // 2
    h[perm[:half]] = 1.0
    if n % 2 == 1:
        h[perm[half]] = 0.5
    return h

def gen_sine_h(n, rng):
    x = np.linspace(0.0, 2.0, n)
    freq = rng.uniform(1.0, 6.0)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * x + phase)
    return project_clamp(h, n / 2.0)

def try_binarize(h):
    n = len(h)
    target = n / 2.0
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    whole = int(np.floor(target))
    frac = target - whole
    h_bin[order[:whole]] = 1.0
    if frac > 1e-9:
        h_bin[order[whole]] = frac
    return h_bin

def swap_local_search(h, rng, max_iters, deadline):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        trial = best.copy()
        trial[i], trial[j] = best[j], best[i]
        val = compute_c5(trial)
        if val < best_val - 1e-13:
            best = trial
            best_val = val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    hard_deadline = t0 + budget_s - 5.0

    # Phase 0: Load initial
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP from initial
    if best_h is not None and time.time() < hard_deadline - 800:
        n0 = len(best_h)
        h_opt = run_slsqp(best_h, n0, 500, 1e-14)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 2: Explore various n with diverse starts
    n_candidates = [40, 44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80]
    for n in n_candidates:
        if time.time() > hard_deadline - 250:
            break
        max_starts = 6
        for s in range(max_starts):
            if time.time() > hard_deadline - 250:
                break
            if s % 3 == 0:
                h0 = gen_binary_h(n, rng)
            elif s % 3 == 1:
                h0 = gen_random_h(n, rng)
            else:
                h0 = gen_sine_h(n, rng)
            h_opt = run_slsqp(h0, n, 250, 1e-13)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 3: Perturbation + re-optimize around best
    if best_h is not None and time.time() < hard_deadline - 120:
        n_best = len(best_h)
        for p in range(15):
            if time.time() > hard_deadline - 120:
                break
            h_pert = best_h + rng.normal(0, 0.1, n_best)
            h_pert = project_clamp(h_pert, n_best / 2.0)
            h_opt = run_slsqp(h_pert, n_best, 300, 1e-14)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_h, best_c5 = h_opt, c5

    # Phase 4: Binarize + SLSQP
    if best_h is not None and time.time() < hard_deadline - 80:
        h_bin = try_binarize(best_h)
        h_opt = run_slsqp(h_bin, len(h_bin), 500, 1e-14)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Phase 5: Swap local search
    if best_h is not None and time.time() < hard_deadline - 30:
        dl = min(hard_deadline - 15, time.time() + 120)
        best_h, best_c5 = swap_local_search(best_h, rng, 200000, dl)

    # Phase 6: Final SLSQP polish
    if best_h is not None and time.time() < hard_deadline - 5:
        h_pol = run_slsqp(best_h, len(best_h), 1000, 1e-15)
        c5 = compute_c5(h_pol)
        if c5 < best_c5:
            best_h, best_c5 = h_pol, c5

    # Ensure constraint satisfaction
    if best_h is not None:
        best_h = project_clamp(best_h, len(best_h) / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
