            thi = min(n - 1, n - 1 + s)
            if tlo <= thi:
                grad[tlo:thi + 1] += (1.0 - h[tlo - s:thi - s + 1])
            tlo2 = max(0, -s)
            thi2 = min(n - 1, n - 1 - s)
            if tlo2 <= thi2:
                grad[tlo2:thi2 + 1] -= h[tlo2 + s:thi2 + s + 1]
        grad /= len(active)
        grad *= dx
        lr = lr0 / (1.0 + 0.02 * it)
        improved = False
        for _ in range(30):
            hn = h - lr * grad
            hn = project_simplex(hn, n * 0.5)
            cn = compute_c5(hn, dx)
            if cn < best - 1e-12 or cn < mx * dx - 1e-12:
                h = hn
                best = min(best, cn)
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        n0 = 70
        h0 = np.full(n0, 0.5)

    dx0 = 2.0 / n0
    h0 = project_simplex(h0, n0 * 0.5)
    best_c5 = compute_c5(h0, dx0)
    best_h = h0.copy()
    best_n = n0

    def _time_check(frac=0.5):
        return (time.time() - t0) < budget_s * frac

    # Phase 1: L-BFGS-B with increasing beta
    for beta in [20, 50, 100, 200, 400]:
        if not _time_check(0.45):
            break
        lam = 200.0
        try:
            res = minimize(
                _c5_soft, h0.copy(), args=(dx0, beta, lam),
                jac=_c5_soft_grad, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n0,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-9, 'maxfun': 2000}
            )
            h_new = project_simplex(res.x, n0 * 0.5)
            c5_new = compute_c5(h_new, dx0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h0 = h_new.copy()
        except Exception:
            pass

    # Phase 2: Active-set gradient descent polish
    if _time_check(0.55):
        h_pol, c5_pol = _active_grad_descent(h0, n_iter=500, lr0=0.05)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
            h0 = h_pol.copy()

    # Phase 3: Random restarts
    if _time_check(0.75):
        n_restarts = 30
        for i in range(n_restarts):
            if not _time_check(0.82):
                break
            scale = 0.12 * max(0.1, 1.0 - i / float(n_restarts))
            h_try = best_h + rng.normal(0, scale, size=best_n)
            h_try = project_simplex(h_try, best_n * 0.5)
            try:
                res = minimize(
                    _c5_soft, h_try, args=(2.0 / best_n, 100.0, 200.0),
                    jac=_c5_soft_grad, method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * best_n,
                    options={'maxiter': 150, 'ftol': 1e-11}
                )
                h_new = project_simplex(res.x, best_n * 0.5)
                c5_new = compute_c5(h_new, 2.0 / best_n)
            except Exception:
                h_new, c5_new = _active_grad_descent(h_try, n_iter=100, lr0=0.03)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Try different n
    if _time_check(0.85):
        for n_try in [50, 60, 80, 90, 100, 120]:
            if not _time_check(0.92):
                break
            if n_try == best_n:
                continue
            x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_simplex(h_interp, n_try * 0.5)
            try:
                res = minimize(
                    _c5_soft, h_interp, args=(2.0 / n_try, 100.0, 200.0),
                    jac=_c5_soft_grad, method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n_try,
                    options={'maxiter': 200, 'ftol': 1e-11}
                )
                h_new = project_simplex(res.x, n_try * 0.5)
            except Exception:
                h_new = h_interp
            h_new, c5_new = _active_grad_descent(h_new, n_iter=200, lr0=0.03)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_try

    # Final polish
    best_h = project_simplex(best_h, best_n * 0.5)
    best_h, best_c5 = _active_grad_descent(best_h, n_iter=200, lr0=0.02)
    best_h = project_simplex(best_h, best_n * 0.5)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
