```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    m = np.max(vals)
    smax = m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (vals - m))))
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return smax + penalty

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time

    def in_budget(frac=0.95):
        return elapsed() < budget_s * frac

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_init = 95
        h0 = np.zeros(n_init)
        h0[:n_init // 2] = 1.0
        h0[n_init // 2] = 0.5
    n = len(h0)
    h0 = make_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # SLSQP from initial
    def slsqp_obj(x):
        return compute_c5(x)

    constr = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
               'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    if in_budget(0.08):
        try:
            res = minimize(slsqp_obj, h0, method='SLSQP', bounds=bounds,
                           constraints=constr, options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = make_feasible(res.x, n)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # L-BFGS-B with smooth max annealing
    def lbfgsb_round(h_start, beta_start, beta_end, n_steps, lam):
        h_cur = h_start.copy()
        for step in range(n_steps):
            frac = step / max(n_steps - 1, 1)
            beta = beta_start * (beta_end / beta_start) ** frac
            lam_cur = lam * (1 + frac)
            try:
                res = minimize(lambda x: smooth_max_obj(x, n, beta, lam_cur),
                               h_cur, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 200, 'ftol': 1e-14})
                h_new = make_feasible(res.x, n)
                h_cur = h_new
            except Exception:
                break
        return h_cur

    if in_budget(0.45):
        # Anneal from best
        h_ann = lbfgsb_round(best_h, 10.0, 200.0, 8, 5.0)
        c = compute_c5(h_ann)
        if c < best_c5:
            best_h, best_c5 = h_ann, c

        # Anneal from initial
        h_ann2 = lbfgsb_round(h0, 5.0, 150.0, 5, 10.0)
        c = compute_c5(h_ann2)
        if c < best_c5:
            best_h, best_c5 = h_ann2, c

    # Parametric initializations + SLSQP
    idx = np.arange(n)
    candidates = []
    for f in [1, 2, 3, 4, 5, 7, 10, 14, 20, 30, 45]:
        for amp in [0.15, 0.25, 0.35, 0.42, 0.48]:
            for phase in [0.0, np.pi / 3, 2.0 * np.pi / 3]:
                h_s = 0.5 + amp * np.cos(2.0 * np.pi * f * idx / n + phase)
                candidates.append(h_s)
    for block in [3, 4, 5, 7, 9, 13, 17, 23, 31, 47]:
        h_b = np.zeros(n)
        step = 2 * block
        for i in range(0, n, step):
            h_b[i:min(i + block, n)] = 1.0
        candidates.append(h_b)
    # Ramp
    candidates.append(np.clip(idx / (n - 1), 0, 1))
    candidates.append(np.clip(1.0 - idx / (n - 1), 0, 1))
    # Binary-like with spread
    for perm_seed in [1, 2, 3, 4, 5]:
        r2 = np.random.default_rng(perm_seed)
        h_bin = r2.permutation(np.sum(h0) - h0).astype(float)
        h_bin[h_bin > 0] = 1.0
        h_bin[h_bin < 0] = 0.0
        candidates.append(h_bin)

    for h_cand in candidates:
        if not in_budget(0.72):
            break
        h_cand = make_feasible(h_cand, n)
        c_init = compute_c5(h_cand)
        if c_init > best_c5 + 0.15:
            continue
        try:
            res = minimize(slsqp_obj, h_cand, method='SLSQP', bounds=bounds,
                           constraints=constr, options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = make_feasible(res.x, n)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Perturbation + SLSQP from best
    for trial in range(25):
        if not in_budget(0.88):
            break
        scale = 0.03 + 0.12 * (trial / 25.0)
        h_trial = best_h + rng.normal(0, scale, n)
        h_trial = make_feasible(h_trial, n)
        try:
            res = minimize(slsqp_obj, h_trial, method='SLSQP', bounds=bounds,
                           constraints=constr, options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = make_feasible(res.x, n)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Final smooth optimization pass
    if in_budget(0.93):
        h_final = lbfgsb_round(best_h, 50.0, 500.0, 6, 20.0)
        c = compute_c5(h_final)
        if c < best_c5:
            best_h, best_c5 = h_final, c

    # Coordinate gradient descent fine-tune
    if in_budget(0.96):
        lr = 0.002
        for step in range(500):
            h = best_h.copy()
            dx_val = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode='full')
            VALS = corr * dx_val
            k_star = np.argmax(VALS)
            lag = k_star - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                for i in range(n - lag):
                    hi = h[i]
                    hi_lag = h[i + lag]
                    grad[i] += hi_lag - hi_lag * hi
                    grad[i + lag] += (1.0 - hi) - hi * (1.0 - hi)
            else:
                lag_abs = -lag
                for i in range(lag_abs, n):
                    hi = h[i]
                    hi_neg = h[i - lag_abs]
                    grad[i] += (1.0 - hi_neg) - hi * (1.0 - hi_neg)
                    grad[i - lag_abs] += hi - hi_neg * hi
            h -= lr * grad
            h = make_feasible(h, n)
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h.copy(), c
            else:
                lr *= 0.8

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
```
