from scipy.special import logsumexp
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def smooth_obj_grad(h, dx, eps):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    C = c * dx
    m = logsumexp(C / eps)
    obj = eps * m
    w = np.exp(C / eps - m)
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1] * dx
        if wk < 1e-12:
            continue
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            jr = np.arange(lo, hi)
            grad[jr] += wk * (1.0 - h[jr - k])
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            jr = np.arange(lo, hi)
            grad[jr] -= wk * h[jr + k]
    return obj, grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_descent(h_init, dx, target_sum, eps_start, max_time, rng):
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    eps = eps_start
    lr = 0.5
    t0 = time.time()
    n = len(h)
    it = 0
    while time.time() - t0 < max_time:
        obj, grad = smooth_obj_grad(h, dx, eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = lr / (gnorm * np.sqrt(n))
        step = min(step, 0.5 / (n ** 0.5))
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.1, 2.0)
        else:
            lr = max(lr * 0.85, 0.01)
            h_new = h.copy()
        h = h_new
        eps = max(eps * 0.9995, 0.003)
        it += 1
        if it % 200 == 0:
            # occasional random nudge
            noise = rng.normal(0, 0.02, n)
            h_nudged = project_feasible(h + noise, target_sum)
            if compute_c5(h_nudged, dx) < best_c5:
                best_h = h_nudged.copy()
                best_c5 = compute_c5(best_h, dx)
                h = best_h.copy()
    return best_h, best_c5

def try_scipy_optimize(h_init, dx, target_sum, max_time, rng):
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    eps = 0.05
    penalty = 10.0
    n = len(h)

    def penalized_obj(hv):
        obj, g = smooth_obj_grad(hv, dx, eps)
        s = np.sum(hv) - target_sum
        return obj + penalty * s * s, g + penalty * 2.0 * s

    t0 = time.time()
    for trial in range(5):
        if time.time() - t0 > max_time:
            break
        res = minimize(penalized_obj, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-7})
        h_res = project_feasible(res.x, target_sum)
        c5_res = compute_c5(h_res, dx)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            h = h_res.copy()
        if trial < 4:
            h = project_feasible(h + rng.normal(0, 0.05, n), target_sum)
        eps = max(eps * 0.7, 0.01)
        penalty = min(penalty * 1.5, 100.0)
    return best_h, best_c5

    h1, c5_1 = gradient_descent(best_h, dx_parent, target_parent, 0.1, t_remain * 0.35, rng)

    resolutions = sorted(set([n_parent, max(50, n_parent - 20), n_parent + 30, n_parent + 70, 200]))
    t_phase2 = budget_s * 0.35
    t_p2_start = elapsed()
    for n_trial in resolutions:
        if n_trial == n_parent:
            continue
        if elapsed() - t_p2_start > t_phase2 or elapsed() > budget_s * 0.75:
            break
        dx_t = 2.0 / n_trial
        target_t = n_trial / 2.0
        # Interpolate best solution to new resolution
        x_old = np.arange(n_parent) * (2.0 / n_parent)
        x_new = np.arange(n_trial) * dx_t
        h_ext = np.concatenate([best_h, best_h[:3]])
        h_interp = np.interp(x_new, x_old, h_ext)
        h_t = project_feasible(h_interp, target_t)
        # Also try binary rounding
        h_bin = (h_interp >= 0.5).astype(float)
        # Fix sum for binary
        idx = np.argsort(h_interp)[::-1]
        h_bin2 = np.zeros(n_trial)
        h_bin2[idx[:int(target_t)]] = 1.0
        h_t_bin = h_bin2.copy()
        c5_bin = compute_c5(h_t_bin, dx_t)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_t_bin.copy()

        # GD on interpolated
        t_alloc = min(t_phase2 - (elapsed() - t_p2_start), (budget_s * 0.75 - elapsed()) * 0.4)
        if t_alloc > 5:
            h_r, c5_r = gradient_descent(h_t, dx_t, target_t, 0.08, t_alloc, rng)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r
        # GD on binary
        if t_alloc > 10:
            h_rb, c5_rb = gradient_descent(h_t_bin, dx_t, target_t, 0.05, t_alloc * 0.5, rng)
            if c5_rb < best_c5:
                best_c5, best_h = c5_rb, h_rb

    # Phase 3: SciPy L-BFGS-B refinement (20% time)
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    t_scipy = budget_s * 0.20
    h3, c5_3 = try_scipy_optimize(best_h, dx_best, n_best / 2.0, min(t_scipy, budget_s - elapsed() - 10), rng)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    # Phase 4: Final fine-tuning (remaining time)
    n_fin = len(best_h)
    dx_fin = 2.0 / n_fin
    t4_start = elapsed()
    while elapsed() < budget_s - 3:
        h_f, c5_f = gradient_descent(best_h, dx_fin, n_fin / 2.0, 0.01, 15, rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        else:
            break
        # try binary rounding of best
        h_br = (best_h >= 0.5).astype(float)
        idx_b = np.argsort(best_h)[::-1]
        h_br2 = np.zeros(n_fin)
        h_br2[idx_b[:int(n_fin / 2.0)]] = 1.0
        c5_br = compute_c5(h_br2, dx_fin)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br2.copy()
            h_br3, c5_br3 = gradient_descent(h_br2, dx_fin, n_fin / 2.0, 0.03, 10, rng)
            if c5_br3 < best_c5:
                best_c5, best_h = c5_br3, h_br3

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
