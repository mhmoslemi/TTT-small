where the sum is over valid i (both indices in [0, n-1]).

So for shift k = m - (n-1), we sum over i where both i and i+k are in [0, n-1].
For k > 0: i ranges from 0 to n-1-k, giving n-k terms.
For k < 0: i ranges from -k to n-1, giving n+k terms.
For k = 0: n terms.

This is correct for the integral ∫₀² h(x)(1-h(x+k))dx when the function is 0 outside [0,2].

OK so the structure is: we have a function on [0,2] that's 0 outside, and we look at all shifts. The "effective" shift range is from -2 to 2 (i.e., -n to n in discrete terms, but really -(n-1) to n-1).

import numpy as np
from scipy import optimize
import time


def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def fix_sum_binary(h, target):
    h = np.clip(h, 0, 1)
    h = (h >= 0.5).astype(float)
    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for idx in np.random.choice(zeros, int(diff), replace=False):
            h[idx] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for idx in np.random.choice(ones, int(-diff), replace=False):
            h[idx] = 0.0
    return h


def sa_search(h_start, dx, target_sum, t_max, rng):
    h_cur = fix_sum_binary(h_start.copy(), target_sum)
    c5_cur = c5_of_h(h_cur, dx)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    T = 0.015
    T_min = 0.00003
    cooling = 0.9997
    stag = 0
    max_stag = 2000
    while time.time() < t_max:
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_cur[i], h_cur[j] = 0.0, 1.0
        new_c5 = c5_of_h(h_cur, dx)
        delta = new_c5 - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
                stag = 0
            else:
                stag += 1
        else:
            h_cur[i], h_cur[j] = 1.0, 0.0
            stag += 1
        T = max(T * cooling, T_min)
        if stag > max_stag:
            T = min(T * 2.5, 0.04)
            stag = 0
    return best_h, best_c5


def two_opt_search(h_start, dx, target_sum, t_max, rng):
    h_cur = fix_sum_binary(h_start.copy(), target_sum)
    c5_cur = c5_of_h(h_cur, dx)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    T = 0.008
    T_min = 0.00003
    cooling = 0.9999
    stag = 0
    while time.time() < t_max:
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(ones, 2, replace=False)
        j1, j2 = rng.choice(zeros, 2, replace=False)
        h_cur[i1] = 0.0; h_cur[i2] = 0.0
        h_cur[j1] = 1.0; h_cur[j2] = 1.0
        new_c5 = c5_of_h(h_cur, dx)
        delta = new_c5 - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
                stag = 0
            else:
                stag += 1
        else:
            h_cur[i1] = 1.0; h_cur[i2] = 1.0
            h_cur[j1] = 0.0; h_cur[j2] = 0.0
            stag += 1
        T = max(T * cooling, T_min)
        if stag > 3000:
            T = min(T * 2.5, 0.03)
            stag = 0
    return best_h, best_c5


def try_n_value(h_source, n_src, n_new, t_max, rng):
    dx_new = 2.0 / n_new
    target_new = n_new // 2
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_source[:n_src], period=2.0)
    h_bin = fix_sum_binary(h_interp, target_new)
    c5_cur = c5_of_h(h_bin, dx_new)
    best_h, best_c5 = sa_search(h_bin, dx_new, target_new, t_max, rng)
    return best_h, best_c5, dx_new


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()
    elapsed = lambda: time.time() - t_start

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 // 2

    best_h = fix_sum_binary(h0, target0)
    best_c5 = c5_of_h(best_h, dx0)
    best_n = n0

    # Phase 1: SA on initial n
    t1 = t_start + budget_s * 0.35
    h_sa, c5_sa = sa_search(h0, dx0, target0, t1, rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa
        best_n = n0

    # Phase 2: 2-opt
    t2 = t_start + budget_s * 0.55
    h_2o, c5_2o = two_opt_search(best_h, 2.0 / best_n, best_n // 2, t2, rng)
    if c5_2o < best_c5:
        best_c5 = c5_2o
        best_h = h_2o

    # Phase 3: Try other n values
    t3 = t_start + budget_s * 0.82
    n_candidates = [22, 26, 34, 38, 42, 46, 52, 58, 64, 74, 82, 94]
    n_list = [n for n in n_candidates if n != best_n and n % 2 == 0]
    time_per_n = max(1.0, (t3 - elapsed()) / len(n_list)) if n_list else 1.0
    for n_try in sorted(n_list, key=lambda x: abs(x - best_n)):
        if elapsed() > t3 - 2:
            break
        t_n = min(t3 - 1, elapsed() + time_per_n)
        h_n, c5_n, dx_n = try_n_value(best_h, best_n, n_try, t_n, rng)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n
            best_n = n_try

    # Phase 4: Refine best with another SA pass
    t4 = t_start + budget_s * 0.92
    dx_best = 2.0 / best_n
    h_ref, c5_ref = sa_search(best_h, dx_best, best_n // 2, t4, rng)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref

    # Phase 5: Continuous refinement via scipy
    if elapsed() < budget_s * 0.95:
        n_c = best_n
        dx_c = 2.0 / n_c
        target_c = n_c / 2.0

        def obj_fn(params):
            h_c = 1.0 / (1.0 + np.exp(-np.clip(params, -30, 30)))
            h_c = h_c / h_c.sum() * target_c
            h_c = np.clip(h_c, 0, 1)
            h_c += (target_c - h_c.sum()) / n_c
            h_c = np.clip(h_c, 0, 1)
            return c5_of_h(h_c, dx_c)

        h_init = np.clip(best_h, 0.05, 0.95)
        x0 = np.log(h_init / (1.0 - h_init))
        try:
            res = optimize.minimize(obj_fn, x0, method='L-BFGS-B',
                                    options={'maxiter': 200, 'ftol': 1e-10})
            h_opt = 1.0 / (1.0 + np.exp(-np.clip(res.x, -30, 30)))
            h_opt = h_opt / h_opt.sum() * target_c
            h_opt = np.clip(h_opt, 0, 1)
            h_opt += (target_c - h_opt.sum()) / n_c
            h_opt = np.clip(h_opt, 0, 1)
            c5_opt = c5_of_h(h_opt, dx_c)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except Exception:
            pass

    # Final: ensure valid output
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    best_h = np.clip(best_h, 0, 1)
    # Enforce sum exactly
    best_h += (target_final - best_h.sum()) / n_final
    best_h = np.clip(best_h, 0, 1)
    # If still off, use binary projection
    if abs(best_h.sum() - target_final) > 1e-9:
        best_h = fix_sum_binary(best_h, int(target_final))
    final_c5 = c5_of_h(best_h, dx_final)
    best_n = len(best_h)

    return (best_h, final_c5, best_n)
