import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corr) / beta * dx)

def _smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k_idx in range(2 * n - 1):
            kag = k_idx - (n - 1)
            i1 = j - kag
            i2 = j + kag
            t1 = (1.0 - h[i1]) if 0 <= i1 < n else 0.0
            t2 = -h[i2] if 0 <= i2 < n else 0.0
            s += w[k_idx] * (t1 + t2)
        grad[j] = s * dx
    return grad

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    step = 0.01
    for _ in range(5000):
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                adj = min(diff, 1.0 - h[i], step)
                h[i] += adj
                diff -= adj
            elif diff < 0 and h[i] > 0.0:
                adj = min(-diff, h[i], step)
                h[i] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                return h
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _coordinate_descent(h, max_rounds=100, time_limit=60.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    t0 = time.time()
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            old = best[j]
            for delta in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1]:
                test = best.copy()
                test[j] = np.clip(old + delta, 0.0, 1.0)
                sd = target - test.sum()
                if abs(sd) < 1e-12:
                    pass
                else:
                    for j2 in range((j + 1) % n, (j + 1 + n) % n, 1):
                        idx = (j + 1 + j2) % n
                        if idx != j:
                            test[idx] = np.clip(test[idx] + sd, 0.0, 1.0)
                            if abs(target - test.sum()) < 1e-10:
                                break
                    if abs(target - test.sum()) > 1e-8:
                        continue
                c5_test = compute_c5(test)
                if c5_test < best_c5 - 1e-10:
                    best = test
                    best_c5 = c5_test
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def _resample(h, new_n):
    n = len(h)
    x_old = np.linspace(0, 1, n, endpoint=False) + 0.5 / n
    x_new = np.linspace(0, 1, new_n, endpoint=False) + 0.5 / new_n
    return np.interp(x_new, x_old, h, period=1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(95, 0.5)

    n = h_init.size
    target = n / 2.0
    h_init = _fix_sum(h_init, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def _time_left():
        return budget_s - (time.time() - t_start) - 30

    # SLSQP optimization with smooth objective
    n_slsqp = 0
    for attempt in range(40):
        if _time_left() < 30:
            break
        n_slsqp += 1
        if attempt == 0:
            h_start = best_h.copy()
        elif attempt <= 5:
            h_start = best_h + rng.normal(0, 0.03, n)
            h_start = _fix_sum(h_start, target)
        elif attempt <= 15:
            h_start = best_h + rng.normal(0, 0.08, n)
            h_start = _fix_sum(h_start, target)
        elif attempt <= 25:
            h_start = best_h + rng.uniform(-0.1, 0.1, n)
            h_start = _fix_sum(h_start, target)
        else:
            # Random binary-ish starts
            h_start = (rng.random(n) < 0.5).astype(float)
            h_start[rng.integers(0, n)] = 0.5  # make sum = n/2
            h_start = _fix_sum(h_start, target)

        beta = 40.0 + 10.0 * (attempt % 5)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(
                lambda x: _smooth_obj(x, beta),
                h_start,
                jac=lambda x: _smooth_obj_grad(x, beta),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12}
            )
            h_new = _fix_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                if attempt % 5 == 0:
                    print(f"SLSQP {attempt}: C5={best_c5:.6f}")
        except Exception:
            pass

    # Coordinate descent refinement
    if _time_left() > 60:
        best_h, best_c5 = _coordinate_descent(best_h, max_rounds=200, time_limit=min(120, _time_left() - 30))
        print(f"After coord descent: C5={best_c5:.6f}")

    # Try a different n (even) by resampling
    for new_n in [96, 100, 120]:
        if _time_left() < 60:
            break
        h_res = _resample(best_h, new_n)
        h_res = _fix_sum(h_res, new_n / 2.0)
        c5_res = compute_c5(h_res)
        # Quick SLSQP on resampled
        n2 = new_n
        cons2 = [{'type': 'eq', 'fun': lambda x, nn=n2: np.sum(x) - nn / 2.0,
                  'jac': lambda x, nn=n2: np.ones(nn)}]
        bounds2 = [(0.0, 1.0)] * n2
        try:
            res2 = minimize(
                lambda x: _smooth_obj(x, 60.0),
                h_res,
                jac=lambda x: _smooth_obj_grad(x, 60.0),
                method='SLSQP',
                bounds=bounds2,
                constraints=cons2,
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            h_new2 = _fix_sum(res2.x, new_n / 2.0)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < best_c5:
                best_h = h_new2
                best_c5 = c5_new2
                n = new_n
                print(f"Resampled n={new_n}: C5={best_c5:.6f}")
        except Exception:
            pass

    # Final coordinate descent
    if _time_left() > 20:
        best_h, best_c5 = _coordinate_descent(best_h, max_rounds=100, time_limit=min(80, _time_left() - 10))

    best_c5 = compute_c5(best_h)
    n_final = len(best_h)
    print(f"Final: n={n_final}, C5={best_c5:.6f}")
    return best_h, best_c5, n_final
