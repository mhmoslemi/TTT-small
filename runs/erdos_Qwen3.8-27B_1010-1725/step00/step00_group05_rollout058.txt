import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c)) * dx

def project_to_constraint(h, target_sum):
    """Project h to {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0, 1)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1 - 1e-15)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    # Get initial
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Ensure feasibility
    h0 = project_to_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP with smooth max
    bounds_list = [(0.0, 1.0)] * n
    
    def make_obj(tau):
        def obj(x):
            c = np.correlate(x, 1-x, mode='full')
            cmax = np.max(c)
            cs = (c - cmax) / tau
            s = np.sum(np.exp(cs))
            return (cmax + tau * np.log(s)) * dx
        return obj
    
    def sum_con(x):
        return np.sum(x) - target_sum
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    
    for tau in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - t0 > budget_s - 100:
            break
        for trial in range(2):
            if time.time() - t0 > budget_s - 100:
                break
            h_curr = best_h.copy()
            if trial == 1 and rng.random() < 0.5:
                h_curr = h_curr + rng.normal(0, 0.02, n)
                h_curr = project_to_constraint(h_curr, target_sum)
            
            try:
                res = minimize(make_obj(tau), h_curr, method='SLSQP', bounds=bounds_list,
                              constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
                if hasattr(res, 'x') and res.x is not None:
                    h_try = project_to_constraint(res.x, target_sum)
                    c5 = compute_c5(h_try)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_try.copy()
            except:
                pass
    
    # Phase 2: Coordinate descent with pair updates
    if time.time() - t0 < budget_s - 20:
        time_left = min(200, budget_s - time.time() - 10)
        h_cd = best_h.copy()
        h_cd, c5_cd = coord_descent(h_cd, n, target_sum, dx, time_left, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    return best_h, best_c5, n

def coord_descent(h, n, target_sum, dx, time_budget, rng):
    t0 = time.time()
    best_c5 = compute_c5(h)
    
    for sweep in range(10000):
        if time.time() - t0 > time_budget:
            break
        improved = False
        for j in range(n):
            k = (j + n // 2) % n
            if k == j:
                k = (j + 1) % n
            
            c = np.correlate(h, 1-h, mode='full')
            
            # Determine valid range of t
            t_lo = max(-h[j], h[k] - 1.0)
            t_hi = min(1.0 - h[j], h[k])
            if t_lo >= t_hi - 1e-12:
                continue
            
            # Objective: max_m c_new[m] * dx where c_new = correlate(h', 1-h', 'full')
            # h'[j] = h[j]+t, h'[k] = h[k]-t
            
            def obj_1d(t):
                h[j] += t
                h[k] -= t
                c_new = np.correlate(h, 1-h, mode='full')
                val = np.max(c_new) * dx
                h[j] -= t
                h[k] += t
                return val
