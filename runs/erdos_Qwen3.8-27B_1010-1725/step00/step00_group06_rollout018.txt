def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0, 1)
    residual = target_sum - h.sum()
    if abs(residual) < 1e-10:
        return h
    n = len(h)
    if residual > 0:
        # Need to increase sum
        mask = h < 1.0
        count = mask.sum()
        if count == 0:
            return h
        room = (1.0 - h[mask])
        total_room = room.sum()
        if total_room < residual:
            h[mask] = 1.0
            return h
        # Scale up proportionally
        h[mask] += residual * room / total_room
    else:
        # Need to decrease sum
        mask = h > 0.0
        count = mask.sum()
        if count == 0:
            return h
        room = h[mask]
        total_room = room.sum()
        if total_room < -residual:
            h[mask] = 0.0
            return h
        h[mask] += residual * room / total_room  # residual is negative
