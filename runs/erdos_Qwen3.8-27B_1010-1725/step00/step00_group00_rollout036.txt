            h_opt = slsqp_local(h_pert, best_n, maxiter=400)
            c5v = compute_c5(h_opt)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_opt

    # Final validation
    n = len(best_h)
    target = n / 2.0
    s = best_h.sum()
    if s > 0:
        best_h = best_h * target / s
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
