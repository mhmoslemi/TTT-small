    sm = np.max(s)
    return cmax + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def project_feasible(h, n):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x, alpha, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(corr, alpha)

def obj_exact(x, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def constraint_func(x, n):
    return np.sum(x) - n / 2.0

def upsample_func(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def slsqp_solve(h, n, use_smooth, alpha, maxiter):
    if use_smooth:
        res = minimize(obj_smooth, h, args=(alpha, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_func, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
    else:
        res = minimize(obj_exact, h, args=(n,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_func, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def local_perturb(h, n, rng, n_rounds):
    best = h.copy()
    best_c5 = compute_c5(best)
    target = n / 2.0
    for _ in range(n_rounds):
        i, j = rng.choice(n, 2, replace=False)
        eps = rng.uniform(0.01, 0.2)
        h_try = best.copy()
        h_try[i] = max(0.0, best[i] - eps)
        h_try[j] = min(1.0, best[j] + eps)
        diff = target - h_try.sum()
        if abs(diff) > 1e-10:
            h_try[i] = np.clip(h_try[i] + diff, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_new = compute_c5(h_try)
        if c5_new < best_c5:
            best = h_try
            best_c5 = c5_new
    return best, best_c5

def push_to_boundary(h, n):
    """Try pushing interior values to 0 or 1 to improve C5."""
    best = h.copy()
    best_c5 = compute_c5(best)
    target = n / 2.0
    for i in range(n):
        if 0.05 < best[i] < 0.95:
            for val in [0.0, 1.0]:
                h_try = best.copy()
                h_try[i] = val
                diff = target - h_try.sum()
                if abs(diff) > 1e-10:
                    spread = np.where(np.abs(h_try - 0.5) > 0.01, np.ones(n), 0.0)
                    if spread.sum() > 0:
                        h_try = h_try + diff / spread.sum() * spread
                    else:
                        h_try = h_try + diff / n
                    h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_new = compute_c5(h_try)
                if c5_new < best_c5:
                    best = h_try
                    best_c5 = c5_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(42) * 0.5
    n_init = len(h)
    h = project_feasible(h, n_init)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP with increasing smoothness at initial resolution
    for alpha in [5, 15, 50, 150, 400, 1000]:
        if time.time() - t0 > budget_s * 0.22:
            break
        h_new = slsqp_solve(h, n_init, True, alpha, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        h = h_new

    # Phase 2: Exact objective refinement
    for _ in range(4):
        if time.time() - t0 > budget_s * 0.35:
            break
        h_new = slsqp_solve(best_h, n_init, False, 0, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Push to boundary + local perturbation
    if time.time() - t0 < budget_s * 0.50:
        best_h, best_c5 = push_to_boundary(best_h, n_init)
    if time.time() - t0 < budget_s * 0.55:
        best_h, best_c5 = local_perturb(best_h, n_init, rng, 500)

    # Phase 4: Multi-resolution refinement
    n = n_init
    for _ in range(3):
        n *= 2
        if n > 400 or time.time() - t0 > budget_s * 0.72:
            break
        h_s = project_feasible(upsample_func(best_h, n), n)
        for alpha in [20, 100, 300]:
            if time.time() - t0 > budget_s * 0.72:
                break
            h_new = slsqp_solve(h_s, n, True, alpha, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_s = h_new
        if time.time() - t0 < budget_s * 0.72:
            h_new = slsqp_solve(h_s, n, False, 0, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Final local search at best resolution
    n_final = len(best_h)
    if time.time() - t0 < budget_s * 0.88:
        best_h, best_c5 = push_to_boundary(best_h, n_final)
    if time.time() - t0 < budget_s * 0.95:
        best_h, best_c5 = local_perturb(best_h, n_final, rng, 300)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
