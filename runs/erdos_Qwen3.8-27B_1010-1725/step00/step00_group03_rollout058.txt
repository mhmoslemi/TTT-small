def compute_c5(h):

    n = len(values)
    n_half = n // 2
    idx = np.argsort(values)[::-1]

    h[i] = 1.0
    h[j] = 0.0

    h[i] = 0.0
    h[j] = 1.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    l = k_star - (n - 1)
    grad = np.zeros(n)
    if l >= 0:
        idx = np.arange(n - l)
        grad[idx] += (1.0 - h[idx + l])
        grad[idx + l] += -h[idx]
    else:
        ll = -l
        idx = np.arange(ll, n)
        grad[idx] += (1.0 - h[idx - ll])
        grad[idx - ll] += -h[idx]
    grad *= dx
    new_h = h - lr * grad
    return project_sum_unit(new_h, n)

        oj = ones[rng.integers(len(ones))]
        cur[zi] = 1.0
        cur[oj] = 0.0
        new_c5 = compute_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        else:
            cur[zi] = 0.0
            cur[oj] = 1.0
        T *= alpha
    return best, best_c5

def kick(h, rng, n_kicks):
    cur = h.copy()
    for _ in range(n_kicks):
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur <= 0.5)[0]
        cur[zeros[rng.integers(len(zeros))]] = 1.0
        cur[ones[rng.integers(len(ones))]] = 0.0
    return cur

def interp_to_n(h_src, n_new, rng):
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_ext = np.concatenate([h_src, [h_src[0]]])
    h_interp = np.interp(x_new, x_src, h_ext)
    return to_binary(h_interp, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except:
        h_init = np.zeros(70, dtype=np.float64)
        h_init[::2] = 1.0

    h_bin = to_binary(h_init, rng)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Hill climb on initial n
    dl1 = t0 + 80
    for _ in range(20):
        if time.time() > dl1:
            break
        hc, c5c = hill_climb(best_h.copy(), rng, 3000)
        if c5c < best_c5 - 1e-12:
            best_c5 = c5c
            best_h = hc

    # Phase 2: SA on original n
    dl2 = t0 + 300
    n_orig = len(best_h)
    for _ in range(4):
        rem = dl2 - time.time()
        if rem < 5:
            break
        hs, c5s = sa_optimize(best_h.copy(), rng, time.time(), min(rem - 1, 90))
        if c5s < best_c5 - 1e-12:
            best_c5 = c5s
            best_h = hs

    # Phase 3: Try multiple n values
    n_candidates = []
    for n in [56, 84, 98, 112, 126, 140, 168, 196, 210, 252, 280, 336, 392, 448, 504, 560, 630, 700]:
        if n % 2 == 0 and n != n_orig and n < 1000:
            n_candidates.append(n)
    n_candidates.sort(key=lambda x: abs(x - n_orig))

    dl3 = t0 + budget_s - 120
    for n in n_candidates:
        rem = dl3 - time.time()
        if rem < 30:
            break
        h_n = interp_to_n(best_h, n, rng)
        h_n, c5_n = hill_climb(h_n, rng, 4000)
        if time.time() < dl3 - 20:
            h_n2, c5_n2 = sa_optimize(h_n.copy(), rng, time.time(), min(60, dl3 - time.time() - 5))
            if c5_n2 < c5_n:
                h_n, c5_n = h_n2, c5_n2
        if c5_n < best_c5 - 1e-12:
            best_c5 = c5_n
            best_h = h_n

    # Phase 4: Random restarts at current best n
    dl4 = t0 + budget_s - 20
    n_best = len(best_h)
    for _ in range(10):
        rem = dl4 - time.time()
        if rem < 10:
            break
        h_rand = to_binary(rng.random(n_best), rng)
        h_rand, c5r = hill_climb(h_rand, rng, 2000)
        h_rand, c5r = sa_optimize(h_rand, rng, time.time(), min(40, rem - 5))
        if c5r < best_c5 - 1e-12:
            best_c5 = c5r
            best_h = h_rand

    # Phase 5: Kicks + SA from best
    rem = dl4 - time.time()
    if rem > 15:
        for _ in range(3):
            rem2 = dl4 - time.time()
            if rem2 < 10:
                break
            h_k = kick(best_h, rng, max(3, n_best // 5))
            hk, c5k = sa_optimize(h_k, rng, time.time(), min(rem2 - 2, 45))
            if c5k < best_c5 - 1e-12:
                best_c5 = c5k
                best_h = hk

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    n_pts = len(best_h)
    assert abs(np.sum(best_h) - n_pts / 2.0) < 1e-9
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, best_c5, n_pts)
