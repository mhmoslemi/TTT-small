import numpy as np
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def obj_func(x):
    return compute_c5(x)


def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0


def do_optimize(h0, maxiter=300, ftol=1e-12):
    n = len(h0)
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': sum_constraint}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)
