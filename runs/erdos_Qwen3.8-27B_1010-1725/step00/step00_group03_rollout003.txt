import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 50

def _smax_obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    smax = m + np.log(np.sum(np.exp(corr - m))) / _g_alpha
    s = np.sum(x) - 0.5 * _g_n
    return smax * dx + 100.0 * s * s

def _con_eq(x):
    return np.sum(x) - 0.5 * _g_n

def _con_eq_jac(x):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = 0.5 * n
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / n
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h) if diff < 0 else np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        s = target - np.sum(h)
        if abs(s) > 1e-15:
            h += s / n
    return h

        h[i] += rng.randn() * 0.1
