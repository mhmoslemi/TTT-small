import numpy as np
import time

def _eval(h, n):
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode="full")
    return float(cr.max() * dx)

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    m = (h < 1.0 - 1e-14) if d > 0 else (h > 1e-14)
    c = m.sum()
    if c == 0:
        return h
    h[m] += d / c
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad(h, n, top_k=3):
    cr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    kk = min(top_k, cr.size)
    idxs = np.argsort(cr)[-kk:]
    g = np.zeros(n)
    for m in idxs:
        lag = int(m) - (n - 1)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            l = -lag
            g[:n - l] += (1.0 - h[l:])
            g[l:] -= h[:n - l]
    g /= float(kk)
    return g, float(cr.max() * dx)

def _upsample(h, no, nn):
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    return _fix(np.interp(xn, xo, h), nn)

def _gd(h, n, t0, budget, rem=40, top_k=3, lr0=0.005):
    h = h.copy()
    best = _eval(h, n)
    lr = lr0
    stuck = 0
    while time.time() - t0 < budget - rem:
        g, c = _grad(h, n, top_k)
        h_try = _fix(h - lr * g, n)
        c_try = _eval(h_try, n)
        if c_try < best - 1e-13:
            h = h_try
            best = c_try
            stuck = 0
            if stuck == 0 and c_try < best - 1e-9:
                lr = min(lr * 1.1, 0.05)
        else:
            stuck += 1
            if stuck > 30:
                lr *= 0.5
                stuck = 0
            if lr < 1e-9:
                h = _fix(h + 0.02 * np.random.randn(n), n)
                lr = lr0
                stuck = 0
        if stuck > 500:
            h = _fix(h + 0.01 * np.random.randn(n), n)
            lr = lr0 * 0.5
            stuck = 0
    return h, best

def _multi_restart(h, n, t0, budget, rem=60, n_restarts=3):
    best_h = h.copy()
    best_c = _eval(h, n)
    per_restart = (budget - 20 - rem - (time.time() - t0)) / max(n_restarts, 1)
    if per_restart < 5:
        n_restarts = 1
    for r in range(n_restarts):
        if time.time() - t0 > budget - rem:
            break
        if r == 0:
            h_start = h.copy()
        else:
            scale = 0.03 * (1.0 + 0.5 * r)
            h_start = _fix(h + scale * np.random.randn(n), n)
        t_rem = rem + (n_restarts - 1 - r) * per_restart * 0.3
        h_opt, c_opt = _gd(h_start, n, t0, budget, t_rem, top_k=3, lr0=0.005)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c
