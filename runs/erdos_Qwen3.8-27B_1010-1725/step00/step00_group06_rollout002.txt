    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(c)
    val = max_val * dx
    mask = (c >= max_val - 1e-12)
    indices = np.where(mask)[0]
    grad = np.zeros(n)
    j = np.arange(n)
    for m in indices:
        s = m - (n - 1)
        i1 = j - s
        i2 = j + s
        g = np.zeros(n)
        v1 = (i1 >= 0) & (i1 < n)
        g[v1] += (1.0 - h[i1[v1]])
        v2 = (i2 >= 0) & (i2 < n)
        g[v2] -= h[i2[v2]]
        grad += g
    grad /= len(indices)
    return val, dx * grad


def rescale_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if np.abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, n, max_iter=3000, ftol=1e-11):
    h0 = rescale_h(h0, n)
    c50 = compute_c5(h0)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for T in [0.05, 0.01, 0.001, 0.0]:
        if T > 0:
            fn = lambda h: soft_max_obj_grad(h, T)
        else:
            fn = lambda h: hard_obj_grad(h)
        try:
            res = minimize(fn, h0, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': ftol})
            h_new = rescale_h(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c50:
                h0 = h_new
                c50 = c5_new
        except Exception:
            pass
    return h0, c50


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(95, 0.5)
    n_parent = len(h_init)
    h_init = rescale_h(h_init, n_parent)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Direct optimization
    if time_left() > 20:
        h_opt, c5_opt = slsqp_optimize(best_h, n_parent, max_iter=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Multi-start
    n_starts = 35
    for i in range(n_starts):
        if time_left() < 10:
            break
        scale = 0.03 + 0.17 * (i / max(n_starts - 1, 1))
        h_pert = best_h + rng.randn(n_parent) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = rescale_h(h_pert, n_parent)
        h_opt, c5_opt = slsqp_optimize(h_pert, n_parent, max_iter=1500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Try different n
    n_trials = [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4, n_parent + 8, n_parent + 15]
    for n_try in n_trials:
        if time_left() < 30:
            break
        if n_try < 20 or n_try > 500:
            continue
        x_old = np.linspace(0.0, 2.0, n_parent)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = rescale_h(h_interp, n_try)
        h_opt, c5_opt = slsqp_optimize(h_interp, n_try, max_iter=2000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n_parent = n_try
            x_old = x_new

    # Phase 4: Fine-tuning
    for i in range(25):
        if time_left() < 5:
            break
        scale = 0.005 + 0.02 * rng.rand()
        h_pert = best_h + rng.randn(len(best_h)) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = rescale_h(h_pert, len(best_h))
        h_opt, c5_opt = slsqp_optimize(h_pert, len(best_h), max_iter=2000, ftol=1e-12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final verification
    best_h = rescale_h(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
