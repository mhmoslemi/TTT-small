import numpy as np
import math

def score_direct(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_box_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target: lo = mid
        else: hi = mid
    y = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - y)
            for idx in order:
                take = min(diff, 1.0 - y[idx])
                if take <= 0: break
                y[idx] += take; diff -= take
                if diff <= 1e-13: break
        else:
            order = np.argsort(y)
            for idx in order:
                take = min(-diff, y[idx])
                if take <= 0: break
                y[idx] -= take; diff += take
                if diff >= -1e-13: break
    return y

import numpy as np
import math
import time

def score_direct(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def make_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * len(h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                take = min(diff, 1.0 - h[idx])
                if take <= 0:
                    break
                h[idx] += take
                diff -= take
                if diff <= 1e-13:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                take = min(-diff, h[idx])
                if take <= 0:
                    break
                h[idx] -= take
                diff += take
                if diff >= -1e-13:
                    break
    return h

def resample_nearest(template, n):
    m = len(template)
    t = np.asarray(template, dtype=float)
    idx = (np.arange(n, dtype=np.int64) * m) // n
    return t[idx]

def resample_linear(template, n):
    m = len(template)
    t = np.asarray(template, dtype=float)
    if m == n:
        return t.copy()
    return np.interp(2.0 * np.arange(n) / n, 2.0 * np.arange(m) / m, t)

def random_balanced(n, rng):
    h = np.zeros(n, dtype=np.int64)
    t = n // 2
    if t > 0:
        h[rng.permutation(n)[:t]] = 1
    return h

def balanced_threshold(scores, n, rng):
    scores = np.asarray(scores, dtype=float)
    scores = scores + rng.normal(0.0, 1e-9, size=n)
    h = np.zeros(n, dtype=np.int64)
    t = n // 2
    if t > 0 and t < n:
        h[np.argsort(scores)[::-1][:t]] = 1
    return h

def initial_candidates(parent, n, rng):
    c = []
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=float)
        for tmpl in (resample_nearest(p, n), resample_linear(p, n)):

    for _ in range(4):

def add_scored(scored, d, h):
    i = 0
    while i < len(scored):
        if d < scored[i][0]:
            scored.insert(i, (d, h))
            break
        i += 1
    else:
        scored.append((d, h))
    if len(scored) > 10:
        del scored[10:]

def evaluate_candidates(cands, n, rng, max_k=8):
    scored = []
    target = n // 2
    for h in cands:
        h = np.asarray(h, dtype=np.int64).copy()
        if int(h.sum()) != target:
            h = balanced_threshold(h.astype(float), n, rng)
        D = initial_D(h, n)
        add_scored(scored, int(D.max()), h)
    return scored[:max_k]

def initial_D(h, n):
    h = np.asarray(h, dtype=np.int64)
    m = n - 1
    D = np.zeros(2 * n - 1, dtype=np.int64)
    for l in range(1, n):
        D[m + l] = np.dot(h[0:n - l], 1 - h[l:n])
        D[m - l] = np.dot(h[l:n], 1 - h[0:n - l])
    return D

def fill_swap_delta(delta, h, p, q, n):
    m = n - 1
    delta.fill(0)
    if p > 0:
        a = m - p
        delta[a:m] += 1
        delta[a:m] -= h[0:p]
    if p < n - 1:
        b = m + n - p
        delta[m + 1:b] += 1
        delta[m + 1:b] -= h[p + 1:n]
    if p > 0:
        delta[m + 1:m + p + 1] -= h[p - 1::-1]
    if p < n - 1:
        a = m + p - n + 1
        delta[a:m] -= h[n - 1:p:-1]
    if q > 0:
        a = m - q
        delta[a:m] -= 1
        delta[a:m] += h[0:q]
    if q < n - 1:
        b = m + n - q
        delta[m + 1:b] -= 1
        delta[m + 1:b] += h[q + 1:n]
    if q > 0:
        delta[m + 1:m + q + 1] += h[q - 1::-1]
    if q < n - 1:
        a = m + q - n + 1
        delta[a:m] += h[n - 1:q:-1]
    delta[m + q - p] += 1
    delta[m + p - q] += 1

def swap_lists(z, o, bp, bq):
    for i, x in enumerate(z):
        if x == bp:
            z[i] = bq
            break
    for i, x in enumerate(o):
        if x == bq:
            o[i] = bp
            break

def local_search_binary(h0, n, rng, deadline, target_c=0.3808, stop_on_target=True):

    t = n // 2

    D = initial_D(h, n)
    L = 2 * n - 1
    currD = int(D.max())
    bestD = currD
    best_h = h.copy()
    bestC = bestD * (2.0 / n)
    zeros = list(np.nonzero(h == 0)[0])
    ones = list(np.nonzero(h == 1)[0])
    delta = np.zeros(L, dtype=np.int64)
    tmp = np.empty(L, dtype=np.int64)
    targetD = int(math.floor(target_c * n / 2.0 + 1e-9))

    q_per_p = 40

            bq = -1

    best_h = np.asarray(best_h)
    best_h = make_feasible(best_h)
    final_c = score_direct(best_h)
    return best_h, float(final_c), int(len(best_h))
