import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smoothed_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - m))))

def project_box_sum(v, target_sum):
    v = np.asarray(v, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = np.clip(v - tau, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    free = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.sum(free) > 0 and abs(diff) > 1e-12:
        h[free] += diff / float(np.sum(free))
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) / float(n_old)
    x_new = np.arange(n_new, dtype=float) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, float(n_new) / 2.0)

def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0.0, 1.0, n)
    return project_box_sum(h, float(n) / 2.0)

def structured_init(n, seed):
    rng = np.random.RandomState(seed)

    # Perturb with random smooth pattern
    freqs = rng.choice(np.arange(1, n // 3 + 1), size=3, replace=False)
    phases = rng.uniform(0, 2 * np.pi, 3)
    amps = rng.uniform(0.1, 0.3, 3)
    x = np.arange(n, dtype=float) / float(n)
    for f, p, a in zip(freqs, phases, amps):
        h += a * np.sin(2 * np.pi * f * x + p)
    return project_box_sum(h, float(n) / 2.0)

def optimize_resolution(h_start, alphas, max_iter, t0, budget_s):
    n = len(h_start)
    h_curr = h_start.copy()
    target_sum = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if (time.time() - t0) > budget_s * 0.88:
            break
        try:
            res = minimize(smoothed_obj, h_curr, args=(alpha,),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-13,
                                   'gtol': 1e-10, 'maxls': 50})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_box_sum(res.x, target_sum)
                if compute_c5(h_new) < compute_c5(h_curr):
                    h_curr = h_new
        except Exception:
            pass
    return h_curr
