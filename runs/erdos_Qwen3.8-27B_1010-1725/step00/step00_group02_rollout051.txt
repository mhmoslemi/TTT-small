import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Exact Euclidean projection onto {x: 0<=x<=1, sum(x)=n/2} via bisection on tau."""
    target = n / 2.0
    h = h.astype(float).copy()
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def obj_wrapper(x, n):
    return compute_c5(project_feasible(x, n))

def coord_desc(h, n, iters, rng):
    """Local coordinate descent with adaptive steps."""
    best = h.copy()
    best_c5 = compute_c5(best)
    base_steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, -0.05, -0.02, -0.01, -0.005, -0.002, -0.001]
    for _ in range(iters):
        i = rng.randint(n)
        improved = False
        for d in base_steps:
            trial = best.copy()
            trial[i] += d
            trial = project_feasible(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                improved = True
                break
    return best, best_c5

def simulated_annealing(h, n, iters, rng, t_start=0.05, t_end=0.001):
    """SA for escaping local optima."""
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    for it in range(iters):
        t = t_start * (t_end / t_start) ** (it / max(iters, 1))
        i = rng.randint(n)
        cur[i] += rng.randn() * t
        cur = project_feasible(cur, n)
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5 or rng.rand() < np.exp((cur_c5 - new_c5) / max(t, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    h_init = project_feasible(np.asarray(initial_h_values, dtype=float), n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nelder-Mead from initial, then with perturbations
    nm_count = 0
    while remaining() > 30:
        start = best_h.copy()
        res = minimize(obj_wrapper, start, args=(n,), method='Nelder-Mead',
                       options={'maxiter': 4000, 'xatol': 1e-9, 'fatol': 1e-13,
                                'adaptive': True, 'restart': False})
        h_opt = project_feasible(res.x, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        nm_count += 1
        # Perturb for next NM run
        scale = 0.02 / (1.0 + 0.1 * nm_count)
        best_h_for_restart = best_h + rng.randn(n) * scale
        best_h_for_restart = project_feasible(best_h_for_restart, n)

    # Phase 2: Simulated annealing attempts
    while remaining() > 20:
        sa_start = best_h + rng.randn(n) * 0.01
        sa_start = project_feasible(sa_start, n)
        sa_h, sa_c5 = simulated_annealing(sa_start, n, 3000, rng, t_start=0.04, t_end=0.0005)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_h = sa_h.copy()
        # NM polish after SA
        res = minimize(obj_wrapper, best_h.copy(), args=(n,), method='Nelder-Mead',
                       options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-14, 'adaptive': True})
        h_opt = project_feasible(res.x, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:

            best_h = h_opt.copy()

    while remaining() > 5:

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - n / 2.0) < 1e-8
