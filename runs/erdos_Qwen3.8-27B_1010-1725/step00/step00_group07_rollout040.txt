    n = len(h)
    c_max = np.max(c_arr)
    # Active lags: those with c[k] > c_max - epsilon
    active_mask = c_arr > c_max - 0.01
    active_lags = np.where(active_mask)[0]
    
    # For each active lag k, compute gradient
    # k in np.correlate "full" mode: index j corresponds to lag j - (n-1)
    grad = np.zeros(n)
    for k_idx in active_lags:
        lag = k_idx - (n - 1)

            # h[i]*(1-h[i+lag]) if i+lag in range

For the current h and c_arr, the "worst" lag is k* = argmax(c_arr). To reduce c[k*], we want to find i where increasing h[i] or decreasing h[i] would help.

The gradient ∂c[k*]/∂h[i] = (1-h[i+k*]) - h[i-k*] (with 0 padding for out-of-range).

If this gradient is positive, decreasing h[i] reduces c[k*].
If this gradient is negative, increasing h[i] reduces c[k*].

So the best single-element change (maintaining sum) would be:
- Find i with largest positive gradient (decrease h[i])
- Find j with most negative gradient (increase h[j])
- Swap: h[i] -= δ, h[j] += δ

This is a greedy coordinate descent step!

def greedy_step(h, c_arr, deadline):
    n = len(h)
    k_max = np.argmax(c_arr)
    lag = k_max - (n - 1)
    
    # Compute gradient for the active lag
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i + lag < n:
            g += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            g += -h[i - lag]
        grad[i] = g

    # To decrease c[k*], we want to decrease h[i] for positive grad[i]
    # and increase h[j] for negative grad[j]
    
    # But we also need to check effect on OTHER lags
    # For now, just try the most extreme pair
    i = np.argmax(grad)  # most positive gradient
    j = np.argmin(grad)  # most negative gradient
    
    if i == j:
        return None, None
    
    # Determine delta to make h[i] decrease and h[j] increase
    delta_max_i = h[i]  # max we can decrease h[i]
    delta_max_j = 1.0 - h[j]  # max we can increase h[j]
    delta = min(delta_max_i, delta_max_j) * 0.1  # small step

        return None, None
    
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, (i, j, -delta)  # return negative delta since we decreased h[i]
