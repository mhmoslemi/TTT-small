import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n-lag] += (1 - h[lag:])
        grad[lag:] += -h[:n-lag]
    elif lag < 0:
        grad[-lag:] += (1 - h[:n+lag])
        grad[:n+lag] += -h[-lag:]
    else:
        grad += (1 - 2*h)
    return dx * grad

def project(h, S):
    """Project h onto {0<=h<=1, sum(h)=S}"""
    h = np.clip(h, 0, 1)
    diff = S - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Water filling: add diff/n to all, then re-clip
    # Iterate a few times
    for _ in range(50):
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
        new_diff = S - np.sum(h)
        if abs(new_diff) < 1e-10:
            break
        # Only adjust non-saturated coordinates
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        h[free] += new_diff / np.sum(free)
        h = np.clip(h, 0, 1)
    return h

def optimize_h(h0, n, max_iters=5000, lr_init=0.01, budget_check=None, elapsed_start=0):
    S = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iters):
        if budget_check is not None and time.time() - elapsed_start > budget_check:
            break
        if it % 50 == 0 and lr < 0.001:
            break
            
        val, grad = c5_objective_and_grad(h)
        
        # Gradient step
        h_new = h - lr * grad
        # Project
        h_new = project(h_new, S)
        
        # Check improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            # Occasionally increase lr if improving fast
        else:
            lr *= 0.95
            # Revert
            h = h  # keep old
        
        # Adaptive lr
        if it % 100 == 0:
            lr = max(lr * 1.05, 0.0001) if val > best_c5 else lr
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_value_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(corr))
    tol = 1e-9
    grad = np.zeros(n)
    count = 0
    for k in range(2 * n - 1):
        if corr[k] >= max_val - tol:
            lag = k - (n - 1)
            if lag > 0:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] += -h[:n - lag]
            elif lag < 0:
                grad[-lag:] += (1.0 - h[:n + lag])
                grad[:n + lag] += -h[-lag:]
            else:
                grad += (1.0 - 2.0 * h)
            count += 1
    if count > 0:
        grad /= count
    return max_val * dx, dx * grad

def project_feasible(h, S):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = S - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = int(np.sum(free))
        if nf == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_descent(h, S, max_iters, lr_init, time_limit, t_start):
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    h = h.copy()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if patience > 800 and lr < 0.005:
            break
        val, grad = c5_value_and_grad(h)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-14:
            patience += 50
            continue
        h_new = h - lr * grad
        h_new = project_feasible(h_new, S)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1
            if patience % 100 == 0:
                lr *= 0.8
        if it % 300 == 0 and patience > 200:
            lr = max(lr * 1.5, 0.01)
            patience = 0
    return best_h, best_c5

def local_search_binary(h, S, n_steps, rng, time_limit, t_start):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        i = int(rng.integers(len(h)))
        j = int(rng.integers(len(h)))
        if i == j:
            continue
        if best_h[i] > 0.5 and best_h[j] < 0.5:
            best_h[i] -= 1.0
            best_h[j] += 1.0
            best_h = project_feasible(best_h, S)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5:
                best_c5 = c5_try
            else:
                best_h[i] += 1.0
                best_h[j] -= 1.0
                best_h = project_feasible(best_h, S)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def check_time(frac):
        return time.time() - t_start > budget_s * frac

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(44)
        h0[::2] = 1.0

    n = len(h0)
    S = n / 2.0
    h = project_feasible(h0, S)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Gradient descent from initial
    h1, c5_1 = gradient_descent(h, S, 30000, 0.08, budget_s * 0.25, t_start)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binarize and local search
    if not check_time(0.35):

        # Randomized local search

        h_bin = project_feasible(h_bin, S)
        h_bin, c5_bin = local_search_binary(h_bin, S, 80000, rng, budget_s * 0.6, t_start)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Refine continuous solution from best
    if not check_time(0.65):
        h3, c5_3 = gradient_descent(best_h, S, 20000, 0.03, budget_s * 0.8, t_start)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Try different n (even) with interpolation
    if not check_time(0.8):
        n2 = n * 2
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        S2 = n2 / 2.0

        h4, c5_4 = gradient_descent(h_interp, S2, 15000, 0.04, budget_s * 0.92, t_start)
        if c5_4 < best_c5:

            best_h = h4.copy()
            n = n2
            S = S2

    # Phase 5: Final polishing
    if not check_time(0.95):
        h5, c5_5 = gradient_descent(best_h, S, 10000, 0.01, budget_s * 0.98, t_start)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, S)
    final_c5 = compute_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
