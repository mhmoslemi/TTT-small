import numpy as np
from scipy.optimize import minimize
import time

_beta = 100.0
_half_n = 22.5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / _beta

def eq_con(x):
    return float(np.sum(x) - _half_n)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h = h + diff / float(n)
    return np.clip(h, 0.0, 1.0)

def perturb_h(h, rng, scale):
    h2 = h + rng.randn(len(h)) * scale

def local_search_binary(h, n, rng, max_iters=2000):

    h_cur = h.copy()

    target = n / 2.0
    if s > target:
        ones = np.where(h_cur == 1.0)[0]
        np.random.shuffle(ones)
        for idx in ones[:int(s - target)]:
            h_cur[idx] = 0.0
    elif s < target:
        zeros = np.where(h_cur == 0.0)[0]
        np.random.shuffle(zeros)
        for idx in zeros[:int(target - s)]:
            h_cur[idx] = 1.0
    # Handle 0.5 for odd n
    if n % 2 == 1:
        frac = target - np.sum(h_cur)
        if abs(frac - 0.5) < 1e-9:
            z = np.where(h_cur == 0.0)[0]
            if len(z) > 0:
                h_cur[z[0]] = 0.5
        elif abs(frac + 0.5) < 1e-9:
            o = np.where(h_cur == 1.0)[0]
            if len(o) > 0:
                h_cur[o[0]] = 0.5
    
    c5_cur = compute_c5(h_cur)
    ones = list(np.where(h_cur == 1.0)[0])
    zeros = list(np.where(h_cur == 0.0)[0])

        if not ones or not zeros:
            break

        j = rng.choice(zeros)
        h_cur[i], h_cur[j] = 0.0, 1.0
        h_cur[j], h_cur[i] = h_cur[j], h_cur[i]
        # Actually just swap
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_new = compute_c5(h_cur)
        if c5_new < c5_cur:
            c5_cur = c5_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h_cur[j] = 0.0
            h_cur[i] = 1.0
    
    return h_cur, c5_cur
