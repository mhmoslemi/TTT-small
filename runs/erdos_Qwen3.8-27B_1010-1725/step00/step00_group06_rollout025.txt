import numpy as np
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_binary_even(h, rng):
    n = len(h)
    h = np.where(h >= 0.5, 1.0, 0.0)
    target = n // 2
    guard = 0
    while h.sum() != target and guard < 2000:
        if h.sum() > target:
            i = rng.randint(n)
            if h[i] == 1:
                h[i] = 0
        else:
            i = rng.randint(n)
            if h[i] == 0:
                h[i] = 1
        guard += 1
    return h

def pair_swap_anneal(h, rng, max_secs):
    n = len(h)
    h = make_binary_even(h, rng)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    T = 0.004
    while time.time() - t0 < max_secs:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                T = max(T * 0.999, 1e-6)
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T * 0.9998, 1e-7)
    return best_h

def continuous_anneal(h, rng, max_secs):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0, 1)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    T = 0.003
    while time.time() - t0 < max_secs:
        k = rng.randint(1, 4)
        idx = rng.choice(n, size=k, replace=False)
        old = h[idx].copy()
        h[idx] += rng.uniform(-0.15, 0.15, size=k)
        h = np.clip(h, 0, 1)
        diff = h.sum() - target
        if abs(diff) > 1e-10:
            mask = (h > 0.02) & (h < 0.98)
            if mask.any():
                h[mask] -= diff / mask.sum()
                h = np.clip(h, 0, 1)
            diff = h.sum() - target
            if abs(diff) > 1e-8:
                h += diff / n
                h = np.clip(h, 0, 1)
        new_c5 = evaluate_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                T = max(T * 0.999, 1e-7)
        else:
            h[idx] = old
        T = max(T * 0.9997, 1e-7)
    return best_h

    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
