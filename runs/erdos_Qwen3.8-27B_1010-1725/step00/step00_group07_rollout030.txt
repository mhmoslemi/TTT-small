def compute_C5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def make_binary(h, rng, n):
    """Convert to binary with exactly n/2 ones"""
    h = (h > 0.5).astype(float)
    target = n // 2
    while h.sum() > target:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
    while h.sum() < target:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
    return h

def local_search_1swap(h, rng, t_limit, max_iters):
    """Greedy 1-swap local search"""
    n = len(h)
    c5 = compute_C5(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_iters):
        if time.time() > t_limit:
            break
        i = ones[rng.randint(len(ones))]

        h[j] = 1
        c5_new = compute_C5(h)
        if c5_new < c5 - 1e-12:
            c5 = c5_new
            if c5 < best:
                best = c5
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0

import numpy as np
import time

def compute_C5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def make_binary(h, rng, n):
    h = (h > 0.5).astype(float)
    target = n // 2
    while h.sum() > target:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
    while h.sum() < target:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
    return h

def local_search_1swap(h, rng, t_limit, max_iters):
    n = len(h)
    c5 = compute_C5(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_iters):
        if time.time() > t_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5_new = compute_C5(h)
        if c5_new < c5 - 1e-12:
            c5 = c5_new
            if c5 < best:
                best = c5
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best

def local_search_2swap(h, rng, t_limit, max_iters):
    n = len(h)
    c5 = compute_C5(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for _ in range(max_iters):
        if time.time() > t_limit:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue
        h[i1] = 0
        h[i2] = 0
        h[j1] = 1
        h[j2] = 1
        c5_new = compute_C5(h)
        if c5_new < c5 - 1e-12:
            c5 = c5_new
            if c5 < best:
                best = c5
                best_h = h.copy()
            ones.remove(i1)
            ones.remove(i2)
            ones.extend([j1, j2])
            zeros.remove(j1)
            zeros.remove(j2)
            zeros.extend([i1, i2])
        else:
            h[i1] = 1
            h[i2] = 1
            h[j1] = 0
            h[j2] = 0
    return best_h, best

def sa_search(h, rng, t_limit, T0, T1):
    n = len(h)
    c5 = compute_C5(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    it = 0
    while time.time() < t_limit:
        it += 1
        T = T0 * (T1 / T0) ** min(1.0, it / 300000.0)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        c5_new = compute_C5(h)
        d = c5_new - c5
        if d < 0 or (T > 1e-9 and rng.random() < np.exp(-d / T)):
            c5 = c5_new
            if c5 < best:
                best = c5
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best

def continuous_polish(h, rng, t_limit, max_iters):
    n = len(h)
    c5 = compute_C5(h)
    best = c5
    best_h = h.copy()
    for _ in range(max_iters):
        if time.time() > t_limit:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        d = rng.uniform(0.005, 0.05) * rng.choice([-1.0, 1.0])
        old_i, old_j = h[i], h[j]
        h[i] = old_i + d
        h[j] = old_j - d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i], h[j] = old_i, old_j
            continue
        c5_new = compute_C5(h)
        if c5_new < c5 - 1e-10:
            c5 = c5_new
            if c5 < best:
                best = c5
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    return best_h, best
