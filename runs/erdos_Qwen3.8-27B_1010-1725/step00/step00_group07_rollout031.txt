def local_search_2swaps(h, time_limit, rng, n_trials=500):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    start = time.time()
    no_imp = 0

    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        best_gain = 1e-12
        best_move = None

        for _ in range(n_trials):
            i1, i2 = rng.choice(ones, size=2, replace=False)
            j1, j2 = rng.choice(zeros, size=2, replace=False)
            h[i1] = 0; h[i2] = 0
            h[j1] = 1; h[j2] = 1
            c5 = compute_c5(h)
            gain = best_c5 - c5
            h[i1] = 1; h[i2] = 1
            h[j1] = 0; h[j2] = 0
            if gain > best_gain:
                best_gain = gain
                best_move = (i1, i2, j1, j2)

        if best_move is not None:
            i1, i2, j1, j2 = best_move
            h[i1] = 0; h[i2] = 0
            h[j1] = 1; h[j2] = 1
            best_c5 = compute_c5(h)
            h_best = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 15:
                break

    return h_best, best_c5

def upsample(h, factor=2):
    return np.repeat(h, factor)

def fix_sum_binary(h, n, rng):
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    h = (h > 0.5).astype(float)
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in rng.choice(ones, size=n_ones - target, replace=False):
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in rng.choice(zeros, size=target - n_ones, replace=False):
            h[idx] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = np.clip(h0, 0, 1)
    target0 = n0 / 2.0
    diff0 = target0 - h0.sum()
    if abs(diff0) > 1e-10:
        mask = (h0 < 1.0 - 1e-8) if diff0 > 0 else (h0 > 1e-8)
        if mask.sum() > 0:
            h0[mask] += diff0 / mask.sum()
        h0 = np.clip(h0, 0, 1)

    best_c5 = compute_c5(h0)
    h_best = h0.copy()

    # Phase 1: Binary local search at n=80
    h_bin = to_binary(h0, n0, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        h_best = h_bin.copy()

    elapsed = time.time() - t0
    budget1 = min(budget_s * 0.40, budget_s - 30)
    if budget1 > 5:
        h1, c5_1 = local_search_swaps(h_best if len(h_best) == n0 and np.allclose(h_best, np.round(h_best)) else to_binary(h0, n0, rng), budget1, rng)
        if c5_1 < best_c5:
            best_c5 = c5_1
            h_best = h1

    # Phase 2: 2-swap moves at n=80
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 20
    budget2 = min(remaining * 0.25, 150)
    if budget2 > 10 and np.allclose(h_best, np.round(h_best), atol=0.01):
        h2, c5_2 = local_search_2swaps(h_best, budget2, rng, n_trials=800)
        if c5_2 < best_c5:
            best_c5 = c5_2
            h_best = h2

    # Phase 3: Upsample to n=160 and search
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    if remaining > 30:
        h_up = upsample(h_best, 2)
        n_up = len(h_up)
        h_up = fix_sum_binary(h_up, n_up, rng)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            h_best = h_up
            best_c5 = c5_up

        budget3 = remaining * 0.7
        h3, c5_3 = local_search_swaps(h_up, budget3, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            h_best = h3

    # Phase 4: Try upsample to n=320 if time remains
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 5
    if remaining > 60 and len(h_best) < 320:
        h_up2 = upsample(h_best, 320 // len(h_best) if 320 % len(h_best) == 0 else 2)
        n_up2 = len(h_up2)
        if 320 % n_up2 == 0:
            pass  # already at 320
        h_up2 = fix_sum_binary(h_up2, n_up2, rng)
        c5_up2 = compute_c5(h_up2)
        if c5_up2 < best_c5:
            h_best = h_up2
            best_c5 = c5_up2
        budget4 = remaining * 0.8
        h4, c5_4 = local_search_swaps(h_up2, budget4, rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            h_best = h4

    # Final: if non-binary might help, try a quick continuous refinement
    elapsed = time.time() - t0
    if remaining > 5 and len(h_best) <= 160:
        n = len(h_best)
        target = n / 2.0
        h_cont = h_best.astype(float).copy()
        # Small perturbations
        for _ in range(200):
            if time.time() - t0 > budget_s - 2:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = rng.uniform(-0.05, 0.05)
            new_hi = h_cont[i] + delta
            new_hj = h_cont[j] - delta
            if 0 <= new_hi <= 1 and 0 <= new_hj <= 1:
                h_cont[i] = new_hi
                h_cont[j] = new_hj
                c5_try = compute_c5(h_cont)
                if c5_try < best_c5 - 1e-10:
                    best_c5 = c5_try
                    h_best = h_cont.copy()
                else:
                    h_cont[i] -= delta
                    h_cont[j] += delta

    # Final verification
    best_c5 = compute_c5(h_best)
    n_final = len(h_best)
    # Ensure sum constraint
    target_f = n_final / 2.0
    actual_sum = h_best.sum()
    if abs(actual_sum - target_f) > 1e-6:
        # Fix by adjusting a few entries
        diff_f = target_f - actual_sum
        if diff_f > 0:
            idxs = np.where(h_best < 1.0)[0]
        else:
            idxs = np.where(h_best > 0.0)[0]
        if len(idxs) > 0:
            h_best[idxs[0]] += diff_f
            h_best = np.clip(h_best, 0, 1)
        best_c5 = compute_c5(h_best)

    return (h_best, best_c5, n_final)
