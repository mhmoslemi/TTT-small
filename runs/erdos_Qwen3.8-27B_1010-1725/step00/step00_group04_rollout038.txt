    lo, hi = h.min() - 1.0, h.max() + 1.0

    """Subgradient using top-k active lags"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Get top-k lags
    idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)

    corr = np.correlate(h, 1.0 - h, mode="full")

        else:
            a = -L
            g[:n-a] += (1.0 - h[a:])
            g[a:] -= h[:n-a]
