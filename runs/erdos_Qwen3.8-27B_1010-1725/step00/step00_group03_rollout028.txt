def pair_line_search(h, a, b):

    # compute O once? In loop we can pass O to save.

def adjust_sum(x, S):
    x = np.array(x, dtype=float, copy=True)
    for _ in range(20):
        d = float(S - x.sum())
        if abs(d) <= 1e-13:
            break
        if d > 0:
            m = x < 1.0 - 1e-12
            if not m.any():
                break
            idx = np.where(m)[0]
            i = idx[int(np.argmax(1.0 - x[idx]))]
            x[i] += min(d, 1.0 - x[i])
        else:
            m = x > 1e-12
            if not m.any():
                break
            idx = np.where(m)[0]
            i = idx[int(np.argmax(x[idx]))]
            x[i] += max(d, -x[i])
        x = np.clip(x, 0.0, 1.0)
    return x


def project_box_sum(v, S):
    v = np.asarray(v, dtype=float)
    x = np.clip(v, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - S) <= 1e-12:
        return adjust_sum(x, S)
    if s < S:
        lo = 0.0
        hi = 1.0
    else:
        lo = -1.0
        hi = 0.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        f = float(np.clip(x + mid, 0.0, 1.0).sum())
        if f < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum(x, S)


def upsample_linear(h, N):
    m = len(h)
    if m == N:
        return h.copy()
    return project_box_sum(
        np.interp(np.linspace(0.0, 1.0, N), np.linspace(0.0, 1.0, m), h),
        0.5 * N
    )


def sine_start(n, rng):
    h = 0.5 * np.ones(n)
    i = np.arange(n)
    for f in (1, 2, 3):
        h += rng.uniform(0.05, 0.30) * np.cos(2.0 * np.pi * f * i / float(n) + rng.uniform(0.0, 2.0 * np.pi))
    return project_box_sum(h, 0.5 * n)


def corr_o(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def c5_value(h):
    return float(corr_o(h).max())


def pick_active(O, p=8):
    k = min(p, len(O))
    idx = np.argpartition(O, -k)[-k:]
    idx = idx[np.argsort(-O[idx])]
    return idx


def sort_by_first(t):
    return t[0]


def add_cand(lst, seen, a, b, h, n):
    a = int(a)
    b = int(b)
    if a < 0 or a >= n or b < 0 or b >= n or a == b:
        return
    hi = min(float(h[a]), float(1.0 - h[b]))
    if hi <= 1e-13:
        return
    key = a * n + b
    if key not in seen:
        seen.add(key)
        lst.append((a, b, hi))


def choose_candidates(h, O, active, n, rng, max_cands=16):
    lst = []
    seen = set()
    for idx in active:
        lag = int(idx) - (n - 1)
        if lag > 0:

                else:
                    tidx = np.argpartition(vals, -2)[-2:]

                    add_cand(lst, seen, t + lag, t, h, n)

def pair_A(h, a, b):
    n = len(h)
    m = 2 * n - 1
    A = np.zeros(m)
    rev = (1.0 - h)[::-1]
    A[a:a + n] -= rev
    A[n - 1 - a:2 * n - 1 - a] += h
    A[b:b + n] += rev
    A[n - 1 - b:2 * n - 1 - b] -= h
    return A


def line_eval(O, A, d1, d2, dx, delta):
    vals = O + delta * A
    q = dx * delta * delta
    vals[d1] += q
    vals[d2] += q
    return float(vals.max())


def golden_pair(O, A, d1, d2, dx, lo, hi):
    if hi - lo <= 1e-12:
        return float(lo)
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = line_eval(O, A, d1, d2, dx, c)
    fd = line_eval(O, A, d1, d2, dx, d)
    for _ in range(20):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = line_eval(O, A, d1, d2, dx, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = line_eval(O, A, d1, d2, dx, d)
    best = 0.5 * (lo + hi)
    vb = line_eval(O, A, d1, d2, dx, best)
    vl = line_eval(O, A, d1, d2, dx, lo)
    vh = line_eval(O, A, d1, d2, dx, hi)
    if vl < vb:
        best = float(lo)
    if vh < vb:
        best = float(hi)
    return float(best)
