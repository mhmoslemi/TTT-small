        if not (0 <= h[i] + eps <= 1 and 0 <= h[j] - eps <= 1):
            continue
        h[i] += eps
        h[j] -= eps
        c5_new = _evaluate_c5(h, n)
        if c5_new > c5 + 1e-14:
            h[i] -= eps
            h[j] += eps
            continue
        c5 = c5_new
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
    return best_h, best_c5

def _binary_local_search(h, n, t0, deadline, target_sum, rng):
    h = np.where(h >= 0.5, 1.0, 0.0)
    if h.sum() != target_sum:
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        if h.sum() > target_sum:
            for a in idx1[:int(h.sum() - target_sum)]:
                h[a] = 0.0
        else:
            for a in idx0[:int(target_sum - h.sum())]:
                h[a] = 1.0
    c5 = _evaluate_c5(h, n)
    best_h, best_c5 = h.copy(), c5
    for it in range(2000000):
        if time.time() > deadline:
            break
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        if len(idx1) == 0 or len(idx0) == 0:
            break
        a = idx1[rng.integers(len(idx1))]
        b = idx0[rng.integers(len(idx0))]
        h[a], h[b] = 0.0, 1.0
        c5_new = _evaluate_c5(h, n)
        if c5_new > c5 + 1e-14:
            h[a], h[b] = 1.0, 0.0
        else:
            c5 = c5_new
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
    return best_h, best_c5

def _slsqp_optimize(h, n, t0, deadline, target_sum, tau=40.0):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        m = corr.max()
        return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-12})
        if res.fun < 10:
            hx = _project(res.x, target_sum)
            return hx, _evaluate_c5(hx, n)
    except Exception:
        pass
    return h, _evaluate_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    ts0 = n0 / 2.0
    h_init = _project(h0, ts0)
    best_h, best_c5 = h_init.copy(), _evaluate_c5(h_init, n0)

    # Phase 1: SLSQP on initial
    d1 = t0 + budget_s * 0.08
    hx, c5x = _slsqp_optimize(h_init, n0, t0, d1, ts0)
    if c5x < best_c5:
        best_h, best_c5 = hx.copy(), c5x

    # Phase 2: Subgradient on n0
    d2 = t0 + budget_s * 0.30
    hx, c5x = _optimize_subgrad(best_h, n0, t0, d2, ts0)
    if c5x < best_c5:
        best_h, best_c5 = hx.copy(), c5x

    # Phase 3: Continuous local search on n0
    d3 = t0 + budget_s * 0.55
    hx, c5x = _local_search(best_h, n0, t0, d3, ts0, rng)
    if c5x < best_c5:
        best_h, best_c5 = hx.copy(), c5x

    # Phase 4: Binary local search on n0
    d4 = t0 + budget_s * 0.70
    hx, c5x = _binary_local_search(best_h, n0, t0, d4, ts0, rng)
    if c5x < best_c5:
        best_h, best_c5 = hx.copy(), c5x

    # Phase 5: Try other n values
    for n_try in [100, 90, 140]:
        if time.time() > t0 + budget_s * 0.88:
            break
        ts_t = n_try / 2.0
        h_t = np.zeros(n_try)
        # Interpolate from best_h
        xs = np.linspace(0, 2, n0)
        xt = np.linspace(0, 2, n_try)
        h_t = np.interp(xt, xs, best_h)
        h_t = _project(h_t, ts_t)
        dt = min(t0 + budget_s * 0.92, time.time() + 30)
        # Quick subgrad
        hx, c5x = _optimize_subgrad(h_t, n_try, t0, dt, ts_t)
        if c5x < best_c5:
            dt2 = min(t0 + budget_s * 0.96, time.time() + 20)
            hx2, c5x2 = _local_search(hx, n_try, t0, dt2, ts_t, rng)
            if c5x2 < best_c5:
                best_h, best_c5 = hx2.copy(), c5x2

    # Final: verify and return
    best_c5 = _evaluate_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
