def project_to_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 with 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-9:
        return h
    # Distribute the difference
    step = 0.01
    for _ in range(10000):
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] + step)
        new_diff = target - np.sum(h)
        if abs(new_diff) < 1e-9:
            break
        # Adjust step
        if abs(new_diff) < abs(diff):
            diff = new_diff
        else:
            step *= 0.5
            diff = new_diff
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        n_fix = int(abs(diff)) + 1
        if diff > 0:
            idx = np.argsort(h)[::-1][:n_fix]
            h[idx] = np.minimum(1.0, h[idx] + diff / len(idx))
        else:
            idx = np.argsort(h)[:n_fix]
            h[idx] = np.maximum(0.0, h[idx] + diff / len(idx))
    return np.clip(h, 0, 1)

    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')  # length 2n-1
    # Soft max
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    obj = (c_max / beta + np.log(np.sum(exp_c)) / beta) * dx
    # Gradient
    # c[m] = sum_i h[i] * (1-h[i+s]) where s = m - (n-1)
    grad = np.zeros(n)
    for m in range(len(c)):

                if j != i:

    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    cross = np.correlate(h, 1.0 - h, mode='full')
    return np.max(cross) * dx

def project_to_constraint(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_parent(parent, n):
    np_ = len(parent)
    if np_ == n:
        return parent.copy()
    x_old = np.linspace(0, 2, np_, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    return np.interp(x_new, x_old, parent)

def soft_max_obj(h, n, beta):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    cmax = np.max(c)
    exp_c = np.exp(beta * (c - cmax))
    s = np.sum(exp_c)
    return (cmax + np.log(s) / beta) * dx

def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    cmax = np.max(c)
    exp_c = np.exp(beta * (c - cmax))
    weights = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for m in range(len(c)):
        w = weights[m]
        if w < 1e-8:
            continue
        s = m - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        for i in range(lo, hi):
            j = i + s
            grad[i] += w * (1.0 - h[j])
            if j != i:
                grad[j] -= w * h[i]
    grad *= dx
    return grad

def objective_func(h, n, beta):
    return soft_max_obj(h, n, beta)

def gradient_func(h, n, beta):
    return soft_max_grad(h, n, beta)

def run_slsqp(h0, n, beta, max_iter, timeout):
    def obj(h):
        return soft_max_obj(h, n, beta)
    def grad(h):
        return soft_max_grad(h, n, beta)
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    def cons_grad(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]
    
    t0 = time.time()
    try:
        res = minimize(
            obj, h0.copy(), jac=grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-10, 'iprint': 0}
        )
    except Exception:
        return None
    
    if res.x is not None:
        xc = project_to_constraint(res.x, n)
        return xc
    return None

def refine_local(h, n, n_steps=500, step_size=0.02):
    """Simple coordinate descent refinement."""
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    h = h.copy()
    
    for _ in range(n_steps):
        idx = np.random.randint(0, n)
        for delta in [step_size, -step_size, step_size*0.5, -step_size*0.5]:
            h_old = h[idx]
            h[idx] = np.clip(h_old + delta, 0.0, 1.0)
            # Adjust to maintain sum
            diff = np.sum(h) - n / 2.0
            if abs(diff) > 1e-10:
                # Undo
                h[idx] = h_old
                continue
            c5_new = evaluate_c5(h)
            if c5_new < best_c5 - 1e-8:
                best_c5 = c5_new
                h = project_to_constraint(h, n)
            else:
                h[idx] = h_old
        step_size *= 0.98
    return h, best_c5

            max_iter = int(min(500, remaining * 15))
            if max_iter < 50:

            if result is not None:
                result = project_to_constraint(result, n)
                c5 = evaluate_c5(result)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = result.copy()
                    best_n = n
                h0 = result
