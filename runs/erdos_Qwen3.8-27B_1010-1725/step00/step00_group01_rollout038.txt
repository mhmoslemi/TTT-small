    c = np.max(corr)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, m):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    else:
        p = -m
        g[p:] += 1.0 - h[:n - p]
        g[:n - p] -= h[p:]
    return g * dx

def softmax_og(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float) * dx
    cm = corr.max()
    e = np.exp((corr - cm) / tau)
    se = e.sum()
    val = cm + tau * np.log(se)
    w = e / se

        g += wk * grad_lag(h, k - (n - 1))

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max()) * dx

def grad_lag(h, m):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    else:
        p = -m
        g[p:] += 1.0 - h[:n - p]
        g[:n - p] -= h[p:]
    return g * dx

def softmax_og(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float) * dx
    cm = float(corr.max())
    e = np.exp((corr - cm) / tau)
    se = float(e.sum())
    val = cm + tau * np.log(se)
    w = e / se
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        g += wk * grad_lag(h, k - (n - 1))
    return val, g

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 - 1e-13 if d > 0 else h > 1e-13
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def pgd_opt(h0, n, tl, seed=0):
    t0 = time.time()
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    tau = 0.05
    lr = 0.1
    mom = np.zeros(n)
    sc = 0.82
    stall = 0
    while time.time() - t0 < tl:
        el = time.time() - t0
        tau = max(0.002, 0.05 * (0.9993 ** el))
        _, g = softmax_og(h, tau)
        mom = sc * mom + (1.0 - sc) * g
        improved = False
        for _ in range(8):
            hn = proj(h - lr * mom, n)
            cn = compute_c5(hn)
            if cn <= bc + 1e-9:
                h = hn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
                    improved = True
                break
            lr *= 0.5
        if improved:
            lr = min(lr / 0.5 * 1.04, 0.4)
            stall = 0
        else:
            lr = max(lr * 0.5, 1e-5)
            stall += 1
        if stall > 350:
            rng_r = np.random.default_rng(hash((n, stall)) & 0xFFFF)
            h = bh + rng_r.randn(n) * 0.04
            h = proj(h, n)
            mom[:] = 0.0
            lr = 0.1
            stall = 0
    return bh, bc

def topk_refine(h0, n, tl, topk=5, seed=0):
    t0 = time.time()
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    lr = 0.06
    while time.time() - t0 < tl:
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full").astype(float) * dx
        idx = np.argsort(corr)[::-1][:topk]
        any_improve = False
        for k in idx:
            m = int(k - (n - 1))
            g = grad_lag(h, m)
            hn = proj(h - lr * g, n)
            cn = compute_c5(hn)
            if cn < bc - 1e-12:
                h = hn
                bc = cn
                bh = h.copy()
                any_improve = True
                lr = min(lr * 1.03, 0.3)
            else:
                lr *= 0.85
        if not any_improve and lr < 0.003:
            break
    return bh, bc

def lsearch(h0, n, tl, seed=0):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    stp = 0.03
    while time.time() - t0 < tl:
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        d = stp * (0.3 + 0.7 * rng.random()) * (1.0 if rng.random() < 0.5 else -1.0)
        h[i] += d
        h[j] -= d
        h = proj(h, n)
        c = compute_c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
            stp = min(stp * 1.04, 0.14)
        else:
            h[i] -= d
            h[j] += d
            h = proj(h, n)
            stp = max(stp * 0.96, 0.003)
    return bh, bc

def xinterp(hs, nd):
    ns = len(hs)
    if ns == nd:
        return hs.copy()
    xs = np.linspace(0, 2, ns, endpoint=False) + 1.0 / ns
    xd = np.linspace(0, 2, nd, endpoint=False) + 1.0 / nd
    return proj(np.interp(xd, xs, hs, left=hs[0], right=hs[-1]), nd)

def rnd_h(n, rng):
    return proj(rng.random(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    tg = time.time()
    rng = np.random.default_rng(seed)
    bh, bc, bn = None, 999.0, 0
    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        c0 = compute_c5(h0)
        bh, bc, bn = h0.copy(), c0, len(h0)
    except Exception:
        pass

    nset = set()
    if h0 is not None:
        nset.add(len(h0))
    for v in (48, 50, 52, 54, 56, 60, 64):
        nset.add(v)
    ns = sorted(nset)

    for ni, n in enumerate(ns):
        rem = budget_s - (time.time() - tg)
        if rem < 10:
            break
        frac = max(0.3, 1.0 - 0.08 * ni)
        inits = []
        if h0 is not None:
            inits.append(xinterp(h0, n))
        inits.append(rnd_h(n, rng))

        for si, sh in enumerate(inits):
            rem = budget_s - (time.time() - tg)
            if rem < 4:
                break
            tp1 = min(rem * 0.30 * frac, 55.0)
            h1, c1 = pgd_opt(sh, n, tp1, seed + n * 10 + si)
            if c1 < bc:
                bh, bc, bn = h1.copy(), c1, n

            rem = budget_s - (time.time() - tg)
            if rem < 3:
                break
            tp2 = min(rem * 0.15 * frac, 20.0)
            h2, c2 = topk_refine(h1, n, tp2, seed + n * 20 + si)
            if c2 < bc:
                bh, bc, bn = h2.copy(), c2, n

            rem = budget_s - (time.time() - tg)
            if rem < 2:
                break
            tp3 = min(rem * 0.12 * frac, 12.0)
            h3, c3 = lsearch(h2, n, tp3, seed + n * 30 + si)
            if c3 < bc:
                bh, bc, bn = h3.copy(), c3, n

    if bh is None:
        bh = rnd_h(50, rng)
        bn = 50
        bc = compute_c5(bh)

    bh = proj(bh, bn)
    bc = compute_c5(bh)
    return bh, bc, bn
