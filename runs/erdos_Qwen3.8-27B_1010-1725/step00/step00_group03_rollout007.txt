Now, for the continuous optimization, the function c_k(h) = sum_n h[n+k] * (1-h[n]) is a quadratic form in h. Specifically:

The first term is linear, the second is a quadratic form (specifically, a "cross" term).

import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 0.01
_g_tgt = 100.0
_g_alpha = 120.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    x = v.astype(float).copy()
    for _ in range(500):
        x = np.clip(x, 0.0, 1.0)
        d = target - np.sum(x)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = x < 1.0 - 1e-15
            c = int(np.sum(m))
            if c == 0:
                break
            x[m] += d / c
        else:
            m = x > 1e-15
            c = int(np.sum(m))
            if c == 0:
                break
            x[m] += d / c
    return np.clip(x, 0.0, 1.0)

def to_binary(h, target, rng):
    idx = np.argsort(-h)
    b = np.zeros_like(h)
    b[idx[:int(target)]] = 1.0
    return b

def binary_ls(h, dx, target, rng, max_it, te):
    hc = h.copy()
    c5c = compute_c5(hc, dx)
    bh2, bc5 = hc.copy(), c5c
    oi = np.where(hc == 1)[0].tolist()
    zi = np.where(hc == 0)[0].tolist()
    for _ in range(max_it):
        if time.time() > te:
            break
        ii = rng.randint(len(oi))
        jj = rng.randint(len(zi))
        i, j = oi[ii], zi[jj]
        hc[i], hc[j] = 0.0, 1.0
        c5n = compute_c5(hc, dx)
        if c5n < c5c - 1e-15:
            c5c = c5n
            oi[ii], zi[jj] = j, i
            if c5n < bc5:
                bc5 = c5n
                bh2 = hc.copy()
        else:
            hc[i], hc[j] = 1.0, 0.0
    return bh2, bc5

def soft_max(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    return soft_max(h, _g_dx, _g_alpha)
