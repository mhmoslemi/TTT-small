        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _c5_val(h):
    dx = 2.0 / _g_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _obj_smooth(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    t = _g_temp
    em = np.exp((corr - m) / t)
    return float(m + t * np.log(np.sum(em)))


def _con_eq(x):
    return float(np.sum(x) - _g_target)


def _con_jac(x):
    return np.ones(_g_n)


def _run_slsqp(h, maxiter, ftol):
    bnds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_obj_smooth, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _fix_feasible(res.x)
    except Exception:
        pass
    return h


def _run_nm(h, maxiter, timeout_s):
    bnds = [(0.0, 1.0)] * _g_n
    t_start = time.time()
    try:
        res = minimize(_obj_smooth, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'fatol': 1e-10,
                                'xatol': 1e-8, 'adaptive': True})
        if time.time() - t_start > timeout_s:
            return h
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _fix_feasible(res.x)
    except Exception:
        pass
    return h


def _run_coord_descent(h, n_steps, step0):
    ht = h.copy()
    best_c = _c5_val(ht)
    step = step0
    stagnant = 0
    for _ in range(n_steps):
        if _time_left() < 5.0:
            break
        i = np.random.randint(_g_n)
        j = np.random.randint(_g_n)
        if i == j:
            continue
        s = step if np.random.random() < 0.5 else -step
        ni = ht[i] + s
        nj = ht[j] - s
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            step = max(step * 0.9, 0.002)
            continue
        old_i, old_j = ht[i], ht[j]
        ht[i] = ni
        ht[j] = nj
        c = _c5_val(ht)
        if c < best_c - 1e-14:
            best_c = c
            stagnant = 0
            step = min(step * 1.02, 0.12)
        elif c <= best_c + 1e-14:
            stagnant += 1
        else:
            ht[i] = old_i
            ht[j] = old_j
            step = max(step * 0.97, 0.002)
            stagnant += 1
        if stagnant > 3000:
            step = step0
            stagnant = 0
    return ht, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_target, _g_temp, _g_t0, _g_budget
    np.random.seed(seed)
    _g_t0 = time.time()
    _g_budget = budget_s
    _g_n = 95
    _g_target = _g_n / 2.0
    _g_temp = 0.01

    h0 = np.array(initial_h_values, dtype=float).copy()
    if len(h0) != _g_n:
        old_n = len(h0)
        xo = np.linspace(0.0, 2.0, old_n, endpoint=False)
        xn = np.linspace(0.0, 2.0, _g_n, endpoint=False)
        h0 = np.interp(xn, xo, h0)
    h0 = _fix_feasible(h0)

    best_h = h0.copy()
    best_c5 = _c5_val(best_h)

    # Phase 1: SLSQP (smooth, coarse)
    if _time_left() > 40:
        h1 = _run_slsqp(best_h, 500, 1e-14)
        c1 = _c5_val(h1)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Coordinate descent (long)
    if _time_left() > 30:
        h2, c2 = _run_coord_descent(best_h, 50000, 0.04)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        else:
            best_c5 = _c5_val(h2)
            if _c5_val(h2) < best_c5:
                best_h = h2

    # Phase 3: SLSQP (finer smoothing)
    if _time_left() > 40:
        _g_temp = 0.003
        h3 = _run_slsqp(best_h, 500, 1e-14)
        c3 = _c5_val(h3)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Nelder-Mead from best
    if _time_left() > 60:
        _g_temp = 0.001
        h4 = _run_nm(best_h, 2000, _time_left() - 10)
        c4 = _c5_val(h4)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Multi-start perturbation + SLSQP
    if _time_left() > 30:
        _g_temp = 0.001
        for _ in range(60):
            if _time_left() < 5:
                break
            h_t = best_h.copy()
            k = int(np.random.randint(3, 12))
            idx = np.random.choice(_g_n, k, replace=False)
            h_t[idx] += np.random.uniform(-0.06, 0.06, k)
            h_t = _fix_feasible(h_t)
            h_t = _run_slsqp(h_t, 200, 1e-12)
            c_t = _c5_val(h_t)
            if c_t < best_c5:
                best_h, best_c5 = h_t, c_t

    # Phase 6: Fine coordinate descent polish
    if _time_left() > 10:
        h6, c6 = _run_coord_descent(best_h, 40000, 0.01)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 7: Final SLSQP polish
    if _time_left() > 20:
        _g_temp = 0.0005
        h7 = _run_slsqp(best_h, 300, 1e-15)
        c7 = _c5_val(h7)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    # Final feasibility fix
    best_h = _fix_feasible(best_h)
    best_c5 = _c5_val(best_h)

    return best_h, best_c5, _g_n
