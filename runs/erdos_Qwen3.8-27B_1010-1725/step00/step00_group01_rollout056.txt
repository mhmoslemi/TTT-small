    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    w = vals - vals.min() + 1e-8
    w = w / w.sum()
    
    grad = np.zeros(n)
    s_arr = np.arange(n)

        vm = (jm >= 0) & (jm < n)
        t2 = np.zeros(n)

import numpy as np
import time

def project_polytope(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def subgradient(h, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    w = vals - vals.min() + 1e-10
    w = w / w.sum()
    s_arr = np.arange(n, dtype=float)
    grad = np.zeros(n)
    for wk, j in zip(w, idx):
        shift = float(j - (n - 1))
        js = s_arr - shift
        vp = (js >= 0) & (js < n)
        t1 = np.zeros(n)
        t1[vp] = 1.0 - h[js[vp].astype(int)]
        jm = s_arr + shift
        vm = (jm >= 0) & (jm < n)
        t2 = np.zeros(n)
        t2[vm] = h[jm[vm].astype(int)]
        grad += wk * (t1 - t2)
    return grad, c

def optimize_subgrad(h, n, time_limit):

    h = project_polytope(h.copy(), n / 2.0)
    best_val = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    top_k = 3
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.time() - t0 < time_limit:
        grad, c = subgradient(h, top_k)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_polytope(h_new, n / 2.0)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        lr *= 0.9995
        top_k = min(15, 3 + int((time.time() - t0) * 2))
    
    return best_h, best_val

    t0 = time.time()
    h = h.copy()
    best_val = compute_c5(h)
    improved = True

import numpy as np
import time

def project_polytope(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def subgradient(h, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    vals = c[idx]
    w = vals - vals.min() + 1e-10
    w = w / w.sum()
    s_arr = np.arange(n, dtype=np.float64)
    grad = np.zeros(n)
    for wk, j in zip(w, idx):
        shift = j - (n - 1)
        js = s_arr - float(shift)
        vp = (js >= 0) & (js < n)
        t1 = np.zeros(n)
        t1[vp] = 1.0 - h[js[vp].astype(np.intp)]
        jm = s_arr + float(shift)
        vm = (jm >= 0) & (jm < n)
        t2 = np.zeros(n)
        t2[vm] = h[jm[vm].astype(np.intp)]
        grad += wk * (t1 - t2)
    return grad, c

def optimize_subgrad(h, n, time_limit):
    t0 = time.time()
    h = project_polytope(h.copy(), n / 2.0)
    best_val = compute_c5(h)
    best_h = h.copy()
    lr = 0.025
    momentum = np.zeros(n)
    beta = 0.85
    step = 0
    while time.time() - t0 < time_limit:
        top_k = min(15, 3 + step // 300)
        grad, c = subgradient(h, top_k)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_polytope(h_new, n / 2.0)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            lr = min(0.06, lr * 1.002)
        h = h_new
        lr *= 0.9997
        step += 1
    return best_h, best_val

def local_search(h, n, time_limit):
    t0 = time.time()
    h = h.copy()
    best_val = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            h_trial = h.copy()
            h_trial[i] = 1.0 - h[i]
            h_trial = project_polytope(h_trial, n / 2.0)
            val = compute_c5(h_trial)
            if val < best_val - 1e-10:
                h = h_trial
                best_val = val
                improved = True
    return h, best_val

def make_random_init(n, rng):
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] = 0.5
    return project_polytope(h, n / 2.0)

def make_interpolated(h_parent, n, rng):
    xp = np.linspace(0.0, 2.0, len(h_parent))
    xn = np.linspace(0.0, 2.0, n)
    h = np.interp(xn, xp, h_parent)
    h += rng.normal(0, 0.04, n)
    return project_polytope(h, n / 2.0)

def make_perturbed(h, rng, frac):
    h_new = h.copy()
    n = len(h_new)
    n_flip = max(1, int(n * frac))
    idx = rng.choice(n, size=n_flip, replace=False)
    h_new[idx] = 1.0 - h_new[idx]
    return project_polytope(h_new, n / 2.0)

def make_sinusoidal(n, rng, freq):
    x = np.linspace(0.0, 2.0, n)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * x + phase)
    return project_polytope(h, n / 2.0)
