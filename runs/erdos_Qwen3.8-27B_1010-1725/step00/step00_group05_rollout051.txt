import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def _pen_obj(h, n, pw):
    hc = np.clip(h, 0.0, 1.0)
    c5 = _compute_c5(hc)
    return c5 + pw * (np.sum(hc) - 0.5 * n) ** 2

def _sm_obj(h, n, beta, pw):
    hc = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    v = beta * corr * dx
    m = np.max(v)
    sm = (m + np.log(np.sum(np.exp(v - m)))) / beta
    return sm + pw * (np.sum(hc) - 0.5 * n) ** 2

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _gen_starts(n, rng, prev):
    starts = []
    if prev is not None:
        starts.append(prev.copy())
    for _ in range(6):
        hb = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        hb[idx] = 1.0
        starts.append(hb)
    for _ in range(5):
        t = np.arange(n, dtype=float) / n * 2.0
        f = rng.integers(1, 8)
        ph = rng.uniform(0, 2.0 * np.pi)
        hs = 0.5 + 0.4 * np.sin(np.pi * f * t + ph)
        starts.append(_fix(hs, n))
    # Weighted-biased patterns
    for _ in range(3):
        t = np.arange(n, dtype=float) / n * 2.0
        hs = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t + rng.uniform(0, 6.28))
        starts.append(_fix(hs, n))
    return starts

def _nm_optimize(h0, n, maxiter, t_deadline):
    start_t = time.time()
    try:
        r = minimize(_pen_obj, h0.copy(), args=(n, 100.0),
                     method='Nelder-Mead',
                     options={'maxiter': maxiter, 'xatol': 1e-9,
                              'fatol': 1e-10, 'adaptive': True})
        return _fix(r.x, n)
    except Exception:
        return h0

def _slq_optimize(h0, n, t_deadline):
    try:
        r = minimize(_sm_obj, h0.copy(), args=(n, 100.0, 500.0),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 400, 'ftol': 1e-12})
        return _fix(r.x, n)
    except Exception:
        return h0

def _mutate(h, n, rng, scale):
    h_m = h.copy()
    mask = rng.random(n) < 0.15
    count = max(1, int(mask.sum()))
    h_m[mask] += rng.normal(0, scale, count)
    return _fix(h_m, n)
