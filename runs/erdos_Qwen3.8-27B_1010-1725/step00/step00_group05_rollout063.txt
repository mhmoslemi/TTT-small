    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(corr)
    exp_c = np.exp(T * (corr - max_val))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)

            h_try = h_cur.copy()
            h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try, dx_val)
            if c5_try < cur_c5 - 1e-14:
                h_cur = h_try
                cur_c5 = c5_try
                if cur_c5 < best_local:
                    best_local = cur_c5
                    best_local_h = h_cur.copy()
                break
        if it % 500 == 499:
            alpha *= 0.8
            if alpha < 0.005:
                alpha = 0.1
    return best_local, best_local_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h = initial_h_values.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = np.clip(h, 0, 1)
    h = project_feasible(h, target_sum)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_n = n

    # Phase 1: Projected gradient at original n
    deadline1 = t0 + budget_s * 0.35
    c5_1, h_1 = optimize_pg(h, n, dx, target_sum, rng, deadline1, steps=5000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h_1
        best_n = n

    # Phase 2: Upsample and optimize at 2n
    remaining = budget_s - (time.time() - t0)
    if remaining > 60 and best_n < 300:
        n2 = min(2 * best_n, 300)
        if n2 > best_n:
            dx2 = 2.0 / n2
            target2 = n2 / 2.0
            h2 = upsample_step(best_h, n2)
            h2 = project_feasible(h2, target2)
            deadline2 = t0 + budget_s * 0.62
            c5_2, h_2 = optimize_pg(h2, n2, dx2, target2, rng, deadline2, steps=4000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_2
                best_n = n2

    # Phase 3: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        deadline3 = t0 + budget_s * 0.82
        c5_3, h_3 = coordinate_descent(best_h, best_n, 2.0 / best_n, best_n / 2.0,
                                       rng, deadline3, iters=8000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_3
            best_n = len(h_3)

    # Phase 4: Random restarts
    remaining = budget_s - (time.time() - t0)
    attempts = 0
    while remaining > 35 and attempts < 8:
        h_r = best_h + rng.normal(0, 0.12, best_n)
        h_r = project_feasible(h_r, best_n / 2.0)
        dx_r = 2.0 / best_n
        deadline_r = t0 + budget_s - 25
        c5_r, h_rb = optimize_pg(h_r, best_n, dx_r, best_n / 2.0, rng, deadline_r, steps=1500)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_rb
            best_n = len(h_rb)
        attempts += 1
        remaining = budget_s - (time.time() - t0)

    # Final: try binary rounding
    if time.time() < t0 + budget_s - 5:
        h_bin = (best_h > 0.5).astype(float)
        # Adjust to meet sum constraint
        diff = int(best_n / 2) - int(h_bin.sum())
        if diff > 0:
            fracs = np.sort(np.where(h_bin == 0)[0])
            for i in range(min(diff, len(fracs))):
                h_bin[fracs[i]] = 1.0
        elif diff < 0:
            fracs = np.sort(np.where(h_bin == 1)[0])
            for i in range(min(-diff, len(fracs))):
                h_bin[fracs[i]] = 0.0
        h_bin = project_feasible(h_bin, best_n / 2.0)
        c5_bin = compute_c5(h_bin, 2.0 / best_n)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    best_n = len(best_h)

    return best_h, best_c5, best_n
