def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _balanced_binary(n, rng):
    """Create a balanced binary sequence using a quasi-random construction."""
    h = np.zeros(n, dtype=float)
    # Use van der Corput-like construction
    m = 1
    while m < n:
        m *= 2
    for i in range(n):
        # Reverse bits of i considering log2(n) bits
        rev = 0
        x = i
        for b in range(int(np.log2(n)) if n > 1 else 1):
            rev = (rev << 1) | (x & 1)
            x >>= 1
        if (rev & (1 << (int(np.log2(n)) - 1))) == 0:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    num_ones = int(np.sum(h))
    if num_ones * 2 > n:
        excess = num_ones - n // 2
        one_positions = np.where(h == 1)[0]
        for j in range(excess):
            h[one_positions[j]] = 0
    elif num_ones * 2 < n:
        deficit = n // 2 - num_ones
        zero_positions = np.where(h == 0)[0]
        for j in range(deficit):
            h[zero_positions[j]] = 1
    return h

import numpy as np
import time


def _corr_max(h):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max())


def _golden_init(n, offset=0.0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = (np.arange(n, dtype=float) + offset) * phi
    ranks = np.argsort(np.mod(x, 1.0))
    h = np.zeros(n)
    h[ranks[:n // 2]] = 1.0
    return h


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        run_len = rng.randint(2, max(3, n // 6))
        if pos + run_len > n:
            run_len = n - pos
        h[pos:pos + run_len] = rng.random()
        pos += run_len
    h = (h >= np.median(h)).astype(float)
    while h.sum() > n / 2:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < n / 2:
        h[rng.choice(np.where(h == 0)[0])] = 1
    return h


def _sa_swap(h, n, rng, deadline, T0=0.06, T_floor=0.002):
    curr = _corr_max(h)
    best_val = curr
    best = h.copy()
    T = T0
    stall = 0
    while time.time() < deadline and stall < 12000:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new = _corr_max(h)
        if new <= best_val:
            stall = 0
            if new < best_val:
                best_val = new
                best = h.copy()
            curr = new
        elif rng.random() < np.exp(-(new - curr) / T):
            curr = new
            stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T = max(T_floor, T * 0.996)
    return best, best_val * 2.0 / n


def _local_polish(h, n, deadline):
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        curr = _corr_max(h)
        for i in ones:
            if time.time() > deadline:
                break
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                new = _corr_max(h)
                if new < curr:
                    curr = new
                    improved = True
                else:
                    h[j] = 0.0
            h[i] = 1.0
        if not improved:
            break
    return h


def _nonbinary_refine(h, n, rng, deadline):
    """Try small non-binary perturbations around the best binary solution."""
    best_val = _corr_max(h)
    best = h.copy()
    for _ in range(200):
        if time.time() > deadline:
            break
        trials = 5
        for _t in range(trials):
            i = rng.randint(n)
            old = h[i]
            delta = (rng.random() - 0.5) * 0.15
            h[i] = np.clip(old + delta, 0.0, 1.0)
            # Renormalize: adjust a random other position
            diff = (n / 2.0) - h.sum()
            j = rng.randint(n)
            if j == i:
                j = (j + 1) % n
            h[j] = np.clip(h[j] + diff, 0.0, 1.0)
            # Check sum constraint approximately
            if abs(h.sum() - n / 2.0) > 0.01:
                h[i], h[j] = old, h[j] - diff + (h.sum() - n / 2.0 + diff)
            c = _corr_max(h)
            if c < best_val:
                best_val = c
                best = h.copy()
            h[i] = old
    return best, best_val * 2.0 / n


def _binarize(h, n, rng):
    h = h.copy()
    med = np.median(h)
    h = (h >= med).astype(float)
    while h.sum() > n / 2:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < n / 2:
        h[rng.choice(np.where(h == 0)[0])] = 1
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    best_c5 = np.inf
    best_h = None

    init = None
    try:
        init = np.asarray(initial_h_values, dtype=float)
    except Exception:
        pass

    sizes = [48, 64, 80, 96, 128]
    if init is not None:
        ni = len(init)
        if ni % 2 == 0:
            if ni in sizes:
                sizes.remove(ni)
            sizes.insert(0, ni)

    # Time allocation per attempt
    n_attempts = len(sizes) * 3
    per_att = max(2.0, (deadline - time.time()) / n_attempts)

    attempt = 0
    for n in sizes:
        for r in range(3):
            attempt += 1
            if time.time() > deadline:
                break

            sub_dl = min(time.time() + per_att, deadline - 5)

            # Initialize
            if r == 0 and init is not None and len(init) == n:
                h = _binarize(init, n, rng)
            elif r == 0:
                h = _golden_init(n)
            elif r == 1:
                h = _block_init(n, rng)
            else:
                h = _golden_init(n, offset=rng.random() * 0.1)

            # SA optimization
            h_opt, c5_binary = _sa_swap(h.copy(), n, rng, sub_dl)

            # Polish for small n
            if n <= 64 and time.time() < deadline - 10:
                polish_dl = min(time.time() + min(30.0, (deadline - time.time()) * 0.1), deadline - 5)
                h_opt = _local_polish(h_opt.copy(), n, polish_dl)
                c5_binary = _corr_max(h_opt) * 2.0 / n

            # Non-binary refinement
            if time.time() < deadline - 5:
                ndl = min(time.time() + min(20.0, (deadline - time.time()) * 0.08), deadline - 3)
                h_nb, c5_nb = _nonbinary_refine(h_opt.copy(), n, rng, ndl)
                if c5_nb < c5_binary:
                    c5_binary = c5_nb
                    h_opt = h_nb

            if c5_binary < best_c5:
                best_c5 = c5_binary
                best_h = h_opt.copy()

    # Final validation
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        # Ensure sum constraint
        diff = len(best_h) / 2.0 - best_h.sum()
        if abs(diff) > 1e-6:
            # Distribute correction to smallest fractional values
            idx = np.argsort(best_h)[:max(1, int(abs(diff)) + 1)]
            for ix in idx:
                adj = np.clip(diff, -(best_h[ix]), 1.0 - best_h[ix])
                best_h[ix] += adj
                diff -= adj
        final_c5 = _corr_max(best_h) * 2.0 / len(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return best_h, best_c5, len(best_h)
