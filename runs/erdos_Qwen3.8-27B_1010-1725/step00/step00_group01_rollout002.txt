            h_bin = project_sum(h_bin, n)
            try:
                h_opt = opt_slsqp(h_bin, n, 100.0, maxiter=200)
                h_opt = project_sum(h_opt, n)
                score = evaluate_c5(h_opt)
                if score < best_score:
                    best_score = score
                    best_h = h_opt.copy()
                    best_n = n
            except Exception:
                pass

    # Final high-temperature refinement
    if best_h is not None and time_left() > 40:
        for temp in [2000.0, 5000.0]:
            if time_left() < 20:
                break
            try:
                h_opt = opt_slsqp(best_h, best_n, temp, maxiter=600)
                h_opt = project_sum(h_opt, best_n)
                score = evaluate_c5(h_opt)
                if score < best_score:
                    best_score = score
                    best_h = h_opt.copy()
            except Exception:
                break

    # One more pass at slightly different n around best
    if best_h is not None and time_left() > 60:
        for dn in [-2, 2, -3, 3]:
            nn = best_n + dn
            if nn < 10 or time_left() < 20:
                continue
            h_try = interp_h(best_h, nn)
            h_try = project_sum(h_try, nn)
            try:
                h_opt = opt_slsqp(h_try, nn, 500.0, maxiter=400)
                h_opt = project_sum(h_opt, nn)
                score = evaluate_c5(h_opt)
                if score < best_score:
                    best_score = score
                    best_h = h_opt.copy()
                    best_n = nn
            except Exception:
                pass

    if best_h is None:
        if parent is not None:
            best_h = parent.copy()
            best_n = len(parent)
            best_score = evaluate_c5(best_h)
        else:
            best_h = np.full(47, 0.5)
            best_n = 47
            best_score = evaluate_c5(best_h)

    return (best_h, best_score, best_n)
