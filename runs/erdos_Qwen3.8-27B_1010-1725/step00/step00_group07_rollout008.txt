import numpy as np
import time

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def project_feasible(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def corr_grad_single(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        s = k
        if n - s > 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    else:
        s = -k
        if n - s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    return grad

def smooth_obj_grad(h, beta):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    n = len(h)
    max_c = corr.max()
    scaled = np.clip(beta * (corr - max_c), -100.0, 0.0)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    obj = max_c + np.log(exp_s.sum()) / beta
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-9:
            continue
        k = m - (n - 1)
        grad += wm * corr_grad_single(h, k)
    return obj, grad

def pg_optimize(h0, budget_s, start_time, max_iters=2000, lr_init=0.003):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 15:
            break
        c5_curr = compute_c5(h, dx)
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 600:
                break
        beta = 15.0 + min(it * 0.15, 40.0)
        _, grad = smooth_obj_grad(h, beta)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < c5_curr + 1e-13:
            h = h_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.75
            if lr < 1e-12:
                break
    return best_h, best_c5

def polish_exact(h0, budget_s, start_time, max_iters=500):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.002
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        c5_curr = corr.max()
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
        max_idx = np.argmax(corr)
        active = np.where(corr > c5_curr - 1e-6)[0]
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            grad += corr_grad_single(h, k)
        grad /= len(active)
        grad *= dx
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < c5_curr + 1e-13:
            h = h_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.7
            if lr < 1e-13:
                break
    return best_h, best_c5

def upsample(h0, new_n):
    old_n = len(h0)
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h0)
    return project_feasible(h_new, new_n / 2.0)

def make_structured_init(n, freq1, freq2, amp1, amp2, phase):
    x = np.arange(n) / n * 2.0
    h = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * x + phase) + amp2 * np.sin(2 * np.pi * freq2 * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = make_structured_init(80, 2, 3, 0.3, 0.2, 0.0)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init / 2.0)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_init)

    # Phase 1: optimize at initial resolution
    h_opt, c5_opt = pg_optimize(h_init, budget_s, start_time)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: multi-resolution refinement
    current_h = best_h.copy()
    for target_n in [160, 256, 384]:
        if time.time() - start_time > budget_s - 80:
            break
        h_up = upsample(current_h, target_n)
        h_up += rng.randn(target_n) * 0.005
        h_up = np.clip(h_up, 0, 1)
        h_up = project_feasible(h_up, target_n / 2.0)
        h_opt, c5_opt = pg_optimize(h_up, budget_s, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            current_h = h_opt.copy()

    # Phase 3: alternative initializations
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n_try = 256
        tries = [(2, 5, 0.3, 0.3, 0.0), (3, 7, 0.25, 0.25, 1.0),
                 (4, 9, 0.3, 0.15, 0.5), (2, 4, 0.4, 0.2, 2.0)]
        for f1, f2, a1, a2, ph in tries:
            if time.time() - start_time > budget_s - 30:
                break
            h_try = make_structured_init(n_try, f1, f2, a1, a2, ph)
            h_opt, c5_opt = pg_optimize(h_try, budget_s, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: polish the best solution
    if time.time() - start_time > budget_s - 20:
        n_final = len(best_h)
        return best_h, compute_c5(best_h, 2.0 / n_final), n_final
    h_pol, c5_pol = polish_exact(best_h, budget_s, start_time)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Final validation
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / n_final)
    return best_h, best_c5, n_final
