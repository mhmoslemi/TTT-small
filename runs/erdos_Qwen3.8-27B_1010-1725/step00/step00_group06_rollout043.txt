import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lag_grad(h, m):
    n = len(h)
    if m == 0:
        return 1.0 - 2.0 * h
    a = abs(m)
    if a >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if m > 0:
        g[a:] = 1.0 - h[:n-a]
        g[:n-a] -= h[a:]
    else:
        g[a:] = 1.0 - h[:n-a]
        g[:n-a] -= h[a:]
    return g

def proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_opt(h0, t_limit=60.0):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h)
    if m == 0:
        return 1.0 - 2.0 * h
    a = abs(m)
    if a >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if m > 0:
        g[a:] = 1.0 - h[:n-a]
        g[:n-a] -= h[a:]
    else:
        g[:n-a] = 1.0 - h[a:]
        g[a:] -= h[:n-a]
    return g

        d = target - h.sum()
        if abs(d) < 1e-13:

import numpy as np
import time
from itertools import combinations

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, m):
    n = len(h)
    if m == 0:
        return 1.0 - 2.0 * h
    a = abs(m)
    if a >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if m > 0:
        g[a:] = 1.0 - h[:n - a]
        g[:n - a] -= h[a:]
    else:
        g[:n - a] = 1.0 - h[a:]
        g[a:] -= h[:n - a]
    return g

def proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_opt(h0, tl=60.0):
    n = len(h0)
    ts = n / 2.0
    h = h0.astype(float).copy()
    bh = h.copy()
    bc = compute_c5(h)
    lr = 0.05
    t0 = time.time()
    for _ in range(100000):
        if time.time() - t0 > tl:
            break
        cv = c5_all(h)
        cm = float(cv.max())
        if cm < bc:
            bc = cm
            bh = h.copy()
        tau = max(0.002, 0.03 * cm)
        act = cv >= cm - tau
        na = max(1, int(act.sum()))
        g = np.zeros(n)
        for ki in np.where(act)[0]:
            g += lag_grad(h, int(ki) - n + 1)
        g /= na
        hn = proj(h - lr * g, ts)
        if compute_c5(hn) < cm + 1e-12:
            h = hn
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return bh, bc

def bin_search(h0, tl=30.0):
    n = len(h0)
    ns = n // 2
    idx = np.argsort(-h0)
    h = np.zeros(n)
    h[idx[:ns]] = 1.0
    bh = h.copy()
    bc = compute_c5(h)
    t0 = time.time()
    imp = True
    while imp and time.time() - t0 < tl:
        imp = False
        oi = list(np.where(h == 1)[0])
        zi = list(np.where(h == 0)[0])
        for i in oi:
            if time.time() - t0 > tl:
                break
            for j in zi:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                if c < bc - 1e-14:
                    bc = c
                    bh = h.copy()
                    imp = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return bh, bc

def brute_small(n, tl=30.0):
    ns = n // 2
    t0 = time.time()
    bc = np.inf
    bh = None
    for combo in combinations(range(n), ns):
        if time.time() - t0 > tl:
            break
        h = np.zeros(n)
        h[list(combo)] = 1.0
        c = compute_c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
    return bh, bc
