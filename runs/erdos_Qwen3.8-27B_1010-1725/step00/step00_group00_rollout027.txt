    n = len(h)

    """Gradient of the objective (max of correlate * dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = max(1e-10, max_val * 1e-8)
    active = np.where(corr >= max_val - tol)[0]

            if 0 <= j + k < n:

def soft_max_obj(h, temp):
    """Smooth approximation of max correlate * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft max
    m = np.max(corr)
    sm = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = max(1e-12, abs(max_val) * 1e-9)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    j = np.arange(n)
    for ai in active:
        k = int(ai) - (n - 1)
        jk1 = j - k
        jk2 = j + k
        mask1 = (jk1 >= 0) & (jk1 < n)
        mask2 = (jk2 >= 0) & (jk2 < n)
        grad[mask1] += (1.0 - h[jk1[mask1]])
        grad[mask2] -= h[jk2[mask2]]
    grad /= max(1, len(active))
    grad *= dx
    return max_val * dx, grad

class SumCon:
    def __init__(self, n):
        self.target = n / 2.0
    def __call__(self, x):
        return x.sum() - self.target

class SoftMaxObj:
    def __init__(self, temp):
        self.temp = temp
    def __call__(self, x):
        n = len(x)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sm = m + self.temp * np.log(np.sum(np.exp((corr - m) / self.temp)))
        return sm * dx

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
