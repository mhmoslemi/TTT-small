import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_objective(h, alpha):
    """Gradient of smoothed max of correlation"""
    n = len(h)
    corr = compute_corr(h)
    m = len(corr)  # 2n-1
    
    # Softmax weights
    cmax = np.max(corr)
    w = np.exp((corr - cmax) * alpha) / np.exp((corr - cmax) * alpha).sum()
    
    # For each corr element at index k (shift s = k - (n-1)):
    # corr[k] = sum_j h[j] * (1 - h[j - s])
    # d/dh[i] corr[k] = (1-h[i-s]) if i-s valid - h[i+s] if i+s valid
    #                    where s = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(m):
        s = k - (n - 1)
        if w[k] < 1e-15:
            continue
        # For each i in 0..n-1:
        # d/dh[i] corr[k] = (1 - h[i-s]) * indicator(0 <= i-s < n) 
        #                    - h[i+s] * indicator(0 <= i+s < n)
        
        # First term: (1-h[i-s]) for i such that 0 <= i-s < n, i.e., max(0,s) <= i <= min(n-1, n-1+s)
        i_lo = max(0, s)
        i_hi = min(n - 1, n - 1 + s)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w[k] * (1.0 - h[idx - s])
        
        # Second term: -h[i+s] for i such that 0 <= i+s < n, i.e., max(0, -s) <= i <= min(n-1, n-1-s)
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] += w[k] * (-h[idx + s])
    
    return grad

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    cmax = np.max(corr)
    return cmax / alpha + np.log(np.exp((corr - cmax) * alpha).sum() / len(corr)) / alpha

            if best_c5 < 0.365:
                break
            scale = 0.03 * (1 + r * 0.5)
            h_pert = h + rng.randn(n) * scale
            h_pert = fix_sum(h_pert, n)
            alpha = 60.0
            try:
                res = minimize(
                    smooth_max_value, h_pert, args=(alpha,),
                    jac=smooth_max_grad, method='SLSQP',
                    bounds=bounds, constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-10})
                if hasattr(res, 'x') and np.isfinite(res.x).all():
                    h_res = fix_sum(res.x, n)
                    c5_res = compute_c5(h_res)
                    if c5_res < best_c5:
                        best_h = h_res.copy()
                        best_c5 = c5_res
                        h = h_res.copy()
            except Exception:
                pass

        # Coordinate descent polish if time allows
        if elapsed() < budget_s * 0.6 and n <= 120:
            h_cd = coord_descent(h, n, n_iter=200)
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd

    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
