    target = n / 2.0

        return h

        room = 1.0 - h
        total_room = np.sum(room)

    else:
        # Need to decrease sum; subtract from elements > 0
        room = h
        total_room = np.sum(room)
        if total_room > 1e-12:

    # Handle any remaining violations
    h = np.clip(h, 0.0, 1.0)

    diff = target - np.sum(h)

            for d in (0.04, 0.08, 0.02):
                if h[i] >= d and h[j] <= 1.0 - d:
                    h2 = h.copy()
                    h2[i] -= d
                    h2[j] += d
                    c2 = compute_c5(h2)
                    if c2 < best_c - 1e-12:
                        h = h2
                        best_c = c2
                        break
        best = h
        # Single-coordinate moves
        h = best.copy()
        for _ in range(40):
            if time.time() - t0 > budget_s:
                break
            i = rng.randint(n)
            for s in step_sizes[:2]:
                h2 = h.copy()
                h2[i] += s
                h2 = project(h2, n)
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-12:
                    h = h2
                    best_c = c2
                    break
        best = h
        # Random kick to escape local minima
        if time.time() - t0 < budget_s:
            kick = 0.015 * rng.randn(n)
            h_k = project(best + kick, n)
            c_k = compute_c5(h_k)
            if c_k < best_c:
                best = h_k
                best_c = c_k
    return best, best_c


def upscale(h, factor):
    n = len(h)
    new_n = n * factor
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1.0, new_n)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def random_h(n, seed=0):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_global = time.time()
    t_deadline = t_global + budget_s * 0.95
    rng = np.random.RandomState(seed)

    # Phase 1: optimize at n=80
    n1 = 80
    t1 = min(150.0, budget_s * 0.12)

    # Start from parent
    h1 = np.array(initial_h_values, dtype=np.float64)
    h1 = project(h1, n1)
    c1 = compute_c5(h1)
    best_h, best_c, best_n = h1.copy(), c1, n1

    # Local search from parent
    h_opt, c_opt = local_search(h1, n1, t1 * 0.5, seed=seed)
    if c_opt < best_c:
        best_h, best_c, best_n = h_opt.copy(), c_opt, n1

    # A few random starts
    remaining = min(t_deadline - time.time(), t1)
    n_rand = max(1, int(remaining / 30.0))
    s = 0
    while s < n_rand and time.time() < t_deadline:
        h_r = random_h(n1, seed=seed + 1000 + s)
        h_o, c_o = local_search(h_r, n1, 25.0, seed=seed + 2000 + s)
        if c_o < best_c:
            best_h, best_c, best_n = h_o.copy(), c_o, n1
        s += 1

    # Phase 2: upscale to n=160
    n2 = 160
    t2 = min(350.0, max(60.0, budget_s * 0.3))
    if time.time() < t_deadline:
        h2 = upscale(best_h, 2)
        h2 = project(h2, n2)
        h2o, c2o = local_search(h2, n2, t2, seed=seed + 500)
        if c2o < best_c:
            best_h, best_c, best_n = h2o.copy(), c2o, n2

        # Also try upscaled random start
        if time.time() < t_deadline:
            h2r = upscale(random_h(n1, seed=seed + 3000), 2)
            h2r = project(h2r, n2)
            h2ro, c2ro = local_search(h2r, n2, t2 * 0.3, seed=seed + 600)
            if c2ro < best_c:
                best_h, best_c, best_n = h2ro.copy(), c2ro, n2

    # Phase 3: upscale to n=320
    n3 = 320
    t3 = min(500.0, max(60.0, budget_s * 0.4))
    if time.time() < t_deadline:
        factor = n3 // best_n if best_n in (80, 160) else 2
        h3 = upscale(best_h, factor)
        h3 = project(h3, n3)
        t3_actual = min(t3, max(10.0, t_deadline - time.time()))
        h3o, c3o = local_search(h3, n3, t3_actual, seed=seed + 700)
        if c3o < best_c:
            best_h, best_c, best_n = h3o.copy(), c3o, n3

    # Final phase: try n=640 if time permits
    n4 = 640
    t4 = max(0.0, t_deadline - time.time() - 20.0)
    if t4 > 60.0 and best_n < n4:
        factor = n4 // best_n if best_n in (80, 160, 320) else 2
        h4 = upscale(best_h, factor)
        h4 = project(h4, n4)
        h4o, c4o = local_search(h4, n4, min(t4, 300.0), seed=seed + 800)
        if c4o < best_c:
            best_h, best_c, best_n = h4o.copy(), c4o, n4

    # Final safety check and return
    elapsed = time.time() - t_global
    if elapsed > budget_s * 0.9:
        pass  # already near deadline
    final_c = compute_c5(best_h)
    best_c = min(best_c, final_c)
    best_h = project(best_h, best_n)
    # Verify constraint
    assert abs(best_h.sum() * (2.0 / best_n) - 1.0) < 1e-6, f"Sum constraint violated: {best_h.sum()*2.0/best_n}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    return best_h, best_c, best_n
