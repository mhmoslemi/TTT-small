import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    comp = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(np.max(comp) * dx)

    h = h.copy().astype(np.float64)
    best_c5 = compute_c5(h, n)
