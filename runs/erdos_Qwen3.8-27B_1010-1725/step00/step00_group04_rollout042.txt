import numpy as np
from scipy.special import logsumexp
import time

def project_to_simplex_box(h, n):
    """Project h onto {0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -max(h), 1 - min(h)
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_corr(h):
    """Compute full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    return (corr * dx).max()

def grad_obj(h, n, temperature=0.02):
    """Gradient of soft-max objective"""
    dx = 2.0 / n
    corr = compute_corr(h)
    vals = corr * dx
    max_val = vals.max()
    log_w = (vals - max_val) / temperature
    w = np.exp(log_w - logsumexp(log_w))
    
    n_corr = len(corr)  # 2n-1
    # corr[k] corresponds to lag = k - (n-1)
    # d corr[k] / d h[m] = (1-h[m-lag]) * [valid] - h[m+lag] * [valid]
    # where lag = k - (n-1)
    
    grad = np.zeros(n)
    lags = np.arange(n_corr) - (n - 1)  # from -(n-1) to (n-1)
    
    for k in range(n_corr):
        lag = lags[k]
        if w[k] < 1e-10:
            continue
        # (1-h[m-lag]) for m where 0<=m-lag<n => lag<=m<n+lag (if lag>0) or 0<=m<n-lag (if lag<0)
        # h[m+lag] for m where 0<=m+lag<n => -lag<=m<n-lag
        if lag >= 0:
            m1_start, m1_end = lag, n  # m-lag in [0,n) => m in [lag, n+lag) ∩ [0,n) = [lag, n)
            m2_start, m2_end = 0, n - lag  # m+lag in [0,n) => m in [-lag, n-lag) ∩ [0,n) = [0, n-lag)
        else:
            m1_start, m1_end = 0, n + lag  # m-lag in [0,n) => m in [lag, n+lag) ∩ [0,n) = [0, n+lag)
            m2_start, m2_end = -lag, n  # m+lag in [0,n) => m in [-lag, n-lag) ∩ [0,n) = [-lag, n)
        
        if m1_end > m1_start:
            grad[m1_start:m1_end] += w[k] * dx * (1 - h[m1_start:m1_end - lag])
        if m2_end > m2_start:
            grad[m2_start:m2_end] -= w[k] * dx * h[m2_start:m2_end + lag]
    
    # Also add gradient of sum constraint penalty
    # We'll handle this via projection instead
    
    return grad

def projected_gradient(h, n, max_iter=2000, lr_init=0.01, temperature=0.02):
    """Projected gradient descent"""
    h = h.copy()
    h = project_to_simplex_box(h, n)
    best_c5 = c5_value(h, n)
    best_h = h.copy()
    
    lr = lr_init
    for it in range(max_iter):
        g = grad_obj(h, n, temperature)
        h_new = h - lr * g
        h_new = project_to_simplex_box(h_new, n)
        
        c5 = c5_value(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Adaptive step size
        if c5 < best_c5 + 1e-10:
            lr = min(lr * 1.1, 0.1)
        else:
            lr = lr * 0.7
            h_new = h  # revert
        
        h = h_new
    
    return best_h, best_c5

def interpolate_to_resolution(h_coarse, n_fine):
    """Upscale from coarse to fine resolution"""
    n_coarse = len(h_coarse)
    ratio = n_fine // n_coarse
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = (i + 1) * ratio if i < n_coarse - 1 else n_fine
        h_fine[start:end] = h_coarse[i]
    # Fix sum
    target = n_fine / 2.0
    h_fine = project_to_simplex_box(h_fine, n_fine)
    return h_fine

def downsample(h, n_coarse):
    """Downsample from fine to coarse"""
    n_fine = len(h)
    ratio = n_fine // n_coarse
    h_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        start = i * ratio
        end = (i + 1) * ratio if i < n_coarse - 1 else n_fine
        h_coarse[i] = h[start:end].mean()
    h_coarse = project_to_simplex_box(h_coarse, n_coarse)
    return h_coarse

def shift_h(h, shift):
    """Circular shift h by 'shift' positions"""
    return np.roll(h, shift)

import numpy as np
from scipy.optimize import minimize
import time

def project(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_val(h, n):
    return float((np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)).max())

def obj_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    mx = vals.max()
    ex = np.exp((vals - mx) / temp)
    s = ex.sum()
    w = ex / s
    f = mx + temp * np.log(s)
    g = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = lags[k]
        if lag >= 0:
            g[lag:] += wk * dx * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * dx * h[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * dx * (1.0 - h[ll:])
            g[ll:] -= wk * dx * h[:n - ll]
    return f, g

def pgd(h, n, iters, lr, temp, decay=0.9995):
    h = project(h.copy(), n)
    bf = c5_val(h, n)
    bh = h.copy()
    step = lr
    stall = 0
    t = temp
    for _ in range(iters):
        t = max(t * decay, 0.001)
        _, g = obj_grad(h, n, t)
        h_new = project(h - step * g, n)
        fn = c5_val(h_new, n)
        if fn < bf - 1e-14:
            bf, bh = fn, h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.5)
            stall = 0
        else:
            step *= 0.7
            stall += 1
            if stall > 300:
                break
    return bh, bf

def upscale(h_c, n_f):
    idx = (np.arange(n_f) * len(h_c)) // n_f
    return project(h_c[idx].copy(), n_f)

def slsqp_solve(h, n):
    dx = 2.0 / n
    def fun(x):
        return float((np.correlate(x, 1.0 - x, mode="full") * dx).max())
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        k = int(corr.argmax())
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += dx * (1.0 - x[:n - lag])
            g[:n - lag] -= dx * x[lag:]
        else:
            ll = -lag
            g[:n - ll] += dx * (1.0 - x[ll:])
            g[ll:] -= dx * x[:n - ll]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 800, 'ftol': 1e-12})
        if res.success:
            hp = project(res.x, n)
            return hp, c5_val(hp, n)
    except Exception:
        pass
    return h.copy(), c5_val(h, n)
