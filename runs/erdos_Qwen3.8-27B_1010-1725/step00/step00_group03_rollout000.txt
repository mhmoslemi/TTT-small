import numpy as np
import scipy.optimize as opt
import time

_opt_n = 100

def _eq_fun(x):
    return np.sum(x) - _opt_n / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def upsample_initial(n_target, h_init):
    n0 = len(h_init)
    xs = np.linspace(0.0, 2.0, n0, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    up = np.interp(xt, xs, h_init)
    up = np.clip(up, 0.0, 1.0)
    diff = n_target / 2.0 - up.sum()
    if abs(diff) > 1e-12:
        mask = (up < 1.0) if diff > 0 else (up > 0.0)
        if mask.sum() > 0:
            up[mask] += diff / mask.sum()
            up = np.clip(up, 0.0, 1.0)
    return up

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def c5_smoothed(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return float(m + temp * np.log(np.exp((vals - m) / temp).sum()))

def c5_smoothed_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    wts = np.exp((vals - m) / temp)
    wts /= wts.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = wts[k] * dx
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        ja_s = max(0, lag)
        ja_e = min(n, n + lag)
        if ja_s < ja_e:
            idx = np.arange(ja_s, ja_e)
            grad[idx] += w * (1.0 - h[idx - lag])
        jb_s = max(0, -lag)
        jb_e = min(n, n - lag)
        if jb_s < jb_e:
            idx = np.arange(jb_s, jb_e)
            grad[idx] -= w * h[idx + lag]
    return grad

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-12:
                m2 = (h > 1e-9) & (h < 1.0 - 1e-9)
                if m2.sum() > 0:
                    h[m2] += diff2 / m2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def make_block_h(n, p1, p0, rng):
    h = np.zeros(n)
    i = 0
    while i < n:
        l1 = max(1, int(rng.exponential(max(1, p1))) + 1)
        h[i:min(i + l1, n)] = 1.0
        i += l1
        if i >= n:

        l0 = max(1, int(rng.exponential(max(1, p0))) + 1)
        h[i:min(i + l0, n)] = 0.0
        i += l0
    return fix_sum(h, n)

    return c5_smoothed(x, temp)

def _grad_wrap(x, temp):
