        target = n / 2.0
        diff = target - np.sum(h_out)
        if abs(diff) > 1e-10:
            idx = np.argmin(np.abs(h_out - 0.5))
            h_out[idx] = np.clip(h_out[idx] + diff, 0.0, 1.0)
        best_c5 = compute_c5(h_out)
        return h_out, float(best_c5), int(n)
