1. Use the "soft max" / log-sum-exp as a smooth approximation
2. Minimize it with gradient descent + projection
3. The temperature controls the approximation quality
4. An anneal the temperature: start high (smooth, easy to minimize) → end low (close to true max)

This is essentially simulated annealing on the objective.

For the log-sum-exp:
L(h, β) = (1/β) * log(sum_k exp(β * c_k)) * dx

As β → ∞, this approaches max_k c_k.

The gradient is:
∂L/∂h[j] = (1/β) * (1/sum_k exp(βc_k)) * sum_k exp(βc_k) * ∂c_k/∂h[j] * dx
= sum_k [exp(βc_k)/sum_j' exp(βc_j')] * ∂c_k/∂h[j] * dx
= sum_k p_k * ∂c_k/∂h[j] * dx

where p_k = softmax(β * c_k)

This is a convex combination of the individual ∂c_k/∂h[j], so it's well-defined and smooth.

def softmax_eval_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = corr.max()
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -100, 0)

    p = exp_s / Z
    c5_soft = (max_c + np.log(Z) / beta) * dx
    grad = np.zeros(n)
    sig = np.where(p > 1e-8)[0]
    for k_idx in sig:
        grad += p[k_idx] * dc_k_over_dh(h, k_idx)
    grad *= dx
    return c5_soft, grad

    best_h = h.copy()

        return compute_c5(x)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, target):
    h = np.array(h, float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                return h
            h[mask] += diff / cnt
            np.minimum(h, 1.0, out=h)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                return h
            h[mask] += diff / cnt
            np.maximum(h, 0.0, out=h)
    return h

def dc_k_dh(h, k):
    n = len(h)
    offset = k - n + 1
    grad = np.zeros(n)
    lo1 = max(0, offset)
    hi1 = min(n, n + offset)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 - offset:hi1 - offset])
    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + offset:hi2 + offset]
    return grad

def softmax_eval(h, beta):

    p = exp_s / Z
    val = (max_c + np.log(Z) / beta) * dx
