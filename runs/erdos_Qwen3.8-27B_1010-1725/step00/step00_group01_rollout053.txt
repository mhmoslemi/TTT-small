def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    cross = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cross) * dx)

def project_constraint(h, n):
    h = np.array(h, dtype=float)
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    cross = np.correlate(h, 1.0 - h, mode="full")
    s = cross / temp
    ms = s.max()
    e = np.exp(s - ms)
    W = e / e.sum()
    obj = temp * (ms + np.log(e.sum())) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = dx * W[k]
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        lo1, hi1 = max(0, lag), min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        lo2, hi2 = max(0, -lag), min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= w * h[idx + lag]
    return obj, grad

        if sc < best_sc:

            best = h.copy()

            best_h, best_sc, best_n = h, sc, n_par

    # Phase 4: Coordinate polish on best solution
    if elapsed() < budget_s - 10:
        h, sc = coordinate_polish(best_h, best_n, max_rounds=30)
        if sc < best_sc:
            best_h, best_sc = h, sc

    # Phase 5: Try even n (allows pure binary)
    for n_try in [46, 50, 52]:
        if elapsed() > budget_s - 40:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        h_i = project_constraint(np.interp(x_new, x_old, best_h), n_try)
        h, sc = pgd(h_i, n_try, 0.06, 4000, 0.002)
        if sc < best_sc:
            best_h, best_sc, best_n = h, sc, n_try
        h, sc = coordinate_polish(h, n_try, max_rounds=20)
        if sc < best_sc:
            best_h, best_sc = h, sc

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s - 10 and best_n <= 60:
        try:
            h, sc = slsqp_opt(best_h, best_n, maxiter=3000)
            if sc < best_sc:
                best_h, best_sc, best_n = h, sc, best_n
        except Exception:
            pass

    # Final feasibility check
    best_h = project_constraint(best_h, best_n)
    best_sc = evaluate_c5(best_h)

    return (best_h, best_sc, best_n)
