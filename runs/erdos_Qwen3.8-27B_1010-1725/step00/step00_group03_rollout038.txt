    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    return np.max(corr) * dx

def compute_gradiants(h, n):
    """Compute gradients for all lags. Return gradient of argmax lag."""
    oh = 1.0 - h
    corr = np.correlate(h, oh, mode="full")
    k_star = np.argmax(corr)
    # k_star is in range [0, 2n-2]
    # The actual lag is k_star - (n-1)
    lag = k_star - (n - 1)
    # Gradient: g[j] = (1-h[j+lag]) if 0<=j+lag<n else 0
    #             - h[j-lag] if 0<=j-lag<n else 0
    g = np.zeros(n)
    if lag >= 0:
        # (1-h[j+lag]) for j such that j+lag < n, i.e., j < n-lag
        g[:n-lag] += (1.0 - h[lag:])
        # -h[j-lag] for j such that j-lag >= 0, i.e., j >= lag
        g[lag:] -= h[:n-lag]
    else:
        lag = -lag  # now positive
        # (1-h[j+lag]) for j < n-lag
        g[:n-lag] += (1.0 - h[lag:])
        # -h[j-lag] for j >= lag
        g[lag:] -= h[:n-lag]
    return g, k_star

        else:

    dx = 2.0 / n

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    e = np.exp(beta * (corr - mx))
    w = e / e.sum()
    G = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            G += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            G[:n - lag] += wk * (1.0 - h[lag:])
            G[lag:] -= wk * h[:n - lag]
        else:
            a = -lag
            G[:n - a] -= wk * h[a:]
            G[a:] += wk * (1.0 - h[:n - a])
    return G * dx

            lo = mid
        else:
            hi = mid
