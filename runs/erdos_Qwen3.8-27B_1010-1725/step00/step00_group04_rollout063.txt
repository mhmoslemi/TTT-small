import numpy as np
from scipy.optimize import minimize
import time

_c5_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(h, alpha=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.mean(np.exp(alpha * (corr - m)))) / alpha

def obj_penalty(h):
    c5 = smooth_c5(h, 80.0)
    n = len(h)
    pen = 1e6 * (np.sum(h) - 0.5 * n) ** 2
    return c5 + pen

def constraint_fun(h):
    return float(np.sum(h) - _c5_target)

def constraint_gra(h):
    return np.ones_like(h)

def repair_sum(h):
    target = 0.5 * len(h)
    h = np.asarray(h, dtype=np.float64)
    for _ in range(3000):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old, dtype=np.float64) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new, dtype=np.float64) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return repair_sum(np.clip(h_new, 0.0, 1.0))

def make_starts(h0, rng, n_t, num_starts=8):
    starts = [h0.copy()]
    for i in range(3):
        amp = 0.02 * (1 + i * 0.5)
        hp = h0 + rng.uniform(-amp, amp, n_t)
        starts.append(repair_sum(np.clip(hp, 0.0, 1.0)))
    bs = max(2, n_t // 12)
    hb = h0.copy()
    for b in range(0, n_t - bs, bs):
        hb[b:b+bs] += rng.uniform(-0.12, 0.12)
    starts.append(repair_sum(np.clip(hb, 0.0, 1.0)))
    hsin = h0.copy()
    f = rng.uniform(1, n_t // 4)
    ph = rng.uniform(0, 2 * np.pi)
    hsin += 0.05 * np.sin(2 * np.pi * f * np.arange(n_t) / n_t + ph)
    starts.append(repair_sum(np.clip(hsin, 0.0, 1.0)))
    hbin = h0.copy()
    mask = rng.random(n_t) < 0.08
    hbin[mask] = 1.0 - hbin[mask]
    starts.append(repair_sum(np.clip(hbin, 0.0, 1.0)))
    hr = rng.uniform(0.0, 1.0, n_t)
    hr = repair_sum(np.clip(hr, 0.0, 1.0))
    starts.append(0.7 * h0 + 0.3 * hr)
    starts[-1] = repair_sum(np.clip(starts[-1], 0.0, 1.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    global _c5_target
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=np.float64)
    else:
        h_best = np.full(42, 0.5)

    h_best = repair_sum(h_best)
    c5_best = compute_c5(h_best)
    n_best = len(h_best)

    base_n = n_best
    resolutions = []
    for pw in range(5):
        r = base_n * (2 ** pw)
        if r <= 336 and (not resolutions or r != resolutions[-1]):
            resolutions.append(r)

    for n_t in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break

        if n_t == n_best:
            h0 = h_best.copy()
        else:
            h0 = upsample_h(h_best, n_t)

        _c5_target = 0.5 * n_t
        bounds = [(0.0, 1.0)] * n_t

        starts = make_starts(h0, rng, n_t)

        for h_s in starts:
            remaining = budget_s - (time.time() - t0)
            if remaining < 25:
                break
            try:
                res = minimize(
                    obj_penalty, h_s, method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 600, 'ftol': 1e-15, 'maxls': 40, 'iprint': -1}
                )
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = repair_sum(h_opt)
                c5_v = compute_c5(h_opt)
                if c5_v < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_v
                    n_best = n_t
                    h0 = h_best.copy()
            except Exception:
                pass

        if n_t <= 200 and (budget_s - (time.time() - t0)) > 40:
            try:
                cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_gra}]
                res2 = minimize(
                    compute_c5, h0, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 300, 'ftol': 1e-14, 'iprint': -1}
                )
                h_opt = np.clip(res2.x, 0.0, 1.0)
                h_opt = repair_sum(h_opt)
                c5_v = compute_c5(h_opt)
                if c5_v < c5_best:

                    c5_best = c5_v
                    n_best = n_t
                    h0 = h_best.copy()
            except Exception:
                pass

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        _c5_target = 0.5 * n_best
        bounds_f = [(0.0, 1.0)] * n_best
        try:
            cons_f = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_gra}]
            res3 = minimize(
                compute_c5, h_best, method='SLSQP', bounds=bounds_f,
                constraints=cons_f, options={'maxiter': 200, 'ftol': 1e-14, 'iprint': -1}
            )
            h_opt = np.clip(res3.x, 0.0, 1.0)
            h_opt = repair_sum(h_opt)
            c5_v = compute_c5(h_opt)
            if c5_v < c5_best:

        except Exception:
            pass

    h_best = repair_sum(np.clip(h_best, 0.0, 1.0))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
