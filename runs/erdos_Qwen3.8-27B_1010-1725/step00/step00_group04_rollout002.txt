import time

_gN = 42

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, N):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = N * 0.5
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        # Water-filling: distribute diff to elements not at bounds
        free = (h > 1e-10) & (h < 1.0 - 1e-10) if diff < 0 else (h > 1e-10) & (h < 1.0)
        # Actually, for adding: free = h < 1; for subtracting: free = h > 0
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        n_free = np.sum(free)
        if n_free == 0:
            break
        h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)

        h[i] += eps
        g[i] = (gp - gm) / (2.0 * eps)

import numpy as np
from scipy.optimize import minimize
import time

_gN = 42

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, N):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = N * 0.5
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            break
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x):
    return compute_c5(repair_h(x, _gN))

def num_gradient(h):
    N = _gN
    g = np.zeros(N)
    eps = 1e-6
    for i in range(N):
        h[i] += eps
        gp = compute_c5(h)
        h[i] -= 2.0 * eps
        gm = compute_c5(h)
        h[i] += eps
        g[i] = (gp - gm) / (2.0 * eps)
    g -= np.mean(g)
    return g

def project_step(h, g, lr):
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_h(h_new, len(h))
    return h_new
