import time

def compute_c5_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, t=50.0):
    """Smooth max approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for idx, w in enumerate(weights):
        if w < 1e-8:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                grad[j] += w * (1.0 - h[j - lag])
                grad[j - lag] += w * (-h[j])
        elif lag < 0:
            m = -lag
            for j in range(n - m):
                grad[j] += w * (1.0 - h[j + m])
                grad[j + m] += w * (-h[j])
        else:
            grad += w * (1.0 - 2.0 * h)
    
    return smooth_obj(h, t), grad * dx

        res = minimize(smooth_obj, h0, args=(n, t), jac=lambda x: smooth_grad(x, n, t)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_feasible(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return h0

def optimize_slsqp(h0, n, maxiter=300):
    dx = 2.0 / n
    target_sum = n / 2.0
    def fun(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(c) * dx)
    def jac(x):
        g, _ = get_subgradient(x, n)
        return g * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return h0

def subgradient_descent(h0, n, n_iter, target_sum):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_exact(h)
    for it in range(n_iter):
        grad, val = get_multi_subgradient(h, n, k=3)
        c5 = val * dx
        if c5 < best_val:
            best_val = c5
            best_h = h.copy()
        lr = 0.8 / (1.0 + 0.02 * it)
        h = h - lr * grad * dx
        h = project_feasible(h, target_sum)
    return best_h, best_val

def binary_local_search(h0, n, target_sum, max_moves=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    best_c5 = compute_c5_exact(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(max_moves):
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = rng.choice(ones)
        zi = rng.choice(zeros)
        h[oi], h[zi] = 0.0, 1.0
        c = compute_c5_exact(h)
        if c < best_c5:
            best_c5 = c
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[oi], h[zi] = 1.0, 0.0
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    time_limit = budget_s - 10.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target_sum = n / 2.0
    h_init = project_feasible(h_init, target_sum)

    best_h = h_init.copy()
    best_c5 = compute_c5_exact(h_init)

    def try_update(h_cand):
        nonlocal best_h, best_c5
        h_cand = project_feasible(np.clip(h_cand, 0, 1), target_sum)
        c = compute_c5_exact(h_cand)
        if c < best_c5:
            best_c5 = c
            best_h = h_cand
        return best_c5

    # Phase 1: Polish initial with multiple methods
    h_s = optimize_slsqp(best_h, n, 500)
    try_update(h_s)
    h_sm = optimize_lsqp_smooth(best_h, n, 300, t=80.0)
    try_update(h_sm)
    h_sg, _ = subgradient_descent(best_h, n, 500, target_sum)
    try_update(h_sg)
    h_s2 = optimize_slsqp(best_h, n, 300)
    try_update(h_s2)

    # Phase 2: Random binary restarts + local search + SLSQP
    n_restarts = 60
    for r in range(n_restarts):
        if time.time() - t0 > time_limit:
            break
        h_r = np.zeros(n)
        idx = rng.choice(n, size=int(target_sum), replace=False)
        h_r[idx] = 1.0
        # Binary local search
        h_bl, _ = binary_local_search(h_r, n, target_sum, max_moves=200, rng=rng)
        # SLSQP polish (allowing continuous values)
        h_pol = optimize_slsqp(h_bl, n, 200)
        c_r = try_update(h_pol)
        # If good, do more polishing
        if c_r < 0.385:
            h_sm2 = optimize_lsqp_smooth(h_pol, n, 400, t=100.0)
            try_update(h_sm2)
            h_s3 = optimize_slsqp(best_h, n, 400)
            try_update(h_s3)

    # Phase 3: Structured patterns
    if time.time() - t0 < time_limit:
        # Alternating with small blocks
        for block in [2, 3, 4, 5, 6, 7]:
            h_b = np.zeros(n)
            period = 2 * block
            fill = int(n / period)
            for i in range(fill * period + min(block, n % period)):
                if i < n and (i % period) < block:
                    h_b[i] = 1.0
            h_b = project_feasible(h_b, target_sum)
            h_pol = optimize_slsqp(h_b, n, 300)
            try_update(h_pol)

    # Phase 4: Perturbations from best
    for scale in [0.05, 0.1, 0.15, 0.2]:
        for _ in range(20):
            if time.time() - t0 > time_limit:
                break
            h_p = best_h + rng.normal(0, scale, n)
            h_p = project_feasible(h_p, target_sum)
            h_p = optimize_slsqp(h_p, n, 150)
            try_update(h_p)

    # Phase 5: Final intensive polishing
    if time.time() - t0 < time_limit:
        for t_val in [30.0, 50.0, 100.0, 200.0]:
            h_f = optimize_lsqp_smooth(best_h, n, 300, t=t_val)
            try_update(h_f)
        h_f = optimize_slsqp(best_h, n, 500)
        try_update(h_f)
        # Subgradient + SLSQP cycle
        for _ in range(5):
            if time.time() - t0 > time_limit:
                break
            h_sg2, _ = subgradient_descent(best_h, n, 200, target_sum)
            try_update(h_sg2)
            h_f2 = optimize_slsqp(best_h, n, 300)
            try_update(h_f2)

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5_exact(best_h)
    return best_h, best_c5, n
