import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h[mask] = np.maximum(h[mask], 0.0)
    return h


def obj_func(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)


def constraint_eq(x, n):
    return np.sum(x) - n / 2.0


def generate_structured_start(n, rng):
    kind = int(rng.integers(5))
    if kind == 0:
        h = rng.uniform(0, 1, n)
    elif kind == 1:
        freq = int(rng.integers(1, 12))

    elif kind == 2:
        nblocks = int(rng.integers(2, 9))

        h = np.tile(vals, n // nblocks + 1)[:n]
    elif kind == 3:

        nmid = n // 5

        h[idx] = rng.uniform(0.1, 0.9, nmid)
    else:
        h = np.zeros(n)
        m = n // 4
        h[:m] = 1.0
        h[n - m:] = 1.0
        h += rng.normal(0, 0.15, n)

    constraints = [{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}]
    try:

                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h_start
