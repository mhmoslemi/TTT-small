        starts.append(0.5 + 0.3 * np.cos(np.pi * x / 2.0) * np.sin(np.pi * x))
        starts.append(0.5 + 0.2 * np.cos(2 * np.pi * x - 1.0))
        for h_s in starts:
            if tl() < 5:
                break
            h_try = project_feasible(h_s, n0)
            try:
                res = minimize(compute_c5, h_try, method='Nelder-Mead',
                              options={'maxiter': 800, 'xatol': 1e-9, 'fatol': 1e-12})
                h_new = project_feasible(res.x, n0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            except Exception:
                pass

    # Phase 5: Higher resolution
    for factor in [2, 3]:
        n_new = n0 * factor
        if n_new > 950 or tl() < 80:
            break
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        bounds_new = [(0.0, 1.0)] * n_new
        cons_new = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian}]

        if tl() > 50:
            try:
                res = minimize(make_smooth_obj, h_interp, method='SLSQP',
                              bounds=bounds_new, constraints=cons_new,
                              options={'maxiter': 4000, 'ftol': 1e-13})
                h_new = project_feasible(res.x, n_new)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            except Exception:
                pass

        if tl() > 40:
            try:
                res = minimize(compute_c5, h_interp, method='Nelder-Mead',
                              options={'maxiter': 600, 'xatol': 1e-9, 'fatol': 1e-11})
                h_new = project_feasible(res.x, n_new)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            except Exception:
                pass

    # Final polish
    if tl() > 5:
        try:
            res = minimize(compute_c5, best_h, method='Nelder-Mead',
                          options={'maxiter': 400, 'xatol': 1e-10, 'fatol': 1e-13})
            h_new = project_feasible(res.x, len(best_h))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
