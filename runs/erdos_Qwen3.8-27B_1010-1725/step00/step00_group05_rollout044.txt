import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, beta):
    corr = get_correlations(h)
    # Soft max with beta scaling
    m = np.max(corr)
    return (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - m)))) + m

def slsqp_optimize(h0, n, beta, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        return soft_max_obj(x, n, beta)
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds, 
                   constraints=cons, maxiter=max_iter, options={'ftol': 1e-10})
    
    if res.success or hasattr(res, 'x'):
        h_opt = project_feasible(res.x, n)
        return h_opt
    return None

    h = rng.uniform(0, 1, n)

        h[i+block_size:i+2*block_size] = rng.uniform(0.0, 0.4, min(block_size, n-i-block_size))
    return project_feasible(h, n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def soft_max_objective(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / beta, 50.0 / beta)
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def slsqp_optimize(h0, n, beta, max_iter=150):
    target = n / 2.0
    c0 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(soft_max_objective, h0.astype(float).copy(), args=(n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       maxiter=max_iter, options={'ftol': 1e-10, 'maxls': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new)
        if c_new <= c0 + 1e-8:
            return h_new
    except Exception:
        pass
    return None

def binary_local_search(n, rng, max_iter=4000, patience=200):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    best = compute_c5(h)
    no_improve = 0
    for _ in range(max_iter):
        if no_improve > patience:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best - 1e-12:
            best = c5
            no_improve = 0
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
    return h.copy(), best
