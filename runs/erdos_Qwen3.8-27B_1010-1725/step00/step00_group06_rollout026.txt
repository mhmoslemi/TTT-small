import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)
    grad = np.zeros(n)
    i1s = max(0, m)
    i1e = min(n, n + m)
    if i1e > i1s:
        grad[i1s:i1e] += (1.0 - h[i1s - m:i1e - m])
    i2s = max(0, -m)
    i2e = min(n, n - m)
    if i2e > i2s:
        grad[i2s:i2e] -= h[i2s + m:i2e + m]
    return dx * grad

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h.copy()
        h *= (target / s)
    else:
        h = np.full(n, 0.5)
    return np.clip(h, 0.0, 1.0)

def make_binary(n, sv):
    rng = np.random.RandomState(sv)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def make_random(n, sv):
    rng = np.random.RandomState(sv)
    return fix_sum(rng.uniform(0.2, 0.8, n))

def obj_func(h):
    return compute_c5(h)

def obj_grad_func(h):
    return get_gradient(h)

def cons_func(h):
    return np.sum(h) - len(h) / 2.0

def cons_jac(h):
    return np.ones(len(h))

def slsqp_solve(h0, max_iter=200, ftol=1e-10):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_func, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h0, jac=obj_grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        return fix_sum(res.x)
    except Exception:
        return h0

def binary_local_search(h_bin, max_iters=2000):
    n = len(h_bin)
    h = h_bin.copy()
    c5 = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for _ in range(max_iters):
        improved = False
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    ones.remove(oi)
                    ones.append(zi)
                    zeros.remove(zi)
                    zeros.append(oi)
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5

    t0 = time.time()
    np.random.seed(seed)

                best_h = h_opt.copy()
