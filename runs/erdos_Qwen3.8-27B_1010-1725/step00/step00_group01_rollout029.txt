import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def objective_and_grad(h, n, eps=0.005):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # log-sum-exp for smooth max
    cmax = np.max(corr)
    exp_corr = np.exp((corr - cmax) / eps)
    obj = (cmax + eps * np.log(np.sum(exp_corr))) * dx
    weights = exp_corr / np.sum(exp_corr)
    
    # Gradient
    grad = np.zeros(n)
    # For each shift m, shift = m - (n-1)
    # d corr[m] / d h[j] = (1-h[j-shift])*I(0<=j-shift<n) - h[j+shift]*I(0<=j+shift<n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        shift = m - (n - 1)
        for j in range(n):
            g = 0.0
            jm = j - shift
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jp = j + shift
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += w * g
    grad *= dx
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    exp_c = np.exp((corr - cmax) / eps)
    s = np.sum(exp_c)
    obj = (cmax + eps * np.log(s)) * dx
    w = exp_c / s
    grad = np.zeros(n)
    offset = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - offset
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            ss = -k
            grad[:n - ss] += wm * (1.0 - h[ss:])
            grad[ss:] -= wm * h[:n - ss]
    return obj, grad * dx

    best_c5 = compute_c5(best_h)

    return (best_h, round(best_c5, 10), best_n)
