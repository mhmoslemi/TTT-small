import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_gradient(h, k_max):
    """Subgradient of c[k_max] w.r.t. h."""
    n = len(h)
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] += (-h[:n - shift])
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    return grad

def upscale_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_sum(h, target):
    """Adjust h to have sum=target while staying in [0,1]."""
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Add/subtract uniformly from non-saturated entries
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])

        h[j] = 1.0

            stall = 0
        else:

            h[j] = 0.0
            stall += 1

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_gradient(h, k_max):
    n = len(h)
    shift = (n - 1) - k_max
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] += (-h[:n - shift])
    else:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    return grad

def upscale_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h.copy()
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
