        _active_set(_p['best_h'].copy(), _p['best_n'], t0, budget)

    bh = np.clip(_p['best_h'], 0.0, 1.0)
    bn = _p['best_n']
    s = np.sum(bh)
    if s > 1e-10:
        bh = bh * (bn / 2.0) / s
    bh = np.clip(bh, 0.0, 1.0)
    bc5 = compute_c5(bh)

    return (bh, bc5, bn)
