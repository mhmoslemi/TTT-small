def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def binarize(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    if n % 2 == 0:
        out[idx[:n // 2]] = 1.0
    else:
        out[idx[:n // 2]] = 1.0
