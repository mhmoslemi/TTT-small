    return np.ones(x.size)

def project_sum(h, target):
    h = h.astype(float).copy()
    n = h.size
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = float(mask.sum())
        if cnt < 1.0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def smooth_start(n, rng):
    x = np.linspace(0.0, 2.0, n)
    f = rng.uniform(1.0, 6.0)
    p = rng.uniform(0.0, 2.0 * np.pi)
    h = 0.5 + 0.35 * np.sin(f * x + p) + 0.15 * np.cos(2.0 * f * x + 0.7 * p)
    return project_sum(np.clip(h, 0.0, 1.0), n * 0.5)

def slsqp_try(h_start, n, cons, bounds, obj_fn, maxiter, ftol):
    target = n * 0.5
    try:
        res = minimize(obj_fn, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(res, 'x') and res.x is not None:
            ht = project_sum(np.clip(res.x, 0.0, 1.0), target)
            return ht, compute_c5(ht)
    except Exception:
        pass
    return None, float('inf')

def coord_descend(h_best, n, target, max_iters, budget_fn):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    eps_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for sweep in range(max_iters):
        if budget_fn() < 8.0:
            break
        improved = False
        for j in range(n):
            for eps in eps_list:
                for sgn in (1.0, -1.0):
                    nv = best_h[j] + sgn * eps
                    if nv < 0.0 or nv > 1.0:
                        continue
                    h_t = best_h.copy()
                    h_t[j] = nv
                    h_t = project_sum(h_t, target)
                    c5_t = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-14:
                        best_h = h_t
                        best_c5 = c5_t
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = h0.size
    target = n * 0.5
    h0 = project_sum(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP hard-max from initial
    if tleft() > 10:
        ht, c5 = slsqp_try(h0, n, cons, bounds, compute_c5, 3000, 1e-14)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 2: Soft-max refinement (two alpha values)
    if tleft() > 10:
        ht, c5 = slsqp_try(best_h, n, cons, bounds, soft_c5_50, 1500, 1e-12)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5
    if tleft() > 10:
        ht, c5 = slsqp_try(best_h, n, cons, bounds, soft_c5_200, 1000, 1e-12)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5

    # Phase 3: Diverse starting points
    starts = []
    for _ in range(8):
        starts.append(project_sum(binary_start(n, rng), target))
    for _ in range(6):
        starts.append(smooth_start(n, rng))
    for _ in range(8):
        starts.append(project_sum(best_h + rng.normal(0, 0.25, n), target))

    for h_s in starts:
        if tleft() < 15:
            break
        ht, c5 = slsqp_try(h_s, n, cons, bounds, compute_c5, 800, 1e-12)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5
        if tleft() < 15:
            break
        ht2, c52 = slsqp_try(h_s, n, cons, bounds, soft_c5_50, 500, 1e-10)
        if ht2 is not None and c52 < best_c5:
            best_h, best_c5 = ht2, c52

    # Phase 4: More random restarts around current best
    for i in range(80):
        if tleft() < 12:
            break
        scale = 0.03 + 0.25 * rng.rand()
        h_r = project_sum(best_h + rng.normal(0, scale, n), target)
        ht, c5 = slsqp_try(h_r, n, cons, bounds, compute_c5, 600, 1e-11)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5
            if tleft() < 12:
                break
            ht2, c52 = slsqp_try(ht, n, cons, bounds, soft_c5_200, 400, 1e-11)
            if ht2 is not None and c52 < best_c5:
                best_h, best_c5 = ht2, c52

    # Phase 5: Coordinate descent polish
    if tleft() > 10:
        best_h, best_c5 = coord_descend(best_h, n, target, 3000, tleft)

    # Phase 6: Final SLSQP from coordinate-descent result
    if tleft() > 10:
        ht, c5 = slsqp_try(best_h, n, cons, bounds, compute_c5, 2000, 1e-14)
        if ht is not None and c5 < best_c5:
            best_h, best_c5 = ht, c5
    if tleft() > 5:
        best_h, best_c5 = coord_descend(best_h, n, target, 2000, tleft)

    # Ensure final feasibility
    best_h = project_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
