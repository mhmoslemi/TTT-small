import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def smooth_max(vals, t):
    tv = t * vals
    mx = float(np.max(tv))
    if mx > 500:
        return mx / t
    return (mx + np.log(np.sum(np.exp(tv - mx)))) / t


def obj_func(x, t, dx):
    g = 1.0 - x
    c = np.correlate(x, g, mode="full")
    return smooth_max(c, t) * dx


def sum_eq(x, target):
    return np.sum(x) - target


def project_sum(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += d / cnt
        else:
            m = h > 0.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_lin(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old, dtype=float)
    xn = np.linspace(0.0, n_old - 1.0, n_new)
    return project_sum(np.interp(xn, xo, h), n_new)


def do_optimize(h0, n, t_lo, t_hi, n_phase, max_iter):
    dx = 2.0 / n
    tgt = n * 0.5
    h = project_sum(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq, 'args': (tgt,)}]
    for i in range(n_phase):
        frac = i / max(n_phase - 1, 1)
        t = t_lo * (t_hi / t_lo) ** frac
        try:
            res = minimize(obj_func, h, args=(t, dx), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': max_iter, 'ftol': 1e-12})
            h_new = project_sum(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5


def perturb_h(h, n, sigma):
    return project_sum(h + sigma * np.random.randn(n), n)


def cos_pattern(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    h = 0.5 + 0.3 * np.cos(x)
    return project_sum(h, n)
