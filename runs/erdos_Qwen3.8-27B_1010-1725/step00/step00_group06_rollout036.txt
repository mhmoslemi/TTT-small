```python
def projected_subgradient(h, step_size, n):
    dx = 2.0 / n
    # Compute all correlations
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    
    k_star = np.argmax(corr)  # index in the 2n-1 array
    k = k_star - (n - 1)  # actual shift
    
    # Compute gradient
    grad = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        for j in range(n):
            g = 0.0
            if k <= j <= n - 1:  # j-k is valid: 0 <= j-k <= n-1-k
                # Actually: 0 <= j-k and j-k <= n-1-k means k <= j <= n-1
                g += (1.0 - h[j-k])
            if 0 <= j <= n-1-k:
                g -= h[j+k]
            grad[j] = g
```

```python
            best_local = c5_new
            step = min(step * 1.05, 0.2)
        else:
            step *= 0.9
```

```python
            if tk == j and t == j:  # k=0, t=j
                # term: h[j]*(1-h[j]) = x - x²
                quad = -1.0
                A += 1.0
            elif tk == j:  # t+k=j, t=j-k
                # term: x*(1-h[t]) = x*(1-h[j-k])
                A += (1.0 - h[j-k])
            elif t == j:  # t=j
                # term: h[j+k]*(1-x) = h[j+k] - h[j+k]*x
                A -= h[j+k]
                C_k += h[j+k]
            else:
                C_k += h[tk] * (1.0 - h[t])
        funcs.append((quad, A, C_k))
    
    # Now minimize max of all funcs over x in [0,1]
    # For efficiency, use a fine grid + local refinement
    x_grid = np.linspace(0, 1, 501)
    best_val = np.inf
    best_x = 0.5
    for x in x_grid:
        val = 0.0
        for quad, A, B in funcs:
            v = quad*x*x + A*x + B
            if v > val:
                val = v
        if val < best_val:
            best_val = val
            best_x = x
    
    h[j] = best_x
    return best_x
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coordinate_sweep(h, n, penalty=100.0, n_grid=301):
    """One coordinate descent sweep. Returns updated h and c5."""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    center = n - 1
    target_sum = n * 0.5
    
    for j in range(n):
        h_j = h[j]
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        A = np.zeros(ncorr)
        B = np.zeros(ncorr)
        q = np.zeros(ncorr)
        
        # k=0
        q[center] = -1.0
        A[center] = 1.0
        B[center] = corr[center] - h_j * (1.0 - h_j)
        
        # Positive shifts
        for s in range(1, n):
            m = n - s
            I1 = 1 if (s <= j) else 0
            I2 = 1 if (j < m) else 0
            idx = center + s
            B[idx] = corr[idx]
            if I1:
                B[idx] -= h_j * (1.0 - h[j - s])
                A[idx] += (1.0 - h[j - s])
            if I2:
                B[idx] -= h[j + s] * (1.0 - h_j)
                A[idx] -= h[j + s]
        
        # Negative shifts
        for s in range(1, n):
            m = n - s
            I1 = 1 if (j < m) else 0
            I2 = 1 if (s <= j) else 0
            idx = center - s
            B[idx] = corr[idx]
            if I1:
                B[idx] -= h_j * (1.0 - h[j + s])
                A[idx] += (1.0 - h[j + s])
            if I2:
                B[idx] -= h[j - s] * (1.0 - h_j)
                A[idx] -= h[j - s]
        
        # Add penalty for sum constraint
        d = np.sum(h) - h_j - target_sum  # sum of other coords minus target
        q[:] += penalty
        A[:] += 2.0 * penalty * d
        B[:] += penalty * d * d
        
        # Minimize over x in [0,1]
        xg = np.linspace(0.0, 1.0, n_grid)
        xm = xg[:, None]
        vals = q[None, :] * xm**2 + A[None, :] * xm + B[None, :]
        mv = np.max(vals, axis=1)
        bi = np.argmin(mv)
        
        lo = max(0, bi - 3)
        hi = min(n_grid - 1, bi + 3)
        xf = np.linspace(xg[lo], xg[hi], 101)
        xfm = xf[:, None]
        vf = q[None, :] * xfm**2 + A[None, :] * xfm + B[None, :]
        mf = np.max(vf, axis=1)
        h[j] = xf[np.argmin(mf)]
    
    return h
```
