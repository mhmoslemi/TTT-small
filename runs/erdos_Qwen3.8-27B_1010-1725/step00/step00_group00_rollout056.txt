    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = t * corr
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    exp_sum = exp_tc.sum()
    softmax = exp_tc / exp_sum
    obj = (tc_max + np.log(exp_sum)) / t * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = softmax[k] * dx
        if w < 1e-14:
            continue
        lag = m - k
        for i in range(n):
            idx1 = i + k - m  # i - lag
            idx2 = i - k + m  # i + lag
            g = 0.0
            if 0 <= idx2 <= n - 1:
                g += 1.0 - h[idx2]
            if 0 <= idx1 <= n - 1:
                g -= h[idx1]
            grad[i] += w * g
    return obj, grad

def _obj(h):
    return soft_obj_and_grad(h, _g_t)[0]

def _grad(h):
    return soft_obj_and_grad(h, _g_t)[1]

def _cons_eq(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(len(h))

def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / float(mask.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def _try_optimize(h_start, n, bounds, cons, t_list, maxiter, ftol):
    global _g_t, _g_n
    best_h = h_start.copy()
    best_obj = compute_c5(best_h)
    for t in t_list:
        _g_t = t
        _g_n = n
        res = minimize(_obj, best_h, jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_h(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_obj:
                best_h = h_new
                best_obj = c5_new
    return best_h, best_obj

def _random_start(n, rng):
    h = rng.random(n)
    h = project_h(h, n)
    return h

def _perturb(h, n, scale, rng):
    h_p = h + rng.normal(0.0, scale, n)
    return project_h(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_t, _g_n
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(45, 0.5)

    n = len(h0)
    _g_n = n
    best_h = project_h(h0, n)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]

    # Phase 1: Annealing from initial
    t_list_full = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 3000.0]
    best_h, best_c5 = _try_optimize(best_h, n, bounds, cons, t_list_full, 250, 1e-14)

    # Phase 2: Multiple perturbations & re-optimization
    phase2_deadline = t_start + budget_s * 0.80
    attempt = 0
    while time.time() < phase2_deadline and attempt < 80:
        attempt += 1
        scale = 0.05 + 0.10 * rng.random()
        h_pert = _perturb(best_h, n, scale, rng)
        t_sub = [100.0, 500.0, 2000.0]
        if attempt % 5 == 0:
            h_pert = _random_start(n, rng)
            t_sub = [10.0, 50.0, 200.0, 1000.0]
        h_new, c5_new = _try_optimize(h_pert, n, bounds, cons, t_sub, 120, 1e-13)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Final fine-tuning at high t
    if time.time() < t_start + budget_s * 0.92:
        _g_t = 5000.0
        _g_n = n
        res = minimize(_obj, best_h, jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 400, 'ftol': 1e-15})
        if res.x is not None:
            h_new = project_h(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Try rounding to nearby grid and re-optimizing
    if time.time() < t_start + budget_s * 0.96:
        grid_vals = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])
        h_round = np.zeros(n)
        for i in range(n):
            h_round[i] = grid_vals[np.argmin(np.abs(grid_vals - best_h[i]))]
        h_round = project_h(h_round, n)
        h_new, c5_new = _try_optimize(h_round, n, bounds, cons,
                                       [50.0, 500.0, 3000.0], 150, 1e-14)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
