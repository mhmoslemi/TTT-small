def compute_c5(h, dx):
    """Compute C5 for given h values."""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h[i]<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute difference
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        room = 1 - h
        total_room = room.sum()
        if total_room < 1e-10:
            return h
        h += diff * room / total_room
    else:
        # Need to decrease sum: subtract from elements that are > 0
        room = h
        total_room = room.sum()
        if total_room < 1e-10:
            return h
        h += diff * room / total_room
    h = np.clip(h, 0, 1)
    # Fine adjustment
    diff = target - h.sum()
    h += diff / n
    return np.clip(h, 0, 1)

The problem says: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

Given two 1-D arrays a and v, compute the full cross-correlation, which is defined as:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_initial(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def make_structured(n, variant=0):
    h = np.zeros(n)
    half = n // 2
    if variant == 0:
        h[0::2] = 1.0
        if h.sum() > half:
            h[h > 0][-1] = 0
    elif variant == 1:
        block = n // 4
        h[:block] = 1.0
        h[n//2:n//2 + block] = 1.0
    elif variant == 2:
        period = 3
        for i in range(n):
            if i % period == 0:
                h[i] = 1.0
        if h.sum() > half:
            idx = np.where(h == 1)[0]
            h[idx[-1]] = 0
        elif h.sum() < half:
            idx = np.where(h == 0)[0]
            h[idx[0]] = 1
    return h

def make_from_guide(n, guide):
    g = len(guide)
    x_g = np.linspace(0, 2, g)
    x_n = np.linspace(0, 2, n)
    h_interp = np.interp(x_n, x_g, guide)
    h = np.zeros(n)
    sorted_idx = np.argsort(-h_interp)
    h[sorted_idx[:n // 2]] = 1.0
    return h

def local_search(h, time_limit, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_h = h.copy()
    curr_c5 = best_c5
    start = time.time()
    T = 0.012
    T_min = 0.0003
    iter_count = 0

    while time.time() - start < time_limit:
        iter_count += 1
        ones = np.where(curr_h == 1)[0]
        zeros = np.where(curr_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if rng.random() < 0.75:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            curr_h[i] = 0
            curr_h[j] = 1
            new_c5 = compute_c5(curr_h)
            accept = new_c5 < curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-9))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = curr_h.copy()
            else:
                curr_h[i] = 1
                curr_h[j] = 0
        else:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            curr_h[i1] = 0
            curr_h[i2] = 0
            curr_h[j1] = 1
            curr_h[j2] = 1
            new_c5 = compute_c5(curr_h)
            accept = new_c5 < curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-9))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = curr_h.copy()
            else:
                curr_h[i1] = 1
                curr_h[i2] = 1
                curr_h[j1] = 0
                curr_h[j2] = 0

        T = max(T * 0.9997, T_min)
        if iter_count % 10000 == 0 and curr_c5 > best_c5:
            curr_h = best_h.copy()
            curr_c5 = best_c5
            for _ in range(3):
                o = np.where(curr_h == 1)[0]
                z = np.where(curr_h == 0)[0]
                if len(o) > 0 and len(z) > 0:
                    a = o[rng.randint(len(o))]
                    b = z[rng.randint(len(z))]
                    curr_h[a], curr_h[b] = curr_h[b], curr_h[a]

    return best_h, best_c5

def polish(h, max_rounds=20, time_limit=60):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_new_c5 = c5
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                if new_c5 < best_new_c5 - 1e-12:
                    best_new_c5 = new_c5
                    best_move = (i, j)
                h[i] = 1
                h[j] = 0
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0
        h[j] = 1
        c5 = best_new_c5

    return h, c5
