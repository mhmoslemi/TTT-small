from scipy.interpolate import interp1d

        return h
    # Distribute the difference
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

        mask = (h > 0) & (h < 1) if diff != 0 else np.ones(n, dtype=bool)

        if not np.any(mask):

        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)

    h = np.zeros(n)

def interpolate_to_n(h, n_new):
    """Interpolate h from its native grid to n_new points"""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = np.arange(n_old) * dx_old + dx_old / 2  # cell centers

def eval_c5(h):

def _obj(x):
