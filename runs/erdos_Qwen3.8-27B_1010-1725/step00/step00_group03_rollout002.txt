def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, n):
    h = np.round(h).astype(float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for k in range(int(diff + 0.5)):
            h[zeros[k]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for k in range(int(-diff + 0.5)):
            h[ones[k]] = 0.0
    return h

def binary_local_search(h, max_rounds=80):
    n = len(h)
    h = make_binary(h, n).copy()
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False

        zeros = np.where(h == 0)[0]
        best_val = best_c5
        best_pair = None

            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                if c5 < best_val - 1e-12:
                    best_val = c5
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = 0.0, 1.0

            improved = True

def continuous_refine(h0, max_iter=3000):

        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr)) * (2.0 / n) + pen

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0.copy(), method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h = np.clip(res.x, 0.0, 1.0)
    h = project_box_sum(h, target_sum)
    return h, compute_c5(h)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    f = interp1d(x_old, h, kind='linear', fill_value="extrapolate")
    h_new = np.clip(f(x_new), 0.0, 1.0)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new
