import numpy as np
from scipy.optimize import minimize
import time

def c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx, corr

def project(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Adjust: add diff/n to elements not at boundary
    if diff > 0:
        mask = h < 1 - 1e-10
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + adj, 1.0)
    else:
        mask = h > 1e-10
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.maximum(h[mask] + adj, 0.0)
    # Final correction: iterative
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1 - 1e-10
        else:
            mask = h > 1e-10
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0, 1)
    return h

def subgradient_c5(h, n, top_k=5):
    """Compute approximate subgradient of C5 using top-k shifts"""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    c5_val = np.max(corr) * dx
    
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        g = np.zeros(n)
        for m in range(n):
            val = 0.0
            if 0 <= m + s < n:
                val += (1 - h[m + s])
            if 0 <= m - s < n:
                val -= h[m - s]
            g[m] = val
        grad += g / top_k
    
    return c5_val, grad * dx

def coordinate_descent(h, n, max_iter=100):
    """Local search by cycling through coordinates"""
    c5_current, _ = c5_and_corr(h)
    for it in range(max_iter):
        improved = False
        perm = np.random.permutation(n)
        for m in perm:
            best_val = h[m]
            best_c5 = c5_current
            # Try a few values
            for trial in [0, 1, h[m] + 0.01, h[m] - 0.01, 0.5]:
                trial = np.clip(trial, 0, 1)
                if abs(trial - h[m]) < 1e-10:
                    continue
                h_new = h.copy()
                h_new[m] = trial
                h_new = project(h_new, n)
                c5_new, _ = c5_and_corr(h_new)
                if c5_new < best_c5 - 1e-9:
                    best_c5 = c5_new
                    best_val = trial
            if best_val != h[m]:
                h[m] = best_val
                h = project(h, n)
                c5_current = best_c5
                improved = True
        if not improved:
            break
    return h, c5_current

def adam_steps(h, n, steps=200, lr=0.01):
    """Adam-style optimization with subgradient"""
    m_assoc = np.zeros(n)
    v_assoc = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    c5_current, _ = c5_and_corr(h)
    
    for t in range(1, steps + 1):
        c5_val, grad = subgradient_c5(h, n, top_k=3)
        c5_current = min(c5_current, c5_val)
        
        m_assoc = b1 * m_assoc + (1 - b1) * grad
        v_assoc = b2 * v_assoc + (1 - b2) * grad**2
        m_hat = m_assoc / (1 - b1**t)
        v_hat = v_assoc / (1 - b2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project(h_new, n)
        
        c5_new, _ = c5_and_corr(h_new)
        if c5_new <= c5_current + 1e-8:
            h = h_new
            c5_current = min(c5_current, c5_new)
        else:
            lr *= 0.95
    
    return h, c5_current

def upsample(h, factor):
    """Upsample h by interpolation"""
    n_old = len(h)
    n_new = n_old * factor
    x_old = np.arange(n_old)
    x_new = np.arange(n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = project(h_new, n_new)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def c5_vector(h):
    """Return all correlation values (unscaled)"""
    return np.correlate(h, 1 - h, mode='full')

def coord_opt(h, n, m):
    """Optimize coordinate m exactly (1D subproblem)"""
    old = h[m]
    h[m] = 0.0
    c0 = c5_vector(h)
    h[m] = 1.0
    c1 = c5_vector(h)
    h[m] = old
    
    slope = c1 - c0  # c_s(v) = c0_s + v * slope_s
    
    # Binary search for optimal max value
    hi = float(np.max(c1))  # max at v=1
    lo = 0.0
    
    for _ in range(40):
        mid = (lo + hi) / 2.0
        # Check if exists v in [0,1] with c0_s + v*slope_s <= mid for all s
        lb, ub = 0.0, 1.0
        feasible = True
        for s in range(len(slope)):
            if slope[s] > 1e-14:
                ub = min(ub, (mid - c0[s]) / slope[s])
            elif slope[s] < -1e-14:
                lb = max(lb, (c0[s] - mid) / (-slope[s]))
            else:
                if c0[s] > mid + 1e-14:
                    feasible = False
                    break
        if lb <= ub + 1e-12 and feasible:
            hi = mid
        else:
            lo = mid
    
    C_opt = (lo + hi) / 2.0
    lb, ub = 0.0, 1.0
    for s in range(len(slope)):
        if slope[s] > 1e-14:
            ub = min(ub, (C_opt - c0[s]) / slope[s])
        elif slope[s] < -1e-14:

def adam_subgrad(h, n, steps=200, lr=0.01, top_k=3):
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for t in range(1, steps + 1):
        corr = c5_vector(h)
        c5_cur = float(np.max(corr)) * (2.0 / n)
        
        # Top-k subgradient
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in top_idx:
            s = idx - (n - 1)
            for m in range(n):
                g = 0.0
                if 0 <= m + s < n:
                    g += 1.0 - h[m + s]
                if 0 <= m - s < n:
                    g -= h[m - s]
                grad[m] += g / top_k
        
        grad *= (2.0 / n)
        
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad**2
        m_hat = m_arr / (1 - b1**t)
        v_hat = v_arr / (1 - b2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0, 1)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_opt(h, n, m):
    old = h[m]
    h[m] = 0.0
    c0 = c5_vec(h)
    h[m] = 1.0
    c1 = c5_vec(h)
    h[m] = old
    slope = c1 - c0
    lo, hi = 0.0, float(np.max(c1))
    for _ in range(40):
        mid = (lo + hi) * 0.5
        pos_idx = slope > 1e-14
        neg_idx = slope < -1e-14
        zero_idx = ~(pos_idx | neg_idx)
        ub = 1.0
        lb = 0.0
        if np.any(pos_idx):
            ub = min(ub, float(np.min((mid - c0[pos_idx]) / slope[pos_idx])))
        if np.any(neg_idx):
            lb = max(lb, float(np.max((c0[neg_idx] - mid) / (-slope[neg_idx]))))
        if zero_idx.any() and np.any(c0[zero_idx] > mid + 1e-14):
            lo = mid
            continue
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    C = (lo + hi) * 0.5
    pos_idx = slope > 1e-14
    neg_idx = slope < -1e-14
    lb, ub = 0.0, 1.0
    if np.any(pos_idx):
        ub = min(ub, float(np.min((C - c0[pos_idx]) / slope[pos_idx])))
    if np.any(neg_idx):
        lb = max(lb, float(np.max((c0[neg_idx] - C) / (-slope[neg_idx]))))
    v_opt = (lb + ub) * 0.5
    return float(np.clip(v_opt, 0.0, 1.0))

def coord_descent(h, n, sweeps=10, rng=None):
    c5_cur = compute_c5(h)
    for _ in range(sweeps):
        imp = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        else:
            np.random.shuffle(order)
        for m in order:
            v = coord_opt(h, n, m)
            if abs(v - h[m]) > 1e-6:
                h[m] = v
                project_sum(h, n)
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    imp = True
        if not imp:
            break
    return c5_cur

def subgrad_step(h, n, lr, top_k=3):
    corr = c5_vec(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        s = int(idx) - (n - 1)
        for m in range(n):
            g = 0.0
            if 0 <= m + s < n:
                g += 1.0 - h[m + s]
            if 0 <= m - s < n:
                g -= h[m - s]
            grad[m] += g / top_k
    h -= lr * grad
    h = np.clip(h, 0.0, 1.0)
    project_sum(h, n)
    return h

def binary_restart(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = rng.choice(n, size=n // 2, replace=False)
    h[pos] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = binary_restart(80, seed)
    n = len(h)
    h = project_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Subgradient steps to escape bad local minima
    lr = 0.02
    while time.time() - t0 < 120:
        for _ in range(30):
            h = subgrad_step(h, n, lr, top_k=3)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        lr *= 0.95

    # Phase 2: Coordinate descent (fast, exact 1D)
    while time.time() - t0 < 450:
        best_c5 = min(best_c5, coord_descent(best_h, n, sweeps=15, rng=rng))

    # Phase 3: Binary restarts
    while time.time() - t0 < 750:
        s = int(time.time() * 1000) + seed
        hb = binary_restart(n, s)
        c5b = coord_descent(hb, n, sweeps=8, rng=rng)
        if c5b < best_c5:
            best_h = hb.copy()
            best_c5 = c5b

    # Phase 4: Try double resolution
    if time.time() - t0 < budget_s - 250:
        n2 = n * 2
        h2 = np.zeros(n2)
        h2[0::2] = best_h
        h2 = project_sum(h2, n2)
        for _ in range(30):
            h2 = subgrad_step(h2, n2, lr=0.005, top_k=3)
        c5_2 = coord_descent(h2, n2, sweeps=10, rng=rng)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2

    # Phase 5: Final polish
    nf = len(best_h)
    while time.time() - t0 < budget_s - 10:
        best_c5 = min(best_c5, coord_descent(best_h, nf, sweeps=5, rng=rng))

    # Round near-binary values to tighten
    best_h[best_h > 0.99] = 1.0
    best_h[best_h < 0.01] = 0.0
    best_h = project_sum(best_h, nf)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, nf
