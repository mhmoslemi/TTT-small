import time

def project_onto_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - target, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_subgradient(h, n):
    """Compute subgradient of max_k corr[k] w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Active indices (within tolerance)
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    lags = active - (n - 1)
    
    grad = np.zeros(n)
    for l in lags:
        # grad[p] += (1 - h[p-l]) * [0<=p-l<n] - h[p+l] * [0<=p+l<n]
        for p in range(n):
            g = 0.0
            if 0 <= p - l < n:
                g += (1.0 - h[p - l])
            if 0 <= p + l < n:
                g -= h[p + l]
            grad[p] += g
    grad /= len(lags)
    return grad, float(max_val)

    best_c5 = compute_c5(h)

        # Project

            stuck_count = 0

= sum_i h[i] + sum_i h[i+k] - 2*sum_i h[i]h[i+k]
= (n/2) + (n/2) - 2*sum_i h[i]h[i+k]  (since sum h = n/2)
= n - 2*sum_i h[i]h[i+k]

So f_k + f_{-k} = n - 2*C_k where C_k = sum_i h[i]h[i+k].

To minimize max(f_k, f_{-k}), we want to minimize n - 2*C_k for all k, i.e., maximize C_k for all k (the autocorrelation).

This is interesting! For binary h, minimizing max_k f_k is equivalent to maximizing min_k C_k (minimum autocorrelation).

But this only works for binary h. For general h in [0,1], the relationship is more complex.

However, this gives us a clue: we want h to have a flat (high) autocorrelation spectrum. This is like a Barker sequence or a sequence with good autocorrelation properties.

For the continuous case, a good strategy might be to start from a binary solution and then smoothly interpolate.

def smooth_max_grad(h, n, T=20.0):
    corr = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    # Softmax weights
    corr_scaled = T * (corr - np.max(corr))  # for numerical stability
    w = np.exp(corr_scaled)
    w /= w.sum()
    
    # Weighted gradient
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        l = k_idx - (n - 1)
        # Contribution to gradient
        p1_start = max(0, l)
        p1_end = min(n, n + l)
        if p1_start < p1_end:
            idx = np.arange(p1_start, p1_end) - l
            g[p1_start:p1_end] += w[k_idx] * (1.0 - h[idx])
        p2_start = max(0, -l)
        p2_end = min(n, n - l)
        if p2_start < p2_end:
            idx = np.arange(p2_start, p2_end) + l
            g[p2_start:p2_end] -= w[k_idx] * h[idx]
    return g

        if nt == n0:
            continue
        xo = np.linspace(0.0, 2.0, n0, endpoint=False)
        xn = np.linspace(0.0, 2.0, nt, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, nt)
        ht, ct = opt_smooth(hi, nt, iters=4000, T=10.0)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
        elapsed = time.time() - t0
        if budget_s - elapsed > 25:
            ht2, ct2 = opt_pairs(ht, nt, iters=15000, seed=seed + nt * 7)
            if ct2 < best_c5:
                best_h, best_c5 = ht2, ct2

    # Phase 4: binary-init starts at a few resolutions
    for nb in [48, 50, 56, 60]:
        elapsed = time.time() - t0
        if budget_s - elapsed < 30:
            break
        hb = make_binary_init(nb, seed + nb)
        ht, ct = opt_smooth(hb, nb, iters=3000, T=8.0)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
        elapsed = time.time() - t0
        if budget_s - elapsed > 20:
            ht2, ct2 = opt_pairs(ht, nb, iters=12000, seed=seed + 100 + nb)
            if ct2 < best_c5:
                best_h, best_c5 = ht2, ct2

    # Phase 5: final high-T polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 15:
        nb = len(best_h)
        h4, c4 = opt_smooth(best_h, nb, iters=4000, T=50.0)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 6: final pair refinement
    elapsed = time.time() - t0
    if budget_s - elapsed > 10:
        h5, c5 = opt_pairs(best_h, len(best_h), iters=15000, seed=seed + 9999)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = c5_val(best_h)

    return best_h, best_c5, len(best_h)
