            best_h, best_c5 = h_try, c5_try

    # Final polish
    best_h, best_c5 = _local_search(best_h, len(best_h), iters=2000)

    if best_h is None:
        best_h = h_init
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
