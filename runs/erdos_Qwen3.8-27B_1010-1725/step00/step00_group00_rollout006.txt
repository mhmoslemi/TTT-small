import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, T):
    mx = np.max(x)
    return float(mx + T * np.log(np.sum(np.exp((x - mx) / T))))

def enforce_constraints(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = h < 1.0 - 1e-9 if diff > 0 else h > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_from, n_to):
    xf = np.arange(n_from, dtype=float)
    xt = np.linspace(0.0, n_from - 1.0, n_to)
    return np.interp(xt, xf, h)

def random_block_init(n, rng, n_blocks=None):
    if n_blocks is None:
        n_blocks = rng.randint(3, 9)
    sizes = rng.dirichlet(np.ones(n_blocks) * 2.0) * n
    h = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        sz = int(round(sizes[i])) if i < n_blocks - 1 else n - pos
        if sz <= 0:
            continue
        v = rng.choice([0.0, 1.0, rng.uniform(0.2, 0.8)])
        h[pos:pos + sz] = v
        pos += sz
    return enforce_constraints(h, n)

def opt_level(h0, n, t_limit, penalty_w=200.0):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h0.copy()
    best_val = compute_c5(h0, n)
    T_list = [n * 0.08, n * 0.04, n * 0.02, n * 0.01, n * 0.005]
    bounds = [(0.0, 1.0)] * n
    for T in T_list:
        if time.time() > t_limit:
            break
        def obj(h):
            hc = np.clip(h, 0.0, 1.0)
            corr = np.correlate(hc, 1.0 - hc, mode='full')
            return smooth_max(corr, T) * dx + penalty_w * (np.sum(hc) - target) ** 2
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'maxfun': 8000, 'ftol': 1e-12, 'gtol': 1e-7})
        h_opt = enforce_constraints(res.x, n)
        v = compute_c5(h_opt, n)
        if v < best_val:
            best_val = v
            best_h = h_opt.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    h_best = np.array(initial_h_values, dtype=float).copy()
    n_best = len(h_best)
    h_best = enforce_constraints(h_best, n_best)
    c5_best = compute_c5(h_best, n_best)

    cur_h = h_best.copy()
    cur_n = n_best

    # Build resolution levels: multiples of n_best up to ~720
    levels = []
    for m in range(1, 20):
        ln = n_best * m
        if ln > 720:
            break
        if ln not in levels and ln >= n_best:
            levels.append(ln)
    # Also add some intermediate sizes
    for m in range(1, 8):
        ln = n_best * m
        if ln not in levels and ln <= 720 and m > 1:
            levels.insert(min(m - 1, len(levels)), ln)
    levels = sorted(set(levels))

    for target_n in levels:
        if time.time() > t_end - 90:
            break
        # Interpolate current best to target resolution
        if target_n != cur_n:
            h_work = interp_h(cur_h, cur_n, target_n)
            h_work = enforce_constraints(h_work, target_n)
        else:
            h_work = cur_h.copy()

        h_opt, v = opt_level(h_work, target_n, t_end - 60)
        if v < c5_best:
            c5_best = v
            h_best = h_opt.copy()
            n_best = target_n
            cur_h = h_opt.copy()
            cur_n = target_n

        # Random restarts at lower resolutions for diversity
        if target_n <= 135 and time.time() < t_end - 150:
            for _ in range(4):
                if time.time() > t_end - 90:
                    break
                h_r = random_block_init(target_n, rng)
                h_r2, v_r = opt_level(h_r, target_n, t_end - 60)
                if v_r < c5_best:
                    c5_best = v_r
                    h_best = h_r2.copy()
                    n_best = target_n
                    cur_h = h_r2.copy()
                    cur_n = target_n

    # Final polish at the best resolution
    if time.time() < t_end - 30:
        h_pol, v_pol = opt_level(h_best, n_best, t_end - 10)
        if v_pol < c5_best:
            c5_best = v_pol
            h_best = h_pol.copy()
            n_best = len(h_pol)

    h_best = enforce_constraints(h_best, n_best)
    c5_best = compute_c5(h_best, n_best)
    return (h_best, c5_best, n_best)
