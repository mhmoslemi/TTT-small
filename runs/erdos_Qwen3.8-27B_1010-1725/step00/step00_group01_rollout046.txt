import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / beta

def eq_constraint_val(h, n):
    return np.sum(h) - n / 2.0

def project_onto_simplex_box(h, n):
    """Project h to satisfy sum(h)=n/2 and 0<=h<=1"""
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0, 1)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Adjust only where we can
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

import numpy as np
from scipy.optimize import minimize
import time

_opt_n = 0
_opt_beta = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_obj(h):
    n = _opt_n
    beta = _opt_beta
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / beta

def _eq_con(h):
    return np.sum(h) - _opt_n / 2.0

def project_onto_simplex_box(h, n):
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def _slsqp_solve(h_init, n, beta, maxiter):
    global _opt_n, _opt_beta
    _opt_n = n
    _opt_beta = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    res = minimize(
        _smooth_obj, h_init,
        method='SLSQP', bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return project_onto_simplex_box(res.x, n)

def _local_refine(h, n, rng, n_steps=20, step_size=0.05):
    """Coordinate-wise local search to polish solution"""
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_steps):
        improved = False
        for i in range(n):
            for delta in [step_size, -step_size, 0.5*step_size, -0.5*step_size]:
                trial = best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_onto_simplex_box(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c5 - 1e-12:
                    best = trial
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _opt_n, _opt_beta
    rng = np.random.default_rng(seed)

    best_n = 0

    else:

    n_list = [47, 50, 55, 60, 70, 80, 100]
    all_h = {len(init_47): init_47.copy()}

    for n in n_list:
        if time.time() - t_start > budget_s - 20:
            break

        if n in all_h:
            h_init = all_h[n].copy()
        else:
            src_n = max(k for k in all_h if k <= n)
            h_init = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, src_n), all_h[src_n])

        h_init = project_onto_simplex_box(h_init, n)
        all_h[n] = h_init.copy()

        # Multi-beta continuation
        for beta, maxit in [(10, 200), (25, 200), (50, 300), (100, 300), (200, 200)]:
            if time.time() - t_start > budget_s - 15:

            h_opt = _slsqp_solve(h_init, n, beta, maxit)
            all_h[n] = h_opt
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n
            h_init = h_opt + rng.normal(0, 0.02, n)
            h_init = project_onto_simplex_box(h_init, n)

            h_r = _slsqp_solve(h_r, n, 50, 200)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n
                all_h[n] = h_r.copy()

        # Local polish on best for this n
