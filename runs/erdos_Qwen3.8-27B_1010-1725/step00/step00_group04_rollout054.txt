def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr)) * dx

def c5_from_corr(corr, n):
    return float(np.max(corr)) * (2.0 / n)

def binarize(h, target_ones):
    """Convert to 0-1 with exactly target_ones ones"""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(len(h))
    h_bin[sorted_idx[:target_ones]] = 1.0
    return h_bin

def local_search_binary(h, time_limit=3.0, seed_rng=None):
    """Hill climbing: random swaps between 0 and 1 positions"""
    rng = seed_rng or np.random
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    start = time.time()
    n = len(h)
    
    while time.time() - start < time_limit:
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.randint(len(ones)))
        j = int(rng.randint(len(zeros)))
        pos_i = ones[i]
        pos_j = zeros[j]

    best_c5 = compute_c5(h)
    start = time.time()
    T = T0
    n = len(h)

    while time.time() - start < time_limit:
        T *= 0.9997
        if T < 1e-6:
            T = 1e-6
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-8)):
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def continuous_refine(h, time_limit, rng):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        hv = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hv, 1.0 - hv, mode='full') * dx
        mx = np.max(corr)
        s = np.sum(np.exp(40.0 * (corr - mx)))
        smooth_max = mx + np.log(s) / 40.0
        pen = 50.0 * (np.sum(hv) - target_sum) ** 2
        return smooth_max + pen

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'maxfun': 20000, 'ftol': 1e-12})
        h_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_out = h.copy()
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-8:
        mask = (h_out < 0.999) if diff > 0 else (h_out > 0.001)
        cnt = np.sum(mask)
        if cnt > 0:
            h_out[mask] += diff / cnt
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out, compute_c5(h_out)

def multi_restart(h0, time_limit, rng, n_restarts=4):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()
    per = time_limit / n_restarts
    n = len(h0)

    for r in range(n_restarts):
        elapsed_r = time.time() - t0
        if elapsed_r >= time_limit:
            break
        remaining_this = min(per, time_limit - elapsed_r)
        if remaining_this < 3:
            break
        if r == 0:
            h_cur = h0.copy()
        else:
            h_cur = h0.copy()
            k_flips = rng.randint(2, max(3, n // 8))
            ones = np.where(h_cur >= 0.999)[0]
            zeros = np.where(h_cur <= 0.001)[0]
            for _ in range(k_flips):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i], h_cur[j] = 0.0, 1.0
                ones = np.where(h_cur >= 0.999)[0]
                zeros = np.where(h_cur <= 0.001)[0]

        h_cur, c5_cur = local_search_binary(h_cur, remaining_this * 0.6, rng)
        if remaining_this * 0.6 < remaining_this - (time.time() - t0 - elapsed_r):
            h_cur, c5_cur2 = sa_binary(h_cur, remaining_this * 0.3, rng, T0=0.03)
            if c5_cur2 < c5_cur:
                c5_cur = c5_cur2
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive search at initial resolution
    if remaining() > 30:
        alloc = min(remaining() * 0.35, 300)
        best_h, best_c5 = multi_restart(best_h, alloc, rng, n_restarts=5)

    # Phase 2: Try higher resolutions
    resolutions = []
    for mult in [2, 3, 4]:
        try_n = n_init * mult
        if try_n <= 400:
            resolutions.append(try_n)

    for try_n in resolutions:
        if remaining() < 40:
            break
        h_up = upsample_binary(best_h, len(best_h), try_n)
        c5_up = compute_c5(h_up)
        # Search on upsampled
        alloc = min(remaining() * 0.25, 200)
        h_up, c5_up = multi_restart(h_up, alloc, rng, n_restarts=3)
        if c5_up < best_c5 - 1e-14:
            best_c5 = c5_up
            best_h = h_up.copy()

    # Phase 3: Final refinement at best resolution
    if remaining() > 15:
        alloc = min(remaining() * 0.5, 200)
        best_h, best_c5 = multi_restart(best_h, alloc, rng, n_restarts=3)

    # Phase 4: Continuous refinement (if still time)
    if remaining() > 20:
        h_cont, c5_cont = continuous_refine(best_h, min(remaining() * 0.3, 60), rng)
        if c5_cont < best_c5 - 1e-8:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Final: ensure sum constraint
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-6:
        if np.sum(best_h) > 0:
            best_h = best_h * (target / np.sum(best_h))
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
