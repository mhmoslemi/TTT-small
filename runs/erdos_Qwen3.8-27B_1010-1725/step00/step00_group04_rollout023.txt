def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if alpha < 1e-6:
        return m, np.zeros(n)
    exp_terms = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_terms)
    p = exp_terms / sum_exp
    obj = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        w = p[lag + (n-1)]
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n-l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n-l]
    grad *= dx
    return obj, grad
