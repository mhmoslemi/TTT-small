import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def lse_max(vals, beta):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp((vals - m) * beta)) / len(vals)) / beta


def obj_smooth(x, n, beta):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return lse_max(corr, beta)


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def fix_feasibility(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)


def coord_descent(h, n, max_sweeps, step, time_limit):
    best_c5 = compute_c5(h)
    t_start = time.time()
    rng_cd = np.random.default_rng(7)
    for sw in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        idx = np.arange(n)
        rng_cd.shuffle(idx)
        for i in idx:
            for delta in (step, -step):
                new_val = h[i] + delta
                if new_val < 0.0 or new_val > 1.0:
                    continue
                h_new = h.copy()
                h_new[i] = new_val
                # redistribute sum diff across others
                diff = -delta
                others = np.delete(np.arange(n), i)
                cap_up = 1.0 - h_new[others]
                cap_dn = h_new[others]
                if diff > 0:
                    alloc = np.minimum(cap_up, diff / len(others))
                    h_new[others] += alloc
                    rem = diff - np.sum(alloc)
                    if rem > 0:
                        rem_mask = cap_up > alloc
                        if np.sum(rem_mask) > 0:
                            h_new[others[rem_mask]] += rem / np.sum(rem_mask)
                else:
                    alloc = np.minimum(cap_dn, -diff / len(others))
                    h_new[others] -= alloc
                    rem = -diff - np.sum(alloc)
                    if rem > 0:
                        rem_mask = cap_dn > alloc
                        if np.sum(rem_mask) > 0:

                h_new = np.clip(h_new, 0.0, 1.0)
                if abs(np.sum(h_new) - n / 2.0) > 1e-6:
                    continue
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    h = h_new.copy()
                    best_c5 = c5_new
                    improved = True
        if not improved:
            step *= 0.75
            if step < 0.0005:
                break
    return h, best_c5
