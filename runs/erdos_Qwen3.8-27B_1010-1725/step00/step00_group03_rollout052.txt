            c5_t = eval_c5(h_trial)
            if c5_t < best - 1e-12:
                h = h_trial
                best = c5_t
                best_h = h_trial.copy()
                stall = 0
                improved = True
                break
            a *= 0.5
        if not improved:
            stall += 1

        if it % 250 == 249:
            alpha *= 0.93
            beta = min(beta * 1.25, 500.0)

        if stall > 400:
            kick = rng.standard_normal(n) * 0.2
            h = project_onto_simplex(h + kick, S)
            stall = 0
            alpha = 0.08
            beta = 40.0

    return best_h, best

def upsample(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old))
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_simplex(h_new, n_new / 2.0)

def try_scipy_smooth(h0, S, n, t_start, t_limit):
    dx = 2.0 / n
    p = 8.0

    def objective(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return float(np.sum(corr**p)) ** (1.0 / p) * dx

    def grad_obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        F = float(np.sum(corr**p)) ** (1.0 / p)
        if F < 1e-15:
            return np.zeros(n)
        factors = p * (corr / F) ** (p - 1.0) / F
        g = np.zeros(n)
        mx = np.max(np.abs(corr))
        for k in np.where(np.abs(corr) > 0.9 * mx)[0]:
            m = k - (n - 1)
            g += factors[k] * lag_grad(hv, m)
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda hv: hv.sum() - S,
             'jac': lambda hv: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h0, jac=grad_obj, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-10})
    if res.success or res.fun < objective(h0):
        h_try = project_onto_simplex(np.clip(res.x, 0, 1), S)
        return h_try, eval_c5(h_try)
    return h0, eval_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    h_init = initial_h_values
    best_c5 = eval_c5(h_init)
    best_h = h_init.copy()
    best_n = len(h_init)

    n_options = [70, 100, 140, 200, 280, 350]

    for idx_n, n in enumerate(n_options):
        if time.time() - t_start > budget_s - 25:
            break
        S = n / 2.0
        if n == len(h_init):
            h0 = h_init.copy()
        else:
            h0 = upsample(h_init, n)

        # Projected gradient
        h_opt, c5 = optimize_pg(h0, S, rng, t_start, budget_s - 10, max_iters=4000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n

        # Scipy SLSQP on smooth approximation (only for small n)
        if n <= 140 and time.time() - t_start < budget_s - 60:
            h_sc, c5_sc = try_scipy_smooth(h_opt, S, n, t_start, budget_s - 5)
            if c5_sc < best_c5:
                best_c5 = c5_sc
                best_h = h_sc.copy()
                best_n = n

        # Random perturbation restart from best
        if time.time() - t_start < budget_s - 40:
            h_pert = best_h.copy() if best_n == n else upsample(h_init, n)
            h_pert = h_pert + rng.standard_normal(n) * 0.3
            h_pert = project_onto_simplex(np.clip(h_pert, 0.0, 1.0), S)
            h_opt2, c5_2 = optimize_pg(h_pert, S, rng, t_start, budget_s - 5, max_iters=2000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_opt2.copy()
                best_n = n

    # Final polish: re-optimize at best_n with smaller steps
    if time.time() - t_start < budget_s - 10:
        S = best_n / 2.0
        h_pol, c5_pol = optimize_pg(best_h, S, rng, t_start, budget_s - 2, max_iters=3000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5, best_n
