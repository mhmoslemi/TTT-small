import numpy as np
from scipy.optimize import minimize
import time


def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = corr - mc
    return float(mc + np.log(np.sum(np.exp(beta * shifted))) / beta)


def _smooth_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    shifted = corr - mc
    exps = np.exp(beta * shifted)
    s_sum = np.sum(exps)
    w = exps / s_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            if s < n:
                t1[s:] = 1.0 - h[:n - s]
                t2[:n - s] = h[s:]
        else:
            ss = -s
            if ss < n:
                t1[:n - ss] = 1.0 - h[ss:]
                t2[ss:] = h[:n - ss]
        grad += w[k] * (t1 - t2)
    grad *= dx
    return grad


def _proj_box_sum(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _eq_fun(h, target):
    return h.sum() - target


def _eq_jac(h, target):
    return np.ones((1, len(h)))


def _slsqp_step(h0, n, dx, target, beta, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}]
    bnds = list(zip(np.zeros(n), np.ones(n)))
    try:
        res = minimize(
            _smooth_obj, h0.copy(),
            jac=_smooth_grad,
            args=(dx, beta),
            method='SLSQP',
            bounds=bnds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if res.x is not None:
            return _proj_box_sum(res.x, target)
    except Exception:
        pass
    return h0


def _round_binary(h_cont, n, target):
    h_bin = np.where(h_cont > 0.5, 1.0, 0.0)
    n_ones = int(h_bin.sum())
    tgt = int(target)
    if n_ones > tgt:
        one_idx = np.where(h_bin == 1.0)[0]
        order = one_idx[np.argsort(h_cont[one_idx])]
        for i in order:
            if n_ones == tgt:
                break
            h_bin[i] = 0.0
            n_ones -= 1
    elif n_ones < tgt:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = zero_idx[np.argsort(-h_cont[zero_idx])]
        for i in order:
            if n_ones == tgt:
                break
            h_bin[i] = 1.0
            n_ones += 1
    return h_bin


def _local_refine(h, n, dx, target, steps=50):
    best = h.copy()
    best_c5 = _eval_c5(best, dx)
    rng = np.random.RandomState(0)
    for _ in range(steps):
        improved = False
        idx = list(range(n))
        rng.shuffle(idx)
        for i in idx:
            for delta in (0.1, -0.1, 0.05, -0.05):
                trial = best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                diff = target - trial.sum()
                if abs(diff) > 1e-9:
                    j = (i + 1) % n
                    if diff > 0 and j != i:
                        trial[j] = np.clip(trial[j] + diff, 0.0, 1.0)
                    elif diff < 0 and j != i:
                        trial[j] = np.clip(trial[j] + diff, 0.0, 1.0)
                    else:
                        continue
                    trial = _proj_box_sum(trial, target)
                c5_t = _eval_c5(trial, dx)
                if c5_t < best_c5 - 1e-8:
                    best = trial
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5
