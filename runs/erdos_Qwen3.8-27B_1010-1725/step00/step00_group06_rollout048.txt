from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-9
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_c5(x, n):
    h = project_feasible(x, n)
    return compute_c5(h)

def coord_descend(h, n, max_outers=3, time_fn=None, budget=0):

            if time_fn is not None and budget - time_fn() < 2:

                c5_cur = compute_c5(h)
