```python
import numpy as np
import time
import math
from scipy.optimize import minimize

TARGET = 0.3808
EPS = 1e-12


def c5_from_h(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / h.size)


def metric_from_bits(b):
    b = np.asarray(b, dtype=np.float32)
    comp = np.ones_like(b) - b
    c = np.correlate(b, comp, mode="full")
    return int(np.max(c)), int(np.dot(c, c))


def metric_bc(b, comp):
    c = np.correlate(b, comp, mode="full")
    return int(np.max(c)), int(np.dot(c, c))


def better_h(h, best_h, best_c, best_n):
    if h is None:
        return best_h, best_c, best_n
    c = c5_from_h(h)
    if not math.isfinite(c):
        return best_h, best_c, best_n
    if c < best_c - EPS:
        h = np.asarray(h, dtype=float)
        return h, float(c), int(h.size)
    return best_h, best_c, best_n


def better_bits(bits, best_h, best_c, best_n):
    if bits is None or bits.size == 0:
        return best_h, best_c, best_n
    cmax, _ = metric_from_bits(bits)
    c5 = float(cmax) * (2.0 / bits.size)
    if c5 < best_c - EPS:
        return bits.astype(float, copy=True), c5, int(bits.size)
    return best_h, best_c, best_n


def project_sum(h):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    target = 0.5 * h.size
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) < 1e-9:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        add = diff / idx.size
        if add > 0:
            room = float((1.0 - h[idx]).min())
            if add > room:
                add = room
        else:
            room = float(-h[idx].min())
            if add < room:
                add = room
        h[idx] += add
    return h


def finalize(h):
    if h is None:
        h = np.zeros(100, dtype=float)
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h)
    c = c5_from_h(h)
    return h, float(c), int(h.size)


def interpolate_parent(parent, n):
    if parent is None:
        return None
    p = np.asarray(parent, dtype=float).ravel()
    pn = p.size
    if pn == 0:
        return None
    if pn == 1:
        return np.full(int(n), float(p[0]), dtype=float)
    xp = np.linspace(0.0, 2.0, pn, dtype=float)
    xn = np.linspace(0.0, 2.0, int(n), dtype=float)
    vals = np.interp(xn, xp, p)
    return np.clip(vals, 0.0, 1.0)


def bits_from_values(vals, n):
    n = int(n)
    m = n // 2
    order = np.argsort(np.asarray(vals, dtype=float))[::-1]
    b = np.zeros(n, dtype=np.float32)
    if m > 0:
        b[order[:m]] = 1.0
    return b


def anti_bits_from_values(vals, n):
    n = int(n)
    m = n // 2
    vals = np.asarray(vals, dtype=float)
    b = np.zeros(n, dtype=np.float32)
    for k in range(m):
        j = n - 1 - k
        if vals[k] > vals[j]:
            b[k] = 1.0
            b[j] = 0.0
        else:
            b[k] = 0.0
            b[j] = 1.0
    return b


def random_full_bits(n, rng):
    n = int(n)
    m = n // 2
    perm = rng.permutation(n)
    b = np.zeros(n, dtype=np.float32)
    if m > 0:
        b[perm[:m]] = 1.0
    return b


def random_anti_bits(n, rng):
    n = int(n)
    m = n // 2
    a = rng.integers(0, 2, m).astype(np.float32)
    b = np.empty(n, dtype=np.float32)
    b[:m] = a
    b[m:] = 1.0 - a[::-1]
    return b


def sine_anti_bits(n, rng):
    n = int(n)
    m = n // 2
    b = np.zeros(n, dtype=np.float32)
    if m <= 0:
        return b
    freq = 0.05 + float(rng.random()) * 0.95
    phase = 2.0 * math.pi * float(rng.random())
    for k in range(m):
        a = 1.0 if math.sin(2.0 * math.pi * freq * k / m + phase) > 0.0 else 0.0
        b[k] = a
        b[n - 1 - k] = 1.0 - a
    return b


def make_bounds(n):
    out = []
    for _ in range(int(n)):
        out.append((0.0, 1.0))
    return out


def continuous_objective(x, alpha):
    hx = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if not np.all(np.isfinite(hx)):
        return 1e12
    c = np.correlate(hx, 1.0 - hx, mode="full")
    cm = float(np.max(c))
    z = c - cm
    return cm + alpha * float(math.log(float(np.sum(np.exp(z / alpha)))))


def sum_le(x, target):
    return target - float(np.sum(x))


def sum_ge(x, target):
    return float(np.sum(x)) - target


def continuous_refine(h, maxiter):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * h.size
    h = project_sum(h)
    try:
        res = minimize(
            continuous_objective,
            h,
            args=(0.25,),
            method="SLSQP",
            bounds=make_bounds(h.size),
            constraints=[
                {"type": "ineq", "fun": sum_le, "args": (target,)},
                {"type": "ineq", "fun": sum_ge, "args": (target,)},
            ],
            options={"maxiter": int(maxiter), "ftol": 1e-9, "disp": False},
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=float)
            if x.size == h.size and np.all(np.isfinite(x)):
                x = np.clip(x, 0.0, 1.0)
                x = project_sum(x)
                return x
    except Exception:
        pass
    return h


def local_anti_search(bits, n, deadline):
    if time.time() >= deadline or n < 4:
        return np.asarray(bits, dtype=np.float32).copy()
    n = int(n)
    m = n // 2
    b = np.asarray(bits, dtype=np.float32).copy()
    comp = np.ones_like(b) - b
    cm, sq = metric_bc(b, comp)
    best = b.copy()
    bcm, bsq = cm, sq
    while time.time() < deadline:
        improved = False
        for k in range(m):
            i = k
            j = n - 1 - k
            bi = b[i]
            bj = b[j]
            if bi == bj:
                continue
            ci = comp[i]
            cj = comp[j]
            b[i] = bj
            b[j] = bi
            comp[i] = cj
            comp[j] = ci
            cm2, sq2 = metric_bc(b, comp)
            if cm2 < cm or (cm2 == cm and sq2 < sq):
                cm, sq = cm2, sq2
                if cm2 < bcm or (cm2 == bcm and sq2 < bsq):
                    best = b.copy()
                    bcm, bsq = cm2, sq2
                improved = True
            else:
                b[i] = bi
                b[j] = bj
                comp[i] = ci
                comp[j] = cj
        if not improved:
            break
    return best


def local_full_search(bits, n, deadline):
    if time.time() >= deadline or n < 4:
        return np.asarray(bits, dtype=np.float32).copy()
    n = int(n)
    b = np.asarray(bits, dtype=np.float32).copy()
    comp = np.ones_like(b) - b
    cm, sq = metric_bc(b, comp)
    best = b.copy()
    bcm, bsq = cm, sq
    while time.time() < deadline:
        improved = False
        for i in range(n - 1):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                bi = b[i]
                bj = b[j]
                if bi == bj:
                    continue
                ci = comp[i]
                cj = comp[j]
                b[i] = bj
                b[j] = bi
                comp[i] = cj
                comp[j] = ci

                if cm2 < cm or (cm2 == cm and sq2 < sq):
                    cm, sq = cm2, sq2
                    if cm2 < bcm or (cm2 == bcm and sq2 < bsq):
                        best = b.copy()
                        bcm, bsq = cm2, sq2
                    improved = True
                else:
                    b[i] = bi
                    b[j] = bj
                    comp[i] = ci
                    comp[j] = cj
        if not improved:
            break
    return best


def optimize_anti_ilc(bits, n, deadline, rng):
    if time.time() >= deadline:
        return np.asarray(bits, dtype=np.float32).copy()
    n = int(n)
    t0 = time.time()
    rem = max(1.0, deadline - t0)
    first = min(deadline, t0 + 0.5 * rem)
    b = local_anti_search(bits, n, first)
    bcm, bsq = metric_from_bits(b)
    m = n // 2
    stall = 0
    while time.time() < deadline - 1.0 and stall < 12 and m > 0:
        kick = 2 + int(rng.integers(0, 6 + min(stall, 12)))
        s = b.copy()
        done = 0
        while done < kick:
            k = int(rng.integers(m))
            p = k
            q = n - 1 - k
            s[p], s[q] = s[q], s[p]
            done += 1
        bonus = max(1.0, min(15.0, 0.3 * (deadline - time.time())))
        b2 = local_anti_search(s, n, min(deadline, time.time() + bonus))
        c2, sq2 = metric_from_bits(b2)
        if c2 < bcm or (c2 == bcm and sq2 < bsq):
            b = b2.copy()
            bcm, bsq = c2, sq2
            stall = 0
        else:
            stall += 1
    return b


def optimize_full_ilc(bits, n, deadline, rng):
    if time.time() >= deadline:
        return np.asarray(bits, dtype=np.float32).copy()
    n = int(n)
    t0 = time.time()
    rem = max(1.0, deadline - t0)
    first = min(deadline, t0 + 0.7 * rem)
    b = local_full_search(bits, n, first)
    bcm, bsq = metric_from_bits(b)
    stall = 0
    while time.time() < deadline - 1.0 and stall < 10:
        kick = 2 + int(rng.integers(0, 6 + min(stall, 12)))
        s = b.copy()
        applied = 0
        tries = 0
        while applied < kick and tries < 6 * kick:
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if s[p] != s[q]:
                s[p], s[q] = s[q], s[p]
                applied += 1
            tries += 1
        bonus = max(1.0, min(20.0, 0.3 * (deadline - time.time())))
        b2 = local_full_search(s, n, min(deadline, time.time() + bonus))
        c2, sq2 = metric_from_bits(b2)
        if c2 < bcm or (c2 == bcm and sq2 < bsq):
            b = b2.copy()
            bcm, bsq = c2, sq2
            stall = 0
        else:
            stall += 1
    return b


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + float(budget_s)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h0 = np.asarray(parent, dtype=float).copy()
        if h0.size > 0:
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_sum(h0)
            best_h, best_c, best_n = better_h(h0, best_h, best_c, best_n)

    if best_h is None:
        b = random_full_bits(100, rng)
        best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)

    if best_c <= TARGET + EPS:
        return finalize(best_h)

    if time.time() < deadline - 120:
        h1 = continuous_refine(best_h, 30)
        best_h, best_c, best_n = better_h(h1, best_h, best_c, best_n)
        if best_c <= TARGET + EPS:
            return finalize(best_h)

    ns = [100, 96, 102, 108, 110, 104]
    fracs = [0.36, 0.24, 0.18, 0.12, 0.07, 0.03]

    for idx, n in enumerate(ns):
        if time.time() > deadline - 35:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.0
        if frac <= 0:
            continue
        slot = (deadline - time.time()) * frac
        if slot < 10:
            break
        end = time.time() + slot
        vals = interpolate_parent(parent, n)

        if vals is not None:
            b = bits_from_values(vals, n)
            best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
            if best_c <= TARGET + EPS:
                return finalize(best_h)
            lim = min(end, time.time() + 0.50 * slot)
            b = optimize_full_ilc(b, n, lim, rng)
            best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
            if best_c <= TARGET + EPS:
                return finalize(best_h)

            b = anti_bits_from_values(vals, n)
            best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
            if best_c <= TARGET + EPS:
                return finalize(best_h)
            lim = min(end, time.time() + 0.20 * slot)
            b = optimize_anti_ilc(b, n, lim, rng)
            best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
            if best_c <= TARGET + EPS:
                return finalize(best_h)

        b = random_anti_bits(n, rng)
        best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
        if best_c <= TARGET + EPS:
            return finalize(best_h)
        if vals is not None:
            lim = min(end, time.time() + 0.15 * slot)
        else:
            lim = min(end, time.time() + 0.50 * slot)
        b = optimize_anti_ilc(b, n, lim, rng)
        best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
        if best_c <= TARGET + EPS:
            return finalize(best_h)

        b = sine_anti_bits(n, rng)
        best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
        if best_c <= TARGET + EPS:
            return finalize(best_h)
        b = optimize_anti_ilc(b, n, end, rng)
        best_h, best_c, best_n = better_bits(b, best_h, best_c, best_n)
        if best_c <= TARGET + EPS:
            return finalize(best_h)

    if best_h is not None and time.time() < deadline - 20:
        rem = deadline - time.time()
        if rem > 300:
            mi = 100
        elif rem > 120:
            mi = 60
        else:
            mi = 20
        h1 = continuous_refine(best_h, mi)
        best_h, best_c, best_n = better_h(h1, best_h, best_c, best_n)

    return finalize(best_h)
```
