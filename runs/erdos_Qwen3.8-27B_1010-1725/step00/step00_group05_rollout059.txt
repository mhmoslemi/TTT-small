import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(full_corr) * dx

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(full_corr))
    k = max_idx - (n - 1)
    
    grad = np.zeros(n)
    # First factor contribution
    lo1 = max(0, -k)
    hi1 = min(n, n - k)
    for j in range(lo1, hi1):
        grad[j] += (1.0 - h[j + k])
    # Second factor contribution
    lo2 = max(0, k)
    hi2 = min(n, n + k)
    for j in range(lo2, hi2):
        grad[j] -= h[j - k]
    
    return grad * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] += diff / count
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] += diff / count
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0

def smooth_max_c5(h, T=50):
    n = len(h)
    dx = 2.0 / n
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    # f_k = full_corr * dx
    f = full_corr * dx
    # Shift for numerical stability
    f_shifted = f - np.max(f)
    exp_f = np.exp(T * f_shifted)
    logsumexp = np.max(f) + np.log(np.sum(exp_f)) / T
    return logsumexp

def smooth_max_c5_grad(h, T=50):
    n = len(h)
    dx = 2.0 / n
    full_corr = np.correlate(h, 1.0 - h, mode="full")
    f = full_corr * dx
    f_max = np.max(f)
    exp_f = np.exp(T * (f - f_max))
    weights = exp_f / np.sum(exp_f)  # softmax weights
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - (n - 1)
        # Gradient of f_k wrt h
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        for j in range(lo1, hi1):
            grad[j] += weights[k_idx] * (1.0 - h[j + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        for j in range(lo2, hi2):
            grad[j] -= weights[k_idx] * h[j - k]
    
    return grad * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)

    w = e / e.sum()
    grad = np.zeros(n)
    k_vals = np.arange(-(n - 1), n)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
