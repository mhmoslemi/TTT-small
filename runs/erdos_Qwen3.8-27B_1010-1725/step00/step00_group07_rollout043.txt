    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        if abs(diff) < 1e-14:

        else:

            break

        h = np.clip(h, 0.0, 1.0)

    return np.interp(xn, np.arange(len(h)), h)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)


def proj_feas(h, ts):
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    for _ in range(3000):
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, nn):
    old = len(h)
    return np.interp(np.arange(nn) * old / nn, np.arange(old), h)


def _sum_eq(x, n):
    return x.sum() - n / 2.0


def _sum_jac(x, n):
    return np.ones(n)


def slsqp_opt(h0, n, tb, rng):
    t0 = time.time()
    ts = n / 2.0
    h0 = proj_feas(h0, ts)
    bc = compute_c5(h0)
    bh = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    maxit = max(50, int(500.0 * 80 / n))
    sg = 0.1
    st = 0
    while time.time() - t0 < tb:
        hi = proj_feas(bh + rng.normal(0, sg, n), ts)
        try:
            r = minimize(compute_c5, hi, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxit, 'ftol': 1e-10})
            if r.x is None:
                st += 1
                continue
            hx = proj_feas(r.x, ts)
            cx = compute_c5(hx)
            if cx < bc - 1e-12:
                bc, bh, st = cx, hx.copy(), 0
                sg = max(0.003, sg * 0.7)
            else:
                st += 1
                sg = min(0.3, sg * 1.3)
        except Exception:
            st += 1
        if st > 15:
            break
    return bh


def coord_descent(h0, n, tb, rng):
    t0 = time.time()
    ts = n / 2.0
    bh = proj_feas(h0, ts)
    bc = compute_c5(bh)
    for _ in range(30):
        if time.time() - t0 > tb:
            break
        imp = False
        for j in rng.permutation(n):
            if time.time() - t0 > tb:
                break
            for v in (0.0, 1.0, np.clip(bh[j] + rng.normal(0, 0.12), 0, 1)):
                ht = bh.copy()
                ht[j] = v
                d = ts - ht.sum()
                if abs(d) > 1e-9:
                    m = (ht < 1.0) if d > 0 else (ht > 0.0)
                    m[j] = False
                    cnt = m.sum()
                    if cnt == 0:
                        continue
                    ht[m] += d / cnt
                    ht = np.clip(ht, 0.0, 1.0)
                    if abs(ht.sum() - ts) > 1e-7:
                        continue
                ct = compute_c5(ht)
                if ct < bc - 1e-13:
                    bc, bh, imp = ct, ht, True
        if not imp:
            break
    return bh


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h
