import numpy as np
import time
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_binary(n, rng, init=None):
    h = np.zeros(n)
    if init is not None:
        x1 = np.linspace(0, 1, len(init))
        x2 = np.linspace(0, 1, n)
        vals = np.interp(x2, x1, init)
        h = (vals > 0.5).astype(float)
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    s = int(h.sum())
    target = n // 2
    if s > target:
        oi = np.where(h == 1)[0]
        h[rng.choice(oi, s - target, replace=False)] = 0.0
    elif s < target:
        zi = np.where(h == 0)[0]
        h[rng.choice(zi, target - s, replace=False)] = 1.0
    return h

def _anneal_binary(h, rng, T_init, cooling, max_iter, deadline):
    n = len(h)
    c5 = evaluate_c5(h)
    T = T_init
    best_c5_local = c5
    for it in range(max_iter):
        if time.time() > deadline:
            return h, c5
        if T < 1e-7:
            break
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)
        dc = new_c5 - c5
        if dc < 0 or rng.rand() < math.exp(-dc / T):
            c5 = new_c5
            if c5 < best_c5_local:
                best_c5_local = c5
        else:
            h[i], h[j] = h[j], h[i]
        T *= cooling
    return h, c5

def _anneal_continuous(h, rng, T_init, step, max_iter, deadline):
    n = len(h)
    c5 = evaluate_c5(h)
    T = T_init
    for it in range(max_iter):

            return h, c5
        if T < 1e-8:
            break
        i = rng.randint(n)
        old = h[i]
        h[i] = np.clip(old + (rng.rand() - 0.5) * step, 0.0, 1.0)
        new_c5 = evaluate_c5(h)
        dc = new_c5 - c5
        if dc < 0 or rng.rand() < math.exp(-dc / max(T, 1e-12)):
            c5 = new_c5
        else:
            h[i] = old
        T *= 0.9998
