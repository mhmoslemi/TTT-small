from scipy.optimize import minimize
import time

_OPT = {'T': 10, 'target': 35.0}

def _obj_grad(x, T, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = np.max(corr)
    shifted = T * (corr - mx)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = mx + log_sum / T
    w = exp_s / exp_s.sum()
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2*n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        # dc[m]/dh[j] = (1-h[j-m+n-1])*I(0<=j-m+n-1<n) + (-h[j+m-n+1])*I(0<=j+m-n+1<n)
        # For term 1: j-m+n-1 in [0,n-1] => m-(n-1) <= j <= m-(n-1)+(n-1) = m
        #   j in [max(0, m-n+1), min(n-1, m)]
        # For term 2: j+m-n+1 in [0,n-1] => n-1-m <= j <= m+n-1
        #   j in [max(0, n-1-m), min(n-1, m+n-1)] = [max(0, n-1-m), n-1]
        
        j1_lo = max(0, m - (n-1))
        j1_hi = min(n-1, m)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi+1)
            idx = js - m + (n-1)
            grad[js] += wm * (1.0 - x[idx])
        
        j2_lo = max(0, (n-1) - m)
        j2_hi = min(n-1, m + (n-1))
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi+1)
            idx = js + m - (n-1)
            grad[js] += wm * (-x[idx])
    
    grad *= dx
    # Add penalty for sum constraint
    s = x.sum() - target
    if abs(s) > 1e-8:
        obj += 50.0 * s * s
        grad += 100.0 * s
    
    return obj, grad

def _objfun(x):
    return _obj_grad(x, _OPT['T'], _OPT['target'])[0]

def _jacfun(x):
    return _obj_grad(x, _OPT['T'], _OPT['target'])[1]

def project_sum(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}"""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-10:
        return x
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(np.clip(x + mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2, 0, 1)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample(h_old, n_old, n_new):

    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def binary_local_search(h, rng, max_iters, deadline):
    """Try random swaps on a binary solution"""
    n = len(h)
    h = (h > 0.5).astype(float)
    # Adjust to exact sum
    target = n // 2
    cur_sum = int(h.sum())
    if cur_sum > target:
        order = np.argsort(h)  # zeros first
        idxs = np.where(h == 1)[0]
        np.random.shuffle(idxs)  # use rng instead
        for i, idx in enumerate(idxs):
            h[idx] = 0
            cur_sum -= 1
            if cur_sum == target:
                break
    elif cur_sum < target:
        idxs = np.where(h == 0)[0]
        rng.shuffle(idxs)
        for i, idx in enumerate(idxs):
            h[idx] = 1
            cur_sum += 1
            if cur_sum == target:
                break
    
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            ones[ones == i] = j  # update indices
            zeros[zeros == j] = i
        else:
            h[i], h[j] = h[j], h[i]  # undo
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_OPT = {'T': 10.0, 'target': 35.0}

def _obj_grad(x, T, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = np.max(corr)
    shifted = T * (corr - mx)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    obj = mx + log_sum / T
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for l in range(n):
        grad += (1.0 - 2.0 * x[l]) * w[l:l+n]
    grad *= dx
    s = x.sum() - target
    if abs(s) > 1e-10:
        obj += 10000.0 * s * s
        grad += 20000.0 * s
    return obj, grad

def _objfun(x):
    return _obj_grad(x, _OPT['T'], _OPT['target'])[0]

def _jacfun(x):
    return _obj_grad(x, _OPT['T'], _OPT['target'])[1]

def project_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x + mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def optimize_at_res(h0, n, T_values, max_iter, deadline):
    target = n / 2.0
    h = project_sum(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for T in T_values:
        if time.time() > deadline - 5:
            break
        _OPT['T'] = T
        _OPT['target'] = target
        res = minimize(_objfun, h, jac=_jacfun, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'maxls': 80,
                                                'ftol': 1e-12, 'gtol': 1e-8})
        h_new = project_sum(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
    return h, best_c5

def upsample(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def to_binary(h, n, rng):
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    cur = int(h_bin.sum())
    if cur > target:
        idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h)[idx]
        rng.shuffle(idx)
        for idx_i in idx:
            h_bin[idx_i] = 0.0
            cur -= 1
            if cur == target:
                break
    elif cur < target:
        idx = np.where(h_bin == 0.0)[0]
        rng.shuffle(idx)
        for idx_i in idx:
            h_bin[idx_i] = 1.0
            cur += 1
            if cur == target:
                break
    return h_bin

def binary_local_search(h, rng, deadline, max_iters=8000):
    n = len(h)
    target = n // 2
    h = to_binary(h, n, rng)
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        if no_improve > 2000:
            break
    return h, best_c5

def make_random_block(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:target]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    h_init = np.asarray(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = project_sum(h_init.copy(), n_init / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize at initial resolution
    t_phase = deadline - (budget_s * 0.20)
    h, c5 = optimize_at_res(h_init, n_init, [5.0, 10.0, 20.0, 50.0], 400, t_phase)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 2: higher resolutions
    n_targets = [100, 120, 150, 200]
    for n_t in n_targets:
        if time.time() > deadline - 80:
            break
        frac = 0.20 / max(1, len(n_targets) - n_targets.index(n_t))
        t_phase = deadline - (deadline - time.time()) * frac
        # From continuous upsample
        h_up = upsample(best_h, len(best_h), n_t)
        h_up = project_sum(h_up, n_t / 2.0)
        h, c5 = optimize_at_res(h_up, n_t, [10.0, 20.0, 50.0], 250, t_phase)
        if c5 < best_c5:
            best_c5, best_h = c5, h
        # From binary upsample
        if time.time() < t_phase - 10:
            h_bin = to_binary(upsample(best_h, len(best_h), n_t), n_t, rng)
            h_b, c5_b = optimize_at_res(h_bin, n_t, [20.0, 50.0], 200, min(t_phase, time.time() + 60))
            if c5_b < best_c5:
                best_c5, best_h = c5_b, h_b
        # Random block start
        if time.time() < t_phase - 5:
            h_rand = make_random_block(n_t, rng)
            h_r, c5_r = optimize_at_res(h_rand, n_t, [10.0, 30.0], 150, min(t_phase, time.time() + 40))
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    # Phase 3: binary local search
    if time.time() < deadline - 10:
        t_dl = deadline - 5
        h_b, c5_b = binary_local_search(best_h, rng, t_dl)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b

    # Final projection
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
