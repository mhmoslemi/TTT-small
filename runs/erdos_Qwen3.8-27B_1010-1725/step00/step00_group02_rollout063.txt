def _c5_dx(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _lbfgs_obj(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode='full') * G_DX
    m = float(np.max(c))
    b = G_BETA
    if b <= 1e-6:
        val = m
    else:
        val = m + float(np.log(np.sum(np.exp(b * (c - m)))) / b)
    val += G_MU * (float(np.sum(x)) - G_TARGET) ** 2
    return val

def _fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        h[idx] += res / float(idx.size)
        h = np.clip(h, 0.0, 1.0)
    return h

def _project_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) <= 1e-13:
        return _fix_sum(h, target)
    if s > target:
        lo = -1.0; hi = 0.0
    else:
        lo = 0.0; hi = 1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        sm = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if sm < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def _linear_resample(h, n):
    m = len(h)

    xn = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(xn, xo, h)

import numpy as np
from scipy.optimize import minimize
import time

G_DX = 0.0
G_TARGET = 0.0
G_BETA = 0.0
G_MU = 0.0
G_DEADLINE = 0.0


def _c5_dx(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def _lbfgs_obj(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode="full") * G_DX
    m = float(np.max(c))
    b = G_BETA
    if b <= 1e-6:
        val = m
    else:
        val = m + float(np.log(np.sum(np.exp(b * (c - m)))) / b)
    val += G_MU * (float(np.sum(x)) - G_TARGET) ** 2
    return float(val)


def _fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        h[idx] += res / float(idx.size)
        h = np.clip(h, 0.0, 1.0)
    return h


def _project_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) <= 1e-13:
        return _fix_sum(h, target)
    if s > target:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        sm = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if sm < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    return _fix_sum(h, target)


def _linear_resample(h, n):
    m = len(h)
    if m == 0:
        return np.zeros(n)
    if m == n:
        return h.copy()
    xo = np.linspace(0.0, 1.0, m, endpoint=False)
    xn = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(xn, xo, h)


def _block_resample(h, n):
    m = len(h)
    if m == 0:
        return np.zeros(n)
    if m == n:
        return h.copy()
    if n % m == 0:
        return np.repeat(h, n // m)
    return _linear_resample(h, n)


def _default_start(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * (2.0 + 0.3 * float(seed % 4)) * x + 0.5 * rng.random())
    h = h + 0.04 * (rng.random(n) - 0.5)
    return _project_sum(h, n / 2.0)


def _optimize_lbfgs(h, dx, target, strong=True):
    global G_DX, G_TARGET, G_BETA, G_MU
    h = _project_sum(h, target)
    n = len(h)
    best = h.copy()
    best_c = _c5_dx(h, dx)
    cur = best.copy()
    bounds = [(0.0, 1.0)] * n

    if strong:
        schedule = [(4.0, 80.0, 2500), (15.0, 80.0, 3000),
                    (60.0, 80.0, 2500), (200.0, 80.0, 2000)]
    else:
        schedule = [(8.0, 80.0, 1500), (60.0, 80.0, 1500)]

    for beta, mu, maxfun in schedule:
        if time.time() >= G_DEADLINE - 150:
            break
        G_DX = dx
        G_TARGET = target
        G_BETA = beta
        G_MU = mu
        maxiter = max(20, int(maxfun // max(1, n + 1)) + 10)
        try:
            res = minimize(
                _lbfgs_obj,
                cur,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxiter, "maxfun": maxfun,
                         "ftol": 1e-8, "gtol": 1e-5, "eps": 1e-6}
            )
            if getattr(res, "x", None) is not None:
                nh = _project_sum(res.x, target)
                nc = _c5_dx(nh, dx)
                if nc < best_c - 1e-12:
                    best = nh.copy()
                    best_c = nc
                cur = nh.copy() if nc <= best_c + 1e-2 else best.copy()
        except Exception:
            pass
    return best, best_c


def _direct_polish(h, dx, target, deadline, grid=33, max_passes=20, seed=0):
    h = _project_sum(h, target)
    n = len(h)
    c = _c5_dx(h, dx)
    if n < 2:
        return h, c

    if n > 150:
        grid = min(grid, 49)
    elif n > 60:
        grid = min(grid, 97)
    else:
        grid = max(11, min(grid, 161))

    work = np.empty_like(h)
    rng = np.random.default_rng(seed)
    stop = False
    no_impr = 0

    for _ in range(max_passes):
        if time.time() >= deadline - 20:
            stop = True
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if stop or time.time() >= deadline - 20:
                stop = True
                break

            mn, mx = 2.0, -2.0
            for k in range(n):
                if k != i:
                    v = h[k]
                    if v < mn:
                        mn = v
                    if v > mx:
                        mx = v

            old = h[i]
            a_lo, a_hi = -mn, 1.0 - mx
            if a_lo < a_hi:
                x_lo = old - a_hi * (n - 1)
                x_hi = old - a_lo * (n - 1)
                if x_lo < 0.0:
                    x_lo = 0.0
                if x_hi > 1.0:
                    x_hi = 1.0
                if x_hi > x_lo:
                    best_loc, best_w, best_x, ev = c, None, old, 0
                    for x in np.linspace(x_lo, x_hi, grid):
                        if (ev & 63) == 0 and time.time() >= deadline - 10:
                            stop = True
                            break
                        alpha = (old - x) / float(n - 1)
                        work[:] = h
                        work += alpha
                        work[i] = x
                        cc = _c5_dx(work, dx)
                        if cc < best_loc - 1e-13:
                            best_loc = cc
                            best_w = work.copy()
                            best_x = x
                            improved = True
                        ev += 1

                    if best_w is not None and not stop:
                        span = x_hi - x_lo
                        win = max(span / float(max(grid, 1)), 1e-4)
                        rlo = max(x_lo, best_x - win)
                        rhi = min(x_hi, best_x + win)
                        for x in np.linspace(rlo, rhi, 41):
                            if (ev & 63) == 0 and time.time() >= deadline - 10:
                                stop = True
                                break
                            alpha = (old - x) / float(n - 1)
                            work[:] = h
                            work += alpha
                            work[i] = x
                            cc = _c5_dx(work, dx)
                            if cc < best_loc - 1e-13:
                                best_loc = cc
                                best_w = work.copy()
                                improved = True
                            ev += 1

                    if best_w is not None:
                        h[:] = best_w
                        c = best_loc
                    if stop:
                        break

            if stop:
                break

            slack = np.minimum(h, 1.0 - h)
            old_slack = slack[i]
            slack[i] = -1.0
            j_slack = int(np.argmax(slack))
            slack[i] = old_slack

            seen = set()
            for j in ((i + 1) % n, (i - 1 + n) % n, j_slack, (i + n // 2) % n):
                if stop or j == i or j in seen:
                    continue
                seen.add(j)
                S = h[i] + h[j]
                t_lo = max(0.0, S - 1.0)
                t_hi = min(1.0, S)
                if t_hi <= t_lo:
                    continue

                best_loc, best_w, best_t, ev = c, None, h[i], 0
                for t in np.linspace(t_lo, t_hi, grid):
                    if (ev & 63) == 0 and time.time() >= deadline - 10:
                        stop = True
                        break
                    work[:] = h
                    work[i] = t
                    work[j] = S - t
                    cc = _c5_dx(work, dx)
                    if cc < best_loc - 1e-13:
                        best_loc = cc
                        best_w = work.copy()
                        best_t = t
                        improved = True
                    ev += 1

                if best_w is not None and not stop:
                    span = t_hi - t_lo
                    win = max(span / float(max(grid, 1)), 1e-4)
                    rlo = max(t_lo, best_t - win)
                    rhi = min(t_hi, best_t + win)
                    for t in np.linspace(rlo, rhi, 41):
                        if (ev & 63) == 0 and time.time() >= deadline - 10:
                            stop = True
                            break
                        work[:] = h
                        work[i] = t
                        work[j] = S - t
                        cc = _c5_dx(work, dx)
                        if cc < best_loc - 1e-13:
                            best_loc = cc
                            best_w = work.copy()
                            improved = True
                        ev += 1

                if best_w is not None:
                    h[:] = best_w
                    c = best_loc
                if stop:
                    break

        if stop:
            break
        if not improved:
            no_impr += 1
            if no_impr >= 3:
                break
        else:
            no_impr = 0
        h = _fix_sum(h, target)
        c = _c5_dx(h, dx)

    h = _fix_sum(h, target)
    return h, _c5_dx(h, dx)


def run(seed=42, budget_s=1100, **kwargs):
    global G_DEADLINE
    t0 = time.time()
    G_DEADLINE = t0 + max(5.0, float(budget_s) - 45.0)

    init = globals().get("initial_h_values", None)
    if init is None:
        init = _default_start(45, seed)
    else:
        init = np.asarray(init, dtype=float)
        init = _fix_sum(init, float(len(init)) / 2.0)
    if len(init) < 2:
        init = _default_start(45, seed)

    best_n = len(init)
    best_h = init.copy()
    best_c = _c5_dx(best_h, 2.0 / best_n)
    parents = [init.copy()]

    if time.time() < G_DEADLINE - 240:
        h1, c1 = _direct_polish(best_h.copy(), 2.0 / best_n, best_n / 2.0,
                                G_DEADLINE, grid=33, max_passes=25, seed=seed)
        if c1 < best_c:
            best_h, best_c = h1.copy(), c1
        parents.append(h1.copy())

        if time.time() < G_DEADLINE - 320:
            h2, c2 = _optimize_lbfgs(h1.copy(), 2.0 / best_n, best_n / 2.0, strong=False)
            if c2 < best_c:
                best_h, best_c = h2.copy(), c2
            h3, c3 = _direct_polish(h2.copy(), 2.0 / best_n, best_n / 2.0,
                                    G_DEADLINE, grid=81, max_passes=10, seed=seed + 7)
            if c3 < best_c:
                best_h, best_c = h3.copy(), c3
            parents.append(h3.copy())

    multis = [2, 3] if best_n * 3 <= 240 else [2]
    for mult in multis:
        n = best_n * mult
        if n <= best_n or n > 540:
            continue
        if time.time() >= G_DEADLINE - 360:
            break

        h0, c0 = None, float("inf")
        bases = [best_h] + parents[-2:]
        for base in bases:
            for hh in (_block_resample(base, n), _linear_resample(base, n)):
                hh = _project_sum(hh, n / 2.0)
                cc = _c5_dx(hh, 2.0 / n)
                if cc < c0:
                    h0, c0 = hh, cc

        if h0 is None:
            h0 = _default_start(n, seed + 50)
            c0 = _c5_dx(h0, 2.0 / n)

        h1, c1 = _direct_polish(h0, 2.0 / n, n / 2.0,
                                G_DEADLINE, grid=25, max_passes=8, seed=seed + 100 * mult)
        if c1 < best_c:
            best_n = n
            best_h, best_c = h1.copy(), c1

        if time.time() < G_DEADLINE - 300 and n <= 180:
            h2, c2 = _optimize_lbfgs(h1, 2.0 / n, n / 2.0, strong=(n <= 90))
            if c2 < best_c:
                best_n = n
                best_h, best_c = h2.copy(), c2
            stage_parent = h2.copy()
        else:
            stage_parent = h1.copy()

        if time.time() < G_DEADLINE - 90:
            h3, c3 = _direct_polish(stage_parent, 2.0 / n, n / 2.0,
                                    G_DEADLINE, grid=57, max_passes=15, seed=seed + 200 * mult)
            if c3 < best_c:
                best_n = n
                best_h, best_c = h3.copy(), c3
            stage_parent = h3.copy()

            if time.time() < G_DEADLINE - 60:
                h4, c4 = _direct_polish(h3, 2.0 / n, n / 2.0,
                                        G_DEADLINE, grid=97, max_passes=6, seed=seed + 300 * mult)
                if c4 < best_c:
                    best_n = n
                    best_h, best_c = h4.copy(), c4
                stage_parent = h4.copy()
        parents.append(stage_parent)

    final_n = len(best_h)
    if time.time() < G_DEADLINE:
        hf, cf = _direct_polish(best_h.copy(), 2.0 / final_n, final_n / 2.0,
                                G_DEADLINE, grid=101, max_passes=3, seed=seed + 999)
        if cf < best_c:
            best_h, best_c = hf, cf

    final_h = _fix_sum(best_h, float(final_n) / 2.0)
    final_dx = 2.0 / final_n
    final_c = _c5_dx(final_h, final_dx)
    return final_h, final_c, int(final_n)
