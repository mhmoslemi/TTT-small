            h_ls, c5_ls = local_search_continuous(cur, nc,
                                                  min(30000, int(time_left() * 3000)), rng, 0.01)
            if c5_ls < best_c5:
                best_h, best_c5, best_n = h_ls, c5_ls, nc

        # Binary on this grid
        if time_left() > 20:
            h_b, c5_b = local_search_binary(best_h if best_n == nc else h_interp,
                                            nc, min(20000, int(time_left() * 2000)), rng)
            if c5_b < best_c5:
                best_h, best_c5, best_n = h_b, c5_b, nc

    # Phase 5: Final refinement with smaller steps
    if time_left() > 20:
        h_f, c5_f = local_search_continuous(best_h, best_n,
                                            min(80000, int(time_left() * 5000)), rng, 0.005)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Phase 6: One more SLSQP polish
    if time_left() > 30:
        h_p, c5_p = slsqp_optimize(best_h, best_n, maxiter=100)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    # Final check
    best_h = project_feasible(best_h, best_n)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
