import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def coord_descent(h, rng, max_rounds=200, time_fn=None):
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h)
    for rnd in range(max_rounds):
        if time_fn and time_fn() < 5:
            break
        improved = False
        for i in rng.permutation(n):
            if time_fn and time_fn() < 5:
                break
            j = rng.randint(n)
            if j == i:
                j = (i + 1) % n
            s = h[i] + h[j]
            L = max(0.0, s - 1.0)
            U = min(1.0, s)
            if U - L < 1e-12:
                continue
            best_val = best_c5
            best_u = h[i]
            ntries = 12
            for t in range(ntries + 1):
                u = L + (U - L) * t / ntries
                if abs(u - h[i]) < 1e-9:
                    continue
                h[i] = u
                h[j] = s - u
                val = evaluate_c5(h)
                if val < best_val:
                    best_val = val
                    best_u = u
            h[i] = best_u
            h[j] = s - best_u
            if best_val < best_c5 - 1e-13:
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5
