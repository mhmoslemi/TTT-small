def project_simplex_box(h, S, lb=0, ub=1):
    """Project h onto {x : lb ≤ x ≤ ub, sum(x) = S}"""
    # Use bisection on the Lagrange multiplier
    def sum_clipped(lam):
        return np.sum(np.clip(h - lam, lb, ub))
    
    lo, hi = h.min() - ub - 1, h.max() - lb + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, lb, ub)

    return float(np.max(corr))

def project_simplex_box(h, S):
    """Project onto {x: 0<=x<=1, sum(x)=S} via bisection on Lagrange multiplier."""
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

    """Subgradient of max_k corr[k] at h (times dx)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_simplex_box(h, S):
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        for m in range(lag, n):
            grad[m] += (1.0 - h[m - lag]) * dx
        for m in range(0, n - lag):
            grad[m] -= h[m + lag] * dx
    else:
        L = -lag
        for m in range(0, n - L):
            grad[m] += (1.0 - h[m + L]) * dx
        for m in range(L, n):
            grad[m] -= h[m - L] * dx
    return grad


def projected_gradient(h, n, max_iter=5000, lr0=0.4, decay=0.95, decay_every=300):
    S = 0.5 * n
    best_h = h.copy()
    best_val = compute_c5(best_h)
    lr = lr0
    h = h.copy()
    for it in range(max_iter):
        grad = subgradient(h, n)
        h = h - lr * grad
        h = project_simplex_box(h, S)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if (it + 1) % decay_every == 0:
            lr *= decay
    return best_h, best_val


def smooth_max_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / T)


def slsqp_optimize(h, n, maxiter=200):
    S = 0.5 * n
    T_val = 40.0

    def obj(x):
        return smooth_max_obj(x, T_val)

    def cons_eq(x):
        return np.sum(x) - S

    cons = [{'type': 'eq', 'fun': cons_eq}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_simplex_box(h_new, S)
        val = compute_c5(h_new)
        if val < compute_c5(h):
            return h_new, val
    except Exception:
        pass
    return h, compute_c5(h)


def interpolate_to_n(h_source, n_target):
    n_src = len(h_source)
    if n_src == n_target:
        return h_source.copy()
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_old, h_source)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_simplex_box(h_new, 0.5 * n_target)
    return h_new


def perturb(h, scale, rng):
    h_p = h + scale * rng.standard_normal(len(h))
    h_p = project_simplex_box(h_p, 0.5 * len(h))
    return h_p
