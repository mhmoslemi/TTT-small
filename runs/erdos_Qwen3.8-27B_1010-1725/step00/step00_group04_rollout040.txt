_G_T = 100.0

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_c5(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def objective_fn(h):
    return soft_c5(h, _G_T)

def cons_fun(x):
    return np.sum(x) - _CONST_N / 2.0

def cons_jac(x):
    return np.ones(_CONST_N)

def get_cons():
    return [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
