import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    h += diff / n
    # After adding uniform offset, re-clip and adjust
    h = np.clip(h, 0.0, 1.0)

        delta = rng.uniform(-step_scale, step_scale)
        h[i] += delta
        h[j] -= delta
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:

            if c5_new < c5:

    """Sort h and assign top n//2 to 1, bottom to 0, middle to 0.5 if odd."""
    h_snapped = np.zeros(n)
    order = np.argsort(h)[::-1]
    n_ones = n // 2
    h_snapped[order[:n_ones]] = 1.0
    h_snapped[order[n_ones:]] = 0.0

        h_snapped[idx] = 0.5

        if c5s < best_c5:
            best_c5, best_h, best_n = c5s, hs.copy(), best_n

    # Phase 4: Multiple resolutions
    resolutions = [40, 50, 54, 60, 64, 70, 80, 90, 100, 120]
    for n_try in resolutions:
        if time.time() - t0 > budget_s - 40:
            break
        if n_try == best_n:
            continue
        h_try = interp_to_n(best_h, n_try)
        x_opt, _ = slsqp_optimize(h_try, max_iter=400, ftol=1e-12)
        x_opt = make_feasible(x_opt, n_try)
        c5 = compute_c5(x_opt)
        if time.time() - t0 < budget_s - 40:
            c5r, hr = local_refine(x_opt, n_iter=6000, step=0.05, seed=seed + n_try)
            if c5r < c5:
                c5, x_opt = c5r, hr
        if time.time() - t0 < budget_s - 30:
            h_snap2 = snap_binary(x_opt, n_try)
            c5b, hb = binary_local_search(h_snap2, n_iter=15000, seed=seed + 100 + n_try)
            if c5b < c5:
                c5, x_opt = c5b, hb
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, x_opt.copy(), n_try

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 5:
        x_f, _ = slsqp_optimize(best_h.copy(), max_iter=400, ftol=1e-14)
        x_f = make_feasible(x_f, best_n)
        c5f = compute_c5(x_f)
        if c5f < best_c5:
            best_c5, best_h = c5f, x_f.copy()

    if time.time() - t0 < budget_s - 2:
        c5f, hf = local_refine(best_h, n_iter=8000, step=0.02, seed=seed + 999)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    return (best_h, best_c5, best_n)
