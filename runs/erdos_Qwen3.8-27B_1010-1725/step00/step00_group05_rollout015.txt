def compute_gradient(h, m_active, dx):
    n = len(h)
    grad = np.zeros(n)
    lag = m_active - (n - 1)
    for j in range(n):
        # First term: h[j] * (1-h)[j + lag]
        idx1 = j + lag
        if 0 <= idx1 < n:
            grad[j] += (1.0 - h[idx1])
        # Second term: -h[j-lag] from (1-h) factor
        idx2 = j - lag
        if 0 <= idx2 < n:

def project_to_constraint(q, target_sum):

        s = proj.sum()
        if s > target_sum:

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def gradient_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:n] += (1.0 - h[0:n-lag])
            grad[0:n-lag] += -h[lag:n]
    else:
        a = -lag
        if n - a > 0:
            grad[0:n-a] += (1.0 - h[a:n])
            grad[a:n] += -h[0:n-a]
    return grad

def smooth_obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = corr.max()
    exps = np.exp(beta * (corr - mx))
    total = exps.sum()
    weights = exps / total
    obj = mx + np.log(total) / beta
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - (n - 1)
        grad += w * gradient_for_lag(h, lag, n)
    grad *= dx
    return obj, grad

def project_constraint(q, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(q - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(q - (lo + hi) * 0.5, 0.0, 1.0)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    best = h0.copy()
    best_c5 = compute_c5(best, dx)

    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = 0.02
    beta = 20.0
    
    while time.time() - start_time < time_limit:

        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad * grad
        m_hat = m_adam / (1 - b1 ** t_adam)
        v_hat = v_adam / (1 - b2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        new_c5 = compute_c5(new_h, dx)

            if t_adam % 100 == 0:
                beta *= 1.05
                lr *= 0.98
        else:
            lr *= 0.95
            if lr < 1e-5:
                break

        remaining = budget_s - (time.time() - start_time) - 30

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[lag:] += 1.0 - h[:L]
            g[:L] -= h[lag:]
    else:
        a = -lag
        L = n - a
        if L > 0:
            g[:L] += 1.0 - h[a:]
            g[a:] -= h[:L]
    return g

def smooth_grad(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    e = np.exp(beta * (c - mx))
    s = e.sum()
    w = e / s
    g = np.zeros(n)
    for m in range(2 * n - 1):

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[lag:] += 1.0 - h[:L]
            g[:L] -= h[lag:]
    else:
        a = -lag
        L = n - a
        if L > 0:
            g[:L] += 1.0 - h[a:]
            g[a:] -= h[:L]
    return g

def smooth_grad(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()

    s = e.sum()
    w = e / s
    g = np.zeros(n)
