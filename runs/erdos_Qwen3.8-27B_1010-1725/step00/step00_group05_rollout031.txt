import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def lag_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[:n-s] += (1.0 - h[s:n])
            grad[s:n] += -h[:n-s]
    else:
        s2 = -s
        if n - s2 > 0:
            grad[s2:n] += (1.0 - h[:n-s2])
            grad[:n-s2] += -h[s2:n]
    return grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return h

def interpolate_h(h0, n_new):
    n0 = len(h0)
    if n0 == n_new:
        return h0.copy()
    x0 = np.linspace(0, 1, n0, endpoint=False)
    x1 = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(x1, x0, h0)

def binary_round(h, n):
    """Round h to binary with exactly n/2 ones"""
    h = h.copy()
    # Sort by value descending, top n/2 become 1
    n_ones = n // 2
    threshold_idx = np.argpartition(h, n - n_ones)[n - n_ones]
    # Handle ties
    h_bin = np.zeros(n)
    h_bin[h >= threshold_idx] = 1.0
    # If too many ones, flip some
    while h_bin.sum() > n_ones:
        candidates = np.where((h_bin == 1) & (h < threshold_idx + 1e-10))[0]
        if len(candidates) == 0:
            candidates = np.where(h_bin == 1)[0]
        h_bin[candidates[0]] = 0
    while h_bin.sum() < n_ones:
        candidates = np.where((h_bin == 0) & (h > threshold_idx - 1e-10))[0]
        if len(candidates) == 0:
            candidates = np.where(h_bin == 0)[0]
        h_bin[candidates[0]] = 1
    return h_bin
