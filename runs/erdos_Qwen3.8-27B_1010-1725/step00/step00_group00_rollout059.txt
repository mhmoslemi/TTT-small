    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(2000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * 2.0 / n)

def _so(x):
    e = _opt['e']
    n = _opt['n']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return float((m + e * np.log(np.exp((corr - m) / e).sum())) * 2.0 / n)

def _rc(x):
    return c5(x)

def _ec(x):
    return x.sum() - _opt['n'] / 2.0

def _po(x):
    n = _opt['n']
    mu = _opt['mu']
    e = _opt['e']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    sm = m + e * np.log(np.exp((corr - m) / e).sum())
    pen = (x.sum() - n / 2.0) ** 2
    return float(sm * 2.0 / n + mu * pen)

def interp_h(h, n):
    x0 = (np.arange(len(h)) + 0.5) / len(h)
    x1 = (np.arange(n) + 0.5) / n
    return project_h(np.interp(x1, x0, h), n)

def slsqp_smooth(h0, n, e, iters):
    _opt['e'] = e
    _opt['n'] = n
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ec}]
    try:
        r = minimize(_so, h0, method='SLSQP', bounds=bnd, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-12})
        if r.x is not None:
            return project_h(r.x, n)
    except Exception:
        pass
    return None

def slsqp_raw(h0, n, iters):
    _opt['n'] = n
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ec}]
    try:
        r = minimize(_rc, h0, method='SLSQP', bounds=bnd, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-10})
        if r.x is not None:
            return project_h(r.x, n)
    except Exception:
        pass
    return None

def lbfgsb_pen(h0, n, e, mu, iters):
    _opt['e'] = e
    _opt['n'] = n
    _opt['mu'] = mu
    bnd = [(0.0, 1.0)] * n
    try:
        r = minimize(_po, h0, method='L-BFGS-B', bounds=bnd,
                     options={'maxiter': iters, 'ftol': 1e-12, 'gtol': 1e-8})
        if r.x is not None:
            return project_h(r.x, n)
    except Exception:
        pass
    return None

def _block_init(n, rng):
    h = np.zeros(n)
    nb = rng.randint(2, max(3, n // 8))
    bs = n // nb
    for b in range(nb):
        if b % 2 == 0:
            h[b * bs:(b + 1) * bs] = 1.0
    return project_h(h, n)

def _rand_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    i1 = np.where(h == 1)[0]
    i0 = np.where(h == 0)[0]
    tgt = n // 2
    if len(i1) > tgt:
        f = rng.choice(i1, len(i1) - tgt, replace=False)
        h[f] = 0.0
    elif len(i1) < tgt:
        f = rng.choice(i0, tgt - len(i1), replace=False)
        h[f] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rs = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    iv = globals().get('initial_h_values')
    if iv is not None:
        best_h = project_h(np.array(iv, dtype=float), len(iv))
        best_c = c5(best_h)

    n_list = []
    if best_h is not None:
        n_list.append(len(best_h))
    for v in [45, 50, 60, 72, 80, 90, 100, 120]:
        if v not in n_list:
            n_list.append(v)
    n_list = sorted(set(n_list))
    tlim = t0 + budget_s * 0.80

    for ni, n in enumerate(n_list):
        if time.time() > tlim:
            break
        starts = []
        if best_h is not None:
            if len(best_h) == n:
                starts.append(best_h.copy())
            else:
                starts.append(interp_h(best_h, n))
            p = starts[0] + 0.04 * rs.randn(n)
            starts.append(project_h(p, n))
            p2 = starts[0] + 0.1 * rs.randn(n)
            starts.append(project_h(p2, n))
        for _ in range(3):
            starts.append(_block_init(n, rs))
        for _ in range(2):
            starts.append(_rand_binary(n, rs))
        if len(starts) == 0:
            starts.append(project_h(0.5 * np.ones(n), n))

        for h0 in starts:
            if time.time() > tlim:
                break
            cur = h0.copy()
            cc = c5(cur)
            if cc < best_c:
                best_c, best_h = cc, cur.copy()

            for e in [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
                if time.time() > tlim:
                    break
                r = slsqp_smooth(cur, n, e, 150)
                if r is not None:
                    cr = c5(r)
                    if cr < c5(cur):
                        cur = r
                    if cr < best_c:
                        best_c, best_h = cr, cur.copy()

            if time.time() < tlim:
                r = lbfgsb_pen(cur, n, 0.05, 100.0, 300)
                if r is not None:
                    cr = c5(r)
                    if cr < c5(cur):
                        cur = r
                    if cr < best_c:
                        best_c, best_h = cr, cur.copy()

            if time.time() < tlim:
                r = slsqp_raw(cur, n, 120)
                if r is not None:
                    cr = c5(r)
                    if cr < best_c:
                        best_c, best_h = cr, r.copy()

    # Final polishing
    if best_h is not None and time.time() < t0 + budget_s * 0.95:
        n = len(best_h)
        for e in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > t0 + budget_s * 0.97:
                break
            r = slsqp_smooth(best_h, n, e, 250)
            if r is not None:
                cr = c5(r)
                if cr < best_c:
                    best_c, best_h = cr, r
        if time.time() < t0 + budget_s * 0.99:
            r = slsqp_raw(best_h, n, 250)
            if r is not None:
                cr = c5(r)
                if cr < best_c:
                    best_c, best_h = cr, r
        # extra L-BFGS pass
        if time.time() < t0 + budget_s * 0.99:
            r = lbfgsb_pen(best_h, n, 0.01, 500.0, 500)
            if r is not None:
                cr = c5(r)
                if cr < best_c:
                    best_c, best_h = cr, r

    return (best_h, best_c, len(best_h))
