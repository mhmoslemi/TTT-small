        res = minimize(penalized_obj, h0, args=(target_sum,), method='Nelder-Mead',
                       options={'maxiter': 15000, 'xatol': 1e-11, 'fatol': 1e-14,
                                'adaptive': True})
        h_nm = project_to_sum(np.clip(res.x, 0, 1), target_sum)
        c5_nm = compute_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm
    except Exception:
        pass

    # Phase 2: Powell refinement
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(penalized_obj, best_h, args=(target_sum,), method='Powell',
                           bounds=bounds, options={'maxiter': 5000, 'xtol': 1e-11, 'ftol': 1e-13})
            h_pw = project_to_sum(np.clip(res.x, 0, 1), target_sum)
            c5_pw = compute_c5(h_pw)
            if c5_pw < best_c5:
                best_c5 = c5_pw
                best_h = h_pw
        except Exception:
            pass

    # Phase 3: Binary local search (step function optimization)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        n_ones = int(round(target_sum))
        # Initialize binary from continuous best
        h_bin = np.zeros(n)
        top_idx = np.argsort(-best_h)[:n_ones]
        h_bin[top_idx] = 1.0
        best_bin_c5 = compute_c5(h_bin)

        # Simulated annealing style local search
        temperature = 0.001
        cooldown = 0.9995
        max_rounds = 30000
        for rd in range(max_rounds):
            if time.time() - t_start > budget_s - 20:
                break
            if rd % 1000 == 0:
                temperature *= cooldown
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            oi, zo = ones[i], zeros[j]
            h_bin[oi] = 0.0
            h_bin[zo] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - best_bin_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-10)):
                best_bin_c5 = new_c5
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            else:
                h_bin[oi] = 1.0
                h_bin[zo] = 0.0
            if best_bin_c5 < best_c5:
                best_c5 = best_bin_c5
                best_h = h_bin.copy()

    # Phase 4: Nelder-Mead from binary solution (if better)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        try:
            res = minimize(penalized_obj, best_h, args=(target_sum,), method='Nelder-Mead',
                           options={'maxiter': 10000, 'xatol': 1e-11, 'fatol': 1e-14,
                                    'adaptive': True})
            h_ref = project_to_sum(np.clip(res.x, 0, 1), target_sum)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
        except Exception:
            pass

    # Phase 5: Multiple Nelder-Mead restarts with small perturbations
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        n_restarts = min(5, int((budget_s - elapsed) / 30))
        for r in range(n_restarts):
            if time.time() - t_start > budget_s - 15:
                break
            h_r = best_h + rng.normal(0, 0.03, n)
            h_r = project_to_sum(np.clip(h_r, 0, 1), target_sum)
            try:
                res = minimize(penalized_obj, h_r, args=(target_sum,), method='Nelder-Mead',
                               options={'maxiter': 8000, 'xatol': 1e-11, 'fatol': 1e-14,
                                        'adaptive': True})
                h_opt = project_to_sum(np.clip(res.x, 0, 1), target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
            except Exception:
                pass

    # Final feasibility check
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
