def directed_values(h):
    n = len(h)
    V = np.zeros(2 * n - 1, dtype=float)
    V[0] = float(np.dot(h, 1.0 - h))
    for s in range(1, n):
        L = n - s
        V[2*s-1] = float(np.dot(h[s:], 1.0 - h[:L]))
        V[2*s] = float(np.dot(h[:L], 1.0 - h[s:]))
    return V

def build_VA(h):
    n=len(h); V=directed_values(h); A=np.zeros((2*n-1,n),dtype=float)
    A[0,:]=1.0-2.0*h
    for s in range(1,n):
       L=n-s; rp=2*s-1; rn=2*s
       A[rp,s:n] += 1.0
       A[rp,s:n] -= h[:L]
       A[rp,0:L] -= h[s:n]
       A[rn,0:L] += 1.0
       A[rn,0:L] -= h[s:n]
       A[rn,s:n] -= h[:L]
    return V,A

    return best_t, best_f


def eval_c(h):
    return float(np.max(directed_values(h))) * (2.0 / len(h))


def random_h_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    else:
        p = int(rng.randint(0, n))
        h[p] = 0.5
        mask = np.ones(n, dtype=bool)
        mask[p] = False
        idx = rng.choice(np.nonzero(mask)[0], size=(n - 1) // 2, replace=False)
        h[idx] = 1.0
    return h


def resample_h(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    return np.interp(np.linspace(0.0, 1.0, m), np.linspace(0.0, 1.0, n), h)


def random_kick(h, rng, strength, moves):
    n = len(h)
    h2 = h.copy()
    for _ in range(int(moves)):
        i = int(rng.randint(0, n))
        j = int(rng.randint(0, n))
        if i == j:
            continue
        a = float(h2[i])
        b = float(h2[j])
        lo = max(-a, b - 1.0)
        hi = min(1.0 - a, b)
        if hi <= lo:
            continue
        t = float(rng.uniform(-strength, strength))
        if t < lo:
            t = lo
        elif t > hi:
            t = hi
        h2[i] = a + t
        h2[j] = b - t
    np.clip(h2, 0.0, 1.0, out=h2)
    return fix_sum(h2, 0.5 * n)


def gradient_improve(h, c, max_steps=4):
    n = len(h)
    target = 0.5 * n
    for _ in range(int(max_steps)):
        v, a = build_va(h)
        vm = float(v.max())
        act = np.nonzero(v >= vm - (1e-7 + 1e-6 * vm))[0]
        if len(act) == 0:
            break
        g = a[act, :].sum(axis=0)
        g = g - g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        d = g / gn
        ok = False
        for step in (0.1, 0.05, 0.02, 0.01, 0.005, 0.002):
            h2 = h - step * d
            h2 = np.clip(h2, 0.0, 1.0)
            h2 = fix_sum(h2, target)
            c2 = eval_c(h2)
            if c2 < c - 1e-10:
                h, c, ok = h2, c2, True
                break
        if not ok:
            for step in (0.05, 0.02, 0.01):
                h2 = h + step * d
                h2 = np.clip(h2, 0.0, 1.0)
                h2 = fix_sum(h2, target)
                c2 = eval_c(h2)
                if c2 < c - 1e-10:
                    h, c, ok = h2, c2, True
                    break
        if not ok:
            break
    return h, c


def polish(h, c, rng, time_limit):
    t0 = time.time()
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(c)
    if n < 2:
        return best_h, best_c

    pairs = []
    dists = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            dists.append(j - i)
    pairs = np.array(pairs, dtype=np.int64)
    dists = np.array(dists, dtype=np.int64)
    m = len(pairs)

    iters = 45 if n <= 80 else 30
    coarse = 31 if n <= 80 else 17
    incidents = build_incidents(n)
    no_improve = 0

    while time.time() - t0 < time_limit:
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()

        h = fix_sum(np.clip(h, 0.0, 1.0), target)
        v, a = build_va(h)
        c = float(v.max()) * dx
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()

        start_c = c
        order = rng.permutation(m)
        tmp = np.empty_like(v)
        cnt = 0

        for idx in order:
            if (cnt & 127) == 0 and time.time() - t0 > time_limit - 5.0:
                break
            cnt += 1
            i = int(pairs[idx, 0])
            j = int(pairs[idx, 1])
            d = int(dists[idx])
            ai = float(h[i])
            bj = float(h[j])
            lo = max(-ai, bj - 1.0) + 1e-12
            hi = min(1.0 - ai, bj) - 1e-12
            if hi <= lo:
                continue

            coeff = a[:, i] - a[:, j]
            r1 = 2 * d - 1
            r2 = 2 * d
            t, val = line_min(v, coeff, r1, r2, lo, hi, iters, coarse, tmp)
            if val * dx < c - 1e-10:
                h[i] = ai + t
                h[j] = bj - t
                new_v = v + coeff * t
                new_v[0] -= 2.0 * t * t
                new_v[r1] += t * t
                new_v[r2] += t * t
                update_a_inc(a, i, t, incidents)
                update_a_inc(a, j, -t, incidents)
                v = new_v
                c = float(v.max()) * dx

        if c < start_c - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                break

            hg, cg = gradient_improve(h, c, 4)
            if cg < c - 1e-10:
                h, c, no_improve = hg, cg, 0
                continue

            temp = 0.0003 if n < 80 else 0.0001
            accepted = False
            for k in range(12):
                strength = 0.05 if k < 6 else 0.12
                h2 = random_kick(h, rng, strength, 2)
                c2 = eval_c(h2)
                if c2 < c - 1e-10 or (c2 < c + temp and float(rng.random()) < float(np.exp(-(c2 - c) / temp))):
                    h, c, accepted = h2, c2, True
                    break
            if not accepted and no_improve >= 4:
                break

    hb = np.clip(h, 0.0, 1.0)
    hb = fix_sum(hb, target)
    cb = eval_c(hb)
    if cb < best_c - 1e-12:
        best_h, best_c = hb, cb
    elif best_h is None:
        best_h, best_c = hb, cb
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(int(seed))
    best_h = None
    best_c = float("inf")
    parent = None

    init = globals().get("initial_h_values", None)
    if init is not None:
        parent = np.asarray(init, dtype=float)
        h = fix_sum(parent, 0.5 * len(parent))
        c = eval_c(h)
        best_h, best_c = h.copy(), c
        left = budget_s - (time.time() - t0)
        if left > 20.0:
            alloc = min(max(left * 0.55, 20.0), 600.0)
            h, c = polish(h, c, rng, alloc)
            if c < best_c:
                best_h, best_c = h.copy(), c

    left = budget_s - (time.time() - t0)
    if parent is not None and left > 60.0:
        n0 = len(parent)
        cands = []
        for m in [n0 + 1, n0 + 2, n0 + 3, 50, 60, 80, 100, 120]:
            if m >= 2 and m % 2 == 0 and m != n0 and m <= 180 and m not in cands:
                cands.append(m)
        for m in cands:
            left = budget_s - (time.time() - t0)
            if left < 50.0:
                break
            h = resample_h(parent, m)
            h = fix_sum(h, 0.5 * m)
            c = eval_c(h)
            if c > best_c + 0.08:
                continue
            alloc = min(left * 0.18, 120.0)
            h2, c2 = polish(h, c, rng, alloc)
            if c2 < best_c:
                best_h, best_c = h2.copy(), c2

    left = budget_s - (time.time() - t0)
    cap = 90.0 if parent is not None else 250.0
    for m in [50, 60, 80, 100]:
        left = budget_s - (time.time() - t0)
        if left < 40.0:
            break
        h = random_h_balanced(m, rng)
        c = eval_c(h)
        if c < best_c + 0.05:
            alloc = min(left * 0.20, cap)
            h2, c2 = polish(h, c, rng, alloc)
            if c2 < best_c:
                best_h, best_c = h2.copy(), c2

    if best_h is None:
        h = random_h_balanced(50, rng)
        c = eval_c(h)
        left = max(budget_s - (time.time() - t0), 1.0)
        h, c = polish(h, c, rng, left)
        best_h, best_c = h, c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, 0.5 * len(best_h))
    best_c = eval_c(best_h)
    return best_h, float(best_c), len(best_h)
