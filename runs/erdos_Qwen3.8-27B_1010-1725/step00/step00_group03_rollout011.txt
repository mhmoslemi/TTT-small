def gradient_c5(h, k, n):
    """Gradient of sum_i h[i]*(1-h[i+k]) with respect to h[j] for a specific lag k"""
    # k is the lag (can be negative)
    # f_k = sum_i h[i]*(1-h[i+k])

        if 0 <= j + k < n:
            g[j] += (1 - h[j + k])
        if 0 <= j - k < n:
            g[j] -= h[j - k]
    return g

    """Project x onto {0<=z<=1, sum(z)=target_sum}"""
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        z = np.clip(x - mid, 0, 1)
        if z.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target_sum - z.sum()
    if abs(diff) < 1e-10:
        return z
    # Distribute remainder
    idx = np.where((z > 0) & (z < 1))[0]
    if len(idx) > 0:
        per = diff / len(idx)
        z[idx] = np.clip(z[idx] + per, 0, 1)
    return z
