    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return np.max(corr) * dx

    n = len(x)
    dx = 2.0 / n
    om = 1.0 - x
    corr = np.correlate(x, om, mode="full") * dx

def c5_grad(x, beta=100.0):
    n = len(x)
    dx = 2.0 / n
    om = 1.0 - x

def c5_grad(x, beta=100.0):
    n = len(x)
    dx = 2.0 / n
    om = 1.0 - x
    corr = np.correlate(x, om, mode="full") * dx
    m = np.max(corr)
    ex = np.exp(beta * (corr - m))
    s = np.sum(ex)
    if s < 1e-300:
        return np.zeros(n)
    w = ex / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):

        else:
            k = -lag
            grad[:n - k] += wi * dx * om[k:]
            grad[k:] -= wi * dx * x[:n - k]
    return grad

def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        mask = x < 1.0
    else:
        mask = x > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return x
    x[mask] += diff / cnt
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def upsample(h, n_target):
    n_src = len(h)
    if n_target == n_src:
        return h.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    hu = np.interp(x_tgt, x_src, h)
    target_sum = n_target / 2.0
    hu = project_feasible(hu, target_sum)
    return hu

def slsqp_optimize(h0, n_pts, maxiter=200, ftol=1e-10):
    target_sum = n_pts / 2.0
    constraints = [{'type': 'eq', 'fun': lambda x, t=target_sum: np.sum(x) - t,
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, 1.0)] * n_pts
    try:
        res = minimize(c5_obj, h0, jac=c5_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(res.x >= -1e-6) and np.all(res.x <= 1 + 1e-6):
            h_out = project_feasible(res.x, target_sum)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h0.copy(), compute_c5(h0)

def coordinate_descent(h, n_pts, time_limit, t_start, steps=None):
    if steps is None:
        steps = [0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.2, -0.2]
    t_limit = t_start + time_limit
    pairs = [(0, 1), (1, 0)]
    n = n_pts
    for sweep in range(500):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            best_val = old_val
            cur_c5 = compute_c5(h)
            for d in steps:
                nv = old_val + d
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                # Maintain sum by adjusting a random other coordinate
                j = (i + 1) % n
                diff = h[i] - old_val
                old_j = h[j]
                h[j] = old_j - diff
                if h[j] < 0 or h[j] > 1:
                    h[j] = np.clip(h[j], 0, 1)
                    # Re-adjust i
                    h[i] = old_val + (h[j] - old_j)
                o = compute_c5(h)
                if o < cur_c5 - 1e-12:
                    cur_c5 = o
                    best_val = h[i].copy()
                else:
                    h[j] = old_j
            h[i] = best_val if best_val is not old_val else old_val
            if best_val is not old_val:
                improved = True
        if not improved:
            break
    return h
