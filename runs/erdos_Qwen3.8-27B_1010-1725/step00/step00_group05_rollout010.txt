import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h


def to_binary_exact(h, n, rng):
    b = (h > 0.5).astype(np.float64)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1.0)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
        s -= 1
    while s < target:
        zeros = np.where(b == 0.0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
        s += 1
    return b
