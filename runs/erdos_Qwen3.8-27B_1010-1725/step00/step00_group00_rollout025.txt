import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_c5_val(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < 1e-14:
        return 0.0
    shifted = t * (corr - m)
    return float(m + np.log(np.exp(shifted).sum()) / t)


def smooth_c5_grad_fn(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m < 1e-14:
        return np.zeros(n)
    shifted = t * (corr - m)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        pm = p[idx]
        if pm < 1e-12:
            continue
        k = idx - (n - 1)
        pm_dx = pm * dx
        if k >= 0:
            if k < n:
                grad[k:] += pm_dx * (1.0 - h[:n - k])
                grad[:n - k] += pm_dx * (-h[k:])
        else:
            k2 = -k
            if k2 < n:
                grad[:n - k2] += pm_dx * (1.0 - h[k2:])
                grad[k2:] += pm_dx * (-h[:n - k2])
    return grad


def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h += (target - s) / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        cand = np.where(h < 1.0)[0]
        if len(cand) > 0:
            idx = cand[np.argmin(1.0 - h[cand])]
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def interp_to(h_coarse, n_fine):
    n_c = len(h_coarse)
    if n_fine == n_c:
        return h_coarse.copy()
    pos = np.linspace(0, n_c - 1, n_fine)
    i0 = np.floor(pos).astype(int)
    i1 = np.minimum(i0 + 1, n_c - 1)
    frac = pos - i0
    h_fine = h_coarse[i0] * (1.0 - frac) + h_coarse[i1] * frac
    return project_h(h_fine, n_fine)


def obj_fun(h, n, t):
    return smooth_c5_val(h, n, t)


def obj_jac(h, n, t):
    return smooth_c5_grad_fn(h, n, t)


def constraint_eq(h, n):
    return h.sum() - n / 2.0


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(44)
        h_init[:22] = 1.0
        h_init = project_h(h_init, 44)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    sizes = [44, 88, 176, 352]
    current_h = h_init.copy()
    start_idx = 0
    for i, s in enumerate(sizes):
        if len(h_init) <= s:
            current_h = h_init.copy() if len(h_init) == s else interp_to(h_init, s)
            start_idx = i
            break
    else:
        current_h = interp_to(h_init, sizes[0])

    for i in range(start_idx, len(sizes)):
        n = sizes[i]
        if time_left() < 50:
            break
        if i > start_idx or len(current_h) != n:
            current_h = interp_to(current_h, n)

        t_smooth = 40.0 + 40.0 * i
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': constraint_eq, 'args': (n,)}

        n_restarts = 3 if n <= 176 else 2
        for r in range(n_restarts):
            if time_left() < 25:
                break
            if r == 0:
                h_start = current_h.copy()
            else:
                h_start = project_h(current_h + rng.randn(n) * 0.08, n)
            try:
                res = minimize(obj_fun, h_start, args=(n, t_smooth),
                               jac=obj_jac, method='SLSQP',
                               bounds=bounds, constraints=[cons],
                               options={'maxiter': 160, 'ftol': 1e-9})
                h_cand = project_h(res.x, n)
                c5_val = compute_c5(h_cand)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_cand.copy()
                current_h = h_cand
            except Exception:
                pass

        if n % 2 == 0 and time_left() > 25:
            h_bin = (current_h >= 0.5).astype(float)
            s = h_bin.sum()
            target = float(n // 2)
            if s != target:
                order = np.argsort(np.abs(current_h - 0.5))
                if s > target:
                    cnt = int(s - target)
                    for j in range(cnt):
                        h_bin[order[j]] = 0.0
                else:
                    cnt = int(target - s)
                    for j in range(cnt):
                        h_bin[order[j]] = 1.0
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                current_h = h_bin.copy()

    # Final polish
    if time_left() > 35:
        n_best = len(best_h)
        t_final = 300.0
        bounds = [(0.0, 1.0)] * n_best
        cons = {'type': 'eq', 'fun': constraint_eq, 'args': (n_best,)}
        try:
            res = minimize(obj_fun, best_h, args=(n_best, t_final),
                           jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=[cons],
                           options={'maxiter': 250, 'ftol': 1e-11})
            h_final = project_h(res.x, n_best)
            c5_final = compute_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
        except Exception:
            pass

    # Verify with pre-imported evaluator if available
    try:
        verified = evaluate_erdos_solution(best_h)
        if isinstance(verified, float) and verified < best_c5:
            best_c5 = verified
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
