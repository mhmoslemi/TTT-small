import time

def _corr(h):
    return correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad(h, dx, beta):
    n = len(h)
    c = _corr(h)
    s = beta * c
    m = np.max(s)
    e = np.exp(s - m)
    w = e / e.sum()
    h_rev = h[::-1]
    one_minus_h = 1.0 - h
    grad = np.empty(n)
    for j in range(n):
        g1 = np.dot(w[n-1-j : 2*n-1-j], one_minus_h)
        g2 = np.dot(w[j : j+n], h_rev)
        grad[j] = dx * (g1 - g2)
    return grad

def _prj(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):

        if np.sum(np.clip(h - t, 0.0, 1.0)) > ts:

    xn = np.linspace(0, 1, n_new)
