from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_logsumexp(x, beta):
    x = beta * x
    x_max = np.max(x)
    if x_max > 500:
        return x_max / beta
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / beta

def make_objective(n, penalty_weight, beta):
    dx = 2.0 / n
    target_sum = n / 2.0
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = smooth_logsumexp(corr, beta) * dx
        penalty = penalty_weight * (np.sum(h) - target_sum) ** 2
        return c5 + penalty
    return obj

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)

            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

    x_scaled = beta * x

    if x_max > 700:

    shifted = x_scaled - x_max
    return (x_max + np.log(np.sum(np.exp(shifted)))) / beta
