import numpy as np
import time

def _project(hv, ts):
    hv = np.clip(np.asarray(hv, dtype=float), 0.0, 1.0)
    for _ in range(50):
        d = ts - hv.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = np.where(hv < 1.0)[0]
        else:
            idx = np.where(hv > 0.0)[0]
        if len(idx) == 0:
            break
        hv[idx] += d / len(idx)
        np.clip(hv, 0.0, 1.0, out=hv)
    return hv

def _c5f(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _get_init(n_pts):
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        if len(h0) == n_pts:
            return h0.copy()
        xo = (np.arange(len(h0)) + 0.5) * 2.0 / len(h0)
        xn = (np.arange(n_pts) + 0.5) * 2.0 / n_pts
        return np.interp(xn, xo, h0)
    except (NameError, UnboundLocalError):
        return np.full(n_pts, 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 10

    bh, bc5, bn = None, 999.0, 0

    sizes = []
    try:
        sizes.append(len(initial_h_values))
    except (NameError, UnboundLocalError):
        pass
    if not sizes:
        sizes.append(45)
    for s in [50, 64, 90]:
        if s not in sizes:
            sizes.append(s)

    total_avail = max(60, budget_s - 40)
    n_sz = len(sizes)
    if n_sz == 1:
        weights = [1.0]
    else:
        weights = [0.45] + [0.55 / (n_sz - 1)] * (n_sz - 1)

    for si, n in enumerate(sizes):
        p_end = min(time.time() + weights[si] * total_avail, t_end - 15)
        if time.time() > p_end:
            continue

        dx = 2.0 / n
        ts = n / 2.0
        h = _project(_get_init(n), ts)
        c5 = _c5f(h, dx)
        if c5 < bc5:
            bh, bc5, bn = h.copy(), c5, n

        T, Tlo = 0.04, 0.002
        stuck = 0
        iter_count = 0

        while time.time() < p_end:
            iter_count += 1
            r = rng.random()
            paired = False
            oi = oj = 0
            fsave = None

            if r < 0.60:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue

                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                e = max(lo + 1e-12, min(hi - 1e-12, e))
                if abs(e) < 1e-12:
                    continue
                oi, oj = h[i], h[j]
                h[i] = oi + e
                h[j] = oj - e
                paired = True
            elif r < 0.75:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                e = rng.uniform(-0.45, 0.45)

                oi, oj = h[i], h[j]
                h[i] = oi + e
                h[j] = oj - e
                c5n = _c5f(h, dx)
                if c5n <= c5 or rng.random() < np.exp(-(c5n - c5) / max(Tp, 1e-6)):
                    c5 = c5n
                    if c5 < bc5:
                        bh, bc5, bn = h.copy(), c5, n
                else:
                    h[i], h[j] = oi, oj
                Tp = max(Tp * 0.9998, 1e-6)
