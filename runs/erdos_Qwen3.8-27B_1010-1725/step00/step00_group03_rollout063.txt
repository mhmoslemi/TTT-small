        cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
        try:
            r = minimize(_smooth_og, h0, jac=True, method='SLSQP',
                         bounds=bds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-10})
            return _project(r.x, S)
        except Exception:
            return h0
    else:
        try:
            r = minimize(_pen_obj, h0, jac=_pen_grad, method='L-BFGS-B',
                         bounds=bds,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            return _project(r.x, S)
        except Exception:
            return h0


def _subgrad_polish(h, S, n_iter):
    c_best = _eval_c5(h)
    h_best = h.copy()
    for _ in range(n_iter):
        _, g = _hard_og(h)
        improved = False
        for lr in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = _project(h - lr * g, S)
            c_try = _eval_c5(h_try)
            if c_try < c_best - 1e-13:
                h_best, c_best = h_try, c_try
                h = h_try
                improved = True
                break
        if not improved:
            break
    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None
    best_h, best_c5 = None, 1e9

    if h_init is not None:
        n0 = len(h_init)
        S0 = n0 / 2.0
        h0 = _project(h_init, S0)
        c = _eval_c5(h0)
        if c < best_c5:
            best_c5, best_h = c, h0.copy()
        for att in range(8):
            if time.time() - t0 > budget_s * 0.22:
                break
            sc = 0.18 / (att + 1)
            hs = h0 + rng.normal(0.0, sc, n0)
            hs = _project(hs, S0)
            hr = _do_opt(hs, S0, 30.0, 500)
            c = _eval_c5(hr)
            if c < best_c5:
                best_c5, best_h = c, hr.copy()
            if c <= _eval_c5(h0):
                h0 = hr

    for n_t in [80, 100, 140, 200, 300, 400]:
        if time.time() - t0 > budget_s * 0.65:
            break
        S = n_t / 2.0
        starts = []
        if best_h is not None and len(best_h) != n_t:
            hi = np.interp(np.linspace(0.0, 2.0, n_t, False),
                           np.linspace(0.0, 2.0, len(best_h), False), best_h)
            starts.append(_project(hi, S))
        x = np.linspace(0.0, 2.0, n_t, endpoint=False)
        for f in [3, 5, 7, 11]:
            starts.append(_project(0.5 + 0.35 * np.sin(np.pi * f * x), S))
        starts.append(_project(rng.uniform(0.0, 1.0, n_t), S))
        for hs in starts:
            if time.time() - t0 > budget_s * 0.65:
                break
            hr = _do_opt(hs, S, 30.0, min(400, 80000 // max(S, 1)))
            c = _eval_c5(hr)
            if c < best_c5:
                best_c5, best_h = c, hr.copy()
            if time.time() - t0 < budget_s * 0.65:
                hr2 = _do_opt(hr, S, 100.0, 250)
                c2 = _eval_c5(hr2)
                if c2 < best_c5:
                    best_c5, best_h = c2, hr2.copy()

    if best_h is not None and time.time() - t0 < budget_s * 0.88:
        nb = len(best_h)
        Sb = nb / 2.0
        for att in range(12):
            if time.time() - t0 > budget_s * 0.93:
                break
            sr = 0.004 / (att + 1) + 0.001
            hs = best_h + rng.normal(0.0, sr, nb)
            hs = _project(hs, Sb)
            hr = _do_opt(hs, Sb, 60.0 + att * 10.0, 400)
            c = _eval_c5(hr)
            if c < best_c5:
                best_c5, best_h = c, hr.copy()

    if best_h is not None and time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = _subgrad_polish(best_h, len(best_h) / 2.0, 500)

    if best_h is not None:
        best_h = _project(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
        best_c5 = _eval_c5(best_h)
    else:
        best_h = np.full(100, 0.5)
        best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
