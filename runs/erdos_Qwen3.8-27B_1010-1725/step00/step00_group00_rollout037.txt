    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0, method='SLSQP', jac=gradient,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return enforce(res.x, ts)
    except Exception:
        return h0

def structured_starts(n, rng):
    starts = []
    # Alternating
    h_alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(enforce(h_alt, n / 2.0))
    # Block
    h_blk = np.zeros(n)
    h_blk[:n // 2] = 1.0
    if n % 2 == 1:
        h_blk[n // 2] = 0.5
    starts.append(enforce(h_blk, n / 2.0))
    # Triangular
    x = np.linspace(0.0, 2.0, n)
    h_tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    starts.append(enforce(h_tri, n / 2.0))
    # Periodic blocks of size 3
    h_per = np.zeros(n)
    for i in range(n):
        h_per[i] = 1.0 if (i % 3 == 0) else 0.0
    starts.append(enforce(h_per, n / 2.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except NameError:
        best_h = np.ones(45) * 0.5

    best_c5 = compute_c5(best_h)
    base_n = len(best_h)

    # Phase 1: optimize at base grid with diverse starts
    n = base_n
    h1 = opt_slsqp(best_h, n, 400)
    c = compute_c5(h1)
    if c < best_c5:
        best_c5, best_h = c, h1.copy()

    starts = structured_starts(n, rng)
    for h_s in starts:
        if time.time() - t0 > budget_s * 0.15:
            break
        h1 = opt_slsqp(h_s, n, 200)
        c = compute_c5(h1)
        if c < best_c5:
            best_c5, best_h = c, h1.copy()

    for i in range(20):
        if time.time() - t0 > budget_s * 0.25:
            break
        h_r = enforce(rng.random(n), n / 2.0)
        h1 = opt_slsqp(h_r, n, 150)
        c = compute_c5(h1)
        if c < best_c5:
            best_c5, best_h = c, h1.copy()

    # Phase 2: multi-resolution refinement
    cur_n = n
    for factor in [2, 3, 4, 6, 8, 12]:
        nn = base_n * factor
        if nn >= 900:
            break
        if time.time() - t0 > budget_s * 0.82:
            break

        x_old = np.linspace(0.0, 2.0, cur_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, nn, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = enforce(h_interp, nn / 2.0)

        h2 = opt_slsqp(h_interp, nn, 300)
        c = compute_c5(h2)
        improved = False
        if c < best_c5:
            best_c5, best_h, cur_n = c, h2.copy(), nn
            improved = True

        # Additional random starts at this resolution
        for ri in range(4):
            if time.time() - t0 > budget_s * 0.82:
                break
            h_r = enforce(rng.random(nn) * 1.8, nn / 2.0)
            h2 = opt_slsqp(h_r, nn, 120)
            c = compute_c5(h2)
            if c < best_c5:
                best_c5, best_h, cur_n = c, h2.copy(), nn
                improved = True

    # Phase 3: fine-tune at best grid
    if time.time() - t0 < budget_s * 0.93:
        fn = len(best_h)
        h3 = opt_slsqp(best_h, fn, 600, ftol=1e-15)
        c = compile_c5 = compute_c5(h3)
        if c < best_c5:
            best_c5, best_h = c, h3.copy()

    # Phase 4: local perturbation refinement
    if time.time() - t0 < budget_s * 0.96:
        fn = len(best_h)
        for pi in range(8):
            if time.time() - t0 > budget_s * 0.96:
                break
            pert = rng.normal(0, 0.05, fn)
            h_p = enforce(best_h + pert, fn / 2.0)
            h3 = opt_slsqp(h_p, fn, 200)
            c = compute_c5(h3)
            if c < best_c5:
                best_c5, best_h = c, h3.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    ts = len(best_h) / 2.0
    s = best_h.sum()
    if s > 0:
        best_h = best_h * (ts / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, len(best_h))
