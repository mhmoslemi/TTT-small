import numpy as np
import time


def evaluate_c5_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def project_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-15:
            h += min(1.0, diff / tr) * room
    else:
        space = h.copy()
        ts = space.sum()
        if ts > 1e-15:
            h -= min(1.0, -diff / ts) * space
    return np.clip(h, 0.0, 1.0)


def make_initial(n, rng):
    h = np.zeros(n)
    seg = n // 5
    for i in range(0, n, 2 * seg):
        h[i:min(i + seg, n)] = 1.0
    return project_constraint(h, n / 2.0)


def subgradient_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    lag = k_star - (n_points - 1)
    grad = np.zeros(n_points)
    jl = np.arange(n_points) - lag
    jr = np.arange(n_points) + lag
    valid_l = (jl >= 0) & (jl < n_points)
    valid_r = (jr >= 0) & (jr < n_points)
    contrib_l = np.zeros(n_points)
    contrib_r = np.zeros(n_points)
    contrib_l[valid_l] = 1.0 - h[jl[valid_l]]
    contrib_r[valid_r] = -h[jr[valid_r]]
    grad = dx * (contrib_l + contrib_r)
    return grad, k_star


def sa_optimize(h, target_sum, n_points, rng, max_time, T_init=0.005):
    t0 = time.time()
    T = T_init
    best_h = h.copy()
    best_c5 = evaluate_c5_h(best_h)
    patience = 0
    while time.time() - t0 < max_time:
        move_type = rng.random()
        h_new = h.copy()
        if move_type < 0.4:
            # Swap two values
            i = rng.randint(n_points)
            j = rng.randint(n_points)
            if i != j:
                h_new[i], h_new[j] = h[j], h[i]
        elif move_type < 0.7:
            # Gaussian perturbation + project
            n_pert = rng.randint(1, max(2, n_points // 8) + 1)
            idx = rng.choice(n_points, size=n_pert, replace=False)
            h_new[idx] += rng.randn(n_pert) * 0.06
            h_new = project_constraint(h_new, target_sum)
        else:
            # Block shift
            blen = rng.randint(2, max(3, n_points // 6) + 1)
            start = rng.randint(0, max(1, n_points - blen))
            shift = rng.choice([-2, -1, 1, 2])
            end = min(start + blen, n_points)
            new_start = start + shift
            new_end = end + shift
            if new_start < 0 or new_end > n_points:
                continue
            if (new_start > start and new_end <= start + blen + abs(shift)) or \
               (new_end < start and new_start >= start - abs(shift)):
                tmp = h_new[start:end].copy()
                h_new[start:new_end] = 0.0
                h_new[new_start:end + shift] = tmp
                h_new = project_constraint(h_new, target_sum)

        c5_new = evaluate_c5_h(h_new)
        if c5_new < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            cur_c5 = evaluate_c5_h(h)
            delta = c5_new - cur_c5
            if delta < 0 or rng.random() < np.exp(-max(0.0, delta) / (T + 1e-10)):
                h = h_new
        T = max(1e-7, T * 0.9997)
        if patience > 8000:
            T = T_init * 0.5
            patience = 0
    return best_h, best_c5


def gradient_polish(h, target_sum, n_points, max_time):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5_h(best_h)
    lr = 0.02
    stuck = 0
    while time.time() - t0 < max_time and stuck < 500:
        grad, k_star = subgradient_c5(h, n_points)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-8:
            break
        for frac in [1.0, 0.5, 0.25, 0.1]:
            h_new = h - lr * frac * grad
            h_new = project_constraint(h_new, target_sum)
            c5_new = evaluate_c5_h(h_new)
            if c5_new < best_c5:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                stuck = 0
                break
            elif c5_new < evaluate_c5_h(h):
                h = h_new
                stuck = 0
                break
        else:
            lr *= 0.9
            stuck += 1
            if lr < 1e-6:
                break
    return best_h, best_c5

    # Get initial solution

    time_alloc3 = min(120, budget_s * 0.2)
    best_h, best_c5 = sa_optimize(best_h, target_sum, n_points, rng, time_alloc3, T_init=0.003)

        old_grid = np.arange(n_points) * (2.0 / n_points)
        new_grid = np.arange(n_new) * (2.0 / n_new)
        h_interp = np.interp(new_grid, old_grid, best_h)
        h_interp = project_constraint(h_interp, n_new / 2.0)
        cur_n = n_new
        cur_target = n_new / 2.0

        # Short SA at new resolution
        remaining = budget_s - 30 - elapsed()

        t_alloc = min(remaining * 0.5, 120)
        h_opt, c5_opt = sa_optimize(h_interp, cur_target, cur_n, rng, t_alloc, T_init=0.003)
