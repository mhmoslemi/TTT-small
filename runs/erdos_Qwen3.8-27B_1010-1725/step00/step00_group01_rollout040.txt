import time

def project_box_sum(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum} via bisection on shift."""
    lo, hi = -5.0, 5.0

        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

import numpy as np
import time

def project_box_sum(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kmax = np.max(corr)
    active = np.where(corr >= kmax - 1e-8)[0]
    grad = np.zeros(n)
    for ka in active:
        k = int(ka) - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += (1.0 - h[:m])
                grad[:m] -= h[k:]
        else:
            m = n + k
            if m > 0:
                grad[:m] += (1.0 - h[-k:])
                grad[-k:] -= h[:m]
    grad /= len(active)
    return grad * dx

def optimize_h(h0, target_sum, max_time, rng):
