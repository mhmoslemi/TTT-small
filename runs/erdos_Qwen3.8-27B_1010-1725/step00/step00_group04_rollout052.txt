        res = minimize(obj_hard, best_h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-14})
        if res.x is not None:
            res.x = np.clip(res.x, 0.0, 1.0)
            res.x = res.x * (target / max(np.sum(res.x), 1e-10))
            res.x = np.clip(res.x, 0.0, 1.0)
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x.copy()

    # Phase 4: Random restarts with binary search
    if time_left() > 120:
        n_cur = len(best_h)
        target_cur = n_cur // 2
        for attempt in range(30):
            if time_left() < 30:
                break
            h_pert = best_h + rng.normal(0, 0.3, n_cur)
            h_pert = (h_pert > 0.5).astype(float)
            while int(np.sum(h_pert)) > target_cur:
                idx = np.where(h_pert == 1)[0]
                h_pert[idx[rng.randint(len(idx))]] = 0.0
            while int(np.sum(h_pert)) < target_cur:
                idx = np.where(h_pert == 0)[0]
                h_pert[idx[rng.randint(len(idx))]] = 1.0
            h_pert, c5_pert = binary_local_search(h_pert, max_rounds=100,
                                                  time_limit=time_left() - 20, t0=t0, seed_rng=rng)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 5: Try higher resolution
    for n2 in [84, 168]:
        if time_left() < 120:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0, 1, n_cur)
        x_new = np.linspace(0, 1, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = (h2 > 0.5).astype(float)
        t2 = n2 // 2
        while int(np.sum(h2)) > t2:
            idx = np.where(h2 == 1)[0]
            h2[idx[rng.randint(len(idx))]] = 0.0
        while int(np.sum(h2)) < t2:
            idx = np.where(h2 == 0)[0]
            h2[idx[rng.randint(len(idx))]] = 1.0

        # Continuous opt at new resolution
        bounds2 = [(0.0, 1.0)] * n2
        for T in [100.0, 50.0]:
            if time_left() < 60:
                break
            def obj_soft2(z):
                corr = np.correlate(z, 1.0 - z, mode='full')
                sm = logsumexp(corr * T) / T
                pen = 1000.0 * (np.sum(z) - t2) ** 2
                return sm * (2.0 / n2) + pen
            res = minimize(obj_soft2, h2.copy(), method='L-BFGS-B',
                           bounds=bounds2, options={'maxiter': 1000, 'ftol': 1e-14})
            res.x = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
                h2 = best_h

        # Binary local search at new resolution
        h2_bin = (h2 > 0.5).astype(float)
        while int(np.sum(h2_bin)) > t2:
            idx = np.where(h2_bin == 1)[0]
            h2_bin[idx[rng.randint(len(idx))]] = 0.0
        while int(np.sum(h2_bin)) < t2:
            idx = np.where(h2_bin == 0)[0]
            h2_bin[idx[rng.randint(len(idx))]] = 1.0
        h2_bin, c5_b2 = binary_local_search(h2_bin, max_rounds=20,
                                            time_limit=time_left() - 40, t0=t0, seed_rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h2_bin.copy()

    # Final validation and repair
    n_final = len(best_h)
    if np.allclose(best_h, (best_h > 0.5).astype(float)):
        # Binary: ensure exact sum
        s = int(round(np.sum(best_h)))
    else:
        # Continuous: repair sum
        tf = n_final / 2.0
        best_h = best_h * (tf / max(np.sum(best_h), 1e-10))
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h[rng.randint(n_final)] += tf - np.sum(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
