    if use_constraint:
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    obj = lambda x: smooth_c5(x, temp)
    res = minimize(obj, h_init, method=method, bounds=bounds,
                   constraints=constraints if use_constraint else [],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    h_new = repair_sum(res.x, target_sum)
    c5_new = compute_c5(h_new)
    return h_new, c5_new


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(42, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = repair_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    
    # Phase 1: SLSQP with equality constraint
    if time.time() - start_time < budget_s * 0.15:
        try:
            h_new, c5_new = try_optimize(best_h, n, target_sum, bounds,
                                         'SLSQP', 800, 100.0, True)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 2: L-BFGS-B with penalty
    if time.time() - start_time < budget_s * 0.3:
        pen = lambda x: smooth_c5(x, 150.0) + 80.0 * (np.sum(x) - target_sum) ** 2
        try:
            res = minimize(pen, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = repair_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 3: Multi-temperature SLSQP
    for temp in [50.0, 250.0, 500.0]:
        if time.time() - start_time > budget_s * 0.5:
            break
        try:
            h_new, c5_new = try_optimize(best_h, n, target_sum, bounds,
                                         'SLSQP', 500, temp, True)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 4: Random restarts with local optimization
    attempts = 30
    for i in range(attempts):
        if time.time() - start_time > budget_s * 0.75:
            break
        scale = 0.03 + 0.005 * i
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = repair_sum(h_pert, target_sum)
        try:
            h_new, c5_new = try_optimize(h_pert, n, target_sum, bounds,
                                         'L-BFGS-B', 200, 200.0, False)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Phase 5: Try binary-like solutions
    if time.time() - start_time < budget_s * 0.6:
        h_bin = np.round(best_h)
        n_ones = int(np.sum(h_bin))
        n_target = int(target_sum)
        if n_ones > n_target:
            order = np.argsort(best_h)
            for idx in order:
                if n_ones <= n_target:
                    break
                if h_bin[idx] == 1:
                    h_bin[idx] = 0
                    n_ones -= 1
        elif n_ones < n_target:
            order = np.argsort(-best_h)
            for idx in order:
                if n_ones >= n_target:
                    break
                if h_bin[idx] == 0:
                    h_bin[idx] = 1
                    n_ones += 1
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            # Refine from binary
            try:
                h_new, c5_new = try_optimize(h_bin.astype(float), n, target_sum,
                                             bounds, 'SLSQP', 300, 150.0, True)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                elif c5_bin < best_c5:
                    best_h, best_c5 = h_bin.astype(float), c5_bin
            except Exception:
                best_h, best_c5 = h_bin.astype(float), c5_bin
    
    # Phase 6: Upscale and optimize
    if time.time() - start_time < budget_s * 0.45:
        n2 = n * 2
        target_sum2 = n2 / 2.0
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
        h_up = np.interp(x_new, x_old, best_h)
        h_up = repair_sum(h_up, target_sum2)
        bounds2 = [(0.0, 1.0)] * n2
        pen2 = lambda x: smooth_c5(x, 100.0) + 50.0 * (np.sum(x) - target_sum2) ** 2
        try:
            res = minimize(pen2, h_up, method='L-BFGS-B', bounds=bounds2,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = repair_sum(res.x, target_sum2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                n = n2
                target_sum = target_sum2
        except Exception:
            pass
    
    # Phase 7: Final polish
    if time.time() - start_time < budget_s * 0.9:
        try:
            h_new, c5_new = try_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                         [(0.0, 1.0)] * len(best_h),
                                         'SLSQP', 500, 300.0, True)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    
    # Final repair
    n_final = len(best_h)
    best_h = repair_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
