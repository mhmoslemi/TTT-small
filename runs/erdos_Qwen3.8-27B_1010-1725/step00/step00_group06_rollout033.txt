import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i + d < n:
            grad[i] += (1.0 - h[i + d]) * dx
        if 0 <= i - d < n:
            grad[i] -= h[i - d] * dx
    return c5, grad

def eq_constraint(h):
    return np.sum(h) - len(h) * 0.5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    k_star = int(np.argmax(corr))
    d = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i + d < n:
            grad[i] += (1.0 - h[i + d]) * dx
        if 0 <= i - d < n:
            grad[i] -= h[i - d] * dx
    return c5, grad

def eq_constraint(h):
    return np.sum(h) - len(h) * 0.5

    try:
        res = minimize(obj_func, h_init, jac=True, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h_init
