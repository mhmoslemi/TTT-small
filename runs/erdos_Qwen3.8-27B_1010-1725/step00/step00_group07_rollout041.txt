import numpy as np
import time

def repair(h, n):
    target = 0.5 * n
    out = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(out)
        d = target - s
        if abs(d) < 1e-10:
            break
        if d > 0:
            m = out < 1.0
            if not m.any(): return out
            out[m] += d / m.sum()
            np.minimum(out, 1.0, out)
        else:
            m = out > 0.0
            if not m.any(): return out
            out[m] += d / m.sum()
            np.maximum(out, 0.0, out)
    return out

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * 2.0/n

def project(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = 0.5 * n
    x = h.copy()
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo+hi)/2.0, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(y)
    if abs(diff) > 1e-8:
        m = (y > 0) & (y < 1)
        if m.any():
            y[m] += diff / m.sum()
            np.clip(y, 0, 1, y)
    return y

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new: return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h)

import numpy as np
import time

def repair(h, n):
    target = 0.5 * n
    out = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target - np.sum(out)
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = out < 1.0
            if not m.any():
                return out
            out[m] += d / m.sum()
            np.minimum(out, 1.0, out)
        else:
            m = out > 0.0
            if not m.any():
                return out
            out[m] += d / m.sum()
            np.maximum(out, 0.0, out)
    return out

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = (np.arange(n_old) + 0.5) / n_old
    xn = (np.arange(n_new) + 0.5) / n_new
    return np.interp(xn, xo, h)

def compute_grad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    maxv = c.max()
    active = np.where(c >= maxv - 0.005)[0]
    g = np.zeros(n)
    for idx in active:
        s = int(idx - (n - 1))
        if s >= 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
        else:
            ss = -s
            g[ss:] += (1.0 - h[:n - ss])
            g[:n - ss] -= h[ss:]
    g /= len(active)
    return g, maxv

def project_simplex(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target - np.sum(y)
    if abs(d) > 1e-10:
        m = (y > 0) & (y < 1)
        if m.any():
            y[m] += d / m.sum()
            np.clip(y, 0.0, 1.0, y)
    return y

def swap_optimize(h, n, rng, max_iter, c5_curr):
    for _ in range(max_iter):
        j1 = rng.integers(n)
        j2 = rng.integers(n)
        if j1 == j2:
            continue
        if h[j1] >= 1.0 or h[j2] <= 0.0:
            continue
        eps = min(0.05, (1.0 - h[j1]) * 0.5, h[j2] * 0.5)
        if eps < 1e-6:
            continue
        h[j1] += eps
        h[j2] -= eps
        c5_new = eval_c5(h)
        if c5_new < c5_curr:
            c5_curr = c5_new
        else:
            h[j1] -= eps
            h[j2] += eps
    return c5_curr

def binarize_pass(h, n):
    c5_curr = eval_c5(h)
    for _ in range(3):
        for j in range(n):
            saved = h[j]
            h[j] = 0.0
            h = repair(h, n)
            c5a = eval_c5(h)
            h[j] = 1.0
            h = repair(h, n)
            c5b = eval_c5(h)
            if c5a <= c5b:
                h[j] = 0.0
                h = repair(h, n)
            else:
                h = repair(h, n)
            c5_curr = min(c5a, c5b)
    return h, c5_curr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    hp = None
    try:
        hp = np.array(initial_h_values, dtype=np.float64)

    if hp is not None:

        n0 = 80
        h = np.zeros(n0)
        h[n0 // 4:3 * n0 // 4] = 1.0
        h = repair(h, n0)

    best_h = h.copy()
    best_c5 = eval_c5(best_h)

    levels = [n0]
    if n0 * 2 <= 400:
        levels.append(n0 * 2)

    for n in levels:
        if time.time() - t0 > budget_s - 20:
            break
        hc = repair(upsample(best_h, n), n)
        c5c = eval_c5(hc)

        remaining = max(10.0, budget_s - 20 - (time.time() - t0))
        t_sub = time.time() + remaining * 0.45
        t_swap = time.time() + remaining * 0.30
        t_bin_end = time.time() + remaining * 0.50

        # Subgradient phase
        lr = 0.02
        while time.time() < t_sub:
            g, _ = compute_grad(hc, n)
            ht = project_simplex(hc - lr * g, n)
            c5t = eval_c5(ht)
            if c5t < c5c:
                hc = ht
                c5c = c5t
                if c5c < best_c5:
                    best_h = hc.copy()
                    best_c5 = c5c
                lr = min(lr * 1.03, 0.15)
            else:
                lr = max(lr * 0.85, 0.002)
                if rng.random() < 0.25:
                    hc = repair(hc + lr * 3 * rng.standard_normal(n), n)
                    c5c = eval_c5(hc)

        # Swap-based local search
        if time.time() < t_swap:
            n_iter = int(max(1000, (t_swap - time.time()) / 0.0003))
            c5c = swap_optimize(hc, n, rng, n_iter, c5c)
            if c5c < best_c5:
                best_h = hc.copy()
                best_c5 = c5c

        # Binarization
        if time.time() < t_bin_end:
            hc, c5c = binarize_pass(hc, n)
            if c5c < best_c5:
                best_h = hc.copy()
                best_c5 = c5c

    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
