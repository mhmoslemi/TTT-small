    d = target_sum - s

    else:
        m = h > 0.0
    if np.sum(m) == 0:
        return h
    h[m] += d / np.sum(m)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    d = target_sum - s
    if abs(d) < 1e-14:
        return h.copy()
    mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
    if np.sum(mask) == 0:
        return h.copy()
    h = h.copy()
    h[mask] += d / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, dx, tol=1e-8):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol * max_val)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        if L == 0:
            grad -= h * dx
            grad += (1 - h) * dx
        elif L > 0:
            grad[L:] += (1 - h[:n-L]) * dx
            grad[:n-L] += -h[L:] * dx
        else:
            M = -L
            grad[:n-M] += (1 - h[M:]) * dx
            grad[M:] += -h[:n-M] * dx
    return grad, max_val * dx

def subgradient_optimize(h0, n_iter=5000, lr0=0.05, decay=0.999):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    
    for it in range(n_iter):
        grad, c5 = compute_grad(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = project_feasible(h - lr * grad, target)
        lr *= decay
        if lr < 1e-8:
            break
    return best_h, best_c5

def coordinate_descent(h, max_iters=200, n_trials=7):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    vals = np.linspace(0, 1, n_trials)
    
    for it in range(max_iters):
        improved = False
        for j in range(n):
            old = h[j]
            best_local_val = old
            best_local_c5 = best_c5
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                # Adjust to maintain sum: change h[j] by (v - old), 
                # distribute among other coordinates
                h[j] = v
                diff = (old - v)
                # Add diff to a random other coordinate (or spread evenly)
                h_other = h.copy()
                h_other[j] = old  # temporarily revert
                mask = h_other > 0.0 if diff > 0 else h_other < 1.0
                mask[j] = False
                if np.sum(mask) == 0:
                    continue
                h_other[mask] += diff / np.sum(mask)
                h_other = np.clip(h_other, 0, 1)
                h_other[j] = v
                # Check sum
                # Actually this is getting complicated. Let me just project.
                h[j] = v
                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < best_local_c5 - 1e-12:
                    best_local_c5 = c
                    best_local_val = v
                    improved = True
                    # keep this h
                    # Actually h was already modified by project_feasible
            if not improved:
                h[j] = old
        # After full pass, ensure sum constraint
        h = project_feasible(h, target)
        best_c5 = compute_c5(h)
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    s = np.sum(h)
    d = target_sum - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += d / cnt
    return np.clip(h, 0.0, 1.0)

def compute_subgrad(h, dx, tol_frac=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol_frac * max_val)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        if L == 0:
            grad += (1.0 - 2.0 * h) * dx
        elif L > 0:
            grad[L:] += (1.0 - h[:n - L]) * dx
            grad[:n - L] -= h[L:] * dx
        else:
            M = -L
            grad[:n - M] += (1.0 - h[M:]) * dx
            grad[M:] -= h[:n - M] * dx
    return grad, max_val * dx

def subgradient_optimize(h0, n_iter, lr0, decay, deadline):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline - 10:
            break
        grad, c5 = compute_subgrad(h, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        h = project_feasible(h - lr * grad, target)
        lr *= decay
        if lr < 1e-10:
            break
        if it % 2000 == 0:
            h = project_feasible(h, target)
    return best_h, best_c5

def coordinate_descent(h, max_iters=300, n_trials=5, deadline=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    vals = np.linspace(0.0, 1.0, n_trials)
    for it in range(max_iters):
        if deadline and time.time() > deadline - 5:
            break
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = old
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                h[j] = v
                h = project_feasible(h, target)
                c = compute_c5(h)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
            h[j] = old
        if improved:
            h = project_feasible(h, target)
        else:
            break
    return h, compute_c5(h)

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n + 1)[:-1]
    new_x = np.linspace(0, 2, new_n + 1)[:-1]
    h_new = np.interp(new_x, old_x, h, left=0, right=0)
    return project_feasible(h_new, new_n / 2.0)
