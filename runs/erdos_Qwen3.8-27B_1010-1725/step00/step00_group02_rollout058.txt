import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    """Adjust h to satisfy sum constraint while keeping [0,1] bounds."""
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute correction to elements with room
    for _ in range(100):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        n_mask = mask.sum()
        adj = diff / n_mask
        # Limit adjustment to not cross bounds
        if diff > 0:
            max_adj = (1.0 - h[mask]).min()
            adj = min(adj, max_adj)
        else:
            max_adj = h[mask].min()
            adj = max(adj, -max_adj)
        h[mask] += adj
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def local_search(h, n, time_limit, rng):
    """Local search to minimize C5."""
    h = h.astype(float).copy()
    h = np.clip(h, 0, 1)
    h = fix_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    
    t_start = time.time()
    step = 0.1
    no_improve = 0
    
    while time.time() - t_start < time_limit:
        if step < 0.0005:
            break
        
        improved = False
        n_tries = 100
        
        for _ in range(n_tries):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            # Determine valid range for delta
            lo = max(-best_h[i], best_h[j] - 1.0)
            hi = min(1.0 - best_h[i], best_h[j])
            if hi <= lo:
                continue
            
            delta = rng.uniform(lo, hi)
            
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta
            
            c5 = compute_c5(cand, n)
            if c5 < best_c5 - 1e-14:
                best_h = cand
                best_c5 = c5
                improved = True
