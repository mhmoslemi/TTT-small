def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Iterative adjustment
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Add to elements that are < 1
            mask = h < 1.0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

    target = n / 2.0
