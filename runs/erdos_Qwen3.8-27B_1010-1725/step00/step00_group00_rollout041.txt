import numpy as np

import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    # Gradient of corr[k_star] with respect to h
    # corr[m] = sum_i h[i] * (1-h[i+m-(n-1)])  (valid indices)
    # d corr[m] / d h[j] = (1-h[j+m-n+1])_valid - h[j-m+n-1]_valid
    g = np.zeros(n)
    m = k_star
    # j appears as first factor: i=j, need j+m-n+1 in [0,n-1]
    j2 = np.arange(n) + m - (n - 1)
    valid1 = (j2 >= 0) & (j2 < n)
    g[valid1] += (1.0 - h[j2[valid1]])
    # j appears as second factor: i+m-n+1=j → i=j-m+n-1, need i in [0,n-1]
    i = np.arange(n) - m + (n - 1)
    valid2 = (i >= 0) & (i < n)
    g[valid2] -= h[i[valid2]]
    return g * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break

        # Adjust uniformly

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = int(np.argmax(corr))
    g = np.ones(n)
    idx = np.arange(n) + m - (n - 1)
    valid = (idx >= 0) & (idx < n)
    g[valid] -= h[idx[valid]]
    idx2 = np.arange(n) - m + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    g[valid2] -= h[idx2[valid2]]
    return g * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(300):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        diff = (target - s) / float(n)
        h = np.clip(h + diff, 0.0, 1.0)
    return h

def upscale(h, n):
    old_n = len(h)
    if old_n == n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x_new, x_old, h), n)

def opt_lbgfb(h_start, n, max_iter, penalty_mult):
    target = n * 0.5
    def obj(x):
        c5v = np.max(np.correlate(x, 1.0 - x, mode="full")) * (2.0 / n)
        pen = penalty_mult * (np.sum(x) - target) ** 2
        return c5v + pen
    def obj_grad(x):
        g = c5_subgrad(x)
        g += 2.0 * penalty_mult * (np.sum(x) - target)
        return g
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def opt_slqp(h_start, n, max_iter):

    def obj_grad(x):
        return c5_subgrad(x)
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)
