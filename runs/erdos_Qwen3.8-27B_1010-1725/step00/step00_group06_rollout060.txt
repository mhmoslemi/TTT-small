from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def soft_max_c5(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c
    return max_c + np.log(np.sum(np.exp(shifted * beta)) / len(corr)) / beta

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix small sum violation
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Add/subtract uniformly where possible
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) > 0:
                add = min(diff, np.sum(1.0 - h[mask]) * 0.5) / np.sum(mask)
                h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 0.0
            if np.sum(mask) > 0:
                sub = min(-diff, np.sum(h[mask]) * 0.5) / np.sum(mask)
                h[mask] = np.maximum(h[mask] - sub, 0.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def soft_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    shifted = corr - mx
    return mx + np.log(np.sum(np.exp(shifted * beta)) / len(corr)) / beta

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, beta):
    n = len(x)
    h = project_feasible(x, n)
    return soft_max_c5(h, beta)

def obj_direct(x, _):
    n = len(x)
    h = project_feasible(x, n)
    return compute_c5(h)

def nm_refine(h, n, maxiter):
    target = n / 2.0
    def _nm_obj(x):
        return obj_direct(x, None)
    try:
        res = minimize(_nm_obj, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11,
                                'adaptive': True})
        return project_feasible(res.x, n)
    except Exception:
        return h

def upscale(h, n_from, n_to):
    x_old = np.arange(n_from) * (2.0 / n_from)
    x_new = np.arange(n_to) * (2.0 / n_to)
    h_up = np.interp(x_new, x_old, h)
    return project_feasible(h_up, n_to)

def optimize_at_resolution(h0, n, rng, time_fn, max_trials, beta):
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

            break

            x0 = best_h.copy()
        elif trial < 8:
            sigma = 0.005 + 0.005 * (trial % 4)
            x0 = best_h + rng.normal(0, sigma, n)
        elif trial < 16:
            sigma = 0.02 + 0.02 * (trial % 3)
            x0 = best_h + rng.normal(0, sigma, n)
        else:

            idx = rng.choice(n, n_pert, replace=False)
            x0 = best_h.copy()
            x0[idx] = rng.uniform(0, 1, n_pert)
        
        x0 = project_feasible(x0, n)
        
        # SLSQP with smooth objective

            res = minimize(obj_smooth, x0, args=(beta,), method='SLSQP',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-12})
            h_trial = project_feasible(res.x, n)
        except Exception:
            h_trial = project_feasible(x0, n)
        
        # Nelder-Mead polish with direct max objective
        h_nm = nm_refine(h_trial, n, 2000)
        c5_nm = compute_c5(h_nm)
        c5_sq = compute_c5(h_trial)
        if c5_nm < c5_sq:
            h_trial = h_nm
            c5_trial = c5_nm
        else:
            c5_trial = c5_sq
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    return best_h, best_c5

    best_n = n

    h_opt, c5_opt = optimize_at_resolution(best_h, n, rng, time_left, 25, 100.0)
    if c5_opt < best_c5:

        best_c5 = c5_opt

            best_h = h_opt2

    c5_final = compute_c5(h_final)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final
    
    return (best_h, best_c5, best_n)
