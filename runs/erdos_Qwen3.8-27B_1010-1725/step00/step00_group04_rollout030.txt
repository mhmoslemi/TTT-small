import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, beta):
    m = np.max(vals)
    shifted = (vals - m) * beta
    shifted = np.clip(shifted, -700, 0)
    return m + np.log(np.sum(np.exp(shifted)) / len(vals)) / beta

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_obj(n, beta, pen_w):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm = smooth_max(corr * dx, beta)
        pen = pen_w * (np.sum(h) - n * 0.5) ** 2
        return sm + pen
    return obj

def optimize_from(h0, n, budget_end, rng):
    h = project_feasible(np.asarray(h0, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    betas = [5, 15, 50, 150, 400, 1000]
    for beta in betas:
        if time.time() > budget_end - 15:
            break
        pen_w = 50.0 if beta < 100 else 200.0
        obj = make_obj(n, beta, pen_w)
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def upsample_periodic(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    h_ext = np.concatenate([h, h, h])
    x_new = np.linspace(0, old_n, new_n, endpoint=False)
    x_old = np.arange(3 * old_n, dtype=float)
    return np.interp(x_new + old_n, x_old, h_ext)

def random_start(n, rng):
    h = np.full(n, 0.5)
    n_bumps = rng.randint(2, 6)
    for _ in range(n_bumps):
        center = rng.randint(n)
        width = rng.randint(3, max(4, n // 4))
        amp = rng.uniform(-0.4, 0.4)
        for i in range(n):
            d = min(abs(i - center), n - abs(i - center))
            if d <= width:
                h[i] += amp * 0.5 * (1.0 + np.cos(np.pi * d / max(width, 1)))
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def coordinate_polish(h, n, budget_end, rng):
    h = project_feasible(h.astype(float).copy(), n)
    best_c5 = compute_c5(h)
    epsilons = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for _ in range(30):
        if time.time() > budget_end - 3:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > budget_end - 3:
                break
            old = h[i]
            for eps in epsilons:
                for sign in (1, -1):
                    trial = old + sign * eps
                    if trial < 0 or trial > 1:
                        continue
                    h[i] = trial
                    h = project_feasible(h, n)
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-9:
                        best_c5 = c5
                        improved = True
                        break
                    h[i] = old
                    h = project_feasible(h, n)
                else:
                    continue
                break
        if not improved:
            break
    return h, best_c5
