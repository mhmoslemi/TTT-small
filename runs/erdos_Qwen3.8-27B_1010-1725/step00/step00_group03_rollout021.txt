import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, N):
    """Project h onto {0<=h<=1, sum(h)=N/2} via bisection on shift."""
    target = N / 2.0
    lo, hi = -1.0 - np.max(h), 1.0 - np.min(h) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_obj(h, tau=0.005, penalty=1e6):
    """Smooth max of corr*dx + penalty for sum constraint."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    max_g = np.max(g)
    exp_g = np.exp((g - max_g) / tau)
    p = exp_g / np.sum(exp_g)
    f = max_g + tau * np.log(np.sum(exp_g))
    grad = np.zeros(N)
    for k in range(len(g)):
        if p[k] < 1e-12:
            continue
        d = k - (N - 1)
        pk = p[k] * dx
        if d >= 0:
            grad[d:] += pk * (1.0 - h[:N - d])
            grad[:N - d] -= pk * h[d:]
        else:
            dp = -d
            grad[:N - dp] += pk * (1.0 - h[dp:])
            grad[dp:] -= pk * h[:N - dp]
    s = np.sum(h) - N / 2.0
    f += penalty * s * s
    grad += 2.0 * penalty * s
    return f, grad

def generate_structured_init(N, p, rng):
    """Create a modular/periodic pattern with N/2 ones."""
    h = np.zeros(N)
    for i in range(N):
        h[i] = 1.0 if (i * p) % (2 * p) < p else 0.0
    h = project_to_constraint(h, N)
    h += rng.uniform(-0.05, 0.05, N)
    return project_to_constraint(h, N)

def local_search(h, N, rng, iters=80):
    """Gaussian perturbation local search."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(iters):
        step = rng.uniform(0, 0.15)
        h_try = best + rng.normal(0, step, N)
        h_try = project_to_constraint(h_try, N)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best = h_try.copy()
            best_c5 = c5_try
    return best, best_c5

    if init is None or len(init) == 0:

        init = np.ones(N_fallback) * 0.5

    init_N = len(init)
    best_h = init.copy()
    best_c5 = compute_c5(best_h)

    N_list = sorted(set([init_N, 84, 100, 140, 200]))

    for N in N_list:
        if time.time() - start_time > budget_s * 0.85:
            break

        # Build initializations for this N
        inits = []
        if N == init_N:
            inits.append(init.copy())
        else:
            x_old = np.linspace(0, 1, init_N)
            x_new = np.linspace(0, 1, N)
            inits.append(np.interp(x_new, x_old, init))

        # Random inits

            inits.append(project_to_constraint(h_r, N))

            try:

                    smooth_obj, h0,
                    jac=True,
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * N,
                    options={'maxiter': 600, 'ftol': 1e-12,

    best_h = project_to_constraint(best_h, N_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, N_final)
