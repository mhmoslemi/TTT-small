import numpy as np
import time

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + d)
    return h

def make_binary(n, rng):
    h = np.zeros(n)
    k = int(n / 2)
    frac = n / 2.0 - k
    p = rng.permutation(n)
    h[p[:k]] = 1.0
    if frac > 1e-10:
        h[p[k]] = frac
    return h

def sa_step(h, n, rng):
    h_new = h.copy()
    m = rng.randint(6)
    if m == 0:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.2, 0.2)
        h_new[i] += d
        h_new[j] -= d
    elif m == 1:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.normal(0, 0.1)
        h_new[i] += d
        h_new[j] -= d
    elif m == 2:
        i = rng.randint(0, max(1, n - 3))
        L = min(rng.randint(3, min(10, n - i)), n - i)
        if L >= 2:
            h_new[i:i + L] = np.roll(h_new[i:i + L].copy(), rng.choice([-1, 1]))
    elif m == 3:
        i, j = rng.choice(n, 2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
    elif m == 4:
        corr = np.correlate(h, 1.0 - h, mode="full")
        mi = int(np.argmax(corr))
        doff = mi - (n - 1)
        if doff != 0:
            a = abs(doff)
            if a < n:
                s = rng.randint(n - a)
                if doff > 0:
                    idx1, idx2 = s + a, s
                else:
                    idx1, idx2 = s, s + a
                step = 0.1 if rng.rand() < 0.5 else -0.1
                h_new[idx1] += step
                h_new[idx2] -= step
        else:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.1, 0.1)
            h_new[i] += d
            h_new[j] -= d

            if ssum > 1e-10:

            h_new[i:i + L] = np.clip(sub, 0.0, 1.0)
