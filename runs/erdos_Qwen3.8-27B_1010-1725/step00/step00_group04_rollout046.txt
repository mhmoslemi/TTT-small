import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max(x, t):
    m = np.max(x)
    return float(m + (1.0 / t) * np.log(np.sum(np.exp(t * (x - m)))))


def objective_fn(h, t, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sm = smooth_max(corr, t)
    s = np.sum(h)
    penalty = 1e5 * (s - target_sum) ** 2
    return sm + penalty


def objective_fn_slsqp(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(corr, t)


def repair_h(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    target = n * 0.5
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return repair_h(h_new)


def make_jac_ones(n):
    return np.ones(n)


def _slsqp_obj(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return m + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - m))))


def optimize_lbfgsb(h, n, t_start, t_end, n_outer, time_limit):
    target_sum = n * 0.5
    bounds = [(0.0, 1.0)] * n
    c5_cur = compute_c5(h)
    for i in range(n_outer):
        if time.time() > time_limit:
            break
        t = t_start - (t_start - t_end) * (i / max(1, n_outer - 1))
        t = max(t, 1.0)

        def obj(x):
            return objective_fn(x, t, target_sum)

        try:
            res = minimize(obj, h.copy(), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9})
            h_new = repair_h(res.x)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h = h_new
                c5_cur = c5_new
        except Exception:
            break
    return h, c5_cur


def optimize_slsqp_fixed(h, n, t, max_iter, time_limit):
    target_sum = n * 0.5
    bounds = [(0.0, 1.0)] * n

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]

    def obj(x):
        return _slsqp_obj(x, t)

    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-9})
        h_new = repair_h(res.x)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 100
        h_best = np.zeros(n_init)
        h_best[:n_init // 2] = 1.0
        h_best = repair_h(h_best)

    c5_best = compute_c5(h_best)
    h_best = repair_h(h_best)

    # Phase 1: SLSQP at initial resolution with decreasing temperature
    n = len(h_best)
    time_limit = t0 + budget_s * 0.25

        if time.time() > time_limit:
            break
        h_opt, c5_opt = optimize_slsqp_fixed(h_best, n, t_val, 300, time_limit)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    # Phase 2: Multi-resolution L-BFGS-B refinement
    resolutions = [84, 168, 336, 672, 1000]
    time_limit = t0 + budget_s * 0.70

    for target_n in resolutions:
        if time.time() > time_limit:
            break
        if target_n == n:
            continue
        h_cur = upsample(h_best, target_n)
        n = target_n
        c5_cur = compute_c5(h_cur)

        h_opt, c5_opt = optimize_lbfgsb(h_cur, n, 30.0, 2.0, 8, time_limit)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt
        else:
            h_cur = upsample(h_best, target_n)

        # SLSQP polish
        if n <= 200 and time.time() < time_limit:
            h_opt, c5_opt = optimize_slsqp_fixed(h_cur, n, 3.0, 200, time_limit)
            if c5_opt < c5_best:
                h_best = h_opt
                c5_best = c5_opt

    # Phase 3: Random restarts at best resolution
    n_best_res = len(h_best)
    time_limit = t0 + budget_s * 0.90
    n_trials = 6 if n_best_res <= 100 else 3

    for trial in range(n_trials):

        h_pert = h_best + rng.randn(n_best_res) * scale
        h_pert = repair_h(h_pert)

            c5_best = c5_opt

    c5_best = compute_c5(h_best)
