def smooth_logsumexp(vals, beta):
    m = np.max(vals)
    if beta < 1:
        beta = 1
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        smax = smooth_logsumexp(corr * dx, beta)
        s = np.sum(h)
        penalty = lambda_sum * (s - target) ** 2

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def Vec_smooth_lse(vals, beta):
    m = np.max(vals)
    e = np.exp(beta * (vals - m))
    return m + np.log(np.sum(e)) / beta

def make_obj(n, dx, beta, lam):
    target = n / 2.0
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = Vec_smooth_lse(corr * dx, beta)
        s = np.sum(h)
        return val + lam * (s - target) ** 2
    return obj

            for i in range(i_min, i_max):

                    pass
            # Simpler: for each valid pair (i, i+shift):
            # contributes (1-h[i+shift]) to d/dh[i] and -h[i] to d/dh[i+shift]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_lse_guided(h, beta, dx, lam, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    se = np.sum(e)
    w = e / se
    val = m + np.log(se) / beta
    s = np.sum(h)
    val += lam * (s - target) ** 2
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        shift = k_idx - (n - 1)
        if shift == 0:
            g += wk * (1.0 - 2.0 * h)
        elif shift > 0:
            g[:n - shift] += wk * (1.0 - h[shift:])
            g[shift:] -= wk * h[:n - shift]
        else:
            st = -shift
            g[st:] += wk * (1.0 - h[:n - st])
            g[:n - st] -= wk * h[st:]
    g += 2.0 * lam * (s - target)
    return val, g

def repair_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

        hp = np.array(initial_h_values, dtype=float)

        x = np.linspace(0, 1, n)

    except (NameError, UnboundLocalError):

def make_init_uniform(n):
    return np.full(n, 0.5)

def make_init_binary(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        n1 = n // 2
    else:
        n1 = n // 2
    ones = rng.choice(n, n1, replace=False)
    h[ones] = 1.0
    if n % 2 != 0:
        h[rng.randint(n)] = 0.5
    return repair_sum(h, n)

def make_init_structured(n, rng):
    h = np.zeros(n)
    n1 = n // 2
    if n1 == 0:
        return repair_sum(h, n)
    pos = np.round(np.linspace(0, n - 1, n1)).astype(int)
    seen = set()
    newpos = []
    for p in pos:
        if p not in seen:
            seen.add(p)
            newpos.append(p)
    while len(newpos) < n1:
        p = rng.randint(n)
        if p not in seen:
            seen.add(p)
            newpos.append(p)
    h[newpos[:n1]] = 1.0
    if n % 2 != 0:
        h[0] = 0.5
    return repair_sum(h, n)

def make_init_cosine(n, freq):
    x = np.linspace(0, 2.0, n)
    h = 0.5 + 0.45 * np.cos(np.pi * freq * x)
    h = np.clip(h, 0.0, 1.0)
    return repair_sum(h, n)

def optimize_from(h0, n, dx, target, max_iter_beta):
    h = h0.copy()
    for beta, miter in max_iter_beta:
        h = repair_sum(h, n)
        def obj(x):
            v, _ = smooth_lse_guided(x, beta, dx, 50.0, target)
            return v
        for _ in range(2):
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': miter, 'ftol': 1e-12,
                                    'gtol': 1e-8, 'maxls': 50})
            h = res.x
            h = repair_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t0

    def try_update(h):
        nonlocal best_h, best_c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Use parent if available

    if hp is not None:

    stages = [(5, 150), (20, 150), (80, 120), (300, 80), (1000, 50)]

    resolutions = [47, 48, 50, 60, 64, 80, 100, 128, 160, 200, 320]
    for n in resolutions:
        if elapsed() > budget_s * 0.82:
            break
        if n < 4:

        dx = 2.0 / n
        target = n / 2.0

        inits = []
        hp_n = make_init_parent(n)

            inits.append(hp_n)
        inits.append(make_init_uniform(n))
        inits.append(make_init_binary(n, rng))
        inits.append(make_init_structured(n, rng))
        inits.append(make_init_cosine(n, 2))
        inits.append(make_init_cosine(n, 3))
        if n >= 80:

            inits.append(make_init_structured(n, rng))

        for h0 in inits:
            if elapsed() > budget_s * 0.9:

            try_update(h_opt)

            # Try rounding to binary
            h_bin = (h_opt >= 0.5).astype(float)
            h_bin = repair_sum(h_bin, n)
            try_update(h_bin)

            # Try soft rounding (0.2/0.8)

            h_soft = repair_sum(h_soft, n)
            try_update(h_soft)

    # Final local polish on best
    if best_h is not None and elapsed() < budget_s * 0.92:
        n = len(best_h)
        dx = 2.0 / n
        target = n / 2.0
        h_polish = optimize_from(best_h, n, dx, target, [(500, 200), (2000, 100), (10000, 50)])
