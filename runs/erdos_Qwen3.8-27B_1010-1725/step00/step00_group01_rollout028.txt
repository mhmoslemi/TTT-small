            h_cs, c5_cs = coord_search(h_ls, 3, seed + 300 * ni)
            if c5_cs < best_c5:
                best_h, best_c5, best_n = h_cs.copy(), c5_cs, n

    if time.time() - t0 < budget_s - 10:
        h_f, c5_f = local_search(best_h, 20000, 0.008, seed + 9999)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    if time.time() - t0 < budget_s - 3:
        h_f2, c5_f2 = coord_search(best_h, 5, seed + 8888)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2

    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
