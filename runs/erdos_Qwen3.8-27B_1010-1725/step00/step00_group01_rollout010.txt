import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_bounds(n):
    return [(0.0, 1.0)] * n


def make_sum_lc(n):
    A = np.ones((1, n))
    v = n / 2.0
    return LinearConstraint(A, [v], [v])


def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    return h


def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new)


def random_binary(n, rng):
    if n % 2 == 0:
        h = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    else:
        h = np.zeros(n)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
        remaining = np.setdiff1d(np.arange(n), idx)
        h[remaining[rng.integers(len(remaining))]] = 0.5
    return h


def try_slsqp(h, n, maxiter, ftol):
    try:
        res = minimize(compute_c5, h, args=(n,), method='SLSQP',
                       bounds=make_bounds(n),
                       constraints=[make_sum_lc(n)],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x, float(res.fun)
    except Exception:
        return h, compute_c5(h, n)


def try_powell(h, n, maxiter, ftol):
    try:
        lo = np.zeros(n)
        hi = np.ones(n)
        res = minimize(compute_c5, h, args=(n,), method='Powell',
                       bounds=list(zip(lo, hi)),
                       constraints=[make_sum_lc(n)],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return fix_sum(res.x, n), compute_c5(fix_sum(res.x, n), n)
    except Exception:
        return h, compute_c5(h, n)


def binary_local_search(h, n, max_iter, rng):
    h = h.copy()
    c5 = compute_c5(h, n)
    for it in range(max_iter):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(15):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            new_c5 = compute_c5(h, n)
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                improved = True
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        if not improved:
            break
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial solution
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n_fallback = 47
        h_init = random_binary(n_fallback, rng)
    n_init = len(h_init)

    best_h = fix_sum(h_init, n_init)
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: SLSQP at initial resolution
    if remaining() > 30:
        h_r, c5_r = try_slsqp(best_h, best_n, 8000, 1e-15)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 2: Powell refinement
    if remaining() > 60:
        h_r, c5_r = try_powell(best_h, best_n, 30000, 1e-15)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 3: SLSQP again from Powell result
    if remaining() > 30:
        h_r, c5_r = try_slsqp(best_h, best_n, 5000, 1e-16)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 4: Random perturbations + SLSQP at same n
    for trial in range(15):
        if remaining() < 30:
            break
        sigma = 0.02 + 0.01 * trial
        pert = rng.normal(0, sigma, best_n)
        h_p = np.clip(best_h + pert, 0, 1)
        h_p = fix_sum(h_p, best_n)
        h_r, c5_r = try_slsqp(h_p, best_n, 3000, 1e-14)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 5: Binary local search at same n
    if remaining() > 20:
        h_b = np.where(np.clip(best_h, 0, 1) > 0.5, 1.0, 0.0)
        h_b = fix_sum(h_b, best_n)
        h_r, c5_r = binary_local_search(h_b, best_n, 3000, rng)
        if c5_r < best_c5:
            h_r, c5_r = try_slsqp(h_r, best_n, 3000, 1e-14)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 6: Higher resolutions
    for n_try in [54, 60, 72, 80, 100, 120, 150, 200]:
        if remaining() < 90:
            break
        if n_try == best_n:
            continue
        # Upsample from best
        h_up = upsample_h(best_h, best_n, n_try)
        h_r, c5_r = try_slsqp(h_up, n_try, 4000, 1e-14)
        if c5_r < best_c5:
            best_h, best_c5, best_n = h_r, c5_r, n_try
        # Also try random binary start
        if remaining() > 60:
            h_rb = random_binary(n_try, rng)
            h_rb = fix_sum(h_rb, n_try)
            h_r2, c5_r2 = try_slsqp(h_rb, n_try, 2000, 1e-14)
            if c5_r2 < best_c5:
                best_h, best_c5, best_n = h_r2, c5_r2, n_try
        if remaining() > 45:
            h_r3, c5_r3 = try_powell(h_up, n_try, 10000, 1e-14)
            if c5_r3 < best_c5:
                best_h, best_c5, best_n = h_r3, c5_r3, n_try

    # Phase 7: Final polish at best resolution
    if remaining() > 20:
        h_r, c5_r = try_slsqp(best_h, best_n, 15000, 1e-16)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
    if remaining() > 15:
        h_r, c5_r = try_powell(best_h, best_n, 20000, 1e-16)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
    if remaining() > 10:
        h_r, c5_r = try_slsqp(best_h, best_n, 10000, 1e-16)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
