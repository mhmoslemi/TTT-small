        elif r < 0.65:
            j, k = rng.integers(0, n, 2)
            if j == k:
                continue
            d = rng.uniform(-0.5, 0.5)
            new_j = np.clip(cur[j] + d, 0.0, 1.0)
            actual_d = new_j - cur[j]
            new_k = np.clip(cur[k] - actual_d, 0.0, 1.0)
            sum_err = (new_j + new_k) - (cur[j] + cur[k])
            cur[j], cur[k] = new_j, new_k
            if abs(sum_err) > 1e-12:
                mask = (cur > 1e-7) & (cur < 1.0 - 1e-7)
                mask[j] = mask[k] = False
                if mask.any():
                    cur[mask] -= sum_err / mask.sum()
                    cur = np.clip(cur, 0.0, 1.0)
                else:
                    cur[j], cur[k] = cur[k], new_j  # revert partially
        elif r < 0.80:
            j, k, m = rng.integers(0, n, 3)
            if len({j, k, m}) < 3:
                continue
            cur[j], cur[k], cur[m] = cur[m], cur[j], cur[k]
        elif r < 0.90:
            j, k, m, p = rng.integers(0, n, 4)
            if len({j, k, m, p}) < 4:
                continue
            cur[j], cur[k], cur[m], cur[p] = cur[p], cur[m], cur[k], cur[j]
        else:
            cnt = rng.integers(2, 6)
            idx = rng.choice(n, cnt, replace=False)
            old_vals = cur[idx].copy()
            cur[idx] = np.clip(cur[idx] + rng.uniform(-0.35, 0.35, cnt), 0.0, 1.0)
            diff = cur[idx].sum() - old_vals.sum()
            if abs(diff) > 1e-9:
                other = np.setdiff1d(np.arange(n), idx)
                mask = (cur[other] > 1e-6) & (cur[other] < 1.0 - 1e-6)
                if mask.any():
                    cur[other[mask]] -= diff / mask.sum()
                    cur = np.clip(cur, 0.0, 1.0)

        new_c5 = eval_c5(cur)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        else:
            if rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
                cur_c5 = new_c5

        T *= 0.99997
        if T < 0.0004:
            T = 0.012
            cur = best.copy()
            cur_c5 = best_c5

    return best, best_c5

def try_pattern(pattern_func, n, target, deadline, rng, seed_offset):
    h = pattern_func(n)
    h = fix_sum(h, target)
    return optimize(h, target, deadline, np.random.RandomState(42 + seed_offset))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.92
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(80)
        h0[:40] = 1.0

    n = len(h0)
    target = n / 2.0
    h0 = fix_sum(h0, target)

    best_h = h0.copy()
    best_c5 = eval_c5(best_h)

    # Phase 1: Main SA from initial (50% budget)
    dl1 = t0 + budget_s * 0.50
    h1, c51 = optimize(h0, target, dl1, rng)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    # Phase 2: Random binary restarts (20% budget)
    if time.time() < t0 + budget_s * 0.72:
        sub_budget = (t0 + budget_s * 0.72) - time.time()
        n_restarts = max(1, int(sub_budget / (budget_s * 0.04)))
        for s in range(min(n_restarts, 8)):
            if time.time() >= t0 + budget_s * 0.72:
                break
            h_rand = np.zeros(n)
            ones = rng.choice(n, n // 2, replace=False)
            h_rand[ones] = 1.0
            h_rand = fix_sum(h_rand, target)
            sub_dl = min(t0 + budget_s * 0.72, time.time() + budget_s * 0.035)
            ht, ct = optimize(h_rand, target, sub_dl, np.random.RandomState(seed + 1000 + s * 13))
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 3: Structured patterns (12% budget)
    if time.time() < t0 + budget_s * 0.84:
        for period in [4, 5, 6, 7, 8, 10, 12, 16, 20, 40]:
            if time.time() >= t0 + budget_s * 0.84:
                break
            if n % period != 0:
                continue
            base = np.zeros(period)
            n1_in_period = (n // 2) // (n // period)
            if n1_in_period <= 0:
                continue
            base[:n1_in_period] = 1.0
            h_pat = np.tile(base, n // period)
            h_pat = fix_sum(h_pat, target)
            sub_dl = min(t0 + budget_s * 0.84, time.time() + budget_s * 0.012)
            ht, ct = optimize(h_pat, target, sub_dl, np.random.RandomState(seed + 2000 + period))
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 4: Final polish from best found (8% budget)
    if time.time() < deadline:
        ht, ct = optimize(best_h, target, deadline, np.random.RandomState(seed + 5000))
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Final constraint enforcement
    best_h = fix_sum(best_h, target)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, n
