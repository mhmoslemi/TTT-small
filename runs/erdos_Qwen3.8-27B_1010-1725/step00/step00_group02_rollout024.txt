def project_onto_constraints(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) < 1e-10:
        return h
    # Need to adjust: add delta to each element
    # But we need to respect [0,1] bounds
    # Simple approach: if sum > target, reduce elements that are > 0
    # If sum < target, increase elements that are < 1
    delta = (target - s) / n
    h = h + delta

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def sg_descent(h, n, t0, budget):
    alpha = 0.03

        dx = 2.0 / n

        lag = k - (n - 1)
        if lag == 0:

            c2[k] = -9999.0

                continue
        grad = np.zeros(n)

            if 0 <= i + lag < n:
