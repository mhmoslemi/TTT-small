    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    k_star = np.argmax(corr)
    s = k_star - (n - 1)  # shift

    for j in range(n):
        val = 0.0
        # Term 1: (1-h[j+s]) if j+s is valid
        if 0 <= j + s < n:
            val += (1.0 - h[j + s])
        # Term 2: -h[j-s] if j-s is valid
        if 0 <= j - s < n:
            val -= h[j - s]
        grad[j] = dx * val
    
    return grad, k_star

        temp2 = 0.0005
        h2 = best.copy()
        deadline = time.time() + remaining - 2
        while time.time() < deadline:
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            eps = rng.uniform(0.001, 0.05)
            d = 1.0 if rng.random() < 0.5 else -1.0
            nh = h2.copy()
            nh[i] += d * eps
            nh[j] -= d * eps
            nh = np.clip(nh, 0.0, 1.0)
            if abs(nh.sum() - target_sum) > 1e-10:
                nh = project(nh)
            c5_new = eval_c5(nh)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = nh.copy()
                h2 = nh
            elif np.log(1.0 - rng.random() + 1e-15) > -(c5_new - best_c5) / max(temp2, 1e-15):
                h2 = nh
            temp2 = max(temp2 * 0.99999, 1e-10)

    return best, best_c5, n
