def _project_box_sum(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} via bisection on threshold."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0.0, 1.0)

def _obj_grad(h, T):

    dx = 2.0 / n
    g = 1.0 - h
    corr_raw = np.correlate(h, g, mode='full')
    corr = corr_raw * dx
    m = np.max(corr)

    # Clip for numerical stability
    z = np.clip(z, -500, 500)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _sg(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    m = np.max(corr)
    z = T * (corr - m)
    z[z > 500] = 500
    z[z < -500] = -500
    e = np.exp(z)
    es = np.sum(e)
    w = e / es
    obj = (1.0 / T) * np.log(es) + m
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k] * dx
        if wk < 1e-14:
            continue

            grad[jlo:jhi] += wk * g[jlo + d:jhi + d]

        if jlo2 < jhi2:
            grad[jlo2:jhi2] -= wk * h[jlo2 - d:jhi2 - d]
    return obj, grad
