import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def grad_corr_lag(h, m):
    """Gradient of the correlation at lag m with respect to h.
    m is the shift: pair (i+m, i) for valid indices.
    """
    n = len(h)
    grad = np.zeros(n)
    # h[j]*(1-h[i]) where j = i+m
    # d/dh[j] = (1-h[i]) when j=i+m valid
    # d/dh[i] = -h[j] when i=j-m valid (i.e., j=i+m)
    # For index p:
    #   as j: i = p-m, valid if 0<=p-m<n -> contribution (1-h[p-m])
    #   as i: j = p+m, valid if 0<=p+m<n -> contribution -h[p+m]
    if m >= 0:
        # i ranges 0 to n-1-m, j = i+m ranges m to n-1
        # d/dh[p] for p in [m, n-1]: (1-h[p-m])
        grad[m:] += (1 - h[:n-m])
        # d/dh[p] for p in [0, n-1-m]: -h[p+m]
        grad[:n-m] -= h[m:]
    else:
        m_pos = -m
        # i ranges m_pos to n-1, j = i-m_pos ranges 0 to n-1-m_pos
        # d/dh[p] for p in [0, n-1-m_pos]: (1-h[p+m_pos])
        grad[:n-m_pos] += (1 - h[m_pos:])
        # d/dh[p] for p in [m_pos, n-1]: -h[p-m_pos]
        grad[m_pos:] -= h[:n-m_pos]
    return grad

def project_sum(h, target):
    """Project h (already in [0,1]) to have exact sum = target."""
    # Use iterative adjustment
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        h = h.copy()
        h[mask] += np.clip(step, -1.0 + h[mask], 1.0 - h[mask])
        h = np.clip(h, 0, 1)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        valid = n - m
        if valid > 0:
            grad[m:] += (1 - h[:valid])
            grad[:valid] -= h[m:]
    else:
        valid = n + m
        if valid > 0:
            grad[:valid] += (1 - h[-valid:])
            grad[m:] -= h[:valid]
    return grad

def project_feasible(h, target):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        valid = n - m
        if valid > 0:
            grad[m:m+valid] += (1.0 - h[:valid])
            grad[:valid] -= h[m:m+valid]
    else:
        valid = n + m
        if valid > 0:
            grad[:valid] += (1.0 - h[n-valid:])
            grad[m:m+valid] -= h[:valid]
    return grad

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for i in range(len(h)):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def upsample(h, new_n):
    n = len(h)
    old_x = np.arange(n) / n
    new_x = np.arange(new_n) / new_n
    h_ext = np.concatenate([h, h[:1]])
    h_new = np.interp(new_x, old_x, h_ext)
    return h_new

        grad = np.zeros(n)

            grad += wi * grad_corr_lag(h, m)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        valid = n - m
        if valid > 0:
            grad[m:] += (1.0 - h[:valid])
            grad[:valid] -= h[m:]
    else:
        s = -m
        valid = n - s
        if valid > 0:
            grad[:valid] += (1.0 - h[s:])
            grad[s:] -= h[:valid]
    return grad

def project_feasible(h, target):

        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for i in range(len(h)):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def upsample(h, new_n):

    h_ext = np.concatenate([h, h[:1]])
    return np.interp(new_x, old_x, h_ext)
