import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def water_fill(y, target_sum):
    h = np.clip(y, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s_mid < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    m = len(corr)
    
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / temp)
    sum_exp = np.sum(exp_c)
    weights = exp_c / sum_exp
    
    obj = max_c + temp * np.log(sum_exp)
    
    idx = np.arange(n)
    grad = np.zeros(n)
    
    for k in range(m):
        w = weights[k]
        if w < 1e-12:
            continue
        mk = k - (n - 1)
        v1 = idx + mk
        v2 = idx - mk
        mask1 = (v1 >= 0) & (v1 < n)
        mask2 = (v2 >= 0) & (v2 < n)
        
        g = np.zeros(n)
        if np.any(mask1):
            g[mask1] += one_minus_h[v1[mask1]]
        if np.any(mask2):
            g[mask2] -= h[v2[mask2]]
        
        grad += (w * dx) * g
    
    return obj, grad

def projected_gradient_descent(h, target_sum, temps, max_iters=500, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for temp in temps:
        h = best_h.copy()
        lr = 0.1 / n
        for it in range(max_iters):

            h_rand = water_fill(h_rand, n_best / 2.0)
            alloc = max(10, int(remaining * 0.15))
            h_opt, c_opt = optimize_h(h_rand, n_best, min(time.time() + alloc, deadline - 15), seed + 2000 + s, max_lbgfs_calls=8)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    # Final: try going to slightly higher n if time permits
    if time.time() < deadline - 40:
        n_hi = min(n_best + 50, 400)
        h_opt, c_opt = optimize_h(best_h, n_hi, deadline - 5, seed + 9999, max_lbgfs_calls=8)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
