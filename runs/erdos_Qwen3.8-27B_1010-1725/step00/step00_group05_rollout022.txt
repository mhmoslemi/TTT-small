from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum} via binary search on offset."""
    lo, hi = -h.max() - 1, 1 - h.min() + 1
    # Actually lo should make all h[i]-lo >= 1 (sum too big), hi should make all h[i]-hi <= 0 (sum too small)
    # Let me just use a wide range

    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) / 2
    return np.clip(h - offset, 0, 1)

def corr_values(h):
    """Compute all correlation values."""
    n = len(h)
    one_minus_h = 1 - h
    return np.correlate(h, one_minus_h, mode='full')

def smooth_max(corr_vals, eps):
    """Smooth max via log-sum-exp."""
    c = corr_vals / eps
    c_max = c.max()
    return eps * (c_max + np.log(np.exp(c - c_max).sum()))

def smooth_max_grad(h, eps):
    """Gradient of smooth max of correlations."""
    n = len(h)
    c = corr_values(h)
    w = np.exp(c / eps - c.max() / eps)
    w /= w.sum()
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        if l >= 0:
            grad[:n-l] += wk * (1 - h[l:])
            grad[l:] -= wk * h[:n-l]
        else:
            ll = -l
            grad[ll:] += wk * (1 - h[:n-ll])
            grad[:n-ll] -= wk * h[ll:]
    return grad

def optimize_h(h0, n, time_budget, seed):
    """Optimize h using projected gradient descent with decreasing epsilon."""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    target_sum = n / 2.0
    best_c5 = np.inf
    best_h = h.copy()
    
    epsilons = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    
    for eps in epsilons:
        if time_budget <= 5:
            break
        # Projected gradient descent
        h = project_onto_constraint(h, target_sum)
        lr = 1.0 / (eps * n)  # initial learning rate
        for it in range(200):
            if time_budget <= 5:
                break
            grad = smooth_max_grad(h, eps)
            subgrad = smooth_max(corr_values(h), eps)
            h_new = project_onto_constraint(h - lr * grad, target_sum)
            
            # Check improvement
            new_obj = smooth_max(corr_values(h_new), eps)
            if new_obj > subgrad + 1e-10:
                lr *= 0.5
                h_new = project_onto_constraint(h - lr * grad, target_sum)
            else:
                h = h_new
            
            # Update best

            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
        
        # Add small perturbation to escape local optima
        h = project_onto_constraint(h + 0.05 * rng.randn(n), target_sum)
    
    return best_h, best_c5
