    # Convert to binary
    sb = (h > 0.5).astype(float)
    # If sum != n/2, adjust
    if n % 2 != 0:
        target = n // 2  # or (n+1)//2
    else:
        target = n // 2
    
    while np.sum(sb) != target:
        if np.sum(sb) > target:
            idx = np.where(sb == 1)[0]
            sb[rng.choice(idx)] = 0
        else:
            idx = np.where(sb == 0)[0]
            sb[rng.choice(idx)] = 1
    
    dx = 2.0 / n
    best = compute_c5(sb, dx)
    current = best
    
    T = 0.1
    T_min = 1e-6
    for i in range(n_iter):
        T = max(T * 0.9999, T_min)
        idx = rng.integers(n)
        # Flip two positions (one 1 to 0, one 0 to 1) to maintain sum
        ones = np.where(sb == 1)[0]
        zeros = np.where(sb == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.integers(len(ones))]
        i0 = zeros[rng.integers(len(zeros))]
        
        sb[i1] = 0
        sb[i0] = 1
        new_c5 = compute_c5(sb, dx)
        
        if new_c5 < current or rng.random() < np.exp((current - new_c5) / T):
            current = new_c5
            if current < best:
                best = current
        else:
            sb[i1] = 1
            sb[i0] = 0
    
    return sb, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_func(h, dx):
    return compute_c5(h, dx)

def sum_eq(h, n):
    return np.sum(h) - n * 0.5

def sum_eq_jac(h, n):
    return np.ones(n)

def make_initial(n, parent, rng):
    if parent is not None and len(parent) == n:
        return parent.astype(float).copy()
    if parent is not None and len(parent) > 0:
        xo = np.linspace(0.0, 2.0, len(parent))
        xn = np.linspace(0.0, 2.0, n)
        h = np.interp(xn, xo, parent)
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h *= (n * 0.5) / s
    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    h += diff / n
    return np.clip(h, 0.0, 1.0)

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        diff = n * 0.5 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        m = (h < 1.0) if diff > 0 else (h > 0.0)
        if not np.any(m):
            break
        h[m] += diff / np.sum(m)
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb(h, rng, sigma):
    h2 = h + rng.normal(0.0, sigma, len(h))
    return project_sum(h2, len(h))

def binary_sa(h_seed, n, rng, max_iter, deadline):
    target = n // 2
    sb = (h_seed > 0.5).astype(float)
    c = int(np.sum(sb))
    while c > target:
        idx = np.flatnonzero(sb == 1)
        sb[rng.integers(len(idx))] = 0.0
        c -= 1
    while c < target:
        idx = np.flatnonzero(sb == 0)
        sb[rng.integers(len(idx))] = 1.0
        c += 1
    dx = 2.0 / n
    cur = compute_c5(sb, dx)
    best = cur
    T = 0.035
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        T = max(T * 0.999995, 1e-9)
        oi = np.flatnonzero(sb)
        zi = np.flatnonzero(1.0 - sb)
        if len(oi) == 0 or len(zi) == 0:
            break
        pi = oi[rng.integers(len(oi))]
        pz = zi[rng.integers(len(zi))]
        sb[pi], sb[pz] = 0.0, 1.0
        nc = compute_c5(sb, dx)
        delta = cur - nc
        if delta > 0 or rng.random() < np.exp(delta / T):
            cur = nc
            if nc < best:
                best = nc
        else:
            sb[pi], sb[pz] = 1.0, 0.0
    return sb, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    best_h, best_c5, best_n = None, float('inf'), 0

    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = np.full(100, 0.5)

    ns = sorted(set([len(parent), 100, 128, 192, 256]))

    for n in ns:
        if time.time() > dl:
            break
        dx = 2.0 / n
        h0 = make_initial(n, parent, rng)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
        ntr = max(2, min(8, 600 // n))
        for tr in range(ntr):
            if time.time() > dl:
                break
            sig = 0.1 * (tr + 1) if tr > 0 else 0.0
            hi = h0 if tr == 0 else perturb(h0, rng, sig)
            try:
                res = minimize(obj_func, hi, args=(dx,), method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-10})
                if np.all(res.x >= -0.05) and np.all(res.x <= 1.05):
                    hc = project_sum(res.x, n)
                    c5 = compute_c5(hc, dx)
                    if c5 < best_c5:
                        best_c5, best_h, best_n = c5, hc.copy(), n
            except Exception:
                pass
        if n % 2 == 0 and time.time() < dl:
            ref = best_h if best_n == n else h0
            iters = 600000 if n <= 128 else 250000
            sb, cs = binary_sa(ref, n, rng, iters, dl)
            if cs < best_c5:
                best_c5, best_h, best_n = cs, sb.copy(), n

    # Extra SA restarts from random binaries if time remains
    if best_h is not None and time.time() < dl * 0.85:
        for extra_n in [100, 128]:
            if time.time() > dl:
                break
            _ = extra_n  # use for SA on best_n or nearby even
        n_sa = best_n if best_n % 2 == 0 else (best_n // 2 + 1) * 2
        for restart in range(3):
            if time.time() > dl:
                break
            rseed = best_h if n_sa == best_n else make_initial(n_sa, parent, rng)
            if restart > 0:
                rseed = np.where(rng.random(n_sa) < 0.5, 1.0, 0.0).astype(float)
                rseed = project_sum(rseed, n_sa)
            iters = 400000 if n_sa <= 128 else 200000
            sb, cs = binary_sa(rseed, n_sa, rng, iters, dl)
            if cs < best_c5:
                best_c5, best_h, best_n = cs, sb.copy(), n_sa

    if best_h is None:
        best_n = len(parent)
        best_h = parent.copy()
        best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
