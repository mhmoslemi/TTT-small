import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    grad = np.zeros(n)
    for idx in range(corr.shape[0]):
        if corr[idx] < mx - 1e-9:
            continue
        lag = idx - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n - 1, n - 1 - lag)
        for i in range(i0, i1 + 1):
            grad[i + lag] += (1.0 - h[i])
            grad[i] -= h[i + lag]
    return mx * dx, grad * dx

def _feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-10
        else:
            m = h > 1e-10
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def _random_h(n, rng):
    h = rng.random(n) * 1.5
    return _feasible(h, n)

def _interp(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _feasible(h_new, n_new)

def _optimize(h0, n, maxiter=300, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        return _c5(x)
    
    def jac(x):
        _, g = _c5_grad(x)
        return g
    
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_best = h0.copy()
    c5_best = _c5(h0)
    if res.fun < c5_best:
        h_best = _feasible(res.x, n)
        c5_best = _c5(h_best)
    return h_best, c5_best

    def fun(x):

    h0 = _feasible(h0, n)

    h_best = _feasible(res.x, n)
    return h_best, _c5(h_best)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    grad = np.zeros(n)
    for idx in range(corr.shape[0]):
        if corr[idx] < mx - 1e-9:
            continue
        lag = idx - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n - 1, n - 1 - lag)
        for i in range(i0, i1 + 1):
            grad[i + lag] += (1.0 - h[i])
            grad[i] -= h[i + lag]
    return mx * dx, grad * dx

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(1000):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def _rand_h(n, rng):
    h = 0.5 + 0.5 * rng.randn(n)
    return _project(h, n)

def _interp(h, n2):
    n1 = len(h)
    x1 = (np.arange(n1) + 0.5) * (2.0 / n1)
    x2 = (np.arange(n2) + 0.5) * (2.0 / n2)
    h2 = np.interp(x2, x1, h)
    return _project(h2, n2)

def _slsqp(h0, n, maxiter=400):
    h0 = _project(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]

    def fun(x):
        return _c5(x)

    def jac(x):
        _, g = _c5_grad(x)
        return g

    try:
        res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_best = _project(res.x, n)
        c5_best = _c5(h_best)
        c5_init = _c5(h0)
        if c5_best > c5_init:
            h_best, c5_best = h0, c5_init
        return h_best, c5_best
    except Exception:
        return h0, _c5(h0)

def _powell(h0, n, maxiter=800):
    h0 = _project(h0, n)
    try:
        res = minimize(_c5, h0, method='Powell',
                       options={'maxiter': maxiter, 'xtol': 1e-10, 'ftol': 1e-12,
                                'disp': False})
        h_best = _project(res.x, n)
        c5_best = _c5(h_best)
        if c5_best > _c5(h0):
            return h0, _c5(h0)
        return h_best, c5_best
    except Exception:
        return h0, _c5(h0)

def _local_search(h, n, rng, n_perturb=10, step=0.02):
    h_best = h.copy()
    c5_best = _c5(h)
    for _ in range(n_perturb):
        h_try = h_best + step * rng.randn(n)
        h_try = _project(h_try, n)
        c5_try = _c5(h_try)
        if c5_try < c5_best:
            h_best, c5_best = h_try, c5_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = 0
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    # Phase 1: optimize the initial 45-point construction
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h0 = _project(h0, n0)

    h_b, c5_b = _slsqp(h0, n0, maxiter=600)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, h_b.copy(), n0

    # Local polish
    h_b, c5_b = _local_search(best_h, n0, rng, n_perturb=20, step=0.03)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, h_b.copy(), n0
    h_b, c5_b = _powell(h_b, n0, maxiter=400)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, h_b.copy(), n0

    # Try random starts at n0
    for i in range(10):
        if elapsed() > budget_s * 0.25:
            break
        h_r = _rand_h(n0, rng)
        h_t, c5_t = _slsqp(h_r, n0, maxiter=200)
        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t.copy(), n0
            h_t2, c5_t2 = _powell(h_t, n0, maxiter=300)
            if c5_t2 < best_c5:
                best_c5, best_h, best_n = c5_t2, h_t2.copy(), n0

    # Phase 2: higher resolutions
    for n_new in [90, 180, 360, 720]:
        rem = budget_s - elapsed()
        if rem < 120:
            break

        h_interp = _interp(best_h, n_new)

        # SLSQP at new resolution
        maxiter_sl = min(400, max(50, int(rem * 0.25 / 0.5)))
        h_t, c5_t = _slsqp(h_interp, n_new, maxiter=maxiter_sl)
        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t.copy(), n_new

        # Powell polish
        rem = budget_s - elapsed()
        if rem > 100:
            h_p, c5_p = _powell(h_t, n_new, maxiter=min(600, int(rem / 0.3)))
            if c5_p < best_c5:
                best_c5, best_h, best_n = c5_p, h_p.copy(), n_new

        # Slightly perturbed restart
        rem = budget_s - elapsed()
        if rem > 80:
            pert = best_h + 0.01 * rng.randn(best_n)
            pert = _project(_interp(pert, n_new), n_new)
            h_t2, c5_t2 = _slsqp(pert, n_new, maxiter=min(300, int(rem / 0.5)))
            if c5_t2 < best_c5:
                best_c5, best_h, best_n = c5_t2, h_t2.copy(), n_new

    # Final: one more SLSQP pass on the best solution
    rem = budget_s - elapsed()
    if rem > 30:
        h_fin, c5_fin = _slsqp(best_h, best_n, maxiter=300)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
