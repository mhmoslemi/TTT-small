Wait, actually c(k) = Σ_i h[i](1-h[i+k]). For a specific k:
- h[j] appears as h[i] (coefficient (1-h[i+k])) when i=j
- h[j] appears as h[i+k] (coefficient -h[i]) when i+k=j, i.e., i=j-k

So ∂c(k)/∂h[j] = (1-h[j+k])*[valid] - h[j-k]*[valid], which is a constant (given other h values).

So c(k) is linear in h[j] for each fixed k. And max_k c(k) is the max of linear functions of h[j], which is convex and piecewise-linear. The optimal h[j] is at a breakpoint or at the boundary [0,1].

The breakpoints are where c(k1)(h[j]) = c(k2)(h[j]) for some pair of lags k1, k2. Since each c(k) is affine in h[j], the intersection points are at h[j] = (c0_k2 - c0_k1)/(a_k1 - a_k2) where a_k is the coefficient and c0_k is the intercept.

This is O(n²) per coordinate (n lags, check all pairs), which is 159² ≈ 25K per coordinate, and 80 coordinates, so 2M per sweep. That's fast!

import numpy as np
import time


def project_h(h, S):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
            g[k:] -= h[:m]
    else:
        m = -k
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
    return g


def lse_gradient_descent(h, S, time_limit, alpha=50.0):
    t0 = time.time()
    n = len(h)
    h = h.astype(float).copy()
    best = eval_c5(h)
    h_best = h.copy()
    lr0 = 0.03
    momentum = np.zeros(n)
    mu = 0.85
    lags = np.arange(2 * n - 1) - (n - 1)
    for it in range(500000):
        if time.time() - t0 > time_limit:
            break
        corr = corr_values(h)
        max_c = np.max(corr)
        scaled = alpha * (corr - max_c)
        exp_vals = np.exp(scaled)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
        g = np.zeros(n)
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w > 1e-8:
                g += w * grad_lag(h, int(lags[idx]))
        momentum = mu * momentum + g
        lr = lr0 / (1.0 + 0.002 * it)
        h = h - lr * momentum
        h = project_h(h, S)
        if it % 300 == 0:
            cur = eval_c5(h)
            if cur < best:
                best = cur
                h_best = h.copy()
    cur = eval_c5(h)
    if cur < best:
        best = cur
        h_best = h.copy()
    return h_best, best


def coord_polish(h, S, time_limit):
    t0 = time.time()
    n = len(h)
    h = h.astype(float).copy()
    best = eval_c5(h)
    h_best = h.copy()
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]
    sweep = 0
    while time.time() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            orig = h[j]
            for d in deltas:
                v = orig + d
                if v < 0.0 or v > 1.0:
                    continue
                h[j] = v
                h_proj = project_h(h, S)
                val = eval_c5(h_proj)
                if val < best - 1e-12:
                    h = h_proj
                    best = val
                    h_best = h.copy()
                    improved = True
                    orig = h[j]
                    break
                else:
                    h[j] = orig
        sweep += 1
        if not improved and sweep > 1:
            break
    return h_best, best


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)
