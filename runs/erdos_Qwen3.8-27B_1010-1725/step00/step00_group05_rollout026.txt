import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def smooth_max_grad(h, temp):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    
    grad = np.zeros(N)
    for idx in range(2*N-1):
        k = idx - (N-1)
        wt = w[idx]
        if wt < 1e-12:
            continue
        lo1 = max(0, k)
        hi1 = min(N, k + N)
        if lo1 < hi1:
            i_vals = np.arange(lo1, hi1)
            grad[i_vals] += wt * (1.0 - h[i_vals - k])
        lo2 = max(0, -k)
        hi2 = min(N, N - k)
        if lo2 < hi2:
            i_vals = np.arange(lo2, hi2)
            grad[i_vals] += wt * (-h[i_vals + k])
    grad *= dx
    return grad

def resample_h(h_old, n_new):
    """Resample h from n_old points to n_new points, preserving integral."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    # Use periodic extension or just interpolate
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    # Renormalize to satisfy sum constraint
    target_sum = n_new / 2.0
    current_sum = np.sum(h_new)
    if current_sum > 0:
        h_new = h_new * (target_sum / current_sum)
    h_new = np.clip(h_new, 0, 1)
    # Re-adjust after clipping
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-10:
        # Distribute difference
        if diff > 0:
            # Add to elements below 1
            mask = h_new < 1.0
            add = diff / np.sum(mask)
            h_new[mask] = np.minimum(h_new[mask] + add, 1.0)
            # Iterate
            for _ in range(10):
                diff = target_sum - np.sum(h_new)
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    mask = h_new < 1.0
                    if not np.any(mask):
                        break
                    add = diff / np.sum(mask)
                    h_new[mask] = np.minimum(h_new[mask] + add, 1.0)
                else:
                    mask = h_new > 0.0
                    if not np.any(mask):
                        break
                    sub = -diff / np.sum(mask)
                    h_new[mask] = np.maximum(h_new[mask] - sub, 0.0)
    return h_new

def optimize_h(h_init, n, temp, max_iter, time_limit):
    """Optimize h with given initial value, size, and temperature."""
    N = n
    target_sum = N / 2.0
    bounds = [(0, 1)] * N
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    def obj(x):
        return smooth_max_obj(x, temp)
    
    def obj_grad(x):
        return smooth_max_grad(x, temp)
    
    h0 = np.clip(h_init, 0, 1).copy()
    # Ensure constraint is satisfied
    h0 = resample_h(h0, N)
    
    result = minimize(
        obj, h0, jac=obj_grad, method='SLSQP',
        bounds=bounds, constraints=[constraint],
        options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False}
    )
    
    h_opt = np.clip(result.x, 0, 1)
    h_opt = resample_h(h_opt, N)
    return h_opt, compute_c5(h_opt)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def smooth_grad(h, temp):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        k = idx - (N - 1)
        wt = w[idx]
        if wt < 1e-14:
            continue
        lo1 = max(0, k)
        hi1 = min(N, k + N)
        if lo1 < hi1:
            iv = np.arange(lo1, hi1)
            grad[iv] += wt * (1.0 - h[iv - k])
        lo2 = max(0, -k)
        hi2 = min(N, N - k)
        if lo2 < hi2:
            iv = np.arange(lo2, hi2)
            grad[iv] += wt * (-h[iv + k])
    grad *= dx
    return grad

def enforce_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return enforce_sum(h_old, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / (n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / (n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return enforce_sum(h_new, n_new)

def optimize_h(h_init, n, temp, max_iter=300, ftol=1e-10):
    N = n
    target_sum = N / 2.0
    bounds = [(0.0, 1.0)] * N

    def obj(x):
        return smooth_obj(x, temp)

    def grad(x):
        return smooth_grad(x, temp)

    def cons_fun(x):
        return np.sum(x) - target_sum

    h0 = enforce_sum(h_init, N)
    result = minimize(
        obj, h0, jac=grad, method='SLSQP',
        bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_fun}],
        options={'maxiter': max_iter, 'ftol': ftol, 'disp': False}
    )
    h_opt = enforce_sum(result.x, N)
    return h_opt, compute_c5(h_opt)
