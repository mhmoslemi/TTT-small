import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, n):
    target_sum = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
    return np.clip(h, 0, 1)

def corr_gradient(h, lag_idx, n):
    """Gradient of corr[lag_idx] w.r.t. h, vectorized."""
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += 1.0
        grad[s:] -= h[:n-s]
        grad[:n-s] -= h[s:]
    else:
        t = -s
        grad[t:] += 1.0
        grad[t:] -= h[:n-t]
        grad[:n-t] -= h[t:]
    return grad

def smooth_max_obj(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = np.max(corr)
    return c_max + (1.0/T) * np.log(np.exp(T*(corr - c_max)).sum())

def smooth_max_grad(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = np.max(corr)
    exp_terms = np.exp(T * (corr - c_max))
    weights = exp_terms / exp_terms.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-10:
            grad += weights[k] * corr_gradient(h, k, n)
    return grad * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_max_obj, h_init, jac=smooth_max_grad,
                  method='SLSQP', bounds=bounds, constraints=cons,
                  options={'maxiter': max_iter, 'ftol': 1e-10})

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h + lam, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        hp = hp + diff / n
        hp = np.clip(hp, 0.0, 1.0)
    return hp

def corr_grad(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += 1.0
        g[s:] -= h[:n - s]
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[t:] += 1.0
        g[t:] -= h[:n - t]
        g[:n - t] -= h[t:]
    return g

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(corr)
    return cmax + (1.0 / T) * np.log(np.sum(np.exp(T * (corr - cmax))))

def smooth_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(corr)
    ex = np.exp(T * (corr - cmax))
    w = ex / ex.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad(h, k, n)
    return g * dx

def sum_jac(x):

    return np.ones(n)

    res = minimize(smooth_obj, h_init, args=(T,), jac=lambda x: smooth_grad(x, T),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    ho = project_feasible(np.clip(res.x, 0, 1), n)
    c5 = compute_c5(ho)[0]
    return ho, c5

    best_h = h.copy()
    best_c5 = compute_c5(h)[0]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        g = corr_grad(h, lag, n)
        lr = lr0 * (1.0 - it / float(max_iters)) + 0.01
        h -= lr * dx * g
        h = project_feasible(h, n)
