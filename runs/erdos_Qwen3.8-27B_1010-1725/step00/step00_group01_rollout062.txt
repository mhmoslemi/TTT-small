```python
def compute_c5_and_grad(h, t=50.0):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    cor = np.correlate(h, one_minus_h, mode="full")
    
    # Soft max
    t_max = cor.max()  # for numerical stability
    weights = np.exp(t * (cor - t_max))
    weights /= weights.sum()
    soft_max_val = t_max + np.log(weights.sum()) / t
    obj = soft_max_val * dx
    
    # Gradient
    grad = np.zeros(n)
    # For each shift s = -(n-1) to n-1, weight = weights[n-1+s]
    # Contribution to grad[j]: w * [(1-h[j-s]) if 0<=j-s<n else 0 - h[j+s] if 0<=j+s<n else 0]
    
    # First part: sum_s w_s * (1-h[j-s]) for valid j-s
    # = convolve-like operation
    # For fixed s, (1-h[j-s]) is valid when 0<=j-s<n, i.e., s<=j<n+s
    # So for each s, we add w_s * (1-h[max(0,-s):n+s]) to grad[0:n]
    
    # More efficiently:
    # Part 1: sum_s w_s * (1-h[j-s]) * [0<=j-s<n]
    # = sum_s w_s * (1-h[j-s]) where j-s in [0,n-1]
    
    # Let me just do it with a loop over shifts (O(n^2) but n is small)
    for s in range(-(n-1), n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        # (1-h[j-s]) for j such that 0<=j-s<n
        lo = max(0, s)  # j >= s (since j-s >= 0)
        hi = min(n, n + s)  # j < n+s (since j-s < n)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo-s:hi-s])
        
        # -h[j+s] for j such that 0<=j+s<n
        lo2 = max(0, -s)  # j >= -s
        hi2 = min(n, n - s)  # j < n-s
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2+s:hi2+s]
    
    grad *= dx
    return obj, grad
```

```python
    for s in range(-(n - 1), n):
        ws = w[n - 1 + s]
        if ws < 1e-14:
            continue
        jlo = max(0, s)
        jhi = min(n - 1, s + n - 1)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += ws * (1.0 - h[jlo - s:jhi - s + 1])
        jlo2 = max(0, -s)
        jhi2 = min(n - 1 - s, n - 1)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] -= ws * h[jlo2 + s:jhi2 + s + 1]
    grad *= dx
    return obj, grad

def adam_optimize(h0, t_deadline, t0=5.0, t1=200.0, lr0=0.04):
    h = h0.copy()
    n = len(h)
    ts = n / 2.0
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    b1, b2, eps = 0.7, 0.9, 1e-9
    step = 0
    best_obj = evaluate_c5(h)
    best_h = h.copy()
    stall = 0
    while time.time() < t_deadline - 1:
        frac = min(1.0, step / 500.0)
        t_val = t0 + (t1 - t0) * frac
        _, grad = c5_softmax_obj_grad(h, t_val)
        step += 1
        lr = lr0 / (1 + 0.005 * step)
        m_a = b1 * m_a + (1 - b1) * grad
        v_a = b2 * v_a + (1 - b2) * grad * grad
        mh = m_a / (1 - b1 ** step)
        vh = v_a / (1 - b2 ** step)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_feasible(h_new, ts)
        obj_true = evaluate_c5(h_new)
        if obj_true < best_obj - 1e-11:
            best_obj = obj_true
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 400:
            break
        if lr < 1e-9:
            break
    return best_h, best_obj

def slsqp_optimize(h0, t_deadline):
    n = len(h)
    ts = n / 2.0
    def slsqp_obj(x):
        cor = np.correlate(x, 1.0 - x, mode="full")
        dx = 2.0 / n
        t_ref = 80.0
        cm = cor.max()
        sm = cm + np.log(np.sum(np.exp(t_ref * (cor - cm)))) / t_ref
        return sm * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts}]
    bounds = [(0, 1)] * n
    res = minimize(slsqp_obj, h0.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
    if res.success:
        h_opt = project_feasible(res.x, ts)
        return h_opt, evaluate_c5(h_opt)
    return h0, evaluate_c5(h0)

def interpolate_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) + 0.5
    x_new = (np.arange(n_new) + 0.5) * n_old / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def block_init(n, rng):
    h = np.zeros(n)
    k = max(2, n // 20)
    period = max(2, n // k)
    half = period // 2
    for i in range(n):
        h[i] = 1.0 if (i % period) < half else 0.0
    return project_feasible(h, n / 2.0)

def random_init(n, rng):
    h = rng.uniform(0, 1, size=n)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    t_deadline = t_start + budget_s
    rng = np.random.default_rng(seed)

    best_obj = float('inf')
    best_h = None
    parent_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
        best_obj = evaluate_c5(parent_h)
        best_h = parent_h.copy()

    n_parent = len(parent_h) if parent_h is not None else 47
    n_list = sorted(set([n_parent, 40, 50, 60, 80, 100]))

    for n in n_list:
        if time.time() > t_deadline - 30:
            break
        is_parent = (n == n_parent and parent_h is not None)
        if is_parent:
            alloc = min(300, (t_deadline - time.time()) * 0.4)
        elif n <= 60:
            alloc = min(120, (t_deadline - time.time()) * 0.15)
        else:
            alloc = min(60, (t_deadline - time.time()) * 0.08)
        seg_end = time.time() + max(alloc, 15)

        inits = []
        if parent_h is not None:
            if n == n_parent:
                inits.append(parent_h.copy())
            else:
                inits.append(interpolate_n(parent_h, n))
        inits.append(block_init(n, rng))
        inits.append(random_init(n, rng))
        inits.append(random_init(n, rng))

        for h_init in inits:
            if time.time() > seg_end - 5:
                break
            # Adam phase
            adam_end = min(seg_end - 5, time.time() + max(20, (seg_end - time.time()) * 0.6))
            h_opt, obj_opt = adam_optimize(h_init, adam_end)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_h = h_opt.copy()
            # SLSQP refinement on best
            if time.time() < seg_end - 10 and obj_opt < 0.42:
                h_sl, obj_sl = slsqp_optimize(h_opt, seg_end - 10)
                if obj_sl < best_obj:
                    best_obj = obj_sl
                    best_h = h_sl.copy()

    if best_h is None:
        n = 47
        best_h = project_feasible(np.full(n, 0.5), n / 2.0)
        best_obj = evaluate_c5(best_h)

    best_obj = evaluate_c5(best_h)
    return (best_h, best_obj, len(best_h))
```
