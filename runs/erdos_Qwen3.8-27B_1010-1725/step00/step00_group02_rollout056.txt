            no_imp += 1
            if no_imp > 200:
                h = best_h.copy()
                c5_cur = best_c5
                no_imp = 0
                t = 0.004
            continue
        
        h[i] += delta
        h[j] -= delta
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        # Fix sum if clipping occurred
        err = np.sum(h) - n / 2.0
        if abs(err) > 1e-10:
            h[i] = np.clip(h[i] + err, 0.0, 1.0)
        
        c5_new = compute_c5(h)
        dc = c5_new - c5_cur
        if dc <= 0:
            c5_cur = c5_new
            no_imp = 0
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            no_imp += 1
            if rng.random() < np.exp(-dc / max(t, 1e-9)):
                c5_cur = c5_new
            else:
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                # Re-fix sum
                err2 = np.sum(h) - n / 2.0
                if abs(err2) > 1e-10:
                    h[i] = np.clip(h[i] + err2, 0, 1)
        
        t = max(t * 0.9995, 1e-4)
        if no_imp > 500:
            h = best_h.copy()
            c5_cur = best_c5
            t = 0.004
            no_imp = 0
            # Also try random restart perturbation
            for _ in range(5):
                a, b_ = rng.choice(n, size=2, replace=False)
                d = rng.uniform(-0.1, 0.1)
                h[a] = np.clip(h[a] + d, 0, 1)
                h[b_] = np.clip(h[b_] - (h[a] - best_h[a]), 0, 1)
            h = np.clip(h, 0, 1)
            err3 = np.sum(h) - n / 2.0
            if abs(err3) > 1e-10:
                h[rng.randint(n)] = np.clip(h[rng.randint(n)] + err3, 0, 1)
    
    return best_h, best_c5

def make_square_wave(n, periods, duty):
    h = np.zeros(n)
    pw = n // periods
    for p in range(periods):
        s = p * pw
        w = int(pw * duty)
        h[s:s + w] = 1.0
    total = pw * periods
    if total < n:
        h[-1] = 1.0
    s_val = h.sum()
    if s_val > 0:
        h *= (n / 2.0) / s_val
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    cands = []
    n0 = 45
    
    try:
        n0 = len(initial_h_values)
        h_init = np.clip(np.array(initial_h_values, dtype=float), 0, 1)
        s = h_init.sum()
        if s > 0:
            h_init = h_init * (n0 / 2.0) / s
            h_init = np.clip(h_init, 0, 1)
        cands.append(h_init)
    except Exception:
        cands.append(np.full(n0, 0.5))
    
    n = n0
    cands.append(np.full(n, 0.5))
    cands.append(make_square_wave(n, 3, 0.5))
    cands.append(make_square_wave(n, 3, 0.4))
    cands.append(make_square_wave(n, 2, 0.5))
    
    for _ in range(3):
        hr = rng.uniform(0.05, 0.95, n)
        hr *= (n / 2.0) / hr.sum()
        cands.append(np.clip(hr, 0, 1))
    
    # Normalize all
    for idx in range(len(cands)):
        c = cands[idx]
        s = c.sum()
        if s > 0:
            c = c * (n / 2.0) / s
        cands[idx] = np.clip(c, 0, 1)
    
    # Score and sort
    scored = []
    for c in cands:
        c_ = np.clip(c, 0, 1)
        s = c_.sum()
        if s > 0 and abs(s - n/2) > 0.5:
            c_ = c_ * (n/2.0) / s
            c_ = np.clip(c_, 0, 1)
        scored.append((compute_c5(c_), c_.copy()))
    scored.sort(key=lambda x: x[0])
    
    best_c5 = scored[0][0]
    best_h = scored[0][1].copy()
    
    # SLSQP from top candidates
    opt_end = t0 + budget_s * 0.40
    for _, h_start in scored[:5]:
        if time.time() > opt_end:
            break
        h_opt = h_start.copy()
        for beta in [6, 12, 25, 50, 100]:
            if time.time() > opt_end:
                break
            try:
                res = minimize(
                    lambda x, nn=n, bb=beta: smooth_max_obj(x, nn, bb),
                    h_opt, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq',
                                  'fun': lambda x, nn=n: x.sum() - nn / 2.0,
                                  'jac': lambda x, nn=n: np.ones(nn)}],
                    options={'maxiter': 150, 'ftol': 1e-10}
                )
                if res.x is not None:
                    h_new = np.clip(res.x, 0, 1)
                    s = h_new.sum()
                    if s > 1 and abs(s - n/2) < 3:
                        h_new = h_new * (n / 2.0) / s
                        h_new = np.clip(h_new, 0, 1)
                    c5_n = compute_c5(h_new)
                    if c5_n <= compute_c5(h_opt) + 1e-7:
                        h_opt = h_new
            except Exception:
                pass
        c5_o = compute_c5(h_opt)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_opt.copy()
    
    # Second round SLSQP from best
    if time.time() < t0 + budget_s * 0.50:
        for beta in [50, 100, 200]:
            if time.time() > t0 + budget_s * 0.50:
                break
            try:
                res = minimize(
                    lambda x, nn=n, bb=beta: smooth_max_obj(x, nn, bb),
                    best_h, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq',
                                  'fun': lambda x, nn=n: x.sum() - nn / 2.0,
                                  'jac': lambda x, nn=n: np.ones(nn)}],
                    options={'maxiter': 200, 'ftol': 1e-11}
                )
                if res.x is not None:
                    h_new = np.clip(res.x, 0, 1)
                    s = h_new.sum()
                    if s > 1 and abs(s - n/2) < 3:
                        h_new = h_new * (n / 2.0) / s
                        h_new = np.clip(h_new, 0, 1)
                    c5_n = compute_c5(h_new)
                    if c5_n < best_c5:
                        best_c5 = c5_n
                        best_h = h_new
            except Exception:
                pass
    
    # Local search (exact pair swaps)
    sa_end = t0 + budget_s * 0.93
    if time.time() < sa_end - 5:
        best_h, best_c5 = local_search(best_h, sa_end, rng)
    
    # Final polish: a few more exact swaps
    if time.time() < t0 + budget_s * 0.97:
        for _ in range(200):
            if time.time() > t0 + budget_s * 0.97:
                break
            i, j = rng.choice(n, size=2, replace=False)
            d = optimal_delta(best_h, n, i, j)
            if abs(d) > 1e-10:
                best_h[i] = np.clip(best_h[i] + d, 0, 1)
                best_h[j] = np.clip(best_h[j] - d, 0, 1)
                c5_t = compute_c5(best_h)
                if c5_t < best_c5:
                    best_c5 = c5_t
    
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
