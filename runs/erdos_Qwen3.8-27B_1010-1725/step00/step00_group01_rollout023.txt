```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max(v, alpha=30):
    v_max = np.max(v)
    v_shifted = v - v_max
    return v_max + np.log(np.sum(np.exp(alpha * v_shifted))) / alpha

def obj_fn(h, n, alpha=30):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return smooth_max(corr * dx, alpha)

def fix_sum(h, target):
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Water-filling style
    for _ in range(1000):
        if diff > 0:
            free = h < 1 - 1e-10
            nfree = np.sum(free)
            if nfree == 0:
                break
            adj = min(diff, (1 - h[free]).min()) if np.any(free) else 0
            h[free] += diff / nfree
            h = np.clip(h, 0, 1)
        else:
            free = h > 1e-10
            nfree = np.sum(free)
            if nfree == 0:
                break
            h[free] += diff / nfree
            h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

def local_search(h, n, max_iters=5000, rng=None):
    """Random pairwise exchange local search"""
    if rng is None:
        rng = np.random.RandomState(0)
    best = compute_c5_raw(h)
    h = h.copy()
    for it in range(max_iters):
        i, j = rng.choice(n, 2, replace=False)
        delta = 0.01 * rng.choice([-1, 1])
        if not (0 <= h[i] + delta <= 1 and 0 <= h[j] - delta <= 1):
            continue
        h[i] += delta
        h[j] -= delta
        c5 = compute_c5_raw(h)
        if c5 < best:
            best = c5
        else:
            h[i] -= delta
            h[j] += delta
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    # Get initial values

    n_list = sorted(set([n0, 40, 45, 50, 55, 60, 70, 80]))

    for n in n_list:

            break

        x_new = np.linspace(0, 2, n)

        else:

        h_init = fix_sum(h_init, n / 2.0)
        c5_init = compute_c5_raw(h_init)
        
        # SLSQP optimization
        alpha = 25
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
        bounds = [(0.0, 1.0)] * n
        
        for attempt in range(3):
            if time.time() - start_time > budget_s - 30:
                break
            
            if attempt == 0:
                h_start = h_init.copy()
            else:
                h_start = h_init + 0.05 * rng.randn(n)
```

c[k] = sum_n a[n] * v[n - k + (len(v) - 1)]  ... no, that's not right either.

Let me look it up mentally: For same-length arrays a, v of length n:
- c has length 2n-1
- c[n-1+k] = sum_i a[i+k] * v[i] for k = -(n-1), ..., (n-1)

So c[n-1] = sum_i a[i]*v[i] (zero lag)
c[n-1+k] = sum_i a[i+k]*v[i] for k > 0 (positive lag, a shifted right by k)
c[n-1-k] = sum_i a[i]*v[i+k] for k > 0 (negative lag)

In our case, a = h, v = 1-h:
c[n-1+k] = sum_i h[i+k] * (1-h[i])

This counts, for shift k, the number of positions where h has a 1 at position i+k and 0 at position i.

The max over all k of c[n-1+k] * dx is our C₅.

OK, my code is correct. Let me finalize.

One issue: for the local search, computing the full correlation each time is O(n^2). For n=50, that's 2500 operations, and with 5000 iterations, that's 12.5M operations. In numpy, this should be fast (< 1 sec).

Let me also add a trick: instead of computing the full correlation for each trial move, I can compute only the affected lags. When I change h[i] by +delta and h[j] by -delta, only the correlations involving positions i and j change. But this is complex to implement correctly, so I'll stick with full recomputation for simplicity.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_max(v, alpha=25):
    vmax = np.max(v)
    return vmax + np.log(np.sum(np.exp(alpha * (v - vmax)))) / alpha

def obj_fn(h, n, alpha=25):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr * dx, alpha)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, max_iters=3000, rng=None, step_init=0.02):
    if rng is None:
        rng = np.random.RandomState(0)
    best = compute_c5_raw(h)
    h = h.copy()
    step = step_init
    stuck = 0
    for it in range(max_iters):
        if stuck > 500:
            step *= 0.7
            stuck = 0
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        delta = step * rng.choice([-1.0, 1.0])
        hi, hj = h[i] + delta, h[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        c5 = compute_c5_raw(h)
        if c5 < best - 1e-12:
            best = c5
            stuck = 0
        else:
            h[i], h[j] = hi - delta, hj + delta
            stuck += 1
    return h, best

def interp_h(h0, n_target, n_src, n0):
    x_src = np.linspace(0, n0, n_src)
    x_tgt = np.linspace(0, n0, n_target)
    return np.interp(x_tgt, x_src, h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.ones(47) * 0.5

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5_raw(best_h)

    n_list = sorted(set([n0, 40, 44, 48, 50, 54, 60, 70, 80]))
    n_list = [n for n in n_list if 30 <= n <= 100]

    for n in n_list:
        if time.time() - t0 > budget_s - 40:
            break

        h_init = interp_h(best_h, n, n0, best_h.size if n0 != best_h.size else n0)
        h_init = fix_sum(h_init, n / 2.0)

        for attempt in range(4):
            if time.time() - t0 > budget_s - 30:
                break

            if attempt == 0:
                h_start = h_init.copy()
            elif attempt == 1:
                h_start = fix_sum(h_init + 0.03 * rng.randn(n), n / 2.0)
            elif attempt == 2:
                h_start = fix_sum(h_init + 0.1 * rng.randn(n), n / 2.0)
            else:
                h_start = fix_sum(0.5 + 0.3 * rng.randn(n), n / 2.0)

            alpha = 25 if attempt < 2 else 40
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                           'jac': lambda x: np.ones(n)}]
            bounds = [(0.0, 1.0)] * n

            try:
                res = minimize(
                    lambda x: obj_fn(x, n, alpha), h_start,
                    method='SLSQP', bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 250, 'ftol': 1e-10}
                )
                h_opt = fix_sum(res.x, n / 2.0)
            except Exception:
                continue

            c5_opt = compute_c5_raw(h_opt)

            time_left = budget_s - (time.time() - t0)
            if time_left > 15:
                h_ls, c5_ls = local_search(h_opt, n, max_iters=int(2000 * time_left / 100),
                                           rng=rng, step_init=0.015)
                if c5_ls < c5_opt:
                    h_opt, c5_opt = h_ls, c5_ls

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_init = h_opt.copy()

    return best_h, float(best_c5), int(len(best_h))
```
