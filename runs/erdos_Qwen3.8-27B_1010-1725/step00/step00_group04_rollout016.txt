    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_obj(N, tgt, T, pen):
    dx = 2.0 / N
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(c, T) * dx + pen * (np.sum(x) - tgt) ** 2
    return f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except:
        h0 = None
    if h0 is None:
        h0 = np.clip(0.5 + rng.randn(42) * 0.1, 0.0, 1.0)
    N = len(h0)
    tgt = N / 2.0
    h0 = project(h0)
    bh = h0.copy()
    bc = compute_c5(bh)
    bounds = [(0.0, 1.0)] * N

    # Phase 1: Multi-scale L-BFGS-B from parent
    for T in [15, 40, 100, 250, 600, 1500]:
        if time.time() - t0 > budget_s * 0.55:
            break
        for _ in range(2):
            f = make_obj(N, tgt, T, 300.0)
            try:
                res = minimize(f, bh, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
            except:
                break
            hn = project(res.x)
            cn = compute_c5(hn)
            if cn < bc:
                bc = cn
                bh = hn.copy()

    # Phase 2: Very high T refinement
    for T in [1500, 4000, 10000]:
        if time.time() - t0 > budget_s * 0.72:
            break
        f = make_obj(N, tgt, T, 5000.0)
        try:
            res = minimize(f, bh, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-14})
        except:
            break
        hn = project(res.x)
        cn = compute_c5(hn)
        if cn < bc:
            bc = cn
            bh = hn.copy()

    # Phase 3: Coordinate-level local search (scan each coord)
    if time.time() - t0 < budget_s * 0.82:
        for sweep in range(4):
            improved = False
            for i in range(N):
                if time.time() - t0 > budget_s * 0.82:
                    break
                old = bh[i]
                candidates = [0.0, 1.0, 0.5, old - 0.05, old - 0.02, old + 0.02, old + 0.05]
                candidates = [max(0.0, min(1.0, c)) for c in candidates]
                best_trial = old
                best_c = bc
                for tval in candidates:
                    if tval == old:
                        continue
                    hh = bh.copy()
                    hh[i] = tval
                    hh = project(hh)
                    c_try = compute_c5(hh)
                    if c_try < best_c:
                        best_c = c_try
                        best_trial = tval
                        bh = hh.copy()
                        improved = True
            if not improved:
                break

    # Phase 4: Random restarts with diverse structures
    deadline = budget_s * 0.93
    while time.time() < deadline:
        mode = int(rng.randint(0, 5))
        if mode == 0:
            h_r = np.clip(0.5 + rng.randn(N) * 0.15, 0, 1)
        elif mode == 1:
            x = np.linspace(0, 2 * np.pi * (1 + rng.randint(1, 5)), N)
            h_r = np.clip(0.5 + 0.35 * np.sin(x) + rng.randn(N) * 0.05, 0, 1)
        elif mode == 2:
            blk = max(1, N // 6)
            h_r = np.zeros(N)
            for b in range(N // blk + 1):
                start = b * blk
                end = min(start + blk, N)
                h_r[start:end] = rng.uniform(0.6, 1.0) if b % 2 == 0 else rng.uniform(0.0, 0.4)
            h_r = np.clip(h_r, 0, 1)
        elif mode == 3:
            h_r = project(bh + rng.randn(N) * 0.05)
        else:
            freq = rng.uniform(1, 4)
            x = np.linspace(0, 2 * np.pi * freq, N)
            h_r = np.clip(0.5 + 0.3 * np.cos(x + rng.random() * 6.28), 0, 1)
            h_r += rng.randn(N) * 0.05
            h_r = np.clip(h_r, 0, 1)

        h_r = project(h_r)
        f = make_obj(N, tgt, 100, 500.0)
        try:
            res = minimize(f, h_r, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-14})
        except:
            continue
        hr = project(res.x)
        cr = compute_c5(hr)
        if cr < bc:
            bc = cr
            bh = hr.copy()

    # Phase 5: Final high-T polish
    if time.time() < budget_s * 0.97:
        for T in [3000, 8000, 20000]:
            if time.time() > budget_s * 0.98:
                break
            f = make_obj(N, tgt, T, 10000.0)
            try:
                res = minimize(f, bh, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 1500, 'ftol': 1e-17, 'gtol': 1e-15})
            except:
                break
            hn = project(res.x)
            cn = compute_c5(hn)
            if cn < bc:
                bc = cn
                bh = hn.copy()

    bh = project(bh)
    bc = compute_c5(bh)
    return (bh, bc, N)
