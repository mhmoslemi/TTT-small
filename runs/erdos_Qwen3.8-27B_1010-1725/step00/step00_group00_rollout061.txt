import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(corr - m))
    return float((m + np.log(s) / beta) * dx)

def penalized_obj(x, dx, beta, n, mu):
    c5v = smooth_obj(x, dx, beta)
    viol = x.sum() - n / 2.0
    return c5v + mu * viol * viol

def eq_constraint(x, n):
    return x.sum() - n / 2.0

def project_constraint(x, n):
