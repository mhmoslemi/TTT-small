def c5_grad(h, tau=10.0):
    """Compute C5 and its gradient using softmax smoothing."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Softmax weighting for smooth max
    tau_local = tau / dx  # scale tau
    # Actually, let's work with corr values directly
    m = np.max(corr)
    exp_corr = np.exp(tau_local * (corr - m))
    weights = exp_corr / np.sum(exp_corr)
    
    c5_smooth = (m + np.log(np.sum(exp_corr)) / tau_local) * dx
    
    # Compute gradient: weighted sum of gradients of each correlation
    # corr[j] = sum_i h[i] * (1-h[i + j - (n-1)])  (valid indices only)
    # shift for corr[j] is k = j - (n-1)
    
    grad = np.zeros(n)
    for j in range(len(corr)):
        if weights[j] < 1e-6:
            continue
        k = j - (n - 1)  # shift
        # corr[j] = sum_i h[i] * (1-h[i+k]) where both indices valid
        # d/dh[i]: (1-h[i+k]) if i+k valid, minus h[i-k] if i-k valid
        # From first factor h[i]: contributes (1-h[i+k]) at position i
        # From second factor (1-h[i+k]): contributes -h[i] at position i+k
        
        # Vectorized: for shift k
        # Positions i where 0<=i<n and 0<=i+k<n:
        lo = max(0, -k)
        hi = min(n, n - k)

            idx = np.arange(lo, hi)
