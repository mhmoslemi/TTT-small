import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full')
    m_star = np.argmax(c)
    s = m_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        js1 = np.arange(0, n - s)
        grad[js1] += (1.0 - h[js1 + s])
        js2 = np.arange(s, n)
        grad[js2] -= h[js2 - s]
    else:
        js1 = np.arange(-s, n)
        grad[js1] += (1.0 - h[js1 + s])
        js2 = np.arange(0, n + s)
        grad[js2] -= h[js2 - s]
    return dx * grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(mask):
            h = h * (target / max(np.sum(h), 1e-15))
            h = np.clip(h, 0, 1)
            break
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def slsqp_optimize(h0, n, maxiter=300, ftol=1e-10):
    target = n / 2.0
    constraints = [{'type': 'eq',
                    'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(c5_func_obj, h0.copy(), jac=c5_grad_func,
                   method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_res = project_feasible(res.x, n)
    return h_res

def c5_func_obj(h):
    return eval_c5(h)

def c5_grad_func(h):
    return c5_grad(h)

        h0 = np.full(n_def, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = eval_c5(best_h)

    # Generate diverse starting points
    starts = [h0.copy()]
    for i in range(6):
        pert = h0 + rng.randn(n) * 0.08 * (1.0 + 0.5 * i)
        starts.append(project_feasible(pert, n))

    # Structured starts
    h_block = np.zeros(n)
    h_block[:n // 2] = 1.0
    starts.append(project_feasible(h_block, n))
