def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    # Simple scaling (might violate bounds)
    # Better: iterative water-filling
    h = h.copy()
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        # Distribute to elements not at bounds
        can_inc = h < 1
        can_dec = h > 0
        if diff > 0:
            mask = can_inc
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = can_dec
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def smooth_max(x, alpha=100):
    """Smooth approximation of max using log-sum-exp"""
    return (1.0/alpha) * np.log(np.sum(np.exp(alpha * x)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
    return np.clip(h, 0.0, 1.0)

def logsumexp(x, alpha):
    x = alpha * x
    m = np.max(x)
    return (m + np.log(np.sum(np.exp(x - m)))) / alpha

def make_objective(alpha):
    pass  # Not used; we pass alpha as arg

def obj_fn(h, n, alpha):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = logsumexp(corr, alpha)
    pen = (np.sum(h) - n / 2.0) ** 2 * 100.0
    return val + pen

def obj_hard(h, n, alpha):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    pen = (np.sum(h) - n / 2.0) ** 2 * 100.0
    return np.max(corr) + pen

def interp_to(h_src, n_new):
    if len(h_src) == n_new:
        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def binary_local_search(h_bin, n, max_iter=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy()
    c5 = compute_c5(h)
    for it in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        # Swap: set h[i]=1, h[j]=0 or vice versa
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
    return h, c5
