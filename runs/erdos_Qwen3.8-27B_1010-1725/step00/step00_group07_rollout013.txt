    no_impr = 0
    while time.time() < t_limit:
        if no_impr > 50000:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        val = compute_c5(cur)
        if val < best_val - 1e-12:
            best = cur.copy()
            best_val = val
            no_impr = 0
        else:
            cur[i], cur[j] = cur[j], cur[i]
            no_impr += 1
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except (NameError, AttributeError, TypeError):
        n0 = 80
        h0 = np.zeros(n0)
        h0[::2] = 1.0

    target = n0 / 2.0
    h0 = project_box_sum(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n0

    t1 = t0 + budget_s * 0.30
    t2 = t0 + budget_s * 0.55
    t3 = t0 + budget_s * 0.80

    # Phase 1: Subgradient descent from parent
    h_sgd, v_sgd = subgradient_optimize(h0, target, t1, 4.0 / n0)
    if v_sgd < best_c5:
        best_h, best_c5, best_n = h_sgd, v_sgd, n0

    # Phase 2: Binary hill climbing from best continuous
    h_bin, v_bin = binary_hill_climb(best_h, n0, t2, rng)
    if v_bin < best_c5:
        best_h, best_c5, best_n = h_bin, v_bin, n0

    # Phase 2b: Binary hill climbing from random restarts
    for restart in range(5):
        if time.time() > t2:
            break
        hr = np.zeros(n0)
        idx = rng.permutation(n0)[:n0 // 2]
        hr[idx] = 1.0
        h_r, v_r = binary_hill_climb(hr, n0, min(t2, time.time() + 20), rng)
        if v_r < best_c5:
            best_h, best_c5, best_n = h_r, v_r, n0

    # Phase 3: Subgradient from best binary
    h_s2, v_s2 = subgradient_optimize(best_h, target, t3, 2.0 / n0)
    if v_s2 < best_c5:
        best_h, best_c5, best_n = h_s2, v_s2, n0

    # Phase 3b: More subgradient with different lr
    h_s3, v_s3 = subgradient_optimize(best_h, target, deadline, 1.0 / n0)
    if v_s3 < best_c5:
        best_h, best_c5, best_n = h_s3, v_s3, n0

    # Phase 4: Quick binary polish
    if time.time() < deadline - 5:
        h_b2, v_b2 = binary_hill_climb(best_h, n0, deadline, rng)
        if v_b2 < best_c5:
            best_h, best_c5, best_n = h_b2, v_b2, n0

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
