import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max(vals, eps):
    m = np.max(vals)
    return float(m + eps * np.log(np.sum(np.exp((vals - m) / eps))))

def obj_smooth(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, eps)

def cons_fun(x):
    return x.sum() - len(x) / 2.0

def cons_jac(x):
    return np.ones(len(x))

def project_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        h += residual / n
        h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h_start, eps, maxiter, n, bounds, cons_list):
    res = minimize(obj_smooth, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons_list, args=(eps,),
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return None
