```python
def binary_hill_climb(h, rng, time_limit):
    """Hill climbing with single swaps for binary h"""
    h = h.copy().astype(int)
    n = len(h)
    current_val = compute_c5(h.astype(float))
    best = h.astype(float).copy()
    best_val = current_val

    stuck = 0
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
```

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def eval_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, rng, n):
    """Round h to binary with exactly n/2 ones"""
    target = n // 2
    b = (h > 0.5).astype(np.int8)
    s = int(np.sum(b))
    if s > target:
        idx = np.where(b == 1)[0]
        rng.shuffle(idx)
        b[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(b == 0)[0]
        rng.shuffle(idx)
        b[idx[:target - s]] = 1
    return b

def sa_binary(h, rng, time_limit, t_init=0.004, t_min=1e-5):
    n = len(h)
    cur = make_binary(h, rng, n).copy()
    cur_val = eval_c5(cur.astype(float))
    best = cur.astype(float).copy()
    best_val = cur_val
    t0 = time.time()
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / (time_limit * 2000))  # ~2000 iters/sec target
        T = t_init * (t_min / t_init) ** frac

        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        cur[i], cur[j] = cur[j], cur[i]
        new_val = eval_c5(cur.astype(float))
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.astype(float).copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_val

def grad_refine(h, n, rng, time_limit, beta=100.0, lr_init=0.005):
    """Projected gradient descent for fractional h"""
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0, 1)
    # Fix sum
    s = np.sum(h)
    if not np.isclose(s, target):
        b = make_binary(h, rng, n).astype(float)
        h = b + 0.0  # start from binary
    
    best = h.copy()
    best_val = eval_c5(h)
    lr = lr_init
    t0 = time.time()
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        corrs = eval_corrs(h)
        max_c = np.max(corrs)
        # Softmax weights
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w /= np.sum(w)
        
        # Compute gradient
        n_corr = len(corrs)  # 2n-1
        grad = np.zeros(n)
        for k in range(n_corr):
            wk = w[k]
            if wk < 1e-14:
                continue
            ell = k - (n - 1)
            # d c_k / d h[j] = (1-h[j-ell]) for valid j-ell - h[j+ell] for valid j+ell
            # First term: j-ell in [0,n-1] => j in [ell, ell+n-1] intersect [0,n-1]
            j_start1 = max(0, ell)
            j_end1 = min(n, ell + n)
            if j_end1 > j_start1:
                grad[j_start1:j_end1] += wk * (1 - h[j_start1 - ell:j_end1 - ell])
            # Second term: j+ell in [0,n-1] => j in [-ell, n-1-ell] intersect [0,n-1]
            j_start2 = max(0, -ell)
            j_end2 = min(n, n - ell)
            if j_end2 > j_start2:
                grad[j_start2:j_end2] -= wk * h[j_start2 + ell:j_end2 + ell]
        
        # Gradient step + projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        # Fix sum constraint
        s = np.sum(h_new)
        diff = target - s
        free = (h_new > 1e-8) & (h_new < 1 - 1e-8)
        if np.any(free):
            h_new[free] += diff / np.sum(free)
        h_new = np.clip(h_new, 0, 1)
        # Final fix
        s = np.sum(h_new)
        diff = target - s
        if abs(diff) > 1e-10:
            # Distribute remaining
            idx = np.argsort(h_new)
            if diff > 0:
                for i in range(len(idx) - 1, -1, -1):
                    add = min(diff, 1 - h_new[idx[i]])
                    h_new[idx[i]] += add
                    diff -= add
                    if abs(diff) < 1e-10:
                        break
            else:
                for i in range(len(idx)):
                    rem = min(-diff, h_new[idx[i]])
                    h_new[idx[i]] -= rem
                    diff += rem
                    if abs(diff) < 1e-10:
                        break
        
        new_val = eval_c5(h_new)
        if new_val < best_val:
            best_val = new_val
            best = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.02)
        else:
            h = h_new  # still accept (momentum-like)
            no_improve += 1
            if no_improve > 50:
                lr *= 0.7
                no_improve = 0
            if lr < 1e-6:
                break
    
    return best, best_val

        h_new[i] = h[j]
```

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def eval_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, rng, n):
    target = n // 2
    b = (h > 0.5).astype(np.int8)
    s = int(np.sum(b))
    if s > target:
        idx = np.where(b == 1)[0]
        rng.shuffle(idx)
        b[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(b == 0)[0]
        rng.shuffle(idx)
        b[idx[:target - s]] = 1
    return b

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
    return h

def sa_binary(h, rng, time_limit, t_init=0.005, t_min=1e-5):
    n = len(h)
    cur = make_binary(h, rng, n).copy()
    cur_val = eval_c5(cur.astype(float))
    best = cur.astype(float).copy()
    best_val = cur_val
    t0 = time.time()
    it = 0
    total_budget = time_limit * 3000
    while time.time() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / max(1, total_budget))
        T = t_init * (t_min / t_init) ** frac
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = cur[j], cur[i]
        new_val = eval_c5(cur.astype(float))
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.astype(float).copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best, best_val

def grad_refine(h, n, rng, time_limit, beta=80.0):
    target = n / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best = h.copy()
    best_val = eval_c5(h)
    lr = 0.008
    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limit:
        corrs = eval_corrs(h)
        max_c = np.max(corrs)
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros_like(corrs)
            w[np.argmax(corrs)] = 1.0
        else:
            w /= wsum
        n_corr = len(corrs)
        grad = np.zeros(n)
        for k in range(n_corr):
            wk = w[k]
            if wk < 1e-14:
                continue
            ell = k - (n - 1)
            js1 = max(0, ell)
            je1 = min(n, ell + n)
            if je1 > js1:
                grad[js1:je1] += wk * (1.0 - h[js1 - ell:je1 - ell])
            js2 = max(0, -ell)
            je2 = min(n, n - ell)
            if je2 > js2:
                grad[js2:je2] -= wk * h[js2 + ell:je2 + ell]
        h_new = h - lr * grad
        h_new = project_sum(h_new, target)
        new_val = eval_c5(h_new)
        if new_val <= best_val + 1e-12:
            h = h_new
            stall = 0
            if new_val < best_val:
                best_val = new_val
                best = h_new.copy()
            lr = min(lr * 1.05, 0.03)
        else:
            stall += 1
            if stall > 80:
                lr *= 0.6
                stall = 0
                if lr < 1e-7:
                    break
    return best, best_val

def upsample_h(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        j = min(int(i * n_old / n_new), n_old - 1)
        h_new[i] = h[j]
    return h_new
```
