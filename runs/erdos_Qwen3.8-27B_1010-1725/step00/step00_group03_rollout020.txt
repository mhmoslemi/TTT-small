where the sum is over i such that 0 ≤ i < n and 0 ≤ i - (m - (n-1)) < n.

So for a given m, the valid range of i is:
max(0, m - (n-1)) ≤ i ≤ min(n-1, m)

And the term is h[i] * (1 - h[i - m + n - 1])

- If j is in the second factor position: h[j+m-n+1] * (1-h[j]) where j is the index in (1-h)
  → contribution: -h[j+m-n+1]
- Could be both if j appears in both positions (when the window overlaps)

def _try(h, best_c5, best_h, budget_s, t0, frac):
    global _c5_temp
    n = len(h)
    hn = _slsqp(h, _c5_temp, 400)
    hn = _proj(hn, n)
    c = compute_c5(hn)
    if c < best_c5:
        best_c5, best_h = c, hn.copy()
    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    best_h = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0).copy()
    n0 = len(best_h)
    best_h = _proj(best_h, n0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP refinement at decreasing temperatures
    for t in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t0 > budget_s * 0.22:
            break
        global _c5_temp
        _c5_temp = t
        hn = _slsqp(best_h, t, 600)
        hn = _proj(hn, n0)
        c = compute_c5(hn)
        if c < best_c5:
            best_c5, best_h = c, hn.copy()

    n = len(best_h)

    # Phase 2: Gaussian perturbations + re-optimize
    for i in range(40):
        if time.time() - t0 > budget_s * 0.55:
            break
        s = 0.2 * np.exp(-i / 25.0) + 0.03
        hp = _proj(best_h + np.random.randn(n) * s, n)
        hn = _slsqp(hp, 0.005, 300)
        hn = _proj(hn, n)
        c = compute_c5(hn)
        if c < best_c5:
            best_c5, best_h = c, hn.copy()

    # Phase 3: Structural transforms (shift, complement)
    for sh in [n // 4, n // 2, 3 * n // 4]:
        if time.time() - t0 > budget_s * 0.65:
            break
        hs = np.roll(best_h, sh)
        hs = _proj(1.0 - hs, n)
        hn = _slsqp(hs, 0.005, 300)
        hn = _proj(hn, n)
        c = compute_c5(hn)
        if c < best_c5:
            best_c5, best_h = c, hn.copy()

    # Phase 4: Binary (0/1) solutions with local search
    if n % 2 == 0 and time.time() - t0 < budget_s * 0.78:
        for i in range(15):
            if time.time() - t0 > budget_s * 0.78:
                break
            idx = np.random.choice(n, n // 2, replace=False)
            hb = np.zeros(n)
            hb[idx] = 1.0
            hb, cb = _bin_ls(hb, 2500)
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()

    # Phase 5: Upscale to higher resolution
    for mult in [2, 3]:
        n2 = min(n * mult, 500)
        if n2 <= n or time.time() - t0 > budget_s * 0.9:
            continue
        hu = _proj(_upscale(best_h, n2), n2)
        for t in [0.01, 0.003, 0.001]:
            if time.time() - t0 > budget_s * 0.97:
                break
            hn = _slsqp(hu, t, 300)
            hn = _proj(hn, n2)
            c = compute_c5(hn)
            if c < best_c5:
                best_c5, best_h = c, hn.copy()
                hu = hn.copy()

    return best_h, best_c5, len(best_h)
