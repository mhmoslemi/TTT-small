def subgradient_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    
    grad = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        grad[i + shift] += (1 - h[i]) * dx
        grad[i] -= h[i + shift] * dx
    else:
        s = -shift
        i = np.arange(s, n)
        grad[i - s] += (1 - h[i]) * dx
        grad[i] -= h[i - s] * dx

Actually, wait - I need to double-check the "full" correlation formula. Let me verify:

For a = [a0, a1, ..., a_{n-1}] and b = [b0, b1, ..., b_{n-1}]:
np.correlate(a, b, "full") has length 2n-1.

The numpy docs say: c_k = sum_n a_{n+k} * conj(b_n)

In full mode, k ranges from -(n_b-1) to (n_a-1) = -(n-1) to n-1.

So c_k = sum_{n: 0≤n<n_and 0≤n+k<n} a[n+k] * b[n]

With a = h, b = 1-h:
c_k = sum_{n: 0≤n<n and 0≤n+k<n} h[n+k] * (1-h[n])

The output is ordered as: c_{-(n-1)}, c_{-(n-2)}, ..., c_0, ..., c_{n-2}, c_{n-1}
So output[0] = c_{-(n-1)}, output[n-1] = c_0, output[2n-2] = c_{n-1}.

And C5 = max(output) * dx.

For k=0: c_0 = sum_n h[n] * (1-h[n]) = sum h - sum h². For binary h, this is 0.

For the subgradient at k*:
∂c_{k*}/∂h[j] = sum over terms involving h[j].

In c_k = sum_n h[n+k] * (1-h[n]):
- h[j] appears as h[n+k] when n+k=j, i.e., n=j-k: term is (1-h[j-k])*dx (if 0≤j-k<n)
- h[j] appears as (1-h[n]) when n=j: term is h[j+k]*(-1)*dx (if 0≤j+k<n)

So: ∂c_k/∂h[j] = (1-h[j-k])*dx * [0≤j-k<n] - h[j+k]*dx * [0≤j+k<n]

And the subgradient of C5 = max c_k * dx at the argmax k* is:
g[j] = dx * [(1-h[j-k*])*[0≤j-k*<n] - h[j+k*]*[0≤j+k*<n]]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def local_search_swap(h, tlimit, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < tlimit:
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
        nc = compute_c5(h)
        if nc <= c5:
            c5 = nc
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5

def local_search_transfer(h, tlimit, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = 0.08
    t0 = time.time()
    while time.time() - t0 < tlimit:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-step, step)
        if h[i]+d < 0 or h[i]+d > 1 or h[j]-d < 0 or h[j]-d > 1:
            continue
        h[i] += d
        h[j] -= d
        nc = compute_c5(h)
        if nc <= c5:
            c5 = nc
            step = min(0.2, step * 1.03)
        else:
            h[i] -= d
            h[j] += d
            step = max(0.005, step * 0.97)
    return h, c5

    c5 = compute_c5(h)
    t0 = time.time()

        d = nc - c5

            c5 = nc
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5

        idx = np.arange(n - shift)
        grad[idx + shift] += (1.0 - h[idx]) * dx
        grad[idx] -= h[idx + shift] * dx
    else:
        s = -shift
        idx = np.arange(s, n)
        grad[idx - s] += (1.0 - h[idx]) * dx
        grad[idx] -= h[idx - s] * dx
    h_new = np.clip(h - alpha * grad, 0.0, 1.0)
    h_new = _fix_sum(h_new, n)
    return h_new
