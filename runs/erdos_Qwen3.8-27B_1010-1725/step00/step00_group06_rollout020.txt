    dx = 2.0 / n

def project_feasible(h, target_sum):
    n = len(h)

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_smooth_max(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    K = len(corr)
    
    # Softmax weights
    c_max = np.max(corr)
    exp_vals = np.exp(T * (corr - c_max))
    w_sum = np.sum(exp_vals)
    if w_sum > 1e-300:
        w = exp_vals / w_sum
    else:
        w = np.zeros(K)
        w[np.argmax(corr)] = 1.0
    
    grad = np.zeros(n)
    for m in range(K):
        if w[m] < 1e-10:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            i_start, i_end = 0, n - 1 - lag
        else:
            i_start, i_end = -lag, n - 1
        if i_start > i_end:
            continue
        i_vals = np.arange(i_start, i_end + 1)
        j1 = i_vals + lag
        grad[j1] += w[m] * dx * (1.0 - h[i_vals])
        grad[i_vals] -= w[m] * dx * h[i_vals + lag]
    
    return grad

    best_h = h.copy()

        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_smooth_max(h, T):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    K = len(corr)
    c_max = np.max(corr)
    exp_vals = np.exp(T * (corr - c_max))
    w_sum = np.sum(exp_vals)
    if w_sum > 1e-300:
        w = exp_vals / w_sum
    else:
        w = np.zeros(K)
        w[np.argmax(corr)] = 1.0
    grad = np.zeros(n)
    for m in range(K):
        if w[m] < 1e-12:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            i_start, i_end = 0, n - 1 - lag
        else:
            i_start, i_end = -lag, n - 1
        if i_start > i_end:
            continue
        i_vals = np.arange(i_start, i_end + 1)
        j1 = i_vals + lag
        grad[j1] += w[m] * dx * (1.0 - h[i_vals])
        grad[i_vals] -= w[m] * dx * h[i_vals + lag]
    return grad

def optimize_gradient(h, t_start, budget_s, max_frac=0.85):
    n = len(h)
    target_sum = n / 2.0
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_iters = 3000
    T_start, T_end = 3.0, 100.0
    lr_start, lr_end = 0.08, 0.002
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(n_iters):
        if time.time() - t_start > t_start + budget_s * max_frac:
            break
        frac = it / max(1, n_iters - 1)
        T = T_start + (T_end - T_start) * frac
        lr = lr_start * (lr_end / lr_start) ** frac
        grad = gradient_smooth_max(h, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

def random_perturb_search(h, t_start, budget_s, max_frac=0.90, n_trials=20000):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

def pair_swap_search(h, t_start, budget_s, max_frac=0.90):
