    # Try to reduce overlap at the worst lag
    # Increase h where h is 1 but h at shift is 0 (reduce the product)
    # Or increase h at the shifted position
    k = worst_lag  # in correlation index, offset by n-1
    actual_lag = k - (n - 1)  # convert to actual lag
    
    h_new = h.copy()
    # Pick a random position and adjust
    for _ in range(3):
        # Find positions contributing to the overlap
        if actual_lag >= 0:
            # h[i] * (1-h[i+k]) for i in [0, n-k-1]
            valid = np.arange(n - abs(actual_lag))
            if actual_lag >= 0:
                contrib = h[valid] * (1 - h[valid + actual_lag])
            else:
                contrib = h[valid + abs(actual_lag)] * (1 - h[valid])
        else:
            valid = np.arange(abs(actual_lag), n)
            if actual_lag < 0:
                contrib = h[valid] * (1 - h[valid + actual_lag])
            else:
                contrib = h[valid + abs(actual_lag)] * (1 - h[valid])

        if contrib.sum() > 0:
            idx = valid[rng.choice(len(valid), p=contrib/contrib.sum())]

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_h(h, n):
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            cnt = int(free.sum())
            if cnt == 0:
                break
            h[free] += diff / cnt
            np.minimum(h[free], 1.0, out=h[free])
        else:
            free = h > 1e-14
            cnt = int(free.sum())
            if cnt == 0:
                break
            h[free] += diff / cnt
            np.maximum(h[free], 0.0, out=h[free])
    return np.clip(h, 0.0, 1.0)
