    # Water-filling approach

def _sum_constraint(x, target):
    return np.sum(x) - target

def _cd_step(h, dx, target, n, n_corr, iters=50):
    """Coordinate descent refinement."""
    for _ in range(iters):
        improved = False
        for i in range(n):
            save = h[i]
            # Try moving h[i] greedily
            best_val = _compute_c5(h, dx)
            best_i_val = save
            # Binary search for best h[i]
            # The objective is piecewise quadratic in h[i]
            # Try a grid
            for trial in [max(0, save-0.05), max(0, save-0.02), save, min(1, save+0.02), min(1, save+0.05)]:
                h[i] = trial
                h = _adjust_sum(h, target, n)
                val = _compute_c5(h, dx)
                if val < best_val - 1e-10:
                    best_val = val
                    best_i_val = trial
            h[i] = best_i_val
            h = _adjust_sum(h, target, n)
            # Check if improved
            if abs(h[i] - save) > 1e-6:
                improved = True
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _adjust_sum(h, target, n):
    h = h.copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr < 1e-15:
                break
            h += diff * room / tr
        else:
            tm = np.sum(h)
            if tm < 1e-15:
                break
            h += diff * h / tm
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max(vals, alpha):
    m = np.max(vals)
    s = np.sum(np.exp(alpha * (vals - m)))
    return m + np.log(s) / alpha

def _sqsqp_obj(x, n, dx, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return _smooth_max(corr, alpha)

def _cd_refine(h, dx, target, n, rounds=8):
    h = h.copy()
    for r in range(rounds):
        for i in range(n):
            save = h[i]
            best_c5 = _compute_c5(h, dx)
            best_t = save
            # Evaluate at candidate values
            candidates = np.concatenate([
                np.linspace(0, 1, 11),
                np.array([max(0.0, save - 0.05), max(0.0, save - 0.02),
                          min(1.0, save + 0.02), min(1.0, save + 0.05)])
            ])
            for t_val in candidates:
                h[i] = t_val
                h = _adjust_sum(h, target, n)
                v = _compute_c5(h, dx)
                if v < best_c5 - 1e-12:
                    best_c5 = v
                    best_t = t_val
            h[i] = best_t
            h = _adjust_sum(h, target, n)
    return h

def _optimize_n(n, h0, seed):
    dx = 2.0 / n
    target = n / 2.0

    # Interpolate h0 to new n if needed
    if len(h0) != n:
        x_old = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n)
        h_start = np.interp(x_new, x_old, h0)
    else:
        h_start = h0.copy()
    h_start = np.clip(h_start, 0, 1)
    h_start = _adjust_sum(h_start, target, n)

    best_h = h_start.copy()
    best_c5 = _compute_c5(h_start, dx)

    # SLSQP with multiple alpha values
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    for alpha in [40.0, 100.0, 200.0]:
        def obj(x):
            return _sqsqp_obj(x, n, dx, alpha)
        try:
            res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-10})
            h_res = np.clip(res.x, 0.0, 1.0)
            h_res = _adjust_sum(h_res, target, n)
            c5_res = _compute_c5(h_res, dx)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                h_start = h_res.copy()
        except Exception:
            pass

    # Coordinate descent refinement
    h_cd = _cd_refine(best_h, dx, target, n, rounds=6)
    c5_cd = _compute_c5(h_cd, dx)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get initial values
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    best_h = np.clip(best_h, 0, 1)
    best_h = _adjust_sum(best_h, target0, n0)
    best_c5 = _compute_c5(best_h, dx0)

    # Phase 1: Optimize at original n
    if time.time() - t0 < budget_s - 120:
        h_opt, c5_opt = _optimize_n(n0, h0, seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Try nearby n values
    tried_ns = [n0]
    for dn in [2, 3, 4, 6, 8, 5, 10]:
        for sign in [1, -1]:
            n_try = n0 + sign * dn
            if n_try < 20 or n_try > 300 or n_try in tried_ns:
                continue
            if time.time() - t0 > budget_s - 90:
                break
            tried_ns.append(n_try)
            h_try, c5_try = _optimize_n(n_try, best_h, seed + n_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Random restarts at best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    target_best = n_best / 2.0

    if time.time() - t0 < budget_s - 60:
        n_restarts = 5
        for rs in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            # Perturbed restart
            h_rand = best_h.copy()
            noise = np.random.randn(n_best) * 0.1
            h_rand = np.clip(h_rand + noise, 0, 1)
            h_rand = _adjust_sum(h_rand, target_best, n_best)
            h_r, c5_r = _optimize_n(n_best, h_rand, seed + 1000 + rs)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Final CD refinement
    if time.time() - t0 < budget_s - 10:
        h_final = _cd_refine(best_h, 2.0 / len(best_h), len(best_h) / 2.0, len(best_h), rounds=4)
        c5_final = _compute_c5(h_final, 2.0 / len(best_h))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    n_final = len(best_h)
    return best_h, best_c5, n_final
