∂f_l/∂h[j] = (1-h[j+l]) * [0≤j<N AND 0≤j+l<N] - h[j-l] * [0≤j-l<N AND 0≤j<N]

This simplifies to:
∂f_l/∂h[j] = (1-h[j+l]) * [0≤j+l<N] - h[j-l] * [0≤j-l<N]

(where we already know 0≤j<N since j is an index of h)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def compute_subgradient(h, dx):
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Get all active indices (within small tolerance)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        l = idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + l < n:
                g += (1.0 - h[j + l])
            if 0 <= j - l < n:
                g -= h[j - l]
            grad[j] += g
    grad /= len(active)
    grad *= dx
    return grad

def project_constraints(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        mask = x < 1.0
        if mask.sum() == 0:
            return x
        x[mask] += diff / mask.sum()
    else:
        mask = x > 0.0
        if mask.sum() == 0:
            return x
        x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)

def pgd(h0, dx, target_sum, max_iter=8000, tol=1e-13):
    n = len(h0)
    h = project_constraints(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step = 0.05
    prev_c5 = best_c5
    stall = 0
    for it in range(max_iter):
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = compute_subgradient(h, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = project_constraints(h - step * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5 - 1e-15:
            h = h_new
            step = min(step * 1.03, 0.5)
        else:
            step *= 0.6
            if step < 1e-13:
                break
        if stall > 2000:
            step = max(step * 2, 0.01)
            stall = 0
    return best_h, best_c5

def slsqp_optimize(h0, dx, target_sum, max_iter=300):
    n = len(h0)
    def obj(x):
        return compute_c5(x, dx)
    def grad(x):
        return compute_subgradient(x, dx)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_opt = project_constraints(res.x, target_sum)
    return h_opt, compute_c5(h_opt, dx)

def make_candidates(n, rng, init_h, target_sum):
    cands = []
    if init_h is not None:
        ni = len(init_h)
        x_old = np.linspace(0, 2, ni, endpoint=False) + 1.0 / ni
        x_new = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        fo = interp1d(x_old, init_h, kind='linear', bounds_error=False,
                      fill_value=(init_h[0], init_h[-1]))
        hr = project_constraints(np.array(fo(x_new), dtype=float), target_sum)
        cands.append(hr)
        # Perturbed resample
        for amp in [0.05, 0.15]:
            hp = project_constraints(hr + amp * rng.randn(n), target_sum)
            cands.append(hp)
    # Binary
    for _ in range(2):
        hb = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        hb[idx] = 1.0
        cands.append(hb)
    # Smooth
    for _ in range(2):
        f1, f2 = rng.randint(1, 6), rng.randint(2, 9)
        hs = 0.5 + 0.35 * np.sin(2 * np.pi * f1 * np.arange(n) / n) + \
             0.25 * np.sin(2 * np.pi * f2 * np.arange(n) / n + rng.rand() * 6.28)
        cands.append(project_constraints(hs, target_sum))
    # Block patterns
    for blocks in [4, 6, 8]:
        bs = n // blocks
        hb = np.zeros(n)
        for b in range(blocks):
            if b % 2 == 0:
                hb[b * bs:(b + 1) * bs] = 1.0
        hb = project_constraints(hb, target_sum)
        cands.append(hb)
    return cands
