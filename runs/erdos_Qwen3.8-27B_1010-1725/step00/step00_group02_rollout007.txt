    t_start = time.time()
    deadline = t_start + budget_s - 5

    h0 = np.array(initial_h_values, dtype=float).copy()

    def project(h_arr, target_sum):
        h_arr = np.clip(h_arr, 0.0, 1.0)
        for _ in range(60):
            diff = target_sum - h_arr.sum()
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                mask = h_arr < 1.0 - 1e-9
            else:
                mask = h_arr > 1e-9
            if not mask.any():
                break
            h_arr[mask] += diff / mask.sum()
            h_arr = np.clip(h_arr, 0.0, 1.0)
        return h_arr

    def evaluate(h_arr, n):
        dx = 2.0 / n
        corr = np.correlate(h_arr, 1.0 - h_arr, mode="full")
        return np.max(corr) * dx

    def run_sa(h_init, n, time_limit, T_init=0.025):
        target = n / 2.0
        h = project(h_init.copy(), target)
        best_h = h.copy()
        best_c5 = evaluate(h, n)
        T = T_init
        stall = 0

        t_end = time.time() + time_limit
        while time.time() < t_end and stall < 400000:
            r = rng.random()
            if r < 0.45:
                n_p = 1 if rng.random() < 0.5 else 2
                idx = rng.choice(n, size=n_p, replace=False)
                h_new = h.copy()
                h_new[idx] += rng.normal(0, T, size=n_p)
            elif r < 0.75:
                i, j = rng.choice(n, size=2, replace=False)
                d = rng.uniform(-2.5 * T, 2.5 * T)
                h_new = h.copy()
                h_new[i] += d
                h_new[j] -= d
            elif r < 0.92:
                n_p = rng.integers(3, min(7, n // 4) + 1)
                idx = rng.choice(n, size=n_p, replace=False)
                h_new = h.copy()
                h_new[idx] += rng.normal(0, T * 1.5, size=n_p)
            else:
                n_p = rng.integers(n // 5, n // 3 + 1)
                idx = rng.choice(n, size=n_p, replace=False)
                h_new = h.copy()
                h_new[idx] += rng.normal(0, 0.12, size=n_p)

            h_new = project(h_new, target)
            c5_new = evaluate(h_new, n)

            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                stall = 0
            else:
                stall += 1
                delta = c5_new - best_c5
                if delta < 0:
                    accept = True
                else:
                    accept = rng.random() < np.exp(-delta / max(T, 1e-12))
                if accept:
                    h = h_new

            T *= 0.99997
            if T < 4e-5:
                T = 4e-5

        return best_h, best_c5

    def polish(h, n, rounds=5):
        target = n / 2.0
        h = h.copy()
        for _ in range(rounds):
            changed = False
            order = rng.permutation(n)
            for i in order:
                h_save = h[i]
                for trial_val in (0.0, 1.0, 0.5, h_save * 0.5, h_save + 0.25, h_save - 0.25):
                    trial_val = max(0.0, min(1.0, trial_val))
                    h_test = h.copy()
                    h_test[i] = trial_val
                    h_test = project(h_test, target)
                    c5_t = evaluate(h_test, n)
                    if c5_t < evaluate(h, n) - 1e-14:
                        h = h_test
                        changed = True
                        break
                if h[i] != h_save:
                    h[i] = h[i]  # keep new value
        return h

    n1 = len(h0)
    total_time = deadline - time.time()

    if total_time < 30:
        h = project(h0.copy(), n1 / 2.0)
        c5 = evaluate(h, n1)
        return h, c5, n1

    # Phase 1: SA at original resolution (55% of time)
    t1 = total_time * 0.55
    h1, c5_1 = run_sa(h0.copy(), n1, t1)

    # Phase 2: Upsample to 2n and refine (35% of time)
    t2 = total_time * 0.35
    n2 = n1 * 2
    h2_init = np.repeat(h1, 2)
    h2, c5_2 = run_sa(h2_init, n2, t2)

    # Phase 3: Polish the best (10% of time)
    t3 = deadline - time.time()
    if t3 > 5:
        if c5_2 < c5_1:
            h2 = polish(h2, n2, rounds=3)
            c5_2 = evaluate(h2, n2)
        else:
            h1 = polish(h1, n1, rounds=3)
            c5_1 = evaluate(h1, n1)

    if c5_2 < c5_1:
        return h2, c5_2, n2
    return h1, c5_1, n1
