import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def soft_max_c5(h, tau):
    """Smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) / tau
    exps = np.exp(shifted - np.max(shifted))  # for numerical stability
    w = exps / exps.sum()
    obj = m + tau * np.log(exps.sum())
    
    # Gradient: sum_d w[d] * grad(c(d))
    grad = np.zeros(n)
    # Only compute for significant weights
    top_k = np.argsort(w)[-20:]  # top 20 lags
    for d_idx in top_k:
        if w[d_idx] < 1e-12:
            continue
        d = d_idx - (n - 1)
        # c(d) = sum_{i: 0<=i<n, 0<=i+d<n} h[i+d]*(1-h[i])
        # grad: for each valid i:
        #   dh[i+d] += (1-h[i])
        #   dh[i] += -h[i+d]
        for i in range(n):
            ip = i + d
            if 0 <= ip < n:
                grad[i + d] += w[d_idx] * (1.0 - h[i]) * dx
                grad[i] += w[d_idx] * (-h[i + d]) * dx
    
    return obj, grad

    """Upsample h from len(h) to new_n using linear interpolation."""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:

        best_n = len(best_h)
    else:

    def elapsed():
        return time.time() - t0
    
    def optimize(h_init, n, time_limit, tau_init=0.05, tau_end=0.005):

        h = h_init.copy()
        best_local = compute_c5(h)

        return h, best_local

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-16
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    s = float(np.sum(np.exp((corr - m) / tau)))
    return m + tau * np.log(s)


def soft_max_grad(h, tau, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    exps = np.exp((corr - m) / tau)
    w = exps / float(np.sum(exps))
    grad = np.zeros(n)
    idxs = np.argsort(w)[::-1][:top_k]
    for di in idxs:
        wi = float(w[di])
        if wi < 1e-12:
            break
        d = int(di) - (n - 1)
        i_all = np.arange(n)
        ip = i_all + d
        valid = (ip >= 0) & (ip < n)
        iv = i_all[valid]
        pv = ip[valid]
        np.add.at(grad, pv, (1.0 - h[iv]) * wi * dx)
        np.add.at(grad, iv, (-h[pv]) * wi * dx)
    return grad


def upsample_linear(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def gradient_optimize(h0, n, t_deadline, tau_start, tau_end, max_iter):
    h = h0.copy()
    best_val = compute_c5(h)
    lr = 0.05
    mu = 0.9
    mom = np.zeros(n)
    it = 0
    stall = 0
    while time.time() < t_deadline and it < max_iter:
        it += 1
        frac = min(float(it) / 5000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        grad = soft_max_grad(h, tau, top_k=30)
        mom = mu * mom - lr * grad
        h_new = project_h(h + mom, n)
        val = compute_c5(h_new)
        if val < best_val - 1e-12:
            best_val = val
            h = h_new
            stall = 0
            lr = min(lr * 1.003, 0.4)
        else:
            lr = max(lr * 0.93, 0.0005)
            mu = max(mu * 0.997, 0.3)
            stall += 1
            if stall > 200:
                h = project_h(h + 0.03 * np.random.randn(n), n)
                best_val = compute_c5(h)
                lr = 0.05
                mu = 0.9
                mom = np.zeros(n)
                stall = 0
    return h, best_val

            else:

    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)
