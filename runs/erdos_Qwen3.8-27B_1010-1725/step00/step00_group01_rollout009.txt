import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: random restarts (binary + fractional)
    remaining = budget_s - (time.time() - t_start) - 5
    n_best = len(best_h)
    if remaining > 40:
        t_per = min(15.0, remaining / 6.0)
        for trial in range(6):
            if time.time() - t_start > budget_s - 5:
                break
            if trial == 0:
                hr = _perturb(best_h, n_best, rng, scale=0.15)
            elif trial == 1:
                hr = _make_binary(n_best, rng, n_best / 2.0)
            elif trial == 2:
                hr = _perturb(best_h, n_best, rng, scale=0.4)
            else:
                hr = _perturb(best_h, n_best, rng, scale=0.25)
            h3, c53 = _optimize(hr, n_best, t_per, rng)
            h3, c53 = _coord_polish(h3, n_best, max_passes=3)
            if c53 < best_c5:
                best_c5 = c53
                best_h = h3

    # Phase 4: final polish at best resolution
    remaining = budget_s - (time.time() - t_start) - 2
    n_best = len(best_h)
    if remaining > 10:
        best_h, best_c5 = _coord_polish(best_h, n_best, max_passes=10)

    # Final projection to ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
