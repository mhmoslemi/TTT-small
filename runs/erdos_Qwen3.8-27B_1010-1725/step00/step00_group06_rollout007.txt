def project_h(h, n):

    target = n / 2.0

        if abs(diff) < 1e-13:

        if diff > 0:

        else:
            mask = h > 0.0

def continuous_opt(h0, n, maxiter=2000):
    h0 = project_h(h0.copy(), n)
    
    def obj(x):
        return compute_c5(project_h(x, n), n)
    
    res = minimize(obj, h0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-7, 'fatol': 1e-9,
                           'initial_simplex_size': 0.05})
    h_out = project_h(res.x, n)
    return h_out, compute_c5(h_out, n)
