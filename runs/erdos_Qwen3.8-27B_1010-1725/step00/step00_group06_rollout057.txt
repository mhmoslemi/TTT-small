import numpy as np
from scipy.optimize import minimize
import time

def corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max(vals, T=0.05):
    m = np.max(vals)
    return T * (m + np.log(np.sum(np.exp((vals - m) / T))))

def objective(h, T=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    return soft_max(corr, T) * dx

def gradient(h, T=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    # Weights for soft max
    shifted = (corr - np.max(corr)) / T
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # grad_f = (dx/T) * sum_k weights[k] * d(corr[k])/dh[i]
    # d(corr[k])/dh[i] = (1-h[i+k])*1[valid] - h[i-k]*1[valid]
    # In terms of the full correlation indexing:
    # corr[k] corresponds to lag k-(n-1) in standard notation
    # corr[k] = sum_j h[j] * (1-h[j + k - (n-1)])
    # Let lag = k - (n-1)
    # d(corr[k])/dh[i] = (1-h[i+lag])*1[0<=i+lag<n] - h[i-lag]*1[0<=i-lag<n]
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        w = weights[k]
        # (1-h[i+lag]) for valid i+lag
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        if i_min < i_max:
            grad[i_min:i_max] += w * (1.0 - h[i_min+lag:i_max+lag])
        # -h[i-lag] for valid i-lag
        i_min2 = max(0, lag)
        i_max2 = min(n, n + lag)
        if i_min2 < i_max2:
            grad[i_min2:i_max2] -= w * h[i_min2-lag:i_max2-lag]
    
    return grad * dx / T

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    diff = target - h.sum()
    n = len(h)
    if abs(diff) < 1e-13:
        return h
    h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            c = np.where(h < 1.0)[0]
        else:
            c = np.where(h > 0.0)[0]
        if len(c) > 0:
            h[c[0]] = np.clip(h[c[0]] + diff2, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, target, T):
    n = len(h)
    h = fix_sum(np.clip(h, 0.0, 1.0), target)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft max
    m = np.max(corr)
    shifted = (corr - m) / T
    exps = np.exp(shifted)
    logsum = m / T + np.log(exps.sum())
    obj = T * logsum * dx
    # Weights
    p = exps / exps.sum()
    # Gradient: d(corr[k])/dh[i] for "full" mode
    # k goes from 0 to 2n-2, lag = k - (n-1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        lag = k - (n - 1)
        pk = p[k]
        if lag >= 0:
            # (1-h[i+lag]) for i in [0, n-1-lag]
            grad[:n-lag] += pk * (1.0 - h[lag:])
            # -h[i-lag] for i in [lag, n-1]
            grad[lag:] -= pk * h[:n-lag]
        else:
            ll = -lag
            # (1-h[i-lag]) = (1-h[i+ll]) for i in [ll, n-1]  -- wait
            # lag < 0, so i+lag = i-ll, valid when i-ll >= 0, i.e., i >= ll
            grad[ll:] += pk * (1.0 - h[:n-ll])
            # -h[i-lag] = -h[i+ll] valid when i+ll < n, i.e., i < n-ll
            grad[:n-ll] -= pk * h[ll:]
    grad *= dx / T
    return obj, grad

    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.asarray(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Nelder-Mead (20% budget)
    t1 = t0 + 0.20 * budget_s
    
    def obj_nm(x):
        return compute_c5(fix_sum(np.clip(x, 0.0, 1.0), target))
    
    for r in range(5):

            break
        x0 = best_h.copy() if r == 0 else fix_sum(np.clip(best_h + rng.randn(n)*0.05, 0, 1), target)
        res = minimize(obj_nm, x0, method='Nelder-Mead',
                       options={'maxiter': 800, 'fatol': 1e-10, 'xatol': 1e-10, 'adaptive': True})
        h_new = fix_sum(np.clip(res.x, 0.0, 1.0), target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
    
    # Phase 2: L-BFGS-B with smooth objective (40% budget)
