    n = len(h)

        if abs(diff) < 1e-14:

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full") * dx
    return float(np.max(c)), c

def smooth_max_val(x, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, "full") * dx
    return float(logsumexp(beta * c) / beta)

def proj_box_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_init, rng, n_iter=5000):
    n = len(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    best = h.copy()
    best_c5, _ = compute_c5(best)
    for it in range(n_iter):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        best[i], best[j] = best[j], best[i]
        c5, _ = compute_c5(best)
        if c5 < best_c5:
            best_c5 = c5
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            best[i], best[j] = best[j], best[i]
    return best, best_c5

def slsqp_optimize(h_start, n, target, bounds, cons, beta, maxiter=300):
    res = minimize(
        lambda x: smooth_max_val(x, beta),
        h_start.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    h_out = proj_box_sum(res.x, target)
    return h_out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_up():
        return (time.time() - t0) >= budget_s * 0.90

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    else:
        best_n = 48
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0

    best_c5, _ = compute_c5(best_h)

    # --- Phase 1: Binary local search for even n values ---
    n_evens = [best_n if best_n % 2 == 0 else best_n + 1]
    for d in [-4, -2, 2, 4, 6, 8, 10]:
        nc = (n_evens[0] + d) if (n_evens[0] + d) % 2 == 0 else (n_evens[0] + d + 1)
        if 24 <= nc <= 120 and nc not in n_evens:
            n_evens.append(nc)

    for n_b in n_evens[:8]:
        if time_up():
            break
        n_half = n_b // 2
        # Try a few random binary starts
        for rep in range(3):
            if time_up():
                break
            h_bin = np.zeros(n_b)
            idx = rng.choice(n_b, n_half, replace=False)
            h_bin[idx] = 1.0
            h_bin, c5_bin = binary_local_search(h_bin, rng, n_iter=3000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_b

    # --- Phase 2: SLSQP continuous optimization ---
    n_candidates = sorted(set([best_n] + [best_n - 1, best_n + 1, best_n - 2, best_n + 2,
                                          best_n - 4, best_n + 4, 48, 50, 52]))
    n_candidates = [n for n in n_candidates if 20 <= n <= 120]

    for n_try in n_candidates:
        if time_up():
            break
        target = n_try / 2.0
        bounds = [(0.0, 1.0)] * n_try
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n_try)}]

        # Initial guess
        if n_try == best_n:
            h_init = best_h.copy()
        else:
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = proj_box_sum(h_init, target)

        # Multi-beta SLSQP
        for beta in [30, 60, 100, 200, 400]:
            if time_up():
                break
            h_out = slsqp_optimize(h_init, n_try, target, bounds, cons, beta, maxiter=400)
            c5_out, _ = compute_c5(h_out)
            if c5_out < best_c5:
                best_c5 = c5_out
                best_h = h_out
                best_n = n_try
                h_init = h_out.copy()
            elif smooth_max_val(h_out, beta) < smooth_max_val(h_init, beta):
                h_init = h_out

        # Random restarts for this n
        for r in range(4):
            if time_up():
                break
            h_r = (rng.random(n_try) < 0.5).astype(float)
            h_r = proj_box_sum(h_r, target)
            for beta in [50, 150]:
                if time_up():
                    break
                h_out = slsqp_optimize(h_r, n_try, target, bounds, cons, beta, maxiter=200)
                c5_out, _ = compute_c5(h_out)
                if c5_out < best_c5:
                    best_c5 = c5_out
                    best_h = h_out
                    best_n = n_try

    # --- Phase 3: Fine-tuning with very high beta ---
    n_f = best_n
    target_f = n_f / 2.0
    bounds_f = [(0.0, 1.0)] * n_f
    cons_f = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_f,
               'jac': lambda x: np.ones(n_f)}]
    for beta in [800, 1500, 3000]:
        if time_up():
            break
        h_out = slsqp_optimize(best_h, n_f, target_f, bounds_f, cons_f, beta, maxiter=500)
        c5_out, _ = compute_c5(h_out)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    # --- Final constraint enforcement ---
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
        else:
            best_h[np.argmin(best_h)] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5, _ = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(best_n)
