import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[s:] += (1.0 - h[:n - s])
            if n > s:
                grad[:n - s] -= h[s:]
    else:
        ns = -s
        if ns < n:
            grad[:n - ns] += (1.0 - h[ns:])
            grad[ns:] -= h[:n - ns]
    return grad * dx


def smooth_objective(h, temp=30.0, pen_weight=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    smooth_max = max_corr + np.log(np.sum(np.exp(temp * shifted))) / temp
    sum_pen = pen_weight * (np.sum(h) - n / 2.0) ** 2
    return smooth_max * dx + sum_pen


def smooth_grad(h, temp=30.0, pen_weight=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    weights = np.exp(temp * shifted)
    weights /= np.sum(weights)
    # Gradient of smooth_max w.r.t. each corr[k] is weights[k]
    # We need d(corr[k])/d(h) for each k, then sum weights[k]*d(corr[k])/d(h)
    # Too expensive for all k. Use only top few.
    top_k = np.argsort(weights)[-20:]
    grad = np.zeros(n)
    for k in top_k:
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += weights[k] * (1.0 - h[:n - s])
                if n > s:
                    grad[:n - s] -= weights[k] * h[s:]
        else:
            ns = -s
            if ns < n:
                grad[:n - ns] += weights[k] * (1.0 - h[ns:])
                grad[ns:] -= weights[k] * h[:n - ns]
    grad *= dx
    # Add penalty gradient
    diff = np.sum(h) - n / 2.0
    grad += 2.0 * pen_weight * diff / n * np.ones(n)
    return grad


def repair_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        frac = (target - s) / len(h)
        h = h + frac
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10 and s > 0:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) / old_n
    x_new = np.arange(new_n) / new_n
    h_new = np.interp(x_new, x_old, h)
    return repair_h(h_new)


def optimize_at_resolution(h, bounds, time_limit, temp=30.0, pen_weight=200.0):
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def obj(x):
        return smooth_objective(x, temp, pen_weight)
    
    def obj_grad(x):
        return smooth_grad(x, temp, pen_weight)
    
    for iteration in range(5):
        if time.time() - start > time_limit - 5:
            break
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 300, 'maxfun': 2000, 'ftol': 1e-14, 'gtol': 1e-10})
            h_try = repair_h(res.x)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            # Increase temperature for tighter max
            temp *= 1.5
        except Exception:
            break
        if time.time() - start > time_limit - 10:
            break
    
    return best_h, best_c5
