from scipy.optimize import minimize
import time

def _corr_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5(h):
    return float(np.max(_corr_values(h)))

def _grad_sub(h, topk=3):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(raw)[-topk:]
    mx = raw[idx[0]]
    wts = np.exp((raw[idx] - mx) / max(mx * 0.1, 1e-9))
    wts /= wts.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, j in zip(wts, idx):
        s = int(j) - (n - 1)
        # vectorized
        # h[m] in C(s): (1-h[m+s]) for valid m+s, and -h[m-s] for valid m-s
        lo, hi = max(0, -s), min(n, n - s)  # range of m where m+s valid
        if lo < hi:
            g[lo:hi] += wi * (1.0 - h[lo+s:hi+s])
        lo2, hi2 = max(0, s), min(n, n + s)  # range of m where m-s valid
        if lo2 < hi2:
            g[lo2:hi2] -= wi * h[lo2-s:hi2-s]
    return g * dx

    h = h.copy()

        d = target - h.sum()

import numpy as np
from scipy.optimize import minimize
import time

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h):
    n = len(h)
    return float(np.max(_corr_raw(h))) * (2.0 / n)

def _c5_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = _corr_raw(h) * dx
    m = np.max(raw)
    if beta < 1e-6:
        return m
    exps = np.exp(beta * (raw - m))
    s = np.sum(exps)
    if s < 1e-300:
        return m
    return m + np.log(s) / beta

def _c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = _corr_raw(h) * dx
    m = np.max(raw)
    exps = np.exp(beta * (raw - m))
    s = np.sum(exps)
    if s < 1e-300:
        wts = np.zeros_like(raw)
        wts[np.argmax(raw)] = 1.0
    else:
        wts = exps / s
    g = np.zeros(n)
    for j in range(len(raw)):
        wi = wts[j]
        if wi < 1e-10:
            continue
        sh = int(j) - (n - 1)
        lo, hi = max(0, -sh), min(n, n - sh)
        if lo < hi:
            g[lo:hi] += wi * (1.0 - h[lo+sh:hi+sh])
        lo2, hi2 = max(0, sh), min(n, n + sh)
        if lo2 < hi2:
            g[lo2:hi2] -= wi * h[lo2-sh:hi2-sh]
    return g * dx

def _project(h, target):
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / len(h)
    return np.clip(h, 0.0, 1.0)

    def obj(x):
        return _c5_smooth(x, beta)

        return _c5_grad(x, beta)

             'jac': lambda x: np.ones(n)}]

    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    h = _project(np.clip(res.x, 0.0, 1.0), target)
    return h, _c5(h)
