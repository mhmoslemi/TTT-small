import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-10:
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(free) > 0:
            h[free[0]] = np.clip(h[free[0]] + diff, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    idx = np.arange(n_new, dtype=float) * n_old / n_new
    lo = np.floor(idx).astype(int)
    hi = np.minimum(lo + 1, n_old - 1)
    fr = idx - lo
    h_new = h[lo] * (1.0 - fr) + h[hi] * fr
    return fix_sum(h_new, n_new / 2.0)
