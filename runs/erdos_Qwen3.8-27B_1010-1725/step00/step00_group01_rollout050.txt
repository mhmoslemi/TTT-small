from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    cross = np.correlate(h, 1 - h, mode='full')
    return float(np.max(cross) * dx)

def objective_and_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    s = tau * (corr - m)

    exp_s = np.exp(s - s_max)

def compute_c5(h):
    dx = 2.0 / len(h)
    cross = np.correlate(h, 1 - h, mode='full')
    return float(np.max(cross) * dx)


def objective_and_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    s = tau * (corr - m)
    s_max = np.max(s)
    exp_s = np.exp(s - s_max)
    log_sum = s_max + np.log(exp_s.sum())
    obj = m + log_sum / tau
    weights = exp_s / exp_s.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        j1 = np.arange(n) - k + n - 1
        j2 = np.arange(n) + k - n + 1
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        t1 = np.zeros(n)
        t1[v1] = 1.0 - h[j1[v1]]
        t2 = np.zeros(n)
        t2[v2] = -h[j2[v2]]
        grad += wk * dx * (t1 + t2)
    return obj, grad
