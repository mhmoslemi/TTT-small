            idx = rng.choice(n, min(k, n), replace=False)
            h_init[idx] = 1.0 - h_init[idx]
        h_init = project_feasible(h_init, n)

        # Quick SLSQP
        hc, c5 = try_slsqp(h_init, n, dx, target, cons, beta=None, maxiter=150, ftol=1e-10)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
            # Refine with coordinate descent
            hc2, c52 = coordinate_descent(best_h, n, dx, target, rng, n_iter=100)
            if c52 < best_c5:
                best_h, best_c5 = hc2, c52

    # Phase 5: If time remains, try different n_points
    if time_left() > 60:
        for n2 in [n + 5, n + 10, n + 15]:
            if time_left() < 30:
                break
            dx2 = 2.0 / n2
            target2 = n2 / 2.0
            cons2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target2}]
            h2_init = np.interp(np.linspace(0, 2, n2, endpoint=False),
                                np.linspace(0, 2, n, endpoint=False), best_h)
            h2_init = project_feasible(h2_init, n2)
            hc2, c52 = try_slsqp(h2_init, n2, dx2, target2, cons2, beta=None, maxiter=300)
            if c52 < best_c5:
                best_h, best_c5, n = hc2, c52, n2
                dx, target, cons = dx2, target2, cons2

    # Final verification
    best_c5 = compute_c5(best_h, n)
    return (best_h, best_c5, n)
