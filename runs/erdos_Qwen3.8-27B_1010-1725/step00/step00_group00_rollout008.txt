import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def c5_smoothed(h, n_points, beta):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    # log-sum-exp for numerical stability
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(corr - max_c))) / beta

def project(h, n_points):
    """Project to [0,1] with sum = n_points/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n_points / 2.0
    if abs(s - target) < 1e-10:
        return h
    # Scale and re-clip iteratively
    for _ in range(100):
        h = np.clip(h * target / s, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        s = np.sum(h)
    return h

def optimize_h(h0, n_points, max_iter=1000, beta=50, time_limit=60):
    """Optimize h using SLSQP with smoothed objective"""
    start_time = time.time()
    
    def objective(h):
        return c5_smoothed(h, n_points, beta)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1)] * n_points
    
    try:
        result = minimize(
            objective, h0, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = project(result.x, n_points)
        else:
            h_opt = h0
    except:
        h_opt = h0
    
    # Refine with projected gradient
    h_opt = projected_gradient(h_opt, n_points, time_limit - (time.time() - start_time))
    return h_opt

def projected_gradient(h, n_points, time_limit=30):
    start_time = time.time()
    dx = 2.0 / n_points
    target = n_points / 2.0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n_points)
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr, n_points)
    
    for iteration in range(10000):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h_curr, 1 - h_curr, mode='full')

        max_idx = np.argmax(corr)

        # ∂c_k/∂h[j] = (1 - h[j+k]) - h[j-k] (for valid indices)

            if 0 <= j + k < n_points:
                g += (1 - h_curr[j + k])
            if 0 <= j - k < n_points:

            grad[j] = g * dx

def project_h(h, n):
    target = n / 2.0
    lo, hi = -max(h.max(), 1) - 1, 1 - min(h.min(), 0) + 1

        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < 1e-12:

    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, n):
    target = n / 2.0
    lo = -max(float(h.max()), 0.5) - 1.0
    hi = 1.0 - min(float(h.min()), 0.5) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < 1e-13:
            break
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp((corr - m) * beta)
    s = np.sum(exps)
    weights = exps / s
    smoothed_val = m + np.log(s) / beta
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        L = idx - (n - 1)
        if L >= 0:
            if n - L > 0:
                grad[:n - L] += w * dx * (1.0 - h[L:])
                grad[L:] += w * dx * (-h[:n - L])
        else:
            Lp = -L
            if n - Lp > 0:
                grad[Lp:] += w * dx * (1.0 - h[:n - Lp])
                grad[:n - Lp] += w * dx * (-h[Lp:])
    return smoothed_val, grad

def opt_momentum(h0, n, time_limit, beta_init=15.0, beta_max=120.0):
    start = time.time()
    h = project_h(h0.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h)
    v = np.zeros(n)
    lr = 0.05
    mu = 0.85
    beta = beta_init
    patience = 0
    for it in range(20000):
        if time.time() - start > time_limit - 1:
            break
        val, grad = smooth_grad(h, beta)
        v = mu * v + grad
        h = project_h(h - lr * v, n)
        actual_val = compute_c5(h)
        if actual_val < best_val:
            best_val = actual_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience == 200:
            lr *= 0.8
            mu = 0.9
            patience = 0
            if lr < 1e-5:
                break
        if patience % 500 == 0:
            beta = min(beta * 1.1, beta_max)
    return best_h, best_val

def opt_slsqp(h0, n, time_limit=60):
    start = time.time()
    h0 = project_h(h0.copy(), n)
    beta = 40.0

    def obj(h):
        return smooth_grad(h, beta)[0]

    def obj_grad(h):
        return smooth_grad(h, beta)[1]

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-9})
        if time.time() - start < time_limit and res.x is not None:
            h_opt = project_h(res.x, n)
        else:
            h_opt = h0
    except Exception:
        h_opt = h0
    return h_opt

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xs_old = np.linspace(0, 1, n_old, endpoint=False)
    xs_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(xs_new, xs_old, h)

def random_feasible(n, rng):
    h = rng.random(n)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - t_start

    best_h = None
    best_c5 = 1e9
    best_n = 0

    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = 0.5 * np.ones(45)
    n_init = len(h_init)
    h_init = project_h(h_init, n_init)
    c0 = compute_c5(h_init)
    update(h_init, c0, n_init)

    # Phase 1: SLSQP at initial resolution
    if elapsed() < budget_s - 200:
        h_s = opt_slsqp(h_init, n_init, time_limit=min(60, budget_s - elapsed() - 100))
        c1 = compute_c5(h_s)
        update(h_s, c1, n_init)

    # Phase 2: Momentum at initial resolution
    if elapsed() < budget_s - 200:
        tl = min(80, budget_s - elapsed() - 100)
        h_m, cm = opt_momentum(h_s, n_init, time_limit=tl)
        update(h_m, cm, n_init)
        # Random restarts at same resolution
        for r in range(5):
            if elapsed() > budget_s - 150:
                break
            hr = project_h(h_s + 0.1 * rng.standard_normal(n_init), n_init)
            h_m2, cm2 = opt_momentum(hr, n_init, time_limit=min(40, budget_s - elapsed() - 100))
            update(h_m2, cm2, n_init)

    # Phase 3: Progressive refinement
    for factor in [2, 4, 8]:
        n_new = n_init * factor
        if n_new > 720:
            break
        if elapsed() > budget_s - 120:
            break
        h_interp = project_h(interp_h(best_h, n_new), n_new)
        c_interp = compute_c5(h_interp)
        update(h_interp, c_interp, n_new)

        n_restarts = 4 if n_new <= 180 else 2
        for r in range(n_restarts):
            if elapsed() > budget_s - 60:
                break
            if r == 0:
                h_start = h_interp
            elif r == 1:
                h_start = project_h(h_interp + 0.05 * rng.standard_normal(n_new), n_new)
            else:
                h_start = random_feasible(n_new, rng)
            tl = min(50, budget_s - elapsed() - 40)
            if tl < 10:
                break
            h_opt, c_opt = opt_momentum(h_start, n_new, time_limit=tl)
            update(h_opt, c_opt, n_new)

        # SLSQP at this resolution if small enough
        if n_new <= 90 and elapsed() < budget_s - 100:
            h_s2 = opt_slsqp(best_h if len(best_h) == n_new else h_interp, n_new,
                             time_limit=min(40, budget_s - elapsed() - 50))
            c_s2 = compute_c5(h_s2)
            update(h_s2, c_s2, n_new)

    # Phase 4: Final polishing at best resolution
    if elapsed() < budget_s - 20 and best_h is not None:
        tl = max(5, budget_s - elapsed() - 5)
        h_pol, c_pol = opt_momentum(best_h, best_n, time_limit=tl,
                                    beta_init=60.0, beta_max=200.0)
        update(h_pol, c_pol, best_n)

    # Final validation
    h_final = project_h(best_h, best_n)
    c5_final = compute_c5(h_final)
    assert np.all(h_final >= -1e-12) and np.all(h_final <= 1.0 + 1e-12)
    assert abs(np.sum(h_final) - best_n / 2.0) < 1e-6
    return (h_final, c5_final, best_n)
