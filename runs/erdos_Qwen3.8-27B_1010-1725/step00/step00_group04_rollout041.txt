    """Smooth max objective."""
    g = 1 - h
    corr = np.correlate(h, g, mode='full') * dx

    """Project h onto {0<=h_i<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def _objgrad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    mc = np.max(corr)
    sh = t * (corr - mc)
    ev = np.exp(sh)
    s = ev.sum()
    obj = mc + np.log(s) / t
    w = ev / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        lo1 = max(0, n - 1 - i)
        hi1 = min(n - 1, 2 * n - 2 - i)
        if lo1 <= hi1:
            idx1 = np.arange(lo1, hi1 + 1) + i - n + 1
            grad[lo1:hi1 + 1] += wi * dx * (1.0 - h[idx1])
        lo2 = max(0, i - n + 1)
        hi2 = min(n - 1, i)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1) - i + n - 1
            grad[lo2:hi2 + 1] -= wi * dx * h[idx2]
    return obj, grad

def _proj(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            if not m.any():
                break
            h[m] += min(d / m.sum(), (1.0 - h[m]).min())
        else:
            m = h > 0.0
            if not m.any():
                break
            h[m] -= min(-d / m.sum(), h[m].min())
    return h

def _scipy_opt(h0, t, mxit=300, ftol=1e-12):
    n = len(h0)
    bd = [(0.0, 1.0)] * n
    ct = [
        {'type': 'eq',
         'fun': lambda x: x.sum() - n * 0.5,
         'jac': lambda x: np.ones(n)}
    ]
    res = minimize(
        lambda x: _objgrad(x, t)[0],
        h0.copy(),
        jac=lambda x: _objgrad(x, t)[1],
        method='SLSQP',
        bounds=bd,
        constraints=ct,
        options={'maxiter': mxit, 'ftol': ftol}
    )
    return res.x

def _coord_descent(h, n_iter=50):
    n = len(h)
    dx = 2.0 / n
    tgt = n / 2.0
    best_c5 = _eval(h)
    for _ in range(n_iter):
        improved = False
        for j in range(n):
            old = h[j]
            # Try pushing h[j] up or down, adjusting one other coordinate
            for direction in [0.01, -0.01, 0.05, -0.05]:
                h_try = h.copy()
                h_try[j] = old + direction
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                # Adjust another coordinate to maintain sum
                rem = tgt - h_try.sum()
                if abs(rem) < 1e-12:
                    l = j
                else:
                    # Find best partner
                    candidates = np.where(np.ones(n, dtype=bool))[0]
                    candidates = candidates[candidates != j]
                    for l in candidates:
                        h_try[l] = h[l] - rem
                        if 0 <= h_try[l] <= 1:
                            break
                    else:
                        continue
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                c5_new = _eval(h_try)
                if c5_new < best_c5 - 1e-9:
                    h = h_try.copy()
                    best_c5 = c5_new
                    improved = True
                    l_restore = l
                    break
            h[j] = old
        if not improved:
            break
    return h, best_c5

    best_v = _eval(best_h)
    n_final = len(best_h)

    return best_h, best_v, n_final
