def project_feasible(h, n):
    """Project h onto {x: 0≤x≤1, sum(x)=n/2}"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    diff = n / 2.0 - np.sum(h)
    # Simple: add uniform correction, then clip
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Iterate a few times for better projection
    for _ in range(10):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Adjust non-bound variables
        free = (h > 1e-8) & (h < 1 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
        else:
            break

        k_star = int(np.argmax(corr))
        c_star = corr[k_star]

def get_subgrad(h, n, shift):
    """Subgradient of corr at given shift w.r.t. h"""
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        # (1-h[i-s]) for s <= i <= n-1
        idx1 = np.arange(s, n)
        grad[idx1] += (1.0 - h[idx1 - s])
        # -h[i+s] for 0 <= i <= n-1-s
        idx2 = np.arange(0, n - s)
        grad[idx2] -= h[idx2 + s]
    else:
        t = -shift
        # (1-h[i+t]) for 0 <= i <= n-1-t
        idx1 = np.arange(0, n - t)
        grad[idx1] += (1.0 - h[idx1 + t])
        # -h[i-t] for t <= i <= n-1
        idx2 = np.arange(t, n)
        grad[idx2] -= h[idx2 - t]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = np.sum(free)
        if nf == 0:
            # Add to zeros or subtract from ones
            if diff > 0:
                zeros = (h < 0.5)
                h[zeros] += diff / max(1, np.sum(zeros))
            else:
                ones = (h > 0.5)
                h[ones] -= (-diff) / max(1, np.sum(ones))
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_descent(h_init, n, max_steps=20000, seed=42, time_limit=60):
    rng = np.random.default_rng(seed)
    h = h_init.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    lr_initial = 0.3
    lr_final = 0.001
    
    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = int(np.argmax(corr))
        shift = m - (n - 1)
        
        grad = get_subgrad(h, n, shift)
        
        # Adaptive step size
        progress = step / max(1, max_steps - 1)
        lr = lr_initial * (lr_final / lr_initial) ** progress
        
        h_new = h - lr * grad * dx
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Accept with some probability for stochasticity
        if c5_new <= compute_c5(h) + 1e-8 or rng.random() < np.exp(-(c5_new - best_c5) * 100):
            h = h_new
    
    return best_h, best_c5

def slsqp_solve(h_init, n, maxiter=300):
    dx = 2.0 / n
    
    def obj(v):
        return v[n]
    
    def constr_ineq(v):
        h = v[:n]
        t = v[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx
    
    def constr_eq(v):
        return np.sum(v[:n]) - n / 2.0
    
    cons = [{'type': 'ineq', 'fun': constr_ineq},
            {'type': 'eq', 'fun': constr_eq}]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    x0 = np.concatenate([h_init, [compute_c5(h_init) + 0.05]])
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if res.fun < compute_c5(h_init):
        return res.x[:n]
    return h_init

def local_search(h_init, n, max_iter=100000, time_limit=60, seed=42):
    rng = np.random.default_rng(seed)
    h = h_init.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    t0 = time.time()
    lr = 0.1
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = rng.uniform(-lr, lr)
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        new_c5 = compute_c5(h)
        if new_c5 <= c5 + 1e-12:
            c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
        if it % 5000 == 0:
            lr *= 0.8
    
    return best_h, best_c5

def interp_to_n(h, new_n):
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) / old_n * 2.0
    new_x = (np.arange(new_n) + 0.5) / new_n * 2.0
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def structured_init(n, seed=0):
    """Generate structured initial guesses"""
    rng = np.random.default_rng(seed)
    candidates = []
    # Sine waves
    for p in [1, 2, 3, 4, 5, 7]:
        for a in [0.3, 0.5, 0.7, 1.0]:
            phi = rng.uniform(0, 2*np.pi)
            h = 0.5 + a * np.cos(2*np.pi*p*np.arange(n)/n + phi)
            h = project_feasible(np.clip(h, 0, 1), n)
            candidates.append(h)
    # Block patterns
    for block in [2, 3, 4, 5, 7]:

        for i in range(n):
            if (i // block) % 2 == 0:
                h[i] = 1.0
        h = project_feasible(h, n)
        candidates.append(h)
    # Random
    candidates.append(rng.uniform(0, 1, n))
    candidates[-1] = project_feasible(candidates[-1], n)
    return candidates
