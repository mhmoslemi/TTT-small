def projected_grad_descent(h, n_points, target_sum, max_iter=8000, time_limit=None, t0=None):
    best = compute_c5(h, n_points)
    h_cur = h.copy()
    lr = 0.05
    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        _, grad = c5_grad_smooth(h_cur, n_points, 5)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, n_points)
        if c5_new < best:
            best = c5_new
            h_cur = h_new.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.85
            if lr < 1e-5:
                break
    return h_cur, best

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _TS
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n_points = len(h_init)
    _N = n_points
    _TS = n_points / 2.0
    target_sum = _TS

    h0 = make_feasible(h_init, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n_points)

    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n_points

    # Phase 1: SLSQP from multiple starts
    starts = [h0.copy()]
    for i in range(6):
        sigma = 0.03 * (i + 1)
        h_p = h0 + np.random.randn(n_points) * sigma
        h_p = make_feasible(h_p, target_sum)
        starts.append(h_p)
    # Binary-ish start
    h_bin = (h0 > 0.5).astype(float)
    h_bin = make_feasible(h_bin, target_sum)
    starts.append(h_bin)
    # Shifted version
    h_shift = np.roll(h0, 1)
    h_shift = make_feasible(h_shift, target_sum)
    starts.append(h_shift)

    for h_start in starts:
        if time.time() - t0 > budget_s * 0.55:
            break
        try:
            res = minimize(_obj, h_start, jac=_grad_single,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 400, 'ftol': 1e-13, 'disp': False})
            if res.fun < best_c5 - 1e-9:
                candidate = np.clip(res.x, 0, 1)
                candidate = make_feasible(candidate, target_sum)
                c5_val = compute_c5(candidate, n_points)
                if c5_val < best_c5:
                    best_h = candidate
                    best_c5 = c5_val
        except Exception:
            pass

    # Phase 2: SLSQP with smooth gradient from best
    if time.time() - t0 < budget_s * 0.7:
        try:
            res = minimize(_obj, best_h, jac=_grad_smooth5,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-14, 'disp': False})
            if res.fun < best_c5 - 1e-9:
                candidate = np.clip(res.x, 0, 1)
                candidate = make_feasible(candidate, target_sum)
                c5_val = compute_c5(candidate, n_points)
                if c5_val < best_c5:
                    best_h = candidate
                    best_c5 = c5_val
        except Exception:
            pass

    # Phase 3: Projected gradient descent
    if time.time() - t0 < budget_s * 0.85:
        h_pg, c5_pg = projected_grad_descent(
            best_h, n_points, target_sum,
            max_iter=10000, time_limit=budget_s * 0.85, t0=t0)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.95:
        try:
            res = minimize(_obj, best_h, jac=_grad_single,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-15, 'disp': False})
            if res.fun < best_c5 - 1e-9:
                candidate = np.clip(res.x, 0, 1)
                candidate = make_feasible(candidate, target_sum)
                c5_val = compute_c5(candidate, n_points)
                if c5_val < best_c5:
                    best_h = candidate
                    best_c5 = c5_val
        except Exception:
            pass

    # Phase 5: Another smooth SLSQP
    if time.time() - t0 < budget_s * 0.98:
        try:
            res = minimize(_obj, best_h, jac=_grad_smooth3,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 600, 'ftol': 1e-15, 'disp': False})
            if res.fun < best_c5 - 1e-9:
                candidate = np.clip(res.x, 0, 1)
                candidate = make_feasible(candidate, target_sum)
                c5_val = compute_c5(candidate, n_points)
                if c5_val < best_c5:
                    best_h = candidate
                    best_c5 = c5_val
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h, n_points)

    return best_h, final_c5, n_points
