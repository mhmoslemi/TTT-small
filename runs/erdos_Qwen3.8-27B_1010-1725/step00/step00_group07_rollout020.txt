def block_swap(h, rng, L=3):
    n = len(h)
    if n < 2 * L:
        return None
    p = int(rng.integers(0, n - L))
    q = int(rng.integers(0, n - L))
    if abs(p - q) < L:
        return None
    # Swap blocks [p:p+L] and [q:q+L]
    # Only swap 1s and 0s (maintain balance)
    block1 = h[p:p+L].copy()
    block2 = h[q:q+L].copy()
    # Check balance: sum(block1) should equal sum(block2) for binary
    if block1.sum() != block2.sum():
        return None
    return p, q, block1, block2
