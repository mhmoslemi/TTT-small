        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    result = np.clip(h - mid, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = initial_h_values.astype(float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = project_constraint(h0, n)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    def remaining():
        return budget_s - (time.time() - t0)

    bounds = [(0.0, 1.0)] * n
    eq_cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with log-sum-exp smoothing at various sharpness levels
    for s_val in [20, 50, 100, 250, 500]:
        if remaining() < 15:
            break

        def obj_smooth(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            return smooth_max(corr, s_val)

        try:
            res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                           constraints=eq_cons,
                           options={'maxiter': 800, 'ftol': 1e-14})
            if res.x is not None:
                c = project_constraint(res.x, n)
                c5 = compute_c5(c, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = c.copy()
                    h = c.copy()
        except Exception:
            pass

    # Phase 2: Minimize sum of top correlations to flatten the peak
    if remaining() > 40:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = min(8, len(corr))

        def obj_top(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            top = np.sort(c)[::-1][:top_k]
            return top[0] + 0.3 * np.mean(top)

        try:
            res = minimize(obj_top, h, method='SLSQP', bounds=bounds,
                           constraints=eq_cons,
                           options={'maxiter': 600, 'ftol': 1e-14})
            if res.x is not None:
                c = project_constraint(res.x, n)
                c5 = compute_c5(c, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = c.copy()
                    h = c.copy()
        except Exception:
            pass

    # Phase 3: Coordinate descent (paired swaps to preserve sum)
    if remaining() > 20:
        for outer in range(200):
            if remaining() < 5:
                break
            improved = False
            step = 0.02 if outer < 50 else (0.01 if outer < 100 else 0.005)
            for i in range(n):
                for j in range(n):
                    if i == j:
                        continue
                    for delta in [step, -step]:
                        hi_val = h[i] + delta
                        hj_val = h[j] - delta
                        if hi_val < -1e-10 or hi_val > 1.0 + 1e-10:
                            continue
                        if hj_val < -1e-10 or hj_val > 1.0 + 1e-10:
                            continue
                        h[i] = np.clip(hi_val, 0, 1)
                        h[j] = np.clip(hj_val, 0, 1)
                        c5 = compute_c5(h, dx)
                        if c5 < best_c5:
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                        if improved:
                            best_c5 = c5
                            best_h = h.copy()
                            break
                if improved:
                    break
            if not improved:
                break

    # Phase 4: Random restarts from perturbed best
    if remaining() > 30:
        for trial in range(8):
            if remaining() < 20:
                break
            scale = 0.03 * (1.0 - trial / 10.0)
            h_pert = h + np.random.randn(n) * scale
            h_pert = project_constraint(h_pert, n)

            def obj_restart(x):
                corr = np.correlate(x, 1.0 - x, mode='full') * dx
                return smooth_max(corr, 150)

            try:
                res = minimize(obj_restart, h_pert, method='SLSQP', bounds=bounds,
                               constraints=eq_cons,
                               options={'maxiter': 400, 'ftol': 1e-13})
                if res.x is not None:
                    c = project_constraint(res.x, n)
                    c5 = compute_c5(c, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = c.copy()
                        h = c.copy()
            except Exception:
                pass

    # Phase 5: Final polish with actual (non-smooth) max
    if remaining() > 10:
        def obj_actual(x):
            return compute_c5(x, dx)
        try:
            res = minimize(obj_actual, h, method='SLSQP', bounds=bounds,
                           constraints=eq_cons,
                           options={'maxiter': 400, 'ftol': 1e-15})
            if res.x is not None:
                c = project_constraint(res.x, n)
                c5 = compute_c5(c, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = c.copy()
        except Exception:
            pass

    # Verify and return
    best_h = project_constraint(best_h, n)
    final_c5 = compute_c5(best_h, dx)

    # Also check initial in case we degraded
    init_c5 = compute_c5(project_constraint(h0, n), dx)
    if init_c5 < final_c5:
        best_h = project_constraint(h0, n)
        final_c5 = init_c5

    return best_h, final_c5, n
