from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), int(np.argmax(corr))

def repair_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0, 1)
    for _ in range(100):
        err = target - h.sum()
        if abs(err) < 1e-12:
            break
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += err / cnt
        h = np.clip(h, 0, 1)
    return h

def gradient_step(h, alpha, seed=0):
    """One subgradient step for the max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    tau = k_max - (n - 1)  # the lag in grid units
    
    # Subgradient for the correlation at lag tau
    # corr_k = sum_i h[i] * (1-h[i+tau]) for valid i
    # grad h[i] = (1-h[i+tau]) - h[i-tau] (with boundary handling)
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears as first factor: pair with h[i+tau]
        j = i + tau
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
        # h[i] appears as second factor (1-h[i]) in pair h[i-tau]*(1-h[i])
        j2 = i - tau
        if 0 <= j2 < n:
            grad[i] -= h[j2]
    
    # Projected gradient descent
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = repair_sum(h_new, n / 2.0)
    return h_new

            eps = max_pos * rng.uniform(0.05, 1.0)
            h[i] += eps
            h[j] -= eps
        elif max_neg > 1e-12:
            eps = max_neg * rng.uniform(0.05, 1.0)
            h[i] -= eps
            h[j] += eps
        else:
            continue
        new_c5 = compute_c5(h)
        if new_c5 < cur:
            cur = new_c5
        else:
            h[i] -= eps
            h[j] += eps
    return h, cur

def make_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.where(rng.random(n) < 0.5, 1.0, 0.0)
    target = n / 2.0
    cur_sum = h.sum()
    diff = cur_sum - target
    k = int(abs(diff))
    if diff > 0:
        ones = np.where(h == 1.0)[0]
        for idx in rng.choice(ones, min(k, len(ones)), replace=False):
            h[idx] = 0.0
    else:
        zeros = np.where(h == 0.0)[0]
        for idx in rng.choice(zeros, min(k, len(zeros)), replace=False):
            h[idx] = 1.0
    rem = target - h.sum()
    if abs(rem) > 1e-10:
        if rem > 0:
            cands = np.where(h < 1.0)[0]
        else:
            cands = np.where(h > 0.0)[0]
        if len(cands) > 0:
            h[rng.choice(cands)] += rem
    return repair_sum(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline_main = t0 + budget_s - 20

    try:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError, TypeError):
        h_parent = None

    best_h = None
    best_c5 = float('inf')

    if h_parent is not None:
        h_p = repair_sum(h_parent, len(h_parent) / 2.0)
        best_h = h_p.copy()
        best_c5 = compute_c5(best_h)

    n_parent = len(h_parent) if h_parent is not None else 47
    res_list = [n_parent]
    for n in [40, 44, 48, 52, 56, 64, 80]:
        if n != n_parent and n not in res_list:
            res_list.append(n)

    time_per_res = (budget_s - 40) / len(res_list)

    for idx, n_res in enumerate(res_list):
        if time.time() > deadline_main:
            break
        res_deadline = min(deadline_main, time.time() + time_per_res)

        if h_parent is not None and n_res == n_parent:
            h = repair_sum(h_parent, n_res / 2.0)
        elif h_parent is not None:
            h = interp_h(h_parent, n_res)
        else:
            h = make_binary(n_res, seed=seed)

        # Main optimization: multiple passes with perturbation
        h, c5 = focused_local_search(h, 800000, seed=seed + idx * 7, deadline=res_deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Perturbation + reoptimize cycles
        n_cycles = max(3, int((res_deadline - time.time()) / 5.0))
        for cyc in range(n_cycles):
            if time.time() > res_deadline:
                break
            h_try = best_h.copy() if best_h is not None and len(best_h) == n_res else h.copy()
            # Small random perturbation
            for _ in range(n_res):
                a = rng.randint(n_res)
                b = rng.randint(n_res)
                if a != b:
                    e = rng.uniform(-0.03, 0.03)
                    h_try[a] = np.clip(h_try[a] + e, 0.0, 1.0)
                    h_try[b] = np.clip(h_try[b] - e, 0.0, 1.0)
            h_try = repair_sum(h_try, n_res / 2.0)
            h_try, c5_try = focused_local_search(
                h_try, 200000, seed=seed + idx * 7 + cyc * 13,
                deadline=res_deadline)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            if c5_try < c5:
                h = h_try

    # Final polish on the global best
    if best_h is not None and time.time() < deadline_main:
        n = len(best_h)
        polish_deadline = t0 + budget_s - 5
        for trial in range(15):
            if time.time() > polish_deadline:
                break
            h_try = best_h.copy()
            for _ in range(n // 3):
                a = rng.randint(n)
                b = rng.randint(n)
                if a != b:
                    e = rng.uniform(-0.02, 0.02)
                    h_try[a] = np.clip(h_try[a] + e, 0.0, 1.0)
                    h_try[b] = np.clip(h_try[b] - e, 0.0, 1.0)
            h_try = repair_sum(h_try, n / 2.0)
            h_try, c5_try = focused_local_search(
                h_try, 100000, seed=seed + 5000 + trial, deadline=polish_deadline)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    if best_h is not None:
        best_h = repair_sum(best_h, len(best_h) / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(len(best_h))
