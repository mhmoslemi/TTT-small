import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_corr = np.exp(corr - m)
    w = exp_corr / np.sum(exp_corr)
    obj = m + np.log(np.sum(exp_corr)) / tau
    
    # Gradient
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wk = w[idx] * dx
        for j in range(n):
            if k >= 0:
                val = 0.0
                if j >= k:
                    val += (1.0 - h[j - k])
                if j < n - k:
                    val -= h[j + k]
            else:
                kp = -k
                val = 0.0
                if j < n - kp:
                    val += (1.0 - h[j + kp])
                if j >= kp:
                    val -= h[j - kp]
            grad[j] += wk * val
    return obj, grad

def project_to_feasible(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def optimize_slsqp(h0, n, tau, max_iter=100):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def fun(h):
        return smooth_obj_and_grad(h, tau)[0]
    
    def fun_and_grad(h):
        o, g = smooth_obj_and_grad(h, tau)
        return o, g
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': lambda h: np.ones(n)}]
    
    res = minimize(fun_and_grad, h0, method='SLSQP', jac=True,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_to_feasible(h_new, target_sum)
    return h_new

def optimize_subgrad(h0, n, nr_iter=2000):

    dx = 2.0 / n
    h = h0.copy()

    best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        
        # Subgradient of the max component
        grad = np.zeros(n)
        if k >= 0:
            for j in range(n):
                val = 0.0
                if j >= k:
                    val += (1.0 - h[j - k])
                if j < n - k:
                    val -= h[j + k]
                grad[j] = val * dx
        else:
            kp = -k
            for j in range(n):
                val = 0.0
                if j < n - kp:
                    val += (1.0 - h[j + kp])
                if j >= kp:
                    val -= h[j - kp]
                grad[j] = val * dx
        
        h = h - lr * it / (it + 50) * grad  # diminishing step size
        h = project_to_feasible(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_corr = np.exp(corr - m)
    w = exp_corr / np.sum(exp_corr)
    obj = m + np.log(np.sum(exp_corr)) / tau
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        wk_dx = wi * dx
        if k >= 0:
            for j in range(n):
                val = 0.0
                if j >= k:
                    val += 1.0 - h[j - k]
                if j < n - k:
                    val -= h[j + k]
                grad[j] += wk_dx * val
        else:
            kp = -k
            for j in range(n):
                val = 0.0
                if j < n - kp:
                    val += 1.0 - h[j + kp]
                if j >= kp:
                    val -= h[j - kp]
                grad[j] += wk_dx * val
    return obj, grad


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def slsqp_opt(h0, n, tau, max_iter=100):
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ts, 'jac': lambda h: np.ones(n)}]

    def fag(h):
        return smooth_obj_and_grad(h, tau)

    res = minimize(fag, h0, method='SLSQP', jac=True, bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    hx = np.clip(res.x, 0.0, 1.0)
    return project_feasible(hx, ts)


def subgrad_opt(h0, n, niter=5000):
    ts = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best = compute_c5(h)
    best_h = h.copy()
    for it in range(niter):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        mi = int(np.argmax(corr))
        k = mi - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for j in range(n):
                v = 0.0
                if j >= k:
                    v += 1.0 - h[j - k]
                if j < n - k:
                    v -= h[j + k]
                grad[j] = v * dx
        else:
            kp = -k
            for j in range(n):
                v = 0.0
                if j < n - kp:
                    v += 1.0 - h[j + kp]
                if j >= kp:
                    v -= h[j - kp]
                grad[j] = v * dx
        lr = 0.02 * n / (it + 50.0)
        h = project_feasible(h - lr * grad, ts)
        c5v = compute_c5(h)
        if c5v < best:
            best = c5v
            best_h = h.copy()
    return best_h, best


def local_search(h0, n, niter, rng):
    ts = n / 2.0
    h = h0.copy()
    bcv = compute_c5(h)
    bh = h.copy()
    for _ in range(niter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        h[i] += d
        h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c5v = compute_c5(h)
            if c5v < bcv:
                bcv = c5v
                bh = h.copy()
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return bh, bcv


def upsample(h_src, n_new):
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_src, h_src, period=2.0)
    return project_feasible(h_new, n_new / 2.0)
