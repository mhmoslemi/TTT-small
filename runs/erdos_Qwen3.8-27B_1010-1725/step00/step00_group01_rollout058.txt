def project_box_sum(h, target_sum):
    h = h.astype(float).copy()
    # Bisection on threshold to project onto {0<=x<=1, sum=target_sum}
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def get_shift_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j - s < n:
            grad[j] += 1.0 - h[j - s]
        if 0 <= j + s < n:
            grad[j] -= h[j + s]
    return grad

def smooth_obj_grad(h, beta):

    dx = 2.0 / n

import numpy as np
import time
from scipy.optimize import minimize as sp_minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def project_box_sum(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def shift_grad(h, s):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        gj = 0.0
        if 0 <= j - s < n:
            gj += 1.0 - h[j - s]
        if 0 <= j + s < n:
            gj -= h[j + s]
        grad[j] = gj
    return grad

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = corr.max()
    exps = np.exp(beta * (corr - cmax))
    Z = exps.sum()
    obj = float(cmax + np.log(Z) / beta)
    p = exps / Z
    grad = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-10:
            continue
        s = k - (n - 1)
        grad += p[k] * shift_grad(h, s) * dx
    return obj, grad

def upsample_h(h_parent, n_target):
    n0 = len(h_parent)
    if n0 == n_target:
        return h_parent.astype(float).copy()
    x_old = np.arange(n0, dtype=float) / n0
    x_new = np.arange(n_target, dtype=float) / n_target
    return np.interp(x_new, x_old, h_parent)

def random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rest = [i for i in range(n) if i not in idx]
        h[rest[rng.integers(len(rest))]] = 0.5
    return h

def sine_init(n, rng):
    phase = rng.uniform(0, 6.2832)
    x = np.arange(n, dtype=float) / n
    h = 0.5 + 0.35 * np.sin(4.0 * 3.14159 * x + phase) + 0.15 * np.sin(8.0 * 3.14159 * x - phase)
    return project_box_sum(h, n / 2.0)

def optimize_smoothed(h0, n, t_end, betas, lr_init, max_inner):
    h = project_box_sum(h0.astype(float).copy(), n / 2.0)
    c5 = compute_c5(h)
    lr = lr_init
    for beta in betas:
        stuck = 0
        for it in range(max_inner):
            if time.time() > t_end - 1.0:
                return h, compute_c5(h)
            obj, grad = smooth_obj_grad(h, beta)
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                lr = min(lr * 1.04, 0.4)
                stuck = 0
            else:
                stuck += 1
                lr *= 0.88
                if lr < 1e-9 or stuck > 80:
                    break
    return h, compute_c5(h)

def local_search(h0, n, t_end, rng):
    h = h0.astype(float).copy()
    c5 = compute_c5(h)
    for _ in range(8000):
        if time.time() > t_end - 0.5:
            break
        n_p = max(1, n // 6)
        idx = rng.choice(n, size=n_p, replace=False)
        h_try = h.copy()
        h_try[idx] += rng.uniform(-0.25, 0.25, size=n_p)
        h_try = project_box_sum(h_try, n / 2.0)
        c5_try = compute_c5(h_try)
        if c5_try < c5:
            h = h_try
            c5 = c5_try
    return h, c5

def scipy_attempt(h0, n, t_end, beta):
    h = project_box_sum(h0.astype(float).copy(), n / 2.0)
    c5_base = compute_c5(h)
    def obj_grad(x):
        hx = np.clip(x, 0.0, 1.0)
        o, g = smooth_obj_grad(hx, beta)
        return o, g
    def obj_only(x):
        return obj_grad(x)[0]
    def grad_only(x):
        return obj_grad(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    if time.time() > t_end - 5:
        return h, c5_base
    try:

                          bounds=bnds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-9})
        if time.time() > t_end - 1:
            return h, c5_base
        h_new = project_box_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5_base:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5_base
