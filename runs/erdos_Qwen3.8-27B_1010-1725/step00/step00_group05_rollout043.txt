def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        t = room.sum()
        if t > 0:
            h += diff * room / t
    else:
        t = h.sum()
        if t > 0:
            h += diff * h / t
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        for i in range(n):
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    elif diff < -1e-14:
        for i in range(n):
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-14:
                break
    return h

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

            h[j] -= d
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_d = d
        h[i] += best_d
        h[j] -= best_d
        if best_val < best:
            best = best_val
    return h, best

def _round_coord(h, n, dx, t0, time_limit):
    """Optimize each coordinate via ternary search on piecewise-linear max"""
    best = compute_c5(h, dx)
    target = n / 2.0
    for i in range(n):
        if time.time() - t0 > time_limit:
            break
        old = h[i]
        lo, hi = 0.0, 1.0
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            h[i] = m1
            h[i] = np.clip(m1, 0, 1)
            f1 = compute_c5(project_feasible(h, n), dx)
            h[i] = m2
            h[i] = np.clip(m2, 0, 1)
            f2 = compute_c5(project_feasible(h, n), dx)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) / 2.0
        h = project_feasible(h, n)
        val = compute_c5(h, dx)
        if val < best:
            best = val
    return h, best

def _upscale(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    h = initial_h_values.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: L-BFGS-B with annealed smoothing at n=70
    phase1_end = t0 + budget_s * 0.35
    for beta in [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]:
        if time.time() > phase1_end:
            break
        hi2, c5 = _lbfgs_step(h, dx, n, beta, 100, t0, phase1_end)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hi2.copy()
            h = hi2.copy()

    # Phase 2: Coordinate descent polish at n=70
    phase2_end = t0 + budget_s * 0.50
    if time.time() < phase2_end:
        h, c5_cd = _coord_descent(h, n, dx, rng, 5000, t0, phase2_end)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 3: Upscale and refine at higher resolutions
    for n_new in [140, 210, 280]:
        phase_end = t0 + budget_s * (0.75 if n_new <= 140 else 0.90)
        if time.time() > phase_end - 5:
            break
        dx_n = 2.0 / n_new
        h_n = _upscale(best_h, n_new)
        h_n = project_feasible(h_n, n_new)
        c5_n = compute_c5(h_n, dx_n)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()

        for beta in [10.0, 50.0, 200.0, 1000.0]:
            if time.time() > phase_end:
                break
            h_n, c5_n = _lbfgs_step(h_n, dx_n, n_new, beta, 60, t0, phase_end)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()

        # Quick coord descent at new resolution
        if time.time() < phase_end - 2:
            h_n, c5_n = _coord_descent(h_n, n_new, dx_n, rng, 2000, t0, phase_end)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()

    # Phase 4: Final polish
    phase4_end = t0 + budget_s * 0.97
    if time.time() < phase4_end:
        nf = len(best_h)
        dxf = 2.0 / nf
        h_f, c5_f = _coord_descent(best_h, nf, dxf, rng, 8000, t0, phase4_end)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
