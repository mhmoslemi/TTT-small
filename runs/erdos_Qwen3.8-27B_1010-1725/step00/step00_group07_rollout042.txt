The first sum contributes 1 if j is in the valid range (i.e., 0≤j<s... no, 0≤j and 0≤j+s<n).

- h[j] appears as the second factor when i+s=j, i=j-s: -h[j-s] (if valid)

    """Gradient of corr_full[k] w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # j in [0, s-1]: 1 - h[j+s]
        if s > 0:
            grad[:s] = 1.0 - h[s:2*s]  # h[j+s] for j=0..s-1 → h[s]..h[2s-1]

        lo2 = max(s, n - s)
        if lo2 < n:
            g[lo2:] = -h[lo2 - s:n - s]
    else:
        s2 = -s
        a = min(s2, n - s2)
        if a > 0:
            g[:a] = -h[s2:s2 + a]
        if s2 < n - s2 and n > 2 * s2:
            g[s2:n - s2] = (1.0 - h[:n - 2 * s2]) - h[2 * s2:n]
        lo2 = max(s2, n - s2)
        if lo2 < n:
            g[lo2:] = 1.0 - h[lo2 - s2:n - s2]
    return g

def _so(h, dx, T):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    e = np.exp((c - m) / T)
    w = e / e.sum()
    n = len(h)
    gr = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            gr += w[k] * _gc(h, k)
    return m + T * np.log(e.sum()), gr * dx

def _pd(h0, dx, budget, T0=0.08):
    n = len(h0)
    tgt = n / 2.0
    h = _proj(h0, tgt)
    bh = h.copy()
    bv = _eval(h)
    t0 = time.time()
    T = T0
    st = 0
    for i in range(200000):
        if time.time() - t0 > budget:
            break
        _, gd = _so(h, dx, T)
        gn = np.linalg.norm(gd, np.inf)
        if gn < 1e-12:
            break
        h2 = _proj(h - 0.4 * gd / gn, tgt)
        v = _eval(h2)
        if v < bv - 1e-12:
            bv, bh, st = v, h2.copy(), 0
        else:
            st += 1
        h = h2
        if st > 400:
            h = bh.copy()
            st = 0
        if i % 3000 == 2999:
            T = max(0.004, T * 0.9)
        if st > 5000:
            break
    return bh, bv

def _ls(h0, dx, budget):
    n = len(h0)
    tgt = n / 2.0
    h = _proj(h0, tgt)
    bv = _eval(h)
    t0 = time.time()
    for _ in range(500):
        if time.time() - t0 > budget:
            break
        mv = False
        _, gd = _so(h, dx, 0.01)
        ix = np.argsort(-np.abs(gd))[:20]
        for a in ix:
            if mv:
                break
            for b in ix:
                if a == b or mv:
                    continue
                for d in (0.06, 0.03, 0.015):
                    for s in (1, -1):
                        h2 = h.copy()
                        h2[a] += s * d
                        h2[b] -= s * d
                        h2 = _proj(h2, tgt)
                        v = _eval(h2)
                        if v < bv - 1e-10:
                            bv, h, mv = v, h2, True
                            break
                    if mv:
                        break
                if mv:
                    break
            if mv:
                break
        if not mv:
            break
    return h, bv

def _mk_bin(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.RandomState(seed)
    try:
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 80
        h0 = _mk_bin(n, rng)
    dx = 2.0 / n
    h0 = _proj(h0, n / 2.0)
    bh, bv = h0.copy(), _eval(h0)
    rem = budget_s - (time.time() - ts)
    # Phase 1: projected gradient descent
    h1, v1 = _pd(h0, dx, rem * 0.3, 0.08)
    if v1 < bv:
        bv, bh = v1, h1
    rem = budget_s - (time.time() - ts)
    # Phase 2: local search
    h2, v2 = _ls(bh, dx, rem * 0.25)
    if v2 < bv:
        bv, bh = v2, h2
    rem = budget_s - (time.time() - ts)
    # Phase 3: random restarts
    n_trials = min(12, max(4, int(rem * 0.3 // 40) + 1))
    per
    = min(50, max(15, int(rem * 0.3 / max(n_trials, 1))))
    for t in range(n_trials):
        if time.time() - ts > budget_s - 80:
            break
        hr = _proj(_mk_bin(n, rng), n / 2.0)
        vr = _eval(hr)
        if vr < bv + 0.1:
            hb, vb = _pd(hr, dx, per, 0.1)
            if vb < bv:
                bv, bh = vb, hb
            hl, vl = _ls(bh, dx, max(5, per // 4))
            if vl < bv:
                bv, bh = vl, hl
    rem = budget_s - (time.time() - ts)
    # Phase 4: refinement
    if rem > 60:
        h4, v4 = _ls(bh, dx, min(rem * 0.35, 80))
        if v4 < bv:
            bv, bh = v4, h4
        rem2 = budget_s - (time.time() - ts)
        if rem2 > 30:
            h5, v5 = _pd(bh, dx, min(rem2 * 0.5, 40), 0.005)
            if v5 < bv:
                bv, bh = v5, h5
    # Phase 5: try n=160 if time allows
    rem = budget_s - (time.time() - ts)
    if rem > 120:
        n2 = 2 * n
        h_coarse = bh
        h_fine = np.zeros(n2)
        h_fine[0::2] = h_coarse
        h_fine[1::2] = h_coarse
        h_fine = _proj(h_fine, n2 / 2.0)
        dx2 = 2.0 / n2
        hf, vf = _pd(h_fine, dx2, min(rem * 0.4, 100), 0.04)
        hf, vf = _ls(hf, dx2, min(rem * 0.15, 30))
        if vf < bv:
            bh, bv = hf, vf
    bh = _proj(bh, len(bh) / 2.0)
    fv = _eval(bh)
    if fv < bv:
        bv = fv
    return bh, bv, len(bh)
