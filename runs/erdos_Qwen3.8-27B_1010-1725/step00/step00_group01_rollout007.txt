import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    lo, hi = np.min(h) - 1.0, np.max(h) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_c5(h, beta=50.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    val = m / beta + np.log(np.sum(np.exp((corr - m) * beta))) / beta
    return val * dx

def finite_diff_grad(h, beta=50.0):
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    for j in range(n):
        hp = h.copy(); hp[j] += eps
        hm = h.copy(); hm[j] -= eps
        grad[j] = (smooth_c5(hp, beta) - smooth_c5(hm, beta)) / (2.0 * eps)
    return grad

def run_slsqp(h0, n, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

        return compute_c5(x)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        return res
    except Exception:
        return None
