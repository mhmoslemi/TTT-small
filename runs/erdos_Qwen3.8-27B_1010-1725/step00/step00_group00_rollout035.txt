    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(2 * n + 10):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not free.any():
                break
            add = diff / float(free.sum())
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = h > 1e-15
            if not free.any():
                break
            sub = -diff / float(free.sum())
            h[free] = np.maximum(h[free] - sub, 0.0)
    return h


def upsample_h(h, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    old_x = (np.arange(n_old) + 0.5) * dx_old
    new_x = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(new_x, old_x, h)
    return project_h(h_new, n_new)


def run_lbfgs(h, n, dx, t, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    obj_p = partial(smooth_max_obj, dx=dx, t=t, penalty=penalty)
    grad_p = partial(smooth_max_grad, dx=dx, t=t, penalty=penalty)
    result = minimize(
        obj_p, h, jac=grad_p,
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                 'maxfun': maxiter * 10}
    )
    return project_h(result.x, n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except NameError:
        n_init = 45
        h_init = np.zeros(n_init)
        h_init[::2] = 1.5
        h_init = project_h(h_init, n_init)

    h_init = project_h(h_init, n_init)
    best_h = h_init.copy()
    best_n = n_init
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    def tl():
        return budget_s - (time.time() - t_start) - 10

    # Build grid list starting from n_init
    candidates = [n_init]
    for g in [60, 90, 120, 150, 180, 240, 360]:
        if g > candidates[-1]:
            candidates.append(g)

    current_h = h_init.copy()
    current_n = n_init

    for n in candidates:
        if tl() < 40:
            break
        if n > current_n:
            current_h = upsample_h(current_h, current_n, n)
        current_h = project_h(current_h, n)
        current_n = n
        dx = 2.0 / n

        # Main optimization sweeps with increasing temperature
        configs = [(50, 200, 50), (100, 250, 100), (200, 200, 100), (500, 150, 200)]
        for t_val, maxit, pen in configs:
            if tl() < 20:
                break
            mi = max(50, maxit if n <= 180 else maxit // 2)
            h_opt = run_lbfgs(current_h, n, dx, t_val, pen, mi)
            c5_opt = compute_c5(h_opt, dx)
            cur_c5 = compute_c5(current_h, dx)
            if c5_opt < cur_c5:
                current_h = h_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

        # Local perturbations to escape local optima
        if tl() > 50 and n <= 240:
            for _ in range(4):
                if tl() < 15:
                    break
                noise = np.random.randn(n) * 0.08
                h_p = project_h(current_h + noise, n)
                h_opt = run_lbfgs(h_p, n, dx, 150, 100, 120)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                    current_h = h_opt

    # Final polish at best grid size with high temperature
    if tl() > 30:
        dx_b = 2.0 / best_n
        for t_val in [300, 1000]:
            if tl() < 15:
                break
            h_opt = run_lbfgs(best_h, best_n, dx_b, t_val, 300, 200)
            c5_opt = compute_c5(h_opt, dx_b)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = best_n

    # Ensure final solution is valid
    best_h = project_h(best_h, best_n)
    final_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, final_c5, best_n)
