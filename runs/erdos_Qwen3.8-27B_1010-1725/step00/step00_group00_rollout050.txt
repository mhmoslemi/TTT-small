                    best_h, best_c5 = h_pw.copy(), c5_pw

    # Final: ensure feasibility and compute exact C5
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n_final = len(best_h)
    return best_h, best_c5, n_final
