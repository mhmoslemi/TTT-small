def apply_change(h,F,G,p,delta):
    if delta == 0.0: return
    n = len(h)
    m = p if p < n - 1 - p else n - 1 - p
    if m >= 0:
        seg = h[p-m:p+1][::-1]
        F[:m+1] += delta * (1.0 - seg)
        G[:m+1] -= delta * seg
    L = n - p

        F[:L] -= delta * tail
        G[:L] += delta * (1.0 - tail)
    d2 = delta * delta
    F[0] -= d2
    G[0] -= d2
    h[p] += delta

def enforce_sum_inplace(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    rem = target - float(h.sum())
    if abs(rem) > 1e-12:
        step = rem / float(h.size)
        if np.all((h + step) >= -1e-12) and np.all((h + step) <= 1.0 + 1e-12):
            h += step
        else:
            order = np.argsort(h) if rem > 0.0 else np.argsort(-h)
            for idx in order:
                cap = (1.0 - h[idx]) if rem > 0.0 else h[idx]
                if cap <= 0.0:
                    continue
                if rem > 0.0:
                    add = cap if cap < rem else rem
                else:
                    add = (-cap if cap < -rem else rem)
                h[idx] += add
                rem -= add
                if abs(rem) <= 1e-12:
                    break
    rem = target - float(h.sum())
    if abs(rem) > 1e-14 and h.size:
        order = np.argsort(h) if rem > 0.0 else np.argsort(-h)
        for idx in order:
            if rem > 0.0 and (1.0 - h[idx]) >= rem:
                h[idx] += rem
                break
            if rem < 0.0 and h[idx] >= -rem:
                h[idx] += rem
                break
    h[h < 1e-12] = 0.0
    h[h > 1.0 - 1e-12] = 1.0
    return h


def adjust_sum_inplace(h, target):
    np.clip(h, 0.0, 1.0, out=h)
    rem = target - float(h.sum())
    if abs(rem) <= 1e-10:

def eval_c(h, n):
    h = np.asarray(h, dtype=float)

        return float("inf")
    return val


def init_fg(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr[n - 1:].copy()
    G = corr[n - 1::-1].copy()
    np.maximum(F, 0.0, out=F)
    np.maximum(G, 0.0, out=G)
    return F, G


def apply_change(h, F, G, p, delta):
    if delta == 0.0:
        return
    n = len(h)
    m = p if p < n - 1 - p else n - 1 - p
    if m >= 0:
        seg = h[p - m:p + 1][::-1]
        F[:m + 1] += delta * (1.0 - seg)
        G[:m + 1] -= delta * seg
    L = n - p
    if L > 0:
        tail = h[p:n]
        F[:L] -= delta * tail
        G[:L] += delta * (1.0 - tail)
    d2 = delta * delta
    F[0] -= d2
    G[0] -= d2
    h[p] += delta


def trial_pair_score(h, F, G, ht, Ft, Gt, a, b, delta):
    fa0 = float(F.max())
    ga0 = float(G.max())
    s0 = fa0 if fa0 >= ga0 else ga0
    if a == b or delta == 0.0:
        if not math.isfinite(s0):
            return float("inf"), fa0, ga0
        return s0, fa0, ga0
    ht[:] = h
    Ft[:] = F
    Gt[:] = G
    apply_change(ht, Ft, Gt, a, delta)
    apply_change(ht, Ft, Gt, b, -delta)
    fa = float(Ft.max())
    ga = float(Gt.max())
    s = fa if fa >= ga else ga

    c_bin = eval_c(h_bin, best_n)
    if c_bin < c_out - 1e-9:
        h_out = h_bin
        c_out = c_bin

    adjust_sum_inplace(h_out, 0.5 * best_n)
    c_out = eval_c(h_out, best_n)
    return h_out, c_out, int(best_n)
