import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(comp), comp


def grad_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    else:
        s = -k
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
    return grad


def smooth_max_and_grad(h, comp, tau):
    n = len(h)
    c_max = np.max(comp)
    shifted = tau * (comp - c_max)
    np.clip(shifted, -700, 700, out=shifted)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total == 0:
        weights = np.zeros_like(comp)
        weights[np.argmax(comp)] = 1.0
    else:
        weights = exp_vals / total
    value = c_max + np.log(total) / tau
    grad = np.zeros(n)
    for k in range(len(comp)):
        w = weights[k]
        if w < 1e-7:
            continue
        shift = k - (n - 1)
        grad += w * grad_shift(h, shift)
    return value, grad
