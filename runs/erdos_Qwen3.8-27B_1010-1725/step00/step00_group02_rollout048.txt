def compute_c5(h):
    """Compute C5 for given h array."""
    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, m, mode="full")

    best_h = ensure_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
