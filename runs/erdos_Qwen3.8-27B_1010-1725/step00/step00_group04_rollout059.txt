        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    for attempt in range(3):
        if time.time() - t0 > t_limit:
            break
        try:
            r = minimize(compute_c5, h_init, method='SLSQP', bounds=bounds,
                        constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            h_init = repair_h(r.x)
            c5 = compute_c5(h_init)
            if c5 < best_c5:
                best_c5, best = c5, h_init.copy()
        except Exception:
            break
    return best, best_c5

def random_perturb(h, rng, scale=0.05):
    h_new = h + rng.uniform(-scale, scale, len(h))
    return repair_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    # Get initial values
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.zeros(42)
        h0[:21] = 1.0
    n0 = len(h0)
    h0 = repair_h(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize at initial resolution n0
    # Try multiple starts
    starts = [h0.copy()]
    for _ in range(4):
        starts.append(random_perturb(h0, rng, 0.08))

    t_per_start = min(80.0, max(20.0, time_left() * 0.4 / len(starts)))
    for s in starts:
        if time_left() < 8:
            break
        if len(s) <= 84:
            h_opt, c5 = opt_slsqp(s.copy(), len(s), t_per_start)
            h_opt2, c5b = opt_lbfgsb(h_opt, len(h_opt), max(10, t_per_start * 0.5))
            if c5b < c5:
                h_opt, c5 = h_opt2, c5b
        else:
            h_opt, c5 = opt_lbfgsb(s.copy(), len(s), t_per_start)
        h_opt = repair_h(h_opt)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5, best_h = c5, h_opt

    # Phase 2: Progressive upsampling
    current_n = len(best_h)
    resolutions = []
    nn = current_n
    for _ in range(5):
        nn = min(2 * nn, 1024)
        resolutions.append(nn)
        if nn >= 1024:
            break

    for target_n in resolutions:
        if time_left() < 40:
            break
        alloc = min(time_left() * 0.5, 180.0)
        if alloc < 15:
            break
        h_start = upsample_h(best_h, target_n)

        # Multiple attempts at this resolution
        h_cur = h_start.copy()
        c5_cur = compute_c5(h_cur)

        if target_n <= 84:
            h_cur, c5_cur = opt_slsqp(h_cur, target_n, alloc * 0.5)
            h_cur2, c5b = opt_lbfgsb(h_cur, target_n, alloc * 0.5)
            if c5b < c5_cur:
                h_cur, c5_cur = h_cur2, c5b
        else:
            h_cur, c5_cur = opt_lbfgsb(h_cur, target_n, alloc * 0.7)
            # Second pass with perturbation if time allows
            if time_left() > 30:
                h_pert = random_perturb(h_cur, rng, 0.03)
                h_cur2, c5b = opt_lbfgsb(h_pert, target_n, alloc * 0.3)
                if c5b < c5_cur:
                    h_cur, c5_cur = h_cur2, c5b

        h_cur = repair_h(h_cur)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h_cur.copy()
        current_n = target_n

    # Phase 3: Fine-tune at best resolution
    if time_left() > 30:
        n_best = len(best_h)
        n_rep = min(4, int(time_left() / 25))
        for _ in range(n_rep):
            if time_left() < 10:
                break
            h_pert = random_perturb(best_h, rng, 0.02)
            if n_best <= 84:
                h_opt, c5 = opt_slsqp(h_pert, n_best, time_left() * 0.4)
                h_opt2, c5b = opt_lbfgsb(h_opt, n_best, max(8, time_left() * 0.2))
                if c5b < c5:
                    h_opt, c5 = h_opt2, c5b
            else:
                h_opt, c5 = opt_lbfgsb(h_pert, n_best, time_left() * 0.6)
            h_opt = repair_h(h_opt)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5, best_h = c5, h_opt

    # Final validation
    best_h = repair_h(best_h)
    best_c5 = compute_c5(best_h)
    n_final = len(best_h)

    # Ensure exact constraint
    target_sum = 0.5 * n_final
    actual_sum = np.sum(best_h)
    if abs(actual_sum - target_sum) > 1e-9:
        best_h = repair_h(best_h)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
