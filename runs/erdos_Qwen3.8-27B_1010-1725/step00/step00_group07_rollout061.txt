import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interterrain_to(h_src, n_dst):
    """Interpolate h from n_src points to n_dst points, preserving integral"""
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def make_binary(h, target_ones):
    """Convert h to binary with exactly target_ones ones"""
    h_bin = (h >= 0.5).astype(float)
    ones = np.sort(np.where(h_bin == 1)[0])
    zeros = np.sort(np.where(h_bin == 0)[0])
    if len(ones) > target_ones:
        excess = len(ones) - target_ones
        # Remove ones with lowest h values
        order = np.argsort(-h[ones])
        for idx in order[:excess]:
            h_bin[ones[idx]] = 0
    elif len(ones) < target_ones:
        deficit = target_ones - len(ones)
        order = np.argsort(-h[zeros])
        for idx in order[:deficit]:
            h_bin[zeros[idx]] = 1
    return h_bin

def local_search(h, rng, max_evals=100000, temperature=0.001):
    """Binary local search with simulated annealing"""
    n = len(h)
    target = n // 2
    h = make_binary(h, target)
    best = compute_c5(h)
    best_h = h.copy()
    current = h.copy()
    current_c5 = best
    evals = 0
    no_improve = 0
    temp = temperature
    
    while evals < max_evals:
        # Pick random swap
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        oi, zj = ones[i], zeros[j]
        
        current[oi] = 0
        current[zj] = 1
        evals += 1
        c = compute_c5(current)
        
        if c < current_c5:

            no_improve = 0
            if c < best:

                current_c5 = c
            else:
                current[oi] = 1
                current[zj] = 0
                no_improve += 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_for_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    j1, j2 = max(0, s), min(n, n + s)
    if j1 < j2:
        idx = np.arange(j1, j2)
        grad[idx] += (1.0 - h[idx - s])
    j3, j4 = max(0, -s), min(n, n - s)
    if j3 < j4:
        idx2 = np.arange(j3, j4)
        grad[idx2] -= h[idx2 + s]
    return grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        mask = h < 1.0 - 1e-9 if diff > 0 else h > 1e-9
        c = mask.sum()
        if c == 0:
            break
        h[mask] += diff / c
    return np.clip(h, 0.0, 1.0)

def to_binary(h, target_ones):
    n = len(h)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target_ones]] = 1.0
    return h_bin

def interp_to(h_src, n_dst):
    n_src = len(h_src)
    x = np.linspace(0, 2, n_dst)
    xs = np.linspace(0, 2, n_src)
    return np.interp(x, xs, h_src)

def binary_sa(h, rng, max_evals=200000, temp0=0.005):

    target = n // 2
    h = to_binary(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur = h.copy()
    cur_c5 = best_c5
    temp = temp0
    stall = 0
    evals = 0
    while evals < max_evals:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        oi, zj = ones[i], zeros[j]
        cur[oi], cur[zj] = 0.0, 1.0
        evals += 1
        c = compute_c5(cur)
        if c < cur_c5 - 1e-14:

            stall = 0

                best_h = cur.copy()
        else:

            if temp > 1e-9 and rng.random() < np.exp(-d / temp):
                cur_c5 = c
            else:

                stall += 1
        if stall > 3000:
            temp *= 0.6
            stall = 0
            if temp < 1e-9:

    return best_h, best_c5

def grad_descent(h, target, rng, max_iter=5000, lr0=0.08):
    n = len(h)
    h = project_sum(h, target)
    best_c5 = compute_c5(h)

    lr = lr0

        ak = int(np.argmax(corr))
        s = ak - (n - 1)
