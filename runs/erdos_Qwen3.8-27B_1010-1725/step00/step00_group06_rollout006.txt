    j = np.arange(n)
    grad = np.zeros(n)
    jp = j + lag
    jm = j - lag
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    grad[vp] += (1.0 - h[jp[vp]])
    grad[vm] -= h[jm[vm]]
    grad *= dx

def sum_constraint(h):
    return np.sum(h) - h.size / 2.0

def sum_jac(h):
    return np.ones(h.size)

def fix_sum(h, n):

        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)

                break
            h_cd = best_h.copy()
            n = len(h_cd)
            order = rng.permutation(n)
            for j in order:
                if t_left() < 5:
                    break
                # Try adjusting h[j]
                h_j = h_cd[j]
                for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]:
                    h_try = h_cd.copy()
                    h_try[j] = np.clip(h_j + delta, 0, 1)
                    diff = n / 2.0 - h_try.sum()
                    # Distribute difference to neighbors
                    if abs(diff) > 1e-4:
                        neighbors = [i for i in [(j-1)%n, (j+1)%n, (j-2)%n, (j+2)%n]
                                     if i != j]
                        room = [1.0 - h_try[i] if diff > 0 else h_try[i] for i in neighbors]
                        total_room = sum(room)
                        if total_room > 0.01:
                            for i, rm in zip(neighbors, room):
                                h_try[i] += diff * rm / total_room
                            h_try = np.clip(h_try, 0, 1)
                    c5_try = compute_c5(h_try)
                    if c5_try < compute_c5(h_cd) - 1e-9:
                        h_cd = h_try
                        h_cd = fix_sum(h_cd, n)
                        break
                h_j = h_cd[j]
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 3: Try higher resolution via interpolation
    for n_new in [192, 384, 512]:
        if t_left() < 80:
            break
        x_old = np.linspace(0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0, 2.0, n_new, endpoint=False)
        h_ext = np.concatenate([best_h, [best_h[0]]])
        h_interp = np.interp(x_new, x_old, h_ext)
        h_interp = fix_sum(h_interp, n_new)

        h_opt, c5_val = slsqp_optimize(h_interp, maxiter=250, ftol=1e-7)
        # Also try a few restarts at this resolution
        for rr in range(3):
            if t_left() < 30:
                break
            h_init = random_perturb(h_opt, rng, frac=0.1, amp=0.2)
            h_opt2, c5_val2 = slsqp_optimize(h_init, maxiter=200, ftol=1e-7)
            if c5_val2 < c5_val:
                h_opt, c5_val = h_opt2, c5_val2

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            n0 = n_new

    # Phase 4: Final fine-tuning
    if t_left() > 20:
        n = len(best_h)
        for rr in range(10):
            if t_left() < 10:
                break
            h_init = random_perturb(best_h, rng, frac=0.05, amp=0.08)
            h_opt, c5_val = slsqp_optimize(h_init, maxiter=500, ftol=1e-9)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

    # Final validation and cleanup
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    best_c5 = compute_c5(best_h)

    # Validate with pre-imported function if available
    try:
        val = evaluate_erdos_solution(best_h, n_final)
        if val is not None and float(val) < best_c5:
            best_c5 = float(val)
    except Exception:
        pass

    return (best_h, best_c5, n_final)
