import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    if s > target_sum:
        excess = s - target_sum
        order = np.argsort(-h)
        for idx in order:
            r = min(h[idx], excess)
            h[idx] -= r
            excess -= r
            if excess < 1e-14:
                break
    else:
        deficit = target_sum - s
        order = np.argsort(h)
        for idx in order:
            a = min(1.0 - h[idx], deficit)
            h[idx] += a
            deficit -= a
            if deficit < 1e-14:
                break
    return h

def compute_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = w[k]
        if abs(wk) < 1e-10:
            continue
        l = k - (n - 1)
        m = n - abs(l)
        if m <= 0:
            continue
        if l >= 0:
            t1 = np.zeros(n)
            t1[:m] = 1.0 - h[:m]
            t2 = np.zeros(n)
            t2[l:] = h[:m]
        else:
            ll = -l
            t1 = np.zeros(n)
            t1[ll:] = 1.0 - h[:m]
            t2 = np.zeros(n)
            t2[:m] = h[ll:]
        grad += wk * (t1 - t2)
    return grad

def subgrad_optimize(h, target_sum, max_time, lr_init, temp_init, rng):
    t_start = time.time()
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    temp = temp_init
    stall = 0
    while time.time() - t_start < max_time:
        over = compute_overlaps(h)
        shifted = over - over.max()
        w = np.exp(shifted / temp)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros_like(over)
            w[np.argmax(over)] = 1.0
        else:
            w /= wsum
        grad = compute_grad(h, w)
        h = project_feasible(h - lr * grad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            # Reduce temperature and lr
            temp = max(temp * 0.7, 0.003)
            lr *= 0.7
            stall = 0
    return best_h, best_c5

def coord_descent(h, target_sum, max_time, delta=0.02):
    t_start = time.time()
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    while time.time() - t_start < max_time:
        improved = False
        for j in range(n):
            for sign in (1, -1):

                    best_c5 = c5t
                    improved = True
                    break
        if not improved:
            delta *= 0.5
            if delta < 1e-6:
                break
    return h, best_c5
