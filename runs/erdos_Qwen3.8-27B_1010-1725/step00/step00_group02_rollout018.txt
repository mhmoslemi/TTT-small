import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = m + (1.0 / beta) * np.log(np.sum(np.exp((corr - m) / beta)))
    return float(s * dx)

def constrain(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, beta, n):
    return smooth_c5(h, beta) + 1e4 * (np.sum(h) - n / 2.0) ** 2

def obj_exact(h, n):
    return compute_c5(h) + 1e4 * (np.sum(h) - n / 2.0) ** 2

def compute_B(h, i, n):
    m_vals = np.arange(2 * n - 1) - (n - 1)
    idx1 = i - m_vals
    idx2 = i + m_vals
    B = np.zeros(2 * n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    B[v1] += (1.0 - h[idx1[v1]])
    B[v2] += (-h[idx2[v2]])
    return B

def coord_descent(h, n, n_sweeps=500, time_limit=None, t0=None):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    xg = np.linspace(0.0, 1.0, 200)
    for sweep in range(n_sweeps):
        if time_limit is not None and (time.time() - t0 > time_limit):
            break
        improved = False
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            B = compute_B(h, i, n)
            A = corr - h[i] * B
            vals = np.max(A[None, :] + xg[:, None] * B[None, :], axis=1) * dx
            idx_best = np.argmin(vals)
            x_new = xg[idx_best]
            if abs(x_new - h[i]) > 1e-9:
                h[i] = x_new
                improved = True
        h = constrain(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved and sweep > 10:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    best_h = constrain(np.array(initial_h_values, dtype=float), n0)
    best_c5 = compute_c5(best_h)
    bounds0 = [(0.0, 1.0)] * n0

    # Phase 1: L-BFGS-B with smooth max at initial resolution
    for beta in [50.0, 200.0, 500.0]:
        if time.time() - t0 > budget_s * 0.25:
            break
        res = minimize(obj_smooth, best_h, args=(beta, n0), method='L-BFGS-B',
                       bounds=bounds0, options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = constrain(res.x, n0)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new

    # Phase 2: Powell (derivative-free, handles non-smooth max)
    if time.time() - t0 < budget_s * 0.45:
        res = minimize(obj_exact, best_h, args=(n0,), method='Powell',
                       bounds=bounds0, options={'maxiter': 2000, 'xtol': 1e-10, 'ftol': 1e-12})
        h_p = constrain(res.x, n0)
        c = compute_c5(h_p)
        if c < best_c5:
            best_c5, best_h = c, h_p

    # Phase 3: Coordinate descent (exact, per-coordinate convex 1D)
    if time.time() - t0 < budget_s * 0.7:
        h_cd, c_cd = coord_descent(best_h, n0, n_sweeps=2000,
                                   time_limit=budget_s * 0.8, t0=t0)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd

    # Phase 4: Higher resolution refinement
    for n_new in [60, 90, 120, 180]:
        if time.time() - t0 > budget_s * 0.92:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_int = constrain(np.interp(x_new, x_old, best_h), n_new)
        bn = [(0.0, 1.0)] * n_new
        for beta in [100.0, 400.0]:
            if time.time() - t0 > budget_s * 0.95:
                break
            res = minimize(obj_smooth, h_int, args=(beta, n_new), method='L-BFGS-B',
                           bounds=bn, options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-11})
            h_i = constrain(res.x, n_new)
            c = compute_c5(h_i)
            if c < best_c5:
                best_c5, best_h, h_int = c, h_i, h_i
        h_cd2, c_cd2 = coord_descent(h_int, n_new, n_sweeps=500,
                                     time_limit=budget_s * 0.95, t0=t0)
        if c_cd2 < best_c5:
            best_c5, best_h = c_cd2, h_cd2

    # Phase 5: Random restarts at original resolution
    if time.time() - t0 < budget_s * 0.95:
        for _ in range(100):
            if time.time() - t0 > budget_s * 0.98:
                break
            h_r = constrain(0.5 + 0.4 * rng.randn(n0), n0)
            res = minimize(obj_smooth, h_r, args=(200.0, n0), method='L-BFGS-B',
                           bounds=bounds0, options={'maxiter': 800, 'ftol': 1e-11})
            h_r = constrain(res.x, n0)
            c = compute_c5(h_r)
            if c < best_c5:
                best_c5, best_h = c, h_r

    # Final coordinate polish
    if time.time() - t0 < budget_s * 0.99:
        n_f = len(best_h)
        h_f, c_f = coord_descent(best_h, n_f, n_sweeps=300,
                                  time_limit=budget_s * 0.99, t0=t0)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    best_h = constrain(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
