import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return float(np.max(c))

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Binary search on lambda
    lo, hi = -10, 10
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(80) * 0.5
    n = len(h0)
    N_target = n // 2  # sum(h) should be N/2
    
    # Evaluate initial
    c5_init = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_init
    
    # Project to ensure feasibility
    best_h = project(best_h, N_target)
    
    # Log-sum-exp parameter
    alpha = 100
    
    # Gradient descent
    lr = 0.01
    for it in range(5000):
        if time.time() - start_time > budget_s - 60:
            break
        if it % 100 == 0:
            alpha *= 1.02  # gradually increase sharpness
        
        # Compute correlation
        c = np.correlate(best_h, 1-best_h, mode='full')
        # Softmax weights
        c_shifted = c - c.max()
        p = np.exp(alpha * c_shifted) / np.exp(alpha * c_shifted).sum()
        
        # Compute gradient: dF/dh[m] = sum_j p[j] * dc_j/dh[m] (times dx, but we'll absorb into lr)
        grad = np.zeros(n)
        for j in range(2*n - 1):
            k = j - (n - 1)
            # dc_j/dh[m] = (1-h[m-k]) * I(0<=m-k<n) - h[m+k] * I(0<=m+k<n)
            # First term: m ranges over max(0,k) to min(n-1, n-1+k)
            m_start1 = max(0, k)
            m_end1 = min(n-1, n-1+k)
            if m_start1 <= m_end1:
                grad[m_start1:m_end1+1] += p[j] * (1 - best_h[m_start1-k:m_end1-k+1])
            
            # Second term: m ranges over max(0,-k) to min(n-1, n-1-k)
            m_start2 = max(0, -k)
            m_end2 = min(n-1, n-1-k)
            if m_start2 <= m_end2:
                grad[m_start2:m_end2+1] -= p[j] * best_h[m_start2+k:m_end2+k+1]
        
        # Gradient step
        best_h = best_h - lr * grad
        # Project
        best_h = project(best_h, N_target)
        
        # Evaluate
        c5_curr = evaluate_c5(best_h)
        if c5_curr < best_c5:
            best_c5 = c5_curr
    
    return best_h, best_c5, n

        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_gradient(h, alpha):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    F = c * dx
    F_shift = F - F.max()
    exp_F = np.exp(alpha * F_shift)
    p = exp_F / exp_F.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (n - 1)
        ms1 = max(0, k)
        me1 = min(n - 1, n - 1 + k)
        if ms1 <= me1:
            grad[ms1:me1 + 1] += w * (1.0 - h[ms1 - k:me1 - k + 1])
        ms2 = max(0, -k)
        me2 = min(n - 1, n - 1 - k)
        if ms2 <= me2:
            grad[ms2:me2 + 1] -= w * h[ms2 + k:me2 + k + 1]
    return grad * dx

def optimize_from(h0, target, budget_end, rng):
    n = len(h0)
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c5 = eval_c5(best_h)
    alpha = 20.0
    lr = 0.05
    max_lr = 2.0
    min_lr = 1e-5
    patience = 0
    max_patience = 200

    while time.time() < budget_end:
        if patience >= max_patience:
            alpha *= 1.8
            lr = 0.05
            patience = 0
            h = best_h + 0.05 * rng.randn(n)
            h = project_feasible(h, target)

        grad = compute_gradient(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            patience += 100
            continue
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = eval_c5(h_new)
        c5_old = eval_c5(h)

        if c5_new < c5_old - 1e-10:
            h = h_new
            lr = min(lr * 1.03, max_lr)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            patience = 0
        else:
            lr *= 0.85
            patience += 1
            if lr < min_lr:
                alpha *= 1.5
                lr = 0.1

    return best_h, best_c5

def make_periodic(n, period, duty):
    one_period = np.zeros(period)
    one_period[:int(duty * period)] = 1.0
    reps = n // period
    h = np.tile(one_period, reps)
    if n > reps * period:
        h = np.concatenate([h, one_period[:n - reps * period]])
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_end = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(80, 0.5)
    n = len(h_init)
    target = n / 2.0

    # Ensure feasibility
    h_init = project_feasible(h_init.copy(), target)
    c5_init = eval_c5(h_init)

    best_h = h_init.copy()
    best_c5 = c5_init

    # Phase 1: Optimize from initial
    h_opt, c5_opt = optimize_from(h_init, target, t0 + 200, rng)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    # Phase 2: Try multiple restarts with perturbations
    n_restarts = 0
    while time.time() < budget_end - 60:
        n_restarts += 1
        # Perturb best solution
        scale = 0.15 / (1 + 0.1 * n_restarts)
        pert = best_h + scale * rng.randn(n)
        pert = project_feasible(pert, target)
        h_opt, c5_opt = optimize_from(pert, target, t0 + 200 * (1 + 0.05 * n_restarts), rng)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt
        else:
            continue

    # Phase 3: Try structured initializations
    structured_starts = [
        make_periodic(n, 6, 0.5),
        make_periodic(n, 8, 0.5),
        make_periodic(n, 10, 0.5),
        make_periodic(n, 12, 0.5),
        make_periodic(n, 4, 0.5),
        make_periodic(n, 14, 0.5),
        make_periodic(n, 16, 0.5),
    ]
    # Add sinusoidal
    x = np.linspace(0, 2 * np.pi * 3, n)
    structured_starts.append(0.5 + 0.3 * np.sin(x))
    structured_starts.append(0.5 + 0.3 * np.cos(x * 1.5))
    structured_starts.append(best_h + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n)))

    for ss in structured_starts:
        if time.time() > budget_end - 30:
            break
        ss = project_feasible(ss, target)
        h_opt, c5_opt = optimize_from(ss, target, min(time.time() + 40, budget_end - 10), rng)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # Phase 4: Fine-tune best solution with very small steps
    h_fine = best_h + 0.02 * rng.randn(n)
    h_fine = project_feasible(h_fine, target)
    h_opt, c5_opt = optimize_from(h_fine, target, budget_end, rng)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    # Final check
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n
