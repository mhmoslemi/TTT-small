import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_c5_grad(h, top_m=1):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    if top_m >= len(corr):
        weights = np.ones(len(corr)) / len(corr)
        idxs = range(len(corr))
    else:
        idxs = np.argsort(corr)[-top_m:]
        w = corr[idxs]
        w = np.exp(w * 10) / np.exp(w * 10).sum()  # softmax weighting
        weights = w
    
    grad = np.zeros(N)
    for wi, ki in zip(weights, idxs):
        lag = ki - (N - 1)
        if lag >= 0:
            for i in range(N - lag):
                grad[i+lag] += wi * (1.0 - h[i])
                grad[i] -= wi * h[i+lag]
        else:
            m = -lag
            for i in range(m, N):
                grad[i] += wi * (1.0 - h[i-m])
                grad[i-m] -= wi * h[i]
    
    max_val = float(np.max(corr))
    return max_val, grad * dx

def project_feasible(h, target_sum):
    N = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    # Fix sum
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # distribute to elements with room
        if diff > 0:
            idxs = np.where(x < 1.0)[0]
        else:
            idxs = np.where(x > 0.0)[0]
        if len(idxs) > 0:
            per = diff / len(idxs)
            x[idxs] += per
            x[idxs] = np.clip(x[idxs], 0.0, 1.0)
    return x

    N = 45
    h0 = initial_h_values.copy()
    target = N / 2.0
    
    beta = 50.0
    
    def obj_and_grad(x):
        h = np.clip(x, 0, 1)  # ensure bounds
        corr = np.correlate(h, 1-h, mode='full') * (2.0/N)
        # log-sum-exp
        max_c = np.max(corr)
        logsum = max_c + np.log(np.sum(np.exp(beta * corr - beta * max_c))) / beta
        # gradient
        weights = np.exp(beta * corr - beta * max_c)
        weights /= weights.sum()
        # weighted gradient
        dx = 2.0 / N
        grad = np.zeros(N)
        for k in range(2*N-1):
            if weights[k] < 1e-10:
                continue
            lag = k - (N-1)
            if lag >= 0:
                for i in range(N - lag):
                    grad[i+lag] += weights[k] * (1-h[i])
                    grad[i] -= weights[k] * h[i+lag]
            else:
                m = -lag
                for i in range(m, N):
                    grad[i] += weights[k] * (1-h[i-m])
                    grad[i-m] -= weights[k] * h[i]
        grad *= dx
        return logsum, grad
    
    from scipy.optimize import minimize
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
                    'jac': lambda x: np.ones(N)}]
    bounds = [(0,1)] * N
    
    result = minimize obj_and_grad, h0, method='SLSQP', 
                      constraints=constraints, bounds=bounds,
