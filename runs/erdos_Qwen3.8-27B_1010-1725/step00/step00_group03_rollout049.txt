import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_feasible(h, n):
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, n, temp):
    om = 1.0 - x
    corr = np.correlate(x, om, mode='full') * (2.0 / n)
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -60.0, 0.0)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new)

    def fun(x):
        return np.sum(x) - n / 2.0
    def jac(x):
        return np.ones(n)
    return [{'type': 'eq', 'fun': fun, 'jac': jac}]
