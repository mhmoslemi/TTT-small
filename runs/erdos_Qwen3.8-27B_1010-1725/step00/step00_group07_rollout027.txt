import time

def compute_c5(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr)) * dx

def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def subgradient(h, n, num_active=5, tol=1e-6):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    c5_val = float(np.max(corr)) * dx
    
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - tol)[0]
    
    # Compute analytic subgradient
    grad = np.zeros(len(corr))  # unused, we want grad w.r.t. h
    grad = np.zeros(n)
    for k in active:

        for j in range(n):

            if 0 <= idx1 <= n-1:
                grad[j] += (1.0 - h[idx1])
            if 0 <= idx2 <= n-1:
                grad[j] -= h[idx2]
        grad /= len(active)
    
    return grad * dx, c5_val
