```python
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)  # weights, length 2n-1
    
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            m = k - (n-1)  # lag
            # First term: (1-h[j+m]) if 0 <= j+m < n
            if 0 <= j+m < n:
                grad[j] += p[k] * (1 - h[j+m])
            # Second term: -h[j-m] if 0 <= j-m < n
            if 0 <= j-m < n:
                grad[j] -= p[k] * h[j-m]
        grad[j] *= dx
    
    return grad
```
