def _fix_sum(h, target):
    """Fix sum(h) to target while keeping h in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass: find elements with room
        room = 1.0 - h
        order = np.argsort(-room)
