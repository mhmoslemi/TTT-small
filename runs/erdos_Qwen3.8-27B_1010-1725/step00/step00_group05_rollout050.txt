import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_feasible(h, target_sum, rng=None):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        h = np.clip(h + diff / len(h), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if diff > 1e-14:
        perm = np.random.permutation(len(h)) if rng is None else rng.permutation(len(h))
        for i in perm:
            room = 1.0 - h[i]
            add = min(room, diff)
            if add > 0:
                h[i] += add
                diff -= add
            if diff <= 1e-14:
                break
    elif diff < -1e-14:
        diff = -diff
        perm = np.random.permutation(len(h)) if rng is None else rng.permutation(len(h))
        for i in perm:
            take = min(h[i], diff)
            if take > 0:
                h[i] -= take
                diff -= take
            if diff <= 1e-14:
                break
    return h

def sa_optimize(h, dx, target_sum, rng, time_budget, T0=0.002, T_floor=1e-9):

    h = h.copy()

    best = cur
    best_h = h.copy()
    T = T0

    step_scale = 0.04
    it = 0
    while time.time() - start < time_budget:
        it += 1
        move_type = rng.integers(3)
        if move_type == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = rng.uniform(-step_scale, step_scale)
            h[i] += delta
            h[j] -= delta
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= delta; h[j] += delta
                continue
        elif move_type == 1:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:

            h[i], h[j] = h[j], h[i]
        else:
