    return float(np.max(corr) * dx)

def project_onto_simplicial(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-14
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def smooth_max_subgrad(h, T=0.01):
    """Compute subgradient of smooth max of correlate(h,1-h,full)."""

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if mask.sum() == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 1e-13
            if mask.sum() == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return h

def smooth_max_grad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    w = np.exp((c - cm) / T)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        l = k - (n - 1)
        i_min = max(0, -l)
        i_max = min(n - 1, n - 1 - l)
        if i_min > i_max:
            continue
        ja = max(0, i_min + l)
        jb = min(n - 1, i_max + l)
        if ja <= jb:

        jc = max(0, i_min)
        jd = min(n - 1, i_max)
        if jc <= jd:
            grad[jc:jd+1] -= wk * h[jc:jd+1 + l]
    return grad, c
