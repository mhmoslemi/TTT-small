                g[lp:] -= h[:n - lp]
                g[:n - lp] -= h[lp:]
        grad += wm * g
    return val, grad * dx

def subgrad_dir(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    ks = np.argmax(corr)
    lag = ks - (n - 1)
    g = np.ones(n)
    if lag >= 0 and n > lag:
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    elif lag < 0:
        lp = -lag
        if n > lp:
            g[lp:] -= h[:n - lp]
            g[:n - lp] -= h[lp:]
    return g, lag, corr[ks]

def binary_init(n, rng, S):
    h = np.zeros(n)
    idx = rng.choice(n, size=S, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is not None and len(h0) > 0:
        n = len(h0)
        S = n // 2
        h0 = proj(h0, S)
    else:
        n = 70
        S = 35
        h0 = binary_init(n, rng, S)

    best_h = h0.copy()
    best_c5 = c5_eval(h0)

    def update_best(h):
        nonlocal best_h, best_c5
        v = c5_eval(h)
        if v < best_c5:
            best_c5 = v
            best_h = h.copy()

    # Phase 1: Smooth gradient descent with decreasing temperature
    h = h0.copy()
    temps = [0.3, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0]
    for T in temps:
        if tl() < 80:
            break
        lr = 0.02 * (1.0 + 1.0 / T)
        for it in range(400):
            if tl() < 30:
                break
            val, grad = smooth_obj_grad(h, T)
            h_new = proj(h - lr * grad, S)
            val_new = c5_eval(h_new)
            if val_new < best_c5:
                best_c5 = val_new
                best_h = h_new.copy()
            if val_new <= val:
                h = h_new
                lr = min(lr * 1.02, 0.5)
            else:
                lr *= 0.75
                if lr < 1e-5:
                    break

    # Phase 2: Exact subgradient with adaptive step
    h = best_h.copy()
    lr = 0.005
    n_stall = 0
    for it in range(8000):
        if tl() < 20:
            break
        g, lag, corr_val = subgrad_dir(h)
        dx = 2.0 / n
        h_new = proj(h - lr * g * dx, S)
        v_new = c5_eval(h_new)
        if v_new < best_c5:
            best_c5 = v_new
            best_h = h_new.copy()
            n_stall = 0
            if v_new < corr_val * dx:
                h = h_new
                lr = min(lr * 1.03, 0.1)
            else:
                lr = max(lr * 0.97, 1e-6)
        else:
            n_stall += 1
            if n_stall > 500:
                lr = max(lr * 0.5, 1e-7)
                n_stall = 0
            h_new2 = proj(h - lr * 0.5 * g * dx, S)
            if c5_eval(h_new2) < corr_val * dx:
                h = h_new2

    # Phase 3: Local search - random perturbations
    h = best_h.copy()
    n_srch = 0
    for it in range(5000):
        if tl() < 15:
            break
        n_srch += 1
        step = max(0.01, 0.1 * np.exp(-0.001 * n_srch))
        h_try = h + rng.normal(0, step, size=n)
        h_try = proj(h_try, S)
        v = c5_eval(h_try)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()
            h = h_try
        elif v < c5_eval(h) + 0.002 * np.exp(-0.0005 * n_srch):
            h = h_try

    # Phase 4: Coordinate refinement
    h = best_h.copy()
    for pass_idx in range(5):
        if tl() < 10:
            break
        for it in range(30):
            if tl() < 10:
                break
            improved = False
            for i in range(n):
                old = h[i]
                for delta in [-0.05, -0.02, 0.02, 0.05, -0.1, 0.1]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                    # Fix sum by adjusting a random other index
                    diff = S - np.sum(h_try)
                    if abs(diff) > 1e-10:
                        j = (i + 1) % n
                        h_try[j] = np.clip(h_try[j] - diff, 0, 1)
                        # If j couldn't absorb all, spread remainder
                        rem = S - np.sum(h_try)
                        if abs(rem) > 1e-10:
                            h_try = proj(h_try, S)
                    else:
                        h_try = proj(h_try, S)
                    v = c5_eval(h_try)
                    if v < c5_eval(h) - 1e-10:
                        h = h_try
                        improved = True
                        if v < best_c5:
                            best_c5 = v
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 5: Try binary relaxation
    if tl() > 30:
        h_bin = (best_h > 0.5).astype(float)
        s_bin = int(np.sum(h_bin))
        if s_bin != S:
            # Adjust to correct sum
            if s_bin > S:
                order = np.argsort(h_bin)
                for i in range(len(order)):
                    if h_bin[order[i]] > 0:
                        h_bin[order[i]] = 0
                        if s_bin == S:
                            break
                        s_bin -= 1
            else:
                order = np.argsort(-best_h)
                for i in range(len(order)):
                    if h_bin[order[i]] < 1:
                        h_bin[order[i]] = 1
                        if s_bin == S:
                            break
                        s_bin += 1
        h_bin = proj(h_bin, S)
        v_bin = c5_eval(h_bin)
        if v_bin < best_c5:
            best_c5 = v_bin
            best_h = h_bin.copy()

    # Phase 6: Final polish with very small steps
    if tl() > 20:
        h = best_h.copy()
        lr = 0.002
        for it in range(3000):
            if tl() < 10:
                break
            g, lag, cv = subgrad_dir(h)
            dx = 2.0 / n
            h_new = proj(h - lr * g * dx, S)
            v_new = c5_eval(h_new)
            if v_new < best_c5:
                best_c5 = v_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.01, 0.02)
            else:
                lr = max(lr * 0.99, 1e-6)
                if lr < 2e-6:
                    break

    # Ensure feasibility
    best_h = proj(best_h, S)
    # Verify
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = c5_eval(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
