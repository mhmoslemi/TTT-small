def project_sum_bound(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h,0,1)
    for _ in range(10):
       res = target - h.sum()
       if abs(res) < 1e-12: break
       if res > 0:
          mask = h < 1.0 - 1e-12
          if not mask.any(): break
          amt = res / mask.sum()
          if amt < 1e-12: break
          add = np.minimum(1.0 - h[mask], amt)
          h[mask] += add
       else:
          mask = h > 1e-12
          if not mask.any(): break
          amt = -res / mask.sum()
          sub = np.minimum(h[mask], amt)
          h[mask] -= sub
    # final exact adjust on one slack element if possible

       elif res<0 and np.any(h>0): i=np.argmin(h); h[i]=max(0,h[i]+res)

def active_gradient(h, idxs, ws=None):
    n=len(h)
    if ws is None:
        ws=np.ones(len(idxs))/len(idxs)
    cx=n-1
    dx=2.0/n
    grad=np.zeros(n)
    for i in range(len(idxs)):
        w=float(ws[i])
        if w==0: continue
        k=int(idxs[i])-cx
        if k>=0:
            m=n-k
            grad[k:n] += w*(1.0 - h[:m])
            grad[:m] -= w*h[k:n]

            grad[:m] += w*(1.0 - h[a:n])
            grad[a:n] -= w*h[:m]
    grad *= dx
    return grad

def project_sum_bound(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(10):
        r = float(target - np.sum(h))
        if abs(r) < 1e-13:
            break
        if r > 0:
            mask = h < 1.0 - 1e-15
            if mask.any():
                h[mask] += np.minimum(1.0 - h[mask], r / float(mask.sum()))
        else:
            mask = h > 1e-15
            if mask.any():
                h[mask] -= np.minimum(h[mask], -r / float(mask.sum()))
    r = float(target - np.sum(h))
    if r > 1e-13:
        if np.any(h < 1.0):
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + r)
    elif r < -1e-13:
        if np.any(h > 0.0):
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + r)
    return h

def fft_convolve(a,b):
    nconv=len(a)+len(b)-1
    size=1
    while size < nconv:
        size <<= 1
    fa=np.fft.rfft(a,size)
    fb=np.fft.rfft(b,size)
    y=np.fft.irfft(fa*fb,size)
    return y[:nconv]

def active_gradient(h, idxs, ws=None):
    n = len(h)
    m = len(idxs)
    if ws is None:
        ws = np.ones(m, dtype=float) / max(1, m)
    cx = n - 1
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    for i in range(m):
        w = float(ws[i])
        if w == 0.0:
            continue
        k = int(idxs[i]) - cx
        if k >= 0:
            length = n - k
            grad[k:n] += w * (1.0 - h[:length])
            grad[:length] -= w * h[k:n]
        else:
            a = -k
            length = n - a
            grad[:length] += w * (1.0 - h[a:n])
            grad[a:n] -= w * h[:length]
    grad *= dx
    return grad

import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

_ST = {}

class _TimeoutErr(Exception):
    pass


def eval_counts(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def eval_c5(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(eval_counts(h)) * (2.0 / len(h)))


def project_sum_bound(h, target):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)

    for _ in range(10):
        r = float(target - np.sum(h))
        if abs(r) < 1.0e-13:
            break
        if r > 0.0:
            mask = h < 1.0 - 1.0e-15
            if mask.any():
                h[mask] += np.minimum(1.0 - h[mask], r / float(mask.sum()))
        else:
            mask = h > 1.0e-15
            if mask.any():
                h[mask] -= np.minimum(h[mask], -r / float(mask.sum()))

    r = float(target - np.sum(h))
    if r > 1.0e-13:
        if np.any(h < 1.0):
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + r)
    elif r < -1.0e-13:
        if np.any(h > 0.0):
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + r)
    return h


def upsample(h, n_new, mode="nearest"):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n_new == n:
        return project_sum_bound(h, 0.5 * n_new)

    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    if mode == "linear":
        x_old = (np.arange(n) + 0.5) / float(n)
        h_new = np.interp(x_new, x_old, h)
    else:
        idx = np.floor(x_new * float(n)).astype(int)
        idx = np.clip(idx, 0, n - 1)
        h_new = h[idx]

    return project_sum_bound(h_new, 0.5 * n_new)


def make_fallback(seed):
    n = 42
    i = np.arange(n)
    s = np.sin(2.0 * np.pi * i / float(n) - 0.3)
    s += 0.5 * np.sin(4.0 * np.pi * i / float(n) + float(seed) * 0.17)
    h = 0.5 + 0.33 * s
    return project_sum_bound(h, 0.5 * n)


def fft_convolve(a, b):
    nconv = len(a) + len(b) - 1
    size = 1
    while size < nconv:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    y = np.fft.irfft(fa * fb, size)
    return y[:nconv]


def _sum_fun(x):
    return float(np.sum(x) - 0.5 * _ST["n"])


def _sum_jac(x):
    return np.ones(int(_ST["n"]), dtype=float)


def _obj(x):
    if time.time() > _ST["deadline"]:
        raise _TimeoutErr

    h = np.asarray(x, dtype=float)
    T = float(_ST["T"])
    dx = float(_ST["dx"])

    c = fft_convolve(h, (1.0 - h)[::-1])
    v = c * dx
    if not np.all(np.isfinite(v)):
        return 1.0e6

    mv = float(np.max(v))
    z = (v - mv) / T
    return float(mv + T * np.log(np.sum(np.exp(z))))


def _jac(x):
    if time.time() > _ST["deadline"]:
        raise _TimeoutErr

    n = int(_ST["n"])
    T = float(_ST["T"])
    dx = float(_ST["dx"])

    h = np.asarray(x, dtype=float)
    c = fft_convolve(h, (1.0 - h)[::-1])
    v = c * dx

    if not np.all(np.isfinite(v)):
        return np.zeros(n, dtype=float)

    mv = float(np.max(v))
    p = np.exp((v - mv) / T)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros(2 * n - 1, dtype=float)
        p[int(np.argmax(v))] = 1.0
    else:
        p /= s

    c0 = n - 1
    ones = np.ones(n, dtype=float)

    a = fft_convolve(ones, p)
    b = fft_convolve(h, p)
    d = fft_convolve(h, p[::-1])

    g = dx * (a[c0:c0 + n] - b[c0:c0 + n] - d[c0:c0 + n])
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=float)
    return g


def run_slsqp(h, T, maxiter, deadline):
    n = len(h)
    h = project_sum_bound(h, 0.5 * n)

    _ST["n"] = n
    _ST["T"] = float(T)
    _ST["dx"] = 2.0 / float(n)
    _ST["deadline"] = float(deadline)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_fun, "jac": _sum_jac}]
    opts = {"maxiter": int(maxiter), "ftol": 1.0e-9}

    try:
        res = minimize(
            _obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=_jac,
            options=opts,
        )
        if hasattr(res, "x") and res.x is not None:
            return project_sum_bound(res.x, 0.5 * n)
    except Exception:
        pass

    return h


def _de_obj(x):
    n = len(x)
    h = project_sum_bound(x, 0.5 * n)
    return eval_c5(h)


def active_gradient(h, idxs, ws=None):
    n = len(h)
    m = len(idxs)
    if ws is None:
        ws = np.ones(m, dtype=float) / max(1, m)

    cx = n - 1
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)

    for i in range(m):
        w = float(ws[i])
        if w == 0.0:
            continue
        k = int(idxs[i]) - cx
        if k >= 0:
            length = n - k
            grad[k:n] += w * (1.0 - h[:length])
            grad[:length] -= w * h[k:n]
        else:
            a = -k
            length = n - a
            grad[:length] += w * (1.0 - h[a:n])
            grad[a:n] -= w * h[:length]

    grad *= dx
    return grad


def polish(h, seconds, rng):
