def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(z, target_sum):
    """Project z onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    n = len(z)
    lo = np.min(z) - 1.0 - 1e-10
    hi = np.max(z) + 1e-10
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical error
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        pass
    else:
        # Add/subtract from the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def subgradient_obj(h):
    """Compute C5 and subgradient (direction) for h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    # Subgradient: g[m] = (1-h[m+lag]) - h[m-lag] (with out-of-bounds = 0)
    g = np.zeros(n)
    # (1-h[m+lag]) term: valid when 0 <= m+lag < n, i.e., -lag <= m < n-lag

    lo2 = max(0, lag)

def smooth_max_obj(h, tau=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = corr * dx
    m = np.max(obj)
    # softmax weights
    shifted = tau * (obj - m)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Gradient: weighted sum of subgradients
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            g[lo1:hi1] += weights[k] * dx * (1.0 - h[lo1+lag:hi1+lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] -= weights[k] * dx * h[lo2-lag:hi2-lag]
    return m + np.log(np.sum(weights)) / tau, g

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(z, target):
    n = len(z)
    if np.sum(z) == target and np.all(z >= -1e-15) and np.all(z <= 1+1e-15):
        return np.clip(z, 0, 1)
    lo = np.min(z) - 1.5
    hi = np.max(z) + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # fix tiny drift
    d = target - h.sum()
    if abs(d) > 1e-12:
        j = np.argmax(h) if d > 0 else np.argmin(h)
        # distribute to many close-to-boundary elements
        h[j] += d
        h[j] = min(max(h[j], 0.0), 1.0)
    return h

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = corr * dx
    m = obj.max()
    # softmax
    s = tau * (obj - m)
    w = np.exp(s - s.max())
    w /= w.sum()
    val = m + np.log(w.sum() * np.exp(s.max())) / tau  # this is just m + log(sum(exp(s)))/tau
    # Actually val = m + log(sum(exp(tau*(obj-m))))/tau
    # Let me just compute it directly
    log_sum = s.max() + np.log(np.exp(s - s.max()).sum())
    val = m + log_sum / tau
    # gradient
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        # d(corr[k])/dh[m] = (1-h[m-lag])*[0<=m-lag<n] - h[m+lag]*[0<=m+lag<n]
        a = max(0, -lag)
        b = min(n, n - lag)

        a2 = max(0, lag)
        b2 = min(n, n + lag)
        if a2 < b2:

def _eval(h):
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _proj(z, tgt):
    lo = float(z.min()) - 1.5
    hi = float(z.max()) + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

def _sgrad(h, tau):
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    o = c * dx
    m = o.max()
    s = tau * (o - m)
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    val = m + (sm + np.log(e.sum())) / tau
    g = np.zeros(n)
    for k in range(c.size):
        if w[k] < 1e-13:
            continue
        ell = k - n + 1
        wt = w[k] * dx
        a = max(0, ell)
        b = min(n, n + ell)
        if a < b:
            g[a:b] += wt * (1.0 - h[a - ell:b - ell])
        a2 = max(0, -ell)
        b2 = min(n, n - ell)
        if a2 < b2:
            g[a2:b2] -= wt * h[a2 + ell:b2 + ell]
    return val, g
