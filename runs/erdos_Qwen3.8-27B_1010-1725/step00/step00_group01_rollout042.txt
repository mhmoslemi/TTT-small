def make_feasible(v, n):
    x = np.asarray(v, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    # Duchi projection
    order = np.argsort(x)[::-1]
    xsorted = x[order]
    rho = np.arange(1, n+1, dtype=float)
    cssv = np.cumsum(xsorted)
    cond = xsorted - (cssv - target) / rho > 0.0
    if not np.any(cond):
        theta = (x.sum() - target) / n
    else:
        rho0 = int(np.max(np.nonzero(cond)[0]))
        theta = (cssv[rho0] - target) / (rho0 + 1.0)
    y = np.clip(x - theta, 0.0, 1.0)

    for _ in range(12):

        if abs(diff) <= 1e-13:

            head = 1.0 - y

        else:
            foot = y
            idx = int(np.argmax(foot))
            sub = min(abs(diff) * 0.5, foot[idx])
            y[idx] -= sub
    return y

import time
import numpy as np
from scipy.optimize import minimize


def make_feasible(v, n):
    x = np.asarray(v, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(x - tau, 0.0, 1.0)
    return fix_sum_small(y, n)


def fix_sum_small(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    for _ in range(12):
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            if add <= 1e-18:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            if sub <= 1e-18:
                break
            h[idx] -= sub
    if abs(target - float(np.sum(h))) > 1e-9:
        return make_feasible(h, n)
    return h


def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    return float(np.max(c))


def scaled_c(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def grad_lag(h, s):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if s == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        g[:-s] += 1.0 - h[s:]
        g[s:] -= h[:-s]
    else:
        r = -s
        g[r:] += 1.0 - h[:-r]
        g[:-r] -= h[r:]
    return g


def objective_t(x):
    return float(x[-1])


def objective_t_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def equality_constraint(x):
    n = len(x) - 1
    return float(np.sum(x[:-1]) - 0.5 * float(n))


def equality_constraint_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def inequality_constraints(x):
    n = len(x) - 1
    h = x[:-1]
    t = x[-1]
    return t - np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def inequality_constraints_jac(x):
    n = len(x) - 1
    h = x[:-1]
    dx = 2.0 / float(n)
    m = 2 * n - 1
    j = np.zeros((m, n + 1), dtype=float)
    for k in range(m):
        s = n - 1 - k
        j[k, :-1] = -dx * grad_lag(h, s)
        j[k, -1] = 1.0
    return j


def slsqp_optimize(h, maxiter=300, ftol=1e-11):
    h = make_feasible(h, len(h))
    n = len(h)
    c = eval_c5(h)
    t0 = min(2.0, c + 1e-7)
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {"type": "eq", "fun": equality_constraint, "jac": equality_constraint_jac},
        {"type": "ineq", "fun": inequality_constraints, "jac": inequality_constraints_jac},
    ]
    try:
        res = minimize(
            objective_t,
            x0,
            jac=objective_t_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "maxls": 20, "disp": False},
        )
        y = res.x[:n]
    except Exception:
        y = h
    y = make_feasible(y, n)
    return y, eval_c5(y)


def poly_search(lo, hi, c0, L, Q, samples=41, max_rounds=3):
    if hi - lo <= 1e-11:
        return 0.0, float(np.max(c0))
    if samples < 2:
        samples = 2
    a = np.linspace(lo, hi, int(samples), dtype=float)
    A = a[:, None]
    vals = np.max(c0[None, :] + A * L[None, :] + (A * A) * Q[None, :], axis=1)
    idx = int(np.argmin(vals))
    best_a = float(a[idx])
    best_c = float(vals[idx])
    span = hi - lo
    step = span / float(max(1, int(samples) - 1))
    for _ in range(int(max_rounds)):
        lo2 = max(lo, best_a - step)
        hi2 = min(hi, best_a + step)
        if hi2 - lo2 <= 1e-8 or step <= 1e-8:
            break
        a = np.linspace(lo2, hi2, int(samples), dtype=float)
        A = a[:, None]
        vals = np.max(c0[None, :] + A * L[None, :] + (A * A) * Q[None, :], axis=1)
        idx = int(np.argmin(vals))
        na = float(a[idx])
        nc = float(vals[idx])
        step = (hi2 - lo2) / float(max(1, int(samples) - 1))
        if nc < best_c:
            best_a = na
            best_c = nc
        else:
            break
    return best_a, best_c


def pair_sweep(h, max_passes, rng, deadline):
    n = len(h)
    h = make_feasible(h, n)
    dx = 2.0 / float(n)
    base_c = scaled_c(h)
    current = float(np.max(base_c))
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for _ in range(int(max_passes)):
        if time.time() >= deadline:
            break
        order = list(range(len(pairs)))
        rng.shuffle(order)
        improved = False
        for idx in order:
            if time.time() >= deadline:
                return h, current
            i, j = pairs[idx]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], -h[j])
            if hi - lo <= 1e-10:
                continue
            d = np.zeros(n, dtype=float)
            d[i] = 1.0
            d[j] = -1.0
            cp = np.correlate(h + d, 1.0 - h - d, mode="full") * dx
            cm = np.correlate(h - d, 1.0 - h + d, mode="full") * dx
            L = (cp - cm) * 0.5
            Q = (cp + cm - 2.0 * base_c) * 0.5
            best_a, best_val = poly_search(lo, hi, base_c, L, Q, 41, 3)
            if best_val < current - 1e-12:
                h2 = h.copy()
                h2[i] += best_a
                h2[j] -= best_a
                np.clip(h2, 0.0, 1.0, out=h2)
                h2 = fix_sum_small(h2, n)
                c2 = eval_c5(h2)
                if c2 < current - 1e-12:
                    h = h2
                    base_c = scaled_c(h)
                    current = float(c2)
                    improved = True
        if not improved:
            break
    return h, current


def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / float(n)
    c = scaled_c(h)
    cmax = float(np.max(c))
    z = (c - cmax) / temp
    ez = np.exp(z)
    wsum = float(np.sum(ez))
    w = ez / wsum
    val = cmax + temp * np.log(wsum)
    g = np.zeros(n, dtype=float)
    for k, wm in enumerate(w):
        if wm > 1e-6:
            s = n - 1 - k
            g += wm * grad_lag(h, s)
    return val, g * dx


def smooth_refine(h, current, rng, deadline, max_evals=2500):
    n = len(h)
    h = make_feasible(h, n)
    current = eval_c5(h)
    best_h = h.copy()
    best_c = float(current)
    temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    alphas = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002]
    eval_count = 0
    for temp in temps:
        if time.time() >= deadline or eval_count >= max_evals:
            break
        stalled = 0
        for _ in range(1200):
            if time.time() >= deadline or eval_count >= max_evals:
                break
            _, g = smooth_gradient(h, float(temp))
            d = -g
            d -= np.mean(d)
            nn = float(np.linalg.norm(d))
            if nn <= 1e-9:
                break
            d /= nn
            improved = False
            for alpha in alphas:
                h2 = make_feasible(h + float(alpha) * d, n)
                eval_count += 1
                c2 = eval_c5(h2)
                if c2 < current - 1e-12:
                    h = h2
                    current = float(c2)
                    improved = True
                    break
            if improved:
                if current < best_c:
                    best_c = current
                    best_h = h.copy()
                stalled = 0
            else:
                h2 = make_feasible(h + rng.normal(0.0, 0.01, size=n), n)
                eval_count += 1
                c2 = eval_c5(h2)
                if c2 < current - 1e-12:
                    h = h2
                    current = float(c2)
                    if current < best_c:
                        best_c = current
                        best_h = h.copy()
                else:
                    stalled += 1
                    if stalled > 20:
                        break
    return best_h, best_c


def random_start(n, rng):
    x = np.zeros(n, dtype=float)
    if n % 2 == 0:
        k = n // 2
        idx = rng.choice(n, size=k, replace=False)
        x[idx] = 1.0
    else:
        k = n // 2
        idx = rng.choice(n, size=k, replace=False)
        x[idx] = 1.0
        rem = np.where(x == 0.0)[0]
        if rem.size:
            x[rem[int(rng.integers(rem.size))]] = 0.5
    return make_feasible(x, n)


def perturb_h(h, scale, rng):
    n = len(h)
    x = h.copy()
    x += rng.normal(0.0, scale, size=n)
    if n > 1 and rng.random() < 0.6:
        i, j = rng.choice(n, size=2, replace=False)
        a = rng.uniform(-0.25, 0.25)
        x[i] += a
        x[j] -= a
    return make_feasible(x, n)


def interpolate_h_to_length(h, m):
    h = np.asarray(h, dtype=float)
    n = len(h)
    m = int(m)
    if m == n:
        return h.copy()
    xo = np.arange(n, dtype=float)
    xn = np.linspace(0.0, float(n - 1), m, dtype=float)
    y = np.interp(xn, xo, h)
    return make_feasible(y, m)


def better_update(best_h, best_c, h, c):
    if c < best_c - 1e-14:
        return np.array(h, dtype=float, copy=True), float(c)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + max(1.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(int(seed))

    if "initial_h_values" in globals():
        h = np.array(globals()["initial_h_values"], dtype=float, copy=True)
    else:
        h = random_start(47, rng)

    h = make_feasible(h, len(h))
    best_h = h.copy()
    best_c = eval_c5(h)

    best_h, best_c = better_update(best_h, best_c, h, best_c)

    if time.time() < deadline:
        h, c = pair_sweep(best_h, 3, rng, deadline)
        best_h, best_c = better_update(best_h, best_c, h, c)

    for _ in range(3):
        if time.time() > deadline - 160.0:
            break
        h, c = slsqp_optimize(best_h, 350, 1e-11)
        best_h, best_c = better_update(best_h, best_c, h, c)
        h, c = pair_sweep(best_h, 4, rng, deadline)
        best_h, best_c = better_update(best_h, best_c, h, c)
        if time.time() > deadline - 80.0:
            break

    if time.time() < deadline - 120.0:
        h, c = smooth_refine(best_h, best_c, rng, deadline, 2200)
        best_h, best_c = better_update(best_h, best_c, h, c)

    scales = [0.005, 0.01, 0.02, 0.05, 0.10, 0.20]
    for scale in scales:
        for rep in range(2):
            if time.time() > deadline - 120.0:
                break
            hp = perturb_h(best_h, float(scale), rng)
            hp, c = slsqp_optimize(hp, 250, 1e-10)
            best_h, best_c = better_update(best_h, best_c, hp, c)
            if time.time() > deadline - 70.0:
                break
            hp, c = pair_sweep(best_h, 3, rng, deadline)
            best_h, best_c = better_update(best_h, best_c, hp, c)

    target_ns = [49, 50, 48, 52, 55, 60]
    for m in target_ns:
        if time.time() > deadline - 150.0:
            break
        hm = interpolate_h_to_length(best_h, m)
        c0 = eval_c5(hm)
        if c0 < best_c + 0.03:
            hm, c = slsqp_optimize(hm, 280, 1e-10)
            best_h, best_c = better_update(best_h, best_c, hm, c)
            if time.time() > deadline - 60.0:
                continue
            hm, c = pair_sweep(best_h, 3, rng, deadline)
            best_h, best_c = better_update(best_h, best_c, hm, c)

    if time.time() < deadline - 10.0:
        h, c = pair_sweep(best_h, 4, rng, deadline)
        best_h, best_c = better_update(best_h, best_c, h, c)

    best_h = make_feasible(best_h, len(best_h))
    best_c = eval_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
