```
def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    e = np.exp(np.clip(beta*(c-m), -50.0, 0.0))
    threshold = e.sum() * 1e-8
    g = np.zeros(n, dtype=float)
    off = n-1
    if n <= 220:
        itr = np.nonzero(e > threshold)[0]
    else:
        # top k
        k = min(80, len(e))
        itr = np.argsort(e)[-k:]
        # optionally threshold
    scale = dx / beta
    for ii in itr:
        w = e[ii]
        if w <= 1e-14: continue
        d = ii - off
        if d >= 0:
            if d == 0:
                g[:] += w * (1.0 - 2.0*h)
            else:
                g[d:] += w * (1.0 - h[:n-d])
                g[:n-d] -= w * h[d:]
        else:
            p = -d
            g[:n-p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n-p]
    return scale*g
```

```python
def smooth_grad(h, beta=20.0):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')

        se = float(np.sum(e))

            e = np.zeros(len(c), dtype=float)
            e[int(np.argmax(c))] = 1.0
            se = 1.0
```

```python
    g = np.zeros(n, dtype=float)
    off = n - 1
    for ii in np.nonzero(e)[0]:
        w = float(e[ii]) / s
        if w <= 1e-12:
            continue
        d = int(ii) - off
        if d >= 0:
            if d < n:
                g[d:] += w * omh[:n - d]
                g[:n - d] -= w * h[d:]
        else:
            p = -d
            if p < n:
                g[:n - p] += w * omh[p:]
                g[p:] -= w * h[:n - p]
    return dx * g


def seq(h):
    return float(np.sum(h)) - EQ


def seu(h):
    return np.ones_like(np.asarray(h, dtype=float), dtype=float)


def opt(h0, beta=20.0, mit=80):
    h0 = proj(h0, len(h0))
    n = len(h0)
    global EQ
    EQ = 0.5 * float(n)
    if n == 0:
        return h0
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": seq, "jac": seu}]
    try:
        r = minimize(
            sob, h0, method="SLSQP", jac=sog, args=(float(beta),),
            bounds=bnds, constraints=cons,
            options={"ftol": 1e-10, "maxiter": int(mit), "disp": False}
        )
    except Exception:
        return h0
    if r is not None and getattr(r, "x", None) is not None and C5(r.x) < 1e90:
        return proj(r.x, n)
    return h0


def rs(v, n):
    v = np.asarray(v, dtype=float).ravel()
    n = int(n)
    if n == 0:
        return v
    m = len(v)
    if m == 0:
        return proj(np.full(n, 0.5), n)
    if m == n:
        return proj(v, n)
    x0 = (np.arange(m) + 0.5) * (2.0 / float(m))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    return proj(np.interp(x1, x0, v), n)


def shape(n, kind=0, p=2.0):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    p = max(0.05, float(p))
    if kind == 0:
        h = np.sin(np.pi * x / 2.0) ** p
    elif kind == 1:
        r = np.sin(np.pi * x / 2.0) ** p
        M = float(r.mean())
        if M <= 0.5:
            h = 1.0 - r
        else:
            h = 1.0 - r / (2.0 * M)
    else:
        h = np.full(n, 0.5)
    return proj(h, n)


def pobj(x, h, j, l, T):
    hh = h.copy()
    hh[j] = x
    hh[l] = T - x
    return C5(hh)


def tpair(cur, j, l, cc):
    if j == l:
        return cur, cc, False
    T = float(cur[j] + cur[l])
    lo = max(0.0, T - 1.0)
    hi = min(1.0, T)
    if hi - lo < 1e-7:
        return cur, cc, False
    try:
        r = minimize_scalar(
            pobj, bounds=(lo, hi), args=(cur, j, l, T),
            method="bounded", options={"maxiter": 20, "xatol": 1e-5}
        )
    except Exception:
        return cur, cc, False
    if not r.success:
        return cur, cc, False
    x = float(r.x)
    y = T - x
    if x < -1e-12 or x > 1.0 + 1e-12 or y < -1e-12 or y > 1.0 + 1e-12:
        return cur, cc, False
    if float(r.fun) >= cc - 1e-12:
        return cur, cc, False
    x = min(max(x, 0.0), 1.0)
    y = min(max(y, 0.0), 1.0)
    nc = cur.copy()
    nc[j] = x
    nc[l] = y
    v = C5(nc)
    if v < cc - 1e-12:
        return nc, v, True
    return cur, cc, False


def pdesc(hin, bud, seed):
    t = time.time()
    bud = max(0.0, float(bud))
    n = len(hin)
    cur = proj(hin, n)
    cc = C5(cur)
    best = cur.copy()
    bc = cc
    rg = np.random.default_rng(int(seed))
    mp = n if n <= 200 else 80
    passes = 0
    stuck = 0
    while time.time() - t < bud and bud > 0.0:
        passes += 1
        imp = False
        if passes % 3 == 0:
            for i in range(mp):
                j = int(rg.integers(0, n))
                l = int(rg.integers(0, n))
                if j == l:
                    l = (l + 1) % n
                cur, cc, ok = tpair(cur, j, l, cc)
                if ok:
                    imp = True
                if cc < bc - 1e-12:
                    best = cur.copy()
                    bc = cc
                if time.time() - t > bud - 1.0:
                    break
        else:
            st = 1 + (passes % 7)
            od = rg.permutation(n)
            for i in range(min(mp, n)):
                j = int(od[i])
                l = (j + st) % n
                cur, cc, ok = tpair(cur, j, l, cc)
                if ok:
                    imp = True
                if cc < bc - 1e-12:
                    best = cur.copy()
                    bc = cc
                if time.time() - t > bud - 1.0:
                    break
        if not imp:
            stuck += 1
            sc = max(1e-5, 1e-3 * (0.6 ** float(min(stuck, 6))))
            pn = proj(cur + rg.normal(0.0, sc, n), n)
            pc = C5(pn)
            if pc < bc + 0.02:
                cur = pn
                cc = pc
            else:
                cur = best.copy()
                cc = bc
            if cc < bc - 1e-12:
                best = cur.copy()
                bc = cc
    return best, bc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + float(budget_s)
    rg = np.random.default_rng(int(seed))

    try:
        pa = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        pa = None

    if pa is not None and len(pa) > 0 and np.all(np.isfinite(pa)):
        bh = proj(pa, len(pa))
        bc = C5(bh)
    else:
        bh = shape(96, 0, 2.0)
        bc = C5(bh)
        pa = bh

    if len(bh) == 0:
        bh = np.full(96, 0.5)
        bc = C5(bh)
        pa = bh

    for beta in (10.0, 30.0, 80.0, 200.0):
        if dl - time.time() < 40.0:
            break
        h = opt(bh.copy(), beta, 80 if len(bh) <= 128 else 50)
        v = C5(h)
        if v < bc - 1e-12:
            bh = h.copy()
            bc = v

    for s in (0.02, 0.05, 0.10):
        if dl - time.time() < 70.0:
            break
        hs = proj(bh + rg.normal(0.0, float(s), len(bh)), len(bh))
        for beta in (20.0, 80.0):
            if dl - time.time() < 45.0:
                break
            hs = opt(hs, beta, 45)
            v = C5(hs)
            if v < bc - 1e-12:
                bh = hs.copy()
                bc = v

    base = pa
    for nar in (32, 64, 96, 128, 192, 256):
        if dl - time.time() < 100.0:
            break
        hs = rs(base, nar)
        for beta in (30.0, 100.0):
            if dl - time.time() < 50.0:
                break
            hs = opt(hs, beta, 60 if nar <= 128 else 40)
            v = C5(hs)
            if v < bc - 1e-12:
                bh = hs.copy()
                bc = v

    ss = ((32, 0, 2.0), (32, 1, 2.0), (64, 1, 2.0), (96, 0, 2.0),
          (96, 1, 2.0), (128, 1, 1.8), (192, 0, 2.5))
    for a, b, c in ss:
        if dl - time.time() < 80.0:
            break
        h = shape(a, b, c)
        for beta in (20.0, 90.0):
            if dl - time.time() < 40.0:
                break
            h = opt(h, beta, 45)
            v = C5(h)
            if v < bc - 1e-12:
                bh = h.copy()
                bc = v

    r = dl - time.time()
    if len(bh) <= 255 and r > 60.0:
        hp, cp = pdesc(bh, min(120.0, r - 30.0), int(seed) + 11)
        if cp < bc - 1e-12:
            bh = hp
            bc = cp

    r = dl - time.time()
    if len(bh) <= 160 and r > 40.0:
        hf = opt(bh, 150.0, 40)
        v = C5(hf)
        if v < bc - 1e-12:
            bh = hf
            bc = v

    r = dl - time.time()
    if len(bh) <= 191 and r > 50.0:
        hp, cp = pdesc(bh, min(60.0, r - 10.0), int(seed) + 23)
        if cp < bc - 1e-12:
            bh = hp
            bc = cp

    bh = proj(bh, len(bh))
    bc = C5(bh)
    return bh, float(bc), len(bh)
```
