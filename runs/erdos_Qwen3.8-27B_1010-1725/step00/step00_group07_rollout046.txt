import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto [0,1]^n with sum = n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(np.abs(h - 0.5))  # adjust the most "free" value
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

    x_to = np.linspace(0, 1, n_to)
