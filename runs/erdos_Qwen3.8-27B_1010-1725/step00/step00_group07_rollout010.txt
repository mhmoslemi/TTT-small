def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def adjust_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            if not m.any():
                break
            step = min(d, (1.0 - h[m]).sum()) / m.sum()
            h[m] += step
        else:
            m = h > 0.0
            if not m.any():
                break
            step = min(-d, h[m].sum()) / m.sum()
            h[m] -= step
    h = np.clip(h, 0.0, 1.0)
    return h
