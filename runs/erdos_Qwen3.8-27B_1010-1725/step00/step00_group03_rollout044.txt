    """Compute c[m] for all lags, return the max."""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def c5_gradient(h, n):
    """Compute subgradient of C5 (w.r.t. h, before dx scaling).
    Uses the argmax subgradient."""
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m_star = np.argmax(c)
    k = m_star - n + 1  # shift
    
    grad = np.zeros(n)
    # dc[m*]/dh[j] = (1-h[j+k]) if 0<=j+k<n minus h[j-k] if 0<=j-k<n
    # First term: for j where 0 <= j+k < n
    if k >= 0:
        # j+k < n => j < n-k, and j >= 0
        js = np.arange(0, n-k)
        grad[js] += (1.0 - h[js + k])
    else:
        # j+k >= 0 => j >= -k, and j+k < n => j < n
        js = np.arange(-k, n)
        grad[js] += (1.0 - h[js + k])
    
    # Second term: -h[j-k] where 0 <= j-k < n
    if k >= 0:
        # j-k >= 0 => j >= k, and j-k < n => j < n+k (always true since j<n)
        js = np.arange(k, n)
        grad[js] -= h[js - k]
    else:

    return h

def upsample(h, new_n):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    f = interp1d(x_old, h, kind='linear', assume_sorted=True)
    h_new = f(x_new)
    return project_constraint(h_new, new_n)

def optimize_h(h0, n, dx, maxiter=400, ftol=1e-10):
    def obj(x):
        return c5_value(x, dx)
    def obj_grad(x):
        return c5_subgradient(x, n) * dx
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_constraint(res.x, n), res.fun if hasattr(res, 'fun') else c5_value(res.x, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_init = np.asarray(initial_h_values, dtype=float)
    n_init = len(h_init)

    # Start at n=140
    n = 140
    h = upsample(h_init, n)
    dx = 2.0 / n
    best_c5 = c5_value(h, dx)
    best_h = h.copy()

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Local refinement at n=140
    noises = [0.0, 0.02, 0.04, 0.06, 0.10, 0.15]
    for i, noise in enumerate(noises):
        if tl() < 40:
            break
        if noise == 0.0:
            h0 = best_h.copy()
        else:
            h0 = best_h + rng.normal(0, noise, n)
            h0 = project_constraint(h0, n)
        try:
            h_c, c5_c = optimize_h(h0, n, dx, maxiter=350, ftol=1e-9)
            c5_c = c5_value(h_c, dx)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
        except Exception:
            pass

    # Phase 2: Try n=200
    if tl() > 80:
        n2 = 200
        h2 = upsample(best_h, n2)
        dx2 = 2.0 / n2
        c5_2 = c5_value(h2, dx2)
        if c5_2 < best_c5 * 1.005:
            for noise in [0.0, 0.02, 0.05]:
                if tl() < 50:
                    break
                h0 = h2 + rng.normal(0, noise, n2) if noise > 0 else h2.copy()
                h0 = project_constraint(h0, n2)
                try:
                    h_c, _ = optimize_h(h0, n2, dx2, maxiter=250, ftol=1e-9)
                    c5_c = c5_value(h_c, dx2)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()
                        n = n2
                except Exception:
                    pass

    # Phase 3: Try n=280 if time permits
    if tl() > 100:
        n3 = 280
        h3 = upsample(best_h, n3)
        dx3 = 2.0 / n3
        c5_3 = c5_value(h3, dx3)
        if c5_3 < best_c5 * 1.005:
            for noise in [0.0, 0.02]:
                if tl() < 40:
                    break
                h0 = h3 + rng.normal(0, noise, n3) if noise > 0 else h3.copy()
                h0 = project_constraint(h0, n3)
                try:
                    h_c, _ = optimize_h(h0, n3, dx3, maxiter=200, ftol=1e-9)
                    c5_c = c5_value(h_c, dx3)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()
                        n = n3
                except Exception:
                    pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_constraint(best_h, len(best_h))
    final_c5 = c5_value(best_h, 2.0 / len(best_h))

    return best_h, final_c5, len(best_h)
