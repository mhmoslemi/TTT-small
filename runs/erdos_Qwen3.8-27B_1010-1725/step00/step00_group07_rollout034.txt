    target = n / 2.0

        s = np.sum(h)
        diff = target - s

            rsum = np.sum(room)

            rsum = np.sum(room)
            if rsum < 1e-15:

            h = h + diff * room / rsum

            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Gradient optimization at initial resolution
    h, c5 = optimize_grad(h0, n0, tau=60, lr=10.0, max_iter=800, t0=t0, bud=budget_s)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Binary local search then refine
    if rem() > 150:
        h, c5 = binary_search(best_h, n0, rng, 5000, t0, budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        if rem() > 80:
            h2, c52 = optimize_grad(h, n0, tau=40, lr=5.0, max_iter=500, t0=t0, bud=budget_s)
            if c52 < best_c5:
                best_c5, best_h = c52, h2.copy()

    # Phase 3: Multiple gradient passes with varying parameters
    params = [(30, 15.0), (80, 5.0), (20, 20.0), (100, 3.0), (50, 8.0)]
    for tau, lr in params:
        if rem() < 50:
            break
        h, c5 = optimize_grad(best_h, n0, tau=tau, lr=lr, max_iter=500, t0=t0, bud=budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Higher resolution attempts
    if rem() > 200:
        for n_try in [n0 * 2, n0 * 4]:
            if rem() < 80 or n_try > 800:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_int = project_h(h_int, n_try)
            h, c5 = optimize_grad(h_int, n_try, tau=40, lr=3.0, max_iter=1500, t0=t0, bud=budget_s)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 5: Diverse random restarts
    for r in range(15):
        if rem() < 40:
            break
        n = n0
        if r % 4 == 0:
            h = project_h(best_h + rng.normal(0, 0.25, n), n)
        elif r % 4 == 1:
            h = project_h(rng.rand(n), n)
        elif r % 4 == 2:
            f = rng.uniform(0.5, 6.0)
            h = 0.5 + 0.45 * np.sin(2.0 * np.pi * f * np.arange(n) / n)
            h = project_h(h, n)
        else:
            # Binary with structure
            period = rng.randint(3, 15)
            h = np.tile([1.0] * rng.randint(1, period) + [0.0] * period, n // period + 1)[:n]
            h = project_h(h, n)
        h, c5 = optimize_grad(h, n, tau=40, lr=8.0, max_iter=400, t0=t0, bud=budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Final fine-tuning
    if rem() > 15:
        n_f = len(best_h)
        h, c5 = optimize_grad(best_h, n_f, tau=150, lr=2.0, max_iter=500, t0=t0, bud=budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
