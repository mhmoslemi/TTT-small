import time

def c5_value(h):
    """Compute C5 for given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr * dx)

def project_onto_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    current_sum = np.sum(h)
    if current_sum == target:
        return h
    diff = target - current_sum
    # Add/subtract uniformly, then clip and redistribute
    for _ in range(100):
        h = h + diff / n
        h = np.clip(h, 0, 1)
        new_sum = np.sum(h)
        if abs(new_sum - target) < 1e-12:
            break
        # Only adjust free variables
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        diff = target - new_sum
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0, 1)
    return h

def subgradient_search(h0, max_iter=500, lr_init=0.01, seed=42):
    """Subgradient method with projection."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    rng = np.random.RandomState(seed)
    
    lr = lr_init
    for it in range(max_iter):

        corr = np.correlate(h, 1-h, mode='full') * dx

corr[m] = sum_i h[i] * (1 - h[i - m + n - 1])

The index i ranges such that both i and i-m+n-1 are in [0, n).

Let me define: for output index m (0 to 2n-2), the valid pairs are:
i ranges from max(0, m-n+1) to min(n-1, m)
And the paired index is j = i - m + n - 1

So corr[m] = sum_{i=max(0,m-n+1)}^{min(n-1,m)} h[i] * (1 - h[i-m+n-1])

The gradient w.r.t. h[j]:
∂corr[m]/∂h[j] = sum over i where j appears:
- When i = j: contributes (1 - h[j - m + n - 1]) if j - m + n - 1 is valid
- When i - m + n - 1 = j, i.e., i = j + m - n + 1: contributes -h[i] if i is valid

So: ∂corr[m]/∂h[j] = (1 - h[j - m + n - 1]) * [valid(j - m + n - 1)] - h[j + m - n + 1] * [valid(j + m - n + 1)]

And [valid(k)] = 1 if 0 ≤ k < n, else 0.

Let me implement this:

```python
def subgradient_step(h, n, step_size):
    """One subgradient descent step against the max correlation term."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    m = np.argmax(corr)
    
    grad = np.zeros(n)
    np1 = n - 1  # n-1
    for j in range(n):
        idx1 = j - m + np1
        if 0 <= idx1 < n:
            grad[j] += 1.0 - h[idx1]
        idx2 = j + m - np1
        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    
    h_new = h - step_size * grad
    return h_new, m
```

This is O(n) per step (the loop over j is O(n)), so very fast.

def compute_gradient(h, m, n):
    """Compute gradient of corr[m] w.r.t. h."""
    grad = np.zeros(n)
    np1 = n - 1
    # First term: (1 - h[j - m + np1]) for valid j - m + np1
    idx1 = np.arange(n) - m + np1
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    # Second term: -h[j + m - np1] for valid j + m - np1
    idx2 = np.arange(n) + m - np1
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad

    j = np.arange(n)
    # Term 1: (1 - h[j + m - n + 1]) where 0 <= j+m-n+1 < n

import numpy as np
import time


def project_box_sum(h, n):
    h = np.asarray(h, dtype=np.float64)
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            continue
        h[free] += diff / nf
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def grad_at_shift(h, m, n):
    grad = np.zeros(n)
    j = np.arange(n)
    idx_t1 = j + m - n + 1
    v1 = (idx_t1 >= 0) & (idx_t1 < n)
    grad[v1] += (1.0 - h[idx_t1[v1]])
    idx_t2 = j - m + n - 1
    v2 = (idx_t2 >= 0) & (idx_t2 < n)
    grad[v2] -= h[idx_t2[v2]]
    return grad


def optimize(h0, budget_s, seed=42, K=4):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = project_box_sum(h0.copy(), n)
    best_h = h.copy()
    best_v = compute_c5(h)
    cur_v = best_v
    T = 0.004
    lr = 0.015
    start = time.time()
    it = 0

    while time.time() - start < budget_s:
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(corr)[-K:]
        # Always include shift 0 for binary push
        if 0 not in top_idx:
            top_idx = np.append(top_idx, 0)

        grad = np.zeros(n)
        for m in top_idx:
            grad += corr[m] * grad_at_shift(h, int(m), n)

        h_new = h - lr * grad
        # Gentle binary push
        h_new += 0.002 * (2.0 * h_new - 1.0)
        # Noise for exploration
        h_new += rng.normal(0, lr * 0.05, n)
        h_new = project_box_sum(h_new, n)

        v_new = compute_c5(h_new)
        delta = v_new - cur_v

        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-max(delta, 1e-15) / T)):
            h = h_new
            cur_v = v_new
            if v_new < best_v:
                best_v = v_new
                best_h = h.copy()

        it += 1
        if it % 3000 == 0:
            T = max(1e-10, T * 0.93)
            lr = max(0.002, lr * 0.98)
        if it % 30000 == 0 and rng.random() < 0.4:
            mask = rng.random(n) < 0.15
            h[mask] = rng.uniform(0, 1, mask.sum())
            h = project_box_sum(h, n)
            cur_v = compute_c5(h)

    h = project_box_sum(h, n)
    best_h = project_box_sum(best_h, n)
    return best_h, compute_c5(best_h)


def make_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return h


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    ones = int(h.sum())
    if ones > n // 2:
        idx = np.where(h == 1)[0]
        zidx = np.where(h == 0)[0]
        for k in range(min(ones - n // 2, len(idx), len(zidx))):
            h[idx[k]] = 0.0
            h[zidx[k]] = 1.0
    elif ones < n // 2:
        zidx = np.where(h == 0)[0]
        oidx = np.where(h == 1)[0]
        for k in range(min(n // 2 - ones, len(zidx), len(oidx))):
            h[zidx[k]] = 1.0
            h[oidx[k]] = 0.0
    return h
