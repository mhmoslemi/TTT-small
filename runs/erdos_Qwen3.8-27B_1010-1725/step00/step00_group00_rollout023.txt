    return float(np.max(corr) * dx)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        return eval_c5(x)

        g = np.zeros(n_points)

            fp = objective(x)

import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_obj_and_grad(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scale = corr * dx
    max_scale = np.max(scale)
    shifted = np.clip(epsilon * (scale - max_scale), -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = max_scale + np.log(sum_exp) / epsilon
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for j in range(len(weights)):
        w = weights[j]
        if w < 1e-10:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k]) * dx
            grad[:n - k] += -w * h[k:] * dx
        else:
            kp = -k
            grad[:n - kp] += w * (1.0 - h[kp:]) * dx
            grad[kp:] += -w * h[:n - kp] * dx
    return obj, grad


def penalized_obj(x, epsilon, pen_lam, target):
    val, _ = smooth_max_obj_and_grad(x, epsilon)
    s = np.sum(x)
    return val + pen_lam * (s - target) ** 2


def penalized_grad(x, epsilon, pen_lam, target):
    _, grad = smooth_max_obj_and_grad(x, epsilon)
    s = np.sum(x)
    return grad + 2.0 * pen_lam * (s - target)


def project_feasible(h, n_points):

    s = np.sum(h)
    if s < 1e-10:
        h = np.zeros(n_points)
        h[:int(target)] = 1.0
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    return np.interp(x_new, x_old, h_old)


def optimize_step(h_start, n_points, max_iter, eps, pen_lam, target):
    bounds = [(0.0, 1.0)] * n_points
    args = (eps, pen_lam, target)
    res = minimize(penalized_obj, h_start, jac=penalized_grad, args=args,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 50})
    return project_feasible(res.x, n_points)


def make_periodic_init(n, n_periods, duty):
    h = np.zeros(n)
    period_len = n / n_periods
    on_len = int(round(duty * period_len))
    for p in range(n_periods):
        start = int(round(p * period_len))
        end = min(start + on_len, n)
        h[start:end] = 1.0
    return project_feasible(h, n)
