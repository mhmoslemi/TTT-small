    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, n):

    h = np.clip(h, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)

    return np.sum(h) - _OPT_TARGET

    _OPT_N = n
    _OPT_TARGET = float(n) / 2.0

        h_out = h0
