    global _g_n, _g_pen, _g_beta
    np.random.seed(seed)
    t0 = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    _g_n = n
    _g_pen = 500.0
    _g_beta = 80.0

    # Phase 1: Main optimization from initial point
    h_new = lbgfs_optimize(best_h, max_iter=400)
    c5_new = compute_c5(h_new)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new

    # Coordinate polish
    if time.time() - t0 < budget_s - 120:
        best_h, best_c5 = coordinate_polish(best_h, n, n_sweeps=8)

    # Phase 2: Multiple restarts
    remaining = budget_s - (time.time() - t0) - 60
    if remaining > 60:
        n_restarts = max(2, int(remaining / 50))
        n_restarts = min(n_restarts, 8)
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 60:
                break
            scale = 0.03 + 0.04 * (r + 1)
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_feasible(h_pert, n)

            _g_n = n
            _g_pen = 300.0 + 100.0 * r
            _g_beta = 80.0 + 20.0 * r
            h_try = lbgfs_optimize(h_pert, max_iter=200)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try

    # Phase 3: Try structured perturbations (shift, flip, blend)
    if time.time() - t0 < budget_s - 120:
        # Shift perturbation
        for shift in [1, 2, 3, -1, -2]:
            h_shift = np.roll(best_h, shift)
            h_shift = project_feasible(h_shift, n)
            _g_n = n
            _g_pen = 800.0
            _g_beta = 100.0
            h_try = lbgfs_optimize(h_shift, max_iter=150)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try

        # Blend with random
        for _ in range(3):
            h_rand = np.random.rand(n)
            for alpha in [0.2, 0.4, 0.6]:
                h_blend = (1 - alpha) * best_h + alpha * h_rand
                h_blend = project_feasible(h_blend, n)
                _g_n = n
                _g_pen = 800.0
                _g_beta = 100.0
                h_try = lbgfs_optimize(h_blend, max_iter=150)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try

    # Phase 4: Try higher resolution
    if time.time() - t0 < budget_s - 200:
        n_high = n * 3
        if n_high % 2 == 0:
            n_high += 1
        n_high = min(n_high, 400)
        x_low = np.linspace(0.0, 2.0, n, endpoint=False)
        x_high = np.linspace(0.0, 2.0, n_high, endpoint=False)
        h_high = np.interp(x_high, x_low, np.concatenate([best_h, best_h[:1]]))
        h_high = project_feasible(h_high, n_high)

        _g_n = n_high
        _g_pen = 300.0
        _g_beta = 100.0
        h_try = lbgfs_optimize(h_high, max_iter=200)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            # Polish at higher res
            if time.time() - t0 < budget_s - 60:
                best_h, best_c5 = coordinate_polish(best_h, n_high, n_sweeps=4)

    # Final polish at current resolution
    if time.time() - t0 < budget_s - 20:
        n_final = len(best_h)
        best_h, best_c5 = coordinate_polish(best_h, n_final, n_sweeps=4)

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
