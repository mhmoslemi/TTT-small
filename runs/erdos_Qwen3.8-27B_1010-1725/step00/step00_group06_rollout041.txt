                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return h, best_c5


def gradient_descent(h, target_sum, dx, n, beta_init, lr_init, budget_end, max_iters):
    beta = beta_init
    lr = lr_init
    momentum = 0.85
    vel = np.zeros(n)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    patience = 0
    current = h.copy()
    for it in range(max_iters):
        if time.time() > budget_end:
            break
        grad = compute_gradient(current, beta)
        vel = momentum * vel - lr * grad
        new_h = project_to_feasible(current + vel, target_sum)
        c5_new = compute_c5(new_h, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = new_h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                lr *= 0.9
                vel *= 0.5
                patience = 0
                if lr < 1e-6:
                    lr = 0.02
                    vel *= 0.1
        current = new_h
        if it % 150 == 0:
            beta = min(50.0, beta + 0.3)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    budget_end = t0 + budget_s - 5

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    h = project_to_feasible(h_init, target_sum)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_n = n

    # Phase 1: Gradient descent on original n
    b_end1 = min(budget_end - 120, t0 + 300)
    best_h, best_c5 = gradient_descent(h, target_sum, dx, n, 5.0, 0.08, b_end1, 50000)
    best_n = n

    # Phase 2: Try nearby even n values for binary solutions
    n_candidates = []
    for offset in [-2, 2, -4, 4, 6, -6, 8, -8, 10]:
        nc = n + offset
        if nc >= 50 and nc <= 200:
            n_candidates.append(nc)
    # Deduplicate and sort by closeness to n
    seen = set()
    unique_candidates = []
    for nc in n_candidates:
        if nc not in seen and nc % 2 == 0:
            seen.add(nc)
            unique_candidates.append(nc)
    unique_candidates.sort(key=lambda x: abs(x - n))

    for nc in unique_candidates:
        if time.time() > budget_end - 30:
            break
        dx_c = 2.0 / nc
        ts_c = nc / 2.0
        x_orig = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, nc, endpoint=False)
        h_c = np.interp(x_new, x_orig, best_h)
        h_c = project_to_feasible(h_c, ts_c)

        # Quick gradient descent
        b_end2 = min(budget_end - 15, t0 + 400)
        h_c, c5_c = gradient_descent(h_c, ts_c, dx_c, nc, 8.0, 0.05, b_end2, 10000)
        if c5_c < best_c5 - 1e-10:
            best_c5 = c5_c
            best_h = h_c.copy()
            best_n = nc
            # Binarize and local search
            if time.time() < budget_end - 10:
                h_bin = binarize(best_h, best_n)
                h_ls, c5_ls = local_search_binary(h_bin, dx_c, budget_end - 5)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Phase 3: Also try binarizing the original solution if n is even
    if best_n % 2 == 0 and time.time() < budget_end - 5:
        h_bin = binarize(best_h, best_n)
        dx_best = 2.0 / best_n
        h_ls, c5_ls = local_search_binary(h_bin, dx_best, budget_end)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: Random restarts with perturbation
    if time.time() < budget_end - 20:
        for restart in range(5):
            if time.time() > budget_end - 10:
                break
            h_pert = best_h + 0.1 * np.random.randn(best_n)
            h_pert = project_to_feasible(h_pert, best_n / 2.0)
            dx_r = 2.0 / best_n
            b_end4 = min(budget_end - 5, time.time() + 30)
            h_r, c5_r = gradient_descent(h_pert, best_n / 2.0, dx_r, best_n, 10.0, 0.03, b_end4, 5000)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
            if best_n % 2 == 0 and time.time() < budget_end - 5:
                h_bin_r = binarize(h_r, best_n)
                h_ls_r, c5_ls_r = local_search_binary(h_bin_r, dx_r, budget_end - 2)
                if c5_ls_r < best_c5:
                    best_c5 = c5_ls_r
                    best_h = h_ls_r.copy()

    # Final verification
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    best_c5 = min(best_c5, final_c5)

    return (best_h, best_c5, best_n)
