import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp(alpha * (c - m))
    return m + np.log(np.sum(e)) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp(alpha * (c - m))
    w = e / np.sum(e)
    d = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        m1 = (d >= j - n + 1) & (d <= j)
        m2 = (d >= -j) & (d < n - j)
        i1 = j - d[m1]
        t1 = np.dot(w[m1], 1.0 - h[i1])
        i2 = j + d[m2]
        t2 = np.dot(w[m2], h[i2])
        grad[j] = dx * (t1 - t2)
    return grad

def _interp(h_src, n_tgt):
    n_src = len(h_src)
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    h = np.clip(np.interp(xt, xs, h_src), 0, 1)
    return _project(h, n_tgt)

def _project(h, n):
    h = np.clip(h.copy(), 0, 1)
    tgt = n / 2.0
    diff = tgt - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 5e-5:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _slsqp(h0, alpha, maxiter, tlim):
    n = len(h0)
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - tgt, 'jac': lambda h: np.ones(n)}]

    def f(h):
        return _smooth_obj(h, alpha)
    def g(h):
        return _smooth_grad(h, alpha)

    t0 = time.time()
    res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def _local_refine(h, n, steps=200, lr=0.01, tlim=30):
    t0 = time.time()
    best = h.copy()
    best_c = _c5(best)
    rng = np.random.RandomState(0)
    for s in range(steps):
        if time.time() - t0 > tlim:

        j = rng.randint(n)
        delta = rng.uniform(-lr, lr)
        h2 = h.copy()
        h2[j] = np.clip(h2[j] + delta, 0, 1)
        h2 = _project(h2, n)
        c2 = _c5(h2)
        if c2 < best_c:
            best = h2.copy()
            best_c = c2
            h = h2.copy()
    return best, best_c
