    val = val_core + penalty * s * s
    weights = exp_sh / sum_exp
    grad = np.zeros(n)
    for j in range(n):
        total = 0.0
        for k in range(2 * n - 1):
            L = n - 1 - k
            if L < 0:
                L = 0
            U = 2 * n - 2 - k
            if U > n - 1:
                U = n - 1
            if L > U:
                continue
            i1 = j - k + n - 1
            if L <= i1 <= U:
                total += weights[k] * (1.0 - h[i1])
            if L <= j <= U:
                total -= weights[k] * h[j + k - n + 1]
        grad[j] = dx * total + 2.0 * penalty * s * dx
    return val, grad

def project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff, np.sum(np.where(mask, h, 0)))
            h[mask] -= step / np.sum(mask)
        else:
            mask = h < 1
            if not np.any(mask):
                break
            step = min(-diff, np.sum(np.where(mask, 1.0 - h, 0)))
            h[mask] += step / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_from(h0, dx, beta, penalty, max_iter, timeout_s):
    n = len(h0)
    target = n / 2.0
    h0 = project_box_sum(np.array(h0, dtype=float).copy(), target)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    def obj(p):
        if time.time() - t0 > timeout_s:
            return np.inf
        return smooth_obj_and_grad(p, dx, beta, penalty)[0]
    def grad(p):
        if time.time() - t0 > timeout_s:
            return np.zeros(len(p))
        return smooth_obj_and_grad(p, dx, beta, penalty)[1]
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxfun': 3000})
    except Exception:
        return h0
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_box_sum(h_opt, target)
    return h_opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = 1.0
    best_h = None

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    def elapsed():
        return time.time() - start_time

    # Strategy 1: Optimize from initial construction with various betas
    if h_init is not None and elapsed() < budget_s - 30:
        n0 = len(h_init)
        dx0 = 2.0 / n0
        for beta in [15.0, 30.0, 50.0, 80.0, 120.0]:
            if elapsed() > budget_s - 60:
                break
            penalty = 100.0 + beta
            timeout = min(120.0, (budget_s - elapsed() - 60) * 0.25)
            if timeout < 5:
                break
            h_opt = optimize_from(h_init, dx0, beta, penalty, 400, timeout)
            c5 = compute_c5(h_opt, dx0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Strategy 2: Refine best found with higher beta
    if best_h is not None and elapsed() < budget_s - 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        for beta in [50.0, 100.0, 200.0]:
            if elapsed() > budget_s - 40:
                break
            timeout = min(80.0, (budget_s - elapsed() - 40) * 0.3)
            if timeout < 5:
                break
            h_opt = optimize_from(best_h, dx_b, beta, 500.0, 300, timeout)
            c5 = compute_c5(h_opt, dx_b)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Strategy 3: Random structured starts at various n
    for n in [70, 100, 140, 200]:
        if elapsed() > budget_s - 40:
            break
        dx = 2.0 / n
        target = n / 2.0
        # Structured starts: alternating blocks, sinusoidal, random
        starts = []
        # Pure random
        starts.append(np.clip(0.5 + 0.4 * rng.randn(n), 0, 1))
        # Block pattern
        block_size = max(1, n // 10)
        h_block = np.zeros(n)
        for b in range(10):
            if b % 2 == 0:
                s, e = b * block_size, min((b + 1) * block_size, n)
                h_block[s:e] = 1.0
        starts.append(h_block)
        # Sinusoidal
        t = np.linspace(0, 4 * np.pi, n)
        starts.append(np.clip(0.5 + 0.3 * np.sin(t), 0, 1))
        # High-freq noise
        starts.append(np.clip(0.5 + 0.45 * rng.randn(n), 0, 1))

        per_start_timeout = max(15.0, (budget_s - elapsed() - 40) / (4 * 3))
        for h_start in starts:
            if elapsed() > budget_s - 30:
                break
            h_s = project_box_sum(h_start, target)
            for beta in [30.0, 60.0]:
                if elapsed() > budget_s - 30:
                    break
                timeout = min(per_start_timeout, (budget_s - elapsed() - 20) * 0.15)
                if timeout < 5:
                    break
                h_opt = optimize_from(h_s, dx, beta, 200.0, 250, timeout)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Strategy 4: Final local refinement of best
    if best_h is not None and elapsed() < budget_s - 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        # Small perturbations + re-optimize
        for _ in range(3):
            if elapsed() > budget_s - 15:
                break
            h_pert = best_h + 0.05 * rng.randn(n_b)
            h_opt = optimize_from(h_pert, dx_b, 80.0, 500.0, 200, 30.0)
            c5 = compute_c5(h_opt, dx_b)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Ensure valid output
    if best_h is None:
        n = 70
        best_h = np.ones(n) * 0.5
        best_c5 = compute_c5(best_h, 2.0 / n)

    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_box_sum(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h, dx_final)
    return best_h, final_c5, n_final
