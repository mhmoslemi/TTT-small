    """Find the active lag and compute subgradient."""
    corr = np.correlate(h, 1-h, mode='full')
    active_k = np.argmax(corr)  # index into the full correlation
    # Lag = active_k - (n-1) where n = len(h)
    n = len(h)
    lag = active_k - (n - 1)
    
    # Subgradient of g_k at h: for each j, d/dh[j] = (1-h[j+k]) - h[j-k]
    # (with appropriate boundary handling)
    grad = np.zeros(n)

        if 0 <= j - lag < n:

                h = h_new
            patience += 1

        T = max(T * 0.99998, 0.0005)

    # Final: try to squeeze out more with very small steps
    T2 = 0.0005
    step2 = 0.005
    h = best_h.copy()
    for _ in range(50000):
        if time.time() - t0 > budget_s - 2:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-step2, step2)
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5 = eval_c5(h_new, dx)
        if c5 <= best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
        elif rng.random() < np.exp(-(c5 - best_c5) / T2):
            h = h_new

    # Verify with evaluate_erdos_solution if available
    if 'evaluate_erdos_solution' in globals() and evaluate_erdos_solution is not None:
        try:
            final_c5 = float(evaluate_erdos_solution(best_h))
            if final_c5 > best_c5:
                best_c5 = final_c5
        except Exception:
            pass

    return best_h, best_c5, n
