import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _smooth_obj(x, n, scale):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp((corr - m) * scale))) / (scale * len(corr)))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Fine correction: adjust a single element
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + diff)
    return h

def _optimize(h0, n, scale, maxiter=400, ftol=1e-12):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)},
        {'type': 'ineq', 'fun': lambda x: np.min(x), 'jac': lambda x: np.where(np.argmin(x) == np.arange(n), 1.0, 0.0)},
        {'type': 'ineq', 'fun': lambda x: np.min(1.0 - x), 'jac': lambda x: -np.where(np.argmin(x) == np.arange(n), 1.0, 0.0)},
    ]
    def obj(x):
        return _smooth_obj(x, n, scale)
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _optimize_simple(h0, n, scale, maxiter=400, ftol=1e-12):
    """Optimization with only sum constraint; bounds handled by projection."""
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    def obj(x):
        return _smooth_obj(x, n, scale)
    res = minimize(obj, h0, method='SLSQP',
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return _fix_sum(res.x, n)

def _random_perturb(h, n, scale=0.25):
    h = h.copy()
    h += np.random.randn(n) * scale
    return _fix_sum(h, n)

def _binary_template(n, rng):
    h = np.zeros(n)
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return _fix_sum(h, n)

def _grad_descent(h, n, steps=200, lr=0.01):
    h = h.copy()
    dx = 2.0 / n
    for s in range(steps):
        loss = _smooth_obj(h, n, 60.0)
        # Numerical gradient
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            hp = h.copy(); hp[i] += eps
            hp = _fix_sum(hp, n)
            hm = h.copy(); hm[i] -= eps
            hm = _fix_sum(hm, n)

        h -= lr * grad
