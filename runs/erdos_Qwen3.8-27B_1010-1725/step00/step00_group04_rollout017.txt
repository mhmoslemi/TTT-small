import numpy as np
from scipy.optimize import minimize

def compute_correlations(h):
    return np.correlate(h, 1-h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_correlations(h)) * dx

def c5_gradient(h, dx):
    n = len(h)
    corr = compute_correlations(h)
    m_star = np.argmax(corr)  # index in [0, 2n-2]
    shift = m_star - (n-1)  # actual shift in [-(n-1), n-1]
    
    grad = np.zeros(n)
    for k in range(n):
        # Contribution from h[k] as h[i+shift] where i = k-shift
        if 0 <= k - shift <= n-1:
            grad[k] += (1 - h[k - shift])
        # Contribution from h[k] in (1-h[i]) where i = k, need h[k+shift]
        if 0 <= k + shift <= n-1:
            grad[k] -= h[k + shift]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    
    global initial_h_values
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    
    # Ensure initial feasibility
    h0 = np.clip(h0, 0, 1)
    s = np.sum(h0)

        h0 = h0 * (n0/2.0) / s
        h0 = np.clip(h0, 0, 1)
    
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)
    
    rng = np.random.RandomState(seed)

        elapsed = time.time() - t0
        if elapsed > budget_s - 60:
            return
        
        def obj(h):
            return compute_c5(h, dx)
        
        def obj_grad(h):
            return c5_gradient(h, dx)
        
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
                       'jac': lambda h: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        
        try:
            res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-10})

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, dx):
    return float(np.max(compute_correlations(h)) * dx)


def soft_c5(h, dx, alpha):
    corr = compute_correlations(h) * dx
    v_max = np.max(corr)
    exps = np.exp(alpha * (corr - v_max))
    return float(v_max + np.log(np.sum(exps)) / alpha)


def soft_c5_grad(h, dx, alpha):
    n = len(h)
    corr = compute_correlations(h) * dx
    v_max = np.max(corr)
    exps = np.exp(alpha * (corr - v_max))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        shift = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-14:
            continue
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + shift:hi2 + shift]
    return grad * dx


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h


class SumConstraint:
    def __init__(self, n):
        self._n = n

    def fun(self, x):
        return float(np.sum(x) - self._n / 2.0)

    def jac(self, x):
        return np.ones(self._n)


def make_constraint(n):
    c = SumConstraint(n)
    return {'type': 'eq', 'fun': c.fun, 'jac': c.jac}


def run_slsqp(h_init, n, dx, alpha, maxiter, ftol):
    elapsed_0 = time.time()
    bounds = [(0.0, 1.0)] * n
    con = make_constraint(n)
    opts = {'maxiter': maxiter, 'ftol': ftol}
    res = minimize(
        soft_c5, h_init, jac=lambda h: soft_c5_grad(h, dx, alpha),
        method='SLSQP', bounds=bounds, constraints=con,
        options=opts
    )
    if res.x is not None:
        return repair_h(res.x, n)
    return None


def refine(h, n, dx, budget_left):
    best = h.copy()
    best_val = compute_c5(best, dx)
    t0 = time.time()
    for alpha in [5.0, 10.0, 20.0, 40.0]:
        if time.time() - t0 > budget_left:
            break
        try:
            h_new = run_slsqp(best.copy(), n, dx, alpha, 300, 1e-10)
            if h_new is not None:
                v = compute_c5(h_new, dx)
                if v < best_val:
                    best_val = v
                    best = h_new.copy()
        except Exception:
            pass
    return best, best_val


def random_restart(h_base, n, rng, scale):
    noise = rng.randn(n) * scale
    h_try = np.clip(h_base + noise, 0.0, 1.0)
    return repair_h(h_try, n)

    t_start = time.time()

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    h0 = repair_h(h0, n0)
    dx0 = 2.0 / n0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)

    rng = np.random.RandomState(seed)

    # Phase 1: Optimize at original resolution
    budget_left = budget_s - (time.time() - t_start) - 60
    if budget_left > 30:
        best_h, best_c5 = refine(best_h, n0, dx0, budget_left)

    # Phase 2: Multiple restarts at original resolution
    n_restarts = 20
    for i in range(n_restarts):
        elapsed = time.time() - t_start
        if elapsed > budget_s - 60:
            break
        scale = 0.03 + 0.15 * (i / max(1, n_restarts))
        h_try = random_restart(best_h, n0, rng, scale)
        h_res, c5_res = refine(h_try, n0, dx0, 8.0)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 3: Try higher resolutions
    for mult in [2, 3, 4]:
        elapsed = time.time() - t_start
        if elapsed > budget_s - 120:
            break
        n = n0 * mult
        dx = 2.0 / n
        # Upsample via linear interpolation
        old_x = np.arange(n0)
        new_x = np.linspace(0, n0 - 1, n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = repair_h(h_up, n)
        h_res, c5_res = refine(h_up, n, dx, 60.0)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 4: Final polish on best solution
    elapsed = time.time() - t_start
    if elapsed < budget_s - 15:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        h_pol, c5_pol = refine(best_h, n_best, dx_best, budget_s - elapsed - 5)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final verification and repair
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = repair_h(best_h, n_final)
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, n_final
