- Actually, the Riemann sum gives: ∫ h(x)(1-h(x+k)) dx ≈ sum_i h[i] * (1-h[i+k]) * dx

- obj = max(cross) * dx (the hard version)
- soft_max_obj = (c_max + mu * log(sum(exp((cross-c_max)/mu)))) * dx

d(soft_max_obj)/dh[i] = dx * sum_k w[k] * dcross[k]/dh[i]

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    cross = np.correlate(h, 1 - h, mode='full')
    return float(np.max(cross) * dx)

def soft_max_obj(h, mu):
    n = len(h)
    dx = 2.0 / n
    cross = np.correlate(h, 1 - h, mode='full')
    c_max = np.max(cross)
    return (c_max + mu * np.log(np.sum(np.exp((cross - c_max) / mu)))) * dx

def soft_max_grad(h, mu):
    n = len(h)
    dx = 2.0 / n
    cross = np.correlate(h, 1 - h, mode='full')
    c_max = np.max(cross)
    w = np.exp((cross - c_max) / mu)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        shift = k - (n - 1)
        for i in range(n):
            j1 = i - shift
            if 0 <= j1 < n:
                grad[i] += wk * (1.0 - h[j1])
            j2 = i + shift
            if 0 <= j2 < n:
                grad[i] += wk * (-h[j2])
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, target_sum, n_iter, rng, t_start, time_limit):
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    step_size = 0.15
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        h_try = best_h.copy()
        n_perturb = int(rng.integers(1, max(2, len(h) // 8)))
        idx = rng.choice(len(h), size=n_perturb, replace=False)
        h_try[idx] += rng.uniform(-step_size, step_size, size=n_perturb)
        h_try = project_feasible(h_try, target_sum)
        c_try = evaluate_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try
            best_c = c_try
            step_size = min(0.3, step_size * 1.02)
        else:
            step_size = max(0.005, step_size * 0.97)
    return best_h, best_c

def coordinate_descent(h, target_sum, n_passes, rng, t_start, time_limit):
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    n = len(h)
    for _ in range(n_passes):
        if time.time() > time_limit:
            break
        for i in range(n):
            if time.time() > time_limit:
                break
            save = best_h[i]
            for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.2, -0.2]:
                best_h[i] = save + delta
                if best_h[i] < 0 or best_h[i] > 1:
                    continue
                best_h = project_feasible(best_h, target_sum)
                c_try = evaluate_c5(best_h)
                if c_try < best_c:
                    best_c = c_try
                    break
            if best_h[i] == save:
                best_h[i] = save
    return best_h, best_c

def optimize_slsqp(h0, target_sum, mu, max_iter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    h0 = project_feasible(h0, target_sum)
    res = minimize(soft_max_obj, h0, args=(mu,), jac=lambda x: soft_max_grad(x, mu),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    x = project_feasible(res.x, target_sum)
    c = evaluate_c5(x)
    c0 = evaluate_c5(project_feasible(h0, target_sum))
    if c < c0:
        return x, c
    return project_feasible(h0, target_sum), c0

def interpolate_to_n(h_src, n_dst, target_sum_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_interp, target_sum_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    time_limit = t_start + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.clip(rng.uniform(0, 1, 47), 0, 1)
    n_init = len(h_init)
    target_sum = n_init / 2.0
    h_init = project_feasible(h_init, target_sum)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    best_n = n_init

    # Phase 1: SLSQP optimization at initial resolution with decreasing mu
    for mu in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.time() > time_limit - 120:
            break
        h_opt, c_opt = optimize_slsqp(best_h, target_sum, mu, max_iter=400)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 2: Local search at initial resolution
    if time.time() < time_limit - 100:
        n_ls = min(50000, int((time_limit - 150 - time.time()) / 0.003))
        best_h, best_c = local_search(best_h, target_sum, n_ls, rng, t_start, time_limit - 150)

    # Phase 3: Coordinate descent polish
    if time.time() < time_limit - 80:
        best_h, best_c = coordinate_descent(best_h, target_sum, 5, rng, t_start, time_limit - 80)

    # Phase 4: Try higher resolutions
    for n_try in [60, 80, 100]:
        remaining = time_limit - 60 - time.time()
        if remaining < 30:
            break
        h_try = interpolate_to_n(best_h, n_try, n_try / 2.0)
        c_try = evaluate_c5(h_try)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, n_try
        for mu in [0.05, 0.02, 0.01]:
            if time.time() > time_limit - 60:
                break
            h_opt, c_opt = optimize_slsqp(best_h, best_n / 2.0, mu, max_iter=200)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, n_try
        if time.time() < time_limit - 60:
            n_ls2 = min(20000, int((time_limit - 80 - time.time()) / 0.003))
            best_h, best_c = local_search(best_h, best_n / 2.0, n_ls2, rng, t_start, time_limit - 60)

    # Phase 5: Final polish at best resolution
    if time.time() < time_limit - 5:
        n_polish = min(30000, int((time_limit - 3 - time.time()) / 0.003))
        best_h, best_c = local_search(best_h, best_n / 2.0, n_polish, rng, t_start, time_limit - 2)
        best_h, best_c = coordinate_descent(best_h, best_n / 2.0, 3, rng, t_start, time_limit - 1)

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c = evaluate_c5(best_h)
    return (best_h, best_c, best_n)
```
