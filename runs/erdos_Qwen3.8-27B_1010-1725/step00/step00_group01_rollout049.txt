    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cor) * dx)

        h = h_init.copy().astype(float)

        x_new = np.arange(n) / n * 2.0
        h = np.interp(x_new, x_old, np.concatenate([h_init, [h_init[0]]]))

    target = n / 2.0
    diff = target - np.sum(h)

import numpy as np
from scipy.optimize import minimize
import time

_curr_n = 47

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cor) * dx)

def _resize_init(h_src, n):
    if len(h_src) == n:
        h = h_src.astype(float).copy()
    else:
        x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_ext = np.concatenate([h_src, [h_src[0]]])
        h = np.interp(x_new, x_old, h_ext)
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_ineq(x):
    n = _curr_n
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - cor

def _slsqp_eq(x):
    n = _curr_n
    return np.sum(x[:n]) - n / 2.0

def _slsqp_obj(x):
    return x[_curr_n]

def _slsqp_jac(x):
    g = np.zeros(_curr_n + 1)
    g[_curr_n] = 1.0
    return g

def _optimize_slsqp(h_init, maxiter=300, ftol=1e-10):
    global _curr_n
    n = len(h_init)
    _curr_n = n
    t0 = evaluate_c5(h_init)
    x0 = np.concatenate([h_init, [t0 + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': _slsqp_ineq},
            {'type': 'eq', 'fun': _slsqp_eq}]
    try:
        result = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_new = _project(result.x[:n], n)
        if evaluate_c5(h_new) < t0:
            return h_new
    except Exception:
        pass
    return h_init

def _local_perturb(h, n, rng, scale=0.08):
    h_p = h.copy().astype(float)
    if len(h_p) != n:
        h_p = _resize_init(h_p, n)
    mask = rng.random(n) < 0.35
    if mask.any():
        h_p[mask] += rng.normal(0.0, scale, mask.sum())
    h_p = _project(h_p, n)
    return h_p

def _swap_perturb(h, n, rng, n_swaps=5):
    h_p = h.copy().astype(float)
    if len(h_p) != n:
        h_p = _resize_init(h_p, n)
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, 2)
        if i != j:
            delta = (h_p[i] - h_p[j]) / 2.0
            h_p[i] -= delta
            h_p[j] += delta
    h_p = _project(h_p, n)
    return h_p
