For coordinate descent:
Fix all h[j] for j≠i. The objective is max_k [C_k + h[i]*d_k] where:
- C_k = c_k with h[i]=0
- d_k = (1-h[i+k])*1[valid(i+k)] + (-h[i-k])*1[valid(i-k)]

We want to minimize max_k [C_k + h[i]*d_k] over h[i] ∈ [0,1]
(ignoring the sum constraint for now, or incorporating it via a Lagrange multiplier)

The minimum of max of linear functions in 1D can be found by:
- The function is piecewise linear, convex in h[i]
- The minimum is at a point where two (or more) of the linear functions cross, or at a boundary

We can find this by:
- Evaluating at h[i]=0 and h[i]=1
- Finding all crossing points of pairs d_k * h[i] + C_k = d_l * h[i] + C_l
  → h[i] = (C_l - C_k) / (d_k - d_l) for d_k ≠ d_l
- Checking which crossing points are in [0,1]
- Evaluating the max at all candidate points
- Taking the minimum

This is O(m^2) where m is the number of lags (2n-1), which is slow for large n. But we only need the top few active lags.

import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def project_onto_constraint(h, S):
    n = len(h)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_subgradient(h, n, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for ki in top_idx:
        ks = int(ki) - (n - 1)
        for j in range(n):
            jp = j + ks
            jm = j - ks
            if 0 <= jp < n:
                grad[j] += 1.0 - h[jp]
            if 0 <= jm < n:
                grad[j] -= h[jm]
    grad /= top_m
    return grad, corr

def subgrad_optimize(h, n, deadline, top_m=5, max_iter=10000):
    S = n / 2.0
    alpha = 0.5
    best = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, corr = get_subgradient(h, n, top_m)
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < best - 1e-10:
            best = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

    best_h = h.copy()
    rng = np.random.RandomState(123)

            break
        order = rng.permutation(n)
        for i in order:

            # The objective as function of h[i] is max of linear functions → convex

            for trial in [0.0, 0.25, 0.5, 0.75, 1.0]:

                h[i] = trial
                obj = compute_c5(h)
                if obj < best_obj:
                    best_obj = obj
                    best_val = trial
            h[i] = best_val

        c5 = compute_c5(h)
        if c5 < best:
            best = c5
            best_h = h.copy()

def binary_local_search(h, n, deadline, seed_val=0, max_trials=50000):
    S = n / 2.0
    h = (h > 0.5).astype(float)
    # Ensure correct number of ones
    n_ones = int(h.sum())
    target = int(S)
    rng = np.random.RandomState(seed_val)
    if n_ones > target:
        excess = np.where(h == 1)[0]
        h[rng.choice(excess, size=n_ones - target, replace=False)] = 0
    elif n_ones < target:
        deficit = np.where(h == 0)[0]
        h[rng.choice(deficit, size=target - n_ones, replace=False)] = 1
    
    best = compute_c5(h)
    best_h = h.copy()
    trials = 0
    cur = h.copy()
    
    while time.time() < deadline and trials < max_trials:
        trials += 1
        i = int(rng.randint(n))
        cur[i] = 1.0 - cur[i]
        c5 = compute_c5(cur)
        if c5 <= best + 1e-12:
            best = c5
            best_h = cur.copy()
        else:
            cur[i] = 1.0 - cur[i]
    
    return best_h, best

def slsqp_optimize(h, n, deadline):
    dx = 2.0 / n
    S = n / 2.0
    
    def obj_func(x):
        x = np.clip(x, 0, 1)
        return float(np.max(np.correlate(x, 1.0 - x, mode='full'))) * dx
    
    def grad_func(x):
        # Numerical gradient (approximate)
        eps = 1e-5
        g = np.zeros(n)
        base = obj_func(x)
        for i in range(n):
            xp = x.copy()
            xp[i] = min(1, x[i] + eps)
            xm = x.copy()
            xm[i] = max(0, x[i] - eps)
            g[i] = (obj_func(xp) - obj_func(xm)) / (xp[i] - xm[i] + 1e-15)
        return g

            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_bl
        # Another restart with different seed
        if time.time() < deadline:
            h_opt2, c5_opt2 = binary_local_search(h_bl, n_bl, deadline, seed_val=seed + 99, max_trials=15000)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2.copy(), c5_opt2, n_bl

    # Phase 5: Larger n via interpolation + subgradient
    for n_large in [140, 200, 300]:
        if time.time() > deadline - 15:
            break
        if best_h is not None:
            h = make_interp_init(best_h, n_large)
        else:
            h = (rng.rand(n_large) < 0.5).astype(float)
        h = project_onto_constraint(h, n_large / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n_large
        h_opt, c5_opt = subgrad_optimize(h.copy(), n_large, deadline, top_m=7, max_iter=4000)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_large

    # Phase 6: Final polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_opt, c5_opt = subgrad_optimize(best_h.copy(), n, deadline, top_m=3, max_iter=2000)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
        h_opt2, c5_opt2 = coord_descent(best_h.copy(), n, deadline, max_sweeps=2)
        if c5_opt2 < best_c5:
            best_h, best_c5, best_n = h_opt2.copy(), c5_opt2, n

    # Ensure constraints
    if best_h is not None:
        best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
