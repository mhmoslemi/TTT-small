        sg = np.zeros(n)
        lo = max(0, -s_max)
        hi = min(n, n - s_max)
        if lo < hi:
            idx = np.arange(lo, hi)
            sg[idx] += (1.0 - h[idx + s_max])
        lo2 = max(0, s_max)
        hi2 = min(n, s_max + n)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            sg[idx2] += -h[idx2 - s_max]
        h = project_feasible(h - alpha * sg, n)
        c5_now = eval_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        alpha *= 0.997
        if it % 50 == 0:
            alpha = max(0.003, alpha)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size < 8 or np.any(np.isnan(h0)):
            h0 = None
    except Exception:
        pass

    if h0 is None:
        n_init = 80
        t = np.arange(n_init) / float(n_init)
        h0 = 0.5 + 0.25 * np.sin(2 * np.pi * 5 * t) + 0.15 * np.sin(2 * np.pi * 3 * t + 1.0)
        h0 = project_feasible(h0, n_init)

    n = len(h0)
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = eval_c5(h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with increasing smoothness parameter
    betas = [3, 5, 10, 20, 40, 80, 160]
    for beta in betas:
        if remaining() < 20:
            break
        n_restarts = 3 if beta <= 10 else 2
        for r in range(n_restarts):
            if remaining() < 10:
                break
            if r == 0:
                x0 = h.copy()
            elif r == 1:
                x0 = project_feasible(h + 0.08 * rng.standard_normal(n), n)
            else:
                x0 = project_feasible(0.7 * h + 0.3 * (0.5 + 0.4 * np.sin(
                    2 * np.pi * (2 + r) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))), n)
            h_new, ok = slsqp_optimize(x0, n, beta, 400, remaining)
            if h_new is not None:
                c5_new = eval_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 2: Subgradient refinement
    if remaining() > 40:
        h_sg, c5_sg = subgrad_descent(h.copy(), n, 800, remaining)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Second round SLSQP at finer beta
    if remaining() > 30:
        h_new, _ = slsqp_optimize(h.copy(), n, 200, 500, remaining)
        if h_new is not None:
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Upsample to higher resolution
    for n_up in [160, 240, 400]:
        if remaining() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_up)
        for beta in [20, 60, 120]:
            if remaining() < 15:
                break
            h_new, _ = slsqp_optimize(h_up, n_up, beta, 300, remaining)
            if h_new is not None:
                c5_new = eval_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_up = h_new.copy()
        if remaining() > 30:
            h_sg, c5_sg = subgrad_descent(h_up, n_up, 400, remaining)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 5: Final polish at current resolution
    if remaining() > 15:
        h_new, _ = slsqp_optimize(best_h.copy(), len(best_h), 300, 300, remaining)
        if h_new is not None:
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
