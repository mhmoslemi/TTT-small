import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, N):
    """Project h onto {0<=h<=1, sum(h)=N/2}"""
    h = np.clip(h, 0, 1)
    target = N / 2
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        if s > target:
            # Decrease: reduce values that are not at 0
            excess = s - target
            mask = h > 0
            n_active = np.sum(mask)
            if n_active == 0:
                break
            reduction = min(excess / n_active, np.min(h[mask]))
            h[mask] -= reduction
        else:
            # Increase: increase values that are not at 1
            deficit = target - s
            mask = h < 1
            n_active = np.sum(mask)
            if n_active == 0:
                break
            increase = min(deficit / n_active, 1 - np.max(h[mask]))
            h[mask] += increase
    return np.clip(h, 0, 1)

def compute_corr(h, dx):
    """Compute all correlation values scaled by dx"""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, beta):
    corr = compute_corr(h, 2.0/len(h))
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / beta

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, N):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = N / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] - diff / cnt, 0.0)
        else:
            mask = h < 1
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] - diff / cnt, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_obj_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    ex = np.exp(corr - mx)
    p = ex / ex.sum()
    obj = float(mx + np.log(ex.sum()) / beta)
    k_range = np.arange(2 * N - 1)
    grad = np.zeros(N)
    for j in range(N):
        i1 = j + k_range - (N - 1)
        i2 = j - k_range + (N - 1)
        m1 = (i1 >= 0) & (i1 < N)
        m2 = (i2 >= 0) & (i2 < N)
        t = np.zeros(2 * N - 1)
        if np.any(m1):
            t[m1] += 1.0 - h[i1[m1]]
        if np.any(m2):
            t[m2] -= h[i2[m2]]
        grad[j] = dx * np.dot(p, t)
    return obj, grad

def slsqp_optimize(h, N, beta, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x, N=N: np.sum(x) - N / 2.0,
             'jac': lambda x, N=N: np.ones(N)}]
    try:
        res = minimize(
            lambda x: smooth_obj_grad(x, beta)[0],
            h,
            jac=lambda x: smooth_obj_grad(x, beta)[1],
            bounds=[(0.0, 1.0)] * N,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, N)
    except Exception:
        return h

def proj_grad_steps(h, N, beta, lr, n_steps, time_check):
    for _ in range(n_steps):
        if time_check():
            break
        _, grad = smooth_obj_grad(h, beta)
        h = project_feasible(h - lr * grad, N)
    return h

def coord_descend(h, N, time_check, max_sweeps=200):
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time_check():
            break
        improved = False
        for j in range(N):

                break

            cands = [0.0, 1.0, orig * 0.4, orig * 1.6,

                    h = h_try
                    orig = cv
                    improved = True
        if not improved:

    return h, best_c5
