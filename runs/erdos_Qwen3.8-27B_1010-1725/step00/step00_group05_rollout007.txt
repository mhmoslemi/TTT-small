            ht = project_onto_simplex(ht, nl)
            c5t = np.max(get_correlation(ht, nl)) * dxl
            if c5t < best_c5:
                best_c5 = c5t
                best_h = ht
                h = ht

    return (best_h, best_c5, best_n)
