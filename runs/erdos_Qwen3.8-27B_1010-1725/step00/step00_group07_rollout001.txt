LAG_MODE = 1

def calibrate_lag_mode():
    global LAG_MODE
    h = np.zeros(6)
    h[0] = 1.0
    c = np.correlate(h, 1.0 - h, mode="full")
    mid = c.size // 2
    if np.sum(c[:mid]) >= np.sum(c[mid + 1:]):
        LAG_MODE = 1
    else:
        LAG_MODE = -1


def c5_of(h):
    n = h.size
    if n == 0:
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def adjust_sum(y, S):
    y = np.asarray(y, dtype=float).copy()
    n = y.size
    if n == 0:
        return y
    for _ in range(n + 12):
        res = S - float(y.sum())
        if abs(res) < 1e-13:
            break
        if res > 0.0:
            i = int(np.argmax(1.0 - y))
            cap = 1.0 - y[i]
            if cap <= 1e-14:
                break
            add = min(cap, res)
            y[i] += add
        else:
            i = int(np.argmax(y))
            if y[i] <= 1e-14:
                break
            add = min(y[i], -res)
            y[i] -= add
    return y


def project_box_sum(x, S):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return x.copy()
    if S <= 0.0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)

    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum()) - S) < 1.0:
        return adjust_sum(y, S)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        yt = np.clip(x - mid, 0.0, 1.0)
        if float(yt.sum()) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(x - tau, 0.0, 1.0)
    return adjust_sum(y, S)


def feasible_copy(h):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    return project_box_sum(h, 0.5 * h.size)


def set_exact_sum(h, S):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    n = h.size
    if n == 0:
        return h
    total = float(h.sum())
    for i in range(n):
        val = S - (total - h[i])
        if -1e-12 <= val <= 1.0 + 1e-12:
            h[i] = float(np.clip(val, 0.0, 1.0))
            return h
    return adjust_sum(h, S)


def selected_lags(corr, n, K):
    L = corr.size
    K = max(1, min(int(K), L))
    if K >= L:
        idx = np.arange(L)
    elif K == 1:
        idx = np.array([int(np.argmax(corr))])
    else:
        idx = np.argpartition(corr, K - 1)[-K:]

    vals = corr[idx].astype(float)
    m = float(np.max(vals))
    w = np.exp((vals - m) / 0.2)
    s = float(w.sum())
    if s < 1e-300:
        w = np.ones_like(w) / float(w.size)
    else:
        w = w / s

    idx_i = idx.astype(int)
    if LAG_MODE == 1:
        lags = (n - 1) - idx_i
    else:
        lags = idx_i - (n - 1)
    return lags, w


def active_gradient(h, lags, weights):
    n = h.size
    g = np.zeros(n, dtype=float)
    for d, w in zip(lags, weights):
        d = int(d)
        w = float(w)
        if w == 0.0:
            continue
        if d >= 0:
            dd = d
            if dd == 0:
                g += w * (1.0 - 2.0 * h)
            else:
                m = n - dd
                if m > 0:
                    g[:m] += w * (1.0 - h[dd:])
                    g[dd:] -= w * h[:m]
        else:
            dd = -d
            m = n - dd
            if m > 0:
                g[dd:] += w * (1.0 - h[:m])
                g[:dd] -= w * h[dd:]
    return g


def continuous_improve(h, end_time, rng):
    n = h.size
    if n < 2 or time.time() >= end_time:
        h = feasible_copy(h)
        return h, c5_of(h)

    S = 0.5 * n
    h = project_box_sum(h, S)
    best_h = h.copy()
    best_c = c5_of(h)

    K = 6
    stagnant = 0

    while time.time() < end_time:
        corr = full_corr(h)
        c = float(np.max(corr) * (2.0 / n))
        if c < best_c - 1e-12:
            best_h = h.copy()
            best_c = c

        lags, w = selected_lags(corr, n, K)
        g = active_gradient(h, lags, w)

        gmax = float(np.max(np.abs(g)))

        if gmax < 1e-10:
            h = project_box_sum(h + rng.normal(0.0, 0.03, n), S)
            K = min(128, K * 2 + 1)
            stagnant += 1
            if stagnant > 10:
                break
            continue

        direction = -g / gmax
        best_x = None
        best_cc = c

        for t in STEP_FRACS:
            x = project_box_sum(h + t * direction, S)
            cx = c5_of(x)
            if cx < best_cc - 1e-13:
                best_cc = cx
                best_x = x

        if best_x is not None:
            h = best_x
            if best_cc < best_c - 1e-12:
                best_h = h.copy()
                best_c = best_cc
            K = min(128, int(K * 1.15) + 1)
            stagnant = 0
            continue

        x = project_box_sum(h + 0.0005 * direction, S)
        cx = c5_of(x)
        if cx < c - 1e-13:
            h = x
            if cx < best_c - 1e-12:
                best_h = h.copy()
                best_c = cx
            stagnant = 0
        else:
            h = project_box_sum(h + rng.normal(0.0, 0.015, n), S)
            K = min(128, K * 2 + 1)
            stagnant += 1
            if stagnant > 14:
                h = project_box_sum(best_h.copy() + rng.normal(0.0, 0.05, n), S)
                K = 12
                stagnant = 0

    return best_h, best_c


def scores_to_binary(n, scores):
    h = np.zeros(n, dtype=float)
    if n <= 0:
        return h
    target = n // 2
    idx = np.argsort(-scores)[:target]
    h[idx] = 1.0
    return h


def initial_random(n, rng):
    x = np.arange(n, dtype=float)
    scores = np.mod(x * 0.61803398875, 1.0) + 0.25 * rng.random(n)
    return feasible_copy(scores_to_binary(n, scores))


def to_binary(h):
    n = h.size
    target = n // 2
    idx = np.argsort(-h)[:target]
    hb = np.zeros(n, dtype=float)
    hb[idx] = 1.0
    return hb


def pair_improve(h, end_time, rng):
    n = h.size
    if n < 2 or time.time() >= end_time:
        h = feasible_copy(h)
        return h, c5_of(h)

    S = 0.5 * n
    h = project_box_sum(h, S)
    best_h = h.copy()
    best_c = c5_of(h)
    cur_c = best_c
    stagnant = 0

    while time.time() < end_time:
        improved = False

        for _ in range(n):
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            while q == p:
                q = int(rng.integers(n))

            a = float(h[p])
            b = float(h[q])
            lo = max(-a, b - 1.0)
            hi = min(1.0 - a, b)
            if hi - lo <= 1e-13:
                continue

            span = hi - lo
            best_d = 0.0
            best_local = cur_c

            for f in PAIR_FRACS:
                d = f * span
                if d < lo:
                    d = lo
                elif d > hi:
                    d = hi
                if abs(d) < 1e-14:
                    continue

                vp = a + d
                vq = b - d
                if vp < -1e-12 or vp > 1.0 + 1e-12 or vq < -1e-12 or vq > 1.0 + 1e-12:
                    continue
                vp = float(np.clip(vp, 0.0, 1.0))
                vq = float(np.clip(vq, 0.0, 1.0))

                h[p] = vp
                h[q] = vq
                cx = c5_of(h)
                if cx < best_local - 1e-13:
                    best_local = cx
                    best_d = d
                h[p] = a
                h[q] = b

            if best_d != 0.0:
                vp = float(np.clip(a + best_d, 0.0, 1.0))
                vq = float(np.clip(b - best_d, 0.0, 1.0))
                h[p] = vp
                h[q] = vq
                cur_c = best_local
                if cur_c < best_c - 1e-13:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True

        if not improved:
            stagnant += 1
            if stagnant > 4:
                break
            h = project_box_sum(h + rng.normal(0.0, 0.02, n), S)
            cur_c = c5_of(h)
            stagnant = 0

    return best_h, best_c


def binary_local_search(hb, end_time, rng):
    n = hb.size
    if n < 4 or n % 2 != 0 or time.time() >= end_time:
        h = feasible_copy(hb)
        return h, c5_of(h)

    target = n // 2
    hb = np.asarray(hb, dtype=float)
    hb = (hb >= 0.5).astype(float)
    if int(hb.sum()) != target:
        hb = to_binary(hb)

    ohb = 1.0 - hb
    ones = np.nonzero(hb > 0.5)[0]
    zeros = np.nonzero(hb <= 0.5)[0]
    dx = 2.0 / n

    c = float(np.max(np.correlate(hb, ohb, mode="full")) * dx)
    best_hb = hb.copy()
    best_c = c
    stagnant = 0

    max_full = 60000 if n <= 320 else 25000

    while time.time() < end_time:
        total = int(len(ones)) * int(len(zeros))
        if total <= 0:
            break

        best_pass = best_c
        best_pair = None

        if total <= max_full:
            for o in ones:
                o = int(o)
                hb[o] = 0.0
                ohb[o] = 1.0
                for z in zeros:
                    z = int(z)
                    hb[z] = 1.0
                    ohb[z] = 0.0
                    val = float(np.max(np.correlate(hb, ohb, mode="full")) * dx)
                    if val < best_pass - 1e-15:
                        best_pass = val
                        best_pair = (o, z)
                    hb[z] = 0.0
                    ohb[z] = 1.0
                hb[o] = 1.0
                ohb[o] = 0.0
        else:
            samples = min(max_full, max(1000, total // 5))
            for _ in range(samples):
                oi = int(ones[int(rng.integers(len(ones)))])
                zi = int(zeros[int(rng.integers(len(zeros)))])
                hb[oi] = 0.0
                ohb[oi] = 1.0
                hb[zi] = 1.0
                ohb[zi] = 0.0
                val = float(np.max(np.correlate(hb, ohb, mode="full")) * dx)
                if val < best_pass - 1e-15:
                    best_pass = val
                    best_pair = (oi, zi)
                hb[oi] = 1.0
                ohb[oi] = 0.0
                hb[zi] = 0.0
                ohb[zi] = 1.0

        if best_pair is not None:
            o, z = best_pair
            o = int(o)
            z = int(z)
            hb[o] = 0.0
            ohb[o] = 1.0
            hb[z] = 1.0
            ohb[z] = 0.0

            ones = ones[ones != o]
            zeros = zeros[zeros != z]
            ones = np.append(ones, z)
            zeros = np.append(zeros, o)

            if best_pass < best_c - 1e-15:
                best_c = best_pass
                best_hb = hb.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 1:
                break

    return best_hb, best_c


def optimize_resolution(h, end_time, rng):
    n = h.size
    if n < 4 or time.time() >= end_time:
        h = feasible_copy(h)
        return h, c5_of(h)

    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = c5_of(h)

    while True:
        rem = end_time - time.time()
        if rem < 4.0:
            break

        if n % 2 == 0 and rem > 12.0:
            hb = to_binary(best_h)
            hb_c = c5_of(hb)
            if hb_c < best_c + 0.12:
                tb = time.time() + min(8.0, rem * 0.2)
                if tb > end_time - 2.0:
                    tb = end_time - 2.0
                if tb > time.time() + 1.0:
                    hb2, cb2 = binary_local_search(hb, tb, rng)
                    if cb2 < best_c - 1e-13:
                        best_h = hb2
                        best_c = cb2

        tc = time.time() + min(25.0, rem * 0.55)
        if tc > end_time - 2.0:
            tc = end_time - 2.0
        if tc > time.time() + 1.0:
            hf, cf = continuous_improve(best_h, tc, rng)
            if cf < best_c - 1e-13:
                best_h = hf
                best_c = cf

        tp = time.time() + min(10.0, rem * 0.2)
        if tp > end_time - 2.0:
            tp = end_time - 2.0
        if tp > time.time() + 1.0:
            hp, cp = pair_improve(best_h, tp, rng)
            if cp < best_c - 1e-13:
                best_h = hp
                best_c = cp

    return best_h, best_c


def resample_h(h, n_new):
    n_old = h.size
    if n_new <= 0:
        return np.empty(0, dtype=float)
    if n_old == 0:
        return feasible_copy(initial_random(n_new, np.random.default_rng(12345)))
    if n_old == n_new:
        return feasible_copy(h)
    if n_new > n_old and n_new % n_old == 0:
        return feasible_copy(np.repeat(h, n_new // n_old))

    x_old = (np.arange(n_old) + 0.5) / float(n_old)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    return feasible_copy(np.interp(x_new, x_old, h))


def try_resolution(n_stage, end_time, current_h, current_c, rng):
    if n_stage <= 0 or time.time() >= end_time:
        return current_h, current_c
    start = resample_h(current_h, n_stage)
    h, c = optimize_resolution(start, end_time, rng)
    if c < current_c - 1e-15:
        return h, c
    return current_h, current_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    reserve = 12.0 if budget > 120.0 else 2.0
    t_end = t0 + max(5.0, budget) - reserve
    if t_end < t0 + 2.0:
        t_end = t0 + 2.0

    calibrate_lag_mode()
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = np.empty(0, dtype=float)

    if parent.size == 0:
        parent = initial_random(80, rng)

    if parent.size % 2 != 0:
        parent = resample_h(parent, parent.size + 1)

    parent = feasible_copy(parent)
    best_h = parent.copy()
    best_c = c5_of(parent)

    if best_h.size >= 8 and best_h.size <= 256 and time.time() < t_end - 5.0:
        stage_end = t0 + min(0.10 * budget, 60.0)
        if stage_end > t0 + 3.0:
            best_h, best_c = try_resolution(best_h.size, min(stage_end, t_end), best_h, best_c, rng)

    if time.time() < t_end - 8.0:
        stage_end = min(t_end - 6.0, time.time() + min(300.0, 0.30 * budget))
        if stage_end > time.time() + 3.0:
            best_h, best_c = try_resolution(160, stage_end, best_h, best_c, rng)

    if time.time() < t_end - 8.0:
        stage_end = min(t_end - 6.0, time.time() + min(250.0, 0.24 * budget))
        if stage_end > time.time() + 3.0:
            best_h, best_c = try_resolution(240, stage_end, best_h, best_c, rng)

    if time.time() < t_end - 8.0:
        stage_end = min(t_end - 6.0, time.time() + min(220.0, 0.20 * budget))
        if stage_end > time.time() + 3.0:
            best_h, best_c = try_resolution(320, stage_end, best_h, best_c, rng)

    if time.time() < t_end - 15.0:
        stage_end = min(t_end - 6.0, time.time() + min(120.0, 0.12 * budget))
        if stage_end > time.time() + 3.0:
            best_h, best_c = try_resolution(480, stage_end, best_h, best_c, rng)

    if time.time() < t_end:
        best_h, best_c = try_resolution(best_h.size, t_end, best_h, best_c, rng)

    best_h = feasible_copy(best_h)
    best_h = set_exact_sum(best_h, 0.5 * best_h.size)
    best_h = np.clip(best_h, 0.0, 1.0)

    if best_h.size == 0 or not np.isfinite(best_c):
        best_h = np.full(80, 0.5, dtype=float)
        best_c = c5_of(best_h)

    best_c = c5_of(best_h)
    return best_h, float(best_c), int(best_h.size)
