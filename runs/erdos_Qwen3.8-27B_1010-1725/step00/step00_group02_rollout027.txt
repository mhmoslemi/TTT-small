    n = len(h)
    for _ in range(max_iters):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        ns = eval_score(h)
        if ns < s:
            s = ns
            improved = True
        else:
            h[a], h[b] = 1.0, 0.0
        if not improved and rng is None:
            break
    return h, s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.asarray(initial_h_values, dtype=np.float64)
    best_h = h0.copy()
    best_score = eval_score(h0)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SA at n=46
    dl1 = t0 + min(180, budget_s * 0.2)
    h46 = make_binary(h0, 46, rng)
    h46, s = sa_binary(h46, rng, dl1)
    if s < best_score:
        best_score, best_h = s, h46

    # Phase 2: SA at n=90 from best
    if remaining() > 80:
        dl2 = t0 + min(350, budget_s * 0.45)
        h90 = make_binary(best_h, 90, rng)
        h90, s = sa_binary(h90, rng, dl2)
        if s < best_score:
            best_score, best_h = s, h90

    # Phase 3: Golden ratio init at n=90
    if remaining() > 60:
        dl3 = t0 + min(500, budget_s * 0.6)
        hg = golden_init(90, rng)
        hg, s = sa_binary(hg, rng, dl3)
        if s < best_score:
            best_score, best_h = s, hg

    # Phase 4: Random restart at n=90
    if remaining() > 40:
        dl4 = t0 + budget_s - 40
        hr = np.zeros(90)
        hr[rng.choice(90, 45, replace=False)] = 1.0
        hr, s = sa_binary(hr, rng, dl4)
        if s < best_score:
            best_score, best_h = s, hr

    # Phase 5: SA at n=180 from best
    if remaining() > 30:
        dl5 = t0 + budget_s - 5
        h180 = make_binary(best_h, 180, rng)
        h180, s = sa_binary(h180, rng, dl5)
        if s < best_score:
            best_score, best_h = s, h180

    # Final: verify and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = eval_score(best_h)
    return best_h, min(best_score, final_score), len(best_h)
