        h3 = make_random_binary(n3, rng)
        deadline = t0 + T - reserve - max(remaining * 0.5, 40.0)
        h3, c5_3 = binary_sa(h3, n3, rng, deadline, t0)
        if c5_3 < overall_c5:
            overall_c5 = c5_3
            overall_best = h3
            overall_n = n3

    # Phase 4: Refine the best solution found
    time_elapsed = time.time() - t0
    remaining = T - reserve - time_elapsed
    if remaining > 30 and overall_best is not None:
        n_best = overall_n
        is_binary = np.allclose(overall_best, np.round(overall_best), atol=1e-8)
        if is_binary:
            deadline = t0 + T - reserve
            overall_best, overall_c5 = binary_sa(overall_best.astype(int), n_best, rng, deadline, t0)
        else:
            deadline = t0 + T - reserve
            overall_best, overall_c5 = sa_loop(overall_best, n_best, rng, deadline, t0)

    # Final cleanup and validation
    overall_best = np.clip(overall_best, 0, 1).astype(float)
    overall_best = fix_sum(overall_best, overall_n / 2.0)
    overall_c5 = compute_c5(overall_best, overall_n)

    return overall_best, overall_c5, overall_n
