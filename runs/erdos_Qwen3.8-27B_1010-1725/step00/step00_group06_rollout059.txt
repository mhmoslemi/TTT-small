    """One subgradient step on max J_k"""
    # Compute all J_k
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    # k_star is 0 to 2n-2, corresponding to lag k_star - (n-1)

        for i in range(n - kk):
            g[i] -= h[i + kk]

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_jk(h, n, k):
    """Gradient of J_k w.r.t. h for a single lag k"""
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n-kk])
        g[:n-kk] -= h[kk:]
    return g

def subgradient_opt(h, n, max_time=50.0, top_m=1):
    t0 = time.time()
    best = h.copy()
    best_c5 = compute_c5(h, n)
    h = h.copy()
    alpha = 0.1
    for it in range(100000):
        if time.time() - t0 > max_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top-m lags
        if top_m == 1:
            idx = int(np.argmax(corr))
            k = idx - (n - 1)
            g = grad_jk(h, n, k)
        else:
            top_idx = np.argsort(corr)[-top_m:]
            g = np.zeros(n)
            for ti in top_idx:
                k = int(ti) - (n - 1)
                g += grad_jk(h, n, k)
            g /= top_m
        h = project(h - alpha * g, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if it % 100 == 99:
            alpha *= 0.995
    return best, best_c5

def coord_descent(h, n, max_time=30.0):
    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h, n)
    steps = [0.0, 1.0, 0.1, -0.1, 0.2, -0.2, 0.3, -0.3]
    while time.time() - t0 < max_time:
        improved = False
        for i in range(n):
            base = h.copy()
            old = h[i]
            best_t = old
            best_c = c5
            for s in steps:
                t = old + s
                if t < 0 or t > 1:
                    continue
                h = base.copy()
                h[i] = t
                h = project(h, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_t = t
            if best_t != old:
                h = base.copy()
                h[i] = best_t
                h = project(h, n)
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def random_search(h, n, rng, max_time=30.0):
    t0 = time.time()
    best = h.copy()
    best_c5 = compute_c5(h, n)
    h = h.copy()
    scale = 0.5
    while time.time() - t0 < max_time:
        h_new = h.copy()
        n_perturb = rng.randint(1, 5)
        idxs = rng.choice(n, n_perturb, replace=False)
        for idx in idxs:
            h_new[idx] += rng.uniform(-scale, scale)
        h_new = project(h_new, n)
