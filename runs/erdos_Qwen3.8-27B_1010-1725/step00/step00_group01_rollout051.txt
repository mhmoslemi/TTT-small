            if 0 <= ik2 < n:
                t -= h[ik2]
            s += wi * t
        grad[j] = dx * s
    return grad

def smooth_obj_grad(h, emax):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp((c - m) / emax)
    s = np.sum(e)
    F = m + emax * np.log(s)
    w = e / s
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for mi in range(2 * n - 1):
            wi = w[mi]
            if wi < 1e-12:
                continue
            k = mi - (n - 1)
            t = 0.0
            ik = j - k
            if 0 <= ik < n:
                t += 1.0 - h[ik]
            ik2 = j + k
            if 0 <= ik2 < n:
                t -= h[ik2]
            val += wi * t
        grad[j] = dx * val
    return F, grad

def project_to_simplex_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to reduce
        idx = np.argsort(h)[::-1]
        for idx_i in idx:
            reduce = min(h[idx_i], -diff)  # h[idx_i] - reduce amounts
            if reduce > 0:
                h[idx_i] -= reduce
                diff += reduce
            if abs(diff) < 1e-12:
                break
    else:
        # need to increase
        idx = np.argsort(h)
        diff = -diff
        for idx_i in idx:
            increase = min(1.0 - h[idx_i], diff)
            if increase > 0:
                h[idx_i] += increase
                diff -= increase
            if abs(diff) < 1e-12:
                break
    return h
