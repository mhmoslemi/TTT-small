    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_sum(h, n):
    target = n / 2.0
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-7
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()

    c = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(c))
    d = k_star - (n - 1)
    
    g = np.zeros(n)
    if d == 0:
        g = 1.0 - 2.0 * h

    else:
        dd = -d
        for i in range(dd, n):
            if h[i] > 0.5 and h[i - dd] < 0.5:
                pairs.append(i)
    if len(pairs) == 0:
        return h
    i = pairs[int(rng.integers(len(pairs)))]
    j = int(rng.integers(n))
    if i == j:
        j = (j + 1) % n
    h[i], h[j] = h[j], h[i]
    return h

def run(seed=42, budget_s=1100, **kw):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    te = t0 + budget_s - 5

    iv = globals().get('initial_h_values', None)
    if iv is not None:
        bh = _proj(np.array(iv, dtype=float).copy(), len(iv))
        bn = len(bh)
    else:
        bn = 100
        bh = np.zeros(bn)
        bh[::2] = 1.0
    bc = _c5(bh)

    # Phase 1: swap local search
    h = bh.copy()
    h, c = _search(h, bn, t0 + budget_s * 0.25, rng)
    if c < bc:
        bc, bh = c, h.copy()

    # Phase 2: subgradient refinement
    h = bh.copy()
    h, c = _sg_phase(h, bn, t0 + budget_s * 0.40, rng)
    if c < bc:
        bc, bh = c, h.copy()

    # Phase 3: more local search with targeted moves
    h = bh.copy()
    T = 0.003
    best_local = bc
    for it in range(80000000):
        if time.time() > t0 + budget_s * 0.65:
            break
        move = rng.integers(3)
        if move == 0:
            i = int(rng.integers(bn))
            j = int(rng.integers(bn))
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
        elif move == 1:
            _seg_rev(h, bn, rng)
        else:
            _targeted_fix(h, bn, rng)
        v = _c5(h)
        if v < best_local or rng.random() < np.exp(-(v - best_local) / max(T, 1e-15)):
            if v < best_local:
                best_local = v
        else:
            if move == 0:
                h[j], h[i] = h[i], h[j]
            elif move == 1:
                pass  # keep reversal (diversity)
            else:
                pass
        if it % 300000 == 0:
            T *= 0.93
    if best_local < bc:
        bc, bh = best_local, h.copy()

    # Phase 4: try other resolutions
    for nt in (140, 200, 300):
        remaining = te - time.time()
        if remaining < 30:
            break
        x_old = np.linspace(0, 1, bn)
        x_new = np.linspace(0, 1, nt)
        h = np.interp(x_new, x_old, bh)
        h = _proj(h, nt)
        seg = min(remaining * 0.45, 90.0)
        h, c = _search(h, nt, time.time() + seg, rng)
        if c < bc:
            bc, bh, bn = c, h.copy(), nt

    # Final polish
    h = bh.copy()
    h, c = _search(h, bn, te, rng)
    if c < bc:
        bc, bh = c, h.copy()

    return bh, float(bc), bn
