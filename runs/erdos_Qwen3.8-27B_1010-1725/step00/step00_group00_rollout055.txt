        if (time.time() - t0) > budget_s * 0.65:
            break
        _g_dx, _g_T, _g_n = dx, float(T), n

        for r in range(n_r):
            if (time.time() - t0) > budget_s * 0.65:
                break
            if r == 0 and T == 200:
                h_s = best_h.copy()
            else:
                h_s = best_h + 0.02 * np.random.randn(n)
                h_s = project_feasible(h_s, n)
            try:
                res = minimize(_obj_smooth, h_s, method='SLSQP',
                              bounds=bounds_n,
                              constraints=[{'type': 'eq', 'fun': _con_eq}],
                              options={'maxiter': 500, 'ftol': 1e-12})
                h_opt = project_feasible(res.x, n)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 2: L-BFGS-B with penalty (faster per iteration)
    if (time.time() - t0) < budget_s * 0.4:
        _g_pen = 5e5
        for T in [20, 10, 5, 3]:
            if (time.time() - t0) > budget_s * 0.75:
                break
            _g_dx, _g_T, _g_n = dx, float(T), n
            for r in range(4):
                if (time.time() - t0) > budget_s * 0.75:
                    break
                h_s = best_h + 0.015 * np.random.randn(n)
                h_s = project_feasible(h_s, n)
                try:
                    res = minimize(_obj_pen, h_s, method='L-BFGS-B',
                                 bounds=bounds_n,
                                 options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10})
                    h_opt = project_feasible(res.x, n)
                    c5_opt = compute_c5(h_opt, dx)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                except Exception:
                    pass

    # Phase 3: Higher resolution attempts
    if (time.time() - t0) < budget_s * 0.3:
        for n_t in [90, 135]:
            if (time.time() - t0) > budget_s * 0.92:
                break
            dx_t = 2.0 / n_t
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.arange(n_t) * (2.0 / n_t)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_t)
            _g_dx, _g_n, _g_pen = dx_t, n_t, 5e5
            bounds_t = [(0.0, 1.0)] * n_t

            for T in [80, 40, 20, 10]:
                if (time.time() - t0) > budget_s * 0.92:
                    break
                _g_T = float(T)
                for r in range(2):
                    if (time.time() - t0) > budget_s * 0.92:
                        break
                    if r == 0 and T == 80:
                        h_s = h_up.copy()
                    else:
                        h_s = h_up + 0.015 * np.random.randn(n_t)
                        h_s = project_feasible(h_s, n_t)
                    try:
                        res = minimize(_obj_pen, h_s, method='L-BFGS-B',
                                     bounds=bounds_t,
                                     options={'maxiter': 400, 'ftol': 1e-14})
                        h_opt = project_feasible(res.x, n_t)
                        c5_opt = compute_c5(h_opt, dx_t)
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            best_h = h_opt.copy()
                            h_up = h_opt.copy()
                    except Exception:
                        pass

    # Phase 4: Final SLSQP polish on best solution
    if (time.time() - t0) < budget_s * 0.95:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        _g_dx, _g_n = dx_f, n_f
        bounds_f = [(0.0, 1.0)] * n_f
        for T in [3, 1.5]:
            if (time.time() - t0) > budget_s * 0.97:
                break
            _g_T = float(T)
            try:
                res = minimize(_obj_smooth, best_h, method='SLSQP',
                              bounds=bounds_f,
                              constraints=[{'type': 'eq', 'fun': _con_eq}],
                              options={'maxiter': 600, 'ftol': 1e-13})
                h_opt = project_feasible(res.x, n_f)
                c5_opt = compute_c5(h_opt, dx_f)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Final verification
    n_ret = len(best_h)
    dx_ret = 2.0 / n_ret
    best_h = project_feasible(best_h, n_ret)
    # Force exact integer sum
    diff = np.sum(best_h) - n_ret / 2.0
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] -= diff / np.count_nonzero(mask)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, dx_ret)
    return (best_h, final_c5, n_ret)
