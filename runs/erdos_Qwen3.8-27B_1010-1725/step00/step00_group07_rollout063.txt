                h = h_adj.copy()
            h[i] -= delta
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if budget_s is None:
        budget_s = 1100
    rng = np.random.default_rng(seed)

    h0 = project_to_feasible(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize from initial (45% budget)
    end1 = t0 + budget_s * 0.45
    h1, c5_1 = opt_single(h0, end1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    # Polish
    if time.time() < end1 + budget_s * 0.03:
        hp, cp = local_polish(best_h, 300)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 2: Random restarts (30% budget)
    end2 = t0 + budget_s * 0.78
    n_rest = 8
    for i in range(n_rest):
        remaining_restarts = n_rest - i
        sub_time = (end2 - time.time()) / remaining_restarts
        if sub_time < 10:
            break
        h_r = random_smooth_h(n, rng)
        sub_end = min(end2, time.time() + sub_time * 0.7)
        hr_opt, c5_r = opt_single(h_r, sub_end, n_betas=4)
        if c5_r < best_c5:
            best_h, best_c5 = hr_opt, c5_r

    # Try binary-like starts
    if time.time() < end2:
        for trial in range(3):
            if time.time() > end2 - 5:
                break
            h_bin = (rng.random(n) < 0.5).astype(float)
            h_bin = project_to_feasible(h_bin, n / 2.0)
            # Smooth it slightly
            from scipy.ndimage import uniform_filter1d
            h_bin = uniform_filter1d(h_bin, size=3)
            h_bin = project_to_feasible(h_bin, n / 2.0)
            sub_end = min(end2, time.time() + 30)
            hb_opt, c5_b = opt_single(h_bin, sub_end, n_betas=4)
            if c5_b < best_c5:
                best_h, best_c5 = hb_opt, c5_b

    # Phase 3: Upscale to 2n (20% budget)
    remaining = budget_s - (time.time() - t0)
    if remaining > 90:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_to_feasible(h_up, n2 / 2.0)
        end3 = t0 + budget_s - 10
        h_up_opt, c5_up = opt_single(h_up, end3, n_betas=4)
        # Also try a direct optimize on the upsampled version with polish
        if time.time() < end3:
            h_up_opt2, c5_up2 = local_polish(h_up_opt, 200)
            if c5_up2 < c5_up:
                h_up_opt, c5_up = h_up_opt2, c5_up2
        if c5_up < best_c5:
            best_h, best_c5 = h_up_opt, c5_up

    # Final polish on best solution
    if time.time() < t0 + budget_s - 5:
        hp2, cp2 = local_polish(best_h, 150)
        if cp2 < best_c5:
            best_h, best_c5 = hp2, cp2

    final_c5 = compute_c5(best_h)
    final_n = len(best_h)
    return best_h, final_c5, final_n
