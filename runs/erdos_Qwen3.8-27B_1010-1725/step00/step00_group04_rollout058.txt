        if jlo2 < jhi2:

    grad *= _g_dx
    grad += 2.0 * _g_penalty_w * (np.sum(x) - _g_target_sum)
    return grad

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return repair_h(h_new)

def optimize_at(h, n, T_list, max_iter, time_budget_left):
    global _g_T, _g_dx, _g_target_sum
    _g_target_sum = n / 2.0
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    for T_val in T_list:
        if time_budget_left() <= 30:
            break
        _g_T = T_val
        res = minimize(_obj, h, method='L-BFGS-B', jac=_grad,
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-13, 'gtol': 1e-9})
        h = repair_h(res.x)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _g_T, _g_dx, _g_target_sum
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    np.random.seed(seed)

    # Start from initial construction
    h = np.array(initial_h_values, dtype=np.float64)
    h = repair_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Optimize at original resolution (n=42)
    n0 = len(h)
    h = optimize_at(h, n0, [30.0, 60.0, 120.0, 250.0, 500.0], 400, time_left)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Upscale and optimize at higher resolutions
    resolutions = [84, 168, 336]
    h_cur = h
    for n_new in resolutions:
        if time_left() < 60:
            break
        h_up = upsample_h(h_cur, n_new)
        h_cur = optimize_at(h_up, n_new, [60.0, 120.0, 250.0], 300, time_left)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()

    # Phase 3: Try perturbation + re-optimization at best resolution
    if time_left() > 120:
        n_best = len(best_h)
        h_pert = best_h.copy()
        mask = np.random.rand(n_best) < 0.3
        h_pert[mask] += np.random.uniform(-0.1, 0.1, mask.sum())
        h_pert = repair_h(h_pert)
        h_pert = optimize_at(h_pert, n_best, [100.0, 200.0], 300, time_left)
        c5 = compute_c5(h_pert)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()

    # Phase 4: Try a few random binary-ish restarts at moderate resolution
    if time_left() > 90:
        n_r = 84
        for _ in range(3):
            if time_left() < 40:
                break
            h_r = np.zeros(n_r)
            idx = np.random.choice(n_r, size=n_r // 2, replace=False)
            h_r[idx] = 1.0
            h_r = repair_h(h_r)
            h_r = optimize_at(h_r, n_r, [50.0, 150.0, 300.0], 250, time_left)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()

    # Final repair and evaluation
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
