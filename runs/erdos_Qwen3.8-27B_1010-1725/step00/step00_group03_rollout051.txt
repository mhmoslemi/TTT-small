    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.asarray(h, dtype=float).copy()

                return soft_c5(x, n, alpha)

                return np.ones(n)

                obj, h_best.copy(),
                method='SLSQP',
                bounds=bounds,

                c5_new = hard_c5(h_new, n)
                if c5_new < c5_best - 1e-12:

                    c5_best = c5_new
        except Exception:
            pass

    if time.time() - t0 < time_limit_s:
        h_best = coord_descent(h_best, n, time.time() - t0, time_limit_s)
        c5_best = hard_c5(h_best, n)
    
    return h_best, c5_best

def coord_descent(h, n, elapsed, total_time):
    """Simple coordinate descent: adjust one coordinate at a time"""
    t0 = time.time()
    c5_curr = hard_c5(h, n)
    step = 0.01
    for _ in range(2000):
        if time.time() - t0 > total_time - elapsed:
            break
        for i in range(n):
            if time.time() - t0 > total_time - elapsed:
                    break
            # Try moving h[i] up or down
            for delta in [step, -step, 2*step, -2*step]:
                h_new = h.copy()
                h_new[i] += delta
                # Fix constraint by adjusting another element
                diff = n / 2.0 - np.sum(h_new)
                # Adjust element j != i
                j = (i + 1) % n
                h_new[j] -= diff
                if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                    c5_new = hard_c5(h_new, n)
                    if c5_new < c5_curr:
                        h = h_new
                        c5_curr = c5_new
    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    final_diff = n / 2.0 - np.sum(h)
    if abs(final_diff) > 1e-14:
        idx = np.argmax(h) if final_diff < 0 else np.argmin(h)
        h[idx] += final_diff
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_c5(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -60.0, 0.0)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(s)))

def interp_to(parent, n):
    x_old = np.linspace(0.0, 1.0, len(parent) + 1)
    h_ext = np.concatenate([parent, [parent[0]]])
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, h_ext)

    c5_best = evaluate_c5(h_best, n)

                lambda x: soft_c5(x, n, alpha),
                h_best.copy(),

                bounds=bounds,

                              'fun': lambda x: np.sum(x) - n / 2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5_new = evaluate_c5(h_new, n)
                if c5_new < c5_best:
                    h_best = h_new
                    c5_best = c5_new
        except Exception:
            pass

        h_best = pair_swap_refine(h_best, n, time.time() - t0, t_limit)
        c5_best = evaluate_c5(h_best, n)
    return h_best, c5_best

def pair_swap_refine(h, n, elapsed, t_limit):
    t0 = time.time()
    c5_curr = evaluate_c5(h, n)
    for _ in range(500):
        if time.time() - t0 + elapsed > t_limit:
            break
        improved = False
        for i in range(0, n, max(1, n // 40)):
            if time.time() - t0 + elapsed > t_limit:
                break
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            for frac in (0.05, 0.1, 0.2, 0.05, 0.1, 0.2):
                for sign in (1, -1):

                    hj = h[j] - sign * frac

                            break
