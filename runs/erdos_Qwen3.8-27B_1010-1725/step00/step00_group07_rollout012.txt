import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(corr, T):
    M = np.max(corr)
    ex = np.exp(T * (corr - M))
    sw = np.sum(ex)
    fval = M + np.log(sw) / T
    w = ex / sw
    return fval, w

def compute_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    fval, w = smooth_max_grad(corr, T)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k] * dx
        if wi < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += wi * (1.0 - h[lag:])
                grad[lag:] -= wi * h[:n - lag]
        else:
            ll = -lag
            if n - ll > 0:
                grad[ll:] += wi * (1.0 - h[:n - ll])
                grad[:n - ll] -= wi * h[ll:]
    return fval, grad, w

def objective_fn(h_flat, n, target, T):
    h = np.clip(h_flat, 1e-10, 1.0 - 1e-10)
    dx = 2.0 / n
    fval, grad, _ = compute_grad(h, dx, T)
    sum_pen = 50.0 * (np.sum(h) - target) ** 2
    fval += sum_pen
    grad = grad + 100.0 * (np.sum(h) - target) * np.ones(n)
    return fval, grad

def optimize_lbfgs(h_start, T_schedule, max_iter_total, time_limit, t_start):
    n = len(h_start)
    target = n / 2.0
    h = project_to_sum(h_start.copy(), target)
    total_iter = 0
    for T in T_schedule:
        if time.time() - t_start > time_limit:
            break
        while total_iter < max_iter_total and time.time() - t_start < time_limit:
            res = minimize(objective_fn, h.copy(), method='L-BFGS-B',
                          args=(n, target, T),
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
            total_iter += res.nit
            h_new = project_to_sum(np.clip(res.x, 0, 1), target)
            if compute_c5_val(h_new) < compute_c5_val(h):
                h = h_new
            else:
                break
    return h

def coordinate_descent(h, max_iter=200, time_limit=None, t_start=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_to_sum(h.copy(), target)
    c5 = compute_c5_val(h)
    for it in range(max_iter):
        if time_limit and t_start and time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(0, n, 3):
            if time_limit and t_start and time.time() - t_start > time_limit:
                break
            j = (i + 1 + (it % (n - 1))) % n
            if i == j:
                continue
            lo = max(-h[i], -(h[j]))
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-10:
                continue
            # Try a few delta values
            deltas = [0.01, -0.01, 0.05, -0.05, (hi + lo) / 2, (hi - lo) * 0.3, -(hi - lo) * 0.3]
            best_delta = 0.0
            best_c5 = c5
            for d in deltas:
                d = np.clip(d, lo, hi)
                if abs(d) < 1e-12:
                    continue
                h_try = h.copy()
                h_try[i] += d
                h_try[j] -= d
                c5_try = compute_c5_val(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_delta = d
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                c5 = best_c5
                improved = True
        if not improved:
            break
    return h
