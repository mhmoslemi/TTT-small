                h_t = project_feasible(h_t, n0)
                c_t = eval_c5(h_t)
                if c_t < best_c5 - 1e-10:
                    best_h, best_c5 = h_t, c_t
                    improved = True
                    break
        if improved:
            stall = 0
            if tleft(0.3):
                h_new = slsqp_opt(best_h, n0, 0.002, 300)
                h_new = project_feasible(h_new, n0)
                c_new = eval_c5(h_new)
                if c_new < best_c5:
                    best_c5, best_h = c_new, h_new
        else:
            stall += 1

    # Phase 5: Finer grid interpolation + optimization
    if tleft(0.25):
        for n_f in [90, 180]:
            if not tleft(0.2):
                break
            x0 = np.linspace(0, 2, n0, endpoint=False)
            x1 = np.linspace(0, 2, n_f, endpoint=False)
            h_f = project_feasible(np.interp(x1, x0, best_h), n_f)
            for tau in [0.015, 0.006, 0.002]:
                if not tleft(0.18):
                    break
                h_new = slsqp_opt(h_f, n_f, tau, 500)
                h_new = project_feasible(h_new, n_f)
                c_new = eval_c5(h_new)
                if c_new < best_c5:
                    best_c5, best_h = c_new, h_new

    # Final: tight SLSQP polish
    if tleft(0.05):
        nf = len(best_h)
        h_new = slsqp_opt(best_h, nf, 0.001, 2000)
        h_new = project_feasible(h_new, nf)
        c_new = eval_c5(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, len(best_h)
