    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def compute_c5_components(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx

def binary_local_search(h_init, n, max_rounds=500, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    target = n // 2
    b = (h_init >= 0.5).astype(float)
    # Fix to exactly target ones
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    if len(ones) > target:
        excess = rng.choice(ones, size=len(ones)-target, replace=False)
        b[excess] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        b[zeros[rng.choice(len(zeros), size=deficit, replace=False)]] = 1
    
    current_c5 = compute_c5(b)
    no_improve = 0
    for _ in range(max_rounds):
        ones_idx = np.where(b == 1)[0]
        zeros_idx = np.where(b == 0)[0]
        best_val = current_c5
        best_swap = None
        # Try all swaps (for small n)
        for oi in ones_idx:
            for zi in zeros_idx:
                b[oi] = 0
                b[zi] = 1
                val = compute_c5(b)
                b[oi] = 1
                b[zi] = 0

                    best_swap = (oi, zi)

            current_c5 = best_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                break
    return b, current_c5

    elif len(ones) < target:
        deficit = target - len(ones)
        b[zeros[rng.choice(len(zeros), size=deficit, replace=False)]] = 1
    
    current_c5 = compute_c5(b)
    best_b = b.copy()
    best_c5 = current_c5
    
    for r in range(max_rounds):
        T = T_init * (T_min / T_init) ** (r / max_rounds)
        ones_idx = np.where(b == 1)[0]
        zeros_idx = np.where(b == 0)[0]
        oi = int(rng.choice(ones_idx))
        zi = int(rng.choice(zeros_idx))
        b[oi] = 0
        b[zi] = 1
        new_c5 = compute_c5(b)
        delta = new_c5 - current_c5
