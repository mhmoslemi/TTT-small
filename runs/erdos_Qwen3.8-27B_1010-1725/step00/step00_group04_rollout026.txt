```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, n, temp):
    """Compute smoothed max of corr*dx and its gradient w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # length 2n-1
    m = np.max(c)
    shifted = (c - m) / temp
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_val = m + temp * np.log(sum_exp / len(c))
    weights = exp_shifted / sum_exp  # softmax weights
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-12:
            continue
        tau = k - (n - 1)
        a = max(0, -tau)
        b = min(n - 1, n - 1 - tau)
        w = weights[k] * dx
        if a <= b:
            # First term: (1-h[m+tau]) for m in [a, b]
            idx1 = np.arange(a, b + 1)
            grad[idx1] += w * (1.0 - h[idx1 + tau])
            # Second term: -h[m-tau] for m in [a+tau, b+tau]
            c_start = max(0, a + tau)
            c_end = min(n - 1, b + tau)
            if c_start <= c_end:
                idx2 = np.arange(c_start, c_end + 1)
                grad[idx2] -= w * h[idx2 - tau]
    
    return smooth_val, grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 100
        h = np.concatenate([np.ones(n//2), np.zeros(n//2)])
    
    n = len(h)
    # Ensure sum constraint
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n/2.0) > 1e-10:
        # Simple projection: adjust to meet sum constraint
        # If sum too high, reduce largest; if too low, increase smallest
        target = n / 2.0
        diff = target - s
        # Distribute uniformly
        h += diff / n
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1: L-BFGS-B with smooth max + penalty
    for temp in [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
        if elapsed() > budget_s * 0.8:
            break
        lam = 10.0
        result = minimize(
            lambda x: smooth_max_and_grad(np.clip(x,0,1), n, temp)[0] + lam*(np.sum(np.clip(x,0,1))-n/2)**2,
            h,
            method='L-BFGS-B',
            bounds=[(0,1)]*n,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        h_new = np.clip(result.x, 0, 1)
        # Project to sum constraint
        s = np.sum(h_new)
        h_new += (n/2.0 - s) / n
        h_new = np.clip(h_new, 0, 1)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return (best_h, best_c5, n)
```
