import numpy as np
from scipy.optimize import minimize
import time

def project_sparse(h, target_sum):
    n = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def sg_subgradient(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(np.max(corr))
    # Use top 5 for robustness
    top_idx = np.argsort(corr)[-5:]
    idx = np.arange(n)
    sgrad = np.zeros(n)
    for j in top_idx:
        k = j - (n - 1)
        pm = (idx + k >= 0) & (idx + k < n)
        mm = (idx - k >= 0) & (idx - k < n)
        sgrad[pm] += dx * (1.0 - h[idx[pm] + k])
        sgrad[mm] -= dx * h[idx[mm] - k]
    sgrad /= 5.0
    return sgrad

def subgrad_descent(h0, n_iter, alpha_init, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    t_start = time.time()
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        sgrad = sg_subgradient(h, dx)
        h = project_sparse(h - alpha * sgrad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        alpha = max(0.0003, alpha * 0.9997)
    return best_h, best_c5

def smooth_objective(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for numerical stability
    m = np.max(corr * beta)
    val = (1.0 / beta) * (m + np.log(np.sum(np.exp(corr * beta - m))))
    return val

def smooth_gradient(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = beta * corr
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    weights = exp_z / np.sum(exp_z)
    # Gradient of sum_j weights[j] * corr[j] * dx w.r.t. h
    idx = np.arange(n)
    grad = np.zeros(n)
    for j in range(len(corr)):
        if weights[j] < 1e-8:
            continue
        k = j - (n - 1)
        pm = (idx + k >= 0) & (idx + k < n)
        mm = (idx - k >= 0) & (idx - k < n)
        grad[pm] += dx * weights[j] * (1.0 - h[idx[pm] + k])
        grad[mm] -= dx * weights[j] * h[idx[mm] - k]
    return grad

def project_sparse(h, target_sum):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def corr_and_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    lags = np.arange(len(corr)) - (n - 1)
    return corr, lags

def subgrad_of_max(h, dx, top_k=5):
    n = len(h)
    corr, lags = corr_and_lags(h)
    top_idx = np.argsort(corr)[-top_k:]
    idx = np.arange(n)
    sgrad = np.zeros(n)
    for j in top_idx:
        k = lags[j]
        mm = (idx - k >= 0) & (idx - k < n)
        pm = (idx + k >= 0) & (idx + k < n)
        sgrad[mm] += (1.0 - h[idx[mm] - k])
        sgrad[pm] -= h[idx[pm] + k]
    sgrad *= dx / top_k
    return sgrad
