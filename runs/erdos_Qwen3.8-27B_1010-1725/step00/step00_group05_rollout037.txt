import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h0 = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h0 + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h0 + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def sa_optimize(h, time_limit, rng, t_hi=0.003, t_lo=1e-7, s0=0.005):
    n = len(h)
    t0 = time.time()
    curr = h.copy()
    curr_v = compute_c5(curr)
    best = h.copy()
    best_v = curr_v
    while time.time() - t0 < time_limit:
        frac = min(1.0, (time.time() - t0) / time_limit)
        t = t_hi * (t_lo / t_hi) ** frac
        s = s0 * (1.0 - 0.9 * frac) + 0.0003
        for _ in range(3000):
            if time.time() - t0 >= time_limit:
                break
            nh = curr + rng.uniform(-s, s, n)
            nh = project_feasible(nh, n)
            nv = compute_c5(nh)
            d = nv - curr_v
            if d < 0.0 or (t > 1e-9 and rng.random() < np.exp(-d / t)):
                curr, curr_v = nh, nv
                if nv < best_v:
                    best, best_v = nh.copy(), nv
    return best, best_v

def subgrad_dir(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    active = np.where(corr > mv - 0.01)[0]
    center = n - 1
    g = np.zeros(n)
    for k in active:
        m = k - center
        if m == 0:
            g += 1.0 - 2.0 * h
        elif m > 0:
            g[m:] += 1.0 - h[:n-m]
            g[:n-m] -= h[m:]
        else:
            m = -m
            g[:n-m] += 1.0 - h[m:]
            g[m:] -= h[:n-m]
    return g / max(1, len(active))

def softmax_grad(h, n, beta=50):
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Soft-max weights
    shifted = corr - np.max(corr)
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    center = n - 1
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-6:

        if m == 0:

            g[:n-m] -= weights[k] * h[m:]
        else:
            m = -m
            g[:n-m] += weights[k] * (1.0 - h[m:])
            g[m:] -= weights[k] * h[:n-m]
    return g
