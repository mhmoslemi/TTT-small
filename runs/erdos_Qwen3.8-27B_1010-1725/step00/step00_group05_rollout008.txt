        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += deficit / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_grad_lag(h, dx, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        return (1.0 - 2.0 * h) * dx
    m = abs(lag)
    if lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad * dx

def topk_subgrad(h, dx, k):
    n = len(h)
    corrs = np.correlate(h, 1-h, mode="full")
    top_idx = np.argsort(corrs)[-k:]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        grad += corr_grad_lag(h, dx, lag)
    return grad / k

def obj_func(x):
    return compute_c5(x, _M[1])

def obj_jac(x):
    return topk_subgrad(x, _M[1], _M[2])

def cons_eq_fun(x):
    return np.sum(x) - _M[0] / 2.0

def cons_eq_jac(x):
    return np.ones(_M[0])

def thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def fix_binary_count(b, n):
    target = n // 2
    oc = int(np.sum(b))
    if oc > target:
        idx = np.where(b == 1)[0]
        b[idx[:oc - target]] = 0.0
    elif oc < target:
        idx = np.where(b == 0)[0]
        b[idx[:target - oc]] = 1.0
    return b

def in_time(t0, limit):
    return (time.time() - t0) < limit

def do_slsqp(h_start, n, max_iter):
    h_start = make_feasible(h_start, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    try:
        res = minimize(obj_func, h_start, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       maxiter=max_iter, ftol=1e-10)
        if not np.isnan(res.fun):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return None

def do_nm(h_start, n, max_iter):
    h_start = make_feasible(h_start, n)
    try:
        res = minimize(obj_func, h_start, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-7, 'fatol': 1e-9})
        if not np.isnan(res.fun):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return None

def gen_random_binary(n, rng):
    b = (rng.rand(n) < 0.5).astype(float)
    return fix_binary_count(b, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _M
    rng = np.random.RandomState(seed)
    t0 = time.time()
    n = 70
    dx = 2.0 / n
    _M = [n, dx, 5]
    
    best_c5 = float('inf')
    best_h = None
    
    starts = []
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.ndim == 1 and h0.shape[0] == n:
            starts.append(h0.copy())
    except Exception:
        pass
    
    tm = thue_morse(n)
    starts.append(fix_binary_count(tm, n))
    for _ in range(4):
        starts.append(gen_random_binary(n, rng))
    starts.append(np.ones(n) * 0.5)
    
    # Phase 1: SLSQP from each start
    for hs in starts:
        if not in_time(t0, budget_s - 80):
            break
        h_opt = do_slsqp(hs, n, 600)
        if h_opt is not None:
            c = compute_c5(h_opt, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
    
    # Phase 2: Nelder-Mead from best and perturbed
    if best_h is not None and in_time(t0, budget_s - 60):
        h_opt = do_nm(best_h, n, 500)
        if h_opt is not None:
            c = compute_c5(h_opt, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
    
    # Phase 3: Random restarts + SLSQP
    for restart_k in range(30):
        if not in_time(t0, budget_s - 40):
            break
        scale = 0.04 * (1.0 - restart_k / 40.0) + 0.01
        h_try = best_h + rng.normal(0, scale, n)
        h_try = make_feasible(h_try, n)
        h_opt = do_slsqp(h_try, n, 400)
        if h_opt is not None:
            c = compute_c5(h_opt, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
        # Also try Nelder-Mead on perturbation
        if in_time(t0, budget_s - 40):
            h_opt2 = do_nm(h_try, n, 300)
            if h_opt2 is not None:
                c2 = compute_c5(h_opt2, dx)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_opt2.copy()
    
    # Phase 4: Fine SLSQP polish
    if best_h is not None and in_time(t0, budget_s - 20):
        for polish_k in range(10):
            if not in_time(t0, budget_s - 10):
                break
            h_opt = do_slsqp(best_h, n, 800)
            if h_opt is not None:
                c = compute_c5(h_opt, dx)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
                else:
                    break
    
    # Fallback
    if best_h is None:
        best_h = make_feasible(starts[0], n)
        best_c5 = compute_c5(best_h, dx)
    
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
