            h_try = fix_sum(h_try, n_best)
        else:
            h_try = best_h.copy()
            swap = rng.choice(n_best, 4, replace=False)
            for j in range(0, 4, 2):
                h_try[swap[j]], h_try[swap[j+1]] = h_try[swap[j+1]], h_try[swap[j]]
            h_try = fix_sum(h_try, n_best)

        h_opt = slsqp_opt(h_try, n_best, max_iter=500, alpha=400.0)
        c5_val = evaluate_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Phase 4: Fine polish at best resolution with tighter alpha
    if remaining() > 10:
        h_opt = slsqp_opt(best_h, n_best, max_iter=2000, alpha=1000.0)
        c5_val = evaluate_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
