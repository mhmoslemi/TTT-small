import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(h, dx, target_sum, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    s = np.sum(np.exp((vals - m) / temp))
    obj = m + temp * np.log(s)
    obj += 1e5 * (np.sum(h) - target_sum) ** 2
    return obj


def _interpolate_to(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h)


def _optimize_phase(h_start, n, temps):
    dx = 2.0 / n
    ts = n / 2.0
    best = h_start.copy()
    best_c5 = _compute_c5(best)
    for temp in temps:
        res = minimize(
            _smooth_obj, best, args=(dx, ts, temp),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 600, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        h_new = _fix_sum(np.clip(res.x, 0.0, 1.0), ts)
        c5 = _compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
    return best, best_c5
