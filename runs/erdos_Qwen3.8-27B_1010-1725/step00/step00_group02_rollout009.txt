def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def adjust_sum(h, target):
    """Adjust h to have sum = target while keeping in [0,1]"""
    n = len(h)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute uniformly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If clipping changed sum, iterate
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        mask = (h < 1) if diff > 0 else (h > 0)
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def local_search(h, max_iters, eps_min, eps_max, time_limit=None, t0=None):
    n = len(h)
    h = h.copy().astype(float)
    best = eval_c5(h)
    
    for it in range(max_iters):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        
        i = np.random.randint(n)
        j = np.random.randint(n - 1) + 1
        while j == i:
            j = np.random.randint(n - 1) + 1
        eps = np.random.uniform(eps_min, eps_max)
        
        improved = False
        # Direction 1: h[i] decreases, h[j] increases
        if h[i] >= eps and h[j] + eps <= 1.0:
            h[i] -= eps
            h[j] += eps
            c5 = eval_c5(h)
            if c5 < best:

                h[i] += eps
                c5 = eval_c5(h)
                if c5 < best:
                    best = c5
                else:
                    h[j] += eps
                    h[i] -= eps
