import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, alpha):
    a = alpha * x
    a_max = float(np.max(a))
    shifted = a - a_max
    return a_max / alpha + float(np.log(np.sum(np.exp(shifted)))) / alpha


def obj_func(params, n, alpha):
    h = np.clip(params, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = smooth_max_val(corr * dx, alpha)
    target = n / 2.0
    penalty = 1e6 * (np.sum(h) - target) ** 2
    return c5 + penalty


def repair_sum(h, n):
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_int = int(np.sum(interior))
        if n_int > 0:
            h[interior] += diff / n_int
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def optimize_res(h_start, n, alpha_list, max_iter_each):
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        try:
            result = minimize(obj_func, best_h.copy(), args=(n, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': max_iter_each,
                                       'maxfun': max_iter_each * 25,
                                       'ftol': 1e-10})
            h_new = repair_sum(result.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new <= best_c5 + 1e-12:
                best_c5 = min(best_c5, c5_new)
                best_h = h_new.copy()
        except Exception:
            continue
    return best_h, best_c5


def try_binarize(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0).copy()
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    excess = n_ones - target_ones
    if excess > 0:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = ones_idx[np.argsort(h[ones_idx])]
        h_bin[order[:excess]] = 0.0
    elif excess < 0:
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = zeros_idx[np.argsort(-h[zeros_idx])]
        h_bin[order[:(-excess)]] = 1.0
    return h_bin


def upsample_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    ratio = n_new / n_old
    h_new = np.zeros(n_new, dtype=np.float64)
    for i in range(n_new):
        pos = i / ratio
        lo = int(np.floor(pos))
        hi = min(lo + 1, n_old - 1)
        frac = pos - lo
        h_new[i] = (1.0 - frac) * h[lo] + frac * h[hi]
    return h_new


def perturb_and_retry(h, n, alpha_list, max_iter, rng):
    sigma = 0.05
    h_pert = h + sigma * rng.standard_normal(n)
    h_pert = repair_sum(h_pert, n)
    return optimize_res(h_pert, n, alpha_list, max_iter)
