import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_gradient(h, n, alpha=50.0):
    """Gradient of smoothed max using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # log-sum-exp for stability
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    # For each k in correlation, compute gradient
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        # d c[k]/dh[i] = (1-h[i+lag])*I - h[i-lag]*I
        # First term: h[i] as first factor
        valid1_start = max(0, -lag)  # i such that i+lag >= 0
        valid1_end = min(n-1, n-1-lag)  # i such that i+lag <= n-1
        if valid1_end >= valid1_start:
            i_range = np.arange(valid1_start, valid1_end + 1)
            grad[i_range] += weights[k] * (1 - h[i_range + lag]) * dx
        
        # Second term: h[i-lag] as part of (1-h[j+lag])
        # This means j+lag = i, so j = i-lag
        # h[i-lag] must be in [0,n-1], so i in [lag, lag+n-1] ∩ [0, n-1]
        valid2_start = max(0, lag)  # i-lag >= 0
        valid2_end = min(n-1, n-1+lag)  # i-lag <= n-1
        if valid2_end >= valid2_start:
            i_range = np.arange(valid2_start, valid2_end + 1)
            grad[i_range] -= weights[k] * h[i_range - lag] * dx
    
    return grad

def c5_smooth_grad(h, n, alpha=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    max_s = np.max(scaled)
    exp_vals = np.exp(alpha * (scaled - max_s))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        # First term
        s1 = max(0, -lag)
        e1 = min(n - 1, n - 1 - lag)
        if e1 >= s1:
            grad[s1:e1+1] += w[k] * (1.0 - h[s1+lag:e1+1+lag])
        # Second term
        s2 = max(0, lag)
        e2 = min(n - 1, n - 1 + lag)
        if e2 >= s2:
            grad[s2:e2+1] -= w[k] * h[s2-lag:e2+1-lag]
    
    return grad * dx

def project_feasible(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on threshold
    lo, hi = -1.0, 1.0  # h = clip(v - t, 0, 1)
    for _ in range(100):
        t = (lo + hi) / 2.0
        h = np.clip(v - t, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = t  # increase t to reduce sum
        else:
            hi = t
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune to get exact sum
    # Adjust the element with most room
    diff = target - np.sum(h)

        idx = np.argmax(np.minimum(h, 1.0 - h))  # element with most room
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_smooth_grad(h, n, alpha=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    max_s = np.max(scaled)
    exp_vals = np.exp(alpha * (scaled - max_s))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        # First term: d/dh[i] of h[i]*(1-h[i+lag])
        s1 = max(0, -lag)
        e1 = min(n - 1, n - 1 - lag)
        if e1 >= s1:
            grad[s1:e1+1] += w[k] * (1.0 - h[s1+lag:e1+1+lag])
        # Second term: d/dh[i] of -h[i-lag]*h[i]  (from expanding)
        # Actually: d/dh[i] of h[j]*(1-h[j+lag]) where j+lag=i, j=i-lag
        s2 = max(0, lag)
        e2 = min(n - 1, n - 1 + lag)
        if e2 >= s2:
            grad[s2:e2+1] -= w[k] * h[s2-lag:e2+1-lag]
    return grad * dx

    lo = v.min() - 0.1
    hi = v.max() + 0.1
    for _ in range(100):
        t = (lo + hi) * 0.5
        s = np.sum(np.clip(v - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) * 0.5
    h = np.clip(v - t, 0.0, 1.0)

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

    for i in range(n_old):
        for j in range(factor):

    n_new = n_old // factor
    h_new = np.zeros(n_new)
    for i in range(n_new):
        h_new[i] = np.mean(h[i*factor:(i+1)*factor])
    return h_new

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def c5_smooth_grad(h, n, alpha=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    max_s = np.max(scaled)
    exp_v = np.exp(alpha * (scaled - max_s))
    w = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        s1 = max(0, -lag)
        e1 = min(n - 1, n - 1 - lag)
        if e1 >= s1:
            grad[s1:e1+1] += w[k] * (1.0 - h[s1+lag:e1+1+lag])
        s2 = max(0, lag)
        e2 = min(n - 1, n - 1 + lag)
        if e2 >= s2:
            grad[s2:e2+1] -= w[k] * h[s2-lag:e2+1-lag]
    return grad * dx

def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 0.5
    hi = v.max() + 0.5
    for _ in range(80):
        t = (lo + hi) * 0.5
        if np.sum(np.clip(v - t, 0.0, 1.0)) > target:
            lo = t
        else:
            hi = t
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_resolution(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def optimize_at_n(n, h0, time_limit, alpha_init=30.0):
    h = project_feasible(h0.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

        v = beta * v + (1 - beta) * g

        c5_new = compute_c5(h_new, n)
        if c5_new < c5 - 1e-12:

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def c5_smooth_grad(h, n, alpha=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    max_s = np.max(scaled)
    exp_v = np.exp(alpha * (scaled - max_s))
    w = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        s1 = max(0, -lag)
        e1 = min(n - 1, n - 1 - lag)
        if e1 >= s1:
            grad[s1:e1 + 1] += wk * (1.0 - h[s1 + lag:e1 + 1 + lag])
        s2 = max(0, lag)
        e2 = min(n - 1, n - 1 + lag)
        if e2 >= s2:
            grad[s2:e2 + 1] -= wk * h[s2 - lag:e2 + 1 - lag]
    return grad * dx


def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 0.5
    hi = v.max() + 0.5
    for _ in range(80):
        t = (lo + hi) * 0.5
        if np.sum(np.clip(v - t, 0.0, 1.0)) > target:
            lo = t
        else:
            hi = t
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        m = (h > 1e-8) & (h < 1.0 - 1e-8)
        if m.any():
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def interp_to_res(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)


def optimize_at_n(n, h0, time_limit, alpha_init=30.0, beta_mom=0.5):
    h = project_feasible(h0.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.1
    alpha = alpha_init
    v = np.zeros(n)
    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limit:
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-9:
            best_c5, best_h, stall = c5, h.copy(), 0
        else:
            stall += 1
        g = c5_smooth_grad(h, n, alpha)
        v = beta_mom * v + (1.0 - beta_mom) * g
        h_new = project_feasible(h - lr * v, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < c5 - 1e-12:
            h = h_new
            lr = min(lr * 1.015, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
        if stall > 200 and stall % 100 == 99:
            h = project_feasible(h + np.random.normal(0, 0.02, n), n)
            v *= 0.5
            stall = 0
        if stall % 400 == 399:
            alpha = min(alpha * 1.5, 600.0)
        if stall > 800:
            break
    return best_h, best_c5
