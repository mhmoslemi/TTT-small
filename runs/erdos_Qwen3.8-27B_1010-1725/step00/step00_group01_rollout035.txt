C[k] = sum_{n: valid} h[n] * (1 - h[n + lag])

where valid means 0 ≤ n < n and 0 ≤ n + lag < n.

dC[k]/dh[j] = sum_{n: valid} [dh[n]/dh[j] * (1-h[n+lag]) + h[n] * d(1-h[n+lag])/dh[j]]
= sum_{n: valid} [δ(n,j) * (1-h[n+lag]) - h[n] * δ(n+lag, j)]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_h(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_init(n, rng):
    h = np.zeros(n)
    k1 = n // 2
    idx = rng.choice(n, k1, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        z = np.where(h == 0.0)[0]
        h[z[rng.integers(len(z))]] = 0.5
    return h

def make_bluen_init(n, rng):
    h = np.zeros(n)
    k1 = n // 2
    positions = np.sort(rng.choice(n, k1, replace=False))
    # Perturb blue noise: swap a few random pairs
    for _ in range(max(1, n // 10)):
        i, j = np.random.default_rng(rng.integers(2**31)).permutation(n)[:2]
        if h[i] != h[j]:
            h[i], h[j] = h[j], h[i]
    if n % 2 == 1:
        z = np.where(h == 0.0)[0]
        if len(z) > 0:
            h[z[rng.integers(len(z))]] = 0.5
    return project_h(h, n)

def interp_to_n(h_src, n_dst):
    ns = len(h_src)
    if ns == n_dst:
        return h_src.copy()
    xs = np.linspace(0.0, 2.0, ns, endpoint=False) + 1.0 / ns
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    return project_h(np.interp(xd, xs, h_src), n_dst)

def anneal(h, n, max_time, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    t0 = time.time()
    dx = 2.0 / n
    t_scale = max(max_time / 3.0, 1.0)
    while time.time() - t0 < max_time:
        temp = 0.04 * np.exp(-(time.time() - t0) / t_scale)
        ones = np.flatnonzero(cur > 0.4)
        zeros = np.flatnonzero(cur < 0.6)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        vi, vj = cur[i], cur[j]
        cur[i], cur[j] = vj, vi
        nc5 = float(np.max(np.correlate(cur, 1.0 - cur, mode='full'))) * dx
        d = nc5 - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(temp, 1e-12)):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best = cur.copy()
        else:
            cur[i], cur[j] = vi, vj
    return best, best_c5

def grad_refine(h, n, max_time, rng):
    cur = h.copy()
    best = h.copy()
    best_c5 = compute_c5(best)
    t0 = time.time()
    dx = 2.0 / n
    while time.time() - t0 < max_time:
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        c5v = float(np.max(corr)) * dx
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best = cur.copy()
        thr = np.max(corr) - 0.3
        active = np.flatnonzero(corr >= thr)
        grad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            for j in range(n):
                if 0 <= j - lag < n:
                    grad[j] += (1.0 - cur[j - lag])
                if 0 <= j + lag < n:
                    grad[j] -= cur[j + lag]
        grad /= max(len(active), 1)
        grad += 0.01 * rng.standard_normal(n)
        cur = project_h(cur - 0.15 * grad, n)
        # Binary projection every few steps
        if int((time.time() - t0) * 10) % 3 == 0:
            hb = (cur > 0.5).astype(float)
            if n % 2 == 1:
                mid = np.argmin(np.abs(cur - 0.5))
                hb[mid] = 0.5
            hb = project_h(hb, n)
            if compute_c5(hb) < compute_c5(cur):
                cur = hb
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf

    n_list = [47, 48, 50, 52, 56, 60, 64, 72, 80, 96, 100, 120, 128, 144, 160, 192, 200, 256]

    # Get initial if available
    init_avail = False
    try:
        _tmp = initial_h_values
        init_avail = _tmp is not None
    except NameError:
        init_avail = False

    total_budget = budget_s - 40
    time_per_n = total_budget / len(n_list)

    for n in n_list:
        elapsed = time.time() - t_start
        remaining = budget_s - 20 - elapsed
        if remaining < 5:
            break
        alloc = min(time_per_n * 1.5, remaining)
        if alloc < 4:
            continue

        n_starts = max(2, int(alloc / 5))
        n_starts = min(n_starts, 12)

        for s in range(n_starts):
            if time.time() - t_start > budget_s - 10:
                break
            t_remaining_for_start = (budget_s - 10 - (time.time() - t_start))
            t_this = min(alloc / n_starts, t_remaining_for_start)
            if t_this < 1.0:
                break

            # Choose initialization
            if s == 0 and init_avail:
                if n == len(initial_h_values):
                    h0 = initial_h_values.astype(float).copy()
                else:
                    h0 = interp_to_n(initial_h_values.astype(float), n)
            elif s == 1 and best_h is not None:
                h0 = interp_to_n(best_h, n)
            elif s == 2:
                h0 = make_bluen_init(n, rng)
            else:
                h0 = make_init(n, rng)

            t_ann = t_this * 0.65
            t_grad = t_this * 0.30

            h1, c5_1 = anneal(h0, n, t_ann, rng)
            h2, c5_2 = grad_refine(h1, n, t_grad, rng)

            if c5_2 < c5_1:
                h_cur, c5_cur = h2, c5_2
            else:
                h_cur, c5_cur = h1, c5_1

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    if best_h is None:
        best_h = make_init(50, rng)
        best_c5 = compute_c5(best_h)

    # Final polish with remaining time
    remaining = budget_s - 10 - (time.time() - t_start)
    if remaining > 3:
        best_h, best_c5 = anneal(best_h, len(best_h), remaining, rng)

    return best_h, best_c5, len(best_h)
