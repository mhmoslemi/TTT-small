def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_correlations(h))

def repair_sum(h, n, tol=1e-10):
    """Adjust h to satisfy sum(h) = n/2 while keeping 0 <= h[i] <= 1"""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < tol:
        return h
    # Distribute the difference evenly
    h = h.copy()
    # First try uniform adjustment
    adj = diff / n
    h_new = np.clip(h + adj, 0, 1)
    diff2 = target - np.sum(h_new)
    if abs(diff2) < tol:
        return h_new
    # If still off, adjust individual elements
    h = h_new.copy()
    # Find elements not at bounds
    free = (h > 0.01) & (h < 0.99)
    if np.any(free):
        # Distribute among free elements
        adj2 = diff2 / np.sum(free)
        h[free] = np.clip(h[free] + adj2, 0, 1)
    # Final adjustment
    diff3 = target - np.sum(h)
    if abs(diff3) > 1e-12:
        # Force adjust one element
        idx = np.argmax(h + (1-h))  # most room
        h[idx] = np.clip(h[idx] + diff3, 0, 1)
    return h

def smooth_max(c, alpha=1000):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def repair_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    return np.interp(x_new, x_old, h)

def to_binary(h, n):
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    excess = len(ones) - target
    if excess > 0:
        for i in range(excess):
            h[ones[-1 - i]] = 0.0
            h[zeros[i]] = 1.0
    elif excess < 0:
        for i in range(-excess):
            h[zeros[-1 - i]] = 1.0
            h[ones[i]] = 0.0
    return h

def local_search(h, n, time_limit, rng):
    h = to_binary(h, n)
    best = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    it = 0
    while time.time() - t_start < time_limit:
        it += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        if r < 0.6:
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            if c5 < best - 1e-14:
                best = c5
                best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        elif r < 0.85:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.integers(len(ones))]
                i2 = ones[rng.integers(len(ones))]
                j1 = zeros[rng.integers(len(zeros))]
                j2 = zeros[rng.integers(len(zeros))]
                if i1 != i2 and j1 != j2:
                    h[i1], h[j1] = h[j1], h[i1]
                    h[i2], h[j2] = h[j2], h[i2]
                    c5 = compute_c5(h)
                    if c5 < best - 1e-14:
                        best = c5
                        best_h = h.copy()
                    else:
                        h[i1], h[j1] = h[j1], h[i1]
                        h[i2], h[j2] = h[j2], h[i2]
        else:
            k_swap = 3
            if len(ones) >= k_swap and len(zeros) >= k_swap:
                idx_o = rng.choice(len(ones), k_swap, replace=False)
                idx_z = rng.choice(len(zeros), k_swap, replace=False)
                for a, b in zip(idx_o, idx_z):
                    h[ones[a]], h[zeros[b]] = h[zeros[b]], h[ones[a]]
                c5 = compute_c5(h)
                if c5 < best - 1e-14:
                    best = c5
                    best_h = h.copy()
                else:
                    for a, b in zip(idx_o, idx_z):
                        h[ones[a]], h[zeros[b]] = h[zeros[b]], h[ones[a]]
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h_parent = np.array(initial_h_values, dtype=float)

        t_total = min(remaining * 0.12 * frac + 5, remaining * 0.35)
        if t_total < 8:

        h_cur = upsample(best_h, n) if n != len(best_h) else best_h.copy()

        n_restarts = max(1, int(t_total / 8))
        t_per = t_total / n_restarts

        for r in range(n_restarts):

            if r == 0:
                h_init = h_cur.copy()
            else:
                h_init = best_h.copy()
                n_pert = 2 + rng.integers(0, 6)
                for _ in range(n_pert):
                    ones = np.where(h_init > 0.5)[0]
                    zeros = np.where(h_init < 0.5)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        ii = ones[rng.integers(len(ones))]

                break

    best_h = repair_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
