def try_pair(h, i, j, current_c, grid_points=13, gold_iters=10):
    vi = float(h[i]); vj = float(h[j])
    lo = max(-vi, vj - 1.0)
    hi = min(1.0 - vi, vj)
    if hi - lo <= 1e-13:
        return current_c, False
    best_t = 0.0
    best_f = float(current_c)
    span = hi - lo
    if grid_points < 2: grid_points = 2
    denom = float(grid_points - 1)
    for k in range(grid_points):
        t = lo + span * (float(k) / denom)
        if abs(t) <= 1e-13:
            continue
        h[i] = vi + t
        h[j] = vj - t
        f = c5_value(h)
        if f < best_f:
            best_f = f
            best_t = t
    half = span / 6.0
    a = max(lo, best_t - half)
    b = min(hi, best_t + half)
    if b - a > 1e-11:
        mu = 0.6180339887498949
        x1 = a + (1.0 - mu) * (b - a)

import numpy as np
import time
import math

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    if n == 0: return v.copy()
    target = float(target)
    if target <= 0.0: return np.zeros(n)
    if target >= float(n): return np.ones(n)
    a = -float(np.max(v)) - 1.0
    b = 2.0 - float(np.min(v))
    for _ in range(60):
        m = 0.5*(a+b)
        if float(np.sum(np.clip(v+m,0.0,1.0))) < target: a=m
        else: b=m
    x = np.clip(v+0.5*(a+b),0.0,1.0)
    for _ in range(10):
        d = target - float(x.sum())
        if abs(d) < 1e-13: break
        if d > 0:
            m = x < 1.0
            c = int(m.sum())
            if c == 0: break
            x[m] = np.minimum(1.0, x[m] + d/c)
        else:
            m = x > 0.0
            c = int(m.sum())
            if c == 0: break
            x[m] = np.maximum(0.0, x[m] + d/c)
    return x

def binary_cost(h, tau):
    c = np.correlate(h, 1.0-h, mode='full')
    m = float(c.max())
    y = np.clip((c-m)*tau, -50.0, 0.0)
    return m + math.log(float(np.exp(y).sum()))/tau, m

def binary_sa(h, deadline, rng, tau=4.0, max_iter=300000):
    h = threshold_binary(h)
    n = h.size
    ones = list(np.flatnonzero(h == 1.0))
    zeros = list(np.flatnonzero(h == 0.0))
    if not ones or not zeros: return h, float(np.max(np.correlate(h,1-h,'full')))
    cost0, max0 = binary_cost(h, tau)
    best = h.copy(); best_max = max0; best_cost = cost0
    T = 1.0; Tmin = 0.05
    for it in range(max_iter):
        if time.time() >= deadline: break
        a = int(rng.integers(len(ones))); b = int(rng.integers(len(zeros)))
        i = ones[a]; j = zeros[b]
        h[i]=0.0; h[j]=1.0
        ones[a]=j; zeros[b]=i
        cost1, max1 = binary_cost(h, tau)
        if max1 < best_max - 1e-12:
            best_max = max1; best_cost = cost1; best = h.copy()
        elif max1 <= best_max + 1e-12 and cost1 < best_cost - 1e-14:
            best_cost = cost1; best = h.copy()
        accept = cost1 <= cost0 + 1e-15
        if not accept:
            d = (cost1-cost0)/T
            if d < 50.0 and rng.random() < math.exp(-d): accept = True
        if not accept:
            h[i]=1.0; h[j]=0.0
            ones[a]=i; zeros[b]=j
        else:
            cost0 = cost1
        T = max(Tmin, T*0.999)
    return best, best_max

import numpy as np
import time
import math


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = int(v.size)
    if n == 0:
        return v.copy()
    target = float(target)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    a = -float(np.max(v)) - 1.0
    b = 2.0 - float(np.min(v))
    for _ in range(64):
        m = 0.5 * (a + b)
        if float(np.sum(np.clip(v + m, 0.0, 1.0))) < target:
            a = m
        else:
            b = m
    x = np.clip(v + 0.5 * (a + b), 0.0, 1.0)
    for _ in range(12):
        d = target - float(x.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            mask = x < 1.0
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            x[mask] = np.minimum(1.0, x[mask] + d / float(cnt))
        else:
            mask = x > 0.0
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            x[mask] = np.maximum(0.0, x[mask] + d / float(cnt))
    return x


def c5_value(h):
    n = int(h.size)
    if n <= 1:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def make_pairs(n):
    n = int(n)
    return [(i, j) for i in range(n) for j in range(i + 1, n)]


def try_pair(h, i, j, current_c, grid_points=13, gold_iters=10):
    vi = float(h[i])
    vj = float(h[j])
    lo = max(-vi, vj - 1.0)
    hi = min(1.0 - vi, vj)
    if hi - lo <= 1e-13:
        return current_c, False

    best_t = 0.0
    best_f = float(current_c)
    span = hi - lo
    if grid_points < 2:
        grid_points = 2
    denom = float(grid_points - 1)

    for k in range(grid_points):
        t = lo + span * (float(k) / denom)
        if abs(t) <= 1e-13:
            continue
        h[i] = vi + t
        h[j] = vj - t
        f = c5_value(h)
        if f < best_f:
            best_f = f
            best_t = t

    half = span / 6.0
    a = max(lo, best_t - half)
    b = min(hi, best_t + half)
    if b - a > 1e-11:
        mu = 0.6180339887498949
        x1 = a + (1.0 - mu) * (b - a)
        x2 = a + mu * (b - a)

        h[i] = vi + x1
        h[j] = vj - x1
        f1 = c5_value(h)
        if f1 < best_f:
            best_f = f1
            best_t = x1

        h[i] = vi + x2
        h[j] = vj - x2
        f2 = c5_value(h)
        if f2 < best_f:
            best_f = f2
            best_t = x2

        for _ in range(gold_iters):
            if f1 <= f2:
                b = x2
                x2 = x1
                f2 = f1
                x1 = a + (1.0 - mu) * (b - a)
                h[i] = vi + x1
                h[j] = vj - x1
                f1 = c5_value(h)
                if f1 < best_f:
                    best_f = f1
                    best_t = x1
            else:
                a = x1
                x1 = x2
                f1 = f2
                x2 = a + mu * (b - a)
                h[i] = vi + x2
                h[j] = vj - x2
                f2 = c5_value(h)
                if f2 < best_f:
                    best_f = f2
                    best_t = x2

    h[i] = vi + best_t
    h[j] = vj - best_t
    if best_f < current_c - 1e-11:
        return best_f, True
    h[i] = vi
    h[j] = vj
    return current_c, False


def pair_descent(h, deadline, rng, max_sweeps=300, grid_points=13, gold_iters=10, random_order=True):
    h = np.array(h, dtype=float, copy=True)
    n = int(h.size)
    target = float(n) / 2.0
    c = c5_value(h)
    if n < 2:
        return h, c

    pairs = make_pairs(n)
    m = len(pairs)
    sweeps = 0

    while sweeps < max_sweeps and time.time() < deadline:
        sweeps += 1
        order = rng.permutation(m) if random_order else range(m)
        improved = False

        for idx in order:
            if time.time() >= deadline:
                break
            p = pairs[int(idx)]
            c2, ok = try_pair(h, p[0], p[1], c, grid_points, gold_iters)
            if ok:
                c = float(c2)
                improved = True

        if abs(float(h.sum()) - target) > 1e-9:
            h = project_box_sum(h, target)
            c = c5_value(h)

        if not improved:
            break

    return h, c


def threshold_binary(h):
    n = int(h.size)
    k = int(round(float(n) / 2.0))
    x = np.zeros(n, dtype=float)
    if k <= 0:
        return x
    if k >= n:
        return np.ones(n, dtype=float)
    x[np.argpartition(h, -k)[-k:]] = 1.0
    return x


def random_binary(n, rng):
    n = int(n)
    k = int(round(float(n) / 2.0))
    x = np.zeros(n, dtype=float)
    if 0 < k < n:
        x[rng.choice(n, k, replace=False)] = 1.0
    return x


def make_continuous_start(base, scale, rng, n):
    if base is None:
        n = int(n)
        v = rng.random(n)
        target = float(n) / 2.0
    else:
        v = base + float(scale) * rng.standard_normal(base.size)
        target = float(base.size) / 2.0
    return project_box_sum(v, target)


def binary_cost(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(c))
    y = np.clip((c - m) * float(tau), -50.0, 0.0)
    return m + math.log(float(np.exp(y).sum())) / float(tau), m


def binary_sa(h, deadline, rng, tau=4.0, max_iter=300000):
    h = threshold_binary(h)
    ones = list(np.flatnonzero(h == 1.0))
    zeros = list(np.flatnonzero(h == 0.0))
    if not ones or not zeros:
        return h, float(np.max(np.correlate(h, 1.0 - h, mode="full")))

    cost0, max0 = binary_cost(h, tau)
    best = h.copy()
    best_max = max0
    best_cost = cost0
    T = 1.0
    Tmin = 0.05

    for _ in range(int(max_iter)):
        if time.time() >= deadline:
            break

        a = int(rng.integers(len(ones)))
        b = int(rng.integers(len(zeros)))
        i = ones[a]
        j = zeros[b]

        h[i] = 0.0
        h[j] = 1.0
        ones[a] = j
        zeros[b] = i

        cost1, max1 = binary_cost(h, tau)

        if max1 < best_max - 1e-12:
            best_max = max1
            best_cost = cost1
            best = h.copy()
        elif max1 <= best_max + 1e-12 and cost1 < best_cost - 1e-14:
            best_cost = cost1
            best = h.copy()

        accept = cost1 <= cost0 + 1e-15
        if not accept:
            d = (cost1 - cost0) / T
            if d < 50.0 and rng.random() < math.exp(-d):
                accept = True

        if not accept:
            h[i] = 1.0
            h[j] = 0.0
            ones[a] = i
            zeros[b] = j
        else:
            cost0 = cost1

        T = max(Tmin, T * 0.999)

    return best, best_max


def finalize_h(h):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 5e-10:
        return project_box_sum(h, target)
    if abs(d) > 1e-12:
        if d > 0.0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            if room[idx] >= d:
                h[idx] += d
            else:
                return project_box_sum(h, target)
        else:
            idx = int(np.argmax(h))
            if h[idx] >= -d:
                h[idx] += d
            else:
                return project_box_sum(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    budget = float(budget_s)
    deadline = start_time + budget
    rng = np.random.default_rng(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = globals().get("initial_h_values")
    n = 70

    if parent is not None:
        h = np.asarray(parent, dtype=float).ravel()
        if h.size > 0:
            n = int(h.size)
            if n > 150:
                nn = 70
                h = np.interp(np.linspace(0.0, float(n) - 1.0, nn), np.arange(float(n)), h)
                n = nn
            h = project_box_sum(h, float(n) / 2.0)
            c = c5_value(h)
            if math.isfinite(c):
                best_h = h.copy()
                best_c = float(c)
                best_n = n

    if best_h is None:
        best_n = 70
        best_h = make_continuous_start(None, 0.0, rng, 70)
        best_c = c5_value(best_h)

    try:
        if budget > 25.0:
            d1 = start_time + min(0.45 * budget, max(25.0, budget - 350.0))
            if time.time() < d1:
                h1, c1 = pair_descent(best_h.copy(), d1, rng, max_sweeps=1000,
                                      grid_points=13, gold_iters=11)
                if math.isfinite(c1) and c1 < best_c:
                    best_h = h1
                    best_c = float(c1)
                    best_n = int(h1.size)

        remaining = deadline - time.time()
        if remaining > 120.0 and best_n * 2 <= 300:
            nu = int(best_n) * 2
            h_up = np.repeat(best_h, 2)
            h_up = project_box_sum(h_up, float(nu) / 2.0)
            up_alloc = min(0.30 * budget, max(50.0, remaining * 0.5))
            up_limit = time.time() + up_alloc
            if up_limit < deadline - 20.0:
                h_u, c_u = pair_descent(h_up, up_limit, rng, max_sweeps=300,
                                        grid_points=10, gold_iters=7)
                if math.isfinite(c_u) and c_u < best_c:
                    best_h = h_u
                    best_c = float(c_u)
                    best_n = int(h_u.size)

        per_start = max(10.0, 0.07 * budget)

        if best_n % 2 == 0 and (deadline - time.time()) > 30.0:
            h_bin = threshold_binary(best_h)
            c_bin = c5_value(h_bin)
            if math.isfinite(c_bin) and c_bin < best_c:
                best_h = h_bin.copy()
                best_c = float(c_bin)

            h_cand = h_bin.copy()
            c_cand = float(c_bin)

            rem = deadline - time.time()
            if rem > 60.0:
                d_sa = time.time() + min(45.0, 0.08 * budget)
                if d_sa < deadline - 10.0:
                    h_sa, _ = binary_sa(h_bin, d_sa, rng, tau=4.0, max_iter=300000)
                    c_sa = c5_value(h_sa)
                    if math.isfinite(c_sa) and c_sa < best_c:
                        best_h = h_sa.copy()
                        best_c = float(c_sa)
                    if c_sa <= c_cand:
                        h_cand = h_sa.copy()
                        c_cand = float(c_sa)

            rem = deadline - time.time()
            alloc = min(per_start, max(5.0, rem * 0.45))
            limit = time.time() + alloc
            if time.time() < limit and limit < deadline - 5.0:
                h2, c2 = pair_descent(h_cand, limit, rng, max_sweeps=400,
                                      grid_points=11, gold_iters=8)
                if math.isfinite(c2) and c2 < best_c:
                    best_h = h2
                    best_c = float(c2)
                    best_n = int(h2.size)

        limit2 = deadline - max(10.0, 0.08 * budget)
        scales = [0.10, 0.05, 0.20, 0.35]
        if best_n > 100:
            scales = scales[:2]

        for s in scales:
            if time.time() >= limit2:
                break
            rem = deadline - time.time()
            alloc = min(per_start, max(5.0, rem * 0.35))
            h_start = make_continuous_start(best_h, s, rng, best_n)
            h3, c3 = pair_descent(h_start, time.time() + alloc, rng,
                                  max_sweeps=400, grid_points=11, gold_iters=8)
            if math.isfinite(c3) and c3 < best_c:
                best_h = h3
                best_c = float(c3)
                best_n = int(h3.size)

        if best_n % 2 == 0 and best_n <= 120:
            for _ in range(2):
                if time.time() >= limit2:
                    break
                rem = deadline - time.time()
                alloc = min(per_start, max(5.0, rem * 0.35))
                h_start = random_binary(best_n, rng)
                h4, c4 = pair_descent(h_start, time.time() + alloc, rng,
                                      max_sweeps=400, grid_points=10, gold_iters=7)
                if math.isfinite(c4) and c4 < best_c:
                    best_h = h4
                    best_c = float(c4)
                    best_n = int(h4.size)

        final_deadline = deadline - 5.0
        if time.time() < final_deadline:
            h5, c5v = pair_descent(best_h.copy(), final_deadline, rng,
                                   max_sweeps=1000, grid_points=12, gold_iters=9)
            if math.isfinite(c5v) and c5v < best_c:
                best_h = h5
                best_c = float(c5v)
                best_n = int(h5.size)

    except Exception:
        pass

    h_final = finalize_h(best_h)
    c_final = c5_value(h_final)
    if not math.isfinite(c_final):
        h_final = best_h
        c_final = best_c

    return h_final, float(c_final), int(h_final.size)
