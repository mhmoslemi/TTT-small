import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, alpha):
    m = np.max(vals)
    s = alpha * (vals - m)
    s = np.clip(s, -60.0, 0.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def fix_feasibility(h, n_pts):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n_pts / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n_pts
        h = np.clip(h, 0.0, 1.0)
    return h

def make_slsqp_obj(n_pts, alpha=50.0):
    dx = 2.0 / n_pts
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max(corr, alpha)
    return obj

def make_eq_fun(n_pts):
    target = n_pts / 2.0
    def eq(x):
        return np.sum(x) - target
    return eq

def try_optimize(h0, n_pts, alpha, method='SLSQP', maxiter=300):
    obj_fun = make_slsqp_obj(n_pts, alpha)
    eq_fun = make_eq_fun(n_pts)
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': eq_fun}]
    try:
        res = minimize(obj_fun, h0, method=method, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_feasibility(res.x, n_pts)
    except Exception:
        return h0

def coord_polish(h, n_pts, n_sweeps=20):
    h = h.copy()
    target = n_pts / 2.0
    dx = 2.0 / n_pts
    for _ in range(n_sweeps):
        improved = False
        for i in range(n_pts):
            orig = h[i]
            best_c5 = compute_c5(h, n_pts)
            best_val = orig
            for trial in [0.0, 1.0, 0.5, (orig+0.0)/2, (orig+1.0)/2]:
                h[i] = trial
                diff = target - np.sum(h)
                if abs(diff) > 1e-10:
                    # distribute to neighbors
                    h[i] = trial
                    others = np.ones(n_pts)
                    others[i] = 0
                    norm = np.sum(others)
                    if norm > 0:
                        h = h + diff * others / norm
                        h = np.clip(h, 0, 1)
                    else:
                        h[i] = orig
                        break
                c5_try = compute_c5(h, n_pts)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_val = trial
            h[i] = best_val
            if best_val != orig:
                improved = True
        if not improved:
            break
    return fix_feasibility(h, n_pts)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution

        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = 0.5 * np.ones(96)

    n = len(h0)

    best_c5 = compute_c5(best_h, n)
    best_n = n

    # Phase 1: SLSQP multi-start at initial resolution
    alpha = 50.0
    n_starts = 12
    for s in range(n_starts):
        if time_left() < 80:
            break

            x0 = best_h.copy()
        elif s == 1:
            x0 = rng.uniform(0, 1, n)
            x0 = fix_feasibility(x0, n)
        elif s == 2:
            # Binary-ish random
            x0 = (rng.rand(n) < 0.5).astype(float)
            # Fix to have exactly n/2 ones
            ones = np.where(x0 == 1)[0]
            zeros = np.where(x0 == 0)[0]
            n_ones_needed = int(n / 2)
            if len(ones) > n_ones_needed:

                x0[flip] = 0
            elif len(ones) < n_ones_needed:

        c5_val = compute_c5(hc, best_n)

            best_h = hc.copy()
            best_n = best_n

    for n_new in [128, 192, 256, 320]:

        h_interp = fix_feasibility(h_interp, n_new)

        # Optimize at new resolution

                best_h = hc2.copy()

    if time_left() > 40:

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = hc.copy()

    return best_h, best_c5, len(best_h)
