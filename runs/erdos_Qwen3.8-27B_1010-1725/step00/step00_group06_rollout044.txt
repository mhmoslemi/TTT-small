import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, lam):
    m = np.max(vals)
    if lam < 1.0:
        return m
    e = np.exp(lam * (vals - m))
    return m + np.log(e.sum()) / lam

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] += min(diff / mask.sum(), 0.5)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] -= min((-diff) / mask.sum(), 0.5)
        diff = target - h.sum()
    return h

def slsqp_solve(h0, n, lam, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr * dx, lam)

    def eq_fun(x):
        return x.sum() - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bnds = [(0.0, 1.0)] * n
    h0_f = project_feasible(h0, n)
    try:
        res = minimize(obj, h0_f, method='SLSQP', bounds=bnds,
                       constraints=cons, maxiter=maxiter,
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        return project_feasible(res.x, n)
    except Exception:
        return h0_f

def nm_polish(h0, n, maxiter=5000):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)

        pen = 50.0 * (h.sum() - target) ** 2

        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

    return np.interp(x_new, x_old, h)
