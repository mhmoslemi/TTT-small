import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    target = n / 2.0
    def sum_clipped(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if sum_clipped(mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n) / old_n * 2.0
    new_x = np.arange(new_n) / new_n * 2.0
    return np.interp(new_x, old_x, h)

def subgradient_obj(h):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    k_max = np.argmax(corr)
    val = corr[k_max] * dx
    lag = k_max - (n - 1)
    t_lo = max(0, -lag)
    t_hi = min(n - 1, n - 1 - lag)
    grad = np.zeros(n)
    if t_hi >= t_lo:
        t = np.arange(t_lo, t_hi + 1)
        h_idx = t + lag
        np.add.at(grad, h_idx, one_h[t])
        np.add.at(grad, t, -h[h_idx])
    grad *= dx
    return val, grad

def subgradient_descent(h, n, max_iters, lr0, decay, t_end):
    best_val = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > t_end:
            break
        val, grad = subgradient_obj(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        lr = lr0 / (1.0 + it * decay)
        h = project_h(h - lr * grad, n)
    return best_h, best_val

def slsqp_refine(h, n, t_end):
    def obj(x):
        hx = project_h(x, n)
        return compute_c5(hx)
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10})
        hx = project_h(res.x, n)
        cv = compute_c5(hx)
        return hx, cv
    except Exception:
        return h, compute_c5(h)

def nm_refine(h, n, t_end):
    def obj(x):
        hx = project_h(x, n)
        return compute_c5(hx)
    try:
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
        hx = project_h(res.x, n)
        cv = compute_c5(hx)
        return hx, cv
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h0 = np.full(70, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = len(h0)

    n_list = [140, 200, 280, 400]
    for n_pts in n_list:
        if time.time() - t0 > budget_s - 60:
            break
        h_up = upsample(h0, n_pts)
        h_up = project_h(h_up, n_pts)

        # Multiple subgradient restarts
        n_restarts = 4
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 40:
                break
            h_cur = h_up.copy()
            if r > 0:
                noise = np.random.randn(n_pts) * 0.15 * (r * 0.3)
                h_cur = project_h(h_cur + noise, n_pts)

            max_it = 80000
            t_end = min(t0 + budget_s - 30, time.time() + 120)
            h_best_r, c5_r = subgradient_descent(h_cur, n_pts, max_it,
                                                  0.8 / np.sqrt(n_pts), 0.003, t_end)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_best_r.copy()
                best_n = n_pts

        # SLSQP on best
        if time.time() - t0 > budget_s - 30:
            break
        t_end = min(t0 + budget_s - 20, time.time() + 60)
        h_ref, c5_ref = slsqp_refine(best_h, best_n, t_end)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

        # Nelder-Mead

        t_end = min(t0 + budget_s - 15, time.time() + 40)
        h_nm, c5_nm = nm_refine(best_h, best_n, t_end)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    # Final subgradient polish
    if time.time() - t0 < budget_s - 5:
        n_f = len(best_h)
        t_end = t0 + budget_s - 2
        h_pol, c5_pol = subgradient_descent(best_h, n_f, 100000,
                                            0.3 / np.sqrt(n_f), 0.001, t_end)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Ensure validity
    best_h = project_h(best_h, len(best_h))
