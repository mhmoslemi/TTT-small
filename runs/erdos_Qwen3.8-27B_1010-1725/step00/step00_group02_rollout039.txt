        elif n_ones < n // 2:
            idx = np.where(h_b == 0)[0]
            order = np.argsort(h[idx])[::-1]
            h_b[idx[order[:n // 2 - n_ones]]] = 1.0
        return h_b
    else:
        h_r = np.round(h * 2.0) / 2.0
        diff = target - h_r.sum()
        if abs(diff) > 1e-10:
            n_fix = int(round(abs(diff) * 2.0))
            if diff > 0:
                halfs = np.where(h_r == 0.5)[0]
                zeros = np.where(h_r == 0.0)[0]
                for k in range(min(n_fix, len(halfs))):
                    h_r[halfs[k]] = 1.0
                rem = n_fix - min(n_fix, len(halfs))
                for k in range(min(rem, len(zeros))):
                    h_r[zeros[k]] = 0.5
            else:
                halfs = np.where(h_r == 0.5)[0]
                ones = np.where(h_r == 1.0)[0]
                for k in range(min(n_fix, len(halfs))):
                    h_r[halfs[k]] = 0.0
                rem = n_fix - min(n_fix, len(halfs))
                for k in range(min(rem, len(ones))):
                    h_r[ones[k]] = 0.5
        return h_r

def greedy_improve(h, n, t_limit):
    h_c = h.copy()
    s = eval_c5(h_c, n)
    vals = np.unique(h_c)
    if len(vals) < 2:
        return h_c, s
    pos_by_val = {v: np.where(h_c == v)[0] for v in vals}
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        all_pairs = []
        for vi in range(len(vals)):
            for vj in range(vi + 1, len(vals)):
                for i in pos_by_val[vals[vi]]:
                    for j in pos_by_val[vals[vj]]:
                        all_pairs.append((i, j, vals[vi], vals[vj]))
        if len(all_pairs) > 50000:
            rng = np.random.RandomState(42)
            idxs = rng.choice(len(all_pairs), size=20000, replace=False)
            all_pairs = [all_pairs[k] for k in idxs]
        best_gain = 1e-15
        best_pi = best_pj = best_vi = best_vj = -1
        for i, j, vi, vj in all_pairs:
            h_c[i], h_c[j] = h_c[j], h_c[i]
            s_new = eval_c5(h_c, n)
            gain = s - s_new
            if gain > best_gain:
                best_gain = gain
                best_pi, best_pj, best_vi, best_vj = i, j, vi, vj
            else:
                h_c[i], h_c[j] = h_c[j], h_c[i]
        if best_pi >= 0:
            h_c[best_pi], h_c[best_pj] = h_c[best_pj], h_c[best_pi]
            s -= best_gain
            improved = True
            pos_by_val = {v: np.where(h_c == v)[0] for v in np.unique(h_c)}
    return h_c, s

def sa_search(h, n, t_limit, seed=0):
    rng = np.random.RandomState(seed)
    h_c = h.copy()
    s_c = eval_c5(h_c, n)
    s_b = s_c
    T = 0.006
    stuck = 0
    while time.time() < t_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h_c[i] == h_c[j]:
            continue
        h_c[i], h_c[j] = h_c[j], h_c[i]
        s_new = eval_c5(h_c, n)
        d = s_new - s_c
        if d < 0 or rng.random() < np.exp(-d / T):
            s_c = s_new
            if s_new < s_b - 1e-12:
                s_b = s_new
                stuck = 0
            else:
                stuck += 1
        else:
            h_c[i], h_c[j] = h_c[j], h_c[i]
            stuck += 1
        T = max(T * 0.99995, 0.0002)
        if stuck > 5000:
            T = 0.006
            stuck = 0
            for _ in range(8):
                a = rng.randint(n)
                b = rng.randint(n)
                if h_c[a] != h_c[b]:
                    h_c[a], h_c[b] = h_c[b], h_c[a]
            s_c = eval_c5(h_c, n)
            s_b = min(s_b, s_c)
    return h_c, s_b

def random_valid(n, seed=0):
    rng = np.random.RandomState(seed)
    if n % 2 == 0:
        h = np.zeros(n)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    else:
        h = np.zeros(n)
        h[rng.choice(n, size=(n - 1) // 2, replace=False)] = 1.0
        remaining = np.where(h == 0)[0]
        h[rng.choice(remaining, size=1, replace=False)] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_s = eval_c5(best_h, n0)

    grids = sorted(set([n0, 60, 90, 120, 180, 240, 360]))
    grids = [n for n in grids if n >= n0]

    for n in grids:
        if time.time() - t0 > budget_s - 25:
            break
        elapsed = time.time() - t0
        remaining = budget_s - 20 - elapsed
        if remaining < 15:
            break
        frac = 0.3 + 0.7 * (len([g for g in grids if g <= n]) / len(grids))
        alloc = remaining * frac * 0.35

        if n == n0:
            h_curr = h0.copy()
        else:
            h_curr = interp_to_n(best_h, len(best_h), n)
            h_curr = project_valid(h_curr, n)

        s_curr = eval_c5(h_curr, n)

        t_gl = min(time.time() + alloc * 0.4, t0 + budget_s - 20)
        if t_gl > time.time() + 3:
            h_curr, s_curr = greedy_improve(h_curr, n, t_gl)

        t_sl = min(time.time() + alloc * 0.5, t0 + budget_s - 15)
        if t_sl > time.time() + 5:
            h_curr, s_curr = sa_search(h_curr, n, t_sl, seed=seed + n)

        t_g2 = min(time.time() + alloc * 0.3, t0 + budget_s - 10)
        if t_g2 > time.time() + 3:
            h_curr, s_curr = greedy_improve(h_curr, n, t_g2)

        if s_curr < best_s - 1e-10:
            best_s = s_curr
            best_h = h_curr.copy()
        print(f"n={n}: C5={s_curr:.6f} | best={best_s:.6f} | t={time.time()-t0:.0f}s")

    # Try a couple random starts at a mid-size grid
    for rs in [7, 13]:
        if time.time() - t0 > budget_s - 30:
            break
        n_r = min(90, max(n0, 90))
        h_r = random_valid(n_r, seed=rs)
        t_gl = min(time.time() + 40, t0 + budget_s - 15)
        h_r, s_r = greedy_improve(h_r, n_r, t_gl)
        t_sl = min(time.time() + 60, t0 + budget_s - 10)
        h_r, s_r = sa_search(h_r, n_r, t_sl, seed=100 + rs)
        if s_r < best_s - 1e-10:
            best_s = s_r
            best_h = h_r.copy()
            print(f"Random start {rs}: n={n_r}, C5={best_s:.6f}")

    final_s = eval_c5(best_h, len(best_h))
    # Ensure sum constraint
    n_f = len(best_h)
    check = best_h.sum() * (2.0 / n_f)
    if abs(check - 1.0) > 1e-6:
        best_h = project_valid(best_h, n_f)
        final_s = eval_c5(best_h, n_f)
    print(f"FINAL: n={n_f}, C5={final_s:.6f}")
    return best_h, final_s, n_f
