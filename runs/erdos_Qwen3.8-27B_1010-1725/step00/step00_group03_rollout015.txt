import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def _c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _obj_fun(x, tau, pen, ntarget):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    tc = tau * corr
    tc_max = np.max(tc)
    lse = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    val = lse / tau + pen * (np.sum(x) - ntarget) ** 2
    return val


def _grad_fun(x, tau, pen, ntarget):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    tc = tau * corr
    tc_max = np.max(tc)
    lse = tc_max + np.log(np.sum(np.exp(tc - tc_max)))
    w = np.exp(tc - lse)
    grad = np.zeros(n)
    s = np.arange(2 * n - 1) - (n - 1)
    for j in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            sm = s[m]
            i1 = j - sm
            if 0 <= i1 < n:
                g += w[m] * (1.0 - x[i1])
            i2 = j + sm
            if 0 <= i2 < n:
                g -= w[m] * x[i2]
        grad[j] = dx * g + 2.0 * pen * (np.sum(x) - ntarget)
    return grad


def _obj_jac(x, tau, pen, ntarget):
    v = _obj_fun(x, tau, pen, ntarget)
    g = _grad_fun(x, tau, pen, ntarget)
    return v, g


def _fix_sum(h, ntarget):
    diff = ntarget - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def _local_optimize(h, tau, pen, ntarget, bounds, maxiter=400):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    res = minimize(_obj_jac, h, args=(tau, pen, ntarget, n),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-10})
    hx = np.clip(res.x, 0.0, 1.0)
    return hx


def _binary_local_search(h, ntarget):
    n = len(h)
    best = h.copy()
    best_c5 = _c5_val(best)
    improved = True
    while improved and improved:
        improved = False
        for j in range(n):
            trial = best.copy()
            if trial[j] > 0.01:
                trial[j] = 0.0
            else:
                trial[j] = 1.0
            s = np.sum(trial)
            if abs(s - ntarget) > ntarget * 0.1:
                continue
            trial = _fix_sum(trial, ntarget)
            c = _c5_val(trial)
            if c < best_c5 - 1e-10:
                best = trial
                best_c5 = c
                improved = True
    return best, best_c5


def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new
