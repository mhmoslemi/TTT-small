def _proj(h, n):
    target = n * 0.5
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        c = int(m.sum())
        if c == 0:

            break

    d = target - h.sum()
    if abs(d) > 1e-13:
        h = np.clip(h + d / n, 0.0, 1.0)
