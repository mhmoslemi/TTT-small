At each iteration:

2. Find k* = argmax f_k
3. Compute gradient of f_{k*}(h)
4. Take a step in the negative gradient direction
5. Project back to feasible set (0≤h≤1, sum=n/2)

The gradient of f_k*(h) = sum_i h[i]*(1-h[i+k*]) is:
∂f/∂h[j] = (1-h[j+k*]) - h[j-k*] (for valid indices, 0 otherwise)

This is a subgradient of the max (since we pick the active k).

This is essentially the "generalized gradient" method or "cutting plane" method for min-max problems.

import time

def c5_eval(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def fix_constraints(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Add mass where there's room
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if room[i] <= 0:
                    break
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            # Remove mass from where there's h
            order = np.argsort(h)
            for i in order:
                if h[i] <= 0:
                    break
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return h

def subgrad_optimize(h, n, max_iters, t_budget):
    t_start = time.time()
    h = h.copy()
    best_c5 = c5_eval(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - t_start > t_budget:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        
        # Compute gradient of f_shift
        j = np.arange(n)
        idx_plus = j + shift
        idx_minus = j - shift
        
        grad = np.ones(n)  # base: 1 - 0 = 1 for out of range
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        grad[mask_plus] -= h[idx_plus[mask_plus]]
        
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        grad[mask_minus] -= h[idx_minus[mask_minus]]
        
        # Learning rate
        lr = 0.1 / (1.0 + 0.01 * it)
        
        h -= lr * grad
        h = fix_constraints(h, n)
        
        c5 = c5_eval(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
