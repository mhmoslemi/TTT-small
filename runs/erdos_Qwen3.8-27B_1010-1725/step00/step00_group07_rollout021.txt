import numpy as np
import time
from scipy.optimize import minimize

_CURRENT_N = 80

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + tau * np.log(np.sum(np.exp((vals - m) / tau)))

def _obj_wrapper(x, tau):
    return _smooth_obj(x, tau)

def _make_obj(tau):
    def f(x):
        return _obj_wrapper(x, tau)
    return f

def _sum_con(x):
    return np.sum(x) - float(_CURRENT_N) / 2.0

def _make_con():
    return {'type': 'eq', 'fun': _sum_con}

def _repair(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(500):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        if s < 1e-15:
            h[:] = 0.5
            break
        if s > target:
            excess = s - target
            mask = h > 0
            if not np.any(mask):
                h = np.full(n, 0.5)
                break
            reduction = excess / np.sum(h[mask])
            h[mask] = np.clip(h[mask] * (1.0 - reduction), 0.0, 1.0)
        else:
            deficit = target - s
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            add = deficit / np.sum(room)
            h[mask] = np.clip(h[mask] + add * room, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    final_s = np.sum(h)
    if abs(final_s - target) > 1e-6:
        h = h * (target / final_s)
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

    best = h.copy()

    t0 = time.time()
    _CURRENT_N = n

            break

            res = minimize(
                _make_obj(tau),
                best,
                method='SLSQP',
                bounds=bounds,
                constraints=[_make_con()],
                options={'maxiter': 400, 'ftol': 1e-12}
            )
            if res.success or res.fun < best_c5 + tau:
                h_new = _repair(res.x, n)
                c5_new = _compute_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    best = h_new.copy()
        except Exception:
            pass
    
    # Random perturbation + re-optimize to escape local minima
    for _ in range(8):
        if time.time() - t0 > max_time:
            break
        sigma = 0.05 * max(1, n) ** (-0.25)
        h_trial = best + sigma * rng.randn(n)
        h_trial = _repair(h_trial, n)
        c5_trial = _compute_c5(h_trial)
        if c5_trial < best_c5:

        # Quick SLSQP refinement from perturbed point
