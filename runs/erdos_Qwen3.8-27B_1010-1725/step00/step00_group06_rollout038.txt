    n = len(h)

            add = min(diff, 1.0 - h[i])
            if add < 1e-15: continue
            h[i] += add; diff -= add
        else:
            sub = min(-diff, h[i])
            if sub < 1e-15: continue
            h[i] -= sub; diff += sub
        if abs(diff) < 1e-14: break
    return h

def grad_lag(h, k, n):
    """Gradient of O(k) w.r.t. h, where k is the index in the full correlate output."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad = 1.0 - 2.0 * h
    elif shift > 0:
        s = shift
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        s = -shift
        if s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    return grad
