```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute the C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_value(vals, alpha):
    """Compute smooth maximum via log-sum-exp."""
    a = alpha * vals
    m = np.max(a)
    return (m + np.log(np.sum(np.exp(a - m)))) / alpha

def smooth_max_grad(vals, alpha):
    """Compute gradient of smooth max wrt vals (returns weights)."""
    a = alpha * vals
    m = np.max(a)
    e = np.exp(a - m)
    return e / np.sum(e)

def obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return smooth_max_value(corr, alpha)

def grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_full = np.correlate(h, 1 - h, mode="full")
    corr = corr_full * dx
    weights = smooth_max_grad(corr, alpha)
    
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        # First term: (1-h[j-m])
        if m >= 0:
            js, je = m, n
        else:
            js, je = 0, n + m
        grad_h[js:je] += wk * (1 - h[js - m:je - m])
        # Second term: -h[j+m]
        if m >= 0:
            js2, je2 = 0, n - m
        else:
            js2, je2 = -m, n
        grad_h[js2:je2] -= wk * h[js2 + m:je2 + m]
    
    return grad_h * dx

def project_onto_polytope(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum}."""
    n = len(h)
    # Use bisection on the Lagrange multiplier
    lo, hi = -1.0, 1.0
    def projected(t):
        return np.clip(h - t, 0, 1)
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.sum(projected(mid))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    return projected((lo + hi) / 2)

def optimize_slsqp(h0, alpha, maxiter=500, time_limit=60):
    """Optimize using SLSQP."""
    n = len(h0)
    target_sum = n / 2.0
    
    # Ensure initial feasibility
    h0 = np.clip(h0, 0, 1)
    h0 = project_onto_polytope(h0, target_sum)
    
    result = minimize(
        fun=lambda x: obj(x, alpha),
        x0=h0,
        jac=lambda x: grad(x, alpha),
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-10}
    )
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        h_opt = project_onto_polytope(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    return h0, compute_c5(h0)

def optimize_projected_grad(h0, alpha, n_iter=5000, lr_init=0.01):
    """Projected gradient descent."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n_iter):
        g = grad(h, alpha)
        h_new = h - lr * g
        h_new = project_onto_polytope(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Decay learning rate
        lr *= 0.9999
    
    return best_h, best_c5

def interpolate_to_n(h, n_new):
    """Interpolate h from its grid to n_new points."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h)
    # Ensure sum constraint
    h_interp = np.clip(h_interp, 0, 1)
    h_interp = project_onto_polytope(h_interp, n_new / 2.0)
    return h_interp

def random_init(n, rng):
    """Generate random initial solution."""
    h = rng.uniform(0, 1, n)
    h = project_onto_polytope(h, n / 2.0)

def block_init(n, rng, n_blocks=4):
    """Generate block-structured initial solution."""
    block_size = n // n_blocks
    h = np.zeros(n)
    # Randomly assign blocks to 0 or 1
    for b in range(n_blocks):
        if b % 2 == 0:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(vals, alpha):
    a = alpha * vals
    m = a.max()
    return (m + np.log(np.exp(a - m).sum())) / alpha

def obj_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return smooth_max(corr, alpha)

def grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_full = np.correlate(h, 1 - h, mode="full")
    corr = corr_full * dx
    a = alpha * corr
    am = a.max()
    e = np.exp(a - am)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n - m]
        else:
            g[-m:] += wk * (1.0 - h[:n + m])
            g[:n + m] -= wk * h[-m:]
    return g * dx

def project_polytope(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def opt_slsqp(h0, alpha, maxiter=400):
    n = len(h0)
    h0 = project_polytope(np.clip(h0, 0.0, 1.0), n / 2.0)
    res = minimize(
        lambda x: obj_smooth(x, alpha),
        h0,
        lambda x: grad_smooth(x, alpha),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0}
    )
    h_opt = project_polytope(np.clip(res.x, 0.0, 1.0), n / 2.0)
    return h_opt, compute_c5(h_opt)

def opt_proj_grad(h0, alpha, n_iter=8000, lr=0.01):
    n = len(h0)
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    for i in range(n_iter):
        g = grad_smooth(h, alpha)
        h_new = project_polytope(h - lr * g, n / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h = h_new
        lr *= 0.9998
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) / n_old * 2.0 + 1.0 / n_old
    xn = np.arange(n_new) / n_new * 2.0 + 1.0 / n_new
    hi = np.interp(xn, xo, h)
    return project_polytope(np.clip(hi, 0.0, 1.0), n_new / 2.0)

    return project_polytope(h, n / 2.0)
```
