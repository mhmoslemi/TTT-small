```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_objective_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    idx = int(np.argmax(corr))
    L = idx - (n - 1)
    
    c5 = corr[idx] * dx
    
    grad = np.zeros(n)
    if L >= 0:
        M = L
        # sum_{n=0}^{n-1-M} h[n+M] * (1-h[n])
        for ni in range(n - M):
            grad[ni + M] += (1.0 - h[ni])
            grad[ni] += -h[ni + M]
    else:
        M = -L
        # sum_{n=M}^{n-1} h[n] * (1-h[n-M])
        for ni in range(M, n):
            grad[ni] += (1.0 - h[ni - M])
            grad[ni - M] += -h[ni]
    
    return c5, grad * dx

def optimize_h(h_init, max_time=300):
    n = len(h_init)
    target_sum = n / 2.0
    
    # Penalty method for sum constraint
    penalty = 100.0
    
    def obj(h):
        c5_val, c5_grad = c5_objective_and_grad(h)
        sum_val = np.sum(h) - target_sum
        total_obj = c5_val + penalty * sum_val * sum_val
        total_grad = c5_grad + 2 * penalty * sum_val * np.ones(n)
        return total_obj, total_grad
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj, h_init, method='L-BFGS-B', jac=True, bounds=bounds,
                      options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-10})
    
    h_opt = np.clip(result.x, 0, 1)
    # Project to satisfy sum constraint
    s = np.sum(h_opt)
    if s > 0:
        # Water-filling style projection
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_obj_grad(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    idx = int(np.argmax(corr))
    L = idx - (n - 1)
    c5 = float(corr[idx] * dx)
    grad = np.zeros(n)
    M = abs(L)
    if M < n:
        if L >= 0:
            grad[M:] += 1.0 - h[:n - M]
            grad[:n - M] += -h[M:]
        else:
            grad[:n - M] += 1.0 - h[M:]
            grad[M:] += -h[:n - M]
    return c5, grad * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_h(h, n_new):
    n_old = len(h)
    t_old = (np.arange(n_old) + 0.5) / n_old
    t_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(t_new, t_old, h)

def _sum_con_fun(h):
    return np.sum(h) - len(h) / 2.0

def _sum_con_jac(h):
    return np.ones(len(h))

def opt_slsqp(h_init, max_iter=3000):
    n = len(h_init)
    target = n / 2.0
    hs = project_sum(h_init, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(c5_obj_grad, hs, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        ho = np.clip(res.x, 0.0, 1.0)
    except Exception:
        ho = hs
    ho = project_sum(ho, target)
    return ho, compute_c5(ho)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = (rng.rand(42) > 0.5).astype(float)

    best_h = project_sum(h0.copy(), len(h0) / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize at base resolution
    h_opt, c5_opt = opt_slsqp(best_h, max_iter=5000)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # Phase 2: multi-resolution upsample + optimize
    h_cur = best_h.copy()
    for _ in range(5):
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        n_new = min(len(h_cur) * 2, 512)
        if n_new <= len(h_cur):
            break
        h_up = project_sum(upsample_h(h_cur, n_new), n_new / 2.0)
        mi = max(400, int(rem * 25))
        h_new, c5_new = opt_slsqp(h_up, max_iter=mi)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        h_cur = h_new

    # Phase 3: random restarts and perturbations
    rem = budget_s - (time.time() - t0)
    n_b = len(best_h)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for attempt in range(30):
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        starts = []
        scale = 0.06 + 0.04 * np.exp(-0.2 * attempt)
        starts.append(project_sum(best_h + scale * rng.randn(n_b), n_b / 2.0))
        starts.append(project_sum((rng.rand(n_b) > 0.5).astype(float), n_b / 2.0))
        gr = (np.mod(np.arange(n_b) * phi, 1.0) < 0.5).astype(float)
        starts.append(project_sum(gr, n_b / 2.0))
        # Thue-Morse inspired
        tm = np.zeros(n_b)
        for i in range(n_b):
            c = 0
            x = i
            while x > 0:
                c += x & 1
                x >>= 1
            tm[i] = 1.0 if c % 2 == 0 else 0.0
        starts.append(project_sum(tm, n_b / 2.0))

        idx_best = 0
        min_c5 = compute_c5(starts[0])
        for si in range(1, len(starts)):
            ci = compute_c5(starts[si])
            if ci < min_c5:
                min_c5 = ci
                idx_best = si

        mi = max(300, int(rem * 12))
        h_new, c5_new = opt_slsqp(starts[idx_best], max_iter=mi)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final validation
    n_f = len(best_h)
    best_h = project_sum(best_h, n_f / 2.0)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_f)
```
