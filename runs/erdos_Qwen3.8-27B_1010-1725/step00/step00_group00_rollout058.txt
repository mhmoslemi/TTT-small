import numpy as np
from scipy.optimize import minimize
import time

def _compute_corr_values(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _c5_objective(x, n_points):
    vals = _compute_corr_values(x, n_points)
    return vals.max()

def _c5_gradient(x, n_points):
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_val = corr.max()
    tol = 1e-8 * max_val + 1e-12
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1.0 - x[j + lag])
            if 0 <= j - lag < n:
                g -= x[j - lag]
            grad[j] += g
    grad *= dx / len(active)
    return grad

def _slsqp_step(h, n_points, maxiter=200):
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}]
    res = minimize(
        lambda x: _c5_objective(x, n_points),
        h,
        method='SLSQP',
        jac=lambda x: _c5_gradient(x, n_points),
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-10}
    )
    if res.success or res.fun < _c5_objective(h, n_points):
        return res.x, res.fun
    return h, _c5_objective(h, n_points)

def _project_sum(h, n_points):
    """Project h to satisfy sum constraint while staying in [0,1]."""
    target = n_points / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Adjust
    diff = target - s
    # Distribute evenly, then clip and re-clip
    h_adj = h + diff / n_points
    h_adj = np.clip(h_adj, 0, 1)
    # If still off, do iterative projection
    for _ in range(50):

        if abs(s - target) < 1e-10:

        diff = target - s
        mask = (h_adj < 1.0) if diff > 0 else (h_adj > 0.0)
        if mask.sum() == 0:
            break
        h_adj[mask] += diff / mask.sum()
        h_adj = np.clip(h_adj, 0, 1)
    return h_adj

    return h

    h = np.clip(h, 0, 1)

    return h

from scipy.optimize import minimize
import time

def _c5_val(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h, n):
    target = n * 0.5
    h = h.copy().astype(float)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            mask = np.ones(n, dtype=bool)
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_c5(h, n):
    dx = 2.0 / n
    corr = _c5_corr(h, n)
    max_val = corr.max()
    tol = 1e-9
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            ip = j + lag
            im = j - lag
            if 0 <= ip < n:
                g += (1.0 - h[ip])
            if 0 <= im < n:
                g -= h[im]
            grad[j] += g
    grad *= dx / len(active)
    return grad

    h = _fix_sum(h, n)

        return _c5_val(x, n)

        return _grad_c5(x, n)

    c5 = _c5_val(h, n)
    T = 0.002
    for s in range(n_steps):
        T_cur = T * (1.0 - s / float(n_steps)) + 0.0001
        n_pert = rng.randint(1, 4)
        idxs = rng.choice(n, n_pert, replace=False)
        h_new = h.copy()
        for idx in idxs:
            h_new[idx] = np.clip(h_new[idx] + rng.uniform(-0.1, 0.1), 0.0, 1.0)
        h_new = _fix_sum(h_new, n)
        c5_new = _c5_val(h_new, n)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T_cur, 1e-8)):
            h = h_new
            c5 = c5_new
    return h, c5

def _rand_h(n, rng):
    raw = rng.random(n) + 0.01
    h = raw / raw.sum() * n * 0.5
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n)

def _struct_h(n, rng, mode):
    x = np.arange(n, dtype=float) / n * 2.0
    if mode == 0:
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    elif mode == 2:
        h = np.where((x % 1.0) < 0.5, 0.9, 0.1)
    else:
        h = 0.5 + 0.35 * np.cos(np.pi * x * 1.5)
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    best_h = h_init.copy()
    best_c5 = _c5_val(best_h, len(best_h))
    best_n = len(best_h)
    
    def ok(frac=1.0):
        return (time.time() - t0) < budget_s * frac
    
    # Phase 1: Optimize initial with SLSQP + local search
    if ok():
        h, c5 = _slsqp_opt(best_h, best_n, 400)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _local_search(best_h, best_n, 400)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _slsqp_opt(best_h, best_n, 300)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 2: Perturbation search from best
    if ok(0.90):
        h, c5 = _perturb_search(best_h, best_n, 300, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _slsqp_opt(best_h, best_n, 300)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 3: Random restarts
    for i in range(30):
        if not ok(0.80):
            break
        h = _rand_h(best_n, rng)
        h, c5 = _slsqp_opt(h, best_n, 120)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
            h, c5 = _local_search(best_h, best_n, 200)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
    
    # Phase 4: Structured starts
    for mode in range(4):
        if not ok(0.72):
            break
        h = _struct_h(best_n, rng, mode)
        h, c5 = _slsqp_opt(h, best_n, 150)
        h, c5 = _local_search(h, best_n, 150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 5: Perturb best and re-optimize
    for i in range(15):
        if not ok(0.62):
            break
        h_p = best_h + rng.normal(0, 0.015, best_n)
        h_p = np.clip(h_p, 0, 1)
        h_p = _fix_sum(h_p, best_n)
        h, c5 = _slsqp_opt(h_p, best_n, 150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 6: Try nearby n values
    for delta_n in [1, -1, 2, -2, 5]:
        if not ok(0.50):
            break
        n_try = best_n + delta_n
        if n_try < 20:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = np.clip(h_up, 0, 1)
        h_up = _fix_sum(h_up, n_try)
        h, c5 = _slsqp_opt(h_up, n_try, 200)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n_try
            h, c5 = _local_search(best_h, best_n, 300)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
    
    # Phase 7: Final polish
    if ok(0.30):
        h, c5 = _slsqp_opt(best_h, best_n, 500)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _local_search(best_h, best_n, 500)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _slsqp_opt(best_h, best_n, 400)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = _perturb_search(best_h, best_n, 300, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _c5_val(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
