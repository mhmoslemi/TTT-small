        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def _sum_cons(x, target):
    return np.sum(x) - target

def _soft_obj(x, dx, t_soft):
    corr = np.correlate(x, 1 - x, mode='full') * dx
    cm = np.max(corr)
    return cm + np.log(np.sum(np.exp(t_soft * (corr - cm)))) / t_soft

def _slsqp(h, n, dx, target, t_soft, maxiter):
    cons = [{'type': 'eq', 'fun': _sum_cons, 'args': (target,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_soft_obj, h.copy(), method='SLSQP', bounds=bounds,
                      constraints=cons, args=(dx, t_soft),
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = np.clip(res.x, 0, 1)
        h_out = fix_sum(h_out, target)
        if np.any(np.isnan(h_out)):
            return h
        return h_out
    except Exception:
        return h

def _sa_loop(h_best, c5_best, n, dx, target, rng, t_start, t_end, T_init, T_floor):
    T = T_init
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        trial = h_best.copy()
        r = rng.random()
        if r < 0.35:
            trial[i], trial[j] = trial[j], trial[i]
        elif r < 0.70:
            eps = rng.uniform(-0.6, 0.6)
            trial[i] = np.clip(trial[i] - eps, 0, 1)
            trial[j] = np.clip(trial[j] + eps, 0, 1)
            trial = fix_sum(trial, target)
        else:
            k = rng.randint(n)
            if k != i and k != j:
                trial[i], trial[j], trial[k] = trial[j], trial[k], trial[i]
            else:
                continue
        c5_try = compute_c5(trial, dx)
        d = c5_try - c5_best
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            h_best, c5_best = trial, c5_try
        frac = min(1.0, (time.time() - t_start) / max(t_end - t_start, 1e-6))
        T = max(T_floor, T_init * np.exp(-3.0 * frac))
    return h_best, c5_best

def _greedy_loop(h_best, c5_best, n, dx, target, rng, t_end):
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        trial = h_best.copy()
        trial[i], trial[j] = trial[j], trial[i]
        c5_try = compute_c5(trial, dx)
        if c5_try < c5_best:
            h_best, c5_best = trial, c5_try
            continue
        trial = h_best.copy()
        eps = rng.uniform(-0.4, 0.4)
        trial[i] = np.clip(trial[i] - eps, 0, 1)
        trial[j] = np.clip(trial[j] + eps, 0, 1)
        trial = fix_sum(trial, target)
        c5_try = compute_c5(trial, dx)
        if c5_try < c5_best:
            h_best, c5_best = trial, c5_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s * 0.96

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: SA with periodic SLSQP (45% budget)
    p1_end = t0 + budget_s * 0.45
    best_h, best_c5 = _sa_loop(best_h, best_c5, n, dx, target, rng, t0, p1_end, 0.012, 0.0004)

    # Periodic SLSQP within remaining phase 1 was skipped; do it now
    h_new = _slsqp(best_h, n, dx, target, 120.0, 1500)
    c5_new = compute_c5(h_new, dx)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new

    # Phase 2: SLSQP with increasing t_soft (25% budget)
    p2_end = t0 + budget_s * 0.72
    for ts in [80.0, 250.0, 800.0, 2000.0]:
        if time.time() > p2_end:
            break
        h_new = _slsqp(best_h, n, dx, target, ts, 4000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-10:
            best_h, best_c5 = h_new, c5_new
        # Quick SA polish
        if time.time() < p2_end - 30:
            best_h, best_c5 = _sa_loop(best_h, best_c5, n, dx, target, rng,
                                        time.time(), min(time.time() + 40, p2_end), 0.003, 0.0005)

    # Phase 3: Try a second SLSQP pass from a slightly perturbed start
    if time.time() < t0 + budget_s * 0.85:
        h_pert = best_h.copy()
        idx = rng.choice(n, 5, replace=False)
        h_pert[idx] = np.clip(h_pert[idx] + rng.uniform(-0.1, 0.1, 5), 0, 1)
        h_pert = fix_sum(h_pert, target)
        h_new = _slsqp(h_pert, n, dx, target, 500.0, 3000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Final greedy hill-climb
    best_h, best_c5 = _greedy_loop(best_h, best_c5, n, dx, target, rng, deadline)

    # Final SLSQP polish if time remains
    if time.time() < t0 + budget_s * 0.98:
        h_new = _slsqp(best_h, n, dx, target, 3000.0, 2000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    return best_h, best_c5, n
