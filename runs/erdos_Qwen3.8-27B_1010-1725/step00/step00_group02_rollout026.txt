            h_opt, c5_opt = optimize_sa(h0, n, dx, target, seed + idx * 7919, t_sa)
            h_opt, c5_opt = polish(h_opt, n, dx, target, time_limit=min(3.0, t_rem * 0.1))
            h_opt, c5_opt = focused_polish(h_opt, n, dx, target, time_limit=min(5.0, t_rem * 0.15))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    if best_h is None:
        h_fb = np.zeros(50)
        h_fb[:25] = 1.0
        best_h = h_fb
        best_c5 = compute_c5(best_h, 0.04)
    return (best_h, best_c5, len(best_h))
