def project_sum(h, target, lo=0.0, hi=1.0):
    """Project h onto {lo <= x <= hi, sum(x) = target}"""
    h = np.clip(h, lo, hi).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    if diff > 0:
        room = hi - h
        active = room > 1e-15

        if total_room <= 0:
            return h
        adj = diff / total_room
        h[active] += adj * room[active]
        # Re-clip for any that exceeded
        h = np.clip(h, lo, hi)
    else:

        active = excess > 1e-15

        if total_excess <= 0:
            return h
        adj = (-diff) / total_excess
        h[active] -= adj * excess[active]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
