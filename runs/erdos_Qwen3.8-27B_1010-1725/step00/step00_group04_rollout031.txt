        _gp['n'] = len(best_h)
        try:
            res = minimize(exact_obj, best_h, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-11,
                                   'adaptive': True})
            h_nm = project_h(res.x, len(best_h))
            c5 = compute_c5(h_nm, len(best_h))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_nm
        except Exception:
            pass

    # Phase 4: Try different resolutions
    n_curr = len(best_h)
    for n_try in [56, 70, 84]:
        if time.time() - t0 > budget_s * 0.92:
            break
        x_old = np.linspace(0, 2, n_curr, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_h(h_up, n_try)
        temps4 = [5, 2, 1, 0.5, 0.2, 0.1, 0.05]
        h_opt4 = anneal_optimize(h_up, n_try, temps4, t0, budget_s, 300)
        c5 = compute_c5(h_opt4, n_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt4.copy()
            n_curr = n_try

    # Phase 5: Final L-BFGS-B polish with very low temperature
    if time.time() - t0 < budget_s * 0.96:
        n_fin = len(best_h)
        _gp['n'] = n_fin
        _gp['t'] = 0.01
        try:
            res = minimize(smooth_obj, best_h, method='L-BFGS-B',
                          options={'maxiter': 300, 'gtol': 1e-10, 'ftol': 1e-14})
            h_fin = project_h(res.x, n_fin)
            c5 = compute_c5(h_fin, n_fin)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fin
        except Exception:
            pass

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
