import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smax(x, alpha):
    m = np.max(x)
    s = np.sum(np.exp((x - m) * alpha))
    return m + np.log(s) / alpha

def project_feasible(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if not np.any(m):
            break
        h[m] += d / np.sum(m)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(h0, n, max_iter, t0, budget_s):
    dx = 2.0 / n
    target = n / 2.0
    best = h0.copy()
    best_c5 = compute_c5(best)
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    for alpha in [15, 30, 60, 120, 250, 500]:
        if time.time() - t0 > budget_s - 20:
            break
        def obj(h, a=alpha):
            c = np.correlate(h, 1.0 - h, mode='full')
            return smax(c, a) * dx
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best = h_new
                    best_c5 = c5
        except Exception:
            pass
    return best, best_c5

def local_refine(h, n, t0, budget_s, iters=2000, step0=0.02):
    """Subgradient / Nelder-Mead style local refinement."""
    best = h.copy()
    best_c5 = compute_c5(best)
    step = step0
    for it in range(iters):
        if time.time() - t0 > budget_s - 10:
            break
        n_move = max(1, n // 15)
        idx = np.random.randint(0, n, size=n_move)
        delta = np.random.randn(n_move) * step
        h_try = best.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best = h_try
            best_c5 = c5_try
            step = min(step * 1.05, 0.1)
        else:
            step = max(step * 0.92, 0.001)
    return best, best_c5

    t0 = time.time()
    global rng
    rng = np.random.RandomState(seed)

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
