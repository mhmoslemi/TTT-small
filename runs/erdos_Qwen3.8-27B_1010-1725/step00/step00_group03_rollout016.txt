import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_block_init(n, a, b):
    pat = np.concatenate([np.ones(a), np.zeros(b)])
    reps = int(np.ceil(n / len(pat)))
    return np.tile(pat, reps)[:n]

def make_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    r = rng.random(n)
    # Sort and threshold for exact sum
    target = n / 2.0
    return project_feasible(r, target)

def smooth_max_grad(corr, h, n, T):
    cm = np.max(corr)
    w = np.exp(T * (corr - cm))
    ws = w.sum()
    if ws < 1e-15:
        return np.zeros(n)
    w = w / ws
    j = np.arange(n)[:, None].astype(np.float64)
    p = np.arange(2 * n - 1)[None, :].astype(np.float64) - (n - 1)
    o1 = j - p
    o2 = j + p
    m1 = (o1 >= 0) & (o1 < n)
    m2 = (o2 >= 0) & (o2 < n)
    t1 = np.where(m1, 1.0 - h[np.clip(o1.astype(int), 0, n - 1)], 0.0)
    t2 = np.where(m2, h[np.clip(o2.astype(int), 0, n - 1)], 0.0)
    return (t1 * w[None, :] - t2 * w[None, :]).sum(axis=1)
