            h = np.clip(h, 0, 1)

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr * dx))

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    target = n_new / 2.0
    for _ in range(50):
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-12:
            break
        mask = (h_new > 1e-10) & (h_new < 1 - 1e-10)
        if mask.sum() == 0:
            mask = np.ones(n_new, dtype=bool)
        h_new[mask] += diff / mask.sum()
        h_new = np.clip(h_new, 0, 1)
    return h_new

def project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() == 0:
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    vmax = vals.max()
    logsum = vmax + np.log(np.sum(np.exp(alpha * (vals - vmax)))) / alpha
    p = np.exp(alpha * (vals - vmax)) / np.sum(np.exp(alpha * (vals - vmax)))
    grad = np.zeros(n)
    centers = np.arange(len(corr)) - (n - 1)
    for idx in range(len(corr)):
        if p[idx] < 1e-12:
            continue
        m_lag = abs(centers[idx])
        if m_lag == 0:
            grad += p[idx] * (1.0 - 2.0 * h)
        else:
            grad[m_lag:] += p[idx] * (1.0 - h[:n - m_lag])
            grad[:n - m_lag] -= p[idx] * h[m_lag:]
    return float(logsum), grad * dx

def smooth_c5_grad(h, alpha):

    vals = corr * dx
    vmax = vals.max()
    shifted = alpha * (vals - vmax)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    p = exp_s / sum_exp
    logsum = vmax + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    centers = np.arange(len(corr)) - (n - 1)
    for idx in range(len(corr)):
        w = p[idx]
        if w < 1e-13:
            continue
        m_lag = abs(centers[idx])
        if m_lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[m_lag:] += w * (1.0 - h[:n - m_lag])
            grad[:n - m_lag] -= w * h[m_lag:]
    return float(logsum), grad * dx

        return smooth_c5_grad(x, alpha)[1]

        return np.sum(x) - target_sum

        return np.ones(n)

                 bounds=[(0, 1)] * n,

                 options={'maxiter': maxiter, 'ftol': 1e-9})
