from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(v, target):
    def f(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0, 1)

def soft_grad_obj(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    sh = np.clip(tau * (corr - mc), -300, 300)
    ev = np.exp(sh)
    se = ev.sum()
    obj = mc + np.log(se) / tau
    w = ev / se
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        if w[l] < 1e-13:
            continue
        lag = l - (n - 1)
        for j in range(n):
            t = 0.0
            a = j - lag
            b = j + lag
            if 0 <= a < n:
                t += 1.0 - h[a]
            if 0 <= b < n:
                t -= h[b]
            if t != 0.0:
                grad[j] += w[l] * dx * t
    return obj, grad
