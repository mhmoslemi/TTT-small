def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def make_objective(n, beta):
    dx = 2.0 / n
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        # clip to avoid overflow in exp

        max_val = np.max(corr_clipped)
        shifted = beta * (corr_clipped - max_val)
        return max_val + np.log(np.sum(np.exp(shifted))) / beta

from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def solve_lp_for_h(g, n):
    # Given fixed g = 1-h, find h that minimizes max_k corr_k
    # min t s.t. A @ h <= t*e, sum(h) = n/2, 0<=h<=1
    # Variables: h[0..n-1], t -> n+1 variables
    # 
    # For each lag k (0 to 2n-2 in correlate output):
    #   corr[k] = sum_i h[i+k_n] * g[i] where k_n = k - (n-1)
    #   This equals sum_i g[i] * h[i + k_n] (for valid indices)
    #   = sum_j g[j - k_n] * h[j] (reindexing)
    #
    # Actually let me just compute the matrix A directly.
    
    m = 2 * n - 1  # number of lags
    # A is m x n, A[k, j] = g[j - (k-n+1)] if valid
    A = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)  # actual lag
        for j in range(n):
            i = j - lag  # index in g
            if 0 <= i < n:
                A[k, j] = g[i]
    
    # min t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub @ [h, t] <= b_ub
    # A[k, j] * h[j] - t <= 0  =>  A[k, :] @ h + [-1] * t <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(h) = n/2
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([n / 2.0])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds_h, method='highs')
    
    if res.success:

    return None

def compute_c5(h):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def lsq_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = float(np.max(corr))
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def project_h(x, n):
    h = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(h0, n, t0, budget_s, seed):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    betas = [8, 20, 50, 120, 300, 600]
    cur = h0.copy()
    for beta in betas:
        if time.time() - t0 > budget_s - 30:
            break
        n_restarts = 3 if beta <= 20 else 2
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 20:
                break
            x0 = cur.copy() if r == 0 else project_h(cur + rng.randn(n) * 0.04, n)
            def obj(x, b=beta):
                return lsq_obj(x, b)
            try:
                res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 400, 'ftol': 1e-10})
                h_new = project_h(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    cur = h_new.copy()

                pass
