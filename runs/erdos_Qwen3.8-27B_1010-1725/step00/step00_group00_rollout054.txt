def soft_obj_grad(x, n, temp):
    dx = 2.0 / n
    c1 = 1.0 - x
    corr = np.correlate(x, c1, mode='full')  # length 2n-1
    
    m = np.max(corr)
    exp_c = np.exp(temp * corr - temp * m)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp  # softmax weights, length 2n-1
    
    obj = (m + np.log(sum_exp) / temp) * dx
    
    # Gradient
    grad = np.zeros(n)
    # corr[k] corresponds to lag k = k_index - (n-1)
    # i.e., corr[k_index] = c_{k_index - (n-1)}
    
    for ki in range(2*n - 1):

        wk = w[ki]

            if 0 <= j + k < n:
                g -= x[j + k]
            grad[j] += wk * g

                    best_h = res.x.copy()
                    best_c5 = c5
                h_rand = res.x

    # Phase 3: Try even n values (allow near-binary)
    for n2 in [50, 44, 46, 60]:
        if check_time(start, budget_s, 0.70):
            break
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.arange(n2) * (2.0 / n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2, n2)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
        for temp in [0.1, 0.02, 0.005]:
            if check_time(start, budget_s, 0.75):
                break
            res = do_slsqp(h2, n2, temp, 200, 1e-12)
            if res.x is not None:
                c5_2 = compute_c5(res.x, n2)
                if c5_2 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_2
                    n = n2
                h2 = res.x

    # Phase 4: Binary local search on even n
    if not check_time(start, budget_s, 0.80):
        n_bin = len(best_h)
        if n_bin % 2 != 0:
            n_bin += 1
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.arange(n_bin) * (2.0 / n_bin)
        h_bin = np.interp(x_new, x_old, best_h)
        h_bin = project_feasible(h_bin, n_bin)
        # Round to nearest binary with exact count
        target_count = n_bin // 2
        idx_sorted = np.argsort(-h_bin)
        h_bin_b = np.zeros(n_bin)
        h_bin_b[idx_sorted[:target_count]] = 1.0
        c5_bin = compute_c5(h_bin_b, n_bin)
        if c5_bin < best_c5:
            best_h = h_bin_b.copy()
            best_c5 = c5_bin

        # Swap-based local search
        improved = True
        max_iter = 2000
        it = 0
        while improved and it < max_iter:
            if check_time(start, budget_s, 0.88):
                break
            improved = False
            it += 1
            ones = np.where(h_bin_b == 1.0)[0]
            zeros = np.where(h_bin_b == 0.0)[0]
            n_swaps = min(200, len(ones) * len(zeros))
            for _ in range(n_swaps):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin_b[i] = 0.0
                h_bin_b[j] = 1.0
                c5_try = compute_c5(h_bin_b, n_bin)
                if c5_try < best_c5 - 1e-12:
                    best_h = h_bin_b.copy()
                    best_c5 = c5_try
                    improved = True
                else:
                    h_bin_b[i] = 1.0
                    h_bin_b[j] = 0.0

    # Phase 5: Final fine-tune with SLSQP on best solution
    if not check_time(start, budget_s, 0.92):
        n_f = len(best_h)
        for temp in [0.005, 0.001]:
            if check_time(start, budget_s, 0.95):
                break
            res = do_slsqp(best_h, n_f, temp, 300, 1e-14)
            if res.x is not None:
                c5_f = compute_c5(res.x, n_f)
                if c5_f < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_f

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, n_final)

    return (best_h, best_c5, n_final)
