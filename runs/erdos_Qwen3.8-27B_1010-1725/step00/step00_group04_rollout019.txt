                best_c5 = c5_ls
                best_h = h_ls

    # Final fine local search
    remaining = budget_s - (time.time() - t_start) - 3
    if remaining > 1:
        best_h, best_c5 = local_search(best_h, n, remaining)

    # Ensure feasibility
    best_h = repair_h(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
