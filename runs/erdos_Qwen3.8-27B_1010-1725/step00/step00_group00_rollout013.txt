def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(h, n):
    h = h.astype(float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - n / 2.0
        if abs(diff) < 1e-15:

        h = h - diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    h[-1] = h[-1] + (n / 2.0 - np.sum(h))
