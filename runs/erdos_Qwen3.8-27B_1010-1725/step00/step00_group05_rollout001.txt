def construct_base(n, kind=0, alpha=1.57079632679, contrast=1.0):

def value_grad(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = float(np.max(c))
    n = h.size
    idx = n - 1
    if tau <= 1e-7:
        act = c >= mx - 1e-8
        if not act.any(): act[idx] = True
        w = act.astype(float); w /= w.sum(); f = mx
    else:
        z = np.exp((c - mx) / tau); s = z.sum(); w = z / s; f = float(mx + tau * np.log(s))
    g = np.zeros(n)
    w0 = w[idx]
    if w0 > 0.0:
        g += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wm = w[idx - p]; wp = w[idx + p]
        if wm == 0.0 and wp == 0.0:
            continue
        s2 = wm + wp
        m = n - p
        b = h[:m]
        sh = h[p:]
        g[:m] += wm - s2 * sh
        g[p:] += wp - s2 * b
    return f, g

import time
import numpy as np


def project_sum(x, S):
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if abs(float(y.sum()) - S) < 1e-12:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(40):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(y - m, 0.0, 1.0)) > S:
            lo = m
        else:
            hi = m
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)


def sym_h(x):
    return 0.5 * (x + x[::-1])


def exact_c5(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def smooth_val(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = float(np.max(c))
    if tau <= 1e-7:
        return mx
    z = np.exp((c - mx) / tau)
    return float(mx + tau * np.log(z.sum()))


def value_grad(h, tau):
    c = np.correlate(h, 1.0 - h, mode="full")
    mx = float(np.max(c))
    n = h.size
    idx = n - 1
    if tau <= 1e-7:
        act = c >= (mx - 1e-8)
        if not act.any():
            act[int(np.argmax(c))] = True
        w = act.astype(float)
        w /= w.sum()
        f = mx
    else:
        z = np.exp((c - mx) / tau)
        s = z.sum()
        w = z / s
        f = float(mx + tau * np.log(s))

    g = np.zeros(n)
    w0 = w[idx]
    if w0 > 0.0:
        g += w0 * (1.0 - 2.0 * h)

    for p in range(1, n):
        wm = w[idx - p]
        wp = w[idx + p]
        if wm == 0.0 and wp == 0.0:
            continue
        s2 = wm + wp
        m = n - p
        b = h[:m]
        sh = h[p:]
        g[:m] += wm - s2 * sh
        g[p:] += wp - s2 * b
    return f, g


def local_optimize(h, n, end_time, sym=False, tau_list=None):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = int(n)
    S = 0.5 * n
    if sym:
        h = sym_h(h)
    h = project_sum(h, S)
    bh = h.copy()
    bc = exact_c5(h)

    if tau_list is None:
        tau_list = (0.7, 0.35, 0.175, 0.0875, 0.04375, 0.021875,
                    0.0109375, 0.00546875, 0.002734375, 0.0013671875)

    step = 0.12
    for tau in tau_list:
        if time.monotonic() >= end_time:
            break
        stall = 0
        it = 0
        while it < 300 and time.monotonic() < end_time:
            it += 1
            fs, gr = value_grad(h, tau)
            nr = float(np.linalg.norm(gr))
            if nr < 1e-9:
                break
            lam = step
            used = 0.0
            ok = False
            for _ in range(12):
                cand = h - (lam / nr) * gr
                cand = project_sum(cand, S)
                if sym:
                    cand = sym_h(cand)
                fs2 = smooth_val(cand, tau)
                if fs2 + 1e-12 < fs:
                    h = cand
                    fs = fs2
                    used = lam
                    ok = True
                    break
                lam *= 0.5
            if ok:
                step = max(0.02, min(0.6, used * 1.1))
                stall = 0
                c = exact_c5(h)
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                step = max(1e-4, step * 0.5)
                stall += 1
                if stall > 10:
                    break
        h = project_sum(h, S)
        if sym:
            h = sym_h(h)
        c = exact_c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
    return bh, bc


def noise_polish(h, n, end_time, seed=1, sym=False):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = int(n)
    S = 0.5 * n
    if sym:
        h = sym_h(h)
    h = project_sum(h, S)
    bh = h.copy()
    bc = exact_c5(h)
    if time.monotonic() >= end_time:
        return bh, bc

    rng = np.random.RandomState(int(seed) % 2147483647 + 1)
    std = 0.025
    temp = 2e-3
    min_temp = 1e-5
    fails = 0

    while time.monotonic() < end_time and std > 1e-5:
        k = int(max(2, min(n, (0.08 + 0.5 * rng.random()) * n)))
        v = h.copy()
        idx = rng.choice(n, k, replace=False)
        v[idx] += std * rng.normal(size=k)
        v = project_sum(v, S)
        if sym:
            v = sym_h(v)
        c = exact_c5(v)

        if c <= bc:
            bh = v.copy()
            bc = c
            h = v
            std = min(0.2, std * 1.15)
            temp = min(temp, 2e-3)
            fails = 0
        else:
            gap = c - bc
            if gap < temp and rng.random() < float(np.exp(-gap / temp)):
                h = v.copy()
                temp = max(min_temp, temp * 0.85)
                fails = 0
                continue
            fails += 1
            temp = max(min_temp, temp * 0.95)
            if fails > max(30, int(80 * std)):
                std *= 0.75
                fails = 0
    return bh, bc


def pair_polish(h, n, end_time, seed=1):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = int(n)
    S = 0.5 * n
    h = project_sum(h, S)
    bh = h.copy()
    bc = exact_c5(h)
    if time.monotonic() >= end_time:
        return bh, bc

    rng = np.random.RandomState(int(seed) % 2147483647 + 1)
    fracs = (0.1, 0.3, 0.5, 0.7, 0.9)
    fails = 0
    infos = 0

    while time.monotonic() < end_time:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            fails += 1
            if fails > 50:
                break
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi <= lo + 1e-10:
            fails += 1
            if fails > 80:
                break
            continue

        lb = bc
        lv = None
        for fr in fracs:
            d = lo + fr * (hi - lo)
            v = h.copy()
            v[i] = h[i] + d
            v[j] = h[j] - d
            if v[i] < 0.0:
                v[i] = 0.0
            elif v[i] > 1.0:
                v[i] = 1.0
            if v[j] < 0.0:
                v[j] = 0.0
            elif v[j] > 1.0:
                v[j] = 1.0
            c = exact_c5(v)
            if c < lb:
                lb = c
                lv = v
        infos += 1
        if lv is not None:
            h = lv.copy()
            bh = h.copy()
            bc = lb
            fails = 0
        else:
            fails += 1
            if fails > 60 or infos > 2000:
                break

    h = project_sum(h, S)
    c = exact_c5(h)
    if c < bc:
        bc = c
        bh = h
    return bh, bc


def construct_base(n, kind=0, alpha=1.57079632679, contrast=1.0):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    u = x - 1.0
    if kind == 0:
        f = np.cos(alpha * u)
    elif kind == 1:
        f = 1.0 - np.abs(u)
    else:
        f = 1.0 - u * u
    y = f - f.mean()
    scale = float(np.max(np.abs(y)))
    if scale < 1e-12:
        return project_sum(np.full(n, 0.5), 0.5 * n)
    y = contrast * y / scale
    h = 0.5 * (1.0 + y)
    return project_sum(np.clip(h, 0.0, 1.0), 0.5 * n)


def best_shape_candidate(n):
    n = int(n)
    best_h = None
    best_c = 1e9
    alphas = (0.8, 1.1, 1.57079632679, 1.9, 2.3, 2.8)
    contrasts = (0.6, 1.0)
    for kind in (0, 1, 2):
        for a in alphas:
            for c in contrasts:
                h = construct_base(n, kind, a, c)
                v = exact_c5(h)
                if v < best_c:
                    best_c = v
                    best_h = h.copy()
    return best_h


def resample_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).ravel()
    n0 = h.size
    if n0 == n:
        return project_sum(h, 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_sum(np.interp(x1, x0, h), 0.5 * n)


def get_candidate(n, kind, parent, best_h):
    n = int(n)
    if kind == 0:
        src = best_h if best_h is not None else parent
        if src is not None:
            return resample_h(src, n)
        return construct_base(n, 0, np.pi * 0.5, 1.0)
    if kind == 1:
        return best_shape_candidate(n)
    src = best_h if best_h is not None else parent
    if src is None:
        return best_shape_candidate(n)
    h = resample_h(src, n)
    rng = np.random.RandomState(1234 + 17 * n)
    h = h + rng.normal(scale=0.10, size=h.size)
    return project_sum(h, 0.5 * n)


def optimize_candidate(h, n, end_time, seed=1, sym=False):
    n = int(n)
    h = np.asarray(h, dtype=float).ravel().copy()
    S = 0.5 * n
    if sym:
        h = sym_h(h)
    h = project_sum(h, S)
    if time.monotonic() >= end_time:
        return h, exact_c5(h)
    stop_local = min(end_time, time.monotonic() + 0.75 * (end_time - time.monotonic()))
    h, c = local_optimize(h, n, stop_local, sym)
    if time.monotonic() + 1.0 < end_time:
        h, c = noise_polish(h, n, end_time, int(seed) + 77, sym)
    return h, c
