def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def objective_and_grad(h, dx, n, beta=50.0):
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    obj_val = max_val * dx
    
    # Softmax weights
    shifted = beta * (corr - max_val)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    g = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        m_lo = max(0, -s)
        m_hi = min(n - 1, n - 1 - s)

        g[m_lo:m_hi+1] += w

        hi2 = m_hi + s
        idx2 = np.arange(max(0, lo2), min(n, hi2 + 1))
        g[idx2] -= w * h[idx2 - s]
    
    return obj_val, g * dx

            step = diff / np.sum(mask)

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0, 1)
    return h

def obj_and_grad(h, dx, n, beta=80.0):
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    obj_val = max_val * dx
    shifted = beta * (corr - max_val)
    exps = np.exp(shifted)
    se = np.sum(exps)
    if se < 1e-300:
        weights = np.zeros(2*n-1)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / se
    g = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lo1 = k
        hi1 = n - 1
        if lo1 <= hi1:
            cnt = hi1 - lo1 + 1
            g[lo1:hi1+1] += w * (1.0 - h[:cnt])
        lo2 = 0
        hi2 = n - 1 - k
        if lo2 <= hi2:
            cnt = hi2 - lo2 + 1
            g[lo2:hi2+1] -= w * h[k:k+cnt]
    return obj_val, g * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(95)
        h0[0:24] = 1.0
        h0[24:48] = 0.5
        h0[48:72] = 1.0
        h0[72:95] = 0.5
    n = len(h0)
    dx = 2.0 / n
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = eval_c5(best_h)

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0, 'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n

    def obj_wrap(x):
        v, _ = obj_and_grad(x, dx, n)
        return v

    def grad_wrap(x):
        _, g = obj_and_grad(x, dx, n)
        return g

    def run_slsqp(h_init, maxiter, ftol):
        res = minimize(obj_wrap, h_init, jac=grad_wrap, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res

    # Multiple starts
    starts = [h0.copy()]

    # Noisy perturbations
    for scale in [0.02, 0.05, 0.1]:
        hp = h0 + rng.uniform(-scale, scale, n)
        starts.append(project_feasible(hp, n))

    # Antisymmetric structure
    h_anti = h0.copy()
    h_anti[:n//2] = 0.5 + 0.3*np.sign(h0[:n//2] - 0.5)
    h_anti[n//2:] = 1.0 - h_anti[:n//2]
    starts.append(project_feasible(h_anti, n))

    # Smooth periodic
    x = np.linspace(0, 2, n, endpoint=False)
    h_sin = 0.5 + 0.25*np.sin(2*np.pi*x) + 0.1*np.sin(4*np.pi*x)
    h_sin = 0.5 + 0.3*np.sin(2*np.pi*x)
    starts.append(project_feasible(h_sin, n))

    for i, hs in enumerate(starts):
        if elapsed() > budget_s - 50:
            break
        hs = project_feasible(hs, n)
        try:
            r = run_slsqp(hs, 300, 1e-10)
            if r.fun < best_c5:
                hn = project_feasible(r.x, n)
                c5n = eval_c5(hn)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = hn
        except Exception:
            pass

    # Iterate: re-optimize from best with tighter tol
    for iteration in range(5):
        if elapsed() > budget_s - 60:

        try:
            r = run_slsqp(best_h, 500, 1e-13)
            if r.fun < best_c5:
                hn = project_feasible(r.x, n)
                c5n = eval_c5(hn)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = hn
        except Exception:
            pass
        # Small perturbation to escape plateau
        if elapsed() > budget_s - 30:
            break
        hp = best_h + rng.uniform(-0.01, 0.01, n)
        hp = project_feasible(hp, n)
        try:
            r2 = run_slsqp(hp, 200, 1e-11)
            if r2.fun < best_c5:
                hn = project_feasible(r2.x, n)
                c5n = eval_c5(hn)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = hn
        except Exception:
            pass

    # Try subgradient: explicitly minimize the max-lag
    if elapsed() < budget_s - 40:
        for _ in range(500):
            if elapsed() > budget_s - 30:
                break
            corr = np.correlate(best_h, 1 - best_h, mode="full")
            active = np.where(corr >= np.max(corr) - 1e-10)[0]
            # Compute gradient at active lags (average)
            g = np.zeros(n)
            for k in active:
                lo1, hi1 = k, n-1
                if lo1 <= hi1:
                    cnt = hi1 - lo1 + 1
                    g[lo1:hi1+1] += (1.0 - h0[:cnt]) if False else (1.0 - best_h[:cnt])
                lo2, hi2 = 0, n-1-k
                if lo2 <= hi2:
                    cnt = hi2 - lo2 + 1
                    g[lo2:hi2+1] -= best_h[k:k+cnt]
            g /= max(1, len(active))
            # Gradient step with constraints
            step = 0.01
            hn = best_h - step * g
            hn = project_feasible(hn, n)
            c5n = eval_c5(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn
            else:
                step *= 0.5

    # Final SLSQP polish
    if elapsed() < budget_s - 10:
        try:
            r = run_slsqp(best_h, 500, 1e-14)
            hn = project_feasible(r.x, n)
            c5n = eval_c5(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn
        except Exception:
            pass

    # Verify with provided evaluator if available
    try:
        if 'evaluate_erdos_solution' in globals() and evaluate_erdos_solution is not None:
            c5_verified = evaluate_erdos_solution(best_h)
            if c5_verified < best_c5:
                best_c5 = c5_verified
    except Exception:
        pass

    # Ensure exact feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h)

    return (best_h, best_c5, n)
