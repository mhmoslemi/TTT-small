    h = np.array(h, dtype=float)

    b = (h >= 0.5).astype(float)

        order = np.argsort(-h[zeros_idx])
        for k in range(n_target - n_ones):
            b[zeros_idx[order[k]]] = 1

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

    b = (h >= 0.5).astype(np.float64)
    target = n // 2

        zeros_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zeros_idx])[:(-diff)]
        b[zeros_idx[order]] = 1
    return b

    best_c5 = cur_c5
    best_h = h.copy()

            break

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def make_binary(h, n, rng):
    b = (h >= 0.5).astype(np.float64)
    target = n // 2
    cur = int(b.sum())
    if cur > target:
        idx = np.where(b == 1.0)[0]
        perm = idx[rng.permutation(len(idx))]
        b[perm[:cur - target]] = 0.0
    elif cur < target:
        idx = np.where(b == 0.0)[0]
        perm = idx[rng.permutation(len(idx))]
        b[perm[:target - cur]] = 1.0
    return b


def local_search_sa(h, n, rng, tlimit):
    h = h.astype(np.float64).copy()
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    t0 = time.time()
    T = 0.004
    T_min = 1e-12
    stall = 0

    while time.time() - t0 < tlimit:

        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

            h[i] = 0.0
            h[j] = 1.0
            nc = compute_c5(h, n)
            d = nc - cur
            if d < -1e-14 or (T > T_min and rng.random_sample() < np.exp(-d / T)):
                cur = nc
                if cur < best - 1e-14:

                    best_h = h.copy()
                    stall = 0

                h[j] = 0.0
                stall += 1
        T *= 0.997
        if stall > 400:

                a = ones2[rng.randint(len(ones2))]
                b2 = zeros2[rng.randint(len(zeros2))]
                h[a] = 0.0
                h[b2] = 1.0
            cur = compute_c5(h, n)
            T = max(T, 0.003)
            stall = 0
    return best_h, best


def local_search_best(h, n, rng, tlimit):
