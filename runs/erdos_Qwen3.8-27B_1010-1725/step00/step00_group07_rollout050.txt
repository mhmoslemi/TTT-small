def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, target=None):
    n = len(h)
    if target is None:
        target = n // 2
    h_b = (h >= 0.5).astype(float)
    ones = np.sum(h_b == 1)
    if ones != target:
        if ones > target:
            idx = np.where(h_b == 1)[0]

            amount *= 0.1
            new_j = h[j] + amount
            new_k = h[k] - amount
            if new_j < 0 or new_j > 1 or new_k < 0 or new_k > 1:
                continue
        h[j] = new_j
        h[k] = new_k
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[j] -= amount
            h[k] += amount
        T = max(T * cooling, T_min)
        step = max(step * 0.99995, step_scale * 0.02)
    return best_h, best_c5

def binary_sa(h_init, time_limit, seed):
    rng = np.random.default_rng(seed)
    n = len(h_init)
    target = n // 2
    h = np.round(h_init).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    diff = target - len(ones)
    if diff > 0:
        for i in rng.choice(zeros, size=diff, replace=False):
            h[i] = 1
    elif diff < 0:
        for i in rng.choice(ones, size=-diff, replace=False):
            h[i] = 0
    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    T = 0.02
    T_min = 0.0005
    cooling = 0.9998
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T = max(T * cooling, T_min)
    return best_h, best_c5

def upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        rng = np.random.default_rng(seed)
        h0 = np.zeros(80)
        h0[rng.choice(80, 40, replace=False)] = 1.0
    best_h = constrain(h0.copy(), len(h0))
    best_c5 = evaluate_c5(best_h)
    # Phase 1: binary SA at n=80 (120s)
    rem = budget_s - (time.time() - t0)
    t1 = min(120, rem * 0.12)
    if t1 > 10:
        hb, c5b = binary_sa(best_h, t1, seed)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb
    # Phase 2: continuous SA at n=80 (250s)
    rem = budget_s - (time.time() - t0)
    t2 = min(250, rem * 0.25)
    if t2 > 10:
        h2, c5_2 = continuous_sa(best_h, t2, seed + 1, step_scale=0.08)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    # Phase 3: polish n=80 (100s)
    rem = budget_s - (time.time() - t0)
    t3 = min(100, rem * 0.12)
    if t3 > 10:
        h3, c5_3 = continuous_sa(best_h, t3, seed + 2, step_scale=0.02)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    # Phase 4: upsample to 2n, SA (350s)
    rem = budget_s - (time.time() - t0)
    t4 = min(350, rem * 0.4)
    n = len(best_h)
    if t4 > 30:
        h_up = upsample(best_h, 2)
        h_up = constrain(h_up, 2 * n)
        h4, c5_4 = continuous_sa(h_up, t4, seed + 3, step_scale=0.06)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    # Phase 5: polish upsampled (200s)
    rem = budget_s - (time.time() - t0)
    t5 = min(200, rem * 0.25)
    if t5 > 10:
        h5, c5_5 = continuous_sa(best_h, t5, seed + 4, step_scale=0.02)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5
    # Final constraint enforcement
    n_final = len(best_h)
    best_h = constrain(best_h, n_final)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_final)
