from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def objective(h, n_points, alpha):
    dx = 2.0 / n_points
    c = np.correlate(h, 1 - h, mode='full')
    # Soft max approximation
    c_scaled = alpha * c
    c_max = np.max(c_scaled)
    soft_max = (c_max + np.log(np.sum(np.exp(c_scaled - c_max))) / len(c_scaled)) / alpha
    return soft_max * dx

def objective_grad(h, n_points, alpha):
    # Numerical gradient
    eps = 1e-7
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        grad[i] = (objective(h_plus, n_points, alpha) - objective(h_minus, n_points, alpha)) / (2*eps)
    return grad

    np.random.seed(seed)
