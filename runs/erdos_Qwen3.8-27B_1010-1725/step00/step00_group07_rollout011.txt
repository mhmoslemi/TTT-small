import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    result = np.zeros(n)
    result[idx] = 1.0
    return result

def local_search_binary(h, timeout):
    n = len(h)
    h = h.copy()
    t0 = time.time()
    while time.time() - t0 < timeout:
        ones = np.where(h >= 0.5)[0].tolist()
        zeros = np.where(h < 0.5)[0].tolist()
        if not ones or not zeros:
            break
        current_c5 = compute_c5(h)
        best_impr = 1e-12
        best_i = -1
        best_j = -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c5 = compute_c5(h)
                if current_c5 - c5 > best_impr:
                    best_impr = current_c5 - c5
                    best_i = i
                    best_j = j
                h[j] = 0.0
            h[i] = 1.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
        else:
            break
    return h

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturb_binary(h, n_swaps, seed):
    h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(n_swaps):
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]
    return h

def multi_start(n, t_budget, n_starts):
    best_h = None
    best_c5 = float('inf')
    t0 = time.time()
    per = t_budget / max(1, n_starts)
    for s in range(n_starts):
        if time.time() - t0 >= t_budget:
            break
        h = random_binary(n, 42 + s * 7919 + n)
        h = local_search_binary(h, max(1.0, per - 1.0))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def subgradient_refine(h, t_budget):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.1
    stalled = 0
    while time.time() - t0 < t_budget:
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        idx_max = int(np.argmax(c))
        lag = idx_max - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(lag, n):
                grad[i] += (1.0 - h[i - lag])
            for i in range(0, n - lag):
                grad[i] -= h[i + lag]
        else:
            p = -lag
            for i in range(0, n - p):
                grad[i] += (1.0 - h[i + p])
            for i in range(p, n):
                grad[i] -= h[i - p]
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_new)

        c5_new = compute_c5(h_new)

            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stalled = 0
        else:
            h = h_new.copy()
            lr *= 0.8
            stalled += 1
            if stalled > 10:
                break
    return best_h, best_c5
