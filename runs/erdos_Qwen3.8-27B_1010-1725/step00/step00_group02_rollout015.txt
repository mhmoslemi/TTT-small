        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
                T = max(T * 0.9, T_min)
        T = max(T * cooling, T_min)

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Initial construction
    h_init = np.clip(np.array(initial_h_values, dtype=float), 0, 1)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    target_init = n_init / 2.0
    h_init = fix_sum(h_init, target_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, dx_init)
    best_n = n_init

    # Phase 1: Optimize on initial grid (continuous)
    if time_left() > 200:
        frac = min(0.3, (time_left() - 100) / budget_s)
        h_c, c5_c = local_search_cont(h_init, dx_init, rng, t0, budget_s, frac)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
            best_n = n_init

    # Phase 2: Try 0-1 grids of various sizes
    grid_sizes = [s for s in range(30, 101, 2) if s != n_init]
    # Sort by proximity to initial size for diversity
    grid_sizes.sort(key=lambda x: abs(x - n_init))

    time_per_grid = max(5, (time_left() - 60) / len(grid_sizes))

    for n_try in grid_sizes:
        if time_left() < 30:
            break
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0

        # Initialize by interpolation from best
        x_best = np.linspace(0, 2, best_n, endpoint=False) + (2.0 / best_n) / 2
        x_try = np.linspace(0, 2, n_try, endpoint=False) + dx_try / 2
        h_try = np.interp(x_try, x_best, best_h)
        h_try = make_01(h_try, n_try, rng)

        try_c5 = compute_c5(h_try, dx_try)

        # Also try a few random 0-1 inits
        if time_left() > 40:
            for _ in range(2):
                h_rand = np.zeros(n_try)
                h_rand[rng.choice(n_try, target_try, replace=False)] = 1.0
                rc5 = compute_c5(h_rand, dx_try)
                if rc5 < try_c5:
                    h_try = h_rand
                    try_c5 = rc5

        # Local search
        h_opt, c5_opt = local_search_01(h_try, dx_try, rng, t0, time_left(), 0.95)

        # Also try a continuous refinement on the best 0-1
        if time_left() > 40:
            h_cont = h_opt + rng.normal(0, 0.05, n_try)
            h_cont, c5_cont = local_search_cont(h_cont, dx_try, rng, t0, time_left(), 0.6)
            if c5_cont < c5_opt:
                h_opt = h_cont
                c5_opt = c5_cont

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
        elif c5_opt < best_c5 + 0.001:
            # Close - try further refinement on best
            if time_left() > 60:
                h2 = best_h + rng.normal(0, 0.03, best_n)
                dx_b = 2.0 / best_n
                h2, c5_2 = local_search_01(h2, dx_b, rng, t0, time_left(), 0.5)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2

    # Final verification
    dx_final = 2.0 / best_n
    final_c5 = compute_c5(best_h, dx_final)
    best_h = np.clip(best_h, 0, 1)
    s = best_h.sum()
    if abs(s - best_n / 2.0) > 1e-9:
        best_h = fix_sum(best_h, best_n / 2.0)
        final_c5 = compute_c5(best_h, dx_final)

    return best_h, final_c5, best_n
