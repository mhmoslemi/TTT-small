def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_box_hyperplane(h, S):
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgradient_worst(h, worst_idx):
    n = len(h)
    m = worst_idx - n + 1
    g = np.zeros(n)
    for t in range(n):
        im = t - m
        ip = t + m
        val = 0.0
        if 0 <= im < n:
            val += 1.0 - h[im]
        if 0 <= ip < n:
            val -= h[ip]
        g[t] = val
    return g

def poly_subgrad(h, time_limit):
