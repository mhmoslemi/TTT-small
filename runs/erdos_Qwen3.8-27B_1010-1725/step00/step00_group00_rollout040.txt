    return float(np.max(r) * dx)

    """Fix h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

            h = np.clip(h, 0.0, 1.0)
