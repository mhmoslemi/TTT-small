                    grad[:n - kk] -= h[kk:]
        grad /= 4.0
        h = _repair(h - step * grad, n)
        step *= 0.9997
        if step < 0.01 / n:
            step = 0.01 / n
    return best, bc

def _optim_multi(h, n, deadline, rng, n_restarts=3):
    best = h.copy()
    bc = _eval_c5(best)
    per_alloc = (deadline - time.time()) / n_restarts
    for r in range(n_restarts):
        if r == 0:
            hc = h.copy()
        elif r == 1:
            hc = _repair(h + 0.2 * (80.0 / n) * rng.standard_normal(n), n)
        else:
            hc = _repair(0.7 * h + 0.3 * rng.random(n), n)
            hc = np.clip(hc, 0, 1)
            hc = _repair(hc, n)
        dl_r = time.time() + per_alloc
        h_r, c_r = _optim(hc, n, dl_r, rng)
        if c_r < bc:
            bc, best = c_r, h_r
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    try:
        p = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        p = np.full(80, 0.5)

    bh = p.copy()
    bc = _eval_c5(bh)
    n0 = len(p)

    # Phase 1: optimize at parent resolution with multiple restarts
    if time.time() < deadline - 80:
        bh, bc = _optim_multi(p, n0, deadline - 80, rng, n_restarts=3)

    # Phase 2: increase resolution progressively
    for nr in [128, 160, 200, 256, 320, 480, 640, 800]:
        tl = deadline - time.time()
        if tl < 30:
            break
        n_restarts = 2 if tl > 200 else 1
        alloc = min(tl * 0.45, 180.0)
        hc = _repair(_interp2(bh, len(bh), nr), nr)
        h_r, c_r = _optim_multi(hc, nr, time.time() + alloc, rng, n_restarts=n_restarts)
        if c_r < bc:
            bh, bc = h_r, c_r

    # Phase 3: final polish
    tl = deadline - time.time()
    if tl > 20:
        h_r, c_r = _optim(bh.copy(), len(bh), time.time() + tl * 0.6, rng)
        if c_r < bc:
            bh, bc = h_r, c_r

    return bh, bc, len(bh)
