import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
            if not m.any():
                break
            s = min(diff / m.sum(), (1.0 - h[m]).min())
            h[m] += s
        else:
            m = h > 0.0
            if not m.any():
                break
            s = min(-diff / m.sum(), h[m].min())
            h[m] -= s
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, rng, time_limit, best_c5):
    t0 = time.time()
    h = h.copy()
    for step in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
        improved = True
        while improved and (time.time() - t0) < time_limit:
            improved = False
            idx = list(range(n))
            rng.shuffle(idx)
            for i in idx:
                if (time.time() - t0) >= time_limit:
                    break
                hi = h[i]
                for j in idx:
                    if i == j:
                        continue
                    hj = h[j]
                    # Try +step on i, -step on j
                    if hi + step <= 1.0 + 1e-12 and hj - step >= -1e-12:
                        h[i] = hi + step
                        h[j] = hj - step
                        c = compute_c5(h)
                        if c < best_c5 - 1e-15:
                            best_c5 = c
                            improved = True
                        else:
                            h[i] = hi
                            h[j] = hj
                    # Try -step on i, +step on j
                    if hi - step >= -1e-12 and hj + step <= 1.0 + 1e-12:
                        h[i] = hi - step
                        h[j] = hj + step
                        c = compute_c5(h)
                        if c < best_c5 - 1e-15:
                            best_c5 = c
                            improved = True
                        else:
                            h[i] = hi
                            h[j] = hj
    return h, best_c5

def make_binary_start(n, rng, seed_offset=0):
    target = n / 2.0
    h = np.zeros(n)
    fr = target - int(target)  # fractional part, 0.5 for odd n
    idx = list(range(n))
    rng.shuffle(idx)
    for k in range(int(target)):
        h[idx[k]] = 1.0
    if fr > 1e-10:
        h[idx[int(target)]] = fr
    return h

def slsqp_optimize(h, n):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        res = minimize(compute_c5, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-13})
        h_opt = project_feasible(np.clip(res.x, 0, 1), n)
        return h_opt
    except Exception:
        return h

def cobyla_optimize(h, n):
    try:
        cons = lambda x: x.sum() - n / 2.0
        res = minimize(compute_c5, h.copy(), method='COBYLA',
                       constraints=[{'type': 'eq', 'fun': cons},
                                    {'type': 'ineq', 'fun': lambda x: x.min()},
                                    {'type': 'ineq', 'fun': lambda x: (1.0 - x).min()}],
                       options={'maxiter': 2000, 'rhobeg': 0.1})
        h_opt = project_feasible(np.clip(res.x, 0, 1), n)
        return h_opt
    except Exception:
        return h

    t_start = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = None

    def rem():
        return budget_s - (time.time() - t_start)

    # Initialize from parent

    best_c5 = compute_c5(best_h)

    if rem() > 40:
        h_s = slsqp_optimize(best_h, n0)

            best_h = h_s

    # Phase 2: Local search at parent

    # Phase 3: Binary rounding + local search (multiple random starts)
    if rem() > 80:
        for trial in range(20):
            if rem() < 40:

            # Perturb best_h then round
            h_pert = best_h + rng.normal(0, 0.15, n0)
            h_pert = np.clip(h_pert, 0, 1)
            h_bin = np.where(h_pert > 0.5, 1.0, 0.0).copy()

    # Ensure exact feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
