def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()

    if diff > 0:
        mask = h < 1.0
    else:

def solve_1d_minimax(A, B, s_lo, s_hi):
    """Minimize max_i (A[i] + B[i]*s) over s in [s_lo, s_hi]"""
    if len(A) == 0:
        return 0.0
    # Evaluate at endpoints
    val_lo = np.max(A + B * s_lo)
    val_hi = np.max(A + B * s_hi)
    best_s = s_lo if val_lo <= val_hi else s_hi
    best_val = min(val_lo, val_hi)
    
    # Check all pairwise intersections
    n = len(A)
    for i in range(n):
        for j in range(i + 1, n):
            db = B[i] - B[j]
            if abs(db) < 1e-15:
                continue
            s_cross = (A[j] - A[i]) / db
            if s_lo < s_cross < s_hi:
                val = np.max(A + B * s_cross)
                if val < best_val:
                    best_val = val
                    best_s = s_cross
    return best_s

            no_improve += 1
        if no_improve > 200:
            break

    return best_h, best_c5


def make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        h[perm[n // 2]] = 1.0
        for i in range(n // 2):
            h[perm[i]] = 0.5
    return project_feasible(h, n / 2.0)


def make_block_start(n, seed):
    rng = np.random.RandomState(seed)
    nblocks = rng.randint(3, min(9, n // 4 + 1))
    blocks = n // nblocks
    h = np.zeros(n)
    pos = 0
    for b in range(nblocks):
        length = blocks if b < nblocks - 1 else n - pos
        if rng.random() < 0.5:
            h[pos:pos + length] = 1.0
        pos += length
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_left():
        return budget_s - (time.time() - t0) - 5.0

    # Try parent construction first
    h_parent = np.array(initial_h_values, dtype=float)
    n_p = len(h_parent)
    h_p = project_feasible(h_parent, n_p / 2.0)
    limit = min(time_left(), 250)
    h_opt, c5 = coordinate_descent(h_p, 8000, limit, t0)
    best_c5, best_h, best_n = c5, h_opt, n_p

    # Upsample and optimize at higher resolution
    if time_left() > 120:
        for n_up in [90, 180]:
            if time_left() < 60:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_up / 2.0)
            limit = min(time_left() * 0.45, 200)
            h_opt2, c5_2 = coordinate_descent(h_up, 8000, limit, t0)
            if c5_2 < best_c5:
                best_c5, best_h, best_n = c5_2, h_opt2, n_up

    # Try diverse starting points
    trial = 0
    while time_left() > 25:
        for n_try in [45, 90]:
            if time_left() < 20:
                break
            s = seed * 31 + trial * 97 + n_try
            if trial % 4 == 0:
                h_init = make_binary_start(n_try, s)
            elif trial % 4 == 1:
                h_init = make_block_start(n_try, s)
            elif trial % 4 == 2:
                h_init = make_binary_start(n_try, s + 7)
            else:
                x = np.linspace(0, 2, n_try)
                freq = 0.5 + (s % 10) * 0.3
                h_init = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x + s * 0.1)
                h_init = np.clip(h_init, 0, 1)
                h_init = project_feasible(h_init, n_try / 2.0)
            limit = min(time_left() * 0.35, 150)
            h_opt3, c5_3 = coordinate_descent(h_init, 6000, limit, t0)
            if c5_3 < best_c5:
                best_c5, best_h, best_n = c5_3, h_opt3, n_try
        trial += 1

    # Final polish on best solution
    if time_left() > 15:
        limit = time_left() - 5
        h_pol, c5_pol = coordinate_descent(best_h.copy(), 10000, limit, t0)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
