    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            return h
        if s > target_sum:
            mask = h > 0
            if not mask.any():
                return h
            h[mask] -= (s - target_sum) / mask.sum()
        else:
            mask = h < 1
            if not mask.any():
                return h
            h[mask] += (target_sum - s) / mask.sum()
    return np.clip(h, 0.0, 1.0)

def coord_descent_step(h, i, dx, target_sum, n, rng):
    """Try to optimize coordinate i with a paired update."""
    c5_before = compute_c5(h, dx)
    
    # Generate candidate values for h[i]
    old_val = h[i]
    candidates = np.unique(np.clip(
        np.concatenate([[old_val], 
                       rng.uniform(0, 1, 15),
                       [0, 1, 0.5],
                       old_val + rng.normal(0, 0.2, 10)]),
        0, 1))
    
    best_c5 = c5_before
    best_h = h.copy()
    
    for new_val in candidates:
        delta = new_val - old_val
        if abs(delta) < 1e-10:
            continue
        # Find a good j to compensate
        # Try a few j's
        for j in rng.choice(np.delete(np.arange(n), i), size=min(8, n-1), replace=False):
            comp = h[j] - delta
            if comp < -1e-9 or comp > 1 + 1e-9:
                continue
            h_test = h.copy()
            h_test[i] = new_val
            h_test[j] = comp
            s = np.sum(h_test)
            if abs(s - target_sum) > 1e-8:
                continue
            c5_test = compute_c5(h_test, dx)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to {0<=h<=1, sum(h)=target} via iterative water-filling."""
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            return h
        if s > target:
            m = h > 1e-15
            if not m.any():
                return h
            h[m] -= (s - target) / m.sum()
        else:
            m = h < 1.0 - 1e-15
            if not m.any():
                return h
            h[m] += (target - s) / m.sum()
    return np.clip(h, 0.0, 1.0)

def paired_step(h, i, j, dx):
    """Optimize paired coords (i,j) preserving sum. Returns (h, new_c5)."""
    s = h[i] + h[j]
    v_lo = max(0.0, s - 1.0)
    v_hi = min(1.0, s)
    if v_lo >= v_hi - 1e-12:
        return h, compute_c5(h, dx)
    # Coarse scan
    v_coarse = np.linspace(v_lo, v_hi, 61)
    best_v = h[i]
    best_c5 = float('inf')
    for v in v_coarse:
        h[i] = v
        h[j] = s - v
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_v = v
    # Fine scan around best
    margin = (v_hi - v_lo) / 60.0
    v_lo2 = max(v_lo, best_v - margin)
    v_hi2 = min(v_hi, best_v + margin)
    v_fine = np.linspace(v_lo2, v_hi2, 101)
    for v in v_fine:
        h[i] = v
        h[j] = s - v
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_v = v
    h[i] = best_v
    h[j] = s - best_v
    return h, best_c5

def optimize_from(h, n, dx, target, rng, deadline):
    """Local optimization via random paired coordinate descent."""
    c5 = compute_c5(h, dx)
    stall = 0
    best_local_c5 = c5
    best_local_h = h.copy()
    for it in range(20000):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if j == i:
            j = (j + 1 + rng.integers(n - 1)) % n
        h, c5 = paired_step(h, i, j, dx)
        if c5 < best_local_c5 - 1e-13:
            best_local_c5 = c5
            best_local_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Random perturbation to escape local optimum
                pp = rng.choice(n, size=3, replace=False)
                h[pp] += rng.uniform(-0.4, 0.4, 3)
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
                stall = 0
            elif stall > 150:
                # Smaller perturbation
                p = rng.integers(n)
                h[p] += rng.uniform(-0.15, 0.15)
                h = project_sum(h, target)
                c5 = compute_c5(h, dx)
    return best_local_h, best_local_c5

def make_initial(n, n_ref, h_ref, rng, mode):
    """Create initial h of size n."""
    if mode == 0:
        # Interpolate from reference
        xo = np.linspace(0, 2, n_ref, endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        return np.interp(xn, xo, h_ref)
    elif mode == 1:
        # Smooth random
        freq = rng.integers(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n, endpoint=False)
        return np.clip(0.5 + 0.3 * np.cos(np.pi * freq * x + phase), 0, 1)
    elif mode == 2:
        # Block structure
        nblocks = rng.integers(2, 6)
        h = np.zeros(n)
        for b in range(nblocks):
            start = int(b * n / nblocks)
            end = int((b + 1) * n / nblocks)
            h[start:end] = rng.uniform(0.2, 0.85)
        return h
    else:
        return rng.uniform(0, 1, n)

    rng = np.random.default_rng(seed)
    t0 = time.time()

    # Get initial construction
    h_init = np.asarray(initial_h_values, dtype=float)
    n_init = len(h_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, 2.0 / n_init)
    best_n = n_init
    
    # Sizes to try (even n for clean constraint)
    n_opts = sorted(set([n_init] + [n for n in range(30, 201, 5) if n % 2 == 0 and n != n_init]),
                    key=lambda x: abs(x - n_init))[:12]
    
    time_budget = budget_s - 40
    time_per_n = time_budget / len(n_opts)
    
    for n in n_opts:
        if time.time() > deadline_global:
            break
        dx = 2.0 / n
        target = n / 2.0
        n_deadline = min(time.time() + time_per_n * 2.0, deadline_global - 20)
        
        # Try multiple initializations
        n_starts = 3 if n == n_init else 2
        for start_idx in range(n_starts):
            if time.time() > n_deadline:
                break
            if n == n_init and start_idx == 0:
                h = h_init.copy()
            else:
                mode = start_idx if n != n_init else 1
                h = make_initial(n, n_init, h_init, rng, mode)
            h = project_sum(h, target)
            
            # Local optimization
            h_opt, c5_opt = optimize_from(h, n, dx, target, rng, n_deadline)
            
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    # Final polish on best solution
    if time.time() < deadline_global - 10:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h = best_h.copy()
        h, c5 = optimize_from(h, n, dx, target, rng, deadline_global - 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    
    return best_h, final_c5, best_n
