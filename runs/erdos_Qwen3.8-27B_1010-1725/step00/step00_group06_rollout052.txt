        idx2 = i - shift

        s2 = h[idx2] if 0 <= idx2 < n else 0.0

    # Use bisection on the derivative (subgradient)
    def upper_envelope(v):
        return np.max(intercepts + v * slopes)
    
    # Find minimum using golden section or bisection
    # The function is convex, so binary search on subgradient
    a, b = 0.0, 1.0
    for _ in range(60):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        if upper_envelope(m1) <= upper_envelope(m2):
            b = m2
        else:
            a = m1
    
    return (a + b) / 2

import numpy as np
from scipy.optimize import minimize
import time

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    for _ in range(300):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
        diff = target - h.sum()
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def coord_descent(h, n, max_sweeps=150, time_limit=None, t_start=None):
    h = fix_sum(h.copy(), n)
    c_curr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(c_curr) * (2.0 / n))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    shifts = np.arange(-(n - 1), n)
    idx_range = np.arange(n)

    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        order = idx_range if sweep % 2 == 0 else idx_range[::-1]
        for i in order:
            old_val = h[i]
            idx1 = i + shifts
            idx2 = i - shifts
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            v1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
            v2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
            slopes = (1.0 - v1) - v2
            intercepts = c_curr - old_val * slopes

            a, b = 0.0, 1.0
            for _ in range(32):
                m1 = a + (b - a) * (1.0 - gr)
                m2 = a + (b - a) * gr
                if np.max(intercepts + m1 * slopes) <= np.max(intercepts + m2 * slopes):
                    b = m2
                else:
                    a = m1
                if b - a < 1e-13:
                    break
            new_val = 0.5 * (a + b)
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                c_curr += delta * slopes
                h[i] = new_val

        h = fix_sum(h, n)
        c_curr = np.correlate(h, 1.0 - h, mode='full').copy()
        c5 = float(np.max(c_curr) * (2.0 / n))
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx * alpha
    vals = np.clip(vals, -400.0, 400.0)
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / alpha
