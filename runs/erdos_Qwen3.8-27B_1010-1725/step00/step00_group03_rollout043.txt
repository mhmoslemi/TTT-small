import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        m = h < 1.0
        av = (1.0 - h[m]).sum()
        if av > 1e-14:
            h[m] += diff * (1.0 - h[m]) / av
    else:
        m = h > 0.0
        av = h[m].sum()
        if av > 1e-14:
            h[m] += diff * h[m] / av
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment for exact sum
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        m = (h > 0) & (h < 1)
        if m.sum() > 0:
            h[m] += diff2 / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, n, T, pen):
    dx = 2.0 / n
    hw = 1.0 - h
    corr = np.correlate(h, hw, mode="full")
    mx = corr.max()
    e = np.exp(T * (corr - mx))
    es = e.sum()
    w = e / es
    val = (mx + np.log(es) / T) * dx + pen * (h.sum() - 0.5 * n) ** 2
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        for p in range(n):
            i1 = p + m - n + 1
            if 0 <= i1 < n:
                g[p] += wm * hw[i1]
            i2 = p - m + n - 1
            if 0 <= i2 < n:
                g[p] -= wm * h[i2]
    g *= dx
    g += 2.0 * pen * (h.sum() - 0.5 * n)
    return val, g

        return project(r.x, n / 2.0)
    except Exception:
        return h0

    h = h.copy()

        corr = np.correlate(h, hw, mode="full")
        k = int(np.argmax(corr))
        g = np.zeros(n)
        for p in range(n):
            i1 = p + k - n + 1
            if 0 <= i1 < n:
                g[p] += hw[i1]
            i2 = p - k + n - 1
            if 0 <= i2 < n:
                g[p] -= h[i2]
        h -= lr * g * dx
        h = project(h, n / 2.0)
    return h
