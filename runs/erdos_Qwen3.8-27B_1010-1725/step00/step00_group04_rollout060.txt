    return float(np.max(corr) * dx)

        cnt = float(np.sum(mask))

    np.clip(proj, 0.0, 1.0, out=proj)
    return proj

    n = len(h)
    dx = 2.0 / n

def project_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    lam = (lo + hi) / 2.0
