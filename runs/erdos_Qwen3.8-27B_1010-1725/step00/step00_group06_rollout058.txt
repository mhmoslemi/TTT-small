import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = int(np.argmax(vals))
    k = m - (n - 1)
    grad = np.zeros(n)
    jj = np.arange(n)
    jk = jj - k
    m1 = (jk >= 0) & (jk < n)
    grad[m1] += (1.0 - h[jk[m1]])
    jk2 = jj + k
    m2 = (jk2 >= 0) & (jk2 < n)
    grad[m2] += (-h[jk2[m2]])
    return vals[m], grad * dx

def smooth_c5_grad(h, tau=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    v_max = vals.max()
    shifted = tau * (vals - v_max)
    log_sum = v_max + np.log(np.sum(np.exp(shifted))) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    jj = np.arange(n)
    for idx in range(len(vals)):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        jk = jj - k
        m1 = (jk >= 0) & (jk < n)
        grad[m1] += w * (1.0 - h[jk[m1]])
        jk2 = jj + k
        m2 = (jk2 >= 0) & (jk2 < n)
        grad[m2] += w * (-h[jk2[m2]])
    return log_sum, grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = np.full(95, 0.5)

    n = len(h0)
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - ts, 'jac': lambda h: np.ones(n)}]

    def fprime(h):
        v, g = c5_grad(h)
        return v, g

    def fprime_smooth(h):
        v, g = smooth_c5_grad(h, tau=30.0)
        return v, g

    # Phase 1: SLSQP with hard max from initial
    if rem() > 120:
        try:
            res = minimize(fprime, h0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13})
            if res.fun < best_c5 - 1e-15:
                best_h = project_feasible(res.x, ts)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    # Phase 2: SLSQP with smooth max from best
    if rem() > 120:
        try:
            res = minimize(fprime_smooth, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13})
            h_try = project_feasible(res.x, ts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
        except Exception:
            pass

    # Phase 3: Multi-start perturbations
    scales = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4]
    for i in range(200):
        if rem() < 40:
            break
        s = scales[i % len(scales)]
        hp = best_h + s * rng.randn(n)
        hp = project_feasible(hp, ts)
        use_smooth = (i % 3 == 0)
        try:
            if use_smooth:
                res = minimize(fprime_smooth, hp, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
            else:
                res = minimize(fprime, hp, method='SLSQP', bounds=bounds,

                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
        except Exception:
            pass

    if rem() > 15:
        for _round in range(50):
            if rem() < 10:

            improved = False

                for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:

                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, ts)

                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break

    # Verify and return
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
