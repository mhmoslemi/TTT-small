                best_h[i] = 0.0
                best_h[j] = 1.0
                v = compute_c5(best_h)
                if v < best_swap_val:
                    best_swap_val = v
                    best_i, best_j = i, j
                best_h[i] = 1.0
                best_h[j] = 0.0
        if best_i >= 0:
            best_h[best_i] = 0.0
            best_h[best_j] = 1.0
            best_val = best_swap_val
        else:
            break
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0.0, n_src - 1.0, n_dst)
    return np.interp(x_dst, x_src, h_src)

def random_binary_init(n, rng):
    h = np.zeros(n)
    perm = rng.permutation(n)
    if n % 2 == 0:
        h[perm[:n // 2]] = 1.0
    else:
        h[perm[:n // 2]] = 1.0
        h[perm[n // 2]] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    t_deadline = t_start + budget_s - 10.0

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h47 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h47 = np.array(initial_h_values, dtype=float).copy()
        h47 = project_feasible(h47, len(h47))

    n_list = [47, 48, 50, 52, 60, 70, 80, 100]
    results = {}

    # Pass 1: gradient optimization for each n
    for n in n_list:
        if time.time() > t_deadline:
            break
        time_left = t_deadline - time.time()
        time_alloc = time_left / max(1, len(n_list) - n_list.index(n) + 1)
        t_dead = min(t_deadline, time.time() + time_alloc)

        if n == 47 and h47 is not None:
            h_init = h47.copy()
        elif h47 is not None:
            h_init = interpolate_to_n(h47, n)
            h_init = project_feasible(h_init, n)
        else:
            h_init = random_binary_init(n, rng)

        h_opt, val_opt = optimize_gradient(h_init, n, 1500, t_dead)
        if val_opt < best_c5:
            best_c5, best_h, best_n = val_opt, h_opt.copy(), n
        results[n] = (h_opt.copy(), val_opt)

    # Pass 2: binary local search for even n + refinement
    for n in sorted([k for k in results if k % 2 == 0]):
        if time.time() > t_deadline:
            break
        time_left = t_deadline - time.time()
        time_alloc = time_left / max(1, len([k for k in results if k % 2 == 0]))
        t_dead = min(t_deadline, time.time() + time_alloc)

        h_curr, _ = results[n]
        h_bin = snap_to_binary(h_curr, n)
        if h_bin is not None:
            h_bl, val_bl = binary_local_search(h_bin, n, 300, t_dead)
            if h_bl is not None and val_bl < best_c5:
                best_c5, best_h, best_n = val_bl, h_bl.copy(), n
                results[n] = (h_bl.copy(), val_bl)

    # Pass 3: refine top solutions with more iterations
    top_ns = sorted(results.keys(), key=lambda k: results[k][1])[:4]
    for n in top_ns:
        if time.time() > t_deadline:
            break
        time_left = t_deadline - time.time()
        time_alloc = time_left / max(1, len(top_ns))
        t_dead = min(t_deadline, time.time() + time_alloc)

        h_curr, _ = results[n]
        h_opt, val_opt = optimize_gradient(h_curr, n, 2000, t_dead)
        if val_opt < best_c5:
            best_c5, best_h, best_n = val_opt, h_opt.copy(), n
            results[n] = (h_opt.copy(), val_opt)

        if n % 2 == 0 and time.time() < t_dead:
            h_bin = snap_to_binary(h_opt, n)
            if h_bin is not None:
                h_bl, val_bl = binary_local_search(h_bin, n, 200, t_dead)
                if h_bl is not None and val_bl < best_c5:
                    best_c5, best_h, best_n = val_bl, h_bl.copy(), n
                    results[n] = (h_bl.copy(), val_bl)

    # Pass 4: random restarts on best n
    if time.time() < t_deadline and best_n > 0:
        n_best = best_n
        h_rand = random_binary_init(n_best, rng)
        t_dead = t_deadline
        h_opt, val_opt = optimize_gradient(h_rand, n_best, 1000, t_dead)
        if val_opt < best_c5:
            best_c5, best_h, best_n = val_opt, h_opt.copy(), n_best
        if n_best % 2 == 0:
            h_bin = snap_to_binary(h_opt, n_best)
            if h_bin is not None:
                h_bl, val_bl = binary_local_search(h_bin, n_best, 200, t_dead)
                if h_bl is not None and val_bl < best_c5:
                    best_c5, best_h, best_n = val_bl, h_bl.copy(), n_best

    # Ensure feasibility
    if best_h is not None:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 48
        best_h = random_binary_init(best_n, rng)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
