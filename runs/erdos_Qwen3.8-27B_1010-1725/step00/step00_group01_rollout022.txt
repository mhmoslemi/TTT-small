import numpy as np
import time
from scipy.optimize import minimize

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def smooth_obj_and_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(tau * (corr - m))
    w = exps / np.sum(exps)
    obj = m + np.log(np.sum(exps)) / tau
    grad_h = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi < 1e-8:
            continue
        if s >= 0:
            nterms = n - s
            idx = np.arange(nterms)
            grad_h[idx] += wi * (1.0 - h[idx + s])
            grad_h[idx + s] -= wi * h[idx]
        else:
            sa = -s
            idx = np.arange(sa, n)
            grad_h[idx] += wi * (1.0 - h[idx - sa])
            grad_h[idx - sa] -= wi * h[idx]
    return obj, dx * grad_h

def project_feasible(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def interp_h(h_orig, n_new):
    n_orig = len(h_orig)
    if n_orig == n_new:
        return h_orig.copy()
    x_orig = np.arange(n_orig, dtype=np.float64) / n_orig
    x_new = np.arange(n_new, dtype=np.float64) / n_new
    return np.interp(x_new, x_orig, h_orig)

def subgradient_opt(h, n, max_iter, tau_start, time_limit):
    h = h.copy()
    best = c5_value(h)
    best_h = h.copy()
    t_start = time.time()
    lr = 0.008
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        tau = tau_start + it * 0.5
        _, grad = smooth_obj_and_grad(h, n, tau)
        h = project_feasible(h - lr * grad, n)
        lr *= 0.9985
        obj = c5_value(h)
        if obj < best - 1e-12:
            best = obj
            best_h = h.copy()
    return best_h, best

def local_search(h, n, max_iter, rng, time_limit):
    h = h.copy()
    best = c5_value(h)
    t_start = time.time()
    eps_list = [0.01, 0.02, 0.05, 0.1, 0.15, 0.25]
    stuck_count = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        improved = False
        n_pairs = 80
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in eps_list:
                for sign in (1, -1):
                    hi_new = h[i] - sign * d
                    hj_new = h[j] + sign * d
                    if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
                        continue
                    h[i] = hi_new
                    h[j] = hj_new
                    c2 = c5_value(h)
                    if c2 < best - 1e-9:
                        best = c2
                        improved = True
                    else:
                        h[i] = hi_new if not improved else h[i]
                        h[j] = hj_new if not improved else h[j]
                        if d > 0.05:
                            h[i] = h[i]  # already restored above
                        else:
                            h[i] = h[i]
                        h[i] = hi_new
                        h[j] = hj_new
                        h[i] = h[i]  # keep as-is, will try next
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stuck_count += 1
            if stuck_count >= 5:
                # Perturb and reset
                idx = rng.choice(n, size=n // 3, replace=False)
                h[idx] = np.clip(h[idx] + rng.randn(len(idx)) * 0.05, 0, 1)
                h = project_feasible(h, n)
                c2 = c5_value(h)
                if c2 < best:
                    best = c2
                stuck_count = 0
                if time.time() - t_start > time_limit:
                    break
        else:
            stuck_count = 0
    return h, best

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = c5_value(best_h)
