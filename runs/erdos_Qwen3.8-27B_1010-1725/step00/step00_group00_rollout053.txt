For k = n-1-j, this is sum_{valid i} h[i] * (1-h[i+j]), which corresponds to shift -j.

The max over all these gives C₅.

                break
            if j > 0:
                h_init = make_feasible(h_init + np.random.randn(n_try) * 0.04, n_try)
            res = do_slsqp(h_init, n_try, maxiter=300, ftol=1e-13)
            if res.x is not None:
                x_new = make_feasible(res.x, n_try)
                c5_new = compute_c5(x_new, n_try)
                if c5_new < best_c5:
                    best_c5, best_h, cur_n = c5_new, x_new, n_try

    # Final polish
    if elapsed() < budget_s * 0.98:
        bnds = [(0.0, 1.0)] * cur_n
        cns = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
        res = minimize(compute_c5, best_h, args=(cur_n,), method='SLSQP',
                       bounds=bnds, constraints=cns,
                       options={'maxiter': 500, 'ftol': 1e-15})
        if res.x is not None:
            x_new = make_feasible(res.x, cur_n)
            c5_new = compute_c5(x_new, cur_n)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, x_new

    return best_h, best_c5, len(best_h)
