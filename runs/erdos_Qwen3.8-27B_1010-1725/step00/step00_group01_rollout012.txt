import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def compute_grad_c5(h):
    """Compute subgradient of C5 = max_k corr(k)*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    k_max = np.argmax(corr)
    shift = k_max - (n - 1)
    
    grad = np.zeros(n)
    # For each j, compute partial derivative of corr(shift) w.r.t. h[j]
    # corr(shift) = sum_i h[i+shift] * (1-h[i])
    # partial w.r.t. h[j]:
    #   if j = i+shift for some valid i (i=j-shift): contributes (1-h[j-shift])
    #   if j = i for some valid i where i+shift valid (i+shift = j+shift): contributes -h[j+shift]
    
    for j in range(n):
        g = 0.0
        i1 = j - shift
        if 0 <= i1 < n:
            g += (1.0 - h[i1])
        i2 = j + shift
        if 0 <= i2 < n:
            g -= h[i2]
        grad[j] = g * dx
    
    return grad, np.max(corr) * dx

def project_box_hyperplane(h, n, tol=1e-12):
    """Project onto 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > tol:
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        cnt = free.sum()
        if cnt > 0:
            adj = diff / cnt
            h[free] += adj
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_step(h, n, lr):
    """One subgradient descent step."""
    grad, c5 = compute_grad_c5(h)
    # Remove mean to preserve sum constraint approximately
    grad = grad - grad.mean()
    h_new = h - lr * grad
    h_new = project_box_hyperplane(h_new, n)
    return h_new, c5

def optimize_from(h_init, n, time_limit, rng):
    """Optimize starting from h_init."""
    h = h_init.copy().astype(float)
    h = project_box_hyperplane(h, n)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.1
    patience = 500
    stuck = 0
    it = 0
    while time.time() - start < time_limit:
        h_new, _ = subgrad_step(h, n, lr)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
            lr = min(lr * 1.05, 1.0)
        else:

            if stuck > patience:

                    idx = rng.choice(n, size=max(1, n//10), replace=False)
                    h[idx] += rng.uniform(-0.2, 0.2, size=len(idx))
                    h = project_box_hyperplane(h, n)
                    lr = 0.1
                    stuck = 0
        h = h_new
        it += 1
    return best_h, best_c5

    h = np.zeros(n)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project to feasible set: 0<=h<=1, sum(n/2)"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return h

def local_search(h, n, max_time=30.0, rng=None):
    """Local search via bit swaps."""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    t_start = time.time()
    dx = 2.0 / n
    
    while time.time() - t_start < max_time:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-14
        best_pair = None
        old_corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = new_corr.max() * dx
                h[i], h[j] = h[j], h[i]
                delta = best_c5 - new_c5
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        best_c5 = best_c5 - best_delta
    
    return h, best_c5

def subgrad_optimize(h, n, max_time=30.0, rng=None):
    """Subgradient descent for continuous h."""
    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    t_start = time.time()
    dx = 2.0 / n
    lr = 0.5
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.time() - t_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        c5 = corr[k_max] * dx
        
        # Subgradient
        shift = k_max - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j - shift < n:
                g += (1.0 - h[j - shift])
            if 0 <= j + shift < n:
                g -= h[j + shift]
            grad[j] = g * dx
        
        # Mean-center to preserve sum
        grad -= grad.mean()
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - lr * momentum
        h_new = project_h(h_new, n)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            lr = min(lr * 1.01, 2.0)
        else:
            lr = max(lr * 0.95, 0.01)
        
        h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def local_search(h, n, max_time=30.0, rng=None):
    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    t_start = time.time()
    stall = 0
    while time.time() - t_start < max_time:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-12
        best_pair = None
        corr_all = np.correlate(h, 1.0 - h, mode='full') * dx
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                nc = np.correlate(h, 1.0 - h, mode='full').max() * dx
                h[i], h[j] = h[j], h[i]
                d = best_c5 - nc
                if d > best_delta:
                    best_delta = d
                    best_pair = (i, j)
        if best_pair is None:
            stall += 1
            if stall > 3 and rng is not None:
                idx1 = rng.choice(ones, size=min(2, len(ones)))
                idx2 = rng.choice(zeros, size=min(2, len(zeros)))
                h[idx1], h[idx2] = h[idx2].copy(), h[idx1].copy()
                best_c5 = evaluate_c5(h)
                h = project_h(h, n)
                stall = 0
            else:
                break
        else:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            best_c5 = best_c5 - best_delta
            stall = 0
    return h, best_c5

def subgrad_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    thresh = corr.max() - 0.5
    active = np.where(corr >= thresh)[0]
    c5 = corr.max() * dx
    grad = np.zeros(n)
    for k_idx in active:
        shift = k_idx - (n - 1)
        for j in range(n):
            if 0 <= j - shift < n:
                grad[j] += (1.0 - h[j - shift])
            if 0 <= j + shift < n:
                grad[j] -= h[j + shift]
    grad *= dx / len(active)
    grad -= grad.mean()
    return grad, c5

def subgrad_optimize(h, n, max_time=30.0, rng=None):
    h = h.copy().astype(float)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    t_start = time.time()
    lr = 0.3
    mom = np.zeros(n)
    beta = 0.85
    no_improve = 0
    while time.time() - t_start < max_time:
        grad, c5 = subgrad_step(h, n)
        mom = beta * mom + (1 - beta) * grad
        h_new = h - lr * mom
        h_new = project_h(h_new, n)
        nc5 = evaluate_c5(h_new)
        if nc5 < best_c5 - 1e-12:
            best_c5 = nc5
            best_h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 1.5)
        else:
            no_improve += 1
            lr = max(lr * 0.97, 0.05)
            if no_improve > 200 and rng is not None:
                h = best_h + rng.normal(0, 0.05, n)
                h = project_h(h, n)
                lr = 0.3
                no_improve = 0
        h = h_new
    return best_h, best_c5
