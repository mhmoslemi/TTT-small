import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = int(mask.sum())
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_worst_shift(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    idx = np.arange(n)
    ip = idx + k
    im = idx - k
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    grad = np.zeros(n)
    grad[mask_p] += (1.0 - h[ip[mask_p]])
    grad[mask_m] -= h[im[mask_m]]
    return grad * dx, float(corr[max_idx] * dx)

def subgrad_descent(h, n_iter=50, lr=0.08):
    n = len(h)
    h_cur = h.copy()
    for i in range(n_iter):
        decay = 1.0 - 0.5 * (i / max(n_iter - 1, 1))
        grad, _ = grad_worst_shift(h_cur)
        h_cur = h_cur - lr * decay * grad
        h_cur = make_feasible(h_cur, n)
    return h_cur

def anneal(h, rng, n_iter=500, t_start=0.05, t_end=0.004):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for i in range(n_iter):
        t = t_start * (t_end / t_start) ** (i / max(n_iter - 1, 1))
        n_p = int(rng.integers(1, max(2, n // 6)))
        idx = rng.choice(n, size=n_p, replace=False)
        trial = cur.copy()
        trial[idx] += rng.normal(0, t, size=n_p)
        trial = make_feasible(trial, n)
        trial_c5 = compute_c5(trial)
        if trial_c5 < cur_c5 or rng.random() < np.exp(-(trial_c5 - cur_c5) / max(t, 1e-12)):
            cur = trial
            cur_c5 = trial_c5
            if trial_c5 < best_c5:
                best = trial.copy()
                best_c5 = trial_c5
    return best, best_c5

def coord_refine(h, rng, n_coords=200):
    n = len(h)
    h_cur = h.copy()
    cur_c5 = compute_c5(h_cur)
    for _ in range(n_coords):
        i = int(rng.integers(n))
        best_trial = h_cur.copy()
        best_val = cur_c5
        for delta in [0.05, -0.05, 0.02, -0.02, 0.1, -0.1]:
            trial = h_cur.copy()
            trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
            trial = make_feasible(trial, n)
            v = compute_c5(trial)
            if v < best_val - 1e-12:
                best_val = v
                best_trial = trial
        if best_val < cur_c5 - 1e-12:
            h_cur = best_trial
            cur_c5 = best_val
    return h_cur, cur_c5
