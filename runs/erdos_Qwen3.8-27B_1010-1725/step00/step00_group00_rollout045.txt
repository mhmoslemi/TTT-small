import numpy as np
from scipy.optimize import minimize
import time


def project_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def evaluate_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def smooth_objective(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    m = float(np.max(g))
    shifted = alpha * (g - m)
    log_terms = np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted))))
    return m + np.exp(log_terms) / alpha


def make_obj(n, alpha):
    def obj(x):
        return smooth_objective(x, n, alpha)
    return obj


def interpolate_h(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xt, xs, h_src)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get initial values
    if 'initial_h_values' in globals():
        iv = globals()['initial_h_values']
        if iv is not None and len(iv) > 0:
            h_init = np.array(iv, dtype=float)
        else:
            n0 = 50
            h_init = np.zeros(n0)
            h_init[:n0 // 2] = 1.0
    else:
        n0 = 50
        h_init = np.zeros(n0)
        h_init[:n0 // 2] = 1.0

    best_h = project_sum(h_init, len(h_init))
    best_n = len(best_h)
    best_c5 = evaluate_c5(best_h)

    # List of n to try (sorted)
    n_list = sorted(set([best_n, 50, 60, 80, 100]))
    n_list = [n for n in n_list if n <= 120]

    for n in n_list:
        if time.time() - t0 > budget_s * 0.85:
            break

        bounds = [(0.0, 1.0)] * n

        # Build multiple starting points
        starts = []
        # Interpolate from best
        h_ip = interpolate_h(best_h, best_n, n)
        starts.append(project_sum(h_ip, n))

        # Random beta start
        h_r = np.random.beta(2.0, 2.0, n)
        starts.append(project_sum(h_r))

        # Structured: alternating blocks
        n_blocks = max(4, n // 8)
        block_len = n / n_blocks
        h_struct = np.zeros(n)
        for i in range(0, n_blocks, 2):
            s = int(i * block_len)
            e = int((i + 1) * block_len)
            h_struct[s:min(e, n)] = 1.0
        starts.append(project_sum(h_struct, n))

        # Another structured: sinusoidal
        x_vals = np.linspace(0, 2 * np.pi * (n // 10), n)
        h_sin = 0.5 + 0.5 * np.sin(x_vals)
        starts.append(project_sum(h_sin, n))

        for h_start in starts:
            if time.time() - t0 > budget_s * 0.93:
                break

            c5_start = evaluate_c5(h_start)

            # Progressive alpha for sharper max approximation
            for alpha in [20, 60, 150, 400]:
                if time.time() - t0 > budget_s * 0.96:
                    break

                cons = [{'type': 'eq',
                         'fun': lambda x, n=n: np.sum(x) - n / 2.0,
                         'jac': lambda x: np.ones(n)}]

                obj_func = make_obj(n, alpha)

                try:
                    res = minimize(obj_func, h_start, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 800, 'ftol': 1e-14})
                    if res.success and res.fun < smooth_objective(h_start, n, alpha):
                        c5_new = evaluate_c5(res.x)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = project_sum(res.x, n)
                            best_n = n
                            h_start = best_h.copy()
                    elif res.success:
                        c5_new = evaluate_c5(res.x)
                        h_start = project_sum(res.x, n)
                except Exception:
                    pass

    # Final polishing: one more SLSQP at the best n with high alpha
    if time.time() - t0 < budget_s * 0.97:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq',
                 'fun': lambda x, n=n: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(make_obj(n, 800), best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-15})
            if res.success:
                c5_new = evaluate_c5(res.x)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = project_sum(res.x, n)
                    best_n = n
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, len(best_h))
