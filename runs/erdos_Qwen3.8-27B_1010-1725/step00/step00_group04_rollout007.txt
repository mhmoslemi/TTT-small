```
def project_feasible(z, S):
    h=np.asarray(z,dtype=float); n=h.size
    h=np.clip(h,0,1)
    for _ in range(50):
       r=S-h.sum()
       if abs(r)<1e-13: break
       if r>0:
          mask=h<1.0-1e-12
          if not np.any(mask): break
          h[mask]=np.minimum(1.0,h[mask]+r/np.count_nonzero(mask))
       else:
          mask=h>1e-12
          if not np.any(mask): break
          h[mask]=np.maximum(0.0,h[mask]+r/np.count_nonzero(mask))
    # final exact-ish: distribute residual to variable with largest slack
    for _ in range(10):
       diff=S-h.sum()
       if abs(diff)<1e-15: break
       if diff>0:
          i=np.argmax(1-h)
          add=min(diff,1-h[i])
          h[i]+=add
       else:
          i=np.argmax(h)
          sub=min(-diff,h[i])
          h[i]-=sub
    return h
```

```python
import time
import numpy as np
from scipy import optimize

_PN_SORT = 0


def n_sort_key(n):
    return (abs(n - _PN_SORT), n)


def project_feasible(z, S):
    h = np.asarray(z, dtype=float)
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        r = S - h.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] = np.minimum(1.0, h[mask] + r / float(np.count_nonzero(mask)))
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] + r / float(np.count_nonzero(mask)))
    for _ in range(30):
        diff = S - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            i = int(np.argmax(h))
            sub = min(-diff, h[i])
            h[i] -= sub
    return h


def resample_to_n(h, n):
    h = np.asarray(h, dtype=float)
    n_old = h.size
    S = 0.5 * float(n)
    if n_old == n:
        return project_feasible(h, S)
    if n_old == 1:
        return project_feasible(np.full(n, h[0]), S)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, S)


def correlate_all(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    out = np.empty(2 * n - 1, dtype=float)
    v = 1.0 - h
    out[n - 1] = np.dot(h, v)
    for l in range(1, n):
        out[n - 1 + l] = np.dot(h[:n - l], v[l:])
        out[n - 1 - l] = np.dot(h[l:], v[:n - l])
    return out


def exact_max_local(h):
    n = h.size
    if n < 2:
        return 0.0
    return float(np.max(correlate_all(h) * (2.0 / float(n))))


def exact_max_np(h):
    n = h.size
    if n < 2:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))))


def grad_weighted_A(h, weights):
    h = np.asarray(h, dtype=float)
    n = h.size
    G = np.zeros(n, dtype=float)
    center = n - 1
    dx = 2.0 / float(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w == 0.0:
            continue
        l = idx - center
        lo1 = -l if l < 0 else 0
        hi1 = n - 1 - l
        if hi1 > n - 1:
            hi1 = n - 1
        if lo1 <= hi1:
            G[lo1:hi1 + 1] += w
            G[lo1:hi1 + 1] -= w * h[lo1 + l:hi1 + l + 1]
        lo2 = l if l > 0 else 0
        hi2 = n - 1 + l
        if hi2 > n - 1:
            hi2 = n - 1
        if lo2 <= hi2:
            G[lo2:hi2 + 1] -= w * h[lo2 - l:hi2 - l + 1]
    return G * dx


def smooth_value(x, args):
    p = float(args[0])
    x = np.asarray(x, dtype=float)
    dx = 2.0 / float(x.size)
    c = correlate_all(x) * dx
    c = np.maximum(c, 0.0)
    return float(np.sum(c ** p) + 1e-18)


def smooth_grad(x, args):
    p = float(args[0])
    x = np.asarray(x, dtype=float)
    dx = 2.0 / float(x.size)
    c = correlate_all(x) * dx
    c = np.maximum(c, 0.0)
    if p == 1.0:
        weights = np.ones_like(c)
    else:
        weights = p * (c ** (p - 1.0))
    return grad_weighted_A(x, weights)


def sum_eq(x, args):
    return float(np.sum(x) - args[0])


def sum_eq_jac(x, args):
    return np.ones_like(np.asarray(x, dtype=float))


def make_candidate(n, kind, seed, parent, best_h):
    S = 0.5 * float(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))

    if kind == 0:
        src = parent if parent is not None else best_h
        if src is None:
            h = 0.5 * np.ones(n)
        else:
            h = resample_to_n(src, n)
        return project_feasible(h, S)

    if kind == 1:
        src = best_h if best_h is not None else parent
        if src is None:
            h = 0.5 * np.ones(n)
        else:
            h = resample_to_n(src, n)
        return project_feasible(h, S)

    if kind == 2:
        a = 0.2 + 0.1 * float((seed + kind) % 6)
        a = min(a, 0.8)
        h = np.empty(n, dtype=float)
        half = n // 2
        h[:half] = 0.5 + 0.5 * a
        h[half:] = 0.5 - 0.5 * a
        return project_feasible(h, S)

    if kind == 3:
        amp = 0.25 + 0.1 * float(seed % 5)
        amp = min(amp, 0.9)
        ph = np.pi * float(seed % 8) / 8.0
        base = np.cos(np.pi * x + ph)
        base = base - base.mean()
        mx = float(np.max(np.abs(base)))
        if mx > 1e-15:
            base = base / mx
        h = 0.5 * (1.0 + amp * base)
        return project_feasible(h, S)

    if kind == 5:
        rng = np.random.default_rng(seed)
        b = np.zeros(n, dtype=float)
        for k in range(1, 4):
            amp = rng.uniform(0.1, 1.0 / float(k ** 0.7))
            ph = rng.uniform(-np.pi, np.pi)
            b += amp * np.cos(np.pi * float(k) * x + ph)
        b = b - b.mean()
        b = np.tanh(b)
        h = 0.5 * (1.0 + b)
        return project_feasible(h, S)

    if kind == 6:
        amp = 0.25 + 0.05 * float(seed % 4)
        amp = min(amp, 0.48)
        z = x.copy()
        z[x >= 1.0] -= 1.0
        cc = np.cos(2.0 * np.pi * z)
        h = np.where(x < 1.0, 0.5 + amp * cc, 0.5 - amp * cc)
        return project_feasible(h, S)

    return project_feasible(0.5 * np.ones(n), S)


def run_slsqp(h0, n, p, maxiter, maxfun):
    h0 = project_feasible(np.asarray(h0, dtype=float), 0.5 * float(n))
    bounds = [(0.0, 1.0)] * n
    cons = (
        {"type": "eq", "fun": sum_eq, "jac": sum_eq_jac, "args": (0.5 * float(n),)},
    )
    opt = {"maxiter": int(maxiter), "fitol": 1e-12, "ftol": 1e-12, "maxfun": int(maxfun)}
    try:
        res = optimize.minimize(
            smooth_value,
            h0,
            method="SLSQP",
            jac=smooth_grad,
            args=(float(p),),
            bounds=bounds,
            constraints=cons,
            options=opt,
        )
    except Exception:
        res = None
    out = h0
    if res is not None and hasattr(res, "x"):
        out = project_feasible(np.asarray(res.x, dtype=float), 0.5 * float(n))
    return out


def active_polish(h, max_iter, deadline):
    h = project_feasible(np.asarray(h, dtype=float), 0.5 * float(h.size))
    n = h.size
    S = 0.5 * float(n)
    best = exact_max_local(h)
    length = 2 * n - 1
    rng = np.random.default_rng(917)

    for _ in range(int(max_iter)):
        if deadline is not None and time.perf_counter() > deadline:
            break
        c = correlate_all(h) * (2.0 / float(n))
        cm = float(np.max(c))
        if cm <= 0.0:
            break
        K = min(32, length)
        idx = np.argsort(c)[-K:]
        top = np.maximum(c[idx], 0.0)
        w = np.zeros(length, dtype=float)
        w[idx] = (top / cm) ** 12.0
        ws = float(np.sum(w))
        if ws <= 0.0:
            break
        w /= ws
        G = grad_weighted_A(h, w)
        gn = float(np.linalg.norm(G))
        if gn < 1e-13:
            break
        u = -G / gn
        improved = False
        for step in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078):
            y = project_feasible(h + step * u, S)
            fy = exact_max_local(y)
            if fy < best - 1e-13:
                h = y
                best = fy
                improved = True
                break
        if not improved:
            y = project_feasible(h + 0.025 * (rng.random(n) - 0.5), S)
            fy = exact_max_local(y)
            if fy < best - 1e-13:
                h = y
                best = fy
            else:
                break
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    global _PN_SORT
    start = time.perf_counter()
    budget = float(budget_s)
    deadline = start + max(1.0, budget)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size == 0:
            parent = None
    except Exception:
        parent = None

    if parent is None:
        n_def = 64
        x = (np.arange(n_def) + 0.5) * (2.0 / float(n_def))
        cc = np.cos(np.pi * x)
        cc -= cc.mean()
        cc /= max(1e-12, float(np.max(np.abs(cc))))
        parent = project_feasible(0.5 * (1.0 + 0.45 * cc), 0.5 * float(n_def))

    parent = project_feasible(parent, 0.5 * float(parent.size))
    best_h = parent.copy()
    best_c = exact_max_np(best_h)

    pn = int(parent.size)
    _PN_SORT = pn
    nset = []

    def add_n(n):
        n = int(round(float(n)))
        if 12 <= n <= 260 and n not in nset:
            nset.append(n)

    add_n(pn)
    add_n(pn - 8)
    add_n(pn + 8)
    add_n(2 * pn)
    add_n(3 * pn)
    add_n(4 * pn)
    add_n(32)
    add_n(48)
    add_n(64)
    add_n(96)
    add_n(128)
    add_n(176)
    add_n(224)
    nset.sort(key=n_sort_key)

    max_calls = 48
    calls = 0
    kind_order = (6, 3, 5, 2)

    for n in nset:
        if calls >= max_calls:
            break
        if time.perf_counter() > deadline - 25.0:
            break

        if n <= 96:
            extra = 4
        elif n <= 160:
            extra = 3
        else:
            extra = 2

        hlist = [resample_to_n(parent, n), resample_to_n(best_h, n)]
        for kind in kind_order[:extra]:
            hlist.append(make_candidate(n, kind, int(seed) + 100 * n + 7 * kind, parent, best_h))

        if n <= 56:
            p_list = (4.0, 12.0, 20.0)
            maxiter, maxfun = 260, 3500
        elif n <= 112:
            p_list = (4.0, 12.0)
            maxiter, maxfun = 180, 2400
        elif n <= 180:
            p_list = (6.0, 14.0)
            maxiter, maxfun = 120, 1600
        else:
            p_list = (8.0, 16.0)
            maxiter, maxfun = 80, 1000

        for h0 in hlist:
            if calls >= max_calls:
                break
            if time.perf_counter() > deadline - 30.0:
                break
            cur = h0
            for p in p_list:
                if calls >= max_calls or time.perf_counter() > deadline - 20.0:
                    break
                calls += 1
                cur = run_slsqp(cur, n, p, maxiter, maxfun)
                cnp = exact_max_np(cur)
                if cnp < best_c - 1e-13:
                    best_h = cur
                    best_c = cnp
                    if time.perf_counter() < deadline - 15.0:
                        hp, _ = active_polish(best_h, 60, deadline - 5.0)
                        hp = project_feasible(hp, 0.5 * float(hp.size))
                        ccp = exact_max_np(hp)
                        if ccp < best_c - 1e-13:
                            best_h = hp
                            best_c = ccp

            if calls < max_calls and time.perf_counter() < deadline - 45.0:
                calls += 1
                extra_start = resample_to_n(best_h, n)
                cur2 = run_slsqp(extra_start, n, p_list[-1], maxiter, maxfun)
                c2 = exact_max_np(cur2)
                if c2 < best_c - 1e-13:
                    best_h = cur2
                    best_c = c2
                    if time.perf_counter() < deadline - 15.0:
                        hp, _ = active_polish(best_h, 60, deadline - 5.0)
                        hp = project_feasible(hp, 0.5 * float(hp.size))
                        ccp = exact_max_np(hp)
                        if ccp < best_c - 1e-13:
                            best_h = hp
                            best_c = ccp

    if time.perf_counter() < deadline - 3.0:
        polish_iter = 500 if best_h.size <= 64 else 280 if best_h.size <= 128 else 140
        best_h, _ = active_polish(best_h, polish_iter, deadline - 0.5)

    best_h = project_feasible(best_h, 0.5 * float(best_h.size))
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = exact_max_np(best_h)
    if not np.isfinite(final_c):
        final_c = exact_max_local(best_h)
    return best_h, float(final_c), int(best_h.size)
```
