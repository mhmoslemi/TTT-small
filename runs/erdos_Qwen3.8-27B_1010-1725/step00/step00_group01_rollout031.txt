def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_c5_shifts(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx, np.argmax(corr * dx)

def project_onto_set(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h, m):
    """Subgradient of c_m = sum_n h[n+m]*(1-h[n]) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if m == 0:
        # c_0 = sum_n h[n]*(1-h[n]) = sum h - sum h^2
        # dc_0/dh[j] = 1 - 2*h[j]
        grad = 1.0 - 2.0 * h
    else:

        if m > 0:
            # j-m valid: m <= j <= n-1
            # j+m valid: 0 <= j <= n-1-m
            for j in range(m, n):
                grad[j] += (1.0 - h[j - m])
            for j in range(0, n - m):
                grad[j] -= h[j + m]
        else:
            k = -m
            # j-m = j+k valid: 0 <= j <= n-1-k

            for j in range(0, n - k):
                grad[j] += (1.0 - h[j + k])
            for j in range(k, n):
                grad[j] -= h[j - k]
    return grad

def optimize_subgradient(h0, target_sum, max_iter=2000, lr0=0.1, time_limit=30):

    h = h0.copy()
    h = project_onto_set(h, target_sum)

def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_onto_set(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, m):
    """Gradient of c_m = sum_i h[i+m]*(1-h[i]) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    else:
        k = -m
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    return grad

def pgd_solve(h0, n, time_limit):
    """Projected gradient descent on max-shift objective."""
    target_sum = n / 2.0
    h = project_onto_set(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = 0.05
    momentum = np.zeros(n)
    for it in range(50000):
        if time.time() - t0 > time_limit:

        corr = compute_corr(h)
        worst_idx = np.argmax(corr)
        m = worst_idx - (n - 1)

        h_new = project_onto_set(h_new, target_sum)

        if c5_new < best_c5:

            h = h_new

        else:

            h = best_h.copy()  # revert

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, "full")
    return float(np.max(corr) * dx)

def project_set(h, ts):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[:n - m] -= h[m:]
        g[m:] += 1.0 - h[:n - m]
    else:
        k = -m
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    return g

def pgd_opt(h0, n, tlim):
    ts = n / 2.0
    h = project_set(h0.copy(), ts)
    bc = compute_c5(h)
    bh = h.copy()
    t0 = time.time()
    lr = 0.05
    mom = np.zeros(n)
    stall = 0
    while time.time() - t0 < tlim:
        corr = np.correlate(h, 1.0 - h, "full")
        m = int(np.argmax(corr)) - (n - 1)
        g = grad_shift(h, m)
        mom = 0.85 * mom + g

        if c2 < bc - 1e-12:
            bc = c2
            bh = h2.copy()
            h = h2
            lr = min(lr * 1.02, 0.4)
            stall = 0

            lr *= 0.82
            stall += 1

    return bh, bc

def random_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] = 0.5
    else:
        rem = half * 2 - n

            h[rng.randint(n)] = 0.5
    return project_set(h, n / 2.0)

def structured_start(n, rng, kind=0):
    h = np.zeros(n)
    target = n / 2.0
    if kind == 0:
        # Alternating blocks
        blen = max(2, n // 8)
        phase = rng.randint(blen)
        for i in range(n):
            if (i + phase) // blen % 2 == 0:
                h[i] = 1.0

        # Two large blocks

        shift = rng.randint(n)
        h[shift:shift + b1] = 1.0
        h[(shift + b1 + 1) % n:] = 0.0

def softmax_pgd(h0, n, tlim, beta=50.0):
    ts = n / 2.0
    h = project_set(h0.copy(), ts)
    bc = compute_c5(h)
    bh = h.copy()

        shifted = corr - np.max(corr)

        weights /= np.sum(weights)
        g = np.zeros(n)
        for m in range(-(n - 1), n):

            h = h2
