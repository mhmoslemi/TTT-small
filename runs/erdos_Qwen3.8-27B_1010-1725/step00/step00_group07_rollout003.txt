import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(20):
        err = target - h.sum()
        if abs(err) < 1e-14:
            break
        if err > 0:
            adj = h < 1.0
        else:
            adj = h > 0.0
        cnt = adj.sum()
        if cnt == 0:
            break
        h[adj] += err / cnt
        h = np.clip(h, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-15:
        idx = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + resid, 0.0, 1.0)
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return fix_sum(h_new, n_new)

def make_periodic_init(n, seed):

    h += rng.normal(0, 0.1, n)
