        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            js2 = np.arange(j_lo2, j_hi2)
            grad[js2] -= dx * h[js2 + d]
    grad /= len(active)
    return grad

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    f = interp1d(x_old, h, kind='linear', fill_value=(h[0], h[-1]), bounds_error=False)
    return f(x_new)

def lbfgs_step(h, n, dx, target, T, max_iter=100, pen_coeff=50.0):
    def obj(x):
        xp = np.clip(x, 0, 1)
        val = softmax_obj_val(xp, dx, T)
        val += pen_coeff * (np.sum(xp) - target) ** 2
        return val

    def obj_grad(x):
        xp = np.clip(x, 0, 1)
        g = softmax_grad(xp, dx, T)
        g += 2.0 * pen_coeff * (np.sum(xp) - target) / n
        return g

    res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7})
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.zeros(70)
        h_init[:35] = 1.0

    n0 = len(h_init)
    best_h = project_feasible(h_init, n0)
    best_c5 = c5_value(best_h, 2.0 / n0)

    n_list = sorted(set([n0, 100, 140, 200, 280, 360]))
    n_list = [n for n in n_list if n % 2 == 0]

    for n in n_list:
        if time.time() - t0 > budget_s - 15:
            break
        remaining = budget_s - 15 - (time.time() - t0)
        frac = remaining / (budget_s - 15)
        alloc = max(30, int(frac * (budget_s - 60) * 0.35))
        t_alloc_end = time.time() + alloc

        if n == len(best_h):
            h = best_h.copy()
        else:
            h = upsample(best_h, len(best_h), n)
            h = project_feasible(h, n)

        dx = 2.0 / n
        target = n / 2.0
        c5 = c5_value(h, dx)

        # Softmax annealing
        T = 40.0
        while time.time() < t_alloc_end - 20 and T > 2.5:
            h_new = lbfgs_step(h, n, dx, target, T, max_iter=min(120, n))
            h_new = project_feasible(h_new, n)
            c5_new = c5_value(h_new, dx)
            if c5_new < c5 + 1e-9:
                h = h_new
                c5 = c5_new
                T = max(2.5, T * 0.65)
            else:
                if rng.random() < 0.25:
                    h = h_new
                    c5 = c5_new
                T = max(2.5, T * 0.8)

        # Subgradient refinement with the true max
        lr = 0.02
        for it in range(500):
            if time.time() > t_alloc_end - 5:
                break
            g = max_subgrad(h, dx)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-10:
                break
            lr = max(0.001, 0.02 * (1 - 0.002 * it))
            h = h - lr * g / (g_norm + 1e-12)
            h = project_feasible(h, n)

        c5 = c5_value(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Random perturbations from current best
        for rep in range(4):
            if time.time() > t_alloc_end - 5:
                break
            h_r = project_feasible(h + 0.08 * rng.randn(n), n)
            T_r = 30.0
            for _ in range(4):
                if time.time() > t_alloc_end - 5:
                    break
                h_r = lbfgs_step(h_r, n, dx, target, T_r, max_iter=60)
                h_r = project_feasible(h_r, n)
                c5_r = c5_value(h_r, dx)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                T_r = max(3.0, T_r * 0.6)

            g_r = max_subgrad(h_r, dx)
            for _ in range(50):
                h_r = h_r - 0.015 * g_r
                h_r = project_feasible(h_r, n)
                c5_r = c5_value(h_r, dx)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                g_r = max_subgrad(h_r, dx)

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = c5_value(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
