import numpy as np
from scipy.optimize import minimize
import time

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(x, temp):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    m = np.max(corr)
    scaled = (corr - m) / temp
    return float((m + temp * np.log(np.sum(np.exp(scaled)))) * dx)

def eq_constraint(x):
    return np.sum(x) - _global_target

def eq_jacobian(x):
    return np.ones(len(x))

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        nm = mask.sum()
        if nm == 0:
            break
        step = diff / nm
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def make_initial(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return fix_sum(h, n / 2.0)

def interpolate_h(h_from, n_to):
    n_from = len(h_from)
    xs = np.linspace(0.0, 1.0, n_to)
    xref = np.linspace(0.0, 1.0, n_from)
    return fix_sum(np.interp(xs, xref, h_from), n_to / 2.0)

def slsqp_optimize(h0, n, temp, target, maxiter):
    global _global_target
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian}]
    try:
        res = minimize(
            smooth_max_obj, h0, args=(temp,),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return res.x
    except Exception:
        return h0
