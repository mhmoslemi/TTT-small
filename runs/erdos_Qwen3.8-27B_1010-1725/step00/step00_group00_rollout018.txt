import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_c5_val_grad(h, n, tau=80.0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    exp_c = np.exp(tau * (c - cmax))
    S = np.sum(exp_c)
    p = exp_c / S
    val = (cmax + np.log(S) / tau) * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        shift = k - nm1
        for m in range(n):
            i1 = m - shift
            if 0 <= i1 < n:
                grad[m] += pk * (1.0 - h[i1])
            i2 = m + shift
            if 0 <= i2 < n:
                grad[m] -= pk * h[i2]
    grad *= dx
    return val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

            hr = 0.5 * interpolate_h(best_h, best_n, n_cur) + 0.5 * hr
            hr = project_feasible(hr, n_cur)
            h_try, c5_try = try_slsqp(hr, n_cur, maxiter=100)
            if c5_try < best_c5:
                best_h, best_c5, best_n = h_try.copy(), c5_try, n_cur

    # Phase 3: final polish with more iterations and larger tau
    remaining = budget_s * 0.95 - (time.time() - t0)
    if remaining > 5:
        h_try, c5_try = try_slsqp(best_h, best_n, maxiter=300)
        if c5_try < best_c5:
            best_h, best_c5 = h_try.copy(), c5_try
        if time.time() - t0 < budget_s * 0.93:
            h_try2, c5_try2 = try_smooth_lbfgsb(best_h, best_n, maxiter=300, tau=100.0)
            if c5_try2 < best_c5:
                best_h, best_c5 = h_try2.copy(), c5_try2

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
