import numpy as np
from scipy.optimize import minimize
import time

BETA = 50

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(corr))
    shifted = corr - m
    return (m + float(np.log(np.sum(np.exp(BETA * shifted)))) / BETA) * dx

def _cons(x):
    return float(np.sum(x) - len(x) / 2.0)

def fix_sum(h, target):
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
