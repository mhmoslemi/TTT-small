    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    # Adjust to maintain sum
    target = n / 2.0
    current = np.sum(h)

                h = np.clip(h, 0, 1)

        h_alt = project_feasible(h_alt, n0)
        c5_alt = evaluate_c5(h_alt)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt.copy(), c5_alt

    # Stage 1: SA at original resolution n0
    el = time.time() - t_start
    t1 = min(300, (budget_s - el - 250) * 0.35)
    if t1 > 15:
        h, c5 = sa_optimize(best_h, n0, t1, seed)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Stage 2: Upsample to 2*n0 and SA
    el = time.time() - t_start
    n1 = 2 * n0
    if budget_s - el > 250:
        h1 = upsample(best_h, n1)
        t2 = min(350, (budget_s - el - 150) * 0.5)
        if t2 > 30:
            h, c5 = sa_optimize(h1, n1, t2, seed + 50)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Stage 3: Upsample to 3*n0 if time remains
    el = time.time() - t_start
    n2 = 3 * n0
    if budget_s - el > 160 and n2 != len(best_h) and n2 <= 280:
        h2 = upsample(best_h, n2)
        t3 = min(250, (budget_s - el - 50) * 0.5)
        if t3 > 30:
            h, c5 = sa_optimize(h2, n2, t3, seed + 100)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Stage 4: Final polish at best resolution
    el = time.time() - t_start
    if budget_s - el > 50:
        nb = len(best_h)
        t4 = budget_s - el - 5
        h, c5 = sa_optimize(best_h, nb, t4, seed + 200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # One more quick local refinement with smaller scale
    el = time.time() - t_start
    if budget_s - el > 20:
        nb = len(best_h)
        rng_f = np.random.RandomState(seed + 999)
        h_f = best_h + 0.01 * rng_f.randn(nb)
        h_f = project_feasible(h_f, nb)
        if evaluate_c5(h_f) < best_c5:
            h_f = project_feasible(h_f, nb)
            best_h, best_c5 = h_f, evaluate_c5(h_f)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
