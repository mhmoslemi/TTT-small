    m = np.max(scaled)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

_op = {'beta': 1.0, 'n': 0}

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_og(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = corr * dx
    m = np.max(sc)
    e = np.exp(beta * (sc - m))
    s = e.sum()
    w = e / s
    obj = m + np.log(s) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        jm = 0 if lag >= 0 else -lag
        jM = n if lag < 0 else n - lag
        for j in range(jm, jM):
            g = 1.0
            if 0 <= j + lag < n:
                g -= h[j + lag]
            if 0 <= j - lag < n:
                g -= h[j - lag]
            grad[j] += wk * g * dx
    return obj, grad

def _obj_fn(x):
    return _smooth_og(x, _op['beta'])[0]

def _grad_fn(x):
    return _smooth_og(x, _op['beta'])[1]

def _eq_con(x):
    return x.sum() - _op['n'] * 0.5

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-13:
            break
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    xo = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    pad = np.concatenate(([h_old[0]], h_old, [h_old[-1]]))
    return project(np.interp(xn, xo, pad), n_new)

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        rest = np.where(h < 0.5)[0]
        h[rng.choice(rest)] += 0.5
    return project(h, n)

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        if c % 2 == 0:
            h[i] = 1.0
    return project(h, n)

def slsqp_run(h, n, betas, maxiter, tlimit):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    for beta in betas:
        if time.time() - t0 > tlimit:
            break
        _op['beta'] = beta
        _op['n'] = n
        try:
            res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h2 = project(res.x, n)
            if eval_c5(h2) <= eval_c5(h):
                h = h2
        except Exception:
            pass
    return h

def subgrad_polish(h, n, niter, tlimit):
    t0 = time.time()
    dx = 2.0 / n
    step = 0.05
    for _ in range(niter):
        if time.time() - t0 > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = int(np.argmax(corr))
        lag = ks - (n - 1)
        sg = np.zeros(n)
        jm = 0 if lag >= 0 else -lag
        jM = n if lag < 0 else n - lag
        for j in range(jm, jM):
            g = 1.0
            if 0 <= j + lag < n:
                g -= h[j + lag]
            if 0 <= j - lag < n:
                g -= h[j - lag]
            sg[j] = g * dx
        h2 = project(h - step * sg, n)
        if eval_c5(h2) < eval_c5(h):
            h = h2
            step = min(step * 1.04, 1.0)
        else:
            step *= 0.85
    return h

def snap_and_check(h, n):
    h2 = h.copy()
    h2[h2 > 0.92] = 1.0
    h2[h2 < 0.08] = 0.0
    h2 = project(h2, n)
    if eval_c5(h2) <= eval_c5(h):
        return h2
    h3 = h.copy()
    h3[h3 > 0.98] = 1.0
    h3[h3 < 0.02] = 0.0
    h3 = project(h3, n)
    return h3 if eval_c5(h3) <= eval_c5(h) else h

def bit_flip_search(h, n, ntrials, tlimit):
    t0 = time.time()
    dx = 2.0 / n
    best = h.copy()
    best_v = eval_c5(best)
    for _ in range(ntrials):
        if time.time() - t0 > tlimit:
            break
        i = np.random.randint(n)
        h2 = best.copy()
        if h[i] > 0.5:
            h2[i] = 0.0
        else:
            h2[i] = 1.0
        h2 = project(h2, n)
        v = eval_c5(h2)
        if v < best_v:
            best = h2
            best_v = v
    return best

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():

        candidates.append((random_binary(n, rng), n))
        candidates.append((thue_morse_init(n), n))

    if rem() > 10:

    best_c5 = eval_c5(best_h)
