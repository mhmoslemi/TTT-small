    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    shifted = T * (vals - m)
    # Clip to prevent overflow

    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    f = m + np.log(sum_exp) / T
    
    g = np.zeros(n)
    # For each i, compute dc_k/dh_i summed over k with weights
    # dc_k/dh_i = [1-h[i-k]]*1{0<=i-k<n} - h[i+k]*1{0<=i+k<n}
    # g_i = dx * sum_k weights[k] * dc_k/dh_i
    
    # Let's compute this efficiently
    # First term: sum_k weights[k] * (1-h[i-k]) * 1{0<=i-k<n}
    #   = sum_{j: 0<=j<n, 0<=i-j+n-1<2n-1} weights[i-j+n-1] * (1-h[j])
    #   where k = i-j, valid when 0<=i-j<=n-1 (for h[i-k] = h[j])
    #   and k+n-1 in [0, 2n-2] (for weight index)
    
    # Actually let me just do the double loop for n=70
    for i in range(n):
        gi = 0.0
        for k in range(-(n-1), n):
            wk = weights[k + n - 1]
            if wk < 1e-15:
                continue
            # h[i] as h[n+k]: n_idx = i-k
            n_idx = i - k
            if 0 <= n_idx < n:
                gi += wk * (1.0 - h[n_idx])
            # h[i] in (1-h[n]): n_idx = i
            if 0 <= i + k < n:
                gi -= wk * h[i + k]

            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            try:
                res = minimize(smooth_obj_wrapper, h_curr, jac=smooth_grad_wrapper,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': 1e-10})
                if not np.any(np.isnan(res.x)) and not np.any(np.isinf(res.x)):
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    c5_curr = compute_c5(h_curr)
                    if c5_new <= c5_curr:
                        h_curr = h_new
            except Exception:
                pass

        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5 - 1e-10:
            best_c5 = c5_curr
            best_h = h_curr.copy()

    # Final polishing on best solution
    if time.time() < t_limit * 0.96:
        h_curr = best_h.copy()
        for T in [800, 1500, 3000]:
            if time.time() > t_limit * 0.98:
                break
            _T_current = T
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            try:
                res = minimize(smooth_obj_wrapper, h_curr, jac=smooth_grad_wrapper,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-12})
                if not np.any(np.isnan(res.x)):
                    h_new = project_feasible(res.x, n)
                    if compute_c5(h_new) < compute_c5(h_curr):
                        h_curr = h_new
            except Exception:
                pass
        c5_final = compute_c5(h_curr)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_curr.copy()

    # Try upsampled refinement if time permits
    if time.time() < t_limit * 0.80:
        n2 = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_feasible(h_up, n2)
        h_curr = h_up.copy()
        c5_up = compute_c5(h_curr)
        for T in [50, 150, 500, 1000]:
            if time.time() > t_limit * 0.97:
                break
            _T_current = T
            bounds = [(0.0, 1.0)] * n2
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            try:
                res = minimize(smooth_obj_wrapper, h_curr, jac=smooth_grad_wrapper,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-10})
                if not np.any(np.isnan(res.x)):
                    h_new = project_feasible(res.x, n2)
                    if compute_c5(h_new) < compute_c5(h_curr):
                        h_curr = h_new
            except Exception:
                pass
        c5_up = compute_c5(h_curr)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_curr.copy()
            n = n2

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
