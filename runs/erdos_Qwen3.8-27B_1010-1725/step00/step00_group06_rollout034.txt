import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    # Final exact adjustment
    diff = n / 2.0 - h.sum()
    if abs(diff) > 0:
        # Find an element to adjust
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def grad_lag(h, m, n):
    g = np.zeros(n)
    if m >= 0:
        if m < n:
            lo = 0
            hi = n - 1 - m
            g[lo:hi+1] = 1.0 - h[lo+m:hi+1+m]
            if m > 0:
                g[m:] -= h[:n-m]
    else:
        mp = -m
        if mp < n:
            g[mp:] = 1.0 - h[:n-mp]
            g[:n-mp] -= h[mp:]
    return g

def optimize_h(h_init, n, rng, deadline, max_iter=200000):
    dx = 2.0 / n
    h = project_feasible(h_init, n)
    best = h.copy()
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    
    step = 0.05
    beta = 80.0
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = np.max(corr)
        
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 1000:
            h = project_feasible(h + rng.normal(0, 0.02, n), n)
            step = 0.05
            patience = 0
        
        # Compute gradient using top lags
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        vals = beta * corr[top_idx]
        vals -= vals.max()
        w = np.exp(vals)
        w /= w.sum()
        
        grad = np.zeros(n)
        for i in range(top_k):
            if w[i] < 1e-10:
                break
            m = top_idx[i] - (n - 1)
            grad += w[i] * grad_lag(h, m, n)
        grad *= dx
        
        # Step
        h_new = project_feasible(h - step * grad, n)
        new_c5 = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
        
        if new_c5 <= c5_val + 1e-12:
            h = h_new
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.7
            if step < 1e-7:
                step = 0.05
                h = project_feasible(h + rng.normal(0, 0.01, n), n)
                
    return best, best_c5

    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h)

        if c5_fin < c5_best:

    h_best = project_feasible(h_best, n_best)
    c5_final = np.max(np.correlate(h_best, 1.0 - h_best, mode='full')) * (2.0 / n_best)
    
    return h_best, c5_final, n_best
