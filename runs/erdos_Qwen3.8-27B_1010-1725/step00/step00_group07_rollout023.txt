import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    mf = f.max()
    bf = beta * (f - mf)
    bf = np.clip(bf, -700.0, 0.0)
    w = np.exp(bf)
    ws = w.sum()
    weights = w / ws
    obj = mf + np.log(ws) / beta
    g = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-13:
            continue
        lag = k - nm1
        if lag == 0:
            g += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            a = n - lag
            g[:a] += wk * dx * (1.0 - h[lag:])
            g[lag:] -= wk * dx * h[:a]
        else:
            m = -lag
            a = n - m
            g[m:] += wk * dx * (1.0 - h[:a])
            g[:a] -= wk * dx * h[m:]
    return obj, g

def project_feasible(h):
    n = len(h)
    tgt = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

    for _ in range(300):
        g += 0.5 - g.mean()
        np.clip(g, 0.0, 1.0, out=g)
    h = np.tile(g, (n + p - 1) // p)[:n]
    return project_feasible(h)

    def __init__(self, beta):
        self._b = beta
    def __call__(self, h):
        return smooth_obj_grad(h, self._b)
