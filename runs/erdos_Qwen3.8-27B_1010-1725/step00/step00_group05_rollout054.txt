    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h = np.ones(70) * 0.5

    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Greedy refinement from initial (30% budget)
    dl1 = start + budget_s * 0.30
    h1, c5_1 = greedy_sweep(h, max_sweeps=300, g_iters=22, deadline=dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Random search with restarts (35% budget)
    dl2 = start + budget_s * 0.65
    for restart in range(20):
        if time.time() > dl2:
            break
        h2 = best_h + rng.normal(0, 0.015, best_h.shape)
        h2 = np.clip(h2, 0.0, 1.0)
        h2 += (best_h.shape[0] / 2.0 - np.sum(h2)) / best_h.shape[0]
        h2 = np.clip(h2, 0.0, 1.0)
        h2, c5_2 = random_phase(h2, 8000, rng, dl2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        # Occasional larger jump
        if restart % 5 == 4:
            h3 = best_h + rng.normal(0, 0.08, best_h.shape)
            h3 = np.clip(h3, 0.0, 1.0)
            h3 += (best_h.shape[0] / 2.0 - np.sum(h3)) / best_h.shape[0]
            h3 = np.clip(h3, 0.0, 1.0)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5 * 1.05:
                h3, c5_3 = random_phase(h3, 3000, rng, dl2)
                if c5_3 < best_c5:
                    best_c5, best_h = c5_3, h3.copy()

    # Phase 3: Fine greedy on best solution (20% budget)
    dl3 = start + budget_s * 0.85
    h3, c5_3 = greedy_sweep(best_h, max_sweeps=400, g_iters=28, deadline=dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Try different resolutions (12% budget)
    dl4 = start + budget_s * 0.97
    best_h, best_c5 = try_different_n(best_h, best_c5, start, dl4)

    # Final polish
    if time.time() < start + budget_s * 0.99:
        best_h, best_c5 = greedy_sweep(best_h, max_sweeps=50, g_iters=25,
                                        deadline=start + budget_s * 0.99)

    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    tgt = nf / 2.0
    s = np.sum(best_h)
    if abs(s - tgt) > 1e-8:
        best_h = best_h * (tgt / s)
        best_h = np.clip(best_h, 0.0, 1.0)
    c5_final = compute_c5(best_h)
    return best_h, c5_final, nf
