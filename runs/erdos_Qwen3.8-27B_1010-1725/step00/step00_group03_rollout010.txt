            best_h, best_c5, best_n = h_bin, c5_bin, best_n

    # Final polish at small n if time permits
    if time.time() - t0 < budget_s * 0.9:
        dx_s = 2.0 / 70
        x_70 = np.arange(70) * dx_s
        x_best = np.arange(len(best_h)) * (2.0 / len(best_h))
        h_70 = np.interp(x_70, x_best, best_h)
        h_70 = project_box_sum(h_70, 35.0)
        h_70_opt, c5_70 = optimize_from(h_70, 70, dx_s, 35.0, t0 + budget_s * 0.97, seed + 4)
        if c5_70 < best_c5:
            best_h, best_c5, best_n = h_70_opt, c5_70, 70

    # Validate and return
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    c5_final = c5_value(best_h, dx_final)
    return best_h, c5_final, best_n
