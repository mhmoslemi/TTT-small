    """Compute smooth max of w_k * corr_k where w depends on alpha."""
    # h already has sum constrained, dx baked in
    corr = np.correlate(h, 1-h, mode="full")

    m = np.max(corr)
    return m + (1.0/alpha) * np.log(np.sum(np.exp(alpha*(corr-m))))

    raw = np.correlate(h, 1-h, mode="full")
    corr = raw * dx
