def repair_integral(h, n):
    """Fix integral constraint while keeping bounds."""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)

    else:

from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = x.max() - target, x.min()  # or wider
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix: adjust to get exact integral
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:
        # Spread the difference
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0, 1)
    return proj

def smooth_obj_and_grad(h, n, beta=50.0):
    """Compute smooth max objective and its gradient."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    dx = 2.0 / n
    obj = (max_corr + np.log(np.sum(exp_vals)) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # d_k[i] = (1-h[i-shift]) * [0<=i-shift<n] - h[i+shift] * [0<=i+shift<n]
        # For i from 0 to n-1:
        # i-shift in [0,n-1] means i in [shift, shift+n-1] intersect [0,n-1]
        # i+shift in [0,n-1] means i in [-shift, n-1-shift] intersect [0,n-1]

        # Part 1: (1-h[i-shift]) for valid i
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)  # i-shift >= 0 -> i >= shift; i-shift < n -> i < n+shift
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-shift:hi1-shift])

        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)  # i+shift >= 0 -> i >= -shift; i+shift < n -> i < n-shift
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2+shift:hi2+shift]
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_obj_and_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    exp_vals = np.exp(beta * (corr - mx))
    weights = exp_vals / np.sum(exp_vals)
    dx = 2.0 / n
    obj_val = (mx + np.log(np.sum(exp_vals)) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + shift:hi2 + shift]
    return obj_val, grad

def slsqp_fun(x, n, beta):
    val, _ = smooth_obj_and_grad(x, n, beta)
    return val

    _, g = smooth_obj_and_grad(x, n, beta)
    return g

    return np.ones(n)

    h_interp = np.interp(x_to, x_from, h)

    t_start = time.time()
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac,
             'args': (n,)}]
    # Fix args in closures - use wrapper
    def _obj(x):
        return slsqp_fun(x, n, beta)
    def _jac(x):
        return slsqp_jac(x, n, beta)
    def _cfn(x):
        return slsqp_con_fun(x, n)
    def _cjac(x):
        return slsqp_con_jac(x, n)
    cons2 = [{'type': 'eq', 'fun': _cfn, 'jac': _cjac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h0, jac=_jac, method='SLSQP',
                   bounds=bounds, constraints=cons2,

    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t_start = time.time()

        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        d = np.zeros(n)
        lo1 = max(0, shift)
        hi1 = min(n, n + shift)
        if hi1 > lo1:
            d[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if hi2 > lo2:

            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
