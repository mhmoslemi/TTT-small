    best_c5, _ = compute_c5(h0)
    best_h = h0.copy()

    def remaining_time():
        return budget_s - (time.time() - t0)

    bounds = Bounds(0.0, 1.0)

    def obj(x):
        c5, _ = compute_c5(x)
        return c5

    def cons_fn(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]

    # Phase 1: Direct SLSQP from initial
    if remaining_time() > 30:
        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            if res.fun < best_c5 - 1e-13:
                best_h = make_feasible(res.x, target_sum)
                best_c5, _ = compute_c5(best_h)
        except Exception:
            pass

    # Phase 2: Diverse random restarts
    if remaining_time() > 30:
        starts = [best_h.copy()]
        # Binary midpoint
        h_bin = np.zeros(n)
        idx = rng.permutation(n)
        h_bin[idx[:n // 2]] = 1.0
        starts.append(make_feasible(h_bin, target_sum))
        # Shifted version
        h_shift = np.roll(h0, n // 7)
        starts.append(make_feasible(h_shift, target_sum))
        # A few random starts
        for _ in range(3):
            h_rand = rng.random(n)
            starts.append(make_feasible(h_rand, target_sum))

        n_restarts = min(int(remaining_time() / 15), 50)
        for r in range(n_restarts):
            if remaining_time() < 20:
                break
            base = starts[r % len(starts)]
            frac = r / max(n_restarts, 1)
            scale = 0.10 * (1.0 - frac) + 0.015
            h_try = base + rng.normal(0, scale, n)
            h_try = make_feasible(h_try, target_sum)
            try:
                res = minimize(obj, h_try, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
                if res.fun < best_c5 - 1e-13:
                    best_h = make_feasible(res.x, target_sum)
                    best_c5, _ = compute_c5(best_h)
            except Exception:
                pass

    # Phase 3: Coordinate refinement on active lags
    if remaining_time() > 20:
        for sweep in range(50):
            if remaining_time() < 25:
                break
            improved = False
            _, vals = compute_c5(best_h)
            max_val = np.max(vals)
            active_mask = vals > max_val - 0.005
            for i in range(n):
                if remaining_time() < 25:
                    break
                orig = best_h[i]
                for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    h_try = best_h.copy()
                    h_try[i] = np.clip(orig + d, 0.0, 1.0)
                    h_try = make_feasible(h_try, target_sum)
                    c5_try, _ = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
            if not improved:
                break

    # Phase 4: Final SLSQP polish
    if remaining_time() > 20:
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            if res.fun < best_c5 - 1e-13:
                best_h = make_feasible(res.x, target_sum)
                best_c5, _ = compute_c5(best_h)
        except Exception:
            pass

    best_h = make_feasible(best_h, target_sum)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
