def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def obj_and_grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    f_val = float(np.max(corr) * dx)
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    if lag > 0:
        grad = np.ones(n) * dx
        grad[:n - lag] -= x[lag:] * dx
        grad[lag:] -= x[:n - lag] * dx
    elif lag < 0:
        k = -lag
        grad = np.ones(n) * dx
        grad[k:] -= x[:n - k] * dx
        grad[:n - k] -= x[k:] * dx
    else:
        grad = (1.0 - 2.0 * x) * dx
    return f_val, grad

def resample_h(h_old, n_new):

    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)

    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n_new)

def fix_sum(h, n):
