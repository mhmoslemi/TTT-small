    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c * dx

def project_simplex(h, S):
    n = len(h)
    if n == 0:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def c5_subgrad(h, k_idx):
    """Subgradient of max_k c[k] * dx at correlation index k_idx."""
    n = len(h)
    dx = 2.0 / n
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # ∂/∂h[j]: (1-h[j-k]) for k<=j<n, minus h[j+k] for 0<=j<n-k
        if n - k > 0:
            grad[k:n] += 1.0 - h[0:n-k]
            grad[0:n-k] -= h[k:n]
    else:
        m = -k
        if n - m > 0:
            grad[m:n] += 1.0 - h[0:n-m]
            grad[0:n-m] -= h[m:n]
    return grad * dx

def lse_objective(h, beta=40.0):
    """Smooth approximation of max correlation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted -= shifted.max()  # numerical stability
    return c_max + np.log(np.exp(shifted).sum()) / beta

def lse_gradient(h, beta=40.0):
    """Gradient of LSE objective."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    # Weighted sum of individual gradients
    # For efficiency, compute as weighted correlate
    # Actually, let's just do it the brute force way for small n
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        if w[k_idx] > 1e-10:
            grad += w[k_idx] * c5_subgrad(h, k_idx)
    return grad
