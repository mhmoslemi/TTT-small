import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if beta < 1:
        return m
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def project_to_simplex_box(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Add uniformly where possible
        n = len(h)
        step = diff / n
        h = h + step
        h = np.clip(h, 0, 1)
        # Iterate
        for _ in range(50):
            diff = target_sum - h.sum()
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def local_binary_search(h, target_sum, max_iter=100):
    """Try setting each coordinate to 0 or 1, keep improvement"""
    n = len(h)
    c5 = compute_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for val in [0.0, 1.0]:
                if h[i] == val:
                    continue
                old = h[i]
                h[i] = val
                h = project_to_simplex_box(h, target_sum)
                new_c5 = compute_c5(h)
                if new_c5 < c5 - 1e-8:
                    c5 = new_c5
                    improved = True
                else:
                    h[i] = old
                    h = project_to_simplex_box(h, target_sum)
                    # Re-project
                    h = project_to_simplex_box(h, target_sum)
        if not improved:
            break
    return h, c5
