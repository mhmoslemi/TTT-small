import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, k, n):
    s = k - (n - 1)
    i_lo = max(0, -s)
    i_hi = min(n, n - s)
    grad = np.zeros(n)
    grad[i_lo:i_hi] = 1.0
    if s >= 0:
        lo = max(i_lo, 0); hi = min(i_hi, n - s)
        if hi > lo: grad[lo:hi] -= h[lo+s:hi+s]
        lo2 = max(i_lo, s); hi2 = min(i_hi, n)
        if hi2 > lo2: grad[lo2:hi2] -= h[lo2-s:hi2-s]
    else:
        t = -s
        lo = max(i_lo, t); hi = min(i_hi, n)
        if hi > lo: grad[lo:hi] -= h[lo-t:hi-t]
        lo2 = max(i_lo, 0); hi2 = min(i_hi, n - t)
        if hi2 > lo2: grad[lo2:hi2] -= h[lo2+t:hi2+t]
    return grad

def lse_grad_weights(corr, temp):
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    return w

        w = lse_grad_weights(corr, temp)
