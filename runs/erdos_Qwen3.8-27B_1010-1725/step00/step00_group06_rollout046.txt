            h[i] = np.clip(h[i], 0.0, 1.0)
            # Fix sum
            diff = n / 2.0 - np.sum(h)
            if abs(diff) > 1e-15:
                mask = (h < 1.0) if diff > 0 else (h > 0.0)
                if mask.any():
                    j2 = np.where(mask)[0]
                    j2 = j2[j2 != i]
                    if len(j2) > 0:
                        share = diff / len(j2)
                        h[j2] = np.clip(h[j2] + share, 0.0, 1.0)
        if abs(np.sum(h) - n / 2.0) > 1e-10:
            h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(95, 0.5)
    n = len(h_init)
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)
    print(f"Initial: n={n}, C5={best_c5:.6f}")

    # Phase 1: Fourier optimization
    for K in [5, 8, 12]:
        if time.time() - t0 > budget_s * 0.2:
            break
        fcoeffs = extract_fourier(best_h, n, K)
        bounds = [(-0.4, 0.4)] * (2 * K)
        res = minimize(lambda p: obj_fourier(p, n, K, 0.004), fcoeffs,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-8})
        h_new = project_feasible(fourier_to_h(res.x, n, K), n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            print(f"Fourier K={K}: C5={best_c5:.6f}")

    # Phase 2: Direct SLSQP
    if time.time() - t0 < budget_s * 0.5:
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                        'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        res = minimize(lambda x: obj_direct(x, n, 0.003), best_h, method='SLSQP',
                       bounds=bnds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            print(f"SLSQP: C5={best_c5:.6f}")

    # Phase 3: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.6:
        best_h = coord_descent_polish(best_h.copy(), n, 0.002, max_rounds=3)
        best_c5 = compute_c5(best_h, n)
        print(f"CoordDesc: C5={best_c5:.6f}")

    # Phase 4: Multi-start SLSQP
    n_iter = 0
    while time.time() - t0 < budget_s * 0.8:
        n_iter += 1
        scale = 0.03 * max(1.0, 1.0 / (1.0 + 0.1 * n_iter))
        pert = best_h + np.random.randn(n) * scale
        pert = project_feasible(pert, n)
        def obj_p(x):
            return obj_direct(x, n, 0.002)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                        'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        res = minimize(obj_p, pert, method='SLSQP', bounds=bnds,
                       constraints=constraints, options={'maxiter': 150, 'ftol': 1e-12})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            if n_iter % 5 == 0:
                print(f"MultiStart {n_iter}: C5={best_c5:.6f}")

    # Polish best with coord descent
    if time.time() - t0 < budget_s - 20:
        best_h = coord_descent_polish(best_h.copy(), n, 0.001, max_rounds=3)
        best_c5 = compute_c5(best_h, n)

    # Phase 5: Try nearby resolutions
    if time.time() - t0 < budget_s - 30:
        for n_try in [n + 10, n - 10, n + 23]:
            if time.time() - t0 > budget_s - 25:
                break
            if n_try < 20 or n_try % 2 == 1 and n_try < 100:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_int = project_feasible(h_int, n_try)
            c5_int = compute_c5(h_int, n_try)
            if c5_int < best_c5 - 0.0001:
                K2 = 8
                fc = extract_fourier(h_int, n_try, K2)
                res2 = minimize(lambda p: obj_fourier(p, n_try, K2, 0.004), fc,
                               method='L-BFGS-B', bounds=[(-0.4, 0.4)]*(2*K2),
                               options={'maxiter': 400, 'ftol': 1e-13})
                h_new = project_feasible(fourier_to_h(res2.x, n_try, K2), n_try)
                c5_new = compute_c5(h_new, n_try)
                if c5_new < best_c5:
                    best_h, best_c5, n = h_new, c5_new, n_try
                    print(f"Res {n_try}: C5={best_c5:.6f}")

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
