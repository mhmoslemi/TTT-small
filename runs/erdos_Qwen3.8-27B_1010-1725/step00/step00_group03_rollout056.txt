import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = float(h.min()) - 2.0, float(h.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_optimize(h, n, max_iters, deadline):
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = evaluate_c5(h, n)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = 2.0 + 60.0 * frac * frac
        corr = compute_corr(h, n)
        max_c = float(corr.max())
        exp_v = np.exp(tau * dx * (corr - max_c))
        s = float(exp_v.sum())
        if s < 1e-300:
            break
        prob = exp_v / s
        grad = np.ones(n)
        for d in range(-(n - 1), n):
            w = prob[d + n - 1]
            if w < 1e-13:
                continue
            if d >= 0:
                if d < n:
                    grad[:n - d] -= w * h[d:]
                    grad[d:] -= w * h[:n - d]
            else:
                dd = -d
                if dd < n:
                    grad[dd:] -= w * h[:n - dd]
                    grad[:n - dd] -= w * h[dd:]
        lr = 0.025 / (1.0 + 0.0008 * it)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        h = h_new
        if it % 150 == 0:
            c5 = evaluate_c5(h, n)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = evaluate_c5(h, n)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def subgrad_polish(h, n, max_iters, deadline):
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = evaluate_c5(h, n)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        max_idx = int(np.argmax(corr))
        d = max_idx - (n - 1)
        grad = np.ones(n)
        if d >= 0:
            if d < n:
                grad[:n - d] -= h[d:]
                grad[d:] -= h[:n - d]
        else:
            dd = -d
            if dd < n:
                grad[dd:] -= h[:n - dd]
                grad[:n - dd] -= h[dd:]
        lr = 0.004 / (1.0 + 0.003 * it)
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        h = h_new
        c5 = evaluate_c5(h, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def coordinate_descent(h, n, max_iters, deadline, rng):
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = evaluate_c5(h, n)
    best_h = h.copy()
    current_c5 = best_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        idx = rng.randint(n)
        old_val = h[idx]
        candidates = [0.0, 1.0, max(0, old_val - 0.1), min(1, old_val + 0.1)]
        if old_val != 0.0:
            candidates.append(0.5 * old_val)
        if old_val != 1.0:
            candidates.append(0.5 * (1.0 + old_val))
        best_val = old_val
        best_local = current_c5
        for cv in candidates:
            h[idx] = cv
            rem_sum = target_sum - cv
            others = h.copy()
            others[idx] = 0.0
            # Redistribute: project others to sum to rem_sum while keeping h[idx]=cv
            others = np.clip(others, 0, 1)
            # Scale to fix sum approximately
            osum = others.sum()
            if osum > 1e-10:
                others = project_box_sum(others, rem_sum)
            h_test = h.copy()
            h_test[idx] = cv
            h_test[np.arange(n) != idx] = others
            c5_test = evaluate_c5(h_test, n)
            if c5_test < best_local:
                best_local = c5_test
                best_val = cv
                best_h_test = h_test.copy()
        if best_val != old_val:
            h = best_h_test
            current_c5 = best_local
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, h.copy()
        else:
            h[idx] = old_val
    return best_h, best_c5
