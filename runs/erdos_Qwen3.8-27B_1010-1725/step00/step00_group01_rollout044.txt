        bounds = [(0.0, 1.0)] * n
        maxiter = max(300, int(tl() / (2.0 * len(grid_sizes))))

        try:
            res = minimize(
                obj, h0, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
            )
            h_opt = project_sum(res.x, n)
            c_opt = c5_value(h_opt)
            if c_opt < c0:
                c0 = c_opt
                h0 = h_opt
        except Exception:
            pass

        if c0 < best_c5:
            best_h = h0.copy()
            best_c5 = c0
            best_n = n

    if tl() > 40:
        n = best_n
        h_cur = best_h.copy()
        obj = make_objective(n)
        cons = [{'type': 'eq', 'fun': make_sum_constraint(n)}]
        bounds = [(0.0, 1.0)] * n
        maxiter = int(max(200, tl() / 2.5))
        try:
            res = minimize(
                obj, h_cur, method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
            )
            h_opt = project_sum(res.x, n)
            c_opt = c5_value(h_opt)
            if c_opt < best_c5:
                best_h = h_opt
                best_c5 = c_opt
        except Exception:
            pass

    if tl() > 25:
        n = best_n
        alpha_vals = [30.0, 80.0, 200.0]
        for alpha in alpha_vals:
            if tl() < 8:
                break
            h_cur = best_h.copy()
            sobj = smooth_max_obj(n, alpha)
            cons = [{'type': 'eq', 'fun': make_sum_constraint(n)}]
            bounds = [(0.0, 1.0)] * n
            maxiter = max(100, int(tl() / 4))
            try:
                res = minimize(
                    sobj, h_cur, method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
                )
                h_opt = project_sum(res.x, n)
                c_opt = c5_value(h_opt)
                if c_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c_opt
            except Exception:
                pass

    if tl() > 15:
        n = best_n
        for attempt in range(30):
            if tl() < 3:
                break
            h_try = best_h + rng.normal(0, 0.015, size=n)
            h_try = project_sum(h_try, n)
            c_try = c5_value(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try

    if tl() > 10:
        n = best_n
        for attempt in range(20):
            if tl() < 3:
                break
            idx = rng.choice(n, size=max(2, n // 8), replace=False)
            h_try = best_h.copy()
            h_try[idx] = rng.uniform(0, 1, size=len(idx))
            h_try = project_sum(h_try, n)
            c_try = c5_value(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try

    best_c5 = c5_value(best_h)
    return (best_h, best_c5, best_n)
