def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    ones = np.ones(n)
    corr = np.correlate(h, ones - h, mode="full")
    return np.max(corr) * dx

def subgradient_at_worst_shift(h):
    """Compute subgradient of C5 at current h using worst shift."""
    n = len(h)
    dx = 2.0 / n
    ones = np.ones(n)
    corr = np.correlate(h, ones - h, mode="full")
    k_worst = np.argmax(corr)
    # k_worst corresponds to shift s = k_worst - (n-1)
    s = k_worst - (n - 1)
    
    grad = np.zeros(n)
    # For valid i where both i and i-s are in [0, n-1]:
    # ∂f_s/∂h[j] = (1-h[j-s]) - h[j+s] (with bounds)
    for j in range(n):
        g = 0.0
        # Term from h[i=j]: (1 - h[j-s]) if j-s is valid
        if 0 <= j - s < n:
            g += (1.0 - h[j - s])
        # Term from h[i=j+s]: -h[j+s] if j+s is valid (as -h[i-s] = -h[j])
        if 0 <= j + s < n:
            g += -h[j + s]
        grad[j] = g
    return grad * dx, k_worst

def project_onto_constraint(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    diff = target - current_sum
    if abs(diff) < 1e-10:
        return h
    
    # Find non-saturated elements
    if diff > 0:
        # Need to increase sum: increase elements < 1
        mask = h < 1.0 - 1e-10
    else:
        # Need to decrease sum: decrease elements > 0
        mask = h > 1e-10
    
    if not np.any(mask):
        return np.clip(h, 0, 1)
    
    count = np.sum(mask)
    adj = diff / count
    h[mask] += adj
    h = np.clip(h, 0.0, 1.0)
    
    # If still not exact, do a second pass
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = h < 1.0 - 1e-10
        else:
            mask2 = h > 1e-10
        if np.any(mask2):
            h[mask2] += diff2 / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    
    return h

def compute_subgradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    # h[j] as h[i=j]: valid if max(0,s) <= j <= min(n-1, n-1+s)
    lo1 = max(0, s)
    hi1 = min(n-1, n-1+s)
    # h[j] as h[i-s=j] i.e. i=j+s: valid if max(0,s) <= j+s <= min(n-1,n-1+s)
    # i.e., max(0,s)-s <= j <= min(n-1,n-1+s)-s
    lo2 = max(0, s) - s
    hi2 = min(n-1, n-1+s) - s
    # Combined: valid range for i is [lo1, hi1]
    # Term 1: for j in [lo1, hi1], add (1-h[j-s]) to grad[j]
    if lo1 <= hi1:
        idx1 = np.arange(lo1, hi1+1)
        grad[idx1] += (1.0 - h[idx1 - s])
    # Term 2: for j in [lo2, hi2] (and j in [0,n-1]), add -h[j+s] to grad[j]
    lo2b = max(0, lo2)
    hi2b = min(n-1, hi2)
    if lo2b <= hi2b:
        idx2 = np.arange(lo2b, hi2b+1)
        grad[idx2] += -h[idx2 + s]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, np.ones(n) - h, mode="full")
    return float(np.max(corr)) * dx

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n - 1, n - 1 + s)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        g[idx] += 1.0 - h[idx - s]
    lo2 = max(0, -s)
    hi2 = min(n - 1, n - 1 - s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 + s]
    return g

def compute_subgrad(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, np.ones(n) - h, mode="full")
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    vals = corr[sorted_idx] - corr[sorted_idx].max()
    weights = np.exp(vals * 2.0)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, sorted_idx):
        s = int(k) - (n - 1)
        grad += w * grad_for_shift(h, s)
    return grad * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def subgrad_descent(h0, max_iter=10000, time_limit=30):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = 0.005
    momentum = 0.9
    t0 = time.time()
    no_improve = 0
    patience = 300

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad = compute_subgrad(h, top_k=5)
        vel = momentum * vel - lr * grad
        h_new = project(h + vel, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
            momentum = min(momentum + 0.001, 0.95)
        else:
            no_improve += 1
        h = h_new
        lr *= 0.9995
        lr = max(lr, 1e-7)
        if no_improve > patience:
            if no_improve > patience * 3:
                break
            # Restart velocity
            vel *= 0.1
            lr = max(lr * 2.0, 1e-5)
            no_improve = patience

    return best_h, best_c5

def make_binary_init(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        remaining = np.setdiff1d(np.arange(n), idx)
        h[remaining[0]] = 0.5
    else:
        # already sum = n/2
        pass
    return h

def make_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    freqs = rng.uniform(0.5, 4.0, 3)
    phases = rng.uniform(0, 2 * np.pi, 3)
    x = np.linspace(0, 1, n)

    h += 0.15 * np.sin(2 * np.pi * freqs[1] * x + phases[1])
    h += 0.10 * np.sin(2 * np.pi * freqs[2] * x + phases[2])
    return project(h, n)
