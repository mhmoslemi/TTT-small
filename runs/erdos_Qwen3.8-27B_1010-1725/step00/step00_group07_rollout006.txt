                x = best_h.copy()
                x[i] = np.clip(x[i] + sign * d, 0.0, 1.0)
                x[j] = np.clip(x[j] - sign * d, 0.0, 1.0)
                c = compute_c5(x)
                if c < best_c5:
                    best_c5, best_h = c, x
                    found = True
            if not found:
                break

    # Phase 4: Resolution increase
    if time.time() - t0 < budget_s * 0.93:
        for mult in [2, 3, 4]:
            n_new = n * mult
            if n_new > 2000:
                break
            if time.time() - t0 > budget_s * 0.93:
                break
            t_old = np.linspace(0.0, 2.0, n)
            t_new = np.linspace(0.0, 2.0, n_new)
            h_interp = np.interp(t_new, t_old, best_h)
            h_interp = repair_h(h_interp)
            c_interp = compute_c5(h_interp)
            x = _slsqp(h_interp, n_new, 400, 1e-11)
            if x is not None:
                c = compute_c5(x)
                if c < best_c5:
                    best_c5, best_h = c, x
                    n = n_new
                    # Second polish at new resolution
                    x2 = _slsqp(best_h, n, 300, 1e-12)
                    if x2 is not None:
                        c2 = compute_c5(x2)
                        if c2 < best_c5:
                            best_c5, best_h = c2, x2

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        x = _slsqp(best_h, len(best_h), 500, 1e-13)
        if x is not None:
            c = compute_c5(x)
            if c < best_c5:
                best_c5, best_h = c, x

    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
