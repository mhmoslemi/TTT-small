def project_to_sum(h, target):
    h=np.asarray(h,float).ravel()
    n=h.size
    if n==0: return h
    if not np.all(np.isfinite(h)): h=np.full(n,0.5)
    target=float(target)
    if target<0: target=0.0
    if target>n: target=float(n)
    y=np.clip(h, -5.0, 5.0)
    low=float(np.min(y))-1.0
    high=float(np.max(y))+1.0
    lam=(low+high)*0.5
    for _ in range(55):
       lam=(low+high)*0.5
       s=float(np.sum(np.clip(y-lam,0.0,1.0)))
       if s>target: low=lam
       else: high=lam
    h=np.clip(y-lam,0.0,1.0)
    diff=target-float(np.sum(h))
    if abs(diff)>1e-13:
       for _ in range(5):
          mask=(h>1e-12)&(h<1.0-1e-12)
          cnt=int(np.sum(mask))
          if cnt==0: break

          h2=h.copy()
          h2[mask]=np.clip(h[mask]+add,0.0,1.0)

import time
import math
import numpy as np
from scipy.optimize import minimize

_G_T = 0.1
_G_D = 1.0
_G_TARGET = 0.0
_G_ONES = np.ones(1, dtype=float)


def _set_state(n, t):
    global _G_T, _G_D, _G_TARGET, _G_ONES
    n = int(n)
    _G_T = float(t)
    _G_D = 2.0 / float(n)
    _G_TARGET = 0.5 * float(n)
    _G_ONES = np.ones(n, dtype=float)


def project_to_sum(h, target):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return h
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)
    y = np.clip(h, -5.0, 5.0)
    low = float(np.min(y)) - 1.0
    high = float(np.max(y)) + 1.0
    lam = 0.5 * (low + high)
    for _ in range(55):
        lam = 0.5 * (low + high)
        s = float(np.sum(np.clip(y - lam, 0.0, 1.0)))
        if s > target:
            low = lam
        else:
            high = lam
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        for _ in range(5):
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            add = diff / float(cnt)
            h2 = h.copy()
            h2[mask] = np.clip(h[mask] + add, 0.0, 1.0)
            s2 = float(np.sum(h2))
            if abs(s2 - target) < abs(diff):
                h = h2
                diff = target - s2
            else:
                break
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        v = h[idx] + diff
        if 0.0 <= v <= 1.0:
            h[idx] = v
    return h


def c5_eval(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / float(n))


def calc_c_raw(h):
    n = h.size
    om = 1.0 - h
    c = np.empty(2 * n - 1, dtype=float)
    base = n - 1
    for t in range(1, n):
        m = n - t
        c[base - t] = float(np.dot(h[t:], om[:m]))
    c[base] = float(np.dot(h, om))
    for s in range(1, n):
        m = n - s
        c[base + s] = float(np.dot(h[:m], om[s:s + m]))
    return c


def calc_grad_raw(h, c, active_idx, weights):
    n = h.size
    om = 1.0 - h
    grad = np.zeros(n, dtype=float)
    base = n - 1
    for idx, w in zip(active_idx, weights):
        if w == 0.0:
            continue
        s = int(idx - base)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            grad[:m] += w * om[s:s + m]
            grad[s:s + m] -= w * h[:m]
        else:
            t = -s
            m = n - t
            if m <= 0:
                continue
            grad[t:t + m] += w * om[:m]
            grad[:m] -= w * h[t:t + m]
    return grad


def calc_cg(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e6, np.zeros(n, dtype=float)
    c = calc_c_raw(h)
    if not np.all(np.isfinite(c)):
        return 1e6, np.zeros(n, dtype=float)
    M = float(np.max(c))
    T = _G_T
    if T <= 1e-9:
        tol = max(1e-10, 1e-7 * max(1.0, M))
        active = c >= (M - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        idx = np.nonzero(active)[0]
        wts = np.full(idx.size, 1.0 / float(idx.size), dtype=float)
        F = M
    else:
        u = np.exp((c - M) / T)
        U = float(np.sum(u))
        if U <= 0.0 or not math.isfinite(U):
            idx = np.array([int(np.argmax(c))], dtype=int)
            wts = np.array([1.0], dtype=float)
            F = M
        else:
            act = u > 1e-12
            if not act.any():
                idx = np.array([int(np.argmax(c))], dtype=int)
                wts = np.array([1.0], dtype=float)
            else:
                idx = np.nonzero(act)[0]
                wts = u[idx] / U
                ws = float(np.sum(wts))
                if ws > 0.0:
                    wts = wts / ws
                else:
                    wts = np.full(idx.size, 1.0 / float(idx.size), dtype=float)
            F = M + T * math.log(U)
    grad = calc_grad_raw(h, c, idx, wts)
    return F, grad


def obj_func(h):
    F, g = calc_cg(h)
    return float(F * _G_D)


def jac_func(h):
    F, g = calc_cg(h)
    return g * _G_D


def con_fun(h):
    return float(np.sum(np.asarray(h, dtype=float).ravel()) - _G_TARGET)


def con_jac(h):
    return _G_ONES.reshape(1, -1).copy()


def slsqp_step(h, tlist, base_iter, deadline):
    n = h.size
    target = 0.5 * float(n)
    h = project_to_sum(h, target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": con_fun, "jac": con_jac}]
    for T in tlist:
        if time.time() > deadline - 20.0:
            break
        if T >= 0.2:
            mi = int(base_iter)
        elif T >= 0.05:
            mi = int(0.7 * base_iter)
        elif T >= 0.01:
            mi = int(0.5 * base_iter)
        else:
            mi = int(0.3 * base_iter)
        mi = max(15, mi)
        _set_state(n, T)
        try:
            res = minimize(
                obj_func, h,
                method="SLSQP",
                jac=jac_func,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": mi, "ftol": 1e-9, "disp": False}
            )
            if res.x is not None:
                h = project_to_sum(res.x, target)
        except Exception:
            pass
    return project_to_sum(h, target)


def replicate_to(parent, n):
    m = parent.size
    if m == n:
        return parent.copy()
    if n % m == 0:
        return np.repeat(parent, n // m)
    pp = np.linspace(0.5, m - 0.5, m)
    pn = np.linspace(0.5, n - 0.5, n)
    return np.interp(pn, pp, parent)


def noisy_start(base, rng, scale):
    n = base.size
    if n <= 1:
        return project_to_sum(base, 0.5 * float(n))
    noise = rng.normal(0.0, scale, n)
    noise = 0.6 * noise + 0.2 * np.roll(noise, 1) + 0.2 * np.roll(noise, -1)
    return project_to_sum(base + noise, 0.5 * float(n))


def highfreq_start(base, amp):
    n = base.size
    if n <= 1:
        return project_to_sum(base, 0.5 * float(n))
    sign = np.where(np.arange(n) % 2 == 0, 1.0, -1.0)
    return project_to_sum(base + amp * sign, 0.5 * float(n))


def wave_start(ref, n, rng, amp):
    base = replicate_to(ref, n)
    x = np.arange(n, dtype=float)
    k = int(rng.integers(1, 4))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    h = base + amp * np.sin(2.0 * np.pi * float(k) * x / float(max(1, n)) + phase)
    return project_to_sum(h, 0.5 * float(n))


def parent_from_initial():
    p = globals().get("initial_h_values", None)
    if p is not None:
        try:
            arr = np.asarray(p, dtype=float).ravel()
        except Exception:
            arr = None
        if arr is not None and arr.size > 0:
            arr = np.where(np.isfinite(arr), arr, 0.5)
            arr = np.clip(arr, 0.0, 1.0)
            return project_to_sum(arr, 0.5 * float(arr.size))
    return project_to_sum(np.full(45, 0.5), 22.5)


def refine_active(h, maxit, deadline):
    n = h.size
    target = 0.5 * float(n)
    h = project_to_sum(h, target)
    val = c5_eval(h)
    if not math.isfinite(val):
        return h, val
    step = 0.02
    for _ in range(int(maxit)):
        if time.time() > deadline - 5.0:
            break
        c = calc_c_raw(h)
        M = float(c.max())
        active = np.nonzero(c >= M - max(1e-8, 1e-7 * M))[0]
        if active.size == 0:
            active = np.array([int(c.argmax())], dtype=int)
        wts = np.full(active.size, 1.0 / float(active.size), dtype=float)
        grad = calc_grad_raw(h, c, active, wts)
        grad = grad - float(np.mean(grad))
        bestv = val
        bestd = None
        for trial in (step, 0.5 * step, 0.25 * step, 0.125 * step, 0.0625 * step):
            hp = project_to_sum(h - trial * grad, target)
            f = c5_eval(hp)
            if math.isfinite(f) and f < bestv - 1e-13:
                bestv = f
                bestd = hp
                break
        if bestd is None:
            step *= 0.5
            if step < 1e-7:
                break
        else:
            h = bestd
            val = bestv
    return h, float(val)


def run(seed=42, budget_s=1100, **kwargs):

    base_n = parent.size if parent.size > 0 else 45

    for mult in (1, 2, 4, 8):

    for n in sizes:

        if math.isfinite(v) and v < best_c5:

        starts.append(wave_start(best_h if best_h.size > 0 else base, n, rng, 0.20))

            tlist = (0.4, 0.15, 0.05, 0.015, 0.005)

            tlist = (0.4, 0.15, 0.05, 0.015)

    best_h = project_to_sum(best_h, 0.5 * float(best_h.size))
    best_c5 = c5_eval(best_h)
    if not math.isfinite(best_c5):
        best_h = project_to_sum(parent, 0.5 * float(parent.size))
        best_c5 = c5_eval(best_h)
    return best_h, float(best_c5), int(best_h.size)
