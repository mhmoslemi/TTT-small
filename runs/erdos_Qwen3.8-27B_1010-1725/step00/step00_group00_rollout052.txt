import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    vals = corr * dx
    return float(logsumexp(beta * vals) / beta)

def smooth_obj_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    one_minus_x = 1 - x
    corr = np.correlate(x, one_minus_x, mode='full')
    vals = corr * dx
    lse = logsumexp(beta * vals)
    p = np.exp(beta * vals - lse)
    f = float(lse / beta)
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            c = 0.0
            ni = i - k
            if 0 <= ni < n:
                c += (1.0 - x[ni])
            ni2 = i + k
            if 0 <= ni2 < n:
                c -= x[ni2]
            s += p[m] * c
        grad[i] = dx * s
    return f, grad

def project_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0)
        else:
            mask = h < 1
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1)
    return h

def make_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = np.sum(h) - target
    if diff > 0:
        # reduce
        order = np.argsort(h)[::-1]  # largest first
        remaining = diff
        for i in order:
            reduce = min(remaining, h[i])
            h[i] -= reduce
            remaining -= reduce
            if remaining < 1e-12:
                break
    else:
        # increase
        order = np.argsort(h)  # smallest first
        remaining = -diff
        for i in order:
            increase = min(remaining, 1 - h[i])
            h[i] += increase
            remaining -= increase
            if remaining < 1e-12:

    return h

            if res.success or res.fun < smooth_obj(x0, beta):

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:

        except Exception as e:

            if time_left() < 60:
                break
            x_old = np.linspace(1.0 / n, 2.0 - 1.0 / n, n)
            x_new = np.linspace(1.0 / n_try, 2.0 - 1.0 / n_try, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = make_feasible(h_interp)
            bounds_t = [(0.0, 1.0)] * n_try
            h_new = optimize_continuous(h_interp, bounds_t, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if time_left() > 30:
                h_bin, c5_bin = binary_local_search(h_interp, max_iter=3000)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()

    # Phase 4: Final polish on best solution
    if time_left() > 40:
        n_f = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_f
        beta_f = 100.0
        h_new = optimize_continuous(best_h, bounds_f, beta_f, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = make_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))

def args_get_maxiter(restart):
    if restart <= 2:
        return 500
    return 300
