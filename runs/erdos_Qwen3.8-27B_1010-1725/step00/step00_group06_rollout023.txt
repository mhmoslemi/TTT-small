import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project onto 0<=h<=1, sum(h)=n/2 using iterative water-filling"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        n_free = np.sum(free)
        if n_free == 0:
            break
        h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_objective(x, n):
    return compute_c5(x)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent
    global initial_h_values
    if initial_h_values is not None:
        h_parent = np.asarray(initial_h_values, dtype=float)
    else:
        n_parent = 95
        h_parent = np.array([0.5]*n_parent)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    best_n = len(h_parent)
    
    n = best_n
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    sum_constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                       'jac': lambda x: np.ones(n)}]
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Local SLSQP from parent
    h_start = project_to_feasible(h_parent, n).copy()
    res = minimize(slsqp_objective, h_start, method='SLSQP',
                   bounds=bounds, constraints=sum_constraint,
                   options={'maxiter': 500, 'ftol': 1e-10})
    if res.success:
        h_new = project_to_feasible(res.x, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    
    if elapsed() > budget_s * 0.15:
        return (best_h, best_c5, best_n)
    
    # Phase 2: Multiple random perturbations + SLSQP
    for i in range(50):
        if elapsed() > budget_s * 0.6:

        # Generate perturbation

        # Also try threshold perturbations

            h_pert[mask] = 1.0 - h_pert[mask]

                best_h = h_new.copy()
    
    if elapsed() > budget_s * 0.75:
        return (best_h, best_c5, best_n)
    
    # Phase 3: Try higher resolution
    n2 = min(2 * best_n, 400)
    if n2 > best_n and elapsed() < budget_s * 0.5:

        h_up = np.interp(x_new, x_old, best_h)

        res = minimize(slsqp_objective, h_up, method='SLSQP',
                      bounds=bounds2, constraints=sc2,
                      options={'maxiter': 300, 'ftol': 1e-9})
        h_new = project_to_feasible(res.x, n2)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    return (best_h, best_c5, best_n)
