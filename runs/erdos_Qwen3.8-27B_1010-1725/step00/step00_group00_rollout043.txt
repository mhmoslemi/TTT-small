import time
import numpy as np
from scipy.optimize import minimize

_DX = 0.0
_T = 1.0


def _force_sum(h, S):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(12):
        diff = float(S) - float(h.sum())
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            step = min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            step = min(-diff, h[idx])
        if step <= 1e-15:
            break
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def _feas_project(y, S=None):
    y = np.asarray(y, dtype=float).copy()
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    n = y.size
    if n == 0:
        return y
    if S is None:
        S = 0.5 * n
    S = float(S)
    if S < 0.0:
        S = 0.0
    if S > float(n):
        S = float(n)
    lo = float(np.min(y) - 1.0)
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(y + mid, 0.0, 1.0)
        if v.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)


def _corr(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def _exact(h, n=None):
    if n is None:
        n = int(h.size)
    if n <= 0:
        return 0.0
    c = _corr(h)
    return float(np.max(c) * (2.0 / float(n)))


def _prepare(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size < n:
            h = np.concatenate((h, np.zeros(n - h.size, dtype=float)))
        else:
            h = h[:n]
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    S = 0.5 * float(n)
    h = _feas_project(h, S)
    h = _force_sum(h, S)
    h = np.clip(h, 0.0, 1.0)
    return h, _exact(h, n)


def _soft_value(c, t):
    if t <= 1e-12:
        return float(c.max())
    z = c / t
    zmax = float(z.max())
    expz = np.exp(z - zmax)
    s = float(expz.sum())
    if s <= 0.0 or not np.isfinite(s):
        return zmax * t
    return float(t * (zmax + np.log(s)))


def _obj(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = _corr(x)
    return _DX * _soft_value(c, _T)


def _obj_jac(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    n = x.size
    c = _corr(x)
    t = _T if _T > 1e-8 else 1e-8
    z = c / t
    zmax = float(z.max())
    expz = np.exp(z - zmax)
    p = expz / float(expz.sum())
    thresh = max(1e-4, float(p.max()) * 1e-3)
    active = np.where(p > thresh)[0]
    if active.size == 0 or active.size > 256:
        active = np.argsort(-p)[:256]
    g = np.zeros(n, dtype=float)
    for idx in active:
        w = float(p[idx])
        k = int(idx) - (n - 1)
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            a = n - k
            g[:a] += w * (1.0 - x[k:])
            g[k:] -= w * x[:-k]
        else:
            ell = -k
            g[ell:] += w * (1.0 - x[:-ell])
            g[:n - ell] -= w * x[ell:]
    return _DX * g


def _exact_obj(x):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    return float(_DX * np.max(_corr(x)))


def _sum_con(x):
    return float(np.sum(x) - 0.5 * float(x.size))


def _sum_jac(x):
    return np.ones(x.shape, dtype=float)


def _smooth_step(start_h, n, t, maxiter):
    global _DX, _T
    h, _ = _prepare(start_h, n)
    _DX = 2.0 / float(n)
    _T = float(t)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_con, "jac": _sum_jac}]
    try:
        res = minimize(
            _obj,
            h,
            jac=_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"ftol": 1e-9, "maxiter": int(max(maxiter, 1)), "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
        if x.size == n and bool(np.all(np.isfinite(x))):
            return _prepare(x, n)[0]
    except Exception:
        pass
    return h


def _exact_step(start_h, n, maxiter):
    global _DX
    h, _ = _prepare(start_h, n)
    _DX = 2.0 / float(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_con, "jac": _sum_jac}]
    try:
        res = minimize(
            _exact_obj,
            h,
            jac=None,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"ftol": 1e-8, "maxiter": int(max(maxiter, 1)), "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
        if x.size == n and bool(np.all(np.isfinite(x))):
            return _prepare(x, n)[0]
    except Exception:
        pass
    return h


def _resample(h_old, n_new):
    n_old = int(h_old.size)
    if n_new == n_old:
        return _feas_project(h_old, 0.5 * n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, np.asarray(h_old, dtype=float))
    return _feas_project(y, 0.5 * n_new)


def _tseq(h):
    cmax = float(np.max(_corr(h)))
    min_t = 0.03 if h.size < 100 else 0.08
    base = max(1.0, 0.45 * cmax)
    seq = []
    t = base
    for _ in range(7):
        if t < min_t:
            t = min_t
        seq.append(float(t))
        t *= 0.5
    return seq


def _to_binary(h, rng):
    n = int(h.size)
    if n % 2:
        n -= 1
        h = np.asarray(h, dtype=float)[:n]
    else:
        h = np.asarray(h, dtype=float).copy()
    S = n // 2
    scores = h + 1e-9 * rng.random(n)
    idx = np.argsort(scores)[::-1]
    b = np.zeros(n, dtype=np.int8)
    b[idx[:S]] = 1
    return b


def _bin_counts(b):
    n = int(b.size)
    p = np.zeros(max(n - 1, 0), dtype=np.int32)
    q = np.zeros(max(n - 1, 0), dtype=np.int32)
    for d in range(1, n):
        left = b[:-d]
        right = b[d:]
        p[d - 1] = np.count_nonzero((left == 1) & (right == 0))
        q[d - 1] = np.count_nonzero((left == 0) & (right == 1))
    return p, q


def _bin_site_flip(b, i, new, p, q):
    old = int(b[i])
    if old == new:
        return
    n = int(b.size)
    if old == 1:
        if i < n - 1:
            a = n - 1 - i
            p[:a] -= (1 - b[i + 1:i + 1 + a])
            q[:a] += b[i + 1:i + 1 + a]
        if i > 0:
            a = i
            rev = b[:a][::-1]
            p[:a] += rev
            q[:a] -= (1 - rev)
    else:
        if i < n - 1:
            a = n - 1 - i
            p[:a] += (1 - b[i + 1:i + 1 + a])
            q[:a] -= b[i + 1:i + 1 + a]
        if i > 0:
            a = i
            rev = b[:a][::-1]
            p[:a] -= rev
            q[:a] += (1 - rev)
    b[i] = new


def _eval_change(b, p, q, i, j):
    _bin_site_flip(b, i, 0, p, q)
    _bin_site_flip(b, j, 1, p, q)
    val = int(max(p.max(), q.max()))
    _bin_site_flip(b, j, 0, p, q)
    _bin_site_flip(b, i, 1, p, q)
    return val


def _binary_sa(b, p, q, stop_time, rng):
    n = int(b.size)
    cur = int(max(p.max(), q.max()))
    best = b.copy()
    best_cur = cur
    it = 0
    k_try = 48 if n <= 80 else 28
    while time.time() < stop_time:
        ones = np.flatnonzero(b == 1)
        zeros = np.flatnonzero(b == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        best_i = -1
        best_j = -1
        best_new = cur
        for _ in range(k_try):
            ci = cj = -1
            if rng.random() < 0.6:
                pidx = int(p.argmax())
                qidx = int(q.argmax())
                pc = int(p[pidx])
                qc = int(q[qidx])
                if pc > 0 and (qc == 0 or rng.random() < pc / float(pc + qc)):
                    d = pidx + 1
                    arr = np.flatnonzero((b[:n - d] == 1) & (b[d:] == 0))
                    if arr.size:
                        pos = int(arr[int(rng.integers(arr.size))])
                        ci = pos
                        cj = pos + d
                if ci < 0 and qc > 0:
                    d = qidx + 1
                    arr = np.flatnonzero((b[:n - d] == 0) & (b[d:] == 1))
                    if arr.size:
                        pos = int(arr[int(rng.integers(arr.size))])
                        ci = pos + d
                        cj = pos
            if ci < 0:
                ci = int(ones[int(rng.integers(ones.size))])
                cj = int(zeros[int(rng.integers(zeros.size))])
            newval = _eval_change(b, p, q, ci, cj)
            if newval < best_new:
                best_new = newval
                best_i = ci
                best_j = cj
        if best_i >= 0 and (best_new < cur or (best_new == cur and rng.random() < 0.35)):
            _bin_site_flip(b, best_i, 0, p, q)
            _bin_site_flip(b, best_j, 1, p, q)
            cur = best_new
            if cur < best_cur:
                best_cur = cur
                best = b.copy()
        it += 1
        if it % 800 == 0:
            p, q = _bin_counts(b)
            cur = int(max(p.max(), q.max()))
            if cur < best_cur:
                best_cur = cur
                best = b.copy()

            b2, cur2 = _binary_sa(b, p, q, stop, rng)
            c2 = float(cur2 * (2.0 / float(nn)))
            if c2 < best_c:
                best_c = c2
                best_n = int(nn)
                best_h = b2.astype(float)

    best_h = np.asarray(best_h, dtype=float)
    best_h = np.nan_to_num(best_h, nan=0.5, posinf=1.0, neginf=0.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _force_sum(best_h, 0.5 * float(best_n))
    best_c = _exact(best_h, int(best_n))
    return best_h, float(best_c), int(best_n)
