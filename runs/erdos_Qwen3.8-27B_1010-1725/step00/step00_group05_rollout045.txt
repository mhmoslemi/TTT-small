def compute_c5(h, dx):
    """Compute C5 = max_k corr(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_simplex_sum(h, target_sum, lo=0.0, hi=1.0):
    """Project h onto {0≤x≤1, sum(x)=target_sum}"""
    h = h.copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < hi
        else:
            mask = h > lo
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, lo, hi)
    return h

    h = 0.5 + 0.3 * rng.standard_normal(n)
    h = np.clip(h, 0, 1)

import numpy as np
import time
from scipy.optimize import minimize

_n_g = 70
_dx_g = 2.0 / 70
_tgt_g = 35.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = h.astype(float).copy()
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary_init(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def make_cont_init(n, rng):
    h = 0.5 + 0.4 * rng.standard_normal(n)
    h = np.clip(h, 0.0, 1.0)
    return project_sum(h, n / 2.0)

def make_perturb_init(h_best, n, rng, scale=0.12):
    h = h_best + scale * rng.standard_normal(n)
    h = np.clip(h, 0.0, 1.0)
    return project_sum(h, n / 2.0)

def _obj(x):
    return x[_n_g]

def _ineq(x):
    h = x[:_n_g]
    t = x[_n_g]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr * _dx_g

def _eq(x):
    return np.sum(x[:_n_g]) - _tgt_g

def coordinate_polish(h, dx, max_iter=200):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h, dx)
    for it in range(max_iter):
        improved = False
        for i in range(n):
            for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                h2 = h.copy()
                h2[i] = np.clip(h2[i] + delta, 0.0, 1.0)
                h2 = project_sum(h2, target)
                c5_new = compute_c5(h2, dx)
                if c5_new < best_c5 - 1e-12:
                    h = h2
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    target = _tgt_g

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_sum(h0, target)
    else:

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    cons = [{'type': 'ineq', 'fun': _ineq},

    restart = 0

        elif restart == 1:
