            new_h = cur_h.copy()
            new_h[st:st + bs] = 1.0 - new_h[st:st + bs]
        elif r < 0.60:
            new_h = cur_h.copy()
            npts = int(rng.integers(1, max(2, n // 4)))
            idxs = rng.choice(n, size=npts, replace=False)
            new_h[idxs] = rng.random(npts)
        elif r < 0.75:
            new_h = cur_h.copy()
            a = int(rng.integers(0, n // 2))
            b = int(rng.integers(n // 2, n))
            bs = int(rng.integers(1, min(4, n // 3)))
            a = min(a, n - bs)
            b = min(b, n - bs)
            tmp = new_h[a:a + bs].copy()
            new_h[a:a + bs] = new_h[b:b + bs]
            new_h[b:b + bs] = tmp
        elif r < 0.88:
            new_h = cur_h[::-1].copy()
            bs = int(rng.integers(1, max(2, n // 8)))
            st = int(rng.integers(0, n - bs))
            new_h[st:st + bs] = 1.0 - new_h[st:st + bs]
        else:
            new_h = binary_round(cur_h, rng)
        new_h = project_feasible(new_h)
        new_c = eval_c5(new_h)
        if new_c < cur_c or (T > T_min and rng.random() < np.exp(-(new_c - cur_c) / T)):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
        T = max(T_min, T * cool)
        it += 1
    # Try binary round at the end
    h_bin = project_feasible(binary_round(best_h, rng))
    c_bin = eval_c5(h_bin)
    if c_bin < best_c:
        best_h, best_c = h_bin, c_bin
    return best_h, best_c

def periodic_pattern(n, period, duty=0.5):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < int(period * duty):
            h[i] = 1.0
    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    if h_init is None or len(h_init) == 0:
        h_init = np.ones(45) * 0.5
    n0 = len(h_init)
    h_init = project_feasible(h_init)
    best_h = h_init.copy()
    best_c = eval_c5(h_init)

    # Phase 1: anneal at initial resolution (40% budget)
    p1_end = t0 + budget_s * 0.35
    h1, c1 = anneal(h_init, n0, rng, t0, int(p1_end))
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: try periodic patterns at initial resolution (10% budget)
    p2_end = t0 + budget_s * 0.45
    for per in range(2, n0 + 1):
        if time.time() > p2_end:
            break
        if n0 % per != 0 and n0 % per > 0:
            continue
        hp = periodic_pattern(n0, per)
        c_p = eval_c5(hp)
        if c_p < best_c:
            best_h, best_c = hp, c_p
            h2, c2 = anneal(hp, n0, rng, t0, int(p2_end))
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 3: refine at higher resolutions (45% budget)
    for n_fine in [2 * n0, 3 * n0, 5 * n0]:
        if time.time() - t0 > budget_s * 0.90:
            break
        p3_end = t0 + budget_s
        h_f = interp_h(best_h, n_fine)
        h3, c3 = anneal(h_f, n_fine, rng, t0, int(p3_end))
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Final
    best_h = project_feasible(best_h)
    final_c = eval_c5(best_h)
    if final_c > best_c:
        final_c = best_c
    return best_h, float(final_c), len(best_h)
