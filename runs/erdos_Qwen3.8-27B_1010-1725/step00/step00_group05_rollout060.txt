    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wv = w[idx]
        if wv < 1e-14:
            continue
        d = idx - (n - 1)
        wdx = wv * dx
        if d >= 0:
            if n - d > 0:
                grad[0:n-d] += wdx * (1.0 - h[d:n])
                grad[d:n] -= wdx * h[0:n-d]
        else:
            ad = -d
            if n + d > 0:
                grad[ad:n] += wdx * (1.0 - h[0:n+d])
                grad[0:n+d] -= wdx * h[ad:n]
    return obj, grad

def optimize_pg(h_init, dx, n, deadline, rng):
    h = project_to_constraint(h_init, n).copy()
    best_c5, _ = compute_c5(h, dx)
    best_h = h.copy()
    temp = 0.015
    lr = 0.2
    mom = 0.8
    v = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        obj, grad = soft_max_obj_grad(h, dx, temp)
        v = mom * v - lr * grad
        h_try = project_to_constraint(h + v, n)
        c_try, _ = compute_c5(h_try, dx)
        alpha = 1.0
        iters = 0
        while c_try > best_c5 + 0.3 * temp and alpha > 0.005 and iters < 8:
            alpha *= 0.5
            h_try = project_to_constraint(h + alpha * v, n)
            c_try, _ = compute_c5(h_try, dx)
            iters += 1
        h = h_try
        temp = max(0.0003, temp * 0.997)
        lr = min(lr * 1.005, 0.8)
        if c_try < best_c5 - 1e-8:
            best_c5 = c_try
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800:
            h = project_to_constraint(h + rng.normal(0, 0.04, n), n)
            v = np.zeros(n)
            temp = 0.02
            lr = 0.15
            mom = 0.7
            stall = 0
    return best_c5, best_h

def optimize_slsqp(h_init, dx, n, max_iter=300):
    def obj_fn(x):
        xp = project_to_constraint(x, n)
        return compute_c5(xp, dx)[0]
    cons = [LinearConstraint(np.ones(n), n / 2, n / 2)]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h_init, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
        h_out = project_to_constraint(res.x, n)
        c_out, _ = compute_c5(h_out, dx)
        return c_out, h_out
    return float('inf'), h_init

def make_init(n, trial, rng):
    if trial == 0:
        k = rng.randint(2, 8)
        x = np.linspace(0, 2.0, n)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * k * x / 2.0 + rng.random() * 6.28)
        h += 0.1 * np.sin(2 * np.pi * (k + 3) * x / 2.0 + rng.random() * 6.28)
    elif trial == 1:
        h = rng.uniform(0.15, 0.85, n)
    elif trial == 2:
        block = max(2, n // 12)
        h = np.tile([1.0] * (block // 2) + [0.0] * block, n // block + 1)[:n]
        h = h * 0.5 + 0.25
    else:
        x = np.linspace(0, 2.0, n)
        h = 0.5 + 0.4 * np.cos(np.pi * x) + 0.1 * rng.random(n)
    return project_to_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Phase 1: Optimize the provided initial construction
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    dx0 = 2.0 / n0
    h0 = project_to_constraint(h0, n0)
    c0, _ = compute_c5(h0, dx0)
    best_c5 = c0
    best_h = h0.copy()
    best_n = n0

    # Projected gradient on initial
    deadline = t0 + min(120, budget_s * 0.15)
    best_c5, best_h = optimize_pg(best_h, dx0, n0, deadline, rng)

    # SLSQP refinement on best so far
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        try:
            c_sl, h_sl = optimize_slsqp(best_h, dx0, n0, max_iter=200)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl
        except Exception:
            pass

    # More PG with tighter params
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        deadline = min(t0 + budget_s - 40, time.time() + min(150, remaining * 0.3))
        best_c5, best_h = optimize_pg(best_h, dx0, n0, deadline, rng)
        try:
            c_sl, h_sl = optimize_slsqp(best_h, dx0, n0, max_iter=200)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl
        except Exception:
            pass

    # Phase 2: Try nearby sizes
    n_candidates = sorted(set([n0 - 10, n0 - 20, n0 + 10, n0 + 20, n0 + 30, n0 + 50, 50, 80, 100, 120, 150, 200]))
    n_candidates = [n for n in n_candidates if 30 <= n <= 300 and n != n0]
    for n_try in n_candidates:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.82:
            break
        dx_t = 2.0 / n_try
        alloc = min(35, max(10, (budget_s - elapsed - 30) * 0.12))
        if alloc < 8:
            break
        for trial in range(2):
            if time.time() - t0 > budget_s * 0.82:
                break
            h_init = make_init(n_try, trial, rng)
            # Also try interpolating best_h to this size
            if trial == 0 and best_h is not None:
                x_best = np.linspace(0, 2.0, len(best_h))
                x_new = np.linspace(0, 2.0, n_try)
                h_interp = np.interp(x_new, x_best, best_h)
                h_init = project_to_constraint(h_interp, n_try)
            c_opt, h_opt = optimize_pg(h_init, dx_t, n_try, time.time() + alloc / 2.0, rng)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n_try
            # Quick SLSQP if promising
            if c_opt < best_c5 + 0.005 and c_opt - best_c5 > -0.001:
                try:
                    c_sl, h_sl = optimize_slsqp(h_opt, dx_t, n_try, max_iter=100)
                    if c_sl < best_c5:
                        best_c5 = c_sl
                        best_h = h_sl
                        best_n = n_try
                except Exception:
                    pass

    # Phase 3: Final polish
    elapsed = time.time() - t0
    if best_h is not None and elapsed < budget_s - 20:
        dx_best = 2.0 / best_n
        deadline = t0 + budget_s - 5
        best_c5, best_h = optimize_pg(best_h, dx_best, best_n, deadline, rng)
        try:
            c_sl, h_sl = optimize_slsqp(best_h, dx_best, best_n, max_iter=150)
            if c_sl < best_c5:
                best_c5 = c_sl
                best_h = h_sl
        except Exception:
            pass

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_n = 70
        best_c5, _ = compute_c5(best_h, 2.0 / 70)

    best_h = project_to_constraint(best_h, best_n)
    best_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, float(best_c5), best_n
