from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(values, temp):
    m = np.max(values)
    if temp < 1e-12:
        return m
    shifted = (values - m) / temp
    shifted = np.clip(shifted, -50, 50)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.count_nonzero(mask)
            h = np.minimum(h, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break

    return h
