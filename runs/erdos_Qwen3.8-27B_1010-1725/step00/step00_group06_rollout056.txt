from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_smooth(h, t=50.0):
    """Smooth approximation of C5 using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Subtract max for numerical stability
    m = np.max(corr)
    return m + (1.0/t) * np.log(np.sum(np.exp(t * (corr - m))))

    """Fix the sum constraint iteratively."""
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Distribute to elements not at bounds

        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_smooth(h, t=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.logaddexp.reduce(t * (corr - m)) / t)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def c5_smooth(h, t=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = t * (corr - m)
    # log-sum-exp for numerical stability
    max_s = np.max(shifted)
    return float(m + max_s / t + np.log(np.sum(np.exp(shifted - max_s))) / t)


def fix_sum(h, n):
    target = 0.5 * n
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def sum_con_fun(v, n):
    return float(v.sum() - 0.5 * n)


def sum_con_jac(v, n):
    return np.ones(n)


def optimize_h(h_init, n, t=80.0, maxiter=2000, ftol=1e-14):
    def obj(v):
        return c5_smooth(v, t)

    cons = [{'type': 'eq', 'fun': lambda v: sum_con_fun(v, n),
             'jac': lambda v: sum_con_jac(v, n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and np.isfinite(res.fun) and np.all(np.isfinite(res.x)):
            return fix_sum(res.x.copy(), n)
    except Exception:
        pass
    return h_init

    t0 = time.time()

    n_best = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Local optimization at initial resolution
    if remaining() > 60:

        c5_check = compute_c5(h_opt)
        if c5_check < best_c5:

    # Phase 2: Multi-start at initial resolution
