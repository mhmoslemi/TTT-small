import numpy as np
from scipy.optimize import minimize
import time

_OBJ_S = 50.0
_OBJ_TARGET = 0.0
_OBJ_N = 0


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_onto_set(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, s):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    dx = 2.0 / len(x)
    shifted = s * (corr - mx)
    return (mx + np.log(np.sum(np.exp(shifted)) / len(corr)) / s) * dx


def _obj_smooth(x):
    return smooth_max_obj(x, _OBJ_S)


def _obj_raw(x):
    return compute_c5(x)


def _sum_eq(x):
    return x.sum() - _OBJ_TARGET


def _sum_jac(x):
    return np.ones(_OBJ_N)


def local_search(h, target, rng, max_time):
    n = len(h)
    t_end = time.time() + max_time
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.05
    no_improve_count = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for delta in (step, -step, 0.5 * step, -0.5 * step):
                ht = best.copy()
                ht[i] += delta
                ht = project_onto_set(ht, target)
                c = compute_c5(ht)
                if c < best_c5 - 1e-12:
                    best, best_c5 = ht, c
                    improved = True
                    break
        for _ in range(150):
            k = int(rng.integers(1, min(4, n)))
            idx = rng.choice(n, size=k, replace=False)
            ht = best.copy()
            for i in idx:
                ht[i] += rng.uniform(-step, step)
            ht = project_onto_set(ht, target)
            c = compute_c5(ht)
            if c < best_c5 - 1e-12:
                best, best_c5 = ht, c
                improved = True
        if not improved:
            no_improve_count += 1
            step *= 0.7
            if step < 1e-5 or no_improve_count > 8:
                break
        else:
            no_improve_count = 0
    return best, best_c5
