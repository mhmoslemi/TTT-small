        n0 = 45
        h_best = random_binary_init(n0, seed=seed)
        h_best = project_feasible(h_best, n0)
        n_best = n0
        c5_best = compute_c5(h_best)

    tl = budget_s - elapsed()
    if tl < 5:
        return (h_best, c5_best, n_best)

    # Phase 1: optimize at initial grid
    h_opt, c5_opt = optimize_grid(h_best, n_best, max_iter=600, time_limit=tl * 0.25)
    if c5_opt < c5_best:
        h_best, c5_best, n_best = h_opt, c5_opt, n_best

    # Phase 2: try random restarts at same grid
    tl = budget_s - elapsed()
    if tl > 20:
        for r in range(4):
            if elapsed() > budget_s - 30:
                break
            h_rand = random_binary_init(n_best, seed=seed + 100 + r)
            h_rand = project_feasible(h_rand, n_best)
            h_r, c5_r = optimize_grid(h_rand, n_best, max_iter=200, time_limit=min(15, tl * 0.15))
            if c5_r < c5_best:
                h_best, c5_best = h_r, c5_r

    # Phase 3: refine at higher grids
    tl = budget_s - elapsed()
    grid_multipliers = [2, 4, 8, 12]
    for mult in grid_multipliers:
        n_new = n_best * mult
        if n_new > 1440:
            n_new = 1440
        if n_new <= n_best or elapsed() > budget_s - 40:
            continue
        tl = budget_s - elapsed()
        h_interp = interp_h(h_best, n_best, n_new)
        h_interp = project_feasible(h_interp, n_new)
        # Snap near-binary values
        h_interp[np.abs(h_interp - 0.5) > 0.3] = np.where(h_interp[np.abs(h_interp - 0.5) > 0.3] > 0.5, 1.0, 0.0)
        h_interp = project_feasible(h_interp, n_new)
        tl_opt = min(tl * 0.4, 200.0)
        h_new, c5_new = optimize_grid(h_interp, n_new, max_iter=800, time_limit=tl_opt)
        if c5_new < c5_best:
            h_best, c5_best, n_best = h_new, c5_new, n_new

    # Phase 4: final polish at best grid
    tl = budget_s - elapsed()
    if tl > 15:
        h_pol, c5_pol = optimize_grid(h_best, n_best, max_iter=1000, time_limit=tl * 0.7)
        if c5_pol < c5_best:
            h_best, c5_best = h_pol, c5_pol

    # Final verification and correction
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n_best)
    assert abs(np.sum(h_best) - n_best / 2.0) < 1e-8, f"Sum={np.sum(h_best)}, target={n_best/2}"
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n_best)
