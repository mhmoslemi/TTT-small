```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_c5(h, n_points):
    n = n_points
    dx = 2.0 / n
    full_len = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    
    grad = np.zeros(n)
    for m in range(full_len):
        if abs(corr[m] - max_val) > 1e-10:
            continue
        l = m - (n - 1)
        n_start = max(0, -l)
        n_end = min(n, n - l)
        idx1 = np.arange(n_start, n_end)
        idx2 = idx1 + l
        np.add.at(grad, idx2, (1 - h[idx1]))
        np.add.at(grad, idx1, -h[idx2])
    
    return grad * dx

def objective(h, n_points):
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    bounds = [(0, 1)] * len(h0)
    
    for n_pts in [45, 60, 90]:
        if n_pts == len(h0):
            h_start = h0.copy()
        else:
            # Interpolate
            h_start = np.interp(np.linspace(0, 1, n_pts), np.linspace(0, 1, len(h0)), h0)
            h_start = np.clip(h_start, 0, 1)
            # Fix sum
            target = n_pts / 2.0
            h_start = h_start - np.mean(h_start) + target / n_pts
            h_start = np.clip(h_start, 0, 1)
        
        n = n_pts
        bounds_local = [(0, 1)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 
                'jac': lambda x: np.ones(n)}]
        
        result = minimize(
            objective, h_start, args=(n,),
            method='SLSQP',
            jac=lambda x: grad_c5(x, n),
            bounds=bounds_local,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        
        if result.success or result.fun < best_c5:
            h_opt = result.x
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        
        if time.time() - t0 > budget_s * 0.7:
            break
    
    return best_h, best_c5, len(best_h)
```

```python
    full_len = 2 * n - 1
    for m in range(full_len):
        if weights[m] < 1e-10:
            continue
        l = m - (n - 1)
        n_start = max(0, -l)
        n_end = min(n, n - l)
        if n_end <= n_start:
            continue
        idx1 = np.arange(n_start, n_end)
        idx2 = idx1 + l
        grad[idx2] += weights[m] * (1.0 - h[idx1])
        grad[idx1] += weights[m] * (-h[idx2])
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(45, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    n_list = [len(h0), 60, 90, 120]
    n_list = sorted(set(n_list))

    for target_n in n_list:
        if time.time() - t_start > budget_s * 0.85:
            break
        _N = target_n
        bounds = [(0.0, 1.0)] * target_n
        constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]

        if target_n == len(h0):
            h_start = h0.copy()
        else:
            x_old = np.linspace(0.0, 1.0, len(h0))
            x_new = np.linspace(0.0, 1.0, target_n)
            h_start = np.interp(x_new, x_old, best_h)
            h_start = np.clip(h_start, 0.0, 1.0)
        h_start = project_feasible(h_start, target_n)

        # SLSQP with hard max
        res = minimize(obj_func, h_start, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 400, 'ftol': 1e-14})
        h_opt = project_feasible(res.x, target_n)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

        # Smooth max approach for better landscape
        if time.time() - t_start < budget_s * 0.55 and target_n <= 90:
            res2 = minimize(smooth_max_obj, h_opt, method='SLSQP',
                            jac=lambda x: smooth_max_grad(x, 100.0),
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 300, 'ftol': 1e-14})
            h_opt2 = project_feasible(res2.x, target_n)
            c5_val2 = compute_c5(h_opt2)
            if c5_val2 < best_c5:
                best_c5 = c5_val2
                best_h = h_opt2

            # Increase beta and refine
            res3 = minimize(smooth_max_obj, h_opt2, method='SLSQP',
                            jac=lambda x: smooth_max_grad(x, 300.0),
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 200, 'ftol': 1e-14})
            h_opt3 = project_feasible(res3.x, target_n)
            c5_val3 = compute_c5(h_opt3)
            if c5_val3 < best_c5:
                best_c5 = c5_val3
                best_h = h_opt3

        # Random restarts for diversity
        for restart in range(2):
            if time.time() - t_start > budget_s * 0.8:
                break
            h_rand_base = (rng.rand(target_n) + 0.3 * np.sin(np.arange(target_n) * rng.rand() * 6.28))
            h_rand_base = np.clip(h_rand_base, 0.0, 1.0)
            h_rand = project_feasible(h_rand_base, target_n)
            res_r = minimize(obj_func, h_rand, method='SLSQP', jac=obj_grad,
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-13})
            h_r = project_feasible(res_r.x, target_n)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Final polish on best solution
    _N = len(best_h)
    bounds_final = [(0.0, 1.0)] * _N
    constraints_final = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    res_final = minimize(obj_func, best_h, method='SLSQP', jac=obj_grad,
                         bounds=bounds_final, constraints=constraints_final,
                         options={'maxiter': 200, 'ftol': 1e-15})
    h_final = project_feasible(res_final.x, _N)
    c5_final = compute_c5(h_final)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final

    return best_h, best_c5, len(best_h)
```
