    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5_smoothed(h, beta=40.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mx = np.max(corr)
    scaled = np.clip((corr - mx) * beta, -60.0, 60.0)
    exps = np.exp(scaled)
    p = exps / np.sum(exps)
    # Keep only significant weights
    active = np.where(p > 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        w = p[k]
        t1 = np.arange(n) - s
        t2 = np.arange(n) + s
        m1 = (t1 >= 0) & (t1 < n)
        m2 = (t2 >= 0) & (t2 < n)
        if np.any(m1):
            grad[m1[0]:m1[-1]+1] += w * (1.0 - h[t1[m1]])
        if np.any(m2):
            grad[m2[0]:m2[-1]+1] -= w * h[t2[m2]]
    grad *= dx
    return grad

def gradient_search(h, max_iters, t0, deadline):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.08
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c5 = compute_c5(best)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        grad = grad_c5_smoothed(best, beta=40.0)
        h_new = best - lr * grad
        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best = h_new
            best_c5 = c5_new
            lr = min(lr * 1.05, 0.6)
            stuck = 0
        else:
            lr *= 0.92
            stuck += 1
            if stuck > 40:
                h_pert = best + np.random.randn(n) * 0.08
                h_pert = project_h(h_pert, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best = h_pert
                    best_c5 = c5_pert
                lr = 0.08
                stuck = 0
    return best, best_c5

def random_search(h, max_iters, t0, deadline):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    temp = 0.12
    for it in range(max_iters):
        if time.time() > deadline:
            break
        n_pert = np.random.randint(1, max(2, n // 20) + 1)
        idx = np.random.choice(n, n_pert, replace=False)
        h_new = best.copy()
        h_new[idx] += np.random.uniform(-temp, temp, n_pert)
        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new
            best_c5 = c5_new
            temp = min(temp * 1.05, 0.25)
        else:
            accept = np.exp(-(c5_new - best_c5) / (temp * 0.008))
            if np.random.random() < accept:
                best = h_new
                best_c5 = c5_new
            temp *= 0.9997
    return best, best_c5

def upscale(h, new_n):
    n = len(h)
    ratios = (np.arange(new_n) + 0.5) * n / new_n
    idx0 = np.floor(ratios).astype(int)
    idx1 = np.minimum(idx0 + 1, n - 1)
    frac = ratios - idx0
    h_new = h[idx0] * (1.0 - frac) + h[idx1] * frac
    return project_h(h_new, new_n)

def swap_local_search(h, max_iters, t0, deadline):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Try swapping values between two indices
        i, j = np.random.choice(n, 2, replace=False)
        h_new = best.copy()
        delta = (h_new[j] - h_new[i]) * np.random.uniform(0.2, 0.8)
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best = h_new
            best_c5 = c5_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution from parent
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.random.randint(0, 2, 80).astype(float)
    
    n = len(h0)
    h0 = project_h(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Gradient search at base resolution (25% budget)
    deadline1 = t0 + budget_s * 0.25
    best_h, best_c5 = gradient_search(best_h, 3000, t0, deadline1)
    
    # Phase 2: Random + swap search (25% budget)
    deadline2 = t0 + budget_s * 0.50
    best_h, best_c5 = random_search(best_h, 20000, t0, deadline2)
    best_h, best_c5 = swap_local_search(best_h, 15000, t0, deadline2)
    
    # Phase 3: Multi-resolution exploration (35% budget)
    deadline3 = t0 + budget_s * 0.85
    for target_n in [2 * n, 3 * n]:
        if time.time() > deadline3:
            break
        h_up = upscale(best_h, target_n)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.005:  # worth trying
            h_up, c5_up = gradient_search(h_up, 800, t0, deadline3)
            h_up, c5_up = random_search(h_up, 8000, t0, deadline3)
            h_up, c5_up = swap_local_search(h_up, 5000, t0, deadline3)
            if c5_up < best_c5:
                best_h = h_up
                best_c5 = c5_up
    
    # Phase 4: Final refinement (remaining budget, use ~95%)
    deadline4 = t0 + budget_s * 0.97
    n_final = len(best_h)
    if time.time() < deadline4 - 10:
        # Additional gradient polish
        best_h, best_c5 = gradient_search(best_h, 1000, t0, deadline4)
    if time.time() < deadline4 - 5:
        best_h, best_c5 = random_search(best_h, 10000, t0, deadline4)
    
    # Final constraint enforcement
    n_final = len(best_h)
    target_sum = n_final / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        # Find index closest to 0.5 for adjustment
        mid_idx = int(np.argmin(np.abs(best_h - 0.5)))
        adj = np.clip(best_h[mid_idx] + diff, 0.0, 1.0)
        best_h[mid_idx] = adj
        diff2 = target_sum - np.sum(best_h)
        if abs(diff2) > 1e-12:
            # Adjust a second entry
            mask = np.ones(n_final, dtype=bool)
            mask[mid_idx] = False
            mid_idx2 = int(np.argmin(np.abs(best_h[mask] - 0.5) + 10*~mask))
            if mid_idx2 != mid_idx:
                adj2 = np.clip(best_h[mid_idx2] + diff2, 0.0, 1.0)
                best_h[mid_idx2] = adj2
    
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, n_final
