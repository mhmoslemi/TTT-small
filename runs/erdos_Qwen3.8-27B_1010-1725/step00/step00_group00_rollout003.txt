    return (m + np.log(np.sum(np.exp(tau * shifted))) / tau) * dx

def _constraint(h):
    return np.sum(h) - _current_n / 2.0

def _c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = (h > 0.0) & (h < 1.0)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _run_slsqp(h0, n, maxiter=300):
    global _current_n
    _current_n = n
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _obj, h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': _constraint}],
        options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
    )
    return res

def _upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    global _current_n
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_p = len(initial_h_values)
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_p = 45
        h_best = np.ones(n_p) / 2.0

    c5_best = _c5(h_best, n_p)

    def _elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from multiple starts at original resolution
    starts = [h_best.copy()]
    for i in range(4):
        h_pert = h_best + 0.05 * (i + 1) * rng.randn(n_p)
        starts.append(_project(h_pert, n_p))
    for i in range(3):
        h_r = 0.3 + 0.4 * rng.rand(n_p)
        starts.append(_project(h_r, n_p))

    for h_start in starts:
        if _elapsed() > budget_s * 0.4:
            break
        res = _run_slsqp(h_start, n_p, maxiter=400)
        h_cand = _project(res.x, n_p)
        c5_cand = _c5(h_cand, n_p)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand
            starts.insert(0, h_best.copy() + 0.02 * rng.randn(n_p))
            starts[0] = _project(starts[0], n_p)

    # Phase 2: Refine with local perturbation at same n
    for i in range(5):
        if _elapsed() > budget_s * 0.6:
            break
        h_start = h_best + 0.01 * rng.randn(n_p)
        h_start = _project(h_start, n_p)
        res = _run_slsqp(h_start, n_p, maxiter=500)
        h_cand = _project(res.x, n_p)
        c5_cand = _c5(h_cand, n_p)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand

    # Phase 3: Try slightly higher resolution (2x)
    n_try = min(2 * n_p, 200)
    if n_try > n_p and _elapsed() < budget_s * 0.7:
        h_up = _upsample(h_best, n_p, n_try)
        h_up = _project(h_up, n_try)
        res = _run_slsqp(h_up, n_try, maxiter=300)
        h_cand = _project(res.x, n_try)
        c5_cand = _c5(h_cand, n_try)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand
            n_p = n_try

    # Phase 4: Final polish
    if _elapsed() < budget_s * 0.9:
        res = _run_slsqp(h_best.copy(), n_p, maxiter=600)
        h_cand = _project(res.x, n_p)
        c5_cand = _c5(h_cand, n_p)
        if c5_cand < c5_best:
            h_best = h_cand
            c5_best = c5_cand

    _current_n = n_p
    final_c5 = _c5(h_best, n_p)
    return (h_best, final_c5, n_p)
