def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return np.max(c) * dx

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    m_max = np.argmax(c)
    k = m_max - (n-1)
    # subgradient
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # from h[i+k] term: (1-h[i-k]) if 0 <= i-k < n
        j1 = i - k
        if 0 <= j1 < n:

        j2 = i + k
        if 0 <= j2 < n:

        grad[i] = g

def c5_val(h):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    """Compute subgradient using top-K active constraints."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top-K indices
    topk = np.argsort(c)[-K:]
    topk = topk[np.argsort(c[topk])[::-1]]  # sort descending
    vals = c[topk]
    # Weights: softmax over top-K
    w = np.exp((vals - vals.max()) * 10)

    n = len(h)
    grad = np.zeros(n)

    if lo2 < hi2:
        j = np.arange(lo2, hi2)

def optimize(h, tgt, rng, time_limit):
    h = proj(h.copy(), tgt)
    n = len(h)
    best = h.copy()
    best_c5 = c5_val(h)
    lr = 0.004
    temp = 0.004
    K = 8
    stuck = 0
    t_end = time.time() + time_limit
    while time.time() < t_end:
        cv = c5_all(h)
        c1 = cv.max()
        topk = np.argsort(cv)[-K:]
        tv = cv[topk]
        w = np.exp(12.0 * (tv - tv.max()))
        w /= w.sum()
        g = np.zeros(n)
        for idx in range(K):
            g += w[idx] * grad_shift(h, topk[idx] - (n - 1))
        ns = lr * 0.4
        h2 = proj(h - lr * g + rng.randn(n) * ns, tgt)
        c2 = c5_val(h2)
        if c2 < best_c5:
            best_c5 = c2
            best = h2.copy()
            h = h2
            stuck = 0
            lr = min(lr * 1.01, 0.04)
        elif c2 <= c1 or rng.random() < np.exp(-(c2 - c1) / max(temp, 1e-10)):
            h = h2
            stuck += 1
        else:
            h = proj(h - 0.25 * lr * g, tgt)
            stuck += 1
        if stuck > 1200:
            temp = max(temp * 0.7, 1e-5)
            lr = max(lr * 0.88, 0.0008)
            stuck = 0
            if best_c5 - c5_val(h) > 0.003:
                h = best.copy()
    return best, best_c5

def local_search(h, tgt, rng, time_limit):
    h = h.copy()
    n = len(h)
    best = h.copy()
    best_c5 = c5_val(h)
    t_end = time.time() + time_limit
    while time.time() < t_end:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(0.02, 0.2) * (1.0 if rng.random() > 0.5 else -1.0)
        h2 = h.copy()
        h2[i] += d
        h2[j] -= d
        h2 = proj(h2, tgt)
        c2 = c5_val(h2)
        if c2 < best_c5:
            best_c5 = c2
            best = h2.copy()
            h = h2
        elif c2 < c5_val(h):
            h = h2
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n = len(h0)
    tgt = n / 2.0
    h0 = proj(h0, tgt)
    # Phase 1: subgradient + annealing (70% budget)
    t1 = min(budget_s * 0.70 - 5, budget_s - 60)
    best, best_c5 = optimize(h0, tgt, rng, t1)
    # Phase 2: local pair search (20% budget)
    elapsed = time.time() - t0
    t2 = min(budget_s * 0.20, budget_s - elapsed - 30)
    if t2 > 10:
        b2, c2 = local_search(best, tgt, rng, t2)
        if c2 < best_c5:
            best, best_c5 = b2, c2
    # Phase 3: random restarts if time remains
    elapsed = time.time() - t0
    t_rem = budget_s - elapsed - 5
    for _ in range(4):
        if t_rem < 25:
            break
        hr = np.zeros(n)
        hr[:n // 2] = 1.0
        if n % 2:
            hr[n // 2] = 0.5
        rng.shuffle(hr)
        hr = proj(hr, tgt)
        alloc = t_rem * 0.4
        br, cr = optimize(hr, tgt, rng, alloc)
        if cr < best_c5:
            best, best_c5 = br, cr
        elapsed = time.time() - t0
        t_rem = budget_s - elapsed - 5
    return best, best_c5, n
