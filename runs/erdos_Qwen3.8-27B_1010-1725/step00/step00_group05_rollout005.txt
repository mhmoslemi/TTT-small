    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    # Find top k indices
    top_idx = np.argsort(corr * dx)[-k:]
    # Compute gradient of each
    grad = np.zeros(n)

        grad += corr_grad(h, one_minus_h, idx)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = h.copy()
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                order = np.argsort(-h)
                for i in order:
                    add = min(1.0 - h[i], diff)
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-15:
                        break
            else:
                order = np.argsort(h)
                for i in order:
                    sub = min(h[i], -diff)
                    h[i] -= sub
                    diff += sub
                    if abs(diff) < 1e-15:
                        break
        h = np.clip(h, 0, 1)
    return h

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / beta

def corr_gradient(h, k_idx):
    """Gradient of correlation at full-output index k_idx w.r.t. h"""
    n = len(h)
    offset = n - 1 - k_idx
    grad = np.zeros(n)
    # First part: (1-h[j+offset]) for valid j
    lo1 = max(0, 0)  # j ≥ 0
    hi1 = n - 1      # j ≤ n-1
    lo1b = max(0, -offset)  # j+offset ≥ 0
    hi1b = n - 1 - offset   # j+offset ≤ n-1
    if hi1b >= lo1b:
        js = np.arange(max(lo1, lo1b), min(hi1, hi1b) + 1)
        grad[js] += (1.0 - h[js + offset])
    # Second part: -h[j-offset] for valid j
    lo2 = max(0, offset)     # j-offset ≥ 0
    hi2 = n - 1              # j ≤ n-1
    if hi2 >= lo2:
        js = np.arange(lo2, hi2 + 1)
        grad[js] += -h[js - offset]
    return grad

def topk_ld_step(h, target_sum, n_top=3, step_size=0.01):
    """One step of top-k line search descent"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    vals = corr * dx
    # Top k indices
    top_idx = np.argsort(vals)[-n_top:]
    # Average gradient
    grad = np.zeros(n)
    for k_idx in top_idx:
        grad += corr_gradient(h, k_idx)
    grad /= n_top
    # Project gradient to sum=0 subspace (to maintain constraint)
    grad -= np.mean(grad)
    # Line search
    best_c5 = compute_c5(h)
    best_step = 0.0
    for try_step in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
        trial = h - try_step * (grad / np.max(np.abs(grad)) * 0.1)
        trial = project_feasible(trial, target_sum)
        c = compute_c5(trial)
        if c < best_c5:
            best_c5 = c
            best_step = try_step
    if best_step > 0:
        h_new = h - best_step * (grad / np.max(np.abs(grad)) * 0.1)
        h_new = project_feasible(h_new, target_sum)
        return h_new, best_c5
    return h, best_c5
