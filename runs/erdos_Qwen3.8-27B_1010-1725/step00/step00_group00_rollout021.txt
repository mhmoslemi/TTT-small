from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    return float(np.max(c)) * dx

def c5_val_grad(h, n, M):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full') * dx
    idx = int(np.argmax(c))
    val = c[idx]
    lag = idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jp = j + lag
        if 0 <= jp < n:
            grad[j] += (1.0 - h[jp]) * dx
        jm = j - lag
        if 0 <= jm < n:
            grad[j] += (-h[jm]) * dx
    s = np.sum(h) - n / 2.0
    val += M * s * s
    grad += 2.0 * M * s * np.ones(n) / n
    return val, grad

        # Distribute the residual
        diff = target - s
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

            best_h = project_h(best_h, n0)
            best_c5 = compute_c5(best_h, 2.0 / n0)

    # Phase 4: Try higher resolutions
    for factor in [2, 3, 4, 5, 6, 8, 10]:
        n_new = n0 * factor
        if n_new > 360:
            n_new = 360
        if n_new <= best_n:
            continue
        if time.time() - t0 > budget_s * 0.88:
            break
        dx_new = 2.0 / n_new
        x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_new = (np.arange(n_new) + 0.5) * dx_new
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_h(h_interp, n_new)

        h_hi = optimize_at(h_interp, n_new, M, 4000, 1e-15)
        c5_hi = compute_c5(h_hi, dx_new)
        if c5_hi < best_c5:
            best_h, best_c5, best_n = h_hi, c5_hi, n_new

        # A few restarts at new resolution
        for r in range(8):
            if time.time() - t0 > budget_s * 0.92:
                break
            h_pert = h_hi + 0.025 * np.random.randn(n_new)
            h_pert = project_h(h_pert, n_new)
            h_r = optimize_at(h_pert, n_new, M, 2000, 1e-13)
            c5_r = compute_c5(h_r, dx_new)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r, c5_r, n_new

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 5:
        best_h = optimize_at(best_h, best_n, M, 5000, 1e-16)

    best_h = project_h(best_h, best_n)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, final_c5, best_n)
