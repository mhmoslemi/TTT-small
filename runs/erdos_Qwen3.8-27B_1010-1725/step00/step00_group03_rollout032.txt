import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_grad_for_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def c5_exact_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(corr))
    obj = corr[k_star]
    lag = k_star - (n - 1)
    grad = corr_grad_for_lag(h, lag, n) * dx
    return obj, grad

def c5_smoothed_grad(h, tau=40.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = tau * (corr - cmax)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    obj = cmax + np.log(Z) / tau
    w = exp_s / Z
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * dx * corr_grad_for_lag(h, lag, n)
    return obj, grad

def project_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def fix_constraint(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask_up = h < 1.0
        mask_dn = h > 0.0
        if diff > 0:
            cnt = np.sum(mask_up)
            if cnt > 0:
                h[mask_up] += diff / cnt
        else:
            cnt = np.sum(mask_dn)
            if cnt > 0:
                h[mask_dn] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    def try_optimize(h0, max_iter=400):
        nonlocal best_h, best_c5
        if time_left() < 5:
            return
        n = len(h0)
        h0 = fix_constraint(h0)
        c0 = compute_c5(h0)
        if c0 < best_c5 - 1e-10:
            best_c5 = c0
            best_h = h0.copy()

        bounds = [(0.0, 1.0)] * n
        con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                'jac': lambda x: np.ones(n)}]

        # Phase 1: smoothed
        res1 = minimize(lambda x: c5_smoothed_grad(x, 40.0)[0], h0,
                        method='SLSQP', jac=lambda x: c5_smoothed_grad(x, 40.0)[1],
                        bounds=bounds, constraints=con,
                        options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
        if res1.fun < compute_c5(best_h) if best_h is not None else False:
            pass
        c1 = compute_c5(res1.x)
        if c1 < best_c5 - 1e-10:
            best_c5 = c1
            best_h = res1.x.copy()

        # Phase 2: exact max
        h2 = fix_constraint(res1.x)
        res2 = minimize(compute_c5, h2,
                        method='SLSQP', jac=lambda x: c5_exact_grad(x)[1],
                        bounds=bounds, constraints=con,
                        options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
        c2 = compute_c5(res2.x)
        if c2 < best_c5 - 1e-10:
            best_c5 = c2
            best_h = res2.x.copy()

    def random_valid_h(n):
        h = np.random.rand(n)
        h = (h - h.min()) / max(h.max() - h.min(), 1e-10)
        h *= (n / 2.0) / max(np.sum(h), 1e-10)
        return fix_constraint(h)

    # Start with initial
    have_initial = 'initial_h_values' in globals() and initial_h_values is not None
    if have_initial:
        h_init = np.array(initial_h_values, dtype=float)
        try_optimize(h_init, max_iter=500)

    # Try initial with perturbation
    if have_initial and time_left() > 30:
        h_init2 = np.array(initial_h_values, dtype=float) + 0.1 * np.random.randn(len(initial_h_values))
        try_optimize(fix_constraint(h_init2), max_iter=300)

    # Try various n with random starts
    for n_try in [70, 100, 140, 200, 280]:
        if time_left() < 20:
            break
        for attempt in range(4):
            if time_left() < 15:
                break
            if have_initial and n_try == len(initial_h_values):
                continue
            hr = random_valid_h(n_try)
            try_optimize(hr, max_iter=300)

    # Try structured starts: piecewise constant
    if time_left() > 30:
        for n_sc in [80, 120, 160]:
            if time_left() < 10:
                break
            for npiece in [2, 3, 4]:
                if time_left() < 5:
                    break
                h_sc = np.zeros(n_sc)
                seg = n_sc // npiece
                vals = np.linspace(0.1, 0.9, npiece)
                for p in range(npiece):
                    lo = p * seg
                    hi = lo + seg if p < npiece - 1 else n_sc
                    h_sc[lo:hi] = vals[p]
                try_optimize(fix_constraint(h_sc + 0.01 * np.random.randn(n_sc)), max_iter=200)

    # Final polish: multi-start neighborhood of best
    if best_h is not None and time_left() > 15:
        n_best = len(best_h)
        for pert in [0.01, 0.03, 0.05]:
            if time_left() < 8:
                break
            hp = best_h + pert * np.random.randn(n_best)
            try_optimize(fix_constraint(hp), max_iter=200)

    # Ensure final solution is valid
    if best_h is not None:
        best_h = fix_constraint(best_h)
        best_c5 = compute_c5(best_h)
        # One more SLSQP with very tight tolerance
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        con = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_best / 2.0,
                'jac': lambda x: np.ones(n_best)}]
        if time_left() > 10:
            res_f = minimize(compute_c5, best_h, method='SLSQP',
                             jac=lambda x: c5_exact_grad(x)[1],
                             bounds=bounds, constraints=con,
                             options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
            cf = compute_c5(res_f.x)
            if cf < best_c5:
                best_c5 = cf
                best_h = res_f.x.copy()

    if best_h is None:
        best_h = fix_constraint(np.full(70, 0.5))
        best_c5 = compute_c5(best_h)

    best_h = fix_constraint(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
