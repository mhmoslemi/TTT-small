    for _ in range(500):
        d = t - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _snap(h, n, lo=0.15, hi=0.85):
    h2 = h.copy()
    h2[h2 < lo] = 0.0
    h2[h2 > hi] = 1.0
    return _feasible(h2, n)

def _slsqp(h, n, iters):
    global _N, _TGT
    _N, _TGT = n, n * 0.5
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': _csum, 'jac': _csum_jac}]
    try:
        r = minimize(_obj, h.copy(), method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': iters, 'ftol': 1e-13})
        if r.fun < 1e5:
            h2 = _feasible(r.x, n)
            return h2, _obj(h2)
    except Exception:
        pass
    return h, _obj(h)

def _local_coord(h, n, steps=50):
    """Coordinate descent: optimize one index at a time via bisection."""
    global _N, _TGT
    _N, _TGT = n, n * 0.5
    h = h.copy()
    for _ in range(steps):
        i = np.random.randint(n)
        lo, hi = 0.0, 1.0
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            h[i] = mid
            val = _obj(h)
            # We want to find the best value at this coordinate
            # Evaluate at lo, mid, hi
            h[i] = lo
            v_lo = _obj(h)
            h[i] = hi
            v_hi = _obj(h)
            h[i] = mid
            v_mid = _obj(h)
            if v_mid < v_lo and v_mid < v_hi:
                lo, hi = lo, mid
            elif v_mid < v_lo and v_mid >= v_hi:
                hi = mid
            elif v_mid >= v_lo and v_mid < v_hi:
                lo = mid
            elif v_lo <= v_mid and v_lo <= v_hi:
                hi = mid
            else:
                lo = mid
        # Set to best of lo, mid, hi
        h[i] = lo
        v_lo = _obj(h)
        mid = 0.5 * (lo + hi)
        h[i] = mid
        v_mid = _obj(h)
        h[i] = hi
        v_hi = _obj(h)
        h[i] = [lo, mid, hi][int(np.argmin([v_lo, v_mid, v_hi]))]
    return _feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _TGT
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h = np.array(initial_h_values, dtype=np.float64)
    n = len(h)
    _N, _TGT = n, n * 0.5
    h = _feasible(h, n)
    best_h = h.copy()
    best_c = _obj(h)
    
    def _time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Main SLSQP from initial
    h, c = _slsqp(h, n, 3000)
    if c < best_c:
        best_h, best_c = h.copy(), c
    
    # Phase 2: Binary snap + re-optimize
    for snap_round in range(5):
        if _time_left() < 60:
            break
        hs = _snap(best_h, n, lo=0.1 * snap_round + 0.1, hi=0.9 - 0.1 * snap_round)
        h2, c2 = _slsqp(hs, n, 1500)
        if c2 < best_c:
            best_h, best_c = h2, c2
        # Coordinate descent polish
        h3 = _local_coord(best_h, n, steps=30)
        c3 = _obj(h3)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h4, c4 = _slsqp(best_h, n, 1000)
        if c4 < best_c:
            best_h, best_c = h4, c4
    
    # Phase 3: Random restarts with annealing
    i = 0
    while _time_left() > 30:
        i += 1
        s = max(0.02, 0.12 * (0.97 ** i))
        ht = _feasible(best_h + rng.normal(0.0, s, n), n)
        h2, c2 = _slsqp(ht, n, 600)
        if c2 < best_c:
            best_h, best_c = h2, c2
            i = 0
    
    # Final polish
    if _time_left() > 10:
        h2, c2 = _slsqp(best_h, n, 2000)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    best_h = _feasible(best_h, n)
    return best_h, _obj(best_h), n
