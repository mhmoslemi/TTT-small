    return np.max(corr), corr

    hi = 2.0

        s = np.sum(np.clip(x - tau, 0, 1))

    corr = np.correlate(h, 1-h, mode='full') * dx
    # Numerically stable log-sum-exp
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    return val

import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project(x, total):
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - tau, 0.0, 1.0))
        if s > total:
            lo = tau
        else:
            hi = tau
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = total - h.sum()
    if abs(d) > 1e-12:
        f = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(f) > 0:
            h[f] += d / len(f)
    return np.clip(h, 0.0, 1.0)

def subgrad(h, n, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k = int(np.argmax(c))
    s = k - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    i1 = j - s
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] = (1.0 - h[i1[m1]])
    i2 = j + s
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g * dx, c[k]

def softmax_grad(h, beta, dx):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    corr = raw * dx
    m = np.max(corr)
    exp_t = np.exp(beta * (corr - m))
    Z = exp_t.sum()
    p = exp_t / Z
    j = np.arange(n)
    g = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-8:
            continue
        s = k - (n - 1)
        i1 = j - s
        m1 = (i1 >= 0) & (i1 < n)
        g[m1] += pk * dx * (1.0 - h[i1[m1]])
        i2 = j + s
        m2 = (i2 >= 0) & (i2 < n)
        g[m2] -= pk * dx * h[i2[m2]]
    return g

def continuous_opt(h, n, dx, tot, rng, t_end):
    h = h.copy()
    bh = h.copy()
    bc5 = compute_c5(h, dx)
    lr = 0.4
    mom = np.zeros(n)
    mu = 0.85
    stall = 0
    
    while time.time() < t_end - 10:
        g, _ = subgrad(h, n, dx)
        mom = mu * mom - lr * g
        hn = project(h + mom, tot)
        c5 = compute_c5(hn, dx)
        if c5 < bc5 - 1e-10:
            bc5 = c5
            bh = hn.copy()
            h = hn
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.8
                mom *= 0.5
            if stall > 1000:
                lr = 0.4
                mom = np.zeros(n)
                h = project(bh + rng.randn(n) * 0.06, tot)
                stall = 0
    
    # Softmax polishing phase
    t_soft_end = min(time.time() + 15, t_end - 5)
    beta = 30.0
    lr2 = 0.05
    for it in range(20000):
        if time.time() > t_soft_end:
            break
        g = softmax_grad(h, beta, dx)
        alpha = lr2
        improved = False
        for _ in range(15):
            hn = project(h - alpha * g, tot)
            c5 = compute_c5(hn, dx)
            if c5 < bc5 - 1e-10:
                h = hn
                bc5 = c5
                bh = h.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            lr2 *= 0.9
            if lr2 < 0.002:
                break
        if it % 200 == 0 and it > 0:
            beta *= 1.08
    return bh, bc5

def discrete_opt(h, n, dx, tot, rng, t_end):
    vals = np.array([0.0, 0.5, 1.0])
    h = np.where(h < 0.25, 0.0, np.where(h < 0.75, 0.5, 1.0))
    # Fix sum
    s = h.sum()
    d = tot - s
    while abs(d) > 0.25 and time.time() < t_end:
        if d > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            i = idx[rng.randint(len(idx))]

        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            i = idx[rng.randint(len(idx))]
            h[i] = max(0.0, h[i] - 0.5)
            d = tot - h.sum()
    
    bh = h.copy()
    bc5 = compute_c5(h, dx)
    stall = 0
    
    while time.time() < t_end - 2:
        if stall > 800:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        old_sum = h.sum()
        best_pair = None
        best_c5_local = bc5
        for vi in vals:

                    continue

                if c5 < best_c5_local - 1e-12:

                attempts = 0

                    else:

                    d = tot - h.sum()
                    attempts += 1
                if abs(h.sum() - tot) > 0.01:
                    h = bh.copy()
                    stall = 0
