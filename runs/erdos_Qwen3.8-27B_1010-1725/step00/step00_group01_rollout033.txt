import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_objective_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    fval = float(np.max(corr) * dx)
    
    max_corr = np.max(corr)
    tol = max(1e-6, 1e-3 * max_corr)
    active = np.where(corr >= max_corr - tol)[0]
    
    grad_corr = np.zeros(n)
    for p in active:
        k = p - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + k < n:
                g += (1.0 - h[j + k])
            if 0 <= j - k < n:
                g -= h[j - k]
            grad_corr[j] += g
    grad_corr /= len(active)
    
    grad = grad_corr * dx
    return fval, grad

def make_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum
            order = np.argsort(1.0 - h)  # elements closest to 1 first
        else:
            # Need to decrease sum
            order = np.argsort(h)  # elements closest to 0 first
        for idx in order:
            remaining = n / 2.0 - np.sum(h)
            if abs(remaining) < 1e-10:
                break
            if remaining > 0:
                h[idx] = min(1.0, h[idx] + remaining)
            else:
                h[idx] = max(0.0, h[idx] + remaining)
    return h

def optimize_n(h_init, n, time_limit):
    """Optimize h for given n."""
    start_time = time.time()
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Use SLSQP
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
                    'jac': lambda h: np.ones(n)]}
    bounds = [(0, 1)] * n
    
    for attempt in range(5):
        if time.time() - start_time > time_limit:
            break
        
        # Perturb for multiple starts
        if attempt == 0:
            h0 = h_init.copy()
        else:
            h0 = h_init + np.random.normal(0, 0.05, n)
            h0 = make_feasible(h0, n)
        
        try:
            result = minimize(
                c5_objective_and_grad,
                h0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            h_new = make_feasible(result.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

    start_time = time.time()

    best_c5 = float('inf')

            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_init

            # Actually, let's just use midpoints
            x_old = (np.arange(n_init) + 0.5) * (2.0/n_init)
            x_new = (np.arange(n_try) + 0.5) * (2.0/n_try)
            
            h_interp = np.interp(x_new, x_old, h_init, left=h_init[0], right=h_init[-1])
            h_interp = make_feasible(h_interp, n_try)

            if c5_opt < best_c5:

                best_h = h_opt
                best_n = n_try
    else:

        h = np.zeros(n)
        h[:n//2] = 1.0
        h[n//2] = n % 2  # For odd adjustment
        # Actually for n=50: sum should be 25

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        total = np.sum(room)
        if total > 1e-15:
            h += diff * room / total
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if residual > 0 else int(np.argmin(h))
        adj = np.clip(residual, -h[idx], 1.0 - h[idx])
        h[idx] += adj
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    m = np.max(s)
    return (np.log(np.sum(np.exp(s - m))) + m) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    m = np.max(s)
    w = np.exp(s - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-12:

        k = p - (n - 1)

                g += 1.0 - h[j - k]
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += w[p] * g
    grad *= dx
    return grad
