def compute_grad(h, n, alpha):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    a = alpha * c
    a -= a.max()
    w = np.exp(a)

    v2 = (i2 >= 0) & (i2 < n)

    h2 = np.where(v2, h[i2.clip(0, n-1)], 0.0)

    s1 = (wr * h1).sum(1)
    s2 = (wr * h2).sum(1)
    grad = dx * (1.0 - s1 - s2)
    return grad, float(a.max()/alpha + np.log(w.sum())/alpha)  # smooth max value

            best = h_new.copy()
        h = h_new

from scipy.optimize import minimize
import time

def evaluate(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h = h + d * room / tr
        else:
            th = h.sum()
            if th < 1e-15:
                break
            h = h + d * h / th
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    a = alpha * c
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    m = n - 1
    j = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    i1 = j + k - m
    i2 = j - k + m
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    term1 = np.where(v1, 1.0 - h[i1.clip(0, n - 1)], 0.0)
    term2 = np.where(v2, -h[i2.clip(0, n - 1)], 0.0)
    wr = w[None, :]
    grad = dx * ((wr * term1).sum(1) + (wr * term2).sum(1))
    sse_val = (a_max + np.log(w.sum())) / alpha
    return grad, float(sse_val)

def proj_grad_descent(h, n, t_limit, alpha0=20, alpha_end=500):
