- Find k* with max correlation
- For the pairs contributing to k*, compute the "gradient" direction
- Move h in that direction (with compensation)

To decrease corr_0, we want to increase sum h[i]², which means pushing h[i] towards 0 or 1.

Gradient w.r.t. h[i]: (1-h[i+k]) - (1-h[i-k]) (if both valid)
To decrease corr_k, adjust h[i] in the negative gradient direction.

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def _proj(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _sa(h, n, budget, rng, target):
    best = h.copy()
    best_c = _c5(h)
    cur_c = best_c
    T = 0.004
    cool = 0.99999
    T_min = 1e-7
    stall = 0
    deadline = time.time() + budget
    while time.time() < deadline:
        ht = h.copy()
        r = rng.random()
        if r < 0.30:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            d = rng.uniform(-0.04, 0.04)
            ht[i] += d
            ht[j] -= d
        elif r < 0.50:
            # swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            tmp = ht[i]
            ht[i] = ht[j]
            ht[j] = tmp
        elif r < 0.68:

            s = int(rng.integers(0, n - m))
            p = rng.uniform(-0.02, 0.02, m)
            ht[s:s+m] += p
            ht[(s + m) % n] -= p.sum()
        elif r < 0.82:

                    ht[i + sh] += 0.03
                else:
                    ht[i - sh] -= 0.03

def _local_refine(h, n, budget, rng, target):
    """Gaussian local refinement with very small steps."""
    best = h.copy()
    best_c = _c5(h)
    deadline = time.time() + budget
    step = 0.01
    while time.time() < deadline:
        h = best + rng.normal(0, step, n)
        h = _proj(h, n)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.98
            if step < 1e-5:
                step = 0.02
    return best, best_c
