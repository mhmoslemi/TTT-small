def project_box_hyperplane(h, target_sum):
    n = len(h)
    # Find lambda such that sum(clamp(h_i - lambda, 0, 1)) = target_sum
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def slsqp_optimize(h0, eps=100.0, maxiter=1000):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': lambda h: np.ones(n)}]
    
    def obj(h):
        return smooth_obj(h, eps)
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_box_hyperplane(h_opt, target_sum)
    return h_opt

def subgradient_optimize(h0, max_iter=5000, lr_init=0.002):
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.5
                no_improve = 0
            if lr < 1e-8:
                break
        
        # Compute gradient for worst shift
        worst = int(np.argmax(corr))
        s = worst - (n - 1)  # actual shift
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j - s < n:
                grad[j] += (1.0 - h[j - s])
            if 0 <= j + s < n:
                grad[j] += -h[j + s]
        grad *= dx
        
        h = h - lr * grad
        h = project_box_hyperplane(h, target_sum)
        lr = lr * 0.9995 + (lr_init - lr) * 0.0005  # slowly increase back
    
    return best_h, best_c5

    return float(np.max(corr) * dx)

def smooth_loss(h, args):
    eps = args[0]
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    return np.sum(h) - args[0]

def subgrad_step(h, dx, n, worst_idx, lr):

            grad[j] += (1.0 - h[j + s])
        if 0 <= j - s < n:
            grad[j] += -h[j - s]
