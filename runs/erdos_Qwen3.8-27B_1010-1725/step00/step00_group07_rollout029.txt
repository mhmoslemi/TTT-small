def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(t * shifted))) / t

def smooth_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(t * shifted)
    total = np.sum(exps)
    weights = exps / total
    
    grad = np.zeros(n)
    omh = 1.0 - h  # one minus h

            if w2 > 1e-15:
                grad[offset:] += w2 * omh[:n - offset]
                grad[:n - offset] += w2 * (-h[offset:])
    return grad

    lo = h.min() - 1.0

        if s > target_sum:

    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(t * shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    omh = 1.0 - h
    for offset in range(n):
        w = weights[n - 1 + offset]
        if w > 1e-15:
            grad[:n - offset] += w * omh[offset:]
            grad[offset:] += w * (-h[:n - offset])
        if offset > 0:
            w2 = weights[n - 1 - offset]
            if w2 > 1e-15:
                grad[offset:] += w2 * omh[:n - offset]
                grad[:n - offset] += w2 * (-h[offset:])
    return grad

def project_feasible(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_sinusoidal(n, k, amp):
    i = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * k * i / n)
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def optimize_h(h0, max_iters, t_init, t_final, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy().astype(float), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(it / max(max_iters - 1, 1), 1.0)
        t = t_init + (t_final - t_init) * frac
        g = smooth_max_grad(h, t)
        cur_val = smooth_max_val(h, t)
        improved = False
        a = alpha
        for _ in range(25):
            h_new = project_feasible(h - a * g, target_sum)
            val = smooth_max_val(h_new, t)
            if val < cur_val - 1e-12:
                h = h_new
                improved = True
                break
            a *= 0.5
        if improved:
            alpha = min(a * 1.8, 0.5)
            stall = 0
        else:
            alpha *= 0.7
            stall += 1
            if stall > 200:
                t = max(t_init * 0.5, 5.0)
                alpha = 0.02
                stall = 0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 999.0
    best_n = 0

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(budget_s - elapsed() - 5.0, 1.0)

    def try_candidate(h0, n_target, time_alloc):
        global initial_h_values
        nonlocal_best = None
        h_work = h0.astype(float).copy()
        if len(h_work) != n_target:
            h_work = upsample(h_work, n_target)
        n = n_target
        target_sum = n / 2.0
        h_work = project_feasible(h_work, target_sum)
        h_best, c5_best = optimize_h(h_work, max_iters=3000, t_init=15.0,
                                      t_final=150.0, time_limit=time_alloc)
        return h_best, c5_best

    # Phase 1: Optimize parent at n=80
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = len(parent)
        rem = remaining()
        t1 = min(200.0, rem * 0.3)
        h1, c5_1 = try_candidate(parent, n_parent, t1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n_parent
        # Also try from parent with perturbation
        if remaining() > 50:
            h_pert = parent + 0.1 * rng.randn(len(parent))
            h_pert = project_feasible(h_pert, len(parent) / 2.0)
            t1b = min(80.0, remaining() * 0.2)
            h1b, c5_1b = try_candidate(h_pert, n_parent, t1b)
            if c5_1b < best_c5:
                best_c5 = c5_1b
                best_h = h1b.copy()
                best_n = n_parent

    # Phase 2: Sinusoidal starts at various n
    for n in [80, 120, 160]:
        if remaining() < 60:
            break
        for k in [2, 3, 5, 7]:
            if remaining() < 40:
                break
            h_sin = make_sinusoidal(n, k, 0.35)
            t2 = min(60.0, remaining() * 0.15)
            h2, c5_2 = try_candidate(h_sin, n, t2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n

    # Phase 3: Upscale best to n=160 and refine
    if best_h is not None and remaining() > 80:
        n_up = 160
        if best_n != n_up:
            h_up = upsample(best_h, n_up)
        else:
            h_up = best_h.copy()
        t3 = min(200.0, remaining() * 0.4)
        h3, c5_3 = try_candidate(h_up, n_up, t3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_up

    # Phase 4: Random restarts at n=80 and n=120
    for n in [80, 120]:
        for restart in range(5):
            if remaining() < 40:
                break
            h_rand = rng.rand(n)
            h_rand = project_feasible(h_rand, n / 2.0)
            # Bias toward structure: mix with sinusoidal
            h_mix = 0.5 * h_rand + 0.5 * make_sinusoidal(n, restart + 2, 0.35)
            h_mix = project_feasible(h_mix, n / 2.0)
            t4 = min(50.0, remaining() * 0.15)
            h4, c5_4 = try_candidate(h_mix, n, t4)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
                best_n = n

    # Phase 5: Fine-tune the overall best at its resolution
    if best_h is not None and remaining() > 100:
        tf = min(200.0, remaining() * 0.5)
        h5, c5_5 = try_candidate(best_h, best_n, tf)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final feasibility check
    if best_h is not None:
        n = best_n
        target_sum = n / 2.0
        best_h = project_feasible(best_h, target_sum)
        # Verify sum is exact (within float precision)
        s = best_h.sum()
        if abs(s - target_sum) > 1e-10:
            # Distribute correction
            diff = (target_sum - s) / n
            best_h = np.clip(best_h + diff, 0.0, 1.0)
            best_h = project_feasible(best_h, target_sum)
        final_c5 = compute_c5(best_h)
        best_c5 = final_c5
    else:
        # Fallback: constant-ish
        n = 80
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
