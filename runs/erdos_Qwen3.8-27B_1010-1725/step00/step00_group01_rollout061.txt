            g = partial(jac_smooth, n=n, beta=beta)
            try:
                res = minimize(f, h, method='L-BFGS-B', jac=g,
                               bounds=[(0, 1)] * n,
                               options={'maxiter': 250, 'gtol': 1e-6, 'ftol': 1e-10})
                h_new = project_box_sum(res.x, n / 2.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
            except Exception:
                pass

        # Subgradient refinement
        for it in range(600):
            if time.time() - t0 > budget_s * 0.92:
                break
            grad = sg_grad(h, n)
            lr = 0.004 / (1.0 + 0.002 * it)
            h_new = project_box_sum(h - lr * grad, n / 2.0)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h)
            if c5_new <= c5_cur:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()

        # Local search: try shifting weight from worst-shift contributors
        if time.time() - t0 > budget_s * 0.92:
            break
        for _ in range(200):
            if time.time() - t0 > budget_s * 0.92:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            ks = int(corr.argmax())
            k = ks - (n - 1)
            # Find positions that contribute most to this shift
            i1 = np.arange(max(0, k), min(n, k + n))
            i2 = np.arange(max(0, -k), min(n, n - k))
            # Try small perturbation at a random position
            m_pos = np.random.randint(n)
            delta = np.random.choice([-0.02, 0.02])
            h_trial = h.copy()
            h_trial[m_pos] += delta
            h_trial = project_box_sum(h_trial, n / 2.0)
            c5_t = compute_c5(h_trial)
            if c5_t < compute_c5(h) - 1e-8:
                h = h_trial
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()

        # Random restart perturbations
        if time.time() - t0 > budget_s * 0.92:
            break
        for _ in range(5):
            h_r = np.clip(h + np.random.randn(n) * 0.04, 0.0, 1.0)
            h_r = project_box_sum(h_r, n / 2.0)
            c5_r = compute_c5(h_r)
            if c5_r < compute_c5(h):
                h = h_r
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h.copy()

    # Final polish with very fine steps on best solution
    if time.time() - t0 < budget_s * 0.97:
        h_f = best_h.copy()
        n_f = len(h_f)
        for it in range(1000):
            if time.time() - t0 > budget_s * 0.97:
                break
            grad = sg_grad(h_f, n_f)
            lr = 0.001 / (1.0 + 0.001 * it)
            h_tri = project_box_sum(h_f - lr * grad, n_f / 2.0)
            c5_tri = compute_c5(h_tri)
            if c5_tri < compute_c5(h_f):
                h_f = h_tri
                if c5_tri < best_c5:
                    best_c5 = c5_tri
                    best_h = h_f.copy()

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), len(best_h)
