So C_k as a function of h[i] is: (quadratic term) + (linear term) + (constant).
Specifically: C_k(h[i]) = -D_k * h[i]² + B_k * h[i] + A_k

where D_k = number of m such that m+k=i AND 0≤m<n (i.e., at most 1 for a given k... wait no)

Actually for a fixed k, h[i] appears:
- As h[m+k] when m = i-k (if valid): contributes h[i](1-h[i-k]) = h[i] - h[i]*h[i-k]
- As h[m] when m = i (if valid): contributes h[i+k](1-h[i]) = h[i+k] - h[i+k]*h[i]

So the terms involving h[i] in C_k are:
- h[i] - h[i]*h[i-k] (if 0≤i-k<n)
- h[i+k] - h[i+k]*h[i] (if 0≤i+k<n)

The quadratic term is: -h[i]*h[i-k] (if valid)
The linear terms are: h[i] - h[i+k]*h[i] (if valid) + h[i+k] (if valid)

So C_k(h[i]) has at most one quadratic term: -h[i]*h[i-k] (coefficient -1 if i-k is valid, 0 otherwise).

This means for a fixed k, C_k is at most quadratic in h[i], so maximizing over h[i] ∈ [0,1] can be done analytically.

But our objective is max over all k, so it's the max of many quadratic functions. Finding the minimum of this max in 1D can be done with bisection or by finding all intersection points and checking.

For a full coordinate descent pass (optimize h[0], then h[1], ..., h[n-1]), each step is O(n) (to evaluate all C_k at the new h[i]) and we need O(n) values of C_k, so O(n²) per coordinate, O(n³) per full pass.

For n=200, that's 8M per pass. Very fast. For n=1000, 1B per pass. A few seconds.

import numpy as np
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())


def compute_c5_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def subgradient(h, k_star):
    n = len(h)
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] += (-h[k:])
    else:
        ka = -k
        if ka < n:
            grad[:n - ka] += (1.0 - h[ka:])
            grad[ka:] += (-h[:n - ka])
    return grad


def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -2.0, 3.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = hp < 1.0
        else:
            mask = hp > 0.0
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp


def upsample(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)


def multi_subgradient(h, c_arr, top_frac=0.1):
    n = len(h)
    n_lags = len(c_arr)
    n_top = max(1, int(n_lags * top_frac))
    top_idx = np.argsort(c_arr)[-n_top:]
    top_idx = top_idx[c_arr[top_idx] > 0]
    if len(top_idx) == 0:
        top_idx = np.array([np.argmax(c_arr)])
    grad = np.zeros(n)
    for ks in top_idx:
        grad += subgradient(h, int(ks))
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = eval_c5(best_h)

    # Build grid sizes: start from init, grow by ~1.5x up to ~3000
    sizes = [n_init]
    n_cur = n_init
    for _ in range(8):
        n_next = int(n_cur * 1.5)
        n_next = min(n_next, 4000)
        if n_next > n_cur and n_next not in sizes:
            sizes.append(n_next)
        n_cur = n_next
    sizes = [s for s in sizes if s <= 4000]

    current_h = best_h.copy()
    current_n = n_init

    for target_n in sizes:
        if time.time() - t0 > budget_s * 0.88:
            break

        if target_n > current_n:
            current_h = upsample(current_h, target_n)
            current_h = project_feasible(current_h, target_n)
            current_n = target_n

        c5_cur = eval_c5(current_h)
        if c5_cur < best_c5:
            best_h = current_h.copy()
            best_c5 = c5_cur

        # Allocation: more steps for smaller grids
        time_left = budget_s * 0.90 - (time.time() - t0)
        if time_left < 5:
            break
        est_cost = max(1.0, current_n ** 2 * 5e-9)
        n_steps = int(min(5000, max(200, time_left * 0.4 / est_cost)))

        lr = 0.08
        stuck = 0
        best_local = c5_cur

        for it in range(n_steps):
            if time.time() - t0 > budget_s * 0.92:
                break

            c_arr = compute_c5_arr(current_h)
            k_star = int(np.argmax(c_arr))

            moved = False
            # Primary: subgradient step
            grad = subgradient(current_h, k_star)
            h_trial = project_feasible(current_h - lr * grad, current_n)
            c5_t = eval_c5(h_trial)
            if c5_t < c5_cur - 1e-9:
                current_h = h_trial
                c5_cur = c5_t
                moved = True
            else:
                # Multi-subgradient
                if rng.random() < 0.3:
                    mg = multi_subgradient(current_h, c_arr)
                    h_trial = project_feasible(current_h - lr * mg / (mg.max() + 1e-8), current_n)
                    c5_t = eval_c5(h_trial)
                    if c5_t < c5_cur - 1e-9:
                        current_h = h_trial
                        c5_cur = c5_t
                        moved = True
                # Swap move
                if not moved and rng.random() < 0.5:
                    i = int(rng.integers(0, current_n))
                    j = int(rng.integers(0, current_n))
                    if i != j:
                        d = float(rng.uniform(0.02, 0.15))
                        h_t2 = current_h.copy()
                        if h_t2[i] >= d and h_t2[j] <= 1.0 - d:
                            h_t2[i] -= d
                            h_t2[j] += d
                            c5_t2 = eval_c5(h_t2)
                            if c5_t2 < c5_cur - 1e-9:
                                current_h = h_t2
                                c5_cur = c5_t2
                                moved = True
                        elif h_t2[j] >= d and h_t2[i] <= 1.0 - d:
                            h_t2[i] += d
                            h_t2[j] -= d
                            c5_t2 = eval_c5(h_t2)
                            if c5_t2 < c5_cur - 1e-9:
                                current_h = h_t2
                                c5_cur = c5_t2
                                moved = True
                # Gaussian perturbation
                if not moved and rng.random() < 0.4:
                    scale = 0.005 + 0.02 * lr
                    h_t3 = project_feasible(current_h + rng.normal(0, scale, current_n), current_n)
                    c5_t3 = eval_c5(h_t3)
                    if c5_t3 < c5_cur - 1e-9:
                        current_h = h_t3
                        c5_cur = c5_t3
                        moved = True
                # Block complement
                if not moved and rng.random() < 0.15:
                    blen = int(rng.integers(2, max(3, current_n // 6)))
                    start = int(rng.integers(0, current_n - blen + 1))
                    h_t4 = current_h.copy()
                    h_t4[start:start + blen] = 1.0 - h_t4[start:start + blen]
                    h_t4 = project_feasible(h_t4, current_n)
                    c5_t4 = eval_c5(h_t4)
                    if c5_t4 < c5_cur - 1e-9:
                        current_h = h_t4
                        c5_cur = c5_t4
                        moved = True

            if moved:
                stuck = 0
                if c5_cur < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_cur
                if c5_cur < best_local:
                    best_local = c5_cur
            else:
                stuck += 1
                lr *= 0.96
                if stuck > 300:
                    break

            if it % 50 == 49:
                lr = max(0.005, min(0.15, lr * (1.02 if stuck == 0 else 0.97)))

        # Reset to best for next grid level
        if len(best_h) != current_n:
            current_h = project_feasible(upsample(best_h, current_n), current_n)
        else:
            current_h = best_h.copy()

    # Final polish: try small refinements at the best grid size
    if time.time() - t0 < budget_s * 0.97:
        best_h = project_feasible(best_h, len(best_h))
        
        for _ in range(200):
            if time.time() - t0 > budget_s * 0.98:
                break
            n_p = len(best_h)
            c_arr = compute_c5_arr(best_h)
            ks = int(np.argmax(c_arr))
            g = subgradient(best_h, ks)
            lr2 = 0.01
            h_t = project_feasible(best_h - lr2 * g, n_p)
            c5_t = eval_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t
                best_c5 = c5_t

    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
