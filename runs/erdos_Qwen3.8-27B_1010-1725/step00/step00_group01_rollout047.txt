```python
            h_minus[j] -= eps
            c_minus = np.correlate(h_minus, 1.0 - h_minus, mode="full")[k_star]
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best = h_new.copy()
            h = h_new
            # Decay learning rate
            lr *= 0.999
        else:
            lr *= 0.9
            h = best.copy()
    
    return best, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_func(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def eq_jac(h):
    return np.ones(len(h))

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    nf = int(np.sum(free))
    if nf > 0:
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(47) * 0.5
    n_parent = len(h_parent)
    h_parent = project_feasible(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h)

    n_list = sorted(set([n_parent, 40, 50, 60, 80, 100]))
    times_per_n = max(30, (budget_s - 60) // len(n_list))

    for n in n_list:
        t_elapsed = time.time() - t_start
        if t_elapsed > budget_s - 15:
            break
        n_deadline = min(t_start + budget_s - 10, time.time() + times_per_n)

        if n == n_parent:
            h0 = h_parent.copy()
        else:
            h0 = interpolate_h(best_h, n)
            h0 = project_feasible(h0, n)

        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

        # SLSQP from best/parent
        try:
            res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 1500, 'ftol': 1e-12})
            h_cand = np.clip(res.x, 0.0, 1.0)
            h_cand = project_feasible(h_cand, n)
            c5_cand = evaluate_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
                h0 = h_cand.copy()
        except Exception:
            pass

        # Nelder-Mead polish on best so far
        t_elapsed = time.time() - t_start
        if t_elapsed < n_deadline - 5:
            try:
                res = minimize(obj_func, h0, method='Nelder-Mead',
                               options={'maxiter': 300, 'xatol': 1e-9, 'fatol': 1e-12})
                h_cand = np.clip(res.x, 0.0, 1.0)
                h_cand = project_feasible(h_cand, n)
                c5_cand = evaluate_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
                    h0 = h_cand.copy()
            except Exception:
                pass

        # Random restarts
        n_restarts = 4
        for r in range(n_restarts):
            t_elapsed = time.time() - t_start
            if t_elapsed > n_deadline:
                break
            h_rand = rng.random(n)
            h_rand = project_feasible(h_rand, n)
            try:
                res = minimize(obj_func, h_rand, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 800, 'ftol': 1e-10})
                h_cand = np.clip(res.x, 0.0, 1.0)
                h_cand = project_feasible(h_cand, n)
                c5_cand = evaluate_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

        # Structured starts
        t_elapsed = time.time() - t_start
        if t_elapsed < n_deadline:
            h_struct = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
            h_struct = project_feasible(h_struct, n)
            try:
                res = minimize(obj_func, h_struct, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 500, 'ftol': 1e-10})
                h_cand = np.clip(res.x, 0.0, 1.0)
                h_cand = project_feasible(h_cand, n)
                c5_cand = evaluate_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

    # Final polish at best grid size
    t_elapsed = time.time() - t_start
    if t_elapsed < budget_s - 10:
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        iters = max(500, int((budget_s - t_elapsed - 5) * 50))
        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': iters, 'ftol': 1e-14})
            h_cand = np.clip(res.x, 0.0, 1.0)
            h_cand = project_feasible(h_cand, n_best)
            c5_cand = evaluate_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
        except Exception:
            pass

        # NM polish
        t_elapsed = time.time() - t_start
        if t_elapsed < budget_s - 5:
            try:
                res = minimize(obj_func, best_h, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-14})
                h_cand = np.clip(res.x, 0.0, 1.0)
                h_cand = project_feasible(h_cand, n_best)
                c5_cand = evaluate_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
