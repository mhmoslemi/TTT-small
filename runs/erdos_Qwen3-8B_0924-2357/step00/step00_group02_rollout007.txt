    start_time = time.time()
    while iteration < max_iter:
        if time.time() - start_time > budget_s:
            break
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_opt = np.argmax(overlaps)
        k_opt = np.clip(k_opt, 1, n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_opt < n_points:
                grad[i] += (1 - h[i + k_opt]) * dx
            if i - k_opt >= 0:
                grad[i] -= h[i - k_opt] * dx
        h_new = h - learning_rate * grad
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        new_c5 = overlaps[k_opt]
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            learning_rate *= 0.998
        if np.random.rand() < 0.1:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        iteration += 1
    overlaps_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(overlaps_best)
    return (best_h, best_c5, n_points)
