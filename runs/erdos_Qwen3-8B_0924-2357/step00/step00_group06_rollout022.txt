            else:
                term2 = 0.0
            grad[i] = term1 - term2

        # Update h_vals
        h_vals = h_vals - learning_rate * grad
        h_vals = np.clip(h_vals, 0, 1)
        scale = (n_points / 2) / np.sum(h_vals)
        h_vals = h_vals * scale

        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

        # Evaluate current C5
        current_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(current_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()

    return (best_h_vals, best_c5, n_points)
