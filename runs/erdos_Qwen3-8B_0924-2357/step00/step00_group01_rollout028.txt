import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    prev_best_c5 = np.inf
    learning_rate = 0.01
    iteration = 0
    time_start = time.time()
    while time.time() - time_start < budget_s:
        corr = np.correlate(h, 1 - h, mode="full")
        c5_current = np.max(corr) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        if abs(c5_current - prev_best_c5) < 1e-6 and iteration > 1000:
            break
        if c5_current >= prev_best_c5:
            learning_rate *= 0.9
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
