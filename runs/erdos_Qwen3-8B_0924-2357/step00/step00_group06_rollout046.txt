def run(seed=42, budget_s=600, **kwargs):

    max_iterations = 500
    start_time = time.time()

    # Compute initial C5
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h_vals = h_vals.copy()

    for _ in range(max_iterations):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
