import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    if not np.isclose(np.sum(h_initial), n_points / 2):
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, n_points)
    time_used = 0.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / len(h)
        return np.max(corr) * dx
    
    def compute_c5(h, n):
        return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)
    
    while time_used < budget_s:
        start = time.time()
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        time_used += time.time() - start
        if time_used >= budget_s:

        current_h = result.x
        current_c5 = compute_c5(current_h, n_points)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        perturb = np.random.normal(0, 0.05, n_points)
        perturb = np.clip(perturb, -0.2, 0.2)
        best_h += perturb

        best_h = best_h / np.sum(best_h) * (n_points / 2)
