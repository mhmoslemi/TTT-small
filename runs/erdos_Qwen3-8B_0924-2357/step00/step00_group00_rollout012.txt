import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)

    dx = 2.0 / n_points
    best_c5 = np.inf

    start_time = time.time()
    
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1)):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.random.rand() < 0.7:
            max_lag = np.argmax(corr)
            corr_k = corr[max_lag]
            if corr_k > 1e-6:
                grad = np.zeros(n_points)
                for j in range(n_points):
                    if j + max_lag < n_points:
