    phi_g = 1.5
    lambda_reg = 0.1
    
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    for _ in range(POP_SIZE):

        h = h_initial + perturbation
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            h = h * (n_points / 2) / sum_h
        particles.append(h)
        velocities.append(np.random.normal(0, 0.05, n_points))
        pbest.append(h.copy())
        pbest_fitness.append(0.0)
    
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    
    start_time = time.time()
    
    for _ in range(MAX_ITER):
        if time.time() - start_time > budget_s - 5:

        for i in range(POP_SIZE):
