import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 50
    dx = 2.0 / n_points
    npr.seed(seed)
    
    # Initialize h from provided values or random if not available
    if 'initial_h_values' in globals():

        h = np.zeros(n_points)

            if i < 25:

            else:
                h[i] = 0.0
    h = h / (n_points / 2.0)
    
    def compute_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * dx
    
    def update_h(h, idx, delta, idx2):
        h[idx] += delta
        h[idx2] -= delta
        h = np.clip(h, 0.0, 1.0)
        return h
    
    # Cost tracking
    best_c5 = np.inf
    best_h = h.copy()
    best_corr = compute_corr(h)
    best_c5 = np.max(best_corr)
    
    start_time = time.time()
    
    for _ in range(10000):  # Iterations with early stopping
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = compute_corr(h)
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        # Find worst lag
        max_idx = np.argmax(corr)
        if max_idx < len(corr) // 2:
            lag = max_idx
        else:
            lag = len(corr) - max_idx - 1
        lag = lag - len(h) + 1
        
        # Gradient for the worst lag
        grad = np.zeros(n_points)

        # Select index with max gradient
