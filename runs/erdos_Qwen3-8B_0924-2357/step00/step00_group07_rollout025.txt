import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 100
        initial_h = np.random.rand(n_points)
        desired_sum = n_points / 2
        initial_h *= desired_sum / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [initial_h.copy()]
    for _ in range(9):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        sum_p = np.sum(perturbed)
        scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
        perturbed *= scale
        population.append(perturbed)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(100):
        evaluated = [(compute_c5(h), h) for h in population]
        evaluated.sort()
        selected = [h for (c5, h) in evaluated[:20]]
        next_pop = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child = (selected[i] + selected[i+1]) / 2
            else:
                child = selected[i]
            idxs = np.random.choice(n_points, 5, replace=False)
            child[idxs] += np.random.normal(0, 0.05, 5)
            child = np.clip(child, 0, 1)
            sum_c = np.sum(child)
            scale = (n_points / 2) / sum_c if sum_c != 0 else 1.0
            child *= scale
            next_pop.append(child)
        while len(next_pop) < len(population):
