import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    target_sum = n_points / 2

    def compute_c5(h_vec):

    best_h = h.copy()
    best_c5 = compute_c5(h)

    time_start = time.time()
    epsilon = 0.02
    iter_count = 0
    max_iter = int(budget_s * 1000)  # 1000 iterations per second

    while iter_count < max_iter and time.time() - time_start < budget_s:
        iter_count += 1
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        if j == i:
            j = (j + 1) % n_points
        delta = np.random.uniform(-epsilon, epsilon)
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if epsilon > 0.0001:
