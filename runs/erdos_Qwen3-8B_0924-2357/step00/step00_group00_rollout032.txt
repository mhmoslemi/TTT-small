import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', [])
    n_points = len(initial_h_values) if initial_h_values else 45
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)
    sum_h = np.sum(h_initial)
    target_sum = n_points / 2
    
    if np.abs(sum_h - target_sum) > 1e-4:
        h_initial = (h_initial / sum_h) * target_sum
    
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    def constraint_func(h_flat):
        return np.sum(h_flat.reshape(n_points)) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    start_time = time.time()
    result = opt.minimize(
        objective_func,
        h_initial,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-9, 'eps': 1e-8}
    )
    
    elapsed_time = time.time() - start_time
    remaining_time = budget_s - elapsed_time
    
    if elapsed_time >= budget_s:
        return (result.x, objective_func(result.x), n_points)
    
    # Perturb and re-optimize if no significant improvement
    best_c5 = objective_func(result.x)
    best_h = result.x
    for _ in range(2):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(result.x + perturbation, 0, 1)
        perturbed_h = (perturbed_h / np.sum(perturbed_h)) * target_sum
        
        result_perturb = opt.minimize(
            objective_func,
            perturbed_h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-9, 'eps': 1e-8}
        )
        
        current_c5 = objective_func(result_perturb.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result_perturb.x
    
    return (best_h, best_c5, n_points)
