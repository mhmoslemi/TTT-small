    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        scale = target_sum / sum_h
        h = np.clip(h * scale, 0, 1)
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        g = np.zeros_like(x)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                g[i] += 1 - x[i + k]
            if i - k >= 0:
                g[i] -= x[i - k]
        return g * dx
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 200})
    h_opt = result.x
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr) * dx
    return (h_opt, c5, n_points)
