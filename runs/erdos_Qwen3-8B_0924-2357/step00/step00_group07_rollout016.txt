        h_initial = np.clip(np.array(h_initial), 0, 1)
        h_initial = h_initial / h_initial.sum() * target_sum
    else:
        n_points = 100
        h_initial = np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        target_sum = n_points / 2
        h_initial = h_initial / h_initial.sum() * target_sum

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def repair(h, target_sum):
        current_sum = h.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        h_scaled = h * scaling
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    population_size = 20
    population = [h_initial.copy() for _ in range(population_size)]

    start_time = time.time()
    best_h = h_initial.copy()
    best_c5 = compute_c5(h_initial)
    gen = 0

    while time.time() - start_time < budget_s:
        gen += 1
        fitness = [compute_c5(h) for h in population]
        indices = np.argsort(fitness)
        top_parents = [population[i] for i in indices[:10]]
        new_population = []

        for i in range(0, len(top_parents), 2):
            if i + 1 < len(top_parents):
                parent1, parent2 = top_parents[i], top_parents[i + 1]
                child = (parent1 + parent2) / 2
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = repair(child, target_sum)
                new_population.append(child)

        while len(new_population) < population_size:
            parent = np.random.choice(top_parents)
            noise = np.random.normal(0, 0.05, n_points)
            child = parent + noise
            child = repair(child, target_sum)
            new_population.append(child)

        population = new_population
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitness.index(current_best)]

    return (best_h, best_c5, n_points)
