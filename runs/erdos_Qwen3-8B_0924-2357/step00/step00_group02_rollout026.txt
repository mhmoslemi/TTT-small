        h = np.clip(h, 0, 1)
        
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            scaling_factor = (n_points / 2.0) / sum_h
            h *= scaling_factor
        
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if iteration % 100 == 0 and current_c5 >= best_c5:
            learning_rate *= 0.995
        
        if time.time() - start_time > budget_s:
            break
    
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.zeros(n_points)
    else:
        h = h * (n_points / 2.0 / sum_h)
    current_overlaps = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(current_overlaps)
    return (h, final_c5, n_points)
