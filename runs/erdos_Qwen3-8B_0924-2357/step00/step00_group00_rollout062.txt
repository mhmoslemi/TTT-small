def run(seed=42, budget_s=600, **kwargs):
    # Initial setup
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.clip(initial_h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2.0) / total
    h = h * scale
    best_c5 = np.inf
    best_h = h.copy()
    learning_rate = 0.01
    iterations = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute correlation and current C5
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        if current_c5 < 1e-12:
            current_c5 = 1e-12
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Gradient computation
        max_lag_idx = np.argmax(correlation)
        k = max_lag_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) * dx / current_c5
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k] * dx / current_c5

        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2.0) / total
        h = h * scale

        # Stochastic perturbation
        if iterations % 100 == 0:
            noise = 0.05 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h + noise, 0, 1)
            total = np.sum(h)
            scale = (n_points / 2.0) / total
            h = h * scale

        iterations += 1

    # Final evaluation
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    return (best_h, final_c5, n_points)
