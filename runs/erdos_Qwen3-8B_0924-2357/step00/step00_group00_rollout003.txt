def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import Bounds, minimize

    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values)
    else:
        n_points = 450

        h_values = np.clip(h_values, 0, 1)

    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h_values.copy()
    start_time = time.time()

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return corr.max()

    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = np.argmax(corr)
        lag = idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):

                if i + lag < n_points:

                if i - lag >= 0:
                    grad[i] -= dx * h[i - lag]
            else:
                m = -lag
                if i - m >= 0:
                    grad[i] -= dx * h[i - m]
                if i + m < n_points:
                    grad[i] += dx * (1 - h[i + m])
        return -grad

    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
