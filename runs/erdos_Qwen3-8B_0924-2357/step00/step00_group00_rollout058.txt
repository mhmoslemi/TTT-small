import numpy as np
import scipy.optimize as opt
import time

# Pre-imported helpers
initial_h_values = None  # Assume to be provided as a global

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine n_points from initial_h_values or default to 450
    n_points = len(initial_h_values) if initial_h_values is not None else 450
    dx = 2.0 / n_points
    
    # Ensure initial_h_values meets the sum constraint
    if initial_h_values is None:
        h = np.ones(n_points) * 0.5
    else:
        h = np.array(initial_h_values)
        total_sum = np.sum(h)
        if abs(total_sum - n_points/2) > 1e-6:
            scale = (n_points / 2) / total_sum
            h *= scale
    
    def obj(h):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(cross_corr)
    
    def grad(h):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        k = np.argmax(cross_corr)
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            if i + k < n:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0., 1.) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = obj(h)
    iter_count = 0
    max_iter = 100
    
    while time.time() - start_time < budget_s and iter_count < max_iter:

        result = opt.minimize(obj, best_h, method='SLSQP', jac=grad, bounds=bounds, constraints=constraints, tol=1e-6)
