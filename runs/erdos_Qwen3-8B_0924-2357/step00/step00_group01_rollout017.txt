def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    np.random.seed(seed)
    initial_h = initial_h_values  # Global provided variable
    n_points = len(initial_h)

    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    eps = 0.01
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        original_h = h.copy()
        i = np.random.randint(n_points)
        original_h_i = h[i]
        delta = np.random.uniform(-eps, eps)
        new_h_i = original_h_i + delta
        if new_h_i < 0:
            new_h_i = 0
            delta_adj = original_h_i
        elif new_h_i > 1:
            new_h_i = 1
            delta_adj = 1 - original_h_i
        else:
            delta_adj = delta
        j = np.random.choice([k for k in range(n_points) if k != i])
        original_h_j = h[j]
        new_h_j = original_h_j - delta_adj
        if new_h_j < 0:
            new_h_j = 0
        elif new_h_j > 1:
            new_h_j = 1
        h[i] = new_h_i
        h[j] = new_h_j
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
