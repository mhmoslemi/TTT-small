import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = 0.0
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate_initial = 0.1
    start_time = time.time()
    
    for iteration in range(max_iter):
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        current_c5 = np.max(corr)
        
        # Update best_c5 and best_h
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Find the lag with maximum value
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] += 1.0 - h[i + k]
            if i - k >= 0 and i - k < n_points:
                grad[i] -= h[i - k]
        
        # Update h
        learning_rate = learning_rate_initial * (1 - iteration / max_iter)
        h -= learning_rate * grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= (n_points / 2.0) / sum_h
        
        # Early stopping if target is reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h, best_c5, n_points)
