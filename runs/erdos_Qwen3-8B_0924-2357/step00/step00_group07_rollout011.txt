    gbest = positions[pbest_fitness.index(best_fitness)].copy()
    
    omega = 0.729
    phi_p = 1.494
    phi_g = 1.494
    start_time = time.time()
    
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        for i in range(pop_size):
            current_h = positions[i]
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            current_fitness = np.max(corr) * 2 / n_points
            if current_fitness < pbest_fitness[i]:
                pbest_fitness[i] = current_fitness
                pbest[i] = current_h.copy()
                if current_fitness < best_fitness:
                    best_fitness = current_fitness
                    gbest = current_h.copy()
        
        for i in range(pop_size):
            current_h = positions[i]
            current_v = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = phi_p * r1 * (pbest[i] - current_h)
            social = phi_g * r2 * (gbest - current_h)
            new_v = omega * current_v + cognitive + social
            velocities[i] = new_v
            new_h = current_h + new_v
            new_h = np.clip(new_h, 0, 1)
            sum_new_h = np.sum(new_h)
            if sum_new_h == 0:
                new_h = np.zeros_like(new_h)
            else:
                scale = (n_points / 2) / sum_new_h
                new_h = new_h * scale
            positions[i] = new_h
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return (gbest, best_fitness, n_points)
