            # Ensure it's not the same as idx
            if min_idx == idx:
                min_idx = np.argmin(new_h[1:])
                min_idx += 1
            new_h[min_idx] += diff
        # Clip values to [0, 1]
        new_h = np.clip(new_h, 0.0, 1.0)
        # Update h
        h = new_h
        # Early stopping
        if np.max(np.abs(h - best_h)) < 1e-5:
            break

    return best_h, best_c5, n_points
