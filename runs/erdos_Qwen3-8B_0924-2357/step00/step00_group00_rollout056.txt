import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    sum_h = np.sum(h_initial)
    adjustment = (n_points / 2 - sum_h) / n_points
    h_initial += adjustment
    bounds = [(0.0, 1.0)] * n_points
    def constraint(x):
        return np.sum(x) - n_points / 2
    def objective(x):
        h = x
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = corr.max()
        return max_corr * (2.0 / n_points)
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    best_h = result.x
    best_c5 = objective(best_h)
    for _ in range(3):
        perturbation = np.random.normal(0, 0.03, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        adjustment = (n_points / 2 - new_sum) / n_points
        new_h += adjustment
        new_h = np.clip(new_h, 0.0, 1.0)
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
        curr_c5 = objective(res.x)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = res.x
    return (best_h, best_c5, n_points)
