            break
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n / 2) > 1e-6:
        dev = s - n / 2
        h[np.argmax(h)] -= dev
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (h, final_c5, n)
