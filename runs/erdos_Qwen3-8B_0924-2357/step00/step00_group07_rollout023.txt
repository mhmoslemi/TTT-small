import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h_clipped = np.clip(h, 0, 1)

            return float('inf')
