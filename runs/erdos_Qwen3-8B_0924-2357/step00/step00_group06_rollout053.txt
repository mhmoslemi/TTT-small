import time

        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        h_vals = np.ones(n_points) / n_points

    scale = (n_points / 2) / np.sum(h_vals)

    tolerance = 1e-5

    best_h_vals = h_vals.copy()
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Compute cross-correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        current_c5 = corr.max() * dx
        
        # Early stopping
        if current_c5 < best_c5 - tolerance:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        elif np.abs(current_c5 - best_c5) < tolerance:

        # If not improved, break early
        if current_c5 >= best_c5:
            break
        
        # Find the most critical lag
        k_max = np.argmax(corr)
        # Compute gradient
        grad = np.zeros(n_points)

            grad[i] = (val1 - val2) / (2 * epsilon)
        
        # Update h_vals
        h_vals -= learning_rate * grad
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize sum constraint
