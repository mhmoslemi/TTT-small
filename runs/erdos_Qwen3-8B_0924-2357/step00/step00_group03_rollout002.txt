import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import minimize

    # Initialize with provided initial values or a random feasible vector
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 500
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
        n_points = len(h)

    dx = 2.0 / n_points

    # Define objective and gradient
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(corr * dx)
        return max_val

    def gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(corr)
        idx = np.argmax(corr)
        n = len(h_vec)
        if idx < n:
            lag = (n - 1) - idx
        else:
            lag = idx - (n - 1)
        grad = np.zeros(n)
        dx_val = dx
        for j in range(n):
            j_plus = j + lag
            if 0 <= j_plus < n:
                t1 = (1.0 - h_vec[j_plus]) * dx_val
            else:
                t1 = 0.0
            j_minus = j - lag
            if 0 <= j_minus < n:
                t2 = -h_vec[j_minus] * dx_val
            else:
                t2 = 0.0
            grad[j] = t1 + t2
        return grad

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    # Optimization settings
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Ensure feasible solution
    h_opt = np.clip(result.x, 0.0, 1.0)
    scale = (n_points / 2) / np.sum(h_opt)
    h_opt *= scale

    # Final evaluation
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr * dx)
    return (h_opt, c5_bound, n_points)
