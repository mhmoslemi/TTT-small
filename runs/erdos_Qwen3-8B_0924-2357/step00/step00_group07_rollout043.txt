import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h = kwargs.get('initial_h_values', None)
    n_points = len(initial_h) if initial_h is not None else 100
    np.random.seed(seed)

    if initial_h is not None:
        initial_h = np.array(initial_h)
        sum_initial = np.sum(initial_h)
        scale = (n_points / 2) / sum_initial
        h_initial = np.clip(initial_h * scale, 0, 1)
    else:
        h_initial = np.random.rand(n_points)
        sum_initial = np.sum(h_initial)
        scale = (n_points / 2) / sum_initial
        h_initial = np.clip(h_initial * scale, 0, 1)

    n_particles = 50
    max_iter = 1000
    omega = 0.729
    phi_p, phi_g = 1.494, 1.494
    dx = 2.0 / n_points

    particles = []
    for _ in range(n_particles):
        if _ == 0 and initial_h is not None:
            h = h_initial.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h
            h = h * scale
        v = np.random.rand(n_points) * 0.1
        p = {'h': h, 'v': v, 'pbest': h.copy(), 'pbest_c5': float('inf')}
        particles.append(p)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for p in particles:
        p['pbest_c5'] = compute_c5(p['h'])

    gbest = min(particles, key=lambda x: x['pbest_c5'])

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time >= budget_s:
            break
        for p in particles:
            c5 = compute_c5(p['h'])
            if c5 < p['pbest_c5']:
                p['pbest_c5'] = c5
                p['pbest'] = p['h'].copy()
        gbest = min(particles, key=lambda x: x['pbest_c5'])
        for p in particles:
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            p['v'] = omega * p['v'] + phi_p * r1 * (p['pbest'] - p['h']) + phi_g * r2 * (gbest['h'] - p['h'])
            p['h'] += p['v']
            p['h'] = np.clip(p['h'], 0, 1)
            sum_h = np.sum(p['h'])
            if sum_h != 0:
                p['h'] *=(n_points/2)/sum_h
            else:
                p['h'] = np.random.rand(n_points)
                sum_h = np.sum(p['h'])
                p['h'] *=(n_points/2)/sum_h

    best_h = gbest['pbest']
    best_c5 = gbest['pbest_c5']
    return (best_h, best_c5, n_points)
