            h = np.clip(h + noise, 0, 1)
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.full(n_points, 0.5)
            else:
                h = h * (target_sum / current_sum)
        if c5 >= best_c5:
            learning_rate *= 0.998
    
    return (best_h, best_c5, n_points)
