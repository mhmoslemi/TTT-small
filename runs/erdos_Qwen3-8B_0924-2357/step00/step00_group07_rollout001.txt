import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def repair(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / s
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    def eval_c5(h):
        o = 1 - h
        c = np.correlate(h, o, mode='full')
        return np.max(c) * dx
    
    pop = []
    for _ in range(20):
        h = np.copy(h_initial)
        h += np.random.normal(0, 0.05, n_points)
        h = repair(h)
        pop.append(h)
    
    best_h = np.copy(h_initial)
    best_c5 = eval_c5(best_h)
    
    import time
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        cand = [(eval_c5(h), h) for h in pop]
        cand.sort()
        top = cand[:10]
        new_pop = []
        for _ in range(20):
            p1, p2 = np.random.choice([c[1] for c in top], 2, replace=False)
            child = (p1 + p2) / 2
            idxs = np.random.choice(n_points, 5, replace=False)
            child[idxs] += np.random.normal(0, 0.05, 5)
            child = repair(child)
            new_pop.append(child)
        pop = new_pop
        for c, h in cand:
            if c < best_c5:
                best_c5 = c
                best_h = h
    
    return (best_h, best_c5, n_points)
