            noise = np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new + noise, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum != 0:
                h_new = h_new * (n_points / 2) / h_new_sum
        
        h = h_new.copy()
        if c5 >= best_c5:
            learning_rate *= 0.998
        iteration += 1
    
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    return (best_h, final_c5, n_points)
