        cross = np.correlate(x, 1 - x, mode='full')
        max_idx = np.argmax(cross)
        lag = max_idx - (n_points - 1)
        g = np.zeros_like(x)
        for i in range(n_points):
            if 0 <= i - lag < n_points:
                g[i] -= x[i - lag] * dx
            if 0 <= i + lag < n_points:
                g[i] += (1 - x[i + lag]) * dx
        return g

    bounds = [(0., 1.) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    start = time.time()
    res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds, constraints=cons, tol=1e-9, options={'maxiter': 1000})
    elapsed = time.time() - start
    if elapsed >= budget_s:
        c5 = obj(h)
        return (h, c5, n_points)
    return (res.x, res.fun, n_points)
