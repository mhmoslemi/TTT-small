    convergence_threshold = 1e-6
    convergence_count = 0
    
    while iteration < max_iter:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            history.append((h.copy(), current_c5))
            convergence_count = 0
        else:
            convergence_count += 1
        
        if convergence_count >= 10:
            break
        
        k_max = np.argmax(corr)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if j + k_max < n_points:
                term1 = 1.0 - h[j + k_max]
            else:
                term1 = 0.0
            if j - k_max >= 0:
                term2 = h[j - k_max]
            else:
                term2 = 0.0
            gradient[j] = term1 - term2
        
        perturb = np.zeros(n_points)
        for j in range(n_points):
            if j + k_max < n_points:
                perturb[j] += 0.5 * (1.0 - h[j]) * gradient[j]
            if j - k_max >= 0:
                perturb[j] -= 0.5 * h[j] * gradient[j]
        
        scale_factor = 0.05 * (1.0 - 0.95 ** (iteration / 100))
        h += perturb * scale_factor
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        iteration += 1
    
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h
    h = h * scaling_factor
    final_c5 = evaluate_erdos_solution(h)
    return (h, final_c5, n_points)
