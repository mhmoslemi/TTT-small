        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
        if iteration % perturbation_interval == 0:
            noise = np.random.normal(0, perturbation_strength, n_points)
            mask = np.random.rand(n_points) < 0.2
            current_h += noise * mask
            current_h = np.clip(current_h, 0.0, 1.0)
            current_sum = np.sum(current_h)
            if current_sum != 0:
                current_h = current_h * (target_sum / current_sum)
        learning_rate *= decay_rate
    return (best_h, best_c5, n_points)
