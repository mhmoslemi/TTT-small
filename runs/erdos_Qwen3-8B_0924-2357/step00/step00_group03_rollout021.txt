    def scale_to_sum(h, n_points):

        if current_sum == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / current_sum
        return h * scale
    
    def objective(x):
        nonlocal best_c5
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)

        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

        return np.clip(x, 0, 1)
