import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
    result = opt.minimize(obj_func, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)

    best_h = h
    best_c5 = obj_func(h)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(3):

            perturbed += (target_sum - np.sum(perturbed)) / n_points
            res = opt.minimize(obj_func, perturbed, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)

                best_h = res.x
                best_c5 = res.fun
    return (best_h, best_c5, n_points)
