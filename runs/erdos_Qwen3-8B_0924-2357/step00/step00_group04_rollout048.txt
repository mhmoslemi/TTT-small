import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iter = int(600 / 1.0)  # 600 seconds budget
    learning_rate = 0.01
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_idx = np.argmax(corr)
        l = lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + l
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = h * (n_points / 2) / total
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, n_points
