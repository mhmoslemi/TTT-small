def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    h = initial_h_values * scale

    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')[-n_points:] * dx)
    learning_rate = 0.07
    max_iter = 600
    iteration = 0
    time_start = time.time()
    
    while iteration < max_iter and time.time() - time_start < budget_s:
        correlation = np.correlate(h, 1 - h, mode='full')
        overlaps = correlation[-n_points:] * dx
        c5 = np.max(overlaps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k_opt = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_opt < n_points:
                grad[i] += (1 - h[i + k_opt]) * dx
            if i - k_opt >= 0:
                grad[i] -= h[i - k_opt] * dx
        h = h - learning_rate * grad

        sum_h = np.sum(h)
