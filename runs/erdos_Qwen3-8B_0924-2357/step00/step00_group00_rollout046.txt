import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals()['initial_h_values']
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values).astype(float)
    
    # Clip and rescale to meet constraints
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0, 1)
    
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    
    learning_rate = 0.01
    for iter in range(1, 2000):  # Conservative upper limit
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient for max lag
        max_lag_index = np.argmax(corr)
        k = max_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += dx * (1 - h[j + k]) / current_c5
            if 0 <= j - k < n_points:
                grad[j] -= dx * h[j - k] / current_c5
        
        # Update h
        h -= learning_rate * grad
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        
        # Stochastic perturbation every 100 iterations
        if iter % 100 == 0:
            noise = 0.05 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
    
    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)
