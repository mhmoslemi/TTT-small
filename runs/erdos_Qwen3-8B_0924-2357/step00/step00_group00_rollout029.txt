        perturb[valid] -= 0.5 * current_h[np.arange(n_points)][valid] * grad[valid]
        scal = 0.05 * (1 - 0.95 ** (iter / 100))
        current_h += perturb * scal
        current_h = np.clip(current_h, 0, 1)
        if np.sum(current_h) < n_points / 2:
            current_h = np.clip(current_h * (n_points / 2) / np.sum(current_h), 0, 1)
        elif np.sum(current_h) > n_points / 2:
            current_h = np.clip(current_h * (n_points / 2) / np.sum(current_h), 0, 1)
        if np.datetime64('now') - time_start > np.timedelta64(budget_s, 's'):
            break
    final_h = best_h if evaluate_erdos_solution(best_h) < evaluate_erdos_solution(current_h) else current_h
    return (final_h, best_c5, n_points)
