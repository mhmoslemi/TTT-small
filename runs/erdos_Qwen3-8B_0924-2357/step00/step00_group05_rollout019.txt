    # Get initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    max_iter = 1000
    lr = 0.1
    decay = 0.999
    early_stop = 50
    no_improvement = 0
    for it in range(max_iter):
        # Compute overlaps
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        # Find maximum value
        c5 = np.max(cross_corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > early_stop:
                break
        # Find lag that gives max
        max_idx = np.argmax(cross_corr)
        # Convert index to lag
        lag = max_idx - (n_points - 1)
        # Compute gradient for this lag
        k_max = lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i + k_max] += h[i]
            if i - k_max >= 0:
                grad[i - k_max] += h[i]
        grad /= n_points
        # Update h
        h -= lr * grad
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to sum n_points/2
        h = h * (n_points / 2) / np.sum(h)

            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        # Decay learning rate
        lr *= decay
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):

        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(cross_corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > early_stop:
                break
        
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)
        k_max = lag
        
        # Compute gradient for lag k_max
        if k_max < 0:
            shift = -k_max
            grad = np.roll(h, shift)
        else:
            shift = k_max
            grad = np.roll(h, -shift)

        h -= lr * grad
