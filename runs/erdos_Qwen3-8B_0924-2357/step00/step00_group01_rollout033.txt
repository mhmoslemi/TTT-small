        current_c5 = compute_c5(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        iter_count += 1
        if iter_count % 1000 == 0:
            epsilon *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
