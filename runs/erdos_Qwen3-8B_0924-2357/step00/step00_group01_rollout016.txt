        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        diff = new_sum - target_sum
        if abs(diff) > 1e-6:
            idx = np.where((new_h > 1e-6) & (new_h < 1 - 1e-6))[0]
            if idx.size > 0:
                idx = idx[0]
                new_h[idx] += diff
        new_h = np.clip(new_h, 0.0, 1.0)
        curr_c5 = compute_c5(new_h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = new_h.copy()
    return (best_h, best_c5, n_points)
