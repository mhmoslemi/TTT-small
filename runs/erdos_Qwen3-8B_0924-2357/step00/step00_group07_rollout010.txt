def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    def scale_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2.0) / total if total != 0 else 1.0
        return h * scale

    pop_size = 50
    particles = []
    velocities = []
    pbest = []
    best_c5 = float('inf')
    gbest = None

    # Initialize particles
    if 'initial_h_values' in globals():
        h_initial = scale_h(initial_h)
        particles.append(h_initial)
        velocities.append(np.zeros(n_points))
        pbest.append(h_initial.copy())
    else:
        for _ in range(pop_size):
            h = np.random.uniform(0, 1, n_points)
            h = scale_h(h)
            particles.append(h)
            velocities.append(np.random.uniform(-1, 1, n_points))
            pbest.append(h.copy())

    # Initialize global best
    gbest = pbest[0].copy()
    best_c5 = float('inf')

    omega = 0.729
    phi_p = 1.494
    phi_g = 1.494
    max_iter = 500
    start_time = time.time()

        if current_time - start_time > budget_s:
            break

        # Evaluate all particles
        for i in range(len(particles)):

            corr = np.correlate(h, 1 - h, mode='full')

            # Update personal best

                pbest[i] = h.copy()
