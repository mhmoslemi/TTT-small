            h[idx2] -= delta
        else:
            idx2 = np.argmin(h) if h[idx] > 0.5 else np.argmax(h)
            h[idx2] -= delta
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h / total * (n_points / 2)
        prev_best_c5 = c5
        iteration += 1

    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h = h / total * (n_points / 2)
    h = np.clip(h, 0, 1)
    best_c5 = compute_c5(best_h)
    return (h, best_c5, n_points)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / len(h))
