            if 0 <= x_plus_k < n_points:
                grad[i] += (1 - h[x_plus_k])
            x_minus_k = i - k
            if 0 <= x_minus_k < n_points:
                grad[i] -= h[x_minus_k]
        eta = initial_lr * (1 - t / max_iter)
        h -= eta * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= target_sum / current_sum
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 <= 0.3807:
            break
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
