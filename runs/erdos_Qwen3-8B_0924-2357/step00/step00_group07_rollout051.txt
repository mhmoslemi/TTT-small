import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values).copy()
        h_initial = np.clip(h_initial, 0, 1)
        scale = (n_points / 2) / np.sum(h_initial)
        h_initial *= scale
    else:
        h_initial = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h_initial)
        h_initial *= scale
    particles = [{'h': h_initial, 'v': np.zeros(n_points), 'pbest': h_initial.copy(), 'pbest_score': 0.0}]
    for _ in range(49):
        h = np.random.rand(n_points)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
        h = np.clip(h, 0, 1)
        particles.append({'h': h, 'v': np.zeros(n_points), 'pbest': h.copy(), 'pbest_score': 0.0})
    gbest = min(particles, key=lambda p: p['pbest_score'])
    start_time = time.time()
    omega = 0.729
    phi_p = 1.494
    phi_g = 1.494
    max_iter = int(budget_s / 2)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            corr = np.correlate(p['h'], 1 - p['h'], mode='full')
            score = np.max(corr) * dx
            p['pbest_score'] = score
        for p in particles:
            if p['pbest_score'] < p['pbest_score']:
                p['pbest'] = p['h'].copy()
                p['pbest_score'] = p['pbest_score']
        gbest = min(particles, key=lambda p: p['pbest_score'])
        for p in particles:
            rand1 = np.random.rand(n_points)
            rand2 = np.random.rand(n_points)
            p['v'] = omega * p['v'] + phi_p * rand1 * (p['pbest'] - p['h']) + phi_g * rand2 * (gbest['pbest'] - p['h'])
