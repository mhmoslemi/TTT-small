import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    n_points = 42
    target_mass = n_points / 2.0
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * target_mass / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 200
    learning_rate = 0.1
    perturb_prob = 0.1

    for _ in range(max_iterations):
        # Compute autocorrelation for valid lags
        valid_lags = np.arange(1, n_points)
        autocorr_values = []
        for k in valid_lags:
            idx = n_points - 1 + k
            autocorr = np.correlate(h, h, mode='full')[idx]
            autocorr_values.append(autocorr)
        min_idx = np.argmin(autocorr_values)
        s = valid_lags[min_idx]
        
        # Compute gradient for lag s
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if j + s < n_points:
                gradient[j] += h[j + s]
            if j - s >= 0:
                gradient[j] += h[j - s]
        
        # Update h
        h_new = h + learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = h_new.sum()
        if h_new_sum != 0:
            h = h_new * (target_mass / h_new_sum)
        else:
            h = h_new
        
        # Random perturbation
        if np.random.rand() < perturb_prob:
            lambda_pert = np.random.uniform(0.05, 0.15) * target_mass
            indices_add = np.random.choice(n_points, size=5, replace=False)

            h_pert = h.copy()
            h_pert[indices_add] += delta
            h_pert[indices_sub] -= delta

                h = h_pert * (target_mass / h_pert_sum)
            else:
                h = h_pert
        
        # Update best solution
        cross_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(cross_corr) * (2.0 / n_points)

            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
