        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i] * 0.7 + 1.5 * r1 * (pbest[i] - population[i]) + 1.5 * r2 * (gbest - population[i])
            velocities[i] = v
            population[i] += v
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
            new_fitness = compute_c5(population[i])
            if new_fitness < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = new_fitness
            if new_fitness < gbest_fitness:
                gbest = population[i].copy()
                gbest_fitness = new_fitness

    initial_c5 = compute_c5(initial_h)
    if initial_c5 < gbest_fitness:
        return (initial_h, initial_c5, n_points)
    else:
        return (gbest, gbest_fitness, n_points)
