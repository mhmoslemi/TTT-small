            if current_c5 < global_c5:
                global_c5 = current_c5
                global_best = h_p.copy()
        for i in range(pop_size):
            v = velocities[i] * 0.8 + np.random.rand() * (personal_best_pos[i] - particles[i]) * 1.5 + np.random.rand() * (global_best - particles[i]) * 1.5
            velocities[i] = v * decay
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = (particles[i] * (n_points/2.0)) / np.sum(particles[i])
        iter_count += 1
        if iter_count > patience and global_c5 >= best_c5:
            break
        best_c5 = global_c5
    return (global_best, best_c5, n_points)
