            current_sum = np.sum(current_h)
            if current_sum == 0:
                current_h = np.zeros(n_points)
            else:
                current_h = current_h * (n_points / 2) / current_sum
        
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
