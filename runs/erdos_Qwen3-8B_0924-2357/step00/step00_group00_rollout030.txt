    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = np.clip(h * scale, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = float('inf')

    learning_rate = 0.01
    perturb_interval = 100
    perturb_magnitude = 0.05

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)

        indices = np.arange(n_points)
        valid_j_plus_k = (indices + k) < n_points
        term1 = np.zeros(n_points)
        term1[valid_j_plus_k] = 1.0 - h[indices[valid_j_plus_k] + k]

        valid_j_minus_k = (indices - k) >= 0
        term2 = np.zeros(n_points)
        term2[valid_j_minus_k] = h[indices[valid_j_minus_k] - k]

        grad = (term1 - term2) * dx / current_c5

        h -= learning_rate * grad

        h = np.clip(h, 0.0, 1.0)
