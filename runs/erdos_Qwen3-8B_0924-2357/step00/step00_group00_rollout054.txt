    dx = 2.0 / n_points

    learning_rate = 0.01

        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)

import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy is available
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = np.clip(h * scale, 0, 1)
    best_c5 = float('inf')
    best_h = h.copy()
    dx = 2.0 / n_points
    start_time = time.time()
    learning_rate = 0.01
    perturb_freq = 100
    perturb_magnitude = 0.05

    for iter in range(int(budget_s * 2)):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        max_lag_index = np.argmax(corr)
        actual_lag = max_lag_index - (n_points - 1)
        current_max_corr = corr[max_lag_index]
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + actual_lag < n_points:
                grad[j] += dx * (1 - h[j + actual_lag]) / current_max_corr
            if 0 <= j - actual_lag < n_points:
                grad[j] -= dx * h[j - actual_lag] / current_max_corr
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        if iter % perturb_freq == 0:
            noise = perturb_magnitude * (np.random.rand(n_points) - 0.5)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    evaluated_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-5)
    return (best_h, evaluated_c5, n_points)
