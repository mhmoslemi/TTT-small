import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    eps = 0.01
    no_improvement = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while j == i:
            j = np.random.randint(n_points)
        delta = np.random.uniform(-eps, eps)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = np.clip(h_new, 0.0, 1.0)
        current_c5 = compute_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > 1000:
            eps *= 0.5
            no_improvement = 0
        h = h_new

    return (best_h, best_c5, n_points)
