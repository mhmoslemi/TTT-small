import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def mutate(h, target_sum):
    noise = np.random.normal(0, 0.05, len(h))
    h = h + noise
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    scale_factor = target_sum / current_sum
    return h * scale_factor

def crossover(parent1, parent2):
    return (parent1 + parent2) / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = [initial_h.copy()]
    for _ in range(19):
        h = initial_h.copy()
        h += np.random.normal(0, 0.05, len(h))
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = target_sum / sum_h
        h *= scale_factor
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()

        c5_values = np.array([compute_c5(h, dx) for h in population])
        top_indices = np.argsort(c5_values)[:10]
