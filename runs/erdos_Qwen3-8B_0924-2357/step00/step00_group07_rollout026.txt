        if time.time() - time_start > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            p['velocity'] = omega * p['velocity'] + phi_p * r1 * (p['pbest'] - p['position']) + phi_g * r2 * (gbest['position'] - p['position'])
            p['position'] += p['velocity']
            p['position'] = np.clip(p['position'], 0, 1)
            sum_pos = np.sum(p['position'])
            if sum_pos != 0:
                scale = (n_points / 2) / sum_pos
                p['position'] *= scale
            p['position'] = np.clip(p['position'], 0, 1)
            current_c5 = evaluate_c5(p['position'])
            if current_c5 < p['pbest_value']:
                p['pbest_value'] = current_c5
                p['pbest'] = p['position'].copy()
        for p in particles:
            if p['pbest_value'] < best_c5:
                best_c5 = p['pbest_value']
                gbest = p
    
    return (gbest['pbest'], best_c5, n_points)
