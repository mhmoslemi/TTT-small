        n_points = 50
        h = np.zeros(n_points)
        n_ones = n_points // 2
        idx = np.random.choice(n_points, n_ones, replace=False)
        h[idx] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    h_sum = h.sum()
    if abs(h_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full') * dx)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        m = np.argmax(corr)
        k = m - n_points + 1
        
        # Compute gradient for worst lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = j + k
            j_minus_k = j - k
            if 0 <= j_plus_k < n_points and 0 <= j_minus_k < n_points:
                grad[j] = (1.0 - h[j_plus_k]) - h[j_minus_k]
            else:
                grad[j] = 0.0
        
        if np.abs(grad).max() < 1e-8:
            break  # No improvement possible
        
        j_max = np.argmax(np.abs(grad))
        if grad[j_max] > 0:
            delta = h[j_max]
        elif grad[j_max] < 0:
            delta = 1.0 - h[j_max]
        else:
            break  # Gradient is zero
        
        # Choose j_prime to compensate
        j_prime = 0
        while j_prime == j_max or h[j_prime] >= 1.0:
            j_prime += 1
            if j_prime >= n_points:
                j_prime = 0
        
        # Apply adjustment
        h[j_max] += -delta if grad[j_max] > 0 else delta
        h[j_prime] += delta if grad[j_max] > 0 else -delta
        
        # Clip and normalize
        h[j_max] = np.clip(h[j_max], 0.0, 1.0)
        h[j_prime] = np.clip(h[j_prime], 0.0, 1.0)
        h_sum = h.sum()
        if abs(h_sum - n_points / 2) > 1e-8:
            # Fix sum
            h_adj = h.copy()
            delta_sum = (n_points / 2) - h_sum
            j_adj = 0
            while j_adj == j_max or j_adj == j_prime or h_adj[j_adj] >= 1.0:
                j_adj += 1
                if j_adj >= n_points:
                    j_adj = 0
            h_adj[j_adj] += delta_sum
            h_adj[j_adj] = np.clip(h_adj[j_adj], 0.0, 1.0)
            h = h_adj
        
        # Update best solution
        curr_corr = np.correlate(h, 1.0 - h, mode='full') * dx
        curr_c5 = curr_corr.max()
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        if abs(curr_c5 - best_c5) < 1e-8:
            break
    
    return best_h, best_c5, len(best_h)
