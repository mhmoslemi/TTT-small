import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    h = initial_h_values.copy()
    dx = 2.0 / n_points

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr * dx)

    def compute_gradient(h_vals, k):
        n = len(h_vals)
        grad = np.zeros(n)
        for i in range(n):
            if i + k < n:
                grad[i] += (1 - h_vals[i + k])
            if i - k >= 0:
                grad[i] -= h_vals[i - k]
        return grad

    def line_search(h_vals, direction, max_iter=50):
        alpha = 1.0
        for _ in range(max_iter):
            h_candidate = h_vals + alpha * direction
            h_candidate = np.clip(h_candidate, 0, 1)
            s = np.sum(h_candidate)
            scale = target_sum / s if s != 0 else 1.0
            h_scaled = h_candidate * scale
            new_c5 = compute_c5(h_scaled)
            if new_c5 < compute_c5(h_vals):
                return alpha
            alpha *= 0.5
        return alpha

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr * dx)
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = compute_gradient(h, k)
        direction = -grad
        alpha = line_search(h, direction)
        h_new = h + alpha * direction
        h_clipped = np.clip(h_new, 0, 1)
        s = np.sum(h_clipped)
        scale = target_sum / s if s != 0 else 1.0
        h = h_clipped * scale
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
