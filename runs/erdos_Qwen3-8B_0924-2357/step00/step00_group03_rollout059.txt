import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    def scale_h(h, target=1.0):
        return h * target / np.sum(h)

    h = scale_h(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, delta=0.05):
        h_pert = h + np.random.normal(0, delta, len(h))
        h_pert = np.clip(h_pert, 0, 1)
        return scale_h(h_pert)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time >= budget_s:
            break
        current_h = perturb(best_h, delta=0.05)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
