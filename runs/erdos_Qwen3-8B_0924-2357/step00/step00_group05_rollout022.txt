def run(seed=42, budget_s=600, **kwargs):

    population_size = 20
    inertia = 0.5
    cognitive = 1.5
    social = 1.5
    lambda_reg = 0.1
    max_iter = 1000
    mutation_scale = 0.1
    
    np.random.seed(seed)
    
    # Initialize particles and velocities
    particles = [h_initial + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    
    # Track personal and global best
    pbest = [h.copy() for h in particles]
    pbest_fitness = [float('inf')] * population_size
    gbest = h_initial
    gbest_fitness = float('inf')
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        for i in range(population_size):

            reg = np.sum(h**2 - h)
