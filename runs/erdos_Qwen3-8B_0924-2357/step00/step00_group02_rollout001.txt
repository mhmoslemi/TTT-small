            learning_rate *= 0.95

        if epoch % 100 == 0:
            perturb_mask = np.random.rand(n_points) < 0.5
            h[perturb_mask] += np.random.uniform(-0.05, 0.05, size=np.sum(perturb_mask))
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h != 0:
                h *= (n_points / 2) / sum_h

        epoch += 1

    return (best_h, best_c5, n_points)
