        current_best = np.argmin(pbest_fitness)
        if pbest_fitness[current_best] < best_fitness:
            gbest = pbest[current_best].copy()
            best_fitness = pbest_fitness[current_best]
        if np.abs(best_fitness - np.inf) < 1e-10:
            break

    final_c5 = evaluate_fitness(gbest)
    return (gbest, final_c5, n_points)
