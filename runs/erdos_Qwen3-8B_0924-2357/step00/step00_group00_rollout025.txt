import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    required_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h_1_minus_h = 1.0 - h
        cross_corr = np.correlate(h, h_1_minus_h, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint_sum(x):
        return np.sum(x) - required_sum

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time.time()
    final_h = h_initial
    best_c5 = objective(h_initial)

    # Initial optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    if result.success:
        final_h = result.x
        best_c5 = result.fun
    else:
        final_h = h_initial

    # Refinement with perturbations
    elapsed = time.time() - start_time
    if elapsed < budget_s - 60:
        np.random.seed(seed)
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(final_h + perturbation, 0.0, 1.0)
        perturbed = perturbed * (required_sum / np.sum(perturbed))
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if result.success:
            final_h = result.x
            best_c5 = result.fun

    final_h = np.clip(final_h, 0.0, 1.0)
    np.sum(final_h) = required_sum

    return (final_h, best_c5, n_points)
