import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def perturb(h, T, n_points):

    j = np.random.choice([x for x in range(n_points) if x != i])

    current_sum = np.sum(new_h)
    scaling = T / current_sum

    new_h = np.clip(new_h, 0, 1)
