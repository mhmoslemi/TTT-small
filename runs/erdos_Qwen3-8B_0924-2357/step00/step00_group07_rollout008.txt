                new_population.append(repair(child))
            else:
                new_population.append(top_candidates[i])
        
        while len(new_population) < len(population):
            parent = random.choice(top_candidates)
            child = parent.copy()
            mutation_indices = np.random.choice(n_points, 5, replace=False)
            noise = np.random.normal(0, 0.05, 5)
            child[mutation_indices] += noise
            new_population.append(repair(child))
        
        population = new_population
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr * dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
