import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    n_points = 42
    target_mass = n_points / 2
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (target_mass / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_mass / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    max_iter = 200
    learn_rate = 0.1
    perturb_prob = 0.1

    for _ in range(max_iter):
        corr = []
        for k in range(1, n_points):
            s = np.roll(h, k)
            corr.append(np.dot(h, s) * dx)
        k_min = np.argmin(corr)
        s = np.roll(h, k_min)

        h += learn_rate * s
        h = np.clip(h, 0, 1)
        s_sum = np.sum(h)
        if s_sum > 0:
            h *= target_mass / s_sum

        if np.random.rand() < perturb_prob:
            lam = np.random.uniform(0.05, 0.15) * target_mass
            add = np.random.choice(n_points, size=5, replace=False)
            sub = np.random.choice(n_points, size=5, replace=False)
            h[add] += lam
            h[sub] -= lam
            h = np.clip(h, 0, 1)
            s_sum = np.sum(h)
            h *= target_mass / s_sum

        curr_c5 = evaluate_erdos_solution(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
