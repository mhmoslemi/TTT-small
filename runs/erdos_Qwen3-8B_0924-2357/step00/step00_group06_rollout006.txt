            if 0 <= i - lag < n_points:
                term2 = -h_vals[i - lag]
            gradient[i] = term1 + term2

        delta = -learning_rate * gradient
        h_vals += delta
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)

        elapsed_time = time.time() - start_time
        iterations += 1

    return (best_h_vals, best_c5, n_points)
