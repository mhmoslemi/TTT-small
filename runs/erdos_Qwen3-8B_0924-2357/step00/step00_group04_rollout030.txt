import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_mass = n_points / 2.0

    learning_rate = 0.1
    learning_rate_decay = 0.95
    perturb_prob = 0.1

    for iteration in range(max_iterations):
        # Compute cross correlations
        cross_corr = np.zeros(n_points - 1)
        for k in range(1, n_points):
            shifted_left = np.roll(h, -k)
            corr = np.dot(h, shifted_left) * dx
            cross_corr[k-1] = corr
        # Find min cross correlation
        k_min = np.argmin(cross_corr)
        # Compute gradient
        shifted_r = np.roll(h, k_min)
        shifted_l = np.roll(h, -k_min)
        grad = dx * (shifted_r + shifted_l)
        # Update h
        h += learning_rate * grad
        # Clip
        h = np.clip(h, 0., 1.)
        # Normalize
        new_mass = np.sum(h)
        h = h * target_mass / new_mass
        # Perturb with some probability

        learning_rate *= learning_rate_decay
