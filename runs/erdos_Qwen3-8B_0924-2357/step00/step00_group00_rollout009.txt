import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    try:
        h_initial = initial_h_values
        n_points = len(h_initial)
    except:
        h_initial = np.zeros(45)
        h_initial[:22] = 1.0
        n_points = 45
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if not np.isclose(sum_h, n_points / 2):
        diff = (n_points / 2 - sum_h)
        h_initial[-1] += diff
        sum_h = np.sum(h_initial)
    
    def objective(x):
        h = np.array(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
