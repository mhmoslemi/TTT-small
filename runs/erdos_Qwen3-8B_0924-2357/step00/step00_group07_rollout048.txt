import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    
    # Initialize population with perturbations
    population = []
    for _ in range(20):
        perturb = np.random.normal(0, 0.05, n_points)
        h = initial_h + perturb
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Sort population by C5
        population.sort(key=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points))
        top_parents = population[:10]
        
        # Generate new population
        new_pop = []
        for i in range(20):
            p1 = top_parents[i % 10]
            p2 = top_parents[(i + 1) % 10]
            child = (p1 + p2) / 2
            # Mutation
            indices = np.random.choice(n_points, 5, replace=False)
            for idx in indices:
                child[idx] += np.random.normal(0, 0.05)
            sum_child = np.sum(child)
            if sum_child == 0:
                sum_child = 1e-10
            scale = (n_points / 2) / sum_child
            child = np.clip(child * scale, 0, 1)
            new_pop.append(child)
        population = new_pop
        
        # Update best solution
        for h in population:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
    return (best_h, best_c5, n_points)
