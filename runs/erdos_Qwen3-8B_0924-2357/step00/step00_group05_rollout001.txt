import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    max_iterations = int(budget_s / 0.05)  # Adjust for time budget
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = corr.max() * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = i + max_lag
            term1 = 1.0 - h[j1] if 0 <= j1 < n_points else 0.0
            j2 = i - max_lag
            term2 = h[j2] if 0 <= j2 < n_points else 0.0
            grad[i] = (term1 - term2) * dx
        learning_rate = 0.1 * (1.0 - iter / max_iterations)
        h += -learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h *= (n_points / 2) / h_sum
    return best_h, best_c5, n_points
