import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values', None)
    n_points = 95
    if initial_h is None:
        h = np.full(n_points, 0.5)
    else:
        h = initial_h.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    scale = target_sum / h_sum if h_sum != 0 else 1.0
    h *= scale

    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_val = np.max(corr) * dx
        return max_val

    def scale_h_vals(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        return h_vals * scale

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - n_points + 1
        if lag < 0:
            lag = 0
        elif lag > n_points:
            lag = n_points

        eps = 0.01
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += eps
            h_perturb = scale_h_vals(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            grad[i] = (corr_perturb[lag] - corr[lag]) / eps

        eta = 0.01
        for i in range(n_points):
            h[i] -= eta * grad[i]
        h = scale_h_vals(h)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if abs(new_c5 - best_c5) < 1e-5:
            break

    return (best_h, best_c5, n_points)
