            v_new = alpha * v + beta * (p_best[i] - current) + gamma * (g_best - current)
            velocities[i] = v_new
            
            new_h = current + v_new
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)
            if sum_new != 0:
                new_h = new_h * (desired_sum / sum_new)
            population[i] = new_h
        
        for i in range(n_particles):
            if p_fitness[i] < g_fitness:
                g_fitness = p_fitness[i]
                g_best = p_best[i].copy()
    
    final_h = g_best
    dx = 2.0 / n_points
    max_overlap = 0.0
    for k in range(n_points):
        shifted = np.roll(final_h, k)
        overlap = np.sum(final_h * (1 - shifted)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    c5_bound = max_overlap
    
    return (final_h, c5_bound, n_points)
