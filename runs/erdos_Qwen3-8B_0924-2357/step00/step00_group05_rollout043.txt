def run(seed=42, budget_s=600, **kwargs):

        overlap = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        c5 = np.max(overlap)

            best_h = h.copy()

        if len(overlap) == 0:

        lag = max_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
