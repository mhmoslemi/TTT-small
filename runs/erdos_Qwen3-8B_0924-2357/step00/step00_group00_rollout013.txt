    history = [(h.copy(), best_c5)]
    start_time = time.time()

    for iteration in range(1, 1000):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            history.append((h.copy(), current_c5))

        if iteration % 100 == 0:
            max_lag = np.argmax(corr)
            grad = np.zeros(n_points)
            for j in range(n_points):
                if j + max_lag < n_points:
                    grad[j] += (1 - h[j + max_lag]) * dx
                if j - max_lag >= 0:
                    grad[j] -= h[j - max_lag] * dx
            scale_factor = 0.05 * (1 - 0.95 ** (iteration / 100))
            h = np.clip(h + grad * scale_factor, 0, 1)
        else:
            noise = np.random.normal(0, 0.03, n_points)
            h = np.clip(h + noise, 0, 1)

        total = np.sum(h)
        scale = (n_points / 2.0) / total
        h = np.clip(h * scale, 0, 1)

        if len(history) >= 10 and np.std([v for _, v in history[-10:]]) < 1e-6:
            break

        if time.time() - start_time > budget_s:
            break

    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2.0) / total
    h = np.clip(h * scale, 0, 1)
    final_c5 = evaluate_erdos_solution(h)
    return h, final_c5, n_points
