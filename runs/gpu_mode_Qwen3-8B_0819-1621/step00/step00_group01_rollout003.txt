def custom_kernel(data):

    norm_bias = weights["norm.bias"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    norm_weight = norm_weight.to(input_tensor.device)
    norm_bias = norm_bias.to(input_tensor.device)

    input_tensor = input_tensor.to(input_tensor.dtype)
    x = input_tensor
    norm_scale = norm_weight
    norm_bias = norm_bias
    x = x * norm_scale + norm_bias

    # Project to hidden_dim
    left_proj_weight = left_proj_weight.to(input_tensor.device)

    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    right = right * mask

        hidden_dim,
        bs,
        mask,

    GROUP_SIZE = tl.constexpr
):

    # We'll use shared memory to reduce memory bandwidth

    # Each block handles a group of seq_len / GROUP_SIZE elements
    # Each block will process a group of N, and for each of those, compute the sum over k
    # We'll use thread blocking to compute the sum
    # We can do this in blocks of GROUP_SIZE, which is the same as batch size
    # This is the number of thread blocks in the sequence dimension

    # Initialize the output tensor
    out = tl.zeros((seq_len, seq_len, hidden_dim), dtype=tl.float32, 
                   num_blocks=tl.cdiv(seq_len * seq_len * hidden_dim, BLOCK * GROUP_SIZE))

    # Each tile will compute a part of the output tensor
    # We'll iterate over the sequence indices in a way that allows for efficient access
    # Since we are computing (i,j) for all i,j, we can iterate over i and j in a way that allows for grouping
    # Then for each (i,j), compute the sum over k of left[i,k] * right[j,k]

    # We start by processing each batch independently
    bid = tl.program_id(axis=0)
    block = bid % GROUP_SIZE
    offset_i = block * seq_len

    # For each batch
    for batch in range(bs):
        # For each i in the sequence
        for i in range(seq_len):
            # For each j in the sequence
            for j in range(seq_len):
                # Check if the mask is active
                if mask_ptr[batch, i, j]:
                    # Fetch left[i,k] and right[j,k]
                    # We can do this efficiently by loading the entire row i of left and row j of right
                    # Then compute the sum over k of left[i,k] * right[j,k]
                    # We do this in blocks
                    # We iterate over k in chunks
                    # This is the core of the computation
                    # We will load the values of left[i,:] and right[j,:]
                    # Then perform the dot product efficiently
                    # We'll use a loop over k, loading values into registers
                    # We'll have to be efficient with memory accesses and computations

                    # We'll use a small block size to compute the dot product efficiently
                    # This is the crux of the algorithm

                    # Get the k indices
                    k = tl.arange(0, seq_len)

                    # Load left[i,:] and right[j,:]
                    left = tl.load(left_ptr + (batch * seq_len * seq_len * hidden_dim) + (i * seq_len * hidden_dim) + (k * hidden_dim), 
                                   shape=(seq_len, hidden_dim), 
                                   align=16)

                                   shape=(seq_len, hidden_dim),

                    # Compute the dot product
                    # This is the same as einsum("k d, k d -> d")
                    dot = tl.dot(left, right)

                    # Apply the gate
                    out_gate_val = out_gate[batch, i, j]
                    dot = dot * out_gate_val

                    # Apply the normalization
                    dot = dot * to_out_norm_weight + to_out_norm_bias

                    # Store the result
                    out[batch, i, j] = dot

    return out
