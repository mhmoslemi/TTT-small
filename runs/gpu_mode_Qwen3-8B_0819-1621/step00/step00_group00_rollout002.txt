import torch
import triton
import triton.language as tl
import math

# This kernel implements the outgoing version of the Triangle Multiplicative
# Update (TriMul) module used in AlphaFold3 and related models for protein
# structure prediction. It processes a 4D input tensor [B, N, N, C] using
# matrix operations, projections, and element-wise multiplicative gates.
# The kernel is optimized for H100 hardware with TMA and wgmma support.

@triton.jit
def triton_trimul_kernel(
    input_ptr, mask_ptr, weights,
    output_ptr,
    B: tl.constexpr,
    N: tl.constexpr,
    dim: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # Compute the offset for the current thread
    pid = tl.program_id(axis=0)
    batch_idx = pid // (N * N)
    i = (pid // N) % N
    j = pid % N

    # Calculate the position in the output tensor
    out_pos = batch_idx * N * N + i * N + j
    out_ptr = output_ptr + out_pos * dim

    # Initialize output
    out = tl.zeros((dim,), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK):
        # Calculate k range
        k_start = k
        k_end = tl.minimum(k + BLOCK, N)

        # Load mask
        mask_idx = batch_idx * N * N + i * N + k_start
        mask_val = tl.load(mask_ptr + mask_idx)

        # Skip if mask is zero (masked out)
        if mask_val == 0:
            continue

        # Load input for this batch
        input_base = batch_idx * N * N * dim
        input_k_start = k_start * dim
        input_k_end = k_end * dim

        # Load left and right projections
        left_start = input_base + i * N * dim + k_start * dim
        left_end = left_start + (k_end - k_start) * dim
        right_start = input_base + j * N * dim + k_start * dim
        right_end = right_start + (k_end - k_start) * dim

        left = tl.load(input_ptr + left_start, mask=tl.arange(0, k_end - k_start) < (k_end - k_start))
        right = tl.load(input_ptr + right_start, mask=tl.arange(0, k_end - k_start) < (k_end - k_start))

        # Apply left and right gates
        left_gate_vals = tl.load(weights + left_gate_weight_offset, mask=tl.arange(0, hidden_dim) < hidden_dim)
        left_gate = tl.sigmoid(left_gate_vals)
        left = left * left_gate

        right_gate_vals = tl.load(weights + right_gate_weight_offset, mask=tl.arange(0, hidden_dim) < hidden_dim)
        right_gate = tl.sigmoid(right_gate_vals)
        right = right * right_gate

        # Compute contribution to output
        # This is the key operation: out[i, j] += left[i, k] * right[j, k] over all k
        # We do this by computing the dot product between left[i, k] and right[j, k]
        # and accumulating the result into out[i, j]
        contrib = tl.dot(left, right)

        # Apply the output gate
        out_gate_vals = tl.load(weights + out_gate_weight_offset, mask=tl.arange(0, hidden_dim) < hidden_dim)
        out_gate = tl.sigmoid(out_gate_vals)
        contrib = contrib * out_gate

        # Accumulate into the output
        out += contrib

    # Write back the output
    tl.store(out_ptr, out)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N = input_tensor.shape[0], input_tensor.shape[1]
    C = dim

    # Set up weight pointers
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute output shape
    output_shape = (B, N, N, dim)
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=torch.float32)

    # Setup offset parameters for weights
    left_gate_weight_offset = 0
    right_gate_weight_offset = left_gate_weight.numel()
    out_gate_weight_offset = right_gate_weight.numel() + left_gate_weight.numel()
    left_proj_weight_offset = out_gate_weight.numel() + out_gate_weight.numel()
    right_proj_weight_offset = left_proj_weight.numel() + left_proj_weight_offset
    to_out_norm_weight_offset = right_proj_weight.numel() + right_proj_weight_offset
    to_out_norm_bias_offset = to_out_norm_weight.numel() + to_out_norm_weight_offset
    to_out_weight_offset = to_out_norm_bias.numel() + to_out_norm_bias_offset
    norm_weight_offset = to_out_weight.numel() + to_out_weight_offset
    norm_bias_offset = norm_weight.numel() + norm_weight_offset

    # Launch kernel
    num_blocks = (B * N * N)
    grid = (num_blocks,)
    triton_trimul_kernel[grid](
        input_tensor, mask, weights,
        output,
        B, N, dim, hidden_dim,
        BLOCK=128,
    )

    # Apply to_out_norm and to_out
    output = output + to_out_norm_bias
    output = torch.matmul(output, to_out_norm_weight.t())
    output = torch.matmul(output, to_out_weight.t())

    return output
