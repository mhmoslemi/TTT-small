import triton
import triton.language as tl

# Docstring summarizing the algorithm
"""
This Triton kernel implements the "outgoing" version of the TriMul operator
used in AlphaFold3 and related structure prediction models. It performs pairwise
operations over a 4D tensor [B, N, N, C] using efficient batched computations
and handles masking for sequence pairs. The implementation leverages triton's
advanced scheduling and memory management capabilities to optimize for performance
on H100 hardware.
"""

@triton.jit
def tri_mul_kernel(
    input_ptr, mask_ptr, out_ptr,
    input_shape, mask_shape, out_shape,
    dim, hidden_dim,
    num_batch, seq_len,
    BLOCK: tl.constexpr, TILE: tl.constexpr,
    num_tiles: tl.constexpr,
):
    # Determine grid and block dimensions for processing
    pid = tl.program_id(axis=0)
    block_id = pid // num_tiles
    tile_id = pid % num_tiles

    # Determine the batch, row, and column indices for this block
    b = block_id // (seq_len // TILE)
    i = (block_id % (seq_len // TILE)) * TILE
    j = (tile_id % (seq_len // TILE)) * TILE

    # Compute the offsets for the current block
    offs_b = tl.arange(0, num_batch)
    offs_i = tl.arange(0, TILE)
    offs_j = tl.arange(0, TILE)
    # Create offsets for the entire block in the output
    offs_idx = (offs_b[:, None, None] * seq_len + offs_i[None, :, None] * seq_len + offs_j[None, None, :])

    # Load data from input and mask
    mask = tl.load(mask_ptr + offs_idx, mask = tl.zeros((num_batch, seq_len, seq_len), dtype=tl.int32))
    mask = ((mask == 1) & (i < seq_len) & (j < seq_len)).astype(tl.float32)

    # Initialize output to zero
    output = tl.zeros((TILE, TILE, hidden_dim), dtype=tl.float32)

    # Iterate over all k in [0, seq_len)
    for k in range(0, seq_len, TILE):
        # Calculate the current k offset
        offs_k = tl.arange(0, TILE)

        # Calculate the indices for left and right
        left_idx = (offs_b[:, None, None] * seq_len + offs_i[None, :, None] * seq_len + offs_k[None, None, :])
        right_idx = (offs_b[:, None, None] * seq_len + offs_k[None, None, :] * seq_len + offs_j[None, :, None])

        # Load left and right projections
        left = tl.load(input_ptr + left_idx, mask = mask)
        right = tl.load(input_ptr + right_idx, mask = mask)

        # Multiply and accumulate into output
        for d in range(hidden_dim):
            for n in range(TILE):
                for m in range(TILE):
                    output[n, m, d] += left[n, m, d] * right[n, m, d]

    # Store the computed output
    tl.store(out_ptr + offs_idx, output)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Extract shape information
    bs, seq_len, _, dim = input_tensor.shape
    input_shape = (bs, seq_len, seq_len, dim)
    mask_shape = (bs, seq_len, seq_len)
    out_shape = (bs, seq_len, seq_len, dim)

    # Determine block and tile sizes
    # We choose a tile size that balances performance and memory constraints
    TILE = 256
    # Number of tiles along the sequence dimension
    num_tiles = seq_len // TILE
    # Total number of blocks (batch * (num_tiles^2))
    num_blocks = bs * (num_tiles * num_tiles)

    # Allocate output tensor
    output = torch.zeros(out_shape, device=input_tensor.device)

    # Launch the kernel
    grid = (num_blocks,)
    tri_mul_kernel[grid](
        input_tensor.data_ptr(), mask.data_ptr(), output.data_ptr(),
        input_shape, mask_shape, out_shape,
        dim, hidden_dim,
        bs, seq_len,
        TILE, TILE,
        num_tiles
    )

    return output
