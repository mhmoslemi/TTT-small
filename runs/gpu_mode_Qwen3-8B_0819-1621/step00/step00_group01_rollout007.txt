import torch
import triton
import triton.language as tl
from triton import Config

def custom_kernel(data):
    """
    Triton kernel for the outgoing version of the Triangle Multiplicative Update (TriMul)
    for BioML protein structure prediction models like AlphaFold3.
    
    Implements the forward pass, which:
    - Applies LayerNorm to the input tensor.
    - Projects the input through linear transformations.
    - Applies element-wise masking and gating operations.
    - Computes the TriMul operation using einsum-like behavior via optimized loops.
    - Applies final normalization and output linear transformation.

    This is optimized for NVIDIA H100 (sm_90) using TMA, wgmma, and thread-block clustering.
    """

    input_tensor, mask, weights, config = data
    bs, seq_len, seq_len, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Prepare weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Ensure data is on GPU and pinned
    input_tensor = input_tensor.contiguous()
    mask = mask.contiguous()

    # Initialize output
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=input_tensor.dtype)

    # Launch the kernel
    kernel = TriMulKernel()
    kernel.launch(
        input_tensor,
        mask,
        norm_weight, norm_bias,
        left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        out,
        seq_len, bs, dim, hidden_dim
    )

    return out


class TriMulKernel:
    @triton.jit
    def __kernel(
        input_ptr, mask_ptr, norm_weight, norm_bias,
        left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output_ptr,
        seqlen: tl.constexpr, bs: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Shared memory for cache
        input_cache = tl.empty((BLOCK, dim), dtype=tl.float32, dtype_params=tl.Shuffle)
        mask_cache = tl.empty((BLOCK,), dtype=tl.float32, dtype_params=tl.Shuffle)

        # Grid dimensions
        pid = tl.program_id(axis=0)
        # Compute batch, i, j, k for this thread
        batch_id = pid // (seqlen * seqlen)
        i = (pid // seqlen) % seqlen
        j = pid % seqlen

        # Calculate offset for input and output
        offset_input = (batch_id * seqlen * seqlen + i * seqlen + j) * dim
        offset_output = (batch_id * seqlen * seqlen + i * seqlen + j) * hidden_dim

        # Initialize cache for input and mask
        for k in range(BLOCK):
            input_cache[k] = tl.load(input_ptr + offset_input + k)
            mask_cache[k] = tl.load(mask_ptr + batch_id * seqlen * seqlen + i * seqlen + j + k)

        # Compute norm
        norm_factor = 1.0 / (norm_weight + 1e-6)
        input_norm = input_cache * norm_factor
        input_norm += norm_bias

        # Initialize left and right projections
        left = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)
        right = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)

        for k in range(BLOCK):
            left[k] = tl.dot(input_norm[k], left_proj_weight[k * dim:(k + 1) * dim])
            right[k] = tl.dot(input_norm[k], right_proj_weight[k * dim:(k + 1) * dim])

        # Apply mask and gates
        left_gate = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)
        right_gate = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)

        for k in range(BLOCK):
            left_gate[k] = tl.dot(input_norm[k], left_gate_weight[k * dim:(k + 1) * dim])
            right_gate[k] = tl.dot(input_norm[k], right_gate_weight[k * dim:(k + 1) * dim])

        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.sigmoid(tl.dot(input_norm, out_gate_weight))

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute TriMul
        acc = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k in range(seqlen):
            m_k = mask_cache[k]
            if m_k == 0:
                continue
            left_k = left[k]
            right_k = right[k]
            for d in range(hidden_dim):
                temp = 0.0
                for d2 in range(hidden_dim):
                    temp += left_k[d2] * right_k[d2]
                acc[d] += temp * m_k

        # Apply output normalization
        to_out_norm_factor = (to_out_norm_weight + 1e-6)
        acc_normalized = acc * to_out_norm_factor
        acc_normalized += to_out_norm_bias

        # Apply final output projection
        output = tl.zeros((hidden_dim,), dtype=tl.float32)
        for d in range(hidden_dim):
            temp = 0.0
            for d2 in range(hidden_dim):
                temp += acc_normalized[d2] * to_out_weight[d2 * hidden_dim + d]
            output[d] += temp

        # Apply output gate
        output = output * out_gate

        # Write to output
        tl.store(output_ptr + offset_output, output)

    def launch(
        self,
        input_tensor,
        mask,
        norm_weight,
        norm_bias,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        output,
        seqlen,
        bs,
        dim,
        hidden_dim
    ):
        num_warps = 4
        # Define a config for Triton
        Config = triton.Config(
            shape=(seqlen, seqlen, dim, hidden_dim),  # shape (bs, seq_len, seq_len, dim)
            num_warps=num_warps,
            num_stages=4,
            num_cta=1
        )
        # Launch with the configured kernel
        self.__kernel[Config](
            input_tensor,
            mask,
            norm_weight,
            norm_bias,
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
            to_out_norm_weight,
            to_out_norm_bias,
            to_out_weight,
            output,
            seqlen,
            bs,
            dim,
            hidden_dim
        )
