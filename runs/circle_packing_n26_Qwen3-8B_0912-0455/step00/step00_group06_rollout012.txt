import cvxpy as cp

    # Initial radius values

    radius = cp.Variable(n)
    center_x = cp.Variable(n)
    center_y = cp.Variable(n)

    # Constraints
    constraints = []

    # Ensure circles are inside the unit square
    for i in range(n):
        constraints.append(center_x[i] >= -1e-12)

        constraints.append(center_y[i] >= -1e-12)
        constraints.append(center_y[i] <= 1 + 1e-12)

    # Ensure circles do not overlap
    for i in range(n):

            dist = cp.sqrt((center_x[i] - center_x[j])**2 + (center_y[i] - center_y[j])**2)
            constraints.append(dist >= radius[i] + radius[j] - 1e-12)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radius))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()

    radii_opt[radii_opt < 0] = 0

    # Return the final packing
