import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Step 1: Initial placement of 26 points in the unit square
    # For simplicity, we use a grid layout but with some randomness to avoid symmetry
    num_circles = 26
    x = np.linspace(0, 1, 6)  # 6 columns
    y = np.linspace(0, 1, 5)  # 5 rows
    grid = np.array([(xi, yi) for xi in x for yi in y])
    grid = np.random.permutation(grid)  # shuffle to avoid symmetry
    centers = grid[:26]  # take first 26 points

    # Initial guess for radii (we assume some overlap, so we need to optimize)
    radii = np.random.uniform(0.01, 0.1, size=num_circles)

    # Define the objective function: maximize the sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # negative for minimization

    # Define constraints:
    # 1. Each circle must be within the unit square
    # 2. Each circle must not overlap with any other
    # 3. All radii must be non-negative

    def constraint1(radii, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return False
        return True

    def constraint2(radii, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Set up the optimization
    bounds = [(0, 1) for _ in range(num_circles)]  # Radii must be between 0 and 1

    # Initial guess for radii
    initial_radii = np.random.uniform(0.01, 0.1, size=num_circles)

    # Optimization using Nelder-Mead
    result = minimize(
        fun=objective,
        x0=initial_radii,
        args=(centers,),
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint1, 'args': (centers,)},
                     {'type': 'ineq', 'fun': constraint2, 'args': (centers,)}],
        method='Nelder-Mead'
    )

    # Extract the final centers and radii
    final_centers = centers
    final_radii = result.x
    final_sum = np.sum(final_radii)

    return final_centers, final_radii, final_sum
