import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Start with a grid-like configuration (approximately 6 rows by 4–5)

    # Let's design initial centers as 6 rows with varying counts of circles per row
    
    centers = []
    for i in range(6):  # 6 rows
        if i % 2 == 0:  # Even rows have more circles
            row_centers = [(x / 5.0, i * 0.2) for x in range(5)]  # 5 circles
        else:  # Odd rows have fewer circles
            row_centers = [(x / 4.0, i * 0.2) for x in range(4)]  # 4 circles
        centers.extend(row_centers)
    
    # Now we'll optimize the positions and radii of circles to maximize the sum of radii
    # We'll use a convex optimization approach with constraints

    # Define optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints:
    # Each circle must lie within the square
    constraints = [0 <= x[i, 0] <= 1 for i in range(n)]
    constraints += [0 <= x[i, 1] <= 1 for i in range(n)]

    # Each circle must have nonnegative radius
    constraints += [r[i] >= 0 for i in range(n)]

    # No overlapping between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(x[i] - x[j])
            constraints += [dist >= r[i] + r[j] - 1e-6]  # Small epsilon for numerical stability

    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK)

    # Extract results
    centers = np.array(x.value)
    radii = np.array(r.value)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
