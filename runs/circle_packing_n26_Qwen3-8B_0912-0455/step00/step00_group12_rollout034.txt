import scipy.optimize as optimize

def run_packing():

    def hex_grid(n, spacing):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j * spacing
                y = i * (np.sqrt(3)/2) * spacing
                if i % 2 == 0:
                    x += spacing / 2
                centers.append((x, y))
        return np.array(centers)

    spacing = 0.3
    centers = hex_grid(n_circles, spacing)

    centers -= np.min(centers, axis=0)

    centers[:, 1] = np.clip(centers[:, 1], 0, 1)

    def objective(radii):
        return -np.sum(radii)

        # Ensure all radii are non-negative
        np.clip(radii, 0, None, out=radii)
        return radii
