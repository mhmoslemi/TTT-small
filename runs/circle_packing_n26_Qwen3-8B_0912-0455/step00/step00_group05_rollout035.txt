import numpy as np
import cvxpy as cp

def run_packing():
    # Initial configuration (row-wise packing of circles)
    n = 26
    rows = 5
    cols = 6  # 5 rows and 6 columns
    spacing = 1.0 / (cols + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // cols
        col = i % cols
        x = spacing * (col + 1)
        y = spacing * (row + 1)
        centers[i] = [x, y]
        # Initial estimate of radius
        radii[i] = 0.1  # Will be optimized later

    # Define optimization variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))
    eps = 1e-6  # Tolerance for numerical stability

    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints: circle boundaries inside [0, 1]^2
    constraints = []
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= -eps)
        constraints.append(x[i, 0] + r[i] <= 1 + eps)
        constraints.append(x[i, 1] - r[i] >= -eps)
        constraints.append(x[i, 1] + r[i] <= 1 + eps)

    # Constraints: no overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - eps)

    # Define and solve the problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4)

    # Extract final values
    final_centers = np.column_stack((x.value, x.value[:, 1]))
    final_radii = r.value
    final_sum = np.sum(final_radii)

    # Return the configuration
    return (final_centers, final_radii, final_sum)
