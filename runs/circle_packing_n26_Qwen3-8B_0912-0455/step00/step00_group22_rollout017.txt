```python
import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Hexagonal close packing: 6 rows of circles (using staggered rows)
    # Each row contains 5 circles, with offset in alternate rows
    # We will fit 6 rows, 4 columns with staggered positions
    # and fill the remaining positions in corners

    # These initial positions are meant to be a base layout for the circles
    centers = np.empty((n, 2))
    radii = np.empty(n)

    # Starting with hexagonal arrangement
    spacing = 0.5  # Base distance between adjacent circle centers
    rows = 6
    cols = 4

    # Define positions in staggered grid pattern
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            shift = spacing / 2 if i % 2 == 1 else 0
            y = i * spacing + shift
            centers[i * cols + j] = [x, y]

    # Add more circles in the corners and borders to fill the space
    # This will help make a more optimal packing
    corner_positions = [
        [0.05, 0.05],
        [0.05, 0.95],
        [0.95, 0.95],
        [0.95, 0.05]
    ]
    for pos in corner_positions:
        centers[n - 4:] = np.vstack(centers[n - 4:])  # Append
        centers[n - 4:] = np.append(centers[n - 4], [pos[0], pos[1]])

    # Now, radius optimization
    # The idea is to maximize the sum of radii, while keeping circles non-overlapping and within the unit square

    # Define decision variables
    r = cp.Variable(n, nonneg=True)
    # Add constraints: circles must be within the unit square
    for i in range(n):
        x, y = centers[i]
        cp.SolverParams(epsilon=1e-8)
        cp.constraints.constraint(cp.Square(x - r[i]) <= 1e-8)
        cp.constraints.constraint(cp.Square(y - r[i]) <= 1e-8)
        cp.constraints.constraint(cp.Square(1 - x - r[i]) <= 1e-8)
        cp.constraints.constraint(cp.Square(1 - y - r[i]) <= 1e-8)

    # Add non-overlap constraints: every pair of circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            cp.constraints.constraint(dist >= r[i] + r[j] - 1e-8)

    # Maximize the total sum of radii
    problem = cp.Problem(cp.Maximize(cp.sum(r)), constraints=cp.constraints)
    problem.solve()

    # Extract results
    radii = r.value
    centers = np.array(centers)

    # Validate packing
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise Exception(f"Packing failed: {message}")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
