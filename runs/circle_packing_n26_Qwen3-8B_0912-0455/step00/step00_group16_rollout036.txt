import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Define the number of circles
    N = 26

    # Step 2: Define a grid-based initial packing
    # We use a 5x5 grid with 25 circles plus one extra (26th) at the bottom right
    grid = np.linspace(0, 1, 6)  # 5 equal divisions of [0, 1]

    y_coords = grid[:-1]
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([x_coords[j], y_coords[i]])
    # Add the 26th circle at the bottom right
    centers.append([x_coords[-1], y_coords[-1]])
    centers = np.array(centers)

    # Step 3: Initial guess for radii
    # We use a spacing-based radius initialization
    max_spacing = 0.1
    radii = np.full(N, max_spacing)

    # Step 4: Define the objective function
    def objective(x: np.ndarray) -> float:
        # x is a flattened array of [x0, y0, r0, x1, y1, r1, ...]
        # Rebuild positions and radii
        unravel = np.reshape(x, (N, 2 + 1))  # [x, y, r]
        positions = unravel[:, :2]
        radii = unravel[:, 2]
        total_radii = np.sum(radii)
        return -total_radii  # Minimize negative to maximize

    # Step 5: Define constraints
    def constraints(x: np.ndarray):
        unravel = np.reshape(x, (N, 2 + 1))
        positions = unravel[:, :2]
        radii = unravel[:, 2]

        # Constraint 1: Circle stays within [0, 1] square
        def in_bounds(x, y, r):
            return (x - r >= 0) & (x + r <= 1) & (y - r >= 0) & (y + r <= 1)

        for i in range(N):
            if not in_bounds(positions[i, 0], positions[i, 1], radii[i]):
                return False

        # Constraint 2: Circle centers are not overlapping
        for i in range(N):
            for j in range(i + 1, N):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

    result = opt.minimize(

    optimal = result.x
    centers = optimal[:2 * N].reshape(N, 2)
    radii = optimal[2 * N:]

    # Step 9: Return the final packing
    return centers, radii, np.sum(radii)
