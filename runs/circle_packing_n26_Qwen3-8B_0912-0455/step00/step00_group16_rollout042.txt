    n = len(centers)
    radii = cp.Variable(n)

    objective = cp.Maximize(cp.sum(radii))

    constraints = []

    for i in range(n):

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)

    if not problem.status == "optimal":
