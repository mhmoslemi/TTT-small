import scipy.optimize as opt

    # Initial guess for dx (horizontal spacing)
    dx = 0.1  # Tune later if needed

    centers = []
    radii = []

    # First row of circles
    for i in range(13):
        x = i * dx
        y = 0.15
        centers.append([x, y])
        # Initial radius guess: 0.05
        radii.append(0.05)

    # Second row of circles (shifted by dx/2 horizontally and dy vertically)
    for i in range(13):
        x = i * dx + dx / 2
        y = 0.15 + dy
        centers.append([x, y])
        # Initial radius guess: 0.05
        radii.append(0.05)

    # Initial configuration
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints:
    # - Each circle must fit in the unit square
    # - No two circles can overlap

    def constraint_circle_in_square(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return False
        return True

    def constraint_no_overlap(radii, centers):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Set bounds: radii must be >= 0 and <= 0.5 (max possible)
    bounds = [(0.0, 0.5) for _ in range(len(radii))]

    # Optimization options
    options = {
        'disp': False,
        'maxiter': 1000,
        'ftol': 1e-10,
        'eps': 1e-10
    }

    # Optimize radii to maximize the sum of radii while keeping centers fixed
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[

        method='SLSQP',
        options=options

    # Extract the result
    optimal_radii = result.x

    return optimal_centers, optimal_radii, sum_radii
