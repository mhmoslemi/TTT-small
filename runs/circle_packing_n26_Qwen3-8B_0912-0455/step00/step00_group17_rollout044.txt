import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # --- Step 1: Generate initial hexagonal grid in square
    n = 26
    centers = []
    radii = []

    # Start from a hexagonal grid with rows, and select the first 26 circles
    rows = 6
    cols = 5
    spacing = 1.0 / (rows + 1)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])

    # Now, adjust the grid to center it in the square
    if len(centers) > n:
        centers = centers[:n]
        radii = [0.0] * n

    # --- Step 2: Optimize positions and radii using gradient descent
    # Define cost function to be minimized: maximize the sum of radii
    # We use a simple greedy expansion where each circle is as big as possible without overlapping

    def objective(x):
        # x contains all [x0, y0, r0, x1, y1, r1, ...]
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)

        # Compute radii such that each circle is as large as possible without overlapping others
        # Start with minimal overlap, then expand as much as possible
        max_radius = 0
        delta = 1e-6  # small epsilon to avoid exact overlaps

        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            # Update the radius to the maximum allowed by square boundary
            r_i = min(r_i, 1 - x_i - delta, x_i - delta, 1 - y_i - delta, y_i - delta)

            # Now compute the maximum radius for each circle based on other circles
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                r_j = radii_flat[j]
                max_r_current = (dist - r_j - delta) if (dist - r_j - delta) > 0 else 0
                r_i = min(r_i, max_r_current)

            # Save the updated radius
            radii_flat[i] = r_i
            max_radius = max(max_radius, r_i)

        return -max_radius  # Minimize negative of max_radius to maximize it

    # Initial positions are centers from the hexagonal grid, and radii all 0
    initial_positions = np.array(centers).flatten()
    initial_radii = np.zeros(n)
    x0 = np.concatenate([initial_positions, initial_radii])

    # Run minimization: minimize the negative of max_radius to maximize it
    result = minimize(
        objective,
        x0,
        method="L-BFGS-B",
        bounds=[(-1, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
    )

    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:].reshape(n)
    sum_radii = np.sum(radii)

    # Validate the packing
    def validate(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False

        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    if not validate(centers, radii):
        raise ValueError("Invalid packing")

    return centers, radii, sum_radii
