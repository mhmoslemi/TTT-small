import scipy.optimize as optimize

def run_packing():

    def hexagonal_grid(nx, ny):

        for y in range(ny):

                y_pos = y / (ny - 1)
                if y % 2 == 0:
                    x_offset = 0
                else:
                    x_offset = 0.5 / (nx - 1)
                centers.append([x_pos + x_offset, y_pos])

        return np.sqrt(2 / (3 * n)) * 0.9  # Adjusting for square boundary effects

        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii, centers):

        n = len(centers)
        constraints = []

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                # Constraint: distance >= r_i + r_j - tolerance
                constraints.append(dist - radii[i] - radii[j] + 1e-12)
        return constraints

    # Initial guess for radii based on density heuristic
    def initial_radii(centers):
