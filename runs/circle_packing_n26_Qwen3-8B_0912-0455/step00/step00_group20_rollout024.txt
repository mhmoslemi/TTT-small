import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    grid_size = 5  # Grid size for 5x5 layout
    x_coords = np.linspace(0.1, 0.9, grid_size)
    y_coords = np.linspace(0.1, 0.9, grid_size)
    centers = []
    for x in x_coords:
        for y in y_coords:
            centers.append([x, y])
    # Add one more circle in the center
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Initial radii based on distance to nearest neighbors
    radii = np.full(num_circles, 0.05)

    # Objective function to maximize the sum of radii
    def objective(x, centers, radii):
        return -np.sum(radii)  # Negative for maximization

    # Constraints function: non-overlapping and boundary conditions
    def constraints(x, centers, radii):
        # x: concatenated [centers, radii]
        centers = x[:centers.shape[0] * 2].reshape((centers.shape[0], 2))
        radii = x[centers.shape[0] * 2:].reshape((radii.shape[0],))
        constraints = []

        # Boundary constraint: x - r >= 0, x + r <= 1
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = radii[i]
            constraints.append(x_i - r_i >= -1e-12)
            constraints.append(x_i + r_i <= 1 + 1e-12)
            constraints.append(y_i - r_i >= -1e-12)
            constraints.append(y_i + r_i <= 1 + 1e-12)

        # Non-overlap constraint: distance between centers >= r_i + r_j
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)

        return constraints

    # Combine centers and radii into a single vector for optimization
    x0 = np.concatenate((centers.flatten(), radii))

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=x0,
        jac=None,
        method='SLSQP',

        constraints=constraints,
        tol=1e-8,
