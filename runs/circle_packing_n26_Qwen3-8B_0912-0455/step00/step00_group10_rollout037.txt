import scipy.optimize as optimize

def run_packing():

    centers = np.random.rand(N, 2)  # Initial guess
    radii = np.random.rand(N) * 0.05 + 0.01  # Random initial radii

    # Constraints:
    # 1. Each circle must be within the unit square
    # 2. Each circle must not overlap with any other
    # 3. Ensure centers are within [0, 1]x[0, 1]

    def constraint1(radii):
        # Ensure circle i is within square
        for i in range(N):
            if np.sqrt((centers[i, 0] - radii[i])**2 + (centers[i, 1] - radii[i])**2) > 1:
                return 0
        return 1

    def constraint2(radii):
        # Ensure non-overlapping
        for i in range(N):
            for j in range(i + 1, N):

                    return 0
        return 1

    bounds = [(0, 1) for _ in range(N * 2)]  # For both x and y positions
    result = optimize.minimize(
        fun=objective,

            {'type': 'ineq', 'fun': constraint2}
