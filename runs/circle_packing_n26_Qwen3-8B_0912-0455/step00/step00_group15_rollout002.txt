from scipy.optimize import minimize

def run_packing():

    # Let's use a systematic approach to place 26 non-overlapping circles

    def objective(radii):
        return -np.sum(radii)  # We want to maximize, so minimize negative sum

        return 0

    def constraint_2(radii):  # Boundary constraint
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:

        return 0

    def constraint_non_overlap(radii):
