def is_point_inside_unit_square(point):
    return np.all(0 <= point) and np.all(point <= 1)

def is_circle_inside_unit_square(center, radius):
    if not is_point_inside_unit_square(center):
        return False
    if (center[0] - radius < 0 or center[0] + radius > 1 or
        center[1] - radius < 0 or center[1] + radius > 1):
        return False
    return True

def is_circle_overlap(c1, r1, c2, r2):
    dist = np.linalg.norm(c1 - c2)
    return dist < r1 + r2 - 1e-8

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
