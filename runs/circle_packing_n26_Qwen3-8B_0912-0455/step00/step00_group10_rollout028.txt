import numpy as np
import cvxpy as cp

def run_packing():
    # Set up problem: 26 circles in a unit square
    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Ensure circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Ensure no circles overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve()

    # Extract the solution
    centers_val = np.array(centers.value)
    radii_val = np.array(radii.value)
    total_radius_sum = np.sum(radii_val)

    return centers_val, radii_val, total_radius_sum
