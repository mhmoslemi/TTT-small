def run_packing():

    # Start with circles near corners
    
    # Row 0 (Top) - 4 circles at corners
    centers = np.array([
        [0.0, 1.0],    # Top-left corner
        [1.0, 1.0],    # Top-right corner
        [0.0, 0.0],    # Bottom-left corner
        [1.0, 0.0],    # Bottom-right corner
    ])
    
    # Midpoint circles near edges
    centers = np.vstack([
        centers,
        [0.5, 1.0],    # Mid-top
        [0.5, 0.0],    # Mid-bottom
        [0.0, 0.5],    # Mid-left
        [1.0, 0.5],    # Mid-right
    ])
    
    # Add additional rows in staggered hexagonal layout
    # Let's define rows with alternating offsets
    row_offsets = [0.0, 0.5]
    num_rows = 8
    row_height = 0.5  # Based on distance between rows assuming circle radii
    
    for i in range(1, num_rows):
        row_offset = row_offsets[i % 2]
        for j in range(1, 6):
            x = j * 0.2 + row_offset
            y = i * row_height
            centers = np.vstack([centers, [x, y]])
    
    # Adjust positions to fit in the square
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    
    # Compute radii based on min distance to neighbor
    n = centers.shape[0]
    radii = np.ones(n) * 0.03  # Initial guess

    # Compute min distances between circles
    min_distances = np.full((n, n), np.inf)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_distances[i, j] = dist
            min_distances[j, i] = dist

    # Use minimum distance to determine radii
    for i in range(n):
        min_dist = np.min(min_distances[i, :])
        radii[i] = min_dist / 2.0 - 0.001  # Margin for safety

    # Ensure radii do not exceed boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = np.min([radii[i], np.min([x, 1 - x, y, 1 - y])])

    # Compute total sum of radii
    sum_radii = np.sum(radii)
    
    # Return the result
    return centers, radii, sum_radii
