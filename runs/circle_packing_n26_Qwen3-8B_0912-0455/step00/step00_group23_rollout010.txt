        return np.array(hex_centers + boundary_centers)

    # Initial guess for radii (uniformly scaled based on initial spacing)
    def initial_radii(centers):
        distances = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                distances.append(dist)
        min_dist = min(distances)
        radius = min_dist * 0.4
        return np.array([radius] * len(centers))
    
    centers = initial_config()
    radii = initial_radii(centers)

    # Optimization parameters
    bounds = Bounds(0, 1)
    
    # Define objective function: maximize sum of radii
    def objective(x):
        x_vals = x[:len(centers)]
        r_vals = x[len(centers):]
        return -np.sum(r_vals)  # Minimize negative sum to maximize

    constraints = []

    def in_bounds(x):
        x_vals = x[:len(centers)]

            x_i, y_i = x_vals[i]
            r_i = r_vals[i]
            if not (0 - r_i <= x_i <= 1 + r_i and 0 - r_i <= y_i <= 1 + r_i):

        return True

    constraints.append({
        'type': 'ineq',

    })
    
    # Run the optimization
