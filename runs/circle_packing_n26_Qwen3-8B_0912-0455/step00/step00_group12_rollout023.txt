import numpy as np
import cvxpy as cp

def run_packing():
    """Packs 26 non-overlapping circles in a unit square to maximize the total sum of radii."""
    n_circles = 26
    num_centers = n_circles

    # Initial configuration: create concentric circles with decreasing radii
    # and cluster them like a flower-like pattern around the square
    centers = []
    radii = []
    radius_sum = 0.0

    # Step 1: Create concentric layers (like flower pattern)
    # Layer 0: 1 large circle in the center
    centers.append([0.5, 0.5])
    radius = 0.22  # Enough to fit without overlapping with others
    radii.append(radius)
    radius_sum += radius

    # Layer 1: 12 medium circles around the center
    # These will be placed in a hexagonal ring around the center
    # We place them at angles spaced every 30 degrees (12 points)
    angles = np.linspace(0, 2 * np.pi, 12, endpoint=False)
    for angle in angles:
        x = 0.5 + 0.25 * np.cos(angle)  # Offset by 0.25 from the center
        y = 0.5 + 0.25 * np.sin(angle)
        centers.append([x, y])
        radius = 0.15
        radii.append(radius)
        radius_sum += radius

    # Layer 2: 12 small circles around the edge
    # These will be placed at the four sides of the square
    # Four sides × three circles each = 12 circles
    for i in range(4):
        angle = i * np.pi / 2
        x = 0.5 + np.cos(angle)
        y = 0.5 + np.sin(angle)
        centers.append([x, y])
        radius = 0.07
        radii.append(radius)
        radius_sum += radius

    # Step 2: Optimize the positions and radii to maximize the sum
    # This is a constrained optimization problem
    # We can treat each circle as a variable with its center and radius

    # Define variables for the problem
    x_vars = cp.Variable((num_centers, 2))
    r_vars = cp.Variable(num_centers, non_negative=True)

    # Constraints
    constraints = []

    # All circles must stay within the unit square
    for i in range(num_centers):
        center = x_vars[i]
        r = r_vars[i]
        constraint = cp.SOC(np.sqrt((center[0] - 0.5)**2 + (center[1] - 0.5)**2) <= r)
        constraints.append(constraint)

    # Circles must be within the square boundaries
    for i in range(num_centers):
        cx, cy = x_vars[i]
        r = r_vars[i]
        constraints.append(cx - r <= 1 + 1e-12)
        constraints.append(cx + r <= 1 + 1e-12)
        constraints.append(cy - r <= 1 + 1e-12)
        constraints.append(cy + r <= 1 + 1e-12)

    # Non-overlapping constraint: distance between centers > sum of radii
    for i in range(num_centers):
        for j in range(i + 1, num_centers):
            dx = x_vars[i][0] - x_vars[j][0]
            dy = x_vars[i][1] - x_vars[j][1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r_vars[i] + r_vars[j] - 1e-12)

    # Objective: maximize the total sum of radii
    objective = cp.Maximize(cp.sum(r_vars))

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP, eps_rel=1e-5, eps_abs=1e-5)

    # Extract the optimal centers and radii
    centers_opt = x_vars.value
    radii_opt = r_vars.value

    # Convert to numpy arrays
    centers = np.array(centers_opt)
    radii = np.array(radii_opt)

    # Ensure all values are finite
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values found in centers or radii")

    # Compute the total sum of radii
    total_radius_sum = np.sum(radii)

    return centers, radii, total_radius_sum
