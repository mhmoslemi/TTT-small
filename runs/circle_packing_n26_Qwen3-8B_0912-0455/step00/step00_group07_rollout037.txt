import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds

def generate_hexagonal_lattice(n, square_size=1.0):

def objective_function(positions, radii, square_size=1.0):

    return np.sum((positions[:, 0] - radii) > 0) + np.sum((positions[:, 1] - radii) > 0) + np.sum(
        (1 - positions[:, 0] - radii) > 0) + np.sum((1 - positions[:, 1] - radii) > 0)
