import scipy.optimize as optimize

def run_packing():

    num_circles = 26

    for i in range(4):
        for j in range(4):
            if i % 2 == 0:
                x = i * grid_size + 0.3
                y = j * grid_size + 0.3
                interior_positions.append([x, y])

                x = i * grid_size + 0.3
                y = j * grid_size + 0.3
                interior_positions.append([x, y])

    # Add boundary circles
    boundary_positions = [
        [0.2, 0.2], [0.2, 0.8], [0.8, 0.8], [0.8, 0.2],
        [0.1, 0.5], [0.5, 0.1], [0.5, 0.9], [0.9, 0.5],
        [0.5, 0.5]
    ]
    initial_centers = np.array(interior_positions + boundary_positions)
    initial_radii = np.full(num_circles, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    def constraints_func(x):
        centers = x[:2 * num_circles].reshape(num_circles, 2)
        radii = x[2 * num_circles:]

            return np.inf

        for i in range(num_circles):
            if (centers[i, 0] - radii[i] < -1e-8 or

                if dist < radii[i] + radii[j] - 1e-8:

        return 0

    # Optimize
