import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers of the 26 circles
    # We use a 2-row hexagonal arrangement: 13 circles in each row
    n = 26
    rows = 2
    cols = 13

    # Initial guess for the centers of the circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Horizontal spacing
    dx = 1 / (cols + 1)
    # Vertical spacing
    dy = 1 / (rows + 1)

    # Place first row
    for i in range(cols):
        x = i * dx + dx / 2
        y = dy / 2
        centers[i] = [x, y]
        radii[i] = 0.05

    # Place second row
    for i in range(cols):
        x = i * dx + dx / 2
        y = dy * (rows - 1) / 2
        centers[cols + i] = [x, y]
        radii[cols + i] = 0.05

    # Initial total sum of radii
    sum_radii = np.sum(radii)

    # Define constraints for optimization
    def objective(radii):
        return -np.sum(radii)  # We minimize -sum to maximize sum

    def constraint_center_in_box(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    def constraint_no_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n)],
        constraints=[
            {'type': 'ineq', 'fun': constraint_center_in_box},
            {'type': 'ineq', 'fun': constraint_no_overlap}
        ]
    )

    radii = result.x
    sum_radii = np.sum(radii)
    centers = np.array(centers)

    # Final check
    assert not np.isnan(radii).any(), "NaN values in radii"
    assert not np.isnan(centers).any(), "NaN values in centers"

    return centers, radii, sum_radii
