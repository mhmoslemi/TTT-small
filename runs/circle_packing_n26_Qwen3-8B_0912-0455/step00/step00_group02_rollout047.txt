import numpy as np
from scipy.optimize import minimize

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    This function attempts to pack 26 circles in a unit square to maximize the total sum of radii.
    The strategy combines random initialization, optimization with constraints, and iterative refinement
    using Voronoi tessellation to guide the packing process.
    """
    # Initial random centers within the unit square
    np.random.seed(42)
    centers = np.random.rand(26, 2)
    
    # Initial guess for radii
    radii = np.full(26, 1 / 26)
    
    # Function to evaluate the sum of radii
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative to maximize the sum

    # Constraints:
    # 1. Each circle must lie within the unit square
    # 2. No two circles can overlap
    # 3. Radii must be non-negative

    # Bounds: radii >= 0
    bounds = [(0, 1) for _ in range(26)]

    # Optimization function with constraints
    def constraint_func(x):
        # x is the concatenated array of centers and radii
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]

        # Check if all circles are inside the unit square
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return -np.inf  # Penalize by returning negative infinity

        # Check for overlaps between all pairs of circles
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -np.inf  # Penalize by returning negative infinity

        return np.sum(radii_flat)  # Return the sum of radii

    # Prepare initial guess
    initial_guess = np.concatenate((centers.flatten(), radii))

    # Optimization with constraints
    result = minimize(
        fun=constraint_func,
        x0=initial_guess,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers = result.x[:52].reshape(26, 2)

        vor_vertices = np.array(vor.vertices)

        dists = np.linalg.norm(vor_vertices[:, np.newaxis, :] - centers, axis=2)
        nearest_vertices = np.argmin(dists, axis=1)
        # Update centers to be the Voronoi vertices
        centers = vor_vertices[nearest_vertices]

        # Re-evaluate radii to maintain non-overlapping and boundary constraints
        # Re-run optimization with updated centers
        initial_guess = np.concatenate((centers.flatten(), radii))
        result = minimize(
            fun=constraint_func,
            x0=initial_guess,
            bounds=bounds,
            method='SLSQP',

    # Final check for validity and return the result
    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]

    sum_radii = np.sum(radii)

    # Final validation
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Final packing failed validation")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
