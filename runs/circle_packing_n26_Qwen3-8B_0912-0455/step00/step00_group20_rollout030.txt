def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    rows = 5
    cols = 5
    spacing = 1.0 / (rows - 1)  # For a square box with 5 rows and 5 columns of circles

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            if i % 2 == 1:

            centers.append([x, y])

    def objective(radii):
