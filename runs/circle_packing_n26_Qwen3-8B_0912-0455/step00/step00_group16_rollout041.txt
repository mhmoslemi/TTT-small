from scipy.optimize import minimize
from scipy.optimize import Bounds
import matplotlib.pyplot as plt

def run_packing():

    n = 26
    # Define initial placements (grid-based + one central)
    grid_size = 5
    grid = np.linspace(0.1, 0.9, grid_size)  # Reserving 0.1 margin for boundary

    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            centers.append([grid[i], grid[j]])

    # Add one central circle
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Define initial radii (all same)
    initial_radii = np.full(n, 0.1)

    # Define the objective function to maximize
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum of radii

    # Define constraints: circle must not cross boundaries
    def boundary_constraint(radii):
        return np.array([radii[i] for i in range(n)])

    # Define circle-to-circle non-overlap constraint
    def circle_overlap(radii):
        result = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                result.append(radii[i] + radii[j] - dist)
        return np.array(result)

    # Constraint manager

        return np.concatenate((boundary_constraint(x), circle_overlap(x)))

    # Create bounds for radii (non-negative)
    bounds = Bounds(lb=0, ub=np.inf)

    # Initial guess
    initial_guess = initial_radii

    # Run optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP"
    )

    # Extract solution
    radii = result.x
    centers = np.array(centers)
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
