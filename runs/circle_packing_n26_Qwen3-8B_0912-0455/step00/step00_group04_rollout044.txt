from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid-based hexagonal packing
    n = 26
    k = int(np.sqrt(n))
    m = (n + k - 1) // k

    centers = []
    radii = []

    # Start with a hexagonal grid of circles in the unit square
    # Rows offset by half the diameter (set to 1 initially)
    for row in range(m):

                x += 0.5 / (k - 1)
            # Ensure circle is within square
            x = max(0, min(1, x))
            y = max(0, min(1, y))
            centers.append([x, y])
            radii.append(1.0 / (k + m))  # Initial small radius for optimization

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(x):
        # x is a flat vector of (centers, radii), reshape for use
        x = x.reshape(-1)
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:].reshape(n)

        # Check if any circles are outside the square or have negative radii
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return 1e6
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return 1e6
            if r_i < 0:
                return 1e6

        # Calculate the total sum of radii (objective to maximize)
        total = np.sum(radii)
        return -total  # Minimize negative of total

    def constraint(x):
        # Same reshape and checks as above
        x = x.reshape(-1)
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:].reshape(n)

        dists = cdist(centers, centers)

        # Check all pairs for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-12:

    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

    x_final = result.x
