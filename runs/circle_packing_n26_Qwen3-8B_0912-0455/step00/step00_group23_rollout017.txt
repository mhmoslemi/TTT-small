def hexagonal_grid(n, spacing, offset_x=0.5, offset_y=0.0):
    """Generate a hexagonal grid of n circles, each with radius spacing/2."""
    # Number of rows

    cols = int(np.ceil(n / rows))
    centers = []

                centers.append((x, y))
    return np.array(centers)
