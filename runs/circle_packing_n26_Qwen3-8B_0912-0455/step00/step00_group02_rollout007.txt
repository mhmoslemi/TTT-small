import numpy as np
import cvxpy as cp

def run_packing():

    n = 26
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    
    # Objective: maximize total sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    
    # Boundary constraints: circle must be inside the unit square
    for i in range(n):
        constraints += [
            centers[i, 0] - radii[i] >= 0,
            centers[i, 0] + radii[i] <= 1,
            centers[i, 1] - radii[i] >= 0,
            centers[i, 1] + radii[i] <= 1
        ]

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints += [dist >= radii[i] + radii[j]]
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6)
    
    # Return centers, radii, and sum of radii
    centers_np = np.array(centers.value).squeeze()
    radii_np = np.array(radii.value).squeeze()
    total_sum = np.sum(radii_np)
    
    return centers_np, radii_np, total_sum
