import numpy as np
from scipy.optimize import minimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Initial placement: a hexagonal grid
    # Row 1: 1 circle
    # Row 2: 2 circles
    # Row 3: 3 circles
    # ...
    # Row 7: 7 circles
    # Total rows: 7 (1 to 7), with 1 to 7 circles per row

    row_height = np.sqrt(3) / 2
    row_start = 0.1  # Leave 10% margin for finer adjustment
    col_width = 0.8  # Leave extra space for adjustment
    row_col = []
    for row in range(7):
        row_col.append([row_start + (col_width / (row + 1)) * i for i in range(row + 1)])
    centers = []
    for row_idx in range(7):
        row_centers = []
        for col_idx in range(len(row_col[row_idx])):
            x = row_col[row_idx][col_idx]
            y = row_start + row_idx * row_height
            row_centers.append([x, y])
        centers.extend(row_centers)

    # Convert to numpy array
    centers = np.array(centers)

    # Initial guess for radii (based on grid size)
    # Start by assuming all circles have the same radius
    initial_radius = 0.1
    radii = np.full(n, initial_radius)

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  #_Minimize negative sum to maximize sum

    # Constraints: 
    # 1. Bounds from square [0, 1] × [0, 1]
    # 2. No overlapping circles
    # 3. No circle outside the square

    # Parametrize the optimization variables
    # x = [radii[0], radii[1], ..., radii[n-1]]
    x0 = radii.copy()

    def constraint(x):
        # Check boundaries
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False

        # Check for overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-12:
                    return False
        return True

    # Use L-BFGS-B algorithm
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "success", "fun": constraint},
        method="L-BFGS-B",
        options={"disp": False}
    )

    final_radii = result.x
    final_centers = centers.copy()

    # Validate the final packing
    if not constraint(final_radii):
        raise ValueError("Packing optimization failed")

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
