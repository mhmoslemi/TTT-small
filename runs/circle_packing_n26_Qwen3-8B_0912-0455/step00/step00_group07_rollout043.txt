import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Define a grid-based initial configuration for 26 circles
    # This is a hybrid approach: 4 rows of 6 circles each with 1 extra circle placed strategically

    # Generate initial centers for a grid-based packing
    # First, create a 2D grid of points spaced with some initial spacing
    # We'll start with a simple square grid
    # First 24 circles arranged in a grid pattern
    grid_radius = 0.35
    grid_spread = 0.85
    row_count = 4
    col_count = 6
    centers = []
    
    # First, create a grid-based initial packing
    for i in range(row_count):
        for j in range(col_count):
            x = j * grid_spread / (col_count - 1)
            y = i * grid_spread / (row_count - 1)
            centers.append([x, y])
    
    # Add an extra circle near the bottom right corner
    extra_circle = [0.9, 0.9]
    centers.append(extra_circle)
    
    # Optimize the configuration to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        # Enforce non-overlapping between circles
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        # Enforce boundary constraints
        for i in range(len(centers)):
            if centers[i][0] - radii[i] < -1e-8 or centers[i][0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i][1] - radii[i] < -1e-8 or centers[i][1] + radii[i] > 1 + 1e-8:
                return False
        return True
    
    # Initial guess: all circles have the same radius
    initial_radii = [grid_radius] * (row_count * col_count + 1)
    
    # Use Sequential Least Squares Programming (SLSQP)
    result = optimize.minimize(
        objective,
        initial_radii,
        constraints=lambda x: constraints(x),
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    # Extract the final centers and radii
    final_centers = np.array(centers)
    final_radii = result.x
    
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 0)
    
    # Calculate the total sum of radii
    total_sum = np.sum(final_radii)
    
    return (final_centers, final_radii, total_sum)
