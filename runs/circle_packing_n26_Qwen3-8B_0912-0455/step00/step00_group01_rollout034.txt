import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    def validate_packing(centers, radii):

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    def objective(radii, centers):
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist - (radii[i] + radii[j]) + 1e-12

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r) + 1e-12

    def constraint_boundary_right(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (1 + 1e-12) - (x + r)

    def constraint_boundary_top(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (1 + 1e-12) - (y + r)

    def constraint_boundary_bottom(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (y - r) + 1e-12

    def get_centers(n):
        centers = []
        for i in range(n):
            x = i / (n - 1) if i % 2 == 0 else (i - 0.5) / (n - 1)
            y = (i // 2) / (n - 1)
            centers.append([x, y])
        return np.array(centers)

    n = 26
    centers = get_centers(n)
    radii = np.ones(n) * 0.1

    bounds = Bounds([0, 0], [1, 1])
    constraints = []

            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(i, j, centers, x)})

        x0=radii,
        bounds=bounds,
        constraints=constraints,
