def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        x, y = centers[i]

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-based configuration with 5 rows and 5 columns
    num_circles = 26
    rows = 5
    cols = 6
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Initial spacing
    row_spacing = 1 / (rows + 1)
    col_spacing = 1 / (cols + 1)
    
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            x = j * col_spacing + col_spacing / 2
            y = i * row_spacing + row_spacing / 2
            centers[idx] = [x, y]
    
    # Create a rough initial radius for each circle
    initial_radius = 0.05
    for i in range(num_circles):
        radii[i] = initial_radius
    
    # Optimize to maximize total radius
    def total_radius(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Ensure radii are non-negative
        if np.any(radii < 0):
            return False
        
        # Ensure all circles are within the unit square
        for i in range(num_circles):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12

                return False
        
        # Ensure no overlaps
        for i in range(num_circles):

                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

        return True

    # Optimization problem
    bounds = [(0, 1) for _ in range(num_circles)]
