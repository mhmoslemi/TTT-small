def generate_hexagonal_grid(n: int, square_size: float = 1.0) -> np.ndarray:
    """
    Generate a hexagonal grid of n circles packed in a square of side length square_size.
    Returns a numpy array of shape (n, 2) with circle centers.
    """
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([[0.5, 0.5]])

    # Start with a base hexagonal pattern
    def hexagonal_points(n: int, spacing: float = 1.0):
        points = []
        for row in range((n + 1) // 2):
            for col in range(n - row):
                x = spacing * (col - row / 2)
                y = spacing * (row + (row % 2) * (row // 2))
                if row % 2 == 0:
                    x = spacing * (row / 2 + 0.5 * (col - row / 2))
                else:
                    x = spacing * ((row + 1) / 2 + 0.5 * (col - (row + 1) / 2))
                points.append((x, y))
        return np.array(points)

    # Adjust spacing so that all points fit in square_size
    spacing = square_size / (np.sqrt(3) if n > 2 else 1)
    points = hexagonal_points(n, spacing)

    # Normalize coordinates to [0, 1]
    points = (points - np.min(points)) / np.max(points)
    return points
