import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based configuration
    n = 26
    grid = np.zeros((n, 2))
    for i in range(n):
        grid[i, 0] = i / (n - 1)
        grid[i, 1] = 0.5  # flat horizontal grid

    # Define variables: x, y, r for each circle
    # To allow optimization, we need to make this a single variable vector
    # We use a parameter x where x = [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]

    # Initial guess: grid x and y positions, and radius = 0.01
    x0 = np.empty(3 * n)
    for i in range(n):
        x0[3*i] = grid[i, 0]
        x0[3*i + 1] = grid[i, 1]
        x0[3*i + 2] = 0.05  # initial radius

    # Define the objective function (maximize sum of radii, we minimize negative sum)
    def objective(x):
        radii = x[2::3]
        return -np.sum(radii)

    # Define the constraints
    def constraints(x):
        # Extract coordinates and radii
        centers = x[::3], x[1::3]
        radii = x[2::3]

        constraints = []
        for i in range(n):
            # Ensure center is within square
            x_i, y_i = centers[i]
            constraints.append(x_i - 1e-12 <= 1)
            constraints.append(x_i + 1e-12 >= 0)
            constraints.append(y_i - 1e-12 <= 1)
            constraints.append(y_i + 1e-12 >= 0)

            # Ensure radius is non-negative
            constraints.append(radii[i] >= 0)

        # Ensure no overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)

        return constraints

    # Run optimization
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 1.0))  # radius

    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method="SLSQP",
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # Extract centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = result.x[3*i]
        centers[i, 1] = result.x[3*i + 1]
        radii[i] = result.x[3*i + 2]

    # Return as requested
    return (centers, radii, np.sum(radii))
