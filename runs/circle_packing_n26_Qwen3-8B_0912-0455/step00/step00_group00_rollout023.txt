import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

    rows = [5, 5, 5, 6, 5]

    grid = []

        for col_idx in range(row):
            x = (col_idx + 0.5) / (cols[row_idx] - 1)

            grid.append((x, y))

    centers = np.array(grid)

    # We'll maximize the sum of radii, so we'll minimize the negative sum
    def objective(radii):
        return -np.sum(radii)
    
    # Optimization bounds: each radius must be non-negative and not cause overlap
    bounds = [(0, 1.0) for _ in range(num_circles)]
    
    # Define constraints:
    # 1. Each circle must be within the unit square
