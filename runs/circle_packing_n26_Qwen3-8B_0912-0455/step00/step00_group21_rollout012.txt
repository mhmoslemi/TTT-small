from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    grid_rows = 5  # We'll arrange 5 rows of circles
    grid_cols = 5  # 5 columns of circles

    # Place circles in a hexagonal grid
    centers = []
    radii = []
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j + 0.5 * i
            y = i + 0.5 * j
            # Ensure we stay within the [0,1]^2 square
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            # Initial radius is chosen to fit in a square of size 1
            r = 0.2
            centers.append((x, y))
            radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization: maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum

    def constraint_overlap(radii, centers):

        dists = cdist(centers, centers)
        return np.array([dists[i, j] - r[i] - r[j] for i in range(len(centers)) for j in range(i + 1, len(centers))])
