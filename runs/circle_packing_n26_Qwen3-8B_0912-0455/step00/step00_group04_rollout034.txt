import cvxpy as cp

def run_packing():

    radii = cp.Variable(n, nonneg=True)

    constraints = []

        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)
    
    # No overlapping circles

            constraints.append(dist >= radii[i] + radii[j] - 1e-12)
    
    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, max_iters=500, eps=1e-6)
