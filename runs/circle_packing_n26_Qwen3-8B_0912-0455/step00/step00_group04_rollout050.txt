import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Initialize centers randomly but ensuring they are in the unit square
    centers = np.random.rand(n, 2)

    # Use a hybrid approach with hexagonal packing and strategic placement at edges and corners
    # First, scale and shift the centers to lie within the square
    centers = centers * 1.0

    # Let the radii be a variable to optimize
    radii = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints:
    # Non-overlap between any two circles
    constraints = []

    for i in range(n):
        # Circle i must be inside the unit square
        constraints.append(centers[i][0] - radii[i] >= 0)
        constraints.append(centers[i][0] + radii[i] <= 1)
        constraints.append(centers[i][1] - radii[i] >= 0)
        constraints.append(centers[i][1] + radii[i] <= 1)
        # Circle i must be positive
        constraints.append(radii[i] >= 0)

    for i in range(n):
        for j in range(i + 1, n):
            # Distance between circle i and j must be >= sum of their radii
            distance = cp.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            constraints.append(distance >= radii[i] + radii[j] - 1e-12)

    # Create and solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()
