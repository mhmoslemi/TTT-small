from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26

    centers = np.zeros((num_circles, 2))

    # Initial hexagonal pattern with row-wise arrangement
    rows = int(np.sqrt(num_circles))
    cols = num_circles // rows + (1 if num_circles % rows else 0)
    num_rows = rows
    num_cols = cols

    hex_width = (num_cols - 1) * 1.0
    hex_height = (num_rows - 1) * np.sqrt(3) / 2.0

    scale_factor = min(1.0 / hex_width, 1.0 / hex_height)

    # Generate the hexagonal grid
    for i in range(num_rows):

            y = i * (np.sqrt(3)/2.0) * scale_factor
            if i % 2 == 1:
                x += 0.5 * scale_factor
            centers[i * num_cols + j] = [x, y]
            radius[i * num_cols + j] = scale_factor / 2.0

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define non-overlap constraints
    def constraints_f(x):
        # x is a vector of [centers, radii]
        centers = x[:2 * num_circles].reshape(num_circles, 2)
        radii = x[2 * num_circles:]

            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
