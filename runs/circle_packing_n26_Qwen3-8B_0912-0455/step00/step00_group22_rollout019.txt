import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def create_initial_grid(num_circles: int, square_size: float = 1.0) -> np.ndarray:
    """
    Create an approximate hexagonal grid layout for the given number of circles.
    The first row is aligned to the bottom of the square.
    """
    # Define circle spacing (approximate hexagonal packing density)
    dx = np.sqrt(3) / 2  # Horizontal spacing between circle centers
    dy = 0.5  # Vertical spacing between horizontal rows

    # Row count and column count based on approximate hexagonal packing
    num_rows = int(np.ceil(np.sqrt(num_circles)))
    num_cols = int(np.ceil(num_circles / num_rows))

    # Create a 2D array of positions
    grid = []
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * dx
            y = row * dy
            if x + 0.5 > square_size:
                x = square_size - 0.5
            if y + 0.5 > square_size:
                y = square_size - 0.5
            grid.append((x, y))
    return np.array(grid)

def validate_packing(centers: np.ndarray, radii: np.ndarray) -> Tuple[bool, str]:
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    initial_centers = create_initial_grid(num_circles)

    # Initial guess for radii (all same for now)
    initial_radii = np.full(num_circles, 0.05)

    # Objective function for maximization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize actual sum

    # Constraints:
    # 1. Non-overlapping: For each pair of circles, distance >= sum of their radii
    # 2. Each circle must fit within [0,1]x[0,1]

    # Define bounds for each circle's radius (non-negative)
    bounds = [(0, 1.0) for _ in range(num_circles)]

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=[
            # Non-overlapping constraint
            optimize.constraints.Constraint(
                fun=lambda x: np.array(
                    [np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2)) - (x[i] + x[j]) for i in range(num_circles) for j in range(i + 1, num_circles)]
                ),
                type='ineq',
                jac=lambda x: np.array(
                    [np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2)) - (x[i] + x[j]) for i in range(num_circles) for j in range(i + 1, num_circles)]
                )
            )
        ],
        method='SLSQP'
    )

    # Extract results
    radii = result.x
    centers = initial_centers

    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
