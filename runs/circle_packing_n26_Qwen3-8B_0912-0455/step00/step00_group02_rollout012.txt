import numpy as np
import scipy.optimize as optimize

        cols = int(np.ceil(N / rows))
        grid = []

                x = c + 0.5 * (r % 2)

                grid.append((x, y))
        return np.array(grid)

        bounds.append((0.0, 1.0))  # x

        bounds.append((0.0, 0.5))  # r  # Max radius is limited to avoid overlapping with square

    # Function to compute the objective: maximize sum of radii
    def objective(x):

        return -np.sum(radii_flat)  # Negative for minimization

    # Constraints: distance between centers >= sum of radii
    def constraint(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
