import numpy as np
import random

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial configuration with hexagonal packing
    # Hexagonal packing with 26 circles
    centers = []
    radii = []

    # Centers of 16 circles arranged in a 4x4 grid (scaled for unit square)
    for i in range(4):
        for j in range(4):
            x = i / 3.0
            y = j / 3.0
            centers.append([x, y])

    # Arrange 10 circles around the 4x4 cluster
    # Adding circles at (0.2, 0.2), (0.8, 0.2), (0.2, 0.8), (0.8, 0.8), and 6 more distributed in the corners
    for i in range(4):
        x = i / 4.0
        y = (i + 3) / 4.0
        centers.append([x, y])
        radii.append(0.05)

    # Finer adjustment for center clustering
    # Use a genetic algorithm to refine the configuration
    # Set population size, generations, and optimization parameters
    pop_size = 100
    generations = 100
    mutation_rate = 0.1

    def fitness(centers, radii):
        valid, msg = validate_packing(centers, radii)
        if not valid:
            return 0.0
        return sum(radii)

    def random_individual():
        centers = np.random.rand(26, 2)
        radii = np.random.rand(26) * 0.1
        return centers, radii

    def crossover(parent1, parent2):
        child_centers = np.zeros((26, 2))
        child_radii = np.zeros(26)
        for i in range(26):
            if random.random() < 0.5:
                child_centers[i] = parent1[0][i]
                child_radii[i] = parent1[1][i]
            else:
                child_centers[i] = parent2[0][i]
                child_radii[i] = parent2[1][i]
        return child_centers, child_radii

    def mutate(individual):
        centers, radii = individual
        for i in range(26):
            if random.random() < mutation_rate:
                centers[i] += np.random.normal(0, 0.05, 2)
                centers[i][0] = np.clip(centers[i][0], 0, 1)
                centers[i][1] = np.clip(centers[i][1], 0, 1)
                radii[i] = np.clip(radii[i], 0, 0.25)
        return centers, radii

    # Initialize population
    population = [random_individual() for _ in range(pop_size)]

    for gen in range(generations):
        # Sort population by fitness
        population.sort(key=lambda x: fitness(x[0], x[1]), reverse=True)
        best_individual = population[0]

        # Create new population
        new_population = []
        for _ in range(pop_size // 2):
            # Select two parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)

        population = new_population

    best_centers, best_radii = best_individual
    return best_centers, best_radii, fitness(best_centers, best_radii)
