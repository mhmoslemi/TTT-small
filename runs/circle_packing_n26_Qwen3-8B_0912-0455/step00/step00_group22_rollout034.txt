import numpy as np
from scipy.optimize import minimize
import time

def run_packing():
    # Initial guess: place circles in a grid with spacing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # First, place circles in a hexagonal grid pattern inside the square
    max_radius = 0.1
    for i in range(num_circles):
        # Place circles in a hexagonal grid pattern
        r = i // 6
        c = i % 6
        x = c * (1 - 2 * max_radius) + max_radius
        y = r * (1 - 2 * max_radius) + max_radius
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = max_radius
    
    # Now, attempt to optimize the radii using a local optimization approach
    def objective(radii):
        return -np.sum(radii)  # maximize the sum of radii

    def constraints(radii):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Reformulate the problem for optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(
        objective,
        radii,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    # Extract the optimized radii
    optimized_radii = result.x
    # Ensure the constraints are satisfied
    if not constraints(optimized_radii):
        raise ValueError("Optimization failed to satisfy constraints")
    
    # Return the centers, radii, and sum of radii
    return (centers, optimized_radii, np.sum(optimized_radii))
