import scipy.optimize as optimize

def run_packing():

    n = 26

    def hex_grid(n):

        cols = int(np.ceil(n / rows))

            for j in range(cols):
                x = j + 0.5 * i
                y = i
                if i % 2 == 0:
                    x = j
                positions.append((x, y))
        return np.array(positions)

    radii = np.full(n, 0.05)

                dy = x[i * 2 + 1] - x[j * 2 + 1]
                dist = np.sqrt(dx**2 + dy**2)

                    return dist - (radii[i] + radii[j] - 1e-6)
