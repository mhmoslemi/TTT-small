import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26
    
    # Define variables for circle centers (x, y) and radii r
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)
    
    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    
    # Boundary constraints: circles must stay within [0,1]×[0,1]
    for i in range(n):
        constraints.append(centers[i, 0] >= 0)
        constraints.append(centers[i, 0] <= 1)
        constraints.append(centers[i, 1] >= 0)
        constraints.append(centers[i, 1] <= 1)

        constraints.append(centers[i, 0] + radii[i] <= 1 + 1e-12)
        constraints.append(centers[i, 1] - radii[i] >= -1e-12)
        constraints.append(centers[i, 1] + radii[i] <= 1 + 1e-12)
    
    # Non-overlapping constraint: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
