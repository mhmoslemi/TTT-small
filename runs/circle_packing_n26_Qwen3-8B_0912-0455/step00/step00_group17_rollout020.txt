import scipy.optimize as opt

    def objective(radii):

    # Constraints:
    # 1. Each circle must be within the unit square
    # 2. No two circles overlap
    # 3. Radii must be non-negative

    def dist(i, j):

                'type': 'ineq',
                'fun': non_overlapping(i, j)
            })

    result = opt.minimize(
        objective,

        bounds=bounds,
        constraints=constraints
