from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial hexagonal packing with spacing
    n = 26
    spacing = 1.0 / np.sqrt(3)  # Hexagonal packing spacing
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Generate initial hexagonal grid
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * spacing * np.sqrt(3) / 2
            if x < 1 and y < 1:
                centers.append([x, y])
                radii.append(1.0 / (np.sqrt(3) * (1 + i + j)))

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Bounds for centers (must be within [0, 1]x[0, 1])
    bounds = Bounds(np.zeros((n, 2)), np.ones((n, 2)))

    # Non-overlapping constraint (distance between centers >= sum of radii)
    def non_overlap_constraint(x):
        # x is a flat array of [x1, y1, x2, y2, ...]
        # Reshape to centers

        radii_flat = x[n * 2:]

        # Compute pairwise distances

    # Constraint: min distance between centers >= sum of radii

    # Initial guess: centers and radii

        bounds=bounds,
        constraints=cons,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
