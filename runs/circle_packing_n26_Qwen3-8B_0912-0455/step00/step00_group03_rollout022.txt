import scipy.optimize as opt

def run_packing():

    def generate_hexagonal_grid(num_rows, num_cols):

                    x = j * (1 + initial_radius)
                    y = i * np.sqrt(3) * initial_radius

                    x = (j + 0.5) * (1 + initial_radius)

                centers.append((x, y))
        return np.array(centers)

    # Create a grid with 5 rows and 5 columns
    centers = generate_hexagonal_grid(5, 5)
    radii = np.full(len(centers), initial_radius)

    # Define objective function: maximize sum of radii
    def objective(x, centers, radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Define constraints: non-overlapping and within square

    def constraint_non_overlap(x, centers, radii):
        # x[0] = radius_0, x[1] = radius_1, ..., x[n-1] = radius_n

        rs = x[:n]

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < rs[i] + rs[j] - 1e-12:
                    return dist - rs[i] - rs[j] + 1e-12
        return 0

    def constraint_boundary(x, centers, radii):
        n = len(centers)
        rs = x[:n]
        for i in range(n):
            x_center, y_center = centers[i]
            r = rs[i]

                return x_center - r + 1e-12

        return 0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
