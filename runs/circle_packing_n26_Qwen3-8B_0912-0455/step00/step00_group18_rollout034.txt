import numpy as np
import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Step 1: Define a hybrid packing structure
    # 26 circles: 7 in a hexagonal cluster, and 19 in a square grid

    # Define the centers for a hexagonal cluster of 7 circles (radius r1)
    r1 = 0.03
    hex_centers = []
    for i in range(3):

            hex_centers.append([i * (1 + r1), j * (1 + r1)])
    for i in range(3):
        hex_centers.append([i * (1 + r1), 0.5 * (1 + r1)])
    centers = np.array(hex_centers)

    # Define the centers for a square grid of 19 circles
    r2 = 0.07
    grid_centers = []
    for i in range(4):
        for j in range(5):
            grid_centers.append([i * (1 + r2), j * (1 + r2)])
    centers = np.vstack((centers, grid_centers))

    # Now, use convex optimization to maximize the sum of radii
    # All circles are in a unit square, with centers in [0,1]^2
    # We will use a simple optimization to adjust radii

    n = centers.shape[0]
    r = cp.Variable(n)
    obj = cp.sum(r)
    constraints = []

    # Distance between centers must be at least radii[i] + radii[j]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # All circles must be within the square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - r[i] >= 0)
        constraints.append(x + r[i] <= 1)
        constraints.append(y - r[i] >= 0)
        constraints.append(y + r[i] <= 1)

    # All radii must be non-negative
    constraints.append(cp.constraints.NonNeg(r))

    # Formulate the optimization problem
    prob = cp.Problem(cp.Maximize(obj), constraints)
    prob.solve()

    # Extract the result
    radii = r.value
    centers = centers
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
