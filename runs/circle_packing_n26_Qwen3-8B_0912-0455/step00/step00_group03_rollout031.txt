import numpy as np
import scipy.optimize as optimize

def run_packing():
    from scipy.optimize import Bounds
    import warnings

    # to allow center circles and avoid overlap
    initial_centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],

        [0.5, 0.9], [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    n = initial_centers.shape[0]

    # Initial radius estimate (based on nearest neighbor distance from initial grid)
    initial_radii = np.array([0.1] * n)
    # This is just a rough estimate for optimization

    def objective(x):
        # x is flattened: [x1, y1, r1, x2, y2, r2, ...]
        x_vals = x[:2*n].reshape((n, 2))
        r_vals = x[2*n:]
        total_sum = np.sum(r_vals)
        return -total_sum  # Minimize negative to maximize

    def constraints(x):
        x_vals = x[:2*n].reshape((n, 2))
        r_vals = x[2*n:]
        constraints = []

        # Boundary constraints
        for i in range(n):
            x_i, y_i = x_vals[i]
            r_i = r_vals[i]
            # Ensure the circle does not exceed the square
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.array([False])
        constraints.append(np.array([True]))

        # Non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((x_vals[i] - x_vals[j])**2))
