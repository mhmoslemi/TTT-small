import scipy.optimize as optimize

def run_packing():
    # Step 1: Initialize a hexagonal grid with 26 circles
    num_circles = 26
    grid = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * row
            y = row
            grid.append((x, y))

    # Convert to NumPy arrays
    centers = np.array(grid)
    radii = np.zeros(num_circles)

    # Step 2: Ensure circles are within the unit square [0, 1] x [0, 1]
    # Start with a simple back-of-the-envelope estimation of radii
    # (this will be refined during optimization)
    max_radius = 0.15
    radii[:] = max_radius

    # Define the objective function to maximize sum of radii
    def objective(x, centers, radii):
        return -np.sum(radii)  # Negative because we minimize

    # Define the constraints:
    # 1. Each circle must be within the unit square
    # 2. The distance between centers must be at least the sum of their radii
    # 3. Radii must be non-negative

    def constraint1(x, i):
        r = x[i + num_circles]
        x_center, y_center = centers[i]
        return x_center - r >= 0
    def constraint2(x, i):
        r = x[i + num_circles]
        x_center, y_center = centers[i]
        return x_center + r <= 1
    def constraint3(x, i):
        r = x[i + num_circles]
        x_center, y_center = centers[i]
        return y_center - r >= 0
    def constraint4(x, i):
        r = x[i + num_circles]
        x_center, y_center = centers[i]
        return y_center + r <= 1

    # Create constraints for all circles
    constraints = []
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint1(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint2(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint3(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint4(x, i)})

    # Define pairwise distance constraints
    def distance_constraint(x, i, j):
        r_i = x[i + num_circles]
        r_j = x[j + num_circles]
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        return np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2) - (r_i + r_j)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
