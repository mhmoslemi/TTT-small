import numpy as np
from scipy.optimize import minimize
import time

_g_n = 45
_g_ts = 22.5
_g_dx = 2.0 / 45

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def adjust_sum(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h):
    return compute_c5(h, _g_dx)

def _con_eq(h):
    return np.sum(h) - _g_ts

def _con_jac(h):

    _g_n = n
    _g_ts = ts
    _g_dx = dx
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n

        r = minimize(_obj, h, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return r.x

        return h
