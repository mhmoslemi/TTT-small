            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n-kk]
        grad += g
    grad *= dx / len(active)
    return c5, grad

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()

            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += room * (diff / s)
        else:
            room = h
            s = room.sum()
            if s < 1e-15:
                break
            h -= room * (-diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    c5 = vals.max()
    active = np.where(vals >= c5 - 1e-8)[0]
    grad = np.zeros(n)
    for p in active:
        lag = n - 1 - p
        g = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            kk = -lag
            if n - kk > 0:
                g[:n - kk] += (1.0 - h[kk:])
                g[kk:] -= h[:n - kk]
        grad += g
    grad *= dx / max(len(active), 1)
    return c5, grad

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += room * (diff / s)
        else:
            room = h
            s = room.sum()
            if s < 1e-15:
                break
            h -= room * (-diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_one(h, target, max_it, t0, budget_s):
    n = len(h)
    lr = 0.08
    best_local = eval_c5(h)
    h_best = h.copy()
    stall = 0
    for it in range(max_it):
        if time.time() - t0 > budget_s * 0.92:
            break
        c5, grad = compute_grad(h)
        h_new = project(h - lr * grad, target)
        c5_new = eval_c5(h_new)
        if c5_new < c5 - 1e-10:
            h = h_new
            lr = min(lr * 1.03, 0.4)

            stall += 1
            if stall > 40:
                h = project(h + np.random.default_rng(it).normal(0, 0.03, n), target)
                lr = 0.04
                stall = 0
    return h_best, best_local
