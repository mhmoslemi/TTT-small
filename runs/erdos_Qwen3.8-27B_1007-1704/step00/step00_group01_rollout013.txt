    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.ones(47) * 0.5

    n0 = len(h_init)
    h_best = project_mass(h_init, n0)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP at native resolution with multiple restarts
    for restart in range(3):
        if elapsed() > budget_s - 80:
            break
        if restart == 0:
            h_cur = h_best
        else:
            h_cur = h_best + rng.randn(n0) * 0.02
            h_cur = project_mass(h_cur, n0)
        h_o, c5_o = slsqp_optimize(h_cur, n0, max_iter=4000)
        if c5_o < c5_best:
            h_best, c5_best = h_o, c5_o

    # Phase 2: Nelder-Mead refinement
    if elapsed() < budget_s - 60:
        h_o, c5_o = nm_optimize(h_best, n0, max_iter=5000)
        if c5_o < c5_best:
            h_best, c5_best = h_o, c5_o

    # Phase 3: Multi-resolution optimization
    for mult in [2, 3]:
        n_hi = mult * n0
        if n_hi > 250 or elapsed() > budget_s - 120:
            break
        h_hi = project_mass(interpolate_h(h_best, n0, n_hi), n_hi)
        h_o, c5_o = slsqp_optimize(h_hi, n_hi, max_iter=2500)
        if c5_o < c5_best:
            c5_best = c5_o
            h_back = project_mass(interpolate_h(h_o, n_hi, n0), n0)
            h_best = h_back
            c5_best = compute_c5(h_best)
        if elapsed() > budget_s - 60:
            break

    # Phase 4: Coordinate descent on dominant lags
    if elapsed() < budget_s - 40:
        for it in range(200):
            if elapsed() > budget_s - 20:
                break
            corr = np.correlate(h_best, 1.0 - h_best, mode="full")
            dx = 2.0 / n0
            # Find near-max lags
            thresh = np.max(corr) * 0.98
            active = np.where(corr > thresh)[0]
            improved = False
            for idx in rng.choice(n0, size=min(10, n0), replace=False):
                for delta in [0.003, -0.003, 0.001, -0.001]:
                    h_try = h_best.copy()
                    h_try[idx] += delta
                    h_try = project_mass(h_try, n0)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_best - 1e-10:
                        h_best, c5_best = h_try, c5_try
                        improved = True
                        break
                if improved:
                    break
            if not improved and it > 50:
                break

    # Phase 5: Final SLSQP polish with very tight tolerance
    if elapsed() < budget_s - 10:
        h_o, c5_o = slsqp_optimize(h_best, n0, max_iter=5000, ftol=1e-15)
        if c5_o < c5_best:
            h_best, c5_best = h_o, c5_o

    # Phase 6: Tiny random perturbations for last-mile improvement
    if elapsed() < budget_s - 5:
        step = 0.002
        for _ in range(200):
            if elapsed() > budget_s - 3:
                break
            h_p = h_best + rng.randn(n0) * step
            h_p = project_mass(h_p, n0)
            c5_p = compute_c5(h_p)
            if c5_p < c5_best:
                h_best, c5_best = h_p, c5_p
            else:
                step *= 0.93
            if step < 1e-7:
                break

    h_best = project_mass(h_best, n0)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n0
