import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = h + (target - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new)

def sa_optimize(h, n, budget_s, rng, T_init=0.04, T_min=0.002):
    t_start = time.time()
    T = T_init
    best_c5 = eval_c5(h)
    best_h = h.copy()
    stall = 0
    iters = 0

    while time.time() - t_start < budget_s:
        # Choose move type
        r = rng.random()
        if r < 0.85:
            # Pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-T, T)
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            h[i] = hi
            h[j] = hj
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
                T = min(T_init, T * 1.005)
            else:
                if rng.random() > np.exp(-(c5 - best_c5) / max(T, 1e-12)):
                    h[i] -= d
                    h[j] += d
                stall += 1
        elif r < 0.95:
            # Segment swap (preserves sum if segments have same sum, or we repair)
            seg_len = rng.randint(2, max(3, n // 5))
            i = rng.randint(n - seg_len)
            j = rng.randint(n - seg_len)
            if abs(i - j) < seg_len:
                continue
            seg_i = h[i:i + seg_len].copy()
            seg_j = h[j:j + seg_len].copy()
            # Swap
            h[i:i + seg_len] = seg_j
            h[j:j + seg_len] = seg_i
            h = np.clip(h, 0.0, 1.0)
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                if rng.random() > np.exp(-(c5 - best_c5) / max(T, 1e-12)):
                    h[i:i + seg_len] = seg_i
                    h[j:j + seg_len] = seg_j

        iters += 1
        if stall > 8000:
            T = max(T_min, T * 0.9)
            stall = 0
        elif iters % 20000 == 0:
            T = max(T_min, T * 0.99)

    return best_h, best_c5

    t0 = time.time()

    def time_left():

    n_candidates = []
    for mult in [2, 3, 4]:
        nc = n_best * mult
        if nc <= 400:

    n_candidates = sorted(set(n_candidates))

    for nc in n_candidates:
