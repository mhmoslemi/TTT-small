        if rem() > 20:
            h, c5 = local_refine(h, n_new, n_iter=200)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Stage 3: random restarts at original grid if time remains
    if rem() > 30:
        x_lo = np.linspace(0, 2, len(best_h))
        x_hi = np.linspace(0, 2, n0)
        h_seed = np.interp(x_hi, x_lo, best_h)
        for restart in range(5):
            if rem() < 20:
                break
            h_try = h_seed + np.random.normal(0, 0.08, n0)
            h_try = normalize_h(h_try, n0)
            h, c5 = optimize_grid(h_try, n0, min(rem() * 0.15, 60))
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Final: ensure best is at reasonable resolution
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
