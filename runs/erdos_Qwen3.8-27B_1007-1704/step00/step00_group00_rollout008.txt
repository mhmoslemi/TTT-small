import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def power_mean_c5(h, dx, p=50):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((np.sum(corr**p) / len(corr))**(1.0/p) * dx)

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        candidates = np.where(h < 1.0)[0]
        room = 1.0 - h[candidates]
        if room.sum() == 0:
            return h
        scale = min(1.0, diff / room.sum())
        h[candidates] += room * scale
    else:
        candidates = np.where(h > 0.0)[0]
        room = h[candidates]
        if room.sum() == 0:
            return h
        scale = min(1.0, -diff / room.sum())
        h[candidates] -= room * scale
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            mask = h > 0.0
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, dx, target_sum, rng, max_iter=50000, time_limit=None, t_start=None):
    best_c5 = compute_c5(h, dx)
    step = 0.05
    for it in range(max_iter):
        if time_limit and time.time() - t_start > time_limit:
            break
        # Pick two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Try moving mass from j to i (or vice versa)
        delta = step * rng.choice([-1, 1])
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            step = min(0.05, step * 1.01)  # slightly grow step
        else:
            step = max(0.001, step * 0.98)
    return h, best_c5

def slsqp_optimize(h, n, dx, target_sum, rng, time_limit=None, t_start=None):
    def objective(x):
        x = np.clip(x, 0, 1)
        return power_mean_c5(x, dx, p=100)
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    bounds = [(0, 1)] * n
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})
    if result.success or result.fun < power_mean_c5(h, dx, p=100):
        h = project_onto_constraint(result.x, target_sum)
    return h

import numpy as np
from scipy.optimize import minimize, minimize_nelder_mead
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def power_mean_obj(h, dx, p=80):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((np.sum(corr**p) / len(corr))**(1.0/p) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, n, dx, target_sum, max_iter=300):
    def obj(x):
        return power_mean_obj(x, dx, p=100)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    out = project_h(res.x, target_sum)
    return out

def coord_descent(h, n, dx, target_sum, rng, max_iter=50000, t_deadline=None):
    best = compute_c5(h, dx)
    step = 0.04
    for _ in range(max_iter):
        if t_deadline and time.time() > t_deadline:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        d = step * rng.choice([-1.0, 1.0])
        hi, hj = h[i] + d, h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            step *= 0.9
            continue
        h[i], h[j] = hi, hj
        c = compute_c5(h, dx)
        if c < best:
            best = c
            step = min(0.08, step * 1.005)
        else:
            h[i], h[j] = hj, hi  # revert
            step = max(0.002, step * 0.97)
    return h, best

def nelder_mead_local(h, n, dx, target_sum, t_deadline=None):
    def obj(x):
        x = project_h(x, target_sum)
        return power_mean_obj(x, dx, p=100)
    res = minimize_nelder_mead(obj, h, maxiter=2000, xatol=1e-8, fatol=1e-10,
                               initial_simplex=None, options={'maxfev': 3000})
    return project_h(res.x, target_sum)

def interpolate_h(h_src, n_dst):
    x_src = np.linspace(0, 1, len(h_src), endpoint=False)
    x_dst = np.linspace(0, 1, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def generate_random_h(n, rng):
    h = rng.uniform(0.3, 0.7, n)
    return project_h(h, n / 2.0)

def generate_block_h(n, rng):
    block = rng.integers(3, max(4, n // 8) + 1)
    h = np.zeros(n)
    idx = 0
    state = 1
    while idx < n:
        length = min(block, n - idx)
        h[idx:idx+length] = state
        state = 1 - state
        idx += length
    return project_h(h, n / 2.0)
