import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        idx = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(idx) > 0:
            h[idx[0]] -= diff
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_ext = np.concatenate([h, h[:1]])
    h_new = np.interp(x_new, x_old, h_ext)
    return h_new

def sa_optimize(h, n, rng, t_end, T_init, T_min, multi_swap=True):

    best_c5 = compute_c5(best)
    T = T_init
    while time.time() < t_end:
        cand = best.copy()
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:

        lo = max(0.0, cand[j] - 1.0 + 1e-14)
        hi = min(cand[i], 1.0 - cand[j] - 1e-14)
        if lo >= hi - 1e-14:
            continue
        delta = rng.uniform(lo, hi)
        if multi_swap:
            i2 = rng.randint(0, n)
            j2 = rng.randint(0, n)
            if i2 == j2:
                i2 = (i2 + 1) % n
            lo2 = max(0.0, cand[j2] - 1.0 + 1e-14)
            hi2 = min(cand[i2], 1.0 - cand[j2] - 1e-14)
            if lo2 < hi2 - 1e-14:
                d2 = rng.uniform(lo2, hi2) * 0.3
                cand[i2] += d2
                cand[j2] -= d2
        cand[i] += delta
        cand[j] -= delta
        cand = project_feasible(cand, n)
        c5 = compute_c5(cand)
        d = c5 - best_c5

            best_c5 = c5
            best = cand
            T = max(T * 0.99998, T_min)
    return best, best_c5
