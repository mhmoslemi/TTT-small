def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += d * room / tr
        else:
            av = h.copy()
            ta = av.sum()
            if ta < 1e-15:
                break
            h += d * av / ta
        h = np.clip(h, 0.0, 1.0)
    return h

    except Exception:
        pass
    return h0c, compute_c5(h0c, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = np.full(47, 0.5)

    n0 = len(parent)
    parent = repair_h(parent, n0)
    best_h = parent.copy()
    best_c5 = compute_c5(parent, 2.0 / n0)
    best_n = n0

    # Phase 1: Coordinate descent at original resolution
    h_cd, c5_cd = coord_descent(parent, n0, t0, budget_s, max_passes=100)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        best_n = n0

    # Phase 2: SLSQP refinement
    if time.time() - t0 < budget_s - 80:
        h_s, c5_s = slsqp_optimize(best_h, n0, t0, budget_s, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        h_cd2, c5_cd2 = coord_descent(h_s, n0, t0, budget_s, max_passes=80)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 3: Random search at original resolution
    if time.time() - t0 < budget_s - 60:
        h_rs, c5_rs = random_search(best_h, n0, t0, budget_s, max_iters=300000)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
        h_cd3, c5_cd3 = coord_descent(h_rs, n0, t0, budget_s, max_passes=60)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 4: Multi-start with different perturbations
    remaining = budget_s - (time.time() - t0)
    n_starts = min(12, int(remaining / 40))
    for si in range(n_starts):
        if time.time() - t0 > budget_s - 30:
            break
        scale = 0.05 + 0.1 * np.random.rand()
        h_start = best_h + np.random.randn(best_n) * scale
        h_start = repair_h(h_start, best_n)
        h_cd_s, c5_cd_s = coord_descent(h_start, best_n, t0, budget_s, max_passes=50)
        if c5_cd_s < best_c5:
            best_c5 = c5_cd_s
            best_h = h_cd_s.copy()
            best_n = best_n
        h_rs_s, c5_rs_s = random_search(h_cd_s, best_n, t0, budget_s, max_iters=80000)
        if c5_rs_s < best_c5:
            best_c5 = c5_rs_s
            best_h = h_rs_s.copy()
            best_n = best_n

    # Phase 5: Try higher resolution (interpolate + optimize)
    remaining = budget_s - (time.time() - t0)
    for n_try in [n0 * 2, min(n0 * 3, 200)]:
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair_h(h_interp, n_try)
        c5_interp = compute_c5(h_interp, 2.0 / n_try)
        # Coordinate descent at new resolution
        h_hd, c5_hd = coord_descent(h_interp, n_try, t0, budget_s, max_passes=50)
        if c5_hd < best_c5:
            best_c5 = c5_hd
            best_h = h_hd.copy()
            best_n = n_try
        # SLSQP at new resolution
        if time.time() - t0 < budget_s - 40:
            h_hs, c5_hs = slsqp_optimize(h_hd, n_try, t0, budget_s, maxiter=150)
            if c5_hs < best_c5:
                best_c5 = c5_hs
                best_h = h_hs.copy()
                best_n = n_try
            h_hd2, c5_hd2 = coord_descent(h_hs, n_try, t0, budget_s, max_passes=40)
            if c5_hd2 < best_c5:
                best_c5 = c5_hd2
                best_h = h_hd2.copy()
                best_n = n_try
        # Quick random search at new resolution
        if time.time() - t0 < budget_s - 30:
            h_hr, c5_hr = random_search(h_hd, n_try, t0, budget_s, max_iters=100000)
            if c5_hr < best_c5:
                best_c5 = c5_hr
                best_h = h_hr.copy()
                best_n = n_try

    # Final: ensure feasibility
    best_h = repair_h(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
