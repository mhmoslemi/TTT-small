            j = rng.integers(n)

                continue

            new_h[i] = new_h[j]
            new_h[j] = tmp
        else:

            i = rng.integers(n)
            j = rng.integers(n)

                new_h[i] -= eps
                new_h[j] += eps
            elif new_h[j] >= eps and new_h[i] <= 1 - eps:
                new_h[j] -= eps
                new_h[i] += eps

                continue

def local_search_binary(h, rng, time_budget=30.0, target_sum=None):
    """Local search for binary/near-binary solutions."""
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    t0 = time.time()

    cur_val = compute_c5(current)
    best = current.copy()
    best_val = cur_val
    
    # Get indices of 1s and 0s (for near-binary)
    T = 0.01
    T_min = 0.0001
    
    no_improve = 0
    streak = 0
    
    while time.time() - t0 < time_budget:
        # Move: swap a 1-position with a 0-position
        ones = np.where(current > 0.5)[0]
        zeros = np.where(current < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
