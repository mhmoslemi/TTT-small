def evaluate_objective(h_vec):
    # Full cross correlation mode
    n = len(h_vec)
    g = 1 - h_vec
    corr = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        lag_idx = lag + (n-1)
        if lag < 0:
            corr[lag_idx] = np.sum(h_vec * g[-lag:])
        elif lag == 0:
            corr[lag_idx] = np.sum(h_vec * g[:])
        else:
            corr[lag_idx] = np.sum(h_vec * g[:lag+1])
    return np.max(corr) * (2.0 / n)  # dx * max_corr  (since dx=2/n, here n_points = len(h))

def compute_c5(h_vec, dx=2.0/None):
    if dx is None:
        dx = 2.0 / len(h_vec)
    n = len(h_vec)
    # Compute correlation in full mode
    g = 1 - h_vec
    # Optimized computation of cross-correlation (without loops)
    # Doing full convolution here with finite width support for this problem
    out_size = 2*n - 1
    corr = np.zeros(out_size)
    # 
    # This method is vectorized and efficient for the small example
    g_padded = np.pad(g, (0, 0), 'constant')  # Just g[:n] but need to do standard convolution alignment
    #
    # Standard way: 
    #   corr[0:n-1] = np.correlate(h_vec, g, mode='valid', method='direct')
    #   but we need full
    #
    # Alternatively, use built-in method if we don't mind the O(n^2) complexity for small n
    # The original simpler approach:
    corr = np.zeros(2*n - 1)
    max_val = 0
    # Or use the initial direct convolution method for clarity
    for lag in range(-(n-1), n):
        lag_idx_full = lag + n - 1  # This shifts positive lags to [n-1, 2*n-2]
        if lag < 0:
            start = -lag
            end = n
            corr[lag_idx_full] = np.sum(h_vec[:start] * g[n + start:])
        elif lag == 0:
            corr[lag_idx_full] = np.sum(h_vec * g)
        else:
            corr[lag_idx_full] = np.sum(h_vec[:lag] * g[:lag+1])
    max_val = np.max(corr) * dx
    return max_val

def grid_transfer(h, n_source, n_target):
    # Transfer between grid sizes
    dx_source = 2.0 / n_source
    dx_target = 2.0 / n_target
    
    # Interpolation along the domain [0,2]
    domain_source = np.linspace(0, 2, n_source, endpoint=False)
    domain_target = np.linspace(0, 2, n_target, endpoint=False)
    
    h_interpolated = np.interp(domain_target, domain_source, h)
    # Repair to enforce physical bounds and integral constraint
    h_repaired = repair_constraint(h_interpolated, dx_target, n_target)
    
    return h_repaired
