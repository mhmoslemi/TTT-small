import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_correlations(h)) * dx

def project_onto_box_sum(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Iterate to fix the sum
    for _ in range(100):
        if diff > 0:
            # Need to increase some values
            can_increase = h < 1.0 - 1e-15
            if not np.any(can_increase):
                break
            val = diff / np.sum(can_increase)
            h[can_increase] = np.clip(h[can_increase] + val, 0, 1)
        else:
            can_decrease = h > 1e-15
            if not np.any(can_decrease):
                break
            val = abs(diff) / np.sum(can_decrease)
            h[can_decrease] = np.clip(h[can_decrease] - val, 0, 1)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
    return h

def subgradient_step(h, step_size):
    """One subgradient step for minimizing max correlation."""
    n = len(h)
    corr = compute_correlations(h)
    max_corr = np.max(corr)
    # Find active lags (within 10% of max)
    threshold = max_corr * 0.9
    active = np.where(corr >= threshold)[0]
    
    # Compute gradient for active lags and average
    grad = np.zeros(n)
    for m in active:
        j = m - (n - 1)  # lag
        # For each l, gradient contribution:
        # (1-h[l+j]) if 0 <= l+j < n
        # -h[l-j] if 0 <= l-j < n
        l = np.arange(n)
        # First term: h[l] appears as first factor when l+j is valid
        valid1 = (l + j >= 0) & (l + j < n)
        # Second term: h[l] appears as second factor when l-j is valid
        valid2 = (l - j >= 0) & (l - j < n)
        
        g = np.zeros(n)
        g[valid1] += (1 - h[l[valid1] + j])
        g[valid2] += -h[l[valid2] - j]
        grad += g
    
    grad /= len(active)
    return h - step_size * grad, max_corr

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def grad_corr_at_lag(h, j):
    """Gradient of corr at lag j w.r.t. h. j is the lag (-n+1 to n-1)."""
    n = len(h)
    l = np.arange(n)
    grad = np.zeros(n)
    # (1-h[l-j]) where 0 <= l-j < n
    valid1 = (l - j >= 0) & (l - j < n)
    grad[valid1] += (1.0 - h[l[valid1] - j])
    # -h[l+j] where 0 <= l+j < n  
    valid2 = (l + j >= 0) & (l + j < n)
    grad[valid2] -= h[l[valid2] + j]
    return grad

def project(h, target_sum):
    """Project onto {0 <= h <= 1, sum(h) = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-14:
        return h
    # Simple iterative projection
    for _ in range(200):

        if not np.any(mask):

        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)

        step = step_base / np.sqrt(1.0 + 0.005 * it)
        h = h - step * grad
        h = project_sum_box(h, target)
        h_avg = 0.995 * h_avg + 0.005 * h

    _, corr_avg = compute_c5(h_avg, dx)
    if float(np.max(corr_avg) * dx) < best_c5:
        best_h = h_avg.copy()
        best_c5 = float(np.max(corr_avg) * dx)

    return best_h, best_c5

def targeted_fix(h, dx, n_targets, deadline):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    _, corr = compute_c5(h, dx)
    best_c5 = float(np.max(corr) * dx)
    for tgt in range(n_targets):
        if time.time() > deadline:
            break
        _, corr = compute_c5(best_h, dx)
        worst_idx = np.argmax(corr)
        worst_lag = worst_idx - (n - 1)
        grad = grad_corr_at_lag(best_h, worst_lag)
        h = best_h - 0.02 * grad
        h = project_sum_box(h, target)
        _, corr_new = compute_c5(h, dx)
        if float(np.max(corr_new) * dx) < best_c5:
            best_c5 = float(np.max(corr_new) * dx)
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    deadline = start + budget_s - 20

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(45) * 0.5

    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    _, best_c5 = compute_c5(best_h, dx0)
    best_n = n0

    # Phase 1: Subgradient at initial resolution
    h_opt, c5_opt = optimize_subgrad(h0.copy(), dx0, 20000, deadline, 0.85)
    if c5_opt < best_c5:
        best_c5, best_h, best_n = c5_opt, h_opt, n0

    # Phase 2: Targeted lag fixing
    h_fix, c5_fix = targeted_fix(best_h.copy(), dx0, 50, deadline)
    if c5_fix < best_c5:
        best_c5, best_h = c5_fix, h_fix

    # Re-run subgrad after fixing
    h_opt, c5_opt = optimize_subgrad(best_h.copy(), dx0, 10000, deadline, 0.80)
    if c5_opt < best_c5:
        best_c5, best_h, best_n = c5_opt, h_opt, n0

    # Phase 3: Try higher resolutions
    for n_pts in [90, 150, 200]:
        if time.time() > deadline:
            break
        dx = 2.0 / n_pts
        n_cur = best_n
        x_old = (np.arange(n_cur) + 0.5) * (2.0 / n_cur)
        x_new = (np.arange(n_pts) + 0.5) * (2.0 / n_pts)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum_box(h_interp, n_pts / 2.0)
        h_opt, c5_opt = optimize_subgrad(h_interp, dx, 8000, deadline, 0.85)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, n_pts
        h_fix, c5_fix = targeted_fix(best_h.copy(), 2.0 / best_n, 30, deadline)
        if c5_fix < best_c5:
            best_c5, best_h = c5_fix, h_fix

    # Phase 4: Random restarts
    if time.time() < deadline - 30:
        dx = 2.0 / best_n
        for restart in range(8):
            if time.time() > deadline:
                break
            noise_scale = 0.05 + 0.02 * restart
            h_r = best_h + rng.randn(best_n) * noise_scale
            h_r = project_sum_box(h_r, best_n / 2.0)
            h_opt, c5_opt = optimize_subgrad(h_r, dx, 5000, deadline, 0.90)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt

    # Phase 5: Final targeted refinement
    dx = 2.0 / best_n
    h_fix, c5_fix = targeted_fix(best_h.copy(), dx, 100, deadline)
    if c5_fix < best_c5:
        best_c5, best_h = c5_fix, h_fix
    h_opt, c5_opt = optimize_subgrad(best_h.copy(), dx, 5000, deadline, 0.75)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    final_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
