import time
import math

def smooth_max(corr_vals, tau):
    m = np.max(corr_vals)
    e = np.exp((corr_vals - m) / tau)
    return m + tau * np.log(np.sum(e)), e / np.sum(e)

    n = len(h)
    rn = 1.0 - h
    corr = np.correlate(h, rn, mode="full") * dx
    sm, weights = smooth_max(corr, tau)
    return sm, weights, corr

def compute_gradient(h, dx, tau):

    rn = 1.0 - h
    corr = np.correlate(h, rn, mode="full") * dx
    sm, weights = smooth_max(corr, tau)
    
    # corr has length 2n-1, indexed 0..2n-2
    # corr[k] corresponds to lag L = k - (n-1)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:

        L = k - (n - 1)
        w = weights[k]
        for j in range(n):
            # g[j] = 1 - h[j+L] if 0<=j+L<n else 1

            if 0 <= j + L < n:

            if 0 <= j - L < n:

import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    rn = 1.0 - h
    corr = np.correlate(h, rn, mode="full")
    return np.max(corr * dx)

def project_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda"""
    h = h.copy()
    # Bisection on lambda: find lambda s.t. sum(clip(h+lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def smooth_max_and_weights(corr_vals, tau):
    m = np.max(corr_vals)
    e = np.exp((corr_vals - m) / tau)
    s = e.sum()
    return m + tau * math.log(s), e / s

def compute_gradient(h, dx, tau):

    rn = 1.0 - h
    corr = np.correlate(h, rn, mode="full") * dx
    _, weights = smooth_max_and_weights(corr, tau)

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def _obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp((corr - m) / tau)
    s = e.sum()
    obj = float(m + tau * math.log(s))
    w = e / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        for j in range(n):
            t = 1.0
            if 0 <= j + L < n:
                t -= h[j + L]
            if 0 <= j - L < n:
                t -= h[j - L]
            grad[j] += wk * dx * t
    return obj, grad

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _scipy_polish(h, target):

        return g
