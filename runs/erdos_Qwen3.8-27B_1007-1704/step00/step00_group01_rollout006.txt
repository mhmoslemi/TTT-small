import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def c5_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = (corr - cmax) * tau
    shifted = np.clip(shifted, -400.0, 400.0)
    return cmax + np.log(np.sum(np.exp(shifted))) / tau

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = np.sum(room)
            if tr < 1e-15:
                break
            h = np.clip(h + diff * room / tr, 0.0, 1.0)
        else:
            tr = np.sum(h)
            if tr < 1e-15:
                break
            h = np.clip(h + diff * h / tr, 0.0, 1.0)
    return h

    x_old = (np.arange(n_old) + 0.5) / (n_old / 2.0)

def run_slsqp(h0, n, tau, max_iter=400):

        return c5_smooth(x, tau)

        h_opt = repair(res.x, n)
        return h_opt, evaluate_c5(h_opt)
    except Exception:

def projected_subgrad(h, n, steps=2000, lr_init=0.02):
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_c = evaluate_c5(h)
    lr = lr_init
    for it in range(steps):
        corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr_vals)
        if cmax * dx < best_c - 1e-12:
            best = h.copy()
            best_c = evaluate_c5(h)
        weights = np.exp((corr_vals - cmax) * 50.0)
        weights /= weights.sum()
        grad = np.zeros(n)
        for k_offset in range(-(n - 1), n):
            if weights[0] < 1e-15:
                break
            w = weights[k_offset + n - 1]
            if w < 1e-12:
                continue
            lo = max(0, -k_offset)
            hi = min(n, n - k_offset)
            for i in range(lo, hi):

                grad[i] += w * (1.0 - h[j]) * dx
                if j != i:
                    grad[j] += w * (-h[i]) * dx
        h = h - lr * grad
        h = repair(h, n)
        lr *= 0.999
    return best, best_c
