            h = np.interp(x_new, x_old, h_init)
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, n / 2.0)

        h_opt, c5_opt = optimize_one(h, n, time_alloc, seed + n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final refinement on best solution
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        n_best = len(best_h)
        h_opt, c5_opt = optimize_one(best_h.copy(), n_best, remaining, seed + 7777)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
