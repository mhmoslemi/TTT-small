import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    return h

def min_max_1d(a, b, n_iter=35):
    """Ternary search for t in [0,1] minimizing max_i(a_i*t + b_i)."""
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, rng, max_rounds=80, t_limit=60):
    t0 = time.time()
    h = h.copy()
    best = compute_c5(h)
    off = n - 1
    N = 2 * n - 1
    for rnd in range(max_rounds):
        if time.time() - t0 > t_limit:
            break
        imp = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            a = np.zeros(N)
            lo1 = max(0, off - j)
            hi1 = min(N - 1, 2 * off - j)
            if lo1 <= hi1:
                ci = j + np.arange(lo1, hi1 + 1) - off
                a[lo1:hi1 + 1] += 1.0 - h[ci]
            lo2 = max(0, j)
            hi2 = min(N - 1, off + j)
            if lo2 <= hi2:
                ci2 = j - np.arange(lo2, hi2 + 1) + off
                a[lo2:hi2 + 1] -= h[ci2]
            corr = np.correlate(h, 1.0 - h, mode="full")
            b = corr - a * old
            t_opt = min_max_1d(a, b)
            h[j] = t_opt
            h = repair(h, n)
            nv = compute_c5(h)
            if nv < best - 1e-13:
                best = nv
                imp = True
            else:
                h[j] = old
                h = repair(h, n)
        if not imp:
            break
    return h, best

def swap_local(h, n, rng, n_iter=20000, t_limit=20):
    t0 = time.time()
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    T = 0.005
    for it in range(n_iter):
        if time.time() - t0 > t_limit:
            break
        if it % 200 == 0:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = 1.0 - old_i, 1.0 - old_j
        h = repair(h, n)
        nv = compute_c5(h)
        delta = nv - best
        if delta < 0 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-6)):
            if nv < best:
                best = nv
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
            h = repair(h, n)
        T *= 0.998
    return best_h, best
