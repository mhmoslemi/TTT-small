import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(r))

def soft_max_val(h, T):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(T * logsumexp(r / T))

def soft_max_grad_val(h, T):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(r)
    p = np.exp((r - m) / T)
    psum = p.sum()
    p = p / psum
    val = m + T * np.log(psum)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for ki in range(2 * n - 1):
            shift = ki - (n - 1)
            if 0 <= i - shift <= n - 1:
                s += p[ki] * (1.0 - h[i - shift])
            if 0 <= i + shift <= n - 1:
                s -= p[ki] * h[i + shift]
        grad[i] = dx * s
    return grad, float(val)

def fix_integral(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(h < 1.0) if diff > 0 else np.argmin(h > 0.0)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def optimize_h(h0, t_remaining, seed=0):
    n = len(h0)
    h = h0.copy()
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    
    temperatures = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for T in temperatures:
        if t_remaining() < 30:
            break
        steps = int(min(300, max(30, t_remaining() * 2.0)))
        
        def obj(x):
            return soft_max_val(x, T) + 200.0 * (x.sum() - n / 2.0) ** 2
        
        def grad(x):
            g, _ = soft_max_grad_val(x, T)
            g += 400.0 * (x.sum() - n / 2.0)
            return g
        
        try:
            res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': steps, 'gtol': 1e-10,
                                   'ftol': 1e-12, 'maxls': 50})
            h_new = fix_integral(res.x)
        except Exception:
            break
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            h = h_new
            best_c5 = c5_new
    
    # Second phase: direct max optimization (no smoothing)
    for T in [0.001, 0.0005]:
        if t_remaining() < 20:
            break
        def obj2(x):
            return soft_max_val(x, T) + 200.0 * (x.sum() - n / 2.0) ** 2
        
        def grad2(x):
            g, _ = soft_max_grad_val(x, T)
            g += 400.0 * (x.sum() - n / 2.0)
            return g
        
        try:
            res = minimize(obj2, h.copy(), jac=grad2, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'gtol': 1e-10,
                                   'ftol': 1e-14, 'maxls': 50})
            h_new = fix_integral(res.x)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    return h, best_c5

def make_initializations(seed, n_sizes):
    rng = np.random.RandomState(seed)
    inits = []
    for n in n_sizes:

        x = np.linspace(0, 2, n, endpoint=False)
        freqs = rng.uniform(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.1, 0.3, size=4)
        h = 0.5
        for f, ph, a in zip(freqs, phases, amps):
            h += a * np.sin(2 * np.pi * f * x / 2.0 + ph)
        h = fix_integral(h)
        inits.append(h)
        
        # More structured: periodic
        for period in [3, 4, 5]:
            h_p = 0.5 + 0.25 * np.sin(2 * np.pi * x * period / 2.0)
            h_p = fix_integral(h_p)
            inits.append(h_p + rng.normal(0, 0.02, n))
            inits[-1] = fix_integral(inits[-1])
    return inits

def upsample(h, factor=2):
    n_old = len(h)
    n_new = n_old * factor
    h_new = np.interp(np.arange(n_new) * n_old / n_new,
                     np.arange(n_old), h)
    return fix_integral(h_new)
