import numpy as np
import time

def _c5_val(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def _fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for k in range(len(idx)):
            rem = diff
            if rem > 0:
                room = 1.0 - h[idx[k]]
                adj = min(rem, room)
                h[idx[k]] += adj
                diff -= adj
            else:
                room = h[idx[k]]
                adj = min(-diff, room)
                h[idx[k]] -= adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    return h

def _sa_search(h, n, rng, deadline):
    best_h = h.copy()
    best_c5 = _c5_val(h, n)
    cur_c5 = best_c5
    temp = 0.003
    base_delta = 0.12
    iters = 0
    stall = 0

    while time.time() < deadline:
        # Adaptive step size
        delta = base_delta * np.exp(-1.5 * rng.random())
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        s = 1.0 if rng.random() < 0.5 else -1.0

        ni = h[i] + s * delta
        nj = h[j] - s * delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            delta *= 0.5
            ni = h[i] + s * delta
            nj = h[j] - s * delta
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue

        h[i] = ni
        h[j] = nj
        new_c5 = _c5_val(h, n)

        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            stall = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                temp = max(temp * 0.995, 0.0005)
        elif rng.random() < np.exp((cur_c5 - new_c5) / max(temp, 1e-12)):
            cur_c5 = new_c5
            stall += 1
        else:
            h[i] = ni - s * delta
            h[j] = nj + s * delta
            stall += 1

        if stall > 5000:
            # Restart temperature occasionally
            temp = 0.003
            base_delta = 0.2
            stall = 0
            # Large perturbation
            for _ in range(3):
                i2 = rng.integers(n)
                j2 = rng.integers(n)
                if i2 == j2:
                    continue
                d2 = rng.uniform(0.1, 0.4)
                s2 = 1.0 if rng.random() < 0.5 else -1.0
                if 0 <= h[i2] + s2 * d2 <= 1 and 0 <= h[j2] - s2 * d2 <= 1:
                    h[i2] += s2 * d2
                    h[j2] -= s2 * d2
                    nc = _c5_val(h, n)
                    if nc < cur_c5:
                        cur_c5 = nc
                        if nc < best_c5:
                            best_c5 = nc
                            best_h = h.copy()
                    else:
                        h[i2] -= s2 * d2
                        h[j2] += s2 * d2

        temp *= 0.999997
        if temp < 0.0008:
            temp = 0.0008
        iters += 1

    return best_h, best_c5, iters

def _binary_start(n, rng):
    h = np.zeros(n)
    k = n // 2
    perm = rng.permutation(n)
    h[perm[:k]] = 1.0
    if n % 2 == 1:
        h[perm[k]] = 0.5
    return h

def _structured_start(n, rng):
    # Golden-ratio based placement
    h = np.zeros(n)
    phi = (1 + 5**0.5) / 2
    k = n // 2
    positions = set()
    for m in range(k):
        pos = int((m * (1.0 / phi)) % 1 * n)
        if pos not in positions:
            positions.add(pos)
    for pos in positions:
        h[pos] = 1.0
    if n % 2 == 1 and h.sum() < n / 2.0 - 0.4:
        free = np.where(h == 0)[0]
        if len(free) > 0:
            h[free[0]] = 0.5
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 8

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 1: Optimize from initial_h_values
    ival = globals().get('initial_h_values', None)
    if ival is not None:
        n0 = len(ival)
        h0 = np.array(ival, dtype=float).copy()
        h0 = _fix_sum(h0, n0)
        h_opt, c5_opt, _ = _sa_search(h0, n0, rng, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n0

    # Phase 2: Try other n values with various starts
    n_attempts = [50, 60, 42, 66, 54, 72] if ival is None else [50, 60, 66, 42]
    for n_try in n_attempts:
        if time.time() > deadline:
            break
        for start_fn in [_binary_start, _structured_start]:
            if time.time() > deadline:
                break
            h_try = start_fn(n_try, rng)
            h_opt, c5_opt, _ = _sa_search(h_try, n_try, rng, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 3: Final polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        h_final, c5_final, _ = _sa_search(best_h.copy(), best_n, rng, deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final sum correction
    if best_h is not None:
        best_h = _fix_sum(best_h.copy(), best_n)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _fix_sum(best_h, best_n)
        final_c5 = _c5_val(best_h, best_n)
        best_c5 = final_c5

    # Cross-check with evaluate_erdos_solution if available
    try:
        ev_fn = globals().get('evaluate_erdos_solution', None)
        if ev_fn is not None:
            chk = float(ev_fn(best_h))
            if chk <= best_c5 + 1e-9:
                best_c5 = chk
    except Exception:
        pass

    return (best_h, best_c5, best_n)
