        c = eval_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(val, temp):
    m = np.max(val)
    return m + temp * np.log(np.sum(np.exp((val - m) / temp)))

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    # Iterative water-filling to satisfy sum exactly
    for _ in range(50):
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-12:
            break
        free = h > 1e-10
        free &= h < 1.0 - 1e-10
        if free.any():
            h[free] += diff / free.sum()
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_resample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def optimize_slsqp(h0, n, temp, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        return smooth_max(corr, temp)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

    h = h.copy()

                h_t[i] = trial
                h_t = fix_sum(h_t, n)
                c = eval_c5(h_t)
                if c < best_local:
                    best_local = c
                    best_val = h_t
            if best_val != h[i] or best_local < current - 1e-10:
                h = best_val
                current = best_local
                improved = True
        if not improved:
            break
    return h
