        hs = bh + rng.normal(0, scale, n)
        hs = _proj(hs, n)
        rem_r = deadline2 - time.time()
        it = max(2000, int(rem_r * 20))
        h, c5 = _anneal(hs, n, it, rng, 0.035)
        if c5 < bc:
            bh, bc = h, c5
            print(f"R{n_restarts}: C5={bc:.6f}")
        n_restarts += 1

    # Phase 3: Higher resolution upsampling
    if time.time() < t0 + budget_s * 0.88:
        xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        for nn in [63, 91, 127]:
            if time.time() > t0 + budget_s * 0.94:
                break
            xn = np.linspace(0, 2, nn, endpoint=False) + 1.0 / nn
            hn = np.interp(xn, xo, bh, period=2.0)
            hn = _proj(hn, nn)
            it = max(3000, int(2.5 * 18))
            h, c5 = _anneal(hn, nn, it, rng, 0.025)
            if c5 < bc:
                bh, bc, n = h, c5, nn
                xo = xn
                print(f"Res{nn}: C5={bc:.6f}")

    # Phase 4: Final polish on best solution
    rem = budget_s - (time.time() - t0) - 4
    if rem > 5:
        it = max(4000, int(rem * 18))
        h, c5 = _anneal(bh, n, it, rng, 0.01)
        if c5 < bc:
            bh, bc = h, c5

    # Final projection and evaluation
    bh = _proj(bh, n)
    bc = _c5(bh)[0]
    elapsed = time.time() - t0
    print(f"Done: C5={bc:.6f} n={n} t={elapsed:.1f}s")
    return bh, bc, n
