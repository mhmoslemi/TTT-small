    return float(np.max(corr)) * dx

def _soft_c5(h, dx, beta=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def _make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_optimize(h_init, n, dx, budget_s, t0):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    best_h = h_init.copy()
    best_c5 = _c5_x(best_h, dx)
    
    # Try both exact max and soft-max objectives
    for obj_type in [0, 1, 0, 1, 0]:
        if remaining() < 15:
            break
        if obj_type == 0:
            obj = lambda h: _c5_x(h, dx)
        else:
            obj = lambda h: _soft_c5(h, dx, 80.0)
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 800, 'ftol': 1e-15})
            if res.success:
                h_new = _make_feasible(res.x, n)
                c5_new = _c5_x(h_new, dx)
                if c5_new < best_c5 - 1e-15:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass
        
        # Nelder-Mead polish
        if remaining() < 15:
            break
        try:
            res2 = minimize(lambda h: _c5_x(h, dx), best_h, method='Nelder-Mead',
                           bounds=bounds, options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-15})
            if res2.success:
                h_new2 = _make_feasible(res2.x, n)
                c5_new2 = _c5_x(h_new2, dx)
                if c5_new2 < best_c5 - 1e-15:
                    best_h = h_new2
                    best_c5 = c5_new2
        except Exception:
            pass
    
    return best_h, best_c5

def _coord_descent(h, n, dx, n_iter=1000):
    """Coordinate descent: update one coordinate, fix sum via spread."""
    best_c5 = _c5_x(h, dx)
    h = h.copy()
    target_sum = n / 2.0
    
    for it in range(n_iter):
        improved = False
        j = it % n
        # Try changing h[j], compensate by adjusting all other coords evenly
        for delta in np.linspace(-0.5, 0.5, 11):
            h_try = h.copy()
            h_try[j] += delta
            if h_try[j] < 0 or h_try[j] > 1:
                continue
            # Distribute -delta over other coords
            others = np.arange(n) != j
            h_try[others] -= delta / (n - 1)
            if np.any(h_try < -1e-12) or np.any(h_try > 1.0 + 1e-12):
                continue
            h_try = np.clip(h_try, 0, 1)
            # Re-fix sum
            h_try = _make_feasible(h_try, n)
            c5_try = _c5_x(h_try, dx)
            if c5_try < best_c5 - 1e-15:
                h = h_try.copy()
                best_c5 = c5_try
                improved = True
        if not improved and it > 100:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h_init = _make_feasible(h_init, n)
    
    best_h = h_init.copy()
    best_c5 = _c5_x(best_h, dx)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP from initial
    if remaining() > 30:
        best_h, best_c5 = _slsqp_optimize(best_h, n, dx, budget_s, t0)
    
    # Phase 2: Coordinate descent polishing
    if remaining() > 10:
        best_h, best_c5 = _coord_descent(best_h, n, dx, n_iter=500)
    
    # Phase 3: Random restarts with increasing perturbation scale
    restart_count = 0
    while remaining() > 20:
        scales = [0.3, 0.2, 0.1, 0.05, 0.15, 0.25, 0.4, 0.1, 0.08, 0.12]
        scale = scales[restart_count % len(scales)]
        h_pert = best_h + np.random.normal(0, scale, n)
        h_pert = _make_feasible(h_pert, n)
        
        if _c5_x(h_pert, dx) < best_c5:
            best_h = h_pert.copy()
            best_c5 = _c5_x(best_h, dx)
        
        # SLSQP from perturbed point
        if remaining() > 25:
            try:
                bounds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(lambda h: _c5_x(h, dx), h_pert, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 500, 'ftol': 1e-15})
                if res.success:
                    h_new = _make_feasible(res.x, n)
                    c5_new = _c5_x(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
            except Exception:
                pass
        
        # Try soft-max objective too
        if remaining() > 25:
            try:
                bounds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(lambda h: _soft_c5(h, dx, 120.0), h_pert,
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 400, 'ftol': 1e-15})
                if res.success:
                    h_new = _make_feasible(res.x, n)
                    c5_new = _c5_x(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
            except Exception:
                pass
        
        restart_count += 1
        if restart_count % 5 == 0 and remaining() > 15:
            best_h, best_c5 = _coord_descent(best_h, n, dx, n_iter=200)
    
    # Final polish
    if remaining() > 5:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(lambda h: _c5_x(h, dx), best_h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 1000, 'ftol': 1e-16})
            if res.success:
                h_new = _make_feasible(res.x, n)
                c5_new = _c5_x(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass
    
    best_h = _make_feasible(best_h, n)
    best_c5 = _c5_x(best_h, dx)
    return best_h, best_c5, n
