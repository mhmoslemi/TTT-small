```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, beta):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def con_sum(h):
    return np.sum(h) - len(h) / 2.0

def con_jac(h):
    return np.ones(len(h))

def obj_b50(h):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h)), 50)

def obj_b100(h):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h)), 100)

def obj_b200(h):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h)), 200)

def obj_b500(h):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h)), 500)

def obj_exact(h):
    return compute_c5(h)

def replicate(h, mult):
    return np.repeat(h, mult)

    best_n = n

    betas = [50, 100, 200, 500, 1000]
```
