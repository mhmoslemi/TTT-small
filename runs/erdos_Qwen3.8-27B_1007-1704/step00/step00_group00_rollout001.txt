import numpy as np
from scipy.optimize import minimize
import time


def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _smooth(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    vals = beta * (corr - m)
    vals[vals < -50] = -50
    return float((m + np.log(np.sum(np.exp(vals))) / beta) * (2.0 / n))


def _fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _rand_start(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return _fix_sum(h, n / 2.0)


def _alt_start(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return _fix_sum(h, n / 2.0)


def _opt(h, n, beta, maxit):
    target = n * 0.5
    res = minimize(
        _smooth, h.copy(), args=(beta,), method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
        options={'maxiter': maxit, 'ftol': 1e-14}
    )
    return res
