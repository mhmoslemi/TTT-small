from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def enforce_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.count_nonzero(free) > 0:
        h[free] += diff / np.count_nonzero(free)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(np.minimum(h, 1.0 - h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolate_h(h0, n_new):
    n_old = len(h0)
    if n_old == n_new:
        return h0.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, np.concatenate([h0, [h0[0]]]))

def smooth_obj(x, n_pts, alpha_val, pen_w):

    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = np.max(corr)
    shifted = alpha_val * (corr - cmax)
    lse = cmax + np.log(np.sum(np.exp(shifted))) / alpha_val
    s = np.sum(x) - n_pts / 2.0
    return lse * dx + pen_w * s * s
