import numpy as np
from scipy.optimize import minimize
import time

_TRUE_N = 45

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_sum(h):
    n = len(h)
    h = h.copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        room = (1.0 - h) if diff > 0 else h.copy()
        total = room.sum()
        if total > 1e-12:
            h = h + diff * room / total
    return np.clip(h, 0.0, 1.0)

def opt_objective(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))

def con_fun(h):
    return float(np.sum(h) - _TRUE_N / 2.0)

def con_jac(h):
    return np.ones(_TRUE_N)

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_s = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_d = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_d, x_s, h_src)

def local_search(h, rng, deadline, max_sweeps=300, n_trials_per_sweep=2000):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    dvals = [0.02, 0.05, 0.1, 0.2, 0.4]
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n_trials_per_sweep):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in dvals:
                for sgn in (1.0, -1.0):
                    delta = sgn * d
                    ni = h[i] + delta
                    nj = h[j] - delta
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i] = ni
                        h[j] = nj
                        c5 = compute_c5(h)
                        if c5 < best:
                            best = c5
                            improved = True
                            break
                        h[i] -= delta
                        h[j] += delta
                if improved:
                    break
        if not improved:
            dvals = [d * 0.5 for d in dvals]
            if dvals[0] < 0.001:
                break
    return h, best

def gen_binary_start(n, rng):
    h = np.zeros(n)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:n // 2]] = 1.0
    if n % 2 == 1:
        h[idx[n // 2]] = 0.5
    return repair_sum(h)

def gen_wave_start(n, freq, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(freq * np.pi * x + phase)
    return repair_sum(h)

def run(seed=42, budget_s=1100, **kwargs):
    global _TRUE_N
    t0 = time.time()
    deadline = t0 + budget_s - 15.0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(45)
        best_h[::2] = 1.0
        best_h = repair_sum(best_h)

    best_c5 = compute_c5(best_h)

    n_list = [45, 50, 54, 60, 70, 80, 90, 100, 120, 150, 200]

    for n in n_list:
        if time.time() > deadline:
            break
        _TRUE_N = n
        cons = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
        bounds = [(0.0, 1.0)] * n

        starts = [repair_sum(interp_h(best_h, n))]
        for _ in range(3):
            starts.append(gen_binary_start(n, rng))
        starts.append(repair_sum((np.zeros(n) + 1.0) * 0.5))
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        starts.append(repair_sum(h_alt))
        for freq in (2, 3, 4):
            starts.append(gen_wave_start(n, freq, 0.0))
            starts.append(gen_wave_start(n, freq, np.pi / 2))

        for hs in starts:
            if time.time() > deadline:
                break
            try:
                res = minimize(opt_objective, hs, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-12})
                h_opt = repair_sum(res.x)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
            except Exception:
                pass

        if time.time() < deadline:
            h_ls, c5_ls = local_search(best_h, rng, deadline,
                                       max_sweeps=300, n_trials_per_sweep=3000)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

    if time.time() < deadline:
        h_final, c5_final = local_search(best_h, rng, deadline,
                                         max_sweeps=500, n_trials_per_sweep=5000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
