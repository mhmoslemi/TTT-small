    while time.time() - t0 < budget_s - 2:
        if local_mode:
            # Local coordinate search
            improved = False
            for _ in range(150):
                if time.time() - t0 > budget_s - 3:
                    break
                i = rng.randint(n)
                for delta in [0.01, -0.01, 0.03, -0.03, 0.07, -0.07, 0.12, -0.12]:
                    h = cur.copy()
                    h = _nudge(h, i, delta, tgt, rng)
                    c5_t = _c5(h, dx)
                    if c5_t < cur_c5 - 1e-14:
                        cur = h; cur_c5 = c5_t
                        if c5_t < best_c5:
                            best = cur.copy(); best_c5 = c5_t
                        improved = True
                        break
                if not improved:
                    break
            n_local += 1

            if improved:
                local_mode = False
                T = min(T * 1.02, T_max)
            else:
                # Perturb and restart SA
                cur = best.copy()
                nflip = rng.randint(3, 10)
                for _ in range(nflip):
                    i, j = rng.choice(n, 2, replace=False)
                    cur[i], cur[j] = cur[j], cur[i]
                cur = _fix_sum(cur, tgt, rng)
                cur_c5 = _c5(cur, dx)
                local_mode = False
                T = T_max * rng.uniform(0.3, 1.0)
                stagnant = 0
        else:
            # SA move
            mt = rng.randint(7)
            h = cur.copy()

            if mt == 0:
                i, j = rng.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            elif mt == 1:
                h = _nudge(h, rng.randint(n), rng.uniform(-0.12, 0.12), tgt, rng)
            elif mt == 2:
                b = rng.randint(2, max(3, n // 5))
                i = rng.randint(max(1, n - b))
                j = rng.randint(max(1, n - b))
                if abs(i - j) >= b:
                    h[i:i+b], h[j:j+b] = h[j:j+b].copy(), h[i:i+b].copy()
            elif mt == 3:
                i, j = rng.choice(n, 2, replace=False)
                d = rng.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i] + d, 0, 1)
                h[j] = np.clip(h[j] - d, 0, 1)
                h = _fix_sum(h, tgt, rng)
            elif mt == 4:
                b = rng.randint(2, max(3, n // 4))
                i = rng.randint(max(1, n - b))
                h[i:i+b] = np.roll(h[i:i+b], rng.randint(1, b))
            elif mt == 5:
                h = _nudge(h, rng.randint(n), rng.choice([-1, 1]) * rng.uniform(0.2, 0.6), tgt, rng)
            else:
                k = rng.randint(2, 5)
                idx = rng.choice(n, k, replace=False)
                deltas = rng.uniform(-0.08, 0.08, k)
                for p in range(k):
                    h[idx[p]] = np.clip(h[idx[p]] + deltas[p], 0, 1)
                h = _fix_sum(h, tgt, rng)

            c5_new = _c5(h, dx)

            if c5_new < cur_c5:
                cur = h; cur_c5 = c5_new
                stagnant = 0
                if c5_new < best_c5:
                    best = cur.copy(); best_c5 = c5_new
                    T = min(T * 1.03, T_max)
            else:
                stagnant += 1
                if rng.random() < math.exp(-(c5_new - cur_c5) / T):
                    cur = h; cur_c5 = c5_new
                    stagnant = 0

            T = max(T * alpha, T_min)

            if stagnant > 25000 and not local_mode:
                local_mode = True

    # Final verification
    best = np.clip(best, 0.0, 1.0)
    best = _fix_sum(best, tgt, rng)
    final_c5 = _c5(best, dx)

    return best, final_c5, n
