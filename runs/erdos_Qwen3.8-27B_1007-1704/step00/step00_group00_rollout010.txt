import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def _ineq_fun(n):
    def f(x):
        h = x[:-1]
        t = x[-1]
        return t - c5_vec(h)
    return f

def _eq_fun(n):
    def f(x):
        return np.sum(x[:-1]) - n / 2.0
    return f

def _obj(x, n):
    return x[-1]

def _fix_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_init, n, max_iter=300):
    x0 = np.concatenate([h_init.copy(), [compute_c5(h_init) + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun(n)},
        {'type': 'eq', 'fun': _eq_fun(n)},
    ]
    res = minimize(_obj, x0, args=(n,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = _fix_sum(res.x[:n].copy(), n)
    return h_new, compute_c5(h_new)

def pair_search(h_init, n, max_iters, rng, t_check):
