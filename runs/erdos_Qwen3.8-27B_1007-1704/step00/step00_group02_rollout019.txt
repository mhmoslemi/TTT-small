def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    """Adjust h to have the correct sum while staying in [0,1]."""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Distribute to elements with room
    if diff > 0:
        # Need to increase: distribute to elements < 1
        room = 1.0 - h
        mask = room > 1e-10
        if not np.any(mask):
            return h
        total_room = np.sum(room[mask])
        if total_room < abs(diff) - 1e-10:
            return h
        h[mask] += diff * room[mask] / total_room
    else:
        # Need to decrease: distribute to elements > 0
        room = h
        mask = room > 1e-10
        if not np.any(mask):
            return h
        total_room = np.sum(room[mask])
        if total_room < abs(diff) - 1e-10:
            return h
        h[mask] += diff * room[mask] / total_room
