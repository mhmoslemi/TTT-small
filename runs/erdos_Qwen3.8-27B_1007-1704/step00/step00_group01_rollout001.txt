    m = np.max(corr)
    shifted = corr - m
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def num_gradient(x, dx, beta):
    n = len(x)
    grad = np.zeros(n)
    eps = 5e-7
    fc = smooth_max_c5(x, dx, beta)
    for i in range(n):
        if x[i] <= 1e-12 or x[i] >= 1.0 - 1e-12:
            continue
        xp = x.copy()
        xp[i] += eps
        grad[i] = (smooth_max_c5(xp, dx, beta) - fc) / eps
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(47, 0.5)

    n_init = len(h_init)
    h_cur = project_feasible(h_init.copy(), n_init / 2.0)
    best_c5 = compute_c5(h_cur, 2.0 / n_init)
    best_h = h_cur.copy()
    best_n = n_init

    def elapsed_ok(frac=0.82):
        return (time.time() - t0) < budget_s * frac

    # Build grid schedule
    grids = [n_init]
    n_mult = n_init
    for _ in range(6):
        n_mult *= 2
        if n_mult > 4096 or not elapsed_ok(0.75):
            break
        grids.append(n_mult)

    for n_g in grids:
        if not elapsed_ok():
            break
        dx = 2.0 / n_g
        target = n_g / 2.0

        if n_g > len(h_cur):
            h_cur = interp_h(h_cur, n_g)
            h_cur = project_feasible(h_cur, target)

        # SLSQP with log-sum-exp smoothed max
        betas = [8, 25, 80, 250]
        for beta in betas:
            if not elapsed_ok():
                break
            # Scale maxiter with available time
            remaining = budget_s * 0.82 - (time.time() - t0)
            per_iter_cost = n_g * 0.0004  # rough estimate in seconds
            max_iter = int(remaining / max(per_iter_cost, 0.01))
            max_iter = max(30, min(max_iter, 250))

            cons = [{'type': 'eq',
                     'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(x.shape[0])}]
            bounds = [(0.0, 1.0)] * n_g

            def obj_fn(x):
                return smooth_max_c5(x, dx, beta)

            def grad_fn(x):
                return num_gradient(x, dx, beta)

            try:
                res = so.minimize(obj_fn, h_cur, jac=grad_fn, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': max_iter, 'ftol': 1e-12,
                                           'disp': False})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_g
                h_cur = h_new
            except Exception:
                pass

        # Nelder-Mead local refinement (direct C5, no smoothing)
        if not elapsed_ok(0.7):
            break
        for _ in range(2):
            if not elapsed_ok(0.75):
                break
            def nm_obj(x):
                h = project_feasible(x, target)
                return compute_c5(h, dx)
            try:
                res = so.minimize(nm_obj, h_cur, method='Nelder-Mead',
                                  options={'maxiter': 400, 'fatol': 1e-10,
                                           'xatol': 1e-9, 'adaptive': True})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_g
                h_cur = h_new
            except Exception:
                pass

        # Stochastic local search
        if not elapsed_ok(0.75):
            break
        for _ in range(8):
            if not elapsed_ok(0.8):
                break
            step = 0.02 / math.sqrt(n_g / n_init)
            h_pert = h_cur + rng.randn(n_g) * step
            h_pert = project_feasible(h_pert, target)
            c5_pert = compute_c5(h_pert, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n_g
                h_cur = h_pert

    # Final validation
    dx_final = 2.0 / best_n
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, dx_final)

    # Quick polish at best resolution if time remains
    if (time.time() - t0) < budget_s * 0.85:
        dx_p = 2.0 / best_n
        target_p = best_n / 2.0
        for beta in [100, 300]:
            if not elapsed_ok(0.9):
                break
            def po(x):
                return smooth_max_c5(x, dx_p, beta)
            def pg(x):
                return num_gradient(x, dx_p, beta)
            cons_p = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_p,
                       'jac': lambda x: np.ones(x.shape[0])}]
            bounds_p = [(0.0, 1.0)] * best_n
            try:
                res = so.minimize(po, best_h, jac=pg, method='SLSQP',
                                  bounds=bounds_p, constraints=cons_p,
                                  options={'maxiter': 80, 'ftol': 1e-12})
                hh = project_feasible(res.x, target_p)
                cc = compute_c5(hh, dx_p)
                if cc < final_c5:
                    final_c5 = cc
                    best_h = hh
            except Exception:
                pass

    return (best_h, final_c5, best_n)
