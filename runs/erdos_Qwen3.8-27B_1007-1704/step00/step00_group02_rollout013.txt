    for temp in [0.04, 0.015, 0.006]:
        if time.time() > budget_end - 60:
            break
        for penalty in [1000.0, 10000.0]:
            if time.time() > budget_end - 40:
                break
            try:
                res = minimize(
                    penalty_obj, best_h.copy(),
                    args=(dx, temp, penalty, n),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': 800, 'maxfun': 40000, 'ftol': 1e-13}
                )
                h_new = fix_sum(res.x, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h_new.copy()
            except Exception:
                pass
    # Local search
    if time.time() < budget_end - 20:
        h_loc = best_h.copy()
        c5_loc = best_c5
        step = 0.025
        iters = min(4000, max(200, int((budget_end - 20 - time.time()) * 80)))
        for it in range(iters):
            if time.time() > budget_end - 15:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (i + 1) % n
            delta = rng.randn() * step
            oi, oj = h_loc[i], h_loc[j]
            h_loc[i] = np.clip(oi + delta, 0.0, 1.0)
            h_loc[j] = np.clip(oj - delta, 0.0, 1.0)
            h_loc = fix_sum(h_loc, n)
            c5_t = compute_c5(h_loc, dx)
            if c5_t < c5_loc - 1e-14:
                c5_loc = c5_t
                best_c5, best_h = c5_t, h_loc.copy()
            else:
                h_loc[i], h_loc[j] = oi, oj
                h_loc = fix_sum(h_loc, n)
            if it % 800 == 799:
                step *= 0.85
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    budget_end = t0 + budget_s
    best_c5 = np.inf
    best_h = None
    best_n = 0

    try:
        h_base = np.array(initial_h_values, dtype=float)
    except NameError:
        h_base = None

    ns = []
    if h_base is not None:
        ns.append(len(h_base))
    for cn in [45, 64, 90, 128, 160]:
        if cn not in ns:
            ns.append(cn)

    time_slice = (budget_s - 90) / len(ns)

    for n in ns:
        if time.time() > budget_end - 50:
            break
        sub_end = min(time.time() + time_slice, budget_end - 40)
        dx = 2.0 / n

        # Build candidate initializations
        candidates = []
        if h_base is not None:
            candidates.append(fix_sum(resample_h(h_base, n), n))
        for h_init in make_inits(n, seed + n):
            candidates.append(fix_sum(h_init, n))

        for h0 in candidates:
            if time.time() > sub_end - 30:
                break
            try:
                h_opt, c5_opt = optimize_one(h0, n, dx, sub_end, rng)
            except Exception:
                continue
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    if best_h is not None:
        best_h = fix_sum(best_h, best_n)
        best_c5 = compute_c5(best_h, 2.0 / best_n)

    return (best_h, best_c5, best_n)
