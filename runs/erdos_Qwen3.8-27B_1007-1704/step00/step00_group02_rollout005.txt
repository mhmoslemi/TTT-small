def _eval(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _proj(h, s):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > s:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = s - hp.sum()
    if abs(d) > 1e-9:
        m = (hp > 1e-7) & (hp < 1.0 - 1e-7)
        if m.sum() > 0:
            hp[m] += d / m.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _subgrad(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    km = int(c.argmax())
    thr = c[km] - 1e-3 * max(float(c.max()), 1.0)
    active = np.where(c >= thr)[0]
    g = np.zeros(n)

        lo_i = max(0, -sh)
        hi_i = min(n, n - sh)
        for i in range(lo_i, hi_i):
            j = i + sh
            g[i] += 1.0 - h[j]
            g[j] -= h[i]
    g /= float(len(active))
    return g, float(c[km] * dx)
