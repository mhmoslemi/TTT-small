def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def local_search_annotate(h, max_iter, temperature=0.0, rng=None):
    """Local search with optional simulated annealing."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    T = temperature
    
    for it in range(max_iter):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or abs(h[i] - h[j]) < 1e-12:
            continue
        # Move mass from higher to lower
        if h[i] > h[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_move = h[src] - h[dst]
        if max_move < 1e-12:
            continue
        delta = rng.uniform(0.05, 1.0) * max_move
        h[src] -= delta
        h[dst] += delta
        c5_new = compute_c5(h)
        if c5_new < best_c5 or (T > 0 and rng.random() < np.exp(-(c5_new - best_c5) / T)):
            best_c5 = min(best_c5, c5_new)
            # Cool

                T *= 0.999
        else:
            h[src] += delta
            h[dst] -= delta
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, max_iter, temp=0.0, rng=None):
    n = len(h)
    h = h.copy().astype(float)
    best = compute_c5(h)
    T = temp
    for _ in range(max_iter):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] > h[j]:
            s, d = i, j
        else:
            s, d = j, i
        mv = h[s] - h[d]
        if mv < 1e-12:
            continue
        delta = rng.uniform(0.05, 1.0) * mv
        h[s] -= delta
        h[d] += delta
        new_val = compute_c5(h)
        if new_val < best or (T > 1e-9 and rng.random() < np.exp(-(new_val - best) / T)):
            best = min(best, new_val)
            T *= 0.999
        else:
            h[s] += delta
            h[d] -= delta
    return h, best

def local_search_neighbors(h, max_iter, temp=0.0, rng=None):
    n = len(h)
    h = h.copy().astype(float)
    best = compute_c5(h)
    T = temp
    shifts = list(range(1, min(n // 2 + 1, 12)))
    for _ in range(max_iter):
        i = rng.integers(n)
        sh = shifts[rng.integers(len(shifts))]
        j = (i + sh) % n
        if i == j:
            continue
        if h[i] > h[j]:
            s, d = i, j
        else:
            s, d = j, i
        mv = h[s] - h[d]
        if mv < 1e-12:
            continue
        delta = rng.uniform(0.1, 1.0) * mv
        h[s] -= delta
        h[d] += delta
        new_val = compute_c5(h)
        if new_val < best or (T > 1e-9 and rng.random() < np.exp(-(new_val - best) / T)):
            best = min(best, new_val)
            T *= 0.999
        else:
            h[s] += delta
            h[d] -= delta
    return h, best

def init_pattern(n, ptype, rng):
    h = np.zeros(n)
    if ptype == 0:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif ptype == 1:
        tm = np.array([0])
        while len(tm) < n:
            tm = np.concatenate([tm, 1 - tm[:min(len(tm), n - len(tm))]])
        h = tm[:n].astype(float)
    elif ptype == 2:
        bs = max(2, n // 6)
        for k in range(0, n, bs):
            h[k:min(k + bs, n)] = (k // bs) % 2
    elif ptype == 3:
        # Periodic with prime period
        p = 7
        for i in range(n):
            h[i] = 1.0 if (i % p) < (p + 1) // 2 else 0.0
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    return fix_sum(h, n)
