import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def c5_along_line(h, i, j, delta):
    """C5 when h[i]+=delta, h[j]-=delta"""
    new_h = h.copy()
    new_h[i] = h[i] + delta
    new_h[j] = h[j] - delta
    if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
        return 1e10
    return compute_c5(new_h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(45, 0.5)
    
    n = len(h_best)
    target_sum = n / 2.0
    c5_best = compute_c5(h_best)
    
    elapsed = lambda: time.time() - t0
    
    # Line search along pair direction
    def optimize_pair(h, i, j):
        """Find best delta for h[i]+=delta, h[j]-=delta"""
        # Bounds: h[i]+delta in [0,1] and h[j]-delta in [0,1]
        lo = max(-h[i], h[j]-1)
        hi = min(1-h[i], h[j])
        if lo >= hi:
            return h, compute_c5(h)
        
        result = minimize_scalar(c5_along_line, bounds=(lo, hi), args=(h, i, j), 
                                  method='bounded', options={'xatol': 1e-5})
        if result.success:
            new_h = h.copy()
            new_h[i] += result.x
            new_h[j] -= result.x
            return new_h, result.fun
        return h, compute_c5(h)
    
    # Main optimization loop
    for outer in range(100):
        if elapsed() > budget_s - 20:
            break
        
        # Multiple passes of pair optimization
        for pass_num in range(3):

        return 1e10
    return compute_c5_fast(nh)

def pair_optimize(h, c5, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-8:
        return h, c5
    try:
        res = minimize_scalar(_c5_line, bounds=(lo, hi), args=(h, i, j),
                              method='bounded', options={'xatol': 1e-5, 'maxiter': 20})
        if res.fun < c5:
            h[i] += res.x
            h[j] -= res.x
            return h, float(res.fun)
    except Exception:
        pass
    return h, c5

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-11:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def full_sweep(h, c5, n_pairs, t0, budget):
    n = len(h)
    idx = np.arange(n * n)
    np.random.shuffle(idx)
    count = 0
    for flat_i in idx:
        if time.time() - t0 > budget - 5:
            break
        i = int(flat_i) // n
        j = int(flat_i) % n
        if i == j:
            continue
        h, c5 = pair_optimize(h, c5, i, j)
        count += 1
        if count >= n_pairs:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(45, 0.5)

    n = len(h_best)
    target = n / 2.0
    c5_best = compute_c5_fast(h_best)

    # Phase 1: Intensive local optimization of initial solution
    p1_end = t0 + budget_s * 0.35
    while time.time() < p1_end:
        h_best, c5_best = full_sweep(h_best, c5_best, n_pairs=1980, t0=t0, budget=budget_s)

    # Phase 2: Multi-restart exploration
    p2_end = t0 + budget_s * 0.85
    n_stall = 0
    while time.time() < p2_end:
        scale = min(0.4, 0.06 + 0.04 * n_stall)
        h_pert = h_best + np.random.randn(n) * scale
        h_pert = fix_sum(h_pert, target)
        h_cand = h_pert.copy()
        c5_cand = compute_c5_fast(h_cand)
        for _ in range(4):
            if time.time() > p2_end:
                break
            h_cand, c5_cand = full_sweep(h_cand, c5_cand, n_pairs=200, t0=t0, budget=budget_s)
        if c5_cand < c5_best - 1e-9:
            h_best = h_cand
            c5_best = c5_cand
            n_stall = 0
        else:
            n_stall += 1
        if n_stall % 8 == 7:
            h_alt = np.clip(np.random.rand(n) * 1.2 + 0.1, 0, 1)
            h_alt = fix_sum(h_alt, target)
            h_alt2 = h_alt.copy()
            c5_alt = compute_c5_fast(h_alt2)
            for _ in range(3):
                if time.time() > p2_end:
                    break
                h_alt2, c5_alt = full_sweep(h_alt2, c5_alt, n_pairs=250, t0=t0, budget=budget_s)
            if c5_alt < c5_best:
                h_best = h_alt2
                c5_best = c5_alt
                n_stall = 0
        if n_stall % 15 == 14:
            h_struct = np.zeros(n)
            ones_pos = np.random.choice(n, n // 2, replace=False)
            h_struct[ones_pos] = 1.0
            h_struct[np.random.randint(n)] += 0.5
            h_struct = fix_sum(h_struct, target)
            h_alt2 = h_struct.copy()
            c5_alt = compute_c5_fast(h_alt2)
            for _ in range(3):
                if time.time() > p2_end:
                    break
                h_alt2, c5_alt = full_sweep(h_alt2, c5_alt, n_pairs=250, t0=t0, budget=budget_s)
            if c5_alt < c5_best:
                h_best = h_alt2
                c5_best = c5_alt
                n_stall = 0

    # Phase 3: Final polish
    while time.time() < t0 + budget_s - 3:
        h_best, c5_best = full_sweep(h_best, c5_best, n_pairs=1980, t0=t0, budget=budget_s)

    h_best = fix_sum(h_best, target)
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = compute_c5_fast(h_best)
    return h_best, c5_final, n
