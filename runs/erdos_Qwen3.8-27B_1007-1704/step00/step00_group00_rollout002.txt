def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, alpha):
    m = np.max(x)
    if not np.isfinite(m):
        return 0.0
    shifted = np.clip(x - m, -60.0, 60.0)
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    h = np.clip(p, 0.0, 1.0)
    h += (target_sum - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    return h
