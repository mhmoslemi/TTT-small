import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def soft_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    scaled = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(scaled))) / beta

def scale_to_constraint(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            return np.clip(h, 0.0, 1.0)
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def optimize_at_n(h_init, n, time_limit):
    t0 = time.time()
    h = scale_to_constraint(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    for beta in [15, 40, 100]:
        if time.time() - t0 > time_limit:
            break
        for _ in range(2):
            if time.time() - t0 > time_limit:
                break
            try:
                res = minimize(soft_obj, h, args=(beta,), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-13})
                if res.x is not None:
                    hn = np.clip(res.x, 0, 1)
                    hn = scale_to_constraint(hn, n)
                    c5v = compute_c5(hn)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = hn.copy()
                        h = hn.copy()
            except Exception:
                pass
    # Final polish with high beta
    if time.time() - t0 < time_limit - 5:
        try:
            res = minimize(soft_obj, best_h, args=(300.0,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-14})
            if res.x is not None:
                hn = np.clip(res.x, 0, 1)
                hn = scale_to_constraint(hn, n)
                c5v = compute_c5(hn)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hn.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.full(47, 0.5)

    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    # Phase 1: Optimize at parent resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        h_opt, c5_opt = optimize_at_n(h_parent, n_parent, min(90, budget_s - elapsed - 10))
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_parent

    # Phase 2: Try higher resolutions via interpolation
    n_tries = sorted(set([n_parent, n_parent * 2, n_parent * 4, n_parent * 8, 200, 400]))
    n_tries = [n for n in n_tries if n < 2000]
    for n_try in n_tries:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 80:
            break
        if n_try <= best_n and best_c5 < 0.381:
            continue
        if n_try == best_n:
            h_init = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_try)
            h_init = np.interp(x_new, x_old, best_h)
        h_init = scale_to_constraint(h_init, n_try)
        time_left = budget_s - elapsed - 30
        time_per = min(time_left * 0.4, 150)
        h_opt, c5_opt = optimize_at_n(h_init, n_try, time_per)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_try

    # Phase 3: Random restarts at best resolution
    n_best = len(best_h)
    for i in range(40):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 12:
            break
        if i % 4 == 0:
            h_r = best_h + np.random.randn(n_best) * 0.05
        elif i % 4 == 1:
            h_r = best_h.copy()
            idx = np.random.choice(n_best, size=max(1, n_best // 5), replace=False)
            h_r[idx] = np.random.uniform(0, 1, size=len(idx))
        elif i % 4 == 2:
            h_r = best_h.copy()
            shift = np.random.randint(1, max(2, n_best // 4))
            h_r = np.roll(best_h, shift)
        else:
            h_r = 0.6 * best_h + 0.4 * (1 - np.roll(best_h, n_best // 3))
        h_r = scale_to_constraint(h_r, n_best)
        time_left = budget_s - elapsed - 5
        t_budget = min(max(5, time_left / max(1, 40 - i)), 45)
        h_opt, c5_opt = optimize_at_n(h_r, n_best, t_budget)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

    # Final verification
    best_h = scale_to_constraint(best_h, len(best_h))
    best_c5_final = compute_c5(best_h)
    return best_h, best_c5_final, len(best_h)
