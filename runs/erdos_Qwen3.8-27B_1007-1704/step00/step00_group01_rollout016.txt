from scipy.optimize import minimize

def compute_correlation(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_correlation(h)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_single(h, idx):
    """Gradient of correlation[idx] w.r.t. h, multiplied by dx"""
    n = len(h)
    dx = 2.0 / n
    lag = idx - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        m = lag
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
        g *= dx
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
        g *= dx
    return g

def subgradient_step(h, n, step, top_k=5):
    c_arr = compute_correlation(h)
    # Get top k indices
    top_idx = np.argsort(c_arr)[-top_k:]
    g = np.zeros(n)
    for idx in top_idx:
        g += grad_single(h, idx)
    g /= top_k
    h_new = project_feasible(h - step * g, n)
    return h_new, c_arr

                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except:
