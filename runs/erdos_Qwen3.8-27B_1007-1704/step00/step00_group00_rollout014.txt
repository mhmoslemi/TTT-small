    deadline = t0 + budget_s * 0.95

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        h0 = None

    if h0 is None or len(h0) == 0:
        n = 45
        h0 = np.ones(n) * 0.5
    else:
        n = len(h0)

    dx = 2.0 / n
    target_sum = n / 2.0

    h_init = normalize(h0, target_sum)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)

    # Build diverse starting points
    starts = [h_init.copy()]
    for _ in range(4):
        hs = h_init + rng.normal(0, 0.08, n)
        starts.append(normalize(hs, target_sum))
    # Block pattern
    hb = np.zeros(n)
    n_blocks = 5
    per_block = int(target_sum / n_blocks)
    extra = int(target_sum) - per_block * n_blocks
    pos = np.linspace(0, n, n_blocks + 1).astype(int)
    for k in range(n_blocks):
        length = per_block + (1 if k < extra else 0)
        hb[pos[k]:pos[k] + length] = 1.0
    starts.append(normalize(hb, target_sum))

    # Phase 1: multi-start (30% time)
    phase1_end = t0 + (deadline - t0) * 0.3
    seg_deadline = max(phase1_end, t0 + 10)
    per_start = (phase1_end - t0) / len(starts)
    for hs in starts:
        sl = max(t0 + 2, t0 + per_start * (starts.index(hs) + 1) + 2)
        best_h, best_c5 = local_search(hs.copy(), dx, target_sum, best_h, best_c5,
                                        min(sl, phase1_end), n, rng)

    # Phase 2: intensive search from best (70% time)
    h = best_h.copy()
    c5 = best_c5
    alpha = 0.02
    beta = 0.003
    streak = 0

    while time.time() < deadline:
        progress = min(1.0, (time.time() - t0) / max(1e-6, deadline - t0))
        alpha = 0.02 * (1.0 - 0.9 * progress) + 0.001
        beta = 0.003 * (1.0 - progress) + 0.0003
        mt = rng.random()

        if mt < 0.85:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            delta = rng.uniform(-alpha, alpha)
            if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta
            h[j] -= delta
            c5_new = compute_c5(h, dx)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                streak = 0
            else:
                if rng.random() < np.exp(-(c5_new - c5) / max(beta, 1e-8)):
                    c5 = c5_new
                else:
                    h[i] -= delta
                    h[j] += delta
                streak += 1
        elif mt < 0.97:
            # Triple move
            idx = rng.choice(n, 3, replace=False)
            a, b, c = int(idx[0]), int(idx[1]), int(idx[2])
            da = rng.uniform(-alpha, alpha)
            db = rng.uniform(-alpha, alpha)
            dc = -(da + db)
            ha, hb2, hc = h[a] + da, h[b] + db, h[c] + dc
            if 0 <= ha <= 1 and 0 <= hb2 <= 1 and 0 <= hc <= 1:
                h[a], h[b], h[c] = ha, hb2, hc
                c5_new = compute_c5(h, dx)
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    streak = 0
                else:
                    if rng.random() < np.exp(-(c5_new - c5) / max(beta, 1e-8)):
                        c5 = c5_new
                    else:
                        h[a], h[b], h[c] = h[a] - da, h[b] - db, h[c] - dc
                    streak += 1
            else:
                streak += 1
        else:
            # Jump perturbation
            js = rng.uniform(0.04, 0.12)
            ii = rng.randint(0, n)
            jj = rng.randint(0, n)
            if ii == jj:
                continue
            h[ii] += js
            h[jj] -= js
            h = normalize(h, target_sum)
            c5 = compute_c5(h, dx)
            streak = 0

        if streak > 100000:
            h = best_h + rng.normal(0, 0.015, n)
            h = normalize(h, target_sum)
            c5 = compute_c5(h, dx)
            streak = 0

    # Final verification
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
