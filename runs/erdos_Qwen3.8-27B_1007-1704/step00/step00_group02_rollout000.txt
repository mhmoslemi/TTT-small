            idx = rng.choice(np.arange(lo, hi), size=n_p, replace=False)
            h[idx] += rng.normal(0.0, sigma * 1.2, size=n_p)
            h = repair(h, n)
        
        c_new = compute_c5(h)
        if c_new < c - 1e-10:
            c = c_new
            stagnation = 0
            sigma = max(0.01, sigma * 0.97)
        elif c_new < c + sigma * 0.15 and rng.random() < 0.08:
            c = c_new  # accept slight worsening (annealing)
            stagnation = 0
        else:
            h = best.copy()
            stagnation += 1
            if stagnation > 300:
                sigma = min(0.35, sigma * 1.12)
                stagnation = 0
                # Escape: large random jump
                n_p = int(rng.integers(5, min(20, n)))
                idx = rng.choice(n, size=n_p, replace=False)
                h[idx] += rng.normal(0.0, sigma * 2.5, size=n_p)
                h = repair(h, n)
                c = compute_c5(h)
        
        if c < best_c:
            best_c = c
            best = h.copy()
    
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    def tl():
        return max(0.0, budget_s - (time.time() - t_start))
    
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    def update(h, n):
        nonlocal best_h, best_c5, best_n
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n
        return c
    
    # Phase 1: Optimize initial construction
    if h_init is not None and len(h_init) >= 10:
        n0 = len(h_init)
        h0 = repair(h_init, n0)
        update(h0, n0)
        t_alloc = min(tl() * 0.28, 280)
        h_opt, c_opt = local_search(h0, n0, t_alloc, rng)
        update(h_opt, n0)
        print(f"Phase1 n={n0}: C5={c_opt:.6f} (left={tl():.0f}s)")
    
    # Phase 2: Explore other resolutions
    n_cands = []
    if best_n:
        n_cands.append(best_n)
    if h_init is not None:
        n_cands.append(len(h_init))
    n_cands += [24, 30, 36, 42, 48, 54, 60, 72, 84, 96, 108, 120, 144, 168]
    seen = set()
    n_list = []
    for nn in n_cands:
        if nn >= 12 and nn <= 300 and nn not in seen:
            seen.add(nn)
            n_list.append(nn)
    
    for n in n_list:
        if tl() < 90:
            break
        if best_h is not None:
            ref = best_h
        elif h_init is not None:
            ref = h_init
        else:
            ref = np.full(n, 0.5)
        if n == len(ref):
            h_start = ref.copy()
        else:
            x_old = np.linspace(0, 2, len(ref), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_start = np.interp(x_new, x_old, ref)
        h_start = repair(h_start, n)
        
        t_alloc = min(tl() * 0.30, 180)
        h_opt, c_opt = local_search(h_start, n, t_alloc, rng)
        update(h_opt, n)
        print(f"Phase2 n={n}: C5={c_opt:.6f} (left={tl():.0f}s)")
    
    # Phase 3: Multi-start at best resolution
    if best_h is not None and tl() > 60:
        for restart in range(4):
            if tl() < 40:
                break
            h_start = best_h.copy()
            # Add random noise
            n_p = int(best_n * 0.15)
            idx = rng.choice(best_n, size=n_p, replace=False)
            h_start[idx] += rng.normal(0.0, 0.05, size=n_p)
            h_start = repair(h_start, best_n)
            t_alloc = min(tl() * 0.25, 120)
            h_opt, c_opt = local_search(h_start, best_n, t_alloc, rng)
            update(h_opt, best_n)
    
    # Phase 4: Final polish with small steps
    if best_h is not None and tl() > 30:
        t_alloc = min(tl() * 0.85, 150)
        h_opt, c_opt = local_search(best_h, best_n, t_alloc, rng)
        if c_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c_opt, best_n
    
    # Final validation
    best_h = repair(best_h, best_n)
    final_c5 = compute_c5(best_h)
    print(f"Final: n={best_n}, C5={final_c5:.6f}")
    return best_h, final_c5, best_n
