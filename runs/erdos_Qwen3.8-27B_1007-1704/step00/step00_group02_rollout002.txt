import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def push_binary(h, n, strength=0.3):
    h = h.copy()
    mask_up = h > 0.5
    mask_dn = h <= 0.5
    h[mask_up] = np.clip(h[mask_up] + strength * (1.0 - h[mask_up]), 0, 1)
    h[mask_dn] = np.clip(h[mask_dn] - strength * h[mask_dn], 0, 1)
    return repair(h, n)

def sa_optimize(h0, n, budget_s, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n)
    curr_c5 = best_c5
    start = time.time()
    t = 0.05
    t_min = 1e-9
    alpha = 0.9997
    while time.time() - start < budget_s and t > t_min:
        h_new = h.copy()
        n_pert = max(1, int(rng.randint(1, n // 8 + 1)))
        for _ in range(n_pert):
            i = int(rng.randint(0, n))
            h_new[i] += rng.normal(0, 0.15)
        h_new = np.clip(h_new, 0, 1)
        h_new = repair(h_new, n)
        c5_new = evaluate_c5(h_new, n)
        if c5_new <= curr_c5:
            h = h_new
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(c5_new - curr_c5) / t):
                h = h_new
                curr_c5 = c5_new
        t *= alpha
    return best_h, best_c5

def coord_descent(h0, n, budget_s, rng):
    h = h0.copy()
    best_c5 = evaluate_c5(h, n)
    start = time.time()
    for _ in range(30):
        if time.time() - start > budget_s:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > budget_s:
                break
            vals = [0.0, 1.0, 0.5, h[i] - 0.3, h[i] - 0.1, h[i] + 0.1, h[i] + 0.3]
            for v in vals:
                v = max(0.0, min(1.0, v))
                h_t = h.copy()
                h_t[i] = v
                h_t = repair(h_t, n)
                c5_t = evaluate_c5(h_t, n)
                if c5_t < best_c5 - 1e-12:
                    h = h_t
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def resample_to(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return repair(h_new, n_new)

def make_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return repair(h, n)

def make_quadratic(n, rng):
    x = (np.arange(n) + 0.5) / n
    h = (0.5 + 0.5 * np.sin(2 * np.pi * 2.0 * x + rng.uniform(0, 6.28))).clip(0, 1)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        n0 = len(parent)
    else:
        n0 = 45
        parent = make_binary(n0, rng)

    best_h = parent.copy()
    best_c5 = evaluate_c5(best_h, n0)

    # Phase 1: SA on parent
    if remaining() > 30:
        b = min(150, remaining() * 0.35)
        h_s, c5_s = sa_optimize(best_h, n0, b, rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 2: Coordinate descent
    if remaining() > 20:
        b = min(60, remaining() * 0.2)
        h_c, c5_c = coord_descent(best_h, n0, b, rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase 3: Binary push + refine
    if remaining() > 20:
        h_b = push_binary(best_h, n0, 0.4)
        c5_b = evaluate_c5(h_b, n0)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        if remaining() > 15:
            h_s, c5_s = sa_optimize(best_h, n0, min(40, remaining() * 0.2), rng)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 4: Higher resolutions
    for n_try in [max(90, n0 * 2), max(180, n0 * 4), max(360, n0 * 8)]:
        if remaining() < 25:
            break
        if n_try > 900:
            break
        candidates = [resample_to(best_h, len(best_h), n_try)]
        if remaining() > 50:
            candidates.append(make_binary(n_try, rng))
        if remaining() > 50:
            candidates.append(make_quadratic(n_try, rng))
        for h_cand in candidates:
            if remaining() < 15:
                break
            c5_init = evaluate_c5(h_cand, n_try)
            if c5_init >= best_c5 + 0.02:
                continue
            b_sa = min(80, remaining() * 0.3)
            h_s, c5_s = sa_optimize(h_cand, n_try, b_sa, rng)
            if remaining() > 10:
                h_cd, c5_cd = coord_descent(h_s, n_try, min(20, remaining() * 0.15), rng)
                if c5_cd < c5_s:
                    h_s, c5_s = h_cd, c5_cd
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 5: Final polish at best resolution
    if remaining() > 8:
        nb = len(best_h)
        b_f = min(remaining() * 0.4, 60)
        h_f, c5_f = sa_optimize(best_h, nb, b_f, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = repair(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h, len(best_h))
    best_c5 = min(best_c5, final_c5)
    return best_h, best_c5, len(best_h)
