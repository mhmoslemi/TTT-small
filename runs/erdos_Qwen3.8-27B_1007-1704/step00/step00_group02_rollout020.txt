import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate, convolve
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def project_onto_feasible(h, target):
    def sum_at_shift(s):
        return np.sum(np.clip(h + s, 0.0, 1.0))
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_at_shift(mid) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', bounds_error=False, fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0, 1)
    return project_onto_feasible(h_new, n_new / 2.0)

def smooth_obj_and_grad(h, dx, alpha):
    n = len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / alpha)
    w = exp_c / np.sum(exp_c)
    val = (max_c + alpha * np.log(np.sum(exp_c))) * dx
    corr_full = correlate(1 - h, w, mode="full")
    conv_full = convolve(h, w, mode="full")
    term1 = corr_full[n - 1:2 * n - 1][::-1]
    term2 = conv_full[n - 1:2 * n - 1]
    grad = dx * (term1 - term2)
    return val, grad

def projected_gradient(h, target, dx, alpha, max_iter, rng, deadline):
    n = len(h)
    lr = 1.0
    h = project_onto_feasible(h, target)
    c5_old = compute_c5(h, dx)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, g = smooth_obj_and_grad(h, dx, alpha)
        h_new = h - lr * g
        h_new = project_onto_feasible(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_old - 1e-12:
            h = h_new
            c5_old = c5_new
            lr = min(lr * 1.08, 5.0)
        else:
            lr *= 0.6
            if lr < 1e-10:
                break
    return h, c5_old

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    deadline = t_start + budget_s * 0.95

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.zeros(50)
        h0[:25] = 1.0

    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)

    sizes = sorted(set([n0, 90, 180, 300]))
    if n0 >= 300:
        sizes = [n0]

    for n in sizes:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        target = n / 2.0

        if n == n0:
            h = h0.copy()
        else:
            h = upsample(h0, n)

        c5_here = compute_c5(h, dx)
        if c5_here < best_c5:
            best_c5 = c5_here
            best_h = h.copy()

        n_restarts = 5 if n <= 100 else (3 if n <= 200 else 2)
        for r in range(n_restarts):
            if time.time() > deadline:
                break

            if r == 0:
                h_curr = h.copy()
            elif r == 1:
                h_curr = h + rng.normal(0, 0.08, n)
                h_curr = project_onto_feasible(h_curr, target)
            elif r == 2:
                h_curr = h + rng.normal(0, 0.2, n)
                h_curr = project_onto_feasible(h_curr, target)
            elif r == 3:
                # Symmetric perturbation
                h_curr = h.copy()
                h_curr[:n//2] += rng.normal(0, 0.1, n//2)
                h_curr[n - n//2:] += rng.normal(0, 0.1, n - n//2)
                h_curr = project_onto_feasible(h_curr, target)
            else:
                # Sinusoidal perturbation
                m = rng.integers(1, max(2, n // 10))
                phase = rng.uniform(0, 2 * np.pi)
                h_curr = h + 0.1 * np.sin(2 * np.pi * m * np.arange(n) / n + phase)
                h_curr = project_onto_feasible(h_curr, target)

            # Annealing schedule
            alphas = [0.08 * n, 0.03 * n, 0.01 * n, 0.003 * n]
            for alpha in alphas:
                if time.time() > deadline:
                    break
                max_it = 300 if n <= 100 else (200 if n <= 200 else 100)
                h_curr, c5_curr = projected_gradient(h_curr, target, dx, alpha, max_it, rng, deadline)
                if c5_curr < best_c5:
                    best_c5 = c5_curr
                    best_h = h_curr.copy()
                if c5_curr < c5_here:
                    h = h_curr.copy()
                    c5_here = c5_curr

            # SLSQP for small n
            if n <= 150 and time.time() < deadline:
                alpha = 0.01 * n
                def obj_slsqp(x):
                    hx = np.clip(x, 0.0, 1.0)
                    corr_s = correlate(hx, 1 - hx, mode="full")
                    mc = np.max(corr_s)
                    return (mc + alpha * np.log(np.sum(np.exp((corr_s - mc) / alpha)))) * dx
                try:
                    res = minimize(obj_slsqp, h_curr, method='SLSQP',
                                   bounds=[(0.0, 1.0)] * n,
                                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                                   options={'maxiter': 200, 'ftol': 1e-12})
                    if res.fun < 1e6:
                        h_s = project_onto_feasible(res.x, target)
                        c5_s = compute_c5(h_s, dx)
                        if c5_s < best_c5:
                            best_c5 = c5_s
                            best_h = h_s.copy()
                        h_curr = h_s
                except Exception:
                    pass

    # Final: ensure exact integral
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    best_h = project_onto_feasible(best_h, n_best / 2.0)
    best_c5 = compute_c5(best_h, dx_best)
    # Ensure sum is exactly n/2 (within float precision)
    diff = n_best / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-14:
        mask = (best_h < 1.0) if diff > 0 else (best_h > 0.0)
        best_h[mask] += diff / np.sum(mask)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h, dx_best)

    return best_h, best_c5, n_best
