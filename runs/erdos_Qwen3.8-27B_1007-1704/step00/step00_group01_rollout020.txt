            w = rng.randint(2, 6)
            si = rng.randint(0, n - w + 1)
            h2[si:si + w] += rng.normal(0, 0.025, w)
        else:
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            d = rng.uniform(-0.06, 0.06)
            h2[i] += d
            h2[j] -= d

        h2 = project_feasible(h2, n)
        c5_2 = compute_c5(h2)
        dc = c5_2 - cur_c5

        if c5_2 < cur_c5:
            h = h2
            cur_c5 = c5_2
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        elif T > 5e-5 and rng.random() < np.exp(-max(dc, 0.0) / T):
            h = h2
            cur_c5 = c5_2

        T *= 0.99996
        if T < 0.0008:
            T = 0.0008
        if rng.random() < 0.0001:
            T = 0.005

    return best_h, best_c5

def gradient_refine(h0, n, maxiter):
    beta = 200.0
    cons = ({'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_jac},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_c5, h0, args=(n, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    hx = project_feasible(res.x, n)
    return hx, compute_c5(hx)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(47, 0.5)

    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SA on initial resolution
    if budget_s - elapsed() > 80:
        alloc = min((budget_s - elapsed()) * 0.30, 280.0)
        h1, c1 = sa_search(h, n, alloc, seed)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Gradient refinement
    if budget_s - elapsed() > 40:
        try:
            hg, cg = gradient_refine(best_h.copy(), n, maxiter=300)
            if cg < best_c5:
                best_c5 = cg
                best_h = hg.copy()
        except Exception:
            pass

    # Phase 3: SA from gradient result
    if budget_s - elapsed() > 60:
        alloc = min((budget_s - elapsed()) * 0.20, 150.0)
        h3, c3 = sa_search(best_h, n, alloc, seed + 1)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Multi-start with structured and random init
    rg = np.random.RandomState(seed + 42)
    for trial in range(25):
        if budget_s - elapsed() < 22:
            break
        if trial == 0:
            ht = np.full(n, 0.5) + rg.normal(0, 0.15, n)
        elif trial == 1:
            x = np.arange(n)
            ht = 0.5 + 0.3 * np.cos(2.0 * np.pi * x / n)
        elif trial == 2:
            x = np.arange(n)
            ht = 0.5 + 0.25 * np.sin(4.0 * np.pi * x / n)
        elif trial == 3:
            ht = best_h + rg.normal(0, 0.10, n)
        elif trial == 4:
            x = np.arange(n)
            ht = 0.5 + 0.2 * np.cos(6.0 * np.pi * x / n) + 0.1 * np.sin(2.0 * np.pi * x / n)
        else:
            ht = best_h + rg.normal(0, 0.05, n)

        ht = project_feasible(ht, n)
        alloc = min(budget_s - elapsed() - 18, 30.0)
        if alloc < 4:
            break
        h2, c2 = sa_search(ht, n, alloc, seed + 100 + trial)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 5: Try one higher resolution
    if budget_s - elapsed() > 60:
        n2 = min(n * 2, 200)
        if n2 != n:
            h_up = upsample(best_h, n2)
            h_up = project_feasible(h_up, n2)
            alloc = min(budget_s - elapsed() - 15, 80.0)
            if alloc > 20:
                h_up2, c_up = sa_search(h_up, n2, alloc, seed + 500)
                if c_up < best_c5:
                    best_c5 = c_up
                    best_h = h_up2
                    n = n2

    # Final gradient polish if time permits
    if budget_s - elapsed() > 20:
        try:
            hg, cg = gradient_refine(best_h.copy(), len(best_h), maxiter=100)
            if cg < best_c5:
                best_c5 = cg
                best_h = hg
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
