def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    """Adjust h to have the target sum while staying in [0,1]"""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h.copy()
    # Distribute correction to interior values
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    if np.sum(interior) > 0:
        h = h.copy()
        adj = diff / np.sum(interior)
        h[interior] += adj
        h = np.clip(h, 0, 1)
    # Second pass: distribute remaining to all
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-12:
        h = h.copy()
        # If diff2 > 0, need to increase some values
        if diff2 > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(45)
        h0[::2] = 1.0
        h0[0] = 0.5

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    if tl() > 150:
        h, c = cont_refine(best_h, 30000, 0.05)
        if c < best_c5:
            best_c5, best_h = c, h
    if tl() > 100:
        h, c = cont_refine(best_h, 30000, 0.02)
        if c < best_c5:
            best_c5, best_h = c, h
    if tl() > 80:
        h, c = cont_refine(best_h, 30000, 0.5)
        if c < best_c5:
            best_c5, best_h = c, h

    sizes = [44, 46, 48, 50, 40, 52, 54, 36, 42]
    for n_try in sizes:
        if tl() < 30:
            break
        n_iter = 120 if n_try <= 48 else 80
        for trial in range(12):
            if tl() < 25:
                break
            kind = trial % 3
            hb = make_pattern(n_try, kind)
            hb, cb = bin_local_search(hb, n_iter)
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()
            if tl() > 35 and trial < 10:
                k1 = np.random.randint(n_try)
                k2 = np.random.randint(n_try)
                while k2 == k1:
                    k2 = np.random.randint(n_try)
                hb[k1] = 1.0 - hb[k1]
                hb[k2] = 1.0 - hb[k2]
                d = n_try / 2.0 - hb.sum()
                if abs(d) > 0.5:
                    hb[k1] = 1.0 - hb[k1]
                    hb[k2] = 1.0 - hb[k2]
                    continue
                hb2, cb2 = bin_local_search(hb, 60)
                if cb2 < best_c5:
                    best_c5, best_h = cb2, hb2.copy()

    if tl() > 40:
        n = len(best_h)
        tgt = n / 2.0
        h = best_h.copy()
        curr = best_c5
        np.random.seed(seed + 1)
        for _ in range(20000):
            if tl() < 10:
                break
            i = np.random.randint(n)
            d = np.random.uniform(-0.25, 0.25)
            ht = h.copy()
            ht[i] = np.clip(h[i] + d, 0, 1)
            ht = fix_sum(ht, tgt)
            if abs(ht.sum() - tgt) > 1e-8:
                continue
            c = compute_c5(ht)
            if c < curr:
                h, curr = ht, c
                if c < best_c5:
                    best_c5, best_h = c, h.copy()

    if tl() > 20:
        n = len(best_h)
        h, c = cont_refine(best_h, 10000, 0.01)
        if c < best_c5:
            best_c5, best_h = c, h

    return best_h, best_c5, len(best_h)
