import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(overlap))

def smooth_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def get_smooth_max_weights(vals, alpha):
    m = np.max(vals)
    exps = np.exp(alpha * (vals - m))
    return exps / (np.sum(exps) + 1e-300)

def project_onto_simplex(y, S):
    """Project y onto {0 <= x <= 1, sum(x) = S}"""
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = S - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Distribute remaining
    if diff > 0:
        mask = x < 1.0
    else:
        mask = x > 0.0
    if mask.sum() > 0:
        x[mask] += diff / mask.sum()
        x = np.clip(x, 0, 1)
    return x

def compute_gradient(h, w, n, dx):
    """Compute gradient of smooth_max(c*dx) where c = correlate(h, 1-h, 'full')"""
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] / dx
        for k in range(n):
            i1 = k + m - n + 1
            i2 = k - m + n - 1
            if 0 <= i1 < n:
                grad[k] += wm * (1.0 - h[i1])
            if 0 <= i2 < n:
                grad[k] -= wm * h[i2]
    return grad

def projected_gradient(h0, n, max_iter=2000, alpha=100, dt_total=None):
    """Projected gradient descent on smooth max of overlap"""
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 0.02
    deadline = dt_total
    
    for it in range(max_iter):
        if deadline and time.time() - deadline[0] > deadline[1]:
            break
        
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        w = get_smooth_max_weights(ov, alpha)
        grad = compute_gradient(h, w, n, dx)
        
        h_new = h - lr * grad
        h_new = project_onto_simplex(h_new, n / 2.0)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Backtracking line search if stuck
        if it % 100 == 0:
            lr *= 0.95
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(overlap))

def smooth_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def smooth_max_weights(vals, alpha):
    m = np.max(vals)
    exps = np.exp(alpha * (vals - m))
    return exps / (np.sum(exps) + 1e-300)

def project_simplex(y, S):
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-13:
        mask = x < 1.0 if diff > 0 else x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad(h, w, n, dx):
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        klo1 = max(0, n - 1 - m)
        khi1 = n - 1
        if klo1 <= khi1:
            idx1 = (np.arange(klo1, khi1 + 1) + m - n + 1).astype(int)
            term1[klo1:khi1+1] += wm * (1.0 - h[idx1])
        klo2 = max(0, m - n + 1)
        khi2 = min(n - 1, m - 1)
        if klo2 <= khi2:
            idx2 = (np.arange(klo2, khi2 + 1) - m + n - 1).astype(int)
            term2[klo2:khi2+1] += wm * h[idx2]
    return (term1 - term2) / dx

def pg_descent(h0, n, max_iter, alpha, t0, budget):
    dx = 2.0 / n
    h = h0.copy()

    best_h = h.copy()
    lr = 0.015
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > budget * 0.88:
            break
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        w = smooth_max_weights(ov, alpha)
        grad = compute_grad(h, w, n, dx)
        h_new = h - lr * grad
        h_new = project_simplex(h_new, n / 2.0)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                stall = 0
        h = h_new
    return best_h, best_c5

def slsqp_optimize(h0, n, alpha, max_iter, ftol, t0, budget_frac):
    if time.time() - t0 > budget_frac:
        return h0, evaluate_c5(h0)
    dx_val = 2.0 / n
    def obj(hf):
        ov = np.correlate(hf, 1.0 - hf, mode='full') * dx_val
        return smooth_max(ov, alpha)
    cons = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_simplex(res.x, n / 2.0)
    return h_new, evaluate_c5(h_new)
