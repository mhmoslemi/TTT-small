```python
import numpy as np
import scipy.optimize as so
import time

_current_n = 47

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = 0.5 * n - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h):
    return _eval_c5(_project(h, _current_n))

def _interpolate(h, n_src, n_tgt):
    xo = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return np.interp(xn, xo, h)

def _try_optimize(hs, n_t, method, maxiter, xatol, fatol, bounds):
    global _current_n
    _current_n = n_t
    try:
        res = so.minimize(_obj, hs.copy(), method=method, bounds=bounds,
                          options={'maxiter': maxiter, 'xatol': xatol, 'fatol': fatol})
        ht = _project(res.x, n_t)
        ct = _eval_c5(ht)
        return ht, ct
    except Exception:
        return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    global _current_n

    # Load parent construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

    n = len(h0)
    best_h = _project(h0, n)
    best_c5 = _eval_c5(best_h)

    def _rem():
        return budget_s - (time.time() - t0)

    # Build list of resolutions to try
    res_list = [n]
    for m in [2, 3, 4, 5, 6, 8]:
        r = n * m
        if r <= 500:
            res_list.append(r)
    for r in [64, 80, 94, 100, 128, 156, 160, 188, 200, 256, 300]:
        if r not in res_list:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if _rem() < 40:
            break

        # Starting point
        if n_t == len(best_h):
            hs = best_h.copy()
        else:
            hs = _project(_interpolate(best_h, len(best_h), n_t), n_t)

        bounds = [(0.0, 1.0)] * n_t

        # Powell optimization (handles non-smooth max well)
        if _rem() > 30:
            mi = 300 if n_t <= 80 else (150 if n_t <= 150 else 80)
            ht, ct = _try_optimize(hs, n_t, 'Powell', mi, 1e-5, 1e-7, bounds)
            if ht is not None and ct < best_c5:
                best_c5, best_h = ct, ht

        # Nelder-Mead
        if _rem() > 30:
            mi = 500 if n_t <= 80 else (200 if n_t <= 150 else 100)
            ht, ct = _try_optimize(hs, n_t, 'Nelder-Mead', mi, 1e-6, 1e-8, bounds)
            if ht is not None and ct < best_c5:
                best_c5, best_h = ct, ht

        # Multi-start with random perturbations
        n_restarts = 5 if n_t <= 80 else (3 if n_t <= 150 else 2)
        for i in range(n_restarts):
            if _rem() < 30:
                break
            sigma = 0.06 + 0.02 * i
            hr = _project(hs + rng.normal(0, sigma, n_t), n_t)
            mi = 100 if n_t <= 100 else 60
            ht, ct = _try_optimize(hr, n_t, 'Powell', mi, 1e-5, 1e-7, bounds)
            if ht is not None and ct < best_c5:
                best_c5, best_h = ct, ht

        # Binary-like perturbation start
        if _rem() > 40 and n_t <= 200:
            h_bin = np.zeros(n_t)
            blk = max(1, n_t // 8)
            for b in range(0, n_t, blk):
                if (b // blk) % 2 == 0:
                    h_bin[b:b+blk] = 1.0
            h_bin = _project(h_bin, n_t)
            ht, ct = _try_optimize(h_bin, n_t, 'Powell', 100, 1e-5, 1e-7, bounds)
            if ht is not None and ct < best_c5:
                best_c5, best_h = ct, ht

    # Final local polish at best resolution
    if _rem() > 10:
        n_f = len(best_h)
        hf = _project(rng.normal(0, 0.03, n_f) + best_h, n_f)
        bounds_f = [(0.0, 1.0)] * n_f
        ht, ct = _try_optimize(hf, n_f, 'Powell', 200, 1e-6, 1e-8, bounds_f)
        if ht is not None and ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = _project(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
```
