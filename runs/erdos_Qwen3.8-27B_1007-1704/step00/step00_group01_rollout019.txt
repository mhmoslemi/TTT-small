```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, dx):
    c = compute_correlations(h) * dx
    return np.max(c)

def smooth_max_objective(h, dx, beta=50.0):
    c = compute_correlations(h) * dx
    # Numerical stability
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    exp_vals = np.exp(shifted)
    return c_max + np.log(np.sum(exp_vals)) / beta

def smooth_max_gradient(h, dx, beta=50.0):
    c = compute_correlations(h) * dx
    n = len(h)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient is weighted sum of individual correlation gradients
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-10:
            continue
        shift = m - (n - 1)
        if shift >= 0:
            s = shift
            # ∂/∂h[i] = (1-h[i-s]) * [s <= i <= n-1] - h[i+s] * [0 <= i <= n-1-s]
            for i in range(s, n):
                grad[i] += weights[m] * (1 - h[i - s]) * dx
            for i in range(0, n - s):
                grad[i] -= weights[m] * h[i + s] * dx
        else:
            s = -shift
            # ∂/∂h[i] = (1-h[i+s]) * [0 <= i <= n-1-s] - h[i-s] * [s <= i <= n-1]
            for i in range(0, n - s):
                grad[i] += weights[m] * (1 - h[i + s]) * dx
            for i in range(s, n):
                grad[i] -= weights[m] * h[i - s] * dx
    return grad
```

```python
    c = compute_c(h) * dx
    n = len(h)
    cm = np.max(c)
    sh = beta * (c - cm)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h, dx):
    return float(np.max(compute_c(h) * dx))

def smooth_cg(h, dx, beta):
    c = compute_c(h) * dx
    n = len(h)
    cm = np.max(c)
    sh = beta * (c - cm)
    ev = np.exp(sh)
    s = ev.sum()
    obj = cm + np.log(s) / beta
    w = ev / s
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-12:
            continue
        s2 = m - (n - 1)
        if s2 >= 0:
            grad[s2:] += wm * (1.0 - h[:n - s2])
            grad[:n - s2] -= wm * h[s2:]
        else:
            s2 = -s2
            grad[:n - s2] += wm * (1.0 - h[s2:])
            grad[s2:] -= wm * h[:n - s2]
    grad *= dx
    return obj, grad

def proj(h, target):
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - p.sum()
    if abs(d) > 1e-12:
        m = (p > 1e-9) & (p < 1.0 - 1e-9)
        if m.sum() > 0:
            p[m] += d / m.sum()
            p = np.clip(p, 0.0, 1.0)

    return r.x
```
