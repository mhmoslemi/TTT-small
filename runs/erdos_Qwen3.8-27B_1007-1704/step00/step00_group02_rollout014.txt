            break
        alloc = min(rem() * 0.11, 100)
        deadline = time.time() + alloc
        while time.time() < deadline and rem() > 50:
            h_new, c5_new = _sa_targeted(h_best, n, rng, sigma, 5000, temp)
            if c5_new < c5_best:
                c5_best, h_best = c5_new, h_new.copy()

    # Phase 3: Gradient-based refinement
    if rem() > 60:
        h_g, c5_g = _grad_phase(h_best, n, 400)
        if c5_g < c5_best:
            c5_best, h_best = c5_g, h_g
    
    # Phase 4: Fine SA polishing
    for sigma, temp in [(0.01, 0.0008), (0.005, 0.0004), (0.003, 0.0002), (0.001, 0.0001)]:
        if rem() < 40:
            break
        alloc = min(rem() * 0.07, 50)
        deadline = time.time() + alloc
        while time.time() < deadline and rem() > 25:
            h_new, c5_new = _sa(h_best, n, rng, sigma, 8000, temp)
            if c5_new < c5_best:
                c5_best, h_best = c5_new, h_new.copy()

    # Phase 5: Second gradient pass
    if rem() > 50:
        h_g, c5_g = _grad_phase(h_best, n, 300)
        if c5_g < c5_best:
            c5_best, h_best = c5_g, h_g

    # Phase 6: Multi-resolution exploration
    for n_h in [n_best * 2, n_best * 4]:
        if rem() < 90 or n_h > 1000:
            break
        xs = np.linspace(0.0, 1.0, n_best)
        xh = np.linspace(0.0, 1.0, n_h)
        h_hi = np.interp(xh, xs, h_best)
        h_hi = _repair(h_hi, n_h)
        c5_hi = _c5max(h_hi)
        for sigma, temp in [(0.04, 0.003), (0.02, 0.001), (0.01, 0.0005)]:
            if rem() < 30:
                break
            alloc = min(rem() * 0.1, 35)
            deadline = time.time() + alloc
            while time.time() < deadline and rem() > 20:
                h_hi, c5_hi = _sa(h_hi, n_h, rng, sigma, 3000, temp)
                if c5_hi < c5_best:
                    c5_best, h_best, n_best = c5_hi, h_hi.copy(), n_h
        if c5_hi < c5_best:
            c5_best, h_best, n_best = c5_hi, h_hi.copy(), n_h

    # Phase 7: Final polish at best resolution
    if rem() > 30:
        for sigma, temp in [(0.01, 0.0005), (0.003, 0.0002)]:
            if rem() < 20:
                break
            alloc = min(rem() * 0.15, 40)
            deadline = time.time() + alloc
            while time.time() < deadline and rem() > 10:
                h_new, c5_new = _sa_targeted(h_best, n_best, rng, sigma, 5000, temp)
                if c5_new < c5_best:
                    c5_best, h_best = c5_new, h_new.copy()

    # Final validation
    h_final = _repair(h_best, len(h_best))
    c5_final = _c5max(h_final)
    return h_final, c5_final, len(h_final)
