min_{h[j]∈[0,1]} max_k (a_k*h[j] + b_k)

    else:

    n0 = len(h0)
    best_h = project_bounded(h0.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: PG optimize initial (20% budget)
    t1 = t0 + budget_s * 0.20
    h1, c5_1 = pg_optimize(best_h, n0, t1, [3, 10, 50, 200], 400)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Coordinate descent refine (20% budget)
    t2 = t0 + budget_s * 0.40
    h2, c5_2 = coordinate_descent(best_h, n0, t2, 30)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2
    # Another PG pass after CD
    h2b, c5_2b = pg_optimize(best_h, n0, t2, [50, 200, 500], 300)
    if c5_2b < best_c5:
        best_c5, best_h = c5_2b, h2b

    # Phase 3: Multiple restarts at n0 (25% budget)
    t3 = t0 + budget_s * 0.65
    n_restarts = 12
    for r in range(n_restarts):
        if time.time() > t3:
            break
        if r == 0:
            hs = np.full(n0, 0.5)
        elif r == 1:
            hs = np.sin(np.linspace(0, np.pi, n0)) ** 2
        elif r == 2:
            mm = max(1, n0 // 3)
            hs = np.zeros(n0)
            hs[mm:2*mm] = 1.0
        elif r == 3:
            hs = np.zeros(n0)
            hs[:n0//3] = 1.0
            hs[2*n0//3:] = 1.0
        else:
            hs = np.random.uniform(0, 1, n0)
        hs = project_bounded(hs, n0 / 2.0)
        h_r, c5_r = pg_optimize(hs, n0, t3, [10, 50, 200], 150)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r
        h_r2, c5_r2 = coordinate_descent(best_h if c5_r >= best_c5 else h_r, n0, t3, 10)
        if c5_r2 < best_c5:
            best_c5, best_h = c5_r2, h_r2

    # Phase 4: Upsample and refine (20% budget)
    t4 = t0 + budget_s * 0.85
    for mult in [2, 4]:
        if time.time() > t4:
            break
        n2 = min(mult * n0, 800)
        h_up = project_bounded(upsample(best_h, n2), n2 / 2.0)
        iters = max(50, 5000 // n2)
        h_up, c5_up = pg_optimize(h_up, n2, t4, [10, 50, 200], iters)
        h_up2, c5_up2 = coordinate_descent(h_up, n2, t4, 10)
        if c5_up2 < c5_up:
            h_up, c5_up = h_up2, c5_up2
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_up

    # Phase 5: Final polish (10% budget)
    t5 = t0 + budget_s * 0.95
    nb = len(best_h)
    hf, cf = pg_optimize(best_h, nb, t5, [200, 500], 400)
    if cf < best_c5:
        best_c5, best_h = cf, hf
    hf2, cf2 = coordinate_descent(best_h, nb, t5, 20)
    if cf2 < best_c5:
        best_c5, best_h = cf2, hf2

    best_h = project_bounded(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
