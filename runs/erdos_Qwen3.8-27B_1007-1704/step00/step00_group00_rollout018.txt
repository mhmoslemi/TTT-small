import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    return compute_c5(x, len(x))

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, maxiter, ftol):
    bounds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options=dict(maxiter=maxiter, ftol=ftol))
    return repair(res.x, len(h))

def _local_search(h, n, rng, n_iter, deadline):
    best = h.copy()
    best_v = compute_c5(best, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        direction = 1.0 if best[i] > 0.5 else -1.0
        step = 0.01 + 0.09 * rng.random()
        ht = best.copy()
        ht[i] += direction * step
        ht = repair(ht, n)
        v = compute_c5(ht, n)
        if v < best_v:
            best = ht
            best_v = v
    return best, best_v

def _swap_search(h, n, rng, n_iter, deadline):
    best = h.copy()
    best_v = compute_c5(best, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        ht = best.copy()
        ht[i], ht[j] = ht[j], ht[i]
        ht = repair(ht, n)
        if np.array_equal(ht, best):
            continue
        v = compute_c5(ht, n)
        if v < best_v:
            best = ht
            best_v = v
    return best, best_v
