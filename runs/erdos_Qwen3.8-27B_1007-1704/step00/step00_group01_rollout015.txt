import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def slsqp_obj(x, T, dx_val):
    c = np.correlate(x, 1.0 - x, mode="full") * dx_val
    m = np.max(c)
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def slsqp_eq(x, n):
    return np.sum(x) - n / 2.0

def slsqp_eq_jac(x, n):
    return np.ones(n)

def nm_obj_proj(x, n):
    h = project_feasible(x, n)
    return compute_c5(h)

            best_c5 = c5_before

                if abs(trial - old) < 1e-4:
                    continue
                h[i] = trial
                rest_sum = target - (target - old) + trial
                if rest_sum > n * 1.0 or rest_sum < 0:
                    h[i] = old
                    continue
                h_tot = np.sum(h)
                if h_tot > 0:

                    h_test = h

                    best_c5 = c5_try
                    best_val = trial
            if best_val != old:
                h[i] = best_val
                improved = True
        h = project_feasible(h, n)
    return h, compute_c5(h)
