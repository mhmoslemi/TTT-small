    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    g = np.zeros(n)

        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            m1 = np.arange(lag, n)
            g[m1] += (1.0 - h[m1 - lag])
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]

        m1 = np.arange(0, n - m_val)
        g[m1] += (1.0 - h[m1 + m_val])
        m2 = np.arange(m_val, n)
        g[m2] -= h[m2 - m_val]

        g[m1] += (1.0 - h[m1 + mv])
        m2 = np.arange(mv, n)
        g[m2] -= h[m2 - mv]

def project_feasible(h):

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        s = h.sum()

            cnt = int(mask.sum())

        else:
            mask = h > 1e-13

            if cnt == 0:
                break

def _slsqp_grad(h):
