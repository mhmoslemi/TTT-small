    best_h, best_c5 = anneal(best_h, best_n, rng, t_lim, t0, budget_s)

    # Phase 2: Hill climb
    t_lim = t0 + min(180, budget_s * 0.15)
    best_h, best_c5 = hill_climb(best_h, best_n, rng, t_lim, t0, budget_s)

    # Phase 3: Lag-focused
    t_lim = t0 + min(100, budget_s * 0.08)
    best_h, best_c5 = lag_focused(best_h, best_n, rng, t_lim, t0, budget_s)

    # Phase 4: Try other n values and perturbations
    n_trials = [44, 46, 50, 56, 64, 80, 96, 120, 160]
    for trial_idx, n in enumerate(n_trials):
        if n == best_n:
            continue
        remaining = budget_s - (time.time() - t0)
        if remaining < 100:
            break
        h_try = perturbed_binary(n, rng, seed + 100 + trial_idx * 7)
        t_lim = t0 + min(100, remaining * 0.35)
        h_try, c5_try = anneal(h_try, n, rng, t_lim, t0, budget_s)
        remaining = budget_s - (time.time() - t0)
        if remaining > 50:
            t_lim2 = t0 + min(60, remaining * 0.35)
            h_try, c5_try = hill_climb(h_try, n, rng, t_lim2, t0, budget_s)
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try.copy(), c5_try, n

    # Phase 5: Second round on best with different perturbations
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        for restart in range(3):
            remaining = budget_s - (time.time() - t0)
            if remaining < 40:
                break
            h_r = best_h.copy()
            noise = rng.uniform(-0.03, 0.03, best_n)
            h_r = np.clip(h_r + noise, 0, 1)
            h_r = fix_sum(h_r, best_n)
            t_lim = t0 + min(60, remaining * 0.25)
            h_r, c5_r = anneal(h_r, best_n, rng, t_lim, t0, budget_s)
            remaining = budget_s - (time.time() - t0)
            if c5_r < best_c5 and remaining > 20:
                t_lim2 = t0 + min(30, remaining * 0.5)
                h_r, c5_r = hill_climb(h_r, best_n, rng, t_lim2, t0, budget_s)
            if c5_r < best_c5:
                best_h, best_c5 = h_r.copy(), c5_r

    # Phase 6: Final fine hill climb
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        t_lim = t0 + min(remaining * 0.8, 120)
        best_h, best_c5 = hill_climb(best_h, best_n, rng, t_lim, t0, budget_s)

    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
