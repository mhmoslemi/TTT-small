        return best, best_c5

    def perturb_search(h0, n, budget):
        h = h0.astype(float).copy()
        h = project(h, n)
        cur_c5 = compute_c5(h)
        best = h.copy()
        best_c5 = cur_c5
        t_start = time.time()
        lr = 0.12
        patience = 0

        while (time.time() - t_start) < budget and patience < 4000:
            n_pert = max(1, n // 8)
            idxs = rng.choice(n, size=n_pert, replace=False)
            deltas = rng.randn(n_pert) * lr
            h_new = h.copy()
            h_new[idxs] += deltas
            h_new = project(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h = h_new
                cur_c5 = c5_new
                patience = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best = h.copy()
                lr = min(lr * 1.03, 0.35)
            else:
                patience += 1
                if patience % 150 == 0:
                    lr = max(lr * 0.93, 0.01)
        return best, best_c5

    # Multi-resolution search
    n_list = []
    if n_parent > 0:
        n_list.append(n_parent)
    for n_try in [40, 50, 55, 60, 65, 72, 80, 90, 100, 112, 120, 144, 160, 200]:
        if n_try not in n_list:
            n_list.append(n_try)
    n_list.sort()

    for n in n_list:
        if remaining() < 15:
            break
        if n == n_parent and h_parent is not None:
            h0 = project(h_parent.copy(), n)
        elif h_parent is not None:
            h0 = np.interp(
                np.linspace(0, n_parent - 1, n),
                np.linspace(0, n_parent - 1, n_parent),
                h_parent
            )
            h0 = project(h0, n)
        else:
            h0 = project(np.full(n, 0.5), n)

        is_parent = (n == n_parent and h_parent is not None)
        frac = 0.35 if is_parent else 0.2
        bud = min(remaining() * frac, 130)
        bud = max(bud, 8)

        h_cd, c5_cd = coord_descent(h0, n, bud * 0.65)
        h_ps, c5_ps = perturb_search(h_cd, n, bud * 0.35)

        if c5_ps < c5_cd:
            c5_here = c5_ps
            h_here = h_ps
        else:
            c5_here = c5_cd
            h_here = h_cd

        if c5_here < best_c5:
            best_c5 = c5_here
            best_h = h_here

    # Final polishing
    if remaining() > 25:
        nb = len(best_h)
        h_cd, c5_cd = coord_descent(best_h, nb, min(remaining() * 0.6, 80))
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd
    if remaining() > 15:
        nb = len(best_h)
        h_ps, c5_ps = perturb_search(best_h, nb, min(remaining() * 0.5, 40))
        if c5_ps < best_c5:
            best_h = h_ps
            best_c5 = c5_ps

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
