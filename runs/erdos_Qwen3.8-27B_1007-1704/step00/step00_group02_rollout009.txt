import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    # Distribute adjustment
    if diff > 0:
        # increase: add to elements with room
        room = 1.0 - h
        mask = room > 1e-10
        if np.sum(room[mask]) < diff:
            h = np.clip(h, 0, 1)  # can't fully satisfy
        else:
            h[mask] += diff / np.sum(room[mask]) * room[mask]
    else:
        # decrease: subtract from elements with room
        room = h
        mask = room > 1e-10
        if np.sum(room[mask]) < -diff:
            h = np.clip(h, 0, 1)
        else:
            h[mask] += diff / np.sum(room[mask]) * room[mask]
    h = np.clip(h, 0.0, 1.0)
    # Exact fix on one element
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def simulate(h0, n, time_limit, rng, T0=0.1, alpha=0.999):
    """Simulated annealing style local search."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T0
    start = time.time()
    
    while time.time() - start < time_limit:
        # Generate candidate
        h_new = h.copy()
        # Pick 1-3 random positions to perturb
        n_mod = rng.integers(1, 4)
        idx = rng.choice(n, size=n_mod, replace=False)
        
        if rng.random() < 0.5:
            # Binary flip
            h_new[idx] = 1.0 - h_new[idx]
        else:
            # Gaussian perturbation
            h_new[idx] += rng.normal(0, 0.1, size=n_mod)
        
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-6)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        
        T *= alpha
        T = max(T, 0.001)
    
    return best_h, best_c5

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += diff / total_room * room
            else:
                h += diff / n
        else:
            room = h
            total_room = np.sum(room)
            if total_room > 0:
                h += diff / total_room * room
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def simulated_anneal(h0, n, time_limit, rng, T0=0.05):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T0
    start = time.time()
    stall = 0
    while time.time() - start < time_limit:
        h_new = h.copy()
        n_mod = int(rng.integers(1, 4))
        idx = rng.choice(n, size=n_mod, replace=False)
        r = rng.random()
        if r < 0.4:
            h_new[idx] = 1.0 - h_new[idx]
        elif r < 0.7:
            h_new[idx] += rng.normal(0, 0.12, size=n_mod)
        else:
            h_new[idx] = rng.random(n_mod)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 or (T > 0.001 and rng.random() < np.exp(-(c5_new - cur_c5) / T)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 5000:
            T = T0 * 0.3
            stall = 0
        T *= 0.9995
        T = max(T, 0.0005)
    return best_h, best_c5

def coord_descent(h0, n, time_limit, rng):
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h)
    start = time.time()
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.5, -0.5]
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                h_new = h.copy()
                h_new[i] += d
                if h_new[i] < 0 or h_new[i] > 1:
                    continue
                h_new = repair(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return h_new

def make_block(n, frac):
    h = np.zeros(n)
    k = int(round(frac * n / 2.0))
    h[:k] = 1.0
    return h

def random_balanced(n, rng):
    h = (rng.random(n) > 0.5).astype(float)
    while np.sum(h) != n // 2:
        if np.sum(h) > n // 2:
            h[int(np.where(h == 1)[0][int(rng.integers(0, n))])] = 0
        else:
            h[int(np.where(h == 0)[0][int(rng.integers(0, n))])] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_good = np.array(initial_h_values, dtype=float)
        n_good = len(h_good)
    else:
        h_good = make_block(50, 0.5)
        n_good = 50

    h_good = repair(h_good, n_good)
    c5_best = compute_c5(h_good)

    # Phase 1: SA on initial resolution
    t_alloc = min(250, budget_s * 0.25)
    if budget_s - (time.time() - t_start) > t_alloc + 100:
        h1, c5_1 = simulated_anneal(h_good, n_good, t_alloc, rng, T0=0.06)
        if c5_1 < c5_best:
            h_good, c5_best = h1, c5_1

    # Phase 2: Coordinate descent polish
    t_alloc = min(120, budget_s * 0.12)
    if time.time() < deadline - 200:
        h2, c5_2 = coord_descent(h_good, n_good, t_alloc, rng)
        if c5_2 < c5_best:
            h_good, c5_best = h2, c5_2

    # Phase 3: Multi-resolution search
    n_candidates = [n_good]
    if n_good <= 60:
        n_candidates.extend([n_good + 5, n_good + 10, n_good + 20, n_good + 35])
    else:
        n_candidates.extend([max(30, n_good - 10), n_good + 15, n_good + 30])
    # Also try some smaller n
    n_candidates.extend([30, 40, 50, 60, 80, 100])
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.time() > deadline - 150:
            break
        if n_try == n_good and n_try in n_candidates[:2]:
            continue
        if n_try < 10 or n_try > 300:
            continue

        # Build initial for this n
        if n_try != len(h_good):
            h_init = interp_to_n(h_good, n_try)
        else:
            h_init = h_good.copy()
        h_init = repair(h_init, n_try)

        # SA
        t_left = deadline - time.time() - 80
        t_alloc = min(80, t_left * 0.35)
        if t_alloc < 20:
            break
        h3, c5_3 = simulated_anneal(h_init, n_try, t_alloc, rng, T0=0.05)
        if c5_3 < c5_best:
            h_good, c5_best = h3, c5_3
            n_good = n_try

        # Quick CD
        if time.time() < deadline - 100:
            h4, c5_4 = coord_descent(h_good, n_good, min(30, deadline - time.time() - 80), rng)
            if c5_4 < c5_best:
                h_good, c5_best = h4, c5_4

    # Phase 4: Random restarts at small n
    for n_try in [30, 35, 40, 45, 50, 55]:
        if time.time() > deadline - 60:
            break
        h_rand = random_balanced(n_try, rng)
        c5_rand = compute_c5(h_rand)
        t_left = deadline - time.time() - 40
        t_alloc = min(25, t_left * 0.4)
        if t_alloc < 10:
            break
        h5, c5_5 = simulated_anneal(h_rand, n_try, t_alloc, rng, T0=0.04)
        if c5_5 < c5_best:
            h_good, c5_best = h5, c5_5
            n_good = n_try

    # Final polish
    if time.time() < deadline - 30:
        h_final, c5_final = coord_descent(h_good, len(h_good), min(20, deadline - time.time() - 10), rng)
        if c5_final < c5_best:
            h_good, c5_best = h_final, c5_final

    n_final = len(h_good)
    h_good = repair(h_good, n_final)
    c5_final = compute_c5(h_good)
    return h_good, c5_final, n_final
