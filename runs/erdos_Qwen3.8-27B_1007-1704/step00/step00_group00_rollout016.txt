import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)


def make_feasible(h, n):
    return project_feasible(h, 0.5 * n)


def nm_objective(x, n, deadline):
    if time.time() > deadline:
        return 10.0
    return compute_c5(make_feasible(x, n))


def coord_descent(h, n, step_sizes, deadline, rng, max_outer=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    plateau = 0
    for outer in range(max_outer):
        if time.time() >= deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() >= deadline:
                break
            for d in step_sizes:
                h_trial = best_h.copy()
                h_trial[i] += d
                h_trial = make_feasible(h_trial, n)
                c5 = compute_c5(h_trial)
                if c5 < best_c5:
                    best_h = h_trial
                    best_c5 = c5
                    improved = True
                    break
        if improved:
            plateau = 0
        else:
            plateau += 1
            if plateau > 10:
                break
    return best_h, best_c5


def random_search(h, n, deadline, rng, max_iter=1000000, init_temp=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    temp = init_temp
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        if it > 0 and it % 50000 == 0:
            temp *= 0.85
            if temp < 0.003:
                temp = 0.003
        k = rng.randint(1, 8)
        idx = rng.choice(n, k, replace=False)
        h_trial = best_h.copy()
        h_trial[idx] += rng.normal(0, temp, k)
        h_trial = make_feasible(h_trial, n)
        c5 = compute_c5(h_trial)
        if c5 < best_c5:
            best_h = h_trial
            best_c5 = c5
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    n = 45
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)
    else:
        h0 = np.full(n, 0.5)

    h0 = make_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nelder-Mead with multiple perturbations
    scales = [0.0, 0.04, 0.08, 0.15]
    for attempt in range(4):
        if time.time() >= deadline - 90:
            break
        if scales[attempt] == 0.0:
            x_start = h0.copy()
        else:
            x_start = best_h + rng.normal(0, scales[attempt], n)
        try:
            res = minimize(nm_objective, x_start, args=(n, deadline - 60),
                           method='Nelder-Mead',
                           options={'maxiter': 25000, 'xatol': 1e-11, 'fatol': 1e-13})
            h_trial = make_feasible(res.x, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
        except Exception:
            pass

    # Phase 2: Coordinate descent (coarse)
    coarse_steps = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.5,
                             -0.05, -0.1, -0.15, -0.2, -0.3, -0.5])
    best_h, best_c5 = coord_descent(best_h, n, coarse_steps,
                                    deadline - 30, rng, max_outer=300)

    # Phase 3: Random search (explore)
    best_h, best_c5 = random_search(best_h, n, deadline - 30, rng,
                                    max_iter=800000, init_temp=0.06)

    # Phase 4: Coordinate descent (fine)
    fine_steps = np.array([0.01, 0.02, 0.05, 0.1,
                           -0.01, -0.002, -0.05, -0.1])
    best_h, best_c5 = coord_descent(best_h, n, fine_steps,
                                    deadline - 15, rng, max_outer=400)

    # Phase 5: Short random polish
    best_h, best_c5 = random_search(best_h, n, deadline - 5, rng,
                                    max_iter=200000, init_temp=0.02)

    # Phase 6: Final fine coordinate polish
    vfine = np.array([0.005, 0.002, 0.001, -0.005, -0.002, -0.001])
    best_h, best_c5 = coord_descent(best_h, n, vfine,
                                    deadline - 2, rng, max_outer=200)

    return (best_h, best_c5, n)
