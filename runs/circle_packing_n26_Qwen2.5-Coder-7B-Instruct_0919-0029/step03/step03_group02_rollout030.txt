import numpy as np
from scipy.optimize import minimize, trust_constr, Bounds
from scipy.spatial import distance_matrix

def validate_packing(centers, radii):
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        if not (0 <= centers[i, 0] <= 1 and 0 <= centers[i, 1] <= 1):
            return False, f"Circle {i} at {centers[i]} is outside the unit square"
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            if d < radii[i] + radii[j]:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def objective(z):
    n = 26
    radii = z[2*n:]
    return -np.sum(radii)

def ineq_constraints(z):
    n = 26
    x, y, radii = z[:n], z[n:2*n], z[2*n:]
    bounds = np.vstack([x-radii, 1-x-radii, y-radii, 1-y-radii]).T.flatten()
    distances = distance_matrix(list(zip(x,y)), list(zip(x,y)))
    np.fill_diagonal(distances, np.inf)
    pairwise = radii[:, None] + radii[None, :] - distances
    return np.concatenate((bounds, pairwise.flatten()))

def run_packing():
    n = 26
    # Warm start: 5x5 grid plus a central circle
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.10)
    centers[:9, 0] = np.array([0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 1.7]) / 2
    centers[:9, 1] = np.zeros(9)
    centers[9:18, 0] = np.array([0.1, 0.3, 0.5]) / 2
    centers[9:18, 1] = np.array([0.1, 0.3, 0.5]) / 2 + 0.5
    centers[18:, 0] = np.array([0.1, 0.3, 0.5]) / 2
    centers[18:, 1] = np.array([0.1, 0.3, 0.5]) / 2 + 1.0
    centers[24, 0] = centers[24, 1] = 0.5
    radii[24] = np.sqrt(0.08) - 0.10

    z0 = np.hstack([centers.ravel(), radii.ravel()])

    bounds = Bounds(0, 1)  # Bounds on x, y
    constraints = {'type': 'ineq', 'fun': ineq_constraints}

    result = minimize(objective, z0, method='trust-constr', jac=None, bounds=bounds, constraints=constraints,
                      options={'max_iter': 2000, 'gtol': 1e-8, 'xtol': 1e-8, 'initial_tr_radius': 0.5})
    if not result.success:
        raise Exception("Optimization failed")

    centers = result.x[:n].reshape(n, 2)
    radii = result.x[n:].reshape(n)
    total = np.sum(radii)
    return centers, radii, total
