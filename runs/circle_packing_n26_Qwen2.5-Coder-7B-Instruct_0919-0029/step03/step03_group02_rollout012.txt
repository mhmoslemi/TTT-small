import numpy as np
from scipy.optimize import linprog
from scipy.stats import multivariate_normal

T0 = 1.0
Tf = 1e-3
alpha = 0.9999
delta = 1e-4
eps_overlap = 1e-12
steps_per_T = 500
max_iter = steps_per_T * int(np.log(T0 / Tf) / np.log(alpha))
sigma_0 = 2 * delta
m = 5

def solve_lp(centers, eps=eps_overlap):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] + eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] + eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - eps)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - eps)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def objective(centers, eps=eps_overlap):
    try:
        radii, sum_radii = solve_lp(centers, eps=eps)
    except Exception as e:
        radii, sum_radii = np.zeros(26), 0.0

    return sum_radii

def anneal(centers, max_iter, T0, Tf, alpha, sigma_0):
    n = centers.shape[0]
    sigma_t = sigma_0
    best_centers = centers
    best_sum = objective(centers)
    best_indices = np.arange(n)

    for t in range(max_iter):
        for _ in range(n // m):
            i = np.random.choice(best_indices)
            proposal = centers[i] + multivariate_normal.rvs(mean=[0,0], cov=sigma_t**2)
            proposal = np.clip(proposal, 0, 1)

            try:
                new_sum = objective(proposal)
                if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T_t):
                    centers[i] = proposal
                    best_sum = new_sum

                    if new_sum > best_sum:
                        best_centers = centers.copy()
                    elif new_sum == best_sum:
                        best_centers[np.where((centers == proposal).all(axis=1))] = proposal

                    if best_sum >= 2.637:
                        return best_centers, solve_lp(best_centers)[0], best_sum

            except Exception as e:
                pass

        sigma_t *= alpha
        T_t = T0 * (alpha ** (t // steps_per_T))

    return centers, solve_lp(centers)[0], best_sum

def run_packing():
    numstarts = 10
    best_centers, best_radii, best_sum = None, None, -np.inf

    for _ in range(numstarts):
        centers = np.random.uniform(0, 1, size=(26, 2))
        initial_r = 0.01
        centers *= initial_r / np.linalg.norm(centers - 0.5, axis=1, keepdims=True)
        try:
            final_centers, final_radii, final_sum = anneal(centers, max_iter, T0, Tf, alpha, sigma_0)
            if final_sum > best_sum:
                best_sum = final_sum
                best_centers = final_centers
                best_radii = final_radii
        except Exception as e:
            pass

    return best_centers, best_radii, best_sum
