import numpy as np
from scipy.optimize import minimize

def penalty(z, mu):
    d = np.sqrt((z[0::3, np.newaxis] - z[0::3]) ** 2 +
                (z[1::3, np.newaxis] - z[1::3]) ** 2 + 1e-24)
    d = d[np.triu_indices_from(d, 1)]
    r = z[2::3]

    penalty_overlap = mu * np.sum((r + r[:, np.newaxis] - d) ** 2 * (r + r[:, np.newaxis] > d))
    penalty_boundary_l = mu * np.sum((r - z[0::3]) ** 2 * (r > z[0::3]))
    penalty_boundary_r = mu * np.sum((r - (1 - z[0::3])) ** 2 * (r > (1 - z[0::3])))
    penalty_boundary_b = mu * np.sum((r - z[1::3]) ** 2 * (r > z[1::3]))
    penalty_boundary_t = mu * np.sum((r - (1 - z[1::3])) ** 2 * (r > (1 - z[1::3])))
    penalty_non_negative = mu * np.sum((-r) ** 2 * (r < 0))

    return penalty_overlap + penalty_boundary_l + penalty_boundary_r + penalty_boundary_b + penalty_boundary_t + penalty_non_negative

def objective(z):
    return -z[2::3].sum()

def gradient(z, mu):

    dr = 2 * mu * (np.where(z[2::3] + z[2::3] > d, 1, 0) + np.where(z[2::3] - z[0::3] > 0, 1, 0) + np.where(z[2::3] - (1 - z[0::3]) > 0, 1, 0) + np.where(z[2::3] - z[1::3] > 0, 1, 0) + np.where(z[2::3] - (1 - z[1::3]) > 0, 1, 0) + np.where(z[2::3] < 0, 1, 0))
    return np.concatenate((dx, dy, dr))
