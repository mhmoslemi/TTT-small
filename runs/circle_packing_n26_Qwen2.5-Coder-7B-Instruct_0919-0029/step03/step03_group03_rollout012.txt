import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def seed_configuration():
    centers = np.array([
        [0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75],
        [0.125, 0.5], [0.375, 0.5], [0.625, 0.5], [0.875, 0.5],
        [0.5, 0.125], [0.5, 0.375], [0.5, 0.625], [0.5, 0.875],
        [0.125, 0.375], [0.375, 0.375], [0.625, 0.375], [0.875, 0.375],
        [0.375, 0.125], [0.375, 0.625], [0.625, 0.125], [0.625, 0.875],
        [0.125, 0.875], [0.875, 0.875]
    ])
    return centers

def lpm(cpos):
    n = cpos.shape[0]
    d = squareform(pdist(cpos))
    bound = np.sqrt(d)-2*(1e-12)
    obj = np.ones(n)
    con1 = {'type': 'ineq', 'fun': lambda r: np.diag(bound) - r}
    con2 = {'type': 'ineq', 'fun': lambda r: r}
    res = linprog(c=obj, x0=np.zeros(n), A=con1['fun'](np.zeros(n)), b=con1['fun'](np.zeros(n)), method='highs')
    return res.x

def sa(cpos, maxiter=20000, sigma=0.001, alpha=0.9999, epsilon=1e-4):
    best_cpos = cpos.copy()
    best_obj = np.sum(lpm(cpos))
    current_cpos = cpos.copy()
    current_obj = best_obj
    T = 0.1
    while T > 1e-6:
        for _ in range(maxiter):
            i = np.random.randint(n)
            xi, yi = current_cpos[i]
            xi_new = np.clip(xi + np.random.normal(0, sigma), 0, 1)
            yi_new = np.clip(yi + np.random.normal(0, sigma), 0, 1)
            new_cpos = current_cpos.copy()
            new_cpos[i] = np.array([xi_new, yi_new])
            non_negative_con = (np.linalg.norm(new_cpos[i] - new_cpos[j]) >= new_cpos[i, 0] + new_cpos[i, 1] for j in range(n) if i != j)
            if all(non_negative_con):
                new_obj = np.sum(lpm(new_cpos))
                if new_obj > current_obj:
                    current_cpos[i] = np.array([xi_new, yi_new])
                    current_obj = new_obj
                else:
                    if np.random.uniform(0, 1) < np.exp((new_obj - current_obj) / T):
                        current_cpos[i] = np.array([xi_new, yi_new])
                        current_obj = new_obj
            if current_obj > best_obj:
                best_cpos = current_cpos.copy()
                best_obj = current_obj
        T *= alpha
    return best_cpos, best_obj

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = seed_configuration()
    radii = lpm(centers)
    current_obj = np.sum(radii)
    for _ in range(3):
        centers, current_obj = sa(centers, maxiter=20000, sigma=0.001, alpha=0.9999, epsilon=1e-4)
        radii = lpm(centers)
    return centers, radii, np.sum(radii)
