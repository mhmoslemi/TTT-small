import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    B = np.minimum(centers[:,0], 1 - centers[:,0])
    B = np.minimum(B, centers[:,1])
    B = np.minimum(B, 1 - centers[:,1])
    A = np.zeros((n*(n-1)//2, 3 * n))
    b = np.zeros(n*(n-1)//2)
    row_index = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row_index, 3*i] = 1
            A[row_index, 3*j] = 1
            A[row_index, 3*i + 2] = -1
            A[row_index, 3*j + 2] = -1
            b[row_index] = np.sqrt(np.sum((centers[i] - centers[j])**2)) - (B[i] + B[j])
            row_index += 1
    res = linprog(-B, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)] + [(0, 0.5) for _ in range(n, 2*n)], method='highs')
    radii = -res.x[n:]
    return radii, np.sum(radii)
