import numpy as np
from scipy.optimize import minimize

def solve_packing_LP(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    c = np.ones(n)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append(np.zeros(n))
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)

        A_ineq.append(np.zeros(n))
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)

        A_ineq.append(np.zeros(n))
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)

        A_ineq.append(np.zeros(n))
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append(np.zeros(n))
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    bounds = [(0, None) for _ in range(n)]
    result = minimize(lambda x: np.sum(x), np.ones(n), method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': lambda x: b_ineq - A_ineq @ x})
    return result.x

def run_packing():
    numstarts = 5
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centers = np.array([
            [5 * i / 4, 5 * j / 4] for i in range(4) for j in range(4)
        ])
        centers[25, :] = [0.5, 0.5]
        radii = solve_packing_LP(centers)
        sum_radii = np.sum(radii)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = radii

    centers, radii = best_centers, best_radii
    centers = centers.repeat((3, 3), axis=0)[:26, :]
    radii = [min(r, 0.5 - np.min([centers[i, 0] - r, centers[i, 1] - r, 1 - (centers[i, 0] + r), 1 - (centers[i, 1] + r)])) for i, r in enumerate(radii)]

    return centers, np.array(radii), np.sum(radii)

centers, radii, sum_radii = run_packing()
