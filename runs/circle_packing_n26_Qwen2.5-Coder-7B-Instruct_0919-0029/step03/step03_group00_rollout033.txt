import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_STARS = 30
    NUM_ITERS = 20000
    INITIAL_STEP = 0.1
    DECAY_STEP = 0.995
    T0 = 0.1
    TEMPERATURE_DECAY = 0.999

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(NUM_STARS):
        centers = np.random.uniform(0.05, 0.95, (n, 2))
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = INITIAL_STEP
        T = T0
        for iter in range(NUM_ITERS):
            idx = np.random.randint(n)
            direction = np.random.randn(2)
            direction /= np.linalg.norm(direction)
            delta = direction * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / T):
                cur_centers = new_c
                cur_radii, cur_sum = new_radii, new_sum
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers
                    best_radii = cur_radii

            step = max(step * DECAY_STEP, step * 0.1)
            T = max(T * TEMPERATURE_DECAY, T * 0.01)

    return best_centers, lp_optimum(best_centers)[0], best_sum
