import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def propose_move(centers, step):
    idx = np.random.randint(26)
    dir = np.random.randn(2)
    dir /= np.linalg.norm(dir)
    Δ = dir * step
    return idx, Δ

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.05
    step_decay = 0.995
    T = 1e-3 * 2.7
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500

    best_sum = -np.inf
    best_centers, best_radii = None, None

    patterns = [
        np.random.rand(n, 2) * 0.9 + 0.05,
        np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)]).flatten().reshape(5, 5, 2),
        np.array([[0.1 + 0.2*i, 0.1 + 0.2*j if i % 2 == 0 else 0.1 + 0.2*j + 0.1] for i in range(5) for j in range(5)]).flatten().reshape(5, 5, 2),
        np.array([[0.15 + 0.15*i, 0.13 + 0.13*j] for i in range(13) for j in range(18)]).flatten().reshape(13, 18, 2)[:26],
        np.vstack((np.random.rand(20, 2) * 0.9 + 0.05, np.array([[0.5, 0.5 + 0.15*i] for i in range(6)])))
    ]

    for start in range(len(patterns)):
        centers = patterns[start].copy()
        radius_sum = 0
        for it in range(ITER_PER_PATTERN):
            idx, Δ = propose_move(centers, step)
            new_centers = centers.copy()
            new_centers[idx] += Δ
            new_centers[idx] = np.clip(new_centers[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_centers)
            delta_sum = new_sum - radius_sum
            if delta_sum >= 0 or np.exp(delta_sum / T) > np.random.rand():
                centers, radius_sum = new_centers, new_sum
                if new_sum > best_sum:
                    best_sum, best_centers, best_radii = new_sum, centers.copy(), new_radii

            T *= T_decay
            step *= step_decay

        if it % MAX_NO_IMPROV == 0:
            centers, radius_sum = patterns[np.random.randint(len(patterns))].copy(), 0

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
