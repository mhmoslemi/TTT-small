import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d_ij - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def random_pattern():
    return np.random.rand(26, 2) * 0.89 + 0.05

def perturbed_grid_pattern():
    grid = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)])
    grid[0, 0] = np.random.rand(2) * 0.89 + 0.05
    return grid[:26]

def diagonal_shift_pattern():
    grid = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)])
    grid[::2, 0] += 0.1
    grid[26:] = np.random.rand(26, 2) * 0.89 + 0.05
    return grid[:26]

def rough_hexagon_pattern():
    rows = np.array([[0.075 * (2 + 2 * np.sin(i)) * (i % 2) + 0.175, 0.13 * (1 + np.cos(i)) + 0.035] for i in range(13)])
    rows = np.vstack([rows, rows + 0.15, rows + np.array([0.15, 0.0]), rows + np.array([0, 0.05, 0.15])])
    np.random.shuffle(rows)
    return rows[:26]

def random_plus_boundary_pattern():
    inner_points = np.random.rand(20, 2) * 0.89 + 0.05
    boundary_points = np.array([[0.15, 0.05] + np.random.randn(2) * 0.04 for _ in range(3)])
    return np.vstack([inner_points, boundary_points])

PATTERNS = [random_pattern, perturbed_grid_pattern, diagonal_shift_pattern, rough_hexagon_pattern, random_plus_boundary_pattern]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(5):
        pattern_fn = PATTERNS[start]
        centers = pattern_fn()
        radii, cur_sum = lp_optimum(centers)

        step, step_min = 0.05, 1e-4
        decay_factor = 0.995
        T = 1e-3 * cur_sum
        T_decay = 0.999
        max_no_improve = 500
        no_improve = 0

        for iter in range(2000):
            idx, dir, Delta = propose_move(centers, step)
            new_cent = centers.copy()
            new_cent[idx] += Delta
            new_cent[idx] = np.clip(new_cent[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_cent)

            Delta_sum = new_sum - cur_sum
            if Delta_sum >= 0 or np.exp(Delta_sum / T) > np.random.rand():
                centers, radii, cur_sum = new_cent, new_radii, new_sum
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            step *= decay_factor
            T *= T_decay

            if no_improve >= max_no_improve:
                step, step_min = 0.05, 1e-4
                T = 1e-3 * best_sum
                no_improve = 0
                centers, radii, cur_sum = pattern_fn(), lp_optimum(centers)[0:2]

    best_radii, best_sum = lp_optimum(best_centers)
    return best_centers, best_radii, best_sum

def propose_move(centers, step):
    idx = np.random.randint(26)
    dir = np.random.randn(2)
    dir /= np.linalg.norm(dir)
    Delta = dir * step
    return idx, Delta, Delta
