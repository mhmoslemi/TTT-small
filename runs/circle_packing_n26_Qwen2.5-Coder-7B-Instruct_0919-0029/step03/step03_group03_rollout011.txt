import numpy as np
from scipy.optimize import linprog, minimize


def compute_distances(positions):
    n = positions.shape[0]
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = positions[i, 0] - positions[j, 0]
            dy = positions[i, 1] - positions[j, 1]
            D[i, j] = D[j, i] = np.sqrt(dx**2 + dy**2)
    return D


def compute_bounds(positions):
    n = positions.shape[0]
    bounds = []
    for i in range(n):
        x, y = positions[i]
        r_min = min(x, 1 - x, y, 1 - y)
        bounds.append((0, r_min))
    return bounds


def evaluate_objective(positions, radii, distances):
    return np.sum(radii)


def objective_for_opt(positions, radii):
    return evaluate_objective(positions, radii, distances)


def constraint_for_opt(positions, radii):
    D = compute_distances(positions)
    L = np.zeros(26)
    for i in range(26):
        for j in range(i + 1, 26):
            L[i] += D[i, j] - radii[i] - radii[j]
    return {'type': 'ineq', 'fun': L}


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    positions = np.array([[0.5, 0.05],
                          [0.95, 0.5],
                          [0.05, 0.5],
                          [-1e-12, -1e-12],
                          [0.5, -1e-12],
                          [1 + 1e-12, -1e-12],
                          [-1e-12, 0.5],
                          [-1e-12, 1 + 1e-12],
                          [0.5, 1 + 1e-12],
                          [1 + 1e-12, 0.5],
                          [1 + 1e-12, 1 + 1e-12]])
    n = 26
    radii = np.ones(n) * 0.01
    D = compute_distances(positions)
    bounds = compute_bounds(positions)

    result = minimize(lambda r: 1 / evaluate_objective(positions, r, D),
                     x0=radii,
                     method='SLSQP',
                     bounds=bounds,
                     constraints=constraint_for_opt(positions, radii)[0])

    if not result.success:
        raise Exception("Optimization failed: ", result.message)

    radii = result.x
    positions = np.array([[np.random.uniform(0, 1), np.random.uniform(0, 1)] for _ in range(26)])
    for _ in range(20000):
        T = 0.1
        alpha = 0.9999
        Best_positions = positions
        Best_radii = radii
        Best_sum = 1 / evaluate_objective(positions, radii, D)
        while T > 10**(-6):
            i = np.random.randint(0, 26)
            updated_positions = positions.copy()
            updated_positions[i] += np.random.normal(0, T / 2, size=2)
            np.clip(updated_positions[i], 0, 1, out=updated_positions[i])

            if np.sum(np.sum((updated_positions - positions) ** 2, axis=1) < 10**(-4)) > 0:
                continue

            D_updated = compute_distances(updated_positions)
            result_updated = minimize(lambda r: 1 / evaluate_objective(updated_positions, r, D_updated),
                                      x0=radii,
                                      method='SLSQP',
                                      bounds=bounds,
                                      constraints=constraint_for_opt(updated_positions, radii)[0])

            if not result_updated.success:
                continue

            updated_radii = result_updated.x
            updated_sum = 1 / evaluate_objective(updated_positions, updated_radii, D_updated)

            if updated_sum > Best_sum and np.random.rand() < np.exp((updated_sum - Best_sum) / T):
                Best_positions = updated_positions
                Best_radii = updated_radii
                Best_sum = updated_sum

            t = np.random.rand()
            if t < alpha:
                positions = Best_positions
                radii = Best_radii

            T *= alpha

    return positions, radii, 1 / evaluate_objective(positions, radii, D)
