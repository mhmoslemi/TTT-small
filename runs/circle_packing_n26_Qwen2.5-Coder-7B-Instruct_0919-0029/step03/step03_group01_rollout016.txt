import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        r = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step_start = 0.1
    step_decay = 0.995
    T0 = 2.75  # Roughly one thousandth of optimum
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500

    best_centers, best_radii, best_sum = None, None, -np.inf

    patterns = [
        np.random.rand(n, 2),  # Random
        np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)]),
        np.array([[0.1 + 0.2*i + (i % 2) * 0.1, 0.1 + 0.2*j] for i in range(5) for j in range(5)]),
        np.array([
            [0.15 * (i % 2 + 1) + 0.075 * j * np.sqrt(3), 0.15 * (j % 2 + 1) + 0.075 * i]
            for i in range(20) for j in range(20) if (i + j) % 2 == 0
        ][:26]),
        np.random.rand(20, 2) + np.array([[0.15, 0.15], [0.15, 0.25], [0.15, 0.35], [0.25, 0.15], [0.25, 0.25], [0.25, 0.35], [0.35, 0.15], [0.35, 0.25], [0.35, 0.35]])
    ]

    for pattern in patterns:
        step = step_start
        T = T0
        centers = pattern
        radii, cur_sum = lp_optimum(centers)

        for _ in range(ITER_PER_PATTERN):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = centers[idx] + delta
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            Δsum = new_sum - cur_sum
            if Δsum >= 0 or np.exp(Δsum / T) > np.random.rand():
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                if cur_sum > best_sum:
                    best_centers, best_radii, best_sum = centers.copy(), new_radii.copy(), cur_sum

            T *= T_decay
            step *= step_decay

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
