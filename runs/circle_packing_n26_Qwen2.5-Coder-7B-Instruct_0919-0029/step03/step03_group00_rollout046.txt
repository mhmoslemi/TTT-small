import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for i inrange(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[i * (n - 1) // 2 + j - i - 1, i] = 1
            A_ub[i * (n - 1) // 2 + j - i - 1, j] = 1
            b_ub[i * (n - 1) // 2 + j - i - 1] = dist - 1e-9

    for i in range(n):
        A_ub[n * (n - 1) // 2 + 4 * i, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i] = 1 - centers[i, 0] - 1e-9
        A_ub[n * (n - 1) // 2 + 4 * i + 1, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 1] = centers[i, 0] - 1e-9
        A_ub[n * (n - 1) // 2 + 4 * i + 2, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 2] = 1 - centers[i, 1] - 1e-9
        A_ub[n * (n - 1) // 2 + 4 * i + 3, i] = 1
        b_ub[n * (n - 1) // 2 + 4 * i + 3] = centers[i, 1] - 1e-9

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def custom_init(n, bounds):
    hex_centers = np.array([
        [0.9, 0.9], [0.9, 0.1], [0.1, 0.9], [0.1, 0.1],
        [0.6, 0.9], [0.6, 0.1], [0.3, 0.3], [0.3, 0.7],
        [0.7, 0.3], [0.7, 0.7], [0.4, 0.95], [0.4, 0.05],
        [0.55, 0.45], [0.45, 0.45], [0.6, 0.15], [0.6, 0.85],
        [0.2, 0.4], [0.8, 0.4]
    ])
    num_hex = hex_centers.shape[0]
    centers = np.repeat(hex_centers, n // num_hex, axis=0)
    centers += np.random.randn(centers.shape[0], 2) * 0.01
    centers[:n, :] = (centers[:n, :] - np.min(centers[:, :], axis=0)) / (np.max(centers[:, :], axis=0) - np.min(centers[:, :], axis=0)) * 1
    return centers

def fitness(c):
    centers = c.reshape((26, 2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r

def local_sa(centers):
    cur_centers = centers.copy()
    cur_radii, cur_sum = lp_optimum(cur_centers)
    step = 0.05
    T = 0.05
    for iter in range(5000):
        idx = np.random.randint(26)
        move = np.random.randn(2) * step
        new_centers = cur_centers.copy()
        new_centers[idx] += move
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or np.random.random() < np.exp((new_sum - cur_sum) / T):
            cur_centers, cur_radii, cur_sum = new_centers, new_radii, new_sum
        step = max(1e-4, step * 0.995)
        T = max(1e-5, T * 0.995)
    return cur_centers, cur_radii, cur_sum

def run_packing():
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    init = custom_init
    tol = 1e-8
    workers = 1

    result = differential_evolution(fitness, bounds, popsize=popsize, maxiter=maxiter, strategy=strategy, init=init, tol=tol, workers=workers)
    if not result.success:
        result.x = np.random.rand(dim)
        result.fun = fitness(result.x)

    best_centers = result.x.reshape((26, 2))
    best_radii, best_sum = lp_optimum(best_centers)
    refined_centers, refined_radii, refined_sum = local_sa(best_centers)

    if refined_sum > best_sum:
        best_centers, best_radii, best_sum = refined_centers, refined_radii, refined_sum

    return best_centers, best_radii, best_sum
