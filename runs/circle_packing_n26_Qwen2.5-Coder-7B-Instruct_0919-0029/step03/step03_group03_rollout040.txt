import numpy as np
from scipy.optimize import linprog, Bounds

def solve_radii(centres):
    n = centres.shape[0]
    B = np.minimum(centres[:, 0], 1 - centres[:, 0])
    B = np.minimum(B, centres[:, 1])
    B = np.minimum(B, 1 - centres[:, 1])

    d_squared = np.sum((centres[i, :][:, np.newaxis] - centres[j, :][np.newaxis, :]) ** 2 for i in range(n) for j in range(i + 1, n))
    d = np.sqrt(d_squared)

    c = np.ones(n)
    A = np.zeros((n + n * (n - 1) // 2, n))
    b = np.zeros(n + n * (n - 1) // 2)

    A[:n, :] = np.eye(n)
    b[:n] = B

    k = n
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1
            A[k, j] = 1
            b[k] = d[i, j]
            k += 1

    res = linprog(c, A_eq=A, b_eq=b, bounds=Bounds(0, B), method='highs', tol=1e-12)
    return res.x, res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial hexagonal layout
    s = 0.099
    x_hex = np.array([])
    y_hex = np.array([])
    for i in range(5):
        if i < 2:
            j_range = 5
        elif i == 2:
            j_range = 4
        else:
            j_range = 4
        for j in range(j_range):
            x_hex = np.append(x_hex, s + j * 2 * s + i % 2 * s)
            y_hex = np.append(y_hex, s + i * np.sqrt(3) * s)
    x_side = np.array([0.5])
    y_side = np.array([0.05, 0.5, 0.5])
    x_hex = np.concatenate([x_hex, x_side])
    y_hex = np.concatenate([y_hex, y_side])

    centres = np.column_stack((x_hex, y_hex))
    radii, sum_radii = solve_radii(centres)

    # Simulated Annealing
    T = 0.1
    T_min = 1e-6
    sigma = T / 2.0
    epsilon = 1e-4
    while T > T_min:
        best_state = centres
        best_sum_radii = sum_radii
        for _ in range(100):
            j = np.random.randint(n)
            dx, dy = np.random.normal(0, sigma, 2)
            new_centres = centres.copy()
            new_centres[j] += np.array([dx, dy])
            new_centres[j] = np.clip(new_centres[j], 0, 1)
            if np.min(np.linalg.norm(new_centres - centres[j], axis=1)) >= epsilon:
                new_radii, new_sum_radii = solve_radii(new_centres)
                if new_sum_radii > sum_radii:
                    centres = new_centres
                    radii = new_radii
                    sum_radii = new_sum_radii
        if new_sum_radii > best_sum_radii:
            best_state = new_centres
            best_sum_radii = new_sum_radii
        if best_sum_radii > sum_radii:
            centres = best_state
            radii = best_state
            sum_radii = best_sum_radii

        sigma *= 0.99
        T *= 0.9999

    new_radii, new_sum_radii = solve_radii(centres)
