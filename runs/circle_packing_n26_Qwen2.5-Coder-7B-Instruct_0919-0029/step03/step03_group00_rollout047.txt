import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-9
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-9
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    dim = 52
    bounds = [(0,1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    seed_centers = generate_seed_centers()
    de_result = differential_evolution(neg_sum_of_radii, bounds, args=(seed_centers,), popsize=popsize, maxiter=maxiter, strategy=strategy)
    best_centers = de_result.x.reshape((26,2))
    local_center = local_sa(best_centers)
    return local_center, lp_optimum(local_center)[0], local_center.shape[0]

def generate_seed_centers():
    h_lattice = np.array([
        (0.26, 0.26), (0.52, 0.26), (0.78, 0.26),
        (0.26, 0.52), (0.52, 0.52), (0.78, 0.52),
        (0.26, 0.78), (0.52, 0.78), (0.78, 0.78),
        (0.3732, 0.65), (0.5247, 0.65), (0.6763, 0.65), # 3 rows
        (0.26, 0.4), (0.52, 0.4), (0.78, 0.4),
        (0.3732, 0.3), (0.5247, 0.3), (0.6763, 0.3),
        (0.26, 0.15), (0.52, 0.15), (0.78, 0.15)
    ], 'float32') * 0.9
    h_lattice = np.repeat(h_lattice[np.newaxis,:], 34, axis=0)
    noise = np.random.normal(0, 0.01, (34, 2))
    pop = h_lattice + noise
    return pop[:26]

def neg_sum_of_radii(x):
    centers = x.reshape((26,2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r

def local_sa(best_centers):
    cur_centers = best_centers.copy()
    cur_radius, cur_sum = lp_optimum(cur_centers)
    step = 0.05
    T = 0.05
    for it in range(5000):
        idx = np.random.randint(26)
        new_centers = cur_centers.copy()
        new_centers[idx] += np.random.randn(2) * step
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radius, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum)/T):
            cur_centers, cur_radius, cur_sum = new_centers, new_radius, new_sum
        step = max(1e-4, step * 0.995)
        T = max(1e-5, T * 0.995)
    return cur_centers
