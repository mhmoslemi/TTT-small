import numpy as np
from scipy.optimize import minimize

def objective(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return -np.sum(r)

def distance_constraint(z, i, j):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return r[i] + r[j] - np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)

def constraint(z):
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[i] - 0.1})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[i] - 0.1})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[26 + i] - 0.1})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - z[26 + i] - 0.1})
        for j in range(i + 1, 26):
            constraints.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: distance_constraint(z, i, j)})
    return constraints

def run_packing():
    # Warm start construction
    centers = np.zeros((26, 2))
    radius = 0.10
    step = 1 / np.sqrt(2)

    for i in range(26):
        if i < 9:
            centers[i, 0] = step * (2 * (i % 3) + 1)
        elif i < 18:
            centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
            centers[i, 1] = step + 0.5 * step
        else:
            centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
            centers[i, 1] = step + 0.5 * step * 2

    centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    centers *= 0.98
    centers += 0.01

    radius_center = np.sqrt(0.08) - radius
    centers[12, 0], centers[12, 1] = 0.5, 0.5
    radii = np.concatenate((np.full(25, radius), np.array([radius_center])))

    initial_vector = np.concatenate((centers.flatten(), radii))

    # Linear programming pre-step (optional but recommended)
    lp_constr = [{'type': 'ineq', 'fun': lambda z, i, j=i: np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2) - r[i] - r[j]} for i in range(26) for j in range(i+1, 26)]
    lp_constr += [{'type': 'ineq', 'fun': lambda z, i=i: r[i] - z[i*2]} for i in range(26)]
    lp_constr += [{'type': 'ineq', 'fun': lambda z, i=i: 1 - z[i*2] - r[i]} for i in range(26)]
    lp_constr += [{'type': 'ineq', 'fun': lambda z, i=i: r[i] - z[i*2+1]} for i in range(26)]
    lp_constr += [{'type': 'ineq', 'fun': lambda z, i=i: 1 - z[i*2+1] - r[i]} for i in range(26)]

    lp_res = minimize(objective, initial_vector, method='L-BFGS-B', bounds=[(0, None)] * 78, constraints=lp_constr)

    if lp_res.success and lp_res.fun > 2:
        initial_vector = lp_res.x

    # Non-linear optimization with trust-constr
    options = {
        'max_iters': 2000,
        'gtol': 1e-8,
        'xtol': 1e-8,
        'disp': False,
    }
    result = minimize(
        fun=objective,
        x0=initial_vector,
        method='trust-constr',
        constraints=constraint(initial_vector),
        options=options,
        jac=None,
        hess=None,
    )

    if not result.success:
        raise ValueError("Optimization failed with message: {}".format(result.message))

    # Post-processing
    x = result.x[:26]
    y = result.x[26:52]
    r = result.x[52:]
    r[r < 1e-12] = 0
    total_radius = np.sum(r)
    return x.reshape(26, 2), y.reshape(26,), total_radius
