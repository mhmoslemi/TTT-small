import numpy as np
from scipy.optimize import linprog, NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step_count = 20000
    cooling_rate = 0.999
    step_scale = 0.1
    r0 = 0.099
    B = np.ones(n) * r0

    # Define the initial centres of the circles with a heuristic layout
    centers = np.array([
        [0.495, 0.495],
        [0.35, 0.35],
        [0.65, 0.35],
        [0.35, 0.65],
        [0.65, 0.65],
        [0.25, 0.5],
        [0.75, 0.5],
        [0.5, 0.25],
        [0.5, 0.75],
        [0.15, 0.15],
        [0.15, 0.65],
        [0.15, 0.85],
        [0.35, 0.15],
        [0.35, 0.85],
        [0.85, 0.15],
        [0.85, 0.35],
        [0.85, 0.85],
        [0.65, 0.15],
        [0.65, 0.85],
        [0.05, 0.05],
        [0.05, 0.5],
        [0.05, 0.95],
        [0.5, 0.05],
        [0.5, 0.95],
        [0.95, 0.05],
        [0.95, 0.5],
        [0.95, 0.95]
    ])

    # Pairwise distance between centers
    distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=-1)

    # Upper limit on radius for each circle
    B = np.min([B, distances / 2 - 1e-6], axis=0)

    # Variables for the radii
    x = np.zeros(n)

    best_centers, best_x = centers, x
    best_sum = np.sum(best_x)

    # Temperature schedule
    T = 0.1

    for _ in range(step_count):
        # Pick a random circle and propose a new position
        i = np.random.randint(n)
        current_center = centers[i]
        new_center = current_center + np.random.randn(2) * step_scale
        new_center = np.clip(new_center, [0, 0], [1, 1])

        # Check if two circles are too close
        if np.any(np.linalg.norm(new_center[:, np.newaxis] - centers, axis=-1) < 1e-4):
            continue

        # Evaluate for the new configuration
        new_positions = np.copy(centers)
        new_positions[i] = new_center
        new_distances = np.linalg.norm(new_positions[:, np.newaxis] - new_positions[np.newaxis, :], axis=-1)
        new_b = np.min([B, new_distances / 2 - 1e-6], axis=0)
        lp_res = linprog(x, A_ub=np.vstack([new_b, new_distances]), b_ub=np.hstack([B + distances, B + distances]), method='highs')
        new_x_new = lp_res.x if lp_res.success else x

        # Compute the objective function for the new configuration
        new_sum = np.sum(new_x_new)

        # Acceptance decision
        if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
            centers, x = new_positions, new_x_new
            best_centers, best_x = centers, x
            best_sum = new_sum

        # Cool the temperature
        T *= cooling_rate

    # Final refinement using potentially finer gradient steps
    for _ in range(20):  # refinement iterations
        for i in range(n):
            original_center = centers[i]
            for axis in range(2):
                for direction in [-1, 1]:
                    # Perturb the center slightly in the current direction
                    perturbed_center = original_center.copy()
                    perturbed_center[axis] += direction * 1e-4
                    perturbed_center = np.clip(perturbed_center, [0, 0], [1, 1])

                    # Evaluate the LP
                    new_positions = np.copy(centers)
                    new_positions[i] = perturbed_center
                    new_distances = np.linalg.norm(new_positions[:, np.newaxis] - new_positions[np.newaxis, :], axis=-1)
                    new_b = np.min([B, new_distances / 2 - 1e-6], axis=0)
                    lp_res = linprog(x, A_ub=np.vstack([new_b, new_distances]), b_ub=np.hstack([B + distances, B + distances]), method='highs')
                    if lp_res.success:
                        new_x_perturbed = lp_res.x
                        new_sum_perturbed = np.sum(new_x_perturbed)

                        # Update the position if there is a gain
                        if new_sum_perturbed > best_sum:
                            centers[i] = perturbed_center
                            x = new_x_perturbed
                            best_sum = new_sum_perturbed

    return best_centers, best_x, best_sum
