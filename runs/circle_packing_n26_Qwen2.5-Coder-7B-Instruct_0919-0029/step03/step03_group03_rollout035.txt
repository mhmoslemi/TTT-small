import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    B = np.min(np.column_stack([centers[:, 0] - centers[:, 1], centers[:, 0] + centers[:, 1], centers[:, 1] - centers[:, 0], centers[:, 1] + centers[:, 0]]), axis=1)
    d = np.linalg.norm(np.diff(centers, axis=0), axis=1)
    A = np.hstack((np.eye(n), -np.eye(n), np.eye(n), -np.eye(n)))
    b = np.hstack((B, -B, d))
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None))
    if not res.success:
        return np.zeros(n), 0.0
    return res.x, np.sum(res.x)

def hex_grid_centers(n):
    r0 = 0.1
    side_space = 2 * r0
    height_space = np.sqrt(3) * r0
    num_rows, num_cols = 5, 4
    centers = np.zeros((n, 2))
    y_offset = 0.5 * height_space
    col_offset = 0.5 * side_space
    row_counter = 0
    for i in range(num_rows):
        for j in range(num_cols):
            x =col_offset * (j + (i % 2)) + 0.25 * height_space
            y = y_offset * i
            centers[row_counter, :] = [x, y]
            row_counter += 1
    return centers

def add_extra_positions(centers):
    new_centers = np.zeros((3, 2))
    new_centers[0, :] = [0.05, 0.5]
    new_centers[1, :] = [0.95, 0.5]
    new_centers[2, :] = [0.5, 0.95]
    return np.vstack((centers, new_centers))

def run_packing():
    centers = hex_grid_centers(23)
    centers = add_extra_positions(centers)
    radii, sum_radii = compute_max_radii(centers)
    return centers, radii, sum_radii

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
