import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            d_ij = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = d_ij - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n=26
    best_centers=None
    best_radii=None
    best_sum=-1
    patterns=[np.random.rand(n, 2), lambda: np.array([[i/5 + 0.1, j/5 + 0.1] for i in range(5) for j in range(5)] + [[0.5, 0.5]])]
    stars=20
    iters_per_star=4000
    step_start=0.1
    step_decay=0.995
    max_steps=1000
    T=1.0
    T_decay=0.999

    for pattern in patterns:
        cent = pattern()
        for i in range(stars):
            current_sum = -np.inf
            for _ in range(max_steps):
                idx = np.random.randint(n)
                dir = np.random.randn(2)
                dir /= np.linalg.norm(dir)
                delta = dir * step_start
                new_cent = cent.copy()
                new_cent[idx] = np.clip(new_cent[idx] + delta, 0, 1)
                new_radii, new_sum = lp_optimum(new_cent)
                if new_sum > best_sum or (new_sum > current_sum and np.random.rand() < np.exp((new_sum - current_sum) / T)):
                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_centers = new_cent.copy()
                        best_radii = new_radii.copy()
                    current_sum = new_sum
                    cent = new_cent.copy()
                else:
                    cent = cent.copy()
                step_start *= step_decay
            T *= T_decay

    return best_centers, best_radii, best_sum
