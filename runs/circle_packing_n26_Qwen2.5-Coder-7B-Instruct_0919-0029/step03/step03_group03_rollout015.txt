import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    radii_bounds = np.zeros((n, 2))
    for i in range(n):
        x, y = centers[i]
        radii_bounds[i] = (0, min(x, 1-x, y, 1-y))
    
    pairwise_dist = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    pairwise_dist -= radii_bounds[:, :, 0] + radii_bounds[:, :, 1] - 1e-12
    
    objective = np.ones(n)
    constraints = []
    for i in range(n):
        constraints.extend([
            {'type': 'ineq', 'fun': lambda z, i=i: np.minimum(z[2*i], z[2*i+1])},
            {'type': 'ineq', 'fun': lambda z, i=i: np.minimum(1 - z[2*i], 1 - z[2*i+1])}
        ])
        for j in range(i+1, n):
            constraints.append({
                'type': 'ineq', 
                'fun': lambda z, i=i, j=j: pairwise_dist[i, j] - (z[2*i+2] + z[2*j+2])
            })
    
    result = linprog(objective, A_ub=np.array([c['fun'](np.ones(3*n)) for c in constraints]), b_ub=np.zeros(len(constraints)), bounds=radii_bounds)
    if result.success:
        return result.x[2::3], np.sum(result.x[2::3])
    else:
        return np.zeros(n), 0.0

def run_packing():
    def hex_grid_centers(n):
        centers = []
        r0 = 0.099
        h = np.sqrt(3) * r0
        for i in range(n // 2 + 1):
            offset = 0 if i % 2 == 0 else r0
            for j in range(n // 2 + 1 - int(i % 2)):
                centers.append([offset + 2 * r0 * j + r0 * (1 - i % 2), i * h + r0])
        return np.array(centers)
    
    n = 26
    hex_centers = hex_grid_centers(n)
    hex_radii, hex_sum = compute_max_radii(hex_centers)
    
    additional_radii = np.ones(3) * 0.05
    additional_centers = np.array([[0.1, 0.5], [0.95, 0.5], [0.5, 0.05]])
    
    combined_centers = np.vstack([hex_centers, additional_centers])
    combined_radii = np.hstack([hex_radii, additional_radii])
    
    n_combined = combined_centers.shape[0]
    radii_bounds = np.zeros((n_combined, 2))
    for i in range(n_combined):
        x, y = combined_centers[i]
        radii_bounds[i] = (0, min(x, 1-x, y, 1-y))
    
    pairwise_dist = np.linalg.norm(combined_centers[:, np.newaxis, :] - combined_centers[np.newaxis, :, :], axis=2)
    pairwise_dist -= radii_bounds[:, :, 0] + radii_bounds[:, :, 1] - 1e-12
    
    objective = np.ones(n_combined)
    constraints = []
    for i in range(n_combined):
        constraints.extend([
            {'type': 'ineq', 'fun': lambda z, i=i: np.minimum(z[2*i], z[2*i+1])},
            {'type': 'ineq', 'fun': lambda z, i=i: np.minimum(1 - z[2*i], 1 - z[2*i+1])}
        ])
        for j in range(i+1, n_combined):
            constraints.append({
                'type': 'ineq', 
                'fun': lambda z, i=i, j=j: pairwise_dist[i, j] - (z[2*i+2] + z[2*j+2])
            })
    
    result = linprog(objective, A_ub=np.array([c['fun'](np.ones(3*n_combined)) for c in constraints]), b_ub=np.zeros(len(constraints)), bounds=radii_bounds)
    if result.success:
        final_radii = result.x[2::3]
    else:
        final_radii = np.zeros(n_combined)
    
    return combined_centers, final_radii, np.sum(final_radii)
