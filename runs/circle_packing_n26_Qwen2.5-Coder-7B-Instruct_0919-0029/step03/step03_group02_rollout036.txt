import numpy as np
from scipy.optimize import linprog

V = np.ones((26, 2), dtype=np.float64) * 0.01
target_radius = 0.15
eps = 1e-12
鲁棒性广播

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - eps)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 2 * eps)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def generate_perturbed_centers(seed, K):
    np.random.seed(seed)
    centers = np.random.rand(26, 2)
    perturbed_centers = centers + np.random.uniform(-0.02, 0.02, size=(26, 2))
    perturbed_centers = np.clip(perturbed_centers, 0, 1)
    perturbed_centers = perturbed_centers[centers == seed, :]
    for i in range(K):
        center = centers[i]
        perturbation = np.random.normal(0, 0.02, size=2)
        perturbation = perturbation / np.linalg.norm(perturbation) * np.clip(center - seed, 0, min(center, 1-center))
        new_center = center + perturbation
        new_center = np.clip(new_center, 0, 1)
        perturbed_centers = np.vstack((perturbed_centers, new_center))
    return perturbed_centers[:26]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    K = 5
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    for seed in range(K):
        centers = generate_perturbed_centers(seed, K)
        while True:
            try:
                radii, sum_radii = solve_lp(centers)
                break
            except Exception:
                pass
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers
            best_radii = radii
    return best_centers, best_radii, best_sum

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
