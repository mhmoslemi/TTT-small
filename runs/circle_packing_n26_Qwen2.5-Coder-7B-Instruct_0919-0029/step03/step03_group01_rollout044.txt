import numpy as np
from scipy.optimize import linprog

def evaluate_radii_and_sum(centers, radii):
    n = centers.shape[0]
    pairwise_dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
    overlap = np.any(pairwise_dists - radii[:, np.newaxis] - radii[np.newaxis, :] < -1e-12)
    if overlap:
        return -np.inf, None
    boundary = np.any(radii > np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]) - 1e-12)
    if boundary:
        return -np.inf, None
    total_radius = np.sum(radii)
    return total_radius, radii

def lp_optimisation(centers):
    n = centers.shape[0]
    A = np.zeros((n*n, n))
    b = np.zeros(n*n)

    # Boundary constraints
    for i in range(n):
        A[i, i] = 1
        b[i] = centers[i, 0] - 1e-12
        A[i+n, i] = 1
        b[i+n] = 1 - centers[i, 0] - 1e-12
        A[i+2*n, i] = 1
        b[i+2*n] = centers[i, 1] - 1e-12
        A[i+3*n, i] = 1
        b[i+3*n] = 1 - centers[i, 1] - 1e-12

    # Pairwise distance constraints
    index = 4*n
    for i in range(n):
        for j in range(i+1, n):
            A[index, i] = 1
            A[index, j] = 1
            b[index] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            index += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), -np.inf

def generate_random_pattern():
    return np.random.rand(26, 2) * 0.9 + 0.05

def propose_random_centre_move(centers, step):
    i = np.random.randint(26)
    d = np.random.randn(2) / np.linalg.norm(np.random.randn(2))
    return (i, step * d)

def run_packing():
    step = 0.05
    step_decay = 0.995
    T0 = 1e-3 * 3 * 2.7  # Estimate of upper bound
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500
    EPS = 1e-12

    best_centers, best_radii, best_sum = None, None, -np.inf

    patterns = [
        generate_random_pattern,
        lambda: generate_random_pattern().reshape(-1, 2).reshape(5, 5, 2).flatten()[:-1],
        lambda: generate_random_pattern().reshape(-1, 2).reshape(5, 5, 2).flatten()[:-2][::2],
        lambda: generate_random_pattern().reshape(-1, 2).reshape(15, 15, 2).flatten()[33:],  # Adjust for 26 points
        lambda: np.vstack((np.random.rand(20, 2) * 0.9 + 0.05, np.array([[k*0.2*0.075 + 0.1 + (k % 2) * 0.1, 0.3 + 0.15 * np.sqrt(3) * (k // 2)] for k in range(6)])
    ])

    for _ in range(5):
        centers = patterns[_]()
        for _ in range(ITER_PER_PATTERN):
            idx, step_vector = propose_random_centre_move(centers, step)
            centers[idx] += step_vector
            centers[idx] = np.clip(centers[idx], 0, 1)
            new_radii, new_sum = lp_optimisation(centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii
            step *= step_decay
            T0 *= T_decay
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii
    
    final_radii, final_sum = lp_optimisation(best_centers)
    return best_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
