import numpy as np
from scipy.optimize import linprog

T0 = 0.1
Tf = 1e-4
alpha = 0.9995
delta = 0.1
steps_per_T = 500
max_iter = steps_per_T * int(np.log(T0 / Tf) / np.log(alpha))

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    numstarts = 5
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centers = np.zeros((26, 2))
        radius = 0.14
        step = 1 / np.sqrt(2)

        for i in range(26):
            if i < 9:
                centers[i, 0] = step * (2 * (i % 3) + 1)
            elif i < 18:
                centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
                centers[i, 1] = step + 0.5 * step
            else:
                centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
                centers[i, 1] = step + 0.5 * step * 2

        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
        centers -= np.min(centers, axis=0)
        centers /= np.max(centers, axis=0)
        centers *= 0.98  # make safe from boundary issues
        centers += 0.01  # simple clamping within range

        try:
            radii, sum_radii = solve_lp(centers)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers
                best_radii = radii
        except Exception as e:
            pass  # accept failure to escape; assume only permissible failures occur

    try:
        final_radii, final_sum = solve_lp(best_centers)
    except Exception as e:
        return np.zeros((26, 2)), np.zeros(26), 0.0  # degenerate

    return best_centers, final_radii, final_sum
