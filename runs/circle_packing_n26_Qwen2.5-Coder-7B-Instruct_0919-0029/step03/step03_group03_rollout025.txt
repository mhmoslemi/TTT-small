import numpy as np
from scipy.optimize import linprog
from scipy.spatial import distance_matrix


def solve_radii(centers):
    n = centers.shape[0]
    distances = distance_matrix(centers, centers)

    B = np.min(np.column_stack((centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])), axis=1)

    c = -np.ones(n)
    A = np.zeros_like(distances)
    for i in range(n):
        A[i, np.arange(n) != i] = distances[i, np.arange(n) != i]
        A[n + i, i] = 1
    b = distances.flatten() - np.ones(distances.size)

    bounds = np.array([[0, B[i]] for i in range(n)])

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'tol': 1e-12})

    if result.success:
        return result.x, -result.fun
    else:
        return None, -np.inf


def run_packing():
    # 52 point grid: 26 hexagon plus 3 additional midpoints
    coords = np.zeros((26, 2))
    coords[:5, 0] = np.linspace(0.1, 0.9, 5, endpoint=False)
    coords[:5, 1] = 0.1 * np.array([0, 1, 2, 1, 0])
    coords[5:9, 0] = np.linspace(0.2, 0.8, 4, endpoint=False)
    coords[5:9, 1] = 0.25 * np.array([2, 3, 2, 3])
    coords[9:19, 0] = np.linspace(0.05, 0.95, 5, endpoint=True)
    coords[9:19, 1] = 0.4 * np.array([1, 1, 2, 2, 1])
    coords[19:22, 0] = np.linspace(0.25, 0.75, 3, endpoint=True)
    coords[19:22, 1] = 0.55 * np.array([2, 3, 2])
    coords[22:26, 0] = np.linspace(0.35, 0.65, 4, endpoint=True)
    coords[22:26, 1] = 0.7 * np.array([1, 1, 2, 2])
    center_pad = np.array([[0.50, 0.05], [0.95, 0.50], [0.05, 0.50]])
    coords = np.vstack((coords, center_pad))

    # SA parameters
    T = 0.1
    alpha = 0.9999
    num_steps = 20000
    epsilon = 1e-4
    delta = 1e-4

    best_score = -np.inf
    best_centers = None

    for _ in range(num_steps):
        centers = coords + np.random.randn(52, 2) * (T / 2)
        centers = np.clip(centers, 0, 1)

        # Quick infeasibility filter
        distances = distance_matrix(centers, centers)
        if np.any(distances < epsilon):
            continue

        radii, score = solve_radii(centers)
        if score > best_score:
            best_score = score
            best_centers = centers.copy()

        if np.random.rand() < np.exp((score - best_score) / T):
            coords = centers
        else:
            continue

        T *= alpha

    # Final polishing step
    for i in range(52):
       (pos_i,) = np.where(np.argmin(np.sum(np.abs(centers - centers[i, np.newaxis]), axis=1)))
        if pos_i != i:
            dist_i, dist_j = np.linalg.norm(centers[i] - centers[pos_i]), np.linalg.norm(centers[i] - centers[i, np.newaxis])
            if dist_i < dist_j - 1e-10:
                score += dist_i - dist_j
                radii = np.where(np.arange(52) == i, dist_i, radii)
                radii = np.where(np.arange(52) == pos_i, dist_j - dist_i, radii)
                centers[i] = centers[pos_i] + (centers[i] - centers[pos_i]) * (radii[i] / dist_i)

    return best_centers[:26, :], radii[:26], np.sum(radii[:26])
