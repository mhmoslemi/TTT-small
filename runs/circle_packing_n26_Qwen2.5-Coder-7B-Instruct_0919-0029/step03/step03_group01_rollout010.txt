import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def pattern_5x5_with_center():
    grid = np.array([[0.1 + 0.2 * i for i in range(5)] for _ in range(5)]).flatten()
    center = np.array([0.5, 0.5])
    centers = np.vstack((grid.reshape(-1, 2), center))
    return centers[:26]

def pattern_6x5():
    r = 0.075
    h_dist = 2 * r
    v_dist = np.sqrt(3) * r
    x_positions = [r + 2 * r * i for i in range(5)]
    y_positions = [r + v_dist * i for i in range(5)]
    centers = np.array([(x, y) for y in y_positions for x in x_positions])
    centers = centers[:25]
    center = np.array([0.5, 0.5])
    return np.vstack((centers, center))

def run_packing():
    n = 26
    best_centers, best_radii = None, None
    best_sum = -np.inf
    patterns = [pattern_5x5_with_center, pattern_6x5]
    stars = 5
    iters_per_star = 2000
    step_start = 0.05
    step_decay = 0.995
    epsilon = 1e-12

    for pattern in patterns:
        centroids = pattern()
        step = step_start
        for _ in range(stars):
            center = centroids.copy()
            current_sum, _ = lp_optimum(center)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = center.copy()
            for i in range(iters_per_star):
                idx = np.random.randint(n)
                dir = np.random.randn(2)
                dir /= np.linalg.norm(dir)
                delta = dir * step
                new_cent = center.copy()
                new_cent[idx] += delta
                new_cent[idx] = np.clip(new_cent[idx], 0, 1)
                new_radii, new_sum = lp_optimum(new_cent)
                if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / epsilon):
                    center = new_cent.copy()
                    current_sum = new_sum
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = center.copy()
            step /= step_decay

    if best_centers is None:
        return np.zeros((n, 2)), np.zeros(n), 0

    best_radii, _ = lp_optimum(best_centers)
    return best_centers, best_radii, best_sum

# Run the packer
centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
