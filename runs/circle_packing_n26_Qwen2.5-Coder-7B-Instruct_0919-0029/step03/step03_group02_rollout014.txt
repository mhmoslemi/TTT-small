import numpy as np
from math import sqrt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define grid coordinates for 5x5 lattice
    centers = np.zeros((26, 2))
    for i in range(5):
        for j in range(5):
            centers[i*5 + j, 0] = 0.1 + 0.2 * i
            centers[i*5 + j, 1] = 0.1 + 0.2 * j
    
    # Add central circle
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    r_new = sqrt(0.08) - 0.10
    
    # Set radii
    radii = np.zeros(26)
    radii[:25] = 0.10
    radii[25] = r_new
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
