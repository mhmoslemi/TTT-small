import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def contact_graph_lp(centers, active_pairs):
    n = centers.shape[0]
    A_ub = np.zeros((len(active_pairs) + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i, j in active_pairs:
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
        idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    A_slack = np.zeros((n*(n-1)//2 - len(active_pairs), n))
    b_slack = np.full(A_slack.shape[0], 1)
    A lucb = np.vstack((A_ub, A_slack))
    b_ub = np.hstack((b_ub, b_slack))

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        return np.zeros(n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_ITERS = 4000
    NUM_STARS = 12

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        cur_centers = centers.copy()
        active_pairs = set()
        while True:
            cur_radii, cur_sum = contact_graph_lp(cur_centers, active_pairs)
            step = 0.1
            improved = False
            for iter in range(NUM_ITERS):
                idx = np.random.randint(n)
                dir = np.random.randn(2)
                dir /= np.linalg.norm(dir)
                delta = dir * step
                new_c = cur_centers.copy()
                new_c[idx] += delta
                new_c[idx] = np.clip(new_c[idx], 0, 1)
                new_active_pairs = set()
                for i, j in active_pairs:
                    if i != idx and j != idx:
                        new_active_pairs.add((i, j))
                else:
                    new_active_pairs.add((idx, i))
                    new_active_pairs.add((idx, j))
                new_centers = new_c
                new_active_pairs.union(set((i, j) for i in range(n) for j in range(i+1, n)))
                new_radii, new_sum = contact_graph_lp(new_centers, new_active_pairs)
                if new_sum > cur_sum:
                    cur_centers = new_c
                    cur_radii, cur_sum = new_radii, new_sum
                    improved = True
                    break
            if not improved:
                break

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers

    return best_centers, contact_graph_lp(best_centers, set((i, j) for i in range(n) for j in range(i+1, n))), best_sum
