import numpy as np
from scipy.optimize import minimize_with_hess
from scipy.optimize.nonlin import l_bfgs_b

def penalty(z, mu):
    n = len(z) // 3
    overlap = 0
    bound = 0
    nzradii = 0
    for i in range(n):
        ri = z[3*i + 2]
        for j in range(i+1, n):
            rj = z[3*j + 2]
            dij = np.sqrt((z[3*i] - z[3*j]) ** 2 + (z[3*i + 1] - z[3*j + 1]) ** 2)
            overlap += np.maximum(0, ri + rj - dij) ** 2
        bound += np.maximum(0, ri - z[3*i]) ** 2 + np.maximum(0, ri - (1 - z[3*i])) ** 2 + np.maximum(0, ri - z[3*i + 1]) ** 2 + np.maximum(0, ri - (1 - z[3*i + 1])) ** 2
        nzradii += np.maximum(0, -ri) ** 2
    return mu * (overlap + bound + nzradii)

def objective(z, mu):
    return -np.sum(z[2::3]) + penalty(z, mu)

def gradient(z, mu):
    n = len(z) // 3
    grad = np.zeros_like(z)
    grad[2::3] = -1
    for i in range(n):
        ri = z[3*i + 2]
        for j in range(i+1, n):
            rj = z[3*j + 2]
            dij = np.sqrt((z[3*i] - z[3*j]) ** 2 + (z[3*i + 1] - z[3*j + 1]) ** 2)
            sij = ri + rj - dij
            grad[3*i] += np.maximum(0, sij) * (z[3*i] - z[3*j]) / dij
            grad[3*i + 1] += np.maximum(0, sij) * (z[3*i + 1] - z[3*j + 1]) / dij
            grad[3*i + 2] += 2 * np.maximum(0, sij)
            grad[3*j] += np.maximum(0, sij) * (z[3*j] - z[3*i]) / dij
            grad[3*j + 1] += np.maximum(0, sij) * (z[3*j + 1] - z[3*i + 1]) / dij
            grad[3*j + 2] += 2 * np.maximum(0, sij)
        grad[3*i] += np.maximum(0, ri - z[3*i]) * (-1)
        grad[3*i + 1] += np.maximum(0, ri - (1 - z[3*i])) * (-1)
        grad[3*i + 2] += 2 * np.maximum(0, ri)
        grad[3*i + 1] += np.maximum(0, ri - (1 - z[3*i + 1])) * (-1)
        grad[3*i + 2] += 2 * np.maximum(0, ri)
        grad[3*i + 1] += np.maximum(0, ri - (1 - z[3*i + 1])) * (-1)
        grad[3*i + 2] += 2 * np.maximum(0, ri)
    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_positions = np.random.uniform(0.05, 0.95, (n, 2))
    initial_radii = np.full(n, 0.05)
    initial_x = initial_positions[:, 0]
    initial_y = initial_positions[:, 1]
    initial_z = np.hstack([initial_x, initial_y, initial_radii])
    best_x, best_y, best_r = initial_x, initial_y, initial_radii
    best_sum = np.sum(best_r)
    mu = 10
    for _ in range(25):
        result = minimize_with_hess(objective, initial_z, args=(mu,), method='L-BFGS-B', jac=gradient, bounds=np.concatenate(([(0, 1)] * n, [(0, None)] * n)))
        x, y, r = result['x'][0:n], result['x'][n:2*n], result['x'][2*n:]
        if np.any(result['success']):
            if np.sum(r) > best_sum:
                best_sum = np.sum(r)
                best_x, best_y, best_r = x, y, r
        alpha = 0.1
        for _ in range(100):
            i = np.random.randint(n)
            new_x = x[i] + alpha * (np.random.randn(2) * 0.01)
            new_x = np.clip(new_x, 0, 1)
            new_r = r[i] + alpha * (0.005 * np.random.randn(1))
            new_r = np.clip(new_r, 0, None)
            new_z = np.hstack([x, y, r])
            new_z[3*i] = new_x[0]
            new_z[3*i + 1] = new_x[1]
            new_z[3*i + 2] = new_r
            result = minimize_with_hess(objective, new_z, args=(mu,), method='L-BFGS-B', jac=gradient, bounds=np.concatenate(([(0, 1)] * n, [(0, None)] * n)))
            x, y, r = result['x'][0:n], result['x'][n:2*n], result['x'][2*n:]
            if np.any(result['success']):
                if np.sum(r) > best_sum:
                    best_sum = np.sum(r)
                    best_x, best_y, best_r = x, y, r
            alpha *= 0.9
            if alpha <= 0:
                break
        if np.all(penalty(np.hstack([best_x, best_y, best_r]), 1e-7) < 1e-7):
            mu *= 10
        else:
            break
    return best_x, best_y, best_r, np.sum(best_r)
