import numpy as np
from scipy.optimize import minimize

def constraint_circles(i, j, centers, radii):
    return np.linalg.norm(centers[i] - centers[j]) - 1e-12 - (radii[i] + radii[j])

def constraint_x(i, centers, radii):
    return centers[i, 0] - 1e-12 - radii[i]

def constraint_y(i, centers, radii):
    return 1 - centers[i, 1] - 1e-12 - radii[i]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    num_stars = 10
    num_iterations = 2000
    learning_rate = 0.001

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(num_stars):
        def objective(centers_and_radii):
            centers = centers_and_radii[:2*n]
            centers = centers.reshape((n, 2))
            radii = centers_and_radii[2*n:]
            sum_radii = np.sum(radii)
            return -sum_radii

        def grad_objective(centers_and_radii):
            centers = centers_and_radii[:2*n]
            centers = centers.reshape((n, 2))
            radii = centers_and_radii[2*n:]
            grad_centers = np.zeros((2*n, 2))
            grad_radii = -np.ones(2*n)
            for i in range(n):
                for j in range(i+1, n):
                    grad_centers[i] += (centers[i] - centers[j]) / max(np.linalg.norm(centers[i] - centers[j]), 1e-12) * grad_radii[2*i] + (centers[j] - centers[i]) / max(np.linalg.norm(centers[i] - centers[j]), 1e-12) * grad_radii[2*j]
                    grad_centers[j] += (centers[j] - centers[i]) / max(np.linalg.norm(centers[i] - centers[j]), 1e-12) * grad_radii[2*i] + (centers[i] - centers[j]) / max(np.linalg.norm(centers[i] - centers[j]), 1e-12) * grad_radii[2*j]
            return np.ravel(grad_centers + grad_radii[2*n:].reshape((-1, 2)))

        centers = np.random.rand(n, 2)
        radii = np.random.rand(n)

        centers_and_radii = np.concatenate((centers.flatten(), radii))

        bounds = [(0, 1)] * (2*n) + [(0, None)] * n
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                constraints.append({'type': 'ineq', 'fun': lambda centers_and_radii, i=i, j=j: constraint_circles(i, j, centers_and_radii[:2*n].reshape((n, 2)), centers_and_radii[2*n:])})
            constraints.append({'type': 'ineq', 'fun': lambda centers_and_radii, i=i: constraint_x(i, centers_and_radii[:2*n].reshape((n, 2)), centers_and_radii[2*n:])})
            constraints.append({'type': 'ineq', 'fun': lambda centers_and_radii, i=i: constraint_y(i, centers_and_radii[:2*n].reshape((n, 2)), centers_and_radii[2*n:])})

        result = minimize(objective, centers_and_radii, method='SLSQP', jac=grad_objective, bounds=bounds, constraints=constraints)

        if result.fun > best_sum:
            best_sum = result.fun
            best_centers = result.x[:2*n].reshape((n, 2))
            best_radii = result.x[2*n:]

    return best_centers, best_radii, -best_sum
