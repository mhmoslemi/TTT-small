# Helper function to compute pairwise distances
def dist_matrix(centers):
    n = len(centers)
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            D[i, j] = np.linalg.norm(centers[i] - centers[j])
            D[j, i] = D[i, j]
    return D
