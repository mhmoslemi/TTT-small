import numpy as np
from scipy.optimize import minimize, linprog

def calc_constraints(positions):
    n = positions.shape[0]
    centers = positions.reshape(n//2, 2)
    radii = np.abs(positions[2*n//2:])
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    inequalities = []

    for i in range(n):
        inequalities.append(-radii[i])  # Non-negativity of radii
        inequalities.append(centers[i, 0] - radii[i])  # Left boundary
        inequalities.append(1 - centers[i, 0] - radii[i])  # Right boundary
        inequalities.append(1 - centers[i, 1] - radii[i])  # Bottom boundary
        inequalities.append(centers[i, 1] - radii[i])  # Top boundary

        for j in range(i + 1, n):
            inequalities.append(radii[i] + radii[j] - dists[i, j])  # Non-overlap

    return np.array(inequalities).T

def objective(positions):
    n = positions.shape[0]
    return -sum(positions[2*n//2:])  # Negative sum of radii for maximization

def run_packing():

    best_centers, best_sum, best_radii = None, -np.inf, None

            result = minimize(objective, positions, method='trust-constr',

                             options={'maxiter': 2000, 'gtol': 1e-8, 'xtol': 1e-8})
