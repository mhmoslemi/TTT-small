import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_eq, b_eq = [], []

    for i in range(n):
        A_eq.append([0] * n)
        A_eq[-1][i] = 1
        b_eq.append(centers[i, 0] - 1e-12)
        A_eq.append([0] * n)
        A_eq[-1][i] = 1
        b_eq.append(1 - centers[i, 0] - 1e-12)
        A_eq.append([0] * n)
        A_eq[-1][i] = 1
        b_eq.append(centers[i, 1] - 1e-12)
        A_eq.append([0] * n)
        A_eq[-1][i] = 1
        b_eq.append(1 - centers[i, 1] - 1e-12)

    A_eq = np.array(A_eq)
    b_eq = np.array(b_eq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")
