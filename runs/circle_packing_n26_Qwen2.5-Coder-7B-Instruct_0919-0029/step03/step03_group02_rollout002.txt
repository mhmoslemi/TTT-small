import numpy as np
from scipy.optimize import minimize, trust_constr

def sum_of_radii(z):
    return -np.sum(z[52:])

def ineq_constraints(z):
    n = 26
    x, y, r = z[:26], z[26:52], z[52:]
    contraints = []

    for i in range(n):
        contraints.append(-r[i])  # Non-negativity of radii
        contraints.append(-r[i] + x[i])  # Boundary constraint x
        contraints.append(-r[i] + (1 - x[i]))  # Boundary constraint x
        contraints.append(-r[i] + y[i])  # Boundary constraint y
        contraints.append(-r[i] + (1 - y[i]))  # Boundary constraint y

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            contraints.append(r[i] + r[j] - dist)

    return contraints

def run_packing():
    numstarts = 5
    best_centers, best_sum, best_radii = None, -np.inf, None

    for _ in range(numstarts):
        centers = np.zeros((26, 2))

            elif i < 18:
                centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
                centers[i, 1] = step + 0.5 * step
            else:
                centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
                centers[i, 1] = step + 0.5 * step * 2

        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))

        centers += 0.01  # simple clamping within range

        initial_guess = centers.flatten().tolist() + (0.7 * np.ones(26)).tolist()
