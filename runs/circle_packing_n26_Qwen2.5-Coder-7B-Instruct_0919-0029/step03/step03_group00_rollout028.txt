import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def halton_sequence(size, base):
    seq = []
    f = 1
    for i in range(1, size+1):
        f /= base
        seq.append(i * f)
    return seq

def generate_initial_centers(n):
    centers = np.zeros((n, 2))
    halton_x = halton_sequence(n, 2) * 0.9 + 0.05
    halton_y = halton_sequence(n, 5) * 0.9 + 0.05
    centers[:, 0] = halton_x
    centers[:, 1] = halton_y
    return centers

def run_packing():
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(30):
        centers = generate_initial_centers(26)
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = 0.1
        T = 0.05
        for iter in range(20000):
            idx = np.random.randint(26)
            new_pos = cur_centers[idx] + step * (np.random.randn(2))
            new_pos = np.clip(new_pos, 0, 1)
            new_centers = cur_centers.copy()
            new_centers[idx] = new_pos
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / T):
                cur_centers, cur_radii, cur_sum = new_centers, new_radii, new_sum
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers
                    best_radii = cur_radii
            step = max(1e-4, step * 0.99)
            T = max(1e-10, T * 0.999)

    return best_centers, best_radii, best_sum
