import numpy as np
from scipy.optimize import linprog, differential_evolution, Bounds

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, 2*n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, 2*i:2*i+2] = centers[i] - centers[j]
            A_ub[idx, 2*j:2*j+2] = centers[j] - centers[i]
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, 2*i:2*i+2] = [1, 0]
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, 2*i:2*i+2] = [-1, 0]
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, 2*i:2*i+2] = [0, 1]
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, 2*i:2*i+2] = [0, -1]
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(2*n)
    bounds = [(0, None)] * 2*n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        radii = np.sqrt(result.x[::2] ** 2 + result.x[1::2] ** 2)
        return radii, -result.fun
    else:
        return np.zeros(n), 0

def generate_initial_population(n, popsize):
    centers = np.empty((popsize, 2*n))
    for i in range(30):
        x, y = i // 5 + 0.5, i % 5
        d_x, d_y = 0.18, np.sqrt(3) * 0.18
        for j in range(4):
            centers[i, 2*j:2*j+2] = [x - j*d_y/2, y + j*d_x]
    for j in range(4):
        centers[30+j*4, 2*j:2*j+2] = [0.03, 0.03 + j * 0.34]
    centers[34:, :] += np.random.normal(0, 0.01, (popsize-34, 2*n))
    np.clip(centers, 0, 1, out=centers)
    return centers

def neg_sum_of_radii(x):
    centers = x.reshape((26,2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r

def local_sa(current, n=5000):
    step = 0.05
    T = 0.1
    for _ in range(n):
        idx = int(np.random.rand() * len(current) // 2)
        direction = np.random.randn(2)
        direction /= np.linalg.norm(direction)
        move = direction * step
        new = current.copy()
        new[idx*2:idx*2+2] += move
        new = np.clip(new, 0, 1)
        new_radii, new_sum = lp_optimum(new.reshape((26, 2)))
        if new_sum > current.sum() or np.exp((new_sum - current.sum()) / T) > np.random.rand():
            current = new
            step = max(1e-4, step * 0.995)
            T = max(1e-5, T * 0.995)
    return current

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    popsize = 15
    maxiter = 200
    bounds = Bounds(low=0, high=1)
    strategy = 'best1bin'
    init = 'latinhypercube'

    pop = generate_initial_population(n, popsize)
    result = differential_evolution(neg_sum_of_radii, bounds=[(0, 1)] * (2 * n), 
                                     init=pop, maxiter=maxiter, tol=1e-8, 
                                     strategy=strategy, popsize=popsize, 
                                     polish=False, workers=1, seed=42)

    centers = result.x.reshape((26, 2))
    current = centers.flatten()
    optimized_centers = local_sa(current)
    radii, sum_r = lp_optimum(optimized_centers.reshape((26, 2)))

    return optimized_centers.reshape((26, 2)), radii, sum_r
