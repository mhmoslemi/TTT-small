import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

def objective(z):
    n = 26
    return -np.sum(z[52:52+n])  # Sum of radii (negative for optimization)

def ineq_constraints(z):
    n = 26
    constraints = []
    
    # Non-negativity of radii
    for i in range(n):
        constraints.append(-z[52+i])
    
    # Boundary constraints
    for i in range(n):
        constraints.append(z[52+i] - z[i*2])
        constraints.append(z[52+i] - (1 - z[i*2]))
        constraints.append(z[52+i] - z[i*2+1])
        constraints.append(z[52+i] - (1 - z[i*2+1]))
    
    # Pairwise non-overlap
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((z[i*2] - z[j*2])**2 + (z[i*2+1] - z[j*2+1])**2)
            constraints.append(z[52+i] + z[52+j] - d)

    return constraints

def run_packing():
   centers = np.zeros((26, 2))
    radius = 0.14
    step = 1 / np.sqrt(2)

    for i in range(26):
        if i < 9:
            centers[i, 0] = step * (2 * (i % 3) + 1)
        elif i < 18:
            centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
            centers[i, 1] = step + 0.5 * step
        else:
            centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
            centers[i, 1] = step + 0.5 * step * 2

    centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    centers *= 0.98  # make safe from boundary issues
    centers += 0.01  # simple clamping within range

    initial_radii = [0.10] * 25 + [np.sqrt(0.08) - 0.10]  # Central circle

    x_bounds = [(0, 1)] * 52
    r_bounds = [(0, 1)] * 26

    x_bounds.extend(r_bounds)

    result = minimize(objective, np.hstack((centers.flatten(), initial_radii)), 
                      bounds=x_bounds, constraints=ineq_constraints)

    if result.success:
        centers = result.x[:52].reshape(26, 2)
        radii = result.x[52:]
        return centers, radii, np.sum(radii)
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0
