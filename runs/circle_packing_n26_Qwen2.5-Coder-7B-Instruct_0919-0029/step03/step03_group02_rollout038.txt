import numpy as np
from math import sqrt, pi

def run_packing():
    # Define grid size and spacing
    grid_size = 5
    spacing = 0.2
    centerX, centerY = 0.5, 0.5

    # Initialize centers and radii
    centers = np.array([
        (0.1 + i * spacing, 0.1 + j * spacing) for i in range(grid_size) for j in range(grid_size)
    ])
    radii = np.full(grid_size * grid_size, 0.1)
    radii[-1] = sqrt(0.08) - 0.1  # Central circle radius

    # Append central circle's center and radius
    centers = np.vstack((centers, (centerX, centerY)))
    radii = np.append(radii, radii[-1])

    # Calculate sum of radii
    sum_radii = np.dot(radii, np.ones(len(radii)))

    return centers, radii, sum_radii
