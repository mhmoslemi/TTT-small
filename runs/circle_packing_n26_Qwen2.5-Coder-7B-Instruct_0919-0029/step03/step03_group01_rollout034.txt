import numpy as np
from scipy.optimize import minimize

def penalty(z, mu):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    d = np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2 + 1e-12**2)
    dist_pen = (r[:, None] + r[None, :] - d) ** 2 * ((r[:, None] + r[None, :] - d) > 0)
    left_pen = (r - x) ** 2 * ((r - x) > 0) + (r - (1 - x)) ** 2 * ((r - (1 - x)) > 0)
    bottom_pen = (r - y) ** 2 * ((r - y) > 0) + (r - (1 - y)) ** 2 * ((r - (1 - y)) > 0)
    non_neg_pen = (-r) ** 2 * (r < 0)
    return mu * (np.sum(dist_pen) + np.sum(left_pen) + np.sum(bottom_pen) + np.sum(non_neg_pen))

def constraint_sum_r(z):
    return np.sum(z[52:])

def func(z, mu):
    return -np.sum(z[52:]) + penalty(z, mu)

def grad(z, mu):
    r = z[52:]
    d = np.sqrt((z[:26][:, None] - z[:26][None, :])**2 + (z[26:52][:, None] - z[26:52][None, :])**2 + 1e-12**2)
    ds = (r[:, None] + r[None, :] - d) * ((r[:, None] + r[None, :] - d) > 0)
    x = z[:26]
    y = z[26:52]
    dx1 = ds * (x - z[:26][None, :]) / d
    dx2 = ds * (z[:26][:, None] - x) / d
    dy1 = ds * (y - z[26:52][None, :]) / d
    dy2 = ds * (z[26:52][:, None] - y) / d
    dr1 = 2 * ds
    dr2 = 2 * ds
    l1, l2, l3 = 2 * (r - x) * ((r - x) > 0) - 2 * (r - (1 - x)) * ((r - (1 - x)) > 0)
    b1, b2, b3 = 2 * (r - y) * ((r - y) > 0) - 2 * (r - (1 - y)) * ((r - (1 - y)) > 0)
    nz = 2 * (-r) * (r < 0)
    return np.hstack([dx1.flatten(), dy1.flatten(), dr1.flatten(),
                      dx2.flatten(), dy2.flatten(), dr2.flatten(),
                      l1.flatten(), l2.flatten(), l3.flatten(),
                      b1.flatten(), b2.flatten(), b3.flatten(),
                      nz.flatten()]) + penalty(z, mu)

def run_packing():
    np.random.seed(42)
    n = 26
    mu = 10
    N_ITER = 200
    N_RUNS = 25
    max_penalty_increments = 10
    grad_check_tol = 1e-6
    tol = 1e-7

    best_centers = None
    best_radii = None
    best_sum = -np.inf

    for seed in range(N_RUNS):
        z0 = np.random.rand(78)
        z0[:26] = np.random.uniform(0.05, 0.95, 26)
        z0[52:] = 0.05

        for _ in range(max_penalty_increments):
            res = minimize(func, z0, args=(mu,), jac=grad, bounds=[(0, None)] * 78, options={'maxiter': N_ITER})

            if res.success:
                violating = penalty(res.x, mu) > tol
                if violating:
                    mu *= 10
                    continue
                else:
                    z = res.x
                    centers = z[:26].reshape(-1, 2)
                    radii = z[52:]
                    current_sum = np.sum(radii)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers
                        best_radii = radii

                    if current_sum - np.max(radii) < grad_check_tol:
                        perturbation = np.random.rand(2)
                        perturbation[0] = perturbation[0] * 0.02 - 0.01
                        perturbation[1] = perturbation[1] * 0.01 - 0.005
                        perturbed_indices = np.random.randint(0, 26)
                        z0 = np.concatenate((best_centers.flatten(), best_radii + perturbation))
                        break

        if best_sum >= 2.637:
            break

    return best_centers, best_radii, best_sum
