from scipy.optimize import minimize

    return -np.sum(r)

def penalty(z, mu):
    x, y, r = z[:26], z[26:52], z[52:]
    n = len(r)
    p_overlap = np.sum(np.maximum(0, r[:,None] + r - np.sqrt((x[:,None] - x) ** 2 + (y[:,None] - y) ** 2)), axis=1) ** 2)
    p_left = np.sum(np.maximum(0, r - x) ** 2)
    p_right = np.sum(np.maximum(0, r - (1 - x)) ** 2)
    p_bottom = np.sum(np.maximum(0, r - y) ** 2)
    p_top = np.sum(np.maximum(0, r - (1 - y)) ** 2)
    p_non_negative = np.sum(np.maximum(0, -r) ** 2)
    return mu * (p_overlap + p_left + p_right + p_bottom + p_top + p_non_negative)

def penalty_gradient(z, mu):
    x, y, r = z[:26], z[26:52], z[52:]
    n = len(r)
    dx = np.zeros_like(x)
    dy = np.zeros_like(y)
    dr = np.zeros_like(r)
    distances = np.sqrt((x[:,None] - x) ** 2 + (y[:,None] - y) ** 2)
    valid = (r[:,None] + r > distances)
    s = r[:,None] + r - distances
    dx[valid] += 2 * s[valid] * (x[:,None] - x) / distances[valid]
    dy[valid] += 2 * s[valid] * (y[:,None] - y) / distances[valid]
    dr[valid] += 2 * s[valid]
    dx += r * ((x[:,None] - x) / distances)
    dy += r * ((y[:,None] - y) / distances)
    d_l = 2 * np.maximum(0, r - x)
    d_r = 2 * np.maximum(0, r - (1 - x))
    d_b = 2 * np.maximum(0, r - y)
    d_t = 2 * np.maximum(0, r - (1 - y))
    dr -= d_l + d_r + d_b + d_t
    dnz = 2 * - np.maximum(0, -r)
    dr -= dnz
    dmu = np.sum(penalty_gradient(z, mu))
    return np.concatenate((dx, dy, dr)), mu * dmu
