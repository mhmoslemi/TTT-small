import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def compute_dual_gradient(centers, radii):
    n = centers.shape[0]
    A = np.zeros((n*(n-1)//2 + 4*n, n))
    b = np.zeros(A.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1
        b[idx] = x - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - x - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = y - 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - y - 1e-12
        idx += 1

    dual, status = linprog(np.zeros(n), A_ub=A, b_ub=b, method='highs')
    if status != 0:
        raise ValueError("Dual computation failed")

    grad = np.zeros(n)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist > 1e-12:
                grad[i] += dual[idx-1] * (centers[i] - centers[j]) / dist
                grad[j] += dual[idx-1] * (centers[j] - centers[i]) / dist
                idx += 1

        for direction in[np.array([1, 0]), np.array([-1, 0]), np.array([0, 1]), np.array([0, -1])]:
            dist = np.dot(centers[i], direction)
            if dist > 1e-12:
                grad[i] += dual[idx-1] * direction / dist
                idx += 1

    return grad

def propose_move(centers, grad, step):
    n = centers.shape[0]
    move = np.zeros_like(centers)
    for i in range(n):
        if np.linalg.norm(grad[i]) > 1e-12:
            move[i] = grad[i] / np.linalg.norm(grad[i])
    move /= np.linalg.norm(move)
    move *= step
    move = np.clip(centers + move, -1e-12, 1 + 1e-12)
    return move

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.1, 1e-4
    decay_factor = 0.995
    NUM_ITERS_PER_PATTERN = 2000
    PATTERNS = [
        ('Random_inner', lambda: np.random.rand(n, 2)),
        ('Grid5x5_cut', lambda: np.array([[0.1 + 0.2 * k, 0.1 + 0.2 * l] for k in range(5) for l in range(5)])),
        ('OffsetGrid', lambda: np.array([[0.1 + 0.2 * k, 0.1 + 0.2 * l] for k in range(5) for l in range(5)]) + np.array([[0.1, 0.0] if l % 2 != 1 else [0.0, 0.0] for k in range(5) for l in range(5)])),
        ('SparseHex', lambda: np.array(np.where(np.array([[np.hypot(k - 0.45, l - 0.45) < 0.15 and k + l < 0.95 for k in np.linspace(0.1, 0.9, 10)] for l in np.linspace(0.1, 0.9, 10)]), k, l) for k, l in np.ndindex((10, 10)))).reshape(-1, 2)[:26]),
        ('CornerExpansion', lambda: np.vstack((np.random.rand(18, 2) * 0.8 + 0.1, np.array([[0.15, 0.25], [0.25, 0.15], [0.45, 0.25], [0.55, 0.15], [0.75, 0.25], [0.85, 0.15], [0.15, 0.85], [0.25, 0.75], [0.45, 0.85], [0.55, 0.75], [0.75, 0.85], [0.85, 0.75]]))))
    ]
    NUM_STARS = len(PATTERNS)
    RESTART_EPOCHS = 500

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        pattern_name, pattern_func = PATTERNS[start % NUM_STARS]
        centers = pattern_func()
        for it in range(NUM_ITERS_PER_PATTERN):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = centers[idx] + delta
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
