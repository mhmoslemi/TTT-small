import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([
        [i * 0.2 + 0.05, (i // 5 + i % 5) * 0.2667] for i in range(n)
    ])
    centers[5:9, 0] = 0.5
    centers[15:19, 0] = 0.95
    centers[10:15, 1] = 0.5
    centers[20:, 1] = 0.5

    def lp_solve(centers):
        def RadiiApprox(vars_amp, centers):
            x, y = vars_amp[:26].reshape(-1, 2), vars_amp[26:]
            A = np.repeat(x - centers, 26, axis=0).reshape(-1, 52) - np.repeat(centers, 26, axis=1).reshape(-1, 52).T
            A1 = A < 0
            A = A ** 3 * 2 * A1 - A1
            B = np.abs(A @ y)
            return np.minimum(B, np.repeat(centers[:, 0] - centers[:, 1], 26))

        vars_bounds = [(0, 0.5) for _ in range(26)] + [(0, 1) for _ in range(26)]
        res = opt.lsq_linear(RadiiApprox, x0=np.random.rand(52), args=(centers), method='highs', bounds=vars_bounds, verbose=0)
        return res.x[26:]

    def objective(vars, centers):
        return np.sum(lp_solve(centers))

    def feasibility(vars, centers):
        radii = lp_solve(centers)
        return optimize_feasibility(radii, centers)

    def optimize_feasibility(radii, centers):
        for i in range(n):
            if not (0 <= centers[i, 0] - radii[i] <= 1 and 0 <= centers[i, 1] - radii[i] <= 1):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                if np.linalg.norm(centers[i] - centers[j]) < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def simulate_annealing():
        best_sum = 0
        best_centers = None
        T = 0.1
        step_count = 20000
        while T > 1e-6:
            for _ in range(step_count):
                i = np.random.randint(n)
                new_center = centers[i] + np.random.randn(2) * T / 2
                new_center = np.clip(new_center, 0, 1)
                if np.any(np.linalg.norm(new_center - centers[:i], axis=1) < 1e-4):
                    continue
                new_centers = centers.copy()
                new_centers[i] = new_center
                new_sum = objective([], new_centers)
                if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
                    best_sum = new_sum
                    best_centers = new_centers
            T *= 0.9999
        return best_centers, best_sum

    optimal_centers, best_sum = simulate_annealing()

    radii = lp_solve(optimal_centers)
    return optimal_centers, radii, best_sum
