import numpy as np
from scipy.optimize import minimize

def objective(z):
    return -np.sum(z[66:])

def constraint(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    
    cons = []
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: -r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: x[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - x[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: y[i] - r[i]})
        cons.append({'type': 'ineq', 'fun': lambda z, i=i: 1 - y[i] - r[i]})
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: r[i] + r[j] - dist})
    
    return cons

def clip_radii(r):
    eps = 1e-12
    return np.maximum(r, eps)

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radius = 0.10
    step = 1 / np.sqrt(2)

    for i in range(5):
        for j in range(5):
            if i < 4:
                centers[i*5+j, 0] = step * (2 * (i % 3) + 1)
            elif j < 4:
                centers[i*5+j, 0] = step * (2 * ((j % 3) + 1))
                centers[i*5+j, 1] = step + 0.5 * step
            else:
                centers[i*5+j, 0] = step * (2 * ((j % 3) + 1))
                centers[i*5+j, 1] = step + 0.5 * step * 2

    center_radius = np.sqrt(0.08) - 0.1
    centers[12] = [0.5, 0.5]
    radii = np.array([radius] * 25 + [center_radius])

    initial_vector = np.concatenate((centers.flatten(), radii))

    cons = constraint(initial_vector)
    
    options = {
        'max_iter': 2000,
        'gtol': 1e-8,
        'xtol': 1e-8,
        'disp': False,
    }
    
    result = minimize(
        fun=objective,
        x0=initial_vector,
        method='trust-constr',
        constraints=cons,
        options=options,
        jac=None,
        hess=None,
    )
    
    if result.success:
        x = result.x[:n]
        y = result.x[n:2*n]
        r = clip_radii(result.x[2*n:])
        sum_radii = -result.fun
        return x.reshape(-1, 1), y.reshape(-1, 1), sum_radii
    else:
        raise Exception("Non-linear optimization failed")
