import numpy as np
from scipy.optimize import differential_evolution, linprog
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])
    
    idx = 0
    for i in range(n):
        A_eq[i, i] = 1
        b_eq[i] = 1
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = dist - 1e-9
            idx += 1
    
    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-9
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_initial_population(n, popsize):
    centers = np.random.rand(n, 2) * 0.18 + 0.09
    centers[0:5, :] = [[0.03, 0.03]] + [[0.22*i + 0.09, 0.44] for i in range(5)]
    centers[5:15, :] = [[0.22*i + 0.09, 0.78] for i in range(5)] + [[1.02, 0.33 + 0.22*i] for i in range(5)]
    centers[15:25, :] = [[0.11 + 0.22*i, 0.44 - 0.22*i] for i in range(5)] + [[0.78, 0.11 + 0.22*i] for i in range(5)]
    centers[25, :] = [0.99, 0.99]
    return np.vstack([centers, np.random.rand(max(0, popsize - len(centers)), 2)]) + np.random.randn(max(0, popsize - len(centers)), 2) * 0.01

def run_packing():
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    
    def neg_sum_of_radii(x):
        centers = x.reshape((26, 2))
        radii, sum_r = lp_optimum(centers)
        return -sum_r
    
    res = differential_evolution(neg_sum_of_radii, bounds, method='dual_annealing', popsize=popsize, maxiter=maxiter, strategy=strategy, init=generate_initial_population)
    best_centers = res.x.reshape((26, 2))
    best_radii, best_sum = lp_optimum(best_centers)
    
    def local_sa(centers):
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = 0.05
        T = 0.05
        for iter in range(5000):
            idx = random.randint(0, 25)
            move = np.random.rand(2) * step - step / 2
            new_centers = cur_centers.copy()
            new_centers[idx] += move
            new_centers[idx] = np.clip(new_centers[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > cur_sum or random.random() < np.exp((new_sum - cur_sum) / T):
                cur_centers, cur_radii, cur_sum = new_centers, new_radii, new_sum
            step = max(1e-4, step * 0.995)
            T = max(1e-5, T * 0.995)
        return cur_centers, cur_radii, cur_sum
    
    best_centers, best_radii, best_sum = local_sa(best_centers)
    
    return best_centers, best_radii, best_sum

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
