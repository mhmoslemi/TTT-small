import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_hexagons(dim, side_length):
    side = np.array([side_length, 0])
    centers = np.array([
        [0.5 * side_length, 0.5 * np.sqrt(3) * side_length],
        [side_length + 0.75 * side_length, 0.25 * np.sqrt(3) * side_length],
        [side_length + 0.75 * side_length, 0.75 * np.sqrt(3) * side_length],
        [2 * side_length, side_length],
        [side_length + 0.75 * side_length, 0.25 * np.sqrt(3) * side_length],
        [side_length + 0.75 * side_length, 0.75 * np.sqrt(3) * side_length],
        [1.5 * side_length, 0.5 * np.sqrt(3) * side_length]
    ])
    
    for k in range(dim):
        centers = np.vstack([centers + np.array([0, k * (2 * np.sqrt(3) * side_length)]), centers])
        
    return centers

def find_optimal_packing():
    n = 26
    num_hexagons = 4
    hexagon_side_length = 0.18
    corner_offset = 0.03
    jitter_std = 0.02
    NUM_RESTARTS = 10
    
    best_sum = -np.inf
    best_centers = None
    
    for restart in range(NUM_RESTARTS):
        if restart == 0:
            centers = generate_hexagons(num_hexagons, hexagon_side_length)
            centers = np.vstack([centers, 
                np.array([[corner_offset, corner_offset], 
                          [corner_offset, 1 - corner_offset], 
                          [1 - corner_offset, corner_offset], 
                          [1 - corner_offset, 1 - corner_offset]])])
        else:
            if restart == 1:
                jitter = jitter_std * np.random.randn(4, 2)
            else:
                jitter = jitter_std * np.random.randn(n, 2)
            centers = np.random.rand(n, 2)
            centers[:4] = np.array([[corner_offset, corner_offset], 
                                  [corner_offset, 1 - corner_offset], 
                                  [1 - corner_offset, corner_offset], 
                                  [1 - corner_offset, 1 - corner_offset]]) + jitter[:4]
            
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = 0.1
        decay = 0.995
        NUM_ITERS = 4000
        
        for iter in range(NUM_ITERS):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum:
                cur_centers = new_c
                cur_sum = new_sum
        
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers
    
    return best_centers, lp_optimum(best_centers)[0], best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = find_optimal_packing()
    return centers, radii, sum_radii
