import numpy as np
from scipy.optimize import minimize

def penalty_vector(z, mu):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    d_ij = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=-1))
    d_ij += 1e-12  # Avoid division by zero
    maxs = np.maximum(radii[:, None] + radii[None, :] - d_ij, 0)
    penalty = mu * np.sum(maxs**2) + mu * np.sum(np.clip(radii - centers[:, 0], 0, None)**2)
    penalty += mu * np.sum(np.clip(radii - centers[:, 1], 0, None)**2)
    penalty += mu * np.sum(np.clip(radii - (1 - centers[:, 0]), 0, None)**2)
    penalty += mu * np.sum(np.clip(radii - (1 - centers[:, 1]), 0, None)**2)
    penalty += mu * np.sum(np.maximum(-radii, 0)**2)
    return penalty

def objective(z):
    return -(np.sum(z[2::3]))  # Minimize negative sum of radii

def grad(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    d_ij = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=-1))
    d_ij += 1e-12  # Avoid division by zero
    maxs = np.maximum(radii[:, None] + radii[None, :] - d_ij, 0)
    
    # Gradient for overlap constraints
    grad_overlap = 4 * maxs * np.where(radii[:, None] + radii[None, :] > d_ij, 1, 0)
    grad_overlap = 2 * (grad_overlap + grad_overlap.transpose((1, 0, 2)))  # Double the sum
    
    # Gradient for left boundary
    grad_left = np.where(radii > centers[:, 0], 4 * (radii - centers[:, 0]), 0)
    
    # Gradient for right boundary
    grad_right = np.where(radii > 1 - centers[:, 0], 4 * (radii - (1 - centers[:, 0])), 0)
    
    # Gradient for bottom boundary
    grad_bottom = np.where(radii > centers[:, 1], 4 * (radii - centers[:, 1]), 0)
    
    # Gradient for top boundary
    grad_top = np.where(radii > 1 - centers[:, 1], 4 * (radii - (1 - centers[:, 1])), 0)
    
    # Gradient for non-negative radii
    grad_radius = np.where(radii < 0, 4 * radii, 0)

    # Combine gradients
    grad_centers = np.sum(grad_overlap + grad_left[:, np.newaxis] + grad_right[:, np.newaxis] +
                         grad_bottom[:, np.newaxis] + grad_top[:, np.newaxis], axis=(1, 2)) / centers.shape[0]
    grad_radii = np.sum(grad_overlap, axis=(0, 1)) / n
    
    return np.hstack((grad_centers.flatten(), grad_radii))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    N_RUNS = 25
    N_RESTARTS = 3
    INIT_RADIUS = 0.05
    INIT_SPACING = 0.15
    PENALTY_MULTIPLIER = 10.0
    CONVERGENCE_TOLERANCE = 1e-7
    MAX_ITERATIONS = 200
    MAX_PENALTY_ROUNDS = 10
    RANDOM_SEED = 42
    
    np.random.seed(RANDOM_SEED)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    def get_initial_state(type='random'):
        if type == 'random':
            centers = np.random.uniform(0.05, 0.95, (n, 2))
            radii = np.full(n, INIT_RADIUS)
        elif type == 'grid':
            centers = np.mgrid[0.1:1.0:INIT_SPACING, 0.1:1.0:INIT_SPACING]
            centers = centers.T.reshape(-1, 2)
            radii = np.full(centers.shape[0], INIT_RADIUS)
        else:
            raise ValueError('Unknown initial state type')
        
        if centers.shape[0] > n:
            centers = centers[:n]
            radii = radii[:n]
        
        z0 = np.hstack((centers.flatten(), radii))
        return z0
    
    for _ in range(N_RUNS):
        z0 = get_initial_state(type='random')
        result = minimize(objective, z0, method='L-BFGS-B', jac=grad, bounds=[(0, None)]*n*3, options={'maxiter': MAX_ITERATIONS})
        
        if not result.success:
            continue
        
        max_violation = np.max([
            np.maximum(result.x[n*2::3] + result.x[:n*2:2] - 1, 0),
            np.maximum(result.x[n*2::3] + result.x[n*2+1::2] - 1, 0),
            np.maximum(result.x[n*2::3] - result.x[:n*2:2], 0),
            np.maximum(result.x[n*2::3] - result.x[n*2+1::2], 0),
            np.maximum(result.x[:n*2::2] + result.x[n*2+1::2] - result.x[n*2::2:n*2+2][:, np.newaxis] - result.x[n*2+1::2:n*2+2].transpose(), 0),
            np.maximum(result.x[n*2::3] - 0, 0)
        ])
        
        if max_violation < CONVERGENCE_TOLERANCE:
            ajusted_sum = -result.fun
            if ajusted_sum > best_sum:
                best_sum = ajusted_sum
                best_centers = result.x[:n*2].reshape(n, 2)
                best_radii = result.x[n*2::3]
    
    for _ in range(N_RESTARTS):
        z0 = best_centers.flatten() + np.random.uniform(-0.01, 0.01, (n, 2)).flatten() + np.random.uniform(-0.005, 0.005, n) * best_radii
        result = minimize(objective, z0, method='L-BFGS-B', jac=grad, bounds=[(0, None)]*n*3, options={'maxiter': MAX_ITERATIONS})
        
        if not result.success:
            continue
        
        max_violation = np.max([
            np.maximum(result.x[n*2::3] + result.x[:n*2:2] - 1, 0),
            np.maximum(result.x[n*2::3] + result.x[n*2+1::2] - 1, 0),
            np.maximum(result.x[n*2::3] - result.x[:n*2:2], 0),
            np.maximum(result.x[n*2::3] - result.x[n*2+1::2], 0),
            np.maximum(result.x[:n*2::2] + result.x[n*2+1::2] - result.x[n*2::2:n*2+2][:, np.newaxis] - result.x[n*2+1::2:n*2+2].transpose(), 0),
            np.maximum(result.x[n*2::3] - 0, 0)
        ])
        
        if max_violation < CONVERGENCE_TOLERANCE:
            ajusted_sum = -result.fun
            if ajusted_sum > best_sum:
                best_sum = ajusted_sum
                best_centers = result.x[:n*2].reshape(n, 2)
                best_radii = result.x[n*2::3]

    return best_centers, best_radii, best_sum
