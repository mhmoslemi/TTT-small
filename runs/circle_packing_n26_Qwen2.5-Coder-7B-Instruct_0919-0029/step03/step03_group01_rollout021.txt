import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    dist_matrix = squareform(pdist(centers, 'euclidean'))
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(A_ub.shape[0])
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist_matrix[i, j] - 1e-12
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_pattern_random(n):
    return np.random.rand(n, 2)

def generate_pattern_perturbed_grid(n):
    grid = np.array([(x, y) for x in np.arange(0.1, 1, 0.2) for y in np.arange(0.1, 1, 0.2)])
    pattern = grid[:n-1]
    pattern = np.vstack((pattern, np.random.rand(1, 2)))
    return pattern

def generate_pattern_diagonal_shift(n):
    grid = np.array([(x, y) for x in np.arange(0.1, 1, 0.2) for y in np.arange(0.1, 1, 0.2)])
    pattern = grid[:n-2]
    pattern = np.vstack((pattern, np.random.rand(2, 2)))
    return pattern

def generate_pattern_rough_hexagon(n):
    grid = np.array([(0.15 + 0.3 * np.cos(i * np.pi / 3), 0.13 * np.sin(i * np.pi / 3)) 
                     for i in range(21)])
    pattern = grid[:n]
    return pattern

def generate_pattern_random_plus_boundary(n):
    centers = np.random.rand(n - 6, 2)
    centers = np.vstack((centers, np.random.rand(6, 2)))
    centers[:2, :] = 0.5 + np.array([[0.15, 0], [-0.15, 0]])
    centers[2:4, :] = 0.5 + np.array([[0, 0.15], [0, -0.15]])
    centers[4:6, :] = 0.5 + np.array([[0, 0], [0, 0]])
    centers[6:, :] = 0.5 + np.random.rand(n - 6, 2)
    centers = np.clip(centers, 0.15, 0.85)
    return centers

def propose_move(centers, step):
    idx = np.random.randint(centers.shape[0])
    direction = np.random.randn(2)
    direction /= np.linalg.norm(direction)
    delta = direction * step
    return idx, delta

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
