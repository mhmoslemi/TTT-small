import numpy as np
from scipy.optimize import differential_evolution, linprog
from scipy.stats import norm
from typing import Tuple

def lp_optimum(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            d = centers[i] - centers[j]
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(d) - 1e-9
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-9
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_de_initial_coords(n: int, popsize: int) -> np.ndarray:
    hex_points = []
    center = np.array([0.5, 0.5])  # Center of the unit square
    radius = np.sqrt(3)/3  # Distance between centroid and edge of hexagon
    vertices = [(0.5, 0.5 + radius), (0.25, 0.25 + radius), (-0.25, 0.25 + radius),
                (0.25, -0.25 + radius), (-0.25, -0.25 + radius), (0, 0.5)]
    for i in range(6):
        for j in range(5):
            if j % 2 == 0:
                hex_points.append((center + np.array(vertices[i]) * (j%5+1)/6,
                                   center + np.array(vertices[i+1]) * (j%5+1)/6))

    distributors = [hex_points[i*5:(i+1)*5] for i in range(6)]
    distributors[0].extend([(0, 0.03), (0, 0.97), (0.97, 0), (0.97, 0.97)])
    distributors[1].extend([(0.03, 0), (0.97, 0), (0.03, 0.97), (0.97, 0.97)])
    distributors[2].extend([(0.5, 0.5 + 0.9 * radius)])
    distributors[3].extend([(0.5, 0.5 - 0.9 * radius)])
    distributors[4].extend([(0.5 + 0.9 * radius, 0.5)])
    distributors[5].extend([(0.5 - 0.9 * radius, 0.5)])

    new_distributors = []
    for i in range(6):
        arr_len = len(distributors[i])
        while len(new_distributors) < (popsize + 5): 
            new_distributors.append(distributors[i][:arr_len])
        
    result = np.array(new_distributors[:popsize])
    result += norm.rvs(0, 0.01, size=(popsize, 2))
    return result.reshape((popsize, 52))

def dual_optimization():
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'

    def objective(x):
        centers = x.reshape((26, 2))
        radii, sum_r = lp_optimum(centers)
        return -sum_r  # Maximize sum of radii by minimizing negative of sum

    optimal = differential_evolution(
        objective, bounds, popsize=popsize, maxiter=maxiter, strategy=strategy,
        init=generate_de_initial_coords, disp=True)

    best_centers = optimal.x.reshape((26, 2))
    best_sum = lp_optimum(best_centers)[1]
    return best_centers, best_sum

def local_sa(centers: np.ndarray):
    cur_centers = centers.copy()
    cur_radii, cur_sum = lp_optimum(cur_centers)
    step = 0.05
    T = 0.05
    for _ in range(5000):
        idx = np.random.randint(26)
        dir = np.random.randn(2)
        dir /= np.linalg.norm(dir)
        delta = dir * step
        new_centers = cur_centers.copy()
        new_centers[idx] += delta
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / T):
            cur_centers, cur_radii, cur_sum = new_centers, new_radii, new_sum
        step = max(1e-4, step * 0.995)
        T = max(1e-5, T * 0.995)
    return cur_centers, cur_radii, cur_sum

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_centers, best_radii = None, None
    
    for _ in range(3):
        centers, sum_r = dual_optimization()
        centers, sum_r = local_sa(centers)
        if sum_r >= >= best_sum:
            best_centers, best_sum = centers, sum_r
                
    return best_centers, best_radii, best_sum

if __name__ == "__main__":
    print(run_packing())
