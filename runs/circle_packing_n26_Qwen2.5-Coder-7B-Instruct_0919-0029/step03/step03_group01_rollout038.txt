import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 10**-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 10**-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 10**-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 10**-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 10**-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def dual_gradient(centers, radii):
    n = centers.shape[0]
    dist = np.linalg.norm(centers[:, np.newaxis] - centers, axis=2)
    non_self = dist != 0
    grad_x = 0 * centers[:, 1]
    grad_y = 0 * centers[:, 0]
    
    for i in range(n):
        contact_pairs = np.where(non_self[i])[0]
        grad_x[i] = np.sum(-radii[i] * (centers[i, 0] - centers[contact_pairs, 0]) / dist[i, contact_pairs], axis=0)
        grad_y[i] = np.sum(-radii[i] * (centers[i, 1] - centers[contact_pairs, 1]) / dist[i, contact_pairs], axis=0)
    
    grad_local_x = (centers[:, 0] - 0.5) * 2
    grad_local_y = (centers[:, 1] - 0.5) * 2

    grad_x -= radii * grad_local_x
    grad_y -= radii * grad_local_y

    return np.sqrt(grad_x**2 + grad_y**2)

def propose_move(centers, grad, step):
    n = centers.shape[0]
    move_x = np random.randn(n) * step
    move_y = np random.randn(n) * step
    move_x = np.clip(move_x, -1, 1) * step
    move_y = np.clip(move_y, -1, 1) * step

    proposal = centers + np.vstack((move_x, move_y)).T
    return np.clip(proposal, 10**-12, 1 - 10**-12)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.1, 1e-4
    decay_factor = 0.995
    MAX_ITERS_PER_PATTERN = 2000
    RESTART_EPOCHS = 500
    PATTERNS = ['Random_inner', 'Grid5x5_cut', 'OffsetGrid', 'SparseHex', 'CornerExpansion']
    
    np.random.seed(0)

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern_name in PATTERNS:
        if pattern_name == 'Random_inner':
            centers = np.random.rand(n, 2) * 0.9 + 0.05
        elif pattern_name == 'Grid5x5_cut':
            centers = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)]).reshape(-1, 2)
            centers[0, :] = np.random.rand(2) * 0.9 + 0.05
        elif pattern_name == 'OffsetGrid':
            centers = np.array([[0.1 + 0.2 * i + (i // 2) * 0.1, 0.1 + 0.2 * j] for i in range(5) for j in range(5)]).reshape(-1, 2)
        elif pattern_name == 'SparseHex':
            hex_radius = 0.15
            vertical_step = np.sqrt(3) * 0.07
            centers = []
            for i in range(5):
                for j in range(10):
                    x = 0.1 + 0.2 * (i + j // 6)
                    y = 0.1 + j * vertical_step
                    if len(centers) < n:
                        centers.append([x, y])
            centers = np.array(centers)
        elif pattern_name == 'CornerExpansion':
            centers = np.random.rand(18, 2) * 0.8 + 0.1
            x_midpoints = np.array([0.15, 0.35, 0.55, 0.75, 0.95])
            y_midpoints = np.array([0.15, 0.35, 0.75, 0.95])
            for i, x in enumerate(x_midpoints):
                if i % 2 == 0:
                    centers[i + 18, :] = [x, 0.15]
                else:
                    centers[i + 18, :] = [x, 0.95]
            for i, y in enumerate(y_midpoints):
                if i % 2 == 0:
                    centers[i + 18 + len(x_midpoints), :] = [0.15, y]
                else:
                    centers[i + 18 + len(x_midpoints), :] = [0.95, y]
            centers = np.unique(centers, axis=0)[:n]
        
        for t in range(MAX_ITERS_PER_PATTERN):
            new_centers = propose_move(centers, dual_gradient(centers, lp_optimum(centers)[1]), step)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
                step = 0.1
            elif np.random.rand() < np.exp((new_sum-best_sum)/0.5):
                centers = new_centers
                new_radii, new_sum = lp_optimum(centers)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = new_radii.copy()
                    step = 0.1
            step = max(step_min, step * decay_factor)
            if step < step_min:
                step = 0.1
            
            if t % 100 == 0:
                print(f"Pattern: {pattern_name}, Iteration: {t}, Best Sum: {best_sum}")

    return best_centers, best_radii, best_sum
