import numpy as np
from scipy.optimize import minimize

# Constants
N_CIRCLES = 26
N_RUNS = 25
N_ITER = 200
MU_INITIAL = 10
MU_MULT = 10
TOLERANCE = 1e-7
DELTA = 1e-12

# Objective and penalty terms
def f(z, mu):
    xs, ys, rs = z[:N_CIRCLES], z[N_CIRCLES:2*N_CIRCLES], z[2*N_CIRCLES:]
    penalty = 0
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d_ij = np.sqrt((xs[i]-xs[j])**2 + (ys[i]-ys[j])**2) + DELTA
            s_ij = rs[i] + rs[j] - d_ij
            penalty += max(0, s_ij)**2 if s_ij > 0 else 0
        penalty += max(0, r - x)**2 if r - x > 0 else 0 for x, r in zip(xs, rs)
        penalty += max(0, r - (1-x))**2 if r - (1-x) > 0 else 0 for x, r in zip(xs, rs)
        penalty += max(0, r - y)**2 if r - y > 0 else 0 for y, r in zip(ys, rs)
        penalty += max(0, r - (1-y))**2 if r - (1-y) > 0 else 0 for y, r in zip(ys, rs)
        penalty += max(0, -r)**2 if r < 0 else 0 for r in rs
    return -np.sum(rs) + mu * penalty

def grad_f(z, mu):
    xs, ys, rs = z[:N_CIRCLES], z[N_CIRCLES:2*N_CIRCLES], z[2*N_CIRCLES:]
    gradient = np.zeros_like(z)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d_ij = np.sqrt((xs[i]-xs[j])**2 + (ys[i]-ys[j])**2) + DELTA
            s_ij = rs[i] + rs[j] - d_ij
            if s_ij > 0:
                d_ij_inv = 1 / d_ij
                gradient[i]++;
                gradient[j]++;
                gradient[N_CIRCLES+i] += 2 * s_ij * (xs[i]-xs[j]) * d_ij_inv;
                gradient[N_CIRCLES+j] += 2 * s_ij * (xs[j]-xs[i]) * d_ij_inv;
        for k in range(N_CIRCLES):
            x, r = xs[k], rs[k]
            x_constraint = max(0, r - x)
            gradient[N_CIRCLES+k] += 2 * x_constraint if x_constraint > 0 else 0
            x_constraint_right = max(0, r - (1-x))
            gradient[N_CIRCLES+k] += 2 * x_constraint_right if x_constraint_right > 0 else 0
            y_constraint = max(0, r - ys[k])
            gradient[2*N_CIRCLES+k] += 2 * y_constraint if y_constraint > 0 else 0
            y_constraint_top = max(0, r - (1-ys[k]))
            gradient[2*N_CIRCLES+k] += 2 * y_constraint_top if y_constraint_top > 0 else 0
            r_constraint = max(0, -r)
            gradient[2*N_CIRCLES+k] += 2 * r_constraint if r_constraint > 0 else 0
    return -gradient + mu * grad_p(z, mu)

def grad_p(z, mu):

    gradient = np.zeros_like(z)
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d_ij = np.sqrt((xs[i]-xs[j])**2 + (ys[i]-ys[j])**2) + DELTA
            s_ij = rs[i] + rs[j] - d_ij
            if s_ij > 0:
                gradient[i] += 2 * s_ij
                gradient[j] += 2 * s_ij
                gradient[N_CIRCLES+i] += 2 * s_ij * (xs[i]-xs[j]) / d_ij
                gradient[N_CIRCLES+j] += 2 * s_ij * (xs[j]-xs[i]) / d_ij
        for k in range(N_CIRCLES):
            x, r = xs[k], rs[k]
            if r represent radius
            x_constraint = max(0, r - x)
            gradient[N_CIRCLES+k] += 2 * x_constraint if x_constraint > 0 else 0
            x_constraint_right = max(0, r - (1-x))
            gradient[N_CIRCLES+k] += 2 * x_constraint_right if x_constraint_right > 0 else 0
            y_constraint = max(0, r - ys[k])
            gradient[2*N_CIRCLES+k] += 2 * y_constraint if y_constraint > 0 else 0
            y_constraint_top = max(0, r - (1-ys[k]))
            gradient[2*N_CIRCLES+k] += 2 * y_constraint_top if y_constraint_top > 0 else 0
            r_constraint = max(0, -r)

# Init random seed for reproducibility
np.random.seed(42)

# Main run function
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for run in range(N_RUNS):
        # Initialize positions and radii
        centers = np.random.uniform(0.05, 0.95, size=(N_CIRCLES, 2))
        radii = np.full(N_CIRCLES, 0.05)
        z = np.concatenate((centers.flatten(), radii))

        mu = MU_INITIAL
        for i in range(N_ITER):
            result = minimize(f, z, method='L-BFGS-B', jac=grad_f, bounds=[(0, None)] * 78, args=(mu))
            zs = result.x
            xs, ys, rs = zs[:N_CIRCLES], zs[N_CIRCLES:2*N_CIRCLES], zs[2*N_CIRCLES:]
            z_new = np.concatenate((xs.flatten(), ys.flatten(), np.maximum(rs, 0.0)))
            if f(z_new, mu) > f(z, mu) + 1e-4:
                z = z_new
            max_violation = max(rs + xs - 1, rs + ys - 1, rs - xs, rs - ys, rs + radii - np.sqrt((xs[:, None] - xs)**2 + (ys[:, None] - ys)**2), -rs)
            if max_violation > 0:
                mu *= MU_MULT
            else:
                sum_radii = np.sum(rs)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers, best_radii = xs.copy(), rs.copy()

    return best_centers.reshape(N_CIRCLES, 2), best_radii, best_sum
