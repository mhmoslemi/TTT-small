import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import minimize

def generate_initial_centers(n):
    return np.random.rand(n, 2)

def calculate_radii_from_centers(centers):
    tri = Delaunay(centers)
    circumcenters = np.array([circumcenter(p) for p in tri.points[tri.simplices]])
    circumradii = np.array([circumradius(p) for p in tri.points[tri.simplices]])
    target_radius = np.mean(circumradii) / 2
    return target_radius

def circumcenter(triangle):
    A = np.array([triangle[0][0], triangle[0][1], 1],
                  [triangle[1][0], triangle[1][1], 1],
                  [triangle[2][0], triangle[2][1], 1])
    B = np.array([triangle[0][0]**2 + triangle[0][1]**2,
                  triangle[1][0]**2 + triangle[1][1]**2,
                  triangle[2][0]**2 + triangle[2][1]**2])
    return np.linalg.solve(A[:, :2], B - A[:, 2][:, None]) / 2

def circumradius(triangle):
    A = np.array([triangle[0][0], triangle[0][1], 1],
                  [triangle[1][0], triangle[1][1], 1],
                  [triangle[2][0], triangle[2][1], 1])
    return np.sqrt(A[:, :2].T @ A[:, :2] - A[:, 2][:, None] * A[:, 2][None, :])

def objective_function(centers):
    return -np.sum(calculate_radii_from_centers(centers))

def constraint_function(centers):
    tri = Delaunay(centers)
    circumcenters = np.array([circumcenter(p) for p in tri.points[tri.simplices]])
    circumradii = np.array([circumradius(p) for p in tri.points[tri.simplices]])
    return circumradii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = generate_initial_centers(n)
    constraints = {'type': 'ineq', 'fun': constraint_function}
    bounds = [(0, None)] * (2 * n)
    result = minimize(objective_function, initial_centers.flatten(), method='SLSQP', bounds=bounds, constraints=constraints)
    if result.success:
        optimized_centers = result.x.reshape(n, 2)
        optimized_radii = calculate_radii_from_centers(optimized_centers)
        return optimized_centers, optimized_radii, np.sum(optimized_radii)
    else:
        return np.zeros(n).reshape(n, 2), np.zeros(n), 0
