import numpy as np
from scipy.optimize import linprog

eps = 1e-12
relax = 1e-10

def solve_lp(centers, is_relaxed=False):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1 - centers[i, 0]
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1 - centers[i, 1]

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            if is_relaxed:
                b_ineq.append(dists[i, j] - eps - relax)
            else:
                b_ineq.append(dists[i, j] - eps)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if not result.success:
        raise Exception("LP optimization failed")

    return result.x

def objective(centers):
    radii = solve_lp(centers)
    return -np.sum(radii)

def run_packing(maxiter=2000):
    n = 26
    p = 0.02
    sigma = 2 * p / np.sqrt(2)
    T = 1.0
    m = 5
    eps = 1e-6

    best_centers = None
    best_radius_sum = -np.inf

    for _ in range(5):
        centers = np.zeros((26, 2))
        for i in range(n):
            if i < n // 3:
                centers[i, 0] = p + (2 * (i % 3) + 1) * (1 - 2 * p) / 3
                centers[i, 1] = p + i // 3 * (1 - 2 * p) / (n // 3 - 1)
            else:
                centers[i, 0] = np.random.uniform(p, 1 - p)
                centers[i, 1] = np.random.uniform(p, 1 - p)

        for it in range(maxiter):
            idx = np.random.randint(n)
            delta = sigma * np.random.randn(2)
            centers[idx] = np.clip(centers[idx] + delta, 0, 1)
            try:
                new_radius_sum = objective(centers)
                if new_radius_sum > best_radius_sum:
                    best_radius_sum = new_radius_sum
                    best_centers = centers.copy()
            except Exception as e:
                pass

            sigma *= 0.999
            T *= 0.999

            if it % 200 == 0:
                sigma *= 0.95
                T *= 0.95

    return best_centers, solve_lp(best_centers), -best_radius_sum
