import numpy as np
from scipy.optimize import linprog
from scipy.spatial import distance_matrix

def solve_radii(centres):
    n = len(centres)
    B = np.minimum(centres[:, 0] - 0.05, 1 - centres[:, 0] - 0.05,
                   centres[:, 1] - 0.05, 1 - centres[:, 1] - 0.05)
    D = distance_matrix(centres, centres)
    A = np.hstack((np.eye(n), -np.eye(n)))
    b = D.ravel()
    A_ub = np.concatenate((A, -A), axis=0)
    b_ub = 2 * B.ravel()
    c = np.ones(2 * n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', tol=1e-12)
    return res.x[:n], res.fun
