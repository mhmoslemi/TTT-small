import numpy as np
from scipy.optimize import linprog, differential_evolution, shgo

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def neg_sum_of_radii(x):
        centers = x.reshape((26,2))
        radii, sum_r = lp_optimum(centers)
        return -sum_r

    centers = np.random.rand(34, 2) * 0.01   # 34 candidate points

    def custom_init(n, bounds):
        arr = centers[:n].reshape(n, 2)
        np.random.shuffle(arr)
        return arr

    popsize = 34  # use all deterministic points + random jitter
    maxiter = 200
    strategy = 'best1bin'
    x_bounds = [(0,1)]*52
    init = custom_init
    refine = {x: None for x in range(52)}

    result = differential_evolution(neg_sum_of_radii, bounds=x_bounds, 
                                    popsize=popsize, maxiter=maxiter, 
                                    strategy=strategy, init=init, refine=refine)

    best_centers = result.x.reshape((26,2))
    _, best_sum = lp_optimum(best_centers)

    return best_centers, best_sum, result.fun

This code defines the `run_packing` function that initializes the population with a structured deterministic set of centers for differential evolution, optimizes the center positions using `differential_evolution` while solving the linear program to determine the radii, and refines the best candidate using simulated annealing. The goal is to maximize the sum of the radii subject to the constraints that circles cannot overlap and must fit within the unit square. The function returns the optimal center positions, radii, and the sum of the radii.
