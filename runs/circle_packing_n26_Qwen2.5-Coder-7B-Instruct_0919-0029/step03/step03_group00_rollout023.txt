import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers, active_pairs):
    n = centers.shape[0]
    A_ub = np.zeros((len(active_pairs) + 2*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i, j in active_pairs:
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = np.linalg.norm(centers[i] - centers[j] - 1e-9)
        idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-9
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def update-contact-graphics(centers, radii):
    n = centers.shape[0]
    active_pairs = set()
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist <= radii[i] + radii[j] - 1e-9:
                active_pairs.add((i, j))
    return active_pairs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_STARS = 20
    TEMP_INIT = 0.15
    TEMP_DECAY = 0.9995
    STEP_DECAY = 0.995
    NUM_ITERS = 10000

    best_sum = -np.inf
    best_centers, best_radii = None, None

    def init_centers():
        circle_radius = np.sqrt(3) * 0.18
        hexagon_step = circle_radius / np.sqrt(3)
        hexagon_center_x = np.arange(0.15, 1, hexagon_step)
        hexagon_center_y = np.array([0.5 - circle_radius, 0.5, 0.5 + circle_radius])
        corners = np.array([[0.03, 0.03], [0.03, 0.97], [0.97, 0.03], [0.97, 0.97]])
        centers = np.vstack([np.repeat(hexagon_center_x, 2), np.tile(hexagon_center_y, (len(hexagon_center_x), 1))]).T
        centers = np.vstack([centers, corners])
        radii = np.minimum(np.linalg.norm(centers - np.mean(centers, axis=0), axis=1), 1e6)
        centers += np.random.normal(0, 0.02, centers.shape)
        return centers[:, :2], radii[:n]

    for _ in range(NUM_STARS):
        centers, _ = init_centers()
        active_pairs = update_contact_graphics(centers, np.zeros(n))
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers, active_pairs)
        temp = TEMP_INIT
        step = TEMP_INIT
        for iter in range(NUM_ITERS):
            idx = np.random.randint(n)
            direction = np.random.randn(2)
            direction /= np.linalg.norm(direction)
            delta = direction * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            active_pairs_new = update_contact_graphics(new_c, np.zeros(n))
            new_radii, new_sum = lp_optimum(new_c, active_pairs_new)
            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / temp):
                cur_centers = new_c
                cur_radii, cur_sum = new_radii, new_sum
                active_pairs = active_pairs_new
            temp *= TEMP_DECAY
            step *= STEP_DECAY

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers

    return best_centers, lp_optimum(best_centers, update_contact_graphics(best_centers, np.zeros(n)))[0], best_sum
