import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_ITERS = 4000
    NUM_STARS = 20
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = 0.1
        for iter in range(NUM_ITERS):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_c[idx][0] = max(0.1, min(0.9, new_c[idx][0]))
            new_c[idx][1] = max(0.1, min(0.9, new_c[idx][1]))
            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum:
                cur_centers = new_c
                cur_radii = new_radii
                cur_sum = new_sum

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers
            best_radii = cur_radii

    return best_centers, best_radii, best_sum
