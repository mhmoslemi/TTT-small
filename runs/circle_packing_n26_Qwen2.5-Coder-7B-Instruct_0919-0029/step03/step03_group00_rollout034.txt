import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def create_hexagon(seed=0.333):
    n = 30
    m = 5
    c = np.array([[i % m + (i // m) / m + seed, i // m / m + seed] for i in range(n)])
    d_x = 0.18
    d_y = np.sqrt(3) * d_x
    r = (1 - d_x) / 2
    scale = (1 - 2 * r) / max(d_x, d_y)
    c *= scale
    c -= 3 * d_x
    c = (c + 1) / 2
    c[:, 0] = np.clip(c[:, 0], 0.03, 0.97)
    c[:, 1] = np.clip(c[:, 1], 0.03, 0.97)
    return c[:26]
