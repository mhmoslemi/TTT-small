import numpy as np
from scipy.optimize import differential_evolution, linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ok = np.eye(n)
    b_ok = centers[:, 0] + centers[:, 1]
    A_bounds = np.array([[1, 0, 0], [-1, 0, 0], [0, 1, 0], [0, -1, 0]])
    b_bounds = np.ones(4) * 1e-12
    A_dist = np.zeros((n * (n - 1) // 2, n))
    b_dist = np.zeros(A_dist.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_dist[idx, i] = 1
            A_dist[idx, j] = 1
            b_dist[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    A_eq = np.vstack((A_ok, A_bounds, A_dist))
    b_eq = np.hstack((b_ok, b_bounds, b_dist))
    A_ub = np.array([])
    b_ub = np.array([])

    c = -np.ones(n)
    bounds = [(0, None)] * n

    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_initial_population():
    centers = np.array([
        (i * 0.18, k * 0.18) for k in range(6) for i in range(5)] +
        [(0.03, 0.03), (0.03, 0.97), (0.97, 0.03), (0.97, 0.97)]
    )
    jitter = np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers + jitter, 0, 1)
    return centers

def neg_sum_of_radii(x):
    centers = x.reshape((26, 2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    init_population = generate_initial_population()

    result = differential_evolution(neg_sum_of_radii, bounds, popsize=popsize, maxiter=maxiter, strategy=strategy, init=init_population)

    best_centers = result.x.reshape((26, 2))
    best_radii, best_sum = lp_optimum(best_centers)
    return best_centers, best_radii, best_sum
