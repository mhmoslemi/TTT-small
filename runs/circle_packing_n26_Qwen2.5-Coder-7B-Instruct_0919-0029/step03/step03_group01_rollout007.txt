import numpy as np
from scipy.optimize import linprog

EPS = 1e-12
step_start = 0.05
step_decay = 0.995
T0 = 1e-3 * 2.7  # Roughly 1000th of the best possible sum (assuming optimal is close to 2.7)
T_decay = 0.999
ITER_PER_PATTERN = 2000
MAX_NO_IMPROV = 500

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - EPS
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - EPS
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - EPS
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - EPS
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - EPS
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def generate_random(start):
    centers = np.random.rand(26, 2) * 0.9 + 0.05
    return centers

def generate_perturbed_grid(start):
    base = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    center_to_remove = base[0]
    base = np.delete(base, 0, axis=0)
    new_center = np.random.rand(2) * 0.9 + 0.05
    base = np.vstack((base, new_center))
    np.random.shuffle(base)
    if start > 0:
        base = np.roll(base, 1, axis=0)
    centers = base[:26]
    return centers

def generate_diagonal_shift(start):
    base = np.array([[0.1 + 0.2*i, 0.1 + 0.2*(i if i%2==0 else i+0.1)] for i in range(5) for j in range(5)])
    base = np.random.shuffle(base)
    if start > 0:
        base = np.roll(base, 1, axis=0)
    centers = base[:26]
    return centers

def generate_rough_hexagon(start):
    hexagon = np.array([[0.15 * np.cos(2 * np.pi * i / 6), 0.13 * np.sin(2 * np.pi * i / 6)] for i in range(6)])
    start = 0.15 * np.cos(2 * np.pi * start / 6)
    base = np.vstack([hexagon + [start + i*0.3, j*0.3] for i in range(5) for j in range(5)])
    centers = base[:26]
    return centers

def generate_random_plus_boundary(start):
    grid = np.array([[0.15 + i*0.3, 0.15 + j*0.3] for i in range(8) for j in range(8)])
    sides = np.array([[0.1 + i*0.2, 0.5] for i in range(5)]
                    + [[0.5, 0.1 + i*0.2] for i in range(5)]
                    + [[0.9 - i*0.2, 0.5] for i in range(5)]
                    + [[0.5, 0.9 - i*0.2] for i in range(5)])
    new_centers = np.vstack((grid[:26], sides[:10]))
    centers = np.random.shuffle(new_centers)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(5):
        if start == 0:
            centers = generate_random(start)
        elif start == 1:
            centers = generate_perturbed_grid(start)
        elif start == 2:
            centers = generate_diagonal_shift(start)
        elif start == 3:
            centers = generate_rough_hexagon(start)
        else:
            centers = generate_random_plus_boundary(start)

        for it in range(ITER_PER_PATTERN):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step_start
            new_c = centers[idx] + delta
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                best_radii = new_radii

            step_start = max(0.01, step_start * step_decay)

            T0 *= T_decay

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
