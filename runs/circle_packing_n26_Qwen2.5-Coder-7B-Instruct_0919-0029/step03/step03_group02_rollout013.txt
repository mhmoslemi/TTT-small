import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance_matrix

def solve_lp(centers):
    n = centers.shape[0]
    dists = distance_matrix(centers, centers)
    radii_vars = np.concatenate([centers[:, 0], centers[:, 1]])

    def lp_objective(vars):
        return -np.sum(vars[:n]) - np.sum(vars[n:])
    
    def lp_constraints(vars):
        constr = []
        for i in range(n):
            constr.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[n+i] - vars[i] - 1e-12})
            constr.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[i] - vars[n+i] - 1e-12})
            constr.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[n+i] - (1 - vars[i]) - 1e-12})
            constr.append({'type': 'ineq', 'fun': lambda vars, i=i: vars[i] - (1 - vars[n+i]) - 1e-12})
        for i in range(n):
            for j in range(i+1, n):
                constr.append({'type': 'ineq', 'fun': lambda vars, i=i, j=j: (vars[:n]**2 + vars[n:n*2]**2 - vars[i]**2 - vars[j]**2)*(1/dists[i, j] - 1 - 1e-12)})
        return constr
    
    bounds = [(0, None) for _ in range(2*n)]
    result = opt.minimize(lp_objective, radii_vars, method='SLSQP', bounds=bounds, constraints=lp_constraints)

    if result.success:
        new_centers = np.sqrt(result.x[:n]**2 - result.x[n:]**2)
        return new_centers, -result.fun
    else:
        print("LP optimization failed")
        return np.zeros(n), 0.0
