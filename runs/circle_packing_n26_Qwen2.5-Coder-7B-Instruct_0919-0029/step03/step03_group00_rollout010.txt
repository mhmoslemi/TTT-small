```python
import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def objective(x, n):
    return -sum(np.sqrt(x[i:i+2]) for i in range(0, 2*n, 2))

def constraint(x, n):
    dists = distance_matrix(x.reshape(-1, 2), x.reshape(-1, 2))
    np.fill_diagonal(dists, np.inf)
    return dists - 2 * np.sqrt(x.reshape(-1, 2)) + 1e-12 > 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(2 * n)
    bounds = [(0, 1) for _ in range(2 * n)]
    con = ({'type': 'ineq', 'fun': constraint, 'args': (n,)})
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[con])
    centers = result.x[:n].reshape(-1, 1)
    centers = np.hstack((centers, centers))
    radii = np.sqrt(result.x[n:])
    sum_radii = result.fun * -1
    return centers, radii, sum_radii
```
