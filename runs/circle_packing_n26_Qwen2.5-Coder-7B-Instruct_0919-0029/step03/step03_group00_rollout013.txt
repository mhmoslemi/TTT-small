import numpy as np
from scipy.optimize import linprog
from scipy.optimize import differential_evolution

def lp_optimum(centers):
    n = centers.shape[0]
    A = []
    b = []

    # Radius constraints
    for i in range(n):
        A.append([0] * n)
        A[-1][i] = 1
        b.append(0)
    
    # Inter-circle distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance = np.sqrt(dx**2 + dy**2)
            A.append([0] * n)
            A[-1][i] = 1
            A[-1][j] = 1
            b.append(distance)

    # Bounds for radius
    bounds = [(0, None)] * n

    # Objective function coefficients (negative to maximize)
    c = [-1] * n

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return result.x, -result.fun if result.success else 0

def fitness(x):
    centers = x.reshape((26, 2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r
