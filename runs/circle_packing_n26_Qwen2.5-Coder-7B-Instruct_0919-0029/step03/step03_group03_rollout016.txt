def evaluate_distance_matrix(centers):
    n = centers.shape[0]
    dists = np.full((n, n), -1.0, dtype=np.float64)
    for i in range(n):
        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
            dists[j, i] = dists[i, j]
    return dists
