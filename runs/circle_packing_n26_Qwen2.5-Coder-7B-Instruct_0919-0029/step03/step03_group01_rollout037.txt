import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def solve_lp(centers):
    return lp_optimum(centers)

def compute_dual_gradient(centers, radii):
    n = centers.shape[0]
    A = np.zeros((n*(n-1), n))
    for i in range(n):
        for j in range(i+1, n):
            A[i*(n-1)+j-1, i] = 1
            A[i*(n-1)+j-1, j] = 1

    Avec = A.getA()
    Avec_T = Avec.T
    dual = np.linalg.inv(A.T @ Avec @ A) @ Avec.T @ (radii[None, :] * radii.T)

    grad = np.zeros(2 * n)
    for i in range(n):
        grad[i*2:i*2+2] = dual @ Avec[i]

    return grad.reshape(n, 2)

def clip_to_square(centers, eps):
    return np.clip(centers, eps, 1-eps).round(4)

def propose_move(centers, grad, step):
    n = centers.shape[0]
    perturbation = np.random.randn(n, 2)
    perturbation /= np.linalg.norm(perturbation)
    new_centers = centers + step * grad + step * perturbation
    return clip_to_square(new_centers, 1e-12)

def alpha(t):
    return 0.8 ** t

def temp(t):
    return 0.5 * (0.995 ** t)

PATTERNS = [
    "Random_inner",
    "Grid5x5_cut",
    "OffsetGrid",
    "SparseHex",
    "CornerExpansion"
]

RESTART_EPOCHS = 500
MAX_ITERS_PER_PATTERN = 2000

def generate_pattern(pattern_name):
    if pattern_name == "Random_inner":
        return np.random.rand(26, 2) * 0.9 + 0.05
    elif pattern_name == "Grid5x5_cut":
        x, y = np.mgrid[0.1:1:0.2, 0.1:1:0.2].reshape(2, -1)
        centers = np.column_stack((x, y))
        centers = np.delete(centers, 0, axis=0)
        centers = np.clip(centers + np.random.rand(25, 2) * 0.02 - 0.01, 0, 1)
        return centers
    elif pattern_name == "OffsetGrid":
        x, y = np.mgrid[0.1:1:0.2, 0.1:1:0.2].reshape(2, -1)
        centers = np.column_stack((x, y))
        centers[::2, 0] -= 0.1
        centers = np.clip(centers + np.random.rand(26, 2) * 0.02 - 0.01, 0, 1)
        return centers
    elif pattern_name == "SparseHex":
        centers = []
        for i in range(26):
            theta = (2 * np.pi * i) / 30
            r = 0.07 * np.sqrt(3) * i
            x = 0.5 + r * np.cos(theta)
            y = 0.25 + r * np.sin(theta)
            if 0 < x < 1 and 0 < y < 1:
                centers.append((x, y))
        centers = np.array(centers[:26])
        centers = np.clip(centers + np.random.rand(26, 2) * 0.04 - 0.02, 0, 1)
        # Remove extreme points
       极端点列表 = [0, 1, 25, 3, 4, 23, 5, 6, 21, 7, 8, 19, 9, 10, 17, 11, 12, 15, 13, 16]
        centers = np.delete(centers, 十极点列表, axis=0)
        return centers
    elif pattern_name == "CornerExpansion":
        centers = np.random.rand(18, 2) * 0.8 + 0.1
        centers = np.vstack([
            centers,
            np.array([
                [0.15, 0.15], [0.85, 0.15], [0.45, 0.35],
                [0.15, 0.85], [0.85, 0.85], [0.45, 0.65],
                [0.15, 0.5], [0.85, 0.5], [0.5, 0.15],
                [0.15, 0.35], [0.85, 0.35], [0.5, 0.65]
            ])
        ])
        return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern_name in PATTERNS:
        centers = generate_pattern(pattern_name)
        radii, cur_sum = solve_lp(centers)
        grad = compute_dual_gradient(centers, radii)

        iterations_since_best = 0
        for t in range(MAX_ITERS_PER_PATTERN):
            new_centers = propose_move(centers, grad, step=alpha(t))
            new_centers = clip_to_square(new_centers, 1e-12)

            new_radii, new_sum = solve_lp(new_centers)
            accept = (new_sum >= cur_sum) or (np.random.rand() < np.exp((new_sum-cur_sum)/temp(t)))

            if accept:
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                grad = compute_dual_gradient(centers, radii)

                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    iterations_since_best = 0
                else:
                    iterations_since_best += 1
            else:
                iterations_since_best += 1

            if iterations_since_best > RESTART_EPOCHS:
                centers = generate_pattern(np.random.choice(PATTERNS))
                radii, cur_sum = solve_lp(centers)
                grad = compute_dual_gradient(centers, radii)
                iterations_since_best = 0
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()

    return best_centers, best_radii, best_sum
