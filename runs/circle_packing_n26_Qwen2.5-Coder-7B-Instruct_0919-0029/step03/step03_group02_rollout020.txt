import numpy as np
from scipy.optimize import linprog, minimize


def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")


def objective(x):
    centers = x.reshape(26, 2)
    try:
        radii, sum_radii = solve_lp(centers)
        return sum_radii
    except Exception as e:
        return np.nan


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    numstarts = 5
    best_centers, best_sum, best_radii = None, -np.inf, None

    for seed in range(numstarts):
        np.random.seed(seed)

        # Step 1: Initial random perturbation of seed configuration
        seed_positions = np.array([[0.5 + (i % 5) / 10., 0.5 + (i // 5) / 5.] for i in range(26)])
        seed_radii = np.full(26, 0.1)
        centers = seed_positions + np.random.uniform(-0.1, 0.1, size=(26, 2))
        centers -= np.min(centers, axis=0)
        centers /= np.max(centers, axis=0)
        centers *= 0.98  # make safe from boundary issues
        centers += 0.01  # simple clamping within range
        
        if seed == 0:
            centers = seed_positions  # Use seed positions for the first run

        # Step 2: Annealing loop
        T0 = 1.0
        Tf = 1e-3
        alpha = 1e-2
        steps_per_T = 1000
        for T in np.logspace(np.log10(T0), np.log10(Tf), steps_per_T * (T0 // Tf)):
            for i in range(26):
                delta = np.random.normal(0, T)
                new_center = centers[i] + np.array([np.cos(delta), np.sin(delta)])
                new_center = np.clip(new_center, 0, 1)
                new_centers = centers.copy()
                new_centers[i] = new_center

                new_sum Radii = objective(new_centers.flatten())
                
                if new_sum_radii > best_sum:
                    best_sum = new_sum_radii
                    best_centers = new_centers
                    best_radii = solve_lp(best_centers)[0]
                    
        # Step 3: Local refinement
        def refine_objective(x):
            return -objective(x)
        best_refinement, _ = minimize(refine_objective, best_centers.flatten(), method='BFGS')

        # Step 4: Final adjustment
        finalradii, _ = solve_lp(best_refinement.reshape(26, 2))

        # Step 5: Early stop if sum is high enough
        if best_sum >= 2.637:
            return best_refinement.reshape(26, 2), finalradii, best_sum

    return best_centers, best_radii, best_sum
