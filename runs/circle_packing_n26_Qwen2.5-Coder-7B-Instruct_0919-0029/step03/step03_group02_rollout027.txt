import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def objective(centers):
    radii, _ = solve_lp(centers)
    return -np.sum(radii)

def run_packing():
    K = 10
    M = 1500
    seeds = [np.random.rand(26, 2) for _ in range(K)]
    best_centers, best_sum, best_radii = None, -np.inf, None

    for seed in seeds:
        random_deltas = np.random.normal(0, 0.04, (K, 26, 2))
        random_deltas = np.clip(random_deltas, -0.06, 0.06)
        random_deltas /= np.max(np.abs(random_deltas), axis=2, keepdims=True)
        random_positions = seed + random_deltas
        random_positions = np.clip(random_positions, 0, 1)

        for step in range(M):
            i = np.random.randint(26)
            dx, dy = np.random.normal(0, 0.02, 2)
            dx, dy = np.clip(dx, -0.04, 0.04)
            dx, dy /= np.max(np.abs(dx), np.abs(dy))
            new_positions = random_positions.copy()
            new_positions[i, 0] += dx
            new_positions[i, 1] += dy
            new_positions[i] = np.clip(new_positions[i], 0, 1)

            new_sum = objective(new_positions)

            if new_sum > best_sum or (np.random.rand() < np.exp((new_sum - best_sum) / (0.1 - step / 1000))):
                random_positions = new_positions
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_positions
                    best_radii = solve_lp(best_centers)[0]

    return best_centers, best_radii, -best_sum

This code implements the described strategy for packing 26 circles in a unit square to maximize the sum of their radii. It uses simulated annealing to explore the space of possible circle placements and then computes the exact radii using linear programming for each configuration. The process is repeated multiple times with different initializations to ensure better exploration.
