import numpy as np
from scipy.optimize import minimize

def get_distances(z):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    d_ij = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)
    return d_ij

def penalty(z):
    n = len(z) // 3
    r = z[2*n:]
    d_ij = get_distances(z)
    s_ij = r[:, np.newaxis] + r[np.newaxis, :] - d_ij
    overlap_penalty = np.sum(np.maximum(s_ij, 0)**2)
    boundary_penalties = np.sum(np.maximum(r - x, 0)**2 + np.maximum(r - (1 - x), 0)**2 + np.maximum(r - y, 0)**2 + np.maximum(r - (1 - y), 0)**2)
    nz_penalty = np.sum(np.maximum(-r, 0)**2)
    return overlap_penalty + boundary_penalties + nz_penalty

def objective(z):
    n = len(z) // 3
    return -np.sum(z[2*n:])

def penalty_gradient(z):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    d_ij = get_distances(z)
    s_ij = r[:, np.newaxis] + r[np.newaxis, :] - d_ij

    d_ij[d_ij <= 0] = np.inf
    grad = np.zeros_like(z)
    grad[2*n:] = -1

    overlaps = (s_ij > 0)[:, :, np.newaxis] * 2 * s_ij
    grad[:n] += np.sum(overlaps * x[:, np.newaxis, np.newaxis] / d_ij, axis=2)
    grad[n:2*n] += np.sum(overlaps * y[:, np.newaxis, np.newaxis] / d_ij, axis=2)

    left_boundary = (r > x)[:, :, np.newaxis] * 2 * (r - x)
    grad[:n] += np.sum(left_boundary, axis=2)
    right_boundary = (r > (1 - x))[:, :, np.newaxis] * 2 * (r - (1 - x))
    grad[:n] += np.sum(right_boundary, axis=2)
    bottom_boundary = (r > y)[:, :, np.newaxis] * 2 * (r - y)
    grad[n:2*n] += np.sum(bottom_boundary, axis=2)
    top_boundary = (r > (1 - y))[:, :, np.newaxis] * 2 * (r - (1 - y))
    grad[n:2*n] += np.sum(top_boundary, axis=2)

    non_negative_radii = (r < 0)[:, :, np.newaxis] * 2 * r
    grad[2*n:] += np.sum(non_negative_radii, axis=2)

    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.1
    step_min = 1e-4
    decay_factor = 0.995
    max_penalty_rounds = 10
    tolerance = 1e-7
    gradient_tolerance = 1e-6
    N_RUNS = 25

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(N_RUNS):
        centers = np.random.uniform(0.05, 0.95, size=(n, 2))
        radii = np.full(n, 0.05)
        z0 = np.concatenate((centers.flatten(), radii))

        mu = 10
        converged = False
        for round in range(max_penalty_rounds):
            res = minimize(objective, z0, method='L-BFGS-B', bounds=[(0, 1)] * 2 * n + [(0, None)] * n,
                           jac=penalty_gradient, args=(mu,))
            if np.linalg.norm(res.jac) < gradient_tolerance:
                converged = True
                break
            if np.all(get_distances(res.x) >= radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-7):
                converged = True
                break
            if np.all(get_distances(res.x)[:, 0:n] >= radii[:, np.newaxis] - x) and \
               np.all(get_distances(res.x)[:, 0:n] >= radii[:, np.newaxis] - (1 - x)) and \
               np.all(get_distances(res.x)[:, n:2*n] >= radii[:, np.newaxis] - y) and \
               np.all(get_distances(res.x)[:, n:2*n] >= radii[:, np.newaxis] - (1 - y)):
                converged = True
                break
            x, y, r = res.x[:n], res.x[n:2*n], res.x[2*n:]
            z0 = np.concatenate((x.flatten(), y.flatten(), r))

        if converged:
            sum_radii = -res.fun
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = x.reshape(-1, 2)
                best_radii = r

    final_radii = np.ones(n) * 0.05
    if best_centers is not None:
        final_radii = best_radii

    return best_centers, final_radii, best_sum
