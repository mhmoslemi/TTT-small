import numpy as np
from scipy.optimize import minimize

def objective(x):
    n = int(len(x) / 2)
    return -np.sum(x[n:]**2)

def constraint_overlaps(x):
    n = int(len(x) / 2)
    A = np.zeros((n*(n-1)//2, 2*n))
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, 2*i:2*i+2] = 1
            A[idx, 2*j:2*j+2] = 1
            idx += 1
    return A, np.sqrt(np.sum((x[0:n] - x[n:2*n])**2, axis=1)) - 1e-12
      
def constraint_bounds(x):
    n = int(len(x) / 2)
    return x[n:] <= 0.5, x[n:] >= 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(2*n)
    constraints = [{'type': 'ineq', 'fun': constraint_overlaps},
                   {'type': 'ineq', 'fun': constraint_bounds}]
    bounds = [(-1, 1), (-1, 1)] * n + [(0, 0.5)] * n
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
    if result.success:
        centers = result.x[:n]
        radii = result.x[n:]
        return centers, radii, -result.fun
    else:
        return np.zeros(2*n), np.zeros(n), 0
