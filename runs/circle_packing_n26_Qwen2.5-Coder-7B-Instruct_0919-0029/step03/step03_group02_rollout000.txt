import numpy as np
import math

def run_packing():
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Place 5x5 grid of circles
    for i in range(5):
        for j in range(5):
            idx = 5 * i + j
            centers[idx, 0] = 0.1 * (2 * i + 1)
            centers[idx, 1] = 0.1 * (2 * j + 1)
            radii[idx] = 0.1

    # Place the center circle
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    radii[25] = math.sqrt(0.08) - 0.1

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
