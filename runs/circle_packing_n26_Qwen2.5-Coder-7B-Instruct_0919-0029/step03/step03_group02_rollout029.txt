import numpy as np
from math import sqrt

def run_packing():
    # Generate the 5x5 grid and central circle coordinates
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append((0.10 + 0.20 * i, 0.10 + 0.20 * j))
    centers.append((0.5, 0.5))
    centers = np.array(centers)

    # Set radii for the 5x5 grid
    radii = np.full(25, 0.10)

    # Calculate the radius for the central circle
    r_new = sqrt(0.08) - 0.10
    radii = np.append(radii, r_new)

    # Calculate the sum of the radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
