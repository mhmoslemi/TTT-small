import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import pairwise_distances
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = dist - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_pop(n, dim):
    # Hexagonal core
    hex_radius = 0.12
    hex_height = hex_radius * np.sqrt(3)
    hex_width = 2 * hex_radius
    hex_circles = 6 * 5
    hex_positions = []

    for j in range(5):
        for i in range(5 if j % 2 else 6):
            x = (0.1 + (i * hex_width / 2)) % 1
            y = (0.1 + (j * hex_height)) % 1
            if i % 2 == 0:
                hex_positions.append((x, y + hex_radius / 2))
            else:
                hex_positions.append((x, y))

    corner_positions = [(0.03, 0.03), (0.03, 0.97), (0.97, 0.03), (0.97, 0.97)]

    positions = np.array(hex_positions + corner_positions)

    if n > hex_circles + 4:
        extra_positions = positions[np.random.choice(hex_circles + 4, n - hex_circles - 4, replace=False)]
        positions = np.concatenate([positions, extra_positions])

    # Add random jitter
    jitter = np.random.normal(0, 0.01, size=(n, dim))
    positions += jitter

    return positions.min(0, 0), positions.max(0, 0)

def neg_sum_of_radii(x):
    centers = x.reshape((26,2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r  # Maximise sum -> minimises negative

def local_sa(centers):
    cur_centers = centers.copy()
    cur_radii, cur_sum = lp_optimum(cur_centers)
    step = 0.05
    T = 0.05
    for _ in range(5000):
        idx = random.randint(0, 25)
        new_centers = cur_centers.copy()
        new_centers[idx] += np.random.randn(2) * step
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or random.random() < np.exp((new_sum - cur_sum) / T):
            cur_centers, cur_radii, cur_sum = new_centers, new_radii, new_sum
        step = max(1e-4, step * 0.995)
        T = max(1e-5, T * 0.995)
    return cur_centers, cur_radii, cur_sum

def run_packing():
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    tol = 1e-8
    workers = 1
    refinement = {x: None}

    def custom_init(n, bounds):
        low, high = initial_pop(n, dim)
        return np.random.uniform(low, high, size=(n, dim))

    result = differential_evolution(neg_sum_of_radii, bounds, popsize=popsize, maxiter=maxiter, strategy=strategy,
                                     tol=tol, workers=workers, init=custom_init)

    best_centers = result.x.reshape((26, 2))
    best_sum = -neg_sum_of_radii(best_centers)
    refined_centers, refined_radii, refined_sum = local_sa(best_centers)

    return refined_centers, refined_radii, refined_sum

best_centers, best_radii, best_sum = run_packing()
