import numpy as np
from scipy.optimize import minimize

def objective(z, n):
    return -np.sum(z[2*n:])

def penalty(z, n):
    mu = 10
    penalty = 0
    for i in range(n):
        for j in range(i+1, n):
            d_ij = np.sqrt((z[i] - z[j])**2 + (z[i+n] - z[j+n])**2)
            s_ij = z[2*n+i] + z[2*n+j] - d_ij
            if s_ij > 0:
                penalty += mu * s_ij**2
        penalty += mu * max(0, z[2*n+i] - z[i])**2
        penalty += mu * max(0, z[2*n+i] - (1 - z[i]))**2
        penalty += mu * max(0, z[2*n+i+n] - z[i+n])**2
        penalty += mu * max(0, z[2*n+i+n] - (1 - z[i+n]))**2
        penalty += mu * max(0, -z[2*n+i])**2
        penalty += mu * max(0, -z[2*n+i+n])**2
    return penalty

def augmented_objective(z, n):
    return objective(z, n) + penalty(z, n)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.05
    NUM_RUNS = 25
    NUM_STARS = 20
    MAX_ITERATIONS = 200
    TOLERANCE = 1e-7

    best_centers = None
    best_radii = None
    best_sum = -np.inf

    for run in range(NUM_RUNS):
        centers = np.random.uniform(0.05, 0.95, size=(n, 2))
        radii = np.full(n, step)
        z0 = np.concatenate((centers.flatten(), radii))

        res = minimize(augmented_objective, z0, jac=None, method="L-BFGS-B", bounds=[(0, None)] * n + [(0, 1)] * n,
                       options={"maxiter": MAX_ITERATIONS}, tol=None)

        if res.success:
            centers = res.x[:n].reshape((n, 2))
            radii = res.x[n:]
            r_sum = np.sum(radii)

            if r_sum > best_sum:
                best_centers = centers
                best_radii = radii
                best_sum = r_sum

                # Optionally, apply random perturbation and re-run
                for iteration in range(3):
                    idx = np.random.randint(n)
                    delta_x = np.random.uniform(-0.01, 0.01)
                    delta_r = np.random.uniform(-0.005, 0.005)
                    delta_r = max(delta_r, 0)
                    p = np.array([-2 * (best_radii[idx] + best_radii[idx] + delta_r - np.sqrt((best_centers[idx, 0] - best_centers[idx:3, 0])**2 + (best_centers[idx, 1] - best_centers[idx:3, 1])**2))) for i in range(n)])
                    if np.any(p > 0):
                        best_centers[idx, 0] += delta_x
                        best_centers[idx, 1] += delta_x
                        best_radii[idx] += delta_r
                        best_centers[idx] = np.clip(best_centers[idx], 0, 1)
                        best_radii[idx] = np.clip(best_radii[idx], 0, 1)

                        res = minimize(augmented_objective, np.concatenate((best_centers.flatten(), best_radii)),
                                       jac=None, method="L-BFGS-B", bounds=[(0, None)] * n + [(0, 1)] * n,
                                       options={"maxiter": MAX_ITERATIONS / 2}, tol=None)
                        if res.success:
                            new_centers = res.x[:n].reshape((n, 2))
                            new_radii = res.x[n:]
                            if np.sum(new_radii) > best_sum:
                                best_centers = new_centers
                                best_radii = new_radii
                                best_sum = np.sum(new_radii)

    return best_centers, best_radii, best_sum
