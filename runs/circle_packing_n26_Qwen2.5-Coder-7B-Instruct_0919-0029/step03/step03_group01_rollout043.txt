import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            A_ub[idx, i+n] = -1
            A_ub[idx, j+n] = -1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        r = centers[i+n]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1
        A_ub[idx, i+n] = 1
        b_ub[idx] = 0
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * (2*n)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x[n:], -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step_start, step_decay = 0.1, 0.995
    T0, T_decay = 1e-3 * 3 * n, 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500
    NUM_STARS = 5

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        if start % 5 == 0:
            centers = np.random.rand(n, 2)
        elif start % 5 == 1:
            centers = np.array([
                (0.1 - 0.05 + i * 0.2 for i in range(5)),
                (0.1 - 0.05 + j * 0.2 for j in range(5)),
                (0.9 - 0.05 + i * 0.2 for i in range(5)),
                (0.9 - 0.05 + j * 0.2 for j in range(5))
            ]).T
            centers[1, 2] = np.random.rand()
            centers[2, 1] = np.random.rand()
        elif start % 5 == 2:
            centers = np.array([
                (0.9 - 0.1 * (i + 1) for i in range(11)),
                (0.1 - 0.1 * (10 - i) for i in range(11))
            ]).T
            centers = centers[:26, :]
        elif start % 5 == 3:
            centers = np.array([
                (0.15 + i * 0.15 for i in range(10)),
                (0.075 + j * 0.13 for j in range(10))
            ]).T
            centers = centers[:26, :]
        else:
            centers = np.random.rand(n, 2)
            centers[:2, :] = centers[:2, :] * 0.2 + 0.15
            centers[2:, :] = centers[2:, :] * 0.2 + 0.35

        step, step_min = step_start, 1e-4
        T, step_decay_rate = T0, step_decay
        cur_sum = np.inf
        no_improvement = 0

        for it in range(ITER_PER_PATTERN):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            Δ = dir * step
            new_c = centers[idx] + Δ
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > cur_sum:
                cur_sum = new_sum
                centers, new_radii = new_centers, new_radii
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers, best_radii = centers.copy(), new_radii.copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1

            step *= step_decay_rate
            T *= T_decay
            if no_improvement >= MAX_NO_IMPROV:
                centers, _ = reinitialize_random()
                no_improvement = 0

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
