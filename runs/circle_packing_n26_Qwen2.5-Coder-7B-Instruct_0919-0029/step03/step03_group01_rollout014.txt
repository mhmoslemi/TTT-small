import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers, radii=None):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + n * 4, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            if radii is None:
                b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            else:
                b_ub[idx] = (centers[i] - centers[j]) @ (radii[i] - radii[j])
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i] if radii is not None else 1
        A_ub[idx, i] = 1
        b_ub[idx] = x - r + 1e-12
        A_ub[idx + 1, i] = 1
        b_ub[idx + 1] = 1 - x + r - 1e-12
        A_ub[idx + 2, i] = 1
        b_ub[idx + 2] = y - r + 1e-12
        A_ub[idx + 3, i] = 1
        b_ub[idx + 3] = 1 - y + r - 1e-12
        idx += 4

    objective = -np.ones(n) if radii is None else -np.sum(radii)
    bounds = [(0, None)] * (n + 1 if radii is None else n)
    result = linprog(objective, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        if radii is None:
            return result.x[:n], -result.fun
        return result.x[:n], result.x[n] - sum(radii[:n])
    else:
        return np.zeros(n) if radii is None else np.zeros(n + 1)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.1, 1e-4
    decay_factor = 0.995
    NUM_ITERS = 4000
    NUM_STARS = 3

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = centers[idx] + delta
            new_c = np.clip(new_c, 0, 1)
            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii

        step = max(step_min, step * decay_factor)

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
