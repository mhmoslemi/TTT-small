import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    B = np.minimum(centers[:, 0], np.minimum(1 - centers[:, 0], np.minimum(centers[:, 1], 1 - centers[:, 1]))).reshape(-1, 1)
    d = cdist(centers, centers)
    c = np.ones(n)
    A_ineq = np.concatenate((-np.eye(n), np.eye(n), -d, d), axis=1)
    b_ineq = np.concatenate((-B.flatten(), B.flatten(), np.zeros(n*(n-1)//2), np.ones(n*(n-1)//2))).flatten()
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, method='highs', tol=1e-12)
    if result.success:
        return result.x, result.fun
    else:
        return np.full(n, -np.inf), -np.inf
