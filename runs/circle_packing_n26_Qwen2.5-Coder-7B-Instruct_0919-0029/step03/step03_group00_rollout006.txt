import numpy as np
from scipy.optimize import linprog, differential_evolution

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def custom_init(n, bounds):
    centers = np.zeros((34, 2))
    d_x = 0.18
    d_y = np.sqrt(3) * d_x
    a, b = -d_x/3, -d_y/3
    for i in range(5):
        for j in range(6):
            centers[0*5 + j, 0] = a + i*d_x
            centers[0*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2
            centers[1*5 + j, 0] = a + i*d_x
            centers[1*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2 + d_y
            centers[2*5 + j, 0] = a + (i + 2)*d_x
            centers[2*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2
            centers[3*5 + j, 0] = a + (i + 2)*d_x
            centers[3*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2 + d_y
            centers[4*5 + j, 0] = a + (i + 4)*d_x
            centers[4*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2
            centers[5*5 + j, 0] = a + (i + 4)*d_x
            centers[5*5 + j, 1] = b + j*d_y + (i % 2) * d_y / 2 + d_y

    centers[13:17, 0] = centers[13:17, 0] + 0.9
    centers[17:21, 1] = centers[17:21, 1] + 0.1
    centers[21:25, 0] = centers[21:25, 0] + 0.9
    centers[25, 1] = centers[25, 1] + 0.1
    centers = centers + np.random.normal(0, 0.01, centers.shape)
    return centers[:n]

def local_sa(centers):
    cur_centers = centers.copy()
    cur_radius, cur_sum = lp_optimum(cur_centers)
    step = 0.05
    T = 0.05
    for _ in range(5000):
        idx = np.random.randint(26)
        move = np.random.randn(2) * step
        new_centers = cur_centers.copy()
        new_centers[idx] += move
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radius, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum)/T):
            cur_centers, cur_radius, cur_sum = new_centers, new_radius, new_sum
        step = max(1e-4, step * 0.995)
        T = max(1e-5, T * 0.995)
    return cur_centers, cur_radius, cur_sum

def run_packing():
    n = 26
    dim = 52
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'

    def neg_sum_of_radii(x):
        centers = x.reshape((26, 2))
        radii, sum_r = lp_optimum(centers)
        return -sum_r

    init = lambda n, bounds: custom_init(n, bounds)
    result = differential_evolution(neg_sum_of_radii, bounds, maxiter=maxiter, popsize=popsize,
                                    strategy=strategy, init=init)

    best_centers = result.x.reshape((26, 2))
    best_radius, best_sum = lp_optimum(best_centers)
    if best_sum < 0:
        best_sum = 0

    return best_centers, best_radius, best_sum
