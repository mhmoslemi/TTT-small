import numpy as np
from scipy.optimize import minimize

def penalty(z, mu):
    n = len(z) // 3
    radii = z[n:]
    dists = np.sqrt(((z[:n].reshape(n, 2)[:, None, :] - z[:n, :].reshape(1, n, 2))**2).sum(axis=2))
    overlapping = np.sum(np.maximum(0, radii[:, None] + radii - dists), axis=1) ** 2
    boundary_sides = np.array([np.minimum(radii[:, None] - x, 0) for x in z[:n, 0]]).max(axis=1) ** 2
    boundary_top = np.array([np.minimum(radii[:, None] - y, 0) for y in z[:n, 1]]).max(axis=1) ** 2
    boundary_right = np.array([np.minimum(radii[:, None] - (1 - x), 0) for x in z[:n, 0]]).max(axis=1) ** 2
    boundary_left = np.array([np.minimum(radii[:, None] - (1 - y), 0) for y in z[:n, 1]]).max(axis=1) ** 2
    non_negative_radius = np.maximum(-radii, 0) ** 2
    return mu * (np.sum(overlapping) + np.sum(boundary_sides) + np.sum(boundary_top) + np.sum(boundary_right) + np.sum(boundary_left) + np.sum(non_negative_radius))

def objective(z):
    return -np.sum(z[len(z) // 3:])

def gradient(z, mu):
    n = len(z) // 3
    radii = z[n:]
    dists = np.sqrt(((z[:n].reshape(n, 2)[:, None, :] - z[:n, :].reshape(1, n, 2))**2).sum(axis=2))
    eps = 1e-8
    grad = np.zeros_like(z)

    for i in range(n):
        delta = np.zeros_like(z)
        delta[i] = [eps, 0]
        grad[i] = (penalty(z + delta, mu) - penalty(z - delta, mu)) / (2 * eps)

        delta[i] = [0, eps]
        grad[i + n] = (penalty(z + delta, mu) - penalty(z - delta, mu)) / (2 * eps)

    for i in range(n):
        delta[i * 3 + 2] = eps
        grad[n + i] = (penalty(z + delta, mu) - penalty(z - delta, mu)) / (2 * eps)

        if radii[i] > dists[i, 0] and radii[i] > dists[i, 3]:
            overlapped_with_left = np.where(dists[i, 0] - radii[i] - eps >= 0)[0]
        else:
            overlapped_with_left = []

        if radii[i] > dists[i, 1] and radii[i] > dists[i, 4]:
            overlapped_with_right = np.where(dists[i, 1] - radii[i] - eps >= 0)[0]
        else:
            overlapped_with_right = []

        if radii[i] > dists[i, 2] and radii[i] > dists[i, 5]:
            overlapped_with_bottom = np.where(dists[i, 2] - radii[i] - eps >= 0)[0]
        else:
            overlapped_with_bottom = []

        if radii[i] > dists[i, 3] and radii[i] > dists[i, 6]:
            overlapped_with_top = np.where(dists[i, 3] - radii[i] - eps >= 0)[0]
        else:
            overlapped_with_top = []

        for j in overlapped_with_left + overlapped_with_right + overlapped_with_bottom + overlapped_with_top:
            delta[j * 3 + 2] = eps
            grad[n + i] += 2 * dists[i, j] * (radii[i] + radii[j] - dists[i, j]) / dists[i, j]

    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    init_radii = 0.05 * np.ones(n)
    initial_guess = np.zeros(2 * n)
    initial_guess[:n] = np.random.uniform(0.05, 0.95, n)
    initial_guess[n:] = init_radii

    results = []
    for _ in range(25):
        result = minimize(objective, initial_guess, jac=gradient, method='L-BFGS-B', bounds=[(0, 1), (0, 1)] * n + [(-0, None)] * n)
        if result.success and result.fun >= 2.637:
            results.append((result.x[n:], result.fun))
        initial_guess[:n] += np.random.normal(0, 0.01, n)
        initial_guess[n:] += np.random.normal(0, 0.005, n)
        initial_guess[:n] = np.clip(initial_guess[:n], 0, 1)
        initial_guess[n:] = np.clip(initial_guess[n:], 0, None)

    best_result = max(results, key=lambda x: x[1])
    return best_result[0][:26], best_result[0][26:], best_result[1]
