import numpy as np
from scipy.optimize import minimize

def objective(z):
    return -np.sum(z[52:])

def constraint_center(z):
    x_c, y_c = z[0], z[1]
    r_c = z[26]
    return np.sqrt(x_c**2 + y_c**2) - r_c - np.sqrt(0.08) + 0.1

def constraint_non_negative_radius(z):
    return z[52:]

def constraint_dist(i, j, z):
    x1, y1, r1 = z[i], z[i + 26], z[52 + i]
    x2, y2, r2 = z[j], z[j + 26], z[52 + j]
    return r1 + r2 - np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

def run_packing():
    centers = np.zeros((26, 2))
    radius = 0.1
    step = 1 / np.sqrt(2)

    for i in range(26):
        if i < 9:
            centers[i, 0] = step * (2 * (i % 3) + 1)
        elif i < 18:
            centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
            centers[i, 1] = step + 0.5 * step
        else:
            centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
            centers[i, 1] = step + 0.5 * step * 2

    centers = centers.flatten()
    centers = np.pad(centers, (0, 26), mode='constant', constant_values=(0.5, 0.1828427125))
    initial_vector = np.concatenate([centers, np.full(26, np.nanmean(centers[26:52]))])

    A_ineq = []
    for i in range(26):
        A_ineq.append([0] * 52)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * 52)
        A_ineq[-1][i + 26] = 1
        A_ineq.append([0] * 52)
        A_ineq[-1][i] = 1
        A_ineq.append([0] * 52)
        A_ineq[-1][i + 26] = 1

    for i in range(26):
        for j in range(i + 1, 26):
            A_ineq.append([0] * 52)
            A_ineq[-1][i] = 1
            A_ineq[-1][i + 26] = 1
            A_ineq[-1][j] = 1
            A_ineq[-1][j + 26] = 1

    A_ineq = np.array(A_ineq)
    c = np.append(np.full(52, -np.inf), np.full(26, -1))
    b_ineq = np.append(np.full(52, 0), np.full(26, 0))
    bounds = [(0, None) for _ in range(78)]

    cv = [{'type': 'ineq', 'fun': constraint_center}, {'type': 'ineq', 'fun': constraint_non_negative_radius}]
    for i in range(26):
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i, i + 1, z)})
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i, i + 2, z)})
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i, i + 3, z)})
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i + 1, i + 2, z)})
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i + 1, i + 3, z)})
        cv.append({'type': 'ineq', 'fun': lambda z, i=i: constraint_dist(i + 2, i + 3, z)})
        for j in range(i + 4, 26):
            cv.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: constraint_dist(i, j, z)})

    result = minimize(
        fun=objective,
        x0=initial_vector,
        method='trust-constr',
        bounds=bounds,
        constraints=cv
    )

    if result.success:
        centers = result.x[:52].reshape((26, 2))
        radii = result.x[52:]
        radii[radii < 0] = 0
        return centers, radii, -result.fun
    else:
        raise Exception("Optimization failed")
