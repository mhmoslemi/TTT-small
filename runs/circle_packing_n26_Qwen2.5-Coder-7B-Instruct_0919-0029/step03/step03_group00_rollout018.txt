import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def objective(r):
    return -np.sum(r)

def constraint(r, centers):
    return np.sum(r) - 2.637

def boundary_constraint(r, i):
    return 0.5 * (1 - r[i] - abs(centers[i, 0] - 0.5) - abs(centers[i, 1] - 0.5))

def pairwise_constraint(r, centers):
    distances = distance_matrix(centers, centers)
    radii_matrix = np.outer(r, r)
    return (distances - radii_matrix) + 1e-12

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                      constraints=[({'type': 'eq', 'fun': lambda r, centers=centers: constraint(r, centers)}), 
                                  {'type': 'ineq', 'fun': lambda r: sum(boundary_constraint(r, i) for i in range(n))}]

    if result.success:
