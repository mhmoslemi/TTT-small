import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    flat_centers = centers.flatten()
    B = np.min(np.column_stack((centers[:, 0] - centers[:, 1], centers[:, 0] + centers[:, 1],
                                centers[:, 1] - centers[:, 0], centers[:, 1] + centers[:, 0])), axis=1)
    d = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d[i, j] = np.linalg.norm(centers[i] - centers[j])
            d[j, i] = d[i, j]
    lhs_ineq = np.zeros((n*(n-1)//2, 2*n))
    for idx, (i, j) in enumerate([(i, j) for i in range(n) for j in range(i+1, n)]):
        lhs_ineq[idx, i*2] = 1
        lhs_ineq[idx, i*2+1] = 1
        lhs_ineq[idx, j*2] = 1
        lhs_ineq[idx, j*2+1] = 1
        lhs_ineq[idx, i*2] *= -1
        lhs_ineq[idx, j*2] *= -1
    b_ineq = d.flatten()
    bounds = [(0, bi) for bi in B]
    result = linprog([-1]*n, A_ub=lhs_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    if result.success:
        radii = -result.x / 2
        return centers, radii
    else:
        return centers, np.full(n, -np.inf)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonal grid seed
    s = 0.099
    centres = np.array([(s + (j + (i % 2)) * 2 * s, s + i * np.sqrt(3) * s)
                        for i in range(5) for j in range(0 if i % 2 == 0 else 4, 5)])
    side_midpoints = np.array([[0.5, 0.05], [0.95, 0.5], [0.05, 0.5]])
    centres = np.vstack((centres, side_midpoints))

    epsilon = 1e-4
    sigma = 0.1
    alpha = 0.9999

    best_sum = -np.inf
    best_centres = None
    T = 0.1

    for _ in range(20000):
        for i in range(26):
            delta_x = np.random.normal(0, sigma)
            delta_y = np.random.normal(0, sigma)
            new_centres = centres.copy()
            new_centres[i] += [delta_x, delta_y]
            new_centres[i] = np.clip(new_centres[i], 0, 1)
            if np.any(np.linalg.norm(new_centres[i] - centres[:i], axis=1) < epsilon
                       if i > 0 else np.linalg.norm(new_centres[i] - centres[i+1:], axis=1) < epsilon):
                continue
            new_result = solve_radii(new_centres)
            new_sum = np.sum(new_result[1])
            if new_sum > best_sum or (np.isclose(new_sum, best_sum) and np.sum(new_result[0]) < np.sum(best_centres)):
                best_sum = new_sum
                best_centres = new_result[0]
        if np.isclose(best_sum, 2.637, atol=1e-3):
            break
        T *= alpha

    return best_centres[:26], best_centres[26:], best_sum
