import numpy as np
from scipy.optimize import linprog

def compute_radius_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    A_eq = np.zeros((n, n))
    b_eq = np.full(n, 1)

    for i in range(n):
        x, y = centers[i]
        A_eq[i, i] = 1
        b_eq[i] = np.clip(1 - x, 0, 1)
        A_eq[i, i] = 1
        b_eq[i] = np.clip(1 - y, 0, 1)
        A_eq[i, i] = 1
        b_eq[i] = np.clip(x, 0, 1)
        A_eq[i, i] = 1
        b_eq[i] = np.clip(y, 0, 1)

    for i in range(n):
        for j in range(i + 1, n):
            A_eq[i, j] = 1
            A_eq[j, i] = 1
            b_eq[i] = np.clip(np.linalg.norm(centers[i] - centers[j]) - 1e-6, 0, 1)

    bounds = [(0, None)] * n
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    patterns = [
        lambda: np.random.rand(n, 2),
        lambda: np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)]),
        lambda: np.array([[0.1 + 0.25 * i if j == 0 or j == 4 else 0.125 + 0.5 * i, 0.1 + 0.5 * j] for j in range(5) for i in range(5)]),
        lambda: np.array([[0.1, 0.3, 0.5, 0.7, 0.9], [0.3, 0.5, 0.7, 0.9, 0.1], [0.5, 0.7, 0.9, 0.1, 0.3], [0.7, 0.9, 0.1, 0.3, 0.5], [0.9, 0.1, 0.3, 0.5, 0.7]]).T
    ]
    stars = 100
    iters_per_star = 2000
    step = 0.1
    step_decay = 0.99
    eps = 1e-6

    for pattern in patterns:
        for _ in range(stars):
            centers = pattern()
            best_of_star = -np.inf
            for _ in range(iters_per_star):
                idx = np.random.randint(n)
                dir = np.random.randn(2)
                dir /= np.linalg.norm(dir)
                delta = dir * step
                new_centers = centers.copy()
                new_centers[idx] += delta
                new_centers = np.clip(new_centers, 0, 1)
                new_radii, new_sum = compute_radius_lp(new_centers)
                if new_sum > best_of_star:
                    best_of_star = new_sum
                    centers = new_centers.copy()
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = new_radii
            step *= step_decay

    return best_centers, best_radii, best_sum
