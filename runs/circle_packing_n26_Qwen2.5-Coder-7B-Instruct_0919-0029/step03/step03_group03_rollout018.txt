import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    s = 0.15  # Base spacing for hexagonal lattice
    centers = np.array([
        [0.5, s * 0.75],
        [0.5, s],
        [0.5 + s * 0.5, s * 1.25],
        [0.5 + s * 0.5, s + s * 1.25],
        [0.5 + s, s * 0.75],
        [0.5 + s, s],
        [0.25, s * 1.5],
        [0.75, s * 1.5],
        [0.5, s * 2.25],
        [0.25, s * 2.5],
        [0.75, s * 2.5],
        [0.5, s * 3],
        [0.25, s * 3.5],
        [0.75, s * 3.5],
        [0.5, s * 4.25],
        [0.25, s * 4.5],
        [0.75, s * 4.5],
        [0.5 + s * 0.5, s * 2.75],
        [0.5 + s * 0.5, s + s * 3.25],
        [0.5 + s, s * 2.75],
        [0.5 + s, s + s * 3.25],
        [0.5 - s, s],
        [0.5 - s, s + s * 1.25],
        [0.5 - s * 0.5, s * 0.75],
        [0.5 - s * 0.5, s + s * 2.5]
    ])

    centers[[20, 21, 22, 23, 24, 25]] += np.array([
        [0.5, 0.05],
        [0.95, 0.5],
        [0.05, 0.5],
        [0.5, 0.05],
        [0.95, 0.5],
        [0.05, 0.5]
    ])

    def lp_solve(centers):
        bounds = [(0, B) for x, y in centers for B in min(x, 1 - x, y, 1 - y)]
        dist = np.array([np.linalg.norm(centers[i] - centers[j]) for i in range(n) for j in range(i + 1, n)])
        c = np.ones(n)
        A = np.column_stack((np.ones(n), dist))
        b = s * np.ones_like(dist)
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        return result.x, result.fun

    def sa(centers):
        T = 0.1
        alpha = 0.9999
        max_steps = 20000
        best_centers = centers
        best_radii = np.array([0.01] * n)
        best_sum = np.sum(best_radii)

        for step in range(max_steps):
            i = np.random.randint(n)
            eps = 1e-4
            new_centers = centers.copy()
            new_centers[i] += np.array([np.random.normal(0, 1 * eps), np.random.normal(0, 1 * eps)])
            new_centers[i] = np.clip(new_centers[i], 0, 1)
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(new_centers[i] - new_centers[j])
                    if dist < 3 * eps:
                        step += max_steps
                        break

            new_radii, new_sum = lp_solve(new_centers)
            if new_sum > best_sum:
                best_centers = new_centers
                best_radii = new_radii
                best_sum = new_sum
            else:
                if np.random.rand() < np.exp((new_sum - best_sum) / T):
                    best_centers = new_centers
                    best_radii = new_radii

            T *= alpha

        return best_centers, best_radii, best_sum

    def gradient_polishing(centers, radii):
        delta = 1e-4
        for i in range(n):
            x, y = centers[i]
            dx = np.zeros_like(centers)
            dy = np.zeros_like(centers)
            dx[i] = delta
            dy[i] = delta
            grad_x = (lp_solve(centers + dx)[1] - lp_solve(centers - dx)[1]) / (2 * delta)
            grad_y = (lp_solve(centers + dy)[1] - lp_solve(centers - dy)[1]) / (2 * delta)
            if grad_x**2 + grad_y**2 > 0:
                if grad_x**2 + grad_y**2 > 1e-6:
                    rv, rr, rs = lp_solve(centers + np.sign([grad_x, grad_y]) * delta * np.array([1, 1]))
                    if rs > lp_solve(centers)[1]:
                        centers[i] = rv
                        radii = rr
        return centers, radii, np.sum(radii)

    centers, radii, sum_radii = sa(centers)
    centers, radii, sum_radii = gradient_polishing(centers, radii)

    return centers, radii, sum_radii
