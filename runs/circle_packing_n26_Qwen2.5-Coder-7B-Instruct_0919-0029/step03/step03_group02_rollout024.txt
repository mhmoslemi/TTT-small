import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def objective(z):
    return -z[26:]

def constraint_distances(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    dists = np.sqrt((x[:, np.newaxis] - x) ** 2 + (y[:, np.newaxis] - y) ** 2)
    np.fill_diagonal(dists, np.inf)
    return r[:, np.newaxis] + r - dists  # Ensure non-overlap

def constraint_boundaries(r, x, y):
    return np.min([x - r, 1 - x - r, y - r, 1 - y - r], axis=0)

def run_packing():
    np.random.seed(0)
    n = 26
    numstarts = 5

    best_centers = np.zeros((26, 2))
    best_radii = np.zeros(26)
    best_sum = -np.inf

    for _ in range(numstarts):
        centers = np.zeros((26, 2))
        radius = 0.10
        step = 1 / np.sqrt(2)
        for i in range(5):
            centers[i      , 0] = step * (2 * (i % 3) + 1)
            centers[5 + i   , 0] = step * (2 * ((i - 0) % 3) + 1)
            centers[10 + i  , 0] = step * (2 * ((i - 0) % 3) + 1)
        for i in range(9):
            centers[i      , 1] = step * (2 * (i % 3) + 1)
            centers[15 + i  , 1] = step * (2 * ((i - 0) % 3) + 1)
            centers[24 + i  , 1] = step * (2 * ((i - 0) % 3) + 1)
        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
        centers -= np.min(centers, axis=0)
        centers /= np.max(centers, axis=0)
        centers *= 0.98
        centers += 0.01

        r_center = np.sqrt(0.08) - 0.1
        centers[12] = np.array([0.5, 0.5])
        radii = np.ones(26) * r_center
        if r_center > 0:
            initial_vector = np.concatenate((centers.ravel(), radii))
        else:
            initial_vector = np.random.rand(78)

        cons = []
        
        for i in range(n):
            for j in range(n):
                if i != j:
                    cons.append(NonlinearConstraint(lambda z, i=i, j=j: z[2*n+i] + z[2*n+j] - np.sqrt((z[i] - z[j])**2 + (z[j+n] - z[i+n])**2), 0, 0))

        for i in range(n):
            cons.append(NonlinearConstraint(lambda z, i=i: (z[i] + z[n+i] - 1) if z[i] >= z[n+i] else (z[n+i] + z[i] - 1), 0, 0))
            cons.append(NonlinearConstraint(lambda z, i=i: (z[i + n] + z[i] - 1) if z[n+i] >= z[i] else (z[i] + z[n+i] - 1), 0, 0))
        
        res = minimize(objective, initial_vector, constraints=cons, method='trust-constr', options={'max_iter': 2000})

        centers = res.x[:26].reshape((26, 2))
        radii = res.x[26:]
        total_radius = np.sum(radii)

        if total_radius > best_sum:
            best_sum = total_radius
            best_centers = centers
            best_radii = radii

    result = minimize(objective, best_centers.ravel(), constraints=cons, method='trust-constr', options={'max_iter': 2000})

    centers = result.x[:26].reshape((26, 2))
    radii = result.x[26:]
    total_radius = np.sum(radii)

    return centers, radii, total_radius
