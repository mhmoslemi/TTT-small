import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([
        [(i + 0.5 * (i % 2 == 1)) * 0.366, (j * 0.577 + (j % 2 == 0)) * 0.289]
        for i in range(5) for j in range(5)
    ])
    centers = np.vstack([centers, [[0.5, 0.05], [0.95, 0.5], [0.05, 0.5]]])
    np.random.shuffle(centers)

    radii = np.ones(n) * 0.01
    best_centers, best_radii = centers.copy(), radii.copy()
    best_sum = evaluate(centers, radii)

    T = 0.1
    while T > 1e-6:
        for _ in range(2000):
            i = np.random.randint(n)
            noise = np.random.randn(2) * (T / 2)
            new_center = centers[i] + noise
            new_center = np.clip(new_center, 0, 1)
            if min([np.linalg.norm(new_center - c) ** 2 for c in centers[:i]]) < 1e-4:
                continue
            new_centers = centers.copy()
            new_centers[i] = new_center
            new_radii = solve_radii(new_centers)
            if new_radii is not None and np.all(new_radii >= 0):
                new_sum = np.sum(new_radii)
                if new_sum > best_sum:
                    best_sum, best_centers, best_radii = new_sum, new_centers, new_radii
                elif np.random.rand() < np.exp((new_sum - best_sum) / T):
                    centers, radii = new_centers, new_radii
        T *= 0.9999

    refine(centers, best_radii)
    return best_centers, best_radii, np.sum(best_radii)

def solve_radii(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    c = np.zeros(26)
    A = np.vstack((np.eye(26), -np.eye(26)))
    b = np.hstack((np.array([0.5] * n + [0.5] * n), np.zeros(2 * n)))
    A_e = np.vstack((np.c_[centers[:, 0], np.zeros(n)], np.c_[centers[:, 1], np.zeros(n)]))
    A_i = np.vstack((np.c_[np.zeros(n), centers[:, 0]], np.c_[np.zeros(n), centers[:, 1]]))
    b_i = np.hstack((1 - centers[:, 0], 1 - centers[:, 1]))
    con = [{'type': 'ineq', 'fun': lambda x, i=i, ci=A_e[i], cii=A_i[i]: ci @ x - cii @ x}, {'type': 'ineq', 'fun': lambda x, c=c: c - x}]
    opt = linprog(c, A_ub=A, b_ub=b, A_eq=A_e, b_eq=c, A_int=A_i, bounds=(0, 0.5), method='highs', options={'abstol': 1e-12})
    return opt.x if opt.success else None

def refine(centers: np.ndarray, radii: np.ndarray):
    n = centers.shape[0]
    for _ in range(100):
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-4:
                    dist = d - (radii[i] + radii[j] - 1e-4)
                    centers[i] -= dist * (centers[i] - centers[j]) / d
                    centers[j] += dist * (centers[j] - centers[i]) / d
                    new_radii = solve_radii(centers)
                    if new_radii is not None:
                        radii = new_radii
                        break
        else:
            break

def evaluate(centers: np.ndarray, radii: np.ndarray) -> float:
    n = centers.shape[0]
    c = np.zeros(2 * n)
    A = np.vstack((np.eye(2 * n), -np.eye(2 * n)))
    b = np.hstack((np.array([0.5] * n + [0.5] * n), np.zeros(2 * n)))
    A_e = np.vstack((np.c_[centers[:, 0], np.zeros(n)], np.c_[centers[:, 1], np.zeros(n)]))
    A_i = np.vstack((np.c_[np.zeros(n), centers[:, 0]], np.c_[np.zeros(n), centers[:, 1]]))
    b_i = np.hstack((1 - centers[:, 0], 1 - centers[:, 1]))
    con = [{'type': 'ineq', 'fun': lambda x, i=i, ci=A_e[i], cii=A_i[i]: ci @ x - cii @ x}, {'type': 'ineq', 'fun': lambda x, c=c: c - x}]
    opt = linprog(c, A_ub=A, b_ub=b, A_eq=A_e, b_eq=c, A_int=A_i, bounds=(0, 0.5), method='highs', options={'abstol': 1e-12})
    return opt.fun if opt.success else -np.inf
