import numpy as np
from scipy.optimize import linprog

def lp_solve(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Boundary constraints
    bounds = [(0, None) for _ in range(n)]
    b_eq = np.concatenate([np.ones(n), -np.ones(n), np.zeros(2*n), np.ones(n), -np.ones(n)])
    c = np.concatenate([np.zeros(n), np.zeros(n), -np.concatenate([radii, radii]), np.zeros(n), np.zeros(n)])
    
    A_eq = np.array([
        np.hstack([np.eye(n), -np.eye(n), np.zeros((n, 2*n)), -np.eye(n), np.zeros((n, n))]),
        np.hstack([np.zeros((n, n)), np.zeros((n, n)), -np.eye(n), np.zeros((n, 2*n)), np.eye(n)]),
        np.hstack([np.zeros((n, n)), np.zeros((n, n)), np.zeros((n, 2*n)), -np.eye(n), np.zeros((n, n))]),
        np.hstack([np.zeros((n, n)), np.zeros((n, n)), np.zeros((n, 2*n)), np.eye(n), -np.eye(n)])
    ])
    
    A = np.vstack([
        A_eq,
        np.hstack([np.zeros((n, n)), np.zeros((n, n)), -np.eye(n), np.zeros((n, 2*n)), np.eye(n)]),
        np.hstack([np.zeros((n, n)), np.zeros((n, n)), -np.eye(n), np.eye(n), -np.eye(n)]),
        np.hstack([np.zeros((n, n)), np.eye(n), -np.eye(n), -np.eye(n), np.eye(n)])
    ])
    
    b = np.hstack([np.fft.fftshift(np.useconds_rfft(np.sum(c[:n]), n, 0)), 1, 1, 1])
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        return None
    radii = res.x[:n]
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([
        (np.arange(5), np.array([0.05, 0.5, 0.95]) + np.arange(5)*0.21650635094610966),
        (np.arange(4), np.array([0.05, 0.5, 0.95])) + np.arange(5, 9)*0.21650635094610966,
        (np.arange(4), np.array([0.05, 0.5, 0.95])) + np.arange(9, 13)*0.21650635094610966,
        (np.arange(3), np.array([0.05, 0.5, 0.95])) + np.arange(13, 15)*0.21650635094610966,
        (np.arange(3), np.array([0.05, 0.5, 0.95])) + np.arange(15, 17)*0.21650635094610966,
        (np.arange(2), np.array([0.05, 0.5])) + np.arange(17, 19)*0.21650635094610966,
        (np.arange(2), np.array([0.05, 0.5])) + np.arange(19, 21)*0.21650635094610966,
        (np.arange(2), np.array([0.05, 0.5])) + np.arange(21, 23)*0.21650635094610966,
        (0.5,), (0.95,), (0.05,)
    ])
    
    best_sum = 0
    best_centers = None
    best_radii = None
    T = 0.1
    cooling_rate = 0.9999
    
    for _ in range(20000):
        idx = np.random.randint(n)
        noise = np.random.randn(2)*T/2
        new_center = np.clip(centers[idx] + noise, 0, 1)
        
        if np.any((new_center - centers[:, :2])**2 < 1e-8).any():
            continue
        
        new_centers = centers.copy()
        new_centers[idx] = new_center
        
        radii = lp_solve(new_centers)
        if radii is None:
            continue
        
        new_sum = np.sum(radii)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
            best_radii = radii
        elif np.random.rand() < np.exp((new_sum - best_sum) / T):
            best_sum = new_sum
            best_centers = new_centers
            best_radii = radii
        
        T *= cooling_rate
    
    return best_centers[:, :2], best_radii, best_sum

best_centers, best_radii, best_sum = run_packing()
print(f"Sum of radii: {best_sum}")
