import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers, radius=0.1):
    n = centers.shape[0]
    A_ub = np.zeros((2 * n, n))
    b_ub = np.zeros(2 * n)

    # Radius constraints
    for i in range(n):
        A_ub[i, :] = 2 * np.ones(n)
        b_ub[i] = radius

    # Distance constraints
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j])
            idx += 1

    c = np.zeros(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub[b_ub > 0], b_ub=b_ub[b_ub > 0], bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_STARS = 10
    NUM_ITERS = 2000
    step_start = 0.05
    step_decay = 0.997
    T_start = 1.0
    T_decay = 0.999

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
        for _ in range(NUM_ITERS):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step_start
            new_c = centers[idx] + delta
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii

            # Simulated annealing
            if np.random.rand() < np.exp((new_sum - best_sum) / T_start):
                centers = new_centers.copy()
            step_start *= step_decay
            T_start *= T_decay

    return best_centers, best_radii, best_sum
