import numpy as np
from scipy.optimize import minimize

def objective(z):
    return -sum(z[26:])

def constraint_plane_dist(z, i, j):
    dx = z[i] - z[j]
    dy = z[i+26] - z[j+26]
    return z[j+26] + z[i+26] - np.sqrt(dx**2 + dy**2)

def constraint_box_dist(z, i, j):
    dx = z[i] - z[j]
    dy = z[i+26] - z[j+26]
    return z[j+26] + z[i+26] - np.sqrt(dx**2 + dy**2) - 1e-12

def run_packing() -> tuple:
    centers = np.zeros((25, 2))
    radius = 0.10
    step = 1 / np.sqrt(2)

    for i in range(5):
        for j in range(5):
            centers[i*5+j, 0] = 0.1 + 0.2 * i
            centers[i*5+j, 1] = 0.1 + 0.2 * j

    center_x, center_y = 0.5, 0.5
    center_radius = np.sqrt(0.08) - 0.1
    centers = np.vstack((centers, [center_x, center_y]))
    radius = np.append(radius * np.ones(25), center_radius)

    initial_vector = centers.flatten().tolist() + radius.tolist()

    warm_start_bounds = [(0, 1) for _ in range(50)]
    center_radius_bound = (0, np.inf)
    bounds = warm_start_bounds + 25 * [center_radius_bound]

    non_overlap_constraints = []
    for i in range(50):
        for j in range(i + 1, 50):
            if i < 25 and j < 25:
                constraint = {'type': 'ineq', 'fun': constraint_plane_dist, 'args': (i, j)}
            else:
                constraint = {'type': 'ineq', 'fun': constraint_box_dist, 'args': (i, j)}
            non_overlap_constraints.append(constraint)

    constraints = [{'type': 'ineq', 'fun': lambda z, i=i: z[i+26] - z[i] - 1e-12} for i in range(25)] + \
                 [{'type': 'ineq', 'fun': lambda z, i=i: z[i+26] - (1 - z[i]) - 1e-12} for i in range(25)] + \
                 [{'type': 'ineq', 'fun': lambda z, i=i: z[i+26] - z[i] - 1e-12} for i in range(25)] + \
                 [{'type': 'ineq', 'fun': lambda z, i=i: z[i+26] - (1 - z[i]) - 1e-12} for i in range(25)] + \
                 non_overlap_constraints

    result = minimize(
        objective,
        x0=initial_vector,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints,
        options={'max_iter': 1000, 'gtol': 1e-8, 'xtol': 1e-8, 'disp': False}
    )

    if result.success:
        optimized_centers = np.reshape(result.x[:50], (25, 2))
        optimized_radii = result.x[50:]
        return optimized_centers, optimized_radii, -result.fun
    else:
        return np.zeros((26, 2)), np.zeros(26), 0.0
