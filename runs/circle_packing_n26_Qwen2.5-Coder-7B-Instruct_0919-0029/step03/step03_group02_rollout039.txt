def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def constraints(x):
    n = 26
    x = x.reshape(n, 3)
    centers = x[:, :2]

    A = np.zeros((n + 4 * n + n * (n - 1) // 2, 3 * n))
    b = np.zeros(n + 4 * n + n * (n - 1) // 2)
    A[:n, np.arange(n) * 3 + 2] = -1
    A[n:n + 4 * n, np.arange(n) * 3 + 2: np.arange(n) * 3 + 5] = np.eye(4 * n)
    for i in range(n):
        for j in range(i + 1, n):
            A[n + 4 * n + i * (n - 1) // 2 + (j - i - 1), np.arange(n) * 3 + 2] = 1
            A[n + 4 * n + i * (n - 1) // 2 + (j - i - 1), j * 3 + 2] = 1

                np.linalg.norm(centers[i] - centers[j]) - radii[i] - radii[j]
            )
    return {'type': 'ineq', 'fun': lambda x: A @ x.flatten() - b}
