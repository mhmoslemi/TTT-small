import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers, active_pairs):
    n = centers.shape[0]
    A_ub = np.zeros((n * (n - 1) // 2 + 4 * n + 1, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            if (i, j) in active_pairs or (j, i) in active_pairs:
                A_ub[idx, i] = 1
                A_ub[idx, j] = 1
                b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
                idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    A_ub[idx, 0:] = 1
    b_ub[idx] = np.inf
    idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def detect_active_pairs(centers, radii):
    active_pairs = set()
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                active_pairs.add((i, j))
    return active_pairs

def run_packing():
    n = 26
    step = 0.15
    step_min = 1e-4
    decay = 0.9995
    NUM_ITERS = 30000
    NUM_STARS = 15

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for rep in range(NUM_STARS):
        if rep == 0:
            # Deterministic hexagon + corners
            hexagon = np.array([
                [1 / 2, np.sqrt(3) / 6], [1 / 2, 2 * np.sqrt(3) / 6], [1 / 2 - np.sqrt(3) / 6, 0],
                [0, 1 / 3], [1 / 3, 1 / 3], [2 / 3, 2 / 3], [1 / 2 + np.sqrt(3) / 6, 1],
                [1, 1 / 3], [3 / 2 - np.sqrt(3) / 6, 1 - 2 * np.sqrt(3) / 6], [3 / 2 - np.sqrt(3) / 6, 1 - np.sqrt(3) / 6],
                [1.5, 2 * np.sqrt(3) / 6], [1.5, np.sqrt(3) / 6], [1, 1 / 3], [1 / 2, 2 * np.sqrt(3) / 6],
                [1 / 2, 1], [1 / 2 - np.sqrt(3) / 6, 1], [0, 1 - 1 / 3], [1 / 2 - np.sqrt(3) / 6, 1.5],
                [3 / 2 - np.sqrt(3) / 6, 1.5], [1 / 2, 1 - np.sqrt(3) / 6], [1.5, 1 - np.sqrt(3) / 6],
                [1.5, 1], [1, 1 - 1 / 3], [2 / 3, 1], [1 / 2, 1 - 2 * np.sqrt(3) / 6]
            ])
            factor_hex = 1 / (1 / (2 * np.linalg.norm(hexagon[0] - hexagon[1])) / (np.cos(np.pi / 3) + np.sqrt(3) / 2) - hexagon[0, 0])
            hexagon *= factor_hex
            hexagon += np.array([0.5, 0.5])
            centers = hexagon[:n]

        elif rep == 1:
            # Deterministic hexagon + jitter
            centers = hexagon + np.random.normal(0, 0.01, size=(n, 2))

        elif rep == 2:
            # Random uniform centers
            centers = np.random.uniform(0.15, 0.85, size=(n, 2))

        elif rep == 3:
            # Quasi-random Halton sequence
            sequence = np.array([
                [0.33204353, 0.38288482], [0.69434338, 0.77582862], [0.32602192, 0.38064618],
                [0.15985385, 0.64946272], [0.99075159, 0.03933564], [0.66823867, 0.01290748],
                [0.67548179, 0.09193422], [6.70415114e-07, 0.06308786], [0.2740874, 0.11820092],
                [0.74632978, 0.84214138], [0.4644911, 0.01613443], [0.53647759, 0.97759029],
                [0.34564429, 0.70723655], [0.0435702, 0.82550237], [0.02869441, 0.84749178],
                [0.22722213, 0.0262682], [0.04707339, 0.25789928], [0.03497632, 0.406909],
                [0.44974276, 0.58121227], [0.72151536, 0.79998854], [0.178607, 0.67952844],
                [0.02198979, 0.65350641], [0.4509495, 0.77192018], [0.98431561, 0.7180665],
                [0.41056958, 0.48617828], [0.04174869, 0.61740719]
            ]) * 0.7
            centers += np.tile(np.array([0.15, 0.15]), (26, 1))

        else:
            centers = hexagon[:n] + np.random.normal(0, 0.01, size=(n, 2))

        for _ in range(NUM_ITERS):
            idx = np.random.randint(n)
            direction = np.random.randn(2)
            direction /= np.linalg.norm(direction)
            delta = direction * step
            new_c = centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_c, active_pairs)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_c
                best_radii = new_radii

            step *= decay
            step = max(step, step_min)

        active_pairs = detect_active_pairs(best_centers, best_radii)

    full_radii, full_sum = lp_optimum(best_centers, active_pairs=active_pairs)
    return best_centers, full_radii, full_sum
