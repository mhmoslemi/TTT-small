import numpy as np
from scipy.optimize import linprog

eps = 1e-12

def solve_lp(centers):
    n = centers.shape[0]
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] + eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] + eps)
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - eps)

        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - eps)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x, -result.fun
    else:
        raise Exception("LP optimization failed")

def objective(centers):
    radii, _ = solve_lp(centers)
    return -radii.sum()

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    temperature = 1.0
    sigma = 2e-2
    iterations = 2000

    for _ in range(iterations):
        for i in range(n):
            delta = np.random.normal(scale=sigma, size=2)
            candidate = centers.copy()
            candidate[i] += delta
            candidate = np.clip(candidate, 0, 1)
            objective_candidate, _ = objective(candidate)
            if objective_candidate > objective(centers):
                centers = candidate.copy()
            else:
                if np.random.rand() < np.exp((objective_candidate - objective(centers)) / temperature):
                    centers = candidate.copy()
        temperature *= 0.99
        sigma *= 0.99

    best_centers, best_sum, best_radii = centers, objective(centers), solve_lp(centers)[0]

    return best_centers, best_radii, -best_sum
