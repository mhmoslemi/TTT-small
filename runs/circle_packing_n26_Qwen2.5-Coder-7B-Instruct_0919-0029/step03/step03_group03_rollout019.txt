import numpy as np
from scipy.optimize import linprog

def solve_radii(centres):
    n = centres.shape[0]
    B = np.minimum(np.minimum(centres[:, 0], 1 - centres[:, 0]), np.minimum(centres[:, 1], 1 - centres[:, 1]))
    D = np.sqrt(((centres[:, np.newaxis, 0] - centres[np.newaxis, :, 0]) ** 2 + (centres[:, np.newaxis, 1] - centres[np.newaxis, :, 1]) ** 2))
    D -= np.eye(n)
    A = np.vstack([B, -D.T, -D]).T
    b = np.hstack([B, np.zeros_like(D.T.flatten())])
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs', tol=1e-12)
    if not res.success:
        return None, float('-inf')
    radii = res.x[:n]
    S = -res.fun
    return radii, S

    current_grid = np.array([[x, y] for y in y_positions for x in x_positions[:n]])
