import numpy as np
from scipy.optimize import linprog

def propose_move(centers, step):
    idx = np.random.randint(26)
    dir = np.random.randn(2)
    dir /= np.linalg.norm(dir)
    Δ = dir * step
    return idx, Δ

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        return np.zeros(n), 0
    return result.x, -result.fun

def reinitialise_random():
    n = 26
    step_min, decay_factor = 1e-4, 0.995
    NUM_ITERS = 4000
    NUM_STARS = 20
    centers = np.random.rand(n, 2)
    for it in range(NUM_ITERS):
        idx = np.random.randint(n)
        dir = np.random.randn(2)
        dir /= np.linalg.norm(dir)
        delta = dir * step_min
        new_c = centers[idx] + delta
        new_c = np.clip(new_c, 0, 1)
        centers[idx] = new_c
    return centers, 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    EPS = 1e-12
    step = 0.05
    step_min, step_decay = 1e-4, 0.995
    T0 = 1e-3 * 2.7  # Roughly a thousandth of the best possible sum
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500
    NUM_STARS = 20
    vecs = [
        np.random.rand(2)*0.9 + 0.05,
        0.1*np.array([[1, 0], [0, 1]]) @ np.random.permutation(np.arange(1, 6)).reshape((5, 5)).T + 0.3,
        (0.1+0.1*np.arange(5))*np.array([[1, np.sqrt(3)], [-np.sqrt(3), 1]]) @ np.random.permutation(np.arange(1, 6)).reshape((5, 5)).T + 0.3,
        np.random.choice(np.array([[1, 0], [0, 1]], dtype=np.float), size=(25,)) * 0.3 + 0.3,
        np.random.rand(20)*0.9 + 0.05 + 0.15 * np.tile(np.array([[0, 1], [-1, 0]]), (6, 1))
    ]

    best_centers, best_radii, best_sum = None, None, -np.inf

    for start in range(NUM_STARS):
        if best_sum >= 2.637:
            break
        centers = vecs[start % len(vecs)]
        if step > step_min:
            for _ in range(ITER_PER_PATTERN):
                idx, Δ = propose_move(centers, step)
                new_c = centers[idx] + Δ
                new_c = np.clip(new_c, 0, 1)
                new_centers = centers.copy()
                new_centers[idx] = new_c
                new_radii, new_sum = lp_optimum(new_centers)
                if new_sum > best_sum:
                    if new_sum > 2.637:
                        return new_centers, new_radii, new_sum
                    best_sum = new_sum
                    best_centers, best_radii = new_centers.copy(), new_radii.copy()
                step *= step_decay

        if step > step_min:
            step *= step_decay
            T = T0 * (step * T_decay) / step
            while True:
                if step < step_min:
                    break
                idx, Δ = propose_move(best_centers, step)
                new_c = best_centers[idx] + Δ
                new_c = np.clip(new_c, 0, 1)
                new_centers = best_centers.copy()
                new_centers[idx] = new_c
                new_radii, new_sum = lp_optimum(new_centers)
                Δsum = new_sum - best_sum
                if Δsum >= 0 or np.exp(Δsum / T) > np.random.rand():
                    best_sum = new_sum
                    best_centers = new_centers.copy()
                    best_radii = new_radii.copy()
                step *= step_decay
                T *= T_decay

        if start % MAX_NO_IMPROV == 0:
            centers, _ = reinitialise_random()
            best_sum = -np.inf

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
