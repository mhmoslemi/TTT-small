import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def solve_lp(centers):
    return lp_optimum(centers)

def compute_dual_gradient(centers, radii):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    result = linprog(None, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    dual = result.dual
    grad = np.zeros((n, 2))
    for i in range(n):
        r_i, x_i, y_i = radii[i], centers[i, 0], centers[i, 1]
        for j in range(n):
            if i != j:
                r_j = radii[j]
                x_j, y_j = centers[j, 0], centers[j, 1]
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < r_i + r_j:
                    grad[i] += (radii[j] - x_j + x_i - y_j + y_i) / (dist * (dist - r_i - r_j))
    return grad

def propose_move(centers, grad, step):
    n = centers.shape[0]
    new_centers = np.copy(centers)
    for i in range(n):
        g_norm = np.linalg.norm(grad[i])
        if g_norm > 1e-12:
            new_centers[i] += step * (grad[i] / g_norm)
    return new_centers

def clip_to_square(centers, eps):
    n = centers.shape[0]
    new_centers = np.copy(centers)
    for i in range(n):
        new_centers[i, 0] = np.clip(new_centers[i, 0], eps, 1 - eps)
        new_centers[i, 1] = np.clip(new_centers[i, 1], eps, 1 - eps)
    return new_centers

def alpha(t):
    return 0.1 * (0.995 ** t)

def temp(t):
    return 0.5 * (0.995 ** t)

def generate_pattern(pattern_name):
    if pattern_name == 'Random_inner':
        return np.random.rand(26, 2) * 0.9 + 0.05
    elif pattern_name == 'Grid5x5_cut':
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        centers = np.zeros((26, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if not (i == 0 and j == 0):
                    centers[idx, 0] = x[i]
                    centers[idx, 1] = y[j]
                    idx += 1
        centers[0] = np.random.rand(2) * 0.9 + 0.05
        return centers
    elif pattern_name == 'OffsetGrid':
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        centers = np.zeros((26, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx, 0] = x[i] + (i % 2) * 0.1
                centers[idx, 1] = y[j]
                idx += 1
        return centers
    elif pattern_name == 'SparseHex':
        x = np.arange(0.15, 0.85, 0.15)
        y = np.arange(0.07, 0.93, 3 * 0.07)
        centers = np.zeros((30, 2))
        idx = 0
        for i in y:
            for j in x:
                centers[idx, 0] = j
                centers[idx, 1] = i
                idx += 1
        centers = centers[:26]
        centers[0:4] = centers[4:8]
        centers[4:8] = centers[8:12]
        centers[8:12] = centers[12:16]
        centers[12:16] = centers[16:20]
        centers[16:20] = centers[20:24]
        return centers
    elif pattern_name == 'CornerExpansion':
        centers = np.random.rand(18, 2) * 0.8 + 0.05
        centers[18:26] = np.array([
            [0.15, 0.45], [0.15, 0.55], [0.15, 0.65], [0.15, 0.75],
            [0.25, 0.4], [0.25, 0.45], [0.25, 0.55], [0.25, 0.65], [0.25, 0.75],
            [0.35, 0.4], [0.35, 0.45], [0.35, 0.55], [0.35, 0.65], [0.35, 0.75],
            [0.85, 0.4], [0.85, 0.45], [0.85, 0.55], [0.85, 0.65], [0.85, 0.75]
        ])
        return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    PATTERNS = [
        'Random_inner', 'Grid5x5_cut', 'OffsetGrid', 'SparseHex', 'CornerExpansion'
    ]
    RESTART_EPOCHS = 500
    MAX_ITERS_PER_PATTERN = 2000
    step, step_min = 0.1, 1e-4

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern_name in PATTERNS:
        centers = generate_pattern(pattern_name)
        radii, cur_sum = solve_lp(centers)
        grad = compute_dual_gradient(centers, radii)

        iterations_since_best = 0
        for t in range(MAX_ITERS_PER_PATTERN):
            new_centers = propose_move(centers, grad, step=alpha(t))
            new_centers = clip_to_square(new_centers, 1e-12)

            new_radii, new_sum = solve_lp(new_centers)
            accept = (new_sum >= cur_sum) or \
                     (np.random.rand() < np.exp((new_sum - cur_sum) / temp(t)))

            if accept:
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                grad = compute_dual_gradient(centers, radii)

                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    iterations_since_best = 0
                else:
                    iterations_since_best += 1
            else:
                iterations_since_best += 1

            if iterations_since_best > RESTART_EPOCHS:
                centers = generate_pattern(np.random.choice(PATTERNS))
                radii, cur_sum = solve_lp(centers)
                grad = compute_dual_gradient(centers, radii)
                iterations_since_best = 0

        if best_sum >= 2.637:
            return best_centers, best_radii, best_sum

    return best_centers, best_radii, best_sum
