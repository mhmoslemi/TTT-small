import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = []  # List to store center coordinates
    radii = [0.1] * 25  # Radii of first 25 circles

    # Generate 5x5 grid of circles
    for i in range(5):
        for j in range(5):
            centers.append((0.1 + 0.2 * i, 0.1 + 0.2 * j))

    # Radii of the 26th circle
    radius_new = math.sqrt(0.08) - 0.10
    radii.append(radius_new)

    # Convert centers to numpy arrays
    centers = np.array(centers)

    # Sum of radii
    sum_radii = np.sum(radii)

    # Return the results
    return centers, np.array(radii), sum_radii
