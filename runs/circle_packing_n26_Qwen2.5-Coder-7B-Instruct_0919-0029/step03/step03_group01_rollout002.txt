import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def alpha(t):
    return 0.1 * (0.95 ** t)

def temp(t):
    return 0.5 * (0.995 ** t)

def compute_dual_gradient(centers, radii):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])
    for i in range(n):
        for j in range(i+1, n):
            A_ub[i*(n-1)//2 + j-1, i] = 1
            A_ub[i*(n-1)//2 + j-1, j] = 1
            A_ub[i*(n-1)//2 + j-1, :] *= (centers[i] - centers[j])

    result = linprog(np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    dual = result.dual
    grad = np.zeros((n, 2))
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist > 1e-12:
                grad[i, 0] -= dual[i*(n-1)//2 + j-1] * (centers[i, 0] - centers[j, 0]) / dist
                grad[i, 1] -= dual[i*(n-1)//2 + j-1] * (centers[i, 1] - centers[j, 1]) / dist
        dist = min(centers[i, 0], 1 - centers[i, 0])
        if dist > 1e-12:
            grad[i, 0] -= dual[i*(n-1)//2 + n*(n-1)//2 + 4*i] / dist
            grad[i, 0] -= dual[i*(n-1)//2 + n*(n-1)//2 + 4*i + 1] / dist
        dist = min(centers[i, 1], 1 - centers[i, 1])
        if dist > 1e-12:
            grad[i, 1] -= dual[i*(n-1)//2 + n*(n-1)//2 + 4*i + 2] / dist
            grad[i, 1] -= dual[i*(n-1)//2 + n*(n-1)//2 + 4*i + 3] / dist
    return grad

def propose_move(centers, grad, step):
    n = centers.shape[0]
    proposed_centers = centers + step * grad
    proposed_centers = np.clip(proposed_centers, 0, 1)
    return proposed_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step, step_min = 0.1, 1e-4
    decay_factor = 0.995
    NUM_ITERS_PER_PATTERN = 2000
    RESTART_EPOCHS = 500
    PATTERNS = [
        np.random.rand(n, 2),
        np.array([[0.1 + 0.2 * k, 0.1 + 0.2 * l] for k in range(5) for l in range(5)]),
        np.array([[0.1 + 0.2 * k + (k % 2) * 0.1, 0.1 + 0.2 * l] for k in range(5) for l in range(5)]),
        np.array([[0.15 * (2 * k + 1), 0.15 * (np.sqrt(3) * e + 1)] for k in range(20) for e in range(3)]),
        np.array([[0.5 * (4 * k + 1), 0.5 * (2 * l + 1)] for k in range(2) for l in range(2)])
    ]

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern in PATTERNS:
        centers = pattern
        radii, cur_sum = lp_optimum(centers)
        grad = compute_dual_gradient(centers, radii)

        iterations_since_best = 0
        for t in range(NUM_ITERS_PER_PATTERN):
            if radii is None or np.isnan(np.sum(radii)):
                raise ValueError("Invalid radius values")

            new_centers = propose_move(centers, grad, alpha(t))
            new_radii, new_sum = lp_optimum(new_centers)
            accept = (new_sum >= cur_sum) or \
                     (np.random.rand() < np.exp((new_sum - cur_sum) / temp(t)))

            if accept:
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                grad = compute_dual_gradient(centers, radii)

                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    iterations_since_best = 0
            else:
                iterations_since_best += 1

            if iterations_since_best > RESTART_EPOCHS:
                centers = np.random.rand(n, 2)
                radii, cur_sum = lp_optimum(centers)
                grad = compute_dual_gradient(centers, radii)
                iterations_since_best = 0

    return best_centers, best_radii, best_sum
