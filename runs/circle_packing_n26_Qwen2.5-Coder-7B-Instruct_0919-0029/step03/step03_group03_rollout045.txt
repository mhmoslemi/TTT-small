import numpy as np
from scipy.optimize import linprog

def build_hexagon_layout(n_rings=4, padding_factor=1.5):
    centers = []
    spacing = 1.0 / (2 * n_rings + 1)
    offset_x, offset_y = spacing * (0.5 + spacing / (2 * n_rings)), spacing + spacing / (2 * n_rings)
    side_midpoints = [(0.5, 0.05), (0.95, 0.5), (0.05, 0.5)]

    for i in range(n_rings + 1):
        for j in range(2 * i + 1):
            x = offset_x + j * spacing
            y = offset_y + i * spacing
            centers.append((x, y))
            
    for x, y in side_midpoints:
        centers.append((x, y + padding_factor * spacing))
    
    return np.array(centers)

def compute_boundaries_and_distances(centers):
    n = centers.shape[0]
    boundaries = np.maximum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                            np.maximum(centers[:, 1], 1 - centers[:, 1]))
    distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=2)
    np.fill_diagonal(distances, np.inf)
    return boundaries, distances

def solve_lp(boundaries, distances):
    n = boundaries.shape[0]
    c = np.ones(n)
    A = np.zeros((n * n, n))
    b = np.zeros(n * n)

    for i in range(n):
        A[i * n, i] = -1
        b[i * n] = -boundaries[i]

    for i in range(n):
        for j in range(i + 1, n):
            A[i * n + j, i] = 1
            A[i * n + j, j] = 1
            b[i * n + j] = distances[i, j]

    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return result.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = build_hexagon_layout()
    boundaries, distances = compute_boundaries_and_distances(centers)
    radii = solve_lp(boundaries, distances)
    sum_radii = np.sum(radii)
