import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def compute_dual_gradient(centers, radii):
    n = centers.shape[0]
    A = np.zeros((n*(n-1)//2 + 4*n, n))
    b = np.zeros(A.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = radii[i]
            A[idx, j] = radii[j]
            b[idx] = np.linalg.norm(centers[i] - centers[j]) - radii[i] - radii[j] + 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        A[idx, i] = 1
        b[idx] = x - r + 1e-12
        idx += 1
        A[idx, i] = -1
        b[idx] = 1 - x - r + 1e-12
        idx += 1
        A[idx, i] = 1
        b[idx] = y - r + 1e-12
        idx += 1
        A[idx, i] = -1
        b[idx] = 1 - y - r + 1e-12
        idx += 1

    result = linprog(A=A, b=b, method='highs')
    if result.success:
        dual = result.x
        gradient = np.zeros(n)
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                gradient[i] += dual[idx] / (1e-12 + np.linalg.norm(centers[i] - centers[j]))
                gradient[j] += dual[idx] / (1e-12 + np.linalg.norm(centers[i] - centers[j]))
                idx += 1
            gradient[i] -= 1 / (1e-12 + centers[i, 0])
            gradient[i] -= -1 / (1e-12 + 1 - centers[i, 0])
            gradient[i] -= 1 / (1e-12 + centers[i, 1])
            gradient[i] -= -1 / (1e-12 + 1 - centers[i, 1])
        return gradient
    else:
        return np.zeros(n)

def propose_move(centers, grad, step):
    n = centers.shape[0]
    direction = grad / np.linalg.norm(grad)
    move = step * direction
    new_centers = centers + move[:, np.newaxis]
    new_centers = np.clip(new_centers, 0, 1)
    return new_centers

def temp(t):
    T0 = 0.5
    eta = 0.995
    return T0 * eta**t

def alpha(t):
    alpha0 = 0.1
    rho = 0.95
    return alpha0 * rho**t

def clip_to_square(centers, eps):
    n = centers.shape[0]
    new_centers = centers.copy()
    new_centers[:, 0] = np.clip(centers[:, 0], eps, 1-eps)
    new_centers[:, 1] = np.clip(centers[:, 1], eps, 1-eps)
    return new_centers

def generate_pattern(pattern_name):
    if pattern_name == 'Random_inner':
        return np.random.rand(26, 2) * 0.9 + 0.05
    elif pattern_name == 'Grid5x5_cut':
        grid = np.array([0.1 + 0.2 * k for k in range(5)])
        centers = np.array(np.meshgrid(grid, grid)).T.reshape(-1, 2)
        centers[12] = np.random.rand(2) * 0.9 + 0.05
        return centers
    elif pattern_name == 'OffsetGrid':
        grid = np.array([0.1 + 0.2 * k for k in range(5)])
        centers = np.array(np.meshgrid(grid, grid)).T.reshape(-1, 2)
        centers[12] = np.random.rand(2) * 0.9 + 0.05
        centers[range(25, -1, -2)] += 0.1
        return centers
    elif pattern_name == 'SparseHex':
        hex_radius = 0.15
        hex_spacing = np.sqrt(3) * hex_radius
        rows = [start + i * hex_spacing for start in [0.1, 0.15, 0.2, 0.5, 0.55, 0.8, 0.85] for i in range(4)]
        cols = [0.1 + j * hex_radius for j in range(int(1.0 / hex_radius))]
        centers = np.array([[col + hex_radius * j for col in cols] for i in range(len(rows)) for j in [0, 1]])
        centers = np.array([[r + 2 * hex_radius for r in rows]])
        centers = centers.flatten().reshape(-1, 2)
        centers = centers[centers[:, 0] >= 0.1]
        centers = centers[centers[:, 0] <= 0.85]
        centers = centers[centers[:, 1] >= 0.1]
        centers = centers[centers[:, 1] <= 0.95]
        return centers[:26]
    elif pattern_name == 'CornerExpansion':
        central_rect = np.random.rand(18, 2) * 0.8 + 0.1
        side_points = np.array([[0.15, side_mid], [side_mid, 0.15], [0.85, side_mid], [side_mid, 0.85]])
        side_mid = 0.55
        return np.vstack((central_rect, side_points)).clip(0, 1)
    else:
        return np.random.rand(26, 2)

def run_packing():
    n = 26
    NUM_PATTERNS = 5
    MAX_ITERS_PER_PATTERN = 2000
    RESTART_EPOCHS = 500
    NUM_STARS = 3
    decay_factor = 0.995
    alpha_min = 1e-4
    eps = 1e-12

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(NUM_STARS):
        best_sum = -np.inf
        for pattern_name in ['Random_inner', 'Grid5x5_cut', 'OffsetGrid', 'SparseHex', 'CornerExpansion']:
            centers = generate_pattern(pattern_name)
            radii, cur_sum = lp_optimum(centers)
            grad = compute_dual_gradient(centers, radii)

            iterations_since_best = 0
            for t in range(MAX_ITERS_PER_PATTERN):
                new_centers = propose_move(centers, grad, step=alpha(t))
                new_centers = clip_to_square(new_centers, eps)
                new_radii, new_sum = lp_optimum(new_centers)
                accept = (new_sum >= cur_sum) or (np.random.rand() < np.exp((new_sum-cur_sum)/temp(t)))
                if accept:
                    centers, radii, cur_sum = new_centers, new_radii, new_sum
                    grad = compute_dual_gradient(centers, radii)
                    if new_sum > best_sum:
                        best_sum = new_sum
                        best_centers = centers.copy()
                        best_radii = radii.copy()
                        iterations_since_best = 0
                else:
                    iterations_since_best += 1
                if iterations_since_best > RESTART_EPOCHS:
                    centers = generate_pattern(pattern_name)
                    radii, cur_sum = lp_optimum(centers)
                    grad = compute_dual_gradient(centers, radii)
                    iterations_since_best = 0
                step = max(alpha_min, alpha(t) * decay_factor)
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        final_radii, final_sum = lp_optimum(best_centers)
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = best_centers.copy()
            best_radii = final_radii.copy()

    return best_centers, best_radii, best_sum
