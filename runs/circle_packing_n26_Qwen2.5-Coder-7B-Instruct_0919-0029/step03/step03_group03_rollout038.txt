        d = cdist(centers[:n], centers[:n])
        A = np.zeros((2 * n, n))
        A[:n, :n] = -np.eye(n)
        A[:n, n:] = np.eye(n)

        b = np.zeros(2 * n)
        b[:n] = B
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, B))
        if not res.success:
            return -np.inf
        return res.fun
    
    # 4. Stochastic exploration - Simulated Annealing
    def sa(centers, initial_temp=0.1, cooling_rate=0.9999):

        best_centers = current_centers.copy()

                new_centers[i] = [np.clip(x + dx, 0, 1), np.clip(y + dy, 0, 1)]
                new_phi = lp_solver(new_centers)
                if new_phi > current_phi or np.random.rand() < np.exp((new_phi - current_phi) / temperature):
                    current_centers = new_centers
                    current_phi = new_phi
                    if current_phi > best_phi:
                        best_centers = current_centers.copy()
                        best_phi = current_phi
            temperature *= cooling_rate
        return best_centers, best_phi
    
    best_centers, best_phi = sa(centers)

        return best_centers
    
    best_centers = gradient_REFINE(best_centers, best_phi, np.zeros(n))
    final_phi = lp_solver(best_centers)
    
    return best_centers, np.ones(n), final_phi
