import numpy as np
from scipy.optimize import minimize

def feasi_dist(z):
    i, j = np.indices((26, 26))
    i = i[i < j].flatten()
    j = j[i < j].flatten()
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
    fx = r[i] + r[j]
    return (fx - d).clip(0)**2

def feasi_boundary(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    fx = r - x
    fy = r - (1 - x)
    fz = r - y
    fw = r - (1 - y)
    return np.maximum(fx, fy, fz, fw).clip(0)**2

def feasi_nonneg(z):
    r = z[52:]
    return (-r).clip(0)**2

def objective(z):
    x = z[:26]
    y = z[26:52]
    return -(x**2 + y**2).sum()

def penalty(z, mu):
    return mu * (feasi_dist(z).sum() + feasi_boundary(z).sum() + 
                feasi_nonneg(z).sum())

def full_objective(z, mu):
    return objective(z) + penalty(z, mu)

def gradient(z, mu):
    n = 26
    r = z[52:]
    x = z[:26]
    y = z[26:52]
    dx = np.zeros(78)
    dy = np.zeros(78)
    dr = np.zeros(78)
    for i in range(n):
        dx[i] = -2 * (x[i]**2 + y[i]**2)
        dy[i] = -2 * (x[i]**2 + y[i]**2)
        dr[i] = -2 * (x[i]**2 + y[i]**2)
    dx[26:52] -= np.repeat(2 * (x**2 + y**2), n)
    dy[26:52] -= np.repeat(2 * (x**2 + y**2), n)
    dr[26:52] -= np.repeat(2 * (x**2 + y**2), n)
    for i in range(n):
        dx[i] += np.sum(2 * (x[i] - x)**2 / np.sqrt((x[i] - x)**2 + (y[i] - y)**2))
        dy[i] += np.sum(2 * (y[i] - y)**2 / np.sqrt((x[i] - x)**2 + (y[i] - y)**2))
        dr[i] += np.sum(2 * (2 * x[i] - x - x) / np.sqrt((x[i] - x)**2 + (y[i] - y)**2))
        dr[i] += np.sum(2 * (2 * y[i] - y - y) / np.sqrt((x[i] - x)**2 + (y[i] - y)**2))
    dx[26:52] -= np.repeat(np.sum(2 * (x**2 + y**2) * 
                                    (np.where(r[j] > x[j], (r[j] - x[j]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2), 0)), 
                     axis=1), n)
    dy[26:52] -= np.repeat(np.sum(2 * (x**2 + y**2) * 
                                    (np.where(r[j] > y[j], (r[j] - y[j]) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2), 0)), 
                     axis=1), n)
    dr[26:52] -= np.repeat(np.sum(2 * (x**2 + y**2) * 
                                    (np.where(r[j] > r[j], (2 * x[j] - x - x) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2), 0)), 
                     axis=1), n)
    dr[26:52] -= np.repeat(np.sum(2 * (x**2 + y**2) * 
                                    (np.where(r[j] > r[j], (2 * y[j] - y - y) / np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2), 0)), 
                     axis=1), n)
    dr[26:52] -= np.sum(2 * (r - (x - np.where(r < x, -np.inf, 0)))**2, axis=1)
    dr[26:52] -= np.sum(2 * (r - (1 - x - np.where(r < (1 - x), -np.inf, 0)))**2, axis=1)
    dr[26:52] -= np.sum(2 * (r - (y - np.where(r < y, -np.inf, 0)))**2, axis=1)
    dr[26:52] -= np.sum(2 * (r - (1 - y - np.where(r < (1 - y), -np.inf, 0)))**2, axis=1)
    return -dx, -dy, -dr

init_r = 0.05 * np.random.rand(26)
init_pos = 0.5 * np.random.rand(26, 2)
inits = np.vstack((init_pos.flatten(), init_r)).flatten()
bounds = [(0, 1)] * (26 * 2) + [(0, None)] * 26

mu = 10
nu = 1e-6
tol = 1e-6
N_RUNS = 25
MAX_ITER = 200
MAX_PEN = 10

best_sum = -np.inf
best_centers, best_radii = None, None

for N in range(N_RUNS):
    if N == 0:
        z = inits
    else:
        steps = 2 * np.random.rand(26, 2) - 1
        radii_steps = np.random.rand(26) * 0.01
        new_centers = z[:52].reshape(26, 2) + steps
        new_radii = z[52:] + radii_steps
        new_radii = np.clip(new_radii, 0, 1)
        z = np.hstack((new_centers.flatten(), new_radii))
    
    constraints = ({'type': 'ineq', 'fun': lambda z: objective(z) + penalty(z, mu)})

    result = minimize(objective, z, method='L-BFGS-B', bounds=bounds, 
                         jac=gradient, args=(mu,), constraints=constraints, 
                         options={'ftol': tol, 'maxiter': MAX_ITER})
    
    while result.fun > -tol:
        mu *= 10
        constraints = ({'type': 'ineq', 'fun': lambda z: objective(z) + penalty(z, mu)})
        result = minimize(objective, result.x, method='L-BFGS-B', bounds=bounds, 
                             jac=gradient, args=(mu,), constraints=constraints, 
                             options={'ftol': tol, 'maxiter': MAX_ITER})
    
    if result.fun < best_sum:
        best_sum = result.fun
        best_centers = result.x[:26].reshape(26, 2)
        best_radii = result.x[26:]

# Ensuring all constraints are satisfied
for _ in range(10):
    new_radii = 0.05 * np.random.rand(26)
    new_centers = 0.5 * np.random.rand(26, 2)
    new_centers = np.clip(new_centers, 0, 1)
    new_centers[best_centers[:, 0] > 0.4] += new_radii[best_centers[:, 0] > 0.4] * 2 * (np.random.rand(26) - 0.5)
    new_centers[best_centers[:, 1] > 0.4] += new_radii[best_centers[:, 1] > 0.4] * 2 * (np.random.rand(26) - 0.5)
    new_centers[:, 1][best_centers[:, 1] < 0.6] -= new_radii[best_centers[:, 1] < 0.6] * 2 * (np.random.rand(26) - 0.5)
    new_centers[:, 0][best_centers[:, 0] < 0.6] -= new_radii[best_centers[:, 0] < 0.6] * 2 * (np.random.rand(26) - 0.5)
    new_centers = np.clip(new_centers, 0, 1)
    constraints = ({'type': 'ineq', 'fun': lambda z: objective(z) + penalty(z, mu)})
    result = minimize(objective, np.hstack((new_centers.flatten(), new_radii)), method='L-BFGS-B', bounds=bounds, 
                         jac=gradient, args=(mu,), constraints=constraints, 
                         options={'ftol': tol, 'maxiter': MAX_ITER})
    if result.fun < best_sum:
        best_sum = result.fun
        best_centers = result.x[:26].reshape(26, 2)
        best_radii = result.x[26:]

print((best_centers, best_radii, best_sum))
