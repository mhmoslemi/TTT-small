import numpy as np

def lp_optimum(centers):
    n = centers.shape[0]
    
    # Constraints: sum of radii
    c = -np.ones(n)
    
    # Bounds for radii
    bounds = [(0, None)] * n
    
    # Upper constraints for sum of constraints
    A_ub = np.zeros((n, n))
    for i in range(n):
        A_ub[i, i] = 1
    
    b_ub = np.ones(n)
    
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            A = np.zeros((3, n))
            A[0, i] = 1
            A[0, j] = -1
            A[1, i] = -2 * (centers[i, 0] - centers[j, 0])
            A[1, j] = 2 * (centers[i, 0] - centers[j, 0])
            A[2, i] = -2 * (centers[i, 1] - centers[j, 1])
            A[2, j] = 2 * (centers[i, 1] - centers[j, 1])
           A_ub = np.vstack((A_ub, A))
            b_ub = np.append(b_ub, np.linalg.norm(centers[i] - centers[j]) - 1e-12)
    
    # Boundary constraints
    for i in range(n):
        A = np.zeros((3, n))
        A[0, i] = 1
        A[1, i] = -2 * centers[i, 0]
        A[2, i] = -2 * centers[i, 1]
        A_ub = np.vstack((A_ub, A))
        b_ub = np.append(b_ub, 1 - 1e-12)
        A = np.zeros((3, n))
        A[0, i] = -1
        A[1, i] = 2 * (1 - centers[i, 0])
        A[2, i] = -2 * centers[i, 1]
        A_ub = np.vstack((A_ub, A))
        b_ub = np.append(b_ub, centers[i, 0] - 1e-12)
        A = np.zeros((3, n))
        A[0, i] = -1
        A[1, i] = -2 * centers[i, 0]
        A[2, i] = 2 * (1 - centers[i, 1])
        A_ub = np.vstack((A_ub, A))
        b_ub = np.append(b_ub, 1 - centers[i, 1] - 1e-12)
        A = np.zeros((3, n))
        A[0, i] = 1
        A[1, i] = 2 * centers[i, 0]
        A[2, i] = 2 * (1 - centers[i, 1])
        A_ub = np.vstack((A_ub, A))
        b_ub = np.append(b_ub, centers[i, 0] + centers[i, 1] - 1 + 1e-12)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0
