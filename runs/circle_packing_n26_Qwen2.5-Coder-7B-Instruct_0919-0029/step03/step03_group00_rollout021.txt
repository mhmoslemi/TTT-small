import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, differential_evolution

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = d - 1e-9
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-9
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-9
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def initial_centers():
    # Deterministic hex lattice of 30 points
    r = d = np.sqrt(3) / 3
    xs, ys = np.mgrid[0:1:d, 0:1:d]
    xs = xs.flatten()[:-1]
    ys = ys.flatten()[:-1]
    # Scale to fit the unit square
    x_height = 2 * r / np.sqrt(3)
    y_height = 2 * 0.5
    scale_x = (1 - 2 * r) / x_height
    scale_y = (1 - 2 * r) / y_height
    xs *= scale_x
    ys *= scale_y
    # Corner anchors
    corner_r = 0.05
    corners = np.array([[corner_r, corner_r],
                       [corner_r, 1 - corner_r],
                       [1 - corner_r, corner_r],
                       [1 - corner_r, 1 - corner_r]])
    # Combine and add noise
    centers = np.vstack((xs + ys[:, None] * d / 2, ys)).T
    centers += np.random.randn(len(centers), 2) * 0.01
    centers[:34, :2] = corners
    centers[:, 0] = np.clip(centers[:, 0], 0, 1)
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)
    return centers

def negate_sum_of_radii(x):
    centers = x.reshape((26, 2))
    radii, sum_r = lp_optimum(centers)
    return -sum_r

def run_packing():
    centers = initial_centers()
    result = differential_evolution(negate_sum_of_radii, bounds=[(0,1)]*52, 
                                     maxiter=200, popsize=15, strategy='best1bin',
                                     callback=lambda pop: (pop.last_population, pop.last_fitness))
    best_centers = result.x.reshape((26,2))
    _, best_sum = lp_optimum(best_centers)
    # Local SA refinement
    for _ in range(3):
        best_centers, _, best_sum = local_sa(best_centers, best_sum)
    return best_centers, lp_optimum(best_centers)[0], best_sum

def local_sa(centers, cur_sum):
    step = 0.05
    T = 0.05
    for _ in range(5000):
        idx = np.random.randint(26)
        move = np.random.randn(2) * step
        new_centers = centers.copy()
        new_centers[idx] += move
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radii, new_sum = lp_optimum(new_centers)
        if new_sum > cur_sum or np.random.random() < np.exp((new_sum - cur_sum) / T):
            centers, cur_sum = new_centers, new_sum
        step *= 0.995
        T *= 0.995
    return centers, cur_sum
