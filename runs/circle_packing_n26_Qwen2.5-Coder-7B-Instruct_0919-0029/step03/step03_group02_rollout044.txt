import numpy as np
from scipy.optimize import minimize, Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def ineq_constraints(z):
    centers = z[:78].reshape(26, 3)
    radii = centers[:, 2]
    boundaries = radii - centers[:, :2]
    distances = np.sqrt((centers[np.newaxis, :, 0] - centers[:, 0]) ** 2 +
                        (centers[np.newaxis, :, 1] - centers[:, 1]) ** 2)
    return np.hstack((radii, boundaries, distances - radii[:, None] - radii[None, :]))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((26, 3))
    centers[1:25, 0] = 0.10 + 0.20 * np.arange(24).reshape(24, 2)[:, 0][:, np.newaxis] + 0.20 * np.arange(25) // 3
    centers[1:25, 1] = 0.10 + 0.20 * np.arange(24).reshape(24, 2)[:, 1][:, np.newaxis] + 0.20 * np.arange(25) % 3
    centers[26, 0:2] = 0.50
    centers[26, 2] = np.sqrt(0.08) - 0.10
    radii = np.full(26, 0.10)
    
    x0 = np.concatenate((centers[:, 0], centers[:, 1], centers[:, 2]))
    
    bounds = Bounds([0, ] * (3 * n), [1, ] * (3 * n))
    
    result = minimize(lambda z: -np.sum(z[52:], initial_guess=x0), method='trust-constr', jac=None, hess=None,
                      bounds=bounds, constraints={'type': 'ineq', 'fun': ineq_constraints}, options={'max_iter': 2000, 'gtol': 1e-8, 'xtol': 1e-8, 'solver': 'slsqp'})
    
    if result.success:
        centers = result.x[:78].reshape(26, 3)
        radii = centers[:, 2]
        total = np.sum(radii)
        return centers[:, 0:2], radii, total
    else:
        raise Exception("OCP optimization failed")
