import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # 5x5 grid of equal circles
    for i in range(5):
        for j in range(5):
            centers[i*5 + j] = np.array([0.1 + 0.2 * i, 0.1 + 0.2 * j])

    # Center circle
    center_radius = np.sqrt(0.08) - 0.1
    centers[25] = np.array([0.5, 0.5])
    radii[25] = center_radius

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
