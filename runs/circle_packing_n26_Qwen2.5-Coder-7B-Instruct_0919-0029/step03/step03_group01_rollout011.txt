import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def propose_move(centers, step):
    idx = np.random.randint(26)
    dir = np.random.randn(2)
    dir /= np.linalg.norm(dir)
    delta = dir * step
    return idx, delta

def clip(x, lo, hi):
    return np.clip(x, lo, hi)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    EPS = 1e-12
    step_start = 0.05
    step_decay = 0.995
    T0 = 1e-3 * 2.7  # Roughly a thousandth of best possible sum
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500

    best_centers, best_radii, best_sum = None, None, -np.inf

    patterns = [
        np.random.rand(n, 2) * 0.9 + 0.05,
        np.random.rand(n-1, 2) * 0.2 + np.array([[0.1, 0.1]] + [[0.1, 0.3], [0.3, 0.1], [0.3, 0.3], [0.5, 0.1], [0.5, 0.3], [0.7, 0.1], [0.7, 0.3], [0.9, 0.1], [0.9, 0.3]])
    ]

    for p in patterns:
        centers = p
        radii, cur_sum = lp_optimum(centers)

        for it in range(ITER_PER_PATTERN):
            idx, delta = propose_move(centers, step_start)
            new_c = centers[idx] + delta
            new_c = clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            Δsum = new_sum - cur_sum
            if Δsum >= 0 or (np.random.rand() < np.exp(Δsum / T0)):
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                if cur_sum > best_sum:
                    best_centers, best_radii, best_sum = centers.copy(), radii.copy(), cur_sum

            T0 *= T_decay
            step_start *= step_decay

    final_radii, final_sum = lp_optimum(best_centers)
    return best_centers, final_radii, final_sum
