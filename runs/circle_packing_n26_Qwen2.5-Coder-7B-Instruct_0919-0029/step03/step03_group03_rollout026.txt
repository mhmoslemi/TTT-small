import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    B = np.minimum(centers[:, 0], 1 - centers[:, 0])
    B = np.minimum(B, centers[:, 1])
    B = np.minimum(B, 1 - centers[:, 1])
    dist_mat = distance_matrix(centers, centers)
    
    # Linear programming formulation
    c = -np ones(n)  # Minimize -sum(radii) is equivalent to maximize sum(radii)
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    A_le = np.zeros((n*(n-1)//2, n))
    b_le = np.zeros(n*(n-1)//2)
    
    for i in range(n):
        A_le[i*(n-1)//2 + i, i] = 1
        b_le[i*(n-1)//2 + i] = B[i]
        for j in range(i+1, n):
            A_le[i*(n-1)//2 + j - i - 1, i] = 1
            A_le[i*(n-1)//2 + j - i - 1, j] = 1
            b_le[i*(n-1)//2 + j - i - 1] = dist_mat[i, j]
    
    res = linprog(c, A_ub=A_le, b_ub=b_le, A_eq=A_eq, b_eq=b_eq, bounds=(0, B), method='highs')
    if not res.success:
        return -np.inf, np.zeros(n)
    return res.fun, res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(123)  # For reproducibility
    hex_radius = 0.099
    centers = np.zeros((26, 2))
    row_coords = np.array([0, 1, 2, 3, 4, 5])
    col_coords = np.empty(26)
    col_coords[:5] = np.arange(5)
    col_coords[5:9] = np.arange(4)
    col_coords[9:14] = np.arange(4)
    col_coords[14:17] = np.arange(3)
    col_coords[17:20] = np.arange(3)
    col_coords[20:22] = np.arange(2)
    col_coords[22:24] = np.arange(2)
    col_coords[24:26] = np.arange(1)
    
    centers[:23, 0] = hex_radius + (col_coords + (row_coords[:, None] % 2)) * 2 * hex_radius
    centers[:23, 1] = hex_radius + row_coords * np.sqrt(3) * hex_radius
    
    side_midpoints = np.array([
        [0.5, 0.05],
        [0.95, 0.5],
        [0.05, 0.5]
    ])
    centers[23:26] = side_midpoints
    
    SA_history = []
    SA_current_state = centers.flatten()
    SA_best_state = SA_current_state
    SA_temperature = 0.1
    SA_cooldown = 0.9999
    SA_max_iterations = 20000
    
    while SA_temperature > 1e-6:
        for _ in range(int(SA_max_iterations)):
            i = np.random.randint(26)
            dx, dy = np.random.normal(0, SA_temperature / 2, 2)
            dx = np.clip(dx, -1, 1)
            dy = np.clip(dy, -1, 1)
            new_centers = SA_current_state.reshape(26, 2).copy()
            new_centers[i, 0] += dx
            new_centers[i, 1] += dy
            new_centers = np.clip(new_centers, 0, 1)
            new_centers = new_centers.flatten()
            
            SA_quick_check = np.min(distance_matrix(new_centers.reshape(26, 2), new_centers.reshape(26, 2)) - np.ones((26, 26)) * 1e-4)
            if SA_quick_check > 0:
                SA_new_state = new_centers
                SA_new_energy, _ = solve_radii(new_centers.reshape(26, 2))
                SA_current_energy, _ = solve_radii(SA_current_state.reshape(26, 2))
                SA_acceptance_probability = min(1, np.exp((SA_new_energy - SA_current_energy) / SA_temperature))
                if np.random.rand() < SA_acceptance_probability:
                    SA_current_state = SA_new_state
                    if SA_new_energy > SA_best_energy:
                        SA_best_state = SA_new_state
                    SA_history.append((SA_current_state, SA_current_energy))
        
        SA_temperature *= SA_cooldown
    
    while any(SA_history[-1][1] > SA_history[-2][1]) and SA_history[-1][1] > 2.63:
        SA_max_iterations *= 2
        SA_temperature *= 0.9999
        for _ in range(int(SA_max_iterations)):
            i = np.random.randint(26)
            dx, dy = np.random.normal(0, SA_temperature / 2, 2)
            dx = np.clip(dx, -1, 1)
            dy = np.clip(dy, -1, 1)
            new_centers = SA_current_state.reshape(26, 2).copy()
            new_centers[i, 0] += dx
            new_centers[i, 1] += dy
            new_centers = np.clip(new_centers, 0, 1)
            new_centers = new_centers.flatten()
            
            SA_quick_check = np.min(distance_matrix(new_centers.reshape(26, 2), new_centers.reshape(26, 2)) - np.ones((26, 26)) * 1e-4)
            if SA_quick_check > 0:
                SA_new_state = new_centers
                SA_new_energy, _ = solve_radii(new_centers.reshape(26, 2))
                SA_current_energy, _ = solve_radii(SA_current_state.reshape(26, 2))
                SA_acceptance_probability = min(1, np.exp((SA_new_energy - SA_current_energy) / SA_temperature))
                if np.random.rand() < SA_acceptance_probability:
                    SA_current_state = SA_new_state
                    if SA_new_energy > SA_best_energy:
                        SA_best_state = SA_new_state
                    SA_history.append((SA_current_state, SA_current_energy))
    
    SA_best_state = SA_best_state.reshape(26, 2).flatten()
    SA_best_energy, SA_best_radii = solve_radii(SA_best_state.reshape(26, 2))
    return SA_best_state.reshape(26, 2), SA_best_radii, SA_best_energy
