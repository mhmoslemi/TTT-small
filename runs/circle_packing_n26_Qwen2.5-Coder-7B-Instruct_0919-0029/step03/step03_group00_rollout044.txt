import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def build_contact_graph(centers, radii):
    n = centers.shape[0]
    contact_graph = np.zeros((n, n), dtype=bool)
    for i in range(n):
        for j in range(i+1, n):
            if radii[i] + radii[j] >= np.linalg.norm(centers[i] - centers[j]) - 1e-12:
                contact_graph[i, j] = contact_graph[j, i] = True
    return contact_graph

def update_contact_graph(contact_graph, active_edges, i, j):
    contact_graph[i, j] = contact_graph[j, i] = False
    new_edges = []
    for k in range(contact_graph.shape[0]):
        if contact_graph[i, k] or contact_graph[j, k]:
            new_edges.append((i, k))
            new_edges.append((j, k))
    contact_graph[active_edges] = False
    contact_graph[new_edges] = True
    return contact_graph

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        cur_radii, cur_sum = lp_optimum(cur_centers)

            idx = np.random.randint(n)

            new_c = cur_centers.copy()
            new_c[idx] += delta

            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum:
                cur_centers = new_c
                cur_radii, cur_sum = new_radii, new_sum

        if cur_sum > best_sum:
            best_sum = cur_sum

    active_edges = build_contact_graph(best_centers, lp_optimum(best_centers)[0])
    for _ in range(NUM_STARS):
        centers = np.random.rand(n, 2)
