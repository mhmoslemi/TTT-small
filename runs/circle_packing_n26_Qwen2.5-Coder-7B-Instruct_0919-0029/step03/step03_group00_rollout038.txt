import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        A_eq[i, i] = 1
        b_eq[i] = centers[i, 0] if i < 2 else centers[i-1, 0] + centers[i-2, 0] + centers[i, 0]
    
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = 2 * np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_trials = 10
    max_iter = 4000
    step_initial = 0.1
    step_min = 1e-4
    decay_step = 0.995
    T_initial = 0.1
    T_min = 1e-4
    T_decay = 0.995

    best_sum = -np.inf
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    for _ in range(max_trials):
        np.random.seed(_)
        centers = np.random.rand(n, 2)
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        
        step = step_initial
        T = T_initial
        radii_max = cur_radii.copy()

        for _iter in range(max_iter):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            
            if np.linalg.norm(new_c[idx] - cur_centers[idx]) > 0.1:
                continue
            
            new_radii, new_sum = lp_optimum(new_c)

            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / T):
                cur_centers = new_c
                cur_radii = new_radii
                cur_sum = new_sum
                
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_c
                    best_radii = new_radii
            
            step = max(step_min, step * decay_step)
            T = max(T_min, T * T_decay)

        for i in range(n):
            if cur_radii[i] < radii_max[i]:
                cur_radii[i] = radii_max[i]

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers
            best_radii = cur_radii

    return best_centers, best_radii, best_sum
