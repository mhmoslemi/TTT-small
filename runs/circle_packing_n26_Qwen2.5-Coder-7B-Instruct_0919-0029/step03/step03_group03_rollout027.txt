import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    n, _ = centers.shape
    r_bounds = centers[:, 0] - 0.5, 1 - centers[:, 1] - 0.5
    a_eq = np.repeat([[1, 0, 0], [0, 1, 0]], n, axis=0)
    b_eq = np.zeros(2 * n)

    A_ineq = np.array([[0, 0, 1]] * n + [[-1, 0, 0]] * n + [[0, -1, 0]] * n + [[0, 0, -1]] * n)
    b_ineq = np.repeat(r_bounds, 4)

    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            A_ineq = np.vstack((A_ineq, np.hstack([diff, [diff @ diff, -1]])))
            b_ineq = np.append(b_ineq, 0)

    res = linprog(-np.ones(3 * n), A_ub=A_ineq, b_ub=b_ineq, A_eq=a_eq, b_eq=b_eq, bounds=r_bounds)
    return res.x.reshape((-1, 3))[:, 2], -res.fun

def repulsive_algorithm(n, max_iter, d_min):
    centers = np.random.rand(n, 2)
    for iteration in range(max_iter):
        dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
        overlaps = dists < d_min
        np.fill_diagonal(overlaps, False)
        offsets = np.random.randn(n, 2)
        offsets /= np.linalg.norm(offsets, axis=1, keepdims=True)
        offsets *= (d_min - dists) / 2
        move = np.sum(offsets, axis=0)
        centers -= move
        centers[np.any(centers < 0, axis=1)] *= -1 + 2 * (centers < 0)
        centers[np.any(centers > 1, axis=1)] *= 2 * (centers > 1) - 1
    return centers
