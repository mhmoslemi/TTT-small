import numpy as np
from scipy.optimize import minimize, trust_regions

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(z):
    n = isinstance(z, np.ndarray) and z.shape[0] or 78
    return -z[n//2:].sum()

def ineq_constraints(z):
    cx, cy, cr = z[:n//2].reshape(-1,2).T, z[n//2:n].reshape(-1), np.sqrt(np.sum((cx[:, np.newaxis, :] - cx[np.newaxis, :, :]) ** 2, axis=2))
    return np.concatenate([-cr, cr - cx.T, cr - (1 - cx.T), cr - cy.T, cr - (1 - cy.T), cr - cr.T + 1e-12])

def run_packing():
    n = 26
    cx0 = np.array([[0.1 * (2 * i + 1) for i in range(5)] + [0.1 * (2 * (i%5) + 1) for i in range(10)] + [0.1 * (2 * (i%5) + 1) for i in range(5)] + [0.5]])
    cy0 = np.array([[0.1 * (2 * i + 1) for i in range(5)] + [0.1 * (2 * (i%5) + 1) + 0.5 for i in range(10)] + [0.1 * (2 * (i%5) + 1) + 1 for i in range(5)] + [0.5]]
    cr0 = np.full(n, 0.1)
    
    initial_guess = np.concatenate([cx0.ravel(), cy0.ravel(), cr0.ravel()])
    
    bounds = [(-1e-8, None) for _ in range(n)] + [(0, None) for _ in range(n)]
    
    result = minimize(objective, initial_guess, method='trust-constr', bounds=bounds, constraints={'type': 'ineq', 'fun': ineq_constraints}, options={'max_iter': 2000, 'gtol': 1e-8, 'xtol': 1e-8})
    
    if result.success:
        centers = result.x[:n].reshape(n, 2)
        radii = result.x[n:]
    else:
        raise Exception("Optimizer failed to converge")
    
    sum_radii = np.sum(radii)
    
    if not validate_packing(centers, radii):
        raise Exception("Invalid packing: failed validation")
    
    return centers, radii, sum_radii

results = run_packing()
print(results)

This code defines the `run_packing` function which, using a combination of a warm-start initial configuration and a nonlinear constrained optimization process, attempts to find a packing of 26 circles within a unit square that maximizes the sum of their radii. The `objective` function and `ineq_constraints` functions are used to define the optimization problem, and `minimize` from `scipy.optimize` is employed with the `trust-constr` method to solve it.
