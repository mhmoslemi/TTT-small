import numpy as np
from scipy.optimize import linprog
import random

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def active_pairs(centers, radii):
    n = centers.shape[0]
    active = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                active.append((i, j))
    return active

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.2
    step_min = 1e-4
    decay = 0.995
    NUM_ITERS = 4000
    NUM_STARS = 15

    best_sum = -np.inf
    best_centers, best_radii = None, None

    def initialize_hexagon_with_corners():
        # Hexagonal seed
        hex_dist = 0.18
        num_cols = 5
        num_rows = 6
        hexagon = []
        for r in range(num_rows):
            for c in range(num_cols):
                if c % 2 == 0:
                    x = c * hex_dist + hex_dist / 2
                    y = r * hex_dist * (3**0.5) / 2
                else:
                    x = c * hex_dist
                    y = r * hex_dist * (3**0.5) / 2
                hexagon.append((x, y))
        # Scale to fit within unit square
        x_min, x_max = min(p[0] for p in hexagon), max(p[0] for p in hexagon)
        y_min, y_max = min(p[1] for p in hexagon), max(p[1] for p in hexagon)
        scale = min((1 - x_min) / (x_max - x_min), (1 - y_min) / (y_max - y_min))
        hexagon = [(x * scale, y * scale) for x, y in hexagon]
        # Add corner points
        corners = [(0.03, 0.03), (0.03, 1-0.03), (1-0.03, 0.03), (1-0.03, 1-0.03)]
        centers = hexagon[:n] + corners[:n-len(hexagon)]
        return np.array(centers)

    for _ in range(NUM_STARS):
        centers = initialize_hexagon_with_corners()
        cur_centers = centers.copy()
        cur_radii, cur_sum = lp_optimum(cur_centers)
        step = 0.1
        for iter in range(NUM_ITERS):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            delta = dir * step
            new_c = cur_centers.copy()
            new_c[idx] += delta
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_radii, new_sum = lp_optimum(new_c)
            if new_sum > cur_sum:
                cur_centers = new_c
                cur_radii, cur_sum = new_radii, new_sum

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers

    best_radii, _ = lp_optimum(best_centers)
    return best_centers, best_radii, best_sum
