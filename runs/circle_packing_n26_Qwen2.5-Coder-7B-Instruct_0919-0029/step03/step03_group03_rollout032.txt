import numpy as np
from scipy.optimize import linprog

def depth(x, y): return min(x, 1-x, y, 1-y)

def new_center(x, y, dx, dy):
    x2 = max(0, min(1, x + dx))
    y2 = max(0, min(1, y + dy))
    return [x2, y2]

def eval_state(centers):
    n = centers.shape[0]
    dists = np.array([[np.linalg.norm(centers[i] - centers[j]) for j in range(i + 1, n)] for i in range(n)])
    radii = np.array([min(depth(centers[i, 0], centers[i, 1]), np.min(dists[i, :])) for i in range(n)])
    return np.sum(radii), radii

def random_move(centers, sigma=0.01):
    i = np.random.randint(len(centers))
    x, y = centers[i]
    new_x, new_y = new_center(x, y, np.random.normal(0, sigma), np.random.normal(0, sigma))
    new_centers = centers.copy()
    new_centers[i] = [new_x, new_y]
    return new_centers

def accept_move(centers, new_centers, old_obj, new_obj):
    if new_obj > old_obj:
        return True, new_centers
    else:
        accept = np.random.rand() < np.exp((new_obj - old_obj) / T)
        return accept, new_centers if accept else centers

def run_packing():
    s = (np.sqrt(3) - 1) / 2
    centers = np.array([[x + s * (y % 2 == 1), 0.1 + 0.66 * y * s] for y in range(5) for x in range(5 + (y % 2))])
    centers = centers[:26]
    centers = np.vstack((centers, np.array([[0.5, 0.05], [0.95, 0.5], [0.05, 0.5]])))
    
    old_obj, old_radii = eval_state(centers)
    best_obj, best_centers, best_radii = old_obj, centers, old_radii
    
    T = 0.1
    alpha = 0.9999
    max_iter = 20000
    
    for _ in range(max_iter):
        new_centers = random_move(centers)
        new_obj, radii = eval_state(new_centers)
        acc, centers = accept_move(centers, new_centers, old_obj, new_obj)
        if acc:
            old_obj = new_obj
            old_radii = radii
            
            if old_obj > best_obj:
                best_obj = old_obj
                best_centers = centers
                best_radii = old_radii
    
    for i in range(50):
        dz = 1e-4
        x = np.array([new_center(best_centers[i, 0], best_centers[i, 1], dz, 0)[0], new_center(best_centers[i, 0], best_centers[i, 1], -dz, 0)[0]])
        y = np.array([new_center(best_centers[i, 0], best_centers[i, 1], 0, dz)[1], new_center(best_centers[i, 0], best_centers[i, 1], 0, -dz)[1]])
        if np.any(x <= 0) or np.any(x >= 1) or np.any(y <= 0) or np.any(y >= 1):
            continue
        
        eval_points = np.stack((x, y)).T
        cur_obj, _ = eval_state(best_centers)
        eval_obj = np.array([eval_state(np.vstack((best_centers[:i], point, best_centers[i+1:]))) for point in eval_points]).flatten()
        max_idx = np.argmax(eval_obj - cur_obj)
        dx, dy = eval_points[max_idx] - best_centers[i]

        if eval_obj[max_idx] - cur_obj > 1e-6:
            best_centers[i] += [dx, dy]
    
    return best_centers, best_radii, np.sum(best_radii)
