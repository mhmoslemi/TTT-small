import numpy as np
import scipy.optimize

def solve_radii(centers):
    n = centers.shape[0]
    B = np.min(np.min(centers, axis=1)[np.newaxis, :] - np.max(centers, axis=1)[:, np.newaxis], axis=0) / 2
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    dists[np.triu_indices(n, k=1)] -= 2 * B[np.triu_indices(n, k=1)]
    bounds = [(0, B[i]) for i in range(n)]
    constraints = [{'type': 'ineq', 'fun': lambda r, i=i: B[i] - r[i]},
                  {'type': 'ineq', 'fun': lambda r, i=i, j=j: dists[i][j] - r[i] - r[j] if j > i else 0} for j in range(i+1, n)]
    result = scipy.optimize.minimize(lambda r: -np.sum(r), np.ones(n), method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-12)
    if not result.success:
        return np.zeros(n), -np.inf
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    hex_radius = 0.099 / np.sqrt(3)
    hex_x = (np.arange(5) * 2 + 1) * hex_radius
    hex_y = np.concatenate([np.arange(5) * np.sqrt(3) * hex_radius, np.arange(4) * np.sqrt(3) * hex_radius + 0.5 * np.sqrt(3) * hex_radius])
    hex_points = np.column_stack([np.tile(hex_x, 5), np.repeat(hex_y, 5)])
    sides_midpoints = np.array([[0.5, 0.05], [0.95, 0.5], [0.05, 0.5]])
    initial_centers = np.vstack([hex_points, sides_midpoints])
    
    initial_radii, initial_energy = solve_radii(initial_centers)
    initial_sum_radii = np.sum(initial_radii)
    
    def lp_objective(centers):
        return -solve_radii(centers)[1]
    
    initial_energy = lp_objective(initial_centers)
    current_centers = initial_centers.copy()
    current_energy = initial_energy
    
    T = 0.1
    alpha = 0.9999
    epsilon = 1e-4
    delta = 1e-4
    
    accepted = False
    for _ in range(20000):
        if T < 1e-6:
            break
        
        i = np.random.randint(26)
        dx, dy = np.random.normal(0, T / 2)
        proposed_centers = current_centers.copy()
        proposed_centers[i] += np.array([dx, dy])
        proposed_centers[i] = np.clip(proposed_centers[i], 0, 1)
        if np.min(np.linalg.norm(proposed_centers[:, np.newaxis, :] - proposed_centers[np.newaxis, :, :], axis=2), axis=1).min() < epsilon:
            continue
        
        proposed_energy = lp_objective(proposed_centers)
        if proposed_energy > current_energy or np.random.rand() < np.exp((proposed_energy - current_energy) / T):
            current_centers = proposed_centers
            current_energy = proposed_energy
            accepted = True
    
    if not accepted:
        return solve_radii(initial_centers)
    
    for i in range(26):
        while True:
            dx, dy = np.random.choice([-delta, delta], size=2)
            proposed_centers = current_centers.copy()
            proposed_centers[i] += np.array([dx, dy])
            proposed_centers[i] = np.clip(proposed_centers[i], 0, 1)
            proposed_energy = lp_objective(proposed_centers)
            if proposed_energy > current_energy or np.random.rand() < np.exp((proposed_energy - current_energy) / T):
                current_centers = proposed_centers
                current_energy = proposed_energy
                break
    
    final_radii, final_sum_radii = solve_radii(current_centers)
    return current_centers, final_radii, final_sum_radii
