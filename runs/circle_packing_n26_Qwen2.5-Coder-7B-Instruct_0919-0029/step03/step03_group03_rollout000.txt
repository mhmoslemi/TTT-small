import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    n = centers.shape[0]
    B = np.minimum(centers[:, 0] - centers[:, 1], centers[:, 0] + centers[:, 1] - 4, centers[:, 2] - centers[:, 0], centers[:, 2] + centers[:, 0] - 1, centers[:, 3] - centers[:, 2], centers[:, 3] + centers[:, 2] - 1)
    dists = np.sqrt(np.sum((centers[:, None, 0:2] - centers[:, 0:2, None])**2, axis=2))
    A = np.c_[B,B,B,dists]
    b = np.zeros_like(np.sum(A,-1))
    bounds = [(0, None) for _ in range(n)] + [(0, None) for _ in range(n)] + [(0, None) for _ in range(n)] + [(0, None) for _ in range(np.size(dists)//3)]
    result = linprog(-np.ones_like(B), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    radii = result.x[:n]
    return radii, np.sum(radii)

def hex_grid_centers(n: int, r: float) -> np.ndarray:
    centers = np.zeros((n, 4))
    h = np.sqrt(3) * r
    k = 1 / (2 * r)
    layer = 0
    while len(centers) < n:
        layer += 1
        for i in range(layer):
            for j in range((layer - 1) // 2, (layer + 1) // 2):
                centers = np.vstack((centers, np.array([(i + (j % 2)) * k * 0.5, (+j * 0.5) * h + (layer - 1) // 4 * h, r, 0.])))
    return centers[:n]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_circles = hex_grid_centers(26, 0.1)
    radii = compute_max_radii(initial_circles)[0]
    return initial_circles[:, :2], radii, np.sum(radii)
