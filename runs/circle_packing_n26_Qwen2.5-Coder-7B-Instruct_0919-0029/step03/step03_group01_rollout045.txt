import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def compute_dual_direction(centers, radii, primal_grad):
    n = centers.shape[0]

    # Compute the dual dual direction
    Aprof = np.zeros((n, n*(n-1)//2 + 4*n))
    K = np.zeros(n*(n-1)//2 + 4*n)
    idx = 0

    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            Aprof[i, idx] = -1 dist
            Aprof[j, idx] = -1 dist
            K[idx] = dist - radii[j] - radii[i]
            idx += 1

    for i in range(n):
        Aprof[i, idx] = -1  # Left boundary constraint
        K[idx] = centers[i, 0] - radii[i]
        idx += 1
        Aprof[i, idx] = -1  # Right boundary constraint
        K[idx] = 1 - centers[i, 0] - radii[i]
        idx += 1
        Aprof[i, idx] = -1  # Bottom boundary constraint
        K[idx] = centers[i, 1] - radii[i]
        idx += 1
        Aprof[i, idx] = -1  # Top boundary constraint
        K[idx] = 1 - centers[i, 1] - radii[i]
        idx += 1

    G = Aprof.dot(primal_grad)
    h = K

    # Solve for the step direction
    step_dual_dir = np.linalg.solve(G.T.dot(G), G.T.dot(h))
    return step_dual_dir

def propose_move_dual(start, step, direction):
    return start + step * direction

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    step, step_min = 0.1, 1e-4
    decay_factor = 0.995
    NUM_ITERS = 4000
    NUM_STARS = 20

    PATTERNS = [

               + np.array([[0.05, 0.05] if (k == 0 and l == 0) else [0, 0] for k in range(5) for l in range(5)]),
        lambda: np.array([[0.1 + 0.2*k, 0.1 + 0.2*l] for k in range(5) for l in range(5)])
               + np.array([[0, 0] if (k % 2 == 0 and l % 2 == 1) else [0.1, 0] for k in range(5) for l in range(5)]),
        lambda: np.array([[0.1 + k*0.15, 0.1 + l*0.07*np.sqrt(3)] for k in range(int(5/0.3)) for l in range(int(3/0.38))])
               [:-4],
        lambda: np.array([[0.1, 0.1], [0.1, 0.9], [0.1, 0.5], [0.9, 0.1], [0.9, 0.9], [0.9, 0.5], [0.5, 0.1], [0.5, 0.9]])
               + 0.15*np.random.rand(8, 2)
    ]

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern in PATTERNS:
        for start in range(NUM_STARS):
            centers = pattern()
            for it in range(NUM_ITERS):
                radii, sum_radii = lp_optimum(centers)
                primal_grad = lp_optimum(np.ones_like(centers))[1]
                dual_direction = compute_dual_direction(centers, radii, primal_grad)
                dir_norm = np.linalg.norm(dual_direction)
                if dir_norm > 0:
                    dual_direction /= dir_norm

                mut = np.random.rand()
                if mut < 0.2:
                    # Random perturbation
                    direct = np.random.randn(2)
                    direct /= np.linalg.norm(direct)
                    direct *= step
                else:
                    # Dual-guided step
                    direct = propose_move_dual(start, step, dual_direction)
                    direct = np.clip(direct, 0, 1)

                new_center = centers + direct
                new_center = np.clip(new_center, 0, 1)
                if validate_packing(np.atleast_2d(new_center), np.atleast_1d(sum_radii[:])[np.newaxis])[0]:
                    centers = new_center

                if it % 500 == 0:
                    new_radii, new_sum = lp_optimum(centers)
                    if new_sum > best_sum:
                        valid, _ = validate_packing(centers, new_radii)
                        if valid:
                            best_sum = new_sum
                            best_centers = new_centers
