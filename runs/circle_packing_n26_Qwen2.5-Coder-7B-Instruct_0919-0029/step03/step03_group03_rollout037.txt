import numpy as np
from scipy.optimize import linprog

def lp_solve(centers):
    n = len(centers)
    B = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    distances = np.full((n, n), -1)
    np.fill_diagonal(distances, 0)
    for i in range(n):
        for j in range(i + 1, n):
            distances[i, j] = distances[j, i] = np.linalg.norm(centers[i] - centers[j])
    
    c = -np.ones(n)
    A_ub = np.vstack((np.diag(B), distances - np.eye(n)))
    b_ub = np.hstack((B, np.zeros(n)))
    x0_bounds = np.column_stack((np.zeros(n), B))
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=x0_bounds, method='highs')
    if result.success:
        radii = result.x
        return radii, sum(radii)
    else:
        return None, float('-inf')

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Seed configuration
    centers = np.array([
        [0.1, 0.2], [0.2, 0.1], [0.2, 0.4], [0.4, 0.2],
        [0.4, 0.5], [0.5, 0.4], [0.5, 0.7], [0.7, 0.5],
        [0.7, 0.8], [0.8, 0.7], [0.15, 0.6], [0.6, 0.15],
        [0.6, 0.9], [0.9, 0.6], [0.3, 0.85], [0.85, 0.3],
        [0.85, 0.95], [0.95, 0.85], [0.125, 0.55], [0.55, 0.125],
        [0.55, 0.65], [0.65, 0.55], [0.375, 0.775], [0.775, 0.375],
        [0.2, 0.75], [0.75, 0.2]
    ])
    centers = np.vstack((centers, np.array([[0.5, 0.05], [0.95, 0.5], [0.05, 0.5]])))
    
    # Initial LP solve
    radii, sum_radii = lp_solve(centers)
    best_sum = sum_radii
    best_centers = centers
    
    # Simulated Annealing
    T = 0.1
    alpha = 0.9999
    while T > 1e-6:
        i = np.random.randint(n)
        dx, dy = np.random.normal(0, 1e-2, 2)
        centers[i] = np.clip(centers[i] + np.array([dx, dy]), 0, 1)
        if np.any(np.linalg.norm(centers - centers[i], axis=1) < 2e-4):
            centers[i] -= np.array([dx, dy])
            continue
        radii, sum_radii = lp_solve(centers)
        if sum_radii > best_sum or np.random.rand() < np.exp((sum_radii - best_sum) / T):
            best_sum = sum_radii
            best_centers = centers
        else:
            centers[i] -= np.array([dx, dy])
        T *= alpha
    
    # Gradient refinement
    delta = 1e-4
    while True:
        improved = False
        for i in range(n):
            for dx, dy in [-delta, delta]:
                centers_copy = best_centers.copy()
                centers_copy[i] += np.array([dx, dy])
                if np.any(centers_copy < 0) or np.any(centers_copy > 1):
                    continue
                radii, sum_radii = lp_solve(centers_copy)
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = centers_copy
                    improved = True
                    break
            if improved:
                break
        else:
            break
    
    return best_centers, best_centers[:, 2], best_sum
