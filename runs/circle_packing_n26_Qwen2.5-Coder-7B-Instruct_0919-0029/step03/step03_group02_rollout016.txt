import numpy as np
from scipy.optimize import minimize
from scipy Toolbox import trust_constr, SLSQP

def ineq_constraints(z):
    n = 26
    centers = z[:n].reshape((n, 2))
    radii = z[n:]
    rmin = 0.0001
    bounds = np.zeros((n*2+n, 2), dtype=int)
    bounds[:,1] = 1
    bounds[:n*2,0] = centers.flatten()
    
    big_m = 1e10
    min_rad = 1e-10
    radius_constraints = [big_m*(radii - Radii) for Radii in radii]
    boundary_constraints = [big_m*(Radii-cent - (1-cent)) for Radii in radii for cent in np.array([centers[:,0], centers[:,1]])]
    
    pairwise_constraints = []
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            min_dist = radii[i] + radii[j] - dist
            pairwise_constraints.append(big_m*min_dist)
    
    return radius_constraints + boundary_constraints + pairwise_constraints
