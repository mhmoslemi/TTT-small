def prepare_initial_guess():
    centers = np.zeros((26, 2))
    radii = np.full(26, 0.1)
    radii[12] = np.sqrt(0.08) - 0.1
    x_center = 0.5
    y_center = 0.5
    for i in range(25):
        radius = 0.1
        r = int(np.sqrt(i // 5))
        c = i % 5
        if i < 9:
            centers[i, 0] = 0.1 + 0.2 * c
            centers[i, 1] = 0.1 + 0.2 * r
        elif i < 18:
            centers[i, 0] = 0.1 + 0.2 * c
            centers[i, 1] = 0.3 + 0.2 * r
        else:
            centers[i, 0] = 0.3 + 0.2 * c
            centers[i, 1] = 0.3 + 0.2 * r
    centers[12, 0] = x_center
    centers[12, 1] = y_center
    initial_guess = np.hstack((centers, radii))
    return initial_guess

def ineq_constraints(z):
    n = 26
    x, y, r = z[:2*n].reshape(2, n), z[2*n:].reshape(n,)
    constraints = np.zeros((n + 4*n + n*(n-1)//2, 3*n))
    for i in range(n):
        constraints[i, i*3:i*3+3] = -2 * r[i]
    
    for i in range(n):
        constraints[i+n, i*3:i*3+3] = np.array([[0, 1, 0], [-1, 0, 0], [0, 0, 1]])
        constraints[i+n+1, i*3:i*3+3] = np.array([[0, 1, 0], [1, 0, 0], [0, 0, -1]])
        constraints[i+n+2, i*3:i*3+3] = np.array([[0, 0, 1], [0, 1, 0], [1, 0, 0]])
        constraints[i+n+3, i*3:i*3+3] = np.array([[0, 0, 1], [0, 1, 0], [-1, 0, 0]])
    
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = x[i, :], y[i, :], r[i]
            xj, yj, rj = x[j, :], y[j, :], r[j]
            diff = np.array([xj-xi, yj-yi])
            length = np.sqrt(np.sum(diff**2))
            constraints[i*n*(n-1)//2 + j - n + 4*n, i*3:i*3+3] = np.array([1, 1, 0])
            constraints[i*n*(n-1)//2 + j - n + 4*n, j*3:j*3+3] = -np.array([1, 1, 0])
            constraints[i*n*(n-1)//2 + j - n + 4*n + 1, i*3:i*3+3] = -diff / length
            constraints[i*n*(n-1)//2 + j - n + 4*n + 1, j*3:j*3+3] = diff / length
    
    return constraints

def objective(z):
