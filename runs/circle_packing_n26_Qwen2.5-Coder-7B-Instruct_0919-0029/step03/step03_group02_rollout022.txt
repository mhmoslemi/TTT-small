import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.array([
        [0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)
    ])
    centers = np.append(centers, [[0.5, 0.5]])
    
    r_new = math.sqrt(0.08) - 0.1
    radii = np.append(np.full(25, 0.1), r_new)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
