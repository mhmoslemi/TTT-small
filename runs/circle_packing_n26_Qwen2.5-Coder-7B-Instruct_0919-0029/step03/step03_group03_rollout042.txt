import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    B = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    d = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    d[d < 1e-8] = 1e-8
    A = np.hstack((np.eye(n), -np.eye(n), -np.eye(n), np.eye(n)))
    b = np.hstack((B, B, B, B))
    c = -np.ones(n)
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda z, i=i, j=j: -(z[i + 2 * n] + z[j + 2 * n]) + np.sqrt((z[i] - z[j])**2 + (z[i + n] - z[j + n])**2)})
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * (3 * n), constraints=cons)
    return result['x'][2::3], np.sum(result['x'][2::3])

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = np.random.rand(2)
        while True:
            radius = np.random.rand() / 100
            valid = True
            for j in range(i):
                if np.sqrt((x - centers[j, 0])**2 + (y - centers[j, 1])**2) < radius + radii[j]:
                    valid = False
                    break
            if valid:
                centers[i] = np.array([x, y])
                break
    
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = 0
    
    for _ in range(10):
        centers = best_centers.copy()
        for i in range(n):
            x, y = centers[i]
            while True:
                radius = np.random.rand() / 100
                valid = True
                for j in range(n):
                    if np.sqrt((x - centers[j, 0])**2 + (y - centers[j, 1])**2) < radius + radii[j]:
                        valid = False
                        break
                if valid:
                    centers[i] = np.array([x, y])
                    break
        
        _, sum_radii = compute_max_radii(centers)
        if sum_radii > best_sum:
            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = sum_radii
    
    return centers, best_radii, best_sum
