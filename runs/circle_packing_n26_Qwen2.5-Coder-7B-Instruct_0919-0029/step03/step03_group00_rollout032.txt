import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform
from numpy.linalg import norm

def lp_optimum(centers):
    n = centers.shape[0]
    A = -np.eye(n)
    b = np.zeros(n)
    D = squareform(pdist(centers)) - 1e-12
    for i in range(n):
        A = np.vstack((A, np.ones(n)))
        b = np.append(b, [0])
    A = np.vstack((A, np.append(np.append(np.zeros(n), 1), [100] * (n - 1))))
    b = np.append(b, [np.inf] + [0] * (n - 1))
    result = opt.linprog(-np.ones(n), A_ub=np.append(A, -D, axis=0), b_ub=np.append(b, -D.sum(axis=0)), bounds=(0, 1), method='highs')
    return result.x, -result.fun if result.success else np.zeros(n), result.success

def custom_init(n, bounds):
    # Hexagonal lattice
    dtheta = np.pi / 3
    d = 2 * np.cos(dtheta)
    half_d = d / 2
    r = 0.1
    half_r = r / 2
    optimal_spacing = r + r * np.tan(dtheta / 2)
    
    hex_anchors = []
    for i in range(5):
        for j in range(6):
            x = i * optimal_spacing + half_r * (j % 2)
            y = j * optimal_spacing
            if x < 1 - half_r:
                hex_anchors.append(np.array([x, y]))
    
    # Corner anchors
    corner_anchors = [
        np.array([0.03, 0.03]),
        np.array([0.03, 1 - 0.03]),
        np.array([1 - 0.03, 0.03]),
        np.array([1 - 0.03, 1 - 0.03])
    ]
    
    # Combine and add jitter
    centers = np.array(hex_anchors + corner_anchors)
    jitter = np.random.randn(34, 2) * 0.01
    centers[:34] += jitter
    
    if n > 34:
        remaining = np.random.rand(n - 34, 2) * 0.94 + 0.03
        centers = np.vstack((centers, remaining))
    
    return centers

def run_packing():
    n = 26
    dim = 2 * n
    bounds = [(0, 1)] * dim
    popsize = 15
    maxiter = 200
    strategy = 'best1bin'
    
    def neg_sum_of_radii(x):
        centers = x.reshape((n, 2))
        radii, sum_r = lp_optimum(centers)
        return -sum_r
    
    def initial_population(n, bounds):
        return custom_init(n, bounds)
    
    result = opt.differential_evolution(neg_sum_of_radii, bounds, args=(n,), popsize=popsize, maxiter=maxiter, strategy=strategy, init=initial_population, polish=False, workers=1)
    
    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0
    
    best_centers = np.array(result.x).reshape((n, 2))
    best_radii, best_sum = lp_optimum(best_centers)
    
    # SA refinement
    step0 = 0.05
    step_min = 1e-4
    decay_step = 0.995
    T0 = 0.05
    T_min = 1e-5
    decay_T = 0.995
    
    cur_centers = best_centers.copy()
    cur_radius, cur_sum = lp_optimum(cur_centers)
    for it in range(5000):
        idx = np.random.randint(n)
        move = np.random.randn(2) * step0
        new_centers = cur_centers.copy()
        new_centers[idx] += move
        new_centers[idx] = np.clip(new_centers[idx], 0, 1)
        new_radius, new_sum = lp_optimum(new_centers)
        acceptance_probability = np.exp((new_sum - cur_sum) / T0)
        if new_sum > cur_sum or np.random.rand() < acceptance_probability:
            cur_centers, cur_radius, cur_sum = new_centers, new_radius, new_sum
        
        step0 = np.maximum(step_min, step0 * decay_step)
        T0 = np.maximum(T_min, T0 * decay_T)
    
    _, refined_sum = lp_optimum(cur_centers)
    return cur_centers, refined_sum

centers, _, sum_radii = run_packing()
