import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1] - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1] - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing():
    n = 26

        cent -= np.min(cent, axis=0)[None, :]
        cent /= np.max(cent, axis=0)[None, :]
        cent *= (1 - 2 * r)
        cent += r
        center nearest to border
        if np.all(centers[:, 0] >= r) and np.all(centers[:, 0] <= 1 - r):
            if np.all(centers[:, 1] >= r) and np.all(centers[:, 1] <= 1 - r):
                for i in range(n):
                    for j in range(i + 1, n):
                        if np.linalg.norm(centers[i] - centers[j]) < radii[i] + radii[j]:
                            radii[i] = np.max([radii[i], r - (np.linalg.norm(centers[i] - centers[j]) / 2)])
                            radii[j] = np.max([radii[j], r - (np.linalg.norm(centers[i] - centers[j]) / 2)])
                return centers, radii, np.sum(radii)
    return centers, radii, np.sum(radii)
