import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def objective(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return -np.sum(r)

def overlap_constraint(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    dists = squareform(pdist(np.column_stack((x, y))))
    return dists - r[:, np.newaxis] - r[np.newaxis, :] + 1e-12

def radius_bounds(z):
    r = z[52:]
    return r

def initial_guess():
    grid_radius = 0.10
    grid_step = 0.2
    warm_start = np.zeros(78)
    for i in range(25):
        row, col = divmod(i, 5)
        warm_start[i*6:i*6+3] = grid_radius + grid_step * np.array([col, row, 0])
        warm_start[i*6+3+i%5+2] = grid_radius + grid_step * np.array([col, row+1, 0])
        warm_start[i*6+3+i//5+0] = grid_radius + grid_step * np.array([col, row, 1])
        warm_start[i*6+3+i//5+1] = grid_radius + grid_step * np.array([col+1, row, 0])

    center_radius = np.sqrt(0.08) - grid_radius
    warm_start[52:77] = center_radius
    
    return warm_start - np.max(warm_start) * 0.01  # slightly perturbed to avoid boundary issues

def run_packing():
    initial_vector = initial_guess()
    n = 26
    initial_bounds = [(0, None) for _ in range(78)]
    linear_constraints = [
        {'type': 'ineq', 'fun': radius_bounds},
    ]

    non_linear_constraints = [{'type': 'ineq', 'fun': overlap_constraint},
                              {'type': 'ineq', 'fun': lambda z: z[:26] - 1e-12},
                              {'type': 'ineq', 'fun': lambda z: 1 - z[:26] - 1e-12},
                              {'type': 'ineq', 'fun': lambda z: z[26:52] - 1e-12},
                              {'type': 'ineq', 'fun': lambda z: 1 - z[26:52] - 1e-12}
                             ]

    options = {
        'max_iter': 2000,
        'gtol': 1e-8,
        'xtol': 1e-8,
        'disp': False,
    }
    result = minimize(objective, initial_vector, method='trust-constr', constraints=linear_constraints + non_linear_constraints, options=options, jac=None, hess=None)

    if not result.success:
        raise Exception("Optimization failed")
    
    centers = result.x[:52].reshape(n, 2)
    radii = -result.fun

    return centers, radii
