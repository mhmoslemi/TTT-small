import numpy as np
from scipy.spatial import Delaunay
from scipy.optimize import linprog

def calculate_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        dists = np.linalg.norm(centers - centers[i], axis=1)
        radii[i] = np.min(dists[dists != 0]) / 2
    return radii

def lp_optimization(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + n, n))
    b_ub = np.zeros(A_ub.shape[0])

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    for i in range(n):
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    NUM_ITERATIONS = 10
    NUM_RETRIES = 5

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for _ in range(NUM_RETRIES):
        initial_centers = np.random.rand(n, 2)
        for _ in range(NUM_ITERATIONS):
            centers = initial_centers.copy()
            radii = calculate_radii(centers)
            new_centers, new_sum = lp_optimization(centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers
                initial_centers = new_centers
            else:
                break

    return best_centers, calculate_radii(best_centers), best_sum
