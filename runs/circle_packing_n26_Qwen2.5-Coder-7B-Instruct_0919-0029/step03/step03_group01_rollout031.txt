import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def generate_center_pattern(pattern_name):
    if pattern_name == 'Random_inner':
        return np.random.uniform(0.05, 0.95, size=(26, 2))
    elif pattern_name == 'Grid5x5_cut':
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        x, y = np.meshgrid(x, y)
        centers = np.array([(x[i, j], y[i, j]) for i in range(5) for j in range(5)]).reshape(25, 2)
        centers[0] = np.random.uniform(0.05, 0.95, 2)
        return centers
    elif pattern_name == 'OffsetGrid':
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        centers_even = np.array([(x[i, j], y[i, j]) for i in range(5) for j in range(5, 0, -1)]).reshape(25, 2)
        centers_odd = np.array([(x[i, j], y[i, j]) for i in range(5) for j in range(5)]).reshape(25, 2)
        centers_odd[:, 0] += 0.1
        return np.concatenate((centers_even, centers_odd), axis=0)
    elif pattern_name == 'SparseHex':
        side = 0.15
        height = np.sqrt(3) * 0.07 * 2
        centers = []
        for i in range(4):
            y = 0.1 + i * height
            for j in range(5):
                x = 0.15 + j * 2 * side + (i % 2) * side
                centers.append((x, y))
        centers = np.array(centers)
        mask = centers[:, 0] >= 0.1 and centers[:, 0] <= 0.9 and centers[:, 1] >= 0.1 and centers[:, 1] <= 0.9
        centers = centers[mask][:26]
        centers = np.delete(centers, [0, 1, 23, 25], axis=0)
        return centers
    elif pattern_name == 'CornerExpansion':
        x_center = np.random.uniform(0.1, 0.9, 18)
        y_center = np.random.uniform(0.1, 0.9, 18)
        centers = np.column_stack((x_center, y_center))
        side = 0.15
        midpoints = np.array([
            (0.5, 0.5),
            (0.1, 0.5), (0.9, 0.5),
            (0.5, 0.1), (0.5, 0.9)
        ])
        for midpoint in midpoints:
            x = midpoint[0]
            y = midpoint[1]
            centers = np.vstack((centers, (x, y - side), (x, y + side), (x - side, y), (x + side, y)))
        return centers
    return None

def lp_optimum(centers):
    n = centers.shape[0]
    radius_bound = 1 / (2 * n)
    radius_start = radius_bound * np.ones(26)
    res = linprog([-sum(radius_start)], A_ub=None, b_ub=None, A_eq=None, b_eq=None, bounds=(0, 0.25), method='simplex')
    radii = res.x
    return radii, -res.fun

def lp_solve(centers):
    n = centers.shape[0]
    A = np.zeros((n*(n-1)//2 + n, n))
    b = np.zeros(A.shape[0])
    for i in range(n):
        for j in range(i+1, n):
            dx, dy = centers[i] - centers[j]
            d = np.sqrt(dx**2 + dy**2)
            A[i*(n-1)//2 + i, i] = 1
            A[i*(n-1)//2 + i, j] = -1
            A[i*(n-1)//2 + j, i] = -1
            A[i*(n-1)//2 + j, j] = 1
            b[i*(n-1)//2 + i] = d - 1e-6
    for i in range(n):
        A[n*(n-1)//2 + i, i] = 1
        b[n*(n-1)//2 + i] = 1
        A[n*(n-1)//2 + i, i] = 1
        b[n*(n-1)//2 + i] = 1
    res = linprog(None, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x

def run_packing():
    eps = 1e-6
    NUM_ITERS_PER_PATTERN, NUM_STARS = 3000, 5
    RESTART_EPOCHS = 1000
    best_sum, best_centers = 0, None
    np.random.seed(0)

    for start in range(NUM_STARS):
        cur_sum, cur_center = 0, None
        for pattern_name in ['SparseHex', 'CornerExpansion', 'Random_inner', 'Grid5x5_cut', 'OffsetGrid']:
            centers = generate_center_pattern(pattern_name)
            radii = lp_solve(centers)
            new_sum = np.sum(radii)
            if new_sum > cur_sum:
                cur_sum = new_sum
                cur_center = centers
                alpha_t, beta_t = 1.0, 1.0
            else:
                alpha_t, beta_t = alpha_t * 0.95, beta_t * 0.995

            for epoch in range(RESTART_EPOCHS):
                new_centers = np.copy(cur_center)
                for i in range(len(cur_center)):
                    dx, dy = np.random.randn(2) * alpha_t, np.random.randn(2) * beta_t
                    dx, dy = dx / np.linalg.norm(dx), dy / np.linalg.norm(dy)
                    new_centers[i, :] += dx + dy
                    new_centers[i, :] = np.clip(new_centers[i, :], eps, 1 - eps)

                for i in range(len(new_centers)):
                    new_centers[i, :] += np.random.randn(2) * alpha_t / np.linalg.norm(cur_center[i, :] - cur_center[i-1, :])
                    new_centers[i, :] = np.clip(new_centers[i, :], eps, 1 - eps)

                if new_sum < cur_sum:
                    alpha_t, beta_t = alpha_t * 0.95, beta_t * 0.995
                else:
                    cur_center = new_centers
                    radii = lp_solve(cur_center)
                    new_sum = np.sum(radii)
                    alpha_t, beta_t = alpha_t * 0.95, beta_t * 0.995
                    beta_t = max(beta_t * 0.995, 1e-4)

                if new_sum > cur_sum and new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = cur_center

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_center
    return best_centers, np.sqrt(best_sum), best_sum

centers, radii, sum_radii = run_packing()
