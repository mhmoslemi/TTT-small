import numpy as np
from scipy.optimize import linprog
from scipy.linalg import cholesky, solve_triangular

def update_positions(x, sigma):
    n = x.shape[0]
    for i in range(n):
        x[i] += np.random.randn(2) * sigma
        x[i] = np.clip(x[i], 0, 1)
        if np.min([x[i, 0] - x[i, 1], x[i, 0] + x[i, 1], x[i, 1] - x[i, 0], x[i, 1] + x[i, 0]]) < 1e-4:
            x[i] -= np.random.randn(2) * sigma
            x[i] = np.clip(x[i], 0, 1)

def compute_radii(x):
    n = x.shape[0]
    r = np.zeros(n)
    d = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d[i, j] = np.sqrt(np.sum((x[i] - x[j]) ** 2))
            d[j, i] = d[i, j]
    for i in range(n):
        L = cholesky(-(d[i][:, None] + d[:, None] - 2 * np.eye(n))), lower=True)
        gamma = np.exp(-np.sum(solve_triangular(L, np.array([r[i]]), lower=True)))
        r[i] = gamma * np.min([x[i, 0] - x[i, 1], x[i, 0] + x[i, 1], x[i, 1] - x[i, 0], x[i, 1] + x[i, 0]])
    return r

def valid(x, r):
    if (x[:, 0] - r < 0).any() or (x[:, 0] + r > 1).any() or (x[:, 1] - r < 0).any() or (x[:, 1] + r > 1).any():
        return False
    for i in range(len(r)):
        for j in range(i + 1, len(r)):
            if np.linalg.norm(x[i] - x[j]) < r[i] + r[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x = np.zeros((n, 2))
    r = np.zeros(n)
    step_scale = 0.1
    sigma = step_scale
    T0 = 0.1
    T = T0
    cooling_rate = 0.9999
    gamma = 0.5
    step_count = 20000
    best_sum = 0
    best_x = np.zeros((n, 2))
    best_r = np.zeros(n)

    # Initial configuration
    for i in range(5):
        for j in range(5):
            if i % 2 == 0:
                x[i * 5 + j] = np.array([i * 2 * 0.099, j * 1.732 * 0.099])
            else:
                x[i * 5 + j] = np.array([(i + 0.5) * 2 * 0.099, j * 1.732 * 0.099])
    for i in [1, 3]:
        x[i * 5 + 0, 0] = 0.5
        x[i * 5 + 3, 0] = 0.95
        x[i * 5 + 1, 1] = 0.5
        x[i * 5 + 2, 1] = 0.5

    # SA loop
    for _ in range(step_count):
        i = np.random.randint(n)
        update_positions(x[[i]], sigma)
        r = compute_radii(x)
        if not valid(x, r):
            step_scale *= gamma
            continue
        new_sum = np.sum(r)
        if new_sum > best_sum or np.random.rand() < np.exp((new_sum - best_sum) / T):
            best_sum = new_sum
            best_x = x.copy()
            best_r = r.copy()
        step_scale *= gamma
        T *= cooling_rate

    # Gradient refinement
    sensitivity = np.zeros_like(x)
    for i in range(n):
        x_p = x.copy()
        for j in range(2):
            x_p[i, j] += 1e-6
            r_p = compute_radii(x_p)
            if not valid(x_p, r_p):
                sensitivity[i, j] = -np.inf
            else:
                sensitivity[i, j] = np.sum(r_p) - best_sum
            x_p[i, j] = x[i, j]

        grad_norm = np.linalg.norm(sensitivity[i])
        if grad_norm > 1e-6:
            x[i] += sensitivity[i] * 1e-6 / grad_norm
            r = compute_radii(x)
            if np.sum(r) > best_sum:
                best_sum = np.sum(r)
                best_x = x.copy()
                best_r = r.copy()

    return best_x, best_r, best_sum
