import numpy as np
from scipy.optimize import minimize
import scipy.spatial as spatial

def objective(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return -np.sum(r)

def constraint_distances(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    n = len(x)
    dists = spatial.distance_matrix(np.column_stack((x, y)), np.column_stack((x, y)))
    np.fill_diagonal(dists, np.inf)
    return r + r - dists

def constraint_radius(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return r

def constraint_square(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    return np.vstack((np.max(r) - 0.5, 0.5 - np.min(r), x + r - 1, 0 - x - r, y + r - 1, 0 - y - r)).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    numstarts = 1
    best_centers, best_sum, best_radii = None, -np.inf, None

            elif i < 18:
                centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
                centers[i, 1] = step + 0.5 * step
            else:
                centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
                centers[i, 1] = step + 0.5 * step * 2

        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
        centers -= np.min(centers, axis=0)
        centers /= np.max(centers, axis=0)
        centers *= 0.98
        centers += 0.01

        r_init = np.full((26,), 0.14)

            A = np.vstack([constraint_radius(z0), C]).T

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return None, None, 0.0

    for i in range(n):
        if radii[i] < 0:
            return None, None, 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return None, None, 0.0

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return None, None, 0.0

    return centers, radii, np.sum(radii)
