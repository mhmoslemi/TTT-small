import numpy as np
from scipy.optimize import linprog

# Constants
eps = 1e-12
RESTART_EPOCHS = 200

def solve_lp(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0], dtype=float)

    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - eps
            idx += 1

    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - eps
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - eps
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - eps
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - eps
        idx += 1

    c = -np.ones(n)
    bounds = [(1e-6, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def dual_gradient(centers, radii):
    n = centers.shape[0]
    grad = np.zeros((n, 2))

    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            grad[i, 0] += (radii[j] + radii[i] - dist) * (centers[i, 0] - centers[j, 0]) / dist
            grad[i, 1] += (radii[j] + radii[i] - dist) * (centers[i, 1] - centers[j, 1]) / dist

    for i in range(n):
        x, y = centers[i]
        if x > 0: grad[i, 0] -= min(radii[i], x - eps) / x
        if x < 1: grad[i, 0] += min(radii[i], 1 - x - eps) / (1 - x)
        if y > 0: grad[i, 1] -= min(radii[i], y - eps) / y
        if y < 1: grad[i, 1] += min(radii[i], 1 - y - eps) / (1 - y)

    return grad

def propose_move(centers, grad, step=0.1):
    n = centers.shape[0]
    new_centers = centers.copy()
    total_grad_norm = np.linalg.norm(grad)

    if total_grad_norm > 1e-12:
        for i in range(n):
            new_centers[i] += step * grad[i] / total_grad_norm
            new_centers[i] = np.clip(new_centers[i], eps, 1-eps)

    return new_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step = 0.1
    decay_factor = 0.95
    max_iters_per_pattern = 2000
    random_patterns = [
        np.random.rand(n, 2),
        np.array([[0.1 + 0.2*k, 0.1 + 0.2*l] for k in range(5) for l in range(5)] +
                  [[0.1, 0.1]]),
        np.array([[0.1 + 0.2*k + 0.1*(k%2), 0.1 + 0.2*l] for k in range(5) for l in range(5)]),
        np.array([[0.1 + 0.15*numpy.cos(k * 2 * numpy.pi / 6 + 0.5), 0.1 + 0.15*numpy.sin(k * 2 * numpy.pi / 6 + 0.5)] for k in range(30)])[:n],
        np.array([[0.1 + 0.4*numpy.cos(k * 2 * numpy.pi / 8 + 0.5), 0.1 + 0.4*numpy.sin(k * 2 * numpy.pi / 8 + 0.5)] for k in range(8)] +
                  [[0.1 + 0.7*numpy.cos(k * 2 * numpy.pi / 8 + 0.5), 0.1 + 0.7*numpy.sin(k * 2 * numpy.pi / 8 + 0.5)] for k in range(8)] +
                  [[0.1 + 0.9*numpy.cos(k * 2 * numpy.pi / 8 + 0.5), 0.1 + 0.9*numpy.sin(k * 2 * numpy.pi / 8 + 0.5)] for k in range(8)] + 
                  [[0.1, 0.95], [0.95, 0.1], [0.95, 0.95], [0.1, 0.05]])
    ]

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for pattern in random_patterns:
        centers = pattern.copy()
        radii, cur_sum = solve_lp(centers)
        grad = dual_gradient(centers, radii)

        for _ in range(max_iters_per_pattern):
            new_centers = propose_move(centers, grad, step=step)
            new_radii, new_sum = solve_lp(new_centers)
            accept_prob = max(0, (new_sum - cur_sum) / (step * n * eps + 1e-6))

            if new_sum > cur_sum or np.random.rand() < np.exp((new_sum - cur_sum) / (step * n * eps + 1e-6)):
                centers, radii, cur_sum = new_centers, new_radii, new_sum
                grad = dual_gradient(centers, radii)

                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()

        step *= decay_factor

    return best_centers, best_radii, best_sum
