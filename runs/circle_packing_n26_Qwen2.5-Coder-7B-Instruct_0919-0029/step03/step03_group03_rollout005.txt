import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    bounds = [(0, None)] * n

    # Compute constraints
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    for i in range(n):
        for j in range(n):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                A_eq[i, j] = 2
                A_eq[i, i] -= 1
                b_eq[i] = dist**2

    # Solve LP problem
    result = linprog(-radii, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    radii = -result.x
    return radii, np.sum(radii)
