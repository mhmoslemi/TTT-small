import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    n = centers.shape[0]
    B = np.minimum(centers[:, 0], 1 - centers[:, 0])
    B = np.minimum(B, centers[:, 1])
    B = np.minimum(B, 1 - centers[:, 1])
    A = np.zeros((n * (n - 1) // 2, n + 1))
    B_ineq = np.zeros(n * (n - 1) // 2)
    for i in range(1, n):
        A[i * (n - 1) // 2:i * (n - 1) // 2 + i, :i + 1] = np.c_[np.ones(i), centers[:i] - centers[i]]
        A[i * (n - 1) // 2:i * (n - 1) // 2 + i, i + 1:n + 1] = -np.c_[np.ones(i), centers[i] - centers[:i]]
        B_ineq[i * (n - 1) // 2:i * (n - 1) // 2 + i] = centers[:i].diagonal().sum() + 1e-6
    res = linprog(-np.ones(n), bounds=(0, None), A_ub=A, b_ub=B_ineq, method='highs')
    if not res.success:
        return np.zeros(n), 0.0
    radii = res.x
    return radii, np.sum(radii)
