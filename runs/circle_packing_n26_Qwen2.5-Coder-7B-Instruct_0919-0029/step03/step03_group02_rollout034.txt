import numpy as np
from scipy.optimize import minimize, trust_constr

T0 = 0.1
Tf = 1e-4
alpha = 0.9995
max_iter = int(np.log(T0 / Tf) / np.log(alpha))

def build_initial_guess():
    centers = np.zeros((26, 2))
    radius = 0.10
    step = 1 / np.sqrt(2)

    for i in range(26):
        if i < 9:
            centers[i, 0] = step * (2 * (i % 3) + 1)
        elif i < 18:
            centers[i, 0] = step * (2 * ((i - 9) % 3) + 1)
            centers[i, 1] = step + 0.5 * step
        else:
            centers[i, 0] = step * (2 * ((i - 18) % 3) + 1)
            centers[i, 1] = step + 0.5 * step * 2

    centers += np.random.uniform(-1e-3, 1e-3, size=(26, 2))
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    centers *= 0.98  # make safe from boundary issues
    centers += 0.01  # simple clamping within range

    radius_center = np.sqrt(0.08) - radius
    radii = np.full(26, radius, dtype=float)
    radii[0] = radius_center
    return centers.flatten(), radii.flatten()

def ineq_constraints(z):
    n = 26
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    const_non_neg = r
    const_boundary = np.maximum(
        0, 
        np.vstack((r-x, x-r, r-y, y-r)).T.reshape(-1)
    )
    dists = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)
    const_pairwise = dists - np.outer(r, r) - 1e-12
    return np.hstack((const_non_neg, const_boundary, const_pairwise))

def objective(z):
    return -np.sum(z[26:])

def lp_refinement(centers, radii):
    dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)
    A_ineq = []
    b_ineq = []

    for i in range(26):
        A_ineq.append([1]*26)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0] - 1e-12)
        A_ineq.append([1]*26)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0] - 1e-12)
        A_ineq.append([1]*26)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 1] - 1e-12)
        A_ineq.append([1]*26)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 1] - 1e-12)

        for j in range(i + 1, 26):
            A_ineq.append([1]*26)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)
    bounds = [(None, None) for _ in range(78)]
    result = linprog([-np.ones(26)], A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    if result.success:
        radii_new = result.x
        idx_center = np.where(centers == centers[0]).item() // 2
        if radii_new[idx_center] > radii[idx_center]:
            radii[idx_center] = radii_new[idx_center]
    return centers.flatten(), radii.flatten()

def run_packing():
    centers, radii = build_initial_guess()
    centers, radii = lp_refinement(centers.reshape(26, 2), radii)
    initial_guess = np.hstack((centers, radii))
    result = minimize(objective, initial_guess, method='trust-constr',
                      jac=None, hess=None, bounds=[(0, None)] * 78,
                      options={'max_iter': max_iter * int(np.log(T0 / Tf) / np.log(alpha)), 'gtol': 1e-8, 'xtol': 1e-8})
    centers = result.x[:26].reshape(26, 2)
    radii = result.x[26:]
    sum_radii = -result.fun
    return centers, radii, sum_radii
