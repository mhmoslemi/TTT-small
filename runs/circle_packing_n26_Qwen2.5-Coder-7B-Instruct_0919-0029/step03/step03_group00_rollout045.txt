import numpy as np
from scipy.spatial import ConvexHull
from scipy.optimize import minimize

def circle_packing_sum(x):
    n = int(len(x) / 2)
    centers = x[:n].reshape (-1, 2)
    radii = x[n:]
    return -np.sum(radii)

def overlap_constraint(x):
    n = int(len(x) / 2)
    centers = x[:n].reshape (-1, 2)
    radii = x[n:]
    return (np.sqrt(np.sum(np.power(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], 2), axis=-1)) - (radii[:, np.newaxis] + radii[np.newaxis, :])).flatten()

def boundary_constraint(x):
    return np.minimum(np.maximum(x[:int(len(x) / 2)], 0), 1) - x[:int(len(x) / 2)]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    x0 = np.random.rand(2 * n)
    constraints = []
    constraints.append({
        'type': 'ineq',
        'fun': overlap_constraint
    })
    constraints.append({
        'type': 'ineq',
        'fun': boundary_constraint
    })
    x0[:n] = np.clip(x0[:n], 0.05, 0.95)
    x0[n:] = np.clip(x0[n:], 0.05, 1 - x0[:n].sum())
    result = minimize(circle_packing_sum, x0, method='SLSQP', constraints=constraints, bounds=[(0, 1)] * (2 * n))
    centers = result.x[:n].reshape(-1, 2)
    radii = result.x[n:]
    return centers, radii, -result.fun
