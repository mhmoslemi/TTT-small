import numpy as np
from math import sqrt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the centers of the 5x5 grid of circles
    centers = np.zeros((25, 2))
    for i in range(5):
        for j in range(5):
            centers[i*5 + j] = [0.10 + 0.20 * i, 0.10 + 0.20 * j]
    
    # Define the centers of the extra circle in the center of the void
    central_center = [0.5, 0.5]
    central_radius = sqrt(0.08) - 0.10
    
    # Combine all centers and radii
    centers = np.vstack((centers, central_center))
    radii = np.concatenate((np.full(25, 0.10), [central_radius]))
    
    # Compute the sum of radii
    sum_radii = np.dot(radii, radii)
    
    return centers, radii, sum_radii
