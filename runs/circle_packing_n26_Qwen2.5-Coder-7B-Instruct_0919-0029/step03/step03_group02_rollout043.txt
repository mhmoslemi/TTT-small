import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the 5x5 grid of circles with radius 0.1
    centers_grid = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    radii_grid = np.full(25, 0.1)

    # Define the central circle with radius sqrt(0.08) - 0.1
    centre_x, centre_y = 0.5, 0.5
    r_new = math.sqrt(0.08) - 0.1
    centre = np.array([centre_x, centre_y])

    # Combine centers and radii
    centers = np.vstack((centers_grid, centre))
    radii = np.hstack((radii_grid, [r_new]))

    # Compute the sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
