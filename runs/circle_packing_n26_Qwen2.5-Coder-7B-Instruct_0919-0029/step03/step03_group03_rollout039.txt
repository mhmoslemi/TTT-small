import numpy as np
from scipy.optimize import linprog

def solve_radii(centres):
    n = centres.shape[0]
    B = np.min(np.vstack((centres[:, 0] + centres[:, 1], 
                          centres[:, 0] - centres[:, 1], 
                          1 - centres[:, 0] + centres[:, 1], 
                          1 - centres[:, 0] - centres[:, 1])), axis=0)
    B = np.minimum(B, 0.1)  # Max radius is 0.1
    diffs = np.array([np.linalg.norm(centres[i] - centres[j]) - 2 * B[j] for j in range(n) for i in range(j)])
    diffs = diffs[diffs > 0]
    c = -np.ones(n)  # Minimise the negative of the sum of radii
    A = np.vstack((np.eye(n), -np.eye(n), diffs.reshape(-1, 1)))
    b = np.hstack((B, -B, np.zeros(len(diffs))))
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    if not res.success or np.isnan(res.x).any():
        return None, float('-inf')
    return res.x, -res.fun  # Return radii and objective value

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centre_grid = np.array([
        [j * 0.099 + 0.0495, 0] for j in range(5)] + 
        [[(2 * j + i + 1) * 0.099 + 0.0495, i * np.sqrt(3) * 0.099 + 0.0495] for i in range(1, 24) for j in range(4)] + 
        [[1, 0.05], [0.599, 0.5], [0.5, 0.5]]
    ])
    centre_grid = centre_grid[:26]

    current_centres = centre_grid.copy()
    current_sum = 0.0
    max_sum = float('-inf')

    T = 0.1
    alpha = 0.9999
    epsilon = 1e-4
    delta = 1e-4

    while T > 1e-6:
        for _ in range(1000):  # Number of moves per temperature phase
            for i in range(26):
                proposal = current_centres[i].copy()
                proposal += np.random.normal(0, T / 2, 2)
                proposal[proposal < 0] = 0
                proposal[proposal > 1] = 1
                
                if np.all(np.linalg.norm(proposal - current_centres[:, np.newaxis], axis=0) > epsilon) and proposal[0] > 0 and proposal[0] < 1 and proposal[1] > 0 and proposal[1] < 1:
                    current_centres[i] = proposal
                    _, current_sum = solve_radii(current_centres)
                    if current_sum > max_sum:
                        max_sum = current_sum
                        best_centres = current_centres.copy()
        
        T *= alpha
    
    current_centres = best_centres.copy()
    current_sum = max_sum

    for _ in range(1000):  # Polishing iterations
        improved = False
        for i in range(26):
            for sign in [-1, 1]:
                for dim in [0, 1]:
                    current_centres[i][dim] += sign * delta
                    _, current_sum = solve_radii(current_centres)
                    if current_sum > max_sum:
                        max_sum = current_sum
                        best_centres = current_centres.copy()
                        improved = True
                current_centres[i][dim] -= sign * delta
        
        if not improved:
            break

    return best_centres, solve_radii(best_centres)[0], max_sum
