import numpy as np
from scipy.optimize import linprog

def compute_max_radii(centers):
    n = centers.shape[0]
    r = np.zeros(n)
    B = np.maximum(np.maximum(centers[:, 0] - r, 1 - centers[:, 0] - r),
                   np.maximum(centers[:, 1] - r, 1 - centers[:, 1] - r))
    A = []
    b = []
    for i in range(n):
        A.append([r[i]] + [0]*n)
        b.append(B[i])
        for j in range(i+1, n):
            A.append([r[i], r[j]] + [0]*(n-2))
            b.append(centers[i] - centers[j])
    c = [-1]*n
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n)
    radii = res.x
    return radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.random.rand(26, 2) * 0.75 + 0.125
    while True:
        for _ in range(10):
            centers[centers > 1] = 1 - (centers[centers > 1] - 1)
            centers[centers < 0] = -centers[centers < 0]
        radii, sum_radii = compute_max_radii(centers)
        if sum_radii > 2.5:
            break
        centers = np.random.rand(26, 2) * 0.75 + 0.125
    return centers, radii, sum_radii
