import numpy as np
import scipy.optimize

def pen_optim(z, mu):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    
    # Objective function
    obj = -np.sum(r)
    
    # Pairwise distance constraints
    d_ij = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)
    s_ij = r[:, np.newaxis] + r[np.newaxis, :] - d_ij
    p_overlap = np.where(s_ij > 0, s_ij**2, 0).sum()

    # Boundary constraints
    p_left = np.where(r - x > 0, (r - x)**2, 0).sum()
    p_right = np.where(r - (1 - x) > 0, (r - (1 - x))**2, 0).sum()
    p_bottom = np.where(r - y > 0, (r - y)**2, 0).sum()
    p_top = np.where(r - (1 - y) > 0, (r - (1 - y))**2, 0).sum()
    p_non_negative = np.where(-r < 0, (-r)**2, 0).sum()

    penalty = mu * (p_overlap + p_left + p_right + p_bottom + p_top + p_non_negative)
    return obj + penalty

def grad_pen_optim(z, mu):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    
    # Pairwise distance constraints
    d_ij = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :])**2 + (y[:, np.newaxis] - y[np.newaxis, :])**2)
    s_ij = r[:, np.newaxis] + r[np.newaxis, :] - d_ij
    
    # Derivative term for overlap
    grad_s_ij = 2 * s_ij
    grad_d_ij = np.zeros((n, n, 2))
    for k in range(n):
        for l in range(k + 1, n):
            denom = d_ij[k, l] + 1e-12
            d_x = (x[k] - x[l]) / denom
            d_y = (y[k] - y[l]) / denom
            grad_d_ij[k, l] = np.array([d_x, d_y])
            grad_d_ij[l, k] = -grad_d_ij[k, l]
    
    grad_overlap = 2 * mu * np.sum(grad_s_ij[:, :, np.newaxis] * grad_d_ij, axis=(0, 1))
    
    # Derivative term for boundary constraints
    grad_left = 2 * mu * np.where(r - x > 0, np.ones_like(r) * (-1), np.zeros_like(r))
    grad_right = 2 * mu * np.where(r - (1 - x) > 0, (1 - 2 * x)[:, np.newaxis], np.zeros_like(r))
    grad_bottom = 2 * mu * np.where(r - y > 0, np.ones_like(r) * (-1), np.zeros_like(r))
    grad_top = 2 * mu * np.where(r - (1 - y) > 0, (1 - 2 * y)[:, np.newaxis], np.zeros_like(r))
    grad_non_negative = 2 * mu * np.where(-r < 0, np.ones_like(r) * (2 * -r), np.zeros_like(r))

    grad_boundary = np.concatenate([grad_left, grad_right, grad_bottom, grad_top, grad_non_negative])
    
    return -(grad_overlap + grad_boundary)

def pel_optimization():
    n = 26
    num_runs = 25
    mu = 10
    tolerance = 1e-8
    max_penalty_rounds = 10

    best_obj = -np.inf
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    for run in range(num_runs):
        # Initialize random centers and small radii
        centers = np.random.rand(n, 2) * 0.9 + 0.05
        radii = np.ones(n) * 0.05
        z = np.concatenate((centers.flatten(), radii))
        
        for _ in range(max_penalty_rounds):
            # Run L-BFGS-B with current z and mu
            res = scipy.optimize.minimize(pen_optim, z, args=(mu), jac=grad_pen_optim, bounds=[(-1e-8, None)]*3*n,
                                          options={'maxfun': 400})
            z = res.x
            mu *= 10
            
            # extract coordinates and radii from result
            centers = z[:n].reshape((n, 2))
            radii = z[n:]
            
            # Check constraints
            d_ij = np.sqrt((centers[:, np.newaxis] - centers[np.newaxis, :])**2 + (radii[:, np.newaxis] - radii[np.newaxis, :])**2)
            s_ij = radii[:, np.newaxis] + radii[np.newaxis, :] - d_ij
            violations = np.max(s_ij[s_ij > 0])
            
            if violations < tolerance:
                obj = -np.sum(radii)
                if obj > best_obj:
                    best_obj = obj
                    best_centers = centers
                    best_radii = radii
            
            # Early exit if converged and outsidepenalty range слишком ранний, убираем его
            if violations < 1e-7:
                break
    
    return best_centers, best_radii, best_obj
