import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    # Pairwise distance constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = np.linalg.norm(centers[i] - centers[j]) - 1e-12
            idx += 1

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y - 1e-12
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y - 1e-12
        idx += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def propose_move(centers, step):
    idx = np.random.randint(26)
    dir = np.random.randn(2)
    dir /= np.linalg.norm(dir)
    Δ = dir * step
    return idx, Δ

def generate_random_pattern():
    return np.random.rand(26, 2) * 0.9 + 0.05

def generate_perturbed_grid():
    grid = np.zeros((5, 5, 2))
    grid[:, :, 0] = np.arange(5) + 0.1
    grid[:, :, 1] = np.arange(5) + 0.1
    grid = grid.flatten()
    np.random.shuffle(grid)
    grid = grid[:26].reshape(26, 2)
    return grid[1:]  # Remove one corner

def generate_diagonal_shift():
    grid = np.zeros((5, 5, 2))
    grid[1::2, 1:, 0] = np.arange(4) + 0.1
    grid[1::2, 1:, 1] = np.arange(4) + 0.105
    grid[::2, :-1, 0] = np.arange(4) + 0.1
    grid[::2, :-1, 1] = np.arange(4) + 0.095
    grid = grid.flatten()
    np.random.shuffle(grid)
    grid = grid[:26].reshape(26, 2)
    return grid

def generate_rough_hexagon():
    hexagon = np.array([
        [0.15, 0.13],
        [0.4, 0.26],
        [0.75, 0.26],
        [1, 0.13],
        [0.85, -0.11],
        [0.5, -0.14],
        [0.15, -0.11],
        [0.15, 0.13]
    ])
    indices = np.arange(8)
    np.random.shuffle(indices)
    hexagon = hexagon[indices]
    return hexagon[:26]

def generate_random_plus_boundary():
    corners = np.array([
        [0.05, 0.5],
        [0.05, 0.05],
        [0.5, 0.05],
        [0.5, 0.5],
        [0.95, 0.5],
        [0.5, 0.95],
        [0.05, 0.95]
    ])
    random_points = np.random.rand(20, 2) * 0.9 + 0.05
    combined = np.vstack((random_points, corners))
    combined = combined[:26]
    np.random.shuffle(combined)
    return combined

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    step_start = 0.05
    step_decay = 0.995
    T0 = 0.003 * 2.7
    T_decay = 0.999
    ITER_PER_PATTERN = 2000
    MAX_NO_IMPROV = 500

    patterns = [
        generate_random_pattern,
        generate_perturbed_grid,
        generate_diagonal_shift,
        generate_rough_hexagon,
        generate_random_plus_boundary
    ]

    best_sum = -np.inf
    best_centers, best_radii = None, None

    for start in range(5):
        centers = patterns[start]()
        for it in range(ITER_PER_PATTERN):
            idx, Δ = propose_move(centers, step_start)
            new_c = centers.copy(); new_c[idx] += Δ
            new_c[idx] = np.clip(new_c[idx], 0, 1)
            new_centers = centers.copy(); new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

        step_start = max(step_start * step_decay, 1e-4)

    final_radii, final_sum = lp_optimum(best_centers.copy())
    return best_centers, final_radii, final_sum
