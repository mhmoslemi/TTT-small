import numpy as np
from scipy.optimize import linprog, hpot

def lp_solve(centers):
    n = centers.shape[0]
    B = np.min(np.column_stack((centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])), axis=1)
    A = np.empty((n * n - n, 2 * n))
    b = np.empty(n * n - n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A[idx, 2 * i] = 1
            A[idx, 2 * j] = 1
            b[idx] = dist
            idx += 1
    
    res = linprog(-B, A_ub=A, b_ub=b, bounds=(0, None), method='highs', res_options={'abstol': 1e-12})
    if not res.success:
        return np.zeros(n), -np.inf
    else:
        return -res.x, res.fun
