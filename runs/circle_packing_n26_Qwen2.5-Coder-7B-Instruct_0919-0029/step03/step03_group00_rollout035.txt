import numpy as np
from scipy.optimize import linprog

# Function to generate centers in a more compact形式
def generate_centers(n):
    x = np.linspace(0, 1, int(np.sqrt(n)))
    y = np.linspace(0, 1, int(np.sqrt(n)))
    centers = np.array([[x_i, y_j] for y_j in y for x_i in x])
    centers = centers[:n]
    return centers

# Function to compute distances between centers
def compute_distances(centers):
    n = centers.shape[0]
    distances = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            distances[i, j] = np.linalg.norm(centers[i] - centers[j])
    distances += distances.T
    np.fill_diagonal(distances, np.inf)
    return distances

# Function to compute the constraints matrix and b vector
def compute_constraints(centers, radii):
    n = centers.shape[0]
    A_eq = np.zeros((n, n))
    for i in range(n):
        A_eq[i, i] = 1
    b_eq = radii

    A_ub = compute_distances(centers) - radii[:, None] - radii[None, :]
    b_ub = 0

    A_bounds = np.vstack((np.eye(n), -np.eye(n)))
    b_bounds = np.hstack((np.ones(n) * 2, -np.zeros(n)))

    return A_eq, b_eq, A_ub, b_ub, A_bounds, b_bounds

# Function to solve the linear programming problem
def lp_solve(centers):
    A_eq, b_eq, A_ub, b_ub, A_bounds, b_bounds = compute_constraints(centers, np.ones(centers.shape[0]))

    c = -np.ones(centers.shape[0])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=A_bounds, options={'disp': False})
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(centers.shape[0]), 0

# Main function to run the packing
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = generate_centers(n)
    radii, sum_radii = lp_solve(centers)
    
    return centers, radii, sum_radii
