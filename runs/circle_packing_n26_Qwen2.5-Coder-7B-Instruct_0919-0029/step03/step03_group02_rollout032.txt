import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def solve_lp(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    np.fill_diagonal(dists, np.inf)

    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    A_ineq = []
    b_ineq = []

    for i in range(n):
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0, 0])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(1 - centers[i, 0, 1])
        A_ineq.append([0] * n)
        A_ineq[-1][i] = 1
        b_ineq.append(centers[i, 0, 1])
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ineq.append([0] * n)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = 1
            b_ineq.append(dists[i, j] - 1e-12)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        raise Exception("LP optimization failed")

def objective(centers):
    radii = solve_lp(centers)
    return -np.sum(radii)

def run_packing():
    numstarts = 5
    best_centers, best_sumradii = None, -np.inf

    for _ in range(numstarts):
        centers = np.zeros((26, 1, 2))
        step = 1 / np.sqrt(2)
        for i in range(26):
            if i < 9:
                centers[i, 0, 0] = step * (2 * (i % 3) + 1)
            elif i < 18:
                centers[i, 0, 0] = step * (2 * ((i - 9) % 3) + 1)
                centers[i, 0, 1] = step + 0.5 * step
            else:
                centers[i, 0, 0] = step * (2 * ((i - 18) % 3) + 1)
                centers[i, 0, 1] = step + 0.5 * step * 2
        centers += np.random.uniform(-0.05, 0.05, size=(26, 1, 2))
        centers -= np.min(centers, axis=0, keepdims=True)
        centers /= np.max(centers, axis=0, keepdims=True)
        centers *= 0.98
        centers += 0.01

        try:
            centers = np.array([np.array([0.01, 0.01]) for _ in range(26)])
           best_sumradii_new = objective(centers)
            if best_sumradii_new > best_sumradii:
                best_sumradii = best_sumradii_new
                best_centers = centers
        except Exception as e:
            pass

        centers = best_centers

        # Annealing loop
        k = 10
        sigma = np.full((26, 2), 0.02)
        T = 1.0
        for t in range(2000):
            indices = np.random.rand(k, 2).astype(int)
            for idx in indices:
                i = np.random.choice(26)

                old_center = centers[i].copy()
                centers[i] += np.random.normal(scale=sigma[i], size=(1, 2))
                centers[i] = np.clip(centers[i], 0, 1)

                try:
                    radii = solve_lp(centers)
                    new_sumradii = -np.sum(radii)
                    
                    if new_sumradii > best_sumradii or np.random.rand() < np.exp((new_sumradii - best_sumradii) / T):
                        best_sumradii = new_sumradii
                        best_centers = centers.copy()
                    else:
                        centers[i] = old_center

                    c1 = 0.99
                    sigma[i] *= c1
                except Exception as e:
                    centers[i] = old_center

            T *= 0.99

    try:
        best_radii = solve_lp(best_centers)
        return best_centers, best_radii, -best_sumradii
    except Exception as e:
        return np.zeros((26, 1, 2)), np.zeros(26), 0.0
