import numpy as np
from scipy.optimize import linprog

def hex_grid_positions(n):
    assert n <= 30, "The maximum number of centers is 30 for hexagonal grid layout"
    h = np.sqrt(3) / 2
    row_lengths = [4 + i % 2 for i in range(n)]
    x_spacings = [(np.arange(length) + i % 2) * 0.2 for i, length in enumerate(row_lengths)]
    y_spacings = [i * h for i in range(sum(row_lengths))]
    x_coords = np.concatenate(x_spacings)
    y_coords = np.concatenate(y_spacings)
    return x_coords, y_coords

def compute_max_radii(centers):
    n = centers.shape[0]
    r_bounds = np.min(np.vstack((centers[:, 0], centers[:, 0] + 1 - centers[:, 0], centers[:, 1], centers[:, 1] + 1 - centers[:, 1])), axis=0)
    r_bounds = np.minimum(r_bounds, centers[:, 0] + centers[:, 1] - 0.01)
    objective = -np.ones(n)
    A = np.hstack((np.eye(n), -np.eye(n), -np.eye(n)))
    b = np.zeros(3 * n)
    c, A, b, x0 = objective, A, b, centers.flatten()
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None))
    radii = np.abs(result.x)
    sum_radii = np.sum(radii)
    return radii, sum_radii

def hex_grid_packing():
    n = 26
    centers = hex_grid_positions(n)
    initial_radii = np.full(n, 0.01)
    radii, sum_radii = compute_max_radii(np.vstack((centers[0].reshape(-1, 1), centers[1].reshape(-1, 1), initial_radii.reshape(-1, 1))).T)
    return centers, radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = hex_grid_packing()
    return centers, radii, sum_radii
