def generate_centers(n: int) -> np.ndarray:
    """Generate a hexagonal grid of centers."""
    radius = 0.1
    side_length = 2 * radius
    vertical_distance = radius * 3**0.5
    horizontal_distance = side_length
    centers = np.array([])
    for row in range(n):
        offsetX = radius if row % 2 else 0
        for col in range(n):
            x = offsetX + col * horizontal_distance
            y = row * vertical_distance + radius * 0.75
            if np.linalg.norm([x - 0.5, y - 0.5]) < 0.5:
                centers = np.append(centers, [[x, y]], axis=0)
        if len(centers) >= n:
            break
    return centers[:n]

def lp_optimum(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """Solve the linear programming problem to find the optimal radii."""
    n = centers.shape[0]
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    for i in range(n):
        A_eq[i, i] = 1
        b_eq[i] = 1

    A_ub = np.zeros((n*(n-1)//2 + 4*n, n))
    b_ub = np.zeros(A_ub.shape[0])

    for k in range(n*(n-1)//2):
        i, j = np.unravel_index(k, (n, n-1))
        A_ub[k, i] = 1
        A_ub[k, j] = 1
        d_ij = np.linalg.norm(centers[i] - centers[j])
        b_ub[k] = d_ij

    for i in range(n):
        x, y = centers[i]
        A_ub[n*(n-1)//2 + 4*i, i] = 1
        b_ub[n*(n-1)//2 + 4*i] = 1 - x
        A_ub[n*(n-1)//2 + 4*i + 1, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 1] = x
        A_ub[n*(n-1)//2 + 4*i + 2, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 2] = 1 - y
        A_ub[n*(n-1)//2 + 4*i + 3, i] = 1
        b_ub[n*(n-1)//2 + 4*i + 3] = y

    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Run the packing algorithm to find the best configuration."""
    n = 26
    centers = generate_centers(n)
    best_sum = -np.inf
    best_centers, best_radii = None, None
    
    for _ in range(30):
        for _ in range(1000):
            idx = np.random.randint(n)
            dir = np.random.randn(2)
            dir /= np.linalg.norm(dir)
            step = 0.05
            new_c = centers[idx] + dir * step
            new_c = np.clip(new_c, 0, 1)
            new_centers = centers.copy()
            new_centers[idx] = new_c

            new_radii, new_sum = lp_optimum(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii

    return best_centers, best_radii, best_sum
