import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def solve_radii(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    B = np.min(centers, axis=1) - np.max(centers, axis=1) / 2
    B = np.min([1 - centers[:, 0], 1 - centers[:, 1], centers[:, 0], centers[:, 1]], axis=0)

    c = -np.ones(n)
    A = np.zeros((n + n * (n - 1) // 2, n))
    A[:n, np.arange(n)] = 1
    for i in range(n):
        for j in range(i + 1, n):
            A[n + i * (n - 1) // 2 + j - i - 1, [i, j]] = 1
    b = np.append(B, dists - np.eye(n)).flatten()
    b[b < 0] = 0
    b = b.astype(np.double)

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')

    if not res.success:
        return None, float('-inf')

    radii = res.x
    return radii, np.sum(radii)

def update_radii(centers, radii):
    n = centers.shape[0]
    c = -np.ones(n)
    A = np.zeros((2 * n + n * (n - 1) // 2, n))
    A[:n, np.arange(n)] = 1
    A[n:n * 2, np.arange(n)] = -1
    for i in range(n):
        for j in range(i + 1, n):
            A[2 * n + i * (n - 1) // 2 + j - i - 1, [i, j]] = 1
    b = np.append(np.append(np.zeros(n), -radii), dists - np.eye(n)).flatten()
    b[b < 0] = 0

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')

    if not res.success:
        return None

    radii_new = res.x
    return radii_new, np.sum(radii_new)

def run_packing():
    np.random.seed(42)
    n = 26
    s = 0.099
    centers = np.array([
        [s + (j + (i % 2)) * 2 * s, s + i * np.sqrt(3) * s]
        for i in range(5) for j in range(5 if i % 2 == 0 else 4)
    ])
    centers = np.vstack((centers,
                          [0.5, 0.05],
                          [0.95, 0.5],
                          [0.05, 0.5]))

    T = 0.1
    alpha = 0.9999
    epsilon = 1e-4
    delta = 1e-4

    best_centers = centers.copy()
    best_radii = np.array([B[i] / 2 for i in range(n)])
    best_sum = np.sum(best_radii)
    while T > 1e-6:
        for _ in range(int(1e5)):
            i = np.random.randint(n)
            x_new = np.clip(centers[i, 0] + np.random.normal(0, T / 2), 0, 1)
            y_new = np.clip(centers[i, 1] + np.random.normal(0, T / 2), 0, 1)
            if np.allclose(np.linalg.norm(centers[i] - centers), np.linalg.norm([x_new, y_new] - centers), atol=epsilon):
                continue
            centers[i, 0] = x_new
            centers[i, 1] = y_new

            new_radii, new_sum = update_radii(centers, best_radii)
            if new_radii is not None:
                new_sum_excluding_i = best_sum - best_radii[i] + new_radii[i]
                delta_sum = new_sum_excluding_i - best_sum
                acceptance_prob = min(1, np.exp(delta_sum / T))
                if delta_sum > 0 or np.random.rand() < acceptance_prob:
                    best_centers = centers.copy()
                    best_radii = new_radii.copy()
                    best_sum = new_sum_excluding_i

        T *= alpha

    for _ in range(10):
        i = np.random.randint(n)
        direction = np.argmax([
            solve_radii(np.vstack((best_centers[:i, :], best_centers[i+1:, :], centers[i, :].reshape(1, -1)))[0],
            solve_radii(np.vstack((best_centers[:i, :], best_centers[i+1:, :], centers[i, :].reshape(1, -1) + [delta, 0]))[0],
            solve_radii(np.vstack((best_centers[:i, :], best_centers[i+1:, :], centers[i, :].reshape(1, -1) + [-delta, 0]))[0],
            solve_radii(np.vstack((best_centers[:i, :], best_centers[i+1:, :], centers[i, :].reshape(1, -1) + [0, delta]))[0],
            solve_radii(np.vstack((best_centers[:i, :], best_centers[i+1:, :], centers[i, :].reshape(1, -1) + [0, -delta]))[0]
        ])
        if direction != 0:
            centers[i, :] += np.array([delta, 0]) if direction == 1 else [-delta, 0] if direction == 2 else [0, delta] if direction == 3 else [0, -delta]

    new_radii, new_sum = update_radii(centers, best_radii)
    if new_radii is not None:
        best_sum = new_sum

    return best_centers, new_radii, best_sum

centers, radii, sum_radii = run_packing()
print(centers, radii, sum_radii)
