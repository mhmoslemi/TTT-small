import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    B = np.exp(-100 * np.sum(centers ** 2, axis=1))
    distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=2)
    A_in = np.eye(n) - np.eye(n, k=1) * (distances < B[:, np.newaxis] + B) * (distances < B[:, np.newaxis] + B[:, np.newaxis]) * (B[:, np.newaxis] - distances) * (B - distances[:, np.newaxis]) * np.log((B[:, np.newaxis] - distances) * (B - distances[:, np.newaxis])) * np.log((B[:, np.newaxis] + B) * (B + B[:, np.newaxis]))
    A_eq = np.zeros((0, n))
    b_eq = []
    A_ub = np.concatenate([np.eye(n), -np.eye(n)], axis=0)
    b_ub = np.concatenate([B, -B])
    result = linprog(np.ones(n), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, A_in=A_in, b_in=np.zeros((distances < B[:, np.newaxis] + B) * (distances < B[:, np.newaxis] + B[:, np.newaxis]) * (B[:, np.newaxis] - distances) * (B - distances[:, np.newaxis])), method='highs', tol=1e-12)
    radii = result.x
    if not result.success:
        return np.zeros(n), -np.inf
    return radii, result.fun

def repair_move(x, y, centers, radii, sigma=0.001):
    proposed = x + np.random.normal(0, sigma), y + np.random.normal(0, sigma)
    proposed = np.clip(proposed, 0, 1)
    for cx, cy in centers:
        distance = np.linalg.norm(proposed - (cx, cy))
        if distance < radii[cx, cy] + radii[cy, cx]:
            proposed = (proposed[0] + cx) / 2, (proposed[1] + cy) / 2
            break
    return proposed

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_seed = np.zeros((n, 2))

    for i in range(5):
        initial_seed[i * 5:(i * 5) + 5, 0] = (i * 2 * 0.099 + 0.099 / 2) * np.ones(5)
        initial_seed[i * 5:(i * 5) + 5, 1] = 0.099 * np.arange(5) + 0.099 * np.sqrt(3) * i

    initial_seed[26, 0] = 0.5
    initial_seed[26, 1] = 0.05
    initial_seed[27, 0] = 0.95
    initial_seed[27, 1] = 0.5
    initial_seed[28, 0] = 0.05
    initial_seed[28, 1] = 0.5

    centers = initial_seed.copy()
    best_centers = centers
    best_radii = np.zeros(n)
    best_sum = -np.inf
    T = 0.1
    alpha = 0.9999

    while T > 1e-6:
        new_centers = centers.copy()

        for i in range(n):
            x, y = centers[i]
            dx, dy = repair_move(x, y, centers, best_radii)
            new_centers[i, 0] = dx
            new_centers[i, 1] = dy

            new_radii, new_sum = solve_radii(new_centers)
            if new_sum > best_sum:
                centers = new_centers
                best_centers = new_centers
                best_radii = new_radii
                best_sum = new_sum
            elif np.random.rand() < np.exp((new_sum - best_sum) / T):
                centers = new_centers

        T *= alpha

    return best_centers, best_radii, best_sum
