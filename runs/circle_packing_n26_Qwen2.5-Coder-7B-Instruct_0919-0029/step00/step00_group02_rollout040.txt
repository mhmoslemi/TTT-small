import numpy as np
from ortools.linear_solver import pywraplp Solver

def build_milp_grid(k: int) -> tuple:
    Delta = 1 / k
    grid_points = [(i * Delta, j * Delta) for i in range(k) for j in range(k)]
    R_max = np.min([[g if i % 2 == 0 else g[::-1] for i in range(2 * k)], axis=0))

    solver = Solver.CreateSolver('CBC')
    b_vars = {g: solver.IntVar(0, 1, f'b{x},{y}') for g, (x, y) in enumerate(grid_points)}
    r_vars = {g: solver.NumVar(0, R_max[g], f'r{x},{y}') for g, (x, y) in enumerate(grid_points)}

    solver.Add(solver.Sum([r_vars[g] for g in b_vars]) == 26)

    for g, (x, y) in enumerate(grid_points):
        solver.Add(r_vars[g] <= b_vars[g] * R_max[g])

    pairs = []
    for i, g1 in enumerate(grid_points):
        for g2 in grid_points[i+1:]:
            dist_squared = (g1[0] - g2[0]) ** 2 + (g1[1] - g2[1]) ** 2
            d_max_squared = (R_max[g1]**2 + R_max[g2]**2) ** 2
            if dist_squared <= d_max_squared + 1e-12:  # Include slight overlaps
                pairs.append((g1, g2, dist_squared, d_max_squared))

    for g1, g2, dist_squared, d_max_squared in pairs:
        M = dist_squared if dist_squared > d_max_squared + 1e-12 else 2.0
        solver.Add(r_vars[g1] + r_vars[g2] <= np.sqrt(dist_squared) + M * (2 - b_vars[g1] - b_vars[g2]))

    return solver, {g: (x, y) for g, (x, y) in enumerate(grid_points)}

def solve_milp(solver: Solver, time_limit: int) -> tuple:
    solver.SetTimeLimit(time_limit * 1000)
    status = solver.Solve()
    best_solution = dict()
    return status, best_solution if status == Solver.OPTIMAL else best_solution

def extract_solution(best_solution: dict, vars_by_grid: dict) -> tuple:
    centers = np.array([np.array(vars_by_grid[g]) for g in best_solution])
    radii = np.array([best_solution[g] for g in best_solution])
    return centers, radii

def polish_solution(centers: np.ndarray, radii: np.ndarray) -> tuple:
    delta_r = 1e-4
    epsilon = 1e-4
    min_radius = 0
    max_radius = np.min(np.min(centers[:, 0] - radii), np.min(centers[:, 1] - radii),
                        np.min(1 - centers[:, 0] - radii), np.min(1 - centers[:, 1] - radii))

    for i in range(len(radii)):
        while True:
            new_radius = min(max_radius, radii[i] + delta_r)
            valid = True
            for j in range(len(radii)):
                if i != j:
                    new_dist_squared = (centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2
                    new_max_radius_sum = new_radius + radii[j]
                    if new_dist_squared <= new_max_radius_sum**2:
                        valid = False
                        break
            if valid:
                radii[i] = new_radius
                break

    for _ in range(3):
        for i in range(len(radii)):
            for j in range(8):  # 4 possible shifts
                new_center = centers[i, :] + epsilon * np.array([[-1, 1][j % 4 < 2], [-1, 1][j % 4 < 2]])
                new_radius = min(max_radius, radii[i])
                valid = all(in_bounds(new_center, new_radius))
                for k in range(len(radii)):
                    if i != k and valid:
                        new_dist_squared = (new_center[0] - centers[k, 0])**2 + (new_center[1] - centers[k, 1])**2
                        new_max_radius_sum = new_radius + radii[k]
                        if new_dist_squared <= new_max_radius_sum**2:
                            valid = False
                            break
                if valid:
                    centers[i, :] = new_center
                    break

    return in_bounds(centers, radii)

def in_bounds(centers: np.ndarray, radii: np.ndarray) -> bool:
    return np.all(((centers[:, 0] - radii >= 0) & (centers[:, 0] + radii <= 1)) &
                   ((centers[:, 1] - radii >= 0) & (centers[:, 1] + radii <= 1)))
