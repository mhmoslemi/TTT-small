import numpy as np
from ortools.linear_solver import pywraplp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Construct the MILP on a k×k grid (k=20).
    solver, grid_vars = build_milp_grid(k=20)

    # 2. Solve it with CBC (OR-Tools).
    status, best_sol = solve_milp(solver, time_limit=600)

    # 3. Extract continuous circles from the solution.
    centers, radii = extract_from_sol(best_sol, grid_vars)

    # 4. Polish the continuous packing (greedy grow + shift).
    centers, radii = polish(centers, radii)

    # 5. Final validation - raises if packing is invalid.
    validate_packing(centers, radii)

    return centers, radii, radii.sum()

def build_milp_grid(k: int) -> tuple[pywraplp.Solver, dict]:
    solver = pywraplp.Solver.CreateSolver('SCIP')
    grid_vars = {}

    for i in range(k):
        for j in range(k):
            b_p = solver.BoolVar(f'b_{i}_{j}')
            r_p = solver.NumVar(0, float('inf'), f'r_{i}_{j}')
            grid_vars[(i, j)] = (b_p, r_p)

    return solver, grid_vars

def solve_milp(solver: pywraplp.Solver, time_limit: int) -> tuple[int, dict]:
    k = int(np.sqrt(len(solver.GetVariable('r_1_1').domain())))
    M = {(i, j): 10000 for i in range(k) for j in range(k)}

    for i in range(k):
        for j in range(k):
            b_p, r_p = grid_vars[(i, j)]
            solver.Add(r_p <= b_p * R_max(i * 1.0 / k, j * 1.0 / k))

    for i in range(k):
        for j in range(k):
            if b_p := grid_vars.get((i, j)) is None:
                continue
            for ii in range(max(0, i-1), min(k, i+2)):
                for jj in range(max(0, j-1), min(k, j+2)):
                    if (ii, jj) == (i, j):
                        continue
                    b_q, r_q = grid_vars.get((ii, jj)) or (1e-6, 0)
                    if dist_sq := (ii-i)**2 + (jj-j)**2 >= (R_max(i * 1.0 / k, j * 1.0 / k) + R_max(ii * 1.0 / k, jj * 1.0 / k))**2:
                        continue
                    solver.Add(r_p + r_q <= np.sqrt(dist_sq) + M[(i, j)] * (2 - b_p - b_q))

    solver.SetObjective(solver.Sum(r_p for _, r_p in grid_vars.values()))
    solver.set_time_limit(time_limit * 1000)
    solver.EnableOutput()

    status = solver.Solve()
    if status not in [pywraplp.Solver.OPTIMAL, pywraplp.Solver.FEASIBLE]:
        raise Exception(f'Solver status: {status}')

    return status, {k: v for k, (v, _) in grid_vars.items()}

def extract_from_sol(best_sol: dict, grid_vars: dict) -> tuple[np.ndarray, np.ndarray]:
    k = int(np.sqrt(len(best_sol['r_1_1'].domain())))
    centers = []
    radii = []

    for i in range(k):
        for j in range(k):
            if best_sol[(i, j)].solution_value() > 1e-6:
                r_p = best_sol[(i, j)].solution_value()
                centers.append([i / k, j / k])
                radii.append(r_p)

    return np.array(centers), np.array(radii)

def polish(centers: np.ndarray, radii: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    epsilon = 1e-4
    delta_r = 1e-4

    def grow(i: int):
        while True:
            r_try = radii[i] + delta_r
            if any((centers[i] - c)**2 min_r**2 for c, min_r in zip(centers[:i] + centers[i+1:], radii[:i] + radii[i+1:])):
                break
            radii[i] = max(0, min(r_try, 1 - radii[i]))

    def shift(i: int):
        for dx, dy in [(-epsilon, 0), (epsilon, 0), (0, -epsilon), (0, epsilon), (-epsilon, -epsilon), (-epsilon, epsilon), (epsilon, -epsilon), (epsilon, epsilon)]:
            c_try = centers[i] + [dx, dy]
            if not any((c_try - c)**2 < min_r**2 for c, min_r in zip(centers[:i] + centers[i+1:], radii[:i] + radii[i+1:])):
                centers[i] = c_try

    for _ in range(6):
        for i in np.random.permutation(len(centers)):
            grow(i)
            shift(i)

        if not any(delta for delta in np.diff(radii)):
            break

    return centers, radii

def R_max(x: float, y: float) -> float:
    return min(x, 1 - x, y, 1 - y)

def validate_packing(centers: np.ndarray, radii: np.ndarray):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        raise Exception("NaN values present")

    for i in range(n):
        if radii[i] < 0:
            raise Exception(f"Circle {i} has negative radius {radii[i]}")

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise Exception(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise Exception(f"Circles {i} and {j} overlap")
