import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k):
    Delta = 1 / k
    centers = np.array([(x * Delta, y * Delta) for y in range(k) for x in range(k)])
    R_max = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    R_max = R_max.reshape(-1, 1)
    model = pywraplp.Solver.CreateSolver('SCIP')
    vars_by_grid = {(i, j): (model.NumVar(0, R_max[i], f'x_{i}_{j}'), model.NumVar(0, 1, f'b_{i}_{j}'))
                    for i, j in enumerate(centers)}
    for i, j in vars_by_grid:
        model.Add(vars_by_grid[i, j][1] * R_max[i] >= vars_by_grid[i, j][0])
    model.Add(sum([vars_by_grid[i, j][1] for i, j in vars_by_grid.keys()]) == 26)
    for i, j in vars_by_grid:
        for i1, j1 in vars_by_grid:
            if i != i1 or j != j1:
                d_max = min(R_max[i], R_max[i1])
                M = d_max + 1e-6
                dist = np.sqrt((centers[i, 0] - centers[i1, 0]) ** 2 + (centers[i, 1] - centers[i1, 1]) ** 2)
                model.Add(vars_by_grid[i, j][0] + vars_by_grid[i1, j1][0] <= dist + M * (1 - vars_by_grid[i, j][1] - vars_by_grid[i1, j1][1]))
    objective = model.Objective()
    objective.SetMaximization()
    for i, j in vars_by_grid:
        objective.SetCoefficient(vars_by_grid[i, j][0], 1)
    return model, vars_by_grid

def solve_milp(model, time_limit=600):
    model.SetTimeLimit(time_limit * 1000)
    model.Solve()
    status = model.Solve()
    if status == pywraplp.Solver.OPTIMAL or status == pywraplp.Solver.FEASIBLE:
        return status, np.array([(vars_by_grid[i, j][0].solution_value(), vars_by_grid[i, j][1].solution_value())
                               for i, j in vars_by_grid.keys()])
    else:
        return status, np.array([(0, 0) for i, j in vars_by_grid.keys()])

def extract_solution(best_solution, vars_by_grid):
    centers = []
    radii = []
    for i, j in vars_by_grid:
        if best_solution[i, j][1] > 0.5:
            centers.append(best_solution[i, j][0])
            radii.append(best_solution[i, j][1])
    return np.array(centers), np.array(radii)

def polish_solution(centers, radii):
    Delta = 1 / 20
    for _ in range(10):
        for i, (cx, cy) in enumerate(centers):
            for dr in [0, 0.0001]:
                for dx in [0, 0.0001]:
                    new_cx = max(0, min(1, cx + dx))
                    new_cy = max(0, min(1, cy + dy))
                    new_r = np.min([np.sqrt((new_cx - other_cx) ** 2 + (new_cy - other_cy) ** 2)
                                    for other_cx, other_cy in centers if not (i == centers.index((other_cx, other_cy)) and dr == 0 and dx == 0)]) - Delta
                    new_r = max(0, new_r)
                    new_centers = np.copy(centers)
                    new_centers[i] = (new_cx, new_cy)
                    new_radii = np.copy(radii)
                    new_radii[i] = new_r
                    if validate_packing(np.array(new_centers), new_radii)[0]:
                        centers = new_centers
                        radii = new_radii
    return np.array(centers), np.array(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Build the MILP model (grid, variables, constraints, objective).
    model, vars_by_grid = build_milp_grid(k=20)

    # 2. Solve with CBC via OR-Tools.
    status, best_solution = solve_milp(model, time_limit=600)

    # 3. Extract feasible centers and radii from best_solution.
    centers, radii = extract_solution(best_solution, vars_by_grid)

    # 4. Deterministic polishing (grow + shift).
    centers, radii = polish_solution(centers, radii)

    # 5. Final validation – raise if invalid.
    assert validate_packing(centers, radii)[0], "Invalid packing"

    return centers, radii, radii.sum()
