import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p=2):
    n = 26
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    penalty = -np.sum(radii)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.linalg.norm(centers[i] - centers[j])
            penalty += λ * max(0, (radii[i] + radii[j]) - d_ij) ** p
        for stage in range(4):
            penalty += λ * max(0, radii[i] - (centers[i, stage] if stage < 2 else 1 - centers[i, stage])) ** p
    return penalty

def lp_optimize_centres(centres):
    n = 26
    bounds = [(0, 1) for _ in range(2*n + n)]
    radius_vars = np.arange(2*n, 2*n + n)
    c = np.zeros_like(radius_vars)
    A = np.zeros((n, n))
    for i in range(n):
        A[i, i] = 1
    b = np.ones(n)
    con = {'type': 'ineq', 'fun': lambda z: b - np.dot(A, (z[radius_vars] + centres[radius_vars])).reshape(-1,)}
    res = minimize(c, np.ones(n), bounds=bounds, method='SLSQP', constraints=con)
    radii = res.x
    sum_radii = np.sum(radii)
    return np.column_stack((centres, radii)), sum_radii
