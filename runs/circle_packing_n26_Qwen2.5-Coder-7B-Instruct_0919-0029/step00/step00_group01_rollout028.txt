import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import differential_evolution, linprog

def solve_lps_for_centres(centres):
    n = centres.shape[0]
    side_length = 1.0
    half_side = side_length / 2.0
    max_radius = 0.25
    constraints = []
    
    for i in range(n):
        r_i = centres[i, 2]
        x_i, y_i = centres[i, 0], centres[i, 1]
        constraints.append({'type': 'ineq', 'fun': lambda *args, i=i, r_i=r_i, x_i=x_i, y_i=y_i: r_i - min(x_i, 1 - x_i, y_i, 1 - y_i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            r_i, r_j = centres[i, 2], centres[j, 2]
            x_i, y_i = centres[i, 0], centres[i, 1]
            x_j, y_j = centres[j, 0], centres[j, 1]
            constraints.append({'type': 'ineq', 'fun': lambda *args, i=i, j=j, r_i=r_i, r_j=r_j, x_i=x_i, y_i=y_i, x_j=x_j, y_j=y_j: r_i + r_j - distance_matrix(np.array([[x_i, y_i]]), np.array([[x_j, y_j]]))[0][0]})
    
    c = -np.ones(n)
    A = np.array(constraints).T[:-1]
    b = np.array(constraints).T[-1]
    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, max_radius))
    
    radii = result.x
    valid_radii = np.clip(radii, 0, max_radius)
    
    return valid_radii, np.sum(valid_radii)

def run_de_search(seed):
    n = 26
    population_size = 50
    max_iterations = 2000
    penalty_weight = 10**7
    max_radius = 0.25
    
    bounds_positions = [(0, 1) for _ in range(2 * n)]
    bounds_radii = [(0, max_radius) for _ in range(n)]
    bounds = bounds_positions + bounds_radii
    
    def de_objective(v):
        positions = v[:2 * n].reshape((-1, 2))
        radii = v[2 * n:]
        penalty = 0.0
        
        # Boundary violations
        penalty += penalty_weight * np.sum(np.clip(-positions[:, 0], 0, None)**2)
        penalty += penalty_weight * np.sum(np.clip(positions[:, 0] - 1, 0, None)**2)
        penalty += penalty_weight * np.sum(np.clip(-positions[:, 1], 0, None)**2)
        penalty += penalty_weight * np.sum(np.clip(positions[:, 1] - 1, 0, None)**2)
        
        # Negative radii
        penalty += penalty_weight * np.sum(np.clip(-radii, 0, None)**2)
        
        # Circle overlap
        distance_matrix = np.linalg.norm(positions[:, np.newaxis, :] - positions[np.newaxis, :, :], axis=-1)
        overlap = distance_matrix - radii[:, np.newaxis] - radii[np.newaxis, :]
        penalty += penalty_weight * np.sum(np.clip(overlap, 0, None)**2)
        
        return -(np.sum(radii) + penalty)
    
    result = differential_evolution(de_objective, bounds, maxiter=max_iterations, popsize=population_size, seed=seed, strategy='intermediate')
    
    return result.x[:2 * n].reshape((-1, 2))

def run_packing() -> tuple np.ndarray, np.ndarray, float:
    seeds = [42, 1337]
    best_centers = None
    best_sum_radii = 0
    
    for seed in seeds:
        centers_best = run_de_search(seed)
        radii, sum_radii = solve_lps_for_centres(centers_best)
        
        if sum_radii > best_sum_radii:
            best_centers = centers_best
            best_sum_radii = sum_radii
            
    return best_centers[:26, :2], best_centers[:26, 2:], best_sum_radii

run_packing()
