import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p):
    n = len(z) // 3
    x = z[:2*n].reshape((-1, 2))
    r = z[2*n:]

    dx = x[:, np.newaxis, :] - x[np.newaxis, :, :]
    dy = np.abs(dx[:, :, 1] - dx[:, :, 1])
    d = np.sqrt(dx**2 + dy**2)

    eta_ij = np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - d)
    beta_i = np.maximum(0, r - x[:, 0]) + np.maximum(0, r - (1 - x[:, 0])) + \
             np.maximum(0, r - x[:, 1]) + np.maximum(0, r - (1 - x[:, 1]))

    return -(np.sum(r) + λ * np.sum(eta_ij**p + beta_i**p))

def compute_gradient(z, λ, p):
    n = len(z) // 3
    x = z[:2*n].reshape((-1, 2))
    r = z[2*n:]

    dx = x[:, np.newaxis, :] - x[np.newaxis, :, :]
    dy = np.abs(dx[:, :, 1] - dx[:, :, 1])
    d = np.sqrt(dx**2 + dy**2)

    eta_ij = np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - d)
    beta_i = np.maximum(0, r - x[:, 0]) + np.maximum(0, r - (1 - x[:, 0])) + \
             np.maximum(0, r - x[:, 1]) + np.maximum(0, r - (1 - x[:, 1]))

    grad_eta_ij = np.zeros((n, n, 2))
    grad_eta_ij[:, :, 0] += (dx / d) * (eta_ij > 0)
    grad_eta_ij[:, :, 1] += (dy / d) * (eta_ij > 0)
    grad_eta_ij = grad_eta_ij / 2 * λ * p * eta_ij**(p-1)

    grad_beta_i = np.zeros((n, 2))
    grad_beta_i[:, 0] += 1 * (beta_i[:, 0] > 0) + \
                         -1 * (beta_i[:, 1] > 0)
    grad_beta_i[:, 1] += 1 * (beta_i[:, 2] > 0) + \
                         -1 * (beta_i[:, 3] > 0)

    grad_eta_all = np.zeros(2*n + n)
    grad_eta_all[:2*n] = grad_eta_ij.sum(axis=0).sum(axis=0)
    grad_eta_all[2*n:] = grad_eta_ij.sum(axis=0).sum(axis=1)

    grad_beta_all = np.zeros(2*n + n)
    grad_beta_all[:2*n] = grad_beta_i.sum(axis=1)
    grad_beta_all[2*n:] = -grad_beta_i.sum(axis=1)

    grad_all = grad_eta_all +.grad_beta_all
    grad_all[:2*n] = grad_all[:2*n] - 2 * r / len(z // 3)
    return grad_all

def lp_optimize_centres(centres):
    n = centres.shape[0]
    c = np.ones(2*n)  # Solve for radii
    A = np.zeros((2*n + 2*n + n, 2*n))
    for i in range(n):

        A[i+n, 2*i:2*i+2] = -1

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    start_guesses = [
        np.random.random((2*n,)) * 0.1,  # Uniform random
        np.random uniform((2*n,)) * 0.2,  # Hexagonal grid
        np.array([
            [0.25, 0.25],
            [0.25, 0.75],
            [0.75, 0.25],
            [0.75, 0.75]
        ].flatten()) * 0.1  # Four corner circles
    ]
    results = []
    λ = 5e4
    p = 2
    tol = 1e-6
    for guess in start_guesses:
        bounds = [(0, 1), ] * 3 * n
        res = minimize(compute_penalty, guess, args=(λ, p),
                      method='L-BFGS-B', jac=compute_gradient,
                      bounds=bounds, tol=tol, callback=lambda x: printrefresh(' '.join(map(str, x))))
        if res.fun < tol:
            centres = res.x.reshape((n, 3))[:, :2]
            original_radii = res.x.reshape((n, 3))[:, 2]
            _, optimal_sum = lp_optimize_centres(centres)
            results.append((centres, original_radii, optimal_sum))
    best_result = max(results, key=lambda x: x[2])
    return best_result

# Helper functions
def printrefresh(s):
    print(s, end='\r', flush=True)

def pairs(n):
    return [(i, j) for i in range(n) for j in range(i+1, n)]

# Main packing function
centres, radii, sum_radii = run_packing()
print('Centres:', centres)
print('Radii:', radii)
print('Sum of radii:', sum_radii)
