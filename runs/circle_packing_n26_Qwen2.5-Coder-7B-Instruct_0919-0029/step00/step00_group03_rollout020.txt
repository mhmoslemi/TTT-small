import numpy as np
import scipy.optimize
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def evaluate_lp(centers):
    n = len(centers)
    distances = distance_matrix(centers, centers)
    wall_distances = np.minimum(centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1])
    A = np.vstack((distances - (wall_distances.reshape(-1, 1) + wall_distances), wall_distances[:, np.newaxis])).T
    b = 1e-12 * np.ones(2 * n)
    res = linprog(np.ones(2 * n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    initial_centers = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    center_guess = initial_centers.flatten()
    radius_guess = np.full(len(center_guess) // 2, np.min(initial_centers[:, 0] - 0.5))

    def objective(z):
        return -np.sum(z[len(center_guess):])

    def constraint(z):
        centers = z[:len(center_guess)].reshape(-1, 2)
        radius_sum = np.sum(z[len(center_guess):])
        wall_constraints = np.clip(centers[:, 0] - radius_sum * 1.01, 0, None)
        wall_constraints += np.clip(centers[:, 1] - radius_sum * 1.01, 0, None)
        wall_constraints += np.clip(1 - centers[:, 0] - radius_sum * 1.01, 0, None)
        wall_constraints += np.clip(1 - centers[:, 1] - radius_sum * 1.01, 0, None)
        pairwise_constraints = distance_matrix(centers, centers) - (radius_sum * 1.01 * np.ones_like(distance_matrix(centers, centers)))
        return np.hstack((wall_constraints, pairwise_constraints.flatten())) - radius_sum * 1.01

    bounds = [(0, 1)] * len(center_guess) + [(0, 0.5)] * len(radius_guess)
    constraints = ({'type': 'ineq', 'fun': constraint})
    result = scipy.optimize.minimize(objective, np.hstack((center_guess, radius_guess)), bounds=bounds, constraints=constraints, options={'maxiter': 20000})

    centers = result.x[:len(center_guess)].reshape(-1, 2)
    radii = np.maximum(0, result.x[len(center_guess):])

    total_radius = np.sum(radii)
    validated, message = validate_packing(centers, radii)
    if not validated:
        print(message)

    return centers, radii, total_radius
