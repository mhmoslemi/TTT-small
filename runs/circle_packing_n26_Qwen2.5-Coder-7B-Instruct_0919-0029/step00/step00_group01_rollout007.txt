import numpy as np
from scipy.optimize import minimize

def lp_interior(c, x):
    """ Compute optimal radii for given centers c. """
    n = len(c)
    A = np.zeros((n, n))
    b = np.zeros(n)
    for i in range(n):
        for j in range(i+1, n):
            A[i, j] = A[j, i] = np.sum((c[i] - c[j]) ** 2)
            b[i] += 2 * c[i]
    A = np.vstack([A, np.eye(n), -np.eye(n)])
    b = np.hstack([b, np.ones(n), -np.ones(n)])
    r = minimize(lambda r: -np.sum(r), np.random.rand(n), method='SLSQP', constraints={'type': 'eq', 'fun': lambda r: np.sum(r) - x}, bounds=[(0, min(x, 0.5)) for x in r] + [(-1, 1)] * (n-1), options={'maxiter': 500})
    return r.x

def minimize_sum_radii():
    n = 26
    max_iter = 200
    tol = 1e-5
    np.random.seed(0)
    for s in range(2):
        c = np.random.rand(n, 2)
        for iter in range(max_iter):
            r = lp_interior(c, np.sum(r))
            S = np.sum(r)
            if iter > max_iter / 2 and abs(S - np.sum(r)) < tol:
                break
        if iter < max_iter / 2:
            break
    c = np.random.rand(n, 2)
    c += 0.1 * (np.random.rand(n, 2) - 0.5)  # random jiggle
    r = lp_interior(c, np.sum(r))
    S = np.sum(r)
    for i in range(10):
        c_copy = c.copy()
        for j in range(n):
            r[j] = 1e-3
            r_copy[j] = 1e-3
            c[j] += 0.1 * (np.random.rand(2) - 0.5)
            r_temp = lp_interior(c_copy, np.sum(r_copy))
            if np.sum(r_temp) > S:
                r = r_temp
                c = c_copy
                S = np.sum(r)
    return c, r, S

centers, radii, sum_radii = minimize_sum_radii()
print("Centers:\n", centers)
print("Radii:\n", radii)
print("Sum of Radii:", sum_radii)
