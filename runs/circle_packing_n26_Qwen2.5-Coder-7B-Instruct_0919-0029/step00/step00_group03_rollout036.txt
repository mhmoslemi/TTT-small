from scipy.optimize import minimize, linprog

def distance(c1, c2):
    return np.sqrt(np.sum((np.array(c1) - np.array(c2)) ** 2))

def objective(z):
    return -np.sum(z[52:])

def constraints(z):
    n = 26
    c = z[:52].reshape(n, 2)
    r = z[52:]
    g = []

    for i in range(n):
        g.append(r[i] - c[i, 0])
        g.append(r[i] - 1 + c[i, 0])
        g.append(r[i] - c[i, 1])
        g.append(r[i] - 1 + c[i, 1])

        for j in range(i + 1, n):
            dist = distance(c[i], c[j])
            g.append(dist - (r[i] + r[j]))

    return g

    r = np.zeros(n)

                   min.distance(c[j], c[i]) / 2 for j in range(n) if i != j)
