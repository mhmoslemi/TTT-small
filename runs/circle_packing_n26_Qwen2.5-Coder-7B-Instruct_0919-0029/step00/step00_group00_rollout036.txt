import numpy as np
from scipy.optimize import linprog

def lp_optimum(centers):
    n = centers.shape[0]
    A outletsb = np.zeros((2 * n * (n - 1) // 2 + 4 * n, n))
    b_ub = np.zeros(2 * n * (n - 1) // 2 + 4 * n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A outletsb[idx, i] = 1
            A outletsb[idx, j] = 1
            d_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            b_ub[idx] = d_ij - 1e-12
            idx += 1
    
    A_ub = np.zeros((4 * n, n))
    b_ub[n * (n - 1) // 2 :]
    for i in range(n):
        A_ub[n * (n - 1) // 2 + i * 4, i] = 1
        b_ub[n * (n - 1) // 2 + i * 4] = centers[i, 0] + 1e-12
        A_ub[n * (n - 1) // 2 + i * 4 + 1, i] = -1
        b_ub[n * (n - 1) // 2 + i * 4 + 1] = 1 - centers[i, 0] - 1e-12
        A_ub[n * (n - 1) // 2 + i * 4 + 2, i] = 1
        b_ub[n * (n - 1) // 2 + i * 4 + 2] = centers[i, 1] + 1e-12
        A_ub[n * (n - 1) // 2 + i * 4 + 3, i] = -1
        b_ub[n * (n - 1) // 2 + i * 4 + 3] = 1 - centers[i, 1] - 1e-12
    
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x, -res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centres = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for start in range(30):
        centres = np.random.rand(n, 2)
        for it in range(2000):
            idx = np.random.randint(n)
            delta = np.random.uniform(-0.05, 0.05, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c
            
            new_radii, new_sum = lp_optimum(new_centres)
            
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                best_radii = new_radii
    
    final_radii, _ = lp_optimum(best_centres)
    return best_centres, final_radii, np.sum(best_radii)
