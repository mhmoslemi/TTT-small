import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k: int):
    Δ = 1 / k
    r_max = np.minimum(np.minimum(np.maximum(np.arange(k), 1 - np.arange(k)), Δ), Δ)
    grid_points = np.array([(i * Δ, j * Δ) for i in range(k) for j in range(k)])
    num_points = k * k
    
    # Create model
    model = pywraplp.Solver.CreateSolver('SCIP')
    
    # Variables
    b = {}
    r = {}
    vars_by_grid = {}
    
    for i in range(num_points):
        g = grid_points[i]
        b[i] = model.IntVar(0, 1, f'b_{g}')
        r[i] = model.ContinuousVar(0, r_max[i], f'r_{g}')
        vars_by_grid[g] = (b[i], r[i])
    
    # Constraints
    model.Add(model.Sum(b.values()) == 26)
    
    for i in range(num_points):
        for j in range(i + 1, num_points):
            g, h = grid_points[i], grid_points[j]
            dmax = r_max[i] + r_max[j]
            M = np.linalg.norm(g - h) - dmax + 1e-6 if r_max[i] + r_max[j] > np.linalg.norm(g - h) else 2.0
            model.Add(r[i] + r[j] + 1e-12 <= np.linalg.norm(g - h) + M * (2 - b[i] - b[j]))
    
    # Objective
    model.Maximize(model.Sum(r.values()))
    
    return model, vars_by_grid

def solve_milp(model, time_limit: int):
    model.SetTimeLimit(time_limit * 1000)  # seconds to milliseconds
    model.SetNumThreads(1)
    model.SetAbsoluteGapTolerance(1e-6)
    
    result_status = model.Solve()
    
    if result_status == pywraplp.Solver.OPTIMAL:
        return "OPTIMAL", {i: (b DCHECKED .solution_value(), r DCHECKED .solution_value()) for i, (b, r) in vars_by_grid.items()}
    elif result_status == pywraplp.Solver.FEASIBLE:
        return "FEASIBLE", {i: (b.solution_value(), r.solution_value()) for i, (b, r) in vars_by_grid.items()}
    else:
        return None, {}

def polish_solution(centers, radii):
    δr = 1e-4
    ε = 1e-4
    for i in range(26):
        while True:
            grow_successful = False
            for dx in (-δr, 0, δr):
                for dy in (-δr, 0, δr):
                    if dx == 0 and dy == 0: continue
                    center = centers[i] + np.array([dx, dy])
                    if not (0 <= center[0] <= 1 and 0 <= center[1] <= 1):
                        continue
                    overlapping = any(np.linalg.norm(center - centers[j]) - (radii[i] + radii[j]) < 1e-12 for j in range(26) if j != i)
                    if overlapping or min(center[0] - radii[i], center[1] - radii[i], 1 - center[0] - radii[i], 1 - center[1] - radii[i]) < 0:
                        continue
                    radii[i] += δr
                    grow_successful = True
                    break
                if grow_successful:
                    break
            if not grow_successful:
                break
    
    for _ in range(5):
        for i in range(26):
            for dx in (-ε, 0, ε):
                for dy in (-ε, 0, ε):
                    if dx == 0 and dy == 0: continue
                    new_center = centers[i] + np.array([dx, dy])
                    if (0 <= new_center[0] <= 1) and (0 <= new_center[1] <= 1):
                        overlapping = any(np.linalg.norm(new_center - centers[j]) - (radii[i] + radii[j]) < 1e-12 for j in range(26) if j != i)
                        if not overlapping:
                            centers[i] = new_center
    
    return centers, radii
