import numpy as np
from scipy.optimize import minimize

def generate_initial_guess():
    centers = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    centers = np.vstack((centers, [(0.5, 0.5)]))
    radii = np.full(26, 0.09)
    return np.vstack((centers, radii))

def make_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+1] - x[3*i+2]})
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt((x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2) - x[3*i+2] - x[3*j+2]})
    return cons

def objective(x):
    return -np.sum(x[2::3])

def grow_step(x, cons):
    n = len(x) // 3
    for i in range(n):

def feasible(x, cons):

        if c['fun'](x) < -1e-12:
            return False
    return True

def optimize_with_initial_guess(x0, cons):
    bounds = [(0,1)]*2*26 + [(0,0.5)]*26
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol':1e-9, 'maxiter':10000, 'disp':False})
    return res

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    k = 0
    best_result = None
    while k < 6:
        x0 = generate_initial_guess()[:, :3].flatten()
        radii = np.random.rand(26) * 0.14 + 0.01
        x0[2::3] = radii
        cons = make_constraints(n)
        res = optimize_with_initial_guess(x0, cons)
        if best_result is None or res.fun < best_result.fun:
            best_result = res
        k += 1
    centers = best_result.x[:n*3].reshape(n, 3)[:, :2]
    radii = best_result.x[:n*3].reshape(n, 3)[:, 2]
    if not feasible(best_result.x, cons):
