import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.maximum(distance_matrix(centers, centers) - np.eye(n), 0)
    
    A_ub = np.vstack((np.diag(-borders), -distances[np.triu_indices(n, k=1)]))
    b_ub = np.hstack((-borders, -distances[np.triu_indices(n, k=1)]))
    
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    radii = np.maximum(result.x, 0)
    sum_radii = -result.fun
    
    return radii, sum_radii

def random_move(centers, step):
    rng = np.random.default_rng()
    i = rng.integers(0, centers.shape[0])
    new_pos = centers[i] + rng.normal(scale=step, size=2)
    new_pos = np.clip(new_pos, 0, 1)
    centers[i] = new_pos
    return centers

def simulated_annealing_centers(max_iter):
    step = 0.05
    T0 = 0.02
    T_end = 1e-5
    rng = np.random.default_rng(42)
    
    best_sum = 0
    best_centers = None
    
    for iteration in range(max_iter):
        centers = random_move(np.random.rand(26, 2), step)
        radii, sum_radii = solve_lp(centers)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers.copy()
        
        if iteration % 5000 == 0:
            step *= 0.9995
        
        if T0 > T_end:
            T0 *= 0.9995
    
    return best_centers, best_radii, best_sum

def run_packing():
    centers, radii, sum_radii = simulated_annealing_centers(30000)
    return centers, radii, sum_radii
