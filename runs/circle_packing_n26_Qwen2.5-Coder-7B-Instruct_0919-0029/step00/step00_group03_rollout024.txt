import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    g = 1e-6
    delta = 5e-7
    max_iter = 200000
    shake_rounds = 20
    eps = 1e-9

    # Seed the configuration
    centers = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    centers[25] = (0.5, 0.5)
    radii = np.zeros((n,))

    # Growth loop
    for _ in range(max_iter):
        # Proposed growth
        radii_new = radii + g
        
        # Constraint checking and conflict resolution
        conflicts = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if radii_new[i] + radii_new[j] > dist:
                    conflicts.append((i, j))
            if not (radii_new[i] > centers[i, 0] and radii_new[i] > 1 - centers[i, 0]
                    and radii_new[i] > centers[i, 1] and radii_new[i] > 1 - centers[i, 1]):
                conflicts.append((i,))
        
        for i, j in conflicts:
            if isinstance(j, tuple):
                closest_neigh = min(range(n), key=lambda k: (k != i) and (-np.sum((centers[i] - centers[k]) ** 2)))
                move_vector = (centers[i] - centers[closest_neigh]) / np.linalg.norm(centers[i] - centers[closest_neigh]) * delta
                centers[i] += move_vector
            else:
                wall_directions = [np.array([1, 0]), np.array([-1, 0]), np.array([0, 1]), np.array([0, -1])]
                move_vector = np.array([-1, -1]) * delta
                closest_wall = min(wall_directions, key=lambda w: np.abs(np.dot(centers[i] - (w >= 0), w)))
                if closest_wall[0] < 0:
                    move_vector[0] /= closest_wall[0]
                elif closest_wall[0] > 0:
                    move_vector[0] /= closest_wall[0]
                elif closest_wall[1] < 0:
                    move_vector[1] /= closest_wall[1]
                elif closest_wall[1] > 0:
                    move_vector[1] /= closest_wall[1]
                centers[i] += move_vector
            radii[i] = min(np.min(np.abs(centers[i] - centers[j]) - radii[j] for j in range(n) if j != i), centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

        # Check for convergence
        sum_radii = np.sum(radii)
        if _ > 0 and abs(sum_radii - prev_sum_radii) < 1e-8:
            break
        prev_sum_radii = sum_radii

        # Optional shake period
        if (_ + 1) % shake_rounds == 0:
            k = np.random.randint(n)
            epsilon = np.random.uniform(-0.02, 0.02, 2)
            epsilon = np.clip(epsilon, 0, 1)
            centers[k] += epsilon
            radii[k] = min(np.min(np.abs(centers[k] - centers[j]) - radii[j] for j in range(n) if j != k), centers[k, 0], 1 - centers[k, 0], centers[k, 1], 1 - centers[k, 1])

    # Final validity check
    centers = np.clip(centers, 0, 1)
    radii = np.maximum(np.maximum(np.maximum(np.maximum(0, radii - eps), radii * 0.9999999), np.maximum(0, radii - eps)), np.abs(radii))

    return centers, radii, np.sum(radii)
