import numpy as np
import scipy.optimize as opt
import scipy.spatial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial 5x5 grid with one extra center at the center
    centers_initial = np.array([
        [0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)
    ])
    centers_initial = np.vstack((centers_initial, [0.5, 0.5]))
    radii_initial = np.ones(26) * 0.1

    # Flatten the initial guess
    z_initial = np.hstack((centers_initial.flatten(), radii_initial))

    # Define the objective function
    def objective(z):
        radii = z[52:]
        return -np.sum(radii)  # We want to maximize

    # Define the constraints
    def constraint_rad(i):
        x_i, y_i, _ = z[2 * i:2 * i + 3]
        return np.array([x_i, 1 - x_i, y_i, 1 - y_i]) - z[52 + i]

    def constraint_dist(i, j):
        x_i, y_i, _ = z[2 * i:2 * i + 3]
        x_j, y_j, _ = z[2 * j:2 * j + 3]
        return np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2) - (z[52 + i] + z[52 + j])

    cons = [{'type': 'ineq', 'fun': constraint_rad, 'args': (i,)} for i in range(26)]
    for i in range(26):
        for j in range(i + 1, 26):
            cons.append({'type': 'ineq', 'fun': constraint_dist, 'args': (i, j)})

    # Perform the optimization
    options = {'maxiter': 2000}
    res = opt.minimize(objective, z_initial, method='SLSQP', bounds=None,
                      constraints=cons, options=options)

    # Get the optimized parameters
    centers_opt = res.x[:52].reshape((26, 2))
    radii_opt = res.x[52:]

    # Convert to actual radii
    distances = scipy.spatial.distance_matrix(centers_opt, centers_opt)
    radii_opt = np.minimum(radii_opt, distances / 2 - 1e-10)
    radii_opt = np.maximum(radii_opt, np.minimum(centers_opt[:, 0], 1 - centers_opt[:, 0]))
