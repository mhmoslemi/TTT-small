import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    solver = pywraplp.Solver.CreateSolver('CBC')
    grid_vars = {}

    n = k * k
    x_vals = np.linspace(0, 1, k)
    y_vals = np.linspace(0, 1, k)

    for i in range(n):
        x, y = x_vals[i % k], y_vals[i // k]
        r_max = min(x, 1 - x, y, 1 - y)
        grid_vars[i] = (solver.NumVar(0, 1, f'b_{i}'), solver.NumVar(0, r_max, f'r_{i}'))

    return solver, grid_vars

def solve_milp(solver, time_limit=600):
    k = 20
    grid_vars = {}
    n = k * k
    x_vals = np.linspace(0, 1, k)
    y_vals = np.linspace(0, 1, k)

    b_vars = [grid_VARS[i][0] for i in range(n)]
    r_vars = [grid_vars[i][1] for i in range(n)]

    solver.SetTimeLimit(time_limit * 1000)

    for i in range(n):
        solver.Add(r_vars[i] <= b_vars[i] * min(x_vals[i % k], 1 - x_vals[i % k], y_vals[i // k], 1 - y_vals[i // k]))

    solver.Add(sum(r_vars) == 26)

    for i in range(n):
        for j in range(i + 1, n):
            d_max = min(x_vals[i % k], 1 - x_vals[i % k], y_vals[i // k], 1 - y_vals[i // k]) + min(x_vals[j % k], 1 - x_vals[j % k], y_vals[j // k], 1 - y_vals[j // k])
            if np.sqrt((x_vals[i % k] - x_vals[j % k]) ** 2 + (y_vals[i // k] - y_vals[j // k]) ** 2) > d_max - 1e-6 + 1e-6:
                continue
            solver.Add(r_vars[i] + r_vars[j] <= np.sqrt((x_vals[i % k] - x_vals[j % k]) ** 2 + (y_vals[i // k] - y_vals[j // k]) ** 2))

    objective = solver.Objective()
    objective maximize sum(r_vars)

    status = solver.Solve()

    return status, [(b_vars[i].solution_value(), r_vars[i].solution_value()) for i in range(n)]

def extract_from_sol(best_sol, grid_vars):
    k = 20
    centers = []
    radii = []

    for i in range(k * k):
        if best_sol[i][0] > 0.5:
            x, y = i % k / k, i // k / k
            r = best_sol[i][1]
            centers.append([x, y])
            radii.append(r)

    return np.array(centers), np.array(radii)

def polish(centers, radii):
    k = 0.0001
    for _ in range(5):
        for i in range(len(centers)):
            for dx, dy in [(-k, 0), (k, 0), (0, -k), (0, k), (-k * np.sqrt(2), -k * np.sqrt(2)), (-k * np.sqrt(2), k * np.sqrt(2)), (k * np.sqrt(2), -k * np.sqrt(2)), (k * np.sqrt(2), k * np.sqrt(2))]:
                new_x, new_y = centers[i][0] + dx, centers[i][1] + dy
                if 0 <= new_x <= 1 - radii[i] and 0 <= new_y <= 1 - radii[i]:
                    overlap = False
                    for j in range(len(centers)):
                        if j != i and 0 < np.sqrt((new_x - centers[j][0]) ** 2 + (new_y - centers[j][1]) ** 2) < radii[i] + radii[j] - 1e-6:
                            overlap = True
                            break
                    if not overlap:
                        centers[i] = [new_x, new_y]
                        break

    for i in range(len(centers)):
        while True:
            r_try = radii[i] + k
            overlap = False
            for j in range(len(centers)):
                if j != i and 0 < np.sqrt((centers[i][0] - centers[j][0]) ** 2 + (centers[i][1] - centers[j][1]) ** 2) < r_try + radii[j] - 1e-6:
                    overlap = True
                    break
            if overlap:
                break
            radii[i] = r_try

    return np.array(centers), np.array(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
