import numpy as np
from scipy.optimize import minimize, linprog, NonlinearConstraint, Bounds
from scipy.spatial import distance_matrix

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial configuration
    grid = np.array([(0.1 + 0.2*i, 0.1 + 0.2*j) for i in range(5) for j in range(5)])
    center_guess = np.tile(grid, (2, 1)).reshape(-1, 2)
    center_guess = np.hstack((center_guess, np.ones(26) * 0.25))
    initial_guess = center_guess.flatten()

    # Define constraints
    def WallConstraint(center):
        return np.clip(center[:26] - center[26:], 0, 1) - center[26:]

    def NonOverlappConstraint(center):
        dist = np.maximum(distance_matrix(center[:26], center[:26]), 1e-9)
        diag = np.eye(26) * center[26:]
        return (dist - diag) - center[26:]

    wall_cons = [NonlinearConstraint(WallConstraint, 0, np.inf)]
    overlap_cons = [NonlinearConstraint(NonOverlappConstraint, 0, np.inf)]

    # COBYLA optimization phase
    bounds = Bounds(0, 1)
    cons = wall_cons + overlap_cons
    solution_cobyla = minimize(lambda x: -np.sum(x[26:]), initial_guess, method='COBYLA', bounds=bounds, constraints=cons, options={'rhobeg': 0.1, 'rhoend': 1e-8, 'maxfun': 20000, 'tol': 1e-7})
    center_cobyla = solution_cobyla.x[:26]
    radius_cobyla = solution_cobyla.x[26:]

    # PSO optimization phase
    def update_radius(center):
        d = distance_matrix(center, center)
        w = np.minimum(center[:, np.newaxis], 1 - center[:, np.newaxis])
        r = linprog(np.ones(26), A_b=(d - w), b_ub=(d - 2 * w), bounds=(0, None), method='highs', options={'tol': 1e-9, 'a': 1e-12}).x
        return r

    N = 30
    T = 2000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_clip = 0.1
    swarm = np.random.rand(N, 52)
    personal_bests = np.full(N, np.inf)
    global_best = np.inf
    velocities = np.zeros((N, 52))

    for _ in range(T):
        r = update_radius(swarm)
        fitness = -np.sum(r, axis=1)
        f_best = np.minimum(fitness, personal_bests)
        personal_bests[fitness < personal_bests] = fitness[fitness < personal_bests]
        swarm[fitness < personal_bests] = swarm[fitness < personal_bests]
        if np.any(fitness < personal_bests):
            g = swarm[np.argmin(fitness)]
        for i in range(N):
            r1, r2 = np.random.rand(2, 52)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_bests[i] - swarm[i]) + c2 * r2 * (g - swarm[i])
            velocities[i] = np.clip(velocities[i], -velocity_clip, velocity_clip)
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)

        if np.any(fitness < global_best):
            global_best = np.min(fitness)
        if global_best < np.inf:
            best_index = np.argmin(fitness)
            r_best = update_radius(swarm[best_index, :26])
            center_best = swarm[best_index, :26]
            return (center_best.reshape(-1, 2), r_best, np.sum(r_best))

    return (center_cobyla.reshape(-1, 2), radius_cobyla, np.sum(radius_cobyla))
