import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def _make_constraints(n):
        cons = []
        for i in range(n):
            x, y, r = 3 * i, 3 * i + 1, 3 * i + 2
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[x] - x[r]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[x] - x[r]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[y] - x[r]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[y] - x[r]})
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt((x[3 * i] - x[3 * j]) ** 2 + (x[3 * i + 1] - x[3 * j + 1]) ** 2) - x[3 * i + 2] - x[3 * j + 2]})
        return cons

    def _obj(x):
        return -np.sum(x[range(26, 78)])

    def _grow_step(x, step, n):
        for i in range(n):
            if x[3 * i + 2] + step > 0 and x[3 * i + 2] + step + min(x[3 * i] - step, 1 - x[3 * i] - step, x[3 * i + 1] - step, 1 - x[3 * i + 1] - step) >= x[3 * i + 2]:
                x[3 * i + 2] += step
        return x

    def _shake_and_refine(x, step, n):
        for i in range(n):
            dx = np.random.uniform(-step, step)
            dy = np.random.uniform(-step, step)
            x[3 * i] += dx
            x[3 * i + 1] += dy
            x[3 * i] = np.clip(x[3 * i], 0, 1)
            x[3 * i + 1] = np.clip(x[3 * i + 1], 0, 1)
        res = minimize(_obj, x, method='SLSQP', bounds=[(0, 1)] * 52 + [(0, 0.5)] * 26, constraints=cons, options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
        return res.x

    def optimize_with_initial_guess(x0):
        n = 26
        x0_flat = np.zeros(3 * n)
        for i in range(n):
            x0_flat[3 * i] = x0[i, 0]
            x0_flat[3 * i + 1] = x0[i, 1]
            x0_flat[3 * i + 2] = 0.09
        bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
        cons = _make_constraints(n)
        res = minimize(_obj, x0_flat, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
        return res

    centers = np.random.rand(26, 2, 4)
    best_sum = -np.inf
    best_x = None

    for k in range(6):
        for j in range(4):
            x0 = 0.1 + 0.2 * np.random.rand(26, 2)
            x0[:, 0] += x0[:, 0].min() / 2
            x0[:, 1] += x0[:, 1].min() / 2
            x0 = x0 * (1 - 0.2) + 0.1
            x0 = np.clip(x0 + np.random.uniform(-0.02, 0.02, (26, 2)), 0, 1)
            r0 = 0.09  # All circles initially have the same radius
            x0 = np.column_stack((x0, r0 * np.ones(26)))
            initial_guess = x0.reshape(-1)
            res = optimize_with_initial_guess(x0)
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x

    centers = best_x.reshape(-1, 3)[:, :2]
    radii = best_x.reshape(-1, 3)[:, 2]
    for _ in range(20):
        centers = _shake_and_refine(centers, 5e-4, 26)
        radii = _grow_step(radii, 1e-4, 26)

    return centers, radii, np.sum(radii)
