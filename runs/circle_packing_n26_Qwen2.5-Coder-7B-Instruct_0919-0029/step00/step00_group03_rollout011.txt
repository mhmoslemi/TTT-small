import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog
import numpy.random as rnd

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    centers = np.vstack([
        centers,
        (0.5, 0.5)
    ])
    
    best_radii = np.zeros(n)
    best_sum_radii = 0
    
    def lp_for_fixed_centers(centers):
        d_wall = np.minimum(
            np.minimum(centers[:, 0], 1 - centers[:, 0]),
            np.minimum(centers[:, 1], 1 - centers[:, 1])
        )
        A_ineq = np.vstack([
            np.eye(n),
            np.full((n, n), -1),
            -np.eye(n),
            np.full((n, n), -1)
        ])
        b_ineq = np.hstack([
            d_wall,
            np.zeros(n),
            d_wall,
            np.zeros(n)
        ])
        
        d_pairs = cdist(centers, centers)
        np.fill_diagonal(d_pairs, np.inf)
        
        A_eq = np.zeros((n-1, n))
        for i in range(n-1):
            A_eq[i, i:i+2] = [1, -1]
        
        res = linprog(
            -np.ones(n),
            A_ub=A_ineq,
            b_ub=b_ineq,
            A_eq=A_eq,
            b_eq=np.ones(n-1),
            bounds=(0, None),
            method='highs'
        )
        
        return res.x
    
    temperature = 1.0
    alpha = 0.995
    max_generations = 2000
    
    while temperature > 1e-6:
        for _ in range(max_generations):
            i = rnd.randint(n)
            p_i_new = centers[i] + rnd.normal(0, 0.1 * temperature, 2)
            p_i_new = np.clip(p_i_new, 0, 1)
            
            d_pairs = cdist(centers, centers)
            np.fill_diagonal(d_pairs, np.inf)
            d_pairs[i, :] = np.inf / 2  # Avoid self-distances
            d_pairs[:, i] = np.inf / 2
            
            centers_old = centers.copy()
            centers[i] = p_i_new
            radii_old = lp_for_fixed_centers(centers)
            sum_radii_old = np.sum(radii_old)
            
            if sum_radii_old > best_sum_radii:
                best_sum_radii = sum_radii_old
                best_radii = radii_old
            else:
                centers = centers_old
        
        temperature *= alpha
        
    return centers, best_radii, best_sum_radii
