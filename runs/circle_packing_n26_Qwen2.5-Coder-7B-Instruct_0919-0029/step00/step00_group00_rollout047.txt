import numpy as np
from scipy.optimize import linprog, minimize

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"

def objective(z):
    n = 26
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    E = -np.sum(radii)
    p = 2
    λ = 5e4
    energy = E
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d_ij == 0:
                d_ij = 1e-12
            energy += λ * (max(0., r_i + r_j - d_ij)**p for r_i, r_j in zip(radii, radii) if r_i + r_j > d_ij)
        for j in range(n):
            energy += λ * max(0., min(radii[j], max(centers[j, 0], 1 - centers[j, 0]) - r_j), max(centers[j, 1], 1 - centers[j, 1]) - r_j, 0)
    return energy

def bounded_objective(z):
    return objective(z)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.01)
    z0 = np.hstack((initial_centers.flatten(), initial_radii))

    d =bounded_objective(z0)
    print("Initial objective: ", d)
    
    res = minimize(bounded_objective, z0, method='Nelder-Mead', bounds=[-1,1] * 2*n + [0,1] * n)

    optimized_centers = res.x[:2*n].reshape(n, 2)
    optimized_radii = res.x[2*n:]

    is_valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not is_valid:
        raise ValueError(msg)

    sum_radii = np.sum(optimized_radii)
    return optimized_centers, optimized_radii, sum_radii

centers, radii, sum_radii = run_packing()
print("Final objective: ", sum_radii)
print("Locally optimal radii sum within the constraints: ", sum_radii)
