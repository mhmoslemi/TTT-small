import numpy as np
from scipy.optimize import linprog
from scipy.stats import multivariate_normal

def solve_lp(centers):
    n = centers.shape[0]
    c = -np.ones(n)  # We want to maximize radii sum, which is equivalent to minimizing -radii sum
    A_wall = -np.concatenate([np.eye(n), -np.eye(n),
                              -np.ones((n, n)), -np.ones((n, n)),
                              -np.ones((n, n)), -np.ones((n, n))], axis=1)
    b_wall = -np.hstack([np.minimum(centers[:, 0], 1 - centers[:, 0]),
                          np.minimum(centers[:, 1], 1 - centers[:, 1]),
                          np.zeros(n), np.zeros(n),
                          np.zeros(n), np.zeros(n)])
    A_dist = -np.zeros((n * (n - 1) // 2, 2 * n))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(2 * n)
            row[i * 2:i * 2 + 2] = 1
            row[j * 2:j * 2 + 2] = 1
            A_dist[j * (j - 1) // 2 + i - (i * (i + 1)) // 2] = row
    b_dist = np.zeros(n * (n - 1) // 2)
    bounds = [(0, None) for _ in range(n)] * 2
    res = linprog(c, A_wall=A_wall, b_wall=b_wall, A_eq=A_dist, b_eq=b_dist, bounds=bounds, method='highs')
    radii = radii_positive(res.x[n:])
    return radii, -res.fun

def radii_positive(radii):
    radii[radii < 0] = 0
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    std_dev = 0.1
    temperature = 1.0
    alpha = 0.995
    max_generations = 2000
    min_temperature = 1e-6

    def simulate_annealing():
        best_centers = None
        best_sum_radii = -np.inf
        while temperature > min_temperature:
            centers = simulate_annealing_step()
            radii, sum_radii = solve_lp(centers)
            if sum_radii > best_sum_radii:
                best_sum_radii = sum_radii
                best_centers = centers
            temperature *= alpha
        return best_centers, best_sum_radii

    def simulate_annealing_step():
        centers = np.random.rand(n, 2)
        for gen in range(max_generations):
            i = np.random.randint(n)
            perturbation = multivariate_normal.rvs(mean=np.zeros(2), cov=std_dev**2 * np.eye(2))
            new_center = centers[i] + perturbation
            new_center = np.clip(new_center, 0, 1)
            centers[i] = new_center
            radii, sum_radii = solve_lp(centers)
            if sum_radii < best_sum_radii:
                centers[i] = new_center
            if gen % 100 == 0 and np.random.rand(1) < 0.1:
                perform_greedy_moves(centers)
        return centers

    def perform_greedy_moves(centers):
        for i in range(n):
            diff = centers[None, i, :] - centers[:, None, :]
            distances = np.sqrt(np.sum(diff ** 2, axis=2))
            for j in range(n):
                if j != i and 0.1 < distances[i, j] < 0.2:
                    direction = diff[i, j] / distances[i, j]
                    new_center = centers[i] - 1e-4 * direction
                    new_center = np.clip(new_center, 0, 1)
                    centers[i] = new_center

    centers, sum_radii = simulate_annealing()
    radii, _ = solve_lp(centers)

    return centers, radii, sum_radii
