import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.spatial.distance import cdist

    boundary = np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - (1 - centers[:, 0])) + \
                 np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii - (1 - centers[:, 1]))
    return np.sum(overlap**2) + np.sum(boundary**2)

def objective(z):
    n = len(z) // 3
    centers = z[:2*n].reshape(-1, 2)
    radii = z[2*n:]
    return -np.sum(radii) + 5e4 * penalty(centers, radii)

def run_packing():
    n = 26
    center_radius_indices = list(range(n))
    restarts = 15
    
    best_centers = None
    best_rsum = -np.inf
    
    for _ in range(restarts):
        initial_centers = (np.random.rand(n, 2) - 0.5) * 0.8 + 0.5
        initial_radii = np.random.rand(n) * 0.01
        initial_z = np.concatenate((initial_centers.flatten(), initial_radii))
        
        res = differential_evolution(objective, bounds=[(0, 1), (0, 1)] * n + list(range(n)), 
                                      x0=initial_z, popsize=50, mutation=(0.7, 1.5), recombination=0.7, 
                                      seed=_)
        
        current_centers = res.x[:2*n].reshape(-1, 2)
        current_radii = res.x[2*n:]
        if np.sum(current_radii) > best_rsum:
            best_centers = current_centers
            best_rsum = np.sum(current_radii)
    
    centers = best_centers
    radii = linprog(-best_rsum, A_ub=np.zeros((n, n)), b_ub=np.zeros(n), A_eq=None, b_eq=None, 
                    bounds=(0, 1), method='highs').x
    return centers, radii, -np.sum(radii)
