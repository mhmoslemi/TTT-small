import numpy as np
from ortools.linear_solver import pywraplp
import itertools

def build_milp_grid(k):
    delta = 1 / k
    R_max = np.minimum.outer(np.minimum(np.arange(k) * delta, 1 - np.arange(k) * delta), np.minimum(np.arange(k) * delta, 1 - np.arange(k) * delta))
    solver = pywraplp.Solver('MILP', pywraplp.Solver.CBC_MIXED_INTEGER_PROGRAMMING)
    variables = {}
    for i in range(k):
        for j in range(k):
            variables[(i, j)] = solver.NumVar(0, R_max[i, j], 'b_%d_%d' % (i, j))
            variables[(i, j + k)] = solver.NumVar(0, 1, 'r_%d_%d' % (i, j))
        variables[(k, j)] = 0

    card = solver.Constraint(26, 26)
    for i, j in variables.keys():
        if j > k:
            card.SetCoefficient(variables[(i, j)], 1)

    for i, j in variables.keys():
        if i < k and j <= k:
            variables[(i, j)].SetIntegration(0)

    for i, j in variables.keys():
        if j > k:
            for a, b in variables.keys():
                if a < k and b <= k:
                    d_pq = np.sqrt((i - a) * (i - a) + (j - b) * (j - b))
                    if d_pq >= max(R_max[i, j], R_max[a, b]):
                        continue
                    M_pq = d_pq - max(R_max[i, j], R_max[a, b]) + 1e-6
                    solver.Add(variables[(i, j)] + variables[(a, b)] <= d_pq + M_pq * (2 - variables[(i, j + k)] - variables[(a, b + k)]))

    return solver, variables

def solve_milp(solver, time_limit):
    solver.SetNumThreads(1)
    solver.SetTimeLimit(time_limit)
    solver.SetRelativeMipGap(1e-6)
    status = solver.Solve()
    return status, [(i[0], i[1], solver.Value(variables[(i[0], i[1])]), solver.Value(variables[(i[0], i[1] + 20)])) for i in variables.keys() if solver.Value(variables[(i[0], i[1] + 20)]) > 1e-10]

def extract_from_sol(best_sol, variables):
    used_centers = [p for p, _, _, r in best_sol if r > 1e-10]
    used_radii = [r for _, _, _, r in best_sol if r > 1e-10]
    return np.array(used_centers), np.array(used_radii)

def is_valid_placement(centers, radii):
    for c1, c2 in itertools.combinations(centers, 2):
        d = np.linalg.norm(c1 - c2)
        if d < sum(radii[centers.tolist().index(c1)], radii[centers.tolist().index(c2)]):
            return False
    for c in centers:
        if c[0] - radii[centers.tolist().index(c)] < 0 or c[0] + radii[centers.tolist().index(c)] > 1 or c[1] - radii[centers.tolist().index(c)] < 0 or c[1] + radii[centers.tolist().index(c)] > 1:
            return False
    return True

def polish(centers, radii):
    delta = 1e-4
    while True:
        changes = False
        for i in range(len(centers)):
            r_new = radii[i] + delta
            if not is_valid_placement(np.insert(centers, i, [centers[i][0], centers[i][1], r_new], axis=0), np.insert(radii, i, r_new)):
                r_new = max(radii[i], 0)
            centers[i][2] = max(0, min(r_new, 1 - r_new))
            if centers[i][2] != r_new:
                changes = True
        if not changes:
            break

    for i in range(len(centers)):
        for dx, dy in [(0, 0), (delta, 0), (-delta, 0), (0, delta), (0, -delta), (delta, delta), (delta, -delta), (-delta, delta), (-delta, -delta)]:
            c_new = np.array([centers[i][0] + dx, centers[i][1] + dy], dtype=np.float64)
            c_new[2] = radii[i]
            if is_valid_placement(np.insert(centers, i, c_new, axis=0), np.insert(radii, i, c_new[2])):
                centers[i][0] = c_new[0]
                centers[i][1] = c_new[1]
                break

    return centers, radii

def run_packing():
    solver, grid_vars = build_milp_grid(k=20)
    status, best_sol = solve_milp(solver, time_limit=600)
    centers, radii = extract_from_sol(best_sol, grid_vars)
    centers, radii = polish(centers, radii)
