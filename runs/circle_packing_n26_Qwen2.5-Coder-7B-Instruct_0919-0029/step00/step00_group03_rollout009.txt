from scipy.optimize import minimize
import numpy as np

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective(x):
    n = 26
    radii = x[n:]
    return -np.sum(radii)

def constraint_rad_i(x):
    n = 26
    centers = x[:2*n].reshape((n, 2))
    radii = x[n:]
    i, = np.where(np.isclose(centers[:, 0] - radii[i], 0))
    return 0

def nonlinear_constraint_dist(x, i, j):
    n = 26
    centers = x[:2*n].reshape((n, 2))
    radii = x[n:]
    return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (radii[i] + radii[j])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    r = 0.1

    # Initial grid configuration
    for i in range(5):
        for j in range(5):
            centers[i*5 + j] = [0.1 + 0.2 * j, 0.1 + 0.2 * i]
    centers[25] = [0.5, 0.5]

    # Compute initial radii
    for i in range(26):
        r_i = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = r_i

    # Create optimization vector
    x0 = np.hstack((centers.flatten(), radii))

    # Constraints
    constraints = []

    # Non-negativity constraints for radii
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: -x[n + i]})
    
    # Constraints for each radial distance to the boundaries
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2] - x[n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*2] - x[n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2 + 1] - x[n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*2 + 1] - x[n + i]})
    
    # Non-overlap constraints for each pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: nonlinear_constraint_dist(x, i, j)})

    # Minimise
    result = minimize(objective, x0, method='SLSQP', constraints=constraints)

    if not result.success:
        raise Exception("Optimisation failed: ", result.message)

    # Extract final centres and radii
    final_centers = result.x[:2 * n].reshape((n, 2))
    final_radii = result.x[2 * n:]

    assert validate_packing(final_centers, final_radii)[0], "Invalid packing returned by optimisation"

    # Compute max linear radii for each centre
    lp_radii = []
    for i in range(n):
        r_i = min(final_centers[i, 0], 1 - final_centers[i, 0], final_centers[i, 1], 1 - final_centers[i, 1])
        for j in range(n):
            if i != j:
                dist_ij = np.sqrt(np.sum((final_centers[i] - final_centers[j]) ** 2))
                r_i = min(r_i, dist_ij / 2)
        lp_radii.append(r_i)
    lp_radii = np.array(lp_radii)

    # Return final result
    return final_centers, lp_radii, np.sum(lp_radii)
