import numpy as np
from scipy.optimize import minimize

def lp_oracle(centers):
    n = centers.shape[0]
    A = np.empty((n*(n-1)//2, n+2))
    b = np.empty(n*(n-1)//2)
    for i in range(n):
        for j in range(i+1, n):
            A[i*(n-1)//2 + j-i-1, i] = 2 * centers[i, 0]
            A[i*(n-1)//2 + j-i-1, n+i] = 2 * centers[i, 1]
            A[i*(n-1)//2 + j-i-1, j] = -2 * centers[j, 0]
            A[i*(n-1)//2 + j-i-1, n+j] = -2 * centers[j, 1]
            A[i*(n-1)//2 + j-i-1, n] = 1
            b[i]*(n*(n+1)//2-1-j*(j+1)//2+i*(i+1)//2-1) = centers[i, 0] + centers[j, 0] - 1
    A[:, n] = np.sqrt((centers[:, 0][:, np.newaxis] - centers[:, 0])**2 + (centers[:, 1][:, np.newaxis] - centers[:, 1])**2).flatten()
    return A, b

def objective(x):
    n = int(np.sqrt(len(x)))
    centers = x.reshape(n, 2)
    A, b = lp_oracle(centers)
    r_best = minimize(lambda x: -x, np.zeros(2*n), A_ub=A, b_ub=b, bounds=[(0, 1), (0, 1)]*n).x
    return np.sum(r_best)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    initial_guess = np.zeros(2*n)
    for i in range(n):
        initial_guess[2*i] = 0.5
        initial_guess[2*i+1] = 0.5
    result = minimize(objective, initial_guess, bounds=[(0, 1), (0, 1)]*n, method='L-BFGS-B', options={'maxiter': 200, 'disp': False})
    centers = result.x.reshape(n, 2)
    A, b = lp_oracle(centers)
    r_best = minimize(lambda x: -x, np.zeros(2*n), A_ub=A, b_ub=b, bounds=[(0, 1), (0, 1)]*n).x
    return centers, r_best, np.sum(r_best)
