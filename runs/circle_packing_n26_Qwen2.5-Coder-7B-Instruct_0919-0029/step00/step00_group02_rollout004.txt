import numpy as np
from scipy.optimize import minimize

def _make_constraints(n):
    # boundary constraints
    bounds = [{'type': 'ineq', 'fun': lambda x, i=i: x[3*i + 0] - x[3*i + 2]},
              {'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i + 0] - x[3*i + 2]},
              {'type': 'ineq', 'fun': lambda x, i=i: x[3*i + 1] - x[3*i + 2]},
              {'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i + 1] - x[3*i + 2]}]
    # pair constraints - evaluate vectorized
    def pair_constr(x, i, j):
        return np.hypot(x[3*i] - x[3*j], x[3*i + 1] - x[3*j + 1]) - x[3*i + 2] - x[3*j + 2]
    dist_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist_matrix[i, j] = pair_constr(None, i, j)
            if dist_matrix[i, j] > -1e-12:
                bounds.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: dist_matrix[i, j]})
    return bounds

def _obj(x):
    return -np.sum(x[2::3])

def _grow_step(x):
    n = x.shape[0] // 3
    changed = False
    for i in range(n):
        if x[3*i+2] >= 0.49:
            continue
        x[3*i+2] += 0.0001
        if not np.all(x[3*i] + x[3*i+2] <= 1) or not np.all(x[3*i+1] + x[3*i+2] <= 1) or np.any((x[3*i+2] + x[3*k+2] for k in range(n)) > np.hypot(x[3*i] - x[3*k], x[3*i+1] - x[3*k+1])):
            x[3*i+2] -= 0.0001
        else:
            changed = True
    return x, changed

def _shake_and_refine(x, n):
    for _ in range(20):
        for i in range(n):
            dx, dy = np.random.uniform(-5e-4, 5e-4, 2)
            if x[i*3] + dx >= 0 and x[i*3] + dx + x[i*3+2] <= 1:
                x[i*3] += dx
            if x[i*3+1] + dy >= 0 and x[i*3+1] + dy + x[i*3+2] <= 1:
                x[i*3+1] += dy
            x, changed = _grow_step(x)
            if changed:
                break
    return x

def run_packing():
    n = 26
    centers = np.random.rand(26, 2)
    radii = np.zeros(26) + 0.09
    for i in range(n):
        centers[i, 0] = 0.1 + 0.2 * i
        centers[i, 1] = 0.1 + 0.2 * (i // 5)
    centers[25] = [0.5, 0.5]
    x0_flat = np.zeros(3 * n)
    for i in range(n):
        x0_flat[3 * i] = centers[i, 0]
        x0_flat[3 * i + 1] = centers[i, 1]
        x0_flat[3 * i + 2] = radii[i]
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    constraints = _make_constraints(n)
    res = minimize(_obj, x0_flat, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 20000, 'disp': False})
    x_best = res.x
    for i in range(n):
        centers[i] = x_best[3 * i:3 * i + 2]
        radii[i] = x_best[3 * i + 2]
    centers[25] = np.array([0.5, 0.5])
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
