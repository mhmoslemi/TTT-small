import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(12345)
    N = 26
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    g = 1e-6
    delta = 0.5e-6
    max_iter = 200000
    shake_rounds = 20
    shake_error_range = 20
    shake_epsilon = 0.04

    # Seed the configuration
    center_grid = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    centers[:25] = center_grid
    centers[25] = (0.5, 0.5)

    def distance(c1, c2):
        return np.sqrt(np.sum((c1 - c2) ** 2))

    def resolve_conflicts(centers, radii):
        for i in range(N):
            for j in range(i + 1, N):
                d_ij = distance(centers[i], centers[j])
                if radii[i] + radii[j] > d_ij:
                    r_i_new = radii[i] + g
                    r_j_new = radii[j] + g
                    if r_i_new + r_j_new > d_ij:
                        delta_c = centers[i] - centers[j]
                        d_c = np.sqrt(np.sum(delta_c ** 2))
                        move_dir = delta_c / d_c
                        centers[i] += move_dir * delta
                        radii[i] = min(
                            distance(centers[i], centers[j]) / 2,
                            centers[i, 0], 1 - centers[i, 0],
                            centers[i, 1], 1 - centers[i, 1]
                        )
                    if r_j_new > r_i_new:
                        move_dir = delta_c / d_c
                        centers[j] -= move_dir * delta
                        radii[j] = min(
                            distance(centers[i], centers[j]) / 2,
                            centers[j, 0], 1 - centers[j, 0],
                            centers[j, 1], 1 - centers[j, 1]
                        )

    for i in range(shake_error_range):
        if i % shake_rounds == 0:
            k = np.random.randint(N)
            centers[k] = np.clip(centers[k] + np.random.uniform(-shake_epsilon, shake_epsilon, 2), 0, 1)
            radii[k] = min(
                distance(centers[k], centers[:k].reshape(k, N)[0, :]).min() / 2,
                centers[k, 0], 1 - centers[k, 0],
                centers[k, 1], 1 - centers[k, 1]
            )

        for iter in range(max_iter):
            resolve_conflicts(centers, radii)
            new_radii_sum = radii.sum()
            if new_radii_sum - radii.sum() < 1e-8:
                break

    if np.isnan(radii).any() or np.any(radii < 0) or not ((centers[:, 0] >= 0) & (centers[:, 0] <= 1) & (centers[:, 1] >= 0) & (centers[:, 1] <= 1)).all():
