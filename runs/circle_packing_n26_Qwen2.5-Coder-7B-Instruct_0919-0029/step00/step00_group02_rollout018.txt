import numpy as np

def build_random_feasible(n, r0=0.05):
    centers = []
    radii = []
    for _ in range(n):
        while True:
            x, y = np.random.rand(2)
            if x >= r0 and x <= 1-r0 and y >= r0 and y <= 1-r0:
                overlaps = any(np.linalg.norm(np.array([x, y]) - np.array(c)) <= r0 + r for c, r in zip(centers, radii))
                if not overlaps:
                    centers.append([x, y])
                    radii.append(r0)
                    break
    return np.array(centers), np.array(radii)

def distance_matrix(centers, radii):
    n = len(centers)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dx, dy = centers[i] - centers[j]
            dists[i, j] = dists[j, i] = np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
    return dists

def random_move(centers, radii, i):
    dx, dy = np.random.uniform(-0.1, 0.1, 2)
    new_x, new_y = centers[i] + np.array([dx, dy])
    new_x = max(radii[i], min(1 - radii[i], new_x))
    new_y = max(radii[i], min(1 - radii[i], new_y))
    return new_x, new_y

def random_radius_change(radii, i):
    dr = np.random.uniform(-0.08, 0.08)
    new_r = max(0.0, min(1 - centers[i][0], min(1 - centers[i][1], radii[i] + dr)))
    return new_r

def simulated_annealing(centers, radii, best_sum):
    n, T0, cool, n_iter, move_prob, shift_max, radius_max = len(centers), 1.0, 0.9995, 200000, 0.7, 0.1, 0.08
    for k in range(n_iter):
        T = T0 * cool**k
        i = np.random.randint(n)
        if np.random.rand() < move_prob:
            new_x, new_y = random_move(centers, radii, i)
        else:
            new_r = random_radius_change(radii, i)
            new_x, new_y = centers[i]
        new_centers = np.array(centers)
        new_radii = np.array(radii)
        new_centers[i] = [new_x, new_y]
        if new_r >= 0.0:
            new_radii[i] = new_r
        dists = distance_matrix(new_centers, new_radii)
        invalid = (dists < -1e-12).any() or (new_x - new_radii[i] < -1e-12 or new_x + new_radii[i] > 1 + 1e-12
                                           or new_y - new_radii[i] < -1e-12 or new_y + new_radii[i] > 1 + 1e-12)
        if not invalid:
            new_sum = new_radii.sum()
            if new_sum >= best_sum:
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers, best_radii = new_centers.copy(), new_radii.copy()
                else:
                    if np.random.rand() < np.exp((new_sum - best_sum) / T):
                        best_sum, best_centers, best_radii = new_sum, new_centers.copy(), new_radii.copy()
            centers, radii = new_centers.copy(), new_radii.copy()
    return centers, radii, best_sum
