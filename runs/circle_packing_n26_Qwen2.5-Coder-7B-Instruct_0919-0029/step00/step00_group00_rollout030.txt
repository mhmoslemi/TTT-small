import numpy as np
from scipy.optimize import linprog

def run_packing() -> (np.ndarray, np.ndarray, float):
    NUM_STARTS = 25
    INIT_R = 0.01
    SIM_ITERS = 2500

    def solve_lp_for_centres(centres) -> (np.ndarray, float):
        n = centres.shape[0]
        vcoeff = np.ones(n)
        sense = ["le"] * n
        bounds = [(0, None)] * n
        constraints = []

        for i in range(n):
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(centres[i] - centres[j])
                    row = np.zeros(n)
                    row[i] = 1.0
                    row[j] = 1.0
                    constants = d - 1e-12
                    constraints.append((row, sense[i*n+j], constants))
                for axis in range(2):
                    constants = centres[i, axis] - 1e-12
                    constraints.append(([0, 0, 0, 0]), sense[2*n+j], constants))
                    constraints.append(([0, 0, 0, 0]), sense[2*n+j], -constants + 1)
        optimal_radii = linprog(-vcoeff, A_ub=np.array(constraints)[:, :-1], b_ub=np.array(constraints)[:, -1], bounds=bounds, method='simplex').x
        optimal_sum = -linprog(-vcoeff, A_ub=np.array(constraints)[:, :-1], b_ub=np.array(constraints)[:, -1], bounds=bounds, method='simplex').fun
        return optimal_radii, optimal_sum

    best_centres = None
    best_radii = None
    best_sum = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)
        radii = np.full(26, INIT_R)

        for iter in range(SIM_ITERS):
            distances = np.linalg.norm(centres[:, np.newaxis] - centres[np.newaxis, :], axis=2)
            overlaps = (radii[:, np.newaxis] + radii[np.newaxis, :] - distances - 1e-12) > 0
            directions = (centres[:, np.newaxis] - centres[np.newaxis, :]) / distances
            displacements = np.where(overlaps[:, :, np.newaxis], 0.5 * overlaps * directions, 0)
            centres += np.sum(displacements, axis=1)
            centres -= np.where(centres[:, 0] - radii < 0, (centres[:, 0] - radii)[:, np.newaxis], 0)
            centres += np.where(centres[:, 0] + radii > 1, (1 - centres[:, 0] - radii)[:, np.newaxis], 0)
            centres -= np.where(centres[:, 1] - radii < 0, (centres[:, 1] - radii)[:, np.newaxis], 0)
            centres += np.where(centres[:, 1] + radii > 1, (1 - centres[:, 1] - radii)[:, np.newaxis], 0)
            radii[np.where(np.sum(displacements, axis=1) > 0)] -= np.minimum(np.sum(displacements, axis=1)[np.where(np.sum(displacements, axis=1) > 0)], radii[np.where(np.sum(displacements, axis=1) > 0)])

        radii = np.full(26, INIT_R)
        distributes = np.where(np.sum(displacements, axis=1) > 0)[0]
        ones = np.ones((1, 26))
        A_ub = np.vstack((np.hstack((ones.T, np.any(displacements, axis=1)[:, np.newaxis] * distances[..., np.newaxis]))), (np.hstack((ones.T, np.any(displacements, axis=1)[:, np.newaxis] * (1 - distances)[..., np.newaxis]))), (np.any(displacements, axis=1)[:, np.newaxis] * ones)])[distributes]
        b = np.hstack((10 ** 11 * np.ones(np.sum(distributes)), 10 ** 11 * np.ones(np.sum(distributes)), 10 ** 6 * np.ones(np.sum(distributes))))
        B = np.arange(np.sum(distributes))[:, np.newaxis]
        A2 = -np.eye(np.sum(distributes))
        b2 = np.zeros(np.sum(distributes))
        A = np.vstack((A_ub, np.hstack((A2, radii[distributes][:, np.newaxis]))))
        b = np.hstack((b, b2))
        redecomposed = linprog(vcoeff[distributes], A_ub=A, b_ub=b, bounds=(0, None)).x
        radii[distributes] = redecomposed

        optimal_radii, optimal_sum = solve_lp_for_centres(centres)
        if optimal_sum > best_sum:
            best_sum = optimal_sum
            best_centres = centres.copy()
            best_radii = optimal_radii.copy()

    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
