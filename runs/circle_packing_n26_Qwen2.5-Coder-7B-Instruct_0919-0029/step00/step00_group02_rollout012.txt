import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k=20):
    solver = pywraplp.Solver.CreateSolver('SCIP')
    grid = np.array([(i / k, j / k) for i in range(k) for j in range(k)])
    R_max = np.minimum(grid[:, 0], 1 - grid[:, 0], grid[:, 1], 1 - grid[:, 1])
    vars_by_grid = {}
    for g in grid:
        b = solver.IntVar(0, 1, f'b{int(g[0]*k)}_{int(g[1]*k)}')
        R = solver.NumVar(0, 1e3, f'r{int(g[0]*k)}_{int(g[1]*k)}')
        solver.Add(R <= b * R_max[[(i, j)]])
        vars_by_grid[tuple(g)] = (b, R)
    return solver, vars_by_grid

def solve_milp(solver, time_limit=600):
    solver.SetNumThreads(1)
    solver.SetTimeLimit(time_limit * 1000)
    solver.SetIntParam(pywraplp.Solver.OPTIMIZATION_GAP, 1e-6)
    status = solver.Solve()
    best_solution = {g: vars_by_grid[g] for g in vars_by_grid if vars_by_grid[g][0].solution_value() == 1}
    return status, best_solution

def extract_solution(best_solution, vars_by_grid):
    centers = np.array([g for g in best_solution])
    radii = np.array([vars_by_grid[g][1].solution_value() for g in best_solution])
    return centers, radii

def polish_solution(centers, radii):
    delta_r = 1e-4
    epsilon = 1e-4
    for _ in range(10):  # grow step
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist2 = np.sum((centers[i] - centers[j]) ** 2)
                max_r = np.minimum(1 - centers[i, 0], centers[i, 0], 1 - centers[i, 1], centers[i, 1], 1 - centers[j, 0], centers[j, 0], 1 - centers[j, 1], centers[j, 1])
                if dist2 > (radii[i] + radii[j] + delta_r) ** 2:
                    continue
                if dist2 < (radii[i] + radii[j] - delta_r) ** 2:
                    max_delta_r = np.sqrt(dist2 / dist2) - radii[i] - radii[j] + delta_r
                    radii[i] += max_delta_r
                    radii[j] += max_delta_r
    for _ in range(3):  # shift step
        for i in range(len(centers)):
            for s in [-epsilon, epsilon]:
                new_center = centers[i].copy()
                for j in range(2):
                    new_center[j] += s
                    if new_center[j] < (radii[i] - epsilon) or new_center[j] > (1 - radii[i] + epsilon):
                        new_center[j] = centers[i, j]  # revert
                if not all((new_center != centers[j]).sum() > 0 for j in range(len(centers)) if j != i):
                    continue
                for j in range(i + 1, len(centers)):
                    dist2 = np.sum((new_center - centers[j]) ** 2)
                    if dist2 < (radii[i] + radii[j] - delta_r) ** 2:
                        new_center[j] = centers[j]  # revert
                centers[i] = new_center
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    model, vars_by_grid = build_milp_grid(k=20)
    status, best_solution = solve_milp(model, time_limit=600)
    if status != pywraplp.Solver.OPTIMAL:
        status, best_solution = solve_milp(model, time_limit=300)
    centers, radii = extract_solution(best_solution, vars_by_grid)
