import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def _make_constraints(n):
        def boundary_constraints(x):
            return x[:, 0] - x[:, 2], 1 - x[:, 0] - x[:, 2], x[:, 1] - x[:, 2], 1 - x[:, 1] - x[:, 2]
        
        def pair_constraints(x):
            dist = np.sqrt((x[:, np.newaxis, 0] - x[np.newaxis, :, 0]) ** 2 + 
                           (x[:, np.newaxis, 1] - x[np.newaxis, :, 1]) ** 2)
            return dist - x[:, np.newaxis, 2] - x[np.newaxis, :, 2] - 1e-15
        
        boundary = np.concatenate([bc.reshape(-1, 1) for bc in boundary_constraints], axis=1)
        dist = np.concatenate([pc.reshape(-1, 1) for pc in pair_constraints], axis=1)
        return np.concatenate([boundary, dist], axis=1)

    def _obj(x):
        return -np.sum(x[2::3])

    def _grow_step(x):
        n = x.shape[0] // 3
        for i in range(n):
            r = x[3*i+2]
            x[3*i+2] = min(r * 1.02, 0.5)
        return x

    def _shake_and_refine(x):
        for _ in range(20):
            x = _grow_step(x)
            dx = (np.random.rand(n, 6) - 0.5) * 1e-4
            new_x = x + dx
            new_x[:, 0] = np.clip(new_x[:, 0], 0, 1)
            new_x[:, 1] = np.clip(new_x[:, 1], 0, 1)
            new_x[:, 2] = np.clip(new_x[:, 2], 0, 0.5)
            new_obj = _obj(new_x)
            if new_obj > _obj(x):
                x = new_x
            minimize(_obj, x.flatten(), method='SLSQP', bounds=[(0,1)]*n*2 + [(0,0.5)]*n, options={'ftol':1e-9,'maxiter':100})
        return x

    n = 26
    x0 = np.concatenate([np.random.rand(n, 2) * 0.8, np.full(n, 0.09)])
    x0 = np.clip(x0, 0, 1)
    x0 = x0.reshape(-1)

    best_x = x0.copy()
    best_obj = _obj(x0)

    for _ in range(5):
        x = x0.copy()
        x[::3] = x[::3] + (np.random.rand(n, 2) - 0.5) * 0.04
        x[:, 0] = np.clip(x[:, 0], 0, 1)
        x[:, 1] = np.clip(x[:, 1], 0, 1)
        minimize(_obj, x.flatten(), method='SLSQP', bounds=[(0,1)]*n*2 + [(0,0.5)]*n, options={'ftol':1e-9,'maxiter':10000})
        if _obj(x) > best_obj:
            best_x = x
            best_obj = _obj(x)
    
    best_x = _shake_and_refine(best_x)
    centers = best_x.reshape(n, 3)[:, :2]
    radii = best_x.reshape(n, 3)[:, 2]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
