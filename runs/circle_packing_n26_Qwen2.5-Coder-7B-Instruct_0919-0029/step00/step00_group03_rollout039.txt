import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    grid_size = 5
    initial_radius = 0.1
    initial_centers = np.array([(0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(grid_size) for j in range(grid_size)])
    initial_centers = np.append(initial_centers, [[0.5, 0.5]])
    initial_radii = np.full(n, initial_radius)
    decision_vector = np.concatenate((initial_centers.flatten(), initial_radii))

    # Phase 1: Global coarse search with COBYLA
    def obj_function(z):
        return -np.sum(z[n * 2:])

    def constraint(z):
        constraints = []
        wall_constraints = [z[2 * i] - z[2 * n + 2 * i] for i in range(2 * n)]
        constraints.extend(wall_constraints)
        overlap_constraints = [
            np.sqrt(np.sum((z[2 * i:2 * i + 2] - z[2 * j:2 * j + 2]) ** 2)) - (z[2 * n + 2 * i] + z[2 * n + 2 * j])
            for i in range(n) for j in range(i + 1, n)
        ]
        constraints.extend(overlap_constraints)
        return constraints

    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]
    cobyla_result = optimize.minimize(
        obj_function, decision_vector, method='COBYLA', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint},
        options={'rhobeg': 0.1, 'rhoend': 1e-8, 'maxfun': 20000, 'tol': 1e-7}
    )
    z_star = cobyla_result.x

    # Phase 2: Particle Swarm optimisation of centres only
    def lp_solve(x):
        x = x.reshape((n, 2))
        distances = np.sqrt(np.sum((x[:, np.newaxis, :] - x[np.newaxis, :, :]) ** 2, axis=-1))
        wall_distances = np.minimum(np.minimum(x[:, 0], 1 - x[:, 0]), np.minimum(x[:, 1], 1 - x[:, 1]))
        wall_distances = np.repeat(wall_distances[:, np.newaxis], n, axis=1)
        A = np.concatenate((distances - (wall_distances.min(axis=0)), -np.eye(n)), axis=0)
        b = wall_distances.max(axis=0)
        c = np.ones(n)
        result = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, 0.5)] * n, method='highs', options={'abstol': 1e-12, 'reltol': 1e-9})
        return -result.fun, result.x

    swarm_size = 30
    max_iterations = 2000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_clip = 0.1
    np.random.seed(42)
    centers_best = np.concatenate(
        (initial_centers.flatten(), initial_radii)
    )
    best_radius = -np.inf
    best_p = centers_best.copy()
    best_g = np.empty_like(centers_best)

    for _ in range(max_iterations):
        velocities = np.random.uniform(-velocity_clip, velocity_clip, size=(swarm_size, 2 * n))
        radius_best = np.inf
        radius_p = np.empty((swarm_size, n))
        positions = np.empty_like(centers_best)

        for i in range(swarm_size):
            r1 = np.random.rand(2 * n)
            r2 = np.random.rand(2 * n)
            velocities[i] = w * velocities[i] + c1 * r1 * (best_p - centers_best[:2 * n]) + c2 * r2 * (best_g - centers_best[:2 * n])
            velocities[i] = np.clip(velocities[i], -velocity_clip, velocity_clip)
            positions[i] = centers_best[:2 * n] + velocities[i]
            positions[i] = np.where(positions[i] < 0, 0, positions[i])
            positions[i] = np.where(positions[i] > 1, 1, positions[i])
            _, r = lp_solve(positions[i])
            if r.sum() < radius_best:
                radius_best = r.sum()
                centers_best = np.concatenate((positions[i], r))
                if r.sum() > radius_p[i].sum():
                    radius_p[i] = r
                    velocities[i] = (centers_best[:2 * n] + w * velocities[i] + c1 * r1 * (best_p - positions[i]) +

                best_radius = radius_p[i].sum()
                best_p = positions[i]
                best_g = np.where(positions[i] < 0, 0, positions[i])
                best_g = np.where(best_g > 1, 1, best_g)

        centers_best = np.concatenate((best_p, best_p))
        radii_best = best_p[-n:]
        centers_best = np.where(centers_best[:2 * n] < 0, 0, centers_best[:2 * n])
        centers_best = np.where(centers_best[:2 * n] > 1, 1, centers_best[:2 * n])
