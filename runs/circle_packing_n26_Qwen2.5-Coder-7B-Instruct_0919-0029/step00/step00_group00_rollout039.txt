import numpy as np
from scipy.optimize import linprog

def solve_lp_for_centres(centres, num_radii=26):
    # Setup LP
    c = [-1] * num_radii  # Minimise negative sum for maximisation
    A = np.array([
        [0 if i == j else abs(centres[i] - centres[j]) - 2 for j in range(num_radii)]
        for i in range(num_radii)
    ])
    b = np.full(num_radii, 1e-12)
    Aeq = np.eye(num_radii)
    beq = np.zeros(num_radii)
    bounds = [(0, None) for _ in range(num_radii)]

    # Solve LP
    result = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
    return result.x, -result.fun

def run_packing() -> (np.ndarray, np.ndarray, float):
    NUM_STARTS = 25
    SIM_ITERS = 2500
    INIT_R = 0.01
    R_INC = 0.001
    ALPHA = 0.99

    best_centres = None
    best_radii = None
    best_sum = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)  # Uniform sample
        radii = np.full(26, INIT_R)

        # Physics simulation
        for iter in range(SIM_ITERS):
            distances = np.linalg.norm(centres[:, np.newaxis, :] - centres[np.newaxis, :, :], axis=2)
            overlaps = np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - distances)

            non_self_overlaps = overlaps != 0
            directions = (centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) / overlaps.nonzero()[2]

            displacements = (0.5 * overlaps.diagonal()) * directions
            centre_displacements = np.sum(displacements, axis=1)

            bounds_check = np.clip(radii + np.sum(np.abs(centre_displacements), axis=0)[:, np.newaxis], np.zeros_like(radii), None)

            max_start_radius = np.min(bounds_check, axis=0)
            delta_r = min(R_INC, max_start_radius - radii)

            radii += delta_r

            updated_displacements = (0.5 * overlaps.diagonal() * delta_r) * directions
            centres += updated_displacements

            centres = np.clip(centres, 0 - 1e-12, 1 + 1e-12 - np.minimum(radii, radii[:, np.newaxis]), centres)

            # Damping
            centre_displacements = np.sum(updated_displacements, axis=1)
            radii += min(centre_displacements * ALPHA, delta_r)

            if not non_self_overlaps.nonzero()[2].size:
                break

        # LP post-processing
        opt_radii, opt_sum = solve_lp_for_centres(centres)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centres = centres.copy()
            best_radii = opt_radii.copy()

    # Final LP to lock in best configuration
    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
