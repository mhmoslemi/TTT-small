import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26
    lambda_ = 5e4
    p = 2
    T0 = 0.5
    alpha = 0.9995
    sigma = 0.04 * T0
    tau = 0.01 * T0
    N = 20000
    restarts = 18

    centers_best = np.zeros((n, 2))
    radii_best = np.zeros(n)
    sum_radii_best = -np.inf

    def energy(z):
        centers = z[:2*n].reshape((n, 2))
        radii = z[2*n:]
        E = -np.sum(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dist_ij = np.linalg.norm(centers[i] - centers[j])
                E += lambda_ * (max(0, (radii[i] + radii[j] - dist_ij) ** p), 0)
                E += lambda_ * max(0, radii[i] - centers[i, 0]) + lambda_ * max(0, radii[i] - (1 - centers[i, 0]))
                E += lambda_ * max(0, radii[i] - centers[i, 1]) + lambda_ * max(0, radii[i] - (1 - centers[i, 1]))
        return E

    def accept_move(old_z, new_z, E_old, E_new, T):
        if E_new <= E_old:
            return True
        else:
            return(np.random.rand() < np.exp(-(E_new - E_old) / T))

    restart_types = ['uniform', 'hexagonal', 'corner']
    for restart_type in restart_types:
        if restart_type == 'uniform':
            centers = np.random.rand(n, 2)
            radii = 0.01 * np.ones(n)
        elif restart_type == 'hexagonal':
            side_length = 0.15
            hex_layer = np.array([[np.cos(np.radians(angle)), np.sin(np.radians(angle))] for angle in np.arange(0, 360, 60)])
            centers = np.repeat(np.expand_dims(hex_layer, 0), n // 6, axis=0)
            centers = centers.reshape(-1, 2) + np.random.rand(len(centers), 2) - 0.5
            centers = centers[:n, :]
            radii = 0.01 * np.ones(n)
        elif restart_type == 'corner':
            centers[0, :] = [0, 0]
            centers[1, :] = [0, 1]
            centers[2, :] = [1, 0]
            centers[3, :] = [1, 1]
            centers[4, :] = [0.5, 0.5]
            radii = [0.15] + [0.01] * 22
        
        for _ in range(restarts):
            z = np.concatenate((centers.flatten(), radii))
            z_best = np.copy(z)
            E_best = energy(z)
            T = T0

            for k in range(N):
                i = np.random.randint(n)
                delta_x = np.random.normal(0, sigma * T)
                delta_y = np.random.normal(0, sigma * T)
                delta_r = np.random.normal(0, tau * T)
                new_z = np.copy(z)
                new_z[2*i:2*i+2] += delta_x, delta_y
                new_z[2*n+i] += delta_r
                new_z[2*n+i] = max(0, min(1, new_z[2*n+i]))
                new_z[2*i:2*i+2] = np.clip(new_z[2*i:2*i+2], 0, 1)
                E_new = energy(new_z)

                if accept_move(z, new_z, E_best, E_new, T):
                    z = new_z
                    if E_new < E_best:
                        E_best = E_new
                        z_best = np.copy(z)

            if E_best > sum_radii_best:
                sum_radii_best = E_best
                centers_best = z_best[:2*n].reshape((n, 2))
                radii_best = z_best[2*n:]

    centers_fixed = centers_best

    A_eq = np.zeros((n, 2*n + n))
    b_eq = np.zeros(n)
    for i in range(n):
        A_eq[i, 2*i:2*i+2] = 1
        A_eq[i, 2*n+i] = -1
        for j in range(n):
            if i != j:
                A_eq[i, 2*j:2*j+2] = -1

    A_ub = np.zeros((n, 2*n + n))
    b_ub = np.zeros(n)
    for i in range(n):
        for j in range(n):
            if i != j:
                A_ub[2*i, 2*j:2*j+2] = 1
                A_ub[2*i+1, 2*j:2*j+2] = 1
                A_ub[2*i, 2*n+j] = -1
                A_ub[2*i+1, 2*n+j] = -1

    for i in range(n):
        A_ub[2*n+i,:] = np.zeros(2*n)
        A_ub[2*n+i,i] = 1
        B = np.block([[A_eq], [A_ub]])
        c = np.hstack((np.zeros(2*n), -np.ones(n)))
        res = linprog(c, A_ub=B, b_ub=np.hstack((b_eq, b_ub)), method='highs')
        if res.success:
            sum_radii_best = -res.fun
            radii_best = res.x[-n:]

    return centers_best, radii_best, sum_radii_best
