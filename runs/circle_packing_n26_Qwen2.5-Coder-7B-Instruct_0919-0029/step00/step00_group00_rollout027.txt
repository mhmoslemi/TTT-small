def compute_distance_matrix(centers):
    n = centers.shape[0]
    D = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            D[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
    D = D + D.T
    np.fill_diagonal(D, 0)
    return D

def compute_penalty(centers, radii, D):
    n = centers.shape[0]
    penalty = 0

        for j in range(i + 1, n):
            diff = radii[i] + radii[j] - D[i, j]
            penalty += diff**2 if diff > 0 else 0
            penalty += max(0, radii[i] - min(centers[i, 0], 1 - centers[i, 0]))
