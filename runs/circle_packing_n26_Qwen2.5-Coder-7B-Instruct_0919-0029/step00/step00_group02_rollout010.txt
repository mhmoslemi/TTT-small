import numpy as np
from scipy.optimize import minimize

def _make_constraints(n, centers, radii):
    def C1(x):
        return centers[:, 0] - radii  # x - r >=0
    def C2(x):
        return 1 - (centers[:, 0] + radii)  # 1-x-r >=0
    def C3(x):
        return centers[:, 1] - radii  # y - r >=0
    def C4(x):
        return 1 - (centers[:, 1] + radii)  # 1-y-r >=0
    def pair_dist(x, i, j):
        return np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - radii[i] - radii[j]
    pair_constraints = [lambda x, i=i, j=j: pair_dist(x, i, j) for i in range(n) for j in range(i+1, n)]
    return [C1, C2, C3, C4] + pair_constraints

def _obj(x):
    return -np.sum(np.split(x, 78)[2::3])  # Negate to maximize sum radii

def optimize_with_initial_guess(x0):
    n = 26
    x0_flat = x0
    bounds = [(0, 1)] * 2 * n + [(0, 0.5)] * n
    constraints = _make_constraints(n, np.zeros((n, 2)), np.zeros(n))
    res = minimize(_obj, x0_flat, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 10000, 'disp': False})
    return res

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    np.random.seed(0)
    initial_guess = np.random.rand(3 * n)

    best_res = None
    best_sum_radii = -np.inf
    for _ in range(10):
        res = optimize_with_initial_guess(initial_guess)
        sum_radii = -res.fun
        if sum_radii > best_sum_radii:
            best_res = res
            best_sum_radii = sum_radii

    # Post-processing
    def _grow_step(x):
        nonlocal best_res
        nonlocal best_sum_radii
        x_flat = best_res.x
        centers = x_flat[:2*n].reshape((n, 2))
        radii = x_flat[2*n:]
        for i in range(n):
            delta = 1e-4
            original_radius = radii[i]
            new_centers = np.copy(centers)
            new_centers[i, 0] += delta
            new_radii = np.copy(radii)
            new_radii[i] += delta
            # Check feasibility
            feasible = all([
                np.all(new_centers[:, 0] - new_radii >= 0),
                np.all(1 - (new_centers[:, 0] + new_radii) >= 0),
                np.all(new_centers[:, 1] - new_radii >= 0),
                np.all(1 - (new_centers[:, 1] + new_radii) >= 0),
                np.all([np.sqrt((new_centers[i, 0] - new_centers[j, 0])**2 + (new_centers[i, 1] - new_centers[j, 1])**2) - new_radii[i] - new_radii[j] >= 0 for j in range(n) if j != i])
            ])
            if feasible:
                new_res = minimize(_obj, new_centers.ravel(), method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000})
                new_sum_radii = -new_res.fun
                if new_sum_radii > best_sum_radii:
                    best_res = new_res
                    best_sum_radii = new_sum_radii
        return best_res

    def _shake_and_refine(x):
        for _ in range(20):
            x_flat = best_res.x
            centers = x_flat[:2*n].reshape((n, 2))
            radii = x_flat[2*n:]
            dx = np.random.uniform(-5e-4, 5e-4, 2*n)
            dy = np.random.uniform(-5e-4, 5e-4, 2*n)
            new_centers = centers + np.vstack((dx, dy)).T
            new_centers = np.clip(new_centers, 0, 1)
            distances = np.sqrt((new_centers[:, 0, np.newaxis] - new_centers[:, 0]) ** 2 + (new_centers[:, 1, np.newaxis] - new_centers[:, 1]) ** 2)
            non_negative_distances = np.maximum(distances, np.eye(n))
            valid_move = np.all(new_centers[:, 0] - radii >= 0) and np.all(1 - (new_centers[:, 0] + radii) >= 0) and np.all(new_centers[:, 1] - radii >= 0) and np.all(1 - (new_centers[:, 1] + radii) >= 0) and np.all(non_negative_distances - radii[i] - radii[j] >= 0 for i in range(n) for j in range(i+1, n))
            if valid_move:
                new_res = minimize(_obj, new_centers.ravel(), method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000})
                if new_res.fun < best_res.fun:
                    best_res = new_res
        return best_res

    best_res = _grow_step(best_res)
    _shake_and_refine(best_res)

    centers = best_res.x[:2*n].reshape((n, 2)) 
    radii = best_res.x[2*n:]
    sum_radii = -best_res.fun
    return centers, radii, sum_radii
