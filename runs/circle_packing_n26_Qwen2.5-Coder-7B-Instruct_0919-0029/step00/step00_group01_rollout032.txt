import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    borders = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    distances = np.linalg.norm(centers[:,None,:]-centers[None,:,:], axis=2)
    np.fill_diagonal(distances, np.inf)

    c = -np.ones(n)
    bounds = [(0, None)] * n

    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)

    for i in range(n):
        A_ub[i, i] = 1
        b_ub[i] = borders[i]

    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = distances[i, j]
            idx += 1

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    radii = np.maximum(result.x, 0)
    sum_radii = -result.fun

    return radii, sum_radii

def random_move(centers, seed, step, rng):
    i, j = rng.choice(centers.shape[0], 2, replace=False)
    new_pos = centers[i] + (rng.normal(scale=step, size=2) * (j % 2 + 0.5))
    new_pos[new_pos < 0] = 0
    new_pos[new_pos > 1] = 1
    centers[i] = new_pos

    return centers

def simulated_annealing_centers(n, max_iter, seed, step, rng):
    centers = rng.random((n, 2))
    best_centers = centers
    best_sum = solve_lp(centers)[-1]

    T = step
    for _ in range(max_iter):
        new_centers = random_move(centers, seed, step, rng)
        new_sum = solve_lp(new_centers)[-1]

        if new_sum > best_sum:
            best_centers = new_centers
            best_sum = new_sum
            step *= 0.9995
        elif rng.rand() < np.exp((best_sum - new_sum) / T):
            best_centers = new_centers
            best_sum = new_sum
            step *= 0.9995
        else:
            step *= 0.9995

    return best_centers, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    max_iter = 30000
    rng = np.random.default_rng(42)
    step = 0.05

    best_centers, best_sum = simulated_annealing_centers(n, max_iter, rng.bit_generator.state, step, rng)

    radii, sum_radii = solve_lp(best_centers)
    return best_centers, radii, sum_radii
