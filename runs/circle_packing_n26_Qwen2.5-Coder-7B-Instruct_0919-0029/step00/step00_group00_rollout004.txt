import numpy as np
from scipy import optimize

def compute_lp_radii(centers):
    n = centers.shape[0]
    A = np.zeros((255 + n, 26))
    b = np.zeros(255 + n)
    u = np.full(n, 0.499, dtype=np.double)
    for i in range(n):
        A[i, i] = 1
        b[i] = u[i]
        for j in range(i):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx**2 + dy**2)
            A[j + 26 * i, i] = 1
            A[j + 26 * i, j] = 1
            b[j + 26 * i] = dist - 1e-12
    res = optimize.linprog(np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x, np.sum(res.x)

def run_packing():
    np.random.seed(0)
    n = 26
    best_centers = np.random.rand(n, 2)
    best_sum = 0
    step = 0.05
    decay_factor = 0.995
    step_min = 1e-8
    num_starts = 25
    num_iters = 2000
    for start in range(num_starts):
        centres = np.random.rand(n, 2)
        for it in range(num_iters):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c
            new_radii, new_sum = compute_lp_radii(new_centres)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centres
                best_radii = new_radii
        step = max(step_min, step * decay_factor)
    final_radii, final_sum = compute_lp_radii(best_centers)
    return best_centers, final_radii, final_sum

centers, radii, sum_radii = run_packing()
