import numpy as np
from scipy.optimize import linprog, differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def penalty(v):
        x = v[:52].reshape(26, 2)
        r = v[52:]
        boundary_penalty = np.sum(np.maximum(0, -x[:, 0]) ** 2) + np.sum(np.maximum(0, x[:, 0] - 1) ** 2) + \
                           np.sum(np.maximum(0, -x[:, 1]) ** 2) + np.sum(np.maximum(0, x[:, 1] - 1) ** 2)
        negative_radius_penalty = np.sum(np.maximum(0, -r) ** 2)
        overlap_penalty = 0
        n = len(r)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((x[i] - x[j]) ** 2))
                overlap_penalty += np.maximum(0, r[i] + r[j] - dist) ** 2
        return -(np.sum(r) + 10_000_000 * (boundary_penalty + negative_radius_penalty + overlap_penalty))

    bounds = [(0, 1)] * 52 + [(0, 0.25)] * 26
    result = differential_evolution(penalty, bounds, strategy='intermediate', popsize=50, 
                                     mutation=0.8, crossover=0.7, maxiter=2000, seed=42)
    best_v = result.x
    
    def lp_objective(r):
        return -np.sum(r)

    A_eq = np.zeros((2 * 26, 26))
    b_eq = np.zeros(2 * 26)
    A_le = []
    b_le = []

    for i in range(26):
        A_eq[i, i] = -1
        b_eq[i] = -best_v[52 + i]

    for i in range(26):
        A_eq[26 + i, i] = 1
        b_eq[26 + i] = best_v[52 + i]

    for i in range(26):
        for j in range(i + 1, 26):
            dx = best_v[i * 2] - best_v[j * 2]
            dy = best_v[i * 2 + 1] - best_v[j * 2 + 1]
            d = np.sqrt(dx ** 2 + dy ** 2)
            A_le.append([1, 1, -dx / d, -dy / d])
            b_le.append(d - best_v[i * 2 + 2] - best_v[j * 2 + 2])

    A_le = np.array(A_le)
    b_le = np.array(b_le)

    result = linprog(lp_objective, A_eq=A_eq, b_eq=b_eq, A_ub=A_le, b_ub=b_le, bounds=[(0, None)] * 26)
    best_r = result.x
    
    return best_v[:52].reshape(26, 2), best_r, -penalty(best_v)
