import numpy as np
from ortools.linear_solver import pywraplp
from scipy.spatial import distance_matrix

def build_milp_grid(k=20):
    delta = 1 / k
    grid = np.array([(i * delta, j * delta) for i in range(k) for j in range(k)])
    max_radii = np.minimum(grid[:, 0], grid[:, 1]) - delta / 2
    max_radii = np.minimum(max_radii, np.maximum(1 - grid[:, 0], 1 - grid[:, 1]) - delta / 2)
    b_vars = {g: pywraplp.Variable.SizedBoolVar(f'b_{g[0]:d}_{g[1]:d}', 0, 1) for g in grid}
    r_vars = {g: pywraplp.Variable(f'r_{g[0]:d}_{g[1]:d}', 0, max_radii[g]) for g in grid}
    vars_by_grid = {g: (b_vars[g], r_vars[g]) for g in grid}
    return b_vars, r_vars, vars_by_grid, grid

def solve_milp(model, time_limit=600):
    for (g, (b, r)) in model.items():
        model.Add(b <= 1.0)
        model.Add(r <= model.max_radii[g])
        model.Add(r <= b * model.max_radii[g])

    for g in model.grid:
        model.Add(model.sum_vars([model.r_vars[h] for h in model.grid if h != g]) +
                  model.r_vars[g] <= model.distance_matrix[g, h] + 2.0 * (1 - model.b_vars[h] - model.b_vars[g])
                  for h in model.grid if h != g and np.linalg.norm(g - h) * 2 - model.max_radii[g] - model.max_radii[h] < 1e-6)

    model.Maximize(model.sum_vars([model.r_vars[g] for g in model.grid]))

    solver = pywraplp.Solver('MILP', pywraplp.Solver.CBC_MIXED_INTEGER_PROGRAMMING)
    solver.SetNumThreads(1)
    solver.SetTimeLimit(time_limit * 1000)
    solver.StopWhenFractionsOfPrimalInfeasibilityGapIsAtMost(1e-6)

    status = solver.Solve()
    if status == pywraplp.Solver.OPTIMAL or status == pywraplp.Solver.FEASIBLE:
        return status, {g: (solver.Value(model.b_vars[g]), solver.Value(model.r_vars[g])) for g in model.grid}
    else:
        return status, None
