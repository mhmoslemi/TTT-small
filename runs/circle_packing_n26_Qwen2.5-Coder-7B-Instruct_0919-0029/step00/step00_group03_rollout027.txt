import numpy as np
from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # initial configuration
    centres = [
        [0.1 + 0.2 * (i % 5), 0.1 + 0.2 * (i // 5)] for i in range(25)
    ] + [[0.5, 0.5]]
    n = len(centres)
    radii_guess = [0.1] * n

    # Optimization constraints
    A = []
    b = []
    
    # Add constraints for each centre to be inside the square
    for i in range(n):
        A.append([1 if j == i else 0 for j in range(4)])
        b.append(1)
        A.append([1 if j == i else 0 for j in range(4, 8)])
        b.append(1)

    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.sqrt((centres[i][0] - centres[j][0]) ** 2 + (centres[i][1] - centres[j][1]) ** 2)
            A.append([0 if j <= 4 else 1 for _ in range(8)] + [0 if j > 4 else 1])
            b.append(dist_ij - 1)
    
    # Define objective function
    def objective(x):
        return -np.sum(x[n:])
    
    # Constraints: all variables must be non-negative
    bnds = [(0, None)] * n + [(0, 0.5)] * n
    
    # Perform SLSQP optimization
    res = minimize(objective, centres + radii_guess, bounds=bnds, method='SLSQP', constraints={'type': 'ineq', 'fun': lambda x: np.array(A) @ x - np.array(b)})
