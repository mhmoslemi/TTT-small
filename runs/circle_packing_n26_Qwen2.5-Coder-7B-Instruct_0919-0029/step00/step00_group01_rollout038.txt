import numpy as np
from scipy.optimize import differential_evolution, linprog

def penalty(v, n):
    centres = v[:2 * n].reshape(n, 2)
    radii = v[2 * n:]
    boundary_penalty = np.sum(np.maximum(0, -centres[:, 0]) ** 2 +
                                np.maximum(0, centres[:, 0] - 1) ** 2 +
                                np.maximum(0, -centres[:, 1]) ** 2 +
                                np.maximum(0, centres[:, 1] - 1) ** 2)
    negative_radius_penalty = np.sum(np.maximum(0, -radii) ** 2)
    overlap_penalty = 1e7 * np.sum(np.maximum(0, radii[:, None] + radii[None] - np.sqrt((centres[:, None, 0] - centres[None, :, 0]) ** 2 + (centres[:, None, 1] - centres[None, :, 1]) ** 2)) ** 2)
    return boundary_penalty + negative_radius_penalty + overlap_penalty

def de_objective(v, n):
    return -penalty(v, n)

def refine_radii(centres, n):
    radii_bounds = [(0, None)] * n
    constraints = [
        {'type': 'ineq', 'fun': lambda r, c, n: r - np.minimum(c[:, 0], 1 - c[:, 0])},
        {'type': 'ineq', 'fun': lambda r, c, n: r - np.minimum(c[:, 1], 1 - c[:, 1])}
    ]
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, c, i, j: c[i, 0] + c[j, 0] + r[i] + r[j] - 2 * np.sqrt((c[i, 0] - c[j, 0]) ** 2 + (c[i, 1] - c[j, 1]) ** 2)})
    result = linprog(c=-radii_bounds, x_bounds=radii_bounds, A_ub=[constraint['fun'](np.zeros(n), centres, n) for constraint in constraints], b_ub=[0] * len(constraints), method='highs')
    return np.maximum(result.x, 0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)] * (2 * n) + [(0, 0.25)] * n
    result1 = differential_evolution(de_objective, bounds, args=(n,), popsize=50, strategy='intermediate', disp=False, maxiter=2000)
    result2 = differential_evolution(de_objective, bounds, args=(n,), popsize=50, strategy='intermediate', disp=False, maxiter=2000)
    centres1 = result1.x[:2 * n].reshape(n, 2)
    radii1 = refine_radii(centres1, n)
    sum_radii1 = np.sum(radii1)
    centres2 = result2.x[:2 * n].reshape(n, 2)
    radii2 = refine_radii(centres2, n)
    sum_radii2 = np.sum(radii2)
    if sum_radii1 > sum_radii2:
        return centres1, radii1, sum_radii1
    else:
        return centres2, radii2, sum_radii2
