import numpy as np
from scipy.optimize import linprog


def lp_optimum(centers):
    n = centers.shape[0]
    r_indices = np.arange(n)
    A ub = np.zeros((n * n, n))
    for i in range(n):
        A ub[n * i + r_indices, i] = 1
        if i > 0:
            A ub[n * i + r_indices, i - 1] = 1
            A ub[n * (i - 1) + r_indices, i] = 1
    A ub = A ub[:-4 * n, :]
    b ub = np.full(n * n, 1e-12)
    c = -np.ones(n)
    
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = 0.5  # Initial guess for radius
        if x_i < r_i or y_i < r_i:
            r_i = x_i if x_i < y_i else y_i
        
        # Boundary distance constraints
        b ub[n * i] = min(x_i, 1 - x_i, y_i, 1 - y_i)
        for j in range(i + 1, n):
            x_j, y_j = centers[j]
            d_ij = np.sqrt((x_i - x_j) ** 2 + (y_i - y_j) ** 2)
            b ub[n * i + n + (j - i - 1)] = d_ij - 1e-12
    
    b ub = -b ub
    
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A ub, b_ub=b ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        raise ValueError("LP solver did not converge")


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(0)  # For reproducibility
    n = 26
    step = 0.05
    step_min = 1e-5
    decay_factor = 0.995
    NUM_STARS = 30
    NUM_ITERS = 2000
    
    best_sum = -np.inf
    best_centres = None
    best_radii = None
    
    for start in range(NUM_STARS):
        centres = np.random.rand(n, 2)
        for it in range(NUM_ITERS):
            idx = np.random.randint(n)
            delta = np.random.uniform(-step, step, size=2)
            new_c = np.clip(centres[idx] + delta, 0, 1)
            new_centres = centres.copy()
            new_centres[idx] = new_c
            
            new_radii, new_sum = lp_optimum(new_centres)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres
                best_radii = new_radii
        
        step = max(step_min, step * decay_factor)
    
    final_radii, _ = lp_optimum(best_centres)
    return best_centres, final_radii, np.sum(final_radii)
