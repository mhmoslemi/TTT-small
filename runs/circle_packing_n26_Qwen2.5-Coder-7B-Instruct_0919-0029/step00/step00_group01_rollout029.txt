import numpy as np
from scipy.optimize import differential_evolution, linprog

def penalty(v):
    x, y, r = v[:26], v[26:52], v[52:]
    pen_boundary = np.sum(np.maximum(0, -x)**2 + np.maximum(0, x-1)**2 + np.maximum(0, -y)**2 + np.maximum(0, y-1)**2)
    pen_negative_r = np.sum(np.maximum(0, -r)**2)
    pen_overlap = np.sum(np.maximum(0, r[:,None] + r - np.sqrt(np.sum((x[:,None] - x)**2 + (y[:,None] - y)**2, axis=2)))**2)
    return pen_boundary + pen_negative_r + pen_overlap

def objective(v):
    return -np.sum(v[52:]) * 1e7 + penalty(v)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    def lp_problem(v):
        x, y, r = v[:26], v[26:52], v[52:]
        constraints = []
        for i in range(26):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: r[i]})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - x[i] - r[i]})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - y[i] - r[i]})
            for j in range(i+1, 26):
                dist = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
                constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: dist - r[i] - r[j]})
        return linprog([-r], A_ub=[], b_ub=[], bounds=[(0, None)], method='highs')

    bounds = [(0, 1)] * 52
    DE_result1 = differential_evolution(objective, bounds, strategy='intermediate', popsize=50, maxiter=2000, seed=42)
    DE_result2 = differential_evolution(objective, bounds, strategy='intermediate', popsize=50, maxiter=2000, seed=51)
    
    if DE_result1.fun > DE_result2.fun:
        v_best = DE_result1.x
    else:
        v_best = DE_result2.x
    
    lp_result = lp_problem(v_best)
    radii = -lp_result.x
    
    return v_best[:26].reshape(26, 2), radii, np.sum(radii)
