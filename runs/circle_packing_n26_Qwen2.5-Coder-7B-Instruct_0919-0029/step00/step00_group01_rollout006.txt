import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26
    # Initial random centers in [0, 1]
    centers = np.random.rand(n, 2)
    # Flags to check if the gradient descent and sparse representation are in use
    use_gradient_descent = False
    use_sparse_representation = False
    # Parameters for gradient descent
    step_size = 0.01
    alpha_decrease_factor = 0.95
    max_iter = 200
    tol = 1e-5
    # Iterations counter
    iteration = 0
    best_sum = 0.0
    best_centers = np.copy(centers)
    
    # Run the algorithm for multiple restarts to avoid getting stuck in local minima
    for restart in range(2):
        # Iterative improvement step
        while iteration < max_iter:
            # Solve the LP to get the best possible radii
            lp_r = lp(centers)[1]
            lp_sum = lp_r.sum()
            
            if lp_sum > best_sum:
                best_sum = lp_sum
                best_centers = np.copy(centers)
            
            # Compute gradient of the sum of radii
            grad = np.zeros((n, 2))
            for i in range(n):
                grad[i] = gradient_at_index(centers, i, lp_r)
            
            # Update the centers
            for i in range(n):
                centers[i] += step_size * grad[i]
                centers[i] = np.clip(centers[i], 0, 1)
            
            iteration += 1
        
        # Optional: large random perturbation to escape small local minima
        if restart == 1:
            perturb_circle(centers)
        
        # Reset iterations for the next restart
        iteration = 0
    
    # Refine the best found configuration
    final_r = lp(best_centers)[1]
    final_sum = final_r.sum()
    final_centers = best_centers
    
    # Perform local radius increase greedy augmentation
    local_increment = 1e-3
    for _ in range(10):
        for i in range(n):
            initial_r = final_r[i]
            final_r[i] += local_increment
            status, _ = lp(np.clip(centers, 0, 1))
            if status:
                final_sum += local_increment
            else:
                final_r[i] = initial_r
    
    return final_centers, final_r, final_sum

def lp(centers) -> tuple[bool, np.ndarray]:
    y = np.zeros(n)
    r = cp.Variable(n)
    center = 0
    
    # Constraints for radius upper bounds (circle's center)
    constraints = []
    for c in centers:
        constraints.append(r >= 0)
        constraints.append(r <= min(c[0], 1-c[0], c[1], 1-c[1]))
    
    # Exact solution using CVXPY
    objective = cp.Maximize(y)
    prob = cp.Problem(objective, constraints)
    result = prob.solve(solver=cp.HS)
    
    if result == -np.inf:
        return False, np.zeros(n)
    else:
        return True, r.value

def gradient_at_index(centers, i, radii):
    grad = np.zeros(2)
    for j in range(len(centers)):
        if i != j:
            distance = np.linalg.norm(centers[i] - centers[j])
            if distance > 1e-6:
                grad += (radii[j] / distance) * (centers[j] - centers[i])
    return grad

def perturb_circle(centers):
    # Find the circle with the smallest radius
    min_radius_index = np.argmin(lp(centers)[1])
    center, radius = centers[min_radius_index], lp(centers)[1][min_radius_index]
    
    # Expand the circle's radius slightly 0.05% to check if better solution found
    centers[min_radius_index, :] += 0.0005 * np.random.normal(size=2)
    centers[min_radius_index, :] = np.clip(centers[min_radius_index, :], 0, 1)
    
    # Recheck the solution with original large circle radius
    _, new_r = lp(centers)
    if new_r.sum() > lp(centers)[1].sum():
        return True
    else:
        centers[min_radius_index, :] = center
        return False
