def compute_penalty(v):
    xs = v[:52//2]
    ys = v[52//2:]
    penalty = 0.0
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None) ** 2 + np.clip(xs - 1, 0, None) ** 2 + np.clip(-ys, 0, None) ** 2 + np.clip(ys - 1, 0, None) ** 2)

            gap = v[i] + v[j] - d

    return penalty

def solve_lps_for_centres(centres):
    n = centres.shape[0]

    b = np.zeros(n*(n-1) + n)

        A[idx, idx] = 1

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1])
            A[i*(n-1) + j, i] = 1
            A[i*(n-1) + j, j] = 1
            b[i*(n-1) + j] = dist
            
    res = linprog(-A.T.dot(b), bounds=(0, None), A_ub=A, b_ub=b, method='highs')

        print("Linprog failed:", res.message)
        return np.zeros(n), 0.0
