import numpy as np
from scipy.spatial import.distance_matrix
from scipy.optimize import linprog

NUM_STARTS = 25
SIM_ITERS = 2500
INIT_R = 0.01

def solve_lp_for_centres(centres):
    n = centres.shape[0]
    d = distance_matrix(centres, centres)
    constr_pairs = d > 0
    constr_sides = np.array([np.min([x,y,1-x,1-y]), 0, 0])[:,np.newaxis] - INIT_R
    A_eq, b_eq = constr_pairs.astype(int), np.ones_like(A_eq)
    A = np.vstack([A_eq, -A_eq, constr_sides[:,:,:1]])
    b = np.hstack([np.ones_like(b_eq), -np.ones_like(b_eq), constr_sides[:,:,:1].flatten()])
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    return res.x, res.fun

def run_packing() -> (np.ndarray, np.ndarray, float):
    best_centres = None
    best_radii   = None
    best_sum     = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26,2)        # uniform sample
        radii   = np.full(26, INIT_R)

        # --- physics simulation ---
        for iter in range(SIM_ITERS):
            # 1. compute pairwise distances
            d = distance_matrix(centres, centres)
            overlap_mask = (d < 2 * radii) & (d > 0)
            distances = d[overlap_mask] - radii[overlap_mask[:,0]] - radii[overlap_mask[:,1]]
            displacements = -distances[:, np.newaxis] * (centres[overlap_mask[:,0]] - centres[overlap_mask[:,1]]) / distances[:, np.newaxis]
            total_displacements = displacements.sum(axis=0)
            centres += total_displacements / 2
            # 2. boundary pushes
            defo_left = np.maximum(INIT_R - centres[:,0], 0)
            defo_right = np.maximum(centres[:,0] + radii - 1, 0)
            defo_bottom = np.maximum(INIT_R - centres[:,1], 0)
            defo_top = np.maximum(centres[:,1] + radii - 1, 0)
            centres[:,0] -= np.minimum(defo_left + defo_right, 0.01)
            centres[:,1] -= np.minimum(defo_bottom + defo_top, 0.01)
            # 3. attempt radius growth
            delta_r = 0.001
            new_radii = radii + delta_r
            new_d = distance_matrix(centres, centres)
            new_overlap_mask = (new_d < 2 * new_radii) & (new_d > 0)
            while new_overlap_mask.any():
                new_overlaps = new_d[new_overlap_mask] - new_radii[new_overlap_mask[:,0]] - new_radii[new_overlap_mask[:,1]]
                new_radii[new_overlap_mask[:,0]] -= new_overlaps / new_overlaps.max()
                new_radii[new_overlap_mask[:,1]] -= new_overlaps / new_overlaps.max()
                delta_r /= 2
                new_radii += delta_r
                new_d = distance_matrix(centres, centres)
                new_overlap_mask = (new_d < 2 * new_radii) & (new_d > 0)
            # 4. damping / cooling
            delta_r *= 0.99

            # Break early if converged
            if np.allclose(overlap_mask, False) and np.allclose(delta_r, 0.001):
                break

        # --- LP post-processing ---
        opt_radii, opt_sum = solve_lp_for_centres(centres)
        if opt_sum > best_sum:
            best_sum   = opt_sum
            best_centres = centres.copy()
            best_radii   = opt_radii.copy()

    # final LP to lock in best configuration
    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
