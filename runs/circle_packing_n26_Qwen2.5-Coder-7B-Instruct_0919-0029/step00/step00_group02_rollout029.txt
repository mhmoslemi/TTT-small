from scipy.optimize import minimize

def _make_constraints(n, centers, radii):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+0] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+0] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+1] - x[3*i+2]})
    for i in range(n):

    return cons

def _obj(x):
    return -np.sum(x[2::3])

                OK = False
