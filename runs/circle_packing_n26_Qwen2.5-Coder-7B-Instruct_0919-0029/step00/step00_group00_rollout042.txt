import numpy as np
from scipy.optimize import linprog, differential_evolution

def objective(z):
    x = z[:26]
    y = z[26:52]
    r = z[52:]
    p = 2
    lambda_val = 5e4
    boundaries = r > np.clip(x, 0, 1) - 1e-6
    boundaries |= r > np.clip(1 - x, 0, 1) - 1e-6
    boundaries |= r > np.clip(y, 0, 1) - 1e-6
    boundaries |= r > np.clip(1 - y, 0, 1) - 1e-6
    overlap = np.clip(r[:, np.newaxis] + r[np.newaxis, :] - np.linalg.norm(x[:, np.newaxis] - x[np.newaxis, :], axis=2), 0, None)
    overlap = np.power(overlap, p)
    penalty = np.sum(np.triu(overlap, k=1)) + np.sum(np.where(boundaries, r, 0))
    energy = -np.sum(r) + lambda_val * penalty
    return energy

def bounds(z, T):
    n = 26
    x_min = 0 + 1e-6
    x_max = 1 - 1e-6
    y_min = 0 + 1e-6
    y_max = 1 - 1e-6
    r_min = 0
    r_max = np.minimum(x_max - np.clip(z[:26], x_min, x_max), y_max - np.clip(z[26:52], y_min, y_max)) - 1e-6
    return zip([x_min] * n + [y_min] * n + [r_min] * n, [x_max] * n + [y_max] * n + r_max)

def preprocess():
    return np.array([0.5] * 52)

def refine_state(x, y, r, T, lambda_val, p):
    A = np.zeros((26, 26))
    A[np.diag_indices(26)] = 1
    for i in range(26):
        for j in range(i):
            A[i, j] = -1
            A[j, i] = -1
    b = 1e-6 + np.cumsum(r)
    A1 = np.zeros((1, 26))
    A1[0, 0] = 1
    b1 = 0.15
    A2 = np.zeros((1, 26))
    A2[0, 26] = 1
    b2 = 0.15
    A3 = np.zeros((1, 26))
    A3[0, 52] = 1
    b3 = 0.15
    all_ineq = np.vstack((A, A1, A2, A3))
    all_b = np.hstack((b, b1, b2, b3))
    all_eq = np.zeros((26, 1))
    all_c = -np.ones((26,))
    res = linprog(all_c, A_ub=all_ineq, b_ub=all_b, A_eq=all_eq, b_eq=[0], bounds=bounds(r, T), method='highs')
    if np.isnan(res.x).any() or np.any(res.x < 0):
        return None, None, None
    r_opt = res.x
    energy = objective(np.concatenate((x, y, r_opt)))
    return r_opt, energy

def simulate_annealing(T0, T, alpha, iterations, nrestarts):
    best_r = None
    best_energy = np.inf
    restarts = ['uniform', 'hexagonal', 'corner']
    for restart in restarts:
        for _ in range(nrestarts):
            x = np.random.rand(26) * 0.99 + 0.005 if restart == 'uniform' else 0.15 * np.arange(1, 27) - 7.5 + np.random.rand(26) * 0.15
            y = np.random.rand(26) * 0.99 + 0.005 if restart == 'uniform' else 0.15 * np.arange(1, 27) - 7.5 + np.random.rand(26) * 0.15
            r = np.full(26, 0.1) if restart == 'uniform' else np.ones(26) * 0.15
            for _ in range(iterations):
                i = np.random.randint(26)
                sigma = 0.04 * T
                delta_x = np.random.normal(0, sigma)
                delta_y = np.random.normal(0, sigma)
                delta_r = np.random.normal(0, 0.01 * T)
                x_new = np.clip(x[i] + delta_x, 0, 1)
                y_new = np.clip(y[i] + delta_y, 0, 1)
                r_new = np.clip(r[i] + delta_r, 0, 1)
                z_old = np.concatenate((x, y, r))
                z_new = np.concatenate((x_new, y_new, r_new))
                E_old = objective(z_old)
                E_new = objective(z_new)
                if E_new <= E_old:
                    acept = 1.0
                else:
                    acept = np.exp(-alpha * (E_new - E_old))
                if np.random.rand() < acept:
                    x[i], y[i], r[i] = x_new, y_new, r_new
                T *= alpha
            r_opt, energy = refine_state(x, y, r, T, 5e4, p=2)
            if r_opt is not None and energy < best_energy:
                best_r = r_opt
                best_energy = energy
    x, y, _ = np.array([0.5] * 52).reshape(3, -1)
    return x, y, best_r, best_energy

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    T0 = 0.5
    T = 1e-4
    alpha = 0.9995
    iterations = 20000
    nrestarts = 18
    x, y, best_r, best_energy = simulate_annealing(T0, T, alpha, iterations, nrestarts)
    best_radii = np.ones(26)
    for i in range(26):
        best_radii[i] = best_r[i]
    if best_radii is None:
        return None, None, 0
    return np.column_stack((x, y)), best_radii, best_energy

 Centers, radii, sum_radii = run_packing()
