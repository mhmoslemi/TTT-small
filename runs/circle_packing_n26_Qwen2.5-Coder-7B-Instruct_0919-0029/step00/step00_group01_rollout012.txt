import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist

def penalty(v):
    x = v[:26]
    y = v[26:52]
    r = v[52:]
    boundary_penalty = np.sum(np.maximum(0, -x)**2) + np.sum(np.maximum(0, x-1)**2) + np.sum(np.maximum(0, -y)**2) + np.sum(np.maximum(0, y-1)**2)
    negative_radius_penalty = np.sum(np.maximum(0, -r)**2)
    overlap_penalty = np.sum(np.maximum(0, r[:, np.newaxis] + r - cdist(x[:, np.newaxis], x)))**2
    return boundary_penalty + negative_radius_penalty + overlap_penalty

def objective(v):
    return -np.sum(v[52:]) + 1e7 * penalty(v)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    x0 = np.random.rand(52)
    bounds = [(0, 1)] * 52
    res = minimize(objective, x0, method='dual_annealing', bounds=bounds)
    x, y, r = res.x[:26], res.x[26:52], res.x[52:]
    
    np.random.seed(51)
    x0 = np.random.rand(52)
    res = minimize(objective, x0, method='dual_annealing', bounds=bounds)
    x_best, y_best, r_best = res.x[:26], res.x[26:52], res.x[52:]
    
    if np.sum(r_best) > np.sum(r):
        x, y, r = x_best, y_best, r_best
    
    # Refine radii using lp
    A = np.zeros((0, 26))
    b = np.zeros(len(A))
    for i in range(26):
        A = np.append(A, np.eye(26)[i], axis=0)
        b = np.append(b, min(x[i], 1-x[i], y[i], 1-y[i]))
    for i in range(26):
        for j in range(i+1, 26):
            A = np.append(A, np.zeros(26), axis=0)
            A[-1, i] = 1
            A[-1, j] = 1
            b = np.append(b, cdist([x[i], y[i]], [x[j], y[j]])[0, 0])
    
    lp_res = linprog(-r, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    r_opt = -lp_res.x
    r_opt[r_opt < 0] = 0
    
    centers = np.column_stack((x, y))
    return centers, r_opt, np.sum(r_opt)
