from ortools.linear_solver import pywraplp

def build_milp_grid(k: int) -> tuple[pywraplp.Solver, dict]:

    delta = 1.0 / k
    grid_points = [(i * delta, j * delta) for i in range(k) for j in range(k)]
    r_maxes = np.array([min(p[0], 1 - p[0], p[1], 1 - p[1]) for p in grid_points])

def solve_milp(model: pywraplp.Solver, time_limit: int) -> tuple[int, dict]:
    model.SetTimeLimit(time_limit * 1000)  # time limit in milliseconds
    model.EnableOutput()
    status = model.Solve()
    
    if status == pywraplp.Solver.OPTIMAL or status == pywraplp.Solver.FEASIBLE:
        best_solution = {
            'b': np.array([model.LookupVariable(str(f'b_{i}_{j}')).solution_value() for i in range(20) for j in range(20)]),
            'r': np.array([model.LookupVariable(str(f'r_{i}_{j}')).solution_value() for i in range(20) for j in range(20)])
        }
    else:
        best_solution = {'b': np.zeros(400), 'r': np.zeros(400)}
    
    return status, best_solution

def extract_solution(best_solution: dict, vars_by_grid: dict) -> tuple[np.ndarray, np.ndarray]:
    centers = []
    radii = []
    for g, (_, r) in enumerate(vars_by_grid.items()):
        if best_solution['b'][g] > 0.5:
            centers.append(vars_by_grid[g][0])
            radii.append(best_solution['r'][g])
    
    return np.array(centers), np.array(radii)

def polish_solution(centers: np.ndarray, radii: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    delta_r = 1e-4
    epsilon = 1e-4
    
    def can_place_circle(center: np.ndarray, radius: float) -> bool:
        for other_center, other_radius in zip(centers, radii):
            if np.linalg.norm(center - other_center) < (radius + other_radius - 1e-12):
                return False
        if center[0] - radius < -1e-12 or center[0] + radius > 1 + 1e-12:
            return False
        if center[1] - radius < -1e-12 or center[1] + radius > 1 + 1e-12:
            return False
        return True
    
    for _ in range(10):
        for i in range(26):
            for change in [-epsilon, epsilon]:
                new_center = centers[i] + np.array([change, 0]) if np.abs(change) < np.abs(centers[i][0] - 0.5) else centers[i] + np.array([0, change])
                if can_place_circle(new_center, radii[i]):
                    centers[i] = new_center
                    for j in range(26):
                        if j != i and np.linalg.norm(centers[i] - centers[j]) < (radii[i] + radii[j] - 1e-12):
                            for dr in [-delta_r, delta_r]:
                                new_center[i] += np.array([dr * np.sign(new_center[i][0] - centers[j][0]), 0])
                                new_center[j] -= np.array([dr * np.sign(new_center[j][0] - centers[i][0]), 0])
                                if not can_place_circle(new_center[i], radii[i]) or not can_place_circle(new_center[j], radii[j]):
                                    new_center[i] -= np.array([dr * np.sign(new_center[i][0] - centers[j][0]), 0])
                                    new_center[j] += np.array([dr * np.sign(new_center[j][0] - centers[i][0]), 0])

                                centers[i] = new_center[i]
                                centers[j] = new_center[j]
                                break

    return centers, radii
