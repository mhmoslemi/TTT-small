import numpy as np
from scipy.spatial import distance_matrix
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def build_random_feasible(n, r0):
    centers = []
    while len(centers) < n:
        u, v = np.random.rand(2)
       (cx, cy) = (u, v)
        valid = True
        for c_old in centers:
            if np.linalg.norm((cx, cy) - c_old) < r0 * 2:
                valid = False
                break
        if valid:
            centers.append([cx, cy])
    return np.array(centers), np.full(n, r0)

def simulated_annealing(centers, radii, best_sum):
    T0 = 1.0
    cool = 0.9995
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    step_size_decay = 0.95
    n_shift = 1000
    best_centers, best_radii = centers, radii

    def to_vector(arr):
        return arr.flatten()

    def from_vector(v):
        return v.reshape((26, 2)), v[52:].reshape((26,))

    def objective(x):
        c, r = from_vector(x)
        return -r.sum()

    def is_feasible(c, r):
        for i in range(26):
            x, y = c[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return False
            for j in range(i + 1, 26):
                if np.sqrt(np.sum((c[i] - c[j]) ** 2)) - r[i] - r[j] < -1e-12:
                    return False
        return True

    def propose_move(x):
        c, r = from_vector(x)
        i = np.random.randint(26)
        move_type = np.random.choice(['shift', 'radius'], p=[move_prob, 1 - move_prob])
        if move_type == 'shift':
            dx, dy = np.random.uniform(-shift_max, shift_max, 2)
            new_c = c.copy()
            new_c[i] += dx, dy
            new_c[i] = np.clip(new_c[i], np.maximum(r[i], 0), np.minimum(1 - r[i], 1))
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_r = r.copy()
            new_r[i] = np.clip(r[i] + dr, 0, min(1 - c[i][0], c[i][0], 1 - c[i][1], c[i][1]))
            new_c = c.copy()
        return to_vector((new_c, new_r))

    for k in range(n_iter):
        T = T0 * cool ** k
        x_prop = propose_move(to_vector((centers, radii)))
        c_prop, r_prop = from_vector(x_prop)
        if is_feasible(c_prop, r_prop):
            ΔObj = r_prop.sum() - radii.sum()
            if ΔObj >= 0:
                accept = True
            else:
                accept = np.random.rand() < np.exp(ΔObj / T)
            if accept:
                centers, radii = c_prop, r_prop
                if r_prop.sum() > best_sum:
                    best_sum = r_prop.sum()
                    best_centers, best_radii = c_prop, r_prop
        # Adaptive step size reduction
        if k % n_shift == 0:
            shift_max *= step_size_decay
            radius_max *= step_size_decay

    return best_centers, best_radii, best_sum

def deterministic_polish(centers, radii):
    def can_grow(i, dr):
        c, r = centers[i] + (0, dr), radii[i] + dr
        if c[1] - r < 0 or c[1] + r > 1:
            return False
        for j in range(26):
            if j != i and np.sqrt(np.sum((c - centers[j]) ** 2)) - r - radii[j] < 0:
                return False
        return True
    
    def can_shift(j, dx, dy):
        c, r = centers[j] + (dx, dy), radii[j]
        if c[0] - r < 0 or c[0] + r > 1 or c[1] - r < 0 or c[1] + r > 1:
            return False
        for i in range(26):
            if i != j and np.sqrt(np.sum((c - centers[i]) ** 2)) - r - radii[i] < 0:
                return False
        return True
    
    improved = True
    while improved:
        improved = False
        for i in range(26):
            if can_grow(i, 1e-4):
                radii[i] += 1e-4
                improved = True
    
    for _ in range(5):
        i = np.random.randint(26)
        dx, dy = 0.001, 0
        for _ in range(8):
            if can_shift(i, dx, dy):
                centers[i] += dx, dy
            dx = -dx
    
    return centers, radii

def run_packing():
    centers, radii = build_random_feasible(26, r0=0.05)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()

    centers, radii, best_sum = simulated_annealing(centers, radii, best_sum)

    centers, radii = deterministic_polish(centers, radii)

    validate_packing(centers, radii)

    return centers, radii, best_sum
