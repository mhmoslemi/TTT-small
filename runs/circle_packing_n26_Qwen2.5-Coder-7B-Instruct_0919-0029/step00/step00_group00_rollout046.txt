import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, λ, p):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]

    d_ij = np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2)
    η_ij = np.maximum(0, r[:, None] + r[None, :] - d_ij)
    β = np.stack([np.maximum(0, r - x), np.maximum(0, r - (1 - x)), 
                  np.maximum(0, r - y), np.maximum(0, r - (1 - y))], axis=-1)
    
    return -np.sum(r) + λ * (np.sum(η_ij**p) + np.sum(β**p))

def compute_gradient(z, λ, p):
    n = len(z) // 3
    x = z[:n]
    y = z[n:2*n]
    r = z[2*n:]
    grad = np.zeros_like(z)

    d_ij = np.sqrt((x[:, None] - x[None, :])**2 + (y[:, None] - y[None, :])**2)
    η_ij = np.maximum(0, r[:, None] + r[None, :] - d_ij)
    β = np.stack([np.maximum(0, r - x), np.maximum(0, r - (1 - x)), 
                  np.maximum(0, r - y), np.maximum(0, r - (1 - y))], axis=-1)
    
    dr = (d_ij > 0) * (1 + (x[:, None] - x[None, :]) / d_ij) * (η_ij**p - 1) + (β[:, :, :p-1] * (np.isclose(η_ij, 0) & (r[:, None] + r[None, :] > d_ij))).sum(-1)
    dx = (-2 * (x[:, None] - x[None, :]) / (d_ij + 1e-12)) * (r[:, None] + r[None, :] - d_ij)**(p-1) * η_ij + (β[:, :, :p-1] * (np.isclose(η_ij, 0) & (r[:, None] + r[None, :] > d_ij))).sum(-1)
    dy = (-2 * (y[:, None] - y[None, :]) / (d_ij + 1e-12)) * (r[:, None] + r[None, :] - d_ij)**(p-1) * η_ij + (β[:, :, :p-1] * (np.isclose(η_ij, 0) & (r[:, None] + r[None, :] > d_ij))).sum(-1)
    
    grad[:n] = (β[:, 0, :p-1] * (np.isclose(η_ij, 0) & (r[:, None] + r[None, :] > d_ij))).sum(-1) + np.sum(dx, axis=0).sum(0)
    grad[n:2*n] = (β[:, 1, :p-1] * (np.isclose(η_ij, 0) & (r[:, None] + r[None, :] > d_ij))).sum(-1) + np.sum(dx, axis=0).sum(0)
    grad[2*n:] = dr
    
    grad /= λ * p
    return grad

def lp_optimize_centres(centres):
    n = len(centres)
    r = np.ones(n)
    A = np.zeros((2*n, n))
    b = np.zeros(2*n)
    for i in range(n):
        for j in range(n):
            if i < j:
                A[i*n+j][i] = 1
                A[i*n+j][j] = 1
                b[i*n+j] = np.sqrt((centres[i] - centres[j])**2)
                A[j*n+i][i] = 1
                A[j*n+i][j] = 1
                b[j*n+i] = np.sqrt((centres[i] - centres[j])**2)
    
    for i in range(n):
        A[i+n][i] = 1
        b[i+n] = 1
    
    result = minimize(lambda r: -np.sum(r), r, args=(), method='highs', bounds=[(0, 1)] * n)
    return result.x, -result.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    λ = 5 * 10**4
    p = 2
    start_configs = [
        np.random.rand(n, 2) * 0.95 + 0.025,
        np.linspace(0, 1, n+1)[1:-1],
        np.array([[0.25, 0.25], [0.25, 0.75], [0.75, 0.25], [0.75, 0.75]] * 6 + [[0.5, 0.5]] * 18)
    ]
    best_score = -np.inf
    best_centres = None
    best_radii = None
    
    for i, start in enumerate(start_configs):
        radii = np.random.rand(n) * 0.01
        z = np.concatenate((start.flatten(), radii))
        
        for _ in range(i):
            res = minimize(compute_penalty, z, args=(λ, p), method='L-BFGS-B', bounds=[(0, 1)] * 2*n + [(0, 1)] * n, jac=compute_gradient, options={'maxfun': 2000})
            penalty_val = compute_penalty(res.x, λ, p)
            
            if penalty_val < 1e-6:
                centres = res.x[:2*n].reshape(n, 2)
                radii = res.x[2*n:]
                score = np.sum(radii)
                
                if score > best_score:
                    best_score = score
                    best_centres = centres
                    best_radii = radii
    
    if best_radii is None:
        return np.zeros((0, 2)), np.zeros(0), 0.0
    
    opt_radii, opt_sum = lp_optimize_centres(best_centres)
    return best_centres, opt_radii, opt_sum

print(run_packing())

It also includes a small linear programming step post-optimization to ensure the radii and positions are feasible and optimal.
