import numpy as np
from scipy.spatial import distance
from scipy.optimize import linprog as lp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def maximal_radii(centers):
    n = centers.shape[0]
    c = np.concatenate((-radii, radii))
    A = []
    b = np.array([])
    for i in range(n):
        A.append([1, 0, 0, 0])
        b = np.append(b, [0])
        A.append([0, 1, 0, 0])
        A.append([0, 0, 1, 0])
        A.append([0, 0, 0, 1])
        b = np.append(b, [d_wall(centers[i])])
    for i in range(n):
        for j in range(i + 1, n):
            A.append([-1, 0, 0, 1])
            A.append([0, -1, 1, 0])
            b = np.append(b, [d_pair(centers[i], centers[j])])
    res = lp(c, A_ub=A, b_ub=b, bounds=[(0, None)] * 2 * n, method='highs')
    if not res.success:
        raise ValueError('Linear programming failed')
    radii = res.x[:n] + res.x[n:]
    return radii

def run_packing():
    centers = np.array([
        (0.1, 0.1), (0.3, 0.1), (0.5, 0.1), (0.7, 0.1), (0.9, 0.1),
        (0.1, 0.3), (0.3, 0.3), (0.5, 0.3), (0.7, 0.3), (0.9, 0.3),
        (0.1, 0.5), (0.3, 0.5), (0.5, 0.5), (0.7, 0.5), (0.9, 0.5),
        (0.1, 0.7), (0.3, 0.7), (0.5, 0.7), (0.7, 0.7), (0.9, 0.7),
        (0.1, 0.9), (0.3, 0.9), (0.5, 0.9), (0.7, 0.9), (0.9, 0.9),
        (0.5, 0.5)
    ])
    n = centers.shape[0]
    for _ in range(500):
        radii = maximal_radii(centers)
        sum_radii = np.sum(radii)
        if sum_radii > 2.637:
            break
        perturbation = np.random.randn(n, 2) * 0.05
        centers += perturbation
        centers = np.clip(centers, 0, 1)
        radii = np.maximum(radii, 0)
    return centers, radii, np.sum(radii)

def d_wall(p):
    return np.min([p[0], 1 - p[0], p[1], 1 - p[1]])

def d_pair(p1, p2):
    return np.linalg.norm(p1 - p2)
