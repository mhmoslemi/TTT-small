import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform
from math import exp, sqrt, pi
from random import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    center_initial = np.array([
        (0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)
    ])
    center_initial = np.append(center_initial, [
        (0.5, 0.5)
    ], axis=0)

    def objective(center):
        dists_wall = np.minimum(center[:, 0], 1 - center[:, 0])
        dists_wall = np.minimum(dists_wall, center[:, 1])
        dists_wall = np.minimum(dists_wall, 1 - center[:, 1])
        dists_pair = squareform(pdist(center))
        return -np.sum(dists_wall)

    def inner_lp(center):
        dists_pair = squareform(pdist(center))
        mat = np.block([
            np.eye(n), np.zeros((n, n)), np.zeros((n, 2 * n)),
            np.zeros((n, n)), -np.eye(n), -np.eye(n),
            np.zeros((n, n)), np.zeros((n, n)), np.eye(n)
        ])
        b = np.concatenate([dists_wall, np.zeros(n), -dists_pair[n:]])
        res = linprog(-dists_wall, A_ub=mat, b_ub=b, bounds=(0, None))
        radii = -res.x[:n]
        return center, radii

    def fitness(center):
        _, radii = inner_lp(center)
        sum_radii = np.sum(radii)
        return sum_radii

    def acceptance_probability(center_new, center_old, temp):
        sum_new, sum_old = fitness(center_new), fitness(center_old)
        return exp((sum_new - sum_old) / temp)

    def make_perturbation(center, sigma):
        return center + sigma * np.random.randn(center.shape[0], center.shape[1])

    def clamp_center(center):
        return np.clip(center, 0.1, 0.9)

    def simulated_annealing():
        center = center_initial
        best_center = center.copy()
        best_fitness = fitness(best_center)
        temp = 1.0
        while temp > 1e-6:
            for _ in range(2000):
                center_new = make_perturbation(center, 0.1 * temp)
                center_new = clamp_center(center_new)
                if acceptance_probability(center_new, center, temp) > random():
                    center = center_new
                yield center
            temp *= 0.995
            if fitness(center) > best_fitness:
                best_fitness = fitness(center)
                best_center = center.copy()
        center = best_center.copy()
        return center

    def refine(center, temp=1e-3):
        for _ in range(100):
            index = np.random.randint(n)
            for _ in range(10):
                center[index, :] += 0.1 * temp * np.random.randn(2)
            center[index, :] = clamp_center(center[index, :])
            if fitness(center) > fitness(best_center):
                best_center = center.copy()

    for center in simulated_annealing():
        _, radii = inner_lp(center)
        sum_radii = np.sum(radii)
        if sum_radii > fitness(best_center):
            best_center = center.copy()
            refine(best_center, 1e-6)

    _, best_radii = inner_lp(best_center)
    return best_center, best_radii, np.sum(best_radii)
