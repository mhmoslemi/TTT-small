import numpy as np
from scipy.optimize import linprog

# Constants
NUM_STARTS = 25
SIM_ITERS = 2500
INIT_R = 0.01
DELTA_R = 0.001
ALPHA = 0.99
EPSILON = 1e-12

def simulate_packing(initial_centres, initial_radii):
    n = len(initial_centres)
    for _ in range(SIM_ITERS):
        overlaps = np.zeros((n, n), dtype=int)
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((initial_centres[i] - initial_centres[j])**2)
                if dist_sq < (initial_radii[i] + initial_radii[j])**2:
                    overlaps[i, j] = overlaps[j, i] = 1
        
        for i, centers in enumerate(initial_centres):
            x, y = centers
            r_x = x - initial_radii[i]
            r_y = y - initial_radii[i]
            l_x = 1 - (x + initial_radii[i])
            l_y = 1 - (y + initial_radii[i])
            for dx in [-1/2, 0, 1/2]:
                for dy in [-1/2, 0, 1/2]:
                    new_x, new_y = x + dx, y + dy
                    if max(x - new_x, new_x - x) <= r_x and max(y - new_y, new_y - y) <= r_y:
                        initial_centres[i] = np.array([new_x, new_y])
                        initial_radii[i] += DELTA_R
                        if i > 0:
                            overlaps[i, np.argmax(overlaps[i])] = overlaps[np.argmax(overlaps[i]), i] = 0
        
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((initial_centres[i] - initial_centres[j])**2)
                if dist_sq < (initial_radii[i] + initial_radii[j])**2:
                    delta = np.sqrt((initial_radii[i] + initial_radii[j])**2 - dist_sq)
                    dx, dy = (initial_centres[j] - initial_centres[i]) / np.sqrt(dist_sq)
                    initial_centres[i][0] += 0.5 * delta * dx
                    initial_centres[i][1] += 0.5 * delta * dy
                    initial_centres[j][0] -= 0.5 * delta * dx
                    initial_centres[j][1] -= 0.5 * delta * dy
        
        DELTA_R *= ALPHA
        if np.all(overlaps == 0) and not np.any(isinstance(r, np.ndarray) for r in initial_radii):
            break
    
    return initial_centres, initial_radii

def solve_lp_for_centres(centres):
    n = len(centres)
    c = [-1] * n  # Maximize radius sum

    A = []
    b = []

    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            A.append([1, 1] + [-2 * (centres[i, 0] - centres[j, 0]), -2 * (centres[i, 1] - centres[j, 1])])
            b.append((centres[i] + centres[j])[0] + (centres[i] + centres[j])[1] - EPSILON)

    # Boundary constraints
    for i in range(n):
        A.append([1, 0, 0, -2, 0])
        b.append(1 - centres[i, 0] - EPSILON)
        A.append([0, 1, 0, 0, -2])
        b.append(1 - centres[i, 1] - EPSILON)
        A.append([0, 0, 1, -2, 0])
        b.append(1 - centres[i, 0] - EPSILON)
        A.append([0, 0, 0, 0, -2])
        b.append(1 - centres[i, 1] - EPSILON)

    result = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    radii = result.x
    sum_radii = -result.fun
    return radii, sum_radii

def run_packing() -> (np.ndarray, np.ndarray, float):
    best_centres = None
    best_radii = None
    best_sum = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26, 2)
        radii = np.full(26, INIT_R)
        centres, radii = simulate_packing(centres, radii)
        radii, sum_radii = solve_lp_for_centres(centres)

        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centres = centres.copy()
            best_radii = radii.copy()

    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum
