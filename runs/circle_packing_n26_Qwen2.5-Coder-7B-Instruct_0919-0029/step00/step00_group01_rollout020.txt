import numpy as np
from scipy.spatial.distance import cdist
from scipy.optimize import linprog, minimize

def get_lp_matrix(n):
    A_eq = np.zeros((n * (n - 1) + 4 * n, n))
    b_eq = np.zeros(n * (n - 1) + 4 * n)
    count = 0
    
    for i in range(n):
        A_eq[count, i] = 1
        b_eq[count] = 1
        count += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            A_eq[count, i] = 1
            A_eq[count, j] = 1
            b_eq[count] = 0.5
            count += 1
    
    for i in range(n):
        A_eq[count, i] = 1
        b_eq[count] = 0.5
        count += 1
    
    for i in range(n):
        A_eq[count, i] = -0.5
        count += 1
    
    A_ub = None
    b_ub = None
    
    return A_eq, b_eq, A_ub, b_ub

def solve_lp(centers):
    n = centers.shape[0]
    A_eq, b_eq, A_ub, b_ub = get_lp_matrix(n)
    c = np.zeros(n)
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if result.success:
        return result.x
    else:
        return np.zeros(n)

def get_gradient(centers, radii):
    n = centers.shape[0]
    epsilon = 1e-4
    grad = np.zeros_like(centers)
    for i in range(n):
        for dim in range(2):
            perturbed_centers = centers.copy()
            perturbed_centers[i, dim] += epsilon
            new_radii = solve_lp(perturbed_centers)
            new_sum = np.sum(new_radii)
            grad[i, dim] = (new_sum - np.sum(radii)) / (2 * epsilon)
    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    best_centers = centers.copy()
    best_sum = 0.0
    tol = 1e-5
    max_iterations = 200
    step_size = 0.01
    
    for _ in range(max_iterations):
        radii = solve_lp(centers)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
        
        grad = get_gradient(centers, radii)
        step = step_size * np.linalg.norm(grad)
        centers = best_centers + step * grad / np.linalg.norm(grad)
        centers = np.clip(centers, 0, 1)
    
    final_radii = solve_lp(best_centers)
    final_sum = np.sum(final_radii)
    
    return best_centers, final_radii, final_sum
