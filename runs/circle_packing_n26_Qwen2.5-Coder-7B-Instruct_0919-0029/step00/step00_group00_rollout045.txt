import numpy as np
from scipy.optimize import linprog

INIT_R = 0.01
SIM_ITERS = 2500
NUM_STARTS = 25
EPSILON = 1e-12

def run_packing():
    best_centres = None
    best_radii   = None
    best_sum     = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26,2)        # uniform sample
        radii   = np.full(26, INIT_R)

        # --- physics simulation ---
        for iter in range(SIM_ITERS):
            # 1. compute distances and overlaps
            distances = np.sqrt(np.sum((centres[:, np.newaxis, :] - centres[np.newaxis, :, :])**2, axis=2))
            overlaps = np.where((distances > 0) & (distances < 2 * radii[:, np.newaxis] + EPSILON), distances, np.inf)
            force = (radii[:, np.newaxis] + radii[np.newaxis, :] - overlaps[~np.eye(26, dtype=bool)]) / 2
            force[force < 0] = 0

            # 2. resolve overlaps by displacements
            def displacement():
                delta_centres = np.sum(force * (centres[:, np.newaxis, :] - centres[np.newaxis, :, :]) /
                                       distances[:, np.newaxis, :], axis=2)
                return delta_centres

            delta_centres = displacement()
            centres += delta_centres / 2

            # 3. boundary pushes
            for i, r in enumerate(radii):
                if centres[i, 0] - r < EPSILON:
                    centres[i, 0] = r + EPSILON
                if centres[i, 0] + r > 1 - EPSILON:
                    centres[i, 0] = 1 - r - EPSILON
                if centres[i, 1] - r < EPSILON:
                    centres[i, 1] = r + EPSILON
                if centres[i, 1] + r > 1 - EPSILON:
                    centres[i, 1] = 1 - r - EPSILON

            # 4. attempt radius growth
            def grow_radii():
                delta_r = np.minimum((centres[1:, :] - centres[:-1, :]) / 2, (1 - centres - radii[:, np.newaxis])[:, 0])
                delta_r = np.sum(delta_r, axis=1)
                return delta_r

            delta_r = grow_radii()
            old_radii = radii.copy()
            radii = old_radii + delta_r

            # 5. damping / cooling
            if iter % 100 == 0 and iter > 0:
                centres *= 0.99
                delta_r *= 0.9

            # 6. break if converged
            if np.allclose(old_radii, radii) and overlaps.max() <= EPSILON:
                break

        # --- LP post-processing ---
        opt_radii, opt_sum = solve_lp_for_centres(centres)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centres = centres.copy()
            best_radii = opt_radii.copy()

    # final LP to lock in best configuration
    final_radii, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, final_radii, final_sum

def solve_lp_for_centres(centres):
    n = centres.shape[0]
    c = -1 * np.ones(n)
    A = np.zeros((n + 2 * n + 4 * n, n))
    b = np.zeros(n + 2 * n + 4 * n)

    for i in range(n):
        A[i, i] = 1
        b[i] = EPSILON

    for i in range(n):
        for j in range(i + 1, n):
            A[n + i, i] = 1
            A[n + i, j] = -1
            b[n + i] = centres[i][0] - centres[j][0] + EPSILON

            A[n + n + i, j] = 1
            A[n + n + i, i] = -1
            b[n + n + i] = centres[i][0] - centres[j][0] - EPSILON

            A[n + n + n + i, i] = 1
            A[n + n + n + i, j] = -1
            b[n + n + n + i] = centres[i][1] - centres[j][1] + EPSILON

            A[n + n + n + n + i, j] = 1
            A[n + n + n + n + i, i] = -1
            b[n + n + n + n + i] = centres[i][1] - centres[j][1] - EPSILON

    A[n + 2 * n :, :] = centres[:, np.newaxis, :]
    b[n + 2 * n :] = 1 - radii[:, np.newaxis]

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res.x, -res.fun

if __name__ == "__main__":
    centre_sample, radius_sample, sum_radius_sample = run_packing()
    print("Centers:\n", centre_sample)
    print("Radii:\n", radius_sample)
    print("Sum of Radii:", sum_radius_sample)
