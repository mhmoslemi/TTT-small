import numpy as np
from scipy.optimize import minimze

def run_packing() -> (np.ndarray, np.ndarray, float):
    NUM_STARTS = 25
    INIT_R = 0.01
    SIM_ITERS = 2500
    INIT_TEMP = 10.0
    COOLING_FACTOR = 0.99
    EPSILON = 1e-12

    def solve_lp_for_centres(centers):
        n = centers.shape[0]
        def objective(x):
            return -np.sum(x[:n])

        def constraint(x):
            cons = []
            for i in range(n):
                x_i = x[i]
                r_i = x[n + i]
                cons.append({'type': 'ineq', 'fun': lambda t, i=i, r=r_i: t[i] - r_i - EPSILON})
                for j in range(i + 1, n):
                    x_j = x[j]
                    r_j = x[n + j]
                    x_ij = centers[i] - centers[j]
                    d_ij = np.sqrt(np.sum(x_ij ** 2))
                    cons.append({'type': 'ineq', 'fun': lambda t, i=i, j=j, r_i=r_i, r_j=r_j, d=d_ij: d_ij - (t[i] + t[j] + EPSILON)})
            return cons

        x0 = np.hstack([radii, np.zeros(n)])
        bounds = [(0, None) for _ in range(2 * n)]
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)
       radii = res.x[n:]
        sum_radii = np.sum(radii)
        return radii, sum_radii

    best_centres = None
    best_radii = None
    best_sum = -np.inf

    for start in range(NUM_STARTS):
        centres = np.random.rand(26,2)
        radii = np.full(26, INIT_R)

        for _ in range(SIM_ITERS):
            distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=-1)
            overlaps = np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - distances - EPSILON)
            shifts = np.zeros_like(centers)
            for i in range(26):
                for j in range(i + 1, 26):
                    if overlaps[i, j] > 0:
                        shift = overlaps[i, j] * (centers[i] - centers[j]) / distances[i, j]
                        shifts[i] += shift
                        shifts[j] -= shift
            shift Magnitudes = np.linalg.norm(shifts, axis=1)
            shift_steps = INIT_TEMP * (shift_magnitudes / np.sum(shift_magnitudes))
            centers -= shifts * shift_steps

            for i in range(26):
                if centers[i, 0] - radii[i] < -EPSILON:
                    centers[i, 0] = radii[i] - EPSILON
                if centers[i, 0] + radii[i] > 1 + EPSILON:
                    centers[i, 0] = 1 + EPSILON - radii[i]
                if centers[i, 1] - radii[i] < -EPSILON:
                    centers[i, 1] = radii[i] - EPSILON
                if centers[i, 1] + radii[i] > 1 + EPSILON:
                    centers[i, 1] = 1 + EPSILON - radii[i]

            for i in range(26):
                delta_r = INIT_TEMP * np.random.rand()
                new_radii = radii.copy()
                new_radii[i] += delta_r
                distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=-1)
                overlaps = np.maximum(0, new_radii[:, np.newaxis] + new_radii[np.newaxis, :] - distances - EPSILON)
                if np.any(overlaps > 0):
                    while np.any(overlaps > 0):
                        delta_r /= 2
                        new_radii[i] += delta_r
                        distances = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=-1)
                        overlaps = np.maximum(0, new_radii[:, np.newaxis] + new_radii[np.newaxis, :] - distances - EPSILON)
            radii += delta_r * np.sign(overlaps)

        for _ in range(int(SIM_ITERS * 0.1)):
            new_radii, new_sum = solve_lp_for_centres(centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = centres.copy()
                best_radii = new_radii.copy()

        INIT_TEMP *= COOLING_FACTOR

    _, final_sum = solve_lp_for_centres(best_centres)
    return best_centres, best_radii, final_sum
