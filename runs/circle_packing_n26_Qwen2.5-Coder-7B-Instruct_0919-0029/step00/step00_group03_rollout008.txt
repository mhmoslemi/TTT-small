import numpy as np
from scipy.optimize import minimize, linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 5x5 grid + center circle configuration
    x = np.concatenate([np.arange(0.1, 0.6, 0.2), [0.5]])
    y = np.concatenate([np.arange(0.1, 0.6, 0.2), [0.5]])
    centers = np.array(np.meshgrid(x, y)).T.reshape(-1, 2)
    radii_in = np.ones_like(x) * 0.1
    z0 = np.concatenate((centers.flatten(), radii_in))

    # Objective: minimize the negative sum of radii
    def objective(z):
        return -np.sum(z[52:])

    # Linear constraints matrix for the lines touching walls/borders
    A_wall = np.zeros((52, 78))
    for i in range(52):
        if i // 10 % 2 == 0:
            A_wall[i, i] = 1
            A_wall[i, i+78] = -1
        else:
            A_wall[i, i] = -1
            A_wall[i, i+78] = 1
        A_wall[i, i+78*2:78*2+10] = np.array([-1 if j % 10 == i % 10 else 0 for j in range(10)])
        A_wall[i, (i+1) % 10*78+78+10:78+10] = np.array([-1 if (i+1) % 10 == j % 10 else 0 for j in range(10)])

    b_wall = np.concatenate((np.ones(52) * 0.1, np.zeros(52)))

    # Nonlinear distance constraints
    def constraint(z):
        n = len(z) // 54
        x, y, r = z[52:].reshape(n, 2), z[:52].reshape(n, 2), z[:52]
        dists = np.linalg.norm(x[:, np.newaxis, :] - x[np.newaxis, :, :], axis=2)
        nonlin_constraints = dists - r[:, np.newaxis] - r[np.newaxis, :]
        nonlin_constraints = np.where(nonlin_constraints > 0, nonlin_constraints, 0)
        nonlin_constraints -= 1e-12
        return nonlin_constraints.ravel()

    constr = {'type': 'ineq', 'fun': constraint}

    # Bounds for variables
    bounds = [(0, None)] * 78

    # SLSQP optimization
    res = minimize(objective, z0, method='SLSQP', jac=None, constraints=[constr], bounds=bounds, options={'disp': False})

    # Extract centers and radii
    centers = res.x[52:].reshape(26, 2)
    radii = res.x[:52]

    # Cap negative radii at 0
    radii = np.maximum(radii, 0)

    # Compute exact maximum radii for given centers
    A_lp = np.zeros((26 + 26*25, 26))
    b_lp = np.zeros(26 + 26*25)
    for i in range(26):
        A_lp[i, i] = 1
        A_lp[26 + i*25:, i] = -1/2
    for i in range(26):
        for j in range(i+1, 26):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            A_lp[i, i] += 1 + 1/2
            A_lp[26 + i*25 + j - i - 1, i] -= 1/2
            A_lp[26 + i*25 + j - i - 1, j] -= 1/2
            b_lp[i] -= dist_ij/2
    max_routine_output = linprog(-radii, A_ub=A_lp, b_ub=b_lp, bounds=(0, None))
