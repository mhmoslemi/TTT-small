import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize

def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    # boundary
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None) ** 2 +
                          np.clip(xs - 1, 0, None) ** 2 +
                          np.clip(-ys, 0, None) ** 2 +
                          np.clip(ys - 1, 0, None) ** 2)
    if np.any(rs < 0):
        penalty += 1e7 * np.sum(np.clip(-rs, 0, None) ** 2)
    # pairwise
    for i in range(26):
        for j in range(i + 1, 26):
            d = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += 1e7 * gap ** 2
    return penalty

def de_objective(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = compute_penalty(v)
    return -np.sum(rs) + penalty

def solve_lps_for_centres(centres):
    A_eq = []
    b_eq = []
    A_ineq = []
    b_ineq = []

    for i in range(26):
        A_ineq.append([1] * 26)
        A_ineq[-1][i] = -1
        b_ineq.append(0)

        A_ineq.append([-1] * 26)
        A_ineq[-1][i] = 1
        b_ineq.append(0)

        A_ineq.append([-1] * 26)
        A_ineq[-1][i] = 1
        b_ineq.append(0.5)

        A_ineq.append([1] * 26)
        A_ineq[-1][i] = -1
        b_ineq.append(0.5)

        for j in range(i + 1, 26):
            dist = np.hypot(centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1])
            A_ineq.append([-1/dist] * 26)
            A_ineq[-1][i] = 1
            A_ineq[-1][j] = -1
            b_ineq.append(0)

            A_ineq.append([1/dist] * 26)
            A_ineq[-1][i] = -1
            A_ineq[-1][j] = 1
            b_ineq.append(0)

    lp_result = linprog(-np.ones(26), A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    return lp_result.x, np.sum(lp_result.x)

def run_de_search(seed):
    np.random.seed(seed)
    bounds = [(0, 1), (0, 1)] * 26 + [(0, 0.25)] * 26
    initial = np.random.rand(52)
    result = differential_evolution(de_objective, bounds, popsize=50, maxiter=2000, strategy='intermediate')
    return result.x

def run_packing():
    np.random.seed(42)
    seeds = [np.random.randint(1000000) for _ in range(2)]
    best_centers = None
    best_sum = -np.inf

    for seed in seeds:
        centers = run_de_search(seed)
        radii, sum_radii = solve_lps_for_centres(centers.reshape((26, 2)))
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers.reshape((26, 2))

    final_radii, final_sum = solve_lps_for_centres(best_centers)
    return best_centers, final_radii, final_sum

centres, radii, sum_radii = run_packing()
print((centres, radii, sum_radii))
