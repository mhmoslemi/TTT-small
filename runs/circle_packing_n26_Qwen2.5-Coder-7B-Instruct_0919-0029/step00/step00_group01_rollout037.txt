import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    penalty += 1e7 * np.sum(np.clip(-xs, 0, None)**2)
    penalty += 1e7 * np.sum(np.clip(xs - 1, 0, None)**2)
    penalty += 1e7 * np.sum(np.clip(-ys, 0, None)**2)
    penalty += 1e7 * np.sum(np.clip(ys - 1, 0, None)**2)
    penalty += 1e7 * np.sum(np.clip(-rs, 0, None)**2)
    for i in range(26):
        for j in range(i + 1, 26):
            d = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += 1e7 * gap**2
    return penalty

def de_objective(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    return -(np.sum(rs) + compute_penalty(v))

def solve_lps_for_centres(centres):
    n = 26
    c = np.ones(n)
    A = []
    b = []
    for i in range(n):
        A.append([-1 if j == i else 0 for j in range(n)])
        b.append(0)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centres[i, 0] - centres[j, 0], centres[i, 1] - centres[j, 1])
            A.append([1 if j == i else -1 if j == k else 0 for k in range(n)])
            b.append(d)
    A = np.array(A)
    b = np.array(b)
    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(c, np.zeros(n), method='highs', bounds=bounds, constraints={"type": 'ineq', "fun": lambda x: b - np.dot(A, x)})
    radii = result.x
    mask = np.abs(radii) < 1e-12
    if mask.any():
        radii[mask] = 0
    return radii, np.sum(radii)

def run_de_search(seed):
    DE_PARAMS = {
        'population_size': 50,
        'maxiter': 2000,
        'tol': 1e-6,
        'strategy': 'intermediate',
        'bounds': [(0, 1) for _ in range(52)],
        'seed': seed
    }
    result = differential_evolution(de_objective, bounds=DE_PARAMS['bounds'], polish=True, maxiter=DE_PARAMS['maxiter'], strategy=DE_PARAMS['strategy'], popsize=DE_PARAMS['population_size'])
    return result.x

def run_packing():
    seeds = [42, 99]
    centres_best =None
    for seed in seeds:
        centres = run_de_search(seed)
        if centres_best is None or de_objective(centres) > de_objective(centres_best):
            centres_best = centres
    radii, sum_radii = solve_lps_for_centres(centres_best.reshape(26, 2))
    return centres_best.reshape(26, 2), radii, sum_radii

# Run the packing function and print result
centers, radii, sum_radii = run_packing()
print("Centers:", centers)
print("Radii:", radii)
print("Sum of radii:", sum_radii)
