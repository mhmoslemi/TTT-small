import numpy as np
from scipy.optimize import basinhopping, linprog
from scipy.spatial import distance_matrix

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def solve_lp(centers):
    n = centers.shape[0]
    obj = [-1] * n
    A = np.zeros((n * (n - 1), n))
    b = np.zeros(n * (n - 1))
    eqIdx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[eqIdx, i] = 1
            A[eqIdx, j] = 1
            b[eqIdx] = distance_matrix([centers[i]], [centers[j]])[0, 0] - 1e-12
            eqIdx += 1
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c=obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        raise ValueError("LP solve failed")
    return -result.fun, result.x

def objective_and_constraints(z):
    n = z.shape[0] // 3
    centers = z.reshape((n, 3))[:, :2]
    radii = z.reshape((n, 3))[:, 2]

    sum_radii = np.sum(radii)
    sum_constraints = np.sum(np.maximum(0, centers[:, 0] - radii + 1e-12)) + \
                      np.sum(np.maximum(0, 1 - centers[:, 0] - radii + 1e-12)) + \
                      np.sum(np.maximum(0, centers[:, 1] - radii + 1e-12)) + \
                      np.sum(np.maximum(0, 1 - centers[:, 1] - radii + 1e-12))
    pair_constraints = np.sum(np.maximum(0, distance_matrix(centers, centers) - radii[:, None] - radii[None, :] + 1e-12))
    return -sum_radii, [sum_constraints, pair_constraints]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers_0 = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j] for i in range(5) for j in range(5)] + [[0.5, 0.5]])
    radii_0 = np.min(distance_matrix(centers_0, centers_0), axis=1)
    z_0 = np.concatenate([centers_0.flatten(), radii_0])

    bounds = [(0, 1) for _ in range(2 * n)]
    bounds.extend([(0, 0.5) for _ in range(n)])

    cobyla_result = basinhopping(objective_and_constraints, z_0, bounds=bounds, niter=1000, stepsize=0.1, minimizer_kwargs={'method': 'SLSQP', 'bounds': bounds})
    if not cobyla_result.success:
        raise ValueError("COBYLA optimization failed")

    centers = cobyla_result.x[:2 * n].reshape((n, 2))
    initial_radii = cobyla_result.x[2 * n:]
    _, best_radii = solve_lp(centers)
    best_radii = np.array(best_radii)
    sum_radii = np.sum(best_radii)
    if not validate_packing(centers, best_radii)[0]:
        max_scaling = 0.9999999
        best_radii *= max_scaling
        sum_radii = np.sum(best_radii)

    return centers, best_radii, sum_radii
