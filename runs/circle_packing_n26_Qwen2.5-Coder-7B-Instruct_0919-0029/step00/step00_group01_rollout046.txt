import numpy as np
from scipy.optimize import linprog

def solve_lp(centers, radii):
    n = len(centers)
    c = [-r for r in radii]  # Maximize sum of radii
    A_eq = np.zeros((n, n))
    b_eq = np.zeros(n)
    A_le = np.zeros((n*(n-1)//2, n))
    b_le = np.zeros(n*(n-1)//2)
    
    for i in range(n):
        A_eq[i, i] = 1
        b_eq[i] = 2 * (min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]))
    
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A_le[idx, i] = 1
            A_le[idx, j] = 1
            b_le[idx] = centers[i] - centers[j]
            idx += 1
    
    res = linprog(c, A_ub=A_le, b_ub=b_le, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs')
    if res.success:
        radii = res.x
    else:
        radii = np.full(n, -1)
    
    return radii, -res.fun

def run_packing():
    np.random.seed(0)
    centers = np.random.rand(26, 2)
    max_sum = 0
    best_centers = None
    best_radii = None
    
    for _ in range(3):
        centers = np.random.rand(26, 2)
        for iter in range(200):
            radii, current_sum = solve_lp(centers, np.ones(26))
            if iter % 10 == 0:
                step_size = 0.01 * 0.95 ** (iter // 10)
                gradient_x = np.array([
                    0.5 * (solve_lp(centers + 1e-4 * np.eye(2), radii)[0] - solve_lp(centers - 1e-4 * np.eye(2), radii)[0]) for centers in centers
                ])
                gradient_y = np.array([
                    0.5 * (solve_lp(centers[:, 0:1] + 1e-4 * np.array([1]), centers[:, 1:2] + 1e-4 * np.array([1]), radii)[0] - solve_lp(centers[:, 0:1] - 1e-4 * np.array([1]), centers[:, 1:2] - 1e-4 * np.array([1]), radii)[0]) for centers in centers
                ])
                step_direction = np.clip(gradient_x + gradient_y, -1, 1)
                step_size = np.maximum(step_size * (1 - np.sum(step_direction < 0)), step_size * (1 + np.sum(step_direction > 0)))
                centers += step_size * step_direction
            if current_sum > max_sum:
                max_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        
        # One last large random jiggle
        idx = np.argmin(best_radii)
        while True:
            perturbation = (np.random.rand(2) - 0.5) * 0.2
            new_center = best_centers[idx] + perturbation
            new_center = np.clip(new_center, 0, 1)
            new_radii, new_sum = solve_lp(np.delete(best_centers, idx, axis=0), np.delete(best_radii, idx))
            if new_sum > max_sum - 1e-5:
                best_radii[idx] = new_radii
                best_centers[idx] = new_center
                max_sum = new_sum
                break
    
    # Local radius increase
    for k in range(10):
        for i in np.random.permutation(26):
            current_radius = best_radii[i]
            current_center = best_centers[i]
            perturbation = np.random.rand(2) * 0.001
            new_center = best_centers[i] + perturbation
            new_center = np.clip(new_center, 0, 1)
            new_radii, current_sum = solve_lp(np.delete(best_centers, i, axis=0) + new_center.reshape(1, 2), np.delete(best_radii, i))
            if new_sum > max_sum:
                best_centers[i] = new_center
                best_radii[i] = current_radius + current_sum
                max_sum = new_sum
    
    return best_centers, best_radii, max_sum
