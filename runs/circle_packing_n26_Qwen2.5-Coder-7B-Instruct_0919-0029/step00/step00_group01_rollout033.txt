import numpy as np
from scipy.optimize import differential_evolution, linprog

def compute_penalty(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    penalty = 0.0
    penalty += np.sum(np.clip(-xs, 0, None)**2 + np.clip(xs-1, 0, None)**2 + np.clip(-ys, 0, None)**2 + np.clip(ys-1, 0, None)**2)
    penalty += np.sum(np.clip(-rs, 0, None)**2)
    for i in range(26):
        for j in range(i+1, 26):
            d = np.hypot(xs[i]-xs[j], ys[i]-ys[j])
            gap = rs[i] + rs[j] - d
            if gap > 0:
                penalty += gap**2
    return penalty

def de_objective(v):
    xs = v[:52:2]
    ys = v[1:52:2]
    rs = v[52:]
    return rs.sum() - compute_penalty(v)

def solve_lps_for_centres(centres):
    A1 = np.zeros((26, 26))
    np.fill_diagonal(A1, 2.0)
    A1 = np.triu(A1, 1)
    A2 = -[np.hypot(centres[::2, np.newaxis] - centres[::2], centres[1::2, np.newaxis] - centres[1::2])]
    A = np.vstack([A1, A2])
    b = np.hstack([np.ones(26), centres.max(axis=0)])

    res = linprog(-np.ones(26), A_ub=A, b_ub=b, bounds=[0, 0.5], method='highs')
    radii = -res.x
    radii[radii < 0] = 0
    sum_radii = radii.sum()
    return radii, sum_radii

def run_de_search(seed):
    bounds = [(0, 1), (0, 1)] * 26 + [(0, 0.25)] * 26
    result = differential_evolution(de_objective, bounds, popsize=50, maxiter=2000, tol=1e-6, seed=seed)
    return result.x[:52]

def run_packing():
    centers_best1 = run_de_search(42)
    centers_best2 = run_de_search(24)
    if solve_lps_for_centres(centers_best1)[1] > solve_lps_for_centres(centers_best2)[1]:
        centers = centers_best1
    else:
        centers = centers_best2
    radii, sum_radii = solve_lps_for_centres(centers)
    return centers.reshape(-1, 2), radii, sum_radii

# Final execution: Run the packing
centers, radii, sum_radii = run_packing()
