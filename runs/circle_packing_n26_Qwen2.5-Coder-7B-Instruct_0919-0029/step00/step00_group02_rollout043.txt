def _ineq_constraints(x):
    n = len(x) // 3
    con = []
    for i in range(n):
        con.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3] - x[i*3+2]})  # x - r >=0
        con.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*3] - x[i*3+2]})  # 1-x-r >=0
        con.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*3+1] - x[i*3+2]}) # y - r >=0
