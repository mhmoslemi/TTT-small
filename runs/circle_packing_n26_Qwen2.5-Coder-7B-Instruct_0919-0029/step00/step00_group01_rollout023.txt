import numpy as np
from scipy.optimize import differential_evolution, linprog

def penalty(v):
    summary, n = 0, 26
    for i in range(n):
        x, y, r = v[2*i:(2*i+2)], v[2*i+2], v[2*n+i]
        summary += max(0, -x)**2 + max(0, x-1)**2 + max(0, -y)**2 + max(0, y-1)**2 + max(0, -r)**2
        for j in range(i):
            dist = np.sqrt(np.sum((x-v[2*j:(2*j+2)])**2))
            summary += max(0, r + v[2*n+j] - dist)**2
    return summary

def evaluate(v):
    penalty_val = penalty(v)
    rad_sum = -np.sum(v[2*n:])
    return rad_sum + 1e7 * penalty_val

def optimize_positions():
    n = 26
    bounds = [(0, 1)] * (2*n) + [(0, 0.25)] * n
    result = differential_evolution(evaluate, bounds, popsize=50, strategy='intermediate', 
                                   mutation=0.8, crossover=0.7, maxiter=2000)
    return result.x

def optimize_radii(centers):
    n = 26
    def lp_objective(radii):
        objective = -np.sum(radii)
        boundary_constraints = [r - min centers[0, 0], r - min(1 - centers[0, 0], centers[0, 1]), r - min(1 - centers[0, 1], centers[1, 1])]
        pair_constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                pair_constraints.append(radii[i] + radii[j] - dist)
        constraints = [{"type": "ineq", "fun": lambda r, b=bc: r - b} for bc in boundary_constraints]
        constraints += [{"type": "ineq", "fun": lambda r, p=pc: r - p} for pc in pair_constraints]
        result = linprog([objective], bounds=[0, None], A_ub=np.vstack(constraints).T, b_ub=np.zeros(len(constraints)), method='highs')
        return result.x

def run_packing():
    centers = optimize_positions().reshape(26, 3)[..., :2]
    radii = optimize_radii(centers)
    return (centers, radii, np.sum(radii))
