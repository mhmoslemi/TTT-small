import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    centers = np.array([
        (0.1 + i * 0.2, 0.1 + j * 0.2) for i in range(5) for j in range(5)
    ])
    centers[-1] = (0.5, 0.5)
    radius_bounds = (0, None)
    
    def objective(x):
        return -x.sum()

    def pairwise_distances(centers):
        n = len(centers)
        dist_matrix = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dist_matrix[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                dist_matrix[j, i] = dist_matrix[i, j]
        return dist_matrix

    def constraints(max_radius):
        n = len(centers)
        c_wall = np.array([min(1 - c[0], c[0], 1 - c[1], c[1]) for c in centers])
        c_distance = np.array([min(max_radius[i] + max_radius[j], d) for i in range(n) for j in range(i + 1, n)])
        return [{'type': 'ineq', 'fun': lambda x, i=i: x[i] - c_distance[np.argmax(x == x[i])]} for i in range(n)] + [{'type': 'ineq', 'fun': lambda x, i=i: c_wall[i] - x[i]} for i in range(n)]

    def optimize_radii(centers):
        dist_matrix = pairwise_distances(centers)
        max_radius = np.zeros(n)
        
        for _ in range(10):  # loop to improve initial bounds
            bounds = np.vstack((max_radius, c_wall)).T
            cons = constraints(max_radius)
            result = linprog([1] * n, A_ub=dist_matrix, b_ub=dist_matrix.flatten(), bounds=bounds, method='highs')
            if not result.success:
                max_radius[min(result.x)] += 1e-4
            else:
                max_radius = result.x
        
        bounds = np.vstack((max_radius, c_wall)).T
        cons = constraints(max_radius)
        result = linprog([1] * n, A_ub=dist_matrix, b_ub=dist_matrix.flatten(), bounds=bounds, method='highs')
        if result.success:
            return result.x
        return max_radius
    
    def sann_search():
        best_centers = centers
        best_t = -np.inf
        for _ in range(1000):
            centers = best_centers + np.random.randn(n, 2) * 0.1
            centers = np.clip(centers, 0, 1)
            radii = optimize_radii(centers)
            t = radii.sum()
            if t > best_t:
                best_t = t
                best_centers = centers
        return best_centers, optimize_radii(best_centers)
    
    centers, radii = sann_search()
    return centers, radii, radii.sum()
