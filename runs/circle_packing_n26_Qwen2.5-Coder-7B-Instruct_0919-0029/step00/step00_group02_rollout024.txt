import numpy as np
from scipy.spatial import distance_matrix
from random import random, uniform, randint

def build_random_feasible(n, r0=0.05, max_trials=10000):
    centres = []
    while len(centres) < n:
        attempts = 0
        while attempts < max_trials:
            c = np.array([uniform(r0, 1-r0), uniform(r0, 1-r0)])
            valid = True
            for centre in centres:
                if np.linalg.norm(c - centre) < 2*r0:
                    valid = False
                    break
            if valid:
                centres.append(c)
                break
            attempts += 1
    r = np.full(n, r0)
    return np.array(centres), r

def simulated_annealing(centers, radii, best_sum, n_iter=200000, T0=1.0, cool=0.9995, move_prob=0.7, shift_max=0.1, radius_max=0.08):
    n circles = len(centers)
    n_shift = 1000
    for k in range(n_iter):
        T = T0 * (cool ** k)
        i = randint(0, n - 1)
        if random() < move_prob:
            dx, dy = uniform(-shift_max, shift_max), uniform(-shift_max, shift_max)
            new_pos = np.clip(centers[i] + np.array([dx, dy]), radius_max, 1 - radius_max)
            new_r = radii[i]
        else:
            dr = uniform(-radius_max, radius_max)
            new_r = max(0.0, radii[i] + dr)
            new_r = min(new_r, min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]))
            new_pos = centers[i]
        tentative_centres = np.copy(centers)
        tentative_centres[i] = new_pos
        dist = distance_matrix(tentative_centres, tentative_centres)
        tentative_radii = np.copy(radii)
        tentative_radii[i] = new_r
        feasible = True
        for i, j in np.ndindex(n, n):
            if i != j and dist[i, j] < tentative_radii[i] + tentative_radii[j] - 1e-12:
                feasible = False
                break
        if feasible:
            new_sum = tentative_radii.sum()
            if new_sum >= best_sum or random() < np.exp((new_sum - best_sum) / T):
                centers = tentative_centres
                radii = tentative_radii
                best_sum = new_sum
    return centers, radii, best_sum

def deterministic_polish(centers, radii, radius_max=0.1):
    n = len(centers)
    def grow(i, step):
        r_try = radii[i] + step
        if r_try <= max(0.0, min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])):
            radii[i] = r_try
            return True
        return False

    def shift(i, step_x, step_y, max_step=1e-4):
        new_pos = centers[i] + np.array([step_x, step_y])
        if np.all(new_pos >= 0) and np.all(new_pos <= 1):
            centers[i] = new_pos
            return True
        return False

    for _ in range(10):
        improved = False
        for i in range(n):
            if grow(i, 1e-4):
                improved = True
        for i in range(n):
            for step_x, step_y in [(1e-4, 0), (-1e-4, 0), (0, 1e-4), (0, -1e-4)]:
                if shift(i, step_x, step_y, max_step=1e-4):
                    improved = True

    return centers, radii
