import numpy as np
from scipy.optimize import differential_evolution, linprog, Bounds

# Constants
N = 26
BOUND = 0.25
PENALTY = 1e7

# Define penalty function
def compute_penalty(v):
    xs = v[:N*2]
    ys = v[N*2:]
    rs = v[-N:]
    penalty = 0.0
    # Boundary violations
    penalty += PENALTY * np.sum(np.maximum(-xs, 0)**2 + np.maximum(xs-1, 0)**2 + np.maximum(-ys, 0)**2 + np.maximum(ys-1, 0)**2)
    if np.any(rs < 0):
        penalty += PENALTY * np.sum(np.maximum(-rs, 0)**2)
    # Pairwise overlaps
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(xs[i]-xs[j], ys[i]-ys[j])
            fancy_gap = -np.maximum(rs[i]+rs[j] - d, 0.0)
            penalty += PENALTY * fancy_gap**2
    return penalty

# Objective function for differential evolution
def de_objective(v):
    xs = v[:N*2]
    ys = v[N*2:]
    penalty = compute_penalty(v)
    return -np.sum(ys) + penalty

    radii = lp_result.x
    return radii, -lp_result.fun
