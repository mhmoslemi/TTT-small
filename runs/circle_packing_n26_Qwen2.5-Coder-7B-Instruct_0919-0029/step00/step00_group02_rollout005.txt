import numpy as np
from scipy.spatial import distance_matrix

def build_random_feasible(n_circles, r0=0.05):
    centers = []
    radii = [r0] * n_circles

    for _ in range(n_circles):
        while True:
            center = np.random.rand(2)
            if all(center - r0 >= 0) and all(center + r0 <= 1):
                valid = all((center - np.array(centers))**2 >= (r0 + radii)**2 + 1e-12)
                if valid:
                    centers.append(center)
                    break
    return np.array(centers), np.array(radii)

def sanitized_too_close(centers, radii, tol):
    dists = distance_matrix(centers, centers)
    np.fill_diagonal(dists, np.inf)
    proximity = np.any(dists <= 2 * radii - tol)
    return proximity

def is_feasible(centers, radii, tol=1e-12):
    return np.all(centers - radii >= 0) and np.all(centers + radii <= 1) and not sanitized_too_close(centers, radii, tol)

def simulated_annealing(centers, radii, best_sum):
    T0, cool, n_iter, move_prob, shift_max, radius_max = 1.0, 0.9995, 200000, 0.7, 0.1, 0.08
    n_shift = 1000

    for k in range(n_iter):
        T = T0 * cool**k
        i = np.random.randint(26)
        if np.random.rand() < move_prob:
            dx, dy = np.random.randn(2) * shift_max
            new_pos = np.clip(centers[i] + [dx, dy], [radii[i], 1-radii[i]], [1-radii[i], radii[i]])
        else:
            dr = np.random.randn() * radius_max
            new_r = np.clip(radii[i] + dr, 0.0, min(centers[i][0], 1-centers[i][0], centers[i][1], 1-centers[i][1]))
            new_pos = centers[i]

        new_radii = radii.copy()
        new_radii[i] = new_r
        new_centers = centers.copy()
        new_centers[i] = new_pos

        if is_feasible(new_centers, new_radii):
            new_sum = new_radii.sum()
            if new_sum >= best_sum:
                best_sum = new_sum
                centers, radii = new_centers, new_radii
            else:
                if np.random.rand() < np.exp((new_sum - best_sum) / T):
                    centers, radii = new_centers, new_radii

    return centers, radii, best_sum

def deterministic_polish(centers, radii):
    approx = 1e-4

    improved = True
    while improved:
        improved = False
        for i in range(26):
            while True:
                r_try = min(1 - centers[i][0], 1 - centers[i][1], centers[i][0], centers[i][1]) - approx
                if all((centers[i] - r_try - centers[j])**2 >= (r_try + radii[j])**2 + 1e-12 for j in range(26) if j != i):
                    radii[i] = r_try
                    improved = True
                else:
                    break

    for _ in range(10):
        i = np.random.randint(26)
        dx, dy = np.random.randn(2) * approx
        new_pos = np.clip(centers[i] + [dx, dy], [radii[i], 1-radii[i]], [1-radii[i], radii[i]])
        if is_feasible(np.array([new_pos if j == i else centers[j] for j in range(26)]), np.array([ radii if j == i else radii[j] for j in range(26)])):
            centers[i] = new_pos

    return centers, radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present")
    for i in range(n):
        if radii[i] < 0:
            raise ValueError(f"Circle {i} has negative radius {radii[i]}")
        x, y = centers[i]
        r = radii[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                raise ValueError(f"Circles {i} and {j} overlap")
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii = build_random_feasible(26, r0=0.05)
    best_centers, best_radii, best_sum = centers, radii, radii.sum()
    centers, radii, best_sum = simulated_annealing(centers, radii, best_sum)
    centers, radii = deterministic_polish(centers, radii)
    validate_packing(centers, radii)
    return centers, radii, best_sum
