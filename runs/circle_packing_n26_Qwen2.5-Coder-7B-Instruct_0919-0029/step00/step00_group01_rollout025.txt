import numpy as np
from highs import solve_miplp

def lp_oracle(centers):
    n = centers.shape[0]
    c = np.zeros(n)
    A_ub = np.zeros((2 * n, n))
    b_ub = np.zeros(2 * n)

    for i in range(n):
        A_ub[i, i] = 1
        A_ub[n + i, i] = 1

        A_ub[i, i] -= 1
        b_ub[i] = centers[i, 0]
        A_ub[n + i, i] -= 1
        b_ub[n + i] = 1 - centers[i, 0]

        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            diff /= dist
            A_ub[i, j] = -diff[0]
            A_ub[i, i] -= diff[0]
            A_ub[n + i, j] = -diff[0]
            A_ub[n + i, i] -= diff[0]
            A_ub[j, i] = diff[0]
            A_ub[j, j] -= diff[0]
            A_ub[n + j, i] = diff[0]
            A_ub[n + j, j] -= diff[0]

            A_ub[i, j] -= diff[1]
            b_ub[i] -= diff[1]
            A_ub[n + i, j] -= diff[1]
            b_ub[n + i] -= diff[1]
            A_ub[j, i] -= diff[1]
            b_ub[j] -= diff[1]
            A_ub[n + j, i] -= diff[1]
            b_ub[n + j] -= diff[1]

    result = solve_miplp(c, A_ub, b_ub)
    return result["x"], result["objective_value"]

def gradient_descent():
    centers = np.random.uniform(0, 1, size=(26, 2))
    best_sum = -np.inf
    best_centers = None
    alpha = 0.01
    step_count = 0
    tol = 1e-5

    for _ in range(2):  # Two restarts
        for _ in range(200):  # Max 200 iterations
            radii, sum_radii = lp_oracle(centers)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers.copy()

            for i in range(26):
                x_plus = np.copy(centers)
                x_minus = np.copy(centers)
                x_plus[i, 0] += 1e-4
                x_minus[i, 0] -= 1e-4
                r_plus, _ = lp_oracle(x_plus)
                r_minus, _ = lp_oracle(x_minus)
                grad_x = (r_plus[i] - r_minus[i]) / (2 * 1e-4)

                x_plus[i, 1] += 1e-4
                x_minus[i, 1] -= 1e-4
                r_plus, _ = lp_oracle(x_plus)
                r_minus, _ = lp_oracle(x_minus)
                grad_y = (r_plus[i] - r_minus[i]) / (2 * 1e-4)

                centers[i] += alpha * np.array([grad_x, grad_y])
                centers[i] = np.clip(centers[i], 0, 1)

            if step_count > 100 and np.abs(np.sum(centers) - np.sum(best_centers)) < tol:
                break

        alpha *= 0.95
        step_count += 1

    # Final jiggle
    if best_centers is not None:
        i = np.argmin(radii)
        jiggle = np.random.uniform(0, 1, 2)
        jiggle -= np.min(jiggle)  # Normalize to [0,1]
        jiggle *= 0.01  # Scale down
        best_centers[i] += jiggle
        best_centers[i] = np.clip(best_centers[i], 0, 1)

    # Post-processing refinement
    for _ in range(10):
        for i in range(26):
            radii, _ = lp_oracle(best_centers)
            best_centers[i, 0] += 1e-3 / best_centers[i, 0]
            best_centers[i, 1] += 1e-3 / best_centers[i, 1]
            best_centers[i], _ = lp_oracle(best_centers)
            new_r, new_s = lp_oracle(best_centers)
            if new_s < sum_radii:
                best_centers[i, 0] -= 2 * 1e-3 / best_centers[i, 0]
                best_centers[i, 1] -= 2 * 1e-3 / best_centers[i, 1]
                best_centers[i], _ = lp_oracle(best_centers)

    return best_centers, radii, sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, _ = gradient_descent()
    return centers, radii, np.sum(radii)
