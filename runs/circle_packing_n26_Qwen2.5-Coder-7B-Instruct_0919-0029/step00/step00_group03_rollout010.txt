import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

# Constants
N = 26
MAX_ITER = 100000
SA_ITER = 2000
GAUSS_SIGMA = 0.1
TEMPERATURE = 1.0
TEMP_DECREASE = 0.995
TEMP_THRESHOLD = 1e-6

# Geometry preprocessing
def wall_distances(centers):
    x, y = centers[:, 0], centers[:, 1]
    return np.column_stack((x, y, 1 - x, 1 - y)).min(axis=1)

def pairwise_distances(centers):
    return squareform(pdist(centers))

# Inner LP - "Maximal radii for fixed centres"
def inner_lp(centers):
    d_wall = wall_distances(centers)
    d_pair = pairwise_distances(centers)
    c = -np.ones(N)  # Maximize sum of radii
    A_ub = np.vstack((np.eye(N), np.triu(np.ones((N, N)), k=1), -np.triu(np.ones((N, N)), k=1)))
    b_ub = np.hstack((d_wall, d_pair[:-1], -d_pair[1:]))
    x0 = np.zeros(N)
    res = linprog(c, A_ub, b_ub, method='highs')
    return res.x

# Outer optimizer - stochastic gradient descent
def sa_optimizer():
    # Initial configuration
    centers = np.array([
        (0.1 + i * 0.2, 0.1 + j * 0.2) for i in range(5) for j in range(5)

    centers[12] = (0.5, 0.5)

        for _ in range(MAX_ITER):
            # Randomly perturb a circle
            i = np.random.randint(N)
            noise = np.clip(np.random.normal(0, GAUSS_SIGMA, 2), 0, 1)
            centers[i] += noise / np.max(noise)

            # Clamp to [0, 1]
            centers = np.clip(centers, 0, 1)

            # Compute new sum of radii
            radii = inner_lp(centers)
            new_sum_radii = np.sum(radii)

            if best_sum_radii is None or new_sum_radii > best_sum_radii:
                best_centers = centers.copy()
                best_sum_radii = new_sum_radii
            elif np.random.rand() < np.exp((new_sum_radii - best_sum_radii) / TEMPERATURE):
                best_centers = centers.copy()
                best_sum_radii = new_sum_radii

            # Periodic gradient-like moves
            if np.random.rand() < 0.01:
                for i in range(N):
                    for j in range(i + 1, N):
                        if np.sum((centers[i] - centers[j]) ** 2) < 0.01:
                            shift = centers[i] - centers[j]
