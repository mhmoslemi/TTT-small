import numpy as np
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import minimize

def build_random_feasible(n: int, r0: float) -> tuple[np.ndarray, np.ndarray]:
    centers = []
    while len(centers) < n:
        c = np.array([np.random.uniform(0, 1), np.random.uniform(0, 1)])
        valid = True
        for cx, cy in centers:
            if np.linalg.norm(c - np.array([cx, cy])) < 2 * r0:
                valid = False
                break
        if valid:
            centers.append(c)
    radii = [r0] * n
    return np.array(centers), np.array(radii)

def is_feasible(centers: np.ndarray, radii: np.ndarray,
               Rmax: float, tol: float = 1e-12) -> bool:
    n, _ = centers.shape
    dists = squareform(pdist(centers))
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i, 0] - radii[i] < -tol or centers[i, 0] + radii[i] > 1 + tol or \
           centers[i, 1] - radii[i] < -tol or centers[i, 1] + radii[i] > 1 + tol:
            return False
        for j in range(i + 1, n):
            if dists[i, j] <= (radii[i] + radii[j] - tol) and dists[i, j] > 2 * Rmax[j] + tol:
                return False
    return True

def simulated_annealing(centers: np.ndarray, radii: np.ndarray,
                        initial_sum: float) -> tuple[np.ndarray, np.ndarray, float]:
    n = len(centers)
    Rmax = np.array([min(c[0], 1 - c[0], c[1], 1 - c[1]) for c in centers])
    temperature = 1.0
    cooling_factor = 0.9995
    n_iter = 200000
    move_prob = 0.7
    shift_max = 0.1
    radius_max = 0.08
    n_shift = 1000

    best_centers = centers
    best_radii = radii
    best_sum = initial_sum

    for k in range(n_iter):
        i = np.random.randint(n)
        if np.random.uniform(0, 1) < move_prob:
            dx = np.random.uniform(-shift_max, shift_max)
            dy = np.random.uniform(-shift_max, shift_max)
            new_pos = np.clip(centers[i] + np.array([dx, dy]), Rmax[i], 1 - Rmax[i])
        else:
            dr = np.random.uniform(-radius_max, radius_max)
            new_r = max(0.0, min(new_pos[0], 1 - new_pos[0], new_pos[1], 1 - new_pos[1]))
            new_r = min(new_r, Rmax[i])
            new_pos = centers[i]

        S_prime = np.copy(centers)
        S_prime[i] = new_pos.copy()
        S_prime[i, 2] = new_r

        if is_feasible(S_prime, best_radii, Rmax):
            current_sum = np.sum(best_radii)
            new_sum = current_sum + new_r - best_radii[i]
            if new_sum >= current_sum or np.random.uniform(0, 1) < np.exp((new_sum - current_sum) / temperature):
                best_centers = S_prime.copy()
                best_radii = best_radii.copy()
                best_sum = max(best_sum, new_sum)

        if k % n_shift == 0:
            shift_max *= 0.95
            radius_max *= 0.95
            temperature *= cooling_factor

    return best_centers, best_radii, best_sum

def deterministic_polish(centers: np.ndarray, radii: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    n = len(centers)
    improved = True
    while improved:
        improved = False
        for i in range(n):
            while True:
                r_try = radii[i] + 1e-4
                if r_try <= min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]):
                    radii[i] = r_try
                    improved = True
                else:
                    break

    for _ in range(5):
        i = np.random.randint(n)
        for dx, dy in [(1e-4, 0), (-1e-4, 0), (0, 1e-4), (0, -1e-4),
                      (1e-4, 1e-4), (-1e-4, 1e-4), (1e-4, -1e-4), (-1e-4, -1e-4)]:
            new_center = np.clip(centers[i] + np.array([dx, dy]), [0, 0], [1, 1])
            if is_feasible(np.insert(centers, i, new_center, axis=0), 
                           np.insert(radii, i, radii[i] + 1e-4), radii):
                centers[i] = new_center

    return centers, radii
