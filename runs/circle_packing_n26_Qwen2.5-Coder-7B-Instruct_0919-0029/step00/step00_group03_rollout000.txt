import numpy as np
from scipy.optimize import minimize, linprog

def solve_optimisation_problem(z_initial):
    n = 26
    z_dim = 2 * n + n

    A = np.zeros((6 * n, z_dim))
    b = np.zeros(6 * n)

    # Borders
    for i in range(n):
        A[2 * i, 2 * i] = 1
        b[2 * i] = 0
        A[2 * i + 1, 2 * i + 1] = 1
        b[2 * i + 1] = 0

    # Distances
    for i in range(n):
        for j in range(i + 1, n):
            row_i = 2 * n + 2 * (j * n + i)
            A[row_i, 2 * i] = -1
            A[row_i, 2 * j] = -1
            A[row_i, 2 * i + 1] = -1
            A[row_i, 2 * j + 1] = -1
            A[row_i, 2 * i + 2] = 1 / (2 * (i // 5 + 1))
            A[row_i, 2 * i + 3] = 1 / (2 * (i // 5 + 1))
            A[row_i, 2 * j + 2] = 1 / (2 * (i // 5 + 1))
            A[row_i, 2 * j + 3] = 1 / (2 * (i // 5 + 1))

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    res = minimize(lambda z: np.sum(z[:-n]), z_initial, bounds=bounds, constraints={'type': 'ineq', 'fun': lambda z: A @ z - b})

    if not res.success:
        return None, None, None

    centers = res.x[:2 * n].reshape(-1, 2)
    radii = res.x[2 * n:]
    return centers, radii, np.sum(radii)


def run_packing():
    centers = []
    y, x = 0.1, 0.1
    for row in range(5):
        for col in range(5):
            centers.append([x + 0.2 * col, y + 0.2 * row])
    centers.append([0.5, 0.5])

    radii_initial = np.array([min(c[i], 1 - c[i], c[3 - i]) for c in centers])
    centers = np.array(centers)

    centers_flipped = centers.flatten()
    result = solve_optimisation_problem(centers_flipped)
    if result[0] is None or result[2] < 2.637:
        scaling_factor = 0.98
        result = solve_optimisation_problem(centers_flipped * scaling_factor)
        if result[2] < 2.637:
            return np.zeros((26, 2)), np.zeros(26), 0

    return result
