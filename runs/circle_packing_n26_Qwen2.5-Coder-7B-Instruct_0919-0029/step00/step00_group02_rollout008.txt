import numpy as np
from ortools.linear_solver import pywraplp

def build_milp_grid(k):
    solver = pywraplp.Solver.CreateSolver('GLOP')
    grid_vars = {}

    for i in range(k):
        for j in range(k):
            p = np.array([i * 1.0 / k, j * 1.0 / k])
            r_max = min(p[0], 1 - p[0], p[1], 1 - p[1])
            b_p = solver.IntVar(0, 1, f'b_{i}_{j}')
            r_p = solver.NumVar(0, r_max, f'r_{i}_{j}')
            grid_vars[(i, j)] = (b_p, r_p)
            solver.Add(r_p <= b_p * r_max)

    return solver, grid_vars

def solve_milp(solver, time_limit):
    solver.NumVar(0, 18, 'n')
    constraint_node = solver.Constraint(26, 26)
    for i in range(400):
        constraint_node.SetCoefficient(i, 1)
    solver.SetTimeLimit(time_limit * 1000)
    status = solver.Solve()
    return status, [(grid_vars[(i//20, i%20)][0].solution_value(), 
                     grid_vars[(i//20, i%20)][1].solution_value()) for i in range(400)]

def extract_from_sol(best_sol, grid_vars):
    centers = []

        if b_p > 0:
            centers.append((i * 1.0 / 20, j * 1.0 / 20))
            radii.append(r_p)
    return np.array(centers), np.array(radii)

def fibonacci_merge(x1, x2, r1, r2):
    phi = (1 + np.sqrt(5)) / 2
    if np.abs(x1 - x2) < 1e-12:
        return x1, r1
    elif x1 < x2:
        if x2 - x1 < 2 * r2:
            x = x1 + r1 / (np.sqrt(phi) + phi)
            r = (np.sqrt(phi) + 1) * r1
        else:
            x = x1 + r1 + r2
            r = r1 + r2
    else:
        if x1 - x2 < 2 * r1:
            x = x2 + r2 / (np.sqrt(phi) + phi)
            r = (np.sqrt(phi) + 1) * r2
        else:
            x = x2 + r1 + r2

    return x, r

def fibonacci_split(x1, r1):
    phi = (1 + np.sqrt(5)) / 2
    r2 = r1 / (phi - 1)
    x2 = x1 - r2
    return x1, x2, r1, r2

def optimal_grow(centers, radii):
    for i in range(len(radii)):
        for attempt in range(10):
            r_try = radii[i] + 1e-4
            conflicts = False
            for j in range(len(radii)):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if d < radii[i] + r_try:
                        conflicts = True
                        break
            if conflicts:
                break
            else:
                radii[i] = r_try
